        # Positions with h near 0 or 1 that have large |grad| are good candidates
        mid_positions = [(abs(grad[i]), i) for i in range(n) if 0.05 < best_h[i] < 0.95]
        edge_positions = [(abs(grad[i]), i) for i in range(n) 
                         if (best_h[i] < 0.05 or best_h[i] > 0.95) and abs(grad[i]) > 1e-6]
        
        mid_positions.sort(reverse=True)
        edge_positions.sort(reverse=True)
        
        # Pick top positions to optimize
        opt_positions = [p for _, p in mid_positions[:10]] + [p for _, p in edge_positions[:15]]
        opt_positions = list(set(opt_positions))
        
        if len(opt_positions) < 2:
            break
        
        # Optimize these positions with L-BFGS
        def sub_obj(sub_x):
            h_tmp = best_h.copy()
            for k, pos in enumerate(opt_positions):
                h_tmp[pos] = sub_x[k]
            h_tmp = project_feasible(h_tmp, n)
            return compute_c5(h_tmp, n)
        
        def sub_grad(sub_x):
            h_tmp = best_h.copy()
            for k, pos in enumerate(opt_positions):
                h_tmp[pos] = sub_x[k]
            h_tmp = project_feasible(h_tmp, n)
            g_full = np.zeros(n)
            c_vec2 = compute_c5_vec(h_tmp, n)
            w2 = smooth_max_w(c_vec2, 500000.0)
            g_full = h_grad_from_w(w2, h_tmp, n)
            return np.array([g_full[p] for p in opt_positions])
        
        x0 = np.array([best_h[p] for p in opt_positions])
        bounds_sub = [(0.0, 1.0)] * len(opt_positions)
        try:
            res = minimize(sub_obj, x0, jac=sub_grad, bounds=bounds_sub,
                          method='L-BFGS-B',
                          options={'maxiter': 2000, 'ftol': 1e-28, 'gtol': 1e-22,
                                   'maxfun': 20000, 'maxls': 40, 'maxcor': 30})
            h_tmp = best_h.copy()
            for k, pos in enumerate(opt_positions):
                h_tmp[pos] = np.clip(res.x[k], 0.0, 1.0)
            h_tmp = project_feasible(h_tmp, n)
            c_new = compute_c5(h_tmp, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_tmp
        except Exception:
            pass
    
    return best_h, best_c

def fine_pair_swap(h, n, deadline):
    """Fine-grained continuous pair swaps"""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            w = smooth_max_w(c_vec, 300000.0)
            grad = h_grad_from_w(w, best_h, n)
            
            pos_idx = np.argsort(-grad)[:50]
            neg_idx = np.argsort(grad)[:50]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx:
                if best_h[i] < step:
                    continue
                for j in neg_idx:
                    if best_h[j] > 1.0 - step:
                        continue
                    amt = min(step, best_h[i], 1.0 - best_h[j])
                    if amt < 1e-12:
                        continue
                    best_h[i] -= amt
                    best_h[j] += amt
                    best_h = project_feasible(best_h, n)
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        best_gain = best_c
                        best_i = i
                        best_j = j
                        improved = True
                        break
                    else:
                        best_h[i] += amt
                        best_h[j] -= amt
                        best_h = project_feasible(best_h, n)
                if improved:
                    break
    return best_h, best_c

def make_beatty_n(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 6 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = min(diff, 1.0)
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = max(diff + 1.0, 0.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    random.seed(seed)
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(400, 0.5)
        n_parent = 400
    
    h_parent = project_feasible(h_parent, n_parent)
    
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: L-BFGS warmup (80s)
    p1_deadline = min(t0 + 80, deadline)
    h1, c1 = run_lbfgs(best_h, best_n, alpha=5000, mu=0.01, maxit=2000)
    if c1 < best_c:
        best_c, best_h = c1, h1
    h1b, c1b = run_lbfgs(best_h, best_n, alpha=50000, mu=0.001, maxit=2000)
    if c1b < best_c:
        best_c, best_h = c1b, h1b
    
    # Phase 2: Binary optimization - simulated annealing (200s)
    p2_deadline = min(t0 + 280, deadline - 400)
    h2, c2 = sa_binary(best_h, best_n, p2_deadline, rng, t0_temp=0.002, t1_temp=0.00005)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()
    
    # Phase 3: Greedy swap binary (150s)
    p3_deadline = min(t0 + 430, deadline - 300)
    h3, c3 = greedy_swap_binary(best_h, best_n, p3_deadline, rng)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()
    
    # Phase 4: More SA from best binary (150s)
    p4_deadline = min(t0 + 580, deadline - 250)
    h4, c4 = sa_binary(best_h, best_n, p4_deadline, rng, t0_temp=0.001, t1_temp=0.00002)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()
    
    # Phase 5: L-BFGS continuous refinement (120s)
    p5_deadline = min(t0 + 700, deadline - 200)
    for alpha in [20000, 100000, 500000]:
        if time.time() > p5_deadline - 5:
            break
        h5, c5 = run_lbfgs(best_h, best_n, alpha=alpha, mu=0.0005, maxit=3000)
        if c5 < best_c:
            best_c, best_h = c5, h5
    
    # Phase 6: Surgical continuous optimization (120s)
    p6_deadline = min(t0 + 820, deadline - 150)
    h6, c6 = surgical_continuous(best_h, best_n, p6_deadline)
    if c6 < best_c:
        best_c, best_h = c6, h6
    
    # Phase 7: Fine pair swap (100s)
    p7_deadline = min(t0 + 920, deadline - 100)
    h7, c7 = fine_pair_swap(best_h, best_n, p7_deadline)
    if c7 < best_c:
        best_c, best_h = c7, h7
    
    # Phase 8: Another round of SA + continuous (80s)
    p8_deadline = min(t0 + 1000, deadline - 40)
    if time.time() < p8_deadline:
        h8, c8 = sa_binary(best_h, best_n, min(time.time() + 40, p8_deadline - 20), rng,
                          t0_temp=0.0005, t1_temp=0.00001)
        if c8 < best_c:
            best_c, best_h = c8, h8.copy()
        if time.time() < p8_deadline:
            h8b, c8b = run_lbfgs(best_h, best_n, alpha=1000000, mu=0.0001, maxit=2000)
            if c8b < best_c:
                best_c, best_h = c8b, h8b
    
    # Phase 9: Final surgical + fine swap (remainder)
    if time.time() < deadline - 5:
        h9, c9 = surgical_continuous(best_h, best_n, deadline - 3)
        if c9 < best_c:
            best_c, best_h = c9, h9
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
