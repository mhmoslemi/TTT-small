def compute_pq(h,n):
    P = np.zeros(n, dtype=float)
    Q = np.zeros(n, dtype=float)
    if n <= 1:
       return P,Q,0.0
    prefix = np.zeros(n+1, dtype=float)
    np.cumsum(h, out=prefix[1:])
    for m in range(1,n):
       L=n-m
       dot=float(np.dot(h[:L], h[m:]))
       P[m]=prefix[L]-dot
       Q[m]=(prefix[n]-prefix[m])-dot
    L0=float(prefix[n]-np.dot(h,h))
    return P,Q,L0

import numpy as np
import time
import math


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(2000):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot <= 1e-14:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)

    diff = target - float(h.sum())
    for _ in range(2000):
        if diff > 1e-15:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, max(0.0, 1.0 - h[idx]))
            if add <= 0.0:
                break
            h[idx] += add
            diff -= add
        elif diff < -1e-15:
            idx = int(np.argmax(h))
            sub = min(-diff, max(0.0, h[idx]))
            if sub <= 0.0:
                break
            h[idx] -= sub
            diff += sub
        else:
            break

    return h


def compute_c5_exact(h, n):
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_pq(h, n):
    P = np.zeros(n, dtype=float)
    Q = np.zeros(n, dtype=float)
    if n <= 1:
        return P, Q, 0.0

    prefix = np.zeros(n + 1, dtype=float)
    np.cumsum(h, out=prefix[1:])

    for m in range(1, n):
        L = n - m
        dot = float(np.dot(h[:L], h[m:]))
        P[m] = prefix[L] - dot
        Q[m] = (prefix[n] - prefix[m]) - dot

    L0 = float(prefix[n] - np.dot(h, h))
    if L0 < 0.0:
        L0 = 0.0
    return P, Q, L0


def metric_count(P, Q, L0):
    pmax = float(P[1:].max()) if P.size > 1 else 0.0
    qmax = float(Q[1:].max()) if Q.size > 1 else 0.0
    pmax = max(0.0, pmax)
    qmax = max(0.0, qmax)
    L0 = max(0.0, float(L0))
    m = max(pmax, qmax, L0)
    return float(m + 0.005 * (pmax + qmax + L0))


def threshold_binary(h, n):
    target = int(n // 2)
    out = np.zeros(n, dtype=float)
    if target <= 0:
        return out
    if target >= n:
        out[:] = 1.0
        return out

    idx = np.argpartition(-np.asarray(h, dtype=float), target)[:target]
    out[idx] = 1.0
    return out


def random_binary(n, rng):
    out = np.zeros(n, dtype=float)
    target = int(n // 2)
    if target <= 0:
        return out
    perm = rng.permutation(n)
    out[perm[:target]] = 1.0
    return out


def interpolate_h(h_src, n_src, n_dst):
    if n_dst <= 0:
        return np.zeros(0, dtype=float)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def apply_binary_swap(h, gh, P, Q, src, dst):
    n = len(h)

    li = n - 1 - src
    if li > 0:
        P[1:li + 1] -= gh[src + 1:src + 1 + li]
        Q[1:li + 1] += h[src + 1:src + 1 + li]

    if src > 0:
        P[1:src + 1] += h[0:src]
        Q[1:src + 1] -= gh[0:src]

    lj = n - 1 - dst
    if lj > 0:
        P[1:lj + 1] += gh[dst + 1:dst + 1 + lj]
        Q[1:lj + 1] -= h[dst + 1:dst + 1 + lj]

    if dst > 0:
        P[1:dst + 1] -= h[0:dst]
        Q[1:dst + 1] += gh[0:dst]

    d = abs(src - dst)
    if 0 < d < n:
        P[d] += 1.0
        Q[d] += 1.0

    h[src] = 0.0
    h[dst] = 1.0
    gh[src] = 1.0
    gh[dst] = 0.0


def apply_continuous_move(h, gh, P, Q, L0, i, j, a):
    if a == 0.0:
        return L0

    b = -a
    old_i = h[i]
    old_j = h[j]
    n = len(h)

    li = n - 1 - i
    if li > 0:
        P[1:li + 1] += a * gh[i + 1:i + 1 + li]
        Q[1:li + 1] -= a * h[i + 1:i + 1 + li]

    if i > 0:
        P[1:i + 1] -= a * h[0:i]
        Q[1:i + 1] += a * gh[0:i]

    lj = n - 1 - j
    if lj > 0:
        P[1:lj + 1] += b * gh[j + 1:j + 1 + lj]
        Q[1:lj + 1] -= b * h[j + 1:j + 1 + lj]

    if j > 0:
        P[1:j + 1] -= b * h[0:j]
        Q[1:j + 1] += b * gh[0:j]

    d = abs(i - j)
    if 0 < d < n:
        cor = -a * b
        P[d] += cor
        Q[d] += cor

    new_i = old_i + a
    new_j = old_j + b
    L0 += (new_i * (1.0 - new_i) - old_i * (1.0 - old_i))
    L0 += (new_j * (1.0 - new_j) - old_j * (1.0 - old_j))
    if L0 < 0.0:
        L0 = 0.0

    h[i] = new_i
    h[j] = new_j
    gh[i] = 1.0 - new_i
    gh[j] = 1.0 - new_j
    return L0


def make_top_entries(h, gh, P, Q, n, binary):
    entries = []
    K = 25
    R = 80

    if n <= 2:
        return entries

    pidx = np.argsort(P[1:])[-K:]
    for idx in pidx:
        m = int(idx) + 1
        L = n - m
        if L <= 0:
            continue
        val = float(P[m])
        if val <= 1e-10:
            continue

        if binary:
            cand = np.nonzero((h[:L] >= 0.5) & (h[m:] <= 0.5))[0]
        else:
            score = h[:L] * gh[m:]
            nz = np.nonzero(score > 1e-8)[0]
            if nz.size == 0:
                continue
            if nz.size <= R:
                cand = nz
            else:
                cand = nz[np.argpartition(-score[nz], R)[:R]]

        if cand.size > 0:
            entries.append((-val, 0, m, cand, len(entries)))

    qidx = np.argsort(Q[1:])[-K:]
    for idx in qidx:
        m = int(idx) + 1
        L = n - m
        if L <= 0:
            continue
        val = float(Q[m])
        if val <= 1e-10:
            continue

        if binary:
            cand = np.nonzero((h[:L] <= 0.5) & (h[m:] >= 0.5))[0]
        else:
            score = gh[:L] * h[m:]
            nz = np.nonzero(score > 1e-8)[0]
            if nz.size == 0:
                continue
            if nz.size <= R:
                cand = nz
            else:
                cand = nz[np.argpartition(-score[nz], R)[:R]]

        if cand.size > 0:
            entries.append((-val, 1, m, cand, len(entries)))

    entries.sort()
    return entries[:K]


def local_search(h_start, n, deadline, rng, binary=False, max_step_init=0.2, max_step_end=0.002):

    if deadline <= t_start + 1.0:
        h = project_feasible(np.asarray(h_start, dtype=float).copy(), n)
        if binary:

        else:
            h = project_feasible(h, n)
        return h, compute_c5_exact(h, n)

    h = project_feasible(np.asarray(h_start, dtype=float).copy(), n)

    if exact0 < 0.0:
        exact0 = 0.0

    best_count = exact0
    best_c = (2.0 / float(n)) * best_count

    best_metric = current_metric
    best_metric_h = h.copy()

    T = 0.1
    T_min = 1e-5
    cool = 0.9997
    no_improve = 0
    reset_interval = 55 if binary else 80

            break
        cnt += 1

        total = max(1.0, float(deadline - t_start))
        frac = min(1.0, max(0.0, (time.time() - t_start) / total))
        max_step_current = max_step_end + (max_step_init - max_step_end) * (1.0 - frac)

        if cnt % reset_interval == 0:
            top_entries = make_top_entries(h, gh, P, Q, n, binary)

        use_move = False
        src = 0

            entry = top_entries[rng.randint(len(top_entries))]

                    break

                    continue
                if side_int == 0:
                    src = u
                    dst = u + m
                    if h[src] > 1e-10 and gh[dst] > 1e-10:
                        use_move = True
                        break
                else:
                    src = u + m
                    dst = u
                    if h[src] > 1e-10 and gh[dst] > 1e-10:
                        use_move = True
                        break

        if not use_move:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                T = max(T_min, T * cool)
                continue
            if h[i] > h[j] + 1e-12:
                src = i

                use_move = True
            elif h[j] > h[i] + 1e-12:
                src = j
                dst = i
                use_move = True
            else:
                if h[i] > 1e-10 and gh[j] > 1e-10:
                    src = i
                    dst = j
                    use_move = True
                elif h[j] > 1e-10 and gh[i] > 1e-10:
                    src = j
                    dst = i
                    use_move = True

        if not use_move:
            T = max(T_min, T * cool)

        used_full = False

        if binary:

                T = max(T_min, T * cool)

            apply_binary_swap(h, gh, P, Q, src, dst)

            used_full = True
        else:

            rstep = rng.random()

            if step <= 1e-9:

                continue
            L0 = apply_continuous_move(h, gh, P, Q, L0, src, dst, -step)

        new_metric = metric_count(P, Q, L0)

                no_improve += 1

        else:

        if cnt % 50000 == 0:
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, n)
            if binary:
                h = threshold_binary(h, n)
            gh = 1.0 - h

            pert = 3 if binary else 5

                    continue

                elif h[jj] > h[ii] + 1e-12:

    best_c = parent_c
    best_n = n_base

    phase_specs = [
        (100, "cont_short", 0.12, 0.002, False, False),

        (40, "final_polish", 0.02, 0.0005, False, False),

    for duration, kind, msi, mse, use_binary, high_res in phase_specs:
        remaining = deadline - time.time()

        if phase_time < 15.0:

        n_try = best_n

            n_try = best_n
            h_try = random_binary(n_try, rng)
        elif high_res:
            if best_n <= 512:
                high_n = 512 if best_n < 512 else best_n
            else:
                high_n = (best_n // 64 + 1) * 64

                    high_n = best_n
                if high_n > 1200:
                    high_n = 1024 if 1024 > best_n else best_n
            n_try = int(high_n)
            if n_try != best_n:
                h_try = interpolate_h(best_h, best_n, n_try)
            else:
                h_try = best_h.copy()
            if use_binary:
                h_try = threshold_binary(h_try, n_try)

        h_out, _ = local_search(
            h_try,
            n_try,
            phase_deadline,
            rng,
            binary=use_binary,
            max_step_init=msi,
            max_step_end=mse
        )

        h_c = project_feasible(h_out, n_try)
        c_c = compute_c5_exact(h_c, n_try)
        candidates.append((h_c.copy(), c_c, n_try))

        if c_c < best_c - 1e-15:
            best_h = h_c.copy()
            best_c = c_c
            best_n = n_try

        if time.time() > deadline:
            break

    final_best_h = None
    final_best_c = float("inf")
    final_best_n = 0

    for h_c, c_c, n_c in candidates:
        h_v = project_feasible(h_c, n_c)
        c_v = compute_c5_exact(h_v, n_c)
        if c_v < final_best_c - 1e-15:
            final_best_h = h_v
            final_best_c = c_v
            final_best_n = n_c

    if final_best_h is None:
        final_best_h = project_feasible(best_h, best_n)
        final_best_c = compute_c5_exact(final_best_h, best_n)
        final_best_n = best_n

    return final_best_h, float(final_best_c), int(final_best_n)
