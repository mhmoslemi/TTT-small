from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(0, 1.0 - h)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # distribute to/from interior points
        interior = (h > 0.01) & (h < 0.99)
        if interior.sum() > 0:
            n_int = interior.sum()
            h[interior] += diff / n_int
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    """Gradient of correlation at shift_idx w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def active_max_obj(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(mx + np.log(np.sum(np.exp(sh))) / alpha)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(0.0, 1.0 - h)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (1.0 + diff / tot)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 0.005) & (h < 0.995)
        cnt = interior.sum()
        if cnt > 0:
            h[interior] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def shift_grad_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def active_obj(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        return float(mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    return float(mx + np.log(np.sum(np.exp(sh))) / alpha)

def active_grad(x, n, active, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi > 1e-15:
            grad += wi * shift_grad_single(x, n, idx)
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-14:
            continue
        grad += w[mm] * shift_grad_single(x, n, mm)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def top_indices(c_vec, k, margin_frac=0.01):
    mx = c_vec.max()
    thresh = mx - margin_frac * mx
    idx = np.where(c_vec >= thresh)[0]
    if len(idx) > k:
        order = np.argsort(-c_vec[idx])
        idx = idx[order[:k]]
    return idx

def active_set_lbfgs(h, n, deadline, top_k_list=None, alpha_list=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if top_k_list is None:
        top_k_list = [20, 10, 5, 3, 5, 3, 8, 4]
    if alpha_list is None:
        alpha_list = [500, 2000, 10000, 50000, 200000]
    
    for tk in top_k_list:
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        active = top_indices(c_vec, tk, margin_frac=0.02)
        if len(active) == 0:
            active = np.argsort(-c_vec)[:tk]
        
        for alpha in alpha_list:
            if time.time() > deadline - 2:
                break
            remaining = deadline - time.time()
            maxiter = max(200, int(min(2000, remaining * 8)))
            try:
                res = minimize(active_obj, h, jac=active_grad,
                              args=(n, active, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': maxiter * 3, 'maxls': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
            except Exception:
                pass
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_sweeps=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for step in steps:
        if time.time() > deadline - 2:
            break
        for sweep in range(max_sweeps):
            if time.time() > deadline - 2:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    break
                old = h[i]
                # Try increasing
                new_val = min(1.0, old + step)
                if new_val > old:
                    h[i] = new_val
                    # Adjust another point to maintain sum
                    diff = new_val - old
                    # pick a point with value > diff to reduce
                    j = (i + n // 3) % n
                    reduction = min(diff, h[j])
                    if reduction > 0:
                        h[j] -= reduction
                        # Compensate remaining diff
                        rem = diff - reduction
                        if rem > 0:
                            candidates = np.where(h > rem / max(1, n))
                            if len(candidates[0]) > 0:
                                for jj in candidates[0][:50]:
                                    if jj == i:
                                        continue
                                    h[jj] -= rem / len(candidates[0])
                                h[i] = project_feasible(h, n)[i]
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] = old
                            h[j] += reduction
                            if rem > 0:
                                for jj in candidates[0][:50]:
                                    if jj == i:
                                        continue
                                    h[jj] += rem / len(candidates[0])
                # Try decreasing
                new_val = max(0.0, old - step)
                if new_val < old:
                    h[i] = new_val
                    diff = old - new_val
                    j = (i + n // 3) % n
                    addition = min(diff, 1.0 - h[j])
                    if addition > 0:
                        h[j] += addition
                        rem = diff - addition
                        if rem > 0:
                            h = project_feasible(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] = old
                            h[j] -= addition
                            if rem > 0:
                                h = project_feasible(h, n)
            if not improved:
                break
    return best_h, best_c

def pair_local_search(h, n, deadline, radius=40, steps=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if steps is None:
        steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for step in steps:
        if time.time() > deadline - 2:
            break
        stagnation = 0
        while stagnation < 5:
            if time.time() > deadline - 2:
                break
            found = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for direction in [1, -1]:
                        hi_i = h[i] + direction * step
                        lo_j = h[j] - direction * step
                        if 0.0 <= hi_i <= 1.0 and 0.0 <= lo_j <= 1.0:
                            h[i] = hi_i
                            h[j] = lo_j
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                best_h = h.copy()
                                found = True
                            else:
                                h[i] = best_h[i]
                                h[j] = best_h[j]
            if not found:
                break
            stagnation = 0
        h = best_h.copy()
    
    return best_h, best_c

def perturb_and_reoptimize(h, n, deadline, n_restarts=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        # Random perturbation
        scale = 0.02 / (r + 1)
        rng = np.random.RandomState(42 + r * 101)
        h_pert = h + scale * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        
        # Quick active-set optimization
        t_sub = min(time.time() + 8, deadline - 5)
        h_opt, c_opt = active_set_lbfgs(h_pert, n, t_sub,
                                         top_k_list=[5, 3],
                                         alpha_list=[5000, 50000])
        if c_opt < best_c - 1e-14:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=200000, maxiter=2000):
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    # Load initial
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except (NameError, Exception):
        h = np.full(500, 0.5)
        n = 500
    
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Quick full L-BFGS (30s)
    t1 = min(t0 + 30, deadline - 800)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in [(200, 50.0), (2000, 500.0)]:
        if time.time() > t1:
            break
        try:
            res = minimize(full_obj, h, jac=full_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 30000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    # Phase 2: Active-set optimization at n (250s)
    t2 = min(time.time() + 250, deadline - 500)
    h, c_local = active_set_lbfgs(h, n, t2,
                                   top_k_list=[30, 15, 10, 5, 8, 4, 6, 3, 10, 5],
                                   alpha_list=[500, 2000, 10000, 50000, 200000, 500000])
    if c_local < best_c:
        best_c = c_local
        best_h = h
    
    # Phase 3: Pair local search at n (80s)
    t3 = min(time.time() + 80, deadline - 400)
    h, c_local = pair_local_search(h, n, t3, radius=50)
    if c_local < best_c:
        best_c = c_local
        best_h = h
    
    # Phase 4: Try n=1000 resolution (200s)
    if time.time() < deadline - 200:
        n2 = 1000
        h2 = upsample(best_h, n, n2)
        c2 = compute_c5(h2, n2)
        
        t4a = min(time.time() + 100, deadline - 150)
        h2, c2 = active_set_lbfgs(h2, n2, t4a,
                                    top_k_list=[20, 10, 5, 3, 5, 3],
                                    alpha_list=[500, 5000, 50000, 200000])
        
        t4b = min(time.time() + 50, deadline - 60)
        h2, c2 = pair_local_search(h2, n2, t4b, radius=30,
                                    steps=[0.01, 0.005, 0.002, 0.001])
        
        if c2 < best_c:
            best_c = c2
            best_h = h2
            n = n2
    
    # Phase 5: Back to best resolution, more active-set rounds (100s)
    if time.time() < deadline - 80:
        t5 = min(time.time() + 100, deadline - 60)
        h, c_local = active_set_lbfgs(best_h, n, t5,
                                       top_k_list=[10, 5, 3, 4, 6, 3, 2, 4],
                                       alpha_list=[2000, 10000, 50000, 200000, 500000, 1000000])
        if c_local < best_c:
            best_c = c_local
            best_h = h
    
    # Phase 6: Perturbation restarts (60s)
    if time.time() < deadline - 40:
        t6 = min(time.time() + 60, deadline - 30)
        h, c_local = perturb_and_reoptimize(best_h, n, t6, n_restarts=4)
        if c_local < best_c:
            best_c = c_local
            best_h = h
    
    # Phase 7: SLSQP polish (40s)
    if time.time() < deadline - 20:
        t7 = min(time.time() + 40, deadline - 10)
        h, c_local = slqp_polish(best_h, n, t7, alpha=500000, maxiter=2000)
        if c_local < best_c:
            best_c = c_local
            best_h = h
    
    # Phase 8: Final pair search
    if time.time() < deadline - 5:
        t8 = deadline - 3
        h, c_local = pair_local_search(best_h, n, t8, radius=30,
                                        steps=[0.001, 0.0005, 0.0002])
        if c_local < best_c:
            best_c = c_local
            best_h = h
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
