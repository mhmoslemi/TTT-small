def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Ensure exactly n/2 ones
    # Thue-Morse has exactly n/2 ones when n is a power of 2
    # For other n, we need to adjust
    ones = int(t.sum())
    target = n // 2
    if ones > target:
        # Remove some ones (from the end)
        idx = np.where(t == 1)[0]
        for i in idx[target:]:
            t[i] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        for i in idx[:target-ones]:
            t[i] = 1
    return t

def binary_sa(h_init, n, deadline, n_restarts=8):
    """Binary simulated annealing."""
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    
    rng = np.random.RandomState(42)
    
    for restart in range(n_restarts):
        if time.time() > deadline - 30:
            break
        
        if restart == 0:
            h = h_init.copy()
        elif restart == 1:
            h = thue_morse(n)
        else:
            h = np.zeros(n)
            ones = rng.choice(n, n//2, replace=False)
            h[ones] = 1.0
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # SA
        T = (best_c - c + 0.05)  # Initial temperature
        T = max(T, 0.02)
        T_min = 0.0005
        alpha = 0.99995  # Cooling rate
        
        n_iter = 500000
        iterations_done = 0
        
        while iterations_done < n_iter and time.time() < deadline - 30:
            # Pick random move
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            j = ones[rng.randint(len(ones))]
            m = zeros[rng.randint(len(zeros))]
            
            if j == m:
                continue
            
            # Try the move
            h[j] = 0
            h[m] = 1
            c_new = compute_c5(h, n)
            
            delta = c_new - c
            
            if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
                c = c_new
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                # Reject
                h[j] = 1
                h[m] = 0
            
            iterations_done += 1
            T = max(T * alpha, T_min)
            
            # Periodic check
            if iterations_done % 10000 == 0 and time.time() > deadline - 30:
                break
    
    return best_h, best_c
