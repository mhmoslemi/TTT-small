import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h2 = np.zeros(n)
    h2[idx[:target]] = 1.0
    return h2

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

Actually, here's a better idea. Instead of checking all n(n/2) pairs, I can:
1. Find the top 5 shifts (largest correlation values)
2. For each top shift s*, compute which (i,j) pairs would reduce c[s*]
3. Check only those pairs

For a fixed shift s*, the change in c[s*] from swapping h[i]:0→1 and h[j]:1→0 is:
delta_c[s*] = (1-h[i+ds]) - h[i-ds] - (1-h[j+ds]) + h[j-ds] - cross
where ds = s* - (n-1) (the actual shift amount), and cross terms handle interactions.

This is O(1) per pair for a fixed shift. So for 5 top shifts, it's O(5) per pair, and we check n/2 * n/2 pairs = O(n²/4) pairs total. For n=250, that's ~15600 pairs, each checked in O(5) = O(78000) operations. Very fast!

But we need to verify that the swap doesn't increase any OTHER shift above the current max. This requires the full O(n) delta computation. So:
1. Fast filter: O(1) per pair for top shifts → candidate pairs
2. Full verification: O(n) per candidate → confirm the swap reduces max

def swap_delta_fast(h, n, i, j):
    """O(n) delta computation for swapping h[i]:0->1, h[j]:1->0"""
    m = 2 * n - 1
    delta = np.zeros(m)
    m_range = np.arange(-(n-1), n)
    
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    
    msk = (im >= 0) & (im < n)
    delta[msk] += (1.0 - h[im[msk]])
    
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= (1.0 - h[jm[msk]])
    
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < m:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < m:
        delta[k2] += 1.0
    
    return delta
