        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(lm_obj, best_h, jac=lm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-14:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_ks=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if top_ks is None:
        top_ks = [4, 6, 8, 12, 16, 24, 32]
    alphas = [20000, 100000, 500000, 2000000, 10000000]
    for top_k in top_ks:
        if time.time() > deadline - 20:
            break
        for alpha in alphas[:3]:
            if time.time() > deadline - 15:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-10, 0.002 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                active = np.array([int(np.argmax(c_vec))])
            maxit = max(200, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def full_coord_descent(h, n, deadline, step_sizes=None, max_passes=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    dx_val = 2.0 / n
    passes = 0
    for step in step_sizes:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline and passes < max_passes:
            improved_any = False
            passes += 1
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx:
                for j in neg_idx:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain < 1e-14:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            old_i = best_h[best_i]
            old_j = best_h[best_j]
            best_h[best_i] = old_i - amt
            best_h[best_j] = old_j + amt
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h[best_i] = np.clip(best_h[best_i], 0, 1)
                best_h[best_j] = np.clip(best_h[best_j], 0, 1)
                best_h = project_feasible(best_h, n)
                improved_any = True
            else:
                best_h[best_i] = old_i
                best_h[best_j] = old_j
    return best_h, best_c

def binary_flip_search(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h < 0.5)[0]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        old_hi = best_h[i]
        old_hj = best_h[j]
        best_h[i] = np.clip(old_hj, 0, 1)
        best_h[j] = np.clip(old_hi, 0, 1)
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
        else:
            best_h[i] = old_hi
            best_h[j] = old_hj
            best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] >= 1.0 - 1e-12 or best_h[j] <= 1e-12:
                        continue
                    best_h[i] += d
                    best_h[j] -= d
                    c = compute_c5_vec(best_h, n).max()
                    if c < best_c - 1e-15:
                        best_c = float(c)
                        improved = True
                    else:
                        best_h[i] -= d
                        best_h[j] += d
                    if best_h[i] >= 1.0 - 1e-12 or best_h[j] <= 1e-12:
                        continue
                    best_h[i] -= d
                    best_h[j] += d
                    c = compute_c5_vec(best_h, n).max()
                    if c < best_c - 1e-15:
                        best_c = float(c)
                        improved = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    seen = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in seen:
            seen.add(pos)
            positions.append(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_block_binary(n, block_size, rng):
    h = np.zeros(n)
    n_ones = n // 2
    if block_size <= 1:
        return make_random_binary(n, rng)
    n_blocks = n // block_size
    n_blocks_ones = n_ones // block_size
    block_idx = rng.choice(n_blocks, size=n_blocks_ones, replace=False)
    for b in block_idx:
        start = b * block_size
        h[start:start + block_size] = 1.0
    return project_feasible(h, n)

def make_thue_morse(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    period = 1
    while period < n:
        period *= 4
    pattern = np.zeros(period)
    tm = 0
    for i in range(period):
        t = bin(i).count('1') % 2
        pattern[i] = t
    h_full = np.tile(pattern, n // period + 1)[:n]
    sorted_idx = np.argsort(-h_full)
    h[sorted_idx[:n_ones]] = 1.0
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    ratio = n_new / n_old
    h_new = np.zeros(n_new)
    for i in range(n_new):
        src = min(n_old - 1, int(i / ratio))
        h_new[i] = h[src]
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Phase 1: Quick L-BFGS from initial (120s)
    p1_end = t0 + 120
    h1, c1 = run_lbfgs(best_h, n, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
        mu=0.01, maxit_list=[800, 1000, 1500, 2000, 2500, 3000, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Diverse binary starts + quick optimization (250s)
    p2_end = min(t0 + 370, deadline - 400)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = []
    starts.append(make_beatty(n, golden / 2.0))
    starts.append(make_beatty(n, np.sqrt(2.0) / 2.0))
    starts.append(make_beatty(n, np.sqrt(3.0) / 2.0))
    starts.append(make_beatty(n, np.pi / (2.0 + np.sqrt(2.0))))
    starts.append(make_thue_morse(n, rng))
    starts.append(make_block_binary(n, 2, rng))
    starts.append(make_block_binary(n, 3, rng))
    starts.append(make_block_binary(n, 5, rng))
    for _ in range(12):
        starts.append(make_random_binary(n, rng))

    for h_s in starts:
        if time.time() > p2_end:
            break
        c_s = compute_c5(h_s, n)
        if c_s < best_c - 0.001:
            quick_end = min(time.time() + 8, p2_end - 5)
            h_q, c_q = run_lbfgs(h_s, n, quick_end,
                alphas=[100, 1000, 10000], mu=0.01,
                maxit_list=[500, 800, 1000])
            if c_q < best_c:
                best_c = c_q
                best_h = h_q
        elif c_s < best_c:
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # Phase 3: Active set optimization (200s)
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 200, deadline - 220)
        h3, c3 = active_set_optimize(best_h, n, p3_end,
            top_ks=[4, 6, 8, 12, 16, 24, 32, 48])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Full coordinate descent (180s)
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 180, deadline - 130)
        h4, c4 = full_coord_descent(best_h, n, p4_end,
            step_sizes=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
            max_passes=800)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Binary flip search (80s)
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 80, deadline - 85)
        h5, c5 = binary_flip_search(best_h, n, p5_end, max_rounds=500)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Active set again after coord descent (80s)
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 80, deadline - 65)
        h6, c6 = active_set_optimize(best_h, n, p6_end,
            top_ks=[4, 6, 8, 12, 16, 24])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine local search (80s)
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 80, deadline - 40)
        h7, c7 = fine_local_search(best_h, n, p7_end,
            radius=50,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Another round of coord descent with tiny steps (50s)
    if time.time() < deadline - 30:
        p8_end = min(time.time() + 50, deadline - 20)
        h8, c8 = full_coord_descent(best_h, n, p8_end,
            step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
            max_passes=500)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final fine polish (20s)
    if time.time() < deadline - 5:
        p9_end = deadline - 2
        h9, c9 = fine_local_search(best_h, n, p9_end,
            radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
