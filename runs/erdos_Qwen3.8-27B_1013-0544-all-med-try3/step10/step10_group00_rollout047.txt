    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()

        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(abs(diff)))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(abs(diff)))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def make_beatty(n, alpha):
    h = np.zeros(n)
    target = n // 2
    seen = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
            if len(seen) >= target:
                break
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for z in zeros[:int(np.ceil(diff))]:
                h[z] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for o in ones[:int(np.ceil(-diff))]:
                h[o] = 0.0
    return project_feasible(h, n)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(i.bit_count() % 2)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for z in zeros[:int(np.ceil(diff))]:
                h[z] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for o in ones[:int(np.ceil(-diff))]:
                h[o] = 0.0
    return project_feasible(h, n)

def sa_binary(h, n, deadline, rng, t_init=0.04, cool=0.99992, max_iters=400000):
    h = binarize(h, n).copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    t = t_init
    tabu = {}
    tabu_len = 80
    n_accepted = 0
    n_rejected = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t *= cool

        c_vec = compute_c5_vec(h, n)
        alpha_hot = 800.0
        w = smooth_max_w(c_vec, alpha_hot)
        grad = h_grad_from_w(w, h, n)

        if rng.random() < 0.75:
            pos_cands = [i for i in ones if grad[i] > 1e-10]
            neg_cands = [j for j in zeros if grad[j] < -1e-10]
            if pos_cands and neg_cands:
                pos_cands.sort(key=lambda x: -grad[x])
                neg_cands.sort(key=lambda x: grad[x])
                pool_p = pos_cands[:60]
                pool_n = neg_cands[:60]
                i = pool_p[rng.randint(len(pool_p))]
                j = pool_n[rng.randint(len(pool_n))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        skey = (min(i, j), max(i, j))
        if skey in tabu and it - tabu[skey] < tabu_len:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            tabu[skey] = it
            n_accepted += 1
            if cur_c < best_c:
                best_c = cur_c
        else:
            h[i] = 1.0
            h[j] = 0.0
            n_rejected += 1

        if it % 5000 == 0 and n_accepted < 50:
            for _ in range(5):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h[ii] = 0.0
                h[jj] = 1.0
                c2 = compute_c5(h, n)
                if c2 < cur_c:
                    cur_c = c2
                    ones.remove(ii)
                    ones.append(jj)
                    zeros.remove(jj)
                    zeros.append(ii)
                else:
                    h[ii] = 1.0
                    h[jj] = 0.0

    return h, best_c

def greedy_binary(h, n, deadline, rng, max_iters=200000):
    h = binarize(h, n).copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    stuck = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        alpha = 600.0
        w = smooth_max_w(c_vec, alpha)
        grad = h_grad_from_w(w, h, n)

        pos_sorted = sorted(ones, key=lambda x: -grad[x])[:50]
        neg_sorted = sorted(zeros, key=lambda x: grad[x])[:50]

        best_delta = 0.0
        best_i = -1
        best_j = -1

        for i in pos_sorted[:35]:
            if grad[i] < 1e-11:
                break
            for j in neg_sorted[:35]:
                if grad[j] > -1e-11:
                    break
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                d = best_c - c
                if d > best_delta:
                    best_delta = d
                    best_i = i
                    best_j = j
                h[i] = 1.0
                h[j] = 0.0

        if best_delta > 1e-15 and best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_c -= best_delta
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            stuck = 0
        else:
            stuck += 1
            if stuck > 300:
                for _ in range(rng.randint(5, 15)):
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    h[ii] = 0.0
                    h[jj] = 1.0
                    c2 = compute_c5(h, n)
                    if c2 < best_c:
                        best_c = c2
                        ones.remove(ii)
                        ones.append(jj)
                        zeros.remove(jj)
                        zeros.append(ii)
                    else:
                        h[ii] = 1.0
                        h[jj] = 0.0
                stuck = 0

    return h, best_c

def lbfgs_soft(h, n, deadline, alpha_val, mu_val, maxit):
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha_val)
        if mu_val > 0:
            val += mu_val * (x.sum() - n / 2.0) ** 2
        return val

    def grad(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha_val)
        g = h_grad_from_w(w, x, n)
        if mu_val > 0:
            g += 2.0 * mu_val * (x.sum() - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-20,
                                'gtol': 1e-16, 'maxfun': maxit * 10,
                                'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def soften_positions(h, n, deadline, num_pos=25):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    w = smooth_max_w(c_vec, 1500.0)
    grad_all = h_grad_from_w(w, h, n)

    positions = np.argsort(-np.abs(grad_all))[:num_pos]

    for pos in positions:
        if time.time() > deadline:
            break
        orig = h[pos]
        for dv in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.04, -0.04, 0.002, -0.002]:
            nv = orig + dv
            if nv < 0 or nv > 1:
                continue
            h[pos] = nv
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[pos] = orig

    # Second pass with finer steps
    for pos in positions[:15]:
        if time.time() > deadline:
            break
        orig = h[pos]
        for dv in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.003, -0.003]:
            nv = orig + dv
            if nv < 0 or nv > 1:
                continue
            h[pos] = nv
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
            else:
                h[pos] = orig

    return h, best_c
