import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

    # For n samples, take the first n terms
    result = np.zeros(n)
    for i in range(n):
        result[i] = bin(i).count('1') % 2
    return result

    # Thue-Morse: t(i) = popcount(i) mod 2
    for i in range(n):
        result[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    # Adjust to have exactly n/2 ones
    target = n // 2
    count = int(result.sum())
    if count > target:
        # Need to flip some 1s to 0s
        ones = np.where(result == 1)[0]
        excess = count - target
        # Flip the ones that are in "low" Thue-Morse positions
        to_flip = ones[-excess:]  # or some heuristic
        result[to_flip] = 0.0
    elif count < target:
        zeros = np.where(result == 0)[0]
        deficit = target - count
        to_flip = zeros[-deficit:]
        result[to_flip] = 1.0
    return result

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def swap_delta_fast(corr, h, i, j, n):
    """Compute change in correlation vector when h[i]:0->1, h[j]:1->0."""
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1)
    
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    
    delta[n-1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    
    return delta

def sim_anneal_binary(h, n, deadline, seed):
    """Simulated annealing on binary h to minimize max correlation."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = compute_corr(h, n)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    
    T_init = 2.0
    T_final = 0.01
    n_steps_target = 50000
    
    # Adaptive: estimate time per step
    # Each step: 1 swap evaluation + update
    # Let's just run until deadline
    
    step = 0
    T = T_init
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    while time.time() < deadline:
        step += 1
        # Anneal temperature
        frac = min(1.0, step / n_steps_target)
        T = T_init * (T_final / T_init) ** frac
        
        # Pick random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(list(zeros))
        j = rng.choice(list(ones))
        
        delta = swap_delta_fast(corr, h, i, j, n)
        new_corr = corr + delta
        new_max = new_corr.max()
        delta_max = new_max - cur_c
        
        if delta_max <= 0 or rng.random() < np.exp(-delta_max / max(T, 1e-10)):
            # Accept
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = new_max
            ones.discard(j)
            ones.add(i)
            zeros.discard(i)
            zeros.add(j)
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    
    return best_h, best_c * dx

def local_search_binary(h, n, deadline, seed):
    """Steepest descent local search: try all swaps, pick best."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = compute_corr(h, n).copy()
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    m_range = np.arange(-(n-1), n)
    
    while time.time() < deadline:
        improved = False
        best_delta_max = 0  # We want most negative (improvement)
        best_i, best_j = -1, -1
        best_new_corr = None
        
        # Try all swaps
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                # Compute delta
                delta = np.zeros(2*n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n-1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                
                new_max = np.max(corr + delta)
                dm = new_max - cur_c
                if dm < best_delta_max - 1e-15:
                    best_delta_max = dm
                    best_i, best_j = i, j
        
        if best_i >= 0:
            # Apply swap
            i, j = best_i, best_j
            delta = np.zeros(2*n - 1)
            im = i + m_range
            msk = (im >= 0) & (im < n)
            delta[msk] += 1.0 - h[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            delta[msk] -= h[imn[msk]]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            delta[msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
