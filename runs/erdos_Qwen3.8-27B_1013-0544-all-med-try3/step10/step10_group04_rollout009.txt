def smooth_objective(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    dx = 2.0 / n
    m = c.max() * dx
    sh = alpha * (c * dx - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_gradient(x, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    cd = c * dx
    m = cd.max()
    sh = alpha * (cd - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx]
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad *= dx
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, configs=None):

    if configs is None:

            h_new = project_feasible(res.x, n)

            if c_new < best_c:

        except Exception:
            pass
