            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # Compute per-pair contributions
            if lag > 0:
                pairs = np.arange(n - lag)
                contrib = h[pairs] * (1.0 - h[pairs + lag])
            else:
                idx = -lag
                pairs = np.arange(idx, n)
                contrib = h[pairs] * (1.0 - h[pairs - idx])
            
            # Try improving the worst pairs
            n_worst = min(20, len(contrib))
            worst_idx = np.argsort(contrib)[-n_worst:]
            
            for wi in worst_idx:
                if lag > 0:
                    i = pairs[wi]
                    j = i + lag
                else:
                    i = pairs[wi]
                    j = i - (-lag)
                
                # Move mass from higher to lower within the pair
                if h[i] > h[j] + 1e-8:
                    for frac in [0.3, 0.5, 0.2, 0.1, 0.4]:
                        d = frac * (h[i] - h[j])
                        if d < 1e-10:
                            continue
                        h[i] -= d
                        h[j] += d
                        h_tmp = project_feasible(h, n)
                        c_tmp = compute_c5(h_tmp, n)
                        if c_tmp < best_c - 1e-15:
                            best_c = c_tmp
                            best_h = h_tmp.copy()
                            h = h_tmp
                            improved = True
                            stall = 0
                            break
                        else:
                            h = h.copy()  # revert
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    for frac in [0.3, 0.5, 0.2, 0.1, 0.4]:
                        d = frac * (h[j] - h[i])
                        if d < 1e-10:
                            continue
                        h[i] += d
                        h[j] -= d
                        h_tmp = project_feasible(h, n)
                        c_tmp = compute_c5(h_tmp, n)
                        if c_tmp < best_c - 1e-15:
                            best_c = c_tmp
                            best_h = h_tmp.copy()
                            h = h_tmp
                            improved = True
                            stall = 0
                            break
                        else:
                            h = h.copy()
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
        
        if not improved:
            stall += 1
            if stall > 50:
                # Add small random perturbation to escape
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                c_check = compute_c5(h, n)
                if c_check > best_c + 1e-8:
                    h = best_h.copy()
                stall = 0
            if stall > 200:
                break
    
    return best_h, best_c

def barrier_method(h, n, deadline, initial_T=None, reduction=0.98, max_iters=5000):
    """Barrier/penalty method: penalize correlations above threshold T, gradually lower T."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    if initial_T is None:
        c_all = compute_all_lags(h, n)
        T = c_all.max() + 0.001
    else:
        T = initial_T
    
    penalty = 1e6
    target_sum = n / 2.0
    
    def barrier_obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        excess = np.maximum(0.0, c - T)
        obj = penalty * np.sum(excess * excess)
        diff = x.sum() - target_sum
        obj += 1e8 * diff * diff
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if c[k] <= T:
                continue
            w_k = 2.0 * penalty * (c[k] - T)
            lag = k - (n - 1)
            if lag == 0:
                grad += w_k * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w_k * (1.0 - x[0:n - lag])
                grad[0:n - lag] += w_k * (-x[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w_k * (1.0 - x[idx:n])
                grad[idx:n] += w_k * (-x[0:n - idx])
        grad += 2.0 * 1e8 * diff * np.ones(n)
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    
    for phase in range(30):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5:
            break
        
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(barrier_obj_and_grad, h.copy(), method='L-BFGS-B',
                           bounds=bounds, jac=True,
                           options={'maxiter': min(maxit, 2000), 'ftol': 1e-20,
                                    'maxls': 30, 'iprint': -1})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        
        # Reduce T
        c_all = compute_all_lags(h, n)
        T_new = max(c_all.max() * 0.9999, T * reduction)
        if T_new < c_all.max() - 0.0001:
            T = T_new
    
    return best_h, best_c

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)

        best_c = compute_c5(best_h, n_init)

    # Phase 1: Multi-alpha L-BFGS-B (budget: ~180s)
    alpha_schedule = [
        (5000, 2000), (20000, 2000), (100000, 2000), (500000, 2000),
        (2000000, 2000), (10000000, 2000), (50000000, 2000), (200000000, 2000)
    ]
    
    for alpha, maxit in alpha_schedule:
        if time.time() > deadline - 400:
            break
        t_phase = time.time()
        h_opt = lbfgsb_run(best_h, n, alpha, time.time() + min(30, (deadline - 400 - t0 - (t_phase - t0)) * 0.3), maxiter)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Gradient descent on top lags (budget: ~150s)
    if time.time() < deadline - 250:
        gd_deadline = min(deadline - 180, time.time() + 120)
        h_opt, c_opt = gradient_descent_top_lags(best_h, n, gd_deadline, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Pair exchange search (budget: ~200s)
    if time.time() < deadline - 200:
        pe_deadline = min(deadline - 120, time.time() + 180)
        h_opt, c_opt = pair_exchange_search(best_h, n, pe_deadline, top_k=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: L-BFGS-B polish after local search
    if time.time() < deadline - 100:
        for alpha in [10000000, 100000000, 1000000000]:
            if time.time() > deadline - 50:
                break
            h_opt = lbfgsb_run(best_h, n, alpha, time.time() + 25, maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 5: Barrier method
    if time.time() < deadline - 100:
        bar_deadline = time.time() + min(80, deadline - time.time() - 50)
        h_opt, c_opt = barrier_method(best_h, n, bar_deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 6: Try higher resolution
    for n_try in [700, 1000]:
        if time.time() > deadline - 120:
            break
        remaining = deadline - 80 - time.time()
        if remaining < 30:
            break
        h_interp = expand_resolution(best_h, len(best_h), n_try)
        # Optimize at new resolution
        t_ph = time.time()
        h_opt = lbfgsb_run(h_interp, n_try, 50000, time.time() + min(20, remaining * 0.3), maxiter=2000)
        h_opt = lbfgsb_run(h_opt, n_try, 5000000, time.time() + min(15, remaining * 0.2), maxiter=2000)
        h_opt = lbfgsb_run(h_opt, n_try, 50000000, time.time() + min(10, remaining * 0.15), maxiter=2000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            # Quick local search at this resolution
            if time.time() < deadline - 60:
                ls_dl = min(deadline - 40, time.time() + 30)
                h_opt2, c_opt2 = pair_exchange_search(best_h, n_try, ls_dl, top_k=5)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2
    
    # Phase 7: More gradient descent at best resolution
    if time.time() < deadline - 40:
        n_final = len(best_h)
        gd_dl = min(deadline - 20, time.time() + 40)
        h_opt, c_opt = gradient_descent_top_lags(best_h, n_final, gd_dl, top_k=3, step_sizes=[5e-4, 2e-4, 1e-4, 5e-5, 1e-5])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 8: Final pair exchange
    if time.time() < deadline - 15:
        n_final = len(best_h)
        pe_dl = min(deadline - 5, time.time() + 30)
        h_opt, c_opt = pair_exchange_search(best_h, n_final, pe_dl, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 9: Final L-BFGS-B with very high alpha
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt = lbfgsb_run(best_h, n_final, 1000000000, deadline - 2, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:

            best_h = h_opt
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
