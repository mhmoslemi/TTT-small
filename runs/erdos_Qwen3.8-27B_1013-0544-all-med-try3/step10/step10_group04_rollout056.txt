def topk_grad(x, n, alpha, mu, topk):
    c = compute_c5_vec(x, n)
    k_idx = np.argpartition(c, -topk)[-topk:]
    c_top = c[k_idx]
    m = c_top.max()
    sh = alpha * (c_top - m)
    mx = sh.max()
    w_top = np.exp(sh - mx)
    w_top /= w_top.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for j in range(topk):
        idx = k_idx[j]
        if w_top[j] < 1e-12:
            continue
        k = idx - center
        wm = w_top[j] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def make_obj_grad(n, alpha, mu, topk=0):
    if topk > 0:
        def obj(x):
            return topk_obj(x, n, alpha, mu, topk)
        def grad(x):
            return topk_grad(x, n, alpha, mu, topk)
    else:
        def obj(x):
            return smooth_obj(x, n, alpha, mu)
        def grad(x):
            return smooth_grad(x, n, alpha, mu)
    return obj, grad

def run_lbfgs(h, n, deadline, configs=None, topk=0):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),

            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
        # After full-max optimization, try top-k focused

                               bounds=bounds,

                    h = h_new2.copy()
            except Exception:

    return h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_local_search(h, n, deadline, max_iters=50000):
    """Local search on binary (or near-binary) solutions via swaps."""
    target = n / 2.0
    h_bin = np.where(h > 0.5, 1.0, 0.0).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones > n / 2:
        # Remove excess ones
        idx = np.where(h_bin == 1.0)[0]
        order = idx[np.argsort(h[idx])]  # remove ones with lowest h values
        for i in range(n_ones - n//2):
            h_bin[order[i]] = 0.0
    elif n_ones < n / 2:
        idx = np.where(h_bin == 0.0)[0]
        order = idx[np.argsort(-h[idx])]  # add zeros with highest h values
        for i in range(n//2 - n_ones):
            h_bin[order[i]] = 1.0
    
    best_c = compute_c5(h_bin, n)
    rng = np.random.RandomState(777)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick random 1 and random 0, try swapping
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c = compute_c5(h_bin, n)
        if c < best_c - 1e-15:
            best_c = c
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    # Project for feasibility (should already be exact for binary)
    return project_feasible(h_bin, n), best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_init(n, seed=0):
    """Create a structured initial guess using Fourier modes."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    x = np.linspace(0, 2, n)
    # Mix of sinusoidal patterns
    for freq in [1, 2, 3, 4, 5]:
        amp = 0.1 / freq
        h += amp * np.sin(np.pi * freq * x + rng.rand() * 2 * np.pi)
    # Add a base level
    h += 0.5
    return project_feasible(h, n)

def random_binary_init(n, seed=0):
    """Create a random binary initial guess."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n//2]] = 1.0
    return project_feasible(h, n)

    target = n / 2.0

            if tot < 1e-14:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def binary_local_search(h, n, deadline, max_iters=80000, seed=999):
    target = n // 2
    h_bin = np.where(h > 0.5, 1.0, 0.0).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h_bin == 1.0)[0]
            order = idx[np.argsort(h[idx])]
            for i in range(n_ones - target):
                h_bin[order[i]] = 0.0
        else:
            idx = np.where(h_bin == 0.0)[0]
            order = idx[np.argsort(-h[idx])]
            for i in range(target - n_ones):
                h_bin[order[i]] = 1.0
    best_c = compute_c5(h_bin, n)
    rng = np.random.RandomState(seed)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c = compute_c5(h_bin, n)
        if c < best_c - 1e-15:
            best_c = c
            no_improve = 0
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1
        if it > 1000 and no_improve > 20000:
            break
    return project_feasible(h_bin, n), best_c

def momentum_optimize(h, n, deadline, lr=0.001, momentum=0.9, alpha=5000, mu=1000.0, topk=0):
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    it = 0
    no_improve = 0
    while time.time() < deadline - 5:
        if topk > 0:
            g = topk_grad(h, n, alpha, mu, topk)
        else:
            g = smooth_grad(h, n, alpha, mu)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            h = h_new
            no_improve = 0
            # Adaptive lr: slow down if improving slowly
            if it > 500 and np.abs(c_new - compute_c5(h - v * 10, n)) < 1e-20:
                lr *= 0.999
        else:
            h = h  # keep old
            no_improve += 1
            lr *= 0.9995
            v *= 0.9
        it += 1
        if it > 1000 and no_improve > 5000:
            break
    return project_feasible(h, n), best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for freq in [1, 2, 3, 5, 7]:
        amp = 0.08 / np.sqrt(freq)
        h += amp * np.sin(np.pi * freq * x + rng.rand() * 2 * np.pi)
    return project_feasible(np.clip(h, 0, 1), n)

def random_binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1.0
    return project_feasible(h, n)

def targeted_refine(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for round_num in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        dx_val = 2.0 / n
        found = False
        if k_worst >= 0:
            affected = list(range(0, n - k_worst)) + list(range(k_worst, n))
        else:
            affected = list(range(-k_worst, n)) + list(range(0, n + k_worst))
        affected = list(set(affected))
        rng_order = np.random.permutation(len(affected))
        for ri in rng_order:
            if time.time() > deadline:
                return best_h, best_c
            i = affected[ri]
            for delta in [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005, 0.005, -0.005]:
                nv = h[i] + delta
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                old_val = h[i]
                h[i] = nv
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = old_val
            if found:
                h = best_h.copy()
                break
        if not found and round_num > 50:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)

    # Phase 1: L-BFGS full smooth max at base resolution (90s)
    t1 = min(t0 + 90, deadline - 950)
    if t1 > time.time() + 5:
        best_h, best_c = run_lbfgs_full(best_h, n_base, t1,
            configs=[(50, 10.0), (200, 50.0), (1000, 200.0),
                     (5000, 1000.0), (20000, 5000.0), (60000, 20000.0)])

    # Phase 2: Top-k focused at base (50s)
    t2 = min(t0 + 140, deadline - 900)
    if t2 > time.time() + 5:
        h_tk, c_tk = run_lbfgs_topk(best_h, n_base, t2,
            configs=[(5000, 2000.0), (20000, 5000.0), (80000, 20000.0)], topk=5)
        if c_tk < best_c:
            best_c = c_tk
            best_h = h_tk

    # Phase 3: SLSQP polish at base (40s)
    t3 = min(t0 + 180, deadline - 860)
    if t3 > time.time() + 5:
        h3, c3 = slqp_polish(best_h, n_base, t3, alpha=8000000, maxiter=4000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Alternative starts (60s)
    t4 = min(t0 + 240, deadline - 800)
    if t4 > time.time() + 15:
        h_struct = structured_init(n_base, seed=seed)
        h_s, c_s = run_lbfgs_full(h_struct, n_base, min(time.time()+20, t4-30),
            configs=[(500, 200.0), (5000, 2000.0), (30000, 10000.0)])
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
        if time.time() < t4 - 5:
            h_rb = random_binary_init(n_base, seed=seed+100)
            h_r, c_r = run_lbfgs_topk(h_rb, n_base, t4,
                configs=[(1000, 500.0), (10000, 5000.0)], topk=8)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 5: Momentum optimization at base (40s)
    t5 = min(t0 + 280, deadline - 760)
    if t5 > time.time() + 10:
        h_m, c_m = momentum_optimize(best_h, n_base, t5,
            lr=0.002, momentum=0.85, alpha=20000, mu=5000.0, topk=5)
        if c_m < best_c:
            best_c = c_m
            best_h = h_m

    # Phase 6: Upsample to n=1000 (180s)
    n1 = 1000
    if time.time() < deadline - 380:
        h1000 = interpolate_to(best_h, len(best_h), n1)
        t6 = min(time.time() + 100, deadline - 200)
        if t6 > time.time() + 10:
            h1000, c1000 = run_lbfgs_full(h1000, n1, t6,
                configs=[(200, 50.0), (1000, 200.0), (5000, 1000.0),
                         (30000, 5000.0), (100000, 20000.0)])
            if c1000 < best_c:
                best_c = c1000
                best_h = h1000
        # Top-k at n=1000
        t6b = min(time.time() + 50, deadline - 130)
        if t6b > time.time() + 8:
            h_tk2, c_tk2 = run_lbfgs_topk(best_h, n1, t6b,
                configs=[(10000, 3000.0), (50000, 10000.0), (200000, 30000.0)], topk=7)
            if c_tk2 < best_c:
                best_c = c_tk2
                best_h = h_tk2
        # SLSQP at n=1000
        t6c = min(time.time() + 40, deadline - 80)
        if t6c > time.time() + 8:
            h6c, c6c = slqp_polish(best_h, n1, t6c, alpha=30000000, maxiter=3000)
            if c6c < best_c:
                best_c = c6c
                best_h = h6c

    # Phase 7: Upsample to n=2000 (200s)
    n2 = 2000
    if time.time() < deadline - 220:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        t7 = min(time.time() + 110, deadline - 100)
        if t7 > time.time() + 10:
            h2000, c2000 = run_lbfgs_topk(h2000, n2, t7,
                configs=[(2000, 500.0), (10000, 2000.0), (50000, 5000.0),
                         (200000, 20000.0)], topk=8)
            if c2000 < best_c:
                best_c = c2000
                best_h = h2000
        t7b = min(time.time() + 50, deadline - 40)
        if t7b > time.time() + 8:
            h7b, c7b = slqp_polish(best_h, n2, t7b, alpha=80000000, maxiter=2000)
            if c7b < best_c:
                best_c = c7b
                best_h = h7b

    # Phase 8: Binary local search (50s)
    if time.time() < deadline - 70:
        t8 = min(time.time() + 40, deadline - 25)
        h8, c8 = binary_local_search(best_h, len(best_h), t8, max_iters=60000, seed=seed+200)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        # Quick polish after binary
        if time.time() < deadline - 15:
            h8b, c8b = run_lbfgs_topk(best_h, len(best_h), deadline - 10,
                configs=[(50000, 10000.0)], topk=6)
            if c8b < best_c:
                best_c = c8b
                best_h = h8b

    # Phase 9: Final targeted refinement
    if time.time() < deadline - 10:
        best_h, best_c = targeted_refine(best_h, len(best_h), deadline - 5)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
