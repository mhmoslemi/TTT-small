import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            # Scale down non-zero elements
            idx = h > 0
            ds = target - s
            if h[idx].sum() < 1e-14:

            room = 1.0 - h[idx]

    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    # Log-sum-exp for smooth max

    sm = sh.max()

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0

    """Project to binary with exactly n//2 ones (for even n)."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def fast_binary_search(h_init, n, deadline, max_sweeps=100000, 
                        batch_size=50, seed=42, sa_temp=0.001, sa_cooling=0.9999):
    """
    Fast binary local search using vectorized A-matrix evaluation.
    h_init should be binary (0s and 1s) with exactly n/2 ones.
    """
    dx = 2.0 / n
    h = h_init.copy()
    h = binary_project(h, n)  # Ensure binary
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    temp = sa_temp
    prev_best_c = cur_c
    no_improve = 0
    
    d = np.arange(2*n-1) - (n-1)  # (2n-1,)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Get current one/zero positions
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Build A matrix (effect of setting position i from 0 to 1)
        # A[i, m] = (1-h[i-d[m]]) - h[i+d[m]] (with boundary handling)
        II = np.arange(n)[:, None]
        DD = d[None, :]
        idx1 = II - DD
        idx2 = II + DD
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        idx1_c = np.clip(idx1, 0, n-1)
        idx2_c = np.clip(idx2, 0, n-1)
        A = (1.0 - h[idx1_c]) * mask1 - h[idx2_c] * mask2  # (n, 2n-1)
        
        # Sample subsets for evaluation
        n_z = len(zeros)
        n_o = len(ones)
        bz = min(batch_size, n_z)
        bo = min(batch_size, n_o)
        zh = rng.choice(zeros, size=bz, replace=False)
        oh = rng.choice(ones, size=bo, replace=False)
        
        # For each i in zh, evaluate all j in oh
        # new_corr = corr + A[i] - A[j]
        # new_max = max(new_corr)
        
        best_i, best_j = -1, -1
        best_new_c = cur_c + temp * rng.standard_exponential()  # SA: allow slightly worse
        # Actually for pure local search, best_new_c should be cur_c (only accept improvements)
        # For SA, we allow some worse moves
        
        # Vectorized evaluation
        # For each i in zh: compute candidate_new_maxs over oh
        zh_A = A[zh]  # (bz, 2n-1)
        oh_A = A[oh]  # (bo, 2n-1)
        
        # new_max for swap (zh[p], oh[q]):
        # max(corr + zh_A[p] - oh_A[q])
        # = max((corr + zh_A[p])[None, :] - oh_A, axis=1)
        
        found = False
        for p in range(bz):
            base = corr + zh_A[p]  # (2n-1,)
            # For all q: new_max = (base[None,:] - oh_A).max(axis=1)
            candidate_maxs = (base[None, :] - oh_A).max(axis=1)  # (bo,)
            q_best = candidate_maxs.argmin()
            c_val = candidate_maxs[q_best]
            if c_val < best_new_c:
                best_new_c = c_val
                best_i = zh[p]
                best_j = oh[q_best]
                found = True
        
        if found and best_new_c < cur_c:  # Only accept if actually improves (or SA)
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr += A[best_i] - A[best_j]
            cur_c = float(corr.max() * dx)
            temp *= sa_cooling
            
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Random perturbation (restart-like)
                pert = rng.randint(3, 8)
                for _ in range(pert):
                    oi = np.where(h == 1.0)[0]
                    zi = np.where(h == 0.0)[0]
                    if len(oi) > 0 and len(zi) > 0:
                        a = rng.choice(oi)
                        b = rng.choice(zi)
                        h[a], h[b] = h[b], h[a]
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = float(corr.max() * dx)
                no_improve = 0
                temp = max(temp, 0.005)
        
        # Also try pure SA acceptance sometimes
        # (already handled by best_new_c threshold)
    
    return best_h, best_c

def slsqp_soft(h, n, alpha, maxiter, deadline):
    """SLSQP with log-sum-exp soft-max."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return float(m + (sm + np.log(np.exp(sh - sm).sum())) / alpha)
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
