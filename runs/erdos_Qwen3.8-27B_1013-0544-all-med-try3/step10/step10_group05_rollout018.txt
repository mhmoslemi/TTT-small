    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h[np.argmax(1-h)] += max(0, diff)

            tot = h.sum()
            if tot < 1e-15:

                          args=(n, active, alpha, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 5,
                                   'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Subgradient method
    p2_end = min(time.time() + 150, deadline - 500)
    if p2_end > time.time() + 30:
        step_size = 0.1
        best_sg_c = best_c
        for _ in range(30000):
            if time.time() > p2_end - 5:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > 20:
                active = np.argsort(-c_vec)[np.argsort(-c_vec[active])[:20]]
            h_new = subgradient_step(best_h, n, c_vec, active, step_size, project_feasible)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                step_size = max(0.005, step_size * 0.998)
            else:
                step_size *= 0.93
                if step_size < 0.002:
                    break

    # Phase 3: Another L-BFGS round with tighter active set
    p3_end = min(time.time() + 150, deadline - 350)
    if p3_end > time.time() + 20:
        for alpha in [50000, 200000, 500000]:
            if time.time() > p3_end - 20:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > 12:
                active = np.argsort(-c_vec)[np.argsort(-c_vec[active])[:12]]
            rem = p3_end - time.time() - 10
            if rem < 15:
                break
            maxit = min(800, max(100, int(rem * 3)))
            try:
                res = minimize(active_obj, best_h, jac=active_obj_grad,
                              args=(n, active, alpha, 0.002), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 4: Targeted pair search
    p4_end = min(time.time() + 200, deadline - 200)
    if p4_end > time.time() + 30:
        best_h, best_c = targeted_pair_search(best_h, n, p4_end)

    # Phase 5: Fine local search
    p5_end = min(time.time() + 150, deadline - 80)
    if p5_end > time.time() + 20:
        best_h, best_c = fine_local_search(best_h, n, p5_end)

    # Phase 6: Another targeted pass
    p6_end = min(time.time() + 80, deadline - 30)
    if p6_end > time.time() + 15:
        best_h, best_c = targeted_pair_search(best_h, n, p6_end)

    # Phase 7: Final subgradient polish
    p7_end = deadline - 3
    if p7_end > time.time() + 10:
        step_size = 0.005
        for _ in range(10000):
            if time.time() > p7_end - 2:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > 8:
                active = np.argsort(-c_vec)[np.argsort(-c_vec[active])[:8]]
            h_new = subgradient_step(best_h, n, c_vec, active, step_size, project_feasible)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                step_size = max(0.0005, step_size * 0.999)
            else:
                step_size *= 0.85
                if step_size < 0.0001:
                    break

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
