import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_for_lag(k, h, n):
    """Compute gradient of c(k) with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) * dx
        grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx
        grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        grad += w[mm] * compute_grad_for_lag(k, h, n)
    return grad

# Full smooth max objective
def full_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    return smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2

def full_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    grad = smooth_max_grad(c, alpha, h, n)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

# Iterated linearization
def solve_lp_active(h, n, active_lags, t_current):
    """Solve LP: min t s.t. c_lin_k(h') <= t for k in active, 0<=h'<=1, sum(h')=n/2"""
    dx = 2.0 / n
    # Linearize each active lag
    # c_lin_k = c(k) + grad_k^T (h' - h)
    # = c(k) + grad_k^T h' - grad_k^T h
    # We want: c(k) + grad_k^T h' - grad_k^T h <= t
    # => grad_k^T h' - t <= grad_k^T h - c(k)
    
    m = len(active_lags)
    # Variables: h[0], ..., h[n-1], t
    # Total: n+1 variables
    
    # Objective: minimize t = variable at index n
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_in @ x <= b_in
    # For each active lag k:
    #   grad_k^T h' - t <= grad_k^T h - c(k)
    
    A_rows = []
    b_rows = []
    
    c_vec = compute_c5_vec(h, n)
    
    for k_idx in active_lags:
        k = k_idx - (n - 1)
        g = compute_grad_for_lag(k, h, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(float(g @ h - c_vec[k_idx]))
    
    A_in = np.array(A_rows)
    b_in = np.array(b_rows)
    
    # Equality constraint: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h'[i] <= 1, t free (but bounded below by 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_in, b_ub=b_in, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        else:

            if tot < 1e-16:

    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:

    return m + np.log(np.sum(np.exp(sh))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_grad_full(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_full(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    return smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2

def grad_full(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    g = smooth_max_grad_full(c, alpha, h, n)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(np.floor(k * alpha_param)) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def compute_subgradient(h, n, active_idx):
    """Compute average gradient of active lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for idx in active_idx:
        k = int(idx) - idx_center
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
    return grad / max(len(active_idx), 1)

def lbfgs_optimize(h, n, deadline, alpha, mu=0.01, maxit_cap=2000):
    """Run L-BFGS optimization and return (h, c)."""
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time() - 3
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = min(max(50, int(rem * 4)), maxit_cap)
    try:
        res = minimize(obj_full, h, jac=grad_full,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)
