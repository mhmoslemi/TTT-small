                h[i] -= d
                h[j] += d
            elif h[j] > h[i] + 1e-6:

                    continue
                h[i] += d
                h[j] -= d
            else:
                T *= cool
                continue
        else:

            d = rng.uniform(-0.01, 0.01)

            h[i] = np.clip(old + d, 0, 1)
            h = project_feasible(h, n)

                    h = project_feasible(h, n)

            continue

        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()

                    else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxfev': maxiter * 3,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

            if h[i] > h[j] + 1e-6:
                d = min(0.003 + T * 2, 0.03) * min(1.0, (h[i] - h[j]) * 3 + 0.3)
                d = min(d, (h[i] - h[j]) * 0.25)
                if d < 1e-8:
                    T *= cool
                    continue
                h[i] -= d
                h[j] += d
            elif h[j] > h[i] + 1e-6:
                d = min(0.003 + T * 2, 0.03) * min(1.0, (h[j] - h[i]) * 3 + 0.3)
                d = min(d, (h[j] - h[i]) * 0.25)
                if d < 1e-8:
                    T *= cool
                    continue
                h[i] += d
                h[j] -= d
            else:
                T *= cool
                continue

        elif r < 0.90:
            # Block perturbation: modify a contiguous block
            block_len = rng.randint(3, min(30, n // 10))
            start = rng.randint(0, n - block_len + 1)
            # Perturb block with a smooth wave
            t = np.linspace(0, 2 * np.pi, block_len)
            amp = 0.005 + T * 2
            h[start:start + block_len] += amp * np.sin(t) * rng.uniform(0.5, 1.0)
            h = np.clip(h, 0, 1)
            h = project_feasible(h, n)

        else:
            # Random single coordinate perturbation
            i = rng.randint(n)
            d = rng.uniform(-0.02, 0.02)
            old = h[i]
            h[i] = np.clip(old + d, 0, 1)
            h = project_feasible(h, n)

        c_new = compute_c5(h, n)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
            it_since_improve = 0
        else:
            # Accept with probability
            c_cur = compute_c5(h, n)  # should be same as c_new
            de = c_cur - best_c
            if de > 0:
                ratio = de / max(T, 1e-10)
                if ratio < 25 and rng.random() < np.exp(-ratio):
                    pass  # accept
                else:
                    # Revert - we need to undo the move
                    # Since we modified in place, just continue with current state
                    # (gradually drifts)
                    pass
            it_since_improve += 1

        T *= cool

        # Restart if stuck
        if it_since_improve > max_stuck:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.002)

    return best_h, best_c

def sweep_local_search(h, n, deadline, top_k=8, step_sizes=None):
    """Systematic sweep: for each of the top lags, try adjusting pairs."""
    h = h.copy()
    best_c = compute_c5(h, n)

    if step_sizes is None:
        step_sizes = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]

    for step in step_sizes:
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]

        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue

            for i in range(0, n - m, max(1, (n - m) // 500)):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 0.005:
                    d = min(step, (h[i] - h[j]) * 0.2)
                    if d < 1e-7:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-13:
                        best_c = c_test
                    else:
                        h[i], h[j] = old_i, old_j
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 0.005:
                    d = min(step, (h[j] - h[i]) * 0.2)
                    if d < 1e-7:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-13:
                        best_c = c_test
                    else:
                        h[i], h[j] = old_i, old_j
                        h = project_feasible(h, n)

        # Re-fetch top lags after this step size round
        if time.time() > deadline:
            break

    return h, best_c

def multi_pair_local(h, n, deadline, top_k=5):
    """Try pairs of coordinates at top lags with adaptive steps."""
    h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0

    for outer in range(200):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False

        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue

            # Find the worst pair at this lag
            c_lag = np.zeros(n - m)
            for i in range(n - m):
                c_lag[i] = h[i + m] * (1 - h[i]) + h[i] * (1 - h[i + m])
            # Actually for the full correlate, at positive lag m:
            # contribution at position i is h[i+m]*(1-h[i])
            # We want to reduce the total, so target the largest individual terms

            h_lag = h[m:n] * (1 - h[0:n - m])
            # Also for negative lag contribution at this offset:
            h_lag2 = h[0:n - m] * (1 - h[m:n])
            total_lag = h_lag + h_lag2

            # Find position with highest contribution
            if total_lag.max() > 0:
                i_worst = int(np.argmax(total_lag))
                j = i_worst + m
                if h[i_worst] > h[j] + 0.01:
                    for d_frac in [0.1, 0.05, 0.02, 0.01]:
                        d = (h[i_worst] - h[j]) * d_frac
                        if d < 1e-7:
                            continue
                        old_i, old_j = h[i_worst], h[j]
                        h[i_worst] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-13:
                            best_c = c_test
                            improved = True
                            break
                        else:
                            h[i_worst], h[j] = old_i, old_j
                            h = project_feasible(h, n)
                elif h[j] > h[i_worst] + 0.01:
                    for d_frac in [0.1, 0.05, 0.02, 0.01]:
                        d = (h[j] - h[i_worst]) * d_frac
                        if d < 1e-7:
                            continue
                        old_i, old_j = h[i_worst], h[j]
                        h[i_worst] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-13:
                            best_c = c_test
                            improved = True
                            break
                        else:
                            h[i_worst], h[j] = old_i, old_j
                            h = project_feasible(h, n)

        if not improved and outer > 20:
            break

    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)

    # Phase 1: L-BFGS-B with increasing alpha
    alphas_p1 = [300, 1000, 5000, 20000, 80000, 300000, 1000000, 5000000]
    for idx, alpha in enumerate(alphas_p1):
        if time.time() > deadline - 500:
            break
        time_per = min(35, (deadline - 500 - time.time()) / max(1, len(alphas_p1) - idx))
        if time_per < 8:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + time_per,
                                maxiter=2000, penalty_coeff=1e8)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Gradient-informed SA
    if time.time() < deadline - 380:
        sa_end = min(deadline - 330, time.time() + 180)
        h_opt, c_opt = gradient_sa(best_h, n, sa_end, rng,
                                   alpha=100000, t_init=0.004, t_min=0.00001,
                                   cool=0.99995, max_stuck=40000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: L-BFGS-B polish
    if time.time() < deadline - 250:
        h_opt = lbfgsb_optimize(best_h, n, 2000000, time.time() + 30, maxiter=2500)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 220:
            h_opt = lbfgsb_optimize(best_h, n, 20000000, time.time() + 30, maxiter=2500)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Multi-pair local search
    if time.time() < deadline - 150:
        h_opt, c_opt = multi_pair_local(best_h, n, min(deadline - 100, time.time() + 80), top_k=6)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Sweep local search
    if time.time() < deadline - 100:
        h_opt, c_opt = sweep_local_search(best_h, n, min(deadline - 60, time.time() + 80),
                                          top_k=8, step_sizes=[0.012, 0.006, 0.003, 0.0015, 0.0008])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try higher resolution
    for n_try in [n + 100, n * 2]:
        if time.time() > deadline - 80:
            break
        if n_try <= n or n_try < 300:
            continue
        remaining = deadline - 70 - time.time()
        if remaining < 25:
            break
        h_interp = interpolate_h(best_h, n, n_try)
        c_interp = compute_c5(h_interp, n_try)
        t_sub = min(20, remaining * 0.45)
        h_opt = lbfgsb_optimize(h_interp, n_try, 50000, time.time() + t_sub, maxiter=1500)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try
            if time.time() < deadline - 40:
                h_opt = lbfgsb_optimize(best_h, n, 5000000, time.time() + min(15, deadline - 40 - time.time()), maxiter=2000)
                c_opt = compute_c5(h_opt, n)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 7: Try lower resolution
    for n_try in [max(200, n // 2), n - 50]:
        if time.time() > deadline - 60:
            break
        if n_try >= n or n_try < 100:
            continue
        remaining = deadline - 50 - time.time()
        if remaining < 20:
            break
        h_interp = interpolate_h(best_h, n, n_try)
        t_sub = min(15, remaining * 0.5)
        h_opt = lbfgsb_optimize(h_interp, n_try, 100000, time.time() + t_sub, maxiter=1500)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try

    # Phase 8: Another SA round (lower temperature)
    if time.time() < deadline - 50:
        n_final = len(best_h)
        sa_end2 = min(deadline - 30, time.time() + 40)
        h_opt, c_opt = gradient_sa(best_h, n_final, sa_end2, rng,
                                   alpha=300000, t_init=0.0015, t_min=0.00001,
                                   cool=0.99998, max_stuck=30000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final L-BFGS-B + local search
    if time.time() < deadline - 20:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, min(deadline - 10, time.time() + 20), maxiter=3000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt, c_opt = sweep_local_search(best_h, n_final, deadline - 2, top_k=5,
                                          step_sizes=[0.004, 0.002, 0.001, 0.0005])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None or len(best_h) == 0:
        best_h = project_feasible(np.full(500, 0.5), 500)
        best_c = compute_c5(best_h, 500)

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    # Ensure sum constraint is tight
    target_sum = n_final / 2.0
    if abs(best_h.sum() - target_sum) > 1e-10:
        best_h = project_feasible(best_h, n_final)
        best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
