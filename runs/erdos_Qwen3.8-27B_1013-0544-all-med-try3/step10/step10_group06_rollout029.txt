def project_feasible(h,n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # fine correction
    for _ in range(100):
        err = target - h.sum()
        if abs(err) <= 1e-15:
            break
        if err > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        add = err
        # If add pushes out, limit
        if add > 0:
            max_add = 1.0 - h[idx]
            if add > max_add: add = max_add
        else:
            max_add = h[idx]
            if add < -max_add: add = -max_add
        h[idx] += add
    return h

def compute_q_array(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(compute_q_array(h, n)) * (2.0 / n))


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(40):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = s
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    h = np.clip(h, 0.0, 1.0)

    for _ in range(200):
        err = target - float(h.sum())
        if abs(err) <= 1e-15:
            break
        if err > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(err, max(0.0, 1.0 - h[idx]))
        else:
            idx = int(np.argmax(h))
            add = max(err, -max(0.0, h[idx]))
        if add == 0.0:
            break
        h[idx] += add

    return h


def threshold_feasible(h, n):
    target = 0.5 * float(n)
    hb = (h > 0.5).astype(float)
    num = float(hb.sum())

    if num > target:
        cnt = int(round(num - target))
        if cnt > 0:
            one_idx = np.where(hb == 1.0)[0]
            order = np.argsort(h[one_idx])
            for idx in one_idx[order[:cnt]]:
                hb[idx] = 0.0
    elif num < target:
        cnt = int(round(target - num))
        if cnt > 0:
            zero_idx = np.where(hb == 0.0)[0]
            order = np.argsort(-h[zero_idx])
            for idx in zero_idx[order[:cnt]]:
                hb[idx] = 1.0

    return project_feasible(hb, n)

def expand_to_resolution(h_src, n_src, n_dst):
    if n_dst == n_src:
        return np.array(h_src, dtype=float).copy()
    x_src = np.arange(n_src, dtype=float) * (2.0 / float(n_src))

    h_dst = np.interp(x_dst, x_src, h_src)

def apply_point(h, q, n, i, new_val):
    old = h[i]
    if old == new_val:
        return
    d = new_val - old

    q[n - 1] += new_val * (1.0 - new_val) - old * (1.0 - old)

    left = i
    if left:
        seg = h[i - left:i][::-1]
        q[n:n + left] += d * (1.0 - seg)
        q[n - left - 1:n - 1][::-1] -= d * seg

    right = n - 1 - i
    if right:
        seg = h[i + 1:i + 1 + right]
        q[n:n + right] -= d * seg
        q[i:n - 1][::-1] += d * (1.0 - seg)

    h[i] = new_val


def target_move(a, b, lag):
    u_lo = max(-a, b - 1.0)
    u_hi = min(1.0 - a, b)
    if u_hi <= u_lo + 1e-13:
        return None

    s = a + b
    if lag > 0:
        if s > 1.0 + 1e-9:
            u_target = u_hi
        elif s < 1.0 - 1e-9:
            u_target = u_lo
        else:
            return None
    else:
        u_target = u_lo

    if abs(u_target) < 1e-12:
        return None

    if u_target < u_lo:
        u_target = u_lo
    if u_target > u_hi:
        u_target = u_hi
    if abs(u_target) < 1e-12:
        return None
    return u_target


def frame_transfer(a, b, u):
    s = a + b
    lo_a = max(0.0, s - 1.0)
    hi_a = min(1.0, s)
    na = a + u
    if na < lo_a:
        na = lo_a
    elif na > hi_a:
        na = hi_a
    nb = s - na
    if nb < 0.0:
        nb = 0.0
        na = s
    elif nb > 1.0:
        nb = 1.0
        na = s - 1.0
    if na < 0.0:
        na = 0.0
    if na > 1.0:
        na = 1.0

    if K <= 0:

            np.empty(0, dtype=int),
            np.empty(0, dtype=int),
            np.empty(0, dtype=int),
        )

    if K == m:
        idxs = np.arange(m, dtype=int)
    else:
        idxs = np.argpartition(q, m - K)[-K:].astype(int)

    pi = []
    pj = []
    pl = []

    for idx in idxs:
        idx = int(idx)
        lag = idx - (n - 1)
        if lag == 0:

        if p >= n:
            continue
        L = n - p
        if L <= 0:
            continue

        if lag > 0:
            contrib = h[p:n] * (1.0 - h[0:n - p])
        else:
            contrib = h[0:n - p] * (1.0 - h[p:n])

        s = float(contrib.sum())
        if s <= 1e-15:
            continue

        take = min(P, L)
        if take <= 0:
            continue
        if take < L:
            top = np.argpartition(contrib, -take)[-take:].astype(int)
        else:
            top = np.arange(L, dtype=int)

        pj.append((top + p).astype(int))
        pl.append(np.full(take, lag, dtype=int))

            np.concatenate(pi).astype(int),
            np.concatenate(pj).astype(int),
            np.concatenate(pl).astype(int),
            idxs.astype(int),
        )
    return (
        np.empty(0, dtype=int),
        np.empty(0, dtype=int),

    cur_c = float(q.max()) * dx
    if cur_c < best_c - 1e-12:
        best_h = h.copy()
        best_c = cur_c

    return best_h, best_c, h.copy(), cur_c


_LB_N = 1
_LB_ALPHA = 100.0
_LB_PEN = 1e5
_LB_TARGET = 0.5
_LB_DEADLINE = 0.0


def _lb_f(x):
    n = int(_LB_N)
    c = compute_q_array(x, n)
    alpha = float(_LB_ALPHA)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    s = float(e.sum())
    f = m + math.log(s) / alpha if s > 0.0 else m
    diff = float(x.sum()) - float(_LB_TARGET)
    f += float(_LB_PEN) * diff * diff
    return float(f)


def _lb_g(x):
    n = int(_LB_N)
    c = compute_q_array(x, n)
    alpha = float(_LB_ALPHA)
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    e = np.exp(sh - mx)
    w = e / float(e.sum())

    g = np.zeros(n, dtype=float)

    if n == 1:
        g[0] = 1.0 - 2.0 * float(x[0])
    else:
        y = 1.0 - x
        w_pos = w[n:2 * n - 1]
        w_neg = w[:n - 1][::-1]
        w0 = float(w[n - 1])

        g += w0 * (1.0 - 2.0 * x)

        g += np.convolve(w_pos, y, "full")[:n]
        g -= np.convolve(x, w_pos[::-1], "full")[n - 1:2 * n - 1]
        g += np.convolve(y, w_neg[::-1], "full")[n - 1:2 * n - 1]
        g -= np.convolve(w_neg, x, "full")[:n]

    diff = float(x.sum()) - float(_LB_TARGET)
    g += 2.0 * float(_LB_PEN) * diff
    return g


def _lb_callback(*args):
    return time.time() > float(_LB_DEADLINE)


def lbfgs_optimize(h, n, alpha, deadline, maxiter=300, penalty=1e5):
    global _LB_N, _LB_ALPHA, _LB_PEN, _LB_TARGET, _LB_DEADLINE

    _LB_N = int(n)
    _LB_ALPHA = float(alpha)
    _LB_PEN = float(penalty)
    _LB_TARGET = 0.5 * float(n)
    _LB_DEADLINE = float(deadline)

    x0 = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(
            _lb_f,
            x0,
            jac=_lb_g,
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": int(maxiter),
                "ftol": 1e-15,
                "gtol": 1e-9,
                "maxls": 35,
                "iprint": -1,
                "callback": _lb_callback,
            },
        )
        x = np.clip(res.x, 0.0, 1.0)
    except Exception:
        x = x0

    return project_feasible(x, n)


def polish_lbfgs(h, n, deadline, best_h, best_c):
    n = int(n)

    if n <= 800:
        alphas = (50.0, 200.0, 1000.0, 5000.0, 20000.0)
        maxits = (80, 120, 180, 240, 320)
        per_call = 45.0
    elif n <= 1500:
        alphas = (100.0, 1000.0, 10000.0)
        maxits = (90, 160, 260)
        per_call = 35.0
    else:
        alphas = (100.0, 3000.0)
        maxits = (70, 160)
        per_call = 25.0

    h_phase = np.array(h, dtype=float).copy()
    c_phase = compute_c5(h_phase, n)
    if c_phase < best_c - 1e-12:
        best_h = h_phase.copy()
        best_c = c_phase

    if c_phase > best_c + 0.01:
        h_phase = np.array(best_h, dtype=float).copy()
        c_phase = best_c

    for idx in range(len(alphas)):
        if time.time() >= deadline:
            break
        call_deadline = min(deadline, time.time() + per_call)
        h_cand = lbfgs_optimize(h_phase, n, alphas[idx], call_deadline, maxits[idx])
        c_cand = compute_c5(h_cand, n)

        if c_cand < best_c - 1e-12:
            best_h = np.array(h_cand, dtype=float).copy()
            best_c = c_cand
            h_phase = best_h.copy()
            c_phase = c_cand
        else:
            if c_cand <= c_phase + 0.005:
                h_phase = np.array(h_cand, dtype=float).copy()
                c_phase = c_cand

    return h_phase, best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1100.0

    rng = np.random.RandomState(int(seed))
    main_deadline = t0 + max(20.0, budget_s - 40.0)
    final_deadline = t0 + max(20.0, budget_s - 5.0)
    if main_deadline > final_deadline - 5.0:
        main_deadline = final_deadline - 5.0

    h_init = get_initial_h()
    if h_init is None or len(h_init) < 10:
        n0 = 500
        best_h = make_fallback_initial(n0, rng)
    else:
        n0 = int(len(h_init))
        best_h = project_feasible(h_init, n0)

    best_c = compute_c5(best_h, n0)

    ns = []
    if n0 >= 2:
        ns.append(n0)
    for mult in (2, 4):
        cand = n0 * mult
        if 200 <= cand <= 2000 and cand not in ns:
            ns.append(cand)

    if (not ns) or max(ns) < 500:
        for cand in (500, 1000):
            if cand <= 2000 and cand not in ns:
                ns.append(cand)

    ns = sorted(set(ns))
    if len(ns) > 3:
        if ns[0] != ns[-1]:
            ns = (ns[0], ns[len(ns) // 2], ns[-1])
        else:
            ns = ns[:3]

    total = max(0.0, main_deadline - time.time())
    if len(ns) == 1:
        times = (total,)
    elif len(ns) == 2:
        times = (0.35 * total, 0.65 * total)
    else:
        times = (0.22 * total, 0.53 * total, 0.25 * total)

    for phase_idx, n in enumerate(ns):
        if time.time() >= main_deadline - 5.0:
            break

        n = int(n)
        phase_time = float(times[phase_idx])
        phase_deadline = min(main_deadline - 5.0, time.time() + phase_time)
        if phase_deadline <= time.time() + 10.0:
            continue

        if len(best_h) == n:
            start_h = np.array(best_h, dtype=float).copy()
        else:
            start_h = expand_to_resolution(best_h, len(best_h), n)

        if phase_idx > 0:
            start_h = project_feasible(start_h + rng.normal(0.0, 1e-4, n), n)

        best_at_n_h = np.array(start_h, dtype=float).copy()
        best_at_n_c = compute_c5(best_at_n_h, n)

        lb_time = min(120.0, 0.25 * phase_time)
        if lb_time > 10.0:
            lb_deadline = min(phase_deadline, time.time() + lb_time)
            h_phase, best_at_n_h, best_at_n_c = polish_lbfgs(
                start_h, n, lb_deadline, best_at_n_h, best_at_n_c
            )
            if best_at_n_c < best_c - 1e-12:
                best_c = best_at_n_c
                best_h = np.array(best_at_n_h, dtype=float).copy()

        if time.time() < phase_deadline - 10.0:
            c_phase = compute_c5(h_phase, n)
            if c_phase <= best_at_n_c + 0.002:
                fs_start = h_phase
            else:
                fs_start = best_at_n_h

            fs_best_h, fs_best_c, _, _ = fast_search(
                fs_start, n, phase_deadline - 3.0, rng, initial_best_h=best_at_n_h
            )
            if fs_best_c < best_at_n_c - 1e-12:
                best_at_n_h = fs_best_h
                best_at_n_c = fs_best_c

            if best_at_n_c < best_c - 1e-12:
                best_c = best_at_n_c
                best_h = np.array(best_at_n_h, dtype=float).copy()

    if time.time() < final_deadline - 20.0:
        n_final = int(len(best_h))
        lb_deadline = min(final_deadline - 10.0, time.time() + (35.0 if n_final > 1500 else 60.0))
        h_pol = lbfgs_optimize(best_h, n_final, 20000.0, lb_deadline, 260)
        c_pol = compute_c5(h_pol, n_final)
        if c_pol < best_c - 1e-12:
            best_c = c_pol
            best_h = h_pol

    if time.time() < final_deadline - 8.0:
        n_final = int(len(best_h))
        g_deadline = min(final_deadline - 4.0, time.time() + (25.0 if n_final > 1500 else 55.0))
        g_best_h, g_best_c, _, _ = greedy_top_search(
            best_h, n_final, g_deadline, initial_best_h=best_h
        )
        if g_best_c < best_c - 1e-12:
            best_c = g_best_c
            best_h = g_best_h

    if time.time() < final_deadline - 2.0:
        n_final = int(len(best_h))
        f_best_h, f_best_c, _, _ = fast_search(
            best_h, n_final, final_deadline - 1.0, rng, initial_best_h=best_h
        )
        if f_best_c < best_c - 1e-12:
            best_c = f_best_c
            best_h = f_best_h

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    try:
        val = evaluate_erdos_solution(best_h)
        if isinstance(val, (int, float, np.floating)):
            val = float(val)
            if math.isfinite(val) and val < best_c:
                best_c = val
    except Exception:
        pass

    return best_h, float(best_c), int(len(best_h))
