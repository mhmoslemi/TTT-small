    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxiter_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxiter_list is not None and i < len(maxiter_list):
            maxit = maxiter_list[i]
        else:
            maxit = max(500, int(remaining * 4))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 6,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_shifts(h, n, deadline, top_k=8, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000]
    for rnd in range(30):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 8:
                break
            maxit = max(400, int((deadline - time.time()) * 3))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def coordinate_descent(h, n, deadline, min_time_s=5.0):
    dx = 2.0 / n
    h = h.copy()
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    total_ks = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    lo_bound = 0.0
    hi_bound = 1.0

    sweep = 0
    while time.time() < deadline - min_time_s:
        sweep += 1
        for i in range(n):
            if time.time() > deadline:
                break
            h_i = h[i]
            idx_p = i + ks
            idx_m = i - ks
            vp = (idx_p >= 0) & (idx_p < n)
            vm = (idx_m >= 0) & (idx_m < n)

            A = np.zeros(total_ks)
            A[vp] += 1.0 - h[idx_p[vp]]
            A[vm] -= h[idx_m[vm]]
            A *= dx

            B = c_vec - A * h_i

            lo_b, hi_b = lo_bound, hi_bound
            for _ in range(32):
                mid = (lo_b + hi_b) * 0.5
                span = hi_b - lo_b
                eps = max(span * 0.002, 1e-12)
                tl = mid - eps
                tr = mid + eps
                if tl < 0.0:
                    tl = 0.0
                if tr > 1.0:
                    tr = 1.0
                f_l = np.max(A * tl + B)
                f_r = np.max(A * tr + B)
                if f_r > f_l:
                    hi_b = mid
                else:
                    lo_b = mid
            t_star = (lo_b + hi_b) * 0.5

            f0 = np.max(B)
            f1 = np.max(A + B)
            ft = np.max(A * t_star + B)
            if f0 <= ft + 1e-16 and f0 <= f1:
                t_star = 0.0
            elif f1 <= ft + 1e-16 and f1 <= f0:
                t_star = 1.0

            delta = t_star - h_i
            if abs(delta) > 1e-13:
                h[i] = t_star
                c_vec += A * delta

        h = project_feasible(h, n)
        c_vec = compute_c5_vec(h, n)
        c_now = c_vec.max()
        if c_now < best_c - 1e-16:
            best_c = c_now

    return h, c_vec, best_c

def binary_projection_optimize(h, n, deadline, rng):
    """Try binary projection and re-optimize."""
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    n_ones = n // 2
    h_bin[sorted_idx[:n_ones]] = 1.0
    h_bin = project_feasible(h_bin, n)
    c_bin = compute_c5(h_bin, n)
    return h_bin, c_bin

def perturbation_restart(h, n, deadline, rng, n_restarts=3):
    """Try restarting from perturbed versions."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for r in range(n_restarts):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        # Random perturbation
        h_p = h.copy()
        n_flip = max(3, n // 100)
        flip_idx = rng.choice(n, size=n_flip, replace=False)
        for idx in flip_idx:
            h_p[idx] = 1.0 - h_p[idx]
        h_p = project_feasible(h_p, n)
        c_p = compute_c5(h_p, n)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()
        # Quick L-BFGS from perturbation
        p_end = min(time.time() + remaining * 0.3, deadline - 15)
        h_new, c_new = run_lbfgs_phase(h_p, n, p_end,
            alphas=[5000, 50000, 500000], mu=0.005,
            maxiter_list=[500, 800, 1000])
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=2000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)

    # Phase 1: Quick L-BFGS at original resolution (40s)
    p1_end = min(t0 + 40, deadline - 300)
    h1, c1 = run_lbfgs_phase(best_h, n0, p1_end,
        alphas=[100, 1000, 10000, 100000], mu=0.01,
        maxiter_list=[800, 1000, 1200, 1500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=2000
    n_hi = 2000
    x_fine = np.linspace(0, 2, n_hi)
    x_coarse = np.linspace(0, 2, len(best_h))
    h_hi = np.interp(x_fine, x_coarse, best_h)
    h_hi = project_feasible(h_hi, n_hi)
    c_hi = compute_c5(h_hi, n_hi)
    if c_hi < best_c:
        best_c = c_hi
        best_h = h_hi.copy()

    # Phase 3: L-BFGS at n=2000 (120s)
    p3_end = min(time.time() + 120, deadline - 350)
    h3, c3 = run_lbfgs_phase(h_hi, n_hi, p3_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000],
        mu=0.005, maxiter_list=[1000, 1200, 1500, 1500, 2000, 2000])
    if c3 < best_c:
        best_c = c3
        best_h = h3
    elif c3 <= best_c:
        best_h = h3

    # Phase 4: Coordinate Descent at n=2000 (250s)
    p4_end = min(time.time() + 250, deadline - 200)
    h4, c_vec4, c4 = coordinate_descent(best_h, n_hi, p4_end, min_time_s=10)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Active set L-BFGS (80s)
    p5_end = min(time.time() + 80, deadline - 120)
    h5, c5 = run_active_shifts(best_h, n_hi, p5_end, top_k=6,
        alphas=[50000, 200000, 1000000])
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: More Coordinate Descent (100s)
    p6_end = min(time.time() + 100, deadline - 80)
    h6, c_vec6, c6 = coordinate_descent(best_h, n_hi, p6_end, min_time_s=5)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Perturbation restarts (60s)
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 60, deadline - 30)
        h7, c7 = perturbation_restart(best_h, n_hi, p7_end, rng, n_restarts=3)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final CD + active set (50s)
    p8_end = min(time.time() + 30, deadline - 20)
    h8, c_vec8, c8 = coordinate_descent(best_h, n_hi, p8_end, min_time_s=3)
    if c8 < best_c:
        best_c = c8
        best_h = h8

    if time.time() < deadline - 5:
        h8b, c8b = run_active_shifts(best_h, n_hi, deadline - 3, top_k=4,
            alphas=[500000, 2000000])
        if c8b < best_c:
            best_c = c8b
            best_h = h8b

    # Phase 9: Final CD sweep
    if time.time() < deadline - 1:
        h9, c_vec9, c9 = coordinate_descent(best_h, n_hi, deadline - 1, min_time_s=0.5)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Also check if n0 solution is still better
    h0_check = project_feasible(h0, n0)
    c0_check = compute_c5(h0_check, n0)
    if c0_check < best_c:
        best_c = c0_check
        best_h = h0_check

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
