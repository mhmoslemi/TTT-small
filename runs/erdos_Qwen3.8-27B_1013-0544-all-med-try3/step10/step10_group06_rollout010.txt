import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[m.argmax()] = 1.0
    else:
        w /= s
    return w

def compute_smooth_max_grad(h, n, alpha):
    """Correct gradient of smooth max of correlation values."""
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            # c[lag] = sum_{j=0}^{n-lag-1} h[j]*(1-h[j+lag])
            # d/dh[j] = (1-h[j+lag]) for j < n-lag
            grad[0:n-lag] += wk * (1.0 - h[lag:n])
            # d/dh[j] = -h[j-lag] for j >= lag
            grad[lag:n] += wk * (-h[0:n-lag])
        else:
            idx = -lag
            # c[-idx] = sum_{j=idx}^{n-1} h[j]*(1-h[j-idx])
            # d/dh[j] = (1-h[j-idx]) for j >= idx
            grad[idx:n] += wk * (1.0 - h[0:n-idx])
            # d/dh[j] = -h[j+idx] for j < n-idx
            grad[0:n-idx] += wk * (-h[idx:n])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000, penalty_coeff=1e10):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = compute_smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 40,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src/2.0
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 2.0/n_dst/2.0
    h_dst = np.interp(x_dst, x_src, h_src, left=h_src[0], right=h_src[-1])
    return project_feasible(h_dst, n_dst)

def targeted_local_search(h, n, deadline, rng, k_top=5, max_rounds=50000):
    """Focused local search reducing top-k lag correlations."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rounds = 0

    c_all = compute_all_lags(h, n)
    top_idx = np.argsort(c_all)[-k_top:]

    while rounds < max_rounds and time.time() < deadline:
        improved = False
        for k_star in top_idx:
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            pair_count = n - m
            if pair_count <= 0:
                continue
            # Try a few random pairs
            for _ in range(5):
                i = rng.randint(0, pair_count)
                if lag > 0:
                    j = i + lag
                    if h[i] > h[j] + 1e-8:
                        d = min(0.01, (h[i] - h[j]) * 0.3)
                        h[i] -= d
                        h[j] += d
                    elif h[j] > h[i] + 1e-8:
                        d = min(0.01, (h[j] - h[i]) * 0.3)
                        h[i] += d
                        h[j] -= d
                        j, i = i, j
                        h[j] -= d
                        h[i] += d
                else:
                    j = i - lag  # lag < 0, j = i + |lag|... wait
                    j = i + m  # i is in [0, n-m-1], j = i+m
                    if h[j] > h[i] + 1e-8:
                        d = min(0.01, (h[j] - h[i]) * 0.3)
                        h[j] -= d
                        h[i] += d
                    elif h[i] > h[j] + 1e-8:
                        d = min(0.01, (h[i] - h[j]) * 0.3)
                        h[i] -= d
                        h[j] += d

                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    c_all = compute_all_lags(h, n)
                    top_idx = np.argsort(c_all)[-k_top:]
                else:
                    # Undo
                    if lag > 0:
                        h[i] += d if h[i] < 1 else 0
                        h[j] -= d if h[j] > 0 else 0
                        # Actually let's just reproject from best
                        h = best_h.copy()
                    else:
                        h = best_h.copy()
                rounds += 1
                if not improved and rounds > 200:
                    break
            if not improved:
                break
        if not improved:
            # Try smaller steps
            break

    return best_h, best_c

def aggressive_local_search(h, n, deadline, rng):
    """More aggressive local search with multiple step sizes."""
    h = h.copy()

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    K = 8
    max_iters = 30000

    for it in range(max_iters):
        if time.time() > deadline:
            break

        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-K:]
        improved = False

        for k_star in top_idx:
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            pair_count = n - m

            for _ in range(3):

                j = i + m

                if h[i] > h[j] + 1e-7:
                    d = rng.uniform(0.002, 0.03)
                    d = min(d, (h[i] - h[j]) * 0.25)
                    h[i] -= d
                    h[j] += d
                elif h[j] > h[i] + 1e-7:
                    d = rng.uniform(0.002, 0.03)
                    d = min(d, (h[j] - h[i]) * 0.25)
                    h[i] += d
                    h[j] -= d
                else:
                    continue

                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    c_all = compute_all_lags(h, n)
                else:
                    h = best_h.copy()

        if not improved:
            # Perturb and retry
            if it > 500:
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
            else:
                break

    return best_h, best_c

def multi_alpha_lbfgsb(h, n, deadline, alpha_list, maxiter_list):
    """Run L-BFGS-B with increasing alpha values."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha, maxit in zip(alpha_list, maxiter_list):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 2:
            break
        time_limit = time.time() + min(remaining * 0.9, maxit * 0.05)
        h_opt = lbfgsb_optimize(best_h, n, alpha, time_limit, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c
