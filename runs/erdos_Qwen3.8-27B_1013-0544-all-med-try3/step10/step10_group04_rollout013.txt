    """Compute change in c_vec when h[p] changes by delta."""
    dx = 2.0 / n
    s = np.arange(-(n-1), n)  # shift values
    dc = np.zeros(2*n-1)
    
    # Case 1: p is first index, need 0 ≤ p+s < n → -p ≤ s ≤ n-1-p
    s_min1 = max(-p, -(n-1))
    s_max1 = min(n-1-p, n-1)
    for si in range(s_min1, s_max1 + 1):
        if 0 <= p + si < n:
            dc[si + (n-1)] += delta * (1.0 - h[p + si])
    
    # Case 2: p+s is second index, so s = p-i, need 0 ≤ i < n
    # i = p-s, need 0 ≤ p-s < n → p-n < s < p
    s_min2 = max(p-n+1, -(n-1))
    s_max2 = min(p-1, n-1)

        i_val = p - si
        if 0 <= i_val < n:
            dc[si + (n-1)] += -delta * h[i_val]
    
    return dc * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_objective(h, n, alpha, top_m=None):
    c = compute_c5_vec(h, n)
    if top_m is not None:
        idx = np.argsort(c)[-top_m:]
        c_sub = c[idx]
    else:
        c_sub = c
    m = c_sub.max()
    shifted = alpha * (c_sub - m)
    max_shift = shifted.max()
    if max_shift > 500:
        val = m + (max_shift + np.log(np.sum(np.exp(shifted - max_shift)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    # Add small feasibility penalty
    val += 1e4 * (h.sum() - n/2.0)**2
    return float(val)

def lse_gradient(h, n, alpha, top_m=None):
    c = compute_c5_vec(h, n)
    if top_m is not None:
        idx = np.argsort(c)[-top_m:]
        c_sub = c[idx]
        # weights for sub-problem
        m = c_sub.max()
        shifted = alpha * (c_sub - m)
        max_shift = shifted.max()
        w_sub = np.exp(shifted - max_shift)
        w_sub /= w_sub.sum()
        # Map back to full weights
        w = np.zeros(len(c))
        w[idx] = w_sub
    else:
        m = c.max()
        shifted = alpha * (c - m)
        max_shift = shifted.max()
        w = np.exp(shifted - max_shift)
        w /= w.sum()
    
    dx = 2.0 / n
    center = n - 1
    n2m1 = 2 * n - 1
    grad = np.zeros(n)
    
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    
    grad += 2e4 * (h.sum() - n/2.0)
    return grad
