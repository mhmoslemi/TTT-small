import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_topk(c, alpha, K):
    """Smooth max using top-K elements with softmax weight."""
    n = len(c)
    if K >= n:
        # Full softmax
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            s = np.exp(shifted - mx)
        else:
            s = np.exp(shifted)
        return float(m + np.log(s.sum()) / alpha)
    else:
        # Top-K softmax
        idx = np.argsort(c)[-K:]
        m = c[idx].max()
        shifted = alpha * (c[idx] - m)
        mx = shifted.max()
        if mx > 500:
            s = np.exp(shifted - mx)
        else:
            s = np.exp(shifted)
        return float(m + np.log(s.sum()) / alpha)

def smooth_grad(h, n, alpha, K):
    """Gradient of smooth max of overlap lags."""
    dx = 2.0 / n
    c = compute_all_lags(h, n) / dx  # raw correlation (without dx)
    n_lags = 2 * n - 1
    
    if K >= n_lags:
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()
    else:
        idx = np.argsort(c)[-K:]
        m = c[idx].max()
        shifted_k = alpha * (c[idx] - m)
        mx = shifted_k.max()
        if mx > 500:
            w_k = np.exp(shifted_k - mx)
        else:
            w_k = np.exp(shifted_k)
        w_k = w_k / w_k.sum()
        w = np.zeros(n_lags)
        w[idx] = w_k
    
    grad = np.zeros(n)
    for k in range(n_lags):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n-lag])
            grad[0:n-lag] += w[k] * (-h[lag:n])
        else:
            idx2 = -lag
            grad[0:n-idx2] += w[k] * (1.0 - h[idx2:n])
            grad[idx2:n] += w[k] * (-h[0:n-idx2])
    return grad * dx

def lbfgsb_topk(h, n, alpha, K, deadline, maxiter=3000):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x2 = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x2, n)
        obj = smooth_max_topk(c, alpha, K)
        diff = x2.sum() - target_sum
        obj += 1e10 * diff * diff
        grad = smooth_grad(x2, n, alpha, K)
        grad += 2e10 * diff * np.ones(n)
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def gradient_descent(h, n, steps, lr_init, deadline, alpha, K):
    """Simple projected gradient descent."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    target_sum = n / 2.0
    
    for step in range(steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        cur_c = c.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        grad = smooth_grad(h, n, alpha, K)
        # Projected gradient step
        h_new = h - lr * grad
        # Project to feasible set
        h_new = project_feasible(h_new, n)
        
        # Check if improved
        c_new = compute_all_lags(h_new, n).max()
        if c_new < cur_c * (1 - 1e-10) or step % 100 == 0:
            h = h_new
            if step % 500 == 0:
                lr *= 0.5
        else:
            # Reject step
            lr *= 0.8
    
    return best_h, best_c

def interpolate(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)  # periodic
    return project_feasible(h_dst, n_dst)

def targeted_local_search(h, n, deadline, rng, top_k=10, max_iters=5000):
    """Targeted local search focusing on the worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    it = 0
    while it < max_iters and time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        cur_max = c_all.max()
        
        improved = False
        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            # For this lag, find the pair that contributes most to the overlap
            # overlap(lag) = sum_{i valid} h[i](1-h[i+lag])*dx
            # Try adjusting individual i to reduce this contribution
            if lag > 0:
                valid = slice(0, n - lag)
                contrib = h[0:n-lag] * (1 - h[lag:n])
            else:
                valid = slice(-lag, n)
                contrib = h[-lag:n] * (1 - h[0:n+lag])
            
            # Find the i with maximum contribution
            i_max = np.argmax(contrib)
            
            # Try reducing h at the "1" position or increasing h at the "0" position
            if lag > 0:
                i1 = i_max  # h[i1] is high
                i0 = i_max + lag  # h[i0] is low
            else:
                i1 = i_max + (-lag)  # h[i1] is high
                i0 = i_max  # h[i0] is low
            
            # Try several step sizes
            for frac in [0.1, 0.2, 0.3, 0.5]:
                d = min(h[i1] - h[i0], 1.0) * frac * 0.1
                if d < 1e-10:
                    continue
                h_old = h.copy()
                h[i1] -= d
                h[i0] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h = h_old
            if improved:
                it += 1
                break
        
        if not improved:
            # Try random small perturbations
            for _ in range(5):
                if time.time() > deadline:
                    break
                i, j = rng.randint(0, n, 2)
                if i == j:
                    continue
                d = rng.choice([-1, 1]) * rng.uniform(0.001, 0.01)
                if h[i] - d < 0 or h[j] + d > 1:
                    continue
                h_old = h.copy()
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                    improved = True
                else:
                    h = h_old
                it += 1
            if not improved:
                break
    
    return best_h, best_c
