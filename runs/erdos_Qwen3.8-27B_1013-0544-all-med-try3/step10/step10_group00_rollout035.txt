def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            tot = room[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += diff * room[mask] / tot
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            tot = h[mask].sum()
            if tot < 1e-14:
                break
            h[mask] += diff * h[mask] / tot
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:

            count = int(np.ceil(diff))
            for idx in order[:count]:
                h_bin[idx] = 1.0
        else:
            order = np.argsort(h)
            count = int(np.ceil(-diff))
            for idx in order[:count]:
                h_bin[idx] = 0.0

def upsample_linear(h, n_new):
    """Linearly upsample h from len(h) to n_new points"""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c == c.max()] = 1.0 / np.sum(c == c.max())
    else:
        w /= s
    return w

def h_grad(w, h, n):
    """Compute gradient of weighted sum of correlation values"""
    dx = 2.0 / n
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:k+nv])
                grad[k:k+nv] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:kk+nv] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:kk+nv])
    return grad

    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    w = smooth_max_w(c, alpha)
    g = h_grad(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

    try:
        res = minimize(lbfgs_objective, h, jac=lbfgs_gradient, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 5,
                               'maxls': 50, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:

    """Compute change in c values for active lags when turning h[i_off] off and h[j_on] on.
    active_lags is an array of lag values r (can be negative, positive, or zero)."""
    dx = 2.0 / n
    deltas = np.zeros(len(active_lags))
    for idx, r in enumerate(active_lags):
        dr = 0.0
        if r >= 0:
            # Turn i_off off
            if r <= i_off <= n-1:
                dr -= (1.0 - h[i_off - r])
            if i_off <= n-1-r:
                dr += h[i_off + r]
            # Turn j_on on
            if r <= j_on <= n-1:
                dr += (1.0 - h[j_on - r])
