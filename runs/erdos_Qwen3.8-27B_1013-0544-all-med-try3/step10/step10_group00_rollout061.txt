def binary_sa(h, n, deadline, rng, initial_temp=0.01, final_temp=0.0001, cooling=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    temp = initial_temp
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    ones_set = set(ones.tolist())
    zeros_set = set(zeros.tolist())
    
    iterations = 0
    while time.time() < deadline:
        iterations += 1
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        
        if new_c < cur_c:
            cur_c = new_c
            ones_set.remove(i); ones_set.add(j)
            zeros_set.add(i); zeros_set.remove(j)
            ones = sorted(ones_set); zeros = sorted(zeros_set)
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Meta acceptance
            delta = new_c - cur_c
            if delta > 0 and rng.random() < np.exp(-delta / max(temp, 1e-15)):
                cur_c = new_c
                ones_set.remove(i); ones_set.add(j)
                zeros_set.add(i); zeros_set.remove(j)
                ones = sorted(ones_set); zeros = sorted(zeros_set)
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        temp *= cooling
        if temp < final_temp:
            temp = final_temp
    
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            cnt = min(int(np.ceil(diff)), len(zeros))
            for idx in range(cnt):
                h_bin[zeros[order[idx]]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            cnt = min(int(np.ceil(-diff)), len(ones))
            for idx in range(cnt):
                h_bin[ones[order[idx]]] = 0.0
    return h_bin

def init_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').astype(float) * dx

def swap_delta_vec(h, i, j, n):
    nf = 2 * n - 1
    m_idx = np.arange(nf)
    k = m_idx - (n - 1)
    dx = 2.0 / n
    d = np.zeros(nf)
    ik = i + k
    v1 = (ik >= 0) & (ik < n)
    d[v1] += h[ik[v1]]
    ik2 = i - k
    v2 = (ik2 >= 0) & (ik2 < n)
    d[v2] += h[ik2[v2]]
    jk = j + k
    v3 = (jk >= 0) & (jk < n)
    d[v3] -= h[jk[v3]]
    jk2 = j - k
    v4 = (jk2 >= 0) & (jk2 < n)
    d[v4] -= h[jk2[v4]]
    return d * dx

def simulated_anneal_binary(h, n, deadline, rng_np, t_mult_init=5.0, t_mult_final=0.1):

    h = h.copy()

    cur_max = c.max()
    best_c = cur_max
    best_h = h.copy()

    zeros = np.where(h < 0.5)[0].copy()

    t_final = t_mult_final * dx
    temp = t_init
    cool_rate = 0.99999
    no_improve = 0
    stall_limit = 50000
    
    while time.time() < deadline:

            cur_max = new_max
            ones[oi] = j
            zeros[oz] = i

                best_c = cur_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

def make_beatty_binary(n, alpha_val):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha_val) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = 1.0
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = 0.0
    return h
