    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def c5_gradient(h, c_vec, worst_k):
    """Gradient of c_k with respect to h, for a specific k."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    k = worst_k - center
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += (1.0 - h[k:n]) 
        grad[k:n] += -h[0:nv]
    else:
        nk = -k
        nv = n - nk
        grad[nk:n] += (1.0 - h[0:nv])
        grad[0:nv] += -h[nk:n]
    return grad * dx

def c5_gradient_all(h, weights):
    """Weighted gradient: sum_k weights[k] * grad(c_k)."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for idx in range(2*n-1):
        if abs(weights[idx]) < 1e-14:
            continue
        k = idx - center
        w = weights[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += w * (1.0 - h[k:n])
                grad[k:n] += w * (-h[0:nv])
        else:

            nv = n - nk
            if nv > 0:
                grad[nk:n] += w * (1.0 - h[0:nv])
                grad[0:nv] += w * (-h[nk:n])
    return grad

    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}."""
    h = h.copy()
    for _ in range(5000):

            break

            if tot < 1e-14:

            eta = diff / tot

        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h)

def pair_coord_descend(h, n, deadline, n_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    center = n - 1
    for rnd in range(n_rounds):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h)
        c_max = c_vec.max()
        top_k = np.argsort(-c_vec)[:5]
        improved = False
        for wi in top_k:
            if time.time() > deadline - 3:
                break
            k = wi - center
            if k >= 0:
                pairs = [(i, i + k) for i in range(n - k)]
            else:
                nk = -k
                pairs = [(i + nk, i) for i in range(n - nk)]
            for (i, j) in pairs:
                if time.time() > deadline - 3:
                    break
                for delta in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.003, -0.003]:
                    vi = h[i] + delta
                    vj = h[j] - delta
                    if vi < -1e-9 or vi > 1.0 + 1e-9 or vj < -1e-9 or vj > 1.0 + 1e-9:
                        continue
                    h[i] = max(0.0, min(1.0, vi))
                    h[j] = max(0.0, min(1.0, vj))
                    c_new = compute_c5(h)
                    if c_new < c_max - 1e-13:
                        c_max = c_new
                        improved = True
                        break
                    else:
                        h[i] = best_h[i]
                        h[j] = best_h[j]
                if improved:
                    break
            if improved:
                h = best_h.copy() if not np.allclose(h, best_h) else h
                # h already updated
                if c_max < best_c:
                    best_c = c_max
                    best_h = h.copy()
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        if not improved and rnd > 20:
            break
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search(h, n, deadline, n_trials=12):
    best_h = h.copy()
    best_c = compute_c5(h)
    rng = np.random.RandomState(777)
    mags = [0.01, 0.02, 0.03, 0.005, 0.04, 0.015, 0.025, 0.05, 0.008, 0.035, 0.012, 0.06]
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        mag = mags[t % len(mags)]
        h_pert = h + mag * (rng.rand(n) - 0.5)
        h_pert = project_feasible(h_pert, n)
        try:
            res = minimize(softmax_obj, h_pert, jac=softmax_grad,
                          args=(n, 30000), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-18, 'maxls': 100})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h)

    # Phase 1: L-BFGS warmup at base (120s)
    t1 = min(t0 + 120, deadline - 880)
    if t1 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            alphas=[200, 1000, 5000, 20000, 100000, 400000],
            maxiter_per=4000)

    # Phase 2: subgradient polish at base (80s)
    t2 = min(t0 + 200, deadline - 800)
    if t2 > time.time() + 5:
        h2, c2 = subgrad_phase(best_h, n_base, t2, lr0=0.004, n_iter=8000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: SLSQP at base (60s)
    t3 = min(t0 + 260, deadline - 740)
    if t3 > time.time() + 5:
        h3, c3 = slqp_phase(best_h, n_base, t3, alpha=500000, maxiter=3000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: perturbation at base (80s)
    t4 = min(t0 + 340, deadline - 660)
    if t4 > time.time() + 10:
        best_h, best_c = perturbation_search(best_h, n_base, t4, n_trials=12)

    # Phase 5: quick SLSQP after perturbation (40s)
    t5 = min(t0 + 380, deadline - 620)
    if t5 > time.time() + 5:
        h5, c5 = slqp_phase(best_h, n_base, t5, alpha=1000000, maxiter=2000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Upscale to n=2000 (350s)
    n_up = 2000
    t6_start = time.time()
    t6_end = min(t0 + 730, deadline - 250)
    if t6_end > t6_start + 30:
        h_up = interpolate_to(best_h, n_base, n_up)
        t6a = min(t6_start + 150, t6_end - 100)
        if t6a > time.time() + 10:
            h_up, c_up = run_lbfgs(h_up, n_up, t6a,
                alphas=[500, 5000, 30000, 100000], maxiter_per=3000)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
            else:
                h_up = best_h
        if time.time() < t6_end - 15:
            h_up, c_up = subgrad_phase(h_up, n_up, t6_end, lr0=0.002, n_iter=5000)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up

    # Phase 7: multi-start at best resolution (150s)
    n_use = len(best_h)
    t7 = min(t0 + 880, deadline - 100)
    if t7 > time.time() + 15:
        best_h, best_c = perturbation_search(best_h, n_use, t7, n_trials=8)

    # Phase 8: final L-BFGS + subgradient (60s)
    t8 = min(t0 + 940, deadline - 40)
    if t8 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_use, min(time.time() + 30, t8 - 5),
            alphas=[50000, 200000], maxiter_per=2000)
        if time.time() < t8 - 3:
            best_h, best_c = subgrad_phase(best_h, n_use, t8, lr0=0.001, n_iter=3000)

    # Phase 9: coordinate descent finish (remaining)
    if time.time() < deadline - 5:
        best_h, best_c = pair_coord_descend(best_h, len(best_h), deadline - 2, n_rounds=100)

    # Final feasibility and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h)

    return best_h, best_c, n_final
