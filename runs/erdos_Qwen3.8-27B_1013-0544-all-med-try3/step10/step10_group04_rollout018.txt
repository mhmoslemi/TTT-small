        if ws < 1e-16:
            continue
        nv = n - s
        grad[0:nv] += ws * (1.0 - x[s:n])
        grad[s:n] += ws * (-x[0:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def weighted_obj(x, n, weights, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    w = np.maximum(weights, 1e-10)
    w = w / w.sum()
    val = np.sum(w * c) + 0.5 * np.sum(w * (c - m) ** 2)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def weighted_grad(x, n, weights, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    w = np.maximum(weights, 1e-10)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wc = w[idx] + w[idx] * (c[idx] - m)
        if abs(wc) < 1e-16:
            continue
        k = idx - center
        wkm = wc * dx
        if k == 0:
            grad += wkm * (1.0 - 2.0 * x)
        elif k > 0:
            nv = n - k
            grad[0:nv] += wkm * (1.0 - x[k:n])
            grad[k:n] += wkm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wkm * (1.0 - x[0:nv])
            grad[0:nv] += wkm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, min_time=10):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 10.0), (200, 50.0), (500, 200.0),
                   (1000, 500.0), (3000, 2000.0), (8000, 10000.0),
                   (20000, 50000.0), (50000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - min_time:
            break
        t_start = time.time()
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 500000, 'maxls': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def run_weighted_lbfgs(h, n, deadline, n_rounds=3):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(n_rounds):
        if time.time() > deadline - 30:
            break
        c_vec = compute_c5_vec(h, n)
        m = c_vec.max()
        weights = np.exp(50.0 * (c_vec - m))
        weights /= weights.sum()
        try:
            res = minimize(weighted_obj, h, jac=weighted_grad,
                          args=(n, weights, 10000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 15000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 200000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=5000, min_time=20):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def slqp_active(h, n, deadline, top_k=10, alpha=2000000, maxiter=4000, min_time=20):
    """SLSQP with explicit constraints on the worst shifts."""
    h = project_feasible(h, n)
    c_vec = compute_c5_vec(h, n)
    m = c_vec.max()
    idx_sorted = np.argsort(-c_vec)
    active_idx = idx_sorted[:top_k]
    dx = 2.0 / n
    center = n - 1

    def obj_fn(x, na):
        c = compute_c5_vec(x, na)
        return float(c.max())

    constraints = []
    for ai in active_idx:
        k = ai - center
        def make_con(ai):
            kk = ai - center
            if kk == 0:
                def con_fun(x, na):
                    c = float(np.sum(x * (1 - x)) * 2.0 / na)
                    return c - m * 1.001
                return {'type': 'ineq', 'fun': con_fun, 'args': (n,)}
            elif kk > 0:
                sl = slice(0, na - kk)
                sr = slice(kk, na)
                def con_fun(x, na):
                    val = float(np.sum(x[kk:] * (1 - x[:na-kk])) * 2.0 / na)
                    return m * 1.001 - val
                return {'type': 'ineq', 'fun': con_fun, 'args': (n,)}
            else:
                kk2 = -kk
                def con_fun(x, na):
                    val = float(np.sum(x[kk2:] * (1 - x[:na-kk2])) * 2.0 / na)
                    return m * 1.001 - val
                return {'type': 'ineq', 'fun': con_fun, 'args': (n,)}
        constraints.append(make_con(ai))

    bounds = [(0.0, 1.0)] * n
    constraints.append({'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)})

    try:
        res = minimize(obj_fn, h, args=(n,), method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def sq_fn(x, na):
    return float(x.sum() - na / 2.0)

def sq_jac_fn(x, na):
    return np.ones(na)

def block_coord_descent(h, n, deadline, block_size=32, n_rounds=200, min_time=30):
    """Block coordinate descent: update one block at a time."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    n_blocks = n // block_size
    actual_bs = n // n_blocks
    rng = np.random.RandomState(777)
    
    for rnd in range(n_rounds):
        if time.time() > deadline - min_time:
            break
        block_idx = rnd % n_blocks
        start = block_idx * actual_bs
        end = start + actual_bs
        
        h_block = h[start:end].copy()
        h_rest = h.copy()
        
        def block_obj(x_block, s, e, hr, nn):
            hx = hr.copy()
            hx[s:e] = x_block
            hx = np.clip(hx, 0.0, 1.0)
            c = compute_c5_vec(hx, nn)
            cv = c.max()
            # Penalty for sum constraint
            target_sum = nn / 2.0
            penalty = 1000.0 * (hx.sum() - target_sum) ** 2
            return float(cv + penalty)
        
        try:
            res = minimize(block_obj, h_block,
                          args=(start, end, h_rest, n),
                          method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * actual_bs,
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
            h_new_block = np.clip(res.x, 0.0, 1.0)
            h_new = h.copy()
            h_new[start:end] = h_new_block
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    
    return h, best_c

def local_shift_refine(h, n, deadline, min_time=30):
    """Targeted refinement: for each worst shift, try small adjustments."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for outer in range(500):
        if time.time() > deadline - min_time:
            break
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        
        if k_worst == 0:
            continue
        
        kk = abs(k_worst)
        nv = n - kk
        
        # The overlap at this shift: sum h[i+kk]*(1-h[i]) for i=0..nv-1
        # To reduce it: decrease h[i+kk] where h[i]<1, or increase h[i] where h[i+kk]>0
        # Try small adjustments
        h_plus = h[kk:kk+nv].copy()
        h_minus = h[0:nv].copy()
        
        # Find positions with highest contribution
        contrib = h_plus * (1.0 - h_minus)
        top_positions = np.argsort(-contrib)[:20]
        
        improved = False
        for pos in top_positions:
            i = int(pos)
            # Try decreasing h[i+kk] or increasing h[i]
            for delta in [0.005, 0.002, 0.001, -0.001, -0.002]:
                h_orig_plus = h[i + kk]
                h[i + kk] = np.clip(h[i + kk] + delta, 0.0, 1.0)
                # Compensate sum
                comp = -(h[i + kk] - h_orig_plus)
                h[i] = np.clip(h[i] + comp, 0.0, 1.0)
                
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                else:
                    h[i + kk] = h_orig_plus
                    h[i] = np.clip(h[i] - comp, 0.0, 1.0)
            
            if improved:
                h = project_feasible(h, n)
                break
        
        if not improved:
            # Try a random perturbation
            for _ in range(10):
                idx = rng_local.randint(n)
                orig = h[idx]
                h[idx] = np.clip(orig + 0.003 * (rng_local * 2 - 1), 0.0, 1.0)
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    break
                else:
                    h[idx] = orig
                    h = project_feasible(h, n)
            else:
                if outer > 50:
                    break
    
    return h, best_c

rng_local = np.random.RandomState(999)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def multi_start_optimize(h_start, n, deadline, min_time=60):
    """Try several perturbations and keep the best."""
    best_h = project_feasible(h_start, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(555)
    
    for trial in range(8):
        if time.time() > deadline - min_time:
            break
        mag = 0.03 * (1 + 0.3 * trial)
        h_p = best_h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 2000, 3000.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-15, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    deadline = t0 + budget_s - 12
    np.random.seed(seed)

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)
    
    # Phase 1: Quick L-BFGS at base resolution (30s)
    t1 = min(t0 + 30, deadline - 600)
    if time.time() < t1 - 5:

    # Phase 2: Upsample to n=1024 (200s total for this phase)
    n2 = 1024
    if time.time() < deadline - 400:
        h1024 = interpolate_to(best_h, len(best_h), n2)
        t2 = min(time.time() + 120, deadline - 380)
        if t2 > time.time() + 15:

                configs=[(100, 20.0), (400, 80.0), (1500, 300.0),

            if c1024 < best_c:

    # Phase 5: Block coordinate descent (100s)
    if time.time() < deadline - 200:
        t5 = min(time.time() + 80, deadline - 180)
        if t5 > time.time() + 20:
            h5, c5 = block_coord_descent(best_h, len(best_h), t5, block_size=16, n_rounds=300, min_time=20)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: Upsample to n=2048 (200s)
    n3 = 2048
    if time.time() < deadline - 250:
        h2048 = interpolate_to(best_h, len(best_h), n3)
        t6 = min(time.time() + 150, deadline - 200)
        if t6 > time.time() + 20:
            h2048, c2048 = run_lbfgs(h2048, n3, t6,
                configs=[(200, 50.0), (800, 200.0), (3000, 800.0),
                         (10000, 3000.0), (30000, 10000.0)],
                min_time=30)
            if c2048 < best_c:
                best_c = c2048
                best_h = h2048
    
    # Phase 7: Weighted L-BFGS at n=2048 (80s)
    if time.time() < deadline - 150:

            h7, c7 = run_weighted_lbfgs(best_h, len(best_h), t7, n_rounds=2)

    # Phase 9: Local shift refinement (80s)

    # Final feasibility check
    n_final = len(best_h)

    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
