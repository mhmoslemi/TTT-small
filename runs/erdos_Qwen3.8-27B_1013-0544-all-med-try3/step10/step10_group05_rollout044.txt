import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smxx = sh.max()
    if smxx > 400:
        val = m + (smxx + np.log(np.sum(np.exp(sh - smxx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smxx = sh.max()
    w = np.exp(sh - smxx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

    grad = smooth_max_grad(c, alpha, x, n)

def gradient_directed_search(h, n, deadline, max_steps=200000):
    """Use gradient to find good pairs to swap mass between."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    idx_center = n - 1
    steps = 0
    
    while time.time() < deadline - 2 and steps < max_steps:
        c_vec = compute_c5_vec(best_h, n)
        cur_max = c_vec.max()
        if cur_max < best_c - 1e-15:
            best_c = cur_max
            best_h = best_h.copy()
        
        # Find active set (top correlations)
        tol = max(1e-10, 0.002 * cur_max)
        active = np.where(c_vec >= cur_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Compute gradient as weighted average of active correlations
        alpha_sm = max(100.0, 10.0 / tol)
        sh = alpha_sm * (c_vec[active] - c_vec[active].max())
        smxx = sh.max()
        w = np.exp(sh - smxx)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, idx in zip(w, active):
            if wi < 1e-14:
                continue
            k = int(idx) - idx_center
            wm = wi * dx
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += wm * (1.0 - best_h[k:n])
                    grad[k:n] += wm * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += wm * (1.0 - best_h[0:nv])
                    grad[0:nv] += wm * (-best_h[kk:n])
        
        # Projected gradient: move mass from high-grad to low-grad
        mean_g = grad.mean()
        g_proj = grad - mean_g
        
        # Find best donors (high g_proj, h > 0) and receivers (low g_proj, h < 1)
        donors = np.where((g_proj > 0) & (best_h > 1e-10))[0]
        receivers = np.where((g_proj < 0) & (best_h < 1.0 - 1e-10))[0]
        
        if len(donors) == 0 or len(receivers) == 0:
            break
        
        # Sort for top candidates
        donors = donors[np.argsort(-g_proj[donors])[:min(50, len(donors))]]
        receivers = receivers[np.argsort(g_proj[receivers])[:min(50, len(receivers))]]
        
        improved_this_round = False
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]

            if time.time() > deadline - 3:
                break

                    continue

                    if best_h[d_idx] < delta or best_h[r_idx] > 1.0 - delta:
                        continue
                    best_h[d_idx] -= delta
                    best_h[r_idx] += delta
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved_this_round = True
                        steps += 1
                        break
                    else:
                        best_h[d_idx] += delta
                        best_h[r_idx] -= delta
        
        if not improved_this_round:

    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        n_orig = 500
        h0 = np.full(n_orig, 0.5)
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Generate alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty(n_orig, golden / 2.0),
        make_beatty(n_orig, np.sqrt(2.0) / 2.0),
        make_beatty(n_orig, 0.4312),
        make_beatty(n_orig, 1.0 / 3.0),
        make_random_binary(n_orig, rng),
        make_random_binary(n_orig, rng),
    ]
    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    
    # Phase 1: Warmup L-BFGS with increasing alpha (60s)
    p1_end = min(t0 + 50, deadline - 800)
    if p1_end > t0 + 10:
        bounds = [(0.0, 1.0)] * best_n
        for alpha in [500, 2000, 10000, 50000, 200000]:
            if time.time() > p1_end:
                break
            rem = p1_end - time.time() - 3
            if rem < 10:
                break
            maxit = max(100, int(rem * 4))
            maxit = min(maxit, 1000)
            try:
                res = minimize(full_obj, best_h, jac=full_grad,
                              args=(best_n, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 2: Active-set focused L-BFGS (200s)
    p2_end = min(time.time() + 200, deadline - 500)
    if p2_end > time.time() + 30:
        cur_n = best_n
        bounds = [(0.0, 1.0)] * cur_n
        configs = [
            (3, 20000, 0.005),
            (4, 50000, 0.002),
            (5, 100000, 0.001),
            (6, 200000, 0.001),
            (8, 500000, 0.0005),
            (10, 1000000, 0.0005),
            (12, 2000000, 0.0003),
            (15, 5000000, 0.0003),
            (20, 10000000, 0.0002),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 15:
                break
            c_vec = compute_c5_vec(best_h, cur_n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-8, 0.001 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            
            rem = p2_end - time.time() - 10
            if rem < 15:
                break
            maxit = max(200, int(rem * 3))
            maxit = min(maxit, 2000)
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(cur_n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, cur_n)
                c_new = compute_c5(h_new, cur_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 3: Gradient-directed local search (250s)
    p3_end = min(time.time() + 250, deadline - 300)
    if p3_end > time.time() + 30:
        cur_n = len(best_h)
        h3, c3 = gradient_directed_search(best_h, cur_n, p3_end, max_steps=100000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        # Second pass with fresh gradient info
        if time.time() < p3_end - 30:
            h3, c3 = gradient_directed_search(best_h, cur_n, p3_end, max_steps=50000)
            if c3 < best_c:
                best_c = c3
                best_h = h3
    
    # Phase 4: Try higher resolution n*2 (150s)
    n_up = best_n * 2
    p4_end = min(time.time() + 150, deadline - 200)
    if p4_end > time.time() + 40 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.001:
            bounds_up = [(0.0, 1.0)] * n_up
            for alpha in [5000, 50000, 500000, 2000000, 10000000]:
                if time.time() > p4_end - 20:
                    break
                rem = p4_end - time.time() - 10
                if rem < 15:
                    break
                maxit = max(200, int(rem * 2))
                maxit = min(maxit, 1500)
                try:
                    res = minimize(full_obj, h_up, jac=full_grad,
                                  args=(n_up, alpha, 0.005), method='L-BFGS-B',
                                  bounds=bounds_up,
                                  options={'maxiter': maxit, 'ftol': 1e-25,
                                           'gtol': 1e-18, 'maxfun': maxit * 5,
                                           'maxls': 40, 'maxcor': 40})
                    h_new = project_feasible(res.x, n_up)
                    c_new = compute_c5(h_new, n_up)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new
                        best_n = n_up
                        h_up = h_new
                        bounds_up = [(0.0, 1.0)] * n_up
                except Exception:
                    pass
            # Active set at higher res
            if best_n == n_up and time.time() < p4_end - 20:
                c_vec = compute_c5_vec(best_h, n_up)
                threshold = c_vec.max() - 0.001 * c_vec.max()
                active = np.where(c_vec >= threshold)[0]
                if len(active) > 10:
                    active = np.array(np.argsort(-c_vec[active])[:10])
                rem = p4_end - time.time() - 10
                if rem > 20:
                    try:
                        res = minimize(active_obj, best_h, jac=active_grad,
                                      args=(n_up, active, 500000, 0.001),
                                      method='L-BFGS-B', bounds=bounds_up,
                                      options={'maxiter': min(int(rem*2), 1000),
                                               'ftol': 1e-25, 'gtol': 1e-18,
                                               'maxfun': 3000, 'maxls': 50, 'maxcor': 50})
                        h_new = project_feasible(res.x, n_up)
                        c_new = compute_c5(h_new, n_up)
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h_new
                    except Exception:
                        pass
            # Gradient-directed at higher res
            if best_n == n_up and time.time() < p4_end - 15:
                h_g, c_g = gradient_directed_search(best_h, n_up, p4_end, max_steps=30000)
                if c_g < best_c:
                    best_c = c_g
                    best_h = h_g
    
    # Phase 5: More active-set rounds at best resolution (120s)
    p5_end = min(time.time() + 120, deadline - 150)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        bounds = [(0.0, 1.0)] * cur_n
        for top_k, alpha, mu in [(3, 5000000, 0.0002), (5, 10000000, 0.0002),
                                  (8, 20000000, 0.0001), (12, 50000000, 0.0001)]:
            if time.time() > p5_end - 15:
                break
            c_vec = compute_c5_vec(best_h, cur_n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                continue
            rem = p5_end - time.time() - 10
            if rem < 15:
                break
            maxit = max(200, int(rem * 3))
            maxit = min(maxit, 1500)
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                              args=(cur_n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 5,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, cur_n)
                c_new = compute_c5(h_new, cur_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 6: Ultra-fine gradient-directed search (150s)
    p6_end = min(time.time() + 150, deadline - 80)
    if p6_end > time.time() + 20:
        cur_n = len(best_h)
        h6, c6 = gradient_directed_search(best_h, cur_n, p6_end, max_steps=200000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Perturbation + reoptimization (60s)
    p7_end = min(time.time() + 60, deadline - 40)
    if p7_end > time.time() + 20:
        cur_n = len(best_h)
        bounds = [(0.0, 1.0)] * cur_n
        for t in range(4):
            if time.time() > p7_end - 15:
                break
            amp = 0.005 * (0.5 ** t)
            h_pert = best_h + rng.uniform(-amp, amp, size=cur_n)
            h_pert = project_feasible(h_pert, cur_n)
            c_pert = compute_c5(h_pert, cur_n)
            if c_pert < best_c + 0.003:
                rem = p7_end - time.time() - 10
                if rem > 20:
                    try:
                        c_vec = compute_c5_vec(h_pert, cur_n)
                        threshold = c_vec.max() - 0.001 * c_vec.max()
                        active = np.where(c_vec >= threshold)[0]
                        if len(active) > 8:
                            active = np.array(np.argsort(-c_vec[active])[:8])
                        res = minimize(active_obj, h_pert, jac=active_grad,
                                      args=(cur_n, active, 1000000, 0.0005),
                                      method='L-BFGS-B', bounds=bounds,
                                      options={'maxiter': min(int(rem*3), 800),
                                               'ftol': 1e-25, 'gtol': 1e-18,
                                               'maxfun': 2000, 'maxls': 40, 'maxcor': 40})
                        h_new = project_feasible(res.x, cur_n)
                        c_new = compute_c5(h_new, cur_n)
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h_new
                    except Exception:
                        pass
    
    # Phase 8: Final gradient-directed polish (remaining time)
    p8_end = deadline - 2
    if p8_end > time.time() + 10:
        cur_n = len(best_h)
        h8, c8 = gradient_directed_search(best_h, cur_n, p8_end, max_steps=100000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
