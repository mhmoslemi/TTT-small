                if mu_decrease and cur_mu > 0.001:

            elif c_new <= best_c + 1e-15:
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run_active_opt(h, n, deadline, top_k=10, alphas=None, rounds=15):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200000, 1000000, 5000000, 20000000, 100000000]
    
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 25:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int((deadline - time.time()) * 3))
            try:
                res = minimize(as_obj, best_h, jac=as_grad,
                              args=(n, active, alpha, 0.002),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 8,
                                       'maxls': 50, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, best_c

def run_pow_opt(h, n, deadline, ps=None, mu=0.005):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if ps is None:
        ps = [10, 20, 50, 100, 200, 500, 1000]
    
    for i, p in enumerate(ps):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        maxit = max(400, int(remaining * 4))
        try:
            res = minimize(pow_obj, best_h, jac=pow_grad,
                          args=(n, p, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def perturbation_restarts(h, n, deadline, seed, n_restarts=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    bounds = [(0.0, 1.0)] * n
    
    time_per_restart = max(30, (deadline - time.time()) / max(1, n_restarts))
    
    for r in range(n_restarts):
        if time.time() > deadline - 20:
            break
        # Generate structured perturbation
        x = np.linspace(0, 2, n)
        amp = 0.02 + 0.01 * r
        freq = rng.randint(3, 15)
        phase = rng.uniform(0, 2 * np.pi)
        pert = amp * np.sin(freq * np.pi * x / 2.0 + phase)
        # Add small random noise
        pert += 0.005 * rng.randn(n)
        h_pert = project_feasible(h + pert, n)
        c_pert = compute_c5(h_pert, n)
        
        if c_pert >= best_c:
            continue
        
        # Quick optimization from perturbed start
        t_sub = min(time.time() + time_per_restart, deadline - 15)
        h_opt, c_opt = run_lbfgs(h_pert, n, t_sub,
            alphas=[5000, 50000, 500000], mu=0.003,
            maxiters=[800, 800, 800])
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = best_c
    # Only consider pairs that affect the max lags
    active_lags = np.where(c_vec >= max_val - 0.005 * max_val)[0]
    
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving from j to i
                    if best_h[j] >= d and best_h[i] + d <= 1.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try moving from i to j
                    if best_h[i] >= d and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h[k:]) * dx
                grad[k:] += (-h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h[:nv]) * dx
                grad[:nv] += (-h[kk:]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.003, 0.01)
        else:
            eta *= 0.97
        
        h = h_new
    
    return best_h, best_c
