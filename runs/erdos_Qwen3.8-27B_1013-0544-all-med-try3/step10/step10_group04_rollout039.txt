        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n):
    """Round to binary maintaining exactly n/2 ones."""
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    target = n // 2
    current_ones = int(h_bin.sum())
    if current_ones > target:
        # Need to remove (current_ones - target) ones: set the ones with smallest h to 0
        one_indices = np.where(h_bin == 1)[0]
        order = np.argsort(-h[one_indices])  # descending h, so reverse gives ascending
        for idx in order[:current_ones - target]:
            h_bin[one_indices[idx]] = 0
    elif current_ones < target:
        # Need to add (target - current_ones) ones: set the zeros with largest h to 1
        zero_indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_indices])  # descending h
        for idx in order[:target - current_ones]:
            h_bin[zero_indices[idx]] = 1
    return h_bin

def thue_morse_binary(n):
    """Generate Thue-Morse sequence of length n with exactly n/2 ones."""
    # Thue-Morse: t(i) = popcount(i) mod 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Adjust to have exactly n/2 ones
    h_bin = np.where(tm == 1, 1.0, 0.0)
    target = n // 2
    current = int(h_bin.sum())
    if current > target:
        one_idx = np.where(h_bin == 1)[0]
        rng = np.random.RandomState(0)
        order = rng.permutation(len(one_idx))
        for i in order[:current - target]:
            h_bin[one_idx[i]] = 0
    elif current < target:
        zero_idx = np.where(h_bin == 0)[0]
        rng = np.random.RandomState(0)
        order = rng.permutation(len(zero_idx))
        for i in order[:target - current]:
            h_bin[zero_idx[i]] = 1
    return h_bin

def binary_sa(h_start, n, deadline, T0=0.05, cooling=0.9999, seed=42):
    """Simulated annealing for binary optimization."""
    h = h_start.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    best_h = h.copy()
    best_c = c_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rng = np.random.RandomState(seed)
    T = T0
    min_T = 1e-10
    
    for it in range(10000000):
        if time.time() > deadline:
            break
        if T < min_T:
            break
        
        ai = rng.randint(len(ones))
        bi = rng.randint(len(zeros))
        a = int(ones[ai])
        b = int(zeros[bi])
        
        h[a] = 0.0
        h[b] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c = float(np.max(new_corr)) * dx
        delta = new_c - c_max
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            corr = new_corr * dx
            c_max = new_c
            ones[ai] = b
            zeros[bi] = a
            if c_max < best_c:
                best_c = c_max
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        
        T *= cooling
    
    return best_h, best_c

def smooth_obj(x, n, alpha, mu, bp=0.0):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    if bp > 0:
        val += bp * np.sum(x * (1.0 - x))
    return float(val)

def smooth_grad(x, n, alpha, mu, bp=0.0):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    if bp > 0:
        grad += bp * (2.0 * x - 1.0)
    return grad
