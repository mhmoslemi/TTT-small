                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10 and n_ones < n:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)

def upsample(h, n_old, n_new):
    x_fine = np.linspace(0, 2, n_new)
    x_coarse = np.linspace(0, 2, n_old)
    h_new = np.interp(x_fine, x_coarse, h)
    return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_focus(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [2000, 10000, 50000, 200000, 1000000]
    for rnd in range(15):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(400, int((deadline - time.time()) * 2.5))
            try:
                res = minimize(focus_obj, best_h, jac=focus_grad,
                              args=(n, active, alpha, 0.005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 35})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def nesterov_subgrad(h, n, deadline, n_iter=5000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    x = best_h.copy()
    v = np.zeros(n)
    eta = 0.003
    t = 1.0
    tol_expand = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(x, n)
        c_max = c_vec.max()
        tol = max(1e-9, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        off = n - 1
        for idx in active:
            k = idx - off
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - x[k:n]) * dx
                grad[k:n] += (-x[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - x[0:nv]) * dx
                grad[0:nv] += (-x[kk:n]) * dx
        grad /= len(active)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = x + beta * (x - (x - eta * grad) + v * 0) if it > 0 else x
        y = x + beta * v
        y = x - eta * grad
        x_new = project_feasible(y, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = x_new.copy()
            eta = min(eta * 1.003, 0.015)
        else:
            eta *= 0.97
        v = x_new - x
        x = x_new
    return best_h, best_c

def coord_descent(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=200000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Try diverse periodic starts at n_orig (120s)
    p1_end = t0 + 120
    periods = [2, 3, 4, 5, 7, 10, 14, 20, 25]
    for per in periods:
        if time.time() > p1_end:
            break
        onp = max(1, per // 2)
        h_ps = periodic_start(n_orig, per, onp)
        c_ps = compute_c5(h_ps, n_orig)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps.copy()
        # Quick L-BFGS on this start
        part_end = min(time.time() + 10, p1_end - 5)
        if part_end > time.time() + 3:
            h_t, c_t = run_lbfgs(h_ps, n_orig, part_end,
                alphas=[100, 1000, 10000], mu=0.01,
                maxiter_per=[500, 500, 800])
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()
    # Also optimize the initial_h_values
    if time.time() < p1_end:
        h_init, c_init = run_lbfgs(h0, n_orig, p1_end,
            alphas=[50, 200, 2000, 20000], mu=0.01,
            maxiter_per=[1500, 1500, 2000, 2000])
        if c_init < best_c:
            best_c = c_init
            best_h = h_init.copy()

    # Phase 2: Full L-BFGS alpha annealing on best (200s)
    p2_end = min(time.time() + 200, deadline - 400)
    h2, c2 = run_lbfgs(best_h, best_n, p2_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
        mu=0.01, maxiter_per=[1500, 1500, 2000, 2000, 2500, 2500, 3000, 3000])
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Upsample to n=1000 and optimize (180s)
    n_high = 1000
    if time.time() < deadline - 300:
        p3_end = min(time.time() + 180, deadline - 250)
        h_up = upsample(best_h, best_n, n_high)
        c_up = compute_c5(h_up, n_high)
        h3, c3 = run_lbfgs(h_up, n_high, p3_end,
            alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
            mu=0.005, maxiter_per=[1000, 1500, 1500, 2000, 2000, 2500, 2500])
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
            best_n = n_high

    # Phase 4: Iterative top-k focus optimization (150s)
    if time.time() < deadline - 150:
        p4_end = min(time.time() + 150, deadline - 120)
        h4, c4 = run_focus(best_h, best_n, p4_end, top_k=5,
            alphas=[5000, 20000, 100000, 500000, 2000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()

    # Phase 5: Nesterov subgradient (120s)
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 120, deadline - 60)
        h5, c5 = nesterov_subgrad(best_h, best_n, p5_end, n_iter=4000)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Phase 6: Try n=750 as alternative resolution (80s)
    n_mid = 750
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 80, deadline - 50)
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        h6, c6 = run_lbfgs(h_mid, n_mid, p6_end,
            alphas=[50, 500, 5000, 50000, 200000],
            mu=0.005, maxiter_per=[1000, 1500, 1500, 2000, 2000])
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
            best_n = n_mid

    # Phase 7: Focus + Nesterov at best resolution (100s)
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 50, deadline - 50)
        h7a, c7a = run_focus(best_h, best_n, p7_end, top_k=3,
            alphas=[50000, 200000, 1000000])
        if c7a < best_c:
            best_c = c7a
            best_h = h7a.copy()
        if time.time() < deadline - 40:
            p7b_end = min(time.time() + 40, deadline - 30)
            h7b, c7b = nesterov_subgrad(best_h, best_n, p7b_end, n_iter=3000)
            if c7b < best_c:
                best_c = c7b
                best_h = h7b.copy()

    # Phase 8: SLSQP polish (40s)
    if time.time() < deadline - 30:
        for alpha in [100000, 500000]:
            if time.time() > deadline - 25:
                break
            h8, c8 = run_slqp(best_h, best_n, min(time.time() + 18, deadline - 20),
                             alpha=alpha, maxiter=800)
            if c8 < best_c:
                best_c = c8
                best_h = h8.copy()

    # Phase 9: Coordinate descent polish (30s)
    if time.time() < deadline - 5:
        h9, c9 = coord_descent(best_h, best_n, deadline - 3, radius=30)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()

    # Phase 10: Final tiny L-BFGS
    if time.time() < deadline - 2:
        h10, c10 = run_lbfgs(best_h, best_n, deadline - 1,
            alphas=[500000, 2000000], mu=0.0, maxiter_per=[300, 300])
        if c10 < best_c:
            best_c = c10
            best_h = h10.copy()

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
