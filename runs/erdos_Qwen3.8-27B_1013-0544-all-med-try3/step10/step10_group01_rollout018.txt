import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbf_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbf_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def penalty_obj(x, n, T, alpha, mu):
    """Penalty: T + alpha * sum(max(0, c_k - T)^2) + mu*(sum-n/2)^2"""
    c = compute_c5_vec(x, n)
    excess = np.maximum(0, c - T)
    val = T + alpha * np.sum(excess**2)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, T, alpha, mu):
    c = compute_c5_vec(x, n)
    excess = np.maximum(0, c - T)
    # d/dx of alpha * sum(excess^2) = alpha * 2 * excess * dc/dx (where excess > 0)
    active = excess > 0
    dx = 2.0 / n
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for idx in active_idx:
        k = idx - (n - 1)
        wm = 2.0 * alpha * excess[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def multi_lbfgs(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = max(500, int(remaining * 3))
        try:
            res = minimize(lbf_obj, best_h, jac=lbf_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def penalty_optimize(h, n, deadline, T_init=None, alphas_pen=None, rounds=3):

    bounds = [(0.0, 1.0)] * n
    if T_init is None:
        T_init = best_c * 0.99
    if alphas_pen is None:
        alphas_pen = [1000, 10000, 100000, 1000000]
    T = T_init

                                       'gtol': 1e-14, 'maxfun': maxit * 8,

                h_new = project_feasible(res.x, n)

                    best_h = h_new
                    T = c_new * 0.995
            except Exception:
                pass
    return best_h, best_c

    best_c = compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_c = best_c

            break

        T = T_start * (T_end / T_start) ** frac
        # Pick random pair: one to decrease, one to increase

        j = np.random.randint(n)

            cur_h[i] += delta

            else:
                cur_h[j] -= delta
                cur_h[i] += delta

def coord_descent_fine(h, n, deadline, step_sizes=None, max_iter=200000):
    """Fine coordinate descent focusing on active lags"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    total_iter = 0
    for step in step_sizes:
        while time.time() < deadline and total_iter < max_iter:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            # Compute gradient for active set
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            # Find best pair: decrease where grad > 0, increase where grad < 0
            pos_mask = (grad > 1e-10) & (best_h > 1e-8)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            # Try top 30 of each
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain < 1e-12:
                break
            amt = min(step, best_h[best_i] * 0.5, (1.0 - best_h[best_j]) * 0.5)
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            total_iter += 1
            if c_new >= best_c - 1e-15:
                # Revert
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) == n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += min(remaining, 1.0 - h[zeros[0]])
    return project_feasible(h, n)

def make_periodic_perturbed(n, period, duty, rng, noise=0.1):
    """Create a periodic pattern with perturbation"""

    h += rng.normal(0, noise, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbf_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lbf_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def penalty_obj(x, n, T, alpha_pen, mu):
    c = compute_c5_vec(x, n)
    excess = np.maximum(0.0, c - T)
    val = T + alpha_pen * np.sum(excess * excess)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, T, alpha_pen, mu):
    c = compute_c5_vec(x, n)
    excess = np.maximum(0.0, c - T)
    active_mask = excess > 1e-14
    dx = 2.0 / n
    grad = np.zeros(n)
    active_idx = np.where(active_mask)[0]
    for idx in active_idx:
        k = idx - (n - 1)
        wm = 2.0 * alpha_pen * excess[idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        if time.time() > deadline:
            break
        frac = min(1.0, it / anneal_div)
        T_curr = T_start * (T_end / T_start) ** frac
        i = rng_sa.randint(n)
        j = rng_sa.randint(n)
        if i == j:
            continue
        delta = rng_sa.uniform(0.002, 0.008)
        if rng_sa.random() < 0.5:
            d = min(delta, cur_h[i], 1.0 - cur_h[j])
        else:
            d = min(delta, cur_h[j], 1.0 - cur_h[i])
        if d < 1e-7:
            continue
        if rng_sa.random() < 0.5:
            cur_h[i] -= d
            cur_h[j] += d
        else:
            cur_h[j] -= d
            cur_h[i] += d
        c_new = compute_c5(cur_h, n)
        accepted = False
        if c_new < cur_c:
            accepted = True
        elif T_curr > 1e-15:
            diff_val = cur_c - c_new
            if diff_val < 50 * T_curr:
                accepted = rng_sa.random() < np.exp(diff_val / T_curr)
        if accepted:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = cur_h.copy()
        else:
            if rng_sa.random() < 0.5 and it % 2 == 0:
                cur_h[i] += d
                cur_h[j] -= d
            elif it % 2 == 0:
                cur_h[j] += d
                cur_h[i] -= d
            else:
                cur_h[i] += d
                cur_h[j] -= d
            cur_h = np.clip(cur_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def coord_descent_fine(h, n, deadline, step_sizes=None, max_iter=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    total_iter = 0
    for step in step_sizes:
        while time.time() < deadline and total_iter < max_iter:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.004 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 1e-10) & (best_h > 1e-8)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                gi = grad[i]
                for j in neg_idx[:40]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain < 1e-12:
                break
            amt = min(step, best_h[best_i] * 0.4, (1.0 - best_h[best_j]) * 0.4)
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            total_iter += 1
            if c_new >= best_c - 1e-15:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def active_set_lbfgs(h, n, deadline, alphas, top_frac=0.008, max_lags=16):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 25:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, top_frac * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > max_lags:
            order = np.argsort(-c_vec[active])
            active = active[order[:max_lags]]
        if len(active) == 0:
            continue
        maxit = max(300, int(remaining * 1.5))
        try:
            res = minimize(lbf_obj, best_h, jac=lbf_grad,
                          args=(n, alpha, 0.001), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-15, 'maxfun': maxit * 7,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 5 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) == n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += min(remaining, 1.0)
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def upsample(h, factor):
    return np.repeat(h, factor)

def downsample_mean(h, n_to):
    h_from = len(h)
    h_new = np.zeros(n_to)
    for i in range(n_to):
        lo = int(i * h_from / n_to)
        hi = int((i + 1) * h_from / n_to)
        if hi <= lo:
            hi = lo + 1
        h_new[i] = h[lo:hi].mean()
    return project_feasible(h_new, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except (NameError, Exception):
        n = 500
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    best_n = n

    # Generate diverse initializations
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        make_beatty(n, golden / 2.0),
        make_beatty(n, golden / 3.0),
        make_beatty(n, np.sqrt(2.0) / 2.0),
        make_beatty(n, np.sqrt(3.0) / 2.0),
        make_beatty(n, np.pi / 4.0),
        make_beatty(n, 0.157),
        make_random_binary(n, rng),
        make_random_binary(n, rng),
        make_random_binary(n, rng),
    ]
    for h_s in starts:
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s.copy()

    # === Phase 1: Multi-resolution warm start ===
    p1_end = min(t0 + 100, deadline - 500)
    if time.time() < p1_end and n > 200:
        n_small = 200
        h_small = downsample_mean(best_h, n_small)
        h_small, c_small = multi_lbfgs(h_small, n_small, min(time.time() + 35, p1_end - 20),
                                       alphas=[10, 50, 200, 2000], mu=0.1)
        h_up = upsample(h_small, n // n_small)
        if len(h_up) > n:
            h_up = h_up[:n]
        elif len(h_up) < n:
            pad = np.full(n - len(h_up), 0.5)
            h_up = np.concatenate([h_up, pad])
        h_up = project_feasible(h_up, n)
        c_up = compute_c5(h_up, n)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up

    # === Phase 2: Full L-BFGS at working resolution ===
    p2_end = min(time.time() + 200, deadline - 450)
    if time.time() < p2_end:
        h2, c2 = multi_lbfgs(best_h, best_n, p2_end,
                            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
                            mu=0.01)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # === Phase 3: Penalty method (quadratic excess) ===
    p3_end = min(time.time() + 180, deadline - 350)
    if time.time() < p3_end:
        h3, c3 = penalty_optimize(best_h, best_n, p3_end,
                                  T_init=best_c * 0.985,
                                  alphas_pen=[10000, 100000, 1000000, 5000000],
                                  rounds=4)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # === Phase 4: High-alpha active set L-BFGS ===
    p4_end = min(time.time() + 180, deadline - 250)
    if time.time() < p4_end:
        h4, c4 = active_set_lbfgs(best_h, best_n, p4_end,
                                  alphas=[5000000, 20000000, 100000000],
                                  top_frac=0.006, max_lags=14)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # === Phase 5: Simulated annealing ===
    p5_end = min(time.time() + 140, deadline - 160)
    if time.time() < p5_end:
        h5, c5 = simulated_annealing(best_h, best_n, p5_end,
                                     n_iter=150000, T_start=0.003, T_end=0.00005)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # === Phase 6: Fine coordinate descent ===
    p6_end = min(time.time() + 100, deadline - 80)
    if time.time() < p6_end:
        h6, c6 = coord_descent_fine(best_h, best_n, p6_end,
                                    step_sizes=[0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                                0.00005, 0.00002, 0.00001],
                                    max_iter=80000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # === Phase 7: Second SA pass with lower temperature ===
    p7_end = min(time.time() + 80, deadline - 60)
    if time.time() < p7_end:
        h7, c7 = simulated_annealing(best_h, best_n, p7_end,
                                     n_iter=80000, T_start=0.001, T_end=0.00002)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # === Phase 8: Try higher resolution ===
    p8_end = min(time.time() + 80, deadline - 40)
    if time.time() < p8_end:
        n2 = best_n * 2
        h2_up = upsample(best_h, 2)
        h2_up = project_feasible(h2_up, n2)
        c2_up = compute_c5(h2_up, n2)
        if c2_up < best_c + 0.005:
            h2_opt, c2_opt = multi_lbfgs(h2_up, n2, min(p8_end - 10, time.time() + 50),
                                         alphas=[500, 5000, 50000], mu=0.005)
            if c2_opt < best_c:
                best_c = c2_opt
                best_h = h2_opt
                best_n = n2

    # === Phase 9: Final coordinate descent polish ===
    if time.time() < deadline - 3:
        h9, c9 = coord_descent_fine(best_h, best_n, deadline - 2,
                                    step_sizes=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                    max_iter=40000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
