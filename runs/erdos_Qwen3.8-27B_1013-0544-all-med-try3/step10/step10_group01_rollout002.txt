    sh = alpha * (c - m)

        max_val = c_vec.max()
        tol = max(1e-8, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 20:
            active = np.array(np.argsort(-c_vec[active])[:20])
        grad = topk_active_grad(c_vec, active, best_h, n)
        h_trial = best_h - step * grad
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-16:
            best_c = c_trial
            best_h = h_trial
            no_improve = 0
        else:
            no_improve += 1
            step *= step_decay
            if no_improve > patience or step < 1e-7:
                break
    return best_h, best_c

def fast_pair_search(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def vectorized_coord_descent(h, n, deadline, step_init=0.01, max_iters=100000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    step = step_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.005 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 30:
            active = np.array(np.argsort(-c_vec[active])[:30])
        grad = topk_active_grad(c_vec, active, best_h, n)
        pos_mask = (grad > 1e-8) & (best_h > 1e-8)
        neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0][:30]
        neg_idx = np.where(neg_mask)[0][:30]
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in pos_idx:
            for j in neg_idx:
                g = grad[i] - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
        if amt < 1e-10:
            break
        best_h[best_i] -= amt
        best_h[best_j] += amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-17:
            best_c = c_new
        else:
            best_h[best_i] += amt
            best_h[best_j] -= amt
            best_h = project_feasible(best_h, n)
            step *= 0.9
            if step < 1e-8:
                break
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def interpolate_start(h, n_target):
    n_orig = len(h)
    if n_orig == n_target:
        return project_feasible(h.copy(), n_target)
    x_orig = np.linspace(0, 2, n_orig, endpoint=False) + 1.0 / n_orig
    x_new = np.linspace(0, 2, n_target, endpoint=False) + 1.0 / n_target
    return project_feasible(np.interp(x_new, x_orig, h), n_target)

def periodic_smooth_start(n, period_frac=0.5, num_periodic=4):
    h = np.zeros(n)
    n_ones = n // 2
    block_size = n // num_periodic
    for blk in range(num_periodic):
        start = blk * block_size
        end = start + block_size
        nblk = end - start
        ones_in_block = n_ones // num_periodic + (1 if blk < n_ones % num_periodic else 0)
        step = nblk / max(1, ones_in_block)
        for k in range(ones_in_block):
            pos = start + int(k * step)
            if pos < end:
                h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h0)
    except Exception:
        h0 = None
        n_parent = 500

    n = n_parent
    best_h = h0.copy() if h0 is not None else np.full(n, 0.5)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    # Generate diverse starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [('parent', best_h.copy())]
    starts.append(('beatty_g', make_beatty_start(n, golden / 2.0)))
    starts.append(('beatty_s2', make_beatty_start(n, np.sqrt(2.0) / 2.0)))
    starts.append(('beatty_s3', make_beatty_start(n, np.sqrt(3.0) / 2.0)))
    starts.append(('rand1', make_random_binary(n, rng)))
    starts.append(('rand2', make_random_binary(n, rng)))
    starts.append(('rand3', make_random_binary(n, rng)))
    starts.append(('periodic', periodic_smooth_start(n)))

    for name, hs in starts:
        c_s = compute_c5(hs, n)
        if c_s < best_c:
            best_c = c_s
            best_h = hs.copy()

    # Phase 1: L-BFGS warmup (200s)
    p1_end = t0 + 180
    h1, c1 = run_lbfgs(best_h, n, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000, 200000],
        mu=0.01, maxit_list=[800, 1200, 1500, 2000, 2500, 3000, 3500])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization (180s)
    if time.time() < deadline - 280:
        p2_end = min(time.time() + 170, deadline - 260)
        h2, c2 = run_active_set(best_h, n, p2_end,
            top_k_list=[4, 6, 8, 12, 16, 24],
            alphas_list=[50000, 200000, 500000, 2000000, 5000000, 10000000],
            rounds_per=3)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient descent (120s)
    if time.time() < deadline - 180:
        p3_end = min(time.time() + 110, deadline - 170)
        h3, c3 = subgradient_descent(best_h, n, p3_end,
            step_init=0.015, step_decay=0.998, patience=300)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Vectorized coordinate descent (100s)
    if time.time() < deadline - 130:
        p4_end = min(time.time() + 90, deadline - 120)
        h4, c4 = vectorized_coord_descent(best_h, n, p4_end,
            step_init=0.005, max_iters=200000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Active set again (80s)
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 70, deadline - 70)
        h5, c5 = run_active_set(best_h, n, p5_end,
            top_k_list=[8, 16, 32],
            alphas_list=[1000000, 5000000, 20000000],
            rounds_per=4)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fast pair search (80s)
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 70, deadline - 50)
        h6, c6 = fast_pair_search(best_h, n, p6_end, radius=50,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine coordinate descent (50s)
    if time.time() < deadline - 35:
        p7_end = min(time.time() + 45, deadline - 30)
        h7, c7 = vectorized_coord_descent(best_h, n, p7_end,
            step_init=0.001, max_iters=100000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Very fine pair search (30s)
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 25, deadline - 15)
        h8, c8 = fast_pair_search(best_h, n, p8_end, radius=30,
            deltas=[0.0001, 0.00005, 0.00002, 0.00001])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final L-BFGS polish (15s)
    if time.time() < deadline - 10:
        p9_end = deadline - 5
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(lm_obj, best_h, jac=lm_grad,
                          args=(n, 50000000, 0.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': 3000,
                                   'maxls': 30, 'maxcor': 30})
            h9 = project_feasible(res.x, n)
            c9 = compute_c5(h9, n)
            if c9 < best_c:
                best_c = c9
                best_h = h9
        except Exception:
            pass

    # Phase 10: Ultimate fine tuning
    if time.time() < deadline - 3:
        h10, c10 = vectorized_coord_descent(best_h, n, deadline - 2,
            step_init=0.0001, max_iters=50000)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final: try a few random perturbations to escape local optima
    if time.time() < deadline - 2:
        for _ in range(5):
            pert = best_h + (rng.randn(n) * 0.001)
            pert = project_feasible(pert, n)
            c_p = compute_c5(pert, n)
            if c_p < best_c:
                best_c = c_p
                best_h = pert
                # Quick polish
                if time.time() < deadline - 3:
                    hp, cp = vectorized_coord_descent(best_h, n, deadline - 2,
                        step_init=0.0005, max_iters=20000)
                    if cp < best_c:
                        best_c = cp
                        best_h = hp

    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
