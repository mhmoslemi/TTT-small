def fast_local_search_binary(h, n, deadline, max_iters=500000):
    """Simulated annealing / hill climbing for binary h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = cur_c * 0.1  # initial temperature
    T_min = 1e-10
    cooling = 0.9995
    
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Compute new correlation
        h[i] = 0
        h[j] = 1
        new_corr = np.correlate(h, 1-h, mode='full')
        new_c = new_corr.max() * dx
        
        if new_c < cur_c:
            # Accept
            cur_c = new_c
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            # i moves to zeros, j moves to ones
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # Maybe accept (simulated annealing)
            if rng.random() < np.exp(-(new_c - cur_c) / max(T, T_min)):
                cur_c = new_c
                ones = ones[ones != i]
                zeros = zeros[zeros != j]
                ones = np.append(ones, j)
                zeros = np.append(zeros, i)
            else:
                # Reject
                h[i] = 1
                h[j] = 0
        
        T *= cooling
    
    return best_h, best_c

def lsqp_smooth(h, n, alpha, maxiter, deadline):
    """SLSQP with soft-max."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

For a binary function on [0,2] with measure 1 (half the domain), the "overlap" at shift k is the measure of points in A whose k-shift is not in A.

If A is an interval of length 1, say [0,1], then:
- For k ∈ [0, 1]: A \ (A+k) = [0, 1-k], measure = 1-k. Max at k→0: ≈1.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    hb = (h >= 0.5).astype(np.float64)
    target = n // 2
    if hb.sum() > target:
        op = np.where(hb == 1)[0]
        excess = int(hb.sum() - target)
        rem = op[np.argsort(h[op])[:excess]]
        hb[rem] = 0.0
    elif hb.sum() < target:
        zp = np.where(hb == 0)[0]
        deficit = int(target - hb.sum())
        ad = zp[np.argsort(-h[zp])[:deficit]]
        hb[ad] = 1.0
    return hb

def vectorized_binary_search(h, n, deadline, max_sweeps=200000, seed=42, use_sa=True, kick_interval=500):
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64).copy()
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)

    n_full = 2 * n - 1
    u = np.arange(n_full) - (n - 1)

    T = 0.3
    T_min = 1e-12
    cool_rate = 0.99998
    sweeps_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep > 0 and sweep % 2000 == 0:
            corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
            cur_c = float(corr.max())

        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]

        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        n_zeros = len(zeros_idx)
        J = zeros_idx[:, None].astype(np.int64)
        U = u[None, :].astype(np.int64)
        idx1_pre = J - U
        idx2_pre = J + U
        mask1_pre = (idx1_pre >= 0) & (idx1_pre < n)
        mask2_pre = (idx2_pre >= 0) & (idx2_pre < n)
        safe1_pre = np.clip(idx1_pre, 0, n - 1)
        safe2_pre = np.clip(idx2_pre, 0, n - 1)

        improved = False
        for i in ones_idx:
            if time.time() > deadline:
                break

            idx_pos = i + u
            idx_neg = i - u
            mp = (idx_pos >= 0) & (idx_pos < n)
            mn = (idx_neg >= 0) & (idx_neg < n)
            sp = np.clip(idx_pos, 0, n - 1)
            sn = np.clip(idx_neg, 0, n - 1)
            base_i = np.where(mp, h[sp], 0.0) - np.where(mn, 1.0 - h[sn], 0.0)

            term1 = np.where(mask1_pre, 1.0 - h[safe1_pre], 0.0)
            term2 = np.where(mask2_pre, h[safe2_pre], 0.0)
            effect = term1 - term2

            new_max_all = (corr[None, :] + base_i[None, :] + effect).max(axis=1)
            best_j_loc = int(new_max_all.argmin())
            best_c_loc = float(new_max_all[best_j_loc])

            if best_c_loc < cur_c - 1e-13:
                j = int(zeros_idx[best_j_loc])
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + base_i + effect[best_j_loc]
                cur_c = best_c_loc
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break

        if improved:
            sweeps_no_improve = 0
        else:
            sweeps_no_improve += 1
            if use_sa and T > T_min:
                if sweeps_no_improve >= kick_interval:
                    n_kicks = min(5, len(ones_idx), len(zeros_idx))
                    for _ in range(n_kicks):
                        oi = np.where(h == 1.0)[0]
                        zi = np.where(h == 0.0)[0]
                        if len(oi) == 0 or len(zi) == 0:
                            break
                        ki = oi[rng.randint(len(oi))]
                        kj = zi[rng.randint(len(zi))]
                        h[ki] = 0.0
                        h[kj] = 1.0
                    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                    cur_c = float(corr.max())
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    T = 0.3
                    sweeps_no_improve = 0
                else:
                    oi = np.where(h == 1.0)[0]
                    zi = np.where(h == 0.0)[0]
                    if len(oi) == 0 or len(zi) == 0:
                        break
                    ki = oi[rng.randint(len(oi))]
                    kj = zi[rng.randint(len(zi))]
                    h[ki] = 0.0
                    h[kj] = 1.0
                    new_corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
                    new_c = float(new_corr.max())
                    if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / T):
                        cur_c = new_c
                        corr = new_corr
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = h.copy()
                    else:
                        h[ki] = 1.0
                        h[kj] = 0.0
                    T *= cool_rate
            else:
                break

    return best_h, best_c * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    try:
