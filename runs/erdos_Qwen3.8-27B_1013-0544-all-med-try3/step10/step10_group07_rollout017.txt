**Iterative projection approach:**
1. Start with h = 0.5 (feasible for box and sum)
2. Repeat:

   b. Project to [0,1]^n

3. Check convergence

To reduce f_k: we want to increase h[i]*h[i+k] for the pairs where they're not both 1. The most effective way is to set h[i] = max(h[i], h[i+k]) for the pairs with small product.

This is essentially a local search! For each k where f_k is too large, find the pair (i, i+k) with the smallest product and try to increase it.

where P1[j] = sum_{k=max(0,-j)}^{min(n-1-j, n-1)} wf[k+n-1] (a "windowed sum" of wf)
and (x ⋆ wf_c)[j] is the cross-correlation (with wf_c[k] = wf[k+n-1]).

Similarly:

where wf_c_rev[k] = wf[-k+n-1] (reversed centered wf).

- P1 using a cumulative sum of wf

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full')
    f = corr * dx
    m_val = f.max()
    sh = alpha * (f - m_val)
    sh_max = sh.max()
    exp_sh = np.exp(sh - sh_max)
    Z = exp_sh.sum()
    val = m_val + (sh_max + np.log(Z)) / alpha
    
    wf = exp_sh / Z
    
    # Gradient (O(n^2) computation)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            if 0 <= j + k < n:
                s += wf[m] * (1.0 - x[j + k])
            if 0 <= j - k < n:
                s -= wf[m] * x[j - k]
        grad[j] = dx * s
    return val, grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_optimize(h, n, alpha, deadline, max_iter=5000):
    """Projected (sub)gradient method with backtracking line search"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    eta = 1.0 / n  # initial step
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, g = lse_obj_and_grad(h, n, alpha)
        
        # Try a sequence of step sizes
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            et = eta * factor
            h_new = h - et * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                eta = et * 1.5  # increase step
                accepted = True
                break
        if not accepted:
            eta *= 0.5
            if eta < 1e-8:
                break
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_sweeps=100):
    """Pair coordinate descent: pick (i,j), redistribute mass"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = np.max(corr) * dx
    h = h.copy()
    rng = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle coordinates
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            # Pick a random partner j
            # Try a few partners
            for attempt in range(3):
                j = rng.randint(n)
                if j == i:
                    continue
                # Try moving mass from i to j (if h[i] > 0) or j to i (if h[j] > 0)
                for direction in [1, -1]:
                    # h[i] += t*direction, h[j] -= t*direction
                    # t in [0, 1] range
                    ts = [0.1, 0.3, 0.5, 0.7, 1.0]
                    if direction == 1:
                        max_t = h[i]  # h[i] - t >= 0
                    else:
                        max_t = h[j]
                    if max_t < 1e-10:
                        continue
                    for t in ts:
                        if t > max_t + 1e-10:
                            continue
                        t = min(t, max_t)
                        if t < 1e-10:
                            continue
                        # Compute new corr
                        # Change: h[i] -> h[i]+t*d, h[j] -> h[j]-t*d
                        # corr[m] changes by:
                        # For index i: (1-h[i+k]) - 1 changes... 
                        # Actually let's just compute the delta
                        d = direction * t
                        # Old contribution of h[i] to corr: h[i] * (1-h[i+k]) for valid k
                        # New: (h[i]+d) * (1-h[i+k])
                        # Delta from i: d * (1-h[i+k]) for valid k
                        # Old contribution of h[j] to corr: h[j] * (1-h[j+k])
                        # New: (h[j]-d) * (1-h[j+k])
                        # Delta from j: -d * (1-h[j+k])
                        # But also: h[i] appears in (1-h[i+k']) when k' = i-j or j-i
                        # corr[m] = sum_r h[r] * (1-h[r+k_m])
                        # h[i] appears as h[r] (r=i): contributes h[i]*(1-h[i+k_m])
                        # h[i] appears in (1-h[r+k_m]) when r+k_m=i, i.e., r=i-k_m
                        #   contributes h[i-k_m]*(-1) [derivative wrt h[i]]
                        # So changing h[i] by d changes corr[m] by:
                        #   d*(1-h[i+k_m]) - d*h[i-k_m] (for valid indices)
                        
                        # This is complex. Let me just compute the full corr for small n.
                        h[i] += d
                        h[j] -= d
                        new_corr = np.correlate(h, 1 - h, mode='full')
                        nc = np.max(new_corr) * dx
                        if nc < cur_c - 1e-14:
                            cur_c = nc
                            corr = new_corr
                            improved = True
                            if cur_c < best_c:
                                best_c = cur_c
                                best_h = h.copy()
                            break
                        else:
                            h[i] -= d
                            h[j] += d
                    if improved:
                        break
                if improved:
                    break
            if improved:
                # Continue with other i's
                pass
            # Actually if improved, we should restart the sweep since indices changed
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

Let me also think about the subgradient more carefully. The objective is:
Φ(h) = max_k f_k(h) where f_k(h) = dx * Σ_i h[i](1-h[i+k_m])

At a point h, let S = {k : f_k(h) = Φ(h)} be the active set. The subdifferential is:
∂Φ(h) = conv{∇f_k(h) : k ∈ S}

A natural subgradient is the average: g = (1/|S|) Σ_{k∈S} ∇f_k(h)

Or for the smoothed version: g = Σ_k p_k ∇f_k(h) where p_k = softmax_k(α·f_k)/Z

The gradient of f_k with respect to h[j] is:
∂f_k/∂h[j] = dx * [(1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]]

(using k_m = m-(n-1) as before)

This can be computed as:
∇f_k = dx * [(1-h shifted by -k) - (h shifted by k)] (with appropriate boundary handling)

For the linear correlation case, the "shift" operations need to handle boundaries (out-of-range → 0).

Let me implement this efficiently using numpy:

```python
def compute_all_grads(h, n):
    """Compute gradients of all f_k for all k. Returns array of shape (2n-1, n)."""
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        # grad[j] = dx * [(1-h[j+k]) if 0<=j+k<n minus h[j-k] if 0<=j-k<n]
        g = np.zeros(n)
        # First term: (1-h[j+k]) for 0<=j+k<n
        if k >= 0:
            g[:n-k] += (1 - h[k:])  # j+k < n means j < n-k, h[j+k] = h[k:]
        else:
            g[-k:] += (1 - h[:n+k])  # j+k >= 0 means j >= -k, h[j+k] = h[:n+k]
        # Second term: -h[j-k] for 0<=j-k<n
        if k >= 0:
            g[k:] -= h[:n-k]  # j-k >= 0 means j >= k
        else:
            g[:-k] -= h[-k:]  # j-k < n means j < n+k
        grads[m] = dx * g
    return grads
```

This is O(n²) total (n shifts × O(n) each). For n=250, that's 250*499 ≈ 125K operations. Fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def lse_smooth(x, n, alpha):
    dx = 2.0 / n
    f = np.correlate(x, 1 - x, mode='full') * dx
    m = f.max()
    sh = alpha * (f - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    def obj(x):
        return lse_smooth(x, n, alpha)
    def con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < lse_smooth(h, n, alpha):
            return project_feasible(res.x, n)
    except:
        pass
    return h

def fourier_obj(params, n, n_harm):
    x = np.linspace(0, 2, n)
    g = np.zeros(n)
    for k in range(1, n_harm + 1):
        g += params[2*(k-1)] * np.cos(k * np.pi * x) + params[2*(k-1)+1] * np.sin(k * np.pi * x)
    h = 0.5 + 0.5 * np.tanh(g)
    c = compute_c5(h, n)
    integral = h.sum() * 2.0 / n
    penalty = 50.0 * (integral - 1.0)**2
    return c + penalty

def optimize_fourier(n, n_harm, deadline, n_restarts=5, seed=42):
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = np.inf
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        scale = 0.5 if r == 0 else 0.3 * (1 + r)
        p0 = scale * rng.randn(2 * n_harm)
        try:
            res = minimize(fourier_obj, p0, method='L-BFGS-B',
                          options={'maxiter': 500, 'ftol': 1e-12})
        except:
            continue
        x = np.linspace(0, 2, n)
        g = np.zeros(n)
        for k in range(1, n_harm + 1):
            g += res.x[2*(k-1)] * np.cos(k * np.pi * x) + res.x[2*(k-1)+1] * np.sin(k * np.pi * x)
        h = 0.5 + 0.5 * np.tanh(g)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def subgradient_step(h, n, alpha, eta):
    """One step of projected subgradient method"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr * dx
    m = f.max()
    sh = alpha * (f - m)
    sm = sh.max()
    wf = np.exp(sh - sm)
    wf /= wf.sum()
    
    # Compute weighted gradient
    # grad[j] = dx * sum_m wf[m] * [(1-h[j+k_m])*[valid] - h[j-k_m]*[valid]]
    # where k_m = m - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            if 0 <= j + k < n:
                s += wf[m] * (1.0 - h[j + k])
            if 0 <= j - k < n:
                s -= wf[m] * h[j - k]
        grad[j] = dx * s
    
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_optimize(h, n, alpha, deadline, max_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    eta = 0.5 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new = subgradient_step(h, n, alpha, eta)
        c_new = compute_c5(h_new, n)
        if c_new < best
