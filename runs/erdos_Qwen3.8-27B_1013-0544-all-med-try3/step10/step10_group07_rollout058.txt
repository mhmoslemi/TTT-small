```python
def proj_grad_smooth_max(h, n, alpha, n_iter, step_size, deadline):
    dx = 2.0 / n
    target = n / 2.0
    lags = np.arange(-(n-1), n)  # 2n-1 lags
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute correlations
        corr = np.correlate(h, 1-h, mode='full') * dx
        m = corr.max()
        
        # Softmax weights
        sh = alpha * corr
        sh -= sh.max()
        p = np.exp(sh)
        p /= p.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for j in range(n):
            for ki, k in enumerate(lags):
                # Term 1: (1-h[j+k]) if j+k in [0,n-1]
                jp = j + k
                if 0 <= jp < n:
                    grad[j] += p[ki] * (1 - h[jp])
                # Term 2: -h[j-k] if j-k in [0,n-1]
                jm = j - k
                if 0 <= jm < n:
                    grad[j] -= p[ki] * h[jm]
        grad *= dx
        
        # Gradient step (minimize, so go opposite to gradient)
        h_new = h - step_size * grad
        
        # Project onto feasible set
        h_new = projection(h_new, n)
        
        h = h_new
    return h
```

```python
    # A[j] = sum over lags k where j+k is valid: p[k+n-1] * (1-h[j+k])

    p_full = np.zeros(N)
```

```python
def compute_gradient_fast(h, n, p_arr, dx):
    # p_arr has length 2n-1, index k corresponds to lag k-(n-1)
    # grad[j] = dx * sum_k p_arr[k] * [(1-h[j+lag_k]) - h[j-lag_k]] (for valid indices)
    
    lags = np.arange(-(n-1), n)
    # Create valid index matrices
    # For each j and each lag k, check if j+k is valid and j-k is valid
    
    # Vectorized approach using broadcasting
    j_idx = np.arange(n)[:, None]  # (n, 1)
    lags_arr = lags[None, :]  # (1, 2n-1)
    
    jp = j_idx + lags_arr  # (n, 2n-1) - positions for h[j+k]
    jm = j_idx - lags_arr  # (n, 2n-1) - positions for h[j-k]
    
    mask_p = (jp >= 0) & (jp < n)  # (n, 2n-1)
    mask_m = (jm >= 0) & (jm < n)  # (n, 2n-1)
    
    # Clip indices for safe indexing
    jp_clipped = np.clip(jp, 0, n-1)
    jm_clipped = np.clip(jm, 0, n-1)
    
    # (1-h)[j+k] for valid positions
    term1 = np.where(mask_p, (1 - h)[jp_clipped], 0) * p_arr[None, :]
    # h[j-k] for valid positions
    term2 = np.where(mask_m, h[jm_clipped], 0) * p_arr[None, :]
    
    grad = dx * (term1.sum(axis=1) - term2.sum(axis=1))
    return grad
```

new_c[k] ≈ corr[k] + flip_plus[i, k] - flip_plus[j, k]

For all pairs (i, j) with i in zeros, j in ones:
new_max[i, j] = max_k (corr[k] + flip_plus[i,k] - flip_plus[j,k])

This is a (n_zero × n_one) matrix where each entry is a max over 2n-1 values.

I can compute this as:
For each i in zeros: A_i[k] = corr[k] + flip_plus[i, k] (precompute, shape n_zero × (2n-1))
For each j in ones: B_j[k] = flip_plus[j, k] (precompute, shape n_one × (2n-1))

new_max[i, j] = max_k (A_i[k] - B_j[k])

This is like a "max of element-wise differences" which can be computed as:
new_max = max over k of (A_i[k] - max over j of B_j[k])? NO, that's wrong.
new_max[i,j] = max_k (A_i[k] - B_j[k])

This can't be simplified in general. For n_zero=125, n_one=125, 2n-1=499:
Total computation: 125 * 125 * 499 ≈ 7.8M operations. In numpy, this can be done with a 3D array or with a loop over k.

Actually, the most efficient way:
new_max[i, j] = max_k (A_i[k] - B_j[k])

For each k: A_i[k] - B_j[k] for all i, j is an outer-difference: A_i[k] (as row) - B_j[k] (as column).
Then new_max = max over k of these outer differences.

So:
new_max = np.full((n_zero, n_one), -np.inf)
for k in range(2*n-1):
    diff = A[:, k, None] - B[None, :, k]  # (n_zero, n_one)
    new_max = np.maximum(new_max, diff)

This is (2n-1) iterations of (n_zero × n_one) operations = 499 * 125 * 125 ≈ 7.8M. In numpy, each iteration creates a (125, 125) array, which is fast.

Total time: ~499 * (time to create and max a 125×125 array) ≈ 499 * ~1μs ≈ 0.5ms per "super-sweep" (all pairs at once).

This is much faster than the O(n³) per-sweep approach! I can find the best swap (i*, j*) in ~0.5ms.

Then I perform the swap, update corr, and repeat. Each swap is O(n²) to recompute corr (or I can update it incrementally).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_max_grad(h, n, alpha, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    lags = np.arange(-(n - 1), n)
    j_idx = np.arange(n)[:, None]
    lags_arr = lags[None, :]
    jp = j_idx + lags_arr
    jm = j_idx - lags_arr
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    jp_c = np.clip(jp, 0, n - 1)
    jm_c = np.clip(jm, 0, n - 1)
    term1 = np.where(mask_p, (1.0 - h)[jp_c], 0.0) * p[None, :]
    term2 = np.where(mask_m, h[jm_c], 0.0) * p[None, :]
    return dx * (term1.sum(axis=1) - term2.sum(axis=1))

def proj_grad_descent(h, n, deadline, alphas, iters_per_alpha):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return h, best_h, best_c
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
            grad = smooth_max_grad(h, n, alpha, dx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = min(0.5 / (alpha * 0.005 + 1.0), 2.0 / max(gnorm, 1e-10))
            for _ in range(8):
                h_new = project_feasible(h - step * grad, n)
                c_new = compute_c5(h_new, n)
                if c_new <= c + 1e-13:
                    h = h_new
                    break
                step *= 0.5
            if step < 1e-14:
                break
    return h, best_h, best_c

def compute_flip_plus_matrix(h, n, dx, j_idx, lags_arr, jp, jm, mask_p, mask_m, jp_c, jm_c):
    return dx * (np.where(mask_p, (1.0 - h)[jp_c], 0.0) - np.where(mask_m, h[jm_c], 0.0))

def fast_binary_search(h_init, n, deadline, seed=42, max_iters=200000):
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()

    lags = np.arange(-(n - 1), n)
    j_idx = np.arange(n)[:, None]
    lags_arr = lags[None, :]
    jp = j_idx + lags_arr
    jm = j_idx - lags_arr
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    jp_c = np.clip(jp, 0, n - 1)
    jm_c = np.clip(jm, 0, n - 1)

    rng = np.random.RandomState(seed)
    no_improve = 0
    max_no_improve = 300
    it_count = 0

    while time.time() < deadline and it_count < max_iters:
        it_count += 1
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz = len(zeros)
        no_ = len(ones)
        if nz == 0 or no_ == 0:
            break

        fp = compute_flip_plus_matrix(h, n, dx, j_idx, lags_arr, jp, jm, mask_p, mask_m, jp_c, jm_c)
        fp_z = fp[zeros]
        fp_o = fp[ones]

        new_max = np.full((nz, no_), -np.inf)
        for k in range(2 * n - 1):
            cv = corr[k]
            outer = (cv + fp_z[:, k])[:, None] - fp_o[:, k][None, :]
            new_max = np.maximum(new_max, outer)
        new_max *= dx

        flat_idx = np.argmin(new_max)
        bi, bj = np.unravel_index(flat_idx, new_max.shape)
        best_val = new_max[bi, bj]

        if best_val < cur_c - 1e-13:
            i = zeros[bi]
            j = ones[bj]
            h[i], h[j] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full')
            cur_c = corr.max() * dx
            no_improve = 0
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > max_no_improve:

                    ones_t = np.where(h == 1)[0]

                        break
                    ri = rng.choice(zeros_t)
                    rj = rng.choice(ones_t)
                    h[ri], h[rj] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c = corr.max() * dx
                no_improve = 0
                max_no_improve = max(50, max_no_improve // 2)

        if time.time() > deadline:
            break

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
