        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in order[:n_ones - target]:
            h_bin[ones_idx[i]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in order[:target - n_ones]:
            h_bin[zeros_idx[i]] = 1
    return h_bin

def project(h, n):
    """Project to feasible set: 0<=h<=1, sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0, 1)

def smooth_max_alpha(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
