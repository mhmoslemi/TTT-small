For a 1-swap at positions i (was 1) and j (was 0):
The new c5 = max_k c_k'. The change in c_k is:
Δc_k = -dx * [h[i+lag_k] contribution change + h[i] contribution change]

Actually, for a swap h[i]: 1→0 and h[j]: 0→1:
c_k' = c_k + dx * [change due to h[i] going 1→0 + change due to h[j] going 0→1]

The change at position i (h[i]: 1→0):
- For each k where i is a valid i index: -(1-h[i+lag_k])
- For each k where i is a valid (i+lag) index: -(-h[i-lag_k]) = +h[i-lag_k]... 

This is getting complex. Let me just recompute c5 from scratch after each swap. For n=500, it's fast enough.

For binary h, the "gradient" at position i tells us: if we increase h[i] (from 0 to 1), how does c5 change? It changes by dx * sum_k w[k] * [(1-h[i+lag_k]) - h[i-lag_k]].

If this is negative, increasing h[i] helps (reduces c5).
If positive, decreasing h[i] helps.

For a swap (i: 1→0, j: 0→1), the approximate change is:
Δc5 ≈ dx * [grad[i] * (-1) + grad[j] * (+1)] = dx * (grad[j] - grad[i])

We want grad[j] < grad[i] (so that Δc5 < 0).

This gives us a fast way to find good swaps without actually evaluating!

        if best_pair is not None and best_gain > 1e-14:
            i1, i2, j1, j2 = best_pair
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                no_improve = 0
                for i_val in (i1, i2):
                    ones.remove(i_val)
                    zeros.append(i_val)
                for j_val in (j1, j2):
                    zeros.remove(j_val)
                    ones.append(j_val)
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                no_improve += 1
        else:
            no_improve += 1
            if no_improve % 20 == 0 and len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    continue
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2:
                    continue
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    no_improve = 0
                    for i_val in (i1, i2):
                        ones.remove(i_val)
                        zeros.append(i_val)
                    for j_val in (j1, j2):
                        zeros.remove(j_val)
                        ones.append(j_val)
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
    return h, best_c

def fine_continuous_refine(h, n, deadline):
    h_work = h.copy()
    best_c = compute_c5(h_work, n)
    
    for alpha_sm in [50000, 200000, 1000000]:
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_work, n)
        w = smooth_max_w(c_vec, alpha_sm)
        g = compute_grad_direct(h_work, n, w)
        
        pos_mask = (g > 1e-12) & (h_work > 1e-8) & (h_work < 1.0 - 1e-8)
        neg_mask = (g < -1e-12) & (h_work > 1e-8) & (h_work < 1.0 - 1e-8)
        
        if not pos_mask.any() or not neg_mask.any():
            continue
        
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for step in step_sizes:
            if time.time() > deadline:
                break
            improved = True
            rounds = 0
            while improved and rounds < 100 and time.time() < deadline:
                improved = False
                rounds += 1
                c_vec2 = compute_c5_vec(h_work, n)
                w2 = smooth_max_w(c_vec2, alpha_sm)
                g2 = compute_grad_direct(h_work, n, w2)
                
                best_gain = 0
                best_i = -1
                best_j = -1
                for i_val in pos_idx[:30]:
                    if g2[i_val] < 1e-12:
                        continue
                    for j_val in neg_idx[:30]:
                        if g2[j_val] > -1e-12:
                            continue
                        gain_val = g2[i_val] - g2[j_val]
                        if gain_val > best_gain:
                            best_gain = gain_val
                            best_i = i_val
                            best_j = j_val
                
                if best_i < 0:
                    break
                
                amt = min(step, h_work[best_i], 1.0 - h_work[best_j])
                if amt < 1e-12:
                    break
                h_work[best_i] -= amt
                h_work[best_j] += amt
                h_work = project_feasible(h_work, n)
                c_new = compute_c5(h_work, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    h_work[best_i] += amt
                    h_work[best_j] -= amt
                    h_work = project_feasible(h_work, n)
    return h_work, best_c

def nemesis_refine(h, n, deadline, num_sens=12):
    h_work = h.copy()
    best_c = compute_c5(h_work, n)
    c_vec = compute_c5_vec(h_work, n)
    w = smooth_max_w(c_vec, 200000.0)
    g = compute_grad_direct(h_work, n, w)
    sens = np.argsort(-np.abs(g))[:num_sens]
    
    def obj_small(x):
        h_tmp = h_work.copy()
        np.clip(x, 0.0, 1.0, out=x)
        h_tmp[sens] = x
        h_tmp = project_feasible(h_tmp, n)
        return compute_c5(h_tmp, n)
    
    x0 = h_work[sens].copy()
    try:
        res = minimize(obj_small, x0, method='Nelder-Mead',
                      options={'maxiter': 5000, 'xatol': 1e-12, 'fatol': 1e-14})
        h_new = h_work.copy()
        h_new[sens] = np.clip(res.x, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h_work, best_c

def make_golden_ratio_bin(n, rng):
    alpha_val = (1.0 + np.sqrt(5.0)) / 2.0
    h = beatty_init(n, alpha_val, rng)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
        h_parent = project_feasible(h_parent, n_parent)
        best_c = compute_c5(h_parent, n_parent)
        best_h = h_parent.copy()
        best_n = n_parent
    except Exception:
        best_h = None
        best_c = 1.0
        best_n = 500
    
    gold_ratio = (1.0 + np.sqrt(5.0)) / 2.0
    
    resolutions = []
    if best_n is not None and best_n % 2 == 0:
        resolutions.append(best_n)
    for rn in [500, 600, 800, 1000]:
        if rn not in resolutions and rn % 2 == 0:
            resolutions.append(rn)
    
    for rn in resolutions:
        if time.time() > deadline - 60:
            break
        t_start_res = time.time()
        time_left_res = deadline - time_start_res
        n = rn
        
        # Initialize: try parent (resampled), Beatty, random
        inits = []
        if best_h is not None and best_n != n:
            h_resamp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, best_n), best_h)
            inits.append(project_feasible(h_resamp, n))
        if best_h is not None and best_n == n:
            inits.append(best_h.copy())
        
        h_beatty = beatty_init(n, gold_ratio, rng)
        inits.append(h_beatty)
        
        h_beatty2 = beatty_init(n, gold_ratio + 0.1, rng)
        inits.append(h_beatty2)
        
        h_rand = np.zeros(n)
        chosen = rng.choice(n, n // 2, replace=False)
        h_rand[chosen] = 1.0
        inits.append(h_rand)
        
        # Pick best init via quick L-BFGS
        best_h_n = None
        best_c_n = 1.0
        lbfgs_time = min(40, time_left_res * 0.15)
        
        for h_init in inits[:4]:
            if time.time() > deadline - 30:
                break
            t_sub = time.time()
            h_lb, c_lb = run_lbfgs_inner(h_init, n, alpha=50000, mu=0.01, maxit=500)
            if c_lb < best_c_n:
                best_c_n = c_lb
                best_h_n = h_lb
        
        if best_h_n is None:
            h_lb, c_lb = run_lbfgs_inner(inits[0], n, alpha=50000, mu=0.01, maxit=500)
            best_c_n = c_lb
            best_h_n = h_lb
        
        if best_c_n < best_c:
            best_c = best_c_n
            best_h = best_h_n.copy()
            best_n = n
        
        # L-BFGS refinement
        if time.time() < deadline - 100:
            for alpha_sm in [100000, 500000, 2000000]:
                if time.time() > deadline - 80:
                    break
                h_tmp, c_tmp = run_lbfgs_inner(best_h_n, n, alpha=alpha_sm, mu=0.001, maxit=2000)
                if c_tmp < best_c_n:
                    best_c_n = c_tmp
                    best_h_n = h_tmp
        
        # Binary local search (main driver)
        if time.time() < deadline - 50:
            h_bin = binarize(best_h_n, n)
            # Multiple restarts of binary search
            num_restarts = 3
            for rr in range(num_restarts):
                if time.time() > deadline - 40:
                    break
                h_br = h_bin.copy()
                ones_list = [i for i in range(n) if h_br[i] > 0.5]
                zeros_list = [i for i in range(n) if h_br[i] < 0.5]
                num_rand_swaps = rng.randint(3, 10)
                for _ in range(num_rand_swaps):
                    if ones_list and zeros_list:
                        i_val = ones_list[rng.randint(len(ones_list))]
                        j_val = zeros_list[rng.randint(len(zeros_list))]
                        h_br[i_val], h_br[j_val] = h_br[j_val], h_br[i_val]
                        ones_list.remove(i_val)
                        ones_list.append(j_val)
                        zeros_list.remove(j_val)
                        zeros_list.append(i_val)
                
                sub_deadline = min(deadline - 30, time.time() + (deadline - 30 - time.time()) / (num_restarts - rr))
                h_bs, c_bs = binary_search_1swap(h_br, n, sub_deadline, rng, max_no_improve=8000)
                if c_bs < best_c_n:
                    best_c_n = c_bs
                    best_h_n = h_bs
                    h_bin = h_bs.copy()
            
            # 2-swap search on best
            if time.time() < deadline - 30:
                h_2s, c_2s = binary_search_2swap(best_h_n, n, deadline - 25, rng)
                if c_2s < best_c_n:
                    best_c_n = c_2s
                    best_h_n = h_2s
            
            # Another round of 1-swap
            if time.time() < deadline - 20:
                h_bs2, c_bs2 = binary_search_1swap(best_h_n, n, deadline - 15, rng, max_no_improve=5000)
                if c_bs2 < best_c_n:
                    best_c_n = c_bs2
                    best_h_n = h_bs2
        
        # Continuous refinement on sensitive positions
        if time.time() < deadline - 15:
            h_cf, c_cf = fine_continuous_refine(best_h_n, n, deadline - 10)
            if c_cf < best_c_n:
                best_c_n = c_cf
                best_h_n = h_cf
        
        # Nelder-Mead on subset
        if time.time() < deadline - 8:
            h_nm, c_nm = nemesis_refine(best_h_n, n, deadline - 5, num_sens=10)
            if c_nm < best_c_n:
                best_c_n = c_nm
                best_h_n = h_nm
        
        # Final: binarize and one more search
        if time.time() < deadline - 4:
            h_final_bin = binarize(best_h_n, n)
            h_fb, c_fb = binary_search_1swap(h_final_bin, n, deadline - 2, rng, max_no_improve=3000)
            if c_fb < best_c_n:
                best_c_n = c_fb
                best_h_n = h_fb
        
        if best_c_n < best_c:
            best_c = best_c_n
            best_h = best_h_n.copy()
            best_n = n
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
