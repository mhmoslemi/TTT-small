import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            pos = room > 1e-10
            if not np.any(pos):
                break
            available = room[pos].sum()
            if available < 1e-15:
                break
            scale = min(diff / available, 1.0)
            h[pos] += scale * room[pos]
        else:
            pos = h > 1e-10
            if not np.any(pos):
                break
            total = h[pos].sum()
            if total < 1e-15:
                break
            frac = min(-diff / total, 1.0)
            h[pos] *= (1.0 - frac)
    return np.clip(h, 0.0, 1.0)

def sm_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def sm_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-14:
            continue
        k = mm - (n - 1)
        wm2 = wm * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] -= wm2 * h[k:n]
                grad[k:n] += wm2 * (1.0 - h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[0:nv] += wm2 * (1.0 - h[kk:n])
                grad[kk:n] -= wm2 * h[0:nv]
    return grad

def act_obj(h, n, active, alpha):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def act_grad(h, n, active, alpha):
    c = compute_c5_vec(h, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm2 = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] -= wm2 * h[k:n]
                grad[k:n] += wm2 * (1.0 - h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[0:nv] += wm2 * (1.0 - h[kk:n])
                grad[kk:n] -= wm2 * h[0:nv]
    return grad

def lbfgs_refine(h, n, deadline, alphas, maxiters, mu=0.0):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = maxiters[i] if i < len(maxiters) else 1000
        maxit = min(maxit, max(10, int(remaining * 2)))
        try:
            if mu > 0:
                def obj_mu(x):
                    c = compute_c5_vec(x, n)
                    m = c.max()
                    sh = alpha * (c - m)
                    mx = sh.max()
                    if mx > 500:
                        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
                    else:
                        val = m + np.log(np.sum(np.exp(sh))) / alpha
                    val += mu * (x.sum() - n / 2.0) ** 2
                    return float(val)
                def grad_mu(x):
                    g = sm_grad(x, n, alpha)
                    g += 2.0 * mu * (x.sum() - n / 2.0)
                    return g
                res = minimize(obj_mu, best_h, jac=grad_mu, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 40})
            else:
                res = minimize(sm_obj, best_h, jac=sm_grad,
                              args=(n, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': maxit * 10,
                                       'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_refine(h, n, deadline, top_k=5, alphas=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000]
    for rnd in range(15):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-6, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(200, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': maxit * 8,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(np.clip(res.x, 0, 1), n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def noise_restart(h, n, deadline, rng, sigma_list):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for sigma in sigma_list:
        if time.time() > deadline - 30:
            break
        h_pert = best_h + sigma * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        alphas = [100000, 500000]
        maxits = [800, 1500]
        end = min(time.time() + 40, deadline - 20)
        h_new, c_new = lbfgs_refine(h_pert, n, end, alphas, maxits, mu=0.0)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def micro_pair_search(h, n, deadline, radius=50, steps=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in steps:
        improved = True
        while improved and time.time() < deadline - 2:
            improved = False
            for i in range(0, n, max(1, n // 200)):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def upsample(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from)
    x_new = np.linspace(0, 2, n_to)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_to)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    def sq_obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
        return float(val)
    def sq_grad(x):
        return sm_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(sq_obj, h, jac=sq_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(np.clip(res.x, 0, 1), n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)
