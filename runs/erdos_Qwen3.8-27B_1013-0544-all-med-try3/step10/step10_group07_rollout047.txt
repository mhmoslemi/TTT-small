import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary: take the n/2 largest values as 1"""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def swap_delta(h, i, j, n, m_range):
    """Compute delta in correlation when h[i]:0->1, h[j]:1->0"""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def best_swap_local_search(h, n, deadline, max_sweeps=2000):
    """Find best swap at each step (not just first improving)"""
    corr = compute_corr(h).astype(float)
    cur_c = corr.max()
    m_range = np.arange(-(n-1), n)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline - 10:
            break
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Shuffle for diversity
        rng.shuffle(zeros)
        rng.shuffle(ones)
        
        best_dc = 0
        best_i = best_j = -1
        best_delta = None
        
        for i in zeros:
            for j in ones:
                d = swap_delta(h, i, j, n, m_range)
                nc = (corr + d).max()
                dc = cur_c - nc  # positive = improvement
                if dc > best_dc:
                    best_dc = dc
                    best_i = i
                    best_j = j
                    best_delta = d
        
        if best_i >= 0 and best_dc > 1e-15:
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr += best_delta
            cur_c -= best_dc
            zeros.remove(best_i)
            ones.remove(best_j)
            zeros.append(best_j)
            ones.append(best_i)
        else:
            break
    
    return h, cur_c
