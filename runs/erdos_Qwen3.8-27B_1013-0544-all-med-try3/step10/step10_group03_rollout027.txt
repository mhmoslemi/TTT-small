import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_anti_periodic(h, n):
    """Project h onto anti-periodic subspace: h[i] + h[i+n//2] = 1."""
    nh = n // 2
    g = np.clip(0.5 * (h[:nh] + 1.0 - h[nh:]), 0.0, 1.0)
    h_out = np.empty(n)
    h_out[:nh] = g
    h_out[nh:] = 1.0 - g
    return h_out

def anti_periodic_expand(g, n):
    """Expand anti-periodic parameter to full function."""
    nh = n // 2
    h = np.empty(n)
    h[:nh] = g
    h[nh:] = 1.0 - g
    return h

def smooth_max_obj(x, n, alpha, mode='full'):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    N = n
    grad = np.zeros(N)
    for s_idx in range(2 * N - 1):
        if w[s_idx] < 1e-15:
            continue
        s = s_idx - (N - 1)
        wm = w[s_idx] * dx
        if s >= 0:
            nv = N - s
            grad[:nv] += wm * (1.0 - x[s:s+nv])
            grad[s:s+nv] += wm * (-x[:nv])
        else:
            kk = -s
            nv = N - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def power_mean_obj(x, n, p):
    c = compute_c5_vec(x, n)
    cp = c ** p
    S = cp.sum()
    if S < 1e-30:
        return 0.0
    return float((S / len(c)) ** (1.0 / p))

def power_mean_grad(x, n, p):
    c = compute_c5_vec(x, n)
    cp = c ** p
    S = cp.sum()
    if S < 1e-30:
        return np.zeros(n)
    m = len(c)
    F = (S / m) ** (1.0 / p)
    # dF/dc_i = c_i^(p-1) / (m^(1/p) * S^(1-1/p))
    pre = 1.0 / (m * S ** (1.0 - 1.0 / p))
    dc = np.zeros(2 * n - 1)
    valid = c > 1e-15
    dc[valid] = (c[valid] ** (p - 1)) * pre
    dx = 2.0 / n
    N = n
    grad = np.zeros(N)
    for s_idx in range(2 * N - 1):
        if dc[s_idx] < 1e-15:
            continue
        s = s_idx - (N - 1)
        wm = dc[s_idx] * dx
        if s >= 0:
            nv = N - s
            grad[:nv] += wm * (1.0 - x[s:s+nv])
            grad[s:s+nv] += wm * (-x[:nv])
        else:
            kk = -s
            nv = N - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def local_search_single(h, n, deadline, max_rounds=200):
    """Local search with single-point moves."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    nr = n // 2
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if h[i] + d <= 1.0:
                    h[i] += d
                    c_new = compute_c5(h, n)
                    h[i] -= d
                    if c_new < best_c - 1e-12:
                        h[i] += d
                        best_c = c_new
                        improved = True
                if h[i] - d >= 0.0:
                    h[i] -= d
                    c_new = compute_c5(h, n)
                    h[i] += d
                    if c_new < best_c - 1e-12:
                        h[i] -= d
                        best_c = c_new
                        improved = True
        if not improved:
            break
    return h, best_c

def binary_local_search(h, n, deadline, max_iters=5000):
    """Simulated annealing on binary (0/1) solutions, then relax."""
    nh = n // 2
    target = n / 2.0
    # Start from rounded version
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    diff = target - h_bin.sum()
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        np.random.shuffle(zeros)
        for idx in zeros[:int(diff)]:
            h_bin[idx] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        np.random.shuffle(ones)
        for idx in ones[:int(-diff)]:
            h_bin[idx] = 0.0
    h_bin = project_feasible(h_bin, n)
    best_c = compute_c5(h_bin, n)
    best_bin = h_bin.copy()
    T = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        if h_bin[i] == 0:
            j_candidates = np.where(h_bin == 1)[0]
            if len(j_candidates) == 0:
                continue
            j = j_candidates[np.random.randint(len(j_candidates))]
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        else:
            j_candidates = np.where(h_bin == 0)[0]
            if len(j_candidates) == 0:
                continue
            j = j_candidates[np.random.randint(len(j_candidates))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c:
            best_c = c_new
            best_bin = h_bin.copy()
        else:
            delta = c_new - best_c
            if np.random.random() < np.exp(-delta / max(T, 1e-8)):
                pass  # accept
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        if it % 100 == 0:
            T *= 0.95
    return best_bin, best_c

def run_lbfgs(h, n, deadline, alphas=None, mu=0.0):
    """Run L-BFGS with increasing alpha."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        def obj(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            if mx > 500:
                val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
            else:
                val = m + np.log(np.sum(np.exp(sh))) / alpha
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return float(val)
        def grad(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            dx = 2.0 / n
            N = n
            g = np.zeros(N)
            for s_idx in range(2 * N - 1):
                if w[s_idx] < 1e-15:
                    continue
                s = s_idx - (N - 1)
                wm = w[s_idx] * dx
                if s >= 0:
                    nv = N - s
                    g[:nv] += wm * (1.0 - x[s:s+nv])
                    g[s:s+nv] += wm * (-x[:nv])
                else:
                    kk = -s
                    nv = N - kk
                    g[kk:] += wm * (1.0 - x[:nv])
                    g[:nv] += wm * (-x[kk:])
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_anti_periodic_lbfgs(h, n, deadline, alphas=None):
    """L-BFGS on anti-periodic subspace."""
    nh = n // 2
    g = h[:nh].copy()
    h_ap = anti_periodic_expand(g, n)
    h_ap = project_feasible(h_ap, n)
    g = h_ap[:nh].copy()
    best_c = compute_c5(anti_periodic_expand(g, n), n)
    best_g = g.copy()
    bounds = [(0.0, 1.0)] * nh
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(anti_periodic_obj, best_g, jac=anti_periodic_grad,
                          args=(n, alpha), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': 30000, 'maxls': 50})
            g_new = np.clip(res.x, 0.0, 1.0)
            h_new = anti_periodic_expand(g_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_g = g_new
        except Exception:
            pass
    h_final = anti_periodic_expand(best_g, n)
    return h_final, best_c

def run_power_mean_lbfgs(h, n, deadline, ps=None):

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def top_k_smooth_obj(x, n, top_indices, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_indices]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def top_k_smooth_grad(x, n, top_indices, alpha):
    c = compute_c5_vec(x, n)
    vals = c[top_indices]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    N = n
    grad = np.zeros(N)
    for wi, idx in zip(w, top_indices):
        s = idx - (N - 1)
        wm = wi * dx
        if s >= 0:
            nv = N - s
            grad[:nv] += wm * (1.0 - x[s:s+nv])
            grad[s:s+nv] += wm * (-x[:nv])
        else:
            kk = -s
            nv = N - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    return grad

def run_top_k_opt(h, n, deadline, top_k=5, alphas=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [2000, 10000, 50000, 200000]
    for _ in range(5):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(best_h, n)
        idx = np.argsort(-c_vec)[:top_k]
        for alpha in alphas:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(top_k_smooth_obj, best_h, jac=top_k_smooth_grad,
                              args=(n, idx, alpha), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': 10000, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c
