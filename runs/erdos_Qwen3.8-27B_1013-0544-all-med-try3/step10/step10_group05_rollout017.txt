import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return c

def compute_c5_max(h, n):
    return compute_c5(h, n).max()

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-14:
                h -= (-diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = h.sum()
    if abs(s - target) > 1e-12:
        diff = target - s
        room = 1.0 - h
        mask = room > 1e-10
        if mask.sum() > 0 and diff > 0:
            h[mask] += diff / mask.sum()
        mask = h > 1e-10
        if mask.sum() > 0 and diff < 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return val

def smooth_max_grad(c, alpha, h, n):
    """Compute gradient of smooth max of c (where c = correlate(h,1-h,'full')*dx)."""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2*n-1):

            continue
        k = idx - (n-1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def smooth_obj(h, n, alpha, mu):
    c = compute_c5(h, n)
    return smooth_max(c, alpha) + mu * (h.sum() - n/2.0)**2

def smooth_obj_grad(h, n, alpha, mu):

    g = smooth_max_grad(c, alpha, h, n)
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n/2.0)
    return g

def subgrad_step(h, n, c, alpha_sm, lr):
    """Take a subgradient step using top-k active correlations."""
    m = c.max()
    top_k = max(10, int(0.05 * (2*n-1)))
    sort_idx = np.argsort(-c)[:top_k]
    # Softmax weights over active
    active_c = c[sort_idx]
    sh = alpha_sm * (active_c - active_c.max())
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0/n
    grad = np.zeros(n)
    for i, idx in enumerate(sort_idx):
        k = int(idx) - (n-1)
        wm = w[i] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return h - lr * grad

def momentum_descent(h, n, deadline, alpha_start, lr_start, momentum):
    """Momentum-based gradient descent with adaptive parameters."""
    best_h = h.copy()
    best_c = compute_c5_max(h, n)
    v = np.zeros(n)
    lr = lr_start
    alpha = alpha_start
    t = 0
    stuck = 0

        cur_c = c.max()

            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

        g += 2.0 * 0.01 * (h.sum() - n/2.0)
        
        v = momentum * v + lr * g
        h_new = project_feasible(h - v, n)
        
        c_new = compute_c5_max(h_new, n)
        if c_new < cur_c:

            v *= 0.5

def make_symmetric_pattern(n, rng):
    """Create a pattern with approximate reflection symmetry."""
    h = np.zeros(n)
    n_ones = n // 2
    half = n // 2
    # Randomly choose positions in first half
    positions = set(rng.choice(half, size=n_ones//2, replace=False).tolist())

        h[p] = 1.0
        h[(p + n//2) % n] = 1.0

        nonzero = np.where(h > 0.5)[0]
        h[rng.choice(nonzero)] = 0
