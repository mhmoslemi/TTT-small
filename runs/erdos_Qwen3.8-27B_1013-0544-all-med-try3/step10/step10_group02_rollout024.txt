        if len(active) > top_k:
            # Keep the top ones
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(300, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def local_pairwise(h, n, deadline, radius=50, deltas=None):
    """Gradient-informed pairwise exchange search in local neighborhoods."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            step = 0
            for i in range(n):
                if time.time() > deadline:
                    break
                if step % 500 == 0 and time.time() > deadline:
                    break
                step += 1
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving mass from i to j
                    if best_h[i] > d and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Try moving mass from j to i
                    if best_h[j] > d and best_h[i] + d <= 1.0:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=10000):
    """Mirror-descent style subgradient with adaptive step."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.01
    min_eta = 0.0001
    tol_expand = 0.002
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, tol_expand * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx_val
                grad[k:n] += (-h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx_val
                grad[0:nv] += (-h[kk:n]) * dx_val
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.03)
        else:
            eta = max(eta * 0.97, min_eta)
        
        h = h_new
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def upsample(h, n_from, n_to):
    """Upsample h from n_from points to n_to points with linear interpolation."""
    x_from = np.linspace(0, 2, n_from, endpoint=False)
    x_to = np.linspace(0, 2, n_to, endpoint=False)
    h_to = np.interp(x_to, x_from, h)
    return project_feasible(h_to, n_to)

def random_perturb(h, n, sigma, rng):
    """Add small random noise and project back."""
    noise = rng.normal(0, sigma, size=n)
    h_new = h + noise
    h_new = project_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS at original resolution
    p1_end = t0 + 120
    h1, c1 = lbfgs_smoothed(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000, 500000],
        mu=0.01, maxit_list=[1500, 2000, 2000, 2500, 2500])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Active set at original resolution
    if time.time() < deadline - 500:
        p2_end = min(time.time() + 80, deadline - 500)
        h2, c2 = active_set_optimize(best_h, best_n, p2_end, top_k=6,
            alphas=[50000, 200000, 1000000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Upsample to n=1000 and optimize
    n_hi = 1000
    if time.time() < deadline - 400:
        h_hi = upsample(best_h, best_n, n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        
        # L-BFGS at high resolution with alpha annealing
        p3_end = min(time.time() + 250, deadline - 300)
        h3, c3 = lbfgs_smoothed(h_hi, n_hi, p3_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxit_list=[1000, 1500, 2000, 2000, 2500, 2500, 2500])
        
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_hi
        else:
            # Keep the high-res version anyway for further refinement
            # (it might improve more with active set)
            best_h = h3
            best_n = n_hi
    
    # Phase 4: Active set optimization at high resolution
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 150, deadline - 200)
        h4, c4 = active_set_optimize(best_h, best_n, p4_end, top_k=8,
            alphas=[50000, 200000, 1000000, 5000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Subgradient refinement
    if time.time() < deadline - 150:
        p5_end = min(time.time() + 80, deadline - 100)
        h5, c5 = subgrad_refine(best_h, best_n, p5_end, n_iter=10000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Local pairwise search
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 60, deadline - 60)
        h6, c6 = local_pairwise(best_h, best_n, p6_end, radius=40,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 50:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 45:
                break
            p7_end = min(time.time() + 25, deadline - 40)
            h7, c7 = slqp_polish(best_h, best_n, p7_end, alpha=alpha, maxiter=1000)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Multi-restart with perturbations
    if time.time() < deadline - 40:
        for trial in range(3):
            if time.time() > deadline - 35:
                break
            sigma = 0.02 * (trial + 1)
            h_pert = random_perturb(best_h, best_n, sigma, rng)
            c_pert = compute_c5(h_pert, best_n)
            
            # Quick active set optimization from perturbed start
            p8_end = min(time.time() + 12, deadline - 30)
            h8, c8 = active_set_optimize(h_pert, best_n, p8_end, top_k=6,
                alphas=[100000, 500000])
            if c8 < best_c:
                best_c = c8
                best_h = h8
    
    # Phase 9: Final subgradient + pairwise
    if time.time() < deadline - 10:
        p9_end = min(time.time() + 8, deadline - 5)
        h9, c9 = subgrad_refine(best_h, best_n, p9_end, n_iter=5000)
        if c9 < best_c:
            best_c = c9
            best_h = h9
    
    if time.time() < deadline - 3:
        p10_end = deadline - 2
        h10, c10 = local_pairwise(best_h, best_n, p10_end, radius=30,
            deltas=[0.0005, 0.0002, 0.0001])
        if c10 < best_c:
            best_c = c10
            best_h = h10
    
    # Final: try to downsample if at high resolution (might give better discrete result)
    if best_n > 500:
        h_down = np.interp(np.linspace(0, 2, 500, endpoint=False),
                           np.linspace(0, 2, best_n, endpoint=False), best_h)
        h_down = project_feasible(h_down, 500)
        c_down = compute_c5(h_down, 500)
        if c_down < best_c:
            best_c = c_down
            best_h = h_down
            best_n = 500
    
    # Also try n=2000 if at n=1000 and still have time
    if best_n == 1000 and time.time() < deadline - 2:
        h_2k = upsample(best_h, 1000, 2000)
        c_2k = compute_c5(h_2k, 2000)
        if c_2k < best_c:
            best_c = c_2k
            best_h = h_2k
            best_n = 2000
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
