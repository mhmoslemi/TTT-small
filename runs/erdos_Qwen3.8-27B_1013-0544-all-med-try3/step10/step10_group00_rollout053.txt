    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = best_c
    T = T0
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    while time.time() < deadline:
        # Budget: do a batch of iterations
        batch_size = 10000
        for _ in range(batch_size):
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h, n)
            delta = c_new - c
            
            if delta < 0 or rng.rand() < np.exp(-delta / T):
                c = c_new
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        T *= cooling
        
    return best_h, best_c
