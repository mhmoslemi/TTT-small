def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def downsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def smax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def eq_fun(x, n):
    return float(x.sum() - n / 2.0)

def eq_jac(x, n):
    return np.ones(n)

def lbfgs_optimize(h, n, deadline, alphas, mus, maxiter_base=3000):
    """Run L-BFGS-B with a sequence of (alpha, mu) pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, (alpha, mu) in enumerate(zip(alphas, mus)):
        if time.time() > deadline - 3:
            break
        # Scale maxiter based on remaining time
        remaining = deadline - time.time()
        maxiter = min(maxiter_base, max(100, int(remaining / 0.05)))
        try:
            res = minimize(smax_obj, h, jac=smax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-12, 'maxfun': maxiter * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

def topk_optimize(h, n, deadline, top_k=5, alphas=None):
    """Optimize focusing on top-k shifts."""
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for _ in range(15):
        if time.time() > deadline - 3:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-5, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(-c_vec[active])][:top_k]
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 2:
                break
            remaining = deadline - time.time()
            maxiter = min(3000, max(100, int(remaining / 0.03)))
            try:
                res = minimize(topk_obj, h, jac=topk_grad,
                              args=(n, active, alpha, 0.05), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-16,
                                       'gtol': 1e-12, 'maxfun': maxiter * 10, 'maxls': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new.copy()
                    best_h = h_new.copy()
            except Exception:
                pass
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=200000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(smax_obj, h, jac=smax_grad,
                      args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
