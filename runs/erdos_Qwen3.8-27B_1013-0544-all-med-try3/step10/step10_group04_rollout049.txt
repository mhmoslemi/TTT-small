```python
    delta[mask1] -= (1 - h[i - k[mask1]])
    
    # Term 2: +h[i+k] where 0 <= i+k < n, i.e., -i <= k <= n-1-i
    mask2 = (k >= -i) & (k <= n-1-i)
    delta[mask2] += h[i + k[mask2]]
    
    # Term 3: +(1-h[j-k]) where 0 <= j-k < n, i.e., j-n+1 <= k <= j
    mask3 = (k >= j-n+1) & (k <= j)
    delta[mask3] += (1 - h[j - k[mask3]])
    
    # Term 4: -h[j+k] where 0 <= j+k < n, i.e., -j <= k <= n-1-j
    mask4 = (k >= -j) & (k <= n-1-j)
    delta[mask4] -= h[j + k[mask4]]
    
    return delta * (2.0/n)  # multiply by dx
```

```python
def binary_ize(h, n):
    """Convert to binary while maintaining exact sum constraint."""
    target = n / 2.0
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:int(target)]] = 1.0
    return h_b


def swap_delta_vec(h, n, i, j):
    """Compute delta in c vector for swapping h[i]:1->0 and h[j]:0->1. O(n) via vectorization."""
    k = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    # Term 1: -(1-h[i-k]) where 0 <= i-k < n -> k in [i-n+1, i]
    m1 = (k >= i - n + 1) & (k <= i)
    if m1.any():
        delta[m1] -= (1.0 - h[i - k[m1]])
    # Term 2: +h[i+k] where 0 <= i+k < n -> k in [-i, n-1-i]
    m2 = (k >= -i) & (k <= n - 1 - i)
    if m2.any():
        delta[m2] += h[i + k[m2]]
    # Term 3: +(1-h[j-k]) where 0 <= j-k < n -> k in [j-n+1, j]
    m3 = (k >= j - n + 1) & (k <= j)
    if m3.any():
        delta[m3] += (1.0 - h[j - k[m3]])
    # Term 4: -h[j+k] where 0 <= j+k < n -> k in [-j, n-1-j]
    m4 = (k >= -j) & (k <= n - 1 - j)
    if m4.any():
        delta[m4] -= h[j + k[m4]]
    return delta * (2.0 / n)


def binary_pair_swap(h, n, deadline, max_rounds=2000000, seed=7):
    \"\"\"Efficient binary pair-swap local search with O(n) per swap evaluation.\"\"\"
    h = h.copy()
    c = compute_c5_vec(h, n)
    best_c = c.max()
    dx = 2.0 / n
    
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h < 0.5)[0])
    
    rng = np.random.RandomState(seed)
    no_improve = 0
    max_no_improve = 30000
    
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        if no_improve > max_no_improve:
            # Reheat: random perturbation
            n_flip = max(5, n // 100)
            for _ in range(n_flip):
                oi = rng.randint(n_ones)
                zi = rng.randint(n_zeros)
                i = ones_list[oi]
                j = zeros_list[zi]
                h[i], h[j] = 0.0, 1.0
                ones_list[oi] = j
                zeros_list[zi] = i
            c = compute_c5_vec(h, n)
            best_c = c.max()
            no_improve = 0
            continue
        
        # Random swap
        oi = rng.randint(n_ones)
        zi = rng.randint(n_zeros)
        i = ones_list[oi]
        j = zeros_list[zi]
        
        # Compute delta
        d = swap_delta_vec(h, n, i, j)
        new_c = c + d
        new_max = new_c.max()
        
        if new_max < best_c - 1e-15:
            # Accept swap
            h[i], h[j] = 0.0, 1.0
            ones_list[oi] = j
            zeros_list[zi] = i
            c = new_c
            best_c = new_max
            no_improve = 0
        else:
            no_improve += 1
    
    return h, best_c


def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):

        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):

    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n

        best_h, best_c = perturbation_search(best_h, len(best_h), t7, n_trials=4,
            mags=[0.02, 0.03, 0.01, 0.04])
    
    # Phase 8: Final SLSQP polish (~30s)
    t8 = min(time.time() + 30, deadline - 10)
    if t8 > time.time() + 5:
        h8, c8 = slqp_polish(best_h, len(best_h), t8, alpha=10000000, maxiter=1500)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final targeted refinement
    if time.time() < deadline - 5:
        best_h, best_c = targeted_refine(best_h, len(best_h), deadline - 2)
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
```
