def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def circular_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    H = rfft(h)
    U = rfft(1.0 - h)
    corr = irfft(np.conj(H) * U, n=n)
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_full(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):

            continue

        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def binary_swap_sa(h, n, deadline, n_iterations=200000, t0_temp=0.5, cooling=0.99999):
    """Simulated annealing on the binary 1/0 structure."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_c = best_c
    ones_mask = cur_h > 0.4
    zeros_mask = ~ones_mask
    ones_idx = np.where(ones_mask)[0]
    zeros_idx = np.where(zeros_mask)[0]
    temperature = t0_temp
    rng = np.random.RandomState(12345)
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        if it % 1000 == 0:
            temperature *= cooling
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        if i == j:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        c_new = compute_c5(cur_h, n)
        delta = c_new - cur_c
        if delta < 0 or (temperature > 1e-8 and rng.random() < np.exp(-delta / max(temperature, 1e-10))):
            cur_c = c_new
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

def spectral_shaping(h, n, deadline, n_iter=1000):
    """Modify h in Fourier domain to reduce peak autocorrelation."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute circular correlation
        corr = circular_corr_vec(best_h)
        corr_max = corr.max()
        # In Fourier domain, the correlation is |H|^2 * (|H|^2 + |U|^2 - 2Re(H*conj(U)))
        # Simplify: power spectrum P = |H|^2
        # We want to flatten the Fourier transform of P
        H = rfft(best_h)
        P = np.abs(H) ** 2
        # Mean of P
        P_mean = P.mean()
        # Dampen peaks in power spectrum
        for freq_idx in range(1, len(P)):
            if P[freq_idx] > P_mean * 1.5:
                P[freq_idx] = P_mean * 1.2
        # Reconstruct: adjust H to have new power spectrum (keep phase)
        H_new = H.copy()
        phases = np.angle(H)
        mags = np.sqrt(P)
        H_new = mags * np.exp(1j * phases)
        h_new = irfft(H_new, n=n)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    for k in range(1, 2 * n + 10):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.append(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def interpolate_h(h_src, n_src, n_dst):

    x_dst = np.linspace(0, 2, n_dst, endpoint=False)

            best_h = h4

    # Phase 5: Active set refinement after SA
    p5_end = min(t0 + 850, deadline - 100)
    if time.time() < p5_end:
        h5, c5 = iterative_active_set(best_h, best_n, p5_end, n_rounds=25)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Structured block optimization
    p6_end = min(t0 + 920, deadline - 60)
    if time.time() < p6_end:
        h6, c6 = structured_block_opt(best_h, best_n, p6_end,
                                       block_sizes=[1, 2, 3, 5, 8])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine pair search
    p7_end = min(t0 + 1010, deadline - 30)
    if time.time() < p7_end:
        h7, c7 = fine_pair_search(best_h, best_n, p7_end, radius=35,
                                   deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final active set polish
    p8_end = deadline - 5
    if time.time() < p8_end:
        h8, c8 = iterative_active_set(best_h, best_n, p8_end, n_rounds=15)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
