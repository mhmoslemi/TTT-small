def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_sum_obj(x, n, k):
    """Sum of top-k values of c + penalty on constraint violation."""
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    # Get top k
    if k >= len(c):
        val = c.sum()
    else:
        # Use partition for speed
        c_sorted = np.partition(c, -k)[-k:]
        val = c_sorted.sum()
    val += 100.0 * (x.sum() - n/2.0)**2
    return float(val)

def topk_sum_grad(x, n, k):
    """Gradient of top-k sum objective."""
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    # Find top k indices

            if nv > 0:
                grad[j:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[j:n])
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def active_set_descent(h, n, deadline, n_iter=5000, step_scale=0.01):
    """Direct subgradient / active-set method on the exact max."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        curr_max = c.max()
        
        if curr_max < best_c:
            best_c = curr_max
            best_h = h.copy()
        
        # Find top few active constraints
        k_active = 3
        top_idx = np.argpartition(c, -k_active)[-k_active:]
        
        # Compute gradient of sum of top-k c values
        grad = np.zeros(n)
        for idx in top_idx:
            kk = idx - center
            if kk >= 0:
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv])
                grad[0:nv] += (-h[kk:n])
            else:
                j = -kk
                nv = n - j
                grad[0:nv] += (1.0 - h[j:n])
                grad[j:n] += (-h[0:nv])
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = step_scale / (gnorm + 1e-10)
        # Try to reduce: move against gradient
        h_new = h - step * dx * grad
        
        # Project to feasible
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < curr_max - 1e-15:
            h = h_new
        else:
            # Try smaller step
            h_new = h - 0.1 * step * dx * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < curr_max - 1e-15:
                h = h_new
            else:
                # Decrease step scale
                step_scale *= 0.9
                if step_scale < 1e-10:
                    break
    
    return best_h, best_c

def push_to_binary(h, n, strength=0.1):
    """Push values towards 0 or 1."""
    # For values > 0.5: push towards 1
    # For values < 0.5: push towards 0
    h_new = h.copy()
    mask_high = h > 0.5
    mask_low = h < 0.5
    h_new[mask_high] += strength * (1.0 - h[mask_high])
    h_new[mask_low] -= strength * h[mask_low]
    return project_feasible(h_new, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def local_refine(h, n, deadline, n_rounds=100):
    """Fine-tune individual or small groups of points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c))
        kk = worst_idx - (n - 1)
        
        # Determine affected range
        if kk >= 0:
            lo, hi = kk, n
            lo2, hi2 = 0, n - kk
        else:
            lo, hi = -kk, n
            lo2, hi2 = 0, n + kk
        
        # Try adjusting a few points in the affected range
        improved = False
        n_try = min(20, (hi - lo) // 3 if (hi - lo) > 6 else hi - lo)
        if n_try < 1:
            break
        indices = np.random.choice(range(max(lo, lo2), min(hi, hi2)), size=min(n_try, min(hi,hi2)-max(lo,lo2)), replace=False) if min(hi,hi2)-max(lo,lo2) > 0 else []
        
        for i in indices:
            if time.time() > deadline:
                return best_h, best_c
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                nv = best_h[i] + delta
                if nv < -1e-9 or nv > 1.0 + 1e-9:
                    continue
                best_h[i] = nv
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = best_h.copy()
                    improved = True
                    break
                else:
                    best_h[i] = h[i]
            if improved:
                break
        if not improved and rnd > 30:
            # Try random perturbation
            idx = np.random.randint(n)
            delta = np.random.uniform(-0.01, 0.01)
            nv = h[idx] + delta
            if 0 <= nv <= 1:
                h[idx] = nv
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
    
    return best_h, best_c
