    dx = 2.0 / n

    dx = 2.0 / n
    # Round h to binary
    h_bin = np.zeros(n)
    # Take the n/2 largest values as 1
    threshold_idx = np.argpartition(h, n//2)[n//2]
    h_bin[h >= threshold_idx] = 1.0
    # Fix: exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones != n//2:
        # Adjust
        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        if n_ones > n//2:
            to_flip = np.random.choice(ones, n_ones - n//2, replace=False)
            h_bin[to_flip] = 0
        else:
            to_flip = np.random.choice(zeros, n//2 - n_ones, replace=False)
            h_bin[to_flip] = 1

    best_c = c_old

        elapsed = time.time() - (deadline - time_budget)
        progress = elapsed / time_budget

        # Choose a random swap
        # Pick a random 1 and a random 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        h_bin[i], h_bin[j] = 0, 1  # h[i]: 1->0, h[j]: 0->1

        delta = c_new - c_old

            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 1, 0  # Undo
        
        # Periodically update lists of ones and zeros
        # (for efficiency)
    
    return best_h, best_c

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        k = mm - (n-1)

        if k >= 0:
            nv = n - k

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad
