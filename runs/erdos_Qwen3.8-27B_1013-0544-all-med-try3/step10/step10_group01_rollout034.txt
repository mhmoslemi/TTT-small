import numpy as np
import time
from scipy.optimize import minimize

_CKEY = None
_CC = None
_SUB_DEADLINE = 1e100


def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h):
    return float(np.max(compute_c5_vec(h)))


def set_sub_deadline(t):
    global _SUB_DEADLINE
    _SUB_DEADLINE = float(t)


def time_callback(xk):
    return time.time() >= _SUB_DEADLINE


def set_c_cache(x, c):
    global _CKEY, _CC
    _CKEY = np.asarray(x, dtype=float).copy()
    _CC = c


def _get_c(x):
    global _CKEY, _CC
    x = np.asarray(x, dtype=float)
    if _CKEY is not None and len(_CKEY) == len(x) and np.array_equal(x, _CKEY):
        return _CC
    c = compute_c5_vec(x)
    _CKEY = x.copy()
    _CC = c
    return c


def project_feasible(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h
    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.5)
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)

    s = float(h.sum())
    if abs(s - target) <= 1e-11:
        diff = target - s
        if abs(diff) > 1e-15:
            interior = (h > 1e-12) & (h < 1.0 - 1e-12)
            cnt = int(np.count_nonzero(interior))
            if cnt > 0:
                h[interior] += diff / cnt
        return np.clip(h, 0.0, 1.0)

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s2 = float(proj.sum())
        if s2 > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    hp = np.clip(h - tau, 0.0, 1.0)
    diff = target - float(hp.sum())

    if abs(diff) > 1e-12:
        interior = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(interior))
        if cnt > 0:
            hp[interior] += diff / cnt
        else:
            if diff > 0:
                room = 1.0 - hp
                tot = float(room.sum())
                if tot > 1e-15:
                    hp += (diff / tot) * room
            else:
                tot = float(hp.sum())
                if tot > 1e-15:
                    hp += (diff / tot) * hp

    return np.clip(hp, 0.0, 1.0)


def upsample_steps(h, nnew):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == nnew:
        return project_feasible(h.copy())
    old_dx = 2.0 / n
    new_dx = 2.0 / nnew
    xold = (np.arange(n, dtype=float) + 0.5) * old_dx
    xnew = (np.arange(nnew, dtype=float) + 0.5) * new_dx
    out = np.interp(xnew, xold, h)
    return project_feasible(out)


def make_sturmian(n, theta, rho):
    x = np.arange(n, dtype=float)
    vals = (x * theta + rho) % 1.0
    h = (vals < 0.5).astype(float)
    return project_feasible(h)


def make_cos_start(n, a):
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h = 0.5 * (1.0 + a * np.cos(np.pi * x))
    return project_feasible(h)


def make_fourier_start(n, p):
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h = 0.5 + 0.65 * np.cos(np.pi * p * x) + 0.15 * np.cos(2.0 * np.pi * p * x + 0.7)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h)


def make_basis_start(n, seed):
    rng = np.random.RandomState(seed)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n)
    for m in range(1, 7):
        amp = 0.35 / float(m * m)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(np.pi * m * x + ph)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        if h.size == 0:
            raise ValueError("empty initial_h_values")
        h = project_feasible(h)
        return h, len(h)
    except Exception:
        n = 1024
        candidates = [
            make_sturmian(n, 0.6180339887, 0.1),
            make_cos_start(n, 0.68),
            make_cos_start(n, 0.80),
            make_fourier_start(n, 1),
            make_basis_start(n, start_seed),
        ]
        best_h = None
        best_c = np.inf
        for h in candidates:
            c = compute_c5(h)
            if np.isfinite(c) and c < best_c:
                best_c = c
                best_h = h
        return best_h, n


def active_smooth_val(c, active, alpha):
    vals = c[active]
    if vals.size == 0:
        return 0.0
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    sm = float(np.max(sh))
    if sm > 400.0:
        sh = sh - sm
        return float(m + (sm + np.log(np.sum(np.exp(sh)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)


def active_softmax(c, active, alpha):
    vals = c[active]
    if vals.size == 0:
        return np.zeros(0, dtype=float)
    m = float(np.max(vals))
    sh = alpha * (vals - m)
    sh = sh - float(np.max(sh))
    w = np.exp(sh)
    s = float(w.sum())
    if s <= 1e-300 or not np.isfinite(s):
        w = np.zeros(vals.size, dtype=float)
        w[0] = 1.0
    else:
        w /= s
    return w


def select_active(c, count):
    n = len(c)
    if n == 0:
        return np.array([], dtype=int)

    cnt = int(max(1, min(n, int(count))))
    top = np.argsort(-c)[:cnt]

    m = float(c.max())
    thr = m - max(1e-12, 5e-4 * m)
    near = np.where(c >= thr)[0]

    if near.size > 0 and near.size <= 600:
        active = np.unique(np.concatenate((top, near)))
    elif near.size > 600:
        active = np.argsort(-c)[:min(600, n)]
    else:
        active = top

    active = active[np.argsort(-c[active])]
    return active


def add_grad_lag(grad, h, idx, weight, dx):
    if weight == 0.0:
        return
    n = len(h)
    k = int(idx) - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (weight * dx) * (1.0 - h[k:n])
            grad[k:n] += (weight * dx) * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (weight * dx) * (1.0 - h[0:nv])
            grad[0:nv] += (weight * dx) * (-h[kk:n])


def grad_from_c(h, c, active, alpha):
    n = len(h)
    if len(active) == 0:
        return np.zeros(n, dtype=float)
    w = active_softmax(c, active, alpha)
    grad = np.zeros(n, dtype=float)
    dx = 2.0 / n
    for idx, wi in zip(active, w):
        if wi > 1e-14:
            add_grad_lag(grad, h, idx, wi, dx)
    return grad


def obj_active(x, n, active, alpha, mu):
    c = _get_c(x)
    val = active_smooth_val(c, active, alpha)
    if mu != 0.0:
        s = x.sum() - 0.5 * n
        val += mu * s * s
    return float(val)


def grad_active(x, n, active, alpha, mu):
    c = _get_c(x)
    grad = grad_from_c(x, c, active, alpha)
    if mu != 0.0:
        grad += 2.0 * mu * (x.sum() - 0.5 * n)
    return grad


def minimize_active_subcall(h, n, active, alpha, mu, sub_deadline):
    if active is None or len(active) == 0:
        h0 = project_feasible(h.copy())
        return h0, compute_c5(h0)

    set_sub_deadline(sub_deadline)
    bounds = [(0.0, 1.0)] * n
    rem = max(0.0, sub_deadline - time.time())
    maxit = int(max(50, min(5000, rem * 120)))

    try:
        options = {
            'maxiter': maxit,
            'ftol': 1e-20,
            'gtol': 1e-14,
            'maxfun': max(1000, maxit * 12),
            'maxls': 30,
            'maxcor': 40,
            'callback': time_callback,
        }
        res = minimize(
            obj_active,
            h,
            jac=grad_active,
            args=(n, active, alpha, mu),
            method='L-BFGS-B',
            bounds=bounds,
            options=options,
        )
        x = np.asarray(res.x, dtype=float)
        if not np.all(np.isfinite(x)):
            h0 = project_feasible(h.copy())
            return h0, compute_c5(h0)
        h_new = project_feasible(x)
        c_new = compute_c5(h_new)
        if not np.isfinite(c_new):
            h0 = project_feasible(h.copy())
            return h0, compute_c5(h0)
        return h_new, c_new
    except Exception:
        h0 = project_feasible(h.copy())
        return h0, compute_c5(h0)

def staged_optimize(h, n, deadline, configs, slice_s=20.0):
    if h is None:
        return None, np.inf
    n = len(h)
    best = project_feasible(h.copy())
    bestc = compute_c5(best)

    max_no_improve = max(20, 5 * len(configs) + 10)
    no_improve = 0
    current_slice = float(max(2.0, slice_s))

    while time.time() < deadline - 6.0:
        if no_improve >= max_no_improve:
            break

        progressed = False
        for cfg in configs:
            if time.time() > deadline - 6.0:
                break

            cvec = compute_c5_vec(best)
            active = select_active(cvec, cfg[2])
            set_c_cache(best, cvec)

            rem = deadline - time.time()
            sl = min(current_slice, max(2.0, rem - 4.0))
            if sl < 1.5:
                break

            sub_deadline = time.time() + sl
            if sub_deadline > deadline - 2.0:
                sub_deadline = deadline - 2.0

            h_new, c_new = minimize_active_subcall(
                best, n, active, float(cfg[0]), float(cfg[1]), sub_deadline
            )
            if np.isfinite(c_new) and c_new < bestc - 1e-14:
                best = h_new
                bestc = float(c_new)
                no_improve = 0
                progressed = True
            else:
                no_improve += 1

        if progressed:
            current_slice = max(3.0, current_slice * 0.97)
        else:
            current_slice = max(3.0, current_slice * 0.90)

    return best, bestc


def coarse_search(parent_h, deadline, seed):
    n = 128
    rng = np.random.RandomState(seed)

    candidates = []
    if parent_h is not None:
        candidates.append(upsample_steps(parent_h, n))

    thetas = [
        0.6180339887, 0.4142135624, 0.293893, 0.539, 0.682,
        0.543, 0.381966, 0.707106, 0.236, 0.764,
    ]
    for th in thetas:
        candidates.append(make_sturmian(n, th, rng.uniform(0.0, 1.0)))

    for _ in range(12):
        candidates.append(make_sturmian(n, rng.uniform(0.2, 0.8), rng.uniform(0.0, 1.0)))

    for p in [1, 2, 3, 4, 5, 6]:
        candidates.append(make_fourier_start(n, p))

    for a in [0.55, 0.65, 0.75, 0.85]:
        candidates.append(make_cos_start(n, a))

    for s in range(4):
        candidates.append(make_basis_start(n, seed + 100 + s))

    hs = []
    cs = []
    for h in candidates:
        if h is None:
            continue
        h = project_feasible(h)
        c = compute_c5(h)
        if np.isfinite(c):
            hs.append(h)
            cs.append(float(c))

    if not hs:
        return None, np.inf

    order = np.argsort(cs)
    order = list(order[:10])

    configs = [
        (10.0, 0.01, 40),
        (50.0, 0.05, 70),
        (200.0, 0.10, 100),
        (1000.0, 0.30, 130),
        (5000.0, 1.0, 100),
        (20000.0, 3.0, 80),
    ]

    best_h = hs[int(order[0])]
    best_c = cs[int(order[0])]

    while time.time() < deadline - 5.0 and len(order) > 0:
        item = order.pop(0)
        h = hs[int(item)]
        rem = deadline - time.time()
        per = max(4.0, rem / float(len(order) + 1))
        end = min(deadline - 2.0, time.time() + per)
        if end <= time.time() + 1.0:
            end = min(deadline - 1.0, time.time() + 6.0)

        h2, c2 = staged_optimize(h, n, end, configs, slice_s=2.0)
        if h2 is not None and np.isfinite(c2) and c2 < best_c:
            best_c = float(c2)
            best_h = project_feasible(h2.copy())

    return best_h, best_c


def pick_best_candidates(cands):
    best_h = None
    best_c = np.inf
    for h in cands:
        if h is None:
            continue
        h = project_feasible(h)
        c = compute_c5(h)
        if np.isfinite(c) and c < best_c:
            best_c = float(c)
            best_h = h
    return best_h, best_c


def pair_refine(h, n, deadline):
    if h is None:
        return None, np.inf
    n = len(h)
    best = project_feasible(h.copy())
    bestc = compute_c5(best)

    steps = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    max_trials = 70
    trials = 0

    for step in steps:
        improved = True
        while improved and trials < max_trials and time.time() < max(0.0, deadline - 1.5):
            improved = False
            cvec = compute_c5_vec(best)
            active = select_active(cvec, 24)
            grad = grad_from_c(best, cvec, active, 20000.0)

            order_hi = np.argsort(-grad)[:60]
            order_lo = np.argsort(grad)[:60]

            hi = []
            lo = []
            minamt = step * 4.0
            for idx in order_hi:
                i = int(idx)
                if best[i] > minamt:
                    hi.append(i)
                    if len(hi) >= 24:
                        break
            for idx in order_lo:
                j = int(idx)
                if best[j] < 1.0 - minamt:
                    lo.append(j)
                    if len(lo) >= 24:
                        break

            if not hi or not lo:
                break

            pairs = []
            for i in hi[:10]:
                for j in lo[:10]:
                    if i == j:
                        continue
                    amt = min(step, float(best[i]), float(1.0 - best[j]))
                    if amt <= 0.0:
                        continue
                    gain = float(grad[i] - grad[j])
                    if gain > 0.0:
                        pairs.append((-gain * amt, float(i), float(j), amt, i, j))

            if not pairs:
                break

            pairs.sort()
            for item in pairs[:4]:
                i = int(item[4])
                j = int(item[5])
                amt = float(item[3])
                trial = best.copy()
                trial[i] -= amt
                trial[j] += amt
                trial = project_feasible(trial)
                ctrial = compute_c5(trial)
                trials += 1
                if np.isfinite(ctrial) and ctrial < bestc - 1e-14:
                    best = trial
                    bestc = float(ctrial)
                    improved = True
                    break

    return best, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    b = float(max(30.0, float(budget_s)))
    deadline = t0 + max(20.0, b - 10.0)

    seed_i = int(seed) % 100000
    best_h, best_n = get_initial(seed_i)
    if best_h is None:
        best_h = np.full(512, 0.5, dtype=float)
        best_n = 512
    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)

    coarse_h = None
    h1 = None
    h2 = None

    try:
        # Cheap multi-start search in low resolution.
        if time.time() < deadline - 520.0:
            coarse_end = min(t0 + 90.0, deadline - 520.0)
            if coarse_end > time.time() + 15.0:
                ch, cc = coarse_search(best_h, coarse_end, seed_i + 1)
                if ch is not None and np.isfinite(cc) and cc < best_c:
                    best_c = float(cc)
                    best_h = project_feasible(ch.copy())
                    best_n = len(best_h)
                coarse_h = ch

        # Polish the parent/direct best at its native resolution if cheap.
        if best_n <= 1024 and time.time() < deadline - 260.0:
            end0 = min(time.time() + 80.0, deadline - 260.0)
            if end0 > time.time() + 10.0:
                h0, c0 = staged_optimize(
                    best_h, best_n, end0, configs_for(best_n), slice_s=5.0
                )
                if h0 is not None and np.isfinite(c0) and c0 < best_c:
                    best_c = float(c0)
                    best_h = project_feasible(h0.copy())
                    best_n = len(best_h)

        # Medium-high resolution: choose the best available 1024 start.
        if time.time() < deadline - 120.0:
            cands = []
            if best_h is not None:
                cands.append(best_h)
                if len(best_h) != 1024:
                    cands.append(upsample_steps(best_h, 1024))
            if coarse_h is not None:
                if len(coarse_h) != 1024:
                    cands.append(upsample_steps(coarse_h, 1024))
                else:
                    cands.append(coarse_h)

            cands.append(make_cos_start(1024, 0.68))
            cands.append(make_cos_start(1024, 0.80))
            cands.append(make_fourier_start(1024, 1))
            cands.append(make_basis_start(1024, seed_i + 7))

            s_h, s_c = pick_best_candidates(cands)
            if s_h is not None:
                if len(s_h) != 1024:
                    s_h = upsample_steps(s_h, 1024)
                    s_c = compute_c5(s_h)

                end1 = min(time.time() + 500.0, deadline - 90.0)
                if end1 > time.time() + 10.0:
                    h1, c1 = staged_optimize(
                        s_h, 1024, end1, configs_for(1024), slice_s=18.0
                    )
                    if h1 is not None and np.isfinite(c1) and c1 < best_c:
                        best_c = float(c1)
                        best_h = project_feasible(h1.copy())
                        best_n = len(best_h)

        # Higher resolution from the best available source.
        if time.time() < deadline - 80.0:
            upc = []
            if best_h is not None and len(best_h) != 2048:
                upc.append(upsample_steps(best_h, 2048))
            if h1 is not None and len(h1) != 2048:
                upc.append(upsample_steps(h1, 2048))

            if upc:
                s2_h, s2_c = pick_best_candidates(upc)
                if s2_h is not None and (s2_c < best_c + 0.003 or best_c > 0.38085):
                    end2 = min(time.time() + 350.0, deadline - 40.0)
                    if end2 > time.time() + 10.0:
                        h2, c2 = staged_optimize(
                            s2_h, 2048, end2, configs_for(2048), slice_s=25.0
                        )
                        if h2 is not None and np.isfinite(c2) and c2 < best_c:
                            best_c = float(c2)
                            best_h = project_feasible(h2.copy())
                            best_n = len(best_h)

        # Optional very-high-resolution polish.
        if time.time() < deadline - 35.0 and best_n < 4096:
            upc = []
            if best_h is not None and len(best_h) != 4096:
                upc.append(upsample_steps(best_h, 4096))
            if h2 is not None and len(h2) != 4096:
                upc.append(upsample_steps(h2, 4096))

            if upc:
                s4_h, s4_c = pick_best_candidates(upc)
                if s4_h is not None and s4_c < best_c + 0.002:
                    end4 = min(time.time() + 90.0, deadline - 8.0)
                    if end4 > time.time() + 8.0:
                        h4, c4 = staged_optimize(
                            s4_h, 4096, end4, configs_for(4096), slice_s=35.0
                        )
                        if h4 is not None and np.isfinite(c4) and c4 < best_c:
                            best_c = float(c4)
                            best_h = project_feasible(h4.copy())
                            best_n = len(best_h)

        # Final exact pair refinement.
        if time.time() < deadline - 3.0:
            hpr, cpr = pair_refine(best_h, len(best_h), deadline - 1.0)
            if hpr is not None and np.isfinite(cpr) and cpr < best_c:
                best_c = float(cpr)
                best_h = project_feasible(hpr.copy())
                best_n = len(best_h)
    except Exception:
        pass

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), len(best_h)
