    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - tau, 0.0, 1.0)

    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in lag_indices:
        k = idx - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    grad /= len(lag_indices)
