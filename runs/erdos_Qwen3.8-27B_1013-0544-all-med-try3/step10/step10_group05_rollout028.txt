import time

def compute_c_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return np.clip(h, 0, 1)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0, 1)
        s_proj = h_proj.sum()
        if abs(s_proj - target) < 1e-15:
            break
        if s_proj > target:
            lo = mid
        else:
            hi = mid
    h_final = np.clip(h - (lo + hi) / 2.0, 0, 1)
    diff = target - h_final.sum()
    if abs(diff) > 1e-12:
        mask = (h_final > 1e-12) & (h_final < 1.0 - 1e-12)
        if mask.any():
            h_final[mask] += diff / mask.sum()
            h_final = np.clip(h_final, 0, 1)
    return h_final

def smooth_max_compute(c, alpha):
    c_max = c.max()
    shifted = alpha * (c - c_max)
    smx = shifted.max()
    exps = np.exp(shifted - smx)
    lse = smx + np.log(exps.sum())
    val = c_max + lse / alpha
    w = exps / exps.sum()
    return val, w

def grad_c_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[:nv] += dx * (1.0 - h[k:])
        grad[k:] += dx * (-h[:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += dx * (1.0 - h[:nv])
        grad[:nv] += dx * (-h[kk:])
    return grad

def smooth_max_obj(h, n, active, alpha, mu):
    c = compute_c_vec(h, n)
    vals = c[active]
    val, w = smooth_max_compute(vals, alpha)
    return val + mu * (h.sum() - n / 2.0) ** 2

def smooth_max_grad(h, n, active, alpha, mu):
    c = compute_c_vec(h, n)
    vals = c[active]
    _, w = smooth_max_compute(vals, alpha)
    grad = np.zeros(n)
    center = n - 1
    for i, idx in enumerate(active):
        if w[i] < 1e-14:
            continue
        k = int(idx) - center
        grad += w[i] * grad_c_k(h, n, k)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def full_obj(h, n, alpha, mu):
    c = compute_c_vec(h, n)
    val, w = smooth_max_compute(c, alpha)
    return val + mu * (h.sum() - n / 2.0) ** 2

def full_grad(h, n, alpha, mu):
    c = compute_c_vec(h, n)
    _, w = smooth_max_compute(c, alpha)
    grad = np.zeros(n)
    center = n - 1
    for i in range(len(c)):
        if w[i] < 1e-14:
            continue
        k = i - center
        grad += w[i] * grad_c_k(h, n, k)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def lbgfs_optimize(h, n, bounds, alpha, mu, deadline, maxit, use_active, top_k):
    c = compute_c_vec(h, n)
    if use_active:
        order = np.argsort(-c)
        active = order[:min(top_k, len(order))]
        obj_fn = smooth_max_obj
        grad_fn = smooth_max_grad
        args = (n, active, alpha, mu)
    else:
        obj_fn = full_obj
        grad_fn = full_grad
        args = (n, alpha, mu)

        return h, compute_c5(h, n)
    maxit = min(maxit, max(50, int(rem * 5)))
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=args, method='L-BFGS-B',
                       bounds=bounds,

        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def get_active_set(h, n, top_k, rel_tol=0.002):
    c = compute_c_vec(h, n)
    c_max = c.max()
    threshold = c_max - max(1e-8, rel_tol * c_max)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c[active])
        active = active[order[:top_k]]
    return active, c, c_max

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    seen = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        if pos not in seen:
            seen.add(pos)
            h[pos] = 1.0
            if len(seen) >= n_ones:
                break
    return project_feasible(h, n)

def make_van_der_corput(n):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    for k in range(1, n * 10):
        rev = 0
        m = k
        while m > 0:
            rev = (rev << 1) | (m & 1)
            m >>= 1
        pos = (rev >> (k.bit_length())) % n
        if pos not in positions:
            positions.append(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

                if time.time() > p3_end - 20:
                    break
                h2, c2 = lbgfs_optimize(h2, n2, bounds2, alpha, mu, p3_end, 800,
                                        use_active=False, top_k=0)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
                    best_n = n2
            for top_k in [8, 15, 25, 40]:
                if time.time() > p3_end - 15:
                    break
                for alpha in [100000, 500000, 2000000]:
                    if time.time() > p3_end - 10:
                        break
                    h2, c2 = lbgfs_optimize(h2, n2, bounds2, alpha, 0.001,
                                            p3_end, 1000, use_active=True, top_k=top_k)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2
                        best_n = n2

    # Phase 4: Try even higher resolution
    n3 = best_n * 2
    p4_end = min(time.time() + 200, deadline - 250)
    if p4_end > time.time() + 60 and n3 <= 4000:
        h3 = upsample(best_h, best_n, n3)
        c3 = compute_c5(h3, n3)
        bounds3 = [(0.0, 1.0)] * n3
        if c3 <= best_c + 0.005:
            for alpha, mu in [(10000, 0.01), (50000, 0.005), (200000, 0.002)]:
                if time.time() > p4_end - 20:
                    break
                h3, c3 = lbgfs_optimize(h3, n3, bounds3, alpha, mu, p4_end, 1000,
                                        use_active=False, top_k=0)
                if c3 < best_c:
                    best_c = c3
                    best_h = h3
                    best_n = n3
            for top_k in [10, 20, 35, 50]:
                if time.time() > p4_end - 15:
                    break
                for alpha in [200000, 1000000, 5000000]:
                    if time.time() > p4_end - 10:
                        break
                    h3, c3 = lbgfs_optimize(h3, n3, bounds3, alpha, 0.001,
                                            p4_end, 1200, use_active=True, top_k=top_k)
                    if c3 < best_c:
                        best_c = c3
                        best_h = h3
                        best_n = n3

    # Phase 5: Directed local search
    p5_end = min(time.time() + 150, deadline - 100)
    if p5_end > time.time() + 30:
        hn = len(best_h)
        h5, c5v = directed_local_search(best_h, hn, p5_end)
        if c5v < best_c:
            best_c = c5v
            best_h = h5

    # Phase 6: More active set optimization
    p6_end = min(time.time() + 120, deadline - 60)
    if p6_end > time.time() + 20:
        hn = len(best_h)
        bounds_n = [(0.0, 1.0)] * hn
        for top_k in [5, 8, 12, 16]:
            if time.time() > p6_end - 10:
                break
            for alpha in [500000, 2000000, 10000000]:
                if time.time() > p6_end - 5:
                    break
                best_h, best_c = lbgfs_optimize(best_h, hn, bounds_n, alpha, 0.0005,
                                                p6_end, 1500, use_active=True, top_k=top_k)

    # Phase 7: Final pair local search
    p7_end = deadline - 2
    if p7_end > time.time() + 10:
        hn = len(best_h)
        h7, c7v = pair_local_search(best_h, hn, p7_end, radius=30,
                                    deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c7v < best_c:
            best_c = c7v
            best_h = h7

    # Final: verify and return
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
