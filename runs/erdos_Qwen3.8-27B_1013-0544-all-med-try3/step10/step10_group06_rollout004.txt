def grad_for_lag(h, n, lag):
    """Gradient of correlation at given lag with respect to h."""
    grad = np.zeros(n)
    if lag == 0:
        # c = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # dc/dh[i] = 1 - 2*h[i]
        grad[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        # c = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
        # dc/dh[i] = (1-h[i+lag]) for i in [0, n-1-lag]
        #           + (-h[i-lag]) for i in [lag, n-1]
        grad[0:n-lag] += (1.0 - h[lag:n])
        grad[lag:n] += (-h[0:n-lag])
    elif -n < lag < 0:
        m = -lag  # m > 0
        # c = sum_{i=m}^{n-1} h[i] * (1-h[i-m])
        #   = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
        # dc/dh[i]:
        #   h[i] appears as h[i+m] in term (sum h[i+m]*(1-h[i])):
        #     contribution: (1-h[i-m]) for i in [m, n-1]
        #   h[i] appears as (1-h[i]) in the same sum:
        #     contribution: -h[i+m] for i in [0, n-1-m]
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += (-h[m:n])
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def corr_grad_for_lag(h, n, lag):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        grad[0:n - lag] += (1.0 - h[lag:n])
        grad[lag:n] += (-h[0:n - lag])
    elif -n < lag < 0:
        m = -lag
        grad[m:n] += (1.0 - h[0:n - m])
        grad[0:n - m] += (-h[m:n])
    return grad


def smooth_max_full_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    m = c_raw.max()
    shifted = alpha * (c_raw - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[0:n - lag] += w[k] * (1.0 - h[lag:n])
            grad[lag:n] += w[k] * (-h[0:n - lag])
        elif -n < lag < 0:
            mi = -lag
            grad[mi:n] += w[k] * (1.0 - h[0:n - mi])
            grad[0:n - mi] += w[k] * (-h[mi:n])
    return grad * dx


def lbfgsb_step(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    pen_coeff = 5e7

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c_raw = np.correlate(x, 1 - x, mode='full')
        obj = smooth_max_val(c_raw, alpha)
        diff = x.sum() - target_sum
        obj += pen_coeff * diff * diff
        grad = smooth_max_full_grad(x, n, alpha)
        grad += 2.0 * pen_coeff * diff
        return obj, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B', bounds=bounds,
                       jac=True, options={'maxiter': maxiter, 'ftol': 1e-20,
                                          'maxls': 50, 'iprint': -1})
        h_out = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_out = h.copy()
    return project_feasible(h_out, n)


def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def directed_local_search(h, n, deadline, rng, top_k=5, max_rounds=300):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    delta_vals = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        improved = False

        best_move_c = best_c
        best_move = None

        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            if lag == 0:
                continue
            m_abs = abs(lag)
            if m_abs >= n:
                continue
            n_pairs = n - m_abs
            for i in range(n_pairs):
                j = i + m_abs
                if h[i] > h[j] + 1e-8:
                    for d in delta_vals:
                        d_actual = min(d, (h[i] - h[j]) * 0.4, h[i] * 0.4, (1.0 - h[j]) * 0.4)
                        if d_actual < 1e-7:
                            continue
                        h[i] -= d_actual
                        h[j] += d_actual
                        c_test = compute_c5(h, n)
                        if c_test < best_move_c - 1e-12:
                            best_move_c = c_test
                            best_move = (i, j, -d_actual)
                        h[i] += d_actual
                        h[j] -= d_actual
                    h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    for d in delta_vals:
                        d_actual = min(d, (h[j] - h[i]) * 0.4, h[j] * 0.4, (1.0 - h[i]) * 0.4)
                        if d_actual < 1e-7:
                            continue
                        h[j] -= d_actual
                        h[i] += d_actual
                        c_test = compute_c5(h, n)
                        if c_test < best_move_c - 1e-12:
                            best_move_c = c_test
                            best_move = (i, j, d_actual)
                        h[j] += d_actual
                        h[i] -= d_actual
                    h = project_feasible(h, n)

            if time.time() > deadline:
                break

        if best_move is not None:
            i, j, d = best_move
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
                improved = True
            else:
                h = best_h.copy()
        if not improved:
            break

    return best_h, best_c

def sa_search(h, n, deadline, rng, t_init=0.005, t_min=0.00001, cool=0.9998):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stuck = 0

    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags(h, n)
        top3 = np.argsort(c_all)[-3:]
        k_star = top3[rng.randint(3)]
        lag = k_star - (n - 1)
        if lag == 0:
            T *= cool
            continue
        m_abs = abs(lag)
        if m_abs >= n:
            T *= cool
            continue
        i = rng.randint(0, n - m_abs)
        j = i + m_abs
        if h[i] > h[j] + 1e-9:
            d = rng.uniform(0.002, 0.06)
            d = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.3, (1.0 - h[j]) * 0.3)
        elif h[j] > h[i] + 1e-9:
            d = -rng.uniform(0.002, 0.06)
            d = max(d, -(h[j] - h[i]) * 0.3, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
        else:
            T *= cool
            continue

        if abs(d) < 1e-9:
            T *= cool
            continue

        c_old = c_all.max()
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        max_new = c_new.max()

        de = max_new - c_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h = best_h.copy()
            stuck += 1

        T *= cool
        if stuck > 50000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            stuck = 0
            T = max(T, 0.003)

    return best_h, best_c

    if time.time() < deadline - 60:
        sa_dead = min(deadline - 30, time.time() + 70)
        h_opt, c_opt = sa_search(best_h, n_cur, sa_dead, rng, t_init=0.003, t_min=0.00002, cool=0.9999)
        if c_opt < best_c - 1e-13:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final L-BFGS-B
    if time.time() < deadline - 25:
        h_opt = lbfgsb_step(best_h, n_cur, 50000000, time.time() + 20, maxiter=3000)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c - 1e-13:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final fine local search
    if time.time() < deadline - 5:
        fg_dead = deadline - 1
        h_opt, c_opt = fine_grain_search(best_h, n_cur, fg_dead, rng)
        if c_opt < best_c - 1e-13:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
