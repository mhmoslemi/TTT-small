            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return coeff * grad

def smooth_max_value(c, dx, alpha):
    vals = c * dx
    m = vals.max()
    shifted = alpha * (vals - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    vals = c * dx
    m = vals.max()
    shifted = alpha * (vals - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(n):
        if w[k] < 1e-15:
            continue
        lag = k - nm1
        if lag == 0:
            grad += w[k] * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * dx * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * dx * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * dx * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * dx * (-h[0:n - idx])
    return grad

def topk_max_value(c, dx, K, alpha):
    """Smooth max focusing only on top K lags."""
    vals = c * dx
    top_idx = np.argpartition(vals, -K)[-K:]
    top_vals = vals[top_idx]
    m = top_vals.max()
    shifted = alpha * (top_vals - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def topk_max_grad(h, n, K, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    vals = c * dx
    top_idx = np.argpartition(vals, -K)[-K:]
    top_vals = vals[top_idx]
    m = top_vals.max()
    shifted = alpha * (top_vals - m)
    mx = shifted.max()
    if mx > 500:
        w_top = np.exp(shifted - mx)
    else:
        w_top = np.exp(shifted)
    w_top = w_top / w_top.sum()
    grad = np.zeros(n)
    nm1 = n - 1
    for ii, k in enumerate(top_idx):
        if w_top[ii] < 1e-15:
            continue
        lag = k - nm1
        if lag == 0:
            grad += w_top[ii] * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w_top[ii] * dx * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w_top[ii] * dx * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w_top[ii] * dx * (1.0 - h[idx:n])
            grad[idx:n] += w_top[ii] * dx * (-h[0:n - idx])
    return grad

def gradient_local_search(h, n, deadline, rng, alpha=50000, max_steps=50000):
    """Gradient-based local search with adaptive steps."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    stuck = 0
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        grad = smooth_max_grad(h, n, alpha)
        gnorm = np.sqrt(np.sum(grad ** 2))
        if gnorm < 1e-15:
            break
        
        # Normalized gradient step
        for mult in [1.0, 0.5, 0.2, 0.1, 0.05, 0.01]:
            h_try = h - lr * mult * grad / gnorm
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try
                stuck = 0
                lr = min(lr * 1.1, 0.5)
                break
        else:
            stuck += 1
            if stuck > 100:
                # Perturb
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                lr = 0.005
                stuck = 0
            elif stuck > 50:
                lr = max(lr * 0.8, 0.001)
            if stuck > 500:
                break
    
    return best_h, best_c

def pair_swap_local_search(h, n, deadline, rng, topK=5):
    """Targeted pair swap local search on critical lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    max_rounds = 200
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-topK:]
        improved = False
        
        for step_scale in [1.0, 0.6, 0.4, 0.2, 0.1, 0.05]:
            if time.time() > deadline:
                break
            base_d = 0.015 * step_scale
            for _ in range(200):
                if time.time() > deadline:
                    break
                k_star = top_idx[rng.randint(topK)]
                lag = k_star - (n - 1)
                if lag == 0:
                    continue
                m = abs(lag)
                if m >= n:
                    continue
                i = rng.randint(0, n - m)
                j = i + m
                if h[i] > h[j] + 1e-8:
                    d = min(base_d, (h[i] - h[j]) * 0.3)
                    if d < 1e-10:
                        continue
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-14:
                        best_c = c_test
                        best_h = h.copy()
                        improved = True
                        c_all = compute_all_lags(h, n)
                        top_idx = np.argsort(c_all)[-topK:]
                    else:
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                elif h[j] > h[i] + 1e-8:
                    d = min(base_d, (h[j] - h[i]) * 0.3)
                    if d < 1e-10:
                        continue
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-14:
                        best_c = c_test
                        best_h = h.copy()
                        improved = True
                        c_all = compute_all_lags(h, n)
                        top_idx = np.argsort(c_all)[-topK:]
                    else:
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
            if improved and step_scale <= 0.2:
                break
        
        if not improved:
            break
    
    return best_h, best_c

def lbfgsb_topk(h, n, K, alpha, deadline, maxiter=2000):
    """L-BFGS-B optimizing top-K smooth max."""
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        c = np.correlate(x_c, 1 - x_c, mode='full')
        vals = c * (2.0 / n)
        obj_val = topk_max_value(c, 2.0/n, K, alpha)
        diff = x_c.sum() - target_sum
        obj_val += 1e7 * diff * diff
        grad = topk_max_grad(x_c, n, K, alpha)
        grad += 2.0 * 1e7 * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def lbfgsb_smoothmax(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        c = np.correlate(x_c, 1 - x_c, mode='full')
        dx = 2.0 / n
        obj_val = smooth_max_value(c, dx, alpha)
        diff = x_c.sum() - target_sum
        obj_val += 1e7 * diff * diff
        grad = smooth_max_grad(x_c, n, alpha)
        grad += 2.0 * 1e7 * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def simulated_annealing(h, n, deadline, rng, alpha_sa=100000, t_init=0.005, t_min=0.00002, cool=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0
    c_all = compute_all_lags(h, n)
    
    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.6:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.08, 0.003 + T * 6)
            if h[i] > h[j] + 1e-8:
                d = rng.uniform(0.002, max_step)
                d = min(d, h[i] * 0.35, (1.0 - h[j]) * 0.35)
            elif h[j] > h[i] + 1e-8:
                d = -rng.uniform(0.002, max_step)
                d = max(d, -h[j] * 0.35, -(1.0 - h[i]) * 0.35)
            else:
                T *= cool
                continue
        else:
            # Structured move at high-overlap lag
            top_k = np.argsort(c_all)[-4:]
            k_star = top_k[rng.randint(4)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-7:
                d = rng.uniform(0.001, 0.03)
                d = min(d, (h[i] - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-7:
                d = -rng.uniform(0.001, 0.03)
                d = max(d, -(h[j] - h[i]) * 0.3)
            else:
                T *= cool
                continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        e_old = smooth_max_value(c_all, 2.0/n, alpha_sa)
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, 2.0/n, alpha_sa)
        max_new = c_new.max() * (2.0/n)

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            c_all = c_new
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            it_since_improve += 1

        T *= cool
        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.008, n)
            h = project_feasible(h, n)
            c_all = compute_all_lags(h, n)
            it_since_improve = 0
            T = max(T, 0.001)

    return best_h, best_c

def exact_max_subgrad_step(h, n, lr):
    """Take a step using the exact max subgradient."""
    c = compute_all_lags(h, n)
    max_idx = np.argmax(c)
    lag = max_idx - (n - 1)
    dx = 2.0 / n
    
    # Gradient of c[max_idx] w.r.t. h
    grad = np.zeros(n)
    if lag == 0:
        grad = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:n] += (1.0 - h[0:n - lag])
        grad[0:n - lag] += (-h[lag:n])
    else:
        idx = -lag
        grad[0:n - idx] += (1.0 - h[idx:n])
        grad[idx:n] += (-h[0:n - idx])
    
    h_new = project_feasible(h - lr * dx * grad, n)
    return h_new

def multi_scale_local_search(h, n, deadline, rng):
    """Local search trying multiple step scales and directions."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for scale in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        improved_any = False
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-8:]
        
        iters = 0
        max_iters = 20000
        while iters < max_iters and time.time() < deadline:
            iters += 1
            k_star = top_idx[rng.randint(8)]
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if abs(h[i] - h[j]) < 1e-9:
                continue
            d = min(scale, abs(h[i] - h[j]) * 0.2)
            if d < 1e-10:
                continue
            if h[i] > h[j]:
                h[i] -= d
                h[j] += d
            else:
                h[i] += d
                h[j] -= d
            h = project_feasible(h, n)
            c_try = compute_c5(h, n)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h.copy()
                improved_any = True
                c_all = compute_all_lags(h, n)
                top_idx = np.argsort(c_all)[-8:]
            else:
                # Undo
                if h[i] > h[j]:
                    h[i] += d
                    h[j] -= d
                else:
                    h[i] -= d
                    h[j] += d
                h = project_feasible(h, n)
        
        if not improved_any:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: Quick optimization at parent resolution with top-K smooth max
    n_cur = len(best_h)
    for K, alpha, mx in [(3, 5000, 800), (5, 10000, 1000), (3, 50000, 1000), (5, 20000, 800)]:
        if time.time() > deadline - 400:
            break
        t_sub = min(60, deadline - 400 - time.time())
        if t_sub < 3:
            break
        h_opt = lbfgsb_topk(best_h, n_cur, K, alpha, t0 + time.time() - t0 + t_sub, maxiter=mx)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Smooth max L-BFGS-B at parent resolution
    for alpha, mx in [(50000, 1000), (200000, 800), (1000000, 600)]:
        if time.time() > deadline - 350:
            break
        t_sub = min(45, deadline - 350 - time.time())
        if t_sub < 3:
            break
        h_opt = lbfgsb_smoothmax(best_h, n_cur, alpha, t0 + time.time() - t0 + t_sub, maxiter=mx)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Gradient local search at parent resolution
    if time.time() < deadline - 300:
        g_deadline = min(deadline - 280, time.time() + 80)
        h_opt, c_opt = gradient_local_search(best_h, n_cur, g_deadline, rng, alpha=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Pair swap local search
    if time.time() < deadline - 250:
        ls_deadline = min(deadline - 230, time.time() + 70)
        h_opt, c_opt = pair_swap_local_search(best_h, n_cur, ls_deadline, rng, topK=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 5: Upscale to higher resolution and optimize
    resolutions = [800, 1000, 1200]
    for n_target in resolutions:
        if time.time() > deadline - 200:
            break
        remaining = deadline - 180 - time.time()
        if remaining < 60:
            break
        
        h_up = expand_to_resolution(best_h, len(best_h), n_target)
        c_up = compute_c5(h_up, n_target)
        
        # Optimize at new resolution
        for K, alpha, mx in [(3, 5000, 500), (5, 20000, 600), (3, 100000, 500)]:
            if time.time() > deadline - 180:
                break
            t_sub = min(50, deadline - 180 - time.time())
            if t_sub < 5:
                break
            h_opt = lbfgsb_topk(h_up, n_target, K, alpha, t0 + time.time() - t0 + t_sub, maxiter=mx)
            c_opt = compute_c5(h_opt, n_target)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h_up = h_opt
            else:
                h_up = h_opt if compute_c5(h_opt, n_target) < c_up else h_up
        
        # Gradient search at new resolution
        if time.time() < deadline - 100:
            g_deadline = min(deadline - 80, time.time() + 40)
            h_opt, c_opt = gradient_local_search(h_up, n_target, g_deadline, rng, alpha=50000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 6: Simulated annealing at best resolution
    n_final = len(best_h)
    if time.time() < deadline - 100:
        sa_deadline = min(deadline - 70, time.time() + 100)
        h_opt, c_opt = simulated_annealing(best_h, n_final, sa_deadline, rng,
                                           alpha_sa=100000, t_init=0.004,
                                           t_min=0.00002, cool=0.9999)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 7: Final L-BFGS-B polish
    if time.time() < deadline - 30:
        for alpha, mx in [(500000, 500), (5000000, 400), (50000000, 300)]:
            if time.time() > deadline - 15:
                break
            t_sub = min(25, deadline - 15 - time.time())
            if t_sub < 3:
                break
            h_opt = lbfgsb_smoothmax(best_h, len(best_h), alpha, t0 + time.time() - t0 + t_sub, maxiter=mx)
            c_opt = compute_c5(h_opt, len(best_h))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 8: Multi-scale local search (very fine)
    if time.time() < deadline - 10:
        ms_deadline = min(deadline - 3, time.time() + 30)
        h_opt, c_opt = multi_scale_local_search(best_h, len(best_h), ms_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 9: Final exact max subgradient steps
    if time.time() < deadline - 2:
        h_cur = best_h.copy()
        n_cur = len(h_cur)
        c_cur = best_c
        for step in range(2000):
            if time.time() > deadline - 1:
                break
            for lr in [0.001, 0.0005, 0.0002, 0.0001]:
                h_try = exact_max_subgrad_step(h_cur, n_cur, lr)
                c_try = compute_c5(h_try, n_cur)
                if c_try < c_cur - 1e-15:
                    h_cur = h_try
                    c_cur = c_try
                    if c_cur < best_c:
                        best_c = c_cur
                        best_h = h_cur.copy()
                    break
    
    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
