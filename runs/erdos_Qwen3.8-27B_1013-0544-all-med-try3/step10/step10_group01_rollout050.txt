def sml_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        d = x.sum() - n / 2.0
        grad += 2.0 * mu * d
    return grad

def topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        d = x.sum() - n / 2.0
        grad += 2.0 * mu * d
    return grad

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += min(remaining, 1.0)
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_spread(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    positions = list(range(0, n, n // n_ones))
    for k in range(n_ones):
        pos = positions[k] + rng.randint(-2, 3)
        pos = pos % n
        while h[pos] > 0:
            pos = (pos + 1) % n
        h[pos] = 1.0
    return project_feasible(h, n)

def upsample(h_coarse, n_coarse, n_fine):
    h_fine = np.zeros(n_fine)
    dx_c = 2.0 / n_coarse
    dx_f = 2.0 / n_fine
    for i in range(n_fine):
        x = (i + 0.5) * dx_f
        idx_c = int(x / dx_c)
        if idx_c >= n_coarse:
            idx_c = n_coarse - 1
        idx_c2 = min(idx_c + 1, n_coarse - 1)
        frac = (x / dx_c) - idx_c
        h_fine[i] = (1 - frac) * h_coarse[idx_c] + frac * h_coarse[idx_c2]
    return project_feasible(h_fine, n_fine)

def run_lbfgs_smooth(h, n, alphas, mu, deadline, maxit_per=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_per is not None and i < len(maxit_per):
            maxit = maxit_per[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(sml_obj, best_h, jac=sml_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-25,
                                   'gtol': 1e-18, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c + 1e-16:
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_topk_opt(h, n, deadline, top_k=8, alphas=None, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(150, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-25,
                                       'gtol': 1e-18, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def coord_descent(h, n, deadline, total_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    steps_done = 0
    while time.time() < deadline and steps_done < total_steps:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.003 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= max(1, len(active))
        pos_mask = (grad > 1e-12) & (best_h > 1e-12)
        neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        best_gain = -1
        best_i = -1
        best_j = -1
        for i in pos_idx[:30]:
            gi = grad[i]
            for j in neg_idx[:30]:
                g = gi - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        if best_i < 0 or best_gain <= 0:
            break
        for step_frac in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]:
            amt = min(step_frac, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-17:
                best_c = c_new
                break
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
        if c_new >= best_c:
            steps_done += 1
    return best_h, best_c

def pair_search(h, n, deadline, radius=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def perturbed_restart(h, n, deadline, rng, alpha=100000, mu=0.01):
    h_pert = h + rng.normal(0, 0.02, size=n)
    h_pert = project_feasible(h_pert, n)
    bounds = [(0.0, 1.0)] * n
    maxit = max(200, int((deadline - time.time()) * 2))
    try:
        res = minimize(sml_obj, h_pert, jac=sml_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 6,
                               'maxls': 30, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h_pert, compute_c5(h_pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    # Get initial solution if available
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 0

    # Primary resolution
    n = 1000
    best_h = None
    best_c = 1.0
    best_n = n

    # Generate initializations at n=1000
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    inits = []
    inits.append(('beatty_g', make_beatty(n, golden / 2.0)))
    inits.append(('beatty_s2', make_beatty(n, np.sqrt(2.0) / 2.0)))
    inits.append(('beatty_pi', make_beatty(n, np.pi / 4.0)))
    inits.append(('rand1', make_random_binary(n, rng)))
    inits.append(('rand2', make_random_binary(n, rng)))
    inits.append(('rand3', make_random_binary(n, rng)))
    inits.append(('spread1', make_spread(n, rng)))
    inits.append(('spread2', make_spread(n, rng)))

    # Upsample from parent if available
    if h_parent is not None and n_parent > 0:
        h_up = upsample(h_parent, n_parent, n)
        inits.append(('parent_up', h_up))

    # Also try at n=500 for comparison
    n500 = 500
    best_h500 = None
    best_c500 = 1.0
    if h_parent is not None and n_parent == n500:
        best_h500 = project_feasible(h_parent, n500)
        best_c500 = compute_c5(best_h500, n500)
    inits_500 = []
    inits_500.append(('beatty_g500', make_beatty(n500, golden / 2.0)))
    inits_500.append(('rand500', make_random_binary(n500, rng)))

    # Evaluate initializations
    for name, h_init in inits:
        c_init = compute_c5(h_init, n)
        if c_init < best_c:
            best_c = c_init
            best_h = h_init.copy()
            best_n = n

    for name, h_init in inits_500:
        c_init = compute_c5(h_init, n500)
        if c_init < best_c500:
            best_c500 = c_init
            best_h500 = h_init.copy()
    if h_parent is not None and n_parent == n500:
        c_p = compute_c5(h_parent, n500)
        if c_p < best_c500:
            best_c500 = c_p
            best_h500 = project_feasible(h_parent, n500)

    # Phase 1: L-BFGS smooth max at n=1000
    if best_h is not None and time.time() < deadline - 400:
        p1_end = min(t0 + 300, deadline - 350)
        h1, c1 = run_lbfgs_smooth(best_h, n,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.01, deadline=p1_end,
            maxit_per=[500, 800, 1000, 1200, 1500, 2000, 2500, 3000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Top-k active set optimization at n=1000
    if best_h is not None and time.time() < deadline - 300:
        p2_end = min(time.time() + 180, deadline - 250)
        h2, c2 = run_topk_opt(best_h, n, p2_end, top_k=6,
            alphas=[50000, 200000, 1000000, 5000000], rounds=15)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Coordinate descent at n=1000
    if best_h is not None and time.time() < deadline - 200:
        p3_end = min(time.time() + 120, deadline - 170)
        h3, c3 = coord_descent(best_h, n, p3_end, total_steps=150000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Another round of top-k after coord descent
    if best_h is not None and time.time() < deadline - 150:
        p4_end = min(time.time() + 80, deadline - 120)
        h4, c4 = run_topk_opt(best_h, n, p4_end, top_k=4,
            alphas=[200000, 1000000, 5000000], rounds=10)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Coordinate descent again (finer)
    if best_h is not None and time.time() < deadline - 100:
        p5_end = min(time.time() + 60, deadline - 80)
        h5, c5 = coord_descent(best_h, n, p5_end, total_steps=100000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Pair search
    if best_h is not None and time.time() < deadline - 60:
        p6_end = min(time.time() + 50, deadline - 40)
        h6, c6 = pair_search(best_h, n, p6_end, radius=40)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try perturbed restarts to escape local optima
    if best_h is not None and time.time() < deadline - 50:
        for r in range(3):
            if time.time() > deadline - 30:
                break
            p7_end = deadline - 25
            h7, c7 = perturbed_restart(best_h, n, p7_end, rng,
                                       alpha=200000, mu=0.01)
            if c7 < best_c - 1e-10:
                best_c = c7
                best_h = h7

    # Phase 8: Also optimize at n=500 in parallel (if time allows)
    if best_h500 is not None and time.time() < deadline - 40:
        p8_end = min(time.time() + 30, deadline - 10)
        h8, c8 = run_lbfgs_smooth(best_h500, n500,
            alphas=[10000, 50000, 200000],
            mu=0.01, deadline=p8_end, maxit_per=[500, 800, 1000])
        if c8 < best_c500:
            best_c500 = c8
            best_h500 = h8

    # Phase 9: Final fine pair search
    if best_h is not None and time.time() < deadline - 5:
        h9, c9 = pair_search(best_h, n, deadline - 2, radius=20)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final: compare n=1000 and n=500 solutions
    if best_h500 is not None:
        c500_final = compute_c5(best_h500, n500)
        if c500_final < best_c:
            best_c = c500_final
            best_h = best_h500.copy()
            best_n = n500

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    # Ensure constraint is met exactly
    assert abs(best_h.sum() - final_n / 2.0) < 1e-6, "Constraint violation"
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15), "Bound violation"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
