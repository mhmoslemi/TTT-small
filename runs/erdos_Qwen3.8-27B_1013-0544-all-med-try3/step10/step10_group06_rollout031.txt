def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    shifted = alpha * (c - m)
    if shifted.max() > 50:
        return float(m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 50:
        w = np.exp(shifted - shifted.max())
    else:
        w = np.exp(shifted)
    return w / w.sum()

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(2*n - 1)
    # For each lag k, the contribution to the gradient
    # c[n-1+k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx for k >= 0
    # dc/dh[i] for lag k>0: 
    #   if i < n-k: dc/dh[i] = (1-h[i+k]) * dx, dc/dh[i+k] = -h[i] * dx
    #   (and symmetrically for k<0)
    
    grad_h = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad_h += w[k] * (1 - 2*h) * dx
        elif lag > 0:
            grad_h[:n-lag] += w[k] * (1 - h[lag:]) * dx
            grad_h[lag:] += w[k] * (-h[:n-lag]) * dx
        else:
            m = -lag
            grad_h[m:] += w[k] * (1 - h[:n-m]) * dx
            grad_h[:n-m] += w[k] * (-h[m:]) * dx
    return grad_h

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        obj_val = smooth_max_val(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e6 * diff * diff
        grad = compute_grad(x, n, alpha)
        grad += 2e6 * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return project_feasible(h.copy(), n)

Now for the key improvement - the "free parameter optimization":

After pushing to near-binary, identify which values are "free" (not at 0 or 1). Then:
1. Fix all binary values
2. Optimize only the free values
3. This is a much smaller problem and can be solved more precisely

But wait, the constraint sum(h) = n/2 must still be satisfied. If I fix most values and only optimize a few, the constraint becomes:
sum(free_values) = n/2 - sum(fixed_values)

This is a simple linear constraint on the free variables.

Let me think about how many free variables there will be. If the solution has, say, 5-10 values at intermediate levels, then I'm optimizing 5-10 variables with a linear constraint. That's trivial for scipy.

But the issue is: which values should be intermediate? That's a combinatorial choice. So I need to search over different sets of "free" positions.

**Refined strategy**:
1. L-BFGS-B with high alpha → near-binary solution
2. Identify "free" positions (h[i] not close to 0 or 1)
3. Try swapping free positions with nearby binary positions (and vice versa)
4. After each swap, re-optimize the free values
5. Accept if C₅ improves

This is a form of "combinatorial + continuous" optimization.

Actually, let me think about this differently. The key insight is:

For a nearly-binary solution, the C₅ is determined by:
- The integer part (from the binary skeleton)
- The fractional part (from the intermediate values)

The integer part gives a "base" C₅, and the intermediate values can reduce it by at most some amount. The question is: what's the optimal "base" binary pattern, and what's the optimal assignment of intermediate values?

For the binary pattern, the max overlap determines the "floor". For n=500, if the max binary overlap is 95, then C₅ ≥ 95/250 = 0.38 (can't go below this with this pattern). If it's 96, C₅ ≥ 0.384.

So the first question: can we find a binary pattern with max overlap ≤ 95?

If yes, then C₅ ≤ 0.38 + (fractional adjustment). The fractional adjustment can only help (reduce the max further), so C₅ ≤ 0.38.

Wait, that's not right. The intermediate values can also INCREASE some overlaps (if a position that was 0 becomes 0.5, it increases the overlap at lags where the other position is 1). So the net effect is complex.

Let me just focus on the practical algorithm and hope it works.

Phase 1: Initial optimization (L-BFGS-B with increasing alpha) - 60s
Phase 2: Push to binary (very high alpha L-BFGS-B) - 60s  
Phase 3: Binary local search (combinatorial, targeting worst lag) - 300s
Phase 4: Continuous relaxation from best binary + polish - 200s
Phase 5: Multi-start (random binary + optimize) - 300s
Phase 6: Final polish - 50s

At n=1000: C₅ = max_overlap * 2/1000 = max_overlap / 500.
For C₅ = 0.3808: max_overlap = 190.4. So we need max_overlap ≤ 190.
For binary: max_overlap would be an integer, so ≤ 190 gives C₅ ≤ 0.38.

At n=500: C₅ = max_overlap / 250.

I think staying at n=500 is fine since that's where we started. The continuous values allow us to get below the integer threshold.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 50:
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    if sm > 50:
        w = np.exp(shifted - sm)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad_h += w[k] * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad_h[:n - lag] += w[k] * (1.0 - h[lag:]) * dx
            grad_h[lag:] += w[k] * (-h[:n - lag]) * dx
        else:
            m = -lag
            grad_h[m:] += w[k] * (1.0 - h[:n - m]) * dx
            grad_h[:n - m] += w[k] * (-h[m:]) * dx
    return grad_h

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        obj_val = smooth_max_val(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty * diff * diff
        grad = compute_grad(x, n, alpha)
        grad += 2.0 * penalty * diff
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    start_t = time.time()
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'maxls': 40,
                                'maxfun': 50000})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def binary_local_search(h, n, deadline, rng, max_iter=200000):
    """Targeted binary local search: always attack the worst lag."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = np.correlate(h, 1 - h, mode='full')
        max_idx = int(c_all.argmax())
        max_val = float(c_all[max_idx])
        k_star = max_idx - (n - 1)

        if k_star == 0:
            break

        m = abs(k_star)
        if m >= n:
            break

        # Find candidate positions for swap at the worst lag
        if k_star > 0:
            # Want h[i]=1, h[i+m]=0 → swap reduces c at lag k_star
            mask = (h[:n - m] == 1) & (h[m:] == 0)
            candidates = np.where(mask)[0]
        else:
            # Want h[i+m]=1, h[i]=0 → swap h[i] and h[i+m]
            # Equivalent: h[i]=0, h[i+m]=1
            mask = (h[:n - m] == 0) & (h[m:] == 1)
            candidates = np.where(mask)[0]

        if len(candidates) == 0:
            no_improve += 1
            if no_improve > 50:
                # Try random perturbation
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j and h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
                    h = project_feasible(h, n)
                continue
            else:
                break

        no_improve = 0
        # Try a random subset of candidates
        n_try = min(len(candidates), max(5, len(candidates) // 5))
        try_idx = rng.choice(candidates, size=n_try, replace=False)

        best_swap_val = max_val
        best_swap_pair = None

        for ci in try_idx:
            if k_star > 0:
                i, j = int(ci), int(ci) + m
            else:
                i, j = int(ci), int(ci) + m

            h[i], h[j] = h[j], h[i]
            c_new = np.correlate(h, 1 - h, mode='full')
            new_max = float(c_new.max())
            h[i], h[j] = h[j], h[i]

            if new_max < best_swap_val:
                best_swap_val = new_max
                best_swap_pair = (i, j)

        if best_swap_pair is not None:
            i, j = best_swap_pair
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)
            c_cur = compute_c5(h, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                # Random restart: flip a random pair
                for _ in range(10):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i != j and h[i] != h[j]:
                        h[i], h[j] = h[j], h[i]
                        h = project_feasible(h, n)
                        break
                no_improve = 0

    return best_h, best_c

def targeted_sa(h, n, deadline, rng, t_init=0.5, t_min=0.01, cool=0.9999, alpha=50000):
    """Simulated annealing on binary, targeting worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    stall = 0

    while time.time() < deadline and T > t_min:
        c_all = np.correlate(h, 1 - h, mode='full')
        top_k = np.argsort(c_all)[-3:]
        k_star = top_k[rng.randint(3)]
        lag = int(k_star) - (n - 1)
        m = abs(lag)

        if lag == 0 or m >= n:
            T *= cool
            continue

        if lag > 0:
            mask = (h[:n - m] == 1) & (h[m:] == 0)
        else:
            mask = (h[:n - m] == 0) & (h[m:] == 1)

        candidates = np.where(mask)[0]
        if len(candidates) == 0:
            T *= cool
            continue

        ci = candidates[rng.randint(len(candidates))]
        i, j = int(ci), int(ci) + m

        e_old = smooth_max_val(c_all * (2.0 / n) / (2.0 / n), alpha)  # just use raw c
        # Actually use the raw correlation for SA
        e_old = float(c_all.max())

        h[i], h[j] = h[j], h[i]
        h = project_feasible(h, n)
        c_new = np.correlate(h, 1 - h, mode='full')
        e_new = float(c_new.max())

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        else:
            ratio = -de / max(T, 1e-10)
            if ratio > -50:
                accept = rng.random() < np.exp(min(0, ratio))

        if accept:
            if e_new < best_c * (n / 2.0):  # compare in same units
                new_c5 = e_new * (2.0 / n)
                if new_c5 < best_c - 1e-14:
                    best_c = new_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, n)
            stall += 1

        T *= cool

        if stall > 5000:
            h = best_h.copy()
            stall = 0
            T = max(T, 0.1)

    return best_h, best_c

def three_lbfgs_multistart(h, n, deadline, rng, n_starts=4):
    """Multi-start: perturb and optimize from multiple points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for s in range(n_starts):
        if time.time() > deadline:
            break
        # Create perturbation
        h_cur = best_h.copy()
        # Flip some random binary values
        n_flips = rng.randint(3, 15)
        for _ in range(n_flips):
            i = rng.randint(n)
            # Find opposite to flip to
            if h_cur[i] > 0.5:
                zeros = np.where(h_cur < 0.5)[0]
            else:
                zeros = np.where(h_cur > 0.5)[0]
            if len(zeros) == 0:
                continue
            j = zeros[rng.randint(len(zeros))]
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]

        h_cur = project_feasible(h_cur, n)
        # Optimize
        h_opt = lbfgsb_optimize(h_cur, n, 500000, time.time() + min(30, (deadline - time.time()) // n_starts), maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

        h_bin2 = project_feasible(h_bin2, n)
        h_bs2, c_bs2 = binary_local_search(h_bin2, n, min(time.time() + 60, deadline - 30), rng, max_iter=300000)

            best_c = c_bs2
            best_h = h_bs2.copy()
