        if len(affected1) + len(affected2) > 200:
            candidates = np.random.choice(n, 200, replace=False)
        else:
            candidates = set(affected1.tolist() + affected2.tolist())
            candidates = np.array(list(candidates))
        
        for i in candidates:
            # Compute d(c_k)/d(h[i])
            dc = 0.0
            if 0 <= i - k_worst < n:
                dc += (1.0 - h[i - k_worst])
            if 0 <= i + k_worst < n:
                dc += (-h[i + k_worst])
            
            if abs(dc) < 1e-10:
                continue
            
            # Direction: decrease h[i] if dc > 0, increase if dc < 0
            if dc > 0:
                # Try decreasing h[i]
                delta = -min(h[i], 0.05)
            else:
                delta = min(1.0 - h[i], 0.05)
            
            # Estimated gain
            gain = -dc * delta
            if gain > best_gain:
                best_gain = gain
                best_delta = delta
                best_pos = i
        
        if best_pos < 0 or best_gain < 1e-10:
            break
        
        h[best_pos] += best_delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
        else:
            h[best_pos] -= best_delta
            h = project_feasible(h, n)
    
    return h, best_c

def simulated_annealing(h, n, deadline, t_start=0.3, t_end=0.01, frac=0.9):
    """Simulated annealing with swaps to maintain sum constraint."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    rng = np.random.RandomState(42)
    
    t0 = time.time()
    total_time = deadline - t0
    
    for iteration in range(1000000):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (iteration / max(1, int(50 * frac)))
        
        # Pick two random positions to swap or adjust
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        
        # Swap-like move: move mass from i to j or vice versa
        delta = 0.05 * rng.choice([-1, 1])
        h_new = h.copy()
        h_new[i] = np.clip(h[i] - delta, 0, 1)
        h_new[j] = np.clip(h[j] + delta, 0, 1)
        
        # Compute new C5
        c_new = compute_c5(h_new, n)
        
        if c_new < current_c or rng.random() < np.exp(-(c_new - current_c) / max(T, 1e-10)):
            h = h_new
            current_c = c_new
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
