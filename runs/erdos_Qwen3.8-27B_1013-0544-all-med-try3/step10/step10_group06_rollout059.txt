import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    return w / w.sum()

def grad_weighted_lags(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m = k - (n - 1)
        wt = w[k]
        if m == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif m > 0:
            grad[0:n-m] += wt * (1.0 - h[m:n])
            grad[m:n] -= wt * h[0:n-m]
        else:
            mm = -m
            grad[0:n-mm] += wt * (1.0 - h[mm:n])
            grad[mm:n] -= wt * h[0:n-mm]
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    pen = 1e7

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_val(c, alpha)
        diff = x.sum() - target_sum
        obj_val += pen * diff * diff
        w = smooth_max_weights(c, alpha)
        grad = grad_weighted_lags(x, n, w)
        grad += 2.0 * pen * diff
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def upsample(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binary_project(h, n):
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def gradient_descent_polish(h, n, deadline, lr_init=0.002, lr_min=0.0001, decay=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    alpha = 200000

    for it in range(200000):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        w = smooth_max_weights(c_all, alpha)
        grad = grad_weighted_lags(h, n, w)

        diff = h.sum() - n / 2.0
        grad += 100.0 * diff

        target_sum = n / 2.0
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)

        s = h_new.sum()
        if s > target_sum:
            excess = s - target_sum
            pos = h_new > 0
            h_new[pos] -= excess / pos.sum() * h_new[pos]
        elif s < target_sum:
            deficit = target_sum - s
            room = 1.0 - h_new
            h_new += deficit / room.sum() * room

        h_new = np.clip(h_new, 0.0, 1.0)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.95

        lr = max(lr, lr_min)

    return best_h, best_c
