def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

def binarize(h, n):

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def penalty_obj(h, n, T, mu):
    c = compute_c5_vec(h, n)
    excess = np.maximum(0, c - T)
    val = np.sum(excess ** 2)
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return val

def penalty_grad(h, n, T, mu):
    c = compute_c5_vec(h, n)
    excess = np.maximum(0, c - T)
    w = 2.0 * excess
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def simulated_annealing(h, n, deadline, rng, block_sizes=(1, 2, 3, 5, 8), T0=0.01, Tend=0.0001, n_iter_max=2000000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    n_ones = int(round(h.sum()))
    
    n_steps = max(1, n_iter_max)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (it / max(1, n_steps // 2))
        if it > n_steps // 2:
            T = Tend
        
        block_size = block_sizes[rng.randint(len(block_sizes))]
        if rng.random() < 0.7:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        else:
            # Block swap
            start_i = rng.randint(n - block_size + 1)
            start_j = rng.randint(n - block_size + 1)
            if abs(start_i - start_j) < block_size:
                # Find non-overlapping position
                if start_i + block_size < start_j:
                    pass
                elif start_j + block_size < start_i:
                    pass
                else:
                    continue
            h[start_i:start_i+block_size], h[start_j:start_j+block_size] = \
                h[start_j:start_j+block_size].copy(), h[start_i:start_i+block_size].copy()
        
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        if delta < 0 or rng.random() < math.exp(-abs(delta) / max(T, 1e-10)):
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        else:
            # Undo
            if block_size == 1 or rng.random() < 0.7:
                pass  # Can't easily undo, just continue with new state
            # Actually for simplicity, we just keep the state (WALKING SA)
            cur_c = c_new
    
    return best_h, best_c

def walk_sa(h, n, deadline, rng, max_iters=500000):
    """Simulated annealing that accepts worse moves (walks the landscape)"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T0 = 0.005
    Tend = 0.00001
    no_improve = 0
    improve_threshold = 20000
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iters)
        T = T0 * (Tend / T0) ** frac
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        if delta < 0 or (T > Tend and rng.random() < math.exp(-delta / T)):
            cur_c = c_new
            no_improve += 1
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            if no_improve > improve_threshold and rng.random() < 0.01:
                # Perturb to escape
                for _ in range(rng.randint(3, 10)):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if a != b:
                        h[a], h[b] = h[b], h[a]
                cur_c = compute_c5(h, n)
                no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c

def block_walk_sa(h, n, deadline, rng, max_iters=300000):
    """Simulated annealing with block moves"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    T0 = 0.008
    Tend = 0.00005
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iters)
        T = T0 * (Tend / T0) ** frac
        
        r = rng.random()
        if r < 0.5:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            swap_type = 'single'
            swap_data = (i, j)
        elif r < 0.8:
            # Block of size 2-4
            bs = rng.choice([2, 3, 4])
            si = rng.randint(n - bs + 1)
            sj = rng.randint(n - bs + 1)
            if abs(si - sj) < bs:
                continue
            old1 = h[si:si+bs].copy()
            old2 = h[sj:sj+bs].copy()
            if np.array_equal(old1, old2):
                continue
            h[si:si+bs] = old2
            h[sj:sj+bs] = old1
            swap_type = 'block'
            swap_data = (si, sj, bs)
        else:
            # Reverse a block
            bs = rng.choice([3, 5, 7, 10])
            si = rng.randint(n - bs + 1)
            old = h[si:si+bs].copy()
            h[si:si+bs] = old[::-1]
            swap_type = 'reverse'
            swap_data = (si, bs)
        
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        if delta < 0 or (T > Tend and rng.random() < math.exp(-delta / max(T, 1e-12))):
            cur_c = c_new
            no_improve += 1
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            if no_improve > 50000 and rng.random() < 0.005:
                for _ in range(rng.randint(5, 15)):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if a != b and h[a] != h[b]:
                        h[a], h[b] = h[b], h[a]
                cur_c = compute_c5(h, n)
                no_improve = 0
        else:
            # Undo
            if swap_type == 'single':
                i, j = swap_data
                h[i], h[j] = h[j], h[i]
            elif swap_type == 'block':
                si, sj, bs = swap_data
                tmp = h[si:si+bs].copy()
                h[si:si+bs] = h[sj:sj+bs]
                h[sj:sj+bs] = tmp
    
    return best_h, best_c

def peak_shaving(h, n, deadline):
    """Identify peak lags and do targeted L-BFGS to reduce them"""
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    top_lags = np.argsort(-c_vec)[:5]
    
    # Minimize the sum of squares of the top lags
    def obj_lags(x):
        x_clip = np.clip(x, 0, 1)
        c = compute_c5_vec(x_clip, n)
        return np.sum(c[top_lags] ** 2)
    
    def grad_lags(x):
        x_clip = np.clip(x, 0, 1)
        c = compute_c5_vec(x_clip, n)
        w = 2.0 * c
        dx = 2.0 / n
        grad = np.zeros(n)
        n_full = len(w)
        for mm in top_lags:
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x_clip[k:n])
                grad[k:n] += wm * (-x_clip[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x_clip[0:nv])
                grad[0:nv] += wm * (-x_clip[kk:n])
        return grad
    
    try:
        res = minimize(obj_lags, h.copy(), jac=grad_lags, method='L-BFGS-B',
                      bounds=[(0,1)]*n, options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def beatty_sequence(n, alpha):
    """Create a Beatty sequence binary pattern"""
    h = np.zeros(n)
    target = n // 2
    positions = []
    seen = set()
    for k in range(1, 200 * n):
        pos = int(k * alpha) % n
        if pos not in seen:
            seen.add(pos)
            positions.append(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    return h
