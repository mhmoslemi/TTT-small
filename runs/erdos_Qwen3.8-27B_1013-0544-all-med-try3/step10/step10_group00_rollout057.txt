import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (target - s) / tot * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (s - target) / tot * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    smx = shifted.max()
    return m + (smx + np.log(np.sum(np.exp(shifted - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    smx = shifted.max()
    w = np.exp(shifted - smx)
    w /= w.sum()
    return w

def grad_weighted(w, h, n):
    dx = 2.0 / n
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-15:
            continue
        m = mm - (n - 1)
        wm = w[mm] * dx
        if m >= 0:
            nv = n - m
            grad[m:m+nv] += wm * (1.0 - h[0:nv])
            grad[0:nv] -= wm * h[m:m+nv]
        else:
            k = -m
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:k+nv])
            grad[k:k+nv] -= wm * h[0:nv]
    return grad

def upsample(h_old, n_old, n_new):
    ratio = n_new / n_old
    h_new = np.zeros(n_new)
    for i in range(n_new):
        src = i / ratio
        i0 = int(src)
        i1 = min(i0 + 1, n_old - 1)
        frac = src - i0
        h_new[i] = (1 - frac) * h_old[i0] + frac * h_old[i1]
    return project_feasible(h_new, n_new)

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = grad_weighted(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, alloct_s, alpha, mu, maxit):
    deadline = time.time() + alloct_s
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 15,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def binarize(h, n):

    if abs(diff) > 1e-10:

            used_i.add(i)
            used_j.add(j)
        
        sub_deadline = min(deadline, time.time() + time_per_start)
        time_left = sub_deadline - time.time()
        if time_left < 1:
            continue
        h_new, c_new = binary_local_search(h_tmp, n, time_left, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def sensitive_positions(h, n, num=20):
    c_vec = compute_c5_vec(h, n)
    alpha = 500000.0
    w = smooth_max_w(c_vec, alpha)
    grad = grad_weighted(w, h, n)
    idx = np.argsort(-np.abs(grad))[:num]
    return idx

def nemesis_subset(h, n, positions, alloct_s, maxit=3000):
    deadline = time.time() + alloct_s
    h_full = h.copy()
    x0 = h_full[positions].copy().astype(float)
    
    def obj_small(x):
        h_tmp = h_full.copy()
        x_clipped = np.clip(x, 0.0, 1.0)
        h_tmp[positions] = x_clipped
        h_tmp = project_feasible(h_tmp, n)
        return compute_c5(h_tmp, n)
    
    try:
        res = minimize(obj_small, x0, method='Nelder-Mead',
                      options={'maxiter': maxit, 'xatol': 1e-14, 'fatol': 1e-14,
                               'maxfev': maxit * len(positions) * 3})
        h_new = h_full.copy()
        h_new[positions] = np.clip(res.x, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h_full, compute_c5(h_full, n)

def sa_binary(h, n, alloct_s, rng):
    """Simulated annealing on binary solutions"""
    deadline = time.time() + alloct_s
    h = binarize(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    t_start = time.time()
    total_time = alloct_s
    T0 = 0.002
    Tend = 0.0001
    
    for it in range(500000):
        if time.time() > deadline:
            break
        frac = (time.time() - t_start) / total_time
        T = T0 * (Tend / T0) ** min(frac, 1.0)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c

def make_structure_init(n, alpha_choice):
    """Create a structured initialization using Beatty-like sequence"""
    h = np.zeros(n)
    target = n // 2
    positions = []
    seen = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha_choice) % n
        if pos not in seen:
            seen.add(pos)
            positions.append(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            h[zeros[0]] = min(diff, 1.0)
        else:
            ones = np.where(h == 1)[0]
            h[ones[0]] = max(diff, -1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = make_structure_init(500, 0.6180339887)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    c_parent = compute_c5(h_parent, n_parent)
    
    best_h = h_parent.copy()
    best_c = c_parent
    best_n = n_parent
    
    # Determine resolutions to try
    n_candidates = []
    if n_parent >= 300:
        n_candidates.append(n_parent)
    if n_parent >= 300:
        n_candidates.append(n_parent * 2)
    if 500 not in n_candidates:
        n_candidates.append(500)
    # Keep at most 2-3 resolutions for time management
    n_candidates = n_candidates[:3]
    
    # Time allocation per resolution
    time_per_res = min(budget_s - 80, 500) / len(n_candidates)
    
    for n in n_candidates:
        res_start = time.time()
        res_deadline = min(res_start + time_per_res, deadline - 30)
        if res_deadline - res_start < 30:
            continue
        
        # Initialize for this resolution
        if n == n_parent:
            h = h_parent.copy()
            c = c_parent
        else:
            h = upsample(h_parent, n_parent, n)
            c = compute_c5(h, n)
        
        # Also try structural initialization
        h_struct = make_structure_init(n, 0.6180339887)
        c_struct = compute_c5(h_struct, n)
        if c_struct < c:
            h = h_struct
            c = c_struct
        h_struct2 = make_structure_init(n, 0.41421356)
        c_struct2 = compute_c5(h_struct2, n)
        if c_struct2 < c:
            h = h_struct2
            c = c_struct2
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_n = n
        
        # Phase A: L-BFGS with smooth max (40% of time)
        pa_time = 0.4 * (res_deadline - res_start)
        pa_deadline = res_start + pa_time
        if time.time() < pa_deadline - 5:
            h, c = run_lbfgs(h, n, min(pa_time, 40), alpha=20000, mu=0.01, maxit=2000)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n
            h, c = run_lbfgs(h, n, min(pa_time * 0.6, 30), alpha=100000, mu=0.001, maxit=2000)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n
        
        # Phase B: Local continuous search (25% of time)
        pb_time = 0.25 * (res_deadline - res_start)
        pb_deadline = min(res_start + time_per_res * 0.65, deadline - 30)
        if time.time() < pb_deadline - 5:
            h, c = local_continuous_search(h, n, pb_time,
                step_scales=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n
        
        # Phase C: Binary + combinatorial search (20% of time)
        pc_deadline = min(res_start + time_per_res * 0.85, deadline - 30)
        if time.time() < pc_deadline - 5:
            time_left = pc_deadline - time.time()
            h, c = binary_local_search(h, n, time_left * 0.4, rng)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n
            # Multi-start
            if time.time() < pc_deadline - 5:
                time_left = pc_deadline - time.time()
                h, c = multistart_binary(h, n, time_left * 0.6, rng, num_starts=2)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    best_n = n
        
        # Phase D: Un-binarize sensitive + Nelder-Mead (10% of time)
        pd_deadline = min(res_start + time_per_res, deadline - 30)
        if time.time() < pd_deadline - 5 and best_n == n:
            sens = sensitive_positions(best_h, n, num=15)
            h, c = nemesis_subset(best_h, n, sens, pd_deadline - time.time() - 2, maxit=2000)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                best_n = n
    
    # Final global polishing phase
    final_deadline = deadline - 5
    if time.time() < final_deadline - 30:
        # L-BFGS with very high alpha
        fin_time = min(60, final_deadline - time.time() - 20)
        h, c = run_lbfgs(best_h, best_n, fin_time, alpha=500000, mu=0.0001, maxit=3000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    if time.time() < final_deadline - 20:
        # Fine coordinate descent
        fin_time = min(60, final_deadline - time.time() - 10)
        h, c = local_continuous_search(best_h, best_n, fin_time,
            step_scales=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    if time.time() < final_deadline - 10 and best_n == best_n:
        # SA on binary
        fin_time = min(40, final_deadline - time.time() - 5)
        h, c = sa_binary(best_h, best_n, fin_time, rng)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    if time.time() < final_deadline - 5:
        # Nelder-Mead on sensitive positions
        sens = sensitive_positions(best_h, best_n, num=12)
        h, c = nemesis_subset(best_h, best_n, sens, final_deadline - time.time() - 2, maxit=2000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
