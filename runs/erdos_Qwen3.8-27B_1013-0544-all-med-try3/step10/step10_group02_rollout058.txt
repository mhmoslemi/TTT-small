def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0

            if tot < 1e-15:

    sweeps_no_improve = 0

def sm_obj(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)  # max is 0
    # log-sum-exp of shifted
    logsum = np.log(np.sum(np.exp(shifted)))  # since max(shifted)=0, this is fine
    val = m + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 6})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_desc_1d(c_vec, slopes, L, U, hi):
    pos_mask = slopes > 1e-18
    neg_mask = slopes < -1e-18
    zero_mask = ~pos_mask & ~neg_mask
    c_zero_max = c_vec[zero_mask].max() if np.any(zero_mask) else -1e18
    f_at_L = np.max(c_vec + slopes * L) if abs(L) > 1e-18 else c_vec.max()
    f_at_U = np.max(c_vec + slopes * U) if abs(U) > 1e-18 else c_vec.max()
    f_0 = c_vec.max()
    t_hi = min(f_at_L, f_at_U, f_0)
    t_lo = min(c_zero_max, t_hi - 0.002)
    best_delta = 0.0
    for _ in range(80):
        t_mid = (t_lo + t_hi) * 0.5
        lo_d = L
        hi_d = U
        if np.any(pos_mask):
            ub_vals = (t_mid - c_vec[pos_mask]) / slopes[pos_mask]
            hi_d = min(hi_d, float(ub_vals.min()))
        if np.any(neg_mask):
            lb_vals = (t_mid - c_vec[neg_mask]) / slopes[neg_mask]
            lo_d = max(lo_d, float(lb_vals.max()))
        if c_zero_max > t_mid + 1e-14:
            t_lo = t_mid
            continue
        if lo_d <= hi_d + 1e-15:
            t_hi = t_mid
            best_delta = max(L, min(U, max(lo_d, min(hi_d, 0.0))))
        else:
            t_lo = t_mid
        if t_hi - t_lo < 1e-15:
            break
    return hi + best_delta

def coordinate_descent(h, n, deadline, max_sweeps=2000, order='sequential'):
    h = h.copy()
    c_vec = compute_c5_vec(h, n)
    best_c = c_vec.max()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    no_improve = 0
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if order == 'random':
            idx_order = rng.permutation(n)
        else:
            idx_order = range(n)
        improved = 0
        for i in idx_order:
            if time.time() > deadline:
                break
            slopes = np.zeros(n_corr)
            m_lo = max(0, n - 1 - i)
            m_hi = min(2 * n - 2, 2 * n - 2 - i)
            if m_lo <= m_hi:
                h_idx = np.arange(m_lo, m_hi + 1) + i - n + 1
                slopes[m_lo:m_hi + 1] += dx * (1.0 - h[h_idx])
            m_lo2 = max(0, i)
            m_hi2 = min(2 * n - 2, i + n - 1)
            if m_lo2 <= m_hi2:
                h_idx2 = i + n - 1 - np.arange(m_lo2, m_hi2 + 1)
                slopes[m_lo2:m_hi2 + 1] += -dx * h[h_idx2]
            hi_val = h[i]
            L = -hi_val
            U = 1.0 - hi_val
            if abs(L) < 1e-15 and abs(U) < 1e-15:
                continue
            new_val = coord_desc_1d(c_vec, slopes, L, U, hi_val)
            delta = new_val - hi_val
            if abs(delta) > 1e-14:
                h[i] = new_val
                c_vec += slopes * delta
                improved += 1
        c_now = c_vec.max()
        if c_now < best_c - 1e-15:
            best_c = c_now
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break
        if (sweep + 1) % 20 == 0:
            c_vec = compute_c5_vec(h, n)
    h = project_feasible(h, n)
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
    return h, best_c

def sm_obj(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    logsum = np.log(np.sum(np.exp(shifted)))
    val = m + logsum / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_phase(h, n, deadline, alphas, mu=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 6})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

def downsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

def random_restart_descent(h, n, deadline, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(42)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = h.copy()
        mask = rng.random(n) < 0.3
        h_pert[mask] = np.clip(h_pert[mask] + rng.uniform(-0.1, 0.1, mask.sum()), 0, 1)
        h_pert = project_feasible(h_pert, n)
        time_per = (deadline - time.time()) / (n_restarts - r) * 0.8
        sub_deadline = time.time() + min(time_per, deadline - time.time())
        h_r, c_r = coordinate_descent(h_pert, n, sub_deadline, max_sweeps=200, order='random')
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
    return best_h, best_c
