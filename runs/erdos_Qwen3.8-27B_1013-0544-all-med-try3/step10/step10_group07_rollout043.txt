def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def apply_swap(corr, h, i, j, n, k_arr, nm1):
    """Apply swap h[i](1->0), h[j](0->1) to corr and h in place."""
    ik = i - k_arr
    jk = j - k_arr
    ikp = i + k_arr
    jkp = j + k_arr
    A = (ik >= 0) & (ik < n)
    B = (jk >= 0) & (jk < n)
    C = (ikp >= 0) & (ikp < n)
    D = (jkp >= 0) & (jkp < n)
    delta = np.zeros(2 * n - 1)
    if A.any():
        delta[A] += h[ik[A]]
    if B.any():
        delta[B] -= h[jk[B]]
    if C.any():
        delta[C] -= (1.0 - h[ikp[C]])
    if D.any():
        delta[D] += (1.0 - h[jkp[D]])
    delta[nm1] -= 2.0
    ki = nm1 + (j - i)
    if 0 <= ki < 2 * n - 1:
        delta[ki] += 1.0
    kj = nm1 + (i - j)
    if 0 <= kj < 2 * n - 1:
        delta[kj] += 1.0
    corr[:] += delta
    h[i] = 0.0
    h[j] = 1.0

def undo_swap(corr, h, i, j, n, k_arr, nm1):
    """Undo: h[i](0->1), h[j](1->0)."""
    ik = i - k_arr
    jk = j - k_arr
    ikp = i + k_arr
    jkp = j + k_arr
    A = (ik >= 0) & (ik < n)
    B = (jk >= 0) & (jk < n)
    C = (ikp >= 0) & (ikp < n)
    D = (jkp >= 0) & (jkp < n)
    delta = np.zeros(2 * n - 1)
    if A.any():
        delta[A] -= h[ik[A]]
    if B.any():
        delta[B] += h[jk[B]]
    if C.any():
        delta[C] += (1.0 - h[ikp[C]])
    if D.any():
        delta[D] -= (1.0 - h[jkp[D]])
    delta[nm1] += 2.0
    ki = nm1 + (j - i)
    if 0 <= ki < 2 * n - 1:
        delta[ki] -= 1.0
    kj = nm1 + (i - j)
    if 0 <= kj < 2 * n - 1:
        delta[kj] -= 1.0
    corr[:] += delta
    h[i] = 1.0
    h[j] = 0.0

def sa_optimize(h, n, deadline, rng, max_restarts=200):
    """Simulated annealing with restarts on binary solution."""
    k_arr = np.arange(-(n - 1), n)
    nm1 = n - 1
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    T = 3.0
    T_min = 0.15
    cool_rate = 0.99985
    stuck_counter = 0
    improve_counter = 0
    restart_count = 0
    steps_since_improve = 0
    
    while time.time() < deadline and restart_count < max_restarts:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        old_corr = corr.copy()
        old_h_i = h[i]
        old_h_j = h[j]
        apply_swap(corr, h, i, j, n, k_arr, nm1)
        new_max = float(corr.max())
        
        if new_max < cur_max:
            # Accept improvement
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            cur_max = new_max
            steps_since_improve = 0
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Maybe accept worsening
            dE = new_max - cur_max
            if rng.random() < np.exp(-dE / T):
                ones[ones.index(i)] = j
                zeros[zeros.index(j)] = i
                cur_max = new_max
            else:
                # Rereject
                corr[:] = old_corr
                h[i] = old_h_i
                h[j] = old_h_j
        
        steps_since_improve += 1
        
        # Cooling
        T *= cool_rate
        if T < T_min:
            T = T_min
        
        # Restart logic
        if steps_since_improve > 8000 + 4000 * restart_count:
            # Reheat and perturb
            restart_count += 1
            T = 2.5
            # Return to best
            h[:] = best_h
            corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
            cur_max = float(corr.max())
            best_max = cur_max
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            # Perturb: random 3-7 swaps
            n_perturb = rng.randint(3, 8)
            for _ in range(n_perturb):
                if len(ones) > 0 and len(zeros) > 0:
                    pi = ones[rng.randint(len(ones))]
                    pj = zeros[rng.randint(len(zeros))]
                    apply_swap(corr, h, pi, pj, n, k_arr, nm1)
                    ones[ones.index(pi)] = pj
                    zeros[zeros.index(p
