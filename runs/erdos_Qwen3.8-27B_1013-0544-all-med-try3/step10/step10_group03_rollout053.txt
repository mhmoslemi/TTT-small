        grad[kk:n] -= h[0:nv] * dx
        grad[0:nv] += (1.0 - h[kk:n]) * dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] = -h[k:n] * dx
        grad[k:n] += (1.0 - h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[0:nv] += (1.0 - h[kk:n]) * dx
        grad[kk:n] = -h[0:nv] * dx
    return grad

def lcm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lcm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    grad = np.zeros(n)
    thresh = 1e-12
    idxs = np.where(w > thresh)[0]
    for idx in idxs:
        g = shift_gradient(x, n, idx)
        grad += w[idx] * g
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

        alpha_list = [500, 2000, 10000, 50000, 200000, 1000000]

            break

            res = minimize(lcm_obj, h, jac=lcm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,

def pair_swap_opt(h, n, deadline, steps=None):

    best_c = compute_c5(h, n)
    if steps is None:

        improved = True
        while improved:

def multiresolve(h, n, deadline):
