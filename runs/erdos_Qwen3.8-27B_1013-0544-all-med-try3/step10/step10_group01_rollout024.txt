def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_smooth_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lse_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w

def grad_c5_weighted(weights, x, n, threshold=1e-12):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < threshold:
            continue
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            grad[:m] += w * (1.0 - x[k:]) * dx
            grad[k:] += w * (-x[:m]) * dx
        else:
            kk = -k
            m = n - kk
            grad[kk:] += w * (1.0 - x[:m]) * dx
            grad[:m] += w * (-x[kk:]) * dx
    return grad

def obj_lse(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = lse_smooth_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    w = lse_weights(c, alpha)
    grad = grad_c5_weighted(w, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_power(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 1e-300)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_power(x, n, p, mu):
    c = compute_c5_vec(x, n)
    c_pos = np.maximum(c, 1e-300)
    w = c_pos ** p
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w = (p / s) * (s ** ((1.0 - p) / p)) * w / len(c)
    w = w / w.sum()
    grad = grad_c5_weighted(w, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upscale_h(h, n_new):
    n_old = len(h)
    h_new = np.zeros(n_new)
    for j in range(n_new):
        i = min(int(j * n_old / n_new), n_old - 1)

            h_work = h5.copy()
    
    # Phase 6: Active set final
    if time.time() < deadline - 100:
        p6_end = min(time.time() + 60, deadline - 80)
        h6, c6 = active_set_optimize(h_work, n_work, p6_end,
                                      top_k=16, alpha_start=1000000, alpha_end=50000000, rounds=10)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
            h_work = h6.copy()
    
    # Phase 7: Pair exchange fine-tuning
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 40, deadline - 30)
        h7, c7 = pair_exchange_search(h_work, n_work, p7_end,
                                       deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                       radius=30)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
            h_work = h7.copy()
    
    # Phase 8: L-BFGS polish with very high alpha
    if time.time() < deadline - 20:
        for alpha in [5000000, 10000000, 20000000]:
            if time.time() > deadline - 10:
                break
            h8, c8 = lbfgs_round(h_work, n_work, deadline - 2, alpha, 0.001, 1000)
            if c8 < best_c:
                best_c = c8
                best_h = h8.copy()
    
    # Phase 9: Final stochastic kick
    if time.time() < deadline - 10:
        h9, c9 = stochastic_local_search(h_work, n_work, deadline - 2, n_trials=20000)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()
    
    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
