def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            tot = h[mask].sum()
            if tot < 1e-13:

def corr_gradient(h, n, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h"""
    dx = 2.0 / n

        w = weights[lag_idx] * dx

            nv = n - k

            grad[k:] += w * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += w * (1.0 - h[:nv])
            grad[:nv] += w * (-h[kk:])
    return grad

def make_quadratic_residues(n):
    """Create binary sequence based on quadratic residues"""
    h = np.zeros(n)
    target = n // 2
    # Use a prime-like number for good distribution

        p += 1
    candidates = [k * k % n for k in range(1, n)]
    candidates = sorted(set(candidates))
    # Pick n/2 from candidates

    for pos in chosen:

    return h
