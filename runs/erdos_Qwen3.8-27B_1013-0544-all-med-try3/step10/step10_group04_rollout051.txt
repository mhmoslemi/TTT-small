        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad_fast(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    W = w
    W_flip = W[::-1].copy()
    corr1 = np.correlate(1.0 - x, W, mode='full')
    corr2 = np.correlate(x, W_flip, mode='full')
    idx = 3 * n - 3 - np.arange(n)
    first_term = corr1[idx]
    second_term = corr2[idx]
    grad = dx * (first_term - second_term)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 100.0), (500, 500.0), (1000, 2000.0),
                   (3000, 5000.0), (8000, 10000.0), (20000, 30000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad_fast,
                           args=(n, alpha, mu), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c
