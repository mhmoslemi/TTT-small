        grad += 2.0 * mu * (h.sum() - n/2.0)
    return grad

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_thue_morse(n):
    """Thue-Morse sequence construction"""
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    # Map to [0,1] values
    h = t.astype(float)
    return project_feasible(h, n)

def projected_gradient_step(h, n, grad, step_size):
    """One step of projected gradient descent"""
    h_new = h - step_size * grad
    return project_feasible(h_new, n)
