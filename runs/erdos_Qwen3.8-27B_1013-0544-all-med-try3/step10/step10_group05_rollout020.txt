    a_k = np.zeros(2*n-1)

        # Second factor: h[i-k] * (1-h[i]) if 0<=i-k<n

    ts = np.linspace(0, 1, n_grid)
    vals = np.max(c_k + a_k * ts[:, None] + b_k * ts[:, None], axis=1)
    best_idx = np.argmin(vals)
    
    # Refine with golden section
    if best_idx == 0:
        return 0.0
    elif best_idx == n_grid - 1:
        return 1.0
    else:

        hi = ts[best_idx+1]
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2

            else:

```python
def optimize_coord(h, n, i, n_grid=500):
    dx = 2.0 / n
    v = h[i]
    ov = compute_c5_vec(h, n)  # (2n-1,)
    
    # Precompute slopes for k≠0
    # m index in ov corresponds to shift k = m - (n-1)
    slopes = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        if k == 0:
            continue
        s = 0.0
        if 0 <= i+k < n:
            s += (1.0 - h[i+k])
        if 0 <= i-k < n:
            s -= h[i-k]
        slopes[m] = dx * s
    
    # For k=0: ov_0(t) = ov[0_offset] + dx*(t - t² - v + v²)
    # where 0_offset = n-1
    ov0 = ov[n-1]
    
    def g(t):
        # Linear parts
        vals = ov + (t - v) * slopes
        # Fix k=0 term
        vals[n-1] = ov0 + dx * (t - t*t - v + v*v)
        return vals.max()
    
    # Grid search
    ts = np.linspace(0, 1, n_grid)
    gs = np.array([g(t) for t in ts])
    best_t = ts[np.argmin(gs)]
    
    # Refine: golden section in a small interval
    idx = np.argmin(gs)
    if idx == 0:
        lo, hi = 0.0, ts[1]
    elif idx == n_grid - 1:
        lo, hi = ts[-2], 1.0
    else:
        lo, hi = ts[idx-1], ts[idx+1]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(60):
        x1 = hi - gr*(hi-lo)
        x2 = lo + gr*(hi-lo)
        if g(x1) < g(x2):
            hi = x2
        else:
            lo = x1
    best_t = (lo+hi)/2
    
    return best_t
```
