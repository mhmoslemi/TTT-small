import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def grad_single_lag(h, n, lag):
    """Compute gradient of overlap at a single lag."""
    n2 = len(h)
    grad = np.zeros(n2)
    if lag > 0:
        # f = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # df/dh[i] = (1-h[i+lag]) for i in [0, n-1-lag]
        # df/dh[i] = -h[i-lag] for i in [lag, n-1]
        grad[0:n-lag] += (1.0 - h[lag:n])
        grad[lag:n] += -h[0:n-lag]
    elif lag < 0:
        m = -lag
        # f = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        # df/dh[i] = (1-h[i-m]) for i in [m, n-1]
        # df/dh[i] = -h[i+m] for i in [0, n-1-m]
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += -h[m:n]
    return grad

def active_set_optimize(h, n, K, deadline, maxiter=2000, lr=0.01, lr_decay=0.999):
    """Optimize using active set (top-K lags) with gradient descent / L-BFGS-B."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Track top K lags
    c_all = compute_all_lags(h, n)
    top_k = np.argsort(c_all)[-K:]
    
    step_size = lr
    patience = 0
    max_patience = 500
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        
        # Update active set periodically
        if it % 50 == 0:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-K:]
            # Adaptive step size
            if patience > max_patience // 2:
                step_size *= 0.8
        
        # Compute gradient as weighted sum of top-K lag gradients
        c_vals = c_all[top_k]
        c_max = c_vals.max()
        # Softmax weights
        w = np.exp(10 * (c_vals - c_max))
        w /= w.sum()
        
        grad = np.zeros(n)
        for idx in range(K):
            lag = top_k[idx] - (n - 1)
            if lag == 0:
                # f = sum h[i]*(1-h[i]) = sum h - sum h^2
                grad += w[idx] * (1.0 - 2.0 * h)
            else:
                grad += w[idx] * grad_single_lag(h, n, lag)
        
        grad *= dx
        
        # Projected gradient step
        h_new = h - step_size * grad
        # Project to feasible: clip and adjust sum
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target_sum - h_new.sum()
        if diff > 0:
            room = 1.0 - h_new
            tot = room.sum()
            if tot > 1e-12:

            tot = h_new.sum()
            if tot > 1e-12:
                h_new += (diff / tot) * h_new
        h_new = np.clip(h_new, 0.0, 1.0)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new  # Still accept for exploration? Or reject?
            # Actually for pure descent, reject
            h = best_h  # revert
            patience += 1
            step_size *= 0.95
        
        if step_size < 1e-10:
            break
    
    return best_h, best_c

def active_set_lbfgsb(h, n, K, deadline, maxiter=5000):
    """Use L-BFGS-B but only compute gradient for top-K lags."""
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Pre-compute which lags to use
    c_all = compute_all_lags(h, n)
    top_k = np.argsort(c_all)[-K:].tolist()
    
    penalty = 1e10
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        # Compute only top-K lags
        c_vals = np.zeros(K)
        for idx, k_pos in enumerate(top_k):
            lag = k_pos - (n - 1)
            if lag > 0:
                c_vals[idx] = np.sum(x[0:n-lag] * (1.0 - x[lag:n])) * dx
            elif lag < 0:
                m = -lag
                c_vals[idx] = np.sum(x[m:n] * (1.0 - x[0:n-m])) * dx
            else:
                c_vals[idx] = np.sum(x * (1.0 - x)) * dx
        
        # Max with smooth approximation
        c_max = c_vals.max()
        obj = c_max
        # Add penalty for constraint violation
        diff = x.sum() - target_sum
        obj += penalty * diff * diff
        
        # Gradient
        grad = np.zeros(n)
        for idx, k_pos in enumerate(top_k):
            lag = k_pos - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * x)
            else:
                grad += grad_single_lag(x, n, lag)
        grad *= dx
        grad += 2.0 * penalty * diff * np.ones(n)
        
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50,
                                'iprint': -1, 'maxfun': maxiter})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def subgradient_polish(h, n, deadline, n_steps=50000, lr_init=0.005):
    """Exact subgradient method: at each step, find the max lag and step against it."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    target_sum = n / 2.0
    lr = lr_init
    no_improve = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if step > 0 and step % 1000 == 0:
            lr *= 0.95
        
        # Find the lag with maximum overlap
        c_all = compute_all_lags(h, n)
        k_star = c_all.argmax()
        lag = k_star - (n - 1)
        
        # Compute subgradient
        if lag == 0:
            sg = (1.0 - 2.0 * h)
        else:
            sg = grad_single_lag(h, n, lag)
        sg *= dx
        
        # Projected subgradient step
        h_new = h - lr * sg
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - h_new
                tot = room.sum()
                if tot > 1e-12:
                    h_new += (diff / tot) * room
            else:
                tot = h_new.sum()
                if tot > 1e-12:
                    h_new += (diff / tot) * h_new
            h_new = np.clip(h_new, 0.0, 1.0)
        
        # Check all lags (not just the one we stepped against)
        c_test = compute_c5(h_new, n)
        if c_test < best_c - 1e-14:
            best_c = c_test
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c

def pairwise_local_search(h, n, deadline, rng, n_rounds=1000):

def diverse_init(n, rng, init_type='random_fourier'):
    """Create diverse initial solutions."""
    if init_type == 'random_fourier':
        # Start with a smooth random function
        freqs = np.fft.rfftfreq(n)
        # Zero out DC
        coeffs = (rng.randn(n // 2 + 1) + 1j * rng.randn(n // 2 + 1)) * 0.1
        coeffs[0] = 0.5  # Mean
        h = np.fft.irfft(coeffs, n)
        h = project_feasible(h, n)
    elif init_type == 'periodic':
        T = rng.choice([4, 8, 16, 32])
        base = rng.uniform(0.2, 0.8, T)
        # Make it sum to T/2
        base = (base - base.mean()) / (base.std() + 1e-10) * 0.3 + 0.5
        base = np.clip(base, 0, 1)
        reps = n // T
        h = np.tile(base[:n % T + (n % T == 0) * T], reps)
        if len(h) < n:
            h = np.concatenate([h, base[:n - len(h)]])
        h = h[:n]
        h = project_feasible(h, n)
    elif init_type == 'blocky':
        # Random blocks of 0 and 1
        block_size = rng.randint(2, 8)
        h = np.zeros(n)
        i = 0
        while i < n:
            blk = min(block_size, n - i)
            if rng.random() < 0.5:
                h[i:i+blk] = rng.uniform(0.7, 1.0)
            else:
                h[i:i+blk] = rng.uniform(0.0, 0.3)
            i += blk
        h = project_feasible(h, n)
    elif init_type == 'gradient':
        # Linear gradient with noise
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x / 2.0) + rng.normal(0, 0.1, n)
        h = project_feasible(h, n)
    else:
        h = project_feasible(rng.uniform(0, 1, n), n)
    return h

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)


def compute_all_lags(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx


def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def grad_single_lag(h, n, lag):
    grad = np.zeros(n)
    if lag > 0:
        grad[0:n - lag] += (1.0 - h[lag:n])
        grad[lag:n] += -h[0:n - lag]
    elif lag < 0:
        m = -lag
        grad[0:n - m] += -h[m:]
        grad[m:n] += (1.0 - h[0:n - m])
    else:
        grad[:] = (1.0 - 2.0 * h)
    return grad


def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= s
    return w

    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)

        else:
            idx = -lag
            grad[0:n - idx] += wk * (1.0 - h[idx:n])
            grad[idx:n] += wk * (-h[0:n - idx])

def lbfgsb_full(h, n, alpha, deadline, maxiter=5000, penalty=1e10):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj += penalty * diff * diff

            no_improve = 0
            momentum = 0.0

        c_all = compute_all_lags(h, n)
        k_star = int(c_all.argmax())
        lag = k_star - (n - 1)
        sg = grad_single_lag(h, n, lag) * dx

        momentum = momentum_decay * momentum + (1 - momentum_decay) * sg
        h_new = h - lr * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h_new
                tot = room.sum()
                if tot > 1e-12:
                    h_new += (diff / tot) * room
            else:
                tot = h_new.sum()
                if tot > 1e-12:
                    h_new += (diff / tot) * h_new
            h_new = np.clip(h_new, 0.0, 1.0)

        c_test = compute_c5(h_new, n)
        if c_test < best_c - 1e-14:
            best_c = c_test
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50000:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-6:
                    break

    return best_h, best_c


def targeted_local_search(h, n, deadline, rng, n_rounds=50000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target_sum = n / 2.0

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break

        if rnd % 200 == 0:
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-6:]
        else:
            c_all = compute_all_lags(h, n)
            top_idx = np.argsort(c_all)[-3:]

        k_star = int(top_idx[rng.randint(len(top_idx))])
        lag = k_star - (n - 1)

        if lag == 0:
            i = rng.randint(n)
            if 0.05 < h[i] < 0.95:
                if h[i] > 0.5:
                    d = min(0.05, (1.0 - h[i]) * 0.5)
                    h[i] += d
                else:
                    d = min(0.05, h[i] * 0.5)
                    h[i] -= d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-14:
                    best_c = c_test
                    best_h = h.copy()
                else:
                    h = best_h.copy()
            continue

        m = abs(lag)
        if m >= n:
            continue

        i = rng.randint(n - m)
        j = i + m

        if h[i] > h[j] + 1e-8:
            d = min(0.04, (h[i] - h[j]) * 0.3, h[i] * 0.4)
            if d < 1e-10:
                continue
            h[i] -= d
            h[j] += d
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c - 1e-14:
                best_c = c_test
                best_h = h.copy()
            else:
                h = best_h.copy()
        elif h[j] > h[i] + 1e-8:
            d = min(0.04, (h[j] - h[i]) * 0.3, h[j] * 0.4)
            if d < 1e-10:
                continue
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            c_test = compute_c5(h, n)
            if c_test < best_c - 1e-14:
                best_c = c_test
                best_h = h.copy()
            else:
                h = best_h.copy()

    return best_h, best_c


def multi_start_optimize(seed, n, budget_s, deadline):
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')

    inits = []
    # Random Fourier
    freqs = np.fft.rfftfreq(n)
    coeffs = (rng.randn(n // 2 + 1) + 1j * rng.randn(n // 2 + 1)) * 0.15
    coeffs[0] = 0.5
    h1 = np.fft.irfft(coeffs, n)
    inits.append(project_feasible(h1, n))

    # Periodic
    T = rng.choice([4, 8, 16])
    base = rng.uniform(0.1, 0.9, T)
    base = 0.5 + 0.4 * (base - base.mean()) / (base.std() + 1e-10)
    base = np.clip(base, 0, 1)
    reps = n // T
    h2 = np.tile(base, reps)
    if len(h2) < n:
        h2 = np.concatenate([h2, base[:n - len(h2)]])
    h2 = h2[:n]
    inits.append(project_feasible(h2, n))

    # Blocky
    block_size = rng.randint(3, 10)
    h3 = np.zeros(n)
    i = 0
    while i < n:
        blk = min(block_size, n - i)
        if rng.random() < 0.5:
            h3[i:i + blk] = rng.uniform(0.6, 1.0)
        else:
            h3[i:i + blk] = rng.uniform(0.0, 0.4)
        i += blk
    inits.append(project_feasible(h3, n))

    time_per_init = max(20, min(60, (deadline - time.time()) // max(1, len(inits)) - 5))

    for h_init in inits:
        if time.time() > deadline - 20:
            break
        h_opt = lbfgsb_full(h_init, n, alpha=5000, deadline=time.time() + time_per_init, maxiter=1000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        best_h = project_feasible(h_init, n_init)
        best_c = compute_c5(best_h, n_init)
    else:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B at parent resolution, multi-alpha
    n = len(best_h)
    alpha_schedule = [200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alpha_schedule:
        if time.time() > deadline - 350:
            break
        remaining = deadline - 350 - time.time()
        if remaining < 10:
            break
        t_call = min(30, remaining * 0.4)
        maxit = min(3000, int(t_call * 40))
        h_opt = lbfgsb_full(best_h, n, alpha, time.time() + t_call, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Subgradient polish at parent resolution
    if time.time() < deadline - 300:
        sg_deadline = min(deadline - 250, time.time() + 40)
        h_opt, c_opt = subgradient_polish(best_h, n, sg_deadline, n_steps=50000, lr_init=0.02)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Upscale to n=1000
    n_up = 1000
    if time.time() < deadline - 250:
        h_up = expand_to_resolution(best_h, len(best_h), n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
        for alpha in [2000, 20000, 200000]:
            if time.time() > deadline - 180:
                break
            remaining = deadline - 180 - time.time()
            if remaining < 15:
                break
            t_call = min(40, remaining * 0.4)
            maxit = min(2000, int(t_call * 30))
            h_opt = lbfgsb_full(best_h, len(best_h), alpha, time.time() + t_call, maxiter=maxit)
            c_opt = compute_c5(h_opt, len(best_h))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Subgradient at n=1000
    if time.time() < deadline - 150:
        sg_deadline = min(deadline - 100, time.time() + 35)
        h_opt, c_opt = subgradient_polish(best_h, len(best_h), sg_deadline, n_steps=30000, lr_init=0.01)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Try n=2000
    if time.time() < deadline - 120:
        n_up2 = 2000
        h_up2 = expand_to_resolution(best_h, len(best_h), n_up2)
        c_up2 = compute_c5(h_up2, n_up2)
        if c_up2 < best_c:
            best_c = c_up2
            best_h = h_up2
        if time.time() < deadline - 80:
            h_opt = lbfgsb_full(best_h, len(best_h), 10000, time.time() + 40, maxiter=1000)
            c_opt = compute_c5(h_opt, len(best_h))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            h_opt = lbfgsb_full(best_h, len(best_h), 100000, time.time() + 25, maxiter=800)
            c_opt = compute_c5(h_opt, len(best_h))
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Multi-start at best resolution
    n_work = len(best_h)
    if time.time() < deadline - 80:
        ms_deadline = min(deadline - 50, time.time() + 50)
        h_ms, c_ms = multi_start_optimize(seed, n_work, (ms_deadline - time.time()), ms_deadline)
        if h_ms is not None and c_ms < best_c:
            best_c = c_ms
            best_h = h_ms

    # Phase 7: Targeted local search
    if time.time() < deadline - 40:
        ls_deadline = min(deadline - 20, time.time() + 30)
        h_opt, c_opt = targeted_local_search(best_h, len(best_h), ls_deadline, rng, n_rounds=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final high-alpha L-BFGS-B
    if time.time() < deadline - 15:
        h_opt = lbfgsb_full(best_h, len(best_h), 5000000, deadline - 8, maxiter=3000)
        c_opt = compute_c5(h_opt, len(best_h))
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final subgradient
    if time.time() < deadline - 5:
        h_opt, c_opt = subgradient_polish(best_h, len(best_h), deadline - 2, n_steps=20000, lr_init=0.005)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
