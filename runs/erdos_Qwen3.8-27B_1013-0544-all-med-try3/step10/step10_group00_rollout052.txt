The BCD idea is: in each constraint, I fix ONE factor. For constraint k:
∑_i h[i]·(1-h[i-k])·dx ≤ t

I can either:
(a) Fix h[i-k] (the "shifted" factor), making it linear in h[i]
(b) Fix h[i] (the "current" factor), making it linear in h[i-k]

Option (a) is what I described. Option (b) is the "dual" where I rearrange:
∑_i h[i-k]·(1-h[i])·dx ≤ t, fixing h[i]

But (b) is actually the same set of constraints just written differently! The correlation at shift k is the same as at shift -k (with h and 1-h swapped). So options (a) and (b) give the same LP.

So the BCD is:
1. Given h^(t), build A^(t) where A^(t)[k,p] = dx·(1-h^(t)[p-j_k])
2. Solve LP: min t s.t. A^(t)@h ≤ t, sum(h)=n/2, 0≤h≤1
3. h^(t+1) = h*_LP (or a blend: α·h*_LP + (1-α)·h^(t))

Let me implement this with over-relaxation α to speed convergence.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def build_A(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(-(n-1), n):
        k = (n-1) + j
        if j >= 0:
            A[k, j:] = dx * (1.0 - h[:n-j])
        else:
            jp = -j
            A[k, :n-jp] = dx * (1.0 - h[jp:])
    return A

def solve_lp_bcd(h_fixed, n):
    """Solve the LP subproblem in BCD"""
    dx = 2.0 / n
    m = 2 * n - 1
    A = build_A(h_fixed, n)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # [A | -1] @ [h; t] <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return h_fixed.copy(), None

def bcd_lp_optimize(h, n, deadline, alpha_relax=0.5, min_improve=1e-10):
    """BCD with LP subproblem"""
    h_curr = project_feasible(h.copy(), n)
    best_h = h_curr.copy()
    best_c = compute_c5(best_h, n)
    stalled = 0
    
    while time.time() < deadline:
        h_lp, t_val = solve_lp_bcd(h_curr, n)
        
        # Blend
        h_new = alpha_relax * h_lp + (1.0 - alpha_relax) * h_curr
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - min_improve:
            best_c = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        
        h_curr = h_new
        
        # Adaptive relaxation
        if stalled > 20:
            alpha_relax = min(1.0, alpha_relax * 1.2)
            if stalled > 50:
                # Perturb and restart
                rng_local = np.random.RandomState(int(time.time() * 1000))
                h_curr = h_curr + rng_local.randn(n) * 0.01
                h_curr = project_feasible(h_curr, n)
                stalled = 0
                alpha_relax = 0.5
        
        # Time check inside loop
        if time.time() > deadline - 2:
            break
    
    return best_h, best_c

def lbfgs_obj_wrapper(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad_wrapper(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj_wrapper, h, jac=lbfgs_grad_wrapper,
                      args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def binary_swap_search(h, n, deadline, rng, max_iters=200000):
    h = binarize(h, n).copy()
    best_c = compute_c5(h, n)
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    ones_list = list(ones)
    zeros_list = list(zeros)
    
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        alpha = 50000.0
        w = smooth_max_w(c_vec, alpha)
        grad = h_grad_from_w(w, h, n)
        
        # Find candidates
        remove_cands = [(grad[i], i) for i in ones_list if grad[i] > 1e-10]
        add_cands = [(grad[j], j) for j in zeros_list if grad[j] < -1e-10]
        
        improved = False
        if remove_cands and add_cands:
            remove_cands.sort(reverse=True)
            add_cands.sort()
            best_gain = 0
            best_i = -1
            best_j = -1
            
            for gi, i in remove_cands[:50]:
                for gj, j in add_cands[:50]:
                    g = gi - gj
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i >= 0:
                h[best_i] = 0.0
                h[best_j] = 1.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(best_i); ones.add(best_j)
                    zeros.discard(best_j); zeros.add(best_i)
                    ones_list = list(ones)
                    zeros_list = list(zeros)
                    improved = True
                    stall = 0
                else:
                    h[best_i] = 1.0
                    h[best_j] = 0.0
                    stall += 1
        
        if not improved:
            stall += 1
            if stall > 5:
                # Random swap
                if ones_list and zeros_list:
                    i = ones_list[rng.randint(len(ones_list))]
                    j = zeros_list[rng.randint(len(zeros_list))]
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        ones.discard(i); ones.add(j)
                        zeros.discard(j); zeros.add(i)
                        ones_list = list(ones)
                        zeros_list = list(zeros)
                        stall = 0
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                if stall > 50:
                    # More aggressive: random multi-swap
                    num_swaps = rng.randint(3, 10)
                    for _ in range(num_swaps):
                        if not ones_list or not zeros_list:
                            break
                        i = ones_list[rng.randint(len(ones_list))]
                        j = zeros_list[rng.randint(len(zeros_list))]
                        h[i], h[j] = h[j], h[i]
                        ones.discard(i); ones.add(j)
                        zeros.discard(j); zeros.add(i)
                    ones_list = list(ones)
                    zeros_list = list(zeros)
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c

                    # Don't revert - accept as new starting point

def upsample(h_old, n_old, n_new):
    """Upsample a solution to higher resolution using linear interpolation"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_val):
    """Create a Beatty sequence initialization"""
    h = np.zeros(n)
    positions = np.zeros(n, dtype=bool)
    for k in range(1, 4 * n + 1):
        pos = int(k * alpha_val) % n
        if not positions[pos]:
            positions[pos] = True
            if positions.sum() >= n // 2:
                break
    h[positions] = 1.0
    # Fix exact sum
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for i in range(int(np.ceil(diff))):
                if i < len(zeros):
                    h[zeros[i]] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for i in range(int(np.ceil(-diff))):
                if i < len(ones):
                    h[ones[i]] = 0.0
    return project_feasible(h, n)

def fine_coord_descent(h, n, deadline):

            if amt < 1e-12:

            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    
    return best_h, best_c

def nemesis_subset(h, n, positions, deadline, maxit=3000):
    h_full = h.copy()
    x0 = h_full[positions].copy()
    
    def obj_small(x):
        h_tmp = h_full.copy()
        x_clipped = np.clip(x, 0.0, 1.0)
        h_tmp[positions] = x_clipped
        h_tmp = project_feasible(h_tmp, n)
        return compute_c5(h_tmp, n)
    
    try:
        res = minimize(obj_small, x0, method='Nelder-Mead',
                      options={'maxiter': maxit, 'xatol': 1e-14, 'fatol': 1e-14})
        h_new = h_full.copy()
        h_new[positions] = np.clip(res.x, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h_full, compute_c5(h_full, n)
