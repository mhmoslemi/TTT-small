from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def shift_grad(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    best_h = h.copy()

            break
        
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Find shifts within tolerance of max

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def shift_grad(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad


def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def topk_obj(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def topk_grad(x, n, top_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = np.array([c[i] for i in top_idx])
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad


def run_lbfgs(h, n, deadline, alphas, mu, maxiter, method='L-BFGS-B'):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(lse_obj, h, jac=lse_grad,
                           args=(n, alpha, mu), method=method,
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-20,
                                    'gtol': 1e-15, 'maxfun': maxiter * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgradient_momentum(h, n, deadline, max_iter=3000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    y = h.copy()
    v = np.zeros(n)
    alpha0 = 0.05
    beta = 0.95

    for t in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(y, n)
        max_val = c_vec.max()
        tol = max(1e-7, 0.02 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]

        g = np.zeros(n)
        for idx in active:
            g += shift_grad(y, n, idx)
        g /= len(active)
        g -= g.mean()

        alpha = alpha0 / (1.0 + 0.05 * t)
        v = beta * v - alpha * g
        y = y + v
        y = project_feasible(y, n)

        c_y = compute_c5(y, n)
        if c_y < best_c:
            best_c = c_y
            best_h = y.copy()

def exact_top2_step(h, n, deadline, max_iter=200):
    """Directly equalize top 2 shifts using line search."""
    best_c = compute_c5(h, n)
    best_h = h.copy()

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        top2 = np.argsort(-c_vec)[:2]
        i1, i2 = top2[0], top2[1]
        c1, c2 = c_vec[i1], c_vec[i2]

        if abs(c1 - c2) < 1e-12:
            continue

        g1 = shift_grad(best_h, n, i1)
        g2 = shift_grad(best_h, n, i2)
        d = -(g1 - g2)
        d -= d.mean()

        improved = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 1e-5]:
            h_trial = best_h + step * d
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                best_c = c_trial
                best_h = h_trial
                improved = True
                break
        if not improved:
            # Try reverse direction
            for step in [0.005, 0.002, 0.001, 0.0005, 0.0001]:
                h_trial = best_h - step * d
                h_trial = project_feasible(h_trial, n)
                c_trial = compute_c5(h_trial, n)
                if c_trial < best_c - 1e-15:
                    best_c = c_trial
                    best_h = h_trial
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c
