def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_obj(x, n, active_set, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_set]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(x, n, active_set, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_set]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_set):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_set(h, n, top_k=15):
    c_vec = compute_c5_vec(h, n)
    c_max = c_vec.max()
    threshold = c_max - max(1e-8, 0.001 * c_max)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    return active

def run_lbfgs(h, n, deadline, alphas, mu=0.005, use_topk=False, top_k=15):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = max(800, int(remaining * 4))
        maxfun = maxit * 10

        if use_topk:
            active = get_active_set(best_h, n, top_k)
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxfun,
                                       'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
                elif c_new <= best_c + 1e-12:
                    best_h = h_new
            except Exception:
                pass
        else:
            try:
                res = minimize(sm_obj, best_h, jac=sm_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxfun,
                                       'maxls': 40, 'maxcor': 50})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new
                elif c_new <= best_c + 1e-12:
                    best_h = h_new
            except Exception:
                pass

        # Update active set periodically
        if use_topk and i < len(alphas) - 1:
            pass  # active set recomputed each call

    return best_h, best_c

def run_nested_optimization(h, n, deadline, alphas_inner, mu_inner, mu_outer=0.001):
    """Two-level: outer smooth max with low alpha, inner with high alpha on active set."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for outer_alpha in [50, 200, 1000, 5000]:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = max(500, int(remaining * 2))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, outer_alpha, mu_outer), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-13, 'maxfun': maxit * 10,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    # Now do fine-tuning with high alpha on active set
    for alpha in alphas_inner:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        active = get_active_set(best_h, n, top_k=20)
        maxit = max(500, int(remaining * 3))
        try:
            res = minimize(topk_obj, best_h, jac=topk_grad,
                          args=(n, active, alpha, mu_inner), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 10,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
            elif c_new <= best_c:
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c

def nesterov_projected(h, n, deadline, alpha=50000, n_iter=3000, eta_init=0.003):
    """Nesterov-accelerated projected gradient."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    y = best_h.copy()
    t = 1.0
    eta = eta_init
    g_prev = sm_grad(y, n, alpha, 0.0)

    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = sm_grad(y, n, alpha, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = y - eta * (g + beta * (g - g_prev))
        h_new = project_feasible(y, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.01)
        else:
            eta *= 0.97
            y = best_h + beta * (y - best_h)
        g_prev = g
        t = t_new
        y = best_h + ((t - 1.0) / t) * (y - best_h)

    return best_h, best_c

def targeted_local_search(h, n, deadline, max_evals=5000):
    """Targeted pair search focusing on high-gradient pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    evals = 0
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for d in deltas:
        improved = True
        while improved and time.time() < deadline and evals < max_evals:
            improved = False
            evals_budget = max_evals - evals
            if evals_budget <= 0:
                break
            # Get gradient to prioritize pairs
            c_vec = compute_c5_vec(best_h, n)
            c_max = c_vec.max()
            active = np.where(c_vec >= c_max - 0.002 * c_max)[0]

            # For each active lag, find the pair with largest "impact"
            for idx in active:
                if evals >= max_evals or time.time() > deadline:
                    break
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    # impact[i] relates to h[i]*(1-h[i+k])
                    impact = best_h[0:nv] * (1.0 - best_h[k:n])
                else:
                    kk = -k
                    nv = n - kk
                    impact = best_h[kk:n] * (1.0 - best_h[0:nv])

                # Find top-impact positions
                top_pos = np.argsort(-impact)
                for pos in top_pos[:20]:
                    if evals >= max_evals or time.time() > deadline:
                        break
                    if k >= 0:
                        i = pos
                        j = pos + k
                    else:
                        i = pos - k
                        j = pos

                    # Try moving mass from i to j and j to i
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        evals += 1
                        if abs(best_h.sum() - n / 2.0) > 1e-10:
                            best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d

                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        evals += 1
                        if abs(best_h.sum() - n / 2.0) > 1e-10:
                            best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def upsample(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 2, n_coarse)
    x_f = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_f, x_c, h_coarse)
    return project_feasible(h_fine, n_fine)

def periodic_pattern(n, period, ones_per_period, phase=0):
    h = np.zeros(n)
    for p in range(ones_per_period):
        start = (p + phase) % period
        h[start::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 1e-12:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < -1e-12:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        return float(val)

    def grad(x):
        return sm_grad(x, n, alpha, 0.0)

    def con_fun(x):
        return x.sum() - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Optimize at n0 (typically 500) - ~120s
    p1_end = min(t0 + 120, deadline - 600)
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,
        alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
        mu=0.005, use_topk=False)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Quick topk refinement at n0
    if time.time() < deadline - 550:
        p1b_end = min(time.time() + 50, deadline - 550)
        h1b, c1b = run_lbfgs(best_h, best_n, p1b_end,
            alphas=[500000, 2000000, 5000000],
            mu=0.002, use_topk=True, top_k=15)
        if c1b < best_c:
            best_c = c1b
            best_h = h1b

    # Phase 2: Upsample to n=1000, optimize - ~350s
    n2 = 1000
    if time.time() < deadline - 400:
        h2_init = upsample(best_h, best_n, n2)
        c2_init = compute_c5(h2_init, n2)
        if c2_init < best_c:
            best_c = c2_init
            best_h = h2_init.copy()
            best_n = n2

        p2_end = min(time.time() + 350, deadline - 250)
        h2, c2 = run_nested_optimization(h2_init, n2, p2_end,
            alphas_inner=[200000, 1000000, 5000000, 20000000],
            mu_inner=0.002)
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n2

    # Phase 3: Try n=2000 if time permits - ~250s
    n3 = 2000
    if time.time() < deadline - 200:
        h3_init = upsample(best_h, best_n, n3)
        c3_init = compute_c5(h3_init, n3)
        if c3_init < best_c:
            best_c = c3_init
            best_h = h3_init.copy()
            best_n = n3

        p3_end = min(time.time() + 250, deadline - 100)
        h3, c3 = run_nested_optimization(h3_init, n3, p3_end,
            alphas_inner=[100000, 500000, 2000000],
            mu_inner=0.001)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n3

    # Phase 4: Try alternative initializations at best_n
    if time.time() < deadline - 150:
        alt_candidates = []
        # Periodic patterns
        for period in [5, 7, 9, 11]:
            if best_n % period == 0:
                ones = period // 2
                h_alt = periodic_pattern(best_n, period, ones)
                alt_candidates.append(h_alt)
        # Random perturbation of best
        h_pert = best_h + rng.normal(0, 0.02, best_n)
        alt_candidates.append(h_pert)

        for h_alt in alt_candidates[:3]:
            if time.time() > deadline - 120:
                break
            h_alt = project_feasible(h_alt, best_n)
            p4_end = min(time.time() + 40, deadline - 100)
            h4, c4 = run_lbfgs(h_alt, best_n, p4_end,
                alphas=[100, 1000, 10000, 100000, 500000],
                mu=0.005, use_topk=False)
            if c4 < best_c:
                best_c = c4
                best_h = h4
                best_n = best_n

    # Phase 5: Nesterov acceleration at best resolution
    if time.time() < deadline - 80:
        p5_end = min(time.time() + 50, deadline - 60)
        h5, c5 = nesterov_projected(best_h, best_n, p5_end,
            alpha=200000, n_iter=2000, eta_init=0.002)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: SLSQP polish
    if time.time() < deadline - 50:
        for alpha_sl in [500000, 2000000]:
            if time.time() > deadline - 40:
                break
            p6_end = min(time.time() + 25, deadline - 35)
            h6, c6 = slqp_polish(best_h, best_n, p6_end,
                alpha=alpha_sl, maxiter=800)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Targeted local search
    if time.time() < deadline - 10:
        p7_end = deadline - 2
        h7, c7 = targeted_local_search(best_h, best_n, p7_end, max_evals=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final: one more SLSQP if time
    if time.time() < deadline - 2:
        h9, c9 = slqp_polish(best_h, best_n, deadline - 1,
            alpha=5000000, maxiter=300)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
