def swap_delta(h, n, i, j):
    """Compute the change in correlation vector when swapping h[i] (1→0) and h[j] (0→1)."""
    # delta[m] = 2*(h[i+m-(n-1)] - h[j+m-(n-1)]) for valid indices
    delta = np.zeros(2*n - 1)
    # For the i part: h[i+m-(n-1)] is valid when 0 <= i+m-(n-1) < n
    # i.e., n-1-i <= m <= n-1-i+n-1 = 2n-2-i
    m_lo_i = max(0, n-1-i)
    m_hi_i = min(2*n-2, n-2-i+n)  # n-1-i+n-1
    # h[i+m-(n-1)] at m gives index i+m-n+1
    # valid: 0 <= i+m-n+1 <= n-1, i.e., n-1-i <= m <= n-2-i+n = 2n-2-i
    m_lo_i = max(0, n-1-i)
    m_hi_i = min(2*n-2, 2*n-2-i)
    if m_lo_i <= m_hi_i:
        idx = i + np.arange(m_lo_i, m_hi_i+1) - (n-1)
        delta[m_lo_i:m_hi_i+1] += 2.0 * h[idx]
    
    # For the j part (negative): -h[j+m-(n-1)]
    m_lo_j = max(0, n-1-j)
    m_hi_j = min(2*n-2, 2*n-2-j)
    if m_lo_j <= m_hi_j:
        idx = j + np.arange(m_lo_j, m_hi_j+1) - (n-1)
        delta[m_lo_j:m_hi_j+1] -= 2.0 * h[idx]
    
    return delta

def simulated_annealing(h_init, n, deadline, seed=42):

    zeros = np.where(h == 0)[0].tolist()

    T_end = 0.0001
    n_iterations = 500000
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / n_iterations)

            zeros.append(j)

                best_h = h.copy()
    
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def swap_delta(h, n, i, j):
    delta = np.zeros(2 * n - 1)
    m_lo_i = max(0, n - 1 - i)
    m_hi_i = min(2 * n - 2, 2 * n - 2 - i)
    if m_lo_i <= m_hi_i:
        idx = i + np.arange(m_lo_i, m_hi_i + 1) - (n - 1)
        delta[m_lo_i:m_hi_i + 1] += 2.0 * h[idx]
    m_lo_j = max(0, n - 1 - j)
    m_hi_j = min(2 * n - 2, 2 * n - 2 - j)
    if m_lo_j <= m_hi_j:
        idx = j + np.arange(m_lo_j, m_hi_j + 1) - (n - 1)
        delta[m_lo_j:m_hi_j + 1] -= 2.0 * h[idx]
    return delta

def fast_binary_search(h_init, n, deadline, max_sweeps=500, n_restarts=3, seed=42):

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def thue_morse_init(n, target_ones, seed=42):
    rng = np.random.RandomState(seed)
    size = 1
    while size < n:
        size *= 2
    tm = np.zeros(size, dtype=np.int8)
    for i in range(1, size):
        tm[i] = tm[i >> 1] ^ (i & 1)
    tm = tm[:n].astype(float)
    ones = np.where(tm == 1)[0]
    zeros = np.where(tm == 0)[0]
    if len(ones) > target_ones:
        rng.shuffle(ones)
        tm[ones[target_ones:]] = 0
    elif len(ones) < target_ones:
        rng.shuffle(zeros)
        tm[zeros[:target_ones - len(ones)]] = 1
    return tm

def periodic_init(n, target_ones, period, n_ones, seed=42):

    h = np.zeros(n)

    pattern[:n_ones] = 1
    for i in range(n):

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target_ones:

        h[ones[target_ones:]] = 0
    elif len(ones) < target_ones:
        rng.shuffle(zeros)
        h[zeros[:target_ones - len(ones)]] = 1
    return h

    dl = t0 + budget_s - 3
    total_time = budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Generate diverse initializations
    inits = []
    inits.append(best_h.copy())
    inits.append(thue_morse_init(n, n // 2, seed=seed))
    inits.append(periodic_init(n, n // 2, period=3, n_ones=1, seed=seed + 1))
    inits.append(periodic_init(n, n // 2, period=4, n_ones=1, seed=seed + 2))
    inits.append(random_binary_init(n, n // 2, seed=seed + 3))

    # Phase 1: Local search from diverse initializations (12% of time)
    p1_end = t0 + 0.12 * total_time
    for init_h in inits:
        if time.time() > p1_end:
            break
        hb, cb = fast_binary_search(init_h, n, p1_end,
                                     max_sweeps=150, n_restarts=1, seed=seed)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 2: Annealed local search from best (25% of time)
    p2_end = t0 + 0.37 * total_time
    hb, cb = annealed_local_search(best_h, n, p2_end, seed=seed, max_cycles=30)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Additional local search
    hb, cb = fast_binary_search(best_h, n, p2_end,
                                 max_sweeps=500, n_restarts=2, seed=seed + 100)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: SLSQP continuous polishing (13% of time)
    p3_end = t0 + 0.50 * total_time
    if time.time() < p3_end:
        hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
        for alpha in [5000, 20000, 80000, 300000]:
            if time.time() > p3_end:
                break
            hp = slsqp_smooth(hp, n, alpha, 1500, p3_end)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 4: Simulated annealing (18% of time)
    p4_end = t0 + 0.68 * total_time
    hb, cb = simulated_annealing(best_h, n, p4_end, seed=seed + 200)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    # Local search after SA
    if time.time() < p4_end:
        hb, cb = fast_binary_search(best_h, n, p4_end,
                                     max_sweeps=400, n_restarts=2, seed=seed + 300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 5: Multi-resolution exploration (12% of time)
    p5_end = t0 + 0.80 * total_time
    for nh in [200, 300, 350]:
        if time.time() > p5_end:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        # Binary-ize and local search
        hb2, cb2 = fast_binary_search(hh, nh, p5_end - 20,
                                       max_sweeps=200, n_restarts=2, seed=seed + nh)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh
        # Also try SLSQP at this resolution
        if time.time() < p5_end - 10:
            hh2 = project_feasible(hh + 0.02 * rng.randn(nh), nh)
            hh2 = slsqp_smooth(hh2, nh, 50000, 500, p5_end - 5)
            ch = compute_c5(hh2, nh)
            if ch < best_c:
                best_c = ch
                best_h = hh2.copy()
                n = nh

    # Phase 6: Extended annealed local search (final stretch)
    p6_end = dl - 5
    hb, cb = annealed_local_search(best_h, n, p6_end, seed=seed + 400, max_cycles=40)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Final SLSQP polish
    if time.time() < dl - 3:
        hp = project_feasible(best_h + 0.01 * rng.randn(n), n)
        hp = slsqp_smooth(hp, n, 500000, 1000, dl - 2)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c = c
            best_h = hp.copy()

    # Final binary local search
    if time.time() < dl:
        hb, cb = fast_binary_search(best_h, n, dl,
                                     max_sweeps=200, n_restarts=1, seed=seed + 500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
