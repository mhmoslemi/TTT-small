import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def make_beatty(n, alpha_frac):
    """Create a Beatty sequence of length n with fraction alpha_frac being 1."""
    h = np.zeros(n)
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    # Use Kronecker sequence: h[i] = 1 if {i * alpha} < alpha_frac
    a = alpha_frac * golden  # or some irrational
    # Better: use the standard construction
    # h[i] = 1 if {i * φ} < 0.5 (gives 50% ones approximately)
    # But we need exactly n/2 ones
    positions = np.zeros(n, dtype=int)
    vals = (np.arange(n) * a) % 1.0
    # Sort and take the smallest n/2
    idx = np.argsort(vals)
    h[idx[:n//2]] = 1.0
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    """Linear upsampling from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def obj_and_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return val, grad

def coordinate_descend(h, n, deadline, max_iter=50000):
    """Project gradient steps in coordinate pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for step in steps:
        iterating = True
        while iterating and time.time() < deadline:
            iterating = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            # Compute weighted gradient
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            
            # Find best pair to swap
            pos = np.where((grad > 1e-8) & (best_h > 1e-8))[0]
            neg = np.where((grad < -1e-8) & (best_h < 1.0 - 1e-8))[0]
            if len(pos) == 0 or len(neg) == 0:
                break
            
            # Try moving amount from pos to neg

            bi, bj = -1, -1
            for i in pos[:30]:

                        bi, bj = i, j

def pair_swap(h, n, deadline, radius=50, deltas=None):
    """Try all pairs within radius, find improving swaps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try: increase i, decrease j
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try: decrease i, increase j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:

def _obj_topk(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad_topk(x, n, active, alpha, mu):

    vals = c[active]

def compute_c5_vec(h, n):

def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:
            break

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Generate alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = []
    for slope in [golden / 2.0, np.sqrt(2.0) / 2.0, (np.sqrt(5.0) - 1.0), 0.4, 0.6, 0.7, 1.1 / 2.0]:
        alt_starts.append(make_beatty(n_orig, slope))

    for s in alt_starts:
        c_s = compute_c5(s, n_orig)
        if c_s < best_c:
            best_c = c_s
            best_h = s.copy()

    # Phase 1: L-BFGS warmup (0-220s)
    p1_end = min(t0 + 220, deadline - 550)
    if p1_end > t0 + 30:
        h1, c1 = phase_lbfgs(best_h, best_n, p1_end,
            alphas=[5, 20, 100, 500, 2000, 10000, 50000, 200000], mu=0.01)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization (220-420s)
    p2_end = min(time.time() + 200, deadline - 380)
    if p2_end > time.time() + 20:
        h2, c2 = phase_active(best_h, best_n, p2_end, top_k=8,
            alphas=[50000, 200000, 1000000, 5000000, 20000000], rounds=20)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Coordinate descent (420-580s)
    p3_end = min(time.time() + 160, deadline - 280)
    if p3_end > time.time() + 15:
        h3, c3 = phase_coord(best_h, best_n, p3_end, max_steps=150000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Binarization + shrink (580-680s)
    p4_end = min(time.time() + 100, deadline - 250)
    if p4_end > time.time() + 10:
        h4, c4 = phase_binarize(best_h, best_n, p4_end)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        if time.time() < p4_end - 10:
            h4b, c4b = phase_shrink(h4, best_n, p4_end)
            if c4b < best_c:
                best_c = c4b
                best_h = h4b

    # Phase 5: Active set again (680-800s)
    p5_end = min(time.time() + 120, deadline - 200)
    if p5_end > time.time() + 15:
        h5, c5 = phase_active(best_h, best_n, p5_end, top_k=6,
            alphas=[100000, 500000, 2000000, 10000000, 50000000], rounds=15)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine pair search (800-920s)
    p6_end = min(time.time() + 120, deadline - 120)
    if p6_end > time.time() + 10:
        h6, c6 = phase_pairs(best_h, best_n, p6_end, radius=50,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final coordinate descent (920-1000s)
    p7_end = min(time.time() + 80, deadline - 50)
    if p7_end > time.time() + 10:
        h7, c7 = phase_coord(best_h, best_n, p7_end, max_steps=100000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Ultra-fine pairs (1000-1070s)
    p8_end = deadline - 5
    if p8_end > time.time() + 10:
        h8, c8 = phase_pairs(best_h, best_n, p8_end, radius=30,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
