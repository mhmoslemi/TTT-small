    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = c.max()
    sh = alpha * (c - m)

        return h, best_c
    
    for i1 in ones:
        if time.time() > deadline:
            break
        for i2 in ones:
            if i1 >= i2:
                continue
            if abs(i1 - i2) > radius * 2:
                continue
            for j1 in zeros:
                if time.time() > deadline:
                    return h, best_c
                for j2 in zeros:
                    if j1 >= j2:
                        continue
                    if abs(j1 - j2) > radius * 2:
                        continue
                    if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                        continue
                    h[i1] = 0.0; h[i2] = 0.0
                    h[j1] = 1.0; h[j2] = 1.0
                    c = compute_c5(h, n)
                    h[i1] = 1.0; h[i2] = 1.0
                    h[j1] = 0.0; h[j2] = 0.0
                    if c < best_c - 1e-13:
                        h[i1] = 0.0; h[i2] = 0.0
                        h[j1] = 1.0; h[j2] = 1.0
                        best_c = c
                        ones = list(np.where(h > 0.5)[0])
                        zeros = list(np.where(h < 0.5)[0])
                        break
                if best_c < compute_c5(h, n) + 1e-10:
                    pass
            ones = list(np.where(h > 0.5)[0])
            zeros = list(np.where(h < 0.5)[0])
    return h, best_c

def binary_sa(h, n, deadline, seed=42, T0=0.008, T_min=0.00005):
    best_c = compute_c5(h, n)
    best_h = h.copy().astype(float)
    cur_h = h.copy().astype(float)
    cur_c = best_c
    rng = np.random.default_rng(seed)
    T = T0
    cool_rate = (T_min / T0) ** (1.0 / 80000)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    step = 0
    while time.time() < deadline:
        if step % 10000 == 0:
            accept_rate = getattr(binary_sa, '_ar', 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        step += 1
        T = max(T * cool_rate, T_min)
    return best_h, best_c

def multi_sweep_sa(h, n, deadline, seeds=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    if seeds is None:
        seeds = [42, 137, 271, 503, 791, 1087]
    remaining = deadline - time.time()
    t_per_seed = remaining / len(seeds)
    for s in seeds:
        if time.time() > deadline - 2:
            break
        h_sa, c_sa = binary_sa(h, n, time.time() + t_per_seed * 0.9, seed=s)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
    return best_h, best_c

def active_shift_lbfgs(h, n, deadline, top_k=8, alphas=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000]
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.003 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        
        try:
            res = minimize(
                active_obj, best_h, jac=active_grad,
                args=(n, active, alpha, 500.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12}
            )
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def active_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgradient_momentum(h, n, deadline, lr_init=0.08, momentum=0.8, seed=99):
    best_c = compute_c5(h, n)
    best_h = h.copy().astype(float)
    vel = np.zeros(n)
    rng = np.random.default_rng(seed)
    lr = lr_init
    
    for it in range(300000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.004 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        k_choice = active[rng.integers(len(active))]
        k = k_choice - (n - 1)
        dx = 2.0 / n
        grad = np.zeros(n)
        if k >= 0:
            nv = n - k
            grad[0:nv] = (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] = (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
        
        cv = h.sum() - n / 2.0
        grad += 0.05 * cv
        
        vel = momentum * vel - lr * grad
        h_new = np.clip(h + vel, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_c if 'best_c' in dir() else c_new
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= 0.99995
    
    return best_h, best_c

def full_lbfgs(h, n, deadline, alphas=None, mus=None):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200, 1000, 5000, 20000, 100000, 500000]
    if mus is None:
        mus = [50.0, 500.0, 5000.0]
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        for mu in mus[:max(1, len(mus)//2)]:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(
                    smooth_obj, best_h, jac=smooth_grad,
                    args=(n, alpha, mu),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12}
                )
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

def gridded_init(n, pattern='blocks'):
    """Create structured initial binary vectors."""
    h = np.zeros(n)
    half = n // 2
    if pattern == 'blocks':
        h[:half] = 1.0
    elif pattern == 'interleaved':
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 1.0 - h.sum() + n/2.0
    elif pattern == 'thue_morse':
        for i in range(n):
            h[i] = 1.0 if (bin(i).count('1') % 2 == 0) else 0.0
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(200, 0.5)
    
    n_parent = len(h_init)
    h_init = project_feasible(h_init, n_parent)
    
    best_c = compute_c5(h_init, n_parent)
    best_h = h_init.copy()
    best_n = n_parent
    
    def update_best(h_new, c_new, n_new):
        nonlocal best_c, best_h, best_n
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            best_n = n_new
    
    resolutions = sorted(set([n_parent, 300, 400, 500, 600, 800]))
    resolutions = [r for r in resolutions if 150 <= r <= 1000]
    n_res = len(resolutions)
    time_per_res = (budget_s - 60) / n_res
    
    for n in resolutions:
        if time.time() > deadline - 10:
            break
        t_res_end = min(time.time() + time_per_res * 1.1, deadline - 10)
        
        # Get starting point
        if n == n_parent:
            h_start = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_parent)
            x_new = np.linspace(0, 2, n)
            h_start = np.interp(x_new, x_old, h_init)
        h_start = project_feasible(h_start, n)
        c_start = compute_c5(h_start, n)
        
        # Also try structured inits
        inits = [h_start]
        for pat in ['blocks', 'thue_morse']:
            hi = gridded_init(n, pattern=pat)
            inits.append(hi)
        
        # Phase A: L-BFGS continuous (if time allows)
        if time.time() < t_res_end - 30:
            t_lb = min(time.time() + time_per_res * 0.25, t_res_end - 30)
            for hi in inits[:2]:
                if time.time() > t_lb:
                    break
                h_lb, c_lb = full_lbfgs(hi, n, t_lb,
                    alphas=[500, 5000, 50000], mus=[100.0, 1000.0])
                update_best(h_lb, c_lb, n)
                if c_lb < c_start:
                    h_start = h_lb
                    c_start = c_lb
        
        # Phase B: Binary optimization
        h_bin = (h_start > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin, n)
        
        # B1: Single swap local search
        if time.time() < t_res_end - 20:
            t_bs = min(time.time() + time_per_res * 0.3, t_res_end - 20)
            h_bs, c_bs = binary_single_swap_search(h_bin, n, t_bs)
            update_best(h_bs, c_bs, n)
            if c_bs < c_bin:
                h_bin = h_bs
                c_bin = c_bs
        
        # B2: Multi-seed SA
        if time.time() < t_res_end - 10:
            t_sa = min(time.time() + time_per_res * 0.35, t_res_end - 10)
            h_sa, c_sa = multi_sweep_sa(h_bin, n, t_sa, seeds=[42, 137, 271, 503, 791])
            update_best(h_sa, c_sa, n)
            if c_sa < c_bin:
                h_bin = h_sa
                c_bin = c_sa
        
        # B3: More single swap after SA
        if time.time() < t_res_end - 5:
            t_bs2 = min(time.time() + 15, t_res_end - 5)
            h_bs2, c_bs2 = binary_single_swap_search(h_bin, n, t_bs2)
            update_best(h_bs2, c_bs2, n)
            if c_bs2 < c_bin:
                h_bin = h_bs2
                c_bin = c_bs2
        
        # Phase C: Active-shift L-BFGS from binary
        if time.time() < t_res_end - 2:
            h_act, c_act = active_shift_lbfgs(h_bin, n, t_res_end - 2,
                top_k=10, alphas=[2000, 10000, 50000, 200000])
            update_best(h_act, c_act, n)
    
    # Final polish at best resolution
    if time.time() < deadline - 5:
        n_f = best_n
        h_f = best_h
        bounds = [(0.0, 1.0)] * n_f
        
        # Try active shift opt
        h_a, c_a = active_shift_lbfgs(h_f, n_f, deadline - 3, top_k=12,
            alphas=[10000, 50000, 200000, 1000000])
        update_best(h_a, c_a, n_f)
        
        # Try from binary of best
        h_bf = (best_h > 0.5).astype(float)
        h_bf = project_feasible(h_bf, n_f)
        h_ab, c_ab = active_shift_lbfgs(h_bf, n_f, deadline - 2, top_k=10,
            alphas=[5000, 50000])
        update_best(h_ab, c_ab, n_f)
    
    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5(best_h, best_n)
    return best_h, final_c, best_n
