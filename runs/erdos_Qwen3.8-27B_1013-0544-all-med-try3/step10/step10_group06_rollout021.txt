t(1-t) - old(1-old) = t - t² - old + old² = (t-old) - (t²-old²) = (t-old)(1-t+old)

But B[lag=0]*t + (c_0 - B[lag=0]*old) should give the new c_0.
B[lag=0] = (1-2*old)*dx
c_0 - B*old + B*t = c_0 + B*(t-old) = c_0 + (1-2*old)*dx*(t-old)

But the actual new c_0 = c_0 - old*(1-old)*dx + t*(1-t)*dx = c_0 + dx*(t - t² - old + old²) = c_0 + dx*(t-old)*(1-t+old)

These are equal only if (1-2*old) = (1-t+old), i.e., t = old. So for t ≠ old, the linear approximation is wrong!

The issue: c_0 is QUADRATIC in h[i], not linear! The other lags (lag ≠ 0) are linear in h[i], but lag 0 is quadratic.

So my linear model c_k(t) = C_base[k] + B[k]*t is only valid for lag ≠ 0. For lag 0, it's:
c_0(t) = c_0 - old*(1-old)*dx + t*(1-t)*dx = (c_0 - dx*old*(1-old)) + dx*(t - t²)

This is a concave quadratic in t (opens downward). So max_k c_k(t) is the max of concave/linear functions, which is NOT convex in general!

This complicates the 1D minimization. The function max_k c_k(t) is the max of:
- One concave quadratic (lag 0)
- (2n-2) linear functions (other lags)

The max of concave and linear functions is not necessarily convex. However, the concave quadratic is "small" compared to the linear functions (it only adds dx * O(1) variation), while the linear terms can have larger slopes.

For practical purposes, I can:
1. Ignore the quadratic term (use the linear approximation) - this might lead to a slightly suboptimal 1D solution but should be close
2. Handle the quadratic term exactly - evaluate the true c_0(t) at the grid points

I'll go with option 2: at the evaluation stage, use the true c_0(t) = c_0_base + dx*(t*(1-t) - old*(1-old)) where c_0_base accounts for all other coordinates.

def compute_B_vectorized(h, n, i, dx):
    B = np.zeros(2 * n - 1)
    lags = np.arange(2 * n - 1) - (n - 1)
    
    # lag = 0
    k0 = n - 1
    B[k0] = (1.0 - 2.0 * h[i]) * dx
    
    # Positive lags
    pos_mask = lags > 0
    l_pos = lags[pos_mask]
    # Term 1: (1-h[i-l])*dx if l <= i
    c1 = l_pos <= i
    if np.any(c1):
        idx1 = i - l_pos[c1]
        B[pos_mask][c1] += (1.0 - h[idx1]) * dx
    # Term 2: -h[i+l]*dx if l <= n-1-i
    c2 = l_pos <= (n - 1 - i)
    if np.any(c2):
        idx2 = i + l_pos[c2]
        B[pos_mask][c2] -= h[idx2] * dx
    
    # Negative lags (m = -lag)
    neg_mask = lags < 0
    m_neg = -lags[neg_mask]
    # Term 1: (1-h[i-m])*dx if m <= i
    c1n = m_neg <= i
    if np.any(c1n):

        B[neg_mask][c1n] += (1.0 - h[idx1n]) * dx

    c2n = m_neg <= (n - 1 - i)
    if np.any(c2n):
        idx2n = i + m_neg[c2n]
        B[neg_mask][c2n] -= h[idx2n] * dx

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx < -500:
        return float(m)
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx < -500:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    elif mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        ww = w[k] * dx
        lag = k - (n - 1)
        if lag == 0:
            grad += ww * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += ww * (1.0 - h[0:n - lag])
            grad[0:n - lag] += ww * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += ww * (1.0 - h[idx:n])
            grad[idx:n] += ww * (-h[0:n - idx])
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000, penalty_coeff=1e8):
    target_sum = n / 2.0
    h = h.copy()
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1, 'gtol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try higher resolution
    for n_try in [1000, 1500]:
        if time.time() > deadline - 120:
            break
        rem = deadline - 100 - time.time()
        if rem < 40:
            break
        h_interp = expand_to_resolution(best_h, len(best_h), n_try)
        h_opt = lbfgsb_optimize(h_interp, n_try, 100000, time.time() + min(25, rem * 0.3), maxiter=1000)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_try
        if time.time() > deadline - 80:
            break
        h_opt2 = lbfgsb_optimize(best_h, n, 2000000, time.time() + 20, maxiter=1500)
        c_opt2 = compute_c5(h_opt2, n)
        if c_opt2 < best_c:
            best_c = c_opt2
            best_h = h_opt2

    # Phase 7: Final coordinate descent at best resolution
    if time.time() < deadline - 40:
        n_final = len(best_h)
        cd_deadline3 = min(deadline - 20, time.time() + 60)
        h_opt, c_opt = coord_descent(best_h, n_final, 50, cd_deadline3, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B
    if time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, deadline - 3, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final tiny CD
    if time.time() < deadline - 2:
        n_final = len(best_h)
        h_opt, c_opt = coord_descent(best_h, n_final, 20, deadline - 1, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
