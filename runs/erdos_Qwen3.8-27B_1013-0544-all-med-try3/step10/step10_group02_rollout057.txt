def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

            break

            tot = room.sum()
            if tot < 1e-15:

            if tot < 1e-15:

def grad_corr(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx

        kk = -k
        nv = n - kk
        grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx

def lp_step(h, n, top_m, deadline):

    c_vec = compute_c5_vec(h, n)
    active = np.argsort(-c_vec)[:top_m]
    c_active = c_vec[active]
    c_max = c_active.max()
    
    # Build LP: min t s.t. g_k^T delta - t <= -c_k, bounds, sum constraint
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((top_m, n + 1))
    b_ub = np.zeros(top_m)
    for i, idx in enumerate(active):
        g = grad_corr(h, n, idx)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -c_vec[idx]
    
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((0.0, c_max + 0.5))

    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0 - h.sum()])

        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):  # step by 1 for thoroughness
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)

                        continue

                        best_h[j] += d

                            best_c = c
                            improved = True
                        else:

                            best_h[j] -= d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_corr(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[:nv] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:] += (1.0 - h[:nv]) * dx
        grad[:nv] += (-h[kk:]) * dx
    return grad

def lp_active_step(h, n, top_m, deadline):
    c_vec = compute_c5_vec(h, n)
    active = np.argsort(-c_vec)[:top_m]
    c_max = c_vec.max()
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((top_m, n + 1))
    b_ub = np.zeros(top_m)
    for i, idx in enumerate(active):
        g = grad_corr(h, n, idx)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -c_vec[idx]
    
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((0.0, c_max + 1.0))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0 - h.sum()])
    
    try:
        remaining = deadline - time.time()
        if remaining < 2:
            return None, False
        tlim = min(3.0, remaining)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',

def smooth_max_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()

def smooth_max_grad(x, n, alpha, mu):

    m = c.max()

    mx = sh.max()

    return grad

def lbfgs_phase(h, n, deadline, alphas=None, mu=0.01):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 50000, 500000, 5000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        maxit = max(500, int((deadline - time.time()) * 6))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_lbfgs(h, n, deadline, top_k=5, alphas=None):
    if alphas is None:
        alphas = [50000, 200000, 1000000, 5000000, 20000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for rnd in range(20):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        threshold = c_max - max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(-c_vec[active])[:top_k]
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            maxit = max(300, int((deadline - time.time()) * 3))
            try:
                res = minimize(active_sm_obj, best_h, jac=active_sm_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-18, 'maxfun': maxit * 10,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgrad_refine(h, n, deadline, n_iter=8000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.004
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-10, 0.0015 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        grad = np.zeros(n)
        for idx in active:
            grad += grad_corr(h, n, idx)
        grad /= max(1, len(active))
        
        h_new = project_feasible(h - eta * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.008, 0.012)
        else:
            eta *= 0.975
        h = h_new
    return best_h, best_c

def pairwise_refine(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def coordinate_block_opt(h, n, deadline, block_size=25):
    """Optimize blocks of coordinates sequentially"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    alpha = 200000
    
    for sweep in range(50):
        if time.time() > deadline - 5:
            break
        for start in range(0, n, block_size):
            if time.time() > deadline - 5:
                return best_h, best_c
            end = min(start + block_size, n)
            
            # Fix all other coordinates, optimize only this block
            h_fixed = best_h.copy()
            
            def block_obj(db, n=n, start=start, end=end, h_fix=h_fixed, alpha=alpha):
                x = h_fix.copy()
                x[start:end] = np.clip(db, 0, 1)
                c = compute_c5_vec(x, n)
                m = c.max()
                sh = alpha * (c - m)
                mx = sh.max()
                if mx > 500:
                    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
                return m + np.log(np.sum(np.exp(sh))) / alpha
            
            x0 = best_h[start:end].copy()
            bl = [(0.0, 1.0)] * (end - start)
            try:
                res = minimize(block_obj, x0, method='L-BFGS-B', bounds=bl,
                              options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14})
                h_try = best_h.copy()
                h_try[start:end] = np.clip(res.x, 0, 1)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_h = h_try
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    total = budget_s - 10
    
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Phase 1: L-BFGS smooth max (25%)
    p1_end = t0 + total * 0.22
    h1, c1 = lbfgs_phase(best_h, n, p1_end,
        alphas=[50, 200, 1000, 5000, 50000, 500000], mu=0.01)
    if c1 < best_c:
        best_c, best_h = c1, h1
    
    # Phase 2: Active set L-BFGS (12%)
    p2_end = min(time.time() + total * 0.12, deadline - total * 0.45)
    h2, c2 = active_lbfgs(best_h, n, p2_end, top_k=5,
        alphas=[100000, 500000, 2000000, 10000000])
    if c2 < best_c:
        best_c, best_h = c2, h2
    
    # Phase 3: LP-based active set refinement (18%)
    p3_end = min(time.time() + total * 0.18, deadline - total * 0.28)
    h3, c3 = lp_refine(best_h, n, p3_end, top_m=4, max_iters=250)
    if c3 < best_c:
        best_c, best_h = c3, h3
    
    # Phase 4: More L-BFGS with higher alpha (10%)
    p4_end = min(time.time() + total * 0.10, deadline - total * 0.18)
    h4, c4 = lbfgs_phase(best_h, n, p4_end,
        alphas=[100000, 1000000, 5000000], mu=0.005)
    if c4 < best_c:
        best_c, best_h = c4, h4
    
    # Phase 5: Active set L-BFGS focused (10%)
    p5_end = min(time.time() + total * 0.10, deadline - total * 0.10)
    h5, c5 = active_lbfgs(best_h, n, p5_end, top_k=3,
        alphas=[500000, 2000000, 10000000, 50000000])
    if c5 < best_c:
        best_c, best_h = c5, h5
    
    # Phase 6: LP again (10%)
    p6_end = min(time.time() + total * 0.10, deadline - total * 0.08)
    h6, c6 = lp_refine(best_h, n, p6_end, top_m=6, max_iters=200)
    if c6 < best_c:
        best_c, best_h = c6, h6
    
    # Phase 7: Subgradient (8%)
    p7_end = min(time.time() + total * 0.08, deadline - total * 0.05)
    h7, c7 = subgrad_refine(best_h, n, p7_end, n_iter=10000)
    if c7 < best_c:
        best_c, best_h = c7, h7
    
    # Phase 8: Final LP (4%)
    p8_end = min(time.time() + total * 0.04, deadline - total * 0.02)
    h8, c8 = lp_refine(best_h, n, p8_end, top_m=5, max_iters=100)
    if c8 < best_c:
        best_c, best_h = c8, h8
    
    # Phase 9: Pairwise fine-tuning (4%)
    p9_end = deadline - 2
    h9, c9 = pairwise_refine(best_h, n, p9_end, radius=30)
    if c9 < best_c:
        best_c, best_h = c9, h9
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
