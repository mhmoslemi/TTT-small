def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert near-binary h to exactly binary with n/2 ones."""
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    # Find the threshold
    h_bin = (h >= 0.5).astype(float)
    # Adjust to get exactly target ones
    curr = int(h_bin.sum())
    if curr > target:
        # Need to turn off some
        order = np.argsort(h)[::-1]  # decreasing
        # Turn off the smallest above 0.5
        above = np.where(h_bin == 1)[0]
        above_sorted = above[np.argsort(h[above])]  # ascending by value
        for idx in above_sorted[curr - target]:
            h_bin[idx] = 0
    elif curr < target:
        # Need to turn on some
        below = np.where(h_bin == 0)[0]
        below_sorted = below[np.argsort(-h[below])]  # descending by value
        for idx in below_sorted[:target - curr]:
            h_bin[idx] = 1
    return h_bin

def binarize(h, n):
    """Binarize with exact constraint."""
    h_bin = to_binary(h, n)
    return h_bin

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def lm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 10:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(lm_obj, best_h, jac=lm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def compute_delta(row, h, n):
    """Compute delta_i(m) for a single position i, for all m.
    Returns array of shape (2n-1,) with values in {-dx, 0, dx} for binary h."""
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    i = row
    N = n
    for m in range(2 * N - 1):
        k = m - (N - 1)
        val = 0.0
        # Term 1: h[i] appears as h[n+k] when n = i-k
        # valid if 0 <= i-k <= N-1-k
        if 0 <= i - k and i - k <= N - 1 - k:
            # coefficient is (1 - h[i-k])
            val += (1.0 - h[i - k])
        # Term 2: h[i] appears as (1-h[n]) when n = i
        # valid if 0 <= i <= N-1-k
        if 0 <= i and i <= N - 1 - k:
            val += (-h[i + k])  # note: i+k must be < N for this to be valid
            # Actually, we need i+k < N, which is i <= N-1-k (same as the condition above)
        delta[m] = val * dx
    return delta

def compute_delta_fast(i, h, n):
    """Fast computation of delta_i for position i."""
    dx = 2.0 / n
    N = n
    delta = np.zeros(2 * N - 1)
    # For k >= 0 (m = N-1+k, k from 0 to N-1):
    # Term 1: n = i-k, valid if 0 <= i-k <= N-1-k → k <= i and k <= N-1-i... 
    #   Actually: 0 <= i-k means k <= i. And i-k <= N-1-k means i <= N-1. Always true.
    #   So term 1 valid for k in [max(0, i-(N-1-k)), min(i, N-1)]... 
    #   Let me just be: 0 <= i-k <= N-1-k
    #   i-k >= 0 → k <= i
    #   i-k <= N-1-k → i <= N-1 (always true)

    # Let j = -k > 0
    # c(-j) = sum_n h[n](1-h[n+j]) dx, n from 0 to N-1-j
    # h[i] appears as h[n] when n=i, valid if 0 <= i <= N-1-j → j <= N-1-i
    #   Contribution: (1-h[i+j])
