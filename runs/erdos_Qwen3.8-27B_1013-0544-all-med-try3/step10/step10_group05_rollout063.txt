```python
def gradient_directed_local_search(h, n, deadline, n_iters=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 500000
    mu = 0.001
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        grad = smooth_max_grad(c_vec, alpha, best_h, n) + 2*mu*(best_h.sum()-n/2)
        
        # Find the most "violated" interior point to decrease
        # and the most "violated" interior point to increase
        interior = (best_h > 1e-10) & (best_h < 1-1e-10)
        if not interior.any():
            break
        
        # Points where gradient is positive (want to decrease)
        to_decrease = interior & (grad > 0)
        to_increase = interior & (grad < 0)
        
        if not to_decrease.any() or not to_increase.any():
            break
        
        # Pick the most extreme pair
        i = np.argmax(np.where(to_decrease, grad, -np.inf))
        j = np.argmax(-np.where(to_increase, grad, np.inf))
        
        # Line search
        best_d = 0
        best_f = best_c
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            test_h = best_h.copy()
            test_h[i] -= d
            test_h[j] += d
            if test_h[i] < 0 or test_h[j] > 1:
                continue
            test_c = compute_c5(test_h, n)
            if test_c < best_f:
                best_f = test_c
                best_d = d
        
        if best_d > 0:
            best_h[i] -= best_d
            best_h[j] += best_d
            best_c = best_f
        else:
            break
    
    return best_h, best_c
```

∂C[k]/∂h[j] = -dx·h[j-k] for j=k,...,n-1
∂C[k]/∂h[j] = dx·((1-h[j+k]) - h[j-k]) for j=k,...,n-1-k (overlap region)

grad[j] = Σ_k w[k]·∂C[k]/∂h[j]

For the "first part" (1-h[j+k]):
Σ_k w[k]·(1-h[j+k])·dx = dx·(Σ_k w[k] - Σ_k w[k]·h[j+k])

The inner sum Σ_k w[k]·h[j+k] is a cross-correlation of w and h, which can be computed with FFT!

Similarly for the "second part":
Σ_k w[k]·h[j-k]·dx is another cross-correlation.

```python
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 40 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def run_lbgfs_round(h, n, deadline, alpha, mu, maxit=2000):
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = min(maxit, max(100, int(rem * 4)))
    try:
        res = minimize(full_obj, h, jac=full_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def run_active_round(h, n, deadline, top_k, alpha, mu=0.001):
    bounds = [(0.0, 1.0)] * n
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-7, 0.0005 * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    if len(active) == 0:
        return h, compute_c5(h, n)
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    maxit = min(3000, max(200, int(rem * 3)))
    try:
        res = minimize(active_obj, h, jac=active_grad,
                      args=(n, active, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 40, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h, compute_c5(h, n)

def gradient_local_search(h, n, deadline, alpha=2000000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    iters = 0
    no_improve_count = 0
    
    while time.time() < deadline - 5:
        c_vec = compute_c5_vec(best_h, n)
        m = c_vec.max()
        sh = alpha * (c_vec - m)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        
        grad = np.zeros(n)
        idx_center = n - 1
        for mm in range(len(w)):
            if w[mm] < 1e-15:
                continue
            k = mm - idx_center
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - best_h[k:n])
                grad[k:n] += wm * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - best_h[0:nv])
                grad[0:nv] += wm * (-best_h[kk:n])
        
        interior = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        n_int = interior.sum()
        if n_int < 2:
            break
        
        sorted_idx = np.argsort(-grad)
        improved = False
        
        # Phase A: Try top 10 decrease / bottom 40 increase pairs
        for i_rank in range(min(10, n_int)):
            i = sorted_idx[i_rank]
            if not interior[i]:
                continue
            for j_rank in range(n - 1, max(n - 41, -1), -1):
                j = sorted_idx[j_rank]
                if not interior[j] or j == i:
                    continue
                if grad[i] <= grad[j]:
                    break
                max_d = min(best_h[i], 1.0 - best_h[j])
                for frac in [1.0, 0.1, 0.01, 0.001, 0.0001]:
                    d = max_d * frac
                    if d < 1e-12:
                        break
                    best_h[i] -= d
                    best_h[j] += d
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                if improved:
                    break
            if improved:
                break
        
        # Phase B: Small random-ish steps guided by gradient
        if not improved:
            for i in range(min(50, n)):
                if time.time() > deadline - 10:
                    break
                if not interior[i] or grad[i] <= 0:
                    continue
                j_start = (i + n // 2) % n
                for jj in range(30):
                    j = (j_start + jj) % n
                    if j == i or not interior[j]:
                        continue
                    if grad[j] >= grad[i]:
                        continue
                    d = 0.005 * min(best_h[i], 1.0 - best_h[j])
                    if d < 1e-12:
                        continue
                    best_h[i] -= d
                    best_h[j] += d
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                if improved:
                    break
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 5:
                break
        else:
            no_improve_count = 0
        iters += 1
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            # Focus on points affecting active constraints
            active_thresh = max_val - 0.002 * max_val
            active = np.where(c_vec >= active_thresh)[0]
            idx_center = n - 1
            # Build set of "important" indices
            imp_set = set()
            for idx in active:
                k = int(idx) - idx_center
                lo = max(0, min(0, -k))
                hi = min(n, n - abs(k))
                for ii in range(lo, hi):
                    imp_set.add(ii)
                    imp_set.add(ii + abs(k))
            imp_list = sorted(imp_set)
            imp_list = [x for x in imp_list if 0 <= x < n]
            
            if len(imp_list) < 2:
                imp_list = list(range(n))
            
            for i in imp_list:
                if time.time() > deadline - 5:
                    break
                if best_h[i] < d or best_h[i] > 1.0 - d:
                    continue
                for j in imp_list:
                    if i == j:
                        continue
                    # Try decrease i, increase j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Try increase i, decrease j
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
        if time.time() > deadline - 5:
            break
    
    return best_h, best_c

def perturb_reopt(h, n, deadline, rng, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for t in range(n_trials):
        if time.time() > deadline - 20:
            break
        scale = 0.02 / (t + 1)
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        if c_pert < best_c + 0.003:
            rem = deadline - time.time() - 10
            if rem < 10:
                break
            per_trial = max(15, int(rem / max(1, n_trials - t)))
            try:
                res = minimize(full_obj, h_pert, jac=full_grad,
                              args=(n, 500000, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': min(per_trial, 800), 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': per_trial * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
            # Quick local polish
            h_pol, c_pol = gradient_local_search(best_h, n, deadline - 15, alpha=5000000)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        h0 = make_beatty(500, golden / 2.0)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Try a few alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alts = [
        make_beatty(n_orig, golden / 2.0),
        make_beatty(n_orig, 0.4312),
        make_beatty(n_orig, np.sqrt(2.0) / 2.0),
    ]
    for h_alt in alts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    
    # Phase 1: L-BFGS warmup (60s)
    p1_end = min(t0 + 60, deadline - 900)
    if p1_end > t0 + 15:
        for alpha in [500, 5000, 50000, 200000]:
            if time.time() > p1_end:
                break
            best_h, best_c = run_lbgfs_round(best_h, best_n, p1_end, alpha, 0.01, maxit=1000)
    
    # Phase 2: Active-set L-BFGS (250s)
    p2_end = min(time.time() + 250, deadline - 550)
    if p2_end > time.time() + 30:
        configs = [
            (4, 100000, 0.002),
            (6, 500000, 0.001),
            (8, 2000000, 0.001),
            (10, 5000000, 0.0005),
            (14, 10000000, 0.0005),
            (20, 30000000, 0.0003),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 20:
                break
            for _ in range(2):
                if time.time() > p2_end - 15:
                    break
                h2, c2 = run_active_round(best_h, best_n, p2_end - 5, top_k, alpha, mu)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
    
    # Phase 3: Try higher resolution n=1000 (200s)
    n_up = min(best_n * 2, 1000)
    p3_start = time.time()
    p3_end = min(p3_start + 200, deadline - 400)
    if p3_end > p3_start + 60 and best_n < 1000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.002:
            for alpha in [10000, 100000, 1000000, 5000000]:
                if time.time() > p3_end - 20:
                    break
                h_new, c_new = run_lbgfs_round(h_up, n_up, p3_end, alpha, 0.005, maxit=1500)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    best_n = n_up
                    h_up = h_new
            # Active set at higher res
            for top_k, alpha in [(8, 5000000), (14, 20000000), (20, 50000000)]:
                if time.time() > p3_end - 15:
                    break
                if best_n == n_up:
                    h3, c3 = run_active_round(best_h, best_n, p3_end - 5, top_k, alpha, 0.0005)
                    if c3 < best_c:
                        best_c = c3
                        best_h = h3
    
    # Phase 4: Gradient-directed local search (250s)
    p4_end = min(time.time() + 250, deadline - 280)
    if p4_end > time.time() + 30:
        cur_n = len(best_h)
        # Multiple passes with different alpha
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > p4_end - 10:
                break
            h4, c4 = gradient_local_search(best_h, cur_n, p4_end, alpha=alpha)
            if c4 < best_c:
                best_c = c4
                best_h = h4
    
    # Phase 5: Fine pair search (180s)
    p5_end = min(time.time() + 180, deadline - 130)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        h5, c5 = fine_pair_search(best_h, cur_n, p5_end,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: Perturbation + re-optimization (80s)
    p6_end = min(time.time() + 80, deadline - 60)
    if p6_end > time.time() + 20:
        cur_n = len(best_h)
        h6, c6 = perturb_reopt(best_h, cur_n, p6_end, rng, n_trials=6)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Ultra-fine final search (50s)
    p7_end = deadline - 3
    if p7_end > time.time() + 15:
        cur_n = len(best_h)
        # Gradient search with very high alpha
        h7, c7 = gradient_local_search(best_h, cur_n, min(time.time() + 25, p7_end), alpha=50000000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        # Tiny pair moves
        rem = p7_end - time.time()
        if rem > 10:
            h7b, c7b = fine_pair_search(best_h, cur_n, p7_end,
                deltas=[0.0001, 0.00005, 0.00002, 0.00001])
            if c7b < best_c:
                best_c = c7b
                best_h = h7b
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
```
