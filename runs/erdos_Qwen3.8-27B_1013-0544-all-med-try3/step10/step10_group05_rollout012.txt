    return best_h, best_c

def perturb_reopt(h, n, deadline, rng, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 40:
            break
        scale = 0.02 * (1 + 0.5 * t)
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.01:
            rem = deadline - time.time() - 10
            if rem < 30:
                break
            per_trial = rem / (n_trials - t)
            h_new, c_new = run_lbfgs(h_pert, n, time.time() + min(per_trial, 120),
                                     soft_obj, soft_grad, (n, 50000, 0.001), bounds, 800)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(256, 0.5)
        n_orig = 256

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Alternative starts
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alphas = [golden/2, np.sqrt(2)/2, 0.431, 0.382, 0.276, 0.618]
    for a in alphas:
        h_alt = make_beatty(n_orig, a)
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    for _ in range(3):
        h_alt = make_random_binary(n_orig, rng)
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS-B warmup with increasing alpha
    p1_end = min(t0 + 100, deadline - 700)
    bounds = [(0.0, 1.0)] * best_n
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > p1_end - 15:
            break
        best_h, best_c = run_lbfgs(best_h, best_n, p1_end - 5,
                                   soft_obj, soft_grad, (best_n, alpha, 0.005),
                                   bounds, 1500)

    # Phase 2: Active set optimization (multiple rounds)
    p2_end = min(time.time() + 250, deadline - 450)
    configs = [
        (3, 20000, 0.001),
        (5, 100000, 0.001),
        (8, 500000, 0.0005),
        (10, 1000000, 0.0005),
        (15, 5000000, 0.0003),
        (20, 10000000, 0.0003),
    ]
    for top_k, alpha, mu in configs:
        if time.time() > p2_end - 10:
            break
        for _ in range(2):
            if time.time() > p2_end - 10:
                break
            c_vec = compute_c5_vec(best_h, best_n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-6, 0.0005 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                continue
            rem = p2_end - time.time() - 5
            if rem < 15:
                break
            maxit = max(150, min(int(rem * 2), 2000))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                             args=(best_n, active, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': maxit, 'gtol': 1e-15,
                                      'maxfun': maxit * 8, 'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 3: Greedy coordinate search
    p3_end = min(time.time() + 150, deadline - 300)
    if p3_end > time.time() + 20:
        best_h, best_c = greedy_coord_search(best_h, best_n, p3_end)

    # Phase 4: Try higher resolution
    n_up = best_n * 2
    p4_end = min(time.time() + 250, deadline - 250)
    if p4_end > time.time() + 80 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.003:
            bounds_up = [(0.0, 1.0)] * n_up
            for alpha in [5000, 50000, 200000, 1000000, 5000000]:
                if time.time() > p4_end - 20:
                    break
                h_up, c_test = run_lbfgs(h_up, n_up, p4_end - 5,
                                         soft_obj, soft_grad, (n_up, alpha, 0.001),
                                         bounds_up, 1500)
                if c_test < best_c:
                    best_c = c_test
                    best_h = h_up
                    best_n = n_up
            # Active set at higher res
            for top_k, alpha in [(5, 200000), (10, 2000000), (15, 10000000)]:
                if time.time() > p4_end - 15:
                    break
                if best_n == n_up:
                    c_vec = compute_c5_vec(best_h, best_n)
                    max_val = c_vec.max()
                    threshold = max_val - max(1e-6, 0.001 * max_val)
                    active = np.where(c_vec >= threshold)[0]
                    if len(active) > top_k:
                        active = np.array(np.argsort(-c_vec[active])[:top_k])
                    rem = p4_end - time.time() - 5
                    if rem < 15:
                        break
                    maxit = max(150, min(int(rem * 1.5), 1200))
                    try:
                        res = minimize(act_obj, best_h, jac=act_grad,
                                     args=(best_n, active, alpha, 0.0005),
                                     method='L-BFGS-B', bounds=bounds_up,
                                     options={'maxiter': maxit, 'gtol': 1e-15,
                                              'maxfun': maxit * 8, 'maxls': 30, 'maxcor': 30})
                        h_new = project_feasible(res.x, best_n)
                        c_new = compute_c5(h_new, best_n)
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h_new
                    except Exception:
                        pass
            # Greedy search at higher res
            if time.time() < p4_end - 20:
                best_h, best_c = greedy_coord_search(best_h, best_n, p4_end - 3)

    # Phase 5: More active set at current resolution
    p5_end = min(time.time() + 120, deadline - 120)
    if p5_end > time.time() + 20:
        cur_n = len(best_h)
        bounds_cur = [(0.0, 1.0)] * cur_n
        for top_k, alpha, mu in [(4, 2000000, 0.0005), (8, 10000000, 0.0003),
                                  (12, 30000000, 0.0002), (15, 50000000, 0.0002)]:
            if time.time() > p5_end - 10:
                break
            c_vec = compute_c5_vec(best_h, cur_n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-7, 0.0003 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            rem = p5_end - time.time() - 5
            if rem < 10:
                break
            maxit = max(100, min(int(rem * 2), 1500))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                             args=(cur_n, active, alpha, mu), method='L-BFGS-B',
                             bounds=bounds_cur,
                             options={'maxiter': maxit, 'gtol': 1e-15,
                                      'maxfun': maxit * 8, 'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, cur_n)
                c_new = compute_c5(h_new, cur_n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 6: Perturbation + reoptimization
    p6_end = min(time.time() + 100, deadline - 80)
    if p6_end > time.time() + 20:
        cur_n = len(best_h)
        best_h, best_c = perturb_reopt(best_h, cur_n, p6_end, rng, n_trials=6)

    # Phase 7: Fine local search
    p7_end = min(time.time() + 80, deadline - 30)
    if p7_end > time.time() + 15:
        cur_n = len(best_h)
        best_h, best_c = fine_pair_local(best_h, cur_n, p7_end, radius=30,
                                         deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    # Phase 8: Final polish
    p8_end = deadline - 2
    if p8_end > time.time() + 15:
        cur_n = len(best_h)
        bounds_final = [(0.0, 1.0)] * cur_n
        for alpha in [5000000, 20000000, 50000000]:
            if time.time() > p8_end - 8:
                break
            best_h, best_c = run_lbfgs(best_h, cur_n, p8_end - 2,
                                       soft_obj, soft_grad, (cur_n, alpha, 0.0003),
                                       bounds_final, 800)
        # Final greedy
        if time.time() < p8_end - 5:
            best_h, best_c = greedy_coord_search(best_h, cur_n, p8_end - 1)

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return (best_h, best_c, final_n)
