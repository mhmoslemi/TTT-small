def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_sinusoidal(n, freq, amp, phase):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n)

def make_multi_freq(n, freqs, amps, phases):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2.0 * np.pi * f * x / 2.0 + p)
    return project_feasible(h, n)

def smooth_max_val(vals, alpha):
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        logsum = smx + math.log(np.sum(np.exp(sh - smx)))
        return float(m + logsum / alpha)
    logsum = math.log(np.sum(np.exp(sh)))
    return float(m + logsum / alpha)

def active_smooth_max_obj(x, n, active, alpha, mu):

    return val

def active_smooth_max_grad(x, n, active, alpha, mu):

    dx = 2.0 / n
