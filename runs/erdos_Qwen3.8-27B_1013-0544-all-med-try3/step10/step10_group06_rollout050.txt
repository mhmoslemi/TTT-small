Let me think about the local search more carefully. The most effective local search for this problem would be:

1. Compute all overlaps
2. Find the top K lags (say K=5)
3. For each top lag, identify the "witness" pairs that contribute most
4. Try to swap h values to reduce those contributions

A specific move: for lag m > 0, the overlap is ∑_{i=0}^{n-1-m} h[i+m]*(1-h[i]).
If h[i+m] = 1 and h[i] = 0, this pair contributes 1.
To reduce the overlap, we want to either make h[i+m] = 0 or h[i] = 1.
But changing h[i] affects other lags too.

A "swap" move: find i, j such that h[i] > h[j], and check if exchanging (partially) h[i] → h[j] reduces the max overlap.

The key insight for the local search: we should focus on the lags that are at or near the maximum, and find moves that reduce those specific lags without increasing others.

Let me code this up properly. I'll use a "top-k lag" approach where I compute the gradient only from the top k lags, which is faster and more focused.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def topk_weights(c, K):
    """Weights concentrated on top K lags with exponential falloff."""
    idx = np.argsort(c)[-K:]
    w = np.zeros(len(c))
    vals = c[idx]
    w_idx = np.exp(-(K - np.arange(K)) * 3.0)
    w[idx] = w_idx
    w = w / w.sum()
    return w

def compute_grad_from_weights(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - nm1
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag >= n:
                continue
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            if idx >= n:
                continue
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = compute_grad_from_weights(x, n, smooth_max_weights(c, alpha))
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 40,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def lbfgsb_topk(h, n, K, alpha, deadline, maxiter=2000, penalty_coeff=1e8):
    """L-BFGS-B focusing on top K lags only."""
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        w = topk_weights(c, K)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = compute_grad_from_weights(x, n, w)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 40,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def greedy_local_search(h, n, deadline, rng, K_top=8, max_passes=2000):
    """Aggressive greedy local search targeting top lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_passes = 0

    while time.time() < deadline and n_passes < max_passes:
        c_all = compute_all_lags(h, n)
        current_max = c_all.max()
        if current_max * (2.0 / n) < best_c - 1e-15:
            best_c = float(current_max * (2.0 / n))
            best_h = h.copy()

        top_idx = np.argsort(c_all)[-K_top:]
        improved = False
        n_passes += 1

        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            # Try swaps to reduce this lag's overlap
            n_pairs = n - m
            if n_pairs <= 0:
                continue
            # Sample candidate pairs
            n_samples = min(n_pairs, 200)
            candidates = rng.choice(n_pairs, size=n_samples, replace=False)
            candidates = np.sort(candidates)

            for i in candidates:
                if time.time() > deadline:
                    break
                j = i + m
                if lag > 0:
                    # overlap includes h[j]*(1-h[i])
                    contribution = h[j] * (1.0 - h[i])
                    if contribution < 1e-6:
                        continue
                    # Try to reduce: lower h[j] or raise h[i]
                    if h[j] > 0.01 and h[i] < 0.99:
                        d = min(0.02, h[j] * 0.3, (1.0 - h[i]) * 0.3)
                        h[j] -= d
                        h[i] += d
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                        else:
                            h[j] += d
                            h[i] -= d
                            h = project_feasible(h, n)
                else:
                    # lag < 0, idx = -lag = m
                    # overlap includes h[i]*(1-h[j]) where j = i + m
                    contribution = h[i] * (1.0 - h[j])
                    if contribution < 1e-6:
                        continue
                    if h[i] > 0.01 and h[j] < 0.99:
                        d = min(0.02, h[i] * 0.3, (1.0 - h[j]) * 0.3)
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c_test = compute_c5(h, n)
                        if c_test < best_c - 1e-15:
                            best_c = c_test
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)

        if not improved and n_passes % 50 == 0:
            # Perturbation to escape local minima
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            c_check = compute_c5(h, n)
            if c_check > best_c + 0.005:
                h = best_h.copy()

    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, alpha_sa=100000, t_init=0.006, t_min=0.00002, cool=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    it_since_improve = 0

    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.7:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.15, 0.005 + T * 10)
            if h[i] > h[j] + 1e-8:
                d = rng.uniform(0.003, max_step)
                d = min(d, h[i] * 0.5, (1.0 - h[j]) * 0.5)
            elif h[j] > h[i] + 1e-8:
                d = -rng.uniform(0.003, max_step)
                max_d = min(h[j] * 0.5, (1.0 - h[i]) * 0.5)
                d = max(d, -max_d)
            else:
                T *= cool
                continue
        else:
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-4:]
            k_star = top_k[rng.randint(4)]
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if lag > 0:
                if h[j] > h[i] + 1e-6:
                    d = rng.uniform(0.002, 0.05)
                    d = min(d, (h[j] - h[i]) * 0.4, 0.05)
                    h[j] -= d
                    h[i] += d
                elif h[i] > h[j] + 1e-6:
                    d = rng.uniform(0.002, 0.05)
                    d = min(d, (h[i] - h[j]) * 0.4, 0.05)
                    h[j] += d
                    h[i] -= d
                else:
                    T *= cool
                    continue
            else:
                if h[i] > h[j] + 1e-6:
                    d = rng.uniform(0.002, 0.05)
                    d = min(d, (h[i] - h[j]) * 0.4, 0.05)
                    h[i] -= d
                    h[j] += d
                elif h[j] > h[i] + 1e-6:
                    d = rng.uniform(0.002, 0.05)
                    d = min(d, (h[j] - h[i]) * 0.4, 0.05)
                    h[i] += d
                    h[j] -= d
                else:
                    T *= cool
                    continue

        if abs(d) < 1e-10:
            T *= cool
            continue

        h = project_feasible(h, n)
        c_new = compute_all_lags(h, n)
        e_new = smooth_max_value(c_new, alpha_sa)

        # Compute old energy (approximate by storing)
        c_old = compute_all_lags(best_h, n)
        e_old = smooth_max_value(c_old, alpha_sa)

        de = e_new - e_old
        accept = False
        if de < 0:
            accept = True
        elif T > t_min * 2:
            ratio = -de / T
            if ratio < 25:
                accept = rng.random() < np.exp(-ratio)

        if accept:
            max_new = c_new.max()
            if max_new < best_c - 1e-14:
                best_c = float(max_new)
                best_h = h.copy()
                it_since_improve = 0
            else:
                it_since_improve += 1
        else:
            h = best_h.copy()
            it_since_improve += 1

        T *= cool

        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.015, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.002)

    return best_h, best_c

def binary_search_optimize(h, n, deadline, rng):
    """Optimize a binary solution directly."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1

    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    h_cur = h_bin.copy()
    no_improve = 0

    while time.time() < deadline and no_improve < 3000:
        c_all = compute_all_lags(h_cur, n)
        top_idx = np.argsort(c_all)[-5:]
        k_star = top_idx[rng.randint(5)]
        lag = k_star - (n - 1)
        if lag == 0:
            no_improve += 1
            continue
        m = abs(lag)
        if m >= n:
            no_improve += 1
            continue

        # Find a pair to swap that reduces overlap
        n_pairs = n - m
        i = rng.randint(0, n_pairs)
        j = i + m

        if h_cur[i] == h_cur[j]:
            no_improve += 1
            continue

        # Try flipping: if h_cur[i]=0, h_cur[j]=1 and lag>0, swap reduces overlap
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c_test = compute_c5(h_cur, n)
        if c_test < best_c - 1e-14:
            best_c = c_test
            best_h = h_cur.copy()
            no_improve = 0
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            no_improve += 1

    return best_h, best_c
