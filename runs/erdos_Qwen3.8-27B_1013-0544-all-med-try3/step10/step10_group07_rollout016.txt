        rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            for jj in ones:
                d = compute_swap_delta(h, i, jj, n, m_range)
                nm = np.max(corr + d)
                if nm * dx < cur_c - 1e-14:
                    h[i] = 1.0
                    h[jj] = 0.0
                    corr += d
                    cur_c = nm * dx
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c

def targeted_greedy(h, n, deadline, max_sweeps=3000, seed=0):
    """Focus on reducing the worst lags."""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    corr = get_correlation(h).copy()
    cur_c = corr.max() * dx
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        worst_lags = np.where(corr > cur_c / dx - 0.5)[0]
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        rng.shuffle(ones)
        for jj in ones:
            if time.time() > deadline:
                break
            best_i = -1
            best_new_max = cur_c / dx
            for i in zeros:
                d = compute_swap_delta(h, i, jj, n, m_range)
                nc_arr = corr + d
                worst_red = 0
                if len(worst_lags) > 0:
                    worst_red = np.min(nc_arr[worst_lags])
                nm = np.max(nc_arr)
                score = nm * 1.0 + worst_red * 0.3
                if score < best_new_max * 1.0 + (cur_c / dx - 0.5) * 0.3 - 1e-14:
                    best_new_max = score
                    best_i = i
            if best_i >= 0:
                d = compute_swap_delta(h, best_i, jj, n, m_range)
                h[best_i] = 1.0
                h[jj] = 0.0
                corr += d
                cur_c = np.max(corr) * dx
                improved = True
                break
        if not improved:
            break
    return h, cur_c

def simulated_anneal(h, n, deadline, seed=42, T0=4.0, T1=0.03, n_steps=200000):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h.copy()
    corr = get_correlation(h).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = float(step) / float(n_steps)
        T = T0 * (T1 / T0) ** frac
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        jj = ones[rng.randint(len(ones))]
        d = compute_swap_delta(h, i, jj, n, m_range)
        corr_new = corr + d
        new_c = np.max(corr_new) * dx
        dc = new_c - cur_c
        if dc <= 0:
            h[i] = 1.0
            h[jj] = 0.0
            corr = corr_new
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            if T > 1e-6:
                accept_prob = np.exp(-dc / T)
                if rng.random() < accept_prob:
                    h[i] = 1.0
                    h[jj] = 0.0
                    corr = corr_new
                    cur_c = new_c
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
    return best_h, best_c

def multi_restart_sa(h_start, n, deadline, n_restarts=5, seed=0):
    best_c = np.inf
    best_h = h_start.copy()
    rng = np.random.RandomState(seed)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_cur = h_start.copy()
        if r > 0:
            n_flips = rng.randint(5, 20)
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            for _ in range(n_flips):
                oi = rng.randint(len(ones))
                zi = rng.randint(len(zeros))
                h_cur[ones[oi]], h_cur[zeros[zi]] = 0.0, 1.0
        T0_r = 3.0 + r * 1.5
        T1_r = 0.02 + 0.01 * r
        h_res, c_res = simulated_anneal(h_cur, n, deadline, seed=seed + 100 * r, T0=T0_r, T1=T1_r, n_steps=150000)
        if c_res < best_c:
            best_c = c_res
            best_h = h_res.copy()
    return best_h, best_c

def slsqp_optimize(h, n, alphas, deadline, maxiter_each=2000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        result = _slsqp_single(h_cur, n, alpha, maxiter_each, deadline)
        c = compute_c5(result, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = result.copy()
    return best_h, best_c

def _slsqp_single(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj_fn(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_eq(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_round(h, n):
    target = n // 2
    h_b = (h > 0.5).astype(float)
    s = h_b.sum()
    if s != target:
        vals = h[h_b == 0]
        if s < target:
            idx = np.argsort(vals)[::-1][:int(target - s)]
            h_b[idx] = 1.0
        else:
            vals1 = h[h_b == 1]
            idx = np.argsort(vals1)[:int(s - target)]
            h_b[idx] = 0.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick greedy to get a binary baseline (60s)
    p1 = min(t0 + 60, deadline - 200)
    h_bin = binary_round(h0, n)
    h_g, c_g = greedy_binary(h_bin, n, p1, max_sweps=3000, seed=seed)
    if c_g < best_c:
        best_c = c_g
        best_h = h_g.copy()

    # Phase 2: Simulated annealing with multiple restarts (300s)
    p2 = min(t0 + 360, deadline - 150)
    h_bin2 = binary_round(best_h, n)
    h_sa, c_sa = multi_restart_sa(h_bin2, n, p2, n_restarts=6, seed=seed + 10)
    if c_sa < best_c:
        best_c = c_sa
        best_h = h_sa.copy()

    # Phase 3: Greedy polish after SA
    p3 = min(t0 + 420, deadline - 100)
    h_bin3 = binary_round(best_h, n)
    h_g3, c_g3 = greedy_binary(h_bin3, n, p3, max_sweeps=5000, seed=seed + 50)
    if c_g3 < best_c:
        best_c = c_g3
        best_h = h_g3.copy()

    # Phase 4: SLSQP non-binary optimization (300s)
    p4 = min(t0 + 720, deadline - 80)
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    h_slq, c_slq = slsqp_optimize(best_h.copy(), n, alphas, p4, maxiter_each=1500)
    if c_slq < best_c:
        best_c = c_slq
        best_h = h_slq.copy()

    # Phase 5: More SA from SLSQP result (binary round)
    p5 = min(t0 + 950, deadline - 50)
    if time.time() < p5 - 30:
        h_bin5 = binary_round(best_h, n)
        h_sa2, c_sa2 = multi_restart_sa(h_bin5, n, p5, n_restarts=4, seed=seed + 200)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2.copy()
        # Final SLSQP peel
        if time.time() < deadline - 10:
            h_fin, c_fin = slsqp_optimize(best_h.copy() + 0.03 * rng.randn(n), n,
                                          [50000, 200000, 1000000], deadline, maxiter_each=2000)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin.copy()

    # Phase 6: Try different resolution
    p6 = deadline - 5
    if time.time() < p6 - 60:
        for n_alt in [300, 375]:
            if time.time() > p6 - 30:
                break
            if n_alt == n:
                continue
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, n_alt)
            h_alt = np.interp(xn, xo, best_h)
            h_alt = project_feasible(h_alt, n_alt)
            h_alt_b = binary_round(h_alt, n_alt)
            h_alt_g, c_alt_g = greedy_binary(h_alt_b, n_alt, p6 - 30, max_sweeps=2000, seed=seed + 300)
            h_alt_sl, c_alt_sl = slsqp_optimize(h_alt_g, n_alt, [5000, 50000, 500000], p6 - 10, maxiter_each=1000)
            c_alt = min(c_alt_g, c_alt_sl)
            if c_alt < best_c:
                best_c = c_alt
                best_h = (h_alt_g if c_alt_g < c_alt_sl else h_alt_sl).copy()
                n = n_alt

    # Final
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
