def compute_c5(h, n):
    dx = 2.0 / n
    c = fast_correlate(h, 1.0 - h)
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    c = fast_correlate(h, 1.0 - h)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def soft_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    return float(val)

    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:

    else:
        w = np.exp(s)

        # dc/dh[j] for j in [lag, n-1]: -h[j-lag] * dx
        g[0:n-lag] += (1.0 - h[lag:n]) * dx
        g[lag:n] += (-h[0:n-lag]) * dx
    else:
        m = -lag
        # c = dx * sum_{i=m}^{n-1} h[i] * (1-h[i-m])
        # = dx * sum_{j=0}^{n-1-m} h[j+m] * (1-h[j])
        # dc/dh[j] for j in [0, n-1-m]: (1-h[j+m])*... wait
        # c = dx * sum_{j=0}^{n-1-m} h[j+m] * (1-h[j])
        # dc/dh[j] (j in [0, n-1-m]): dx * (1-h[j+m])  [from the (1-h[j]) term] ... no
        # Let me redo: c = dx * sum_{i=-lag}^{n-1} h[i] * (1-h[i+lag]) where lag<0
        # Let m = -lag > 0: c = dx * sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
        # dc/dh[j]: 
        #   if j in [m, n-1]: h[j] appears as h[i+m] with i=j-m: contribution dx*(1-h[j-m])
        #   if j in [0, n-1-m]: h[j] appears as h[i] with i=j: contribution dx*(-h[j+m])
        g[0:n-m] += (-h[m:n]) * dx
        g[m:n] += (1.0 - h[0:n-m]) * dx
    return g

def soft_max_gradient(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = soft_max_weights(c, alpha)

        if w[k] < 1e-12:
            continue
        lag = k - (n-1)
        g += w[k] * gradient_c_k(h, n, lag)
    return g

x_{k+1} = P_C(x_k - α_k * g_k)

where g_k is a subgradient at x_k, α_k = α_0 / sqrt(k), and the output is the running average.

But for our problem, the subgradient of max_k c_k at h is ∇c_{k*}(h) where k* = argmax. This is O(n) to compute.

The step size needs to be chosen carefully. Since c_k is bounded by [0, 2*2/n * dx] = [0, px/n * 2/n]... let me think.

import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def fast_correlate(a, b):
    return fftconvolve(a, b[::-1])

def compute_c5(h, n):
    dx = 2.0 / n
    c = fast_correlate(h, 1.0 - h)
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return fast_correlate(h, 1.0 - h) * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def soft_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    return w / w.sum()

def gradient_c_lag(h, n, lag):
    """Gradient of the lag-specific overlap integral w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        return (1.0 - 2.0 * h) * dx
    elif lag > 0:
        m = lag
        g[0:n-m] += (1.0 - h[m:n]) * dx
        g[m:n] += (-h[0:n-m]) * dx
    else:
        m = -lag
        g[0:n-m] += (-h[m:n]) * dx
        g[m:n] += (1.0 - h[0:n-m]) * dx
    return g

    if c_all is None:
        c_all = compute_all_lags(h, n)
    w = soft_max_weights(c_all, alpha)
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        g += w[k] * gradient_c_lag(h, n, lag)
    return g

def lbfgsb(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x_c, n)
        obj = soft_max(c, alpha)
        diff = x_c.sum() - target_sum
        obj += penalty_coeff * diff * diff
        g = soft_max_grad(x_c, n, alpha, c)
        g += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj, g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return project_feasible(h.copy(), n)

import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def fast_corr(a, b):
    return fftconvolve(a, b[::-1])

def compute_c5(h, n):
    dx = 2.0 / n
    c = fast_corr(h, 1.0 - h)
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return fast_corr(h, 1.0 - h) * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def soft_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    return w / w.sum()

def gradient_c_lag(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        return (1.0 - 2.0 * h) * dx
    elif lag > 0:
        m = lag
        g[0:n-m] += (1.0 - h[m:n]) * dx
        g[m:n] += (-h[0:n-m]) * dx
    else:
        m = -lag
        g[0:n-m] += (-h[m:n]) * dx
        g[m:n] += (1.0 - h[0:n-m]) * dx
    return g

def soft_max_grad(h, n, alpha, c_all):
    w = soft_max_weights(c_all, alpha)
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        lag = k - (n-1)
        g += w[k] * gradient_c_lag(h, n, lag)
    return g

def lbfgsb_opt(h, n, alpha, deadline, maxiter=3000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x_c, n)
        obj = soft_max(c, alpha)
        diff = x_c.sum() - target_sum
        obj += penalty_coeff * diff * diff
        g = soft_max_grad(x_c, n, alpha, c)
        g += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def nesterov_subgrad(h_init, n, deadline, total_steps=200000, t0=4.0):
    x = h_init.copy()
    y = h_init.copy()
    h_best = h_init.copy()
    c_best = compute_c5(h_init, n)
    t = 1.0
    for k in range(total_steps):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(y, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n-1)
        g = gradient_c_lag(y, n, lag)
        alpha = t0 / (np.sqrt(k+1.0))
        x_new = project_feasible(y - alpha * g, n)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
        theta = (t - 1.0) / t_new
        y = project_feasible(x_new + theta * (x_new - x), n)
        x = x_new
        t = t_new
        if (k+1) % 5000 == 0:
            h_check = project_feasible((x + y) / 2.0, n)
            c_check = compute_c5(h_check, n)
            if c_check < c_best:
                c_best = c_check
                h_best = h_check.copy()
    c1 = compute_c5(x, n)
    if c1 < c_best:
        c_best = c1
        h_best = x.copy()
    c2 = compute_c5(y, n)
    if c2 < c_best:
        c_best = c2
        h_best = y.copy()
    return h_best, c_best

def targeted_ls(h_init, n, deadline, rng, K=10):
    h = h_init.copy()
    h_best = h.copy()
    c_best = compute_c5(h, n)
    for epoch in range(500):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        top_k = np.argsort(c_all)[-K:]
        improved = False
        for lag_idx in top_k:
            if time.time() > deadline:
                break
            lag = lag_idx - (n-1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            g = gradient_c_lag(h, n, lag)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                continue
            g_dir = -g / g_norm
            for s in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
                h_try = project_feasible(h + s * g_dir, n)
                c_try = compute_c5(h_try, n)
                if c_try < c_best - 1e-14:
                    h = h_try
                    c_best = c_try
                    h_best = h.copy()
                    improved = True
                    break
        if not improved:
            for _ in range(80):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i == j:
                    continue
                d = rng.uniform(-0.015, 0.015)
                h_try = h.copy()
                h_try[i] -= d
                h_try[j] += d
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < c_best - 1e-14:
                    h = h_try
                    c_best = c_try
                    h_best = h.copy()
                    improved = True
            if not improved:
                break
    return h_best, c_best

def steep_descent(h_init, n, deadline, max_steps=100000):
    h = h_init.copy()
    h_best = h.copy()
    c_best = compute_c5(h, n)
    step = 0.02
    stagnant = 0
    for it in range(max_steps):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        k_star = int(np.argmax(c_all))
        lag = k_star - (n-1)
        g = gradient_c_lag(h, n, lag)
        found = False
        for s in [step, step*0.5, step*0.25, step*0.1]:
            h_try = project_feasible(h - s * g, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_best - 1e-15:
                h = h_try
                c_best = c_try
                h_best = h.copy()
                stagnant = 0
                found = True
                break
        if not found:
            stagnant += 1
            step *= 0.8
            if step < 1e-7:
                step = 0.01
                stagnant = 0
            if stagnant > 300:
                break
    return h_best, c_best

def expand_res(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
