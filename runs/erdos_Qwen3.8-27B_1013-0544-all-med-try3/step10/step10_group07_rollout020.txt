from scipy.optimize import minimize
import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h_init, n):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def swap_delta(h, i, j, n, m_range):
    """Compute the change in correlation when h[i]: 0->1 and h[j]: 1->0."""
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def local_search(h, deadline, max_sweeps=500, seed=42, sweep_mode='first'):
    """Local search with single swaps. First-improvement or best-per-zero."""
    n = len(h)
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        z_list = list(zeros)
        rng.shuffle(z_list)
        
        if sweep_mode == 'first':
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = list(ones)
                rng.shuffle(o_list)
                for j in o_list:
                    delta = swap_delta(h, i, j, n, m_range)
                    new_corr = corr + delta
                    nc = float(new_corr.max()) * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
        elif sweep_mode == 'best_per_zero':
            # For each zero, find the best one to swap with
            best_swap = None
            best_nc = cur_c
            for i in z_list:
                if time.time() > deadline:
                    break
                for j in ones:
                    delta = swap_delta(h, i, j, n, m_range)
                    new_corr = corr + delta
                    nc = float(new_corr.max()) * dx
                    if nc < best_nc - 1e-15:
                        best_nc = nc
                        best_swap = (i, j, new_corr)
            if best_swap is not None:
                i, j, new_corr = best_swap
                h[i], h[j] = 1.0, 0.0
                corr = new_corr
                cur_c = best_nc
                ones.discard(j)
                ones.add(i)
                zeros.discard(i)
                zeros.add(j)
                improved = True
        if not improved:
            break
    return h, cur_c

def two_swap_search(h, deadline, max_iters=2000, seed=42):
    """Try random 2-for-2 swaps."""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    ones = set(np.where(h == 1.0)[0].tolist())

            break
        if len(zeros) < 2 or len(ones) < 2:
            break
        z_list = list(zeros)
        o_list = list(ones)
        i1, i2 = rng.choice(z_list, size=2, replace=False)
        j1, j2 = rng.choice(o_list, size=2, replace=False)
        
        new_h = h.copy()
        new_h[i1] = 1.0
        new_h[i2] = 1.0
        new_h[j1] = 0.0
        new_h[j2] = 0.0
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        nc = float(new_corr.max()) * dx

            h = new_h
            corr = new_corr
            cur_c = nc
            ones.discard(j1)
            ones.discard(j2)
            ones.add(i1)
            ones.add(i2)
            zeros.discard(i1)
            zeros.discard(i2)
            zeros.add(j1)
            zeros.add(j2)

def block_move_search(h, deadline, max_iters=1000, seed=42):
    """Try moving contiguous blocks of 1s."""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick a random contiguous block of 1s
        o_arr = np.sort(np.array(list(ones)))
        if len(o_arr) < 2:
            break
        # Find contiguous segments
        segs = []
        start = o_arr[0]
        prev = o_arr[0]
        for val in o_arr[1:]:
            if val == prev + 1:
                prev = val
            else:
                segs.append((start, prev))
                start = val
                prev = val
        segs.append((start, prev))
        
        s, e = rng.choice(segs)
        block_len = e - s + 1
        if block_len < 1:
            continue
        # Try shifting the block left or right
        direction = rng.choice([-1, 1])
        shift = direction * rng.randint(1, 4)
        new_s = s + shift
        new_e = e + shift
        if new_s < 0 or new_e >= n:
            continue
        # Check that the target positions are all 0
        target = set(range(new_s, new_e + 1))
        if target & ones:
            continue
        # Check that the source positions become 0 (they will be)
        
        new_h = h.copy()
        new_h[s:e+1] = 0.0
        new_h[new_s:new_e+1] = 1.0
        # Verify feasibility
        if new_h.sum() != n / 2.0:
            continue
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        nc = float(new_corr.max()) * dx
        if nc < cur_c - 1e-15:
            h = new_h
            corr = new_corr
            cur_c = nc
            ones = set(np.where(h == 1.0)[0].tolist())
            zeros = set(np.where(h == 0.0)[0].tolist())
    return h, cur_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
