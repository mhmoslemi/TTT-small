```python
def simulated_annealing_binary(h, n, deadline, T0=1.0, cooling=0.999):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_h = h.copy()
    current_c = best_c
    T = T0
    ones = set(np.where(current_h > 0.5)[0].tolist())
    zeros = set(np.where(current_h < 0.5)[0].tolist())
    
    while time.time() < deadline and T > 1e-6:
        # Propose a swap
        i = random.choice(list(ones))
        j = random.choice(list(zeros))
        
        # Apply swap
        current_h[i] = 0
        current_h[j] = 1
        new_c = compute_c5(current_h, n)
        
        # Accept or reject
        delta = new_c - current_c
        if delta < 0 or random.random() < np.exp(-delta / T):
            current_c = new_c
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            if new_c < best_c:
                best_c = new_c
                best_h = current_h.copy()
        else:
            current_h[i] = 1
            current_h[j] = 0
        
        T *= cooling
    
    return best_h, best_c
```

```python
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k=8, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [100000, 500000, 2000000, 10000000, 50000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.0005 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(100, int((deadline - time.time()) * 2))
            try:
                res = minimize(topK_obj, best_h, jac=topK_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def binary_swap_sa(h, n, deadline, T0=0.001, cooling=0.9995, max_iterations=500000):
    """Simulated annealing on binary swaps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    curr_c = best_c
    ones = set(np.where(curr_h > 0.5)[0].tolist())
    zeros = set(np.where(curr_h < 0.5)[0].tolist())
    T = T0
    rng = np.random.RandomState(12345)
    iterations = 0
    no_improve = 0
    
    while time.time() < deadline and T > 1e-8 and iterations < max_iterations:
        # Focus on positions near worst shifts
        c_vec = compute_c5_vec(curr_h, n)
        max_val = c_vec.max()
        worst_shifts = np.where(c_vec >= max_val - 0.01 * max_val)[0]
        
        # Propose swap: pick from ones near a worst shift contribution
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[list(ones)[rng.randint(len(ones))]]
        j = zeros[list(zeros)[rng.randint(len(zeros))]]
        
        curr_h[i] = 0.0
        curr_h[j] = 1.0
        new_c = compute_c5(curr_h, n)
        delta = new_c - curr_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c = new_c
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if new_c < best_c:
                best_c = new_c
                best_h = curr_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            curr_h[i] = 1.0
            curr_h[j] = 0.0
            no_improve += 1
        
        T *= cooling
        iterations += 1
        
        if no_improve > 50000:
            T = max(T * 2, 0.0005)
            no_improve = 0
    
    return best_h, best_c

def directed_swap_search(h, n, deadline, max_rounds=200):
    """Directed local search: find swaps that reduce the worst shifts."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones = set(np.where(best_h > 0.5)[0].tolist())
    zeros = set(np.where(best_h < 0.5)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        # Find positions contributing to worst shifts
        worst_k = np.argmax(c_vec)
        k_worst = worst_k - (n - 1)
        
        # Find candidates: i where best_h[i]=1 and best_h[(i+k_worst)%n]=0 (mod n not applicable for linear)
        improve_found = False
        ones_list = list(ones)
        zeros_list = list(zeros)
        
        # Try swaps that reduce the contribution at the worst shift
        if 0 <= k_worst < n:
            for idx in range(n - k_worst):
                i = idx
                j = idx + k_worst
                if best_h[i] > 0.5 and best_h[j] < 0.5:
                    # Swapping i and j would reduce contribution at shift k_worst
                    # But check overall effect
                    best_h[i] = 0.0
                    best_h[j] = 1.0
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        ones.discard(i)
                        ones.add(j)
                        zeros.discard(j)
                        zeros.add(i)
                        improve_found = True
                    else:
                        best_h[i] = 1.0
                        best_h[j] = 0.0
                elif best_h[i] < 0.5 and best_h[j] > 0.5 and (n-1-i+k_worst) < n:
                    # Try the reverse
                    best_h[i] = 1.0
                    best_h[j] = 0.0
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        onesdiscard_i = i
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improve_found = True
                    else:
                        best_h[i] = 0.0
                        best_h[j] = 1.0
        elif k_worst < 0:
            kk = -k_worst
            for idx in range(n - kk):
                i = idx
                j = idx + kk
                if best_h[j] > 0.5 and best_h[i] < 0.5:
                    best_h[j] = 0.0
                    best_h[i] = 1.0
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improve_found = True
                    else:
                        best_h[j] = 1.0
                        best_h[i] = 0.0
    
        if not improve_found:
            # Try random swaps as perturbation
            for _ in range(100):
                if time.time() > deadline:
                    break
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = list(ones)[np.random.randint(len(ones))]
                j = list(zeros)[np.random.randint(len(zeros))]
                best_h[i] = 0.0
                best_h[j] = 1.0
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                else:
                    best_h[i] = 1.0
                    best_h[j] = 0.0
    
    return best_h, best_c

def grad_descent_refine(h, n, deadline, step_sizes=None):
    """Gradient-guided pair exchange refinement."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    
    for step in step_sizes:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            
            pos_mask = (grad > 1e-10) & (best_h > 1e-10)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=30, deltas=None):
    """Fine-grained pair exchange search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                        if time.time() > deadline:
                            break
                if time.time() > deadline:
                    break
    return best_h, best_c

def make_beatty(n, alpha_param):
    """Generate Beatty sequence: positions where floor(k*alpha) mod n is hit."""
    h = np.zeros(n)
    positions = set()
    for k in range(1, 2 * n + 10):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n // 2:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_thue_morse(n):
    """Generate Thue-Morse based binary sequence."""
    # Find the largest power of 2 <= n
    # Use the Thue-Morse sequence to create a balanced binary sequence
    h = np.zeros(n)
    # Thue-Morse: t(i) = popcount(i) % 2
    ones = np.array([i for i in range(n) if bin(i).count('1') % 2 == 1])
    zeros = np.array([i for i in range(n) if bin(i).count('1') % 2 == 0])
    n_ones_target = n // 2
    if len(ones) >= n_ones_target:
        h[ones[:n_ones_target]] = 1.0
    else:
        h[ones] = 1.0
        need = n_ones_target - len(ones)
        if need > 0 and len(zeros) > 0:
            h[zeros[:need]] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def upsample(h, factor):
    """Upsample h by repeating each value 'factor' times."""
    n_new = len(h) * factor
    return np.repeat(h, factor)

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def slqp_con_fun(x, n):
    return x.sum() - n / 2.0

def slqp_con_jac(x, n):
    return np.ones(n)

def run_slqp(h, n, deadline, alpha=1000000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_con_fun, 'jac': slqp_con_jac, 'args': (n,)}]
    try:
        maxit = max(100, int((deadline - time.time()) * 5))
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': min(maxit, 2000), 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def multi_resolution_optimize(candidates, deadline):
    """Optimize multiple candidates and return the best."""
    results = []
    remaining_per = max(30, (deadline - time.time()) / max(1, len(candidates)))
    for name, h_init, n in candidates:
        if time.time() > deadline - 20:
            break
        phase_end = min(time.time() + remaining_per, deadline - 15)
        
        # Quick L-BFGS
        h1, c1 = run_lbfgs(h_init, n, phase_end * 0.4,
                          alphas=[50, 500, 5000, 50000], mu=0.01)
        # Active set
        if time.time() < phase_end * 0.7:
            h2, c2 = active_set_optimize(h1, n, phase_end * 0.7, top_k=6, rounds=10)
            if c2 < c1:
                h1, c1 = h2, c2
        # Gradient refinement
        if time.time() < phase_end * 0.9:
            h3, c3 = grad_descent_refine(h1, n, phase_end * 0.9,
                    step_sizes=[0.001, 0.0005, 0.0002, 0.0001])
            if c3 < c1:
                h1, c1 = h3, c3
        results.append((c1, h1, n, name))
    
    results.sort(key=lambda x: x[0])
    return results

        h5, c5 = directed_swap_search(best_h, best_n, min(time.time() + 80, phase5_end - 30),
                                      max_rounds=100)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        if time.time() < phase5_end - 10:
            h5b, c5b = grad_descent_refine(best_h, best_n, phase5_end - 5,
                    step_sizes=[0.0005, 0.0002, 0.0001, 0.00005, 0.00001])
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
    
    # Phase 6: Fine pair search
    phase6_end = t0 + 1000
    if time.time() < phase6_end:
        print("Phase 6: Fine pair search, current: %.6f" % best_c)
        h6, c6 = fine_pair_search(best_h, best_n, min(time.time() + 70, phase6_end - 10),
                                  radius=25, deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00001])
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: SLSQP polish
    if time.time() < deadline - 20:
        print("Phase 7: SLSQP, current: %.6f" % best_c)
        for alpha in [5000000, 50000000]:
            if time.time() > deadline - 15:
                break
            h7, c7 = run_slqp(best_h, best_n, deadline - 5, alpha=alpha)
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 8: Final micro-adjustments
    if time.time() < deadline - 3:
        h8, c8 = fine_pair_search(best_h, best_n, deadline - 1, radius=15,
                                  deltas=[0.00005, 0.00002, 0.00001, 0.000005])
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final projection and evaluation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    print("Final: C5 = %.8f (n=%d)" % (best_c, final_n))
    return best_h, best_c, final_n
```
