def project_box_sum(x,n):
    x=np.asarray(x,dtype=float).copy()
    target=n/2.0
    # Euclidean projection via bisection
    lo=float(np.min(x)-1.0)
    hi=float(np.max(x)+1.0)
    # ensure sum(lo) <= target <= sum(hi)
    # sum clip is monotone. 100 iters
    for _ in range(100):
        mid=(lo+hi)/2
        if np.sum(np.clip(x+mid,0.0,1.0)) < target:
             lo=mid
        else: hi=mid
    val=np.clip(x+(lo+hi)/2,0.0,1.0)
    # exact adjust
    val=exact_sum(val,n)
    return val

def force_exact_sum(h,n):
    h=np.asarray(h,dtype=float)
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for _ in range(10):
       diff=target - float(h.sum())
       if abs(diff) < 1e-14: break
       if diff > 0:
          order=np.argsort(1.0-h)[::-1]
          add=diff
          for idx in order:
             if add <= 1e-15: break
             room=1.0 - h[idx]
             if room <= 1e-15: continue
             step=add if add<room else room
             h[idx]+=step; add-=step
       else:
          order=np.argsort(h)[::-1]
          sub=-diff
          for idx in order:
             if sub <= 1e-15: break
             room=h[idx]
             if room <=1e-15: continue
             step=sub if sub<room else room
             h[idx]-=step; sub-=step
    # final tiny adjust one
    diff=target - float(h.sum())
    if abs(diff) > 1e-12:
       if diff >0:

          add=min(diff,1.0-h[idx])
          h[idx]+=add

          idx=int(np.argmax(h))

    return h

                add -= step
        else:
            order = np.argsort(h)[::-1]
            sub = float(-diff)
            for idx in order:
                if sub <= 1e-15:
                    break
                room = h[idx]
                if room <= 1e-15:
                    continue
                step = sub if sub < room else room
                h[idx] -= step
                sub -= step
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            step = min(diff, 1.0 - h[idx])
            h[idx] += step
        else:
            idx = int(np.argmax(h))
            step = min(-diff, h[idx])
            h[idx] -= step
    return h


def project_box_sum(x, n):
    x = np.array(x, dtype=float, copy=True)
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    if n <= 0:
        return x
    if n == 1:
        x[0] = target
        return x
    lo = float(np.min(x) - 1.0)
    hi = float(np.max(x) + 1.0)
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    x = force_exact_sum(x, n)
    return x


def compute_f(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    L = 2 * n - 1
    off = n - 1
    f = np.zeros(L, dtype=float)
    om = 1.0 - h
    f[off] = float(np.dot(h, om))
    for t in range(1, n):
        f[off + t] = float(np.dot(h[:n - t], om[t:]))
        f[off - t] = float(np.dot(h[t:], om[:n - t]))
    return f


def compute_c5(h, n):
    f = compute_f(h, n)
    return float(np.max(f) * (2.0 / n))


def compute_f_G(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    L = 2 * n - 1
    off = n - 1
    f = np.zeros(L, dtype=float)
    G = np.zeros((n, L), dtype=float)
    om = 1.0 - h
    f[off] = float(np.dot(h, om))
    G[:, off] = 1.0 - 2.0 * h
    for t in range(1, n):
        left = h[:n - t]
        right_h = h[t:]
        right_om = om[t:]
        left_om = om[:n - t]

        f[off + t] = float(np.dot(left, right_om))
        G[:n - t, off + t] += right_om
        G[t:, off + t] += -left

        f[off - t] = float(np.dot(right_h, left_om))
        G[t:, off - t] += left_om
        G[:n - t, off - t] += -right_h
    return f, G


def _sl_con_fun(x):
    return float(np.sum(x) - 0.5 * _SL_N)


def _sl_con_jac(x):
    return np.ones(_SL_N, dtype=float)


def _sl_obj(x):
    f, G = compute_f_G(x, _SL_N)
    maxf = float(f.max())
    sh = _SL_ALPHA * (f - maxf)
    sm = float(sh.max())
    e = np.exp(sh - sm)
    logsum = float(sm + np.log(e.sum()))
    return float(maxf + logsum / _SL_ALPHA)


def _sl_jac(x):
    f, G = compute_f_G(x, _SL_N)
    maxf = float(f.max())
    sh = _SL_ALPHA * (f - maxf)
    sm = float(sh.max())
    e = np.exp(sh - sm)
    p = e / e.sum()
    grad = G.T @ p
    return grad / _SL_ALPHA


def slake_polish(h, n, deadline, maxiter=40, alphas=(20.0, 60.0, 130.0)):
    global _SL_N, _SL_ALPHA
    n = int(n)
    h = project_box_sum(np.array(h, dtype=float, copy=True), n)
    best = h.copy()
    best_cnt = float(np.max(compute_f(h, n)))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sl_con_fun, 'jac': _sl_con_jac, 'bounds': (0.0, 0.0)}]

    for a in alphas:
        if time.time() > float(deadline) - 10.0:
            break
        _SL_N = n
        _SL_ALPHA = float(a)
        try:
            res = minimize(
                _sl_obj,
                best,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': int(maxiter), 'ftol': 1e-12}
            )
            x = getattr(res, 'x', None)
            if x is not None and np.all(np.isfinite(x)):
                hp = project_box_sum(x, n)
                cnt = float(np.max(compute_f(hp, n)))
                if cnt < best_cnt - 1e-12:
                    best_cnt = cnt
                    best = hp
        except Exception:
            pass
    return best, best_cnt * (2.0 / n)


def pair_value(theta, Fm, Am, F0, A, id0, id1, id2):
    if Fm.size:
        base = float(np.max(Fm + Am * theta))
    else:
        base = -1e30
    t2 = theta * theta
    v0 = F0[id0] + A[id0] * theta - 2.0 * t2
    v1 = F0[id1] + A[id1] * theta + t2
    v2 = F0[id2] + A[id2] * theta + t2
    m = base
    if v0 > m:
        m = v0
    if v1 > m:
        m = v1
    if v2 > m:
        m = v2
    return float(m)


def golden_pair(a, b, Fm, Am, F0, A, id0, id1, id2, iters=24):
    if b - a < 1e-14:
        return pair_value(a, Fm, Am, F0, A, id0, id1, id2), float(a)
    gr = 0.61803398874989484
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = pair_value(c, Fm, Am, F0, A, id0, id1, id2)
    fd = pair_value(d, Fm, Am, F0, A, id0, id1, id2)
    for _ in range(int(iters)):
        if fc <= fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = pair_value(c, Fm, Am, F0, A, id0, id1, id2)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = pair_value(d, Fm, Am, F0, A, id0, id1, id2)
    t = 0.5 * (a + b)
    return pair_value(t, Fm, Am, F0, A, id0, id1, id2), float(t)


def minimize_pair(F0, A, lo, hi, id0, id1, id2):
    if hi <= lo:
        th = float(lo)
        return pair_value(th, np.zeros(0), np.zeros(0), F0, A, id0, id1, id2), th

    mask = np.ones(F0.shape, dtype=bool)
    mask[id0] = False
    mask[id1] = False
    mask[id2] = False
    Fm = F0[mask]
    Am = A[mask]

    gnum = 65
    grid = np.linspace(lo, hi, gnum + 1)
    if Fm.size:
        base_grid = np.max(Fm[None, :] + Am[None, :] * grid[:, None], axis=1)
    else:
        base_grid = np.full(grid.shape, -1e30, dtype=float)
    t2 = grid * grid
    v0 = F0[id0] + A[id0] * grid - 2.0 * t2
    v1 = F0[id1] + A[id1] * grid + t2
    v2 = F0[id2] + A[id2] * grid + t2
    Fgrid = np.maximum(base_grid, v0)
    Fgrid = np.maximum(Fgrid, v1)
    Fgrid = np.maximum(Fgrid, v2)

    k = int(np.argmin(Fgrid))
    best_val = float(Fgrid[k])
    best_theta = float(grid[k])

    # local polish around two best grid neighborhoods
    order = np.argsort(Fgrid, kind='stable')[:6]
    neighborhoods = []
    last = -1e30
    for idx in order:
        idx = int(idx)
        aa = grid[max(0, idx - 2)]
        bb = grid[min(gnum, idx + 2)]
        if last < aa - 0.05 * (hi - lo):
            neighborhoods.append((float(aa), float(bb)))
        last = float(bb)
        if len(neighborhoods) >= 2:
            break

    if not neighborhoods:
        neighborhoods = [(float(lo), float(hi))]

    for aa, bb in neighborhoods:
        if bb - aa > 1e-12:
            v, t = golden_pair(aa, bb, Fm, Am, F0, A, id0, id1, id2, 24)
            if v < best_val:
                best_val = v
                best_theta = t

    for t in (lo, hi):
        v = pair_value(float(t), Fm, Am, F0, A, id0, id1, id2)
        if v < best_val:
            best_val = v
            best_theta = float(t)

    return best_val, best_theta


def coordinate_search(h, n, deadline, seed=1):
    n = int(n)
    h = project_box_sum(np.array(h, dtype=float, copy=True), n)
    rng = np.random.RandomState(int(seed))
    f = compute_f(h, n)
    best_cnt = float(f.max())
    best_h = h.copy()
    alpha_smooth = 45.0
    stuck = 0

    while time.time() < float(deadline) - 6.0:
        f, G = compute_f_G(h, n)
        fmax = float(f.max())
        if fmax < best_cnt - 1e-12:
            best_cnt = fmax
            best_h = h.copy()
        if not np.isfinite(fmax):
            break

        L = len(f)
        tol = 1e-5 * max(1.0, fmax)
        active = np.where(f >= fmax - tol)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(f))])
        weights = np.zeros(L, dtype=float)
        if active.size <= 60:
            weights[active] = 1.0 / float(active.size)
        else:
            sh = alpha_smooth * (f - fmax)
            p = np.exp(sh - float(sh.max()))
            weights = p / float(p.sum())

        grad = G.T @ weights
        g = grad
        score = np.abs(g[:, None] - g[None, :])
        room = (
            np.minimum(1.0 - h[:, None], h[None, :]) +
            np.minimum(h[:, None], 1.0 - h[None, :])
        )
        pot = score * room
        np.fill_diagonal(pot, -1.0)
        pot[pot < 1e-12] = -1.0

        if n <= 260:
            K = 60
        else:
            K = 42
        flat_pot = pot.ravel()
        K = int(max(0, min(K, flat_pot.size)))
        if K > 0:
            if flat_pot.size <= K:
                part = np.argsort(-flat_pot, kind='stable')
            else:
                part = np.argpartition(-flat_pot, K - 1)[:K]
        else:
            part = np.array([], dtype=int)

        improved = False
        for code in part:
            if time.time() > float(deadline) - 8.0:
                break
            code = int(code)
            i = code // n
            j = code % n
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-11:
                continue
            d = j - i
            id1 = d + n - 1
            id2 = -d + n - 1
            id0 = n - 1
            A = G[i] - G[j]
            newv, th = minimize_pair(f, A, float(lo), float(hi), id0, id1, id2)
            if newv < fmax - 1e-10:
                h[i] = float(np.clip(h[i] + th, 0.0, 1.0))
                h[j] = float(np.clip(h[j] - th, 0.0, 1.0))
                if abs(float(h.sum()) - 0.5 * n) > 1e-10:
                    h = project_box_sum(h, n)
                improved = True
                break

        if not improved:
            for _ in range(120):
                if time.time() > float(deadline) - 8.0:
                    break
                i = int(rng.randint(0, n))
                j = int(rng.randint(0, n))
                if i == j:
                    continue
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-11:
                    continue
                d = j - i
                id1 = d + n - 1
                id2 = -d + n - 1
                id0 = n - 1
                A = G[i] - G[j]
                newv, th = minimize_pair(f, A, float(lo), float(hi), id0, id1, id2)
                if newv < fmax - 1e-10:
                    h[i] = float(np.clip(h[i] + th, 0.0, 1.0))
                    h[j] = float(np.clip(h[j] - th, 0.0, 1.0))
                    if abs(float(h.sum()) - 0.5 * n) > 1e-10:
                        h = project_box_sum(h, n)
                    improved = True
                    break

        if improved:
            stuck = 0
            continue

        stuck += 1
        alpha_smooth = min(500.0, alpha_smooth * 1.6)
        if stuck >= 8:
            if time.time() > float(deadline) - 30.0:
                break
            h = best_h + 0.03 * rng.randn(n)
            h = project_box_sum(h, n)
            stuck = 0
        elif stuck >= 3:
            if time.time() > float(deadline) - 30.0:
                break
            idx = rng.choice(n, size=2, replace=False)
            ii, jj = int(idx[0]), int(idx[1])
            th = 0.05 * float(rng.uniform(-1.0, 1.0))
            h[ii] += th
            h[jj] -= th
            h = project_box_sum(h, n)
            stuck = 0

    f = compute_f(best_h, n)
    best_cnt = float(f.max())
    return best_h, best_cnt * (2.0 / n)


def binary_compute_f(b, n):
    b = np.asarray(b, dtype=np.int32)
    n = int(n)
    L = 2 * n - 1
    off = n - 1
    f = np.zeros(L, dtype=np.int32)
    oh = 1 - b
    for t in range(1, n):
        f[off + t] = int(np.dot(b[:n - t], oh[t:]))
        f[off - t] = int(np.dot(b[t:], oh[:n - t]))
    return f


def binary_prepare(b, n):
    b = np.asarray(b, dtype=np.int32)
    n = int(n)
    L = 2 * n - 1
    off = n - 1
    f = binary_compute_f(b, n)
    Bpos = np.zeros((n, L), dtype=np.int32)
    Cpos = np.zeros((n, L), dtype=np.int32)
    rows = np.arange(n, dtype=np.int64)
    ones = np.nonzero(b)[0]
    zeros = np.nonzero(1 - b)[0]

    for z in zeros:
        cols = z - rows + off
        Bpos[rows, cols] -= 1
        Cpos[rows, cols] += 1
    for o in ones:
        cols = rows - o + off
        Bpos[rows, cols] += 1
        Cpos[rows, cols] -= 1
    return f, Bpos, Cpos


def binary_sweep_once(b, n):
    b = np.asarray(b, dtype=np.int32)
    n = int(n)
    f, Bpos, Cpos = binary_prepare(b, n)
    off = n - 1
    cur = int(np.max(f))

    ones = np.nonzero(b)[0]
    zeros = np.nonzero(1 - b)[0]
    if ones.size == 0 or zeros.size == 0:
        return False, b, cur

    B = Bpos[ones]
    C = Cpos[zeros]
    fbase = f.astype(np.int32)
    fbase[off] = -200000

    if n <= 250:
        chunk = 80
    elif n <= 350:
        chunk = 56
    else:
        chunk = 36

    best_count = cur
    best_i = -1
    best_j = -1
    zlen = int(zeros.size)

    for s in range(0, zlen, chunk):
        zc = zeros[s:s + chunk]
        Cc = C[s:s + chunk]
        base = fbase[None, None, :] + B[:, None, :] + Cc.T[None, :, :]
        base_max = base.max(axis=2)

        d_mat = zc[None, :] - ones[:, None]
        d_idx = d_mat + off
        mind_idx = -d_mat + off

        Bd = np.take(B, d_idx, axis=1)
        Cd = np.take(Cc, d_idx.T, axis=1).T
        fd = f[d_idx]
        based = fd + Bd + Cd

        Bm = np.take(B, mind_idx, axis=1)
        Cm = np.take(Cc, mind_idx.T, axis=1).T
        fm = f[mind_idx]
        basem = fm + Bm + Cm

        special = np.maximum(based + 1, basem + 1)
        exact = np.maximum(base_max, special)
        np.maximum(exact, 0, out=exact)

        min_val = int(np.min(exact))
        if min_val < best_count:
            flat = int(np.argmin(exact))
            best_count = min_val
            ii = flat // len(zc)
            jj = flat % len(zc)
            best_i = int(ones[ii])
            best_j = int(zc[jj])

    if best_i >= 0 and best_count < cur:
        nb = b.copy()
        nb[best_i] = 0
        nb[best_j] = 1
        return True, nb, best_count
    return False, b, cur


def to_binary(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        old = np.linspace(0.0, 2.0, len(h), endpoint=False)
        new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(new, old, h)
    N = n // 2
    order = np.argsort(h, kind='stable')[::-1]
    b = np.zeros(n, dtype=np.int8)
    b[order[:N]] = 1
    return b


def random_binary(n, rs):
    n = int(n)
    N = n // 2
    b = np.zeros(n, dtype=np.int8)
    idx = rs.choice(n, N, replace=False)
    b[idx] = 1
    return b


def random_binary_sym(n, rs):
    n = int(n)
    if n % 4 != 0:
        return random_binary(n, rs)
    m = n // 2
    half = np.zeros(m, dtype=np.int8)
    idx = rs.choice(m, m // 2, replace=False)
    half[idx] = 1
    b = np.zeros(n, dtype=np.int8)
    b[:m] = half
    b[m:] = half
    return b


def binary_perturb(b, rs, k):
    b = np.array(b, dtype=np.int8, copy=True)
    k = int(k)
    for _ in range(k):
        ones = np.nonzero(b)[0]
        zeros = np.nonzero(1 - b)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        a = int(rs.choice(ones))
        z = int(rs.choice(zeros))
        b[a] = 0
        b[z] = 1
    return b


def binary_search(start_h, n, deadline, seed=0, max_restarts=5):
    n = int(n)
    if n < 4 or (n % 2 != 0):
        return np.zeros(n, dtype=float), np.inf
    rs = np.random.RandomState(int(seed))
    b = to_binary(start_h, n)
    cur = int(binary_compute_f(b, n).max())
    best_c = cur * (2.0 / n)
    best_b = b.copy()
    used_sym = False

    for rest in range(int(max_restarts)):
        if time.time() > float(deadline) - 10.0:
            break
        if rest == 0:
            cur_b = to_binary(start_h, n)
        elif (not used_sym) and (n % 4 == 0) and (time.time() < float(deadline) - 60.0):
            cur_b = random_binary_sym(n, rs)
            used_sym = True
        elif rest == 1:
            cur_b = binary_perturb(best_b, rs, 3)
        elif rest == 2:
            cur_b = random_binary(n, rs)
        else:
            cur_b = binary_perturb(best_b, rs, rs.randint(3, 12))

        sweeps = 0
        while time.time() < float(deadline) - 10.0 and sweeps < 4000:
            improved, cur_b, cnt = binary_sweep_once(cur_b, n)
            sweeps += 1
            if not improved:
                break
            valid_c = compute_c5(cur_b.astype(float), n)
            if valid_c < best_c - 1e-12:
                best_c = valid_c
                best_b = cur_b.copy()
    return best_b.astype(float), best_c


def resample_linear(h, n, nz):
    h = np.asarray(h, dtype=float)
    if len(h) == nz:
        return h.copy()
    old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new = np.linspace(0.0, 2.0, nz, endpoint=False)
    return np.interp(new, old, h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + total - 8.0
    rs_seed = int(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is None or len(parent) < 10:
        n0 = 250
        h0 = np.full(n0, 0.5)
    else:
        n0 = int(len(parent))
        h0 = parent.copy()

    h0 = project_box_sum(h0, n0)
    best_h = h0
    best_n = n0
    best_c = compute_c5(h0, n0)

    # Short smooth polish from the parent.
    rem = deadline - time.time()
    if rem > 90.0:
        ns = best_n if (best_n % 2 == 0 and 50 <= best_n <= 320) else 250
        hs = resample_linear(best_h, best_n, ns) if ns != best_n else best_h.copy()
        d1 = t0 + min(60.0, total * 0.08)
        hs, cs = slake_polish(hs, ns, d1, maxiter=35, alphas=(20.0, 70.0))
        cs = compute_c5(hs, ns)
        if cs < best_c - 1e-12:
            best_c = cs
            best_h = hs.copy()
            best_n = ns

    # Continuous pairwise coordinate search at the current resolution.
    rem = deadline - time.time()
    if rem > 80.0:
        alloc = float(min(rem * 0.38, 300.0))
        hc, cc = coordinate_search(best_h, best_n, t0 + alloc, seed=rs_seed + 1)
        cc = compute_c5(hc, best_n)
        if cc < best_c - 1e-12:
            best_c = cc
            best_h = hc.copy()
            best_n = best_n

    # Binary local search at a stable even resolution.
    rem = deadline - time.time()
    if rem > 80.0:
        nb = best_n
        if (nb % 2 != 0) or nb < 8 or nb > 400:
            nb = 250
        alloc = float(min(rem * 0.38, 280.0))
        hb, cb = binary_search(best_h, nb, t0 + alloc, seed=rs_seed + 2, max_restarts=4)
        if cb < best_c - 1e-12:
            best_c = cb
            best_h = hb
            best_n = nb

    # Multi-resolution attempts.
    tried = set()
    if best_n not in (256, 300, 320):
        tried.add(best_n)
    cand_ns = []
    for nh in (256, 300, 320):
        if nh not in tried:
            cand_ns.append(nh)

    for nh in cand_ns:
        rem = deadline - time.time()
        if rem < 110.0:
            break
        try:
            nh = int(nh)
        except Exception:
            continue
        if nh < 80 or nh > 400 or (nh % 2 != 0):
            continue
        hr = resample_linear(best_h, best_n, nh)
        hr = project_box_sum(hr, nh)

        alloc_c = float(min(rem * 0.38, 160.0))
        hc2, cc2 = coordinate_search(hr, nh, t0 + alloc_c, seed=rs_seed + 50 + nh)
        cc2 = compute_c5(hc2, nh)
        start2 = hc2
        if cc2 < best_c - 1e-12:
            best_c = cc2
            best_h = hc2.copy()
            best_n = nh

        rem = deadline - time.time()
        if rem < 70.0:
            continue
        alloc_b = float(min(rem * 0.55, 180.0))
        hb2, cb2 = binary_search(start2, nh, t0 + alloc_b, seed=rs_seed + 100 + nh, max_restarts=3)
        if cb2 < best_c - 1e-12:
            best_c = cb2
            best_h = hb2
            best_n = nh

    # Final continuous polish.
    rem = deadline - time.time()
    if rem > 70.0:
        nn = best_n
        hh = best_h.copy()
        if (nn % 2 != 0) or nn > 400:
            nn = 250
            hh = resample_linear(best_h, best_n, nn)
        alloc = float(min(rem * 0.55, 170.0))
        hf, cf = coordinate_search(project_box_sum(hh, nn), nn, t0 + alloc, seed=rs_seed + 200)
        cf = compute_c5(hf, nn)
        if cf < best_c - 1e-12:
            best_c = cf
            best_h = hf
            best_n = nn

    # Final binary polish, if time remains.
    rem = deadline - time.time()
    if rem > 55.0:
        nb = best_n
        if (nb % 2 != 0) or nb < 8 or nb > 400:
            nb = 250
        alloc = float(min(rem * 0.6, 130.0))
        hb, cb = binary_search(best_h, nb, t0 + alloc, seed=rs_seed + 300, max_restarts=3)
        if cb < best_c - 1e-12:
            best_c = cb
            best_h = hb
            best_n = nb

    # Final sanitation and exact evaluation.
    best_h = np.asarray(best_h, dtype=float)
    best_n = int(best_n)
    rounded = np.round(best_h)
    if np.max(np.abs(best_h - rounded)) < 1e-6:
        best_h = rounded.copy()
        N = best_n // 2
        s = int(np.sum(best_h))
        if s > N:
            ones = np.where(best_h == 1)[0]
            for idx in ones[:s - N]:
                best_h[idx] = 0
        elif s < N:
            zeros = np.where(best_h == 0)[0]
            need = N - s
            for idx in zeros[:need]:
                best_h[idx] = 1
        best_h = best_h.astype(float)
    else:
        best_h = project_box_sum(best_h, best_n)

    best_c = compute_c5(best_h, best_n)
    if not np.isfinite(best_c) or best_c > 1.0:
        best_h = project_box_sum(h0, n0)
        best_n = n0
        best_c = compute_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
