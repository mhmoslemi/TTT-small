def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        diff = x.sum() - target_sum
        obj_val += 1e8 * diff * diff
        
        # Gradient of smooth max
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)

            if w[k] < 1e-12:

            lag = k - (n - 1)

                grad[idx:n] += w[k] * (-x[0:n - idx])

        grad += 2e8 * diff * np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_smooth_max_grad_lags(c, alpha, n):
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return *dx, w

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e8 * diff * diff

        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
        else:
            w = np.exp(shifted)
        w = w / w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[lag:n] += w[k] * (1.0 - x[0:n - lag])
                grad[0:n - lag] += w[k] * (-x[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w[k] * (1.0 - x[idx:n])
                grad[idx:n] += w[k] * (-x[0:n - idx])
        grad *= dx
        grad += 2e8 * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)

def sweep_local_search(h, n, deadline, rng):
    """Systematic sweep over pairs at top lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    top_k = 5
    steps = [0.005, 0.003, 0.002, 0.001, 0.0005]
    rounds = 0

    while time.time() < deadline:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-top_k:]
        best_c = compute_c5(h, n)
        improved = False
        rounds += 1

        for k_star in top_idx:
            if time.time() > deadline:
                return best_h, best_c
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue

            if lag > 0:
                diffs = h[0:n-m] - h[m:n]
            else:
                diffs = h[-lag:n] - h[0:n+lag]

            pos_cands = np.where(diffs > 0.01)[0]
            if len(pos_cands) == 0:
                pos_cands = np.where(diffs > 0.003)[0]
            if len(pos_cands) == 0:
                continue

            # Pick top imbalanced + random
            order = np.argsort(-np.abs(diffs[pos_cands]))
            n_try = min(25, len(pos_cands))
            selected = pos_cands[order[:n_try]]
            if n_try < len(pos_cands):
                extra_n = min(10, len(pos_cands) - n_try)
                extra = rng.choice(pos_cands[n_try:], extra_n, replace=False)
                selected = np.concatenate([selected, extra])

            for i in selected:
                if time.time() > deadline:
                    return best_h, best_c

                if lag > 0:
                    j = i + m
                    if h[i] <= h[j] + 1e-8:
                        continue
                else:
                    j = i + (-lag)
                    if h[j] <= h[i] + 1e-8:
                        continue

                for d in steps:
                    if d > np.abs(h[max(i,j)] - h[min(i,j)]) * 0.4:
                        continue
                    if lag > 0:
                        h[i] -= d
                        h[j] += d
                    else:
                        h[j] -= d
                        h[i] += d
                    h = project_feasible(h, n)
                    c_test = compute_c5(h, n)
                    if c_test < best_c - 1e-15:
                        best_c = c_test
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        if lag > 0:
                            h[i] += d
                            h[j] -= d
                        else:
                            h[j] += d
                            h[i] -= d
                        h = project_feasible(h, n)

        if not improved and rounds > 20:
            # Perturb to escape
            h = best_h + rng.normal(0, 0.003, n)
            h = project_feasible(h, n)
            rounds = 0

    return best_h, best_c
