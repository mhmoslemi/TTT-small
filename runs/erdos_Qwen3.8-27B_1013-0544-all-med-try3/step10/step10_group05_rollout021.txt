import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def smooth_max_obj(x, n, alpha, mu, c_vec=None):
    if c_vec is None:
        c_vec = compute_c5_vec(x, n)
    m = c_vec.max()
    sh = alpha * (c_vec - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            if k == 0:
                grad += wm * (1.0 - 2.0 * x)
            else:
                nv = n - k
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            if kk == 0:
                grad += wm * (1.0 - 2.0 * x)
            else:
                nv = n - kk
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def compute_gradient_active(h, n, active_indices, c_vec):

    if active_indices is None:

    else:

        active_mask = np.zeros(len(c_vec), dtype=bool)
        active_mask[top] = True

            else:

def greedy_pair_descent(h, n, deadline, max_iter=50000, n_top_lags=30):
    """Greedy pair descent: find best pair (i,j) to transfer mass."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for iteration in range(max_iter):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        if c_max < best_c:
            best_c = c_max
        
        # Compute gradient
        G = compute_gradient_active(best_h, n, n_top_lags, c_vec)
        
        # Find best pair: move from high G to low G
        # ΔJ ≈ δ * (G[i] - G[j]) for adding δ to i, subtracting from j
        # To decrease: G[i] < G[j], move from j to i
        j = np.argmax(G)
        i = np.argmin(G)
        
        if G[j] - G[i] < 1e-10:
            break  # No improvement possible
        
        # Maximum step
        delta_max = min(best_h[j], 1.0 - best_h[i])
        if delta_max < 1e-12:
            # Try to find another pair
            # Second best
            G_sorted = np.sort(G)
            break
        
        # Fractional step to avoid oscillation
        delta = delta_max * 0.5
        if delta < 1e-12:
            break
            
        best_h[j] -= delta
        best_h[i] += delta
        
        # Periodic projection
        if iteration % 100 == 0:
            best_h = project_feasible(best_h, n)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def batch_pair_descent(h, n, deadline, max_rounds=500, n_top_lags=20, batch_size=50):
    """Batch version: do multiple pair moves per round."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        
        G = compute_gradient_active(best_h, n, n_top_lags, c_vec)
        
        # Sort by G
        order = np.argsort(G)  # low to high
        n_pairs = min(batch_size, n // 2)
        
        improved = False
        for p in range(n_pairs):
            i = order[p]  # low G → add here
            j = order[n - 1 - p]  # high G → subtract here
            
            if G[j] - G[i] < 1e-10:
                break
            
            delta = min(best_h[j], 1.0 - best_h[i]) * 0.3
            if delta < 1e-12:
                continue
            
            test_h = best_h.copy()
            test_h[j] -= delta
            test_h[i] += delta
            new_c = compute_c5(test_h, n)
            
            if new_c < c_max - 1e-14:
                best_h = test_h
                improved = True
        
        if not improved:
            # Try smaller batch
            for p in range(min(10, n // 2)):
                i = order[p]
                j = order[n - 1 - p]
                if G[j] - G[i] < 1e-10:
                    break
                delta = min(best_h[j], 1.0 - best_h[i]) * 0.1
                if delta < 1e-14:
                    continue
                best_h[j] -= delta
                best_h[i] += delta
                if time.time() > deadline:
                    break
        
        best_h = project_feasible(best_h, n)
        new_best_c = compute_c5(best_h, n)
        if new_best_c < best_c:
            best_c = new_best_c
        else:
            # No improvement, break
            if rnd > 10:
                break
    
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_periodic(n, period, duty):
    """Periodic pattern with given period and duty cycle."""
    h = np.zeros(n)
    n_ones_per_period = int(period * duty)
    for i in range(n):
        if i % period < n_ones_per_period:
            h[i] = 1.0
    return project_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-12:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            if k == 0:
                grad += wm * (1.0 - 2.0 * x)
            else:
                nv = n - k
                grad[:k] += wm * 1.0
                grad[k:] += wm * (1.0 - x[:nv])
                grad[:nv] += wm * (-x[k:])
        else:
            kk = -k
            nv = n - kk
            grad[nv:] += wm * 1.0
            grad[:nv] += wm * (1.0 - x[kk:])
            grad[kk:] += wm * (-x[:nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def compute_gradient_active(h, n, c_vec, n_top_lags=30, alpha_sm=5000):
    dx = 2.0 / n
    idx_center = n - 1
    c_max = c_vec.max()
    top_idx = np.argsort(-c_vec)[:n_top_lags]
    sh_all = alpha_sm * (c_vec - c_max)
    smx = sh_all.max()
    w_all = np.exp(sh_all - smx)
    w = np.zeros(len(c_vec))
    w[top_idx] = w_all[top_idx]
    w_sum = w.sum()
    if w_sum < 1e-15:
        w[top_idx] = 1.0 / len(top_idx)
        w_sum = 1.0
    w /= w_sum
    G = np.zeros(n)
    for mm in top_idx:
        wm = w[mm] * dx
        if wm < 1e-15:
            continue
        k = mm - idx_center
        if k >= 0:
            if k == 0:
                G += wm * (1.0 - 2.0 * h)
            else:
                nv = n - k
                G[:k] += wm * 1.0
                G[k:] += wm * (1.0 - h[:nv])
                G[:nv] += wm * (-h[k:])
        else:
            kk = -k
            nv = n - kk
            G[nv:] += wm * 1.0
            G[:nv] += wm * (1.0 - h[kk:])
            G[kk:] += wm * (-h[:nv])
    return G

def batch_pair_descent(h, n, deadline, max_rounds=500, n_top_lags=30, batch_size=30, step_frac=0.4):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_step = step_frac
    stalled = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        G = compute_gradient_active(best_h, n, c_vec, n_top_lags)
        order = np.argsort(G)
        n_pairs = min(batch_size, n // 2)
        applied = 0
        for p in range(n_pairs):
            i_idx = int(order[p])
            j_idx = int(order[n - 1 - p])
            diff_g = G[j_idx] - G[i_idx]
            if diff_g < 1e-10:
                break
            delta = min(best_h[j_idx], 1.0 - best_h[i_idx]) * cur_step
            if delta < 1e-13:
                continue
            best_h[j_idx] -= delta
            best_h[i_idx] += delta
            applied += 1
        if applied == 0:
            break
        best_h = project_feasible(best_h, n)
        new_c = compute_c5(best_h, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
            cur_step = min(1.0, cur_step * 1.1)
            stalled = 0
        else:
            stalled += 1
            cur_step *= 0.7
            if cur_step < 0.02:
                cur_step = step_frac
                if stalled > 3:
                    break
    return best_h, best_c

def single_pair_polish(h, n, deadline, n_top_lags=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(50000):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        G = compute_gradient_active(best_h, n, c_vec, n_top_lags)
        j_idx = int(np.argmax(G))
        i_idx = int(np.argmin(G))
        if G[j_idx] - G[i_idx] < 1e-12:
            break
        delta = min(best_h[j_idx], 1.0 - best_h[i_idx]) * 0.5
        if delta < 1e-14:
            # try second best
            G_sorted = np.sort(G)
            i2 = int(np.argsort(G)[1])
            j2 = int(np.argsort(-G)[1])
            if G[j2] - G[i2] < 1e-12:
                break
            delta = min(best_h[j2], 1.0 - best_h[i2]) * 0.5
            if delta < 1e-14:
                break
            best_h[j2] -= delta
            best_h[i2] += delta
        else:
            best_h[j_idx] -= delta
            best_h[i_idx] += delta
        best_h = project_feasible(best_h, n)
        new_c = compute_c5(best_h, n)
        if new_c < best_c - 1e-15:
            best_c = new_c
        else:
            # revert
            if delta > 0:
                pass  # already applied, need to revert
            break
    return best_h, best_c

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 50 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_periodic(n, period, duty_num, duty_den):
    h = np.zeros(n)
    n_ones_pp = int(period * duty_num / duty_den)
    for i in range(n):
        if i % period < n_ones_pp:
            h[i] = 1.0
    return project_feasible(h, n)
