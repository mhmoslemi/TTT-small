def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def active_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

                            j -= n
                        if best_h[j] - diff_neg >= 0.0:
                            best_h[i] = cv
                            old_j = best_h[j]
                            best_h[j] = old_j - diff_neg
                            c_try = compute_c5(best_h, n)
                            best_h[j] = old_j
                            if c_try < best_c_j:
                                best_c_j = c_try
                                best_j = j
                    if best_j >= 0:
                        best_h[i] = cv
                        best_h[best_j] -= diff_neg
                        best_c = best_c_j

    return best_h, best_c

def upsample_to(h_coarse, n_fine):
    x_c = np.linspace(0, 2, len(h_coarse), endpoint=False)
    x_f = np.linspace(0, 2, n_fine, endpoint=False)
    h_f = np.interp(x_f, x_c, h_coarse)
    return project_feasible(h_f, n_fine)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_start = np.array(initial_h_values, dtype=float).copy()
        n_start = len(h_start)
    except Exception:
        h_start = np.full(500, 0.5)
        n_start = 500

    h_start = project_feasible(h_start, n_start)
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n_start)
    best_n = n_start

    # Phase 1: L-BFGS at n_start with progressive refinement
    if time.time() < deadline - 450:
        p1_end = min(t0 + 280, deadline - 450)
        h1, c1 = run_lbfgs_phase(best_h, best_n, p1_end,
            alphas=[100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
            mu=0.01,
            top_k_list=[None, None, 25, 20, 15, 10, 8, 5],
            maxit_list=[2000, 2500, 2500, 3000, 3000, 3500, 3500, 4000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
            best_n = n_start

    # Phase 2: Try at n=1000 for finer discrete approximation
    n_fine = 1000
    if time.time() < deadline - 300:
        h_fine = upsample_to(best_h, n_fine)
        c_fine = compute_c5(h_fine, n_fine)

        p2_end = min(time.time() + 180, deadline - 200)
        h2, c2 = run_lbfgs_phase(h_fine, n_fine, p2_end,
            alphas=[50, 500, 5000, 50000, 200000, 1000000, 5000000, 20000000],
            mu=0.005,
            top_k_list=[None, 25, 20, 15, 10, 8, 6, 4],
            maxit_list=[1500, 2000, 2500, 2500, 3000, 3000, 3000, 3000])

        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_fine

    # Phase 3: Also try n=2000 if time allows
    n_vfine = 2000
    if time.time() < deadline - 250:
        h_vf = upsample_to(best_h, n_vfine)
        c_vf = compute_c5(h_vf, n_vfine)
        # Just check if it's already better
        if c_vf < best_c:
            p3_end = min(time.time() + 100, deadline - 150)
            h3, c3 = run_lbfgs_phase(h_vf, n_vfine, p3_end,
                alphas=[100, 2000, 20000, 200000, 2000000],
                mu=0.003,
                top_k_list=[None, 20, 15, 10, 6],
                maxit_list=[1000, 1500, 2000, 2000, 2500])
            if c3 < best_c:
                best_c = c3
                best_h = h3
                best_n = n_vfine
        elif c_vf < best_c + 0.0001:
            # Close enough to try a bit
            p3_end = min(time.time() + 60, deadline - 150)
            h3, c3 = run_lbfgs_phase(h_vf, n_vfine, p3_end,
                alphas=[1000, 10000, 100000],
                mu=0.005,
                top_k_list=[15, 10, 6],
                maxit_list=[1000, 1500, 1500])
            if c3 < best_c:
                best_c = c3
                best_h = h3
                best_n = n_vfine

    # Phase 4: SLSQP refinement
    if time.time() < deadline - 120:
        n_best = len(best_h)
        p4_end = min(time.time() + 60, deadline - 100)
        h4, c4 = slqp_phase(best_h, n_best, p4_end,
            alphas_to_try=[500000, 2000000, 10000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_best

    # Phase 5: Very tight active-set (top 3-4 shifts)
    if time.time() < deadline - 80:
        n_best = len(best_h)
        p5_end = min(time.time() + 50, deadline - 60)
        h5, c5 = run_lbfgs_phase(best_h, n_best, p5_end,
            alphas=[5000000, 50000000, 500000000],
            mu=0.001,
            top_k_list=[5, 4, 3],
            maxit_list=[2500, 2500, 2500])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n_best

    # Phase 6: Coordinate descent
    if time.time() < deadline - 40:
        n_best = len(best_h)
        p6_end = min(time.time() + 30, deadline - 30)
        h6, c6 = coordinate_descent(best_h, n_best, p6_end, n_sweeps=30)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = n_best

    # Phase 7: Fine pair search
    if time.time() < deadline - 8:
        n_best = len(best_h)
        p7_end = deadline - 3
        h7, c7 = fine_pair_search(best_h, n_best, p7_end, radius=25)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n_best

    # Final validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
