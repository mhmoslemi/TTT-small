import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_weights(c_raw / dx, alpha)
    grad = np.zeros(n)
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            il = -lag
            grad[0:n - il] += wk * (1.0 - h[il:n])
            grad[il:n] += wk * (-h[0:n - il])
    return grad * dx

def subgrad_single(h, n, k_idx):
    """Subgradient of c_{k_idx} with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h)
    elif lag > 0:
        grad[lag:n] += (1.0 - h[0:n - lag])
        grad[0:n - lag] += (-h[lag:n])
    else:
        il = -lag
        grad[0:n - il] += (1.0 - h[il:n])
        grad[il:n] += (-h[0:n - il])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c_raw = np.correlate(x, 1 - x, mode='full')
        c = c_raw * (2.0 / n)
        obj_val = smooth_max(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 40,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def subgradient_descent(h, n, deadline, rng, n_iters_max=200000, eta0=0.05):
    """Subgradient method targeting the worst lag with Nesterov momentum."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_prev = h.copy()
    eta = eta0
    no_improve = 0

    for t in range(1, n_iters_max + 1):
        if time.time() > deadline:
            break
        # Nesterov momentum
        beta = (t - 1) / (t + 2)
        y = h + beta * (h - h_prev) if t > 1 else h.copy()
        y = np.clip(y, 0.0, 1.0)

        # Find worst lag
        c_raw = np.correlate(y, 1 - y, mode='full')
        k_star = np.argmax(c_raw)
        g = subgrad_single(y, n, k_star)

        # Adaptive step size
        gnrm = np.linalg.norm(g)
        if gnrm > 1e-15:
            eta = eta0 / np.sqrt(t + 1)
            eta = min(eta, 0.5 / gnrm)
        step = y - eta * g
        h_new = project_feasible(step, n)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                # Add noise to escape
                h_new = best_h + rng.normal(0, 0.005, n)
                h_new = project_feasible(h_new, n)
                no_improve = 0

        h_prev = h
        h = h_new

    return best_h, best_c

def dual_guided_optimization(h, n, deadline, rng, n_iters=50000):
    """Optimize using a dual distribution over lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Initialize dual: uniform over all lags
    nfull = 2 * n - 1
    p = np.ones(nfull) / nfull
    eta = 0.1
    no_improve = 0

    for t in range(1, n_iters + 1):
        if time.time() > deadline:
            break
        # Compute overlaps
        c_all = compute_all_lags(h, n)
        # Update dual: tilt toward high-overlap lags
        alpha_dual = 0.1 + 0.5 * t / n_iters
        p_new = np.exp(alpha_dual * c_all)
        p_new = p_new / p_new.sum()
        p = 0.5 * p + 0.5 * p_new

        # Compute weighted gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for k in range(nfull):
            pk = p[k]
            if pk < 1e-12:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += pk * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:n] += pk * (1.0 - h[0:n - lag])
                grad[0:n - lag] += pk * (-h[lag:n])
            else:
                il = -lag
                grad[0:n - il] += pk * (1.0 - h[il:n])
                grad[il:n] += pk * (-h[0:n - il])
        grad *= dx

        # Step
        gnrm = np.linalg.norm(grad)
        if gnrm > 1e-15:
            step_size = min(eta / np.sqrt(t + 1), 0.3 / gnrm)
        else:
            step_size = 0.0
        h_new = project_feasible(h - step_size * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                h_new = best_h + rng.normal(0, 0.003, n)

    return best_h, best_c

        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
