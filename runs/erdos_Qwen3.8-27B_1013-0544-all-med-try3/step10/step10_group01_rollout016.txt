    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()

    # We want sum(clip(h - lambda, 0, 1)) = target

    for _ in range(100):

        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = h_trial.sum()
        if s > target:

        else:

    h_cur = h_cur.copy() if hasattr(h_cur, 'copy') else h.copy()

            break
        c_vec = compute_c5_vec(h_cur, n)
        max_val = c_vec.max()

        active = np.where(c_vec >= max_val - tol)[0]

            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - h_cur[k:n]) * dx_val
                grad[k:] += (-h_cur[:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:] += (1.0 - h_cur[:nv]) * dx_val
                grad[:nv] += (-h_cur[kk:]) * dx_val
        grad /= len(active)
        gnorm = grad.sum() ** 2
        if gnorm < 1e-24:
            break
        step_size = step_scale * (max_val - 0.35) / np.sqrt(gnorm)
        step_size = min(step_size, 0.05)
        h_next = h_cur - step_size * grad
        h_next = project_feasible(h_next, n)
        c_new = compute_c5(h_next, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_next.copy()
            h_cur = h_new
            patience = 0
        else:
            h_cur = h_next
            patience += 1
            if patience > 500:
                step_scale *= 0.7
                patience = 0
                if step_scale < 1e-6:
                    break
    return best_h, best_c

                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_proj
                    c_vec = compute_c5_vec(best_h, n)
                    max_val = max_val
                    improved_any = True
                    break
                else:
                    best_h = h_proj.copy()
                    best_h[i] += amt
                    best_h[j] -= amt
                    best_h = project_feasible(best_h, n)
        if not improved_any:
            break
    return best_h, best_c


def fine_pair_local(h, n, deadline, radius=30, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        hp = project_feasible(best_h, n)
                        c = compute_c5(hp, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = hp
                            improved = True
                        else:
                            best_h = h.copy()
                            best_h[i] += d
                            best_h[j] -= d
                            best_h = project_feasible(best_h, n)
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        hp = project_feasible(best_h, n)
                        c = compute_c5(hp, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = hp
                            improved = True
                        else:
                            best_h = h.copy()
                            best_h[i] -= d
                            best_h[j] += d
                            best_h = project_feasible(best_h, n)
    return best_h, best_c


def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 3 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)


def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)


def mutate_and_retry(h, n, deadline, rng, n_tries=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(n_tries):
        if time.time() > deadline:
            break
        noise_scale = 0.05 * (1.0 - t / n_tries)
        h_mut = best_h + rng.randn(n) * noise_scale
        h_mut = project_feasible(h_mut, n)
        c_mut = compute_c5(h_mut, n)
        if c_mut < best_c + 0.005:
            h_sub, c_sub = run_lbfgs(h_mut, n, min(deadline, time.time() + 40),
                                     alphas=[10000, 50000, 200000],
                                     mu=0.01, maxits=[300, 500, 800])
            if c_sub < best_c:
                best_c = c_sub
                best_h = h_sub
    return best_h, best_c


def resample_h(h_from, n_from, n_to):
    if n_from == n_to:
        return h_from.copy()
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False) + 1.0 / n_from
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False) + 1.0 / n_to
    h_inter = np.interp(x_new, x_old, h_from, period=2.0)
    return project_feasible(h_inter, n_to)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    alt_starts = []
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts.append(make_beatty_start(n, golden / 2.0))
    alt_starts.append(make_beatty_start(n, np.sqrt(2.0) / 2.0))
    alt_starts.append(make_beatty_start(n, (1.0 + np.sqrt(3.0)) / 4.0))
    alt_starts.append(make_random_binary(n, rng))
    alt_starts.append(make_random_binary(n, rng))

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: L-BFGS warmup with increasing alpha
    p1_end = t0 + 180
    if time.time() < p1_end:
        h1, c1 = run_lbfgs(best_h, n, p1_end,
                           alphas=[20, 100, 500, 2000, 10000, 50000, 200000, 1000000],
                           mu=0.01, maxits=[500, 800, 1000, 1500, 2000, 2500, 3000, 4000])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization
    p2_end = t0 + 420
    if time.time() < p2_end:
        configs = [
            (4, [50000, 200000, 1000000, 5000000], 4),
            (8, [200000, 1000000, 5000000], 4),
            (16, [1000000, 5000000], 3),
            (24, [5000000, 20000000], 3),
        ]
        h2, c2 = run_active_set(best_h, n, p2_end, configs)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Subgradient method
    p3_end = t0 + 650
    if time.time() < p3_end:
        h3, c3 = subgradient_optimize(best_h, n, p3_end, max_steps=30000)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Exhaustive pairwise
    p4_end = t0 + 800
    if time.time() < p4_end:
        h4, c4 = exhaustive_pair_search(best_h, n, p4_end, max_rounds=4)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Randomized restart
    p5_end = t0 + 920
    if time.time() < p5_end:
        h5, c5 = mutate_and_retry(best_h, n, p5_end, rng, n_tries=4)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Final L-BFGS polish with very high alpha
    p6_end = t0 + 1020
    if time.time() < p6_end:
        h6, c6 = run_active_set(best_h, n, p6_end,
                                configs=[(4, [10000000, 50000000], 3),
                                         (8, [5000000, 20000000], 3)])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Fine pair local search
    p7_end = t0 + budget_s - 15
    if time.time() < p7_end:
        h7, c7 = fine_pair_local(best_h, n, p7_end, radius=25,
                                 deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final micro adjustments
    if time.time() < deadline - 2:
        h8, c8 = fine_pair_local(best_h, n, deadline - 1, radius=15,
                                 deltas=[0.00001, 0.000005, 0.000002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    final_h = project_feasible(best_h, n)
    final_c = compute_c5(final_h, n)

    return final_h, final_c, n
