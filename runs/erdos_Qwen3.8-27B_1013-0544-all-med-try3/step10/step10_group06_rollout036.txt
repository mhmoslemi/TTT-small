Let me think about this differently. The problem is:
min max_k g_k(h)

At the optimum, several lags will be "active" (achieving the max). To improve, we need to find a direction that reduces all active lags simultaneously.

This is like a **Chebyshev approximation** problem. We want to find h such that all g_k(h) are as small as possible, with the maximum being minimized.

A practical approach:
1. Find the active lags (those achieving the max)
2. Solve a small QP: minimize t subject to g_k(h) ≤ t for all active k, 0≤h≤1, sum(h)=n/2
   But this is non-linear (g_k is quadratic), so we linearize.
3. Linearize: g_k(h + δ) ≈ g_k(h) + ∇g_k(h)^T δ
4. Solve: min t subject to g_k(h) + ∇g_k(h)^T δ ≤ t, 0≤h+δ≤1, sum(h+δ)=n/2
   This is a LP! We can solve it with scipy.
5. Take a fraction of the solution (line search) to ensure we don't overshoot.

This is essentially **linear programming-based local search** or **cutting plane** approach. It should be very effective for fine-tuning.

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

    if lag == 0:
        g = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:n] += (1.0 - h[0:n - lag]) * dx
        g[0:n - lag] += (-h[lag:n]) * dx
    else:
        idx = -lag
        g[0:n - idx] += (1.0 - h[idx:n]) * dx
        g[idx:n] += (-h[0:n - idx]) * dx
    return g

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    penalty_coeff = 1e8

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:

            best_c = c_opt
            best_h = h_opt

    # Phase 3: High-alpha L-BFGS-B (100s)
    if time.time() < deadline - 200:
        for alpha in [1000000, 10000000, 100000000]:
            if time.time() > deadline - 120:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 25, maxiter=3000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: LP + pair search (80s)
    if time.time() < deadline - 100:
        dl = min(deadline - 80, time.time() + 40)
        h_opt, c_opt = lp_local_search(best_h, n, dl, max_iters=5000, top_k=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 50:
            dl2 = min(deadline - 40, time.time() + 30)
            h_opt, c_opt = pair_local_search(best_h, n, dl2, rng, top_k=5)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Gradient-guided SA (70s)
    if time.time() < deadline - 60:
        dl = min(deadline - 40, time.time() + 70)
        h_opt, c_opt = gradient_guided_sa(best_h, n, dl, rng,
                                           alpha=200000, t_init=0.002,
                                           t_min=0.0001, cool=0.99995)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Post-SA polish (40s)
    if time.time() < deadline - 30:
        for alpha in [5000000, 50000000]:
            if time.time() > deadline - 15:
                break
            h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + 10, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: Final LP search (20s)
    if time.time() < deadline - 8:
        h_opt, c_opt = lp_local_search(best_h, n, deadline - 5, max_iters=3000, top_k=10)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Try alternate resolutions
    for n_alt in [400, 600]:
        if time.time() > deadline - 10:
            break
        h_alt = expand_to_resolution(best_h, n, n_alt)
        h_opt = lbfgsb_optimize(h_alt, n_alt, 500000, time.time() + 8, maxiter=1000)
        h_opt = lbfgsb_optimize(h_opt, n_alt, 10000000, time.time() + 5, maxiter=2000)
        c_opt = compute_c5(h_opt, n_alt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_alt

    # Final validation
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
