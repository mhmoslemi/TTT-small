import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    c = compute_c5_vec(h, n)
    return float(c.max())

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            nv = n - k
            grad[:nv] += (1.0 - h[k:]) * dx
            if k > 0:
                grad[k:] -= h[:nv] * dx
    else:
        kk = -k
        if kk < n:
            nv = n - kk
            grad[:nv] -= h[kk:] * dx
            grad[kk:] += (1.0 - h[:nv]) * dx
    return grad

def smooth_max_value(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    z = alpha * (c - m)
    mz = z.max()
    if mz > 400:
        val = m + (mz + np.log(np.sum(np.exp(z - mz)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(z))) / alpha
    return float(val)

def smooth_max_gradient(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    z = alpha * (c - m)
    mz = z.max()
    w = np.exp(z - mz)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        grad += w[idx] * grad_single_k(h, n, k)
    return grad

def topk_smooth_value(h, n, active, alpha):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    z = alpha * (vals - m)
    if z.max() > 400:
        val = m + (z.max() + np.log(np.sum(np.exp(z - z.max())))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(z))) / alpha
    return float(val)

def topk_smooth_grad(h, n, active, alpha):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    z = alpha * (vals - m)
    w = np.exp(z - z.max())
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        grad += wi * grad_single_k(h, n, k)
    return grad

def phase_lbfgs(h, n, deadline, alphas, mu=0.01, maxit_base=800):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 15:
            break
        a = alpha
        def obj_fn(x):
            return smooth_max_value(x, n, a) + mu * (x.sum() - n/2.0)**2
        def grad_fn(x):
            return smooth_max_gradient(x, n, a) + 2.0*mu*(x.sum() - n/2.0)
        rem = max(5, int(deadline - time.time()))
        maxit = min(maxit_base + 200*i, max(200, rem*2))
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-15, 'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def phase_active_set(h, n, deadline, top_k=5, alphas=None, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [20000, 100000, 500000, 2000000, 10000000]
    for rnd in range(rounds):
        if time.time() > deadline - 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            a = alpha
            ak = active
            def obj_fn(x):
                return topk_smooth_value(x, n, ak, a) + 0.001*(x.sum()-n/2.0)**2
            def grad_fn(x):
                return topk_smooth_grad(x, n, ak, a) + 0.002*(x.sum()-n/2.0)
            rem = max(5, int(deadline - time.time()) / max(1, len(alphas)-1))
            maxit = max(200, min(1500, int(rem * 3)))
            try:
                res = minimize(obj_fn, best_h, jac=grad_fn, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-15, 'maxls': 30, 'maxcor': 40})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def subgradient_momentum(h, n, deadline, n_iter=50000, top_k=3):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.002
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-12, 5e-4 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            grad += grad_single_k(best_h, n, k)
        grad /= len(active)
        v = momentum * v - lr * grad
        step = best_h - v
        step = np.clip(step, 0.0, 1.0)
        step = project_feasible(step, n)
        c_new = compute_c5(step, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = step
            lr = min(lr * 1.05, 0.01)
        else:
            lr = max(lr * 0.95, 1e-5)
    return best_h, best_c

def pair_coordinate_descent(h, n, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    active_pairs = []
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-11, 0.005 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        improved = False
        for idx in active:
            k = idx - (n - 1)
            if k == 0:
                continue
            if k > 0:
                nv = n - k
                for i in range(0, nv, max(1, nv // 30)):
                    if time.time() > deadline:
                        break
                    j = i + k
                    if best_h[i] > 1e-6 and best_h[j] < 1.0 - 1e-6:
                        amt = min(0.005, best_h[i] * 0.1, (1.0 - best_h[j]) * 0.1)
                        best_h[i] -= amt
                        best_h[j] += amt
                        best_h = project_feasible(best_h, n)
                        c_new = compute_c5(best_h, n)
                        if c_new < best_c - 1e-16:
                            best_c = c_new
                            improved = True
                        else:
                            best_h[i] += amt
                            best_h[j] -= amt
                            best_h = project_feasible(best_h, n)
            else:
                kk = -k
                nv = n - kk
                for i in range(kk, n, max(1, nv // 30)):
                    if time.time() > deadline:
                        break
                    j = i - kk
                    if best_h[i] > 1e-6 and best_h[j] < 1.0 - 1e-6:
                        amt = min(0.005, best_h[i] * 0.1, (1.0 - best_h[j]) * 0.1)
                        best_h[i] -= amt
                        best_h[j] += amt
                        best_h = project_feasible(best_h, n)

                        if c_new < best_c - 1e-16:

                            improved = True
                        else:
                            best_h[i] += amt
                            best_h[j] -= amt
                            best_h = project_feasible(best_h, n)

def micro_adjust(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()

            active = np.where(c_vec >= max_val - tol)[0]
            for idx in active[:8]:
                if time.time() > deadline:
                    break
                k = idx - (n - 1)
                if k == 0:
                    continue
                if k > 0:
                    nv = n - k
                    positions = np.random.choice(nv, size=min(10, nv), replace=False)
                else:
                    kk = -k
                    nv = n - kk
                    positions = np.random.choice(nv, size=min(10, nv), replace=False)
                    k = -k

                    else:
                        j = i - k
                    if j < 0 or j >= n:

                        best_h[j] -= d
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)

                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d

                        best_h = project_feasible(best_h, n)

                            improved = True
                        else:

def upsample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new)

def make_beatty_binary(n, alpha_param, rng=None):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h < 0.5)[0]
        if len(zeros) > 0:
            idx = zeros[np.argmin(remaining)]
            h[idx] = min(1.0, h[idx] + remaining)
    return project_feasible(h, n)

def anneal_refine(h, n, deadline, n_iter=300000, t_start=0.003, t_end=0.0001):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    n_iter_actual = max(1000, n_iter)
    for it in range(n_iter_actual):
        if time.time() > deadline:
            break
        progress = it / n_iter_actual
        t = t_start * (t_end / t_start) ** progress
        # Propose a move: transfer mass between two positions
        i = np.random.randint(n)
        window = max(1, n // 50)
        j = np.random.randint(max(0, i-window), min(n, i+window+1))
        if i == j:
            continue
        amt = np.random.uniform(0.001, 0.01)
        # Decide direction
        if np.random.random() < 0.5:
            di, dj = -amt, amt
        else:
            di, dj = amt, -amt
        if cur_h[i] + di < 0 or cur_h[i] + di > 1:
            continue
        if cur_h[j] + dj < 0 or cur_h[j] + dj > 1:
            continue
        cur_h[i] += di
        cur_h[j] += dj
        cur_h = project_feasible(cur_h, n)
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
            cur_c = new_c
            if new_c < best_c - 1e-16:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] -= di
            cur_h[j] -= dj
            cur_h = project_feasible(cur_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Try alternative starts at n0
    alt_starts = []
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts.append(make_beatty_binary(n0, golden / 2.0))
    alt_starts.append(make_beatty_binary(n0, np.sqrt(2.0) / 2.0))
    h_rand = np.zeros(n0)
    h_rand[rng.choice(n0, size=n0//2, replace=False)] = 1.0
    alt_starts.append(project_feasible(h_rand, n0))

    for hs in alt_starts:
        c_s = compute_c5(hs, n0)
        if c_s < best_c:
            best_c = c_s
            best_h = hs.copy()

    # Phase 1: Quick L-BFGS at n0
    p1_end = min(t0 + 80, deadline - 600)
    if p1_end > t0 + 20:
        h1, c1 = phase_lbfgs(best_h, n0, p1_end,
            alphas=[50, 200, 1000, 5000, 20000], mu=0.01, maxit_base=600)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Upsample to 2*n0
    n_new = 2 * n0
    p2_start = time.time()
    h_up = upsample_h(best_h, n_new)
    c_up = compute_c5(h_up, n_new)
    if c_up < best_c:
        best_c = c_up
        best_h = h_up.copy()
        best_n = n_new
    else:
        # Still use upsampled as working copy at new resolution
        h_up = upsample_h(best_h, n_new)

    # Phase 3: L-BFGS at n_new
    p3_end = min(t0 + 350, deadline - 400)
    if p3_end > time.time() + 20:
        h3, c3 = phase_lbfgs(h_up, n_new, p3_end,
            alphas=[100, 500, 2000, 10000, 50000, 200000], mu=0.005, maxit_base=1000)
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_new

    # Phase 4: Active set optimization at n_new
    p4_end = min(t0 + 550, deadline - 250)
    if p4_end > time.time() + 20:
        h4, c4 = phase_active_set(best_h, best_n, p4_end,
            top_k=4, alphas=[50000, 200000, 1000000, 5000000], rounds=15)
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = best_n

    # Phase 5: Subgradient momentum
    p5_end = min(t0 + 700, deadline - 180)
    if p5_end > time.time() + 15:
        h5, c5 = subgradient_momentum(best_h, best_n, p5_end, n_iter=40000, top_k=3)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Another active set pass
    p6_end = min(t0 + 820, deadline - 130)
    if p6_end > time.time() + 15:
        h6, c6 = phase_active_set(best_h, best_n, p6_end,
            top_k=3, alphas=[200000, 1000000, 5000000, 20000000], rounds=12)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Pair coordinate descent
    p7_end = min(t0 + 920, deadline - 80)
    if p7_end > time.time() + 10:
        h7, c7 = pair_coordinate_descent(best_h, best_n, p7_end, max_rounds=100)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Anneal refine
    p8_end = min(t0 + 1020, deadline - 40)
    if p8_end > time.time() + 15:
        h8, c8 = anneal_refine(best_h, best_n, p8_end, n_iter=100000,
                               t_start=0.002, t_end=0.0002)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Final micro adjustments
    p9_end = deadline - 2
    if p9_end > time.time() + 8:
        h9, c9 = micro_adjust(best_h, best_n, p9_end)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and validation
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    # Ensure constraints are met
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c, final_n
