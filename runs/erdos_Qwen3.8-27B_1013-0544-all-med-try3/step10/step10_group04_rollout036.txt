```python
    import cvxpy as cp
    
    c_vec = compute_c5_vec(h, n)
    # Get top M indices
    top_idx = np.argsort(-c_vec)[:M]
    
    # Compute gradients of c_k at h for k in top_idx
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        return float(m + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def run_lbfgs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            def obj(x):
                v = smooth_obj(x, n, alpha)
                v += mu * (x.sum() - n/2.0)**2
                return v
            def grad(x):
                g = smooth_grad(x, n, alpha)
                g += 2.0*mu*(x.sum() - n/2.0)
                return g
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 100000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 20000, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coordinate_descent(h, n, deadline, max_passes=200):
    """Exact coordinate descent exploiting the affine structure of c_k in each h[i]."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c_vec = compute_c5_vec(h, n)
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    center = n - 1
    k_arr = np.arange(n2m1) - center  # actual shift values
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        any_improved = False
        for i in range(n):
            # Compute slopes: dc_k/dh[i] for all k
            # slope_k = dx * (1 - h[i+k] - h[i-k]) with OOB = 0
            idx_a = i + k_arr
            idx_b = i - k_arr
            a_vals = np.zeros(n2m1)
            valid_a = (idx_a >= 0) & (idx_a < n)
            a_vals[valid_a] = h[idx_a[valid_a]]
            b_vals = np.zeros(n2m1)
            valid_b = (idx_b >= 0) & (idx_b < n)
            b_vals[valid_b] = h[idx_b[valid_b]]
            slopes = dx * (1.0 - a_vals - b_vals)
            
            h_i = h[i]
            # Grid search
            t_grid = np.linspace(0, 1, 21)
            best_t = h_i
            best_val = c_vec.max()
            for t in t_grid:
                v = (c_vec + (t - h_i) * slopes).max()
                if v < best_val:
                    best_val = v
                    best_t = t
            
            # Ternary search refine
            if best_t > 0 and best_t < 1:
                # Find neighborhood of best_t
                grid_idx = np.argmin(np.abs(t_grid - best_t))
                lo = t_grid[max(0, grid_idx-1)]
                hi = t_grid[min(len(t_grid)-1, grid_idx+1)]
                for _ in range(15):
                    m1 = lo + (hi - lo) / 3
                    m2 = hi - (hi - lo) / 3
                    v1 = (c_vec + (m1 - h_i) * slopes).max()
                    v2 = (c_vec + (m2 - h_i) * slopes).max()
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1
                t_star = (lo + hi) / 2
                v_star = (c_vec + (t_star - h_i) * slopes).max()
                if v_star < best_val:
                    best_val = v_star
                    best_t = t_star
            
            if best_val < best_c - 1e-13 and abs(best_t - h_i) > 1e-8:
                h[i] = best_t
                c_vec += (best_t - h_i) * slopes
                cur_c = c_vec.max()
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    any_improved = True
        
        if not any_improved:
            break
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturbation_search(h, n, deadline, n_trials=10, mags=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    if mags is None:
        mags = [0.02, 0.04, 0.01, 0.06, 0.03, 0.05, 0.08, 0.015]
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 2000, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c
