import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu, top_m=100):

    active = w > 1e-13
    if active.sum() > top_m:
        active_idx = np.nonzero(active)[0]
        top_idx = active_idx[np.argsort(-w[active_idx])[:top_m]]
        active = np.zeros(2*n-1, dtype=bool)
        active[top_idx] = True
    for idx in np.nonzero(active)[0]:
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:

            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def subgradient_step(h, n, step_size, top_k=10):
    c = compute_c5_vec(h, n)
    top_indices = np.argsort(-c)[:top_k]
    center = n - 1
    grad = np.zeros(n)
    c_sum = c[top_indices].sum()
    if c_sum < 1e-15:
        return h
    for idx in top_indices:
        k = idx - center
        wt = c[idx] / c_sum
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wt * (1.0 - h[k:n])
                grad[k:n] += wt * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wt * (1.0 - h[0:nv])
                grad[0:nv] += wt * (-h[kk:n])
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

    h_new = np.interp(x_new, x_old, h_old)

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):
    def obj_fn(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 400:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def sq_f(x):
        return float(x.sum() - n / 2.0)
    def sq_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_f, 'jac': sq_j}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)
