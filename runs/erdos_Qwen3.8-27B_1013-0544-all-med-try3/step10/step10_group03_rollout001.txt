    if time.time() < deadline - 200:
        t_p4 = min(time.time() + 120, deadline - 200)
        best_h, best_c = adaptive_local_search(best_h, n, t_p4, top_k=8)

    # Phase 5: Subgradient polishing
    if time.time() < deadline - 100:
        t_p5 = min(time.time() + 50, deadline - 100)
        best_h, best_c = subgradient_steps(best_h, n, t_p5, n_steps=10000, lr_init=0.003)

    # Phase 6: Another round of adaptive local search
    if time.time() < deadline - 80:
        t_p6 = min(time.time() + 60, deadline - 80)
        best_h, best_c = adaptive_local_search(best_h, n, t_p6, top_k=5)

    # Phase 7: Final Nesterov
    if time.time() < deadline - 30:
        t_p7 = min(time.time() + 25, deadline - 30)
        h_nest2, c_nest2 = nesterov_smooth(best_h, n, t_p7, alpha=500000, mu=0.01, max_iter=10000)
        if c_nest2 < best_c:
            best_h, best_c = h_nest2, c_nest2

    # Phase 8: Final subgradient
    if time.time() < deadline - 5:
        h_sg2, c_sg2 = subgradient_steps(best_h, n, deadline - 1, n_steps=5000, lr_init=0.001)
        if c_sg2 < best_c:
            best_h, best_c = h_sg2, c_sg2

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
