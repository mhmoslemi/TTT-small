import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu, top_k=None):
    c = compute_c5_vec(x, n)
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        ct = c[idx]
        m = ct.max()
        sh = alpha * (ct - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
    else:
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu, top_k=None):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        ct = c[idx]
        m = ct.max()
        sh = alpha * (ct - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        # Map back to full correlation indices
        w_full = np.zeros(n2m1)
        for j, i in enumerate(idx):
            w_full[i] = w[j]
    else:
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        w_full = w
    
    for idx_i in range(n2m1):
        if w_full[idx_i] < 1e-13:
            continue
        k = idx_i - center
        wm = w_full[idx_i] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_h(h, n, deadline, configs=None, top_k=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (10000, 2000.0), (50000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu, top_k), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 50000, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coord_descent_worst_shifts(h, n, deadline, n_rounds=200):

    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0

                return best_h, best_c
            k = shift_idx - center
            # For this shift, overlap = sum h[x]*(1-h[x+k]) * dx
            # Try adjusting each point
            for i in range(min(n, 400)):  # Limit for speed
                if time.time() > deadline + 5:
                    return best_h, best_c

                        continue
                    h[i] = nv
                    # Recompute just the worst shift
                    if k >= 0:
                        nv_pts = n - k
                        ov = np.sum(h[0:nv_pts] * (1.0 - h[k:n])) * (2.0/n)
                    else:
                        kk = -k
                        nv_pts = n - kk
                        ov = np.sum(h[kk:n] * (1.0 - h[0:nv_pts])) * (2.0/n)
                    if ov < c_vec[shift_idx] - 1e-12:
                        # Check if overall max improved
                        # Quick check: is this still the worst?
                        improved = True
                        break
                    else:
                        h[i] = best_h[i]
                if improved:
                    h = best_h.copy()
                    c_vec = compute_c5_vec(h, n)

            c_now = compute_c5(h, n)

            else:
                h = best_h.copy()

def binary_local_search(h, n, deadline, max_flips=50000):
    """Local search: try flipping pairs of near-binary values."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    center = n - 1
    rng = np.random.RandomState(99)
    
    # Identify near-binary points
    near_one = np.where(h > 0.8)[0]
    near_zero = np.where(h < 0.2)[0]
    
    flips = 0
    while flips < max_flips and time.time() < deadline:
        # Pick a random near-one and near-zero point
        if len(near_one) == 0 or len(near_zero) == 0:
            break
        i1 = near_one[rng.randint(len(near_one))]
        i0 = near_zero[rng.randint(len(near_zero))]
        if i1 == i0:
            continue
        
        # Try: set i1 to 0 (if h[i1] > 0.5) and adjust i0
        # Or: push i1 towards 0, i0 towards 1
        orig_i1 = h[i1]
        orig_i0 = h[i0]
        
        # Option 1: flip i1 -> 0, i0 -> 1 (if it makes sense)
        if orig_i1 > 0.5 and orig_i0 < 0.5:
            h[i1] = 0.0
            h[i0] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h.copy()
            else:
                h = best_h.copy()
        
        # Option 2: partial push
        step = 0.05
        h[i1] = max(0.0, orig_i1 - step)
        h[i0] = min(1.0, orig_i0 + step)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h.copy()
        else:
            h[i1] = orig_i1
            h[i0] = orig_i0
        
        flips += 1
    
    return best_h, best_c

def slqp_polish2(h, n, deadline, alpha=2000000, maxiter=2000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def multi_start_optimize(h_init, n, deadline, n_starts=4):
