def dc_iteration(h, n, max_iter=100):
    """One round of DC algorithm."""
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        # Compute all correlation values and gradients
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        
        # Compute gradients for active shifts (top few)
        threshold = max_val - 0.001 * max_val
        active = np.where(c_vec >= threshold)[0]
        
        # Build LP: min t s.t. grad_k · h <= t - c_k + grad_k · h_old
        # Variables: h[0..n-1], t
        # c matrix: [0, 0, ..., 0, 1] (minimize t)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: grad_k · h - t <= -c_k + grad_k · h_old
        # For each active k:
        A_ub = []
        b_ub = []
        for k_idx in active:
            grad_k = shift_grad_single(h, n, k_idx)
            c_k = c_vec[k_idx]
            row = np.zeros(n + 1)
            row[:n] = grad_k
            row[-1] = -1.0
            A_ub.append(row)
            b_ub.append(-c_k + grad_k @ h)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Equality constraint: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= h_i <= 1, t >= 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_trial = (1 - alpha) * h + alpha * h_new
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c:
                h = h_trial
                best_c = c_trial
                improved = True
                break
        if not improved:
            break
    
    return h, best_c

        old_c = c_vec.max()

    T = T0

    stagnation = 0
    max_stagnation = 200000

    for step in range(n_steps):
        if time.time() > deadline:
            break
        if stagnation > max_stagnation:
            break

        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))

        delta = grad[:, i] - grad[:, j]
        new_c = float((c_vec + delta).max())
        old_c = float(c_vec.max())

        if new_c <= old_c + 1e-15:
            accept = True
        elif T > Tend * 10:
            delta_e = new_c - old_c
            if delta_e < 50 * T:
                accept = (rng.random() < np.exp(-delta_e / T))
            else:
                accept = False
        else:
            accept = False

        if accept:
            h[i] = 1.0
            h[j] = 0.0
            h_bin[i] = 1
            h_bin[j] = 0
            zeros.remove(i)
            ones.append(i)
            ones.remove(j)
            zeros.append(j)
            c_vec = c_vec + delta
            update_grad_flip(grad, h, n, i, j)

            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        T *= cooling

    return best_h, best_c

def binary_greedy(h_start, n, deadline, seed=42):
    """Greedy binary local search targeting max shifts."""
    rng = np.random.RandomState(seed)
    h_bin = (h_start > 0.5).astype(np.int8)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        rng.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target]:
            h_bin[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        rng.shuffle(zeros_idx)
        for idx in zeros_idx[:target - n_ones]:
            h_bin[idx] = 1

    h = h_bin.astype(np.float64)
    c_vec = compute_c5_vec(h, n)
    best_c = float(c_vec.max())
    best_h = h.copy()

    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    grad = compute_grad_matrix(h, n)

    no_improve = 0
    max_no_improve = 500000

    while time.time() < deadline and no_improve < max_no_improve:
        # Find active shifts
        max_val = float(c_vec.max())
        threshold = max_val - 0.005
        active = np.where(c_vec >= threshold)[0]

        # Try swaps that reduce active shifts
        found = False
        n_try = min(5000, len(zeros) * len(ones))
        for _ in range(n_try):
            if time.time() > deadline:
                return best_h, best_c
            i = int(rng.choice(zeros))
            j = int(rng.choice(ones))
            delta = grad[:, i] - grad[:, j]
            # Check if this reduces the max
            new_max = float((c_vec + delta).max())
            if new_max < best_c - 1e-13:
                h[i] = 1.0
                h[j] = 0.0
                h_bin[i] = 1
                h_bin[j] = 0
                zeros.remove(i)
                ones.append(i)
                ones.remove(j)
                zeros.append(j)
                c_vec = c_vec + delta
                update_grad_flip(grad, h, n, i, j)
                best_c = new_max
                best_h = h.copy()
                found = True
                no_improve = 0
                break

        if not found:
            no_improve += 1

    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs=None):
    """L-BFGS-B optimization with smooth max."""
    if configs is None:
        configs = [(500, 100.0), (2000, 1000.0), (5000, 10000.0), (15000, 50000.0)]

    bounds = [(0.0, 1.0)] * n
    best_c = compute_c5(h, n)

    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        h0 = h.copy()

        def obj(x, a=alpha, m=mu):
            c = compute_c5_vec(x, n)
            mx = float(c.max())
            sh = a * (c - mx)
            smx = float(sh.max())
            if smx > 500:
                val = mx + (smx + float(np.log(np.sum(np.exp(sh - smx))))) / a
            else:
                val = mx + float(np.log(np.sum(np.exp(sh)))) / a
            val += m * (x.sum() - n / 2.0) ** 2
            return val

        def obj_grad(x, a=alpha, m=mu):
            c = compute_c5_vec(x, n)
            mx = float(c.max())
            sh = a * (c - mx)
            smx = float(sh.max())
            w = np.exp(sh - smx)
            w /= w.sum()
            dx = 2.0 / n
            g = np.zeros(n)
            for mm in range(2 * n - 1):
                if w[mm] < 1e-14:
                    continue
                k = mm - (n - 1)
                wm = w[mm] * dx
                if k >= 0:
                    nv = n - k
                    g[:nv] += wm * (1.0 - x[k:])
                    g[k:] += wm * (-x[:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:] += wm * (1.0 - x[:nv])
                    g[:nv] += wm * (-x[kk:])
            g += 2.0 * m * (x.sum() - n / 2.0)
            return g

        try:
            res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass

    return h, best_c

def dc_algorithm(h, n, deadline, max_rounds=50):
    """DC algorithm: successive linearization with LP subproblem."""
    from scipy.optimize import linprog
    best_c = compute_c5(h, n)
    h = h.copy()
    n_full = 2 * n - 1

    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break

        c_vec = compute_c5_vec(h, n)
        max_val = float(c_vec.max())
        # Use active shifts (within 1% of max)
        threshold = max_val - max(0.005, 0.01 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])

        # Build LP: min t s.t. grad_k . h <= t - c_k + grad_k . h_old
        # vars: h[0..n-1], t (index n)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub_list = []
        b_ub_list = []
        for k_idx in active:
            k = k_idx - (n - 1)
            c_k = float(c_vec[k_idx])
            # Gradient of c_k at current h
            g_k = np.zeros(n)
            if k >= 0:
                nv = n - k
                g_k[:nv] = (1.0 - h[k:]) * (2.0 / n)
                g_k[k:] += (-h[:nv]) * (2.0 / n)
            else:
                kk = -k
                nv = n - kk
                g_k[kk:] = (1.0 - h[:nv]) * (2.0 / n)
                g_k[:nv] += (-h[kk:]) * (2.0 / n)

            row = np.zeros(n + 1)
            row[:n] = g_k
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(g_k @ h - c_k)

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds_lp = [(0, 1)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
            if res.status != 0:
                break
            h_lp = res.x[:n]
        except Exception:
            break

        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_trial = (1 - alpha) * h + alpha * h_lp
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-14:
                h = h_trial
                best_c = c_trial
                improved = True
                break
        if not improved:
            break

    return h, best_c

def multi_start_binary(n, deadline, seed=0):
    """Generate diverse binary initial points and run SA on each."""
    rng = np.random.RandomState(seed)
    results = []

    # Start 1: random
    h_rand = (rng.rand(n) > 0.5).astype(np.float64)
    h_rand = project_feasible(h_rand, n)

    # Start 2: block pattern
    h_block = np.zeros(n)
    h_block[:n//2] = 1.0

    # Start 3: interleaved
    h_inter = np.zeros(n)
    h_inter[::2] = 1.0
    h_inter = project_feasible(h_inter, n)

    # Start 4: golden ratio based
    golden = 0.6180339887498949
    h_gold = np.zeros(n)
    for i in range(n // 2):
        pos = int((i * golden) % 1.0 * n)
        h_gold[pos] = 1.0
    h_gold = project_feasible(h_gold, n)

    candidates = [h_rand, h_block, h_inter, h_gold]

    # Split time among candidates
    time_per = (deadline - time.time()) / len(candidates) - 2
    for idx, h_cand in enumerate(candidates):
        if time.time() > deadline - 5:
            break
        t_end = time.time() + max(10, time_per)
        h_opt, c_opt = binary_sa(h_cand, n, t_end, seed=seed + idx)
        results.append((h_opt, c_opt))

    # Return the best
    best = min(results, key=lambda x: x[1])
    return best[0], best[1]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_orig = 200

    n = 500
    # Interpolate to n=500 if needed
    if n_orig != n:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n)
        h_init = np.interp(x_new, x_old, h_init)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    # Phase 1: Quick continuous optimization on initial (60s)
    t1 = t0 + 60
    if deadline > t1:
        h1, c1 = lbfgs_optimize(h_init, n, t1,
                                configs=[(500, 100.0), (2000, 1000.0), (5000, 10000.0)])
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Binary SA on the continuous solution (300s)
    t2 = t0 + 360
    if deadline > t2 + 30:
        h2, c2 = binary_sa(best_h, n, t2, seed=seed)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Continuous optimization from binary (150s)
    t3 = t0 + 510
    if deadline > t3 + 20:
        h3, c3 = lbfgs_optimize(best_h, n, t3,
                                configs=[(1000, 500.0), (5000, 5000.0), (15000, 50000.0)])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Binary SA with different seed (200s)
    t4 = t0 + 710
    if deadline > t4 + 30:
        h4, c4 = binary_sa(best_h, n, t4, seed=seed + 1000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: DC algorithm (150s)
    t5 = t0 + 860
    if deadline > t5 + 10:
        h5, c5 = dc_algorithm(best_h, n, t5, max_rounds=30)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Binary greedy refinement (120s)
    t6 = t0 + 980
    if deadline > t6 + 10:
        h6, c6 = binary_greedy(best_h, n, t6, seed=seed + 2000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final continuous polish (remaining time)
    if time.time() < deadline - 10:
        h7, c7 = lbfgs_optimize(best_h, n, deadline - 3,
                                configs=[(5000, 10000.0), (20000, 100000.0)])
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
