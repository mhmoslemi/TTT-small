        if time.time() > deadline - 400:
            break
        t_phase = min(time.time() + 40, deadline - 350)
        h_opt = lbfgsb_optimize(best_h, n, alpha, K, t_phase, maxiter=800)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Multi-start with diverse perturbations
    if time.time() < deadline - 350:
        for trial in range(4):
            if time.time() > deadline - 300:
                break
            if trial == 0:
                h_pert = best_h + rng.normal(0, 0.03, n)
            elif trial == 1:
                h_pert = best_h + rng.normal(0, 0.05, n)
                h_pert[:n//4] += 0.02
                h_pert[-n//4:] -= 0.02
            elif trial == 2:
                # Phase shift perturbation
                shift = rng.randint(n//20, n//10)
                h_pert = np.roll(best_h, shift)
                h_pert = project_feasible(h_pert, n)
            else:
                # Binary snap then perturb
                h_bin = (best_h > 0.5).astype(float)
                h_bin = project_feasible(h_bin, n)
                h_pert = h_bin + rng.normal(0, 0.04, n)

            h_pert = project_feasible(h_pert, n)
            t_phase = min(time.time() + 30, deadline - 250)
            h_opt = lbfgsb_optimize(h_pert, n, 50000, 5, t_phase, maxiter=1000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            else:
                h_opt2 = lbfgsb_optimize(h_pert, n, 500000, 4, min(time.time() + 20, deadline - 230), maxiter=1000)
                c_opt2 = compute_c5(h_opt2, n)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    # Phase 3: Targeted local search
    if time.time() < deadline - 200:
        ls_deadline = min(deadline - 150, time.time() + 200)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: L-BFGS-B polish after local search
    if time.time() < deadline - 120:
        for alpha, K in [(100000, 5), (1000000, 4), (10000000, 3)]:
            if time.time() > deadline - 60:
                break
            t_phase = min(time.time() + 25, deadline - 50)
            h_opt = lbfgsb_optimize(best_h, n, alpha, K, t_phase, maxiter=1500)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Simulated annealing
    if time.time() < deadline - 100:
        sa_deadline = min(deadline - 80, time.time() + 100)
        h_opt, c_opt = simulated_annealing(best_h, n, sa_deadline, rng,
                                           alpha_sa=100000, t_init=0.004,
                                           t_min=0.00003, cool=0.9999)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 6: Try higher resolution
    for n_try in [800, 1200]:
        if time.time() > deadline - 80:
            break
        remaining = deadline - 50 - time.time()
        if remaining < 30:
            break
        h_interp = interpolate_resolution(best_h, len(best_h), n_try)
        t_phase = min(time.time() + max(15, remaining * 0.4), deadline - 30)
        h_opt = lbfgsb_optimize(h_interp, n_try, 50000, 5, t_phase, maxiter=1000)
        t_phase2 = min(time.time() + max(10, remaining * 0.3), deadline - 15)
        h_opt = lbfgsb_optimize(h_opt, n_try, 500000, 4, t_phase2, maxiter=1500)
        c_opt = compute_c5(h_opt, n_try)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final targeted local search
    if time.time() < deadline - 15:
        n_final = len(best_h)
        h_opt, c_opt = targeted_local_search(best_h, n_final, min(deadline - 3, time.time() + 80), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_final, 50000000, 3, deadline - 3, maxiter=2000)
        c_opt = compute_c5(h_opt, n_final)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
