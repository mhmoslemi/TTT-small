    target=n/2.0; diff=target-h.sum()
    if abs(diff) < 1e-12: return h

       h=h.copy()

       if rs>1e-15: h += diff*room/rs

import numpy as np
import math
import time

GOLDEN_RATIO = (math.sqrt(5.0) - 1.0) / 2.0


def compute_c5_exact(h, n):
    vals = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(vals)) * (2.0 / float(n))


def current_max_cp(cp, cn):
    if cp.size <= 1:
        return 0.0
    return max(float(np.max(cp[1:])), float(np.max(cn[1:])))


def compute_direct_counts(h, n):
    cp = np.zeros(n, dtype=float)
    cn = np.zeros(n, dtype=float)
    g = 1.0 - h
    for m in range(1, n):
        L = n - m
        cp[m] = float(np.dot(h[:L], g[m:]))
        cn[m] = float(np.dot(h[m:], g[:L]))
    return cp, cn


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * float(n)

    s = float(x.sum())
    if abs(s - target) < 1e-14:
        return x

    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        sy = float(y.sum())
        if sy > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(6):
        diff = target - float(y.sum())
        if abs(diff) < 1e-14:
            break
        room = np.minimum(y, 1.0 - y)
        rs = float(room.sum())
        if rs > 1e-15:
            y += diff * (room / rs)
        else:
            idx = int(np.argmax(y))
            y[idx] += diff
        y = np.clip(y, 0.0, 1.0)

    return y


def exact_sum_fix(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if abs(diff) < 1e-8:
        room = np.minimum(h, 1.0 - h)
        rs = float(room.sum())
        if rs > 1e-15:
            h += diff * (room / rs)
        else:
            idx = int(np.argmax(h))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
        if abs(target - float(h.sum())) > 1e-10:
            h = project_feasible(h, n)
        return h
    return project_feasible(h, n)


def binarize_exact(h, n):
    target = int(n // 2)
    hb = (h >= 0.5).astype(float)
    cnt = int(round(float(hb.sum())))
    diff = target - cnt

    if diff > 0:
        zeros = np.where(hb < 0.5)[0]
        if zeros.size > 0:
            order = np.argsort(-h[zeros], kind="mergesort")
            k = min(diff, zeros.size)
            hb[zeros[order[:k]]] = 1.0
    elif diff < 0:
        ones = np.where(hb > 0.5)[0]
        if ones.size > 0:
            order = np.argsort(h[ones], kind="mergesort")
            k = min(-diff, ones.size)
            hb[ones[order[:k]]] = 0.0

    return hb


def make_rotation(n, alpha, phi):
    x = np.arange(n, dtype=float)
    frac = np.mod((x + float(phi)) * float(alpha), 1.0)
    h = (frac < 0.5).astype(float)
    return binarize_exact(h, n)


def make_random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    target = int(n // 2)
    if target > 0:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1.0
    return binarize_exact(h, n)


def add_noise_project(h, sigma, rng):
    noise = rng.normal(0.0, float(sigma), size=h.size)
    return project_feasible(h + noise, h.size)


def upsample_h(h, n0, n1):
    if n0 == n1:
        return project_feasible(h, n1)
    x0 = np.linspace(0.0, 2.0, n0)
    x1 = np.linspace(0.0, 2.0, n1)
    y = np.interp(x1, x0, h)
    return project_feasible(y, n1)


def search_params_for(n):
    if n <= 550:
        return 45, 45, 70, 3
    if n <= 750:
        return 35, 35, 55, 3
    return 25, 25, 40, 2


def smooth_weights(c, M, tau):
    w = np.zeros_like(c, dtype=float)
    if c.size <= 1:
        return w
    ex = (c[1:] - float(M)) / float(tau)
    ex = np.where(ex > 0.0, 0.0, ex)
    ex = np.where(ex < -50.0, -50.0, ex)
    ww = np.exp(ex)
    ww[ww < 1e-12] = 0.0
    w[1:] = ww
    return w


def weighted_gradient(h, wp, wn, n):
    d = np.zeros(n, dtype=float)
    active = np.where((wp[1:] + wn[1:]) > 1e-12)[0] + 1
    for mm in active:
        m = int(mm)
        L = n - m
        w = float(wp[m])
        if w > 1e-12:
            d[:L] += w * (1.0 - h[m:])
            d[m:] -= w * h[:L]
        w = float(wn[m])
        if w > 1e-12:
            d[m:] += w * (1.0 - h[:L])
            d[:L] -= w * h[m:]
    return d


def max_after_transfer(cp, cn, h, a, b, eps):
    if abs(eps) < 1e-15:
        return current_max_cp(cp, cn)

    n = h.size
    ncp = cp.copy()
    ncnn = cn.copy()

    # Positive lag counts: edge i -> i+m
    if a < n - 1:
        ncp[1:n - a] += -eps * (1.0 - h[a + 1:n])
    if a > 0:
        ncp[1:a + 1] += eps * h[:a]
    if b < n - 1:
        ncp[1:n - b] += eps * (1.0 - h[b + 1:n])
    if b > 0:
        ncp[1:b + 1] += -eps * h[:b]

    # Negative lag counts: edge i+m -> i
    if a > 0:
        ncnn[1:a + 1] += -eps * (1.0 - h[:a])
    if a < n - 1:
        ncnn[1:n - a] += eps * h[a + 1:n]
    if b > 0:
        ncnn[1:b + 1] += eps * (1.0 - h[:b])
    if b < n - 1:
        ncnn[1:n - b] += -eps * h[b + 1:n]

    m = abs(a - b)
    if m > 0:
        e2 = eps * eps
        ncp[m] += e2
        ncnn[m] += e2

    return current_max_cp(ncp, ncnn)


def apply_transfer(cp, cn, h, a, b, eps):
    if abs(eps) < 1e-15:
        return

    n = h.size

    if a < n - 1:
        cp[1:n - a] += -eps * (1.0 - h[a + 1:n])
    if a > 0:
        cp[1:a + 1] += eps * h[:a]
    if b < n - 1:
        cp[1:n - b] += eps * (1.0 - h[b + 1:n])
    if b > 0:
        cp[1:b + 1] += -eps * h[:b]

    if a > 0:
        cn[1:a + 1] += -eps * (1.0 - h[:a])
    if a < n - 1:
        cn[1:n - a] += eps * h[a + 1:n]
    if b > 0:
        cn[1:b + 1] += eps * (1.0 - h[:b])
    if b < n - 1:
        cn[1:n - b] += -eps * h[b + 1:n]

    m = abs(a - b)
    if m > 0:
        e2 = eps * eps
        cp[m] += e2
        cn[m] += e2

    h[a] -= eps
    h[b] += eps

    if h[a] < 0.0 and h[a] > -1e-10:
        h[a] = 0.0
    if h[a] > 1.0 and h[a] < 1.0 + 1e-10:
        h[a] = 1.0
    if h[b] < 0.0 and h[b] > -1e-10:
        h[b] = 0.0
    if h[b] > 1.0 and h[b] < 1.0 + 1e-10:
        h[b] = 1.0


def pair_transfer_golden(cp, cn, h, a, b, maxit=35):
    ha = float(h[a])
    hb = float(h[b])
    max_pos = min(ha, 1.0 - hb)
    max_neg = min(1.0 - ha, hb)
    margin = 1e-12

    if max_pos > margin:
        hi = max_pos - margin
    else:
        hi = 0.0
    if max_neg > margin:
        lo = -(max_neg - margin)
    else:
        lo = 0.0

    if hi <= lo:
        return 0.0, current_max_cp(cp, cn)

    invphi = (math.sqrt(5.0) - 1.0) / 2.0
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = max_after_transfer(cp, cn, h, a, b, c)
    fd = max_after_transfer(cp, cn, h, a, b, d)

    for _ in range(int(maxit)):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = max_after_transfer(cp, cn, h, a, b, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = max_after_transfer(cp, cn, h, a, b, d)

    best_eps = 0.0
    best_val = current_max_cp(cp, cn)

    candidates = (0.0, lo, hi, c, d, 0.5 * (lo + hi))
    for e in candidates:
        e = float(e)
        if e < lo:
            e = lo
        if e > hi:
            e = hi
        v = max_after_transfer(cp, cn, h, a, b, e)
        if v < best_val - 1e-13:
            best_val = v
            best_eps = e

    return best_eps, best_val

                    best_n = n_s

    if time.time() < final_deadline - 40.0:
        pd3, pi3, pe3, px3 = search_params_for(best_n)
        pe3 = max(30, pe3 - 20)
        px3 = max(1, px3 - 1)
        cp, cn = compute_direct_counts(best_h, best_n)
        hb3, _ = continuous_transfer_search(best_h, cp, cn, final_deadline - 10.0,
                                            rng, pd3, pi3, pe3, px3)
        if hb3 is not None:
            hb3 = exact_sum_fix(hb3, best_n)
            cb3 = compute_c5_exact(hb3, best_n)
            if cb3 < best_c:
                best_c = cb3
                best_h = hb3

    final_h = exact_sum_fix(best_h, best_n)
    final_c = compute_c5_exact(final_h, best_n)

    for threshold in (1e-6, 1e-4, 5e-4):
        h_clean = final_h.copy()
        mask = (h_clean < threshold) | (h_clean > 1.0 - threshold)
        if mask.any():
            h_clean[h_clean < threshold] = 0.0
            h_clean[h_clean > 1.0 - threshold] = 1.0
            h_clean = project_feasible(h_clean, best_n)
            c_clean = compute_c5_exact(h_clean, best_n)
            if c_clean < final_c:
                final_h = h_clean
                final_c = c_clean

    h_bin = binarize_exact(final_h, best_n)
    c_bin = compute_c5_exact(h_bin, best_n)
    if c_bin < final_c:
        final_h = h_bin
        final_c = c_bin

    final_h = exact_sum_fix(final_h, best_n)
    final_c = compute_c5_exact(final_h, best_n)

    return final_h, float(final_c), int(best_n)
