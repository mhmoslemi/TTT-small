```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_lag_gradient(c_idx, h, n):
    """Compute gradient of correlation at a specific lag index c_idx."""
    n = len(h)
    lag = c_idx - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # c = sum_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
        # d/dh[i] = (1-h[i+lag]) for i in [0, n-1-lag)
        # d/dh[i+lag] = -h[i] for i in [0, n-1-lag)
        grad[0:n-lag] += (1.0 - h[lag:n])
        grad[lag:n] += (-h[0:n-lag])
    else:
        idx = -lag
        # c = sum_{i=0}^{n-1-idx} h[i+idx] * (1 - h[i])
        # d/dh[i+idx] = (1-h[i]) for i in [0, n-1-idx)
        # d/dh[i] = -h[i+idx] for i in [0, n-1-idx)
        grad[idx:n] += (1.0 - h[0:n-idx])
        grad[0:n-idx] += (-h[idx:n])
    return grad

    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * compute_lag_gradient(k, h, n)
    return grad * dx

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:

                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)

def lp_polish(h, n, top_m=None, deadline=None):
    """Use LP linearization to polish the solution."""
    if top_m is None:
        top_m = 10
    dx = 2.0 / n
    c_all = compute_all_lags(h, n) / dx  # work in unnormalized units
    c_all_norm = c_all * dx  # normalized

    top_indices = np.where(c_all_norm >= thresh)[0]
    if len(top_indices) < top_m:
        top_indices = np.argsort(c_all_norm)[-top_m:]
    else:
        top_indices = top_indices[:top_m]
    
    # Compute gradients for these lags
    grads = np.array([compute_lag_gradient(k, h, n) * dx for k in top_indices])
    c_vals = c_all_norm[top_indices]
    
    m = len(top_indices)
    
    # LP: min t

        order = np.argsort(-h[zero_idx])

    return h_bin
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def lag_gradient(k, h):
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[0:n-lag] += (1.0 - h[lag:n])
        grad[lag:n] += (-h[0:n-lag])
    else:
        m = -lag
        grad[m:n] += (1.0 - h[0:n-m])
        grad[0:n-m] += (-h[m:n])
    return grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c_vals, alpha):
    m = c_vals.max()
    shifted = alpha * (c_vals - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad_vec(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * lag_gradient(k, h)
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e8 * diff * diff
        grad = smooth_max_grad_vec(x, n, alpha)
        grad += 2e8 * diff * np.ones(n)
        return obj_val, grad
    
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def lp_polish(h, n, top_m=15, max_iter=30):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        c_all = compute_all_lags(best_h, n)
        c_max = c_all.max()
        
        # Find active lags (within 1% of max)
        thresh = c_max - 0.01 * c_max
        active = np.where(c_all >= thresh)[0]
        if len(active) > top_m:
            # Keep only the top `top_m`
            active = np.argsort(c_all)[-top_m:]
        elif len(active) == 0:
            active = np.argsort(c_all)[-top_m:]
        
        m_act = len(active)
        grads = np.array([lag_gradient(k, best_h) * dx for k in active])
        c_vals = c_all[active]
        
        # Solve LP
        try:
            t_var = cp.Variable()
            delta_var = cp.Variable(n)
            
            constraints = [
                delta_var >= -best_h,
                delta_var <= 1.0 - best_h,
                cp.sum(delta_var) == 0
            ]
            for i, k in enumerate(active):
                constraints.append(c_vals[i] + grads[i] @ delta_var <= t_var)
            
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            
            solved = False
            for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
                try:
                    prob.solve(solver=solver)
                    if t_var.value is not None and delta_var.value is not None:
                        solved = True
                        break
                except Exception:
                    continue
            
            if not solved:
                break
            
            delta = np.clip(delta_var.value, -best_h - 1e-10, 1.0 - best_h + 1e-10)
            delta -= delta.sum() / n  # re-center
            
            # Line search
            improved = False
            for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                h_test = best_h + frac * delta
                h_test = project_feasible(h_test, n)
                c_test = compute_c5(h_test, n)
                if c_test < best_c - 1e-15:
                    best_c = c_test
                    best_h = h_test
                    improved = True
                    break
            
            if not improved:
                break
        except Exception:
            break
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def lag_gradient(k, h):
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[0:n - lag] += (1.0 - h[lag:n])
        grad[lag:n] += (-h[0:n - lag])
    else:
        m = -lag
        grad[m:n] += (1.0 - h[0:n - m])
        grad[0:n - m] += (-h[m:n])
    return grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c_vals, alpha):
    m = c_vals.max()
    shifted = alpha * (c_vals - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * lag_gradient(k, h)
    return grad * dx

def _obj_and_grad(x, n, alpha, target_sum):
    x = np.clip(x, 0.0, 1.0)
    c = compute_all_lags(x, n)
    obj_val = smooth_max_value(c, alpha)
    diff = x.sum() - target_sum
    obj_val += 1e8 * diff * diff
    grad = smooth_max_grad(x, n, alpha)
    grad += 2e8 * diff * np.ones(n)
    return obj_val, grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_and_grad, h, args=(n, alpha, target_sum),
                       method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)
```

def binary_project(h, n):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin
