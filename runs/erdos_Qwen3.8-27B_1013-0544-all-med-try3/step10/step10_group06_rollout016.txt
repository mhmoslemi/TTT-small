def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000):
    dx = 2.0 / n
    target_sum = n / 2.0

    def obj_and_grad(x):
        xc = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(xc, n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
        else:
            logsumexp = np.log(np.sum(np.exp(shifted)))
        obj_val = m + logsumexp / alpha
        diff = xc.sum() - target_sum
        obj_val += 1e8 * diff * diff

        w = np.exp(alpha * (c - m))
        w = w / w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * xc)
            elif lag > 0:
                grad[lag:n] += w[k] * (1.0 - xc[0:n - lag])
                grad[0:n - lag] += w[k] * (-xc[lag:n])
            else:
                idx = -lag
                grad[0:n - idx] += w[k] * (1.0 - xc[idx:n])
                grad[idx:n] += w[k] * (-xc[0:n - idx])
        grad *= dx
        grad += 2.0 * 1e8 * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)


def binary_swap_search(h, n, deadline, K=10, max_rounds=500):
    h = h.copy()
    best_h = h.copy()
    best_c = float(compute_all_lags(h, n).max())

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[n - 1] = False
        top_K = np.argsort(c_all * mask - (1 - mask) * 1e10)[-K:]
        current_max = c_all.max()

        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break

        improved = False

        for k_idx in top_K[::-1]:
            if time.time() > deadline:
                break
            L = abs(int(k_idx - (n - 1)))
            if L == 0 or L >= n:
                continue

            nc = np.zeros(n)
            nc[:n - L] += h[L:]
            nc[L:] += h[:n - L]

            nc_ones = nc[one_idx]
            nc_zeros = nc[zero_idx]

            a_order = np.argsort(nc_ones)[:min(100, len(one_idx))]
            b_order = np.argsort(-nc_zeros)[:min(100, len(zero_idx))]

            found = False
            for ai in a_order:
                if found or time.time() > deadline:
                    break
                a = one_idx[ai]
                for bi in b_order:
                    b = zero_idx[bi]
                    if a == b or nc[a] >= nc[b]:
                        continue
                    h[a] = 0.0
                    h[b] = 1.0
                    new_max = compute_all_lags(h, n).max()
                    if new_max < best_c - 1e-15:
                        best_c = float(new_max)
                        best_h = h.copy()
                        one_idx = np.where(h == 1)[0]
                        zero_idx = np.where(h == 0)[0]
                        improved = True
                        found = True
                        break
                    else:
                        h[a] = 1.0
                        h[b] = 0.0

        if not improved:
            broke_out = False
            c_all = compute_all_lags(h, n)
            worst_k = int(np.argmax(c_all * mask - (1 - mask) * 1e10))
            L = abs(int(worst_k - (n - 1)))
            if L > 0 and L < n:
                nc = np.zeros(n)
                nc[:n - L] += h[L:]
                nc[L:] += h[:n - L]
                for a in one_idx:
                    if broke_out or time.time() > deadline:
                        break
                    for b in zero_idx:
                        if a == b or nc[a] >= nc[b]:
                            continue
                        h[a] = 0.0
                        h[b] = 1.0
                        new_max = compute_all_lags(h, n).max()
                        if new_max < best_c - 1e-15:
                            best_c = float(new_max)
                            best_h = h.copy()
                            one_idx = np.where(h == 1)[0]
                            zero_idx = np.where(h == 0)[0]
                            broke_out = True
                            break
                        else:
                            h[a] = 1.0
                            h[b] = 0.0
            if not broke_out:
                break

    return best_h, best_c


def targeted_continuous_search(h, n, deadline, K=10):
    h = h.copy()
    best_h = h.copy()
    best_c = float(compute_all_lags(h, n).max())
    steps = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for step in steps:
        if time.time() > deadline:
            break
        progress = True
        while progress and time.time() < deadline:
            progress = False
            c_all = compute_all_lags(h, n)
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[n - 1] = False
            top_K = np.argsort(c_all * mask - (1 - mask) * 1e10)[-K:]

            for k_idx in top_K:
                if time.time() > deadline:
                    break
                lag = int(k_idx - (n - 1))
                if lag == 0 or abs(lag) >= n:
                    continue
                L = abs(lag)
                n_pairs = n - L
                stride = max(1, n_pairs // 300)
                for i in range(0, n_pairs, stride):
                    if time.time() > deadline:
                        break
                    if lag > 0:
                        j = i + L
                    else:
                        j = i - lag
                    if h[i] > h[j] + 0.005:
                        d = min(step, (h[i] - h[j]) * 0.25)
                        if d < 1e-8:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = np.clip(h, 0, 1)
                        nc = compute_all_lags(h, n).max()
                        if nc < best_c - 1e-15:
                            best_c = float(nc)
                            best_h = h.copy()
                            progress = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = np.clip(h, 0, 1)
                    elif h[j] > h[i] + 0.005:
                        d = min(step, (h[j] - h[i]) * 0.25)
                        if d < 1e-8:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0, 1)
                        nc = compute_all_lags(h, n).max()
                        if nc < best_c - 1e-15:
                            best_c = float(nc)
                            best_h = h.copy()
                            progress = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = np.clip(h, 0, 1)

    return best_h, best_c


def perturb_binary(h, n, rng, n_perturb=10):
    h = h.copy()
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    n_p = min(n_perturb, len(one_idx), len(zero_idx))
    a_choice = rng.choice(one_idx, n_p, replace=False)
    b_choice = rng.choice(zero_idx, n_p, replace=False)
    h[a_choice] = 0.0
    h[b_choice] = 1.0
    return h


def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B warmup with increasing alpha
    for alpha, maxit, tlim in [(200, 200, 30), (2000, 300, 40), (20000, 400, 40), (200000, 500, 40)]:
        if time.time() > deadline - 700:
            break
        t_take = min(tlim, deadline - 700 - time.time())
        if t_take < 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + t_take, maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Binary swap search
    if time.time() < deadline - 500:
        h_bin = binary_project(best_h, n)
        sa_dead = min(deadline - 450, time.time() + 250)
        h_opt, c_opt = binary_swap_search(h_bin, n, sa_dead, K=10, max_rounds=300)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            h_rel = lbfgsb_optimize(best_h, n, 1000000, time.time() + 30, maxiter=1500)
            c_rel = compute_c5(h_rel, n)
            if c_rel < best_c:
                best_c = c_rel
                best_h = h_rel

    # Phase 3: L-BFGS-B with high alpha
    for alpha, tlim in [(1000000, 35), (10000000, 35), (50000000, 40)]:
        if time.time() > deadline - 250:
            break
        t_take = min(tlim, deadline - 250 - time.time())
        if t_take < 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + t_take, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Binary swap again from new continuous solution
    if time.time() < deadline - 150:
        h_bin = binary_project(best_h, n)
        sa_dead = min(deadline - 100, time.time() + 100)
        h_opt, c_opt = binary_swap_search(h_bin, n, sa_dead, K=8, max_rounds=200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            h_rel = lbfgsb_optimize(best_h, n, 5000000, time.time() + 25, maxiter=1000)
            c_rel = compute_c5(h_rel, n)
            if c_rel < best_c:
                best_c = c_rel
                best_h = h_rel

    # Phase 5: Perturbation restarts with binary search
    for restart in range(5):
        if time.time() > deadline - 80:
            break
        h_pert = perturb_binary(best_h, n, rng, n_perturb=8)
        h_opt, c_opt = binary_swap_search(h_pert, n, time.time() + 30, K=8, max_rounds=100)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            h_rel = lbfgsb_optimize(best_h, n, 5000000, time.time() + 20, maxiter=1000)
            c_rel = compute_c5(h_rel, n)
            if c_rel < best_c:
                best_c = c_rel
                best_h = h_rel

    # Phase 6: Targeted continuous search
    if time.time() < deadline - 50:
        h_opt, c_opt = targeted_continuous_search(best_h, n, min(deadline - 10, time.time() + 40), K=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Final L-BFGS-B polish
    if time.time() < deadline - 10:
        h_opt = lbfgsb_optimize(best_h, n, 100000000, deadline - 3, maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final binary nudge
    if time.time() < deadline - 2:
        h_bin = binary_project(best_h, n)
        h_opt, c_opt = binary_swap_search(h_bin, n, deadline - 1, K=5, max_rounds=50)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final projection and validation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
