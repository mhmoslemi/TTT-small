def block_swap_annealing(h, n, deadline, rng):

def resample_h(h_orig, n_orig, n_new):

    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, n_new)
    return h_new

        max_val = c_raw[top_k_idx[0]]
        mean_top = c_raw[top_k_idx].mean()
        min_top = c_raw[top_k_idx[-1]]
        
        # If already well-equalized, break
        if max_val - min_top < 0.01:
            break
        
        # Weight: heavier on the larger correlations
        top_vals = c_raw[top_k_idx]
        w_top = (top_vals - min_top + 0.1) ** 2
        w_top /= w_top.sum()
        
        w = np.zeros(len(c_raw))
        w[top_k_idx] = w_top
        grad = h_gradient(w, h, n)
        
        # Line search: find best step
        best_step = 0.001
        best_c_step = compute_c5(h, n)
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = project_feasible(h - step * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c_step - 1e-14:
                best_step = step
                best_c_step = c_try
        
        if best_c_step < best_c:
            h = project_feasible(h - best_step * grad, n)
            best_c = compute_c5(h, n)
        else:
            # Try opposite direction on the min-top correlations
            w2 = np.zeros(len(c_raw))
            w2[top_k_idx[-k//3:]] = 1.0 / (k//3 + 1)
            grad2 = h_gradient(w2, h, n)
            for step in [0.005, 0.002, 0.001]:
                h_try = project_feasible(h + step * grad2, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    h = h_try
                    best_c = c_try
                    break
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    random.seed(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: L-BFGS at parent resolution (two passes with different alpha)
    p1_deadline = min(t0 + 120, deadline - 800)
    h1, c1 = run_lbfgs(best_h, best_n, p1_deadline, alpha=20000, mu=0.005, maxit=5000)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()
    
    p1b_deadline = min(t0 + 200, deadline - 750)
    h1b, c1b = run_lbfgs(best_h, best_n, p1b_deadline, alpha=200000, mu=0.001, maxit=5000)
    if c1b < best_c:
        best_c, best_h = c1b, h1b.copy()
    
    # Phase 2: Active set optimization
    p2_deadline = min(t0 + 380, deadline - 600)
    h2, c2 = active_set_optimize(best_h, best_n, p2_deadline, top_k=25, max_iters=80000)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()
    
    # Phase 3: Pair swap search
    p3_deadline = min(t0 + 500, deadline - 500)
    h3, c3 = pair_swap_search(best_h, best_n, p3_deadline, max_iters=60000)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()
    
    # Phase 4: Top-k equalization
    p4_deadline = min(t0 + 620, deadline - 400)
    h4, c4 = top_k_equalization(best_h, best_n, p4_deadline, k=15)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()
    
    # Phase 5: Try different resolutions
    resolutions_to_try = []
    if best_n != 400:
        resolutions_to_try.append(400)
    if best_n != 600:
        resolutions_to_try.append(600)
    if best_n != 700:
        resolutions_to_try.append(700)
    
    remaining = max(60, deadline - time.time())
    time_per_res = remaining / (len(resolutions_to_try) + 2)  # +2 for buffer and final
    
    for n_try in resolutions_to_try:
        if time.time() > deadline - 80:
            break
        sub_deadline = min(time.time() + time_per_res, deadline - 80)
        h_init = resample_h(best_h, best_n, n_try)
        
        # Quick optimization at new resolution
        h_r, c_r = run_lbfgs(h_init, n_try, min(sub_deadline - 20, sub_deadline),
                           alpha=50000, mu=0.005, maxit=3000)
        if time.time() < sub_deadline - 10:
            h_r2, c_r2 = active_set_optimize(h_r, n_try, sub_deadline, top_k=20, max_iters=30000)
            if c_r2 < c_r:
                h_r, c_r = h_r2, c_r2
        
        if c_r < best_c:
            best_c, best_h, best_n = c_r, h_r.copy(), n_try
    
    # Phase 6: Final L-BFGS polish on best solution
    p6_deadline = deadline - 30
    if time.time() < p6_deadline - 20:
        h6, c6 = run_lbfgs(best_h, best_n, p6_deadline, alpha=1000000, mu=0.0005, maxit=3000)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()
    
    # Phase 7: Final fine adjustments
    if time.time() < deadline - 5:
        h7, c7 = pair_swap_search(best_h, best_n, deadline - 3, max_iters=20000)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    # Verify constraint
    assert abs(best_h.sum() - best_n / 2.0) < 1e-10, f"Constraint violated: {best_h.sum()} vs {best_n/2}"
    assert best_h.min() >= -1e-15 and best_h.max() <= 1 + 1e-15
    
    return best_h, best_c, best_n
