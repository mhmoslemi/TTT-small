Let me think about what a good local search looks like:

1. Compute all overlaps: overlap[s] for s = -(n-1) to (n-1), or equivalently using the 2n-1 correlation values.
2. Find the worst shift s*.
3. Try to reduce overlap[s*] by moving a 1 that contributes to it.
4. Check that the move doesn't make other shifts worse.

For step 3: if overlap[s*] is due to specific positions (i, i+s*) where h[i]=0, h[i+s*]=1, then moving h[i+s*] from 1 to 0 (and putting 1 somewhere else) would reduce overlap[s*] by 1. But it might increase other overlaps.

A simple approach: for each "bad" pair contributing to the worst shift, try to undo it by swapping the 1 at position i+s* with a 0 at some other position. Accept if the overall max doesn't increase.

def correlate_delta(h, p, q, n):
    """Compute the change in each correlation value when h[p]:1→0, h[q]:0→1."""
    # h is the current state BEFORE the swap
    # p has h[p]=1, q has h[q]=0
    
    delta = np.zeros(2*n-1)
    
    for s in range(1, n):
        # Positive shift s: corr at index (n-1+s)
        d = 0
        # p as "1" position (h[k+s] slot): k=p-s, valid if 0<=p-s<=n-1-s
        if s <= p <= n-1-s:
            d -= (1 - h[p-s])
        # p as "0" position (1-h[k] slot, after swap): k=p, valid if 0<=p<=n-1-s  
        # Wait, before swap h[p]=1, so (1-h[p])=0. After swap h[p]=0, so (1-h[p])=1.
        # Term (1-h[p])*h[p+s] goes from 0 to h[p+s]. Delta: +h[p+s]
        if 0 <= p <= n-1-s:
            d += h[p+s]
        # q as "1" position (h[k+s] slot, after swap): k=q-s, valid if 0<=q-s<=n-1-s
        if s <= q <= n-1-s:
            d += (1 - h[q-s])
        # q as "0" position (before swap): k=q, valid if 0<=q<=n-1-s
        # Term (1-h[q])*h[q+s] goes from h[q+s] to 0. Delta: -h[q+s]
        if 0 <= q <= n-1-s:
            d -= h[q+s]
        
        delta[n-1+s] = d
    
    for s in range(1, n):
        # Negative shift -s: corr at index (n-1-s)
        d = 0
        # For overlap(-s) = sum h[k]*(1-h[k+s]):
        # p (1→0): 
        #   k=p: h[p]*(1-h[p+s]) → 0. Delta: -(1-h[p+s])
        if 0 <= p <= n-1-s:
            d -= (1 - h[p+s])
        #   k+s=p, k=p-s: h[p-s]*(1-h[p]) → h[p-s]*1. Delta: +h[p-s]
        if s <= p <= n-1-s:
            d += h[p-s]
        # q (0→1):
        #   k=q: h[q]*(1-h[q+s]) → 1*(1-h[q+s]). Delta: +(1-h[q+s])
        if 0 <= q <= n-1-s:

        #   k+s=q, k=q-s: h[q-s]*(1-h[q]) → h[q-s]*0. Delta: -h[q-s]
        if s <= q <= n-1-s:
            d -= h[q-s]
        
        delta[n-1-s] = d

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_ones = n // 2
    if n % 2 == 0:
        n_ones = n // 2
    else:
        n_ones = (n - 1) // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones] if n_ones < n else 0
        h_bin[next_idx] += remaining
    return h_bin

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def binary_sa(h, n, t_start, deadline, rng, focused_frac=0.5, n_restarts=3):
    """Simulated annealing on binary patterns with focused moves."""
    best_h = h.copy()
    best_corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    best_max = best_corr.max()
    best_c = float(best_max * (2.0 / n))
    
    ones_list = np.where(h == 1)[0].copy()
    zeros_list = np.where(h == 0)[0].copy()
    
    for restart in range(n_restarts):
        if time.time() >= deadline:
            break
        h = best_h.copy()
        corr = best_corr.copy()
        cur_max = best_max
        ones_list = np.where(h == 1)[0].copy()
        zeros_list = np.where(h == 0)[0].copy()
        
        t = t_start
        total_est = max(1, int((deadline - time.time()) * 3000))
        cooling_rate = 0.9998 if total_est > 1e6 else 0.999
        
        no_improve = 0
        max_no_improve = 8000
        
        while time.time() < deadline:
            if no_improve > max_no_improve:
                h = best_h.copy()
                corr = best_corr.copy()
                cur_max = best_max
                ones_list = np.where(h == 1)[0].copy()
                zeros_list = np.where(h == 0)[0].copy()
                no_improve = 0
                max_no_improve = min(max_no_improve + 5000, 30000)
                t = max(t, 0.3 * t_start)
                continue
            
            use_focused = rng.rand() < focused_frac
            
            if use_focused and cur_max > 0:
                # Find worst shift
                worst_idx = int(np.argmax(corr))
                s = worst_idx - (n - 1)
                # Find contributing 1-positions
                if s >= 0:
                    mask = (h[s:] == 1) & (h[:-s if s > 0 else 0] == 0)
                    contrib_ones = np.where(mask)[0] + s
                else:
                    ss = -s
                    mask = (h[:n-ss] == 1) & (h[ss:] == 0)
                    contrib_ones = np.where(mask)[0]
                
                if len(contrib_ones) == 0:
                    p_idx = rng.randint(len(ones_list))
                    p = ones_list[p_idx]
                else:
                    p = contrib_ones[rng.randint(len(contrib_ones))]
                    p_idx = int(np.where(ones_list == p)[0][0]) if p in ones_list else rng.randint(len(ones_list))
                    if p != ones_list[p_idx]:
                        p = ones_list[p_idx]
            else:
                p_idx = rng.randint(len(ones_list))
                p = ones_list[p_idx]
            
            q_idx = rng.randint(len(zeros_list))
            q = zeros_list[q_idx]
            
            if p == q:
                continue
            
            # Perform swap
            h[p] = 0
            h[q] = 1
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = new_corr.max()

            ones_list[p_idx] = q
            zeros_list[q_idx] = p
            
            if new_max < cur_max:
                cur_max = new_max
                corr = new_corr
                no_improve = 0
            else:
                delta = new_max - cur_max
                if delta > 0 and rng.rand() > np.exp(-delta / max(t, 1e-10)):
                    # Reject
                    h[p] = 1
                    h[q] = 0
                    ones_list[p_idx] = p
                    zeros_list[q_idx] = q

                best_h = h.copy()
                best_corr = corr.copy()
                best_c = float(best_max * (2.0 / n))
                no_improve = 0
            
            t = max(t * cooling_rate, 1e-4)
    
    return best_h, best_c

def binary_hill_climb(h, n, deadline, rng, patience=5000):
    """Focused hill climbing on binary patterns."""
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    best_max = corr.max()
    best_c = float(best_max * (2.0 / n))
    
    while time.time() < deadline:
        worst_idx = int(np.argmax(corr))
        s = worst_idx - (n - 1)
        
        # Find contributing positions to worst shift
        if s >= 0:
            mask = (h[s:] == 1) & (h[:n-s] == 0)
            contrib = np.where(mask)[0] + s
            target_positions = contrib
        else:
            ss = -s
            mask = (h[:n-ss] == 1) & (h[ss:] == 0)
            contrib = np.where(mask)[0]
            target_positions = contrib + ss
        
        if len(target_positions) == 0:
            break
        
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        
        improved = False
        tried = set()
        
        # Try moving contributing 1s to other zero positions
        for _ in range(min(50, len(target_positions))):
            if time.time() > deadline:
                break
            p = target_positions[rng.randint(len(target_positions))]
            if h[p] != 1:
                continue
            key = p
            if key in tried:
                continue
            tried.add(key)
            
            for _ in range(3):
                q = zeros_pos[rng.randint(len(zeros_pos))]
                if q == p or q in tried:
                    continue
                
                h[p] = 0
                h[q] = 1
                nc = np.correlate(h, 1 - h, mode='full')
                nm = nc.max()
                
                if nm < best_max:
                    best_max = nm
                    best_h = h.copy()
                    best_c = float(best_max * (2.0 / n))
                    corr = nc.astype(float)
                    improved = True
                    ones_pos = np.where(h == 1)[0]
                    zeros_pos = np.where(h == 0)[0]
                    break
                else:
                    h[p] = 1
                    h[q] = 0
        
        if improved:
            continue
        
        # If stuck, do random perturbation
        p = ones_pos[rng.randint(len(ones_pos))]
        q = zeros_pos[rng.randint(len(zeros_pos))]
        h[p], h[q] = h[q], h[p]
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
    
    return best_h, best_c

def continuous_refine(h, n, deadline, alpha_list=None, mu=0.01):
    """L-BFGS continuous refinement starting from near-binary h."""
    if alpha_list is None:
        alpha_list = [100, 500, 2000, 10000, 50000, 200000]
    
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alpha_list:
        remaining = deadline - time.time()
        if remaining < 10:
            break
        maxit = max(200, int(remaining * 3))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 5,
                                   'maxls': 30, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def thue_morse_binary(n, rng=None):
    """Generate a binary pattern based on Thue-Morse sequence."""
    h = np.zeros(n)
    # Thue-Morse: t(k) = parity of popcount(k)
    for i in range(n):
        pc = bin(i).count('1')
        if pc % 2 == 0:
            h[i] = 1.0
    # Adjust to get exactly n//2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        to_flip = ones_idx[rng.randint(n_ones - target + 1, size=target - target)] if False else rng.choice(ones_idx, size=n_ones - target, replace=False)
        h[to_flip] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zeros_idx, size=target - n_ones, replace=False)
        h[to_flip] = 1
    return h

def upsample_binary(h_old, n_old, n_new):
    """Upsample a binary pattern to higher resolution by repeating/interleaving."""
    h_new = np.zeros(n_new)
    scale = n_new / n_old
    for i in range(n_old):
        lo = int(i * scale)
        hi = int((i + 1) * scale)
        hi = min(hi, n_new)
        h_new[lo:hi] = h_old[i]
    # Adjust to get exactly n_new//2 ones
    n_ones = int(h_new.sum())
    target = n_new // 2
    if n_ones > target:
        ones_idx = np.where(h_new == 1)[0]
        to_flip = np.random.choice(ones_idx, size=n_ones - target, replace=False)
        h_new[to_flip] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_new == 0)[0]
        to_flip = np.random.choice(zeros_idx, size=target - n_ones, replace=False)
        h_new[to_flip] = 1
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0
    
    # === PHASE 1: Binary SA at n=500 ===
    n1 = 500
    h_bin1 = to_binary(h0, n1)
    c_bin1 = compute_c5(h_bin1, n1)
    if c_bin1 < best_c:
        best_c = c_bin1
        best_h = h_bin1.copy()
        best_n = n1
    
    p1_end = min(t0 + 220, deadline - 300)
    h_bin1, c_bin1 = binary_sa(h_bin1, n1, 1.0, p1_end, rng, focused_frac=0.5, n_restarts=3)
    if c_bin1 < best_c:
        best_c = c_bin1
        best_h = h_bin1.copy()
        best_n = n1
    
    # === PHASE 2: Binary hill climbing at n=500 ===
    p2_end = min(t0 + 340, deadline - 200)
    h_hc, c_hc = binary_hill_climb(best_h if best_n == n1 else to_binary(best_h, n1), n1, p2_end, rng)
    if c_hc < best_c:
        best_c = c_hc
        best_h = h_hc.copy()
        best_n = n1
    
    # === PHASE 3: Continuous refinement at n=500 ===
    p3_end = min(t0 + 440, deadline - 150)
    h_cont, c_cont = continuous_refine(best_h, n1, p3_end,
        alpha_list=[50, 200, 1000, 5000, 20000, 100000], mu=0.005)
    if c_cont < best_c:
        best_c = c_cont
        best_h = h_cont.copy()
        best_n = n1
    
    # === PHASE 4: Try n=1000 ===
    n4 = 1000
    p4_start = time.time()
    if p4_start < deadline - 250:
        h_up = upsample_binary(best_h if best_n == n1 else to_binary(best_h, n4), n1, n4)
        # Also try a fresh binary at n=1000
        h_rand4 = to_binary(h0, n1)
        h_up2 = upsample_binary(h_rand4, n1, n4)
        
        # SA on upsampled
        p4_end1 = min(p4_start + 150, deadline - 150)
        h4a, c4a = binary_sa(h_up, n4, 1.0, p4_end1, rng, focused_frac=0.5, n_restarts=2)
        
        # Hill climb
        p4_end2 = min(p4_end1 + 80, deadline - 80)
        h4b, c4b = binary_hill_climb(h4a, n4, p4_end2, rng)
        
        if c4b < best_c:
            best_c = c4b
            best_h = h4b.copy()
            best_n = n4
    
    # === PHASE 5: Continuous refinement at best resolution ===
    p5_end = min(time.time() + 120, deadline - 60)
    if time.time() < p5_end:
        h5, c5 = continuous_refine(best_h, best_n, p5_end,
            alpha_list=[200, 1000, 5000, 20000, 100000, 500000], mu=0.003)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
    
    # === PHASE 6: Try a different initialization (Thue-Morse) ===
    if time.time() < deadline - 80:
        h_tm = thue_morse_binary(n1, rng)
        c_tm = compute_c5(h_tm, n1)
        p6_end = min(time.time() + 60, deadline - 30)
        h6, c6 = binary_sa(h_tm, n1, 0.8, p6_end, rng, focused_frac=0.6, n_restarts=2)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()
            best_n = n1
    
    # === PHASE 7: Final continuous polish ===
    if time.time() < deadline - 5:
        h7, c7 = continuous_refine(best_h, best_n, deadline - 3,
            alpha_list=[5000, 20000, 100000, 500000], mu=0.002)
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()
    
    # Final verification and projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
