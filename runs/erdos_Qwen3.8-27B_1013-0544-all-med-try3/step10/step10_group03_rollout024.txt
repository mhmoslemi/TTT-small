```python
        hi = 1.0
        for _ in range(tern_iters):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            f1 = np.max(B + coeff * t1)
            f2 = np.max(B + coeff * t2)
            if f1 < f2:
                hi = t2
            else:
                lo = t1
        
        new_val = (lo + hi) / 2.0
        
        if new_val != old_val:
            h[i] = new_val
            c_vec += coeff * (new_val - old_val)
    
    return h
```

This is clean and efficient. The c_vec is updated incrementally, so I don't need to recompute it from scratch each sweep.

Now, one issue: the sum constraint. After modifying h[i], the sum changes. I should project periodically. But if I project in the middle of a sweep, it changes many coordinates and the c_vec becomes stale. So I should only project between sweeps.

f(t) = max(B + coeff*t) + μ*(S_old - old_val + t - n/2)^2

where S_old = sum(h) before this update.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent_sweep(h, n, c_vec, order, tern_iters=30):
    n_c = 2 * n - 1
    k_arr = np.arange(-(n - 1), n)
    mm_arr = k_arr + (n - 1)
    neg_mask = k_arr < 0
    pos_mask = k_arr >= 0
    m_arr = -k_arr

    for i in order:
        old_val = h[i]

        coeff = np.zeros(n_c)
        v1 = pos_mask & (i + k_arr < n)
        if np.any(v1):
            coeff[mm_arr[v1]] += (1.0 - h[i + k_arr[v1]])
        v2 = pos_mask & (i >= k_arr) & (k_arr > 0)
        if np.any(v2):
            coeff[mm_arr[v2]] += -h[i - k_arr[v2]]
        v3 = neg_mask & (i >= m_arr)
        if np.any(v3):
            coeff[mm_arr[v3]] += (1.0 - h[i - m_arr[v3]])
        v4 = neg_mask & (i + m_arr < n)
        if np.any(v4):
            coeff[mm_arr[v4]] += -h[i + m_arr[v4]]

        B = c_vec - coeff * old_val

        lo = 0.0
        hi = 1.0
        for _ in range(tern_iters):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            f1 = np.max(B + coeff * t1)
            f2 = np.max(B + coeff * t2)
            if f1 < f2:
                hi = t2
            else:
                lo = t1

        new_val = (lo + hi) / 2.0
        if new_val != old_val:
            h[i] = new_val
            c_vec += coeff * (new_val - old_val)

    return h

def coord_descent_with_penalty(h, n, c_vec, order, tern_iters=30, mu=0.01):
    n_c = 2 * n - 1
    k_arr = np.arange(-(n - 1), n)
    mm_arr = k_arr + (n - 1)
    neg_mask = k_arr < 0
    pos_mask = k_arr >= 0
    m_arr = -k_arr
    target_sum = n / 2.0

    for i in order:
        old_val = h[i]
        S = h.sum() - old_val

        coeff = np.zeros(n_c)
        v1 = pos_mask & (i + k_arr < n)
        if np.any(v1):
            coeff[mm_arr[v1]] += (1.0 - h[i + k_arr[v1]])
        v2 = pos_mask & (i >= k_arr) & (k_arr > 0)
        if np.any(v2):
            coeff[mm_arr[v2]] += -h[i - k_arr[v2]]
        v3 = neg_mask & (i >= m_arr)
        if np.any(v3):
            coeff[mm_arr[v3]] += (1.0 - h[i - m_arr[v3]])
        v4 = neg_mask & (i + m_arr < n)
        if np.any(v4):
            coeff[mm_arr[v4]] += -h[i + m_arr[v4]]

        B = c_vec - coeff * old_val
        a_pen = mu
        b_pen = 2.0 * mu * (S - target_sum)
        c_pen = mu * (S - target_sum) ** 2

        lo = 0.0
        hi = 1.0
        for _ in range(tern_iters):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            f1 = np.max(B + coeff * t1) + a_pen * t1 * t1 + b_pen * t1 + c_pen
            f2 = np.max(B + coeff * t2) + a_pen * t2 * t2 + b_pen * t2 + c_pen
            if f1 < f2:
                hi = t2
            else:
                lo = t1

        new_val = (lo + hi) / 2.0
        if new_val != old_val:
            h[i] = new_val
            c_vec += coeff * (new_val - old_val)

    return h

def active_shift_order(c_vec, n, h, rng, top_frac=0.05):
    max_val = c_vec.max()
    threshold = max_val - top_frac * max_val
    active = np.where(c_vec >= threshold)[0]

    active_coords = np.zeros(n, dtype=bool)
    for mm in active:
        k_val = mm - (n - 1)
        if k_val >= 0:
            active_coords[0:min(n, n - k_val)] = True
            if k_val > 0:
                active_coords[k_val:n] = True
        else:
            m_val = -k_val
            active_coords[m_val:n] = True
            active_coords[0:n - m_val] = True

    active_idx = np.where(active_coords)[0]
    other_idx = np.where(~active_coords)[0]
    rng.shuffle(active_idx)
    rng.shuffle(other_idx)

    n_active_use = min(len(active_idx), int(n * 0.7))
    n_other_use = min(len(other_idx), n - n_active_use)
    order = np.concatenate([active_idx[:n_active_use], other_idx[:n_other_use]])
    rng.shuffle(order)
    return order

def random_perturb(h, n, rng, n_perturb=5, magnitude=0.1):
    h_new = h.copy()
    indices = rng.choice(n, size=n_perturb, replace=False)
    for idx in indices:
        delta = rng.uniform(-magnitude, magnitude)
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
    return project_feasible(h_new, n)

def block_flip(h, n, rng):
    h_new = h.copy()
    block_len = rng.integers(10, n // 4)
    start = rng.integers(0, n - block_len)
    h_new[start:start + block_len] = 1.0 - h_new[start:start + block_len]
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        rng0 = np.random.default_rng(seed)
        h_init = rng0.uniform(0, 1, 500)
        h_init = project_feasible(h_init, 500)
        n_orig = 500

    n = n_orig
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    rng = np.random.default_rng(seed)
    c_vec = compute_c5_vec(h, n)

    # Phase 1: Main coordinate descent at original resolution
    sweep_count = 0
    no_improve_count = 0
    while time.time() < deadline - 80:
        if sweep_count % 50 == 0:
            order = active_shift_order(c_vec, n, h, rng, top_frac=0.08)
        else:
            order = rng.permutation(n)

        h = coord_descent_sweep(h, n, c_vec, order, tern_iters=25)
        h = project_feasible(h, n)
        c_vec = compute_c5_vec(h, n)
        c = np.max(c_vec)

        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sweep_count += 1

        if no_improve_count > 80:
            h = random_perturb(h, n, rng, n_perturb=10, magnitude=0.2)
            c_vec = compute_c5_vec(h, n)
            no_improve_count = 0

    # Phase 2: Try penalty-based coordination for cleaner sum constraint
    if time.time() < deadline - 50:
        h = best_h.copy()
        c_vec = compute_c5_vec(h, n)
        for _ in range(20):
            if time.time() > deadline - 50:
                break
            order = rng.permutation(n)
            h = coord_descent_with_penalty(h, n, c_vec, order, tern_iters=25, mu=0.05)
            h = project_feasible(h, n)
            c_vec = compute_c5_vec(h, n)
            c = np.max(c_vec)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Phase 3: Upsample to higher resolution and refine
    if time.time() < deadline - 30:
        n2 = n * 2
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)

        if c2 < best_c + 0.005:
            c_vec2 = compute_c5_vec(h2, n2)
            for _ in range(30):
                if time.time() > deadline - 10:
                    break
                order = rng.permutation(n2)
                h2 = coord_descent_sweep(h2, n2, c_vec2, order, tern_iters=20)
                h2 = project_feasible(h2, n2)
                c_vec2 = compute_c5_vec(h2, n2)
                c = np.max(c_vec2)
                if c < best_c:
                    best_c = c
                    best_h = h2.copy()
                    n = n2

    # Final check
    best_h = project_feasible(best_h, len(best_h))
    final_c = compute_c5(best_h, len(best_h))
    if final_c < best_c:
        best_c = final_c

    return best_h, best_c, len(best_h)
```
