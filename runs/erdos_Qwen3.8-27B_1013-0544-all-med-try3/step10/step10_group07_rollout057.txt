            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_from_continuous(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_delta_pos(h, n, i):
    """Compute delta contribution of position i going 0->1 in the correlation."""
    delta = np.zeros(2 * n - 1)
    s = np.arange(-(n - 1), n)
    im = i + s
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - s
    msk2 = (imn >= 0) & (imn < n)
    delta[msk2] -= h[imn[msk2]]
    return delta

def greedy_local_search(h, n, deadline, max_sweeps=50000, rng=None):
    """Fast greedy binary local search - accept first improving swap found."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        rng.shuffle(zeros)
        
        for i in zeros:
            if time.time() > deadline:
                break
            delta_i = compute_delta_pos(h, n, i)
            base = corr + delta_i
            for j in ones:
                if j == i:
                    continue
                # Effect of j going 1->0
                s = np.arange(-(n - 1), n)
                jm = j + s
                msk = (jm >= 0) & (jm < n)
                dj = -(1.0 - h[jm[msk]])
                jm2 = j - s
                msk2 = (jm2 >= 0) & (jm2 < n)
                dj_at = np.zeros(2 * n - 1)
                dj_at[msk] = dj
                dj_at[msk2] += h[jm2[msk2]]
                
                # Cross terms
                new_corr = base + dj_at
                new_corr[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                new_corr[k1] += 1.0
                k2 = n - 1 + (i - j)
                new_corr[k2] += 1.0
                
                nc = new_corr.max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = new_corr
                    cur_c = nc
                    improved = True
                    break
            if improved:
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
    
    return best_h, best_c

def annealed_local_search(h, n, deadline, T0=0.005, cool_rate=0.99995, max_sweeps=100000, rng=None):
    """Binary local search with simulated annealing."""
    if rng is None:
        rng = np.random.RandomState(42)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    T = T0
    
    target = n // 2
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        i = rng.choice(zeros)
        j = rng.choice(ones)
        if i == j:
            continue
        
        delta_i = compute_delta_pos(h, n, i)
        base = corr + delta_i
        s = np.arange(-(n - 1), n)
        jm = j + s
        msk = (jm >= 0) & (jm < n)
        dj_at = np.zeros(2 * n - 1)
        dj_at[msk] = -(1.0 - h[jm[msk]])
        jm2 = j - s
        msk2 = (jm2 >= 0) & (jm2 < n)
        dj_at[msk2] += h[jm2[msk2]]
        
        new_corr = base + dj_at
        new_corr[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        new_corr[k1] += 1.0
        k2 = n - 1 + (i - j)
        new_corr[k2] += 1.0
        
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        
        if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        T *= cool_rate
    
    return best_h, best_c

def continuous_optimize(h, n, deadline, alpha_seq=None, maxiter_each=500):
    """Optimize continuous h using SLSQP with smooth min-max."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    if alpha_seq is None:
        alpha_seq = [1000, 5000, 20000, 100000, 500000, 2000000]
    
    cur = h.copy()
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            sh = alpha * (c - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        def cons_fun(x):
            return x.sum() - n / 2.0
        
        try:
            res = minimize(obj, cur, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cons_fun}],
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            cur = project_feasible(res.x, n)
            c = compute_c5(cur, n)
            if c < best_c:
                best_c = c
                best_h = cur.copy()
        except Exception:
            pass
    
    return best_h, best_c

def try_resolution(h_best, n_best, c_best, n_try, deadline, rng):
    """Try optimizing at a different resolution."""
    if n_try == n_best:
        return h_best, c_best, n_best
    
    xo = np.linspace(0, 2, n_best)
    xn = np.linspace(0, 2, n_try)
    hh = np.interp(xn, xo, h_best)
    hh = project_feasible(hh, n_try)
    
    # Quick continuous optimization
    t_sub = deadline - 30
    hh, ch = continuous_optimize(hh, n_try, t_sub, alpha_seq=[5000, 50000, 500000], maxiter_each=300)
    
    # Binary local search
    hb = binary_from_continuous(hh, n_try)
    t_bin = deadline - 10
    hb, cb = greedy_local_search(hb, n_try, t_bin, max_sweeps=30000, rng=rng)
    
    if cb < c_best:
        return hb, cb, n_try
    if ch < c_best:
        return project_feasible(hh, n_try), ch, n_try
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    
    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Continuous optimization (8% of time)
    p1 = t0 + 0.08 * (budget_s - 5)
    h_cont, c_cont = continuous_optimize(best_h, n, p1,
                                          alpha_seq=[2000, 10000, 50000, 200000],
                                          maxiter_each=800)
    if c_cont < best_c:
        best_c = c_cont
        best_h = h_cont.copy()
    
    # Phase 2: Best-swap local search (35% of time)
    p2 = t0 + 0.43 * (budget_s - 5)
    hb = binary_from_continuous(best_h, n)
    
    # First: greedy local search to convergence
    hb, cb = greedy_local_search(hb, n, min(p2, t0 + 0.25 * (budget_s - 5)),
                                  max_sweeps=50000, rng=rng)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Then: simulated annealing to escape local minima
    p2a = min(p2, t0 + 0.32 * (budget_s - 5))
    if time.time() < p2a:
        hb2, cb2 = annealed_local_search(best_h, n, p2a, T0=0.003, cool_rate=0.99997,
                                          max_sweeps=200000, rng=rng)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    
    # More greedy after annealing
    if time.time() < p2 - 5:
        hb3, cb3 = greedy_local_search(best_h, n, p2 - 5, max_sweeps=50000, rng=rng)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
    
    # Phase 3: Multi-resolution exploration (20% of time)
    p3 = t0 + 0.63 * (budget_s - 5)
    for n_try in [200, 225, 275, 300, 325, 248, 252, 256]:
        if time.time() > p3 - 40:
            break
        if n_try == n:
            continue
        t_sub = min(p3 - 20, time.time() + 60)
        best_h, best_c, n = try_resolution(best_h, n, best_c, n_try, t_sub, rng)
    
    # Phase 4: Extended binary search on best (25% of time)
    p4 = t0 + 0.88 * (budget_s - 5)
    # Round 1: Extended greedy
    hb, cb = greedy_local_search(best_h, n, p4, max_sweeps=100000, rng=rng)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    
    # Round 2: Annealing with lower temperature
    if time.time() < p4 - 10:
        hb2, cb2 = annealed_local_search(best_h, n, p4 - 10, T0=0.001,
                                          cool_rate=0.99998, max_sweeps=100000, rng=rng)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
    
    # Phase 5: Final polish (remaining time)
    # Continuous optimization from best binary
    if time.time() < dl - 15:
        h_cont2, c_cont2 = continuous_optimize(best_h, n, dl - 15,
                                                alpha_seq=[100000, 500000, 2000000],
                                                maxiter_each=1000)
        if c_cont2 < best_c:
            best_c = c_cont2
            best_h = h_cont2.copy()
    
    # Final binary search
    if time.time() < dl - 5:
        hb_final, cb_final = greedy_local_search(best_h, n, dl - 5,
                                                  max_sweeps=50000, rng=rng)
        if cb_final < best_c:
            best_c = cb_final
            best_h = hb_final.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
