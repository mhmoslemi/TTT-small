import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def c5_grad(c, x, n, weights=None):
    """Gradient of smooth max of C(k) w.r.t. x"""
    dx_val = 2.0 / n
    n_lags = len(c)
    if weights is None:
        weights = np.ones(n_lags) / n_lags
    grad = np.zeros(n)
    for mm in range(n_lags):
        w = weights[mm] * dx_val
        if abs(w) < 1e-15:
            continue
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += w * (1.0 - x[k:n])
            grad[k:n] += w * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += w * (1.0 - x[0:nv])
            grad[0:nv] += w * (-x[kk:n])
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = c5_grad(c, x, n, w)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    full_w = np.zeros(len(c))
    full_w[active] = w
    grad = c5_grad(c, x, n, full_w)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def compute_grad_full(c, x, n, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return weighted_grad(w, x, n)

def weighted_grad(w, x, n):
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx_val
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_full_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    g = compute_grad_full(c, x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_active(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_active_fn(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    full_w = np.zeros(len(c))
    full_w[active] = w
    g = weighted_grad(full_w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def obj_single_lag(x, n, lag_idx, mu):
    c = compute_c5_vec(x, n)
    val = c[lag_idx]
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def grad_single_lag(x, n, lag_idx, mu):
    c = compute_c5_vec(x, n)
    full_w = np.zeros(len(c))
    full_w[lag_idx] = 1.0
    g = weighted_grad(full_w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_staggered(n, offset_pattern):
    h = np.zeros(n)
    period = offset_pattern
    for i in range(n):
        if (i % period) < period // 2:
            h[i] = 1.0
    return project_feasible(h, n)

def lbfgs_phase(h, n, deadline, alphas, mu, maxits=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxits is not None and i < len(maxits):
            maxit = maxits[i]
        else:
            maxit = max(200, int(remaining * 3))
        try:
            res = minimize(obj_full, best_h, jac=grad_full_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (4, 50000, 6),
        (6, 200000, 5),
        (8, 1000000, 5),
        (10, 5000000, 4),
        (14, 10000000, 4),
        (20, 30000000, 3),
        (30, 100000000, 3),
    ]
    for top_k, alpha, rounds in configs:
        if time.time() > deadline - 10:
            break
        for rnd in range(rounds):
            if time.time() > deadline - 8:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(5e-9, 0.0003 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                active = np.array(np.argsort(-c_vec[active])[:top_k])
            if len(active) == 0:
                break
            maxit = max(200, int((deadline - time.time()) * 1.2))
            try:
                res = minimize(obj_active, best_h, jac=grad_active_fn,
                              args=(n, active, alpha, 0.0005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def gradient_pursuit(h, n, deadline, step_scales=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_scales is None:
        step_scales = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    bounds = [(0.0, 1.0)] * n
    for scale in step_scales:
        improved_count = 0
        for _ in range(5000):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            for idx in active:
                if time.time() > deadline:
                    break
                try:
                    res = minimize(obj_single_lag, best_h, jac=grad_single_lag,
                                  args=(n, idx, 0.01), method='L-BFGS-B',
                                  bounds=bounds,
                                  options={'maxiter': 50, 'ftol': 1e-20,
                                           'gtol': 1e-14, 'maxfun': 200,
                                           'maxls': 10, 'maxcor': 10})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        best_h = h_new
                        improved_count += 1
                    elif c_new <= best_c + 1e-15:
                        best_h = h_new
                except Exception:
                    pass
        if improved_count == 0:
            break
    return best_h, best_c

def pairwise_local_search(h, n, deadline, radius=None, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = 30
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    n_lags = 2 * n - 1
    for d in deltas:
        improved_this_delta = True
        while improved_this_delta and time.time() < deadline:
            improved_this_delta = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Move mass from i to j
                    if best_h[i] >= d and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved_this_delta = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Move mass from j to i
                    if best_h[j] >= d and best_h[i] + d <= 1.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved_this_delta = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
    return best_h, best_c

def binary_swap_search(h, n, deadline, max_iters=100000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones_mask = best_h > 0.99
    zeros_mask = best_h < 0.01
    iters = 0
    while iters < max_iters and time.time() < deadline:
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        rng = np.random.RandomState(iters)
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        best_h[i] = 0.0
        best_h[j] = 1.0
        c = compute_c5(best_h, n)
        if c < best_c - 1e-14:
            best_c = c
            ones_mask[i] = False
            ones_mask[j] = True
            zeros_mask[i] = True
            zeros_mask[j] = False
        else:
            best_h[i] = 1.0
            best_h[j] = 0.0
        iters += 1
    return best_h, best_c

def perturb_and_optimize(h, n, deadline, perturb_scale=0.05):
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    h_pert = h.copy()
    mask = rng.rand(n) < 0.1
    h_pert[mask] += rng.randn(mask.sum()) * perturb_scale
    h_pert = project_feasible(h_pert, n)

    h_opt, c_opt = lbfgs_phase(h_pert, n, deadline, [1000, 10000, 100000], 0.01, [200, 300, 400])
    return h_opt, c_opt
