import numpy as np
import time

def fft_correlate_max(h):
    """Compute max of full correlation of h and 1-h using FFT."""
    n = len(h)
    M = 1
    while M < 2 * n - 1:
        M *= 2
    H = np.fft.rfft(h, M)
    Oh = np.fft.rfft(1.0 - h, M)
    corr = np.fft.irfft(H * np.conj(Oh), M)
    return corr.max()

    dx = 2.0 / n
    return dx * fft_correlate_max(h)

    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 1e-15:
                h += (diff / tot) * h

        tm[i] = tm[i >> 1] ^ (i & 1)

import numpy as np
from scipy.optimize import minimize
import time

def fft_corr_max(h):
    """Compute max of full linear correlation of h and (1-h) using FFT."""
    n = len(h)
    # Need M >= 2n-1 for linear correlation
    M = 1
    while M < 2 * n - 1:
        M <<= 1
    H = np.fft.rfft(h, M)
    Oh = np.fft.rfft(1.0 - h, M)
    corr = np.fft.irfft(H * np.conj(Oh), M)
    return float(corr.max())

def c5_of(h):
    """Compute C5 bound for given h."""
    n = len(h)
    return (2.0 / n) * fft_corr_max(h)

def project_feasible(h, n):
    """Project h to be feasible: 0<=h<=1, sum(h)=n/2."""
    target = n / 2.0
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += (diff / s) * room
        else:
            s = h.sum()
            if s > 1e-14:
                h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def make_thue_morse(n):
    """Create binary Thue-Morse based initialization."""
    h = np.zeros(n)
    # Thue-Morse: t(n) = popcount(n) mod 2
    for i in range(n):
        t = i
        c = 0
        while t:
            c ^= 1
            t >>= 1
        h[i] = 1.0 if c == 0 else 0.0
    # Adjust to have exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        # Remove some ones
        ones = np.where(h == 1.0)[0]
        for i in range(n_ones - target):
            h[ones[i]] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        for i in range(target - n_ones):
            h[zeros[i]] = 1.0
    return h

def upsample(h, new_n):
    """Upsample h to new_n by nearest-neighbor interpolation, preserving sum."""
    n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        src = int(i * n / new_n)
        h_new[i] = h[src]
    # Adjust to have exactly new_n/2 ones
    n_ones = int(np.sum(h_new > 0.5))
    target = new_n // 2
    if n_ones > target:
        ones = np.where(h_new > 0.5)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        for i in range(n_ones - target):
            h_new[ones[i]] = 0.0
    elif n_ones < target:
        zeros = np.where(h_new <= 0.5)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        for i in range(target - n_ones):
            h_new[zeros[i]] = 1.0
    return h_new
