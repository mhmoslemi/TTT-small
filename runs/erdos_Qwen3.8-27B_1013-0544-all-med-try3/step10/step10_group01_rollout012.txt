def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_smooth(h, n, k=3, alpha=100000):
    c = compute_c5_vec(h, n)
    # Sort and take top k
    sorted_c = np.sort(c)[::-1]
    top = sorted_c[:k]
    # Smooth max of top k
    m = top.max()
    sh = alpha * (top - m)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def topk_grad(h, n, k=3, alpha=100000):
    c = compute_c5_vec(h, n)
    idx = np.argsort(c)[::-1][:k]
    top = c[idx]
    m = top.max()
    sh = alpha * (top - m)
    w = np.exp(sh)
    w /= w.sum()

    grad = np.zeros(n)

            grad[kk2:n] += wm * (1.0 - h[0:nv])

For the correlation: c[k_idx] = Σ_i h[i] * (1 - h[i + k_idx - (n-1)])
where the sum is over valid i (0 ≤ i < n and 0 ≤ i + k_idx - (n-1) < n).

Let shift = k_idx - (n-1).
If shift ≥ 0: valid i are 0 to n-1-shift, so c = Σ_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
If shift < 0: let s = -shift, valid i are s to n-1, so c = Σ_{i=s}^{n-1} h[i]*(1-h[i-s])

dc/dh[j] for shift ≥ 0:
- For j in [0, n-1-shift]: dc/dh[j] = (1 - h[j+shift]) [from h[j] term]
- For j in [shift, n-1]: dc/dh[j] = -h[j-shift] [from -(1-h[j-shift])... wait]

Let me redo this. c = Σ_{i=0}^{n-1-shift} h[i] * (1 - h[i+shift])

dc/dh[j] = Σ_i [δ_{i,j}(1-h[i+shift]) + h[i] * (-δ_{i+shift,j})]
= (1-h[j+shift]) * [j ≤ n-1-shift] - h[j-shift] * [j-shift ≥ 0 and j-shift ≤ n-1-shift]
= (1-h[j+shift]) * [0 ≤ j ≤ n-1-shift] - h[j-shift] * [shift ≤ j ≤ n-1]

For shift < 0, let s = -shift > 0:
c = Σ_{i=s}^{n-1} h[i] * (1 - h[i-s])

dc/dh[j] = Σ_i [δ_{i,j}(1-h[i-s]) + h[i] * (-δ_{i-s,j})]
= (1-h[j-s]) * [s ≤ j ≤ n-1] - h[j+s] * [j+s ≥ s and j+s ≤ n-1]
= (1-h[j-s]) * [s ≤ j ≤ n-1] - h[j+s] * [0 ≤ j ≤ n-1-s]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    s = np.sum(np.exp(sh))
    return float(m + np.log(s) / alpha)

def smooth_max_grad(c, alpha, x, n, top_k=None):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w_sum = w.sum()
    if top_k is not None and top_k < len(w):
        idx_threshold = np.partition(w, len(w) - top_k)[len(w) - top_k]
        w[w < idx_threshold] = 0.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            return np.zeros(n)
    w = w / w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for ii in active:
        kk = int(ii) - (n - 1)
        wm = w[ii] * dx
        if kk >= 0:
            nv = n - kk
            grad[0:nv] += wm * (1.0 - x[kk:n])
            grad[kk:n] += wm * (-x[0:nv])
        else:
            kk2 = -kk
            nv = n - kk2
            grad[kk2:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk2:n])
    return grad

def obj_full(x, n, alpha, top_k=None):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def jac_full(x, n, alpha, top_k=None):
    c = compute_c5_vec(x, n)
    return smooth_max_grad(c, alpha, x, n, top_k)

def obj_topk_conc(x, n, k, alpha):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[::-1][:k]
    top = c[idx]
    m = top.max()
    sh = alpha * (top - m)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def jac_topk_conc(x, n, k, alpha):
    c = compute_c5_vec(x, n)
    idx = np.argsort(c)[::-1][:k]
    top = c[idx]
    m = top.max()
    sh = alpha * (top - m)
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, ii in zip(w, idx):
        kk = int(ii) - (n - 1)
        wm = wi * dx
        if kk >= 0:
            nv = n - kk
            grad[0:nv] += wm * (1.0 - x[kk:n])
            grad[kk:n] += wm * (-x[0:nv])
        else:
            kk2 = -kk
            nv = n - kk2
            grad[kk2:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk2:n])
    return grad

def make_beatty(n, slope):
    h = np.zeros(n)
    n_ones = n // 2
    used = np.zeros(n, dtype=bool)
    count = 0
    for k in range(1, 6 * n):
        pos = int(k * slope) % n
        if not used[pos]:
            used[pos] = True
            h[pos] = 1.0
            count += 1
            if count >= n_ones:
                break
    return project_feasible(h, n)

def make_random(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def downsample(h_old, n_old, n_new):
    if n_new == n_old:
        return h_old.copy()
    if n_new < n_old:
        ratios = np.linspace(0, n_old, n_new + 1)
        h_new = np.zeros(n_new)
        for i in range(n_new):
            lo = int(ratios[i])
            hi = int(ratios[i + 1])
            if hi > lo:
                h_new[i] = np.mean(h_old[lo:hi])
            else:
                h_new[i] = h_old[lo]
        return project_feasible(h_new, n_new)
    else:
        pos = np.linspace(0, 2.0, n_new)
        pos_old = np.linspace(0, 2.0, n_old)
        h_new = np.interp(pos, pos_old, h_old)
        return project_feasible(h_new, n_new)

def run_lbfgs(h, n, deadline, alpha, maxit=None, top_k=None):
    bounds = [(0.0, 1.0)] * n
    if maxit is None:
        maxit = max(200, int((deadline - time.time()) * 3))
    try:
        res = minimize(obj_full, h, jac=jac_full,
                      args=(n, alpha, top_k), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-22,
                               'gtol': 1e-15, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 60})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run_topk_lbfgs(h, n, deadline, k, alpha, maxit=None):

def active_set_optimize(h, n, deadline, configs=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if configs is None:
        configs = [(3, 50000, 4), (5, 200000, 4), (8, 1000000, 3), (12, 5000000, 3)]
    for k, alpha, rounds in configs:
        for rnd in range(rounds):
            if time.time() > deadline - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0005 * max_val)
            active_idx = np.where(c_vec >= threshold)[0]
            if len(active_idx) > min(k, len(active_idx)):
                active_idx = np.array(np.argsort(-c_vec[active_idx])[:k])
            if len(active_idx) == 0:
                break
            maxit = max(300, int((deadline - time.time()) * 1.5))
            h_new, c_new = run_topk_lbfgs(best_h, n, deadline, len(active_idx), alpha, maxit)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    return best_h, best_c

def grad_perturbation(h, n, deadline, n_steps=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) > 8:
            active = np.array(np.argsort(-c_vec[active])[:8])
        grad = np.zeros(n)
        for idx in active:
            kk = int(idx) - (n - 1)
            if kk >= 0:
                nv = n - kk
                grad[0:nv] += (1.0 - best_h[kk:n]) * dx_val
                grad[kk:n] += (-best_h[0:nv]) * dx_val
            else:
                kk2 = -kk
                nv = n - kk2
                grad[kk2:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk2:n]) * dx_val
        grad /= len(active)
        pos_mask = (grad > 1e-12) & (best_h > 1e-12)
        neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        best_gain = 0
        best_i = -1
        best_j = -1
        for i in pos_idx[:30]:
            for j in neg_idx[:30]:
                g = grad[i] - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        for amt in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if amt > best_h[best_i] or amt > 1.0 - best_h[best_j]:
                continue
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                break
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def pair_neighbor_search(h, n, deadline, radius=30, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def slsqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def slsqp_eq_fun(x, n):

def slsqp_eq_jac(x, n):

            if c6 < best_c:
                best_c = c6
                best_h = h6.copy()

    # Phase 7: Final pair search
    if time.time() < deadline - 10:
        h7, c7 = pair_neighbor_search(best_h, best_n, deadline - 3, radius=25,
                                      deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        if c7 < best_c:
            best_c = c7
            best_h = h7.copy()

    # Final: one more quick L-BFGS with very high alpha
    if time.time() < deadline - 3:
        h_final, c_final = run_lbfgs(best_h, best_n, deadline - 1, 50000000, maxit=300, top_k=5)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
