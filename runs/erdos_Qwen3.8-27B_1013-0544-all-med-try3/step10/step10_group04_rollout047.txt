∫h(x)h(x+k)dx ≈ ∫(h⁰(x)h(x+k) + h(x)h⁰(x+k) - h⁰(x)h⁰(x+k))dx

This gives a linear constraint in h (at each iterate), leading to a cutting-plane or successive linearization approach.

At each iteration:
1. Given current h⁰, form the linearized constraints
2. Solve the resulting LP: min t s.t. linearized constraints, 0≤h≤1, ∫h=1
3. Update h⁰ and repeat

This is essentially a DVP (Dinkelbach-type) or cutting-plane method for the non-convex problem.

The LP at each step would have n + 2n-1 + 1 variables (h[0..n-1], t, and maybe some slacks) and 2n-1 + 3 constraints. For n=1000, that's 3001 variables and 2004 constraints. Solvable by CBC/GLPK in reasonable time.

But wait, the linearization might not be tight enough. Let me think...

Actually, a simpler approach: since h(1-h) is concave in h (for h in [0,1]), the function h(1-h(x+k)) is NOT concave in h (it's bilinear). So the problem is genuinely non-convex.

However, we can use the fact that at the optimum, many of the h values will be at the boundary (0 or 1) due to the max structure. This suggests a combinatorial structure.

Let me try a different practical approach:

**Hybrid approach:**
1. Start with the parent construction (n=500)
2. Quick L-BFGS to improve it
3. Upsample to n=1000
4. At n=1000, alternate between:
   a. L-BFGS optimization (continuous)
   b. "Snap to binary" + local search (combinatorial)
   c. SLSQP with equality constraint
5. Try multiple random restarts from different structured initializations
6. Final coordinate-wise refinement

For the "snap to binary + local search" step:
- Round h to {0,1} (with the sum constraint)
- Then do a local search: flip individual bits or swap pairs of bits
- Accept if C₅ improves
- This is like a simulated annealing / hill climbing on the binary solutions

The binary local search is fast (each evaluation is O(n²) for the correlate, but we can update incrementally when flipping one bit).

When we flip h[j] from 0 to 1, how does c[k] change?
- c[n-1+s] changes by: (1-h[j+s] if j<=n-1-s) - h[j+s] (the old value was h[j]*(1-h[j+s])=0, new is 1*(1-h[j+s]))
  So delta = (1-h[j+s]) for the s where j is in range, and similarly for the -s shift.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def binary_local_search(h, n, deadline, max_iter=5000):
    """Local search on binary patterns."""
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    # Fix sum
    s = h_b.sum()
    if s < target:
        order = np.argsort(h)[::-1]  # largest fractional values first
        for i in order:
            if h_b[i] < 1.0:
                h_b[i] = 1.0
                if h_b.sum() >= target:
                    break
    elif s > target:
        order = np.argsort(h)
        for i in order:
            if h_b[i] > 0.0:
                h_b[i] = 0.0
                if h_b.sum() <= target:
                    break
    
    best_c = compute_c5(h_b, n)
    best_h = h_b.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Try flipping individual pairs
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Randomly try a swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        c = compute_c5(h_b, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_b.copy()
        else:
            h_b[j] = 0.0
            h_b[i] = 1.0
    
    return best_h, best_c

def run_lbfgs(h, n, deadline, alphas=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [(100, 100.0), (500, 500.0), (2000, 2000.0),
                  (5000, 5000.0), (10000, 10000.0), (30000, 30000.0)]
    for alpha, mu in alphas:
        if time.time() > deadline - 10:

        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 10000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

    def sq_fn(x):
        return float(x.sum() - n / 2.0)

        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',

    except Exception:
        pass

def structured_initial(n, seed=0):
    """Create a structured initial point using a periodic pattern."""
    # Try different period lengths
    rng = np.random.RandomState(seed)
    period = rng.choice([3, 4, 5, 6, 7, 8, 10, 12])

    if len(h) < n:
        h = np.concatenate([h, np.ones(n - len(h))])
    return project_feasible(h, n)
