import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n

        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)

        h0 = np.full(500, 0.5)

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Phase 1: Quick optimization at original resolution (n=500)
    p1_end = t0 + 80
    h1, c1 = run_lbfgs(best_h, best_n, p1_end,

        mu=0.01, maxiter_per=[1000, 1500, 2000, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Upsample to n=1000 and optimize
    n1000 = 1000
    h1000 = upsample(best_h, best_n, n1000)
    c1000 = compute_c5(h1000, n1000)
    if c1000 < best_c:
        best_c = c1000
        best_h = h1000
        best_n = n1000

    p2_end = min(t0 + 250, deadline - 300)
    h2, c2 = run_lbfgs(h1000, n1000, p2_end,

        mu=0.005, maxiter_per=[1500, 2000, 2500, 3000, 3000, 3000])
    if c2 < best_c:
        best_c = c2
        best_h = h2
        best_n = n1000

    # Phase 3: Continue at n=1000 with higher alpha
    p3_end = min(t0 + 450, deadline - 200)
    h3, c3 = run_lbfgs(best_h, best_n, p3_end,
        alphas=[50000, 200000, 1000000, 5000000],
        mu=0.002, maxiter_per=[3000, 4000, 4000, 4000])
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Active set refinement at n=1000
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 120, deadline - 150)
        h4, c4 = run_active_refine(best_h, best_n, p4_end, top_k=10,
            alphas=[20000, 100000, 500000, 2000000, 10000000])
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Try n=2000
    n2000 = 2000
    if time.time() < deadline - 250:
        h2000 = upsample(best_h, best_n, n2000)
        c2000 = compute_c5(h2000, n2000)
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
            best_n = n2000

        p5_end = min(time.time() + 200, deadline - 150)
        h5, c5 = run_lbfgs(h2000, n2000, p5_end,
            alphas=[50, 200, 1000, 5000, 50000, 500000],
            mu=0.003, maxiter_per=[1500, 2000, 2500, 3000, 3000, 3000])
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = n2000

        # Active set at n=2000
        if time.time() < deadline - 100:
            p5b_end = min(time.time() + 80, deadline - 80)
            h5b, c5b = run_active_refine(best_h, best_n, p5b_end, top_k=8,
                alphas=[50000, 200000, 1000000, 5000000])
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: Perturbation restarts at best resolution
    if time.time() < deadline - 120:
        p6_end = min(time.time() + 80, deadline - 80)
        h6, c6 = perturb_and_reopt(best_h, best_n, p6_end,
                                    n_perturbs=3, noise_scales=[0.003, 0.002, 0.001])
        if c6 < best_c:

            best_h = h6

    # Phase 7: Final active set + L-BFGS
    if time.time() < deadline - 60:
        p7_end = min(time.time() + 40, deadline - 40)
        h7, c7 = run_active_refine(best_h, best_n, p7_end, top_k=6,

            p7b_end = min(time.time() + 15, deadline - 15)
            h7b, c7b = run_lbfgs(best_h, best_n, p7b_end,
                alphas=[5000000, 20000000],
                mu=0.001, maxiter_per=[2000, 2000])
            if c7b < best_c:
                best_c = c7b
                best_h = h7b

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
