```python
def logsumexp_smoother(corr, alpha):
    """Smooth approximation to max of corr using log-sum-exp."""
    m = np.max(corr)
    shifted = alpha * (corr - m)
    sm = np.max(shifted)
    lse = (m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha)
    return lse

    weights = np.exp(alpha_shift - sm)
```

```python
def gradient_of_max_vec(h, n, top_k=50, alpha=500):
    corr = np.correlate(h, 1 - h, mode='full')
    alpha_shift = alpha * (corr - corr.max())
    sm = alpha_shift.max()
    weights = np.exp(alpha_shift - sm)
    total_w = weights.sum()
    if total_w < 1e-30:
        return np.zeros(n)
    weights /= total_w
    
    # Only use top_k
    top_idx = np.argsort(weights)[-top_k:]
    w_sum = weights[top_idx].sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for m in top_idx:
        w = weights[m] / w_sum
        s = m - (n - 1)
        # Contribution from (1-h[l+s]): positive
        if s >= 0:
            # l from 0 to n-1-s
            if n - 1 - s >= 0:
                grad[:n-s] += w * (1 - h[s:])
        else:
            # l from -s to n-1
            req = -s
            grad[req:] += w * (1 - h[:n-req])
        # Contribution from -h[l-s]: negative
        if s >= 0:
            # l from s to n-1
            if s <= n-1:
                grad[s:] -= w * h[:n-s]
        else:
            # l from 0 to n-1+s
            req = -s
            grad[:n-req] -= w * h[req:]
    return grad
```

    """Compute delta corr when h[i]:0→1, h[j]:1→0. O(n) using numpy."""
    delta = np.zeros(2*n-1)
    center = n - 1
    
    # i flip (0 to 1):
    # Adds (1 - h_new[i+s]) at positions where i+s is valid
    # h_new[i+s] = 0 if i+s==j, else h[i+s]
    lo1 = max(0, -i)  # s >= 0 - i... wait s can be negative

    s_lo = -i
    s_hi = n - 1 - i
    s = np.arange(s_lo, s_hi + 1)
    idx = center + s
    vals = 1.0 - h[i + s]
    # Cross: where i+s == j
    has_cross = (i + s == j)
    vals[has_cross] = 1.0  # 1 - h_new[j] = 1 - 0 = 1
    delta[idx] += vals
    
    # Removes h[i-s] at positions where i-s is valid (the (1-h[i]) went from 1 to 0)
    # i-s in [0,n) means s in [i-(n-1), i]

    s_hi2 = i
    s2 = np.arange(s_lo2, s_hi2 + 1)
    idx2 = center + s2
    # The term h[i-s]*(1-h[i]) went from h[i-s]*1 to h[i-s]*0. Change: -h[i-s]
    # But if i-s == j, this overlaps with the j flip handling
