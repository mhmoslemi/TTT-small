import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_subgradient(c_vec, h, n, active_idx, epsilon=1e-8):
    """Compute the subgradient of max_k c_k at the given point.
    For the active set, compute weighted gradient."""
    dx = 2.0 / n
    g = np.zeros(n)
    max_val = c_vec.max()
    weights = np.zeros(len(c_vec))
    for idx in active_idx:
        if c_vec[idx] >= max_val - epsilon:
            weights[idx] = 1.0
    if weights.sum() == 0:
        weights[active_idx[0]] = 1.0
    weights /= weights.sum()
    
    for idx in active_idx:
        if weights[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        w = weights[idx] * dx
        if k >= 0:
            nv = n - k
            g[0:nv] += w * (1.0 - h[k:n])
            g[k:n] += w * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += w * (1.0 - h[0:nv])
            g[0:nv] += w * (-h[kk:n])
    return g

def active_set_descent(h, n, deadline, max_iters=5000):
    """Exact subgradient descent with line search on active set."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    step = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Identify active set
        tol = max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        
        # Compute subgradient
        g = exact_subgradient(c_vec, best_h, n, active, tol)
        
        # Find best step: move mass from high-g to low-g positions
        # subject to 0<=h<=1 and sum(h)=n/2
        # Simple approach: move step amount from highest g to lowest g
        pos_mask = (g > 1e-12) & (best_h > 1e-12)
        neg_mask = (g < -1e-12) & (best_h < 1.0 - 1e-12)
        
        if not pos_mask.any() or not neg_mask.any():
            break
        
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        # Find best pair
        best_gain = 0
        best_i, best_j = -1, -1
        for i in pos_idx[:30]:
            for j in neg_idx[:30]:
                gain = g[i] - g[j]
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
        
        if best_i < 0 or best_gain <= 0:
            break
        
        amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
        best_h[best_i] -= amt
        best_h[best_j] += amt
        best_h = project_feasible(best_h, n)
        
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            # Revert
            best_h[best_i] += amt
            best_h[best_j] -= amt
            best_h = project_feasible(best_h, n)
            step *= 0.7
    
    return best_h, best_c

- Distribute the "increase" proportionally to -g_j among the increase positions
- Scale so that total decrease = total increase

    if not dec_mask.any() or not inc_mask.any():
        return h_new
    
    dec_amt = g[dec_mask] * step_size  # amount to remove from each
    inc_amt = (-g[inc_mask]) * step_size  # amount to add to each

    total_inc = inc_amt.sum()

    h_new[dec_mask] -= dec_amt
    h_new[inc_mask] += inc_amt
    
    # Project
    h_new = project_feasible(h_new, n)

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_phase(h, n, deadline, max_iters=10000):

            break

                nv = n - kk

        neg_mask = (g < -1e-10) & (best_h < 1.0 - 1e-10)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def topk_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def topk_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    if mu > 0:
        g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def corr_gradient_single(k, h, n):
    """Gradient of c_k with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        nv = n - k
        g[:nv] += (1.0 - h[k:]) * dx
        g[k:] += (-h[:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        g[kk:] += (1.0 - h[:nv]) * dx
        g[:nv] += (-h[kk:]) * dx
    return g

def lbfgs_phase(h, n, deadline, alphas, mu=0.1, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list and i < len(maxit_list):
            maxit = min(maxit_list[i], max(200, int(remaining * 3)))
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-15, 'maxfun': maxit * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def subgradient_phase(h, n, deadline, max_iters=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    step_sizes = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    step_idx = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.002 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        n_active = len(active)
        if n_active == 0:
            break
        
        # Compute subgradient: average of gradients of active c_k
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            w = dx_val / n_active
            if k >= 0:
                nv = n - k
                g[:nv] += w * (1.0 - best_h[k:])
                g[k:] += w * (-best_h[:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:] += w * (1.0 - best_h[:nv])
                g[:nv] += w * (-best_h[kk:])
        
        # Proportional step: decrease where g>0, increase where g<0
        pos_idx = np.where((g > 1e-10) & (best_h > 1e-10))[0]
        neg_idx = np.where((g < -1e-10) & (best_h < 1.0 - 1e-10))[0]
        
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        
        improved = False
        for si in range(step_idx, len(step_sizes)):
            step = step_sizes[si]
            # Try proportional step
            dec_amt = np.minimum(g[pos_idx] * step, best_h[pos_idx] * 0.5)
            inc_amt = np.minimum((-g[neg_idx]) * step, (1.0 - best_h[neg_idx]) * 0.5)
            total_dec = dec_amt.sum()
            total_inc = inc_amt.sum()
            if total_dec < 1e-14 or total_inc < 1e-14:
                continue
            scale = min(total_inc, total_dec) / max(total_inc, total_dec)
            best_h[pos_idx] -= dec_amt * scale
            best_h[neg_idx] += inc_amt * scale
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved = True
                step_idx = si
                no_improve = 0
                break
            else:
                # Revert
                best_h[pos_idx] += dec_amt * scale
                best_h[neg_idx] -= inc_amt * scale
                best_h = project_feasible(best_h, n)
        
        if not improved:
            no_improve += 1
            if no_improve > 8 and step_idx < len(step_sizes) - 1:
                step_idx += 1
                no_improve = 0
            if step_idx >= len(step_sizes) - 1 and no_improve > 30:
                break
    
    return best_h, best_c

def pair_swap_search(h, n, deadline, radius=None, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = min(60, n // 8)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        changed = True
        while changed and time.time() < deadline:
            changed = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        old_i, old_j = best_h[i], best_h[j]
                        best_h[i] = min(1.0, old_i + d)
                        best_h[j] = max(0.0, old_j - d)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            changed = True
                        else:
                            best_h[i], best_h[j] = old_i, old_j
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        old_i, old_j = best_h[i], best_h[j]
                        best_h[i] = max(0.0, old_i - d)
                        best_h[j] = min(1.0, old_j + d)
                        best_h = project_feasible(best_h, n)

                        if c < best_c - 1e-15:

                            changed = True
                        else:
                            best_h[i], best_h[j] = old_i, old_j
    return best_h, best_c
