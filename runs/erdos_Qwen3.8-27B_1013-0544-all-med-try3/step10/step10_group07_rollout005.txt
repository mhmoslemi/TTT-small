def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_to_binary(h, n):
    """Project to binary: pick top n/2 values to be 1."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    """Project h to [0,1] with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            break

            if s < 1e-14:
