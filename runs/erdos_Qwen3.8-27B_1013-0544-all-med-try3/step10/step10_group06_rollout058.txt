import time

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

            h += (diff / tot) * h

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):

    def obj_and_grad(x):

        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

def snap_to_binary(h, n):
    """Snap to binary, ensuring exactly n//2 ones."""
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[one_idx])
        for idx in order[:num_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif num_ones < target:

        # Add the zeros with largest h values
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - num_ones]:
            h_bin[zero_idx[idx]] = 1
    assert int(h_bin.sum()) == target

```python
def binary_swap_delta(h, n, i, j):
    """Compute change in c_m (positive lags only) for swapping h[i]:1->0, h[j]:0->1.
    Returns array of length n where dxc[m] is the change for lag m (m=1,...,n-1)."""
    dxc = np.zeros(n)
    
    # i: 1->0
    if i + 1 < n:
        dxc[1:n-i] -= (1.0 - h[i+1:n])
    if i >= 1:
        dxc[1:i+1] += h[0:i]
    
    # j: 0->1
    if j + 1 < n:
        dxc[1:n-j] += (1.0 - h[j+1:n])
    if j >= 1:
        dxc[1:j+1] -= h[0:j]
    
    # Correction for double counting at lag |j-i|
    d = abs(j - i)
    if 1 <= d < n:
        dxc[d] += 1.0
    
    return dxc
```

Let me also add: **pairwise swap within a lag**. For a specific lag m, the overlap c_m = Σ h[i](1-h[i+m]). - For each such i, find a good target j where h[j]=0 and (h[j+m*]=1 or h[j-m*]=1) [meaning j is already "covered"]
- If such j exists, swap i and j
- This reduces c_{m*} by 1 (in count units) without increasing other lags too much

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    m = c.max()

    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()
