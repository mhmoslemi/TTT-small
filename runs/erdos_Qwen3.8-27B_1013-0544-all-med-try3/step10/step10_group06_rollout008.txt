from scipy.optimize import minimize
import time

def fft_correlate(h, n):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""
    N = 1
    while N < 2*n - 1:
        N *= 2
    f_h = np.fft.fft(h, N)
    f_v = np.fft.fft(1.0 - h, N)
    r = np.fft.ifft(f_h * np.conj(f_v)).real
    # Reorder: r[0:k] gives lags 0 to k-1 (positive)
    # r[N-k+1:N] gives negative lags

    # Negative lags (-(n-1) to -1) → correlate indices 0 to n-2

One more important consideration: the smooth_max function. For very high alpha (e.g., 10^8), the log-sum-exp can overflow. I need to be careful:

log(sum exp(alpha * c_k)) = alpha * max(c_k) + log(sum exp(alpha * (c_k - max(c_k))))

This is numerically stable. And the smooth max value is:
(1/alpha) * [alpha * max(c) + log(sum exp(alpha * (c - max(c))))]
= max(c) + (1/alpha) * log(sum exp(alpha * (c - max(c))))

The weights are: w_k = exp(alpha * (c_k - max(c))) / sum_j exp(alpha * (c_j - max(c)))

For the gradient:
grad = sum_k w_k * dc_k/dh * dx

This is the weighted sum of the "exact" gradients at each lag.

import numpy as np
from scipy.optimize import minimize
import time

# Pre-computed FFT size (next power of 2 >= 2n-1)
def next_pow2(m):
    p = 1
    while p < m:
        p *= 2
    return p

def fft_correlate(h, n):
    N = next_pow2(2*n - 1)
    f_h = np.fft.fft(h, N)
    f_v = np.fft.fft(1.0 - h, N)
    r = np.fft.ifft(f_h * np.conj(f_v)).real
    out = np.zeros(2*n-1)
    out[n-1:] = r[0:n]
    out[:n-1] = r[N-n+1:N] if n > 1 else np.zeros(0)
    return out

def compute_c5(h, n):
    dx = 2.0 / n
    c = fft_correlate(h, n)
    return float(np.max(c) * dx)

def smooth_max_vec(c, alpha):
    """Compute smooth max and weights of vector c."""
    m = c.max()
    shifted = alpha * (c - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    val = m + log_sum / alpha
    weights = exp_vals / np.sum(exp_vals)
    return val, weights

def gradient_smooth_max(h, n, c, weights, alpha):
    """Compute gradient of smooth max of correlation with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only process lags with non-negligible weight
    # Find top lags
    top_count = min(n, 20)  # top 20 lags should suffice for high alpha
    top_idx = np.argpartition(-weights, top_count)[:top_count]
    
    for k in top_idx:
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            # dc/dh[j] = (1-h[j-m]) * [j>=m] - h[j+m] * [j+m<n]
            grad[m:n] += w * (1.0 - h[0:n-m])
            grad[0:n-m] += w * (-h[m:n])
        else:
            m = -lag
            # dc/dh[j] = (1-h[j+m]) * [j+m<n] - h[j-m] * [j>=m]
            grad[0:n-m] += w * (1.0 - h[m:n])
            grad[m:n] += w * (-h[0:n-m])
    
    return grad * dx
