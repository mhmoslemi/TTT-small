        trial_dl = max(deadline, min(deadline, time.time() + (deadline - time.time()) / max(1, n_trials - trial) * 1.2))
        h_ls, c_ls = binary_local_search(h_ls, n, trial_dl,
                                         max_sweeps=3000, seed=seed + 1000 + trial)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP polish from initial (short)
    p1_end = t0 + 0.08 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [500, 5000, 50000]:
        if time.time() > p1_end:
            break
        h = slsqp_smooth(h, best_n, alpha, 1500, p1_end)
        c = compute_c5(h, best_n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Binary local search on parent resolution
    p2_end = t0 + 0.30 * (budget_s - 3)
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    target = best_n // 2
    excess = int(h_bin.sum() - target)
    if excess > 0:
        idx = np.where(h_bin == 1)[0]
        for k in range(excess):
            h_bin[idx[k]] = 0.0
    elif excess < 0:
        idx = np.where(h_bin == 0)[0]
        for k in range(-excess):
            h_bin[idx[k]] = 1.0
    hb, cb = binary_local_search(h_bin, best_n, p2_end, max_sweeps=5000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: SLSQP continuation from binary
    p3_end = t0 + 0.42 * (budget_s - 3)
    if time.time() < p3_end:
        hp = best_h + 0.04 * rng.randn(best_n)
        hp = project_feasible(hp, best_n)
        for alpha in [5000, 50000, 500000, 5000000]:
            if time.time() > p3_end:
                break
            hp = slsqp_smooth(hp, best_n, alpha, 2500, p3_end)
            c = compute_c5(hp, best_n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 4: Multi-resolution binary search
    p4_end = t0 + 0.65 * (budget_s - 3)
    for nh in [300, 400, 500]:
        if time.time() > p4_end - 20:
            break
        n_cur = len(best_h)
        xo = np.arange(n_cur) * 2.0 / n_cur
        xn = np.arange(nh) * 2.0 / nh
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, nh)
        h_bn = np.where(hh > 0.5, 1.0, 0.0)
        target_n = nh // 2
        excess = int(h_bn.sum() - target_n)
        if excess > 0:
            idx = np.where(h_bn == 1)[0]
            for k in range(excess):
                h_bn[idx[k]] = 0.0
        elif excess < 0:
            idx = np.where(h_bn == 0)[0]
            for k in range(-excess):
                h_bn[idx[k]] = 1.0
        remain = p4_end - 20
        per_res = remain / 3.0
        sweep_dl = time.time() + per_res
        hbn, cbn = binary_local_search(h_bn, nh, sweep_dl,
                                       max_sweeps=3000, seed=seed + nh)
        if cbn < best_c:
            best_c = cbn
            best_h = hbn.copy()
            best_n = nh
        else:
            c_cont = compute_c5(hh, nh)
            if c_cont < best_c:
                best_c = c_cont
                best_h = hh.copy()
                best_n = nh

    # Phase 5: Iterated local search at best resolution
    p5_end = t0 + 0.82 * (budget_s - 3)
    n_cur = best_n
    h_bin_cur = np.where(best_h > 0.5, 1.0, 0.0)
    target = n_cur // 2
    excess = int(h_bin_cur.sum() - target)
    if excess > 0:
        idx = np.where(h_bin_cur == 1)[0]
        for k in range(excess):
            h_bin_cur[idx[k]] = 0.0
    elif excess < 0:
        idx = np.where(h_bin_cur == 0)[0]
        for k in range(-excess):
            h_bin_cur[idx[k]] = 1.0
    n_trials = max(5, int((p5_end - time.time()) / 15))
    hb5, cb5 = iterated_local_search(h_bin_cur, n_cur, p5_end,
                                     n_trials=n_trials, seed=seed + 50)
    if cb5 < best_c:
        best_c = cb5
        best_h = hb5.copy()

    # Phase 6: Final SLSQP polish
    if time.time() < dl - 8:
        n_cur = len(best_h)
        h_fin = best_h + 0.02 * rng.randn(n_cur)
        h_fin = project_feasible(h_fin, n_cur)
        for alpha in [100000, 1000000, 10000000, 50000000]:
            if time.time() > dl - 8:
                break
            h_fin = slsqp_smooth(h_fin, n_cur, alpha, 3000, dl - 8)
            c = compute_c5(h_fin, n_cur)
            if c < best_c:
                best_c = c
                best_h = h_fin.copy()

    # Phase 7: Last binary search
    if time.time() < dl - 2:
        n_cur = len(best_h)
        h_bin_fin = np.where(best_h > 0.5, 1.0, 0.0)
        target = n_cur // 2
        excess = int(h_bin_fin.sum() - target)
        if excess > 0:
            idx = np.where(h_bin_fin == 1)[0]
            for k in range(excess):
                h_bin_fin[idx[k]] = 0.0
        elif excess < 0:
            idx = np.where(h_bin_fin == 0)[0]
            for k in range(-excess):
                h_bin_fin[idx[k]] = 1.0
        hb7, cb7 = binary_local_search(h_bin_fin, n_cur, dl,
                                       max_sweeps=2000, seed=seed + 999)
        if cb7 < best_c:
            best_c = cb7
            best_h = hb7.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
