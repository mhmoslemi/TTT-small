    # Step
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new

def targeted_local_search(h, n, deadline, max_iter=50000):
    """Iteratively reduce the max correlation by targeting worst lags."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    step = 0.01
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        cur_max = c_vec.max()
        if cur_max < best_c - 1e-12:
            best_c = cur_max
            best_h = best_h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            step *= 0.7
            if step < 1e-8:
                break
            no_improve = 0
        # Find top-5 worst lags
        top_idx = np.argsort(-c_vec)[:5]
        for aidx in top_idx:
            if time.time() > deadline - 5:
                break
            center = n - 1
            k = int(aidx) - center
            # Compute gradient of this specific lag
            dx = 2.0 / n
            grad = np.zeros(n)
            if k == 0:
                grad = 1.0 - 2.0 * best_h
            elif k > 0:
                nv = n - k
                grad[k:] += (1.0 - best_h[:nv])
                grad[:nv] -= best_h[k:]
            else:
                kk = -k
                nv = n - kk
                grad[:nv] += (1.0 - best_h[kk:])
                grad[kk:] -= best_h[:nv]
            grad *= dx
            # Take step in -grad direction
            for trial_step in [step, step * 0.5, step * 0.2]:
                h_try = best_h - trial_step * grad
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5_vec(h_try, n).max()
                if c_try < best_c - 1e-14:
                    best_h = h_try
                    best_c = c_try
                    break
    return best_h, best_c

def pair_dense_search(h, n, deadline, radius, delta):
    """Dense pairwise search with small delta."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    for d in [delta]:
        improved_any = True
        while improved_any and time.time() < deadline - 5:
            improved_any = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            # Focus on positions relevant to worst lags
            top_lags = np.argsort(-c_vec)[:8]
            relevant = set()
            center = n - 1
            for aidx in top_lags:
                k = int(aidx) - center
                if k >= 0:
                    for i in range(min(n - k, 200)):
                        relevant.add(i)
                        if i + k < n:
                            relevant.add(i + k)
                else:
                    kk = -k
                    for i in range(min(n - kk, 200)):
                        relevant.add(i)
                        if i + kk < n:
                            relevant.add(i + kk)
            rel_list = list(relevant)[:300]
            for i in rel_list:
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try move mass from i to j
                    if best_h[i] >= d and best_h[j] <= 1.0 - d:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5_vec(best_h, n).max()
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Try move mass from j to i
                    if best_h[j] >= d and best_h[i] <= 1.0 - d:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5_vec(best_h, n).max()
                        if c < best_c - 1e-15:
                            best_c = c
                            improved_any = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    return best_h, best_c

def full_lbfgs(h, n, alpha, mu, maxiter, ftol=1e-20):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: obj_and_grad(x, n, alpha, mu)[0],
            h, jac=lambda x: obj_and_grad(x, n, alpha, mu)[1],
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': ftol,
                     'gtol': 1e-18, 'maxfun': maxiter * 10,
                     'maxls': 50, 'maxcor': 50}
        )
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def active_lbfgs(h, n, active, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: active_obj_and_grad(x, n, active, alpha, mu)[0],
            h, jac=lambda x: active_obj_and_grad(x, n, active, alpha, mu)[1],
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-20,
                     'gtol': 1e-18, 'maxfun': maxiter * 10,
                     'maxls': 50, 'maxcor': 50}
        )
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def subgradient_iteration(h, n, deadline, n_iter=2000):
    """Pure subgradient method targeting the max."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    step = 0.05
    for it in range(n_iter):
        if time.time() > deadline - 10:
            break
        c_vec = compute_c5_vec(best_h, n)
        cur_max = c_vec.max()
        if cur_max < best_c:
            best_c = cur_max
            best_h = best_h.copy()
        # Subgradient: gradient of the (near-)max lags
        tol = 0.01 * cur_max
        active = np.where(c_vec >= cur_max - tol)[0]
        center = n - 1
        dx = 2.0 / n
        grad = np.zeros(n)
        for aidx in active:
            k = int(aidx) - center
            if k == 0:
                grad += (1.0 - 2.0 * best_h)
            elif k > 0:
                nv = n - k
                grad[k:] += (1.0 - best_h[:nv])
                grad[:nv] -= best_h[k:]
            else:
                kk = -k
                nv = n - kk
                grad[:nv] += (1.0 - best_h[kk:])
                grad[kk:] -= best_h[:nv]
        grad *= dx / len(active)
        step *= 0.999
        h_new = best_h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        best_h = h_new
    return best_h, compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig

    # Try alternative starts
    alt_starts = [
        make_random_binary(n_orig, rng),
        make_soft_binary(n_orig, rng, 0.5),
        make_soft_binary(n_orig, rng, 0.8),
    ]
    for ha in alt_starts:
        ca = compute_c5(ha, n_orig)
        if ca < best_c:
            best_c = ca
            best_h = ha.copy()

    # Phase 1: L-BFGS smoothing at original resolution (60s)
    p1_end = min(t0 + 60, deadline - 800)
    if p1_end > t0 + 5:
        for alpha in [5000, 50000, 500000, 2000000]:
            if time.time() > p1_end - 5:
                break
            rem = p1_end - time.time() - 3
            if rem < 5:
                break
            maxit = max(100, int(rem * 4))
            maxit = min(maxit, 2000)
            h1, c1 = full_lbfgs(best_h, best_n, alpha, 0.01, maxit)
            if c1 < best_c:
                best_c = c1
                best_h = h1

    # Phase 2: Active-set L-BFGS rounds (180s)
    p2_end = min(t0 + 240, deadline - 600)
    if p2_end > time.time() + 20:
        for round_i in range(8):
            if time.time() > p2_end - 10:
                break
            c_vec = compute_c5_vec(best_h, best_n)
            max_val = c_vec.max()
            threshold = max_val - 0.005 * max_val
            active = np.where(c_vec >= threshold)[0]
            if len(active) > 12:
                active = np.array(np.argsort(-c_vec[active])[:12])
            if len(active) == 0:
                continue
            alpha = 50000 * (10 ** round_i)
            alpha = min(alpha, 50000000)
            rem = p2_end - time.time() - 5
            if rem < 5:
                break
            maxit = max(200, int(rem * 3))
            maxit = min(maxit, 3000)
            h2, c2 = active_lbfgs(best_h, best_n, active, alpha, 0.001, maxit)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Upsample to 2x and optimize (200s)
    n_up = best_n * 2
    p3_end = min(time.time() + 200, deadline - 350)
    use_higher = False
    if p3_end > time.time() + 60 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        # Quick optimization at higher res
        if c_up <= best_c + 0.002:
            for alpha in [10000, 100000, 1000000]:
                if time.time() > p3_end - 10:
                    break
                rem = p3_end - time.time() - 5
                if rem < 5:
                    break
                maxit = max(200, int(rem * 3))
                maxit = min(maxit, 2000)
                hu, cu = full_lbfgs(h_up, n_up, alpha, 0.005, maxit)
                if cu < c_up:
                    h_up = hu
                    c_up = cu
            # Active set at higher res
            c_vec_u = compute_c5_vec(h_up, n_up)
            max_u = c_vec_u.max()
            thr_u = max_u - 0.005 * max_u
            act_u = np.where(c_vec_u >= thr_u)[0]
            if len(act_u) > 15:
                act_u = np.array(np.argsort(-c_vec_u[act_u])[:15])
            rem_u = p3_end - time.time() - 5
            if rem_u > 10 and len(act_u) > 0:
                maxit_u = max(200, int(rem_u * 2))
                maxit_u = min(maxit_u, 2000)
                hu2, cu2 = active_lbfgs(h_up, n_up, act_u, 1000000, 0.001, maxit_u)
                if cu2 < c_up:
                    h_up = hu2
                    c_up = cu2
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                best_n = n_up
                use_higher = True

    # Phase 4: Targeted local search (350s) - the main improvement phase
    p4_end = min(time.time() + 350, deadline - 200)
    if p4_end > time.time() + 30:
        h4, c4 = targeted_local_search(best_h, best_n, p4_end, max_iter=100000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Subgradient iteration (100s)
    p5_end = min(time.time() + 100, deadline - 120)
    if p5_end > time.time() + 20:
        h5, c5 = subgradient_iteration(best_h, best_n, p5_end, n_iter=5000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Another round of active L-BFGS after local search (80s)
    p6_end = min(time.time() + 80, deadline - 80)
    if p6_end > time.time() + 15:
        c_vec = compute_c5_vec(best_h, best_n)
        max_val = c_vec.max()
        threshold = max_val - 0.003 * max_val
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 10:
            active = np.array(np.argsort(-c_vec[active])[:10])
        if len(active) > 0:
            rem = p6_end - time.time() - 5
            if rem > 5:
                maxit = max(100, int(rem * 3))
                maxit = min(maxit, 2000)
                h6, c6 = active_lbfgs(best_h, best_n, active, 20000000, 0.0005, maxit)
                if c6 < best_c:
                    best_c = c6
                    best_h = h6

    # Phase 7: Ultra-fine pair search (100s)
    p7_end = min(time.time() + 100, deadline - 20)
    if p7_end > time.time() + 15:
        h7, c7 = pair_dense_search(best_h, best_n, p7_end, radius=50, delta=0.0005)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final micro-adjustments
    p8_end = deadline - 1
    if p8_end > time.time() + 5:
        h8, c8 = targeted_local_search(best_h, best_n, p8_end, max_iter=200000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
