def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def c5_gradient_single_k(h, n, k):
    """Gradient of c_k with respect to h, for a single shift k.
    k is the shift (can be negative)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    # c_k = sum_i h[i]*(1-h[i+k]) for valid i
    # dc_k/dh[j] = (1-h[j+k])*[0<=j+k<n] - h[j-k]*[0<=j-k<n]

    grad[valid1] += (1.0 - h[j[valid1] + k])
    
    # Second term: -h[j-k] for 0<=j-k<n
    valid2 = (j - k >= 0) & (j - k < n)
    grad[valid2] -= h[j[valid2] - k]

def c5_gradient_smooth_max(h, n, alpha, top_k=50):
    """Gradient of smooth max of c_k using top-k active set."""
    c = compute_c5_vec(h, n)
    # Get top-k indices
    if len(c) <= top_k:
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]

    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)

    w = np.exp(sh - smx)

        valid2 = (j - k >= 0) & (j - k < n)

Let me also think about the **complementary slackness** conditions. At the optimum, the active correlations (at the maximum) should have their gradients pointing in a feasible direction. This means the weighted sum of gradients of active constraints should be zero (at the minimum of the max).

OK, I think the best strategy is:

1. Start from initial_h_values (n=500)
2. Do a quick L-BFGS-B optimization with moderate alpha
3. Switch to active-set optimization (top-k) with increasing alpha
4. Do local pair search
5. If time allows, try n=1000
6. Final squeeze with very fine pair search

The key difference from the previous code:
- More efficient gradient (vectorized, no inner Python loops)
- Better time allocation
- More diverse exploration at the start
- Aggressive local search at the end

Actually, let me reconsider. The previous code's main bottleneck was the Python loop in `smooth_max_grad` over all 2n-1 = 999 correlation values. My vectorized version loops over only `top_k` values (10-50), each with O(n) numpy operations. This should be 20-100x faster per gradient evaluation.

With faster gradients, L-BFGS-B can do many more iterations in the same time, leading to better convergence.

Let me also add a **cycling method**:
- For each of the top few k values, do a few gradient steps on just that c_k
- This is like a block coordinate method

And a **greedy local search**:
- For the worst k, find the best single-coordinate move (change h[j] by delta)
- Apply it if it reduces the max
- Cycle through coordinates

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_single_k(h, n, k):
    """Gradient of c_k w.r.t. h (before multiplying by dx)."""
    dx = 2.0 / n
    j = np.arange(n)
    grad = np.zeros(n)
    v1 = (j + k >= 0) & (j + k < n)
    if v1.any():
        grad[v1] += (1.0 - h[j[v1] + k])
    v2 = (j - k >= 0) & (j - k < n)
    if v2.any():
        grad[v2] -= h[j[v2] - k]
    return grad * dx

def smooth_max_and_grad(h, n, alpha, top_k=30):
    """Returns (smooth_max_value, gradient) using top-k active set."""
    c = compute_c5_vec(h, n)
    nc = len(c)
    if nc <= top_k:
        top_idx = np.arange(nc)
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    wsum = w.sum()
    w /= wsum
    
    val = m + (smx + np.log(wsum)) / alpha
    
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    
    for wi, idx in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k = int(idx) - (n - 1)
        v1 = (j + k >= 0) & (j + k < n)
        if v1.any():
            grad[v1] += wi * dx * (1.0 - h[j[v1] + k])
        v2 = (j - k >= 0) & (j - k < n)
        if v2.any():
            grad[v2] -= wi * dx * h[j[v2] - k]
    
    return float(val), grad

def obj_func(h, n, alpha, top_k, mu):
    val, _ = smooth_max_and_grad(h, n, alpha, top_k)
    penalty = mu * (h.sum() - n / 2.0) ** 2
    return val + penalty

def grad_func(h, n, alpha, top_k, mu):
    _, g = smooth_max_and_grad(h, n, alpha, top_k)
    if mu > 0:
        g = g + 2.0 * mu * (h.sum() - n / 2.0)
    return g

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def lbbfgs_round(h, n, deadline, alpha, top_k, mu, maxit=None):
    bounds = [(0.0, 1.0)] * n
    rem = deadline - time.time()
    if rem < 10:
        return h, compute_c5(h, n)
    if maxit is None:
        maxit = max(200, int(rem * 5))
        maxit = min(maxit, 5000)
    try:
        res = minimize(obj_func, h.copy(), jac=grad_func,
                      args=(n, alpha, top_k, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-25,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def cyclic_greedy(h, n, deadline, n_iters=None):
    """Cycle through worst k values, do greedy single-coordinate moves."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n_iters is None:
        n_iters = 100000
    dx = 2.0 / n
    c_vec = compute_c5_vec(best_h, n)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Find worst k
        worst_idx = c_vec.argmax()
        worst_k = int(worst_idx) - (n - 1)
        worst_val = c_vec[worst_idx]
        
        # Compute gradient of c_{worst_k}
        g = grad_single_k(best_h, n, worst_k)
        
        # Find the coordinate with largest negative gradient (decreasing h there helps)
        # or largest positive gradient (increasing h there hurts most, so decrease)
        # We want to decrease c_{worst_k}: move h in -g direction
        # But we need to maintain sum(h) = n/2, so we need a pair move
        
        # Find best pair (i, j) to move: decrease h[i], increase h[j]
        # Effect on c_{worst_k}: -g[i] + g[j] (approximately, for small delta)
        # We want -g[i] + g[j] < 0, i.e., g[i] > g[j]
        
        # For efficiency, just find the coordinate with max g (most to decrease)
        # and min g (most to increase)
        i_dec = np.argmax(g)  # largest positive gradient → most to decrease
        i_inc = np.argmin(g)  # largest negative gradient → most to increase
        
        # Try a small move
        delta = 0.01
        if best_h[i_dec] >= delta and best_h[i_inc] <= 1.0 - delta:
            best_h[i_dec] -= delta
            best_h[i_inc] += delta
            c_new = compute_c5_vec(best_h, n)
            if c_new.max() < best_c - 1e-15:
                best_c = c_new.max()
                c_vec = c_new
            else:
                best_h[i_dec] += delta
                best_h[i_inc] -= delta
                # Try smaller delta
                delta = 0.001
                if best_h[i_dec] >= delta and best_h[i_inc] <= 1.0 - delta:
                    best_h[i_dec] -= delta
                    best_h[i_inc] += delta
                    c_new = compute_c5_vec(best_h, n)
                    if c_new.max() < best_c - 1e-15:
                        best_c = c_new.max()
                        c_vec = c_new
                    else:
                        best_h[i_dec] += delta
                        best_h[i_inc] -= delta
        # Recompute periodically
        if it % 100 == 0:
            c_vec = compute_c5_vec(best_h, n)
    
    return best_h, best_c
