def smooth_max_topk(c, k, alpha):
    # Take top k entries, smooth max over them
    if k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(-c)[:k]
    vals = c[idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val, idx

def smooth_max_topk_grad(x, n, top_k, alpha, mu):
    c = compute_c5_vec(x, n)
    m_all = c.max()
    threshold = m_all - max(1e-8, 0.001 * m_all)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c[active])[:top_k])
    
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

For shift k ≥ 0:

= ∫₀^{2-k} h(x) dx - ∫₀^{2-k} h(x)h(x+k) dx

∂/∂h_j = dx * [∫δ(x_j) (1-h(x_j+k)) dx - ∫(δ(x_j) h(x_j+k) + ... ) dx]

Actually, in the discrete case:

More carefully:

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active_set(c, top_k, margin_frac=0.002):
    c_max = c.max()
    threshold = c_max - max(1e-10, margin_frac * c_max)
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c[active])[:top_k])
    return active

def obj_fn(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_grad(x, n, active, alpha, mu):

    vals = c[active]

        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            g[0:nv] += wm * (1.0 - x[k:n])
            g[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:n] += wm * (1.0 - x[0:nv])
            g[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):
    return np.ones(n)

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def phase_lbfgs(h, n, deadline, alphas, maxit_per, mu, use_full):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        rem = deadline - time.time()
        if rem < 10:
            break
        mi = maxit_per[i] if i < len(maxit_per) else 2000
        mi = min(mi, max(500, int(rem * 3)))
        
        if use_full:
            try:
                res = minimize(full_obj, best_h, jac=full_grad,
                             args=(n, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': mi, 'ftol': 1e-20,
                                      'gtol': 1e-16, 'maxfun': mi * 8,
                                      'maxls': 40, 'maxcor': 40})
            except Exception:
                continue
        else:
            c_cur = compute_c5_vec(best_h, n)
            active = get_active(c_cur, 8, 0.002)
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                             args=(n, active, alpha, mu), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': mi, 'ftol': 1e-20,
                                      'gtol': 1e-16, 'maxfun': mi * 8,
                                      'maxls': 40, 'maxcor': 40})
            except Exception:
                continue
        
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def phase_active(h, n, deadline, top_k, alphas, rounds):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for r in range(rounds):
        rem = deadline - time.time()
        if rem < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        active = get_active(c_vec, top_k, 0.002)
        if len(active) == 0:
            break
        
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            mi = max(500, int((deadline - time.time()) * 2))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                             args=(n, active, alpha, 0.005), method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': mi, 'ftol': 1e-20,
                                      'gtol': 1e-16, 'maxfun': mi * 8,
                                      'maxls': 40, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def phase_subgrad(h, n, deadline, n_iter):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        cmax = c_vec.max()
        tol = max(1e-10, 0.002 * cmax)
        active = np.where(c_vec >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                g[0:nv] += (1.0 - h[k:n]) * dx
                g[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += (1.0 - h[0:nv]) * dx
                g[0:nv] += (-h[kk:n]) * dx
        g /= len(active)
        
        h_new = h - eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.015)
        else:
            eta *= 0.99
        
        h = h_new
    return best_h, best_c

def phase_slqp(h, n, deadline, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        res = minimize(slqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)
