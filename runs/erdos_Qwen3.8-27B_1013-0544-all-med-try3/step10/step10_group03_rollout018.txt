```python
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def find_active_shifts(h, n, top_k=5, rel_tol=0.005):
    c = compute_c5_vec(h, n)
    max_val = c.max()
    threshold = max_val - rel_tol * max_val
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c[active])
        active = active[order[:top_k]]
    return active.tolist()

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def binary_project(h, n):
    """Project to nearest binary solution with n/2 ones."""
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def pair_search(h, n, deadline, n_pairs=50000):
    """Try random pair moves."""
    best_c = compute_c5(h, n)
    np.random.seed(123)
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    count = 0
    while count < n_pairs and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        for step in steps:
            for sign in [1, -1]:
                hi_new = h[i] + sign * step
                hj_new = h[j] - sign * step
                if 0 <= hi_new <= 1 and 0 <= hj_new <= 1:
                    h[i] = hi_new
                    h[j] = hj_new
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                    else:
                        h[i] = hi_new - sign * step
                        h[j] = hj_new + sign * step
        count += 1
    return h, best_c
```

```python
        # With zero-padding to size, the irfft gives the circular conv, but with enough padding it's linear
        B = Bx_full[:n] - W_pos_padded[0] * x  # remove k=0 term
        
        # C[i] = sum_{k=1}^{i} W_neg[k] * omx[i-k]
        W_neg_full_padded = np.zeros(size)
        W_neg_full_padded[:n] = np.concatenate([[0], W_neg_padded[:n-1]])  # W_neg_full with [0] at start
        W_neg_full_f = np.fft.rfft(W_neg_full_padded)
        Comx_full = np.fft.irfft(W_neg_full_f * omx_f, n=size)
        C = Comx_full[:n] - 0 * one_minus_x  # W_neg_full[0] = 0, so no subtraction needed
        
        grad = dx * (A - B + C - D)
    
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
```

            if tot < 1e-15:
