    """Fine-tune using coordinate descent on pairs."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    steps = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    c_vec = compute_c5_vec(best_h, n)
    active = np.where(c_vec >= 0.99 * c_vec.max())[0]
    
    # Focus on coordinates near the "interesting" regions
    # The most important coordinates are those that appear in the active shifts
    important_coords = set()
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            important_coords.update(range(n - k))
            important_coords.update(range(k, n))
        else:
            kk = -k
            important_coords.update(range(kk, n))
            important_coords.update(range(n - kk))
    
    coord_list = sorted(important_coords)
    
    for step in steps:
        if time.time() > deadline:
            break

        while improved and time.time() < deadline:
            improved = False

                    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def upsample(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def active_obj(x, n, active_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 500:
        return float(m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha)
    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def active_grad(x, n, active_idx, alpha):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()

    else:
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def full_smooth_grad(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):

        k = mm - (n - 1)

        pert = scale * rng.randn(n)
        h_pert = project_feasible(h + pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.005:
            t_sub = min(time.time() + 40, deadline - 20)
            h_opt, c_opt = run_active_lbfgs(h_pert, n, t_sub, top_k=5,
                                             alpha_list=[5000, 50000, 200000])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick optimization at native resolution
    if time.time() < deadline - 500:
        t1 = min(time.time() + 50, deadline - 500)
        best_h, best_c = run_active_lbfgs(best_h, n_orig, t1, top_k=5,
                                           alpha_list=[1000, 5000, 20000])
    
    # Phase 2: Upsample to n=1000
    n = 1000
    h1000 = upsample(best_h, n_orig, n)
    c1000 = compute_c5(h1000, n)
    
    # Phase 3: Broad L-BFGS at n=1000
    if time.time() < deadline - 400:
        t3 = min(time.time() + 120, deadline - 400)
        h1000, c1000 = run_full_lbfgs(h1000, n, t3, alpha_list=[30, 100, 300])
    
    if c1000 < best_c:
        best_c = c1000
        best_h = h1000
    
    # Phase 4: Focused active-set L-BFGS at n=1000
    if time.time() < deadline - 300:
        t4 = min(time.time() + 200, deadline - 300)
        best_h, best_c = run_active_lbfgs(best_h, n, t4, top_k=5,
                                           alpha_list=[5000, 20000, 100000, 500000])
    
    # Phase 5: Penalty method to push down active constraints
    if time.time() < deadline - 200:
        t5 = min(time.time() + 100, deadline - 200)
        best_h, best_c = penalty_refine(best_h, n, t5)
    
    # Phase 6: Another active-set round (re-identify active set)
    if time.time() < deadline - 150:
        t6 = min(time.time() + 80, deadline - 150)
        best_h, best_c = run_active_lbfgs(best_h, n, t6, top_k=3,
                                           alpha_list=[100000, 500000, 2000000])
    
    # Phase 7: Multi-start from current best
    if time.time() < deadline - 100:
        t7 = min(time.time() + 60, deadline - 100)
        best_h, best_c = multi_start_optimize(best_h, n, t7, n_starts=3, scale=0.005)
    
    # Phase 8: Fine pair search
    if time.time() < deadline - 30:
        t8 = min(time.time() + 35, deadline - 20)
        best_h, best_c = fine_pair_search(best_h, n, t8, focus_k=5)
    
    # Phase 9: Final active-set polish
    if time.time() < deadline - 5:
        t9 = deadline - 2
        best_h, best_c = run_active_lbfgs(best_h, n, t9, top_k=5,
                                           alpha_list=[500000, 2000000, 5000000])
    
    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
