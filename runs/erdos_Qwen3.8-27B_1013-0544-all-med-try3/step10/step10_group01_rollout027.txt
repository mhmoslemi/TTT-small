import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def objective_fn(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def gradient_fn(x, n, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def topk_objective(x, n, active, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topk_gradient(x, n, active, alpha, mu=0.0):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        pos = int(k * alpha) % n
        if h[pos] < 0.9:

def upsample(h_low, n_low, n_high):
    """Upsample from n_low to n_high using linear interpolation."""
    x_low = np.linspace(0, 2, n_low, endpoint=False) + 1.0/n_low
    x_high = np.linspace(0, 2, n_high, endpoint=False) + 1.0/n_high
    h_high = np.interp(x_high, x_low, h_low, period=2.0)
    return project_feasible(h_high, n_high)

def lbfgs_phase(h, n, deadline, alphas, mu=0.01, maxit_per=None):

    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        if maxit_per is not None and i < len(maxit_per):

        else:

        try:
            res = minimize(objective_fn, best_h, jac=gradient_fn,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit*10,
                                   'maxls': 50, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

                for j in zeros_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_pair_gain:
                        best_pair_gain = g
                        best_i = i
                        best_j = j
            if best_pair_gain <= 0:
                break
        
        best_h[best_i] = 0.0
        best_h[best_j] = 1.0
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        iters += 1
        
        if c_new < best_c - 1e-15:
            best_c = c_new
        else:
            # Revert
            best_h[best_i] = 1.0
            best_h[best_j] = 0.0
            best_h = project_feasible(best_h, n)
            # Try next best
            # For simplicity, break
            break
    
    return best_h, best_c

def grad_pair_search(h, n, deadline, max_steps=100000):
    """Gradient-guided pair search for fractional solutions."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx_val = 2.0 / n
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in step_sizes:
        if time.time() > deadline:
            break
        stuck = 0
        while time.time() < deadline and stuck < 200:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.001 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            
            pos_mask = (grad > 1e-10) & (best_h > 1e-8)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-8)
            
            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0:
                break
            
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            
            if c_new < best_c - 1e-16:
                best_c = c_new
                stuck = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stuck += 1
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):  # step by 2 for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        old_i, old_j = best_h[i], best_h[j]
                        best_h[i] = min(1.0, old_i + d)
                        best_h[j] = max(0.0, old_j - d)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i], best_h[j] = old_i, old_j
                            best_h = project_feasible(best_h, n)
    
    return best_h, best_c
