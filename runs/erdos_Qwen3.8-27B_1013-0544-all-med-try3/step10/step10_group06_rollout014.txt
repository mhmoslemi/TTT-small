def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

                  eps=1e-8, penalty_coeff=1e6):

    v = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000, penalty_coeff=1e8):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def adam_optimize(h, n, alpha, deadline, lr=0.03, beta1=0.85, beta2=0.999,
                  eps=1e-8, penalty_coeff=1e7):
    h = h.copy()
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    target_sum = n / 2.0

    while time.time() < deadline:
        t += 1
        c = compute_all_lags(h, n)
        obj_val = smooth_max_value(c, alpha)
        diff = h.sum() - target_sum
        obj_val += penalty_coeff * diff * diff

        grad = smooth_max_grad(h, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        h = h_new

        if t % 500 == 0:
            lr *= 0.95

    return h

        if time.time() > deadline - 50:
            break
        rem = min(deadline - 45 - time.time(), 25)
        if rem < 3:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, time.time() + rem, maxiter=2000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 6: Perturbations
    if time.time() < deadline - 60:
        h_opt, c_opt = try_perturbations(best_h, n, time.time() + min(40, deadline - time.time() - 20), rng, num_trials=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 7: Targeted local search (hard max)
    if time.time() < deadline - 25:
        ls_deadline = min(deadline - 10, time.time() + 60)
        h_opt, c_opt = targeted_local_search(best_h, n, ls_deadline, rng, top_k=6, max_iters=300000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 8: Pair swap local search
    if time.time() < deadline - 15:
        h_opt, c_opt = pair_swap_local_search(best_h, n, min(deadline - 5, time.time() + 30), rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 9: Final L-BFGS-B with very high alpha
    if time.time() < deadline - 8:
        h_opt = lbfgsb_optimize(best_h, n, 500000000, deadline - 4, maxiter=3000)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 10: Tiny final local search
    if time.time() < deadline - 2:
        h_opt, c_opt = targeted_local_search(best_h, n, deadline - 1, rng, top_k=3, max_iters=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Final validation
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
