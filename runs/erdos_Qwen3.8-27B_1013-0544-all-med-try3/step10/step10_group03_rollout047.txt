                                'gtol': 1e-14, 'maxfun': 30000, 'maxls': 30})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def run_lbfgs_full(h, n, alpha, mu, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(full_obj, h, jac=full_grad,
                       args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 20000, 'maxls': 30})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, steps=None):
    best_c = compute_c5(h, n)
    if steps is None:
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 + 1e-15 and h[j] - d >= -1e-15:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= -1e-15 and h[j] + d <= 1.0 + 1e-15:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def single_coord_opt(h, n, idx, deadline, lo=-0.01, hi=0.01):
    def obj(d):
        h2 = h.copy()
        h2[idx] += d
        h2 = np.clip(h2, 0.0, 1.0)
        diff = n / 2.0 - h2.sum()
        if diff > 0:
            room = 1.0 - h2
            tot = room.sum()
            if tot > 1e-15:
                h2 += (diff / tot) * room
        elif diff < 0:
            tot = h2.sum()
            if tot > 1e-15:
                h2 += (diff / tot) * h2
        h2 = np.clip(h2, 0.0, 1.0)
        return compute_c5(h2, n)
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                          options={'maxiter': 100, 'xatol': 1e-10})
    if res.success and res.fun < compute_c5(h, n) - 1e-15:
        h2 = h.copy()
        h2[idx] += res.x
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < compute_c5(h, n) - 1e-15:
            return h2, c2
    return h, compute_c5(h, n)

def randomized_restart(h, n, deadline, seed, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    bounds = [(0.0, 1.0)] * n
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        # Small random perturbation
        eps = rng.uniform(0, 0.02, n)
        h_pert = h + eps - eps.mean()
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick optimize: top shifts with moderate alpha
        sidx, c_vec = get_active_shifts(h_pert, n, tol_frac=0.005, tol_abs=2e-5, top_k=5)
        t_sub = min(60, (deadline - time.time()) / max(1, n_restarts - r))
        h_new, c_new = run_lbfgs_top(h_pert, n, sidx, 50000, 10.0, t_sub)
        
        # Then very high alpha on refined active set
        if c_new < best_c and time.time() < deadline - 30:
            sidx2, _ = get_active_shifts(h_new, n, tol_frac=0.002, tol_abs=1e-5, top_k=4)
            h_new, c_new = run_lbfgs_top(h_new, n, sidx2, 500000, 5.0, min(30, deadline - time.time()))
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def upsample_refine(h, n, deadline):
    n_new = n * 2
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = project_feasible(h_up, n_new)
    c_up = compute_c5(h_up, n_new)
    
    if time.time() > deadline - 10:
        return None, None, 0
    
    # Optimize at higher resolution
    sidx, _ = get_active_shifts(h_up, n_new, tol_frac=0.01, tol_abs=5e-5, top_k=6)
    t_opt = min(120, (deadline - time.time()) * 0.6)
    h_up, c_up = run_lbfgs_top(h_up, n_new, sidx, 20000, 5.0, t_opt)
    
    if time.time() < deadline - 10:
        sidx2, _ = get_active_shifts(h_up, n_new, tol_frac=0.003, tol_abs=2e-5, top_k=4)
        h_up, c_up = run_lbfgs_top(h_up, n_new, sidx2, 200000, 2.0, min(60, deadline - time.time()))
    
    # Downsample back
    h_dn = np.interp(x_old, x_new, h_up)
    h_dn = project_feasible(h_dn, n)
    c_dn = compute_c5(h_dn, n)
    
    # Also evaluate at n_new
    return h_up, c_up, n_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    # Phase 0: Initialize
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(500, 0.5)
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    # Phase 1: L-BFGS full with moderate alpha (establish good baseline)
    if time.time() < deadline - 400:
        for alpha, mu in [(200, 10.0), (2000, 50.0)]:
            if time.time() > deadline - 380:
                break
            t_end = min(time.time() + 80, deadline - 370)
            h_new, c_new = run_lbfgs_full(h, n, alpha, mu, t_end)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 2: L-BFGS full with higher alpha
    if time.time() < deadline - 300:
        for alpha, mu in [(10000, 100.0), (50000, 500.0)]:
            if time.time() > deadline - 280:
                break
            t_end = min(time.time() + 60, deadline - 260)
            h_new, c_new = run_lbfgs_full(h, n, alpha, mu, t_end)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Phase 3: Top-shifts focused with escalating alpha
    for alpha in [50000, 200000, 1000000, 5000000]:
        if time.time() > deadline - 200:
            break
        sidx, _ = get_active_shifts(h, n, tol_frac=0.002, tol_abs=1e-5, top_k=6)
        if len(sidx) == 0:
            break
        t_end = min(time.time() + 40, deadline - 180)
        h_new, c_new = run_lbfgs_top(h, n, sidx, alpha, 1.0, t_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new

    # Phase 4: Re-identify active shifts, very high alpha polish
    for round in range(3):
        if time.time() > deadline - 150:
            break
        sidx, _ = get_active_shifts(h, n, tol_frac=0.001, tol_abs=5e-6, top_k=4)
        if len(sidx) == 0:
            break
        t_end = min(time.time() + 30, deadline - 130)
        h_new, c_new = run_lbfgs_top(h, n, sidx, 10000000, 0.5, t_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new

    # Phase 5: Randomized restarts
    if time.time() < deadline - 120:
        t_end = deadline - 100
        h_rr, c_rr = randomized_restart(h, n, t_end, seed=42, n_restarts=4)
        if c_rr < best_c:
            best_c = c_rr
            best_h = h_rr
            h = h_rr

    # Phase 6: Single-coordinate optimization on critical indices
    if time.time() < deadline - 60:
        c_vec = compute_c5_vec(h, n)
        grad_top = get_active_shifts(h, n, tol_frac=0.005, tol_abs=1e-4, top_k=5)[0]
        # Compute combined gradient to find most critical coordinates
        combined_grad = np.zeros(n)
        for si in grad_top:
            combined_grad += _shift_grad_raw(h, n, si)
        top_coords = np.argsort(-np.abs(combined_grad))[:40]
        
        for idx in top_coords:
            if time.time() > deadline - 30:
                break
            h2, c2 = single_coord_opt(h, n, int(idx), deadline)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                h = h2

    # Phase 7: Fine pair search
    if time.time() < deadline - 40:
        t_end = min(time.time() + 40, deadline - 10)
        h, best_c = pair_search(h, n, t_end, steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    # Phase 8: Final top-shifts polish
    if time.time() < deadline - 5:
        sidx, _ = get_active_shifts(h, n, tol_frac=0.001, tol_abs=2e-6, top_k=5)
        if len(sidx) > 0:
            h_new, c_new = run_lbfgs_top(h, n, sidx, 5000000, 0.1, deadline)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new

    # Final feasibility
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    if best_c > compute_c5(best_h, n):
        h = best_h
        best_c = compute_c5(h, n)

    return h, best_c, n
