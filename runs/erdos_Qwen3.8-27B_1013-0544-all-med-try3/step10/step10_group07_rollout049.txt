import time

            if s < 1e-14:

            h = h + diff / s * h

def binary_from_continuous(h_cont, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h_cont)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def sa_binary_search(h_bin, n, deadline, max_steps=2000000, seed=42):
    """Simulated annealing on binary landscape with fixed count."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    two_n_minus_1 = 2 * n - 1
    
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max()) * dx
    
    best_c = cur_c
    best_h = h.copy()
    
    ones_pos = np.where(h == 1)[0].copy()
    zeros_pos = np.where(h == 0)[0].copy()
    n_ones = len(ones_pos)
    
    # SA schedule
    T0 = 0.15 * dx
    T_min = 1e-9 * dx
    n_eff = min(max_steps, 2000000)
    cooling = (T_min / T0) ** (1.0 / n_eff)

    best_T = T0
    
    for step in range(n_eff):
        if time.time() > deadline:
            break

        ii = rng.randint(n_ones)
        ji = rng.randint(n_ones)
        i = zeros_pos[ii]  # position to flip 0->1
        j = ones_pos[ji]   # position to flip 1->0

        im = i + m_range
        imn = i - m_range
        jm = j + m_range
        jmn = j - m_range

        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        msk = (jm >= 0) & (jm < n)

        msk = (jmn >= 0) & (jmn < n)

        delta[n-1] -= 2.0

        delta[k1] += 1.0
        k2 = n - 1 + (i - j)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh_max = sh.max()
        lse = sh_max + np.log(np.exp(sh - sh_max).sum())
        return m + lse / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
