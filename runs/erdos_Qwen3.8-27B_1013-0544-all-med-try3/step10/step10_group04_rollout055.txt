            if tot < 1e-14:

    if mx > 500:

                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    
    return float(val), grad

def lbfgs_optimize(h, n, deadline, alphas=(100, 500, 2000, 10000, 50000), maxiter_per_alpha=5000):

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_fn(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    val = m + np.log(np.sum(np.exp(sh - mx)) + mx) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] -= wm * x[0:nv]

            nv = n - kk

                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] -= wm * x[kk:n]
    if mu > 0:

    return float(val), grad

def msequence_init(n, rng):
    """Generate an m-sequence-like binary pattern."""
    # Use a simple LFSR to generate a pseudo-random binary sequence
    m = max(1, n.bit_length() - 1)
    # Generate 2^m - 1 bits using LFSR
    poly = {8, 14} if m >= 4 else {1, 2}
    state = 1
    seq = []
    for _ in range(2**m - 1):
        seq.append(state & 1)
        # LFSR step (simplified)
        fb = ((state >> 0) ^ (state >> 3) ^ (state >> 4) ^ (state >> 5)) & 1
        state = (state >> 1) | (fb << (m-1))
        state &= (1 << m) - 1
    
    # Tile to fill n positions
    h = np.zeros(n)
    for i in range(n):
        h[i] = seq[i % len(seq)]
    
    # Fix to have exactly n/2 ones
    target = n // 2
    ones = np.sum(h)
    if ones > target:
        # Remove excess ones
        idx = np.where(h == 1)[0]
        to_remove = rng.choice(idx, int(ones - target), replace=False)
        h[to_remove] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        to_add = rng.choice(idx, int(target - ones), replace=False)
        h[to_add] = 1
    
    return h

def hill_climb_binary(h, n, deadline):
    """Hill climbing on binary solutions via pair swaps."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    pos_ones = np.where(h > 0.5)[0]
    pos_zeros = np.where(h < 0.5)[0]
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        best_c = compute_c5(best_h, n)
        pos_ones = np.where(best_h > 0.5)[0]
        pos_zeros = np.where(best_h < 0.5)[0]
        
        # Try all pairs (swap a 1 and a 0)
        # For efficiency, only try a subset per pass
        n_tries = min(len(pos_ones) * len(pos_zeros), 50000)
        
        for trial in range(n_tries):
            if time.time() > deadline:
                return best_h, best_c
            
            if trial < len(pos_ones) * len(pos_zeros):
                i = pos_ones[trial % len(pos_ones)]
                j = pos_zeros[(trial // len(pos_ones)) % len(pos_zeros)]
            else:
                break
            
            if i == j:
                continue
            
            # Swap: h[i] 1→0, h[j] 0→1
            best_h[i] = 0.0
            best_h[j] = 1.0
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                improved = True
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_val(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def obj_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    val = m + np.log(np.sum(np.exp(sh - mx)) + mx) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] -= wm * x[0:nv]
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] -= wm * x[kk:n]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return float(val), grad

def lbfgs_optimize(h, n, deadline, alphas=None, maxiter=5000):
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 80000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        mu = max(alpha * 0.05, 1.0)
        try:
            res = minimize(obj_val, best_h, jac=obj_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def threshold_binary(h, n):
    h_bin = (h >= 0.5).astype(float)
    target = n / 2.0
    diff = target - h_bin.sum()
    if abs(diff) < 1e-12:
        return h_bin
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            if h_bin[i] < 1.0 and diff > 0.5:
                h_bin[i] = 1.0
                diff -= 1.0
    else:
        order = np.argsort(h)

            if h_bin[i] > 0.0 and diff < -0.5:
                h_bin[i] = 0.0
                diff += 1.0
    return project_feasible(h_bin, n)

def slqp_optimize(h, n, deadline, alpha=500000):
