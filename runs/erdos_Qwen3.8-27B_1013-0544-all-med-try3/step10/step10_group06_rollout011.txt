def block_perturbation(h, n, rng, block_sizes=None):
    """Apply a block-based perturbation."""
    if block_sizes is None:
        block_sizes = [5, 10, 20, 50]
    
    h = h.copy()
    b = block_sizes[rng.randint(len(block_sizes))]
    i = rng.randint(0, max(1, n - b - 1))
    
    r = rng.random()
    if r < 0.3:
        # Reverse block
        h[i:i+b] = h[i:i+b][::-1]
    elif r < 0.5:
        # Circular shift block
        shift = rng.randint(1, b // 2 + 1) * (1 if rng.random() < 0.5 else -1)
        h[i:i+b] = np.roll(h[i:i+b], shift)
    elif r < 0.7:
        # Swap two blocks
        b2 = min(b, n - i - b - 1)
        if b2 > 2:
            j = i + b + rng.randint(0, max(1, b2 - 1))
            tmp = h[i:i+b].copy()
            h[i:i+b] = h[j:j+b].copy()
            h[j:j+b] = tmp
    elif r < 0.85:
        # Add noise to block
        h[i:i+b] += rng.normal(0, 0.05, b)
    else:
        # Exchange two positions
        i2 = rng.randint(0, n)
        h[i], h[i2] = h[i2], h[i]
    
    return project_feasible(h, n)

        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            T = max(T, 0.002)
            it_since_improve = 0
    
    return best_h, best_c
