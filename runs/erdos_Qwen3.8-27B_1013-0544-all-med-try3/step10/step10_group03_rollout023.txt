def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def full_grad(h, n, alpha, mu, active=None):
    """Corrected gradient of soft-max of correlation overlaps."""
    c = compute_c5_vec(h, n)
    if active is not None:
        vals = c[active]
        m = vals.max()
        sh = alpha * (vals - m)
        smx = sh.max()
        w_full = np.zeros(2 * n - 1)
        w_active = np.exp(sh - smx)
        w_active /= w_active.sum()
        for wi, idx in zip(w_active, active):
            w_full[idx] = wi
    else:
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()
        w_full = np.exp(sh - smx)
        w_full /= w_full.sum()

    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w_full[mm]
        if wm < 1e-15:
            continue
        k = mm - (n - 1)
        wm_dx = wm * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[k:n] += wm_dx * (1.0 - h[0:nv])
                grad[0:nv] += wm_dx * (-h[k:n])
        else:
            s = -k
            nv = n - s
            if nv > 0:
                grad[0:nv] += wm_dx * (1.0 - h[s:n])
                grad[s:n] += wm_dx * (-h[0:nv])
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def full_obj(h, n, alpha, mu, active=None):

        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
    else:
        m = c.max()
        sh = alpha * (c - m)
        smx = sh.max()

        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def run_lbfgs(h, n, deadline, configs, active=None):

    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:

            break

def simulated_anneal(h, n, deadline, seed=42):

    best_c = compute_c5(h, n)
    curr_c = best_c
    rng = np.random.default_rng(seed)
    t_start = time.time()
    elapsed_total = max(1, deadline - t_start)
    n_iter = int(elapsed_total * 8000)

    T0 = 0.005
    Tend = 0.00001

    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter, 1)
        T = T0 * (Tend / T0) ** frac

        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        d = rng.uniform(0.002, 0.03) * (1 if rng.random() < 0.5 else -1)

        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue

        h[i] += d
        h[j] -= d
        curr_c = compute_c5(h, n)

        delta = curr_c - best_c
        if delta < 0:
            best_c = curr_c
            best_h = h.copy()
        elif rng.random() < np.exp(-delta / max(T, 1e-12)):
            pass
        else:
            h[i] -= d
            h[j] += d

    return best_h, best_c

def antisymmetric_init(n, seed=42):
