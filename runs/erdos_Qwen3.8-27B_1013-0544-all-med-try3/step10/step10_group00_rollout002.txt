import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def fast_c5_binary(h, n):
    """Compute C5 for binary h more efficiently by tracking counts"""
    # For binary h, correlate(h, 1-h) at shift k is:
    # sum_i h[i] * (1-h[i+k]) for valid i
    # = sum_i h[i] - h[i] * h[i+k] (where both are defined)
    # For h in {0,1}: h[i] - h[i]*h[i+k]
    # So the correlation at shift k is: (number of ones in the overlap) - (number of 1,1 pairs at shift k)
    # Actually for full mode, the overlap shrinks.
    # Let's just use numpy for simplicity but it's O(n^2)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0/n))

def simulated_annealing_binary(h, n, deadline, rng, T0=0.001, T_end=1e-8, max_iters=200000):
    """Simulated annealing on binary configuration"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    dx = 2.0 / n
    
    n_iter = 0
    total_iters = max_iters
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max(1, total_iters - 1))
        
        # Pick random 1 and random 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
        delta = c - cur_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # Accept
            cur_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c

def beatty_init(n, alpha, rng):
    """Initialize using Beatty sequence"""
    h = np.zeros(n)
    target = n // 2
    used = set()
    k = 1
    while len(used) < target:
        pos = int(k * alpha) % n
        if pos not in used:
            used.add(pos)
            h[pos] = 1.0
        k += 1
        if k > 10 * n:
            break
    # Fix if we didn't get exactly target
    diff = target - int(h.sum())
    if diff > 0:
        remaining = [i for i in range(n) if h[i] == 0]
        rng.shuffle(remaining)
        for i in remaining[:diff]:
            h[i] = 1.0
    elif diff < 0:
        remaining = [i for i in range(n) if h[i] == 1]
        rng.shuffle(remaining)
        for i in remaining[:(-diff)]:
            h[i] = 0.0
    return h

def thue_morse_init(n, bit=0):
    """Initialize using Thue-Morse sequence"""
    tm = np.zeros(1)
    while len(tm) < n:
        tm = np.concatenate([tm, 1.0 - tm])
    h = tm[:n].copy()
    diff = n // 2 - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        rng_local = np.random.RandomState(0)
        rng_local.shuffle(zeros)
        for i in zeros[:diff]:
            h[i] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng_local = np.random.RandomState(0)
        rng_local.shuffle(ones)
        for i in ones[:(-diff)]:
            h[i] = 0.0
    return h
