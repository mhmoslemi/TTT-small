    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:

            if tot < 1e-15:

            h += (diff / tot) * h

            if final_diff > 0:
                add = min(final_diff, 1.0 - h[i])

        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def get_active_set(c_vec, top_k, rel_tol=0.001):
    max_val = c_vec.max()
    threshold = max_val - max(1e-10, rel_tol * max_val)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    return active

def lbfgs_call(obj, grad, x0, n, args, bounds, maxiter):
    try:
        res = minimize(obj, x0, jac=grad, args=args, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-24,
                                'gtol': 1e-18, 'maxfun': maxiter * 12,
                                'maxls': 40, 'maxcor': 50})
        return np.array(res.x)
    except Exception:
        return x0.copy()

def do_active_set_phase(h, n, deadline, top_k_list, alpha_list, rel_tol=0.001,
                        mu=0.001, max_rounds=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    offset = n - 1
    rnd = 0
    while rnd < max_rounds and time.time() < deadline - 20:
        c_vec = compute_c5_vec(best_h, n)
        for top_k in top_k_list:
            if time.time() > deadline - 15:
                break
            active = get_active_set(c_vec, top_k, rel_tol)
            if len(active) == 0:
                continue
            for alpha in alpha_list:
                if time.time() > deadline - 12:
                    break
                maxit = max(200, int(min(5000, (deadline - time.time()) * 3)))
                x_new = lbfgs_call(topK_obj, topK_grad, best_h, n,
                                   (n, active, alpha, mu), bounds, maxit)
                x_new = project_feasible(x_new, n)
                c_new = compute_c5(x_new, n)
                if c_new < best_c - 1e-16:
                    best_c = c_new
                    best_h = x_new
        rnd += 1
    return best_h, best_c

def do_single_lag_phase(h, n, deadline, n_lags=3, mu=0.01, alpha=5000000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    offset = n - 1
    while time.time() < deadline - 15:
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        active = get_active_set(c_vec, n_lags, 0.0005)
        if len(active) == 0:
            break
        for idx in sorted(active, key=lambda x: -c_vec[x]):
            if time.time() > deadline - 12:
                break
            single = np.array([idx])
            others = np.array([x for x in active if x != idx])
            # Minimize this lag + penalty on others
            def obj_single(x, n, act1, act2, al, m):
                c = compute_c5_vec(x, n)
                v1 = c[act1].max()
                penalty = 0.0
                if len(act2) > 0:
                    excess = np.maximum(0, c[act2] - v1)
                    penalty = np.sum(excess ** 2)
                val = v1 + al * penalty
                if m > 0:
                    val += m * (x.sum() - n / 2.0) ** 2
                return float(val)
            def grad_single(x, n, act1, act2, al, m):
                c = compute_c5_vec(x, n)
                v1 = c[act1].max()
                w1 = smooth_max_weights(c[act1], 1000)
                dx = 2.0 / n
                g = np.zeros(n)
                for wi, a in zip(w1, act1):
                    k = a - offset
                    g += wi * grad_for_lag(x, n, k, dx)
                if len(act2) > 0:
                    excess = np.maximum(0.0, c[act2] - v1)
                    for j, a in enumerate(act2):
                        if excess[j] > 1e-15:
                            k = a - offset
                            g += al * 2.0 * excess[j] * grad_for_lag(x, n, k, dx)
                if m > 0:
                    g += 2.0 * m * (x.sum() - n / 2.0)
                return g
            maxit = max(100, int(min(2000, (deadline - time.time()) * 2)))
            x_new = lbfgs_call(obj_single, grad_single, best_h, n,
                               (n, single, others, 50.0, mu), bounds, maxit)
            x_new = project_feasible(x_new, n)
            c_new = compute_c5(x_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = x_new
    return best_h, best_c

def do_coord_descent(h, n, deadline, max_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    offset = n - 1
    dx = 2.0 / n
    step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                  0.00002, 0.00001, 0.000005, 0.000002, 0.000001]
    steps_done = 0
    for step in step_sizes:
        stall = 0
        while time.time() < deadline - 5 and steps_done < max_steps and stall < 500:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-10, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > 5:
                active = np.array(np.argsort(-c_vec[active])[:5])
            grad = np.zeros(n)
            for idx in active:
                k = idx - offset
                grad += grad_for_lag(best_h, n, k, dx)
            grad /= max(1, len(active))
            pos_mask = (grad > 1e-12) & (best_h > 1e-10)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-10)
            if not pos_mask.any() or not neg_mask.any():
                stall += 100
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            # Find best pair by gradient difference
            best_gain = 0.0
            best_i = -1
            best_j = -1
            p_sorted = pos_idx[np.argsort(-grad[pos_idx])][:30]
            n_sorted = neg_idx[np.argsort(grad[neg_idx])][:30]
            for pi in p_sorted:
                for ni in n_sorted:
                    g = grad[pi] - grad[ni]
                    if g > best_gain:
                        best_gain = g
                        best_i = pi
                        best_j = ni
            if best_i < 0 or best_gain < 1e-14:
                stall += 100
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-14:
                stall += 50
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            steps_done += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
                stall = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stall += 1
    return best_h, best_c

def do_fine_pair(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002,
                  0.00001, 0.000005, 0.000002]
    for d in deltas:
        improved_any = False
        for i in range(n):
            if time.time() > deadline - 3:
                return best_h, best_c
            lo = max(0, i - radius)
            hi = min(n, i + radius + 1)
            for j in range(lo, hi):
                if i == j:
                    continue
                # Try moving mass from i to j
                if best_h[i] >= d and best_h[j] + d <= 1.0 + 1e-12:
                    best_h[i] -= d
                    best_h[j] = min(best_h[j] + d, 1.0)
                    # Fix sum
                    diff = (n / 2.0) - best_h.sum()
                    if abs(diff) > 1e-14:
                        best_h[i] += d
                        best_h[j] -= d
                        continue
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved_any = True
                    else:
                        best_h[i] += d
                        best_h[j] -= d
                # Try moving mass from j to i
                if best_h[j] >= d and best_h[i] + d <= 1.0 + 1e-12:
                    best_h[j] -= d
                    best_h[i] = min(best_h[i] + d, 1.0)
                    diff = (n / 2.0) - best_h.sum()
                    if abs(diff) > 1e-14:
                        best_h[j] += d
                        best_h[i] -= d
                        continue
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-16:
                        best_c = c
                        improved_any = True
                    else:
                        best_h[j] += d
                        best_h[i] -= d
        if not improved_any and d < 0.0001:
            break
    return best_h, best_c

def slqp_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha)

def slqp_sum_eq(x, n):
    return x.sum() - n / 2.0

def slqp_sum_jac(x, n):
    return np.ones(n)

def do_slqp(h, n, deadline, alpha_list=None, maxiter=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if alpha_list is None:
        alpha_list = [1000000, 10000000, 100000000]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slqp_sum_eq, 'jac': slqp_sum_jac, 'args': (n,)}]
    for alpha in alpha_list:
        if time.time() > deadline - 10:
            break
        try:
            rem = max(5, int(deadline - time.time() - 5))
            maxit_use = min(maxiter, max(100, rem * 5))
            res = minimize(slqp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit_use, 'ftol': 1e-22,
                                    'iprint': 0})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = []
    seen = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in seen:
            seen.add(pos)
            positions.append(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Try alternative starting points
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty_start(n, golden / (2.0 * np.sqrt(2))),
        make_beatty_start(n, np.sqrt(2.0) - 1.0),
        make_beatty_start(n, 1.0 / np.sqrt(3.0)),
    ]
    for _ in range(3):
        hb = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        hb[idx] = 1.0
        alt_starts.append(project_feasible(hb, n))

    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()

    # Phase 1: Full smooth max L-BFGS warmup (0 - 120s)
    p1_deadline = t0 + 120
    p1_alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 5000000]
    bounds = [(0.0, 1.0)] * n
    for alpha in p1_alphas:
        if time.time() > p1_deadline - 10:
            break
        maxit = max(300, int(min(3000, (p1_deadline - time.time()) * 4)))
        x_new = lbfgs_call(full_obj, full_grad, best_h, n, (n, alpha, 0.01), bounds, maxit)
        x_new = project_feasible(x_new, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = x_new

    # Phase 2: Aggressive active-set optimization (120s - 450s)
    p2_deadline = t0 + 450
    best_h, best_c = do_active_set_phase(
        best_h, n, p2_deadline,
        top_k_list=[2, 3, 4, 6, 10],
        alpha_list=[500000, 2000000, 10000000, 50000000, 200000000],
        rel_tol=0.001,
        mu=0.001,
        max_rounds=60
    )

    # Phase 3: Single-lag alternating (450s - 650s)
    p3_deadline = t0 + 650
    best_h, best_c = do_single_lag_phase(
        best_h, n, p3_deadline,
        n_lags=3, mu=0.01, alpha=5000000
    )

    # Phase 4: Another round of active set with higher precision (650s - 780s)
    p4_deadline = t0 + 780
    best_h, best_c = do_active_set_phase(
        best_h, n, p4_deadline,
        top_k_list=[2, 3, 5],
        alpha_list=[10000000, 100000000, 500000000],
        rel_tol=0.0005,
        mu=0.0005,
        max_rounds=40
    )

    # Phase 5: Gradient-guided coordinate descent (780s - 920s)
    p5_deadline = t0 + 920
    best_h, best_c = do_coord_descent(best_h, n, p5_deadline, max_steps=300000)

    # Phase 6: Fine pair search (920s - 1020s)
    p6_deadline = t0 + 1020
    best_h, best_c = do_fine_pair(best_h, n, p6_deadline, radius=50,
        deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])

    # Phase 7: SLSQP polish (1020s - 1070s)
    p7_deadline = t0 + 1070
    best_h, best_c = do_slqp(best_h, n, p7_deadline,
        alpha_list=[10000000, 100000000, 1000000000], maxiter=400)

    # Phase 8: Final coordinate descent + pair (1070s - deadline)
    p8_deadline = deadline - 3
    if time.time() < p8_deadline - 30:
        best_h, best_c = do_coord_descent(best_h, n, p8_deadline - 30, max_steps=100000)
    if time.time() < p8_deadline - 5:
        best_h, best_c = do_fine_pair(best_h, n, p8_deadline - 3, radius=30,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002])

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
