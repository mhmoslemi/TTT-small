        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=(100, 500, 2000, 10000, 50000)):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        mu = alpha * 0.01
        try:
            res = minimize(smooth_objective, h, jac=smooth_gradient,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def lp_optimize(h, n, deadline, top_k=15, max_iters=5000):
    """Linearized LP optimization."""
    h_curr = project_feasible(h, n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_overlap_vec(h_curr, n)
        worst_idx = np.argsort(-c_vec)[0:top_k]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        
        for idx in worst_idx:
            k = int(idx) - (n - 1)
            f_val = c_vec[idx]
            f_grad = subgrad_at_shift(h_curr, n, k) * dx
            # Linearization: f_k(h_new) >= f_k(h) + grad * (h_new - h) for convex f_k
            # But f_k is NOT convex. Still, use it as a heuristic direction.
            cons.append(f_val + f_grad @ (h_var - h_curr) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = project_feasible(np.array(h_var.value).flatten(), n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        
        # If no progress after many iterations, break
        if c_new > best_c + 1e-6:
            h_curr = best_h.copy()
        else:
            h_curr = h_new
        
        if it > 100 and c_new >= best_c - 1e-10:
            # stagnation
            # try a different set of constraints
            top_k = min(top_k + 5, 50)
    
    return best_h, best_c

def subgradient_refine(h, n, deadline, n_trials=3000):
    """Subgradient descent with momentum on exact max."""
    h_curr = project_feasible(h, n)
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    
    eta = 0.02
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(n_trials):
        if time.time() > deadline:
            break
        
        c_vec = compute_overlap_vec(h_curr, n)
        k_star = int(np.argmax(c_vec)) - (n - 1)
        
        # Use top 3 worst shifts for better direction
        top_idx = np.argsort(-c_vec)[0:3]
        sg = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            sg += subgrad_at_shift(h_curr, n, k)
        sg /= 3.0
        
        v = momentum * v - eta * sg
        h_new = project_feasible(h_curr + v, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Adaptive step
        if it % 100 == 0:
            if c_new < best_c:
                eta = min(eta * 1.1, 0.1)
            else:
                eta = max(eta * 0.8, 1e-5)
        
        h_curr = h_new
    
    return best_h, best_c

def binary_search(h, n, deadline, max_iters=50000):
    """Local search on binary solutions."""
    h_bin = make_binary(h, n)
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    no_improve = 0
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_overlap_vec(best_h, n)
        k_star = int(np.argmax(c_vec)) - (n - 1)
        
        # Find contributing positions for worst shift
        if k_star > 0:
            valid = np.arange(0, n - k_star)
            contributing = valid[(best_h[valid] > 0.5) & (best_h[valid + k_star] < 0.5)]
        elif k_star < 0:
            kk = -k_star
            valid = np.arange(kk, n)
            contributing = valid[(best_h[valid] > 0.5) & (best_h[valid - kk] < 0.5)]
        else:
            break  # can't happen for binary
        
        if len(contributing) == 0:
            break
        
        # Try flipping a contributing 1 to 0, and a random 0 to 1
        i = contributing[rng.randint(len(contributing))]
        zeros = np.where(best_h < 0.5)[0]
        if len(zeros) == 0:
            break
        j = zeros[rng.randint(len(zeros))]
        
        best_h[i], best_h[j] = best_h[j], best_h[i]
        c_new = compute_c5(best_h, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            no_improve = 0
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
            no_improve += 1
        
        if no_improve > 500:
            # Restart with different random seed
            no_improve = 0
            # Temporarily accept a worse solution to escape
            i = contributing[rng.randint(len(contributing))]
            zeros = np.where(best_h < 0.5)[0]
            j = zeros[rng.randint(len(zeros))]
            best_h[i], best_h[j] = best_h[j], best_h[i]
    
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
