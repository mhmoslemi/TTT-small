def softmax_top(vals, beta=20.0):

    s=exp.sum()
    if s<=0: return np.ones(len(v))/len(v)

            free = (1.0 - x) > 1e-12
            if not np.any(free): break
            room = 1.0 - x
            s = room[free].sum()
            if s <= 0: break
            step = diff / s
            x[free] += room[free] * step
            x[free] = np.minimum(1.0, x[free])

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def next_fast_len(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _adjust_sum_inplace(x, target):
    diff = target - float(x.sum())
    for _ in range(8):
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            free = (1.0 - x) > 1e-12
            if not np.any(free):
                break
            room = 1.0 - x
            s = float(room[free].sum())
            if s <= 0.0:
                break
            step = diff / s
            x[free] += room[free] * step
            x[free] = np.minimum(1.0, x[free])
        else:
            free = x > 1e-12
            if not np.any(free):
                break
            s = float(x[free].sum())
            if s <= 0.0:
                break
            step = (-diff) / s
            x[free] -= x[free] * step
            x[free] = np.maximum(0.0, x[free])
        diff = target - float(x.sum())

    diff = target - float(x.sum())
    if abs(diff) > 1e-15:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            add = min(1.0 - x[idx], diff)
            x[idx] += add
        else:
            idx = int(np.argmax(x))
            sub = min(x[idx], -diff)
            x[idx] -= sub
    return x


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    target = float(target)
    n = y.size
    if n == 0:
        return y

    ymin = float(np.min(y))
    ymax = float(np.max(y))

    if ymin >= -1e-12 and ymax <= 1.0 + 1e-12:
        x = np.clip(y, 0.0, 1.0)
        if abs(float(x.sum()) - target) < 1e-8:
            return _adjust_sum_inplace(x, target)

    lo = ymin - 1.0
    hi = ymax + 1.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)
        s = float(z.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    return _adjust_sum_inplace(x, target)


def c_counts_fft(x, n, P):
    y = 1.0 - x
    X = np.fft.rfft(x, P)
    Y = np.fft.rfft(y, P)
    r = np.fft.irfft(X * np.conj(Y), P)
    if n == 1:
        return r[:1].copy()
    return np.concatenate((r[-(n - 1):], r[:n]))


def c_counts_arr(x, n, use_fft, P):
    if use_fft:
        return c_counts_fft(x, n, P)
    return np.correlate(x, 1.0 - x, mode="full")


def c5_from_counts(c, n):
    return float(np.max(c)) * (2.0 / float(n))


def c5_exact(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / float(n))


def setup_c(n):
    if n > 600:
        return True, next_fast_len(2 * n - 1)
    return False, 0


def resample_h(h, n_from, n_to):
    if n_from == n_to:
        return np.array(h, dtype=float).copy()
    x = (np.arange(n_from) + 0.5) * (2.0 / float(n_from))
    y = (np.arange(n_to) + 0.5) * (2.0 / float(n_to))
    return np.interp(y, x, h)


def random_balanced(n, seed):
    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    target = 0.5 * n
    m = int(np.floor(target))
    idx = rng.choice(n, size=m, replace=False)
    h = np.zeros(n, dtype=float)
    h[idx] = 1.0
    rem = target - m
    if abs(rem) > 1e-12 and n > 0:
        used = np.zeros(n, dtype=bool)
        used[idx] = True
        cand = np.where(~used)[0]
        if cand.size > 0:
            h[cand[0]] = rem
    return h


def top_indices(c, m):
    L = len(c)
    if m <= 0:
        return np.array([], dtype=int)
    if m >= L:
        return np.arange(L)
    return np.argpartition(c, -m)[-m:]


def softmax_top(vals, beta=20.0):
    vals = np.asarray(vals, dtype=float)
    m = vals.size
    if m == 0:
        return vals
    mx = float(np.max(vals))
    z = beta * (vals - mx)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(m, dtype=float) / float(m)
    return e / s


def gradient_top_lags(x, idxs, weights, n):
    g = np.zeros(n, dtype=float)
    if idxs.size == 0:
        return g
    y = 1.0 - x
    center = n - 1
    for pos in range(idxs.size):
        idx = int(idxs[pos])
        w = float(weights[pos])
        if w == 0.0:
            continue
        k = idx - center
        if k == 0:
            g += w * (y - x)
        elif k > 0:
            m = n - k
            g[k:] += w * y[:m]
            g[:m] -= w * x[k:]
        else:
            q = -k
            m = n - q
            g[:m] += w * y[q:]
            g[q:] -= w * x[:m]
    return g


def unit_plus(p, h, one_minus, n, ar):
    L = 2 * n - 1
    up = np.zeros(L, dtype=float)
    idx1 = n - 1 + p - ar
    up[idx1] += one_minus
    idx2 = n - 1 + ar - p
    up[idx2] -= h
    return up


def find_best_pair(c, h, n, g, pos, neg, pos_units, neg_units, fracs, tmp):
    if n < 2:
        return None

    L = 2 * n - 1
    base_max = float(np.max(c))
    base_score = base_max + 1e-7 * float(np.sum(c))
    best_primary = base_max - 1e-12
    best_score = base_score - 1e-12
    best = None

    for pi in range(len(pos)):
        ip = int(pos[pi])
        ua = pos_units[pi]
        xip = float(h[ip])
        if xip <= 1e-9:
            continue
        for qi in range(len(neg)):
            iq = int(neg[qi])
            if ip == iq:
                continue
            xiq = float(h[iq])
            if xiq >= 1.0 - 1e-9:
                continue
            if float(g[iq] - g[ip]) > -1e-8:
                continue

            dmax = xip
            room = 1.0 - xiq
            if room < dmax:
                dmax = room
            if dmax <= 1e-11:
                continue

            ub = neg_units[qi]
            lin = ub - ua
            for f in fracs:
                d = float(f) * dmax
                if d <= 1e-13:
                    continue

                np.multiply(lin, d, out=tmp)
                tmp += c
                d2 = d * d
                r1 = n - 1 + ip - iq
                r2 = n - 1 + iq - ip
                rc = n - 1

                old1 = tmp[r1]
                old2 = tmp[r2]
                oldc = tmp[rc]
                tmp[r1] += d2
                tmp[r2] += d2

                            take = True

                if take:
                    best = (ip, iq, d, lin.copy())

                tmp[r1] = old1
                tmp[r2] = old2
                tmp[rc] = oldc

def try_gradient_step(h, n, idxs, weights, target, use_fft, P, c, base_max, step_fracs):
    g = gradient_top_lags(h, idxs, weights, n)
    if float(np.max(np.abs(g))) < 1e-10:
        return None

    best_ret = None
    best_mx = base_max - 1e-11
    for eta in step_fracs:
        xnew = project_box_sum(h - float(eta) * g, target)
        cnew = c_counts_arr(xnew, n, use_fft, P)
        mx = float(np.max(cnew))
        if mx < best_mx:
            best_mx = mx
            best_ret = (xnew, cnew, mx)
    return best_ret

                    scale = 0.06
                h2 = project_box_sum(h + scale * (rng.rand(n) * 2.0 - 1.0), target)
                c2 = c_counts_arr(h2, n, use_fft, P)
                mx2 = float(np.max(c2))
                if mx2 * dx < best_c + 0.08:
                    h = h2
                    c = c2
                else:
                    h = best_h.copy()
                    c = c_counts_arr(h, n, use_fft, P)
                stuck = 0
                no_primary = 0

    h = project_box_sum(h, target)
    c = c_counts_arr(h, n, use_fft, P)
    cval = c5_from_counts(c, n)
    if cval < best_c:
        best_c = cval
        best_h = h.copy()

    bh = project_box_sum(best_h, target)
    bh_exact = c5_exact(bh, n)
    if bh_exact < best_c:
        best_c = bh_exact
        best_h = bh
    else:
        best_h = bh

    hh = project_box_sum(h, target)
    hh_exact = c5_exact(hh, n)
    if hh_exact < best_c:
        best_c = hh_exact
        best_h = hh

    if n % 2 == 0:
        rounded = np.round(best_h)
        if float(np.max(np.abs(best_h - rounded))) < 1e-7:
            rproj = project_box_sum(rounded, target)
            rc = c5_exact(rproj, n)
            if rc <= best_c + 1e-12:
                best_h = rproj
                if rc < best_c:
                    best_c = rc

    return best_h, float(best_c)


def slsqp_obj(x, idxs, weights, n, use_fft, P):
    c = c_counts_arr(x, n, use_fft, P)
    return float(np.dot(weights, c))


def slsqp_jac(x, idxs, weights, n, use_fft, P):
    return gradient_top_lags(x, idxs, weights, n)


def eq_constraint(x, target):
    return float(x.sum() - float(target))


def eq_constraint_jac(x, target):
    return np.ones(len(x), dtype=float)


def try_slsqp_polish(h, n, use_fft, P, target):
    if minimize is None or n > 450:
        return h, c5_exact(h, n)
    try:
        c = c_counts_arr(h, n, use_fft, P)
        idxs = top_indices(c, 4)
        weights = softmax_top(c[idxs], 10.0)
        bounds = [(0.0, 1.0)] * n
        cons = [
            {
                "type": "eq",
                "fun": eq_constraint,
                "jac": eq_constraint_jac,
                "args": (target,),
            }
        ]
        res = minimize(
            slsqp_obj,
            h.copy(),
            args=(idxs, weights, n, use_fft, P),
            jac=slsqp_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 90, "ftol": 1e-12},
        )
        hnew = project_box_sum(res.x, target)
        cnew = c5_exact(hnew, n)
        return hnew, cnew
    except Exception:
        return h, c5_exact(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    safe_end = t0 + max(10.0, budget - 8.0)

    best_h = None

        n0 = nr

    best_n = n0
    best_h = project_box_sum(h0, 0.5 * float(best_n))
    best_c = c5_exact(best_h, best_n)

    def update_candidate(cand_h, cand_c, cand_n):

        nonlocal_best_n = best_n
        if cand_h is None:

        if cand_c < nonlocal_best_c:

            nonlocal_best_n = cand_n
            nonlocal_best_c = cand_c

    # The above helper closure would violate the requested style, so perform updates inline below.

                              seed=int(seed))

            best_h = hh

    # Phase 1b: small perturbation restart at the same resolution.

    if time.time() < t1b - 10.0 and best_n > 0:
        hp = project_box_sum(best_h + 0.02 * (rng.rand(best_n) * 2.0 - 1.0), 0.5 * float(best_n))
        usef, P = setup_c(best_n)
        hh, cc = local_search(hp, best_n, t1b, usef, P,
                              m_lags=6 if best_n <= 300 else 5,
                              K=5 if best_n <= 300 else 4,
                              seed=int(seed) + 1)
        if hh is not None and cc < best_c:
            best_c = cc
            best_h = hh
            best_n = best_n

    # Phase: short SLSQP polish on a fixed leading-lag objective.
    t2 = t0 + 0.50 * total
    if time.time() < t2 - 30.0 and 0 < best_n <= 450 and minimize is not None:
        usef, P = setup_c(best_n)
        hh, cc = try_slsqp_polish(best_h, best_n, usef, P, 0.5 * float(best_n))
        if hh is not None and cc < best_c:
            best_c = cc
            best_h = hh
            best_n = best_n

    # Phase 2: try a nearby higher resolution.
    if time.time() < t2 - 20.0 and best_n != 350:
        nh = 350
        hstart = project_box_sum(resample_h(best_h, best_n, nh), 0.5 * float(nh))
        usef2, P2 = setup_c(nh)

    best_c = c5_exact(best_h, best_n)
