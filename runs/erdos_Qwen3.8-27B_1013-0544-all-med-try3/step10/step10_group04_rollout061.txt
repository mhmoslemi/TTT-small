            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        widx = w[idx]
        if widx < 1e-13:
            continue
        k = idx - center
        wm = widx * dx_val
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, bounds_override=None):
    h = project_feasible(h, n)
    best_c = compute_c5_fft(h, n)
    if bounds_override is not None:
        bounds = bounds_override
    else:
        bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (5000, 5000.0), (20000, 20000.0), (50000, 50000.0),
                   (100000, 100000.0), (200000, 200000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 30000, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': 300000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5_fft(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def binary_freeze_optimize(h, n, deadline, thresholds=None):
    if thresholds is None:
        thresholds = [0.02, 0.05, 0.1]
    best_h = h.copy()
    best_c = compute_c5_fft(h, n)
    for thr in thresholds:
        if time.time() > deadline - 20:
            break
        frozen_low = h < thr
        frozen_high = h > 1.0 - thr
        n_frozen = frozen_low.sum() + frozen_high.sum()
        if n_frozen < 10 or n_frozen > n - 10:
            continue
        bounds = [(0.0, 1.0)] * n
        for i in range(n):
            if frozen_low[i]:
                bounds[i] = (0.0, 0.0)
            elif frozen_high[i]:
                bounds[i] = (1.0, 1.0)
        h_work = best_h.copy()
        h_work[frozen_low] = 0.0
        h_work[frozen_high] = 1.0
        h_work = project_feasible(h_work, n)
        h_new, c_new = run_lbfgs(h_work, n, deadline - 20,
                                 configs=[(5000, 2000.0), (20000, 10000.0),
                                          (50000, 50000.0), (100000, 100000.0)],
                                 bounds_override=bounds)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5_fft(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_optimize(h, n, deadline, n_trials=6, mags=None, alpha=10000, mu=5000.0):
    best_h = h.copy()
    best_c = compute_c5_fft(h, n)
    rng = np.random.RandomState(54321)
    if mags is None:
        mags = [0.02, 0.04, 0.06, 0.01, 0.08, 0.03]
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5_fft(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def group_refine(h, n, deadline, group_size=3):
    best_c = compute_c5_fft(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(999)
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        worst_idx = int(np.argmax(c_vec))
        k_worst = worst_idx - (n - 1)
        if k_worst >= 0:
            lo, hi = 0, n - k_worst
        else:
            lo, hi = -k_worst, n
        if hi - lo < 1:
            hi, lo = min(hi, n), max(lo, 0)
        candidates = list(range(max(lo, 0), min(hi, n - group_size + 1)))
        if not candidates:
            candidates = list(range(max(0, n - group_size), min(n, n)))
        rng.shuffle(candidates)
        found = False
        for start in candidates[:40]:
            if time.time() > deadline:
                break
            for delta in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                ok = True
                for gi in range(group_size):
                    nv = h[start + gi] + delta
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        ok = False
                        break
                if not ok:
                    continue
                old = h[start:start+group_size].copy()
                for gi in range(group_size):
                    h[start + gi] += delta
                h = project_feasible(h, n)
                c = compute_c5_fft(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h = best_h.copy()
            if found:
                h = best_h.copy()
                break
        if not found:
            break
    return best_h, best_c

def slqp_final(h, n, deadline, alpha=5000000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5_fft(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5_fft(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500
    
    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5_fft(best_h, n_base)
    
    # Phase 1: L-BFGS warmup at base resolution
    t1 = min(t0 + 100, deadline - 800)
    if t1 > time.time() + 10:
        best_h, best_c = run_lbfgs(best_h, n_base, t1,
            configs=[(50, 10.0), (200, 50.0), (1000, 200.0),
                     (5000, 1000.0), (20000, 5000.0), (50000, 20000.0)])
    
    # Phase 2: Binary freeze at base
    t2 = min(t0 + 180, deadline - 700)
    if t2 > time.time() + 15:
        h2, c2 = binary_freeze_optimize(best_h, n_base, t2,
            thresholds=[0.02, 0.05, 0.1])
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 3: Perturbation at base
    t3 = min(t0 + 260, deadline - 600)
    if t3 > time.time() + 10:
        h3, c3 = perturb_optimize(best_h, n_base, t3, n_trials=4,
            mags=[0.03, 0.05, 0.02, 0.04], alpha=5000, mu=2000.0)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Upscale to n=1000
    n1 = 1000
    if time.time() < deadline - 450:
        h1000 = interpolate_to(best_h, n_base, n1)
        t4 = min(time.time() + 130, deadline - 320)
        if t4 > time.time() + 10:
            h1000, c1000 = run_lbfgs(h1000, n1, t4,
                configs=[(500, 200.0), (2000, 1000.0), (5000, 5000.0),
                         (20000, 10000.0), (50000, 30000.0), (100000, 50000.0)])
        if c1000 < best_c:
            best_c = c1000
            best_h = h1000
        
        # Binary freeze at n=1000
        t5 = min(time.time() + 70, deadline - 250)
        if t5 > time.time() + 15:
            h5, c5 = binary_freeze_optimize(best_h, n1, t5,
                thresholds=[0.01, 0.03, 0.05])
            if c5 < best_c:
                best_c = c5
                best_h = h5
        
        # Perturbation at n=1000
        t5b = min(time.time() + 50, deadline - 200)
        if t5b > time.time() + 10:
            h5b, c5b = perturb_optimize(best_h, n1, t5b, n_trials=3,
                mags=[0.02, 0.03, 0.04], alpha=20000, mu=10000.0)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b
    
    # Phase 5: Upscale to n=2000
    n2 = 2000
    if time.time() < deadline - 280:
        h2000 = interpolate_to(best_h, len(best_h), n2)
        t6 = min(time.time() + 120, deadline - 150)
        if t6 > time.time() + 10:
            h2000, c2000 = run_lbfgs(h2000, n2, t6,
                configs=[(2000, 1000.0), (10000, 5000.0), (30000, 15000.0),
                         (80000, 40000.0), (150000, 80000.0)])
        if c2000 < best_c:
            best_c = c2000
            best_h = h2000
        
        # Binary freeze at n=2000
        t7 = min(time.time() + 50, deadline - 70)
        if t7 > time.time() + 12:
            h7, c7 = binary_freeze_optimize(best_h, n2, t7,
                thresholds=[0.01, 0.03])
            if c7 < best_c:
                best_c = c7
                best_h = h7
    
    # Phase 6: SLSQP polish
    if time.time() < deadline - 50:
        t8 = min(time.time() + 35, deadline - 25)
        if t8 > time.time() + 8:
            h8, c8 = slqp_final(best_h, len(best_h), t8,
                alpha=10000000, maxiter=2000)
            if c8 < best_c:
                best_c = c8
                best_h = h8
    
    # Phase 7: Group refinement
    if time.time() < deadline - 20:
        best_h, best_c = group_refine(best_h, len(best_h), deadline - 8, group_size=2)
    
    # Phase 8: Final single-point refinement
    if time.time() < deadline - 5:
        n_f = len(best_h)
        rng_final = np.random.RandomState(42)
        for _ in range(max(1, int((deadline - 3) - time.time()))):
            c_vec = compute_c5_vec(best_h, n_f)
            worst_idx = int(np.argmax(c_vec))
            k_w = worst_idx - (n_f - 1)
            if k_w >= 0:
                lo_i, hi_i = 0, n_f - k_w
            else:
                lo_i, hi_i = -k_w, n_f
            idxs = np.random.choice(n_f, min(20, n_f), replace=False)
            improved = False
            for i in idxs:
                if lo_i <= i < hi_i or (k_w < 0 and (i - k_w >= 0 and i - k_w < n_f)):
                    for d in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]:
                        nv = best_h[i] + d
                        if nv < -1e-12 or nv > 1.0 + 1e-12:
                            continue
                        old_v = best_h[i]
                        best_h[i] = nv
                        best_h = project_feasible(best_h, n_f)
                        c_new = compute_c5_fft(best_h, n_f)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                            break
                        else:
                            best_h[i] = old_v
                            best_h = project_feasible(best_h, n_f)
                    if improved:
                        break
            if not improved:
                break
    
    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5_fft(best_h, n_final)
    
    return best_h, best_c, n_final
