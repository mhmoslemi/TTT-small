                d = hi_val if hi_val > 0 else 1.0 - hi_val
                if j < n and 0 <= d:
                    h_try = best_h.copy()
                    if hi_val < 1.0:
                        mv = min(1.0 - hi_val, best_h[j])
                        if mv > 1e-12:
                            h_try[i] += mv
                            h_try[j] -= mv
                            h_try = project_feasible(h_try, n)
                            c_try = compute_c5(h_try, n)
                            if c_try < best_c - 1e-16:
                                best_c = c_try
                                best_h = h_try
                                improved = True
                    if hi_val > 0.0:
                        mv = min(hi_val, 1.0 - best_h[j])
                        if mv > 1e-12:
                            h_try[i] -= mv
                            h_try[j] += mv
                            h_try = project_feasible(h_try, n)
                            c_try = compute_c5(h_try, n)
                            if c_try < best_c - 1e-16:
                                best_c = c_try
                                best_h = h_try
                                improved = True
        if not improved:
            break
    return best_h, best_c

def pair_refine(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                for j in range(max(0, i - radius), min(n, i + radius + 1)):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
            best_h = project_feasible(best_h, n)
    return best_h, best_c

def subgradient_hedge(h, n, deadline, n_iter=8000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-8, 0.002 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h[k:n]) * dx
                grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h[0:nv]) * dx
                grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.01)
        else:
            eta *= 0.97
        h = h_new
    return best_h, best_c

def make_periodic(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n - 1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Aggressive L-BFGS at parent resolution
    p1_end = t0 + 250
    h1, c1 = run_lbfgs(best_h, n0, p1_end,
        alphas=[20, 50, 100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000],
        mu=0.01, maxit_per=[1000, 1000, 1500, 2000, 2000, 2500, 2500, 3000, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Focused optimization on top peaks at parent resolution
    if time.time() < deadline - 300:
        p2_end = min(time.time() + 100, deadline - 300)
        h2, c2 = run_focused(best_h, n0, p2_end, top_k=6,
            alphas=[50000, 200000, 1000000, 5000000], n_rounds=12)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Upsample to n=1000 and optimize
    n_fine = 1000
    if time.time() < deadline - 300:
        h_fine = upsample(best_h, best_n, n_fine)
        c_fine = compute_c5(h_fine, n_fine)
        p3_end = min(time.time() + 280, deadline - 250)
        h3, c3 = run_lbfgs(h_fine, n_fine, p3_end,
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.005, maxit_per=[800, 1000, 1500, 1500, 2000, 2000, 2500, 2500, 3000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = n_fine

    # Phase 4: Focused on top peaks at fine resolution
    if time.time() < deadline - 200:
        p4_end = min(time.time() + 100, deadline - 180)
        h4, c4 = run_focused(best_h, best_n, p4_end, top_k=5,
            alphas=[200000, 1000000, 5000000, 20000000], n_rounds=10)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: SLSQP polish
    if time.time() < deadline - 120:
        for alpha in [500000, 2000000, 10000000]:
            if time.time() > deadline - 100:
                break
            p5_end = min(time.time() + 30, deadline - 90)
            h5, c5 = run_slqp(best_h, best_n, p5_end, alpha=alpha, maxiter=800)
            if c5 < best_c:
                best_c = c5
                best_h = h5

    # Phase 6: Subgradient refinement
    if time.time() < deadline - 80:
        p6_end = min(time.time() + 60, deadline - 60)
        h6, c6 = subgradient_hedge(best_h, best_n, p6_end, n_iter=10000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Pair refinement (fine local search)
    if time.time() < deadline - 40:
        p7_end = min(time.time() + 35, deadline - 30)
        h7, c7 = pair_refine(best_h, best_n, p7_end, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final SLSQP
    if time.time() < deadline - 5:
        h8, c8 = run_slqp(best_h, best_n, deadline - 2, alpha=10000000, maxiter=500)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
