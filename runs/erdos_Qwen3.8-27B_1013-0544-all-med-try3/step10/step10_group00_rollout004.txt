        if time.time() > deadline:
            break
        
        if it % 50 == 0:
            c_vec = compute_c5_vec(h, n)
            alpha = 100000.0
            w = smooth_max_w(c_vec, alpha)
            grad = h_grad_from_w(w, h, n)
        
        found = False
        
        # Single swap: guided by gradient
        if ones_list and zeros_list:
            idx_ones = np.array(ones_list)
            idx_zeros = np.array(zeros_list)
            g_ones = grad[idx_ones]
            g_zeros = grad[idx_zeros]
            
            # Best candidates to remove (high positive grad)
            top_rm = np.argsort(-g_ones)[0:min(25, len(idx_ones))]
            # Best candidates to add (low/negative grad)
            top_ad = np.argsort(g_zeros)[0:min(25, len(idx_zeros))]
            
            for ti in top_rm:
                if found:
                    break
                for tj in top_ad:
                    i = idx_ones[ti]
                    j = idx_zeros[tj]
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        oi = ones_list.index(i)
                        oj = zeros_list.index(j)
                        ones_list[oi] = j
                        zeros_list[oj] = i
                        found = True
                        stall_count = 0
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
        
        # Double swap every 30 iterations if stalled
        if not found and it % 30 == 15 and len(ones_list) >= 2 and len(zeros_list) >= 2:
            idx_ones = np.array(ones_list)
            idx_zeros = np.array(zeros_list)
            g_ones = grad[idx_ones]
            g_zeros = grad[idx_zeros]
            
            top_rm = np.argsort(-g_ones)[0:min(15, len(idx_ones))]
            top_ad = np.argsort(g_zeros)[0:min(15, len(idx_zeros))]
            
            for ti1 in top_rm[:8]:
                if found:
                    break
                for ti2 in top_rm[:8]:
                    if ti2 <= ti1:
                        continue
                    if found:
                        break
                    for tj1 in top_ad[:8]:
                        if found:
                            break
                        for tj2 in top_ad[:8]:
                            if tj2 <= tj1:
                                continue
                            i1, i2 = idx_ones[ti1], idx_ones[ti2]
                            j1, j2 = idx_zeros[tj1], idx_zeros[tj2]
                            if i1 in (j1, j2) or i2 in (j1, j2):
                                continue
                            h[i1] = 0.0
                            h[i2] = 0.0
                            h[j1] = 1.0
                            h[j2] = 1.0
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                for x in (i1, i2):
                                    ones_list.remove(x)
                                    zeros_list.append(x)
                                for x in (j1, j2):
                                    zeros_list.remove(x)
                                    ones_list.append(x)
                                found = True
                                stall_count = 0
                                break
                            else:
                                h[i1] = 1.0
                                h[i2] = 1.0
                                h[j1] = 0.0
                                h[j2] = 0.0
        
        # Random swaps to escape local optima
        if not found:
            stall_count += 1
            if stall_count % 100 == 50:
                num_rand = rng.randint(1, 4)
                for _ in range(num_rand):
                    if time.time() > deadline:
                        break
                    if not ones_list or not zeros_list:
                        break
                    i = ones_list[rng.randint(len(ones_list))]
                    j = zeros_list[rng.randint(len(zeros_list))]
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        oi = ones_list.index(i)
                        oj = zeros_list.index(j)
                        ones_list[oi] = j
                        zeros_list[oj] = i
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                        stall_count += 1
                    if stall_count > 500:
                        break
    
    return h, best_c

def multistart_binary_search(h_start, n, deadline, rng, n_starts=5):
    best_h = h_start.copy()
    best_c = compute_c5(h_start, n)
    per_start = max(10.0, (deadline - time.time()) / n_starts)
    
    for si in range(n_starts):
        if time.time() > deadline - 10:
            break
        h_tmp = h_start.copy()
        ones_tmp = np.where(h_tmp > 0.5)[0].tolist()
        zeros_tmp = np.where(h_tmp < 0.5)[0].tolist()
        
        # Perturb
        num_swaps = rng.randint(3, 12)
        for _ in range(num_swaps):
            if not ones_tmp or not zeros_tmp:
                break
            i = ones_tmp[rng.randint(len(ones_tmp))]
            j = zeros_tmp[rng.randint(len(zeros_tmp))]
            h_tmp[i] = 0.0
            h_tmp[j] = 1.0
            oi = ones_tmp.index(i)
            oj = zeros_tmp.index(j)
            ones_tmp[oi] = j
            zeros_tmp[oj] = i
        
        sub_dl = min(deadline - 30, time.time() + per_start)
        h_new, c_new = binary_local_search_enhanced(h_tmp, n, sub_dl, rng, max_iters=200000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def continuous_refine(h, n, deadline):
    """Refine using L-BFGS with smooth max"""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    
    best_h = h.copy()
    
    for alpha in [50000, 200000, 1000000, 5000000]:
        if time.time() > deadline - 5:
            break
        
        def obj_func(x):
            xx = np.clip(x, 0.0, 1.0)
            c = compute_c5_vec(xx, n)
            val = smooth_max(c, alpha)
            val += 0.01 * (xx.sum() - n / 2.0) ** 2
            return val
        
        def grad_func(x):
            xx = np.clip(x, 0.0, 1.0)
            c = compute_c5_vec(xx, n)
            w = smooth_max_w(c, alpha)
            g = h_grad_from_w(w, xx, n)
            g += 0.02 * (xx.sum() - n / 2.0)
            return g
        
        bounds = [(0.0, 1.0)] * n
        remaining = deadline - time.time()
        maxit = max(200, int(remaining * 30))
        
        try:
            res = minimize(obj_func, best_h, jac=grad_func,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': min(maxit, 3000), 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': min(maxit * 5, 15000),
                                   'maxls': 40, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
        
        if time.time() > deadline - 5:
            break
    
    return best_h, best_c

def fine_pair_search(h, n, deadline):
    """Fine-tuned pair swap search with small continuous adjustments"""
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in steps:
        if time.time() > deadline - 1:
            break
        improved = True
        while improved and time.time() < deadline - 1:
            improved = False
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            w = smooth_max_w(c_vec, 300000.0)
            grad = h_grad_from_w(w, best_h, n)
            
            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            
            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0][:50]
            neg_idx = np.where(neg_mask)[0][:50]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx:
                for j in neg_idx:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0 or best_gain < 1e-14:
                break
            
            amt = min(step, best_h[best_i] - 1e-10, 1.0 - best_h[best_j] - 1e-10)
            if amt < 1e-10:
                break
            
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    
    return best_h, best_c

def try_resolution(h_best, n_best, c_best, n_try, deadline):
    """Try to optimize at a different resolution"""
    if time.time() > deadline - 30:
        return h_best, c_best, n_best
    
    # Resample from best_h to new resolution
    x_old = np.linspace(0, 2, n_best, endpoint=False) + 1.0 / n_best
    x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
    
    h_new = np.interp(x_new, x_old, h_best, left=0.0, right=0.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    
    # Binarize
    if n_try % 2 == 0:
        h_bin = binarize(h_new, n_try)
        c_bin = compute_c5(h_bin, n_try)
        
        # Quick local search
        sub_dl = min(deadline - 20, time.time() + 60)
        h_opt, c_opt = binary_local_search_enhanced(h_bin, n_try, sub_dl,
                                                    np.random.RandomState(123), max_iters=50000)
        if c_opt < c_best:
            return h_opt, c_opt, n_try
        else:
            # Try continuous refinement
            sub_dl2 = min(deadline - 10, time.time() + 30)
            h_cont, c_cont = continuous_refine(h_bin, n_try, sub_dl2)
            if c_cont < c_best:
                return h_cont, c_cont, n_try
    else:
        # Odd resolution: use continuous
        h_new = project_feasible(h_new, n_try)
        sub_dl = min(deadline - 20, time.time() + 40)
        h_opt, c_opt = continuous_refine(h_new, n_try, sub_dl)
        if c_opt < c_best:
            return h_opt, c_opt, n_try
    
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent
    
    # Phase 1: Binarize (10s)
    t1_end = t0 + 10
    h_bin = binarize(best_h, best_n)
    c_bin = compute_c5(h_bin, best_n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    # Phase 2: Intensive binary local search (350s)
    t2_end = min(t0 + 360, deadline - 400)
    h_phase2, c_phase2 = binary_local_search_enhanced(best_h, best_n, t2_end, rng, max_iters=2000000)
    if c_phase2 < best_c:
        best_c = c_phase2
        best_h = h_phase2.copy()
    
    # Phase 3: Multi-start binary search (200s)
    t3_end = min(t0 + 560, deadline - 250)
    if time.time() < t3_end:
        h_ms, c_ms = multistart_binary_search(best_h, best_n, t3_end, rng, n_starts=6)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms.copy()
    
    # Phase 4: Try other resolutions (150s)
    t4_end = min(t0 + 710, deadline - 150)
    alt_rs = [400, 600, 800]
    for r in alt_rs:
        if r == best_n or time.time() > t4_end - 30:
            continue
        best_h, best_c, best_n = try_resolution(best_h, best_n, best_c, r, t4_end)
    
    # Phase 5: Continuous refinement (150s)
    t5_end = min(t0 + 860, deadline - 100)
    if time.time() < t5_end:
        h_cont, c_cont = continuous_refine(best_h, best_n, t5_end)
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()
    
    # Phase 6: Fine pair search (100s)
    t6_end = min(t0 + 960, deadline - 50)
    if time.time() < t6_end:
        h_fine, c_fine = fine_pair_search(best_h, best_n, t6_end)
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine.copy()
    
    # Phase 7: Final binary polish
    if time.time() < deadline - 10:
        h_final_bin = binarize(best_h, best_n)
        c_final_bin = compute_c5(h_final_bin, best_n)
        if c_final_bin < best_c:
            best_c = c_final_bin
            h_sub_dl = deadline - 5
            h_pol, c_pol = binary_local_search_enhanced(h_final_bin, best_n, h_sub_dl, rng, max_iters=100000)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol.copy()
    
    # Ensure constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
