import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_smooth(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max_val(c, 0)  # placeholder, need alpha
        return val
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(200, int(remaining * 3))
        
        def obj_alpha(x, a=alpha):
            c = compute_c5_vec(x, n)
            val = smooth_max_val(c, a)
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return val

            g = smooth_max_grad(c, a, x, n)
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
        
        try:
            res = minimize(obj_alpha, best_h, jac=grad_alpha,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,

                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

        g += 2.0 * mu * (x.sum() - n / 2.0)

def _topk_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _topk_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def fourier_init(n, n_modes, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, n_modes + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def beatty_init(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def random_binary_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def upsample(h, n_from, n_to):
    xs_from = np.linspace(0, 2.0, n_from, endpoint=False)
    xs_to = np.linspace(0, 2.0, n_to, endpoint=False)
    h_up = np.interp(xs_to, xs_from, h, period=2.0)
    return project_feasible(h_up, n_to)

def downsample(h, n_from, n_to):
    xs_from = np.linspace(0, 2.0, n_from, endpoint=False)
    xs_to = np.linspace(0, 2.0, n_to, endpoint=False)
    h_dn = np.interp(xs_to, xs_from, h, period=2.0)
    return project_feasible(h_dn, n_to)

def multi_res_optimize(h, n, deadline, n_coarse=100):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    h_coarse = downsample(best_h, n, n_coarse)
    c_coarse = compute_c5(h_coarse, n_coarse)
    bounds_coarse = [(0.0, 1.0)] * n_coarse
    alphas_coarse = [10, 50, 200, 1000, 5000, 20000]
    for alpha in alphas_coarse:
        if time.time() > deadline - 30:
            break
        maxit = max(200, int((deadline - time.time()) * 2))
        try:
            res = minimize(_sm_obj, h_coarse, jac=_sm_grad,
                          args=(n_coarse, alpha, 0.01), method='L-BFGS-B',
                          bounds=bounds_coarse,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxls': 40, 'maxcor': 40})
            h_coarse = project_feasible(res.x, n_coarse)
            c_coarse = compute_c5(h_coarse, n_coarse)
        except Exception:
            pass
    h_up = upsample(h_coarse, n_coarse, n)
    c_up = compute_c5(h_up, n)
    if c_up < best_c:
        best_c = c_up
        best_h = h_up
    bounds_full = [(0.0, 1.0)] * n
    alphas_full = [100, 1000, 10000, 100000, 1000000]
    for alpha in alphas_full:
        if time.time() > deadline - 20:
            break
        maxit = max(300, int((deadline - time.time()) * 2.5))
        try:
            res = minimize(_sm_obj, best_h, jac=_sm_grad,
                          args=(n, alpha, 0.005), method='L-BFGS-B',
                          bounds=bounds_full,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, configs=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [
            (4, 50000, 4),
            (6, 200000, 4),
            (8, 1000000, 3),
            (12, 5000000, 3),
            (16, 20000000, 2),
            (20, 50000000, 2),
            (30, 100000000, 2),
        ]
    for top_k, alpha, rounds in configs:
        if time.time() > deadline - 20:
            break
        for rnd in range(rounds):
            if time.time() > deadline - 15:
                break
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            threshold = max_val - max(1e-9, 0.0003 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > top_k:
                sorted_active = np.argsort(-c_vec[active])
                active = active[sorted_active[:top_k]]
            if len(active) == 0:
                break
            maxit = max(200, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(_topk_obj, best_h, jac=_topk_grad,
                              args=(n, active, alpha, 0.0005), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def balancing_search(h, n, deadline, max_iters=50000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx_val = 2.0 / n
    iters = 0
    while time.time() < deadline and iters < max_iters:
        iters += 1
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-10, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        if len(active) == 0:
            break
        w = c_vec[active] - (max_val - tol)
        w = w / (w.sum() + 1e-30)
        grad = np.zeros(n)
        for wi, idx in zip(w, active):
            k = idx - (n - 1)
            wm = wi * dx_val
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - best_h[k:n])
                grad[k:n] += wm * (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - best_h[0:nv])
                grad[0:nv] += wm * (-best_h[kk:n])
        pos_mask = (grad > 1e-12) & (best_h > 1e-12)
        neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-12)
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        best_gain = 0
        best_i = -1
        best_j = -1
        pos_sample = pos_idx[:30]
        neg_sample = neg_idx[:30]
        for i in pos_sample:
            gi = grad[i]
            if gi <= 0:
                continue
            for j in neg_sample:
                g = gi - grad[j]
                if g > best_gain:
                    best_gain = g
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        amt = min(0.005, best_h[best_i], 1.0 - best_h[best_j])
        if amt < 1e-12:
            break
        best_h[best_i] -= amt
        best_h[best_j] += amt
        best_h = project_feasible(best_h, n)
        c_new = compute_c5(best_h, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
        else:
            best_h[best_i] += amt
            best_h[best_j] -= amt
            best_h = project_feasible(best_h, n)
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=40, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000):
    def sq_fun(x, nn):
        return x.sum() - nn / 2.0
    def sq_jac(x, nn):
        return np.ones(nn)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    try:
        rem = max(10, deadline - time.time())
        maxit = max(200, int(rem * 8))
        res = minimize(_sm_obj, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500

    n = n_orig
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        ('parent', h0.copy()),
        ('beatty_g', beatty_init(n, golden / 2.0)),
        ('beatty_s2', beatty_init(n, np.sqrt(2.0) / 2.0)),
        ('beatty_pi', beatty_init(n, np.pi / 6.0)),
        ('rand1', random_binary_init(n, rng)),
        ('rand2', random_binary_init(n, rng)),
        ('fourier1', fourier_init(n, 8, rng)),
        ('fourier2', fourier_init(n, 12, rng)),
    ]

    for name, hs in starts:
        c_s = compute_c5(hs, n)
        if c_s < best_c:
            best_c = c_s
            best_h = hs.copy()

    # Phase 1: Multi-resolution optimization from best start
    p1_deadline = t0 + 180
    if time.time() < p1_deadline:
        h1, c1 = multi_res_optimize(best_h, n, p1_deadline, n_coarse=80)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: Active set optimization
    p2_deadline = t0 + 380
    if time.time() < p2_deadline:
        h2, c2 = active_set_optimize(best_h, n, p2_deadline)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Try multi-res from a different start
    p3_deadline = t0 + 520
    if time.time() < p3_deadline:
        alt_start = beatty_init(n, golden / 2.0 + 0.01)
        h3, c3 = multi_res_optimize(alt_start, n, p3_deadline, n_coarse=120)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        h3b, c3b = active_set_optimize(best_h, n, min(time.time() + 60, p3_deadline))
        if c3b < best_c:
            best_c = c3b
            best_h = h3b

    # Phase 4: Balancing search
    p4_deadline = t0 + 700
    if time.time() < p4_deadline:
        h4, c4 = balancing_search(best_h, n, p4_deadline, max_iters=80000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Active set refinement
    p5_deadline = t0 + 850
    if time.time() < p5_deadline:
        configs5 = [
            (6, 100000, 5),
            (10, 1000000, 4),
            (16, 10000000, 3),
            (24, 100000000, 2),
        ]
        h5, c5 = active_set_optimize(best_h, n, p5_deadline, configs=configs5)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Fine pair search
    p6_deadline = t0 + 980
    if time.time() < p6_deadline:
        h6, c6 = fine_pair_search(best_h, n, p6_deadline, radius=30,
            deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: SLSQP polish
    p7_deadline = deadline - 15
    if time.time() < p7_deadline:
        for alpha in [500000, 5000000, 50000000]:
            if time.time() > p7_deadline - 10:
                break
            h7, c7 = slqp_polish(best_h, n, deadline - 5, alpha=alpha)
            if c7 < best_c:
                best_c = c7
                best_h = h7

    # Phase 8: Final fine adjustments
    if time.time() < deadline - 3:
        h8, c8 = fine_pair_search(best_h, n, deadline - 2, radius=20,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final projection
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
