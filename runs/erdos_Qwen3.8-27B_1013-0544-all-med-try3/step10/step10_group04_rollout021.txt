```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def fast_smooth_grad(x, n, alpha, mu):
    """Compute gradient of smooth objective using FFT-based operations."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx  # length 2n-1
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    
    # W[m] = w[m] for m in 0..2n-2, corresponding to shift k = m-(n-1)
    W = w  # length 2n-1
    
    # g_j = prefix_W[j+1] - (W*x)[j] - (W*x_rev)[n-1-j]
    x_rev = x[::-1]
    
    conv1 = fftconvolve(W, x, mode='full')  # length 3n-2
    conv2 = fftconvolve(W, x_rev, mode='full')  # length 3n-2
    
    prefix_W = np.concatenate(([0.0], np.cumsum(W)))
    
    # g_j = prefix_W[j+1] - conv1[j] - conv2[n-1-j] for j in 0..n-1
    grad = prefix_W[1:n+1] - conv1[0:n] - conv2[n-1::-1]
    
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(50, 10.0), (200, 50.0), (500, 200.0), (1000, 500.0), (3000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_objective, h, jac=fast_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 200000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c
```
