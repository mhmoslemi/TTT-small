import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (15000, 8000.0), (50000, 30000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_passes=200):
    """Coordinate descent with exact C5 evaluation."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Adaptive delta sizes
    base_deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        
        h_cur = best_h.copy()
        pass_improved = False
        scale = max(0.1, 1.0 - 0.5 * p / max_passes)  # Reduce step size over passes
        
        deltas = [d * scale for d in base_deltas] + [-d * scale for d in base_deltas[::-1]]
        
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            
            hi = h_cur[i]
            if hi < 1e-8 or hi > 1 - 1e-8:
                # Already at boundary, only try moving inward
                if hi < 1e-8:
                    deltas_i = [d for d in deltas if d > 0]
                else:
                    deltas_i = [d for d in deltas if d < 0]
            else:
                deltas_i = deltas
            
            best_local_c = best_c
            best_local_val = hi
            
            for d in deltas_i:
                nv = hi + d
                if nv < -1e-10 or nv > 1 + 1e-10:
                    continue
                h_cur[i] = nv
                c_new = compute_c5(h_cur, n)
                if c_new < best_local_c - 1e-16:
                    best_local_c = c_new
                    best_local_val = nv
                    pass_improved = True
            
            h_cur[i] = best_local_val
        
        # Project to feasibility
        h_cur = project_feasible(h_cur, n)
        c_final = compute_c5(h_cur, n)
        if c_final < best_c:
            best_c = c_final
            best_h = h_cur.copy()
        
        if not pass_improved:
            break
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):

        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b
    if diff > 0:
        order = np.argsort(-h)
        for i in order[:int(np.ceil(abs(diff)))]:
            if h_b[i] < 1.0:
                h_b[i] = 1.0
    else:
        order = np.argsort(h)
        for i in order[:int(np.ceil(abs(diff)))]:
            if h_b[i] > 0.0:
                h_b[i] = 0.0
    return project_feasible(h_b, n)

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w_full = np.exp(sh - mx)
    w_full /= w_full.sum()
    # Keep only top K
    top_idx = np.argsort(-w_full)[:K]
    w = np.zeros_like(w_full)
    w[top_idx] = w_full[top_idx]
    w /= w.sum()
    
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:

            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu, K):
    c = compute_c5_vec(x, n)
    m = c.max()
    if K < len(c):
        top_vals = np.sort(c)[-K:]
        sh = alpha * (top_vals - top_vals.max())
        mx = sh.max()
        val = top_vals.max() + np.log(np.sum(np.exp(sh - mx)) + mx) / alpha
    else:
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu, K):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w_full = np.exp(sh - mx)
    w_full /= w_full.sum()
    if K < len(c):
        top_idx = np.argsort(-w_full)[:K]
        w = np.zeros_like(w_full)
        w[top_idx] = w_full[top_idx]
        w /= w.sum()
        active = top_idx
    else:
        active = np.where(w_full > 1e-12)[0]
        w = w_full
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for idx in active:
        if w[idx] < 1e-14:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None, K=50):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 20.0), (500, 100.0), (2000, 500.0),
                   (5000, 2000.0), (15000, 8000.0), (50000, 30000.0),
                   (100000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 8:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu, K), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-19,
                                   'gtol': 1e-15, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def coord_descent(h, n, deadline, max_passes=200):
    """Coordinate descent with exact C5 evaluation and adaptive steps."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        
        h_cur = best_h.copy()
        pass_improved = False
        
        # Adaptive scale: start larger, shrink over passes
        scale = max(0.02, 1.0 * (0.95 ** p))
        base_deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        deltas = [d * scale for d in base_deltas]
        all_deltas = deltas + [-d for d in reversed(deltas)]
        
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            
            hi = h_cur[i]
            
            # Determine valid deltas based on boundaries
            if hi <= 1e-9:
                deltas_i = [d for d in all_deltas if d > 0]
            elif hi >= 1 - 1e-9:
                deltas_i = [d for d in all_deltas if d < 0]
            else:
                deltas_i = all_deltas
            
            if not deltas_i:
                continue
            
            best_local_c = best_c
            best_local_val = hi
            
            for d in deltas_i:
                nv = hi + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h_cur[i] = nv
                c_new = compute_c5(h_cur, n)
                if c_new < best_local_c - 1e-16:
                    best_local_c = c_new
                    best_local_val = nv
                    pass_improved = True
            
            h_cur[i] = best_local_val
        
        # Project to feasibility after full pass
        h_cur = project_feasible(h_cur, n)
        c_final = compute_c5(h_cur, n)
        
        if c_final < best_c:
            best_c = c_final
            best_h = h_cur.copy()
        elif not pass_improved:
            break
    
    return best_h, best_c

def targeted_coord_descent(h, n, deadline, max_rounds=50):
    """Coordinate descent focusing on coordinates involved in worst shifts."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    center = n - 1
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        # Get top 5 worst shifts
        top_worst = np.argsort(-c_vec)[:5]
        
        # Determine which coordinates are involved in these shifts
        affected = set()
        for idx in top_worst:
            k = idx - center
            if k >= 0:
                affected.update(range(0, n - k))
                affected.update(range(k, n))
            else:
                kk = -k
                affected.update(range(0, n - kk))
                affected.update(range(kk, n))
        
        affected_list = sorted(affected)
        if not affected_list:
            break
        
        h_cur = best_h.copy()
        improved = False
        
        deltas = [0.003, 0.001, 0.0005, -0.0005, -0.001, -0.003,
                  0.0002, -0.0002]
        
        for i in affected_list:
            if time.time() > deadline:
                break
            
            hi = h_cur[i]
            for d in deltas:
                nv = hi + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                h_cur[i] = nv
                c_new = compute_c5(h_cur, n)
                if c_new < compute_c5(best_h, n) - 1e-16:
                    hi = nv
                    improved = True
                else:
                    h_cur[i] = hi
            h_cur[i] = hi
        
        h_cur = project_feasible(h_cur, n)
        c_new = compute_c5(h_cur, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_cur.copy()
        elif not improved:
            break
    
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=3000000, maxiter=3000):
    def obj_fn(x, na, al):
        c = compute_c5_vec(x, na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)
    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)
    def sq_jac_fn(x, na):
        return np.ones(na)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-19, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def binary_push(h, n):
    h_b = np.where(h > 0.5, 1.0, 0.0).astype(float)
    target = n / 2.0
    diff = target - h_b.sum()
    if abs(diff) < 1e-12:
        return h_b.copy()
    if diff > 0:
        order = np.argsort(-h)
        cnt = 0
        for i in order:
            if h_b[i] < 1.0 and cnt < int(np.ceil(abs(diff))):
                h_b[i] = 1.0
                cnt += 1
    else:
        order = np.argsort(h)
        cnt = 0
        for i in order:
            if h_b[i] > 0.0 and cnt < int(np.ceil(abs(diff))):
                h_b[i] = 0.0
                cnt += 1
    return project_feasible(h_b, n)

def perturbation_search(h, n, deadline, n_trials=5, mags=None, K=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    if mags is None:
        mags = [0.03, 0.06, 0.02, 0.01, 0.05]
    for t in range(n_trials):
        if time.time() > deadline - 15:
            break
        mag = mags[t % len(mags)]
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0, K), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 1500, 'ftol': 1e-18, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c
