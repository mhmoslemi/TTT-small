```python
    k_max = int(np.argmax(c))
    # Compute gradient of g_{k_max}
    k = k_max - (n-1)
    dx = 2.0/n

    if k >= 0:
        for i in range(n-k):
            # d/dh[i] of h[i]*(1-h[i+k]) = 1 - h[i+k]
            # d/dh[i+k] of h[i]*(1-h[i+k]) = -h[i]
            grad[i] += dx * (1 - h[i+k])

            grad[i+kk] += dx * (1 - h[i])
            grad[i] += dx * (-h[i+kk])
    h -= step * grad
    h = project_feasible(h, n)
```

- The total 1s in the overlap region for lag k is not exactly n/2 (it depends on the boundary)
- The (1,1) pairs at lag k is the "autocorrelation"

To minimize the max overlap, we want the autocorrelation to be as uniform as possible across all lags, with the maximum being as small as possible.

This is related to "Golay sequences" and "perfect sequences" in signal processing!

A perfect binary sequence of length N has the property that its periodic autocorrelation is 0 for all non-zero lags. If such a sequence exists with exactly N/2 ones, it would give a vanishingly small C₅.

But perfect sequences only exist for very specific lengths. For our problem, we have a non-periodic (finite) correlation, so the boundaries matter.

```python
def smooth_obj(x, n, alpha):
    c = compute_c5_vec_fast(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += 1000.0 * (x.sum() - n/2.0)**2
    return float(val)

def smooth_grad(x, n, alpha):
    c = compute_c5_vec_fast(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    
    # For high alpha, only top few w matter
    threshold = 1e-12
    active = np.where(w > threshold)[0]
    
    for idx in active:
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    
    grad += 2000.0 * (x.sum() - n/2.0)
    return grad
```

```python
def pair_swap_descent(h, n, deadline, n_rounds=20):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(777)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        
        improved = False
        n_trials = 2000
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            
            i, j = rng.randint(n, size=2)
            if i == j or abs(best_h[i] - best_h[j]) < 1e-10:
                continue
            
            # Swap
            tmp = best_h[i]
            best_h[i] = best_h[j]
            best_h[j] = tmp
            
            c_new = compute_c5_vec_fast(best_h, n).max()
            if c_new < best_c - 1e-10:
                best_c = float(c_new)
                improved = True
            else:
                # Revert
                best_h[j] = best_h[i]
                best_h[i] = tmp
        
        if not improved:
            break
    
    return best_h, best_c
```
