```python
def fft_correlate(a, b):
    n = len(a)
    m = len(b)
    size = 2 ** int(np.ceil(np.log2(n + m - 1)))
    fa = np.fft.fft(a, size)
    fb = np.fft.fft(b[::-1], size)
    return np.fft.ifft(fa * fb).real[:n+m-1]
```

```python
def smooth_max_grad(h, n, w):
    """Compute gradient of smooth_max(C) wrt h, where C = correlate(h, 1-h, 'full')*dx."""
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k in range(2*n - 1):
        if w[k] < 1e-14:
            continue
        if k < n:
            # C[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # dC/dh[j] = (1-h[j-k]) for k<=j<=n-1, minus h[j+k] for 0<=j<=n-1-k
            if k < n:
                # First part: j ranges from k to n-1
                grad[k:] += w[k] * (1 - h[:n-k])
                # Second part: j ranges from 0 to n-1-k
                grad[:n-k] -= w[k] * h[k:]
        else:
            kp = k - n + 1  # 1 to n-1
            # C[k] = sum_{i=kp}^{n-1} h[i]*(1-h[i-kp])
            # dC/dh[j] = (1-h[j-kp]) for kp<=j<=n-1, minus h[j+kp] for 0<=j<=n-1-kp
            grad[kp:] += w[k] * (1 - h[:n-kp])
            grad[:n-kp] -= w[k] * h[kp:]
    
    return grad * dx
```

This is O(n²) but each inner operation is vectorized, so it should be fast.

import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(h, omh):
    n = len(h)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fh = np.fft.rfft(h, size)
    fo = np.fft.rfft(omh[::-1], size)
    return np.fft.irfft(fh * fo, size)[:2*n-1]

def compute_all_lags(h, n):
    dx = 2.0 / n
    c = fft_correlate(h, 1.0 - h)
    return c * dx

def compute_c5(h, n):
    return float(np.max(compute_all_lags(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > -700:
        return float(m + np.log(np.sum(np.exp(s - mx))) / alpha)
    else:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    return w / w.sum()

def grad_smooth_max(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        if k < n:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            kp = k - n + 1
            grad[kp:] += wk * (1.0 - h[:n-kp])
            grad[:n-kp] -= wk * h[kp:]
    return grad * dx

def lbfgsb(h0, n, alpha, deadline, maxiter=5000, pen=1e6):
    target = n / 2.0
    def f_and_g(x):
        x = np.clip(x, 0, 1)
        c = compute_all_lags(x, n)
        val = smooth_max(c, alpha)
        d = x.sum() - target
        val += pen * d * d
        w = smooth_max_weights(c, alpha)
        g = grad_smooth_max(x, n, w) + 2.0 * pen * d
        return val, g
    try:
        res = minimize(f_and_g, h0, method='L-BFGS-B', jac=True,
                      bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return project_feasible(h0.copy(), n)

    y = np.linspace(0, 2, n_dst)
    h_new = np.interp(y, x, h)
    return project_feasible(h_new, n_dst)
