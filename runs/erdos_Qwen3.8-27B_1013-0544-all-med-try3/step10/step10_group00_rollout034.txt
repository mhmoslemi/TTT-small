```python
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def c5_grad(h, n, w):
    """Gradient of weighted sum of correlations w.r.t. h"""
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = c5_grad(x, n, w)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-30,
                               'gtol': 1e-20, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 100})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def subgradient_descent(h, n, deadline, num_iters=5000):
    """Subgradient descent focusing on the argmax correlation"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    
    for it in range(num_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        k_max = c.argmax()
        w = np.zeros(len(c))
        w[k_max] = 1.0
        grad = c5_grad(best_h, n, w)
        
        # Adaptive step size
        step = 0.005 / (1 + it * 0.001)
        
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def multi_pair_descent(h, n, deadline, max_rounds=500):
    """Coordinate descent using multi-pair moves"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        if time.time() > deadline:
            break
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(best_h, n)
            alpha = 50000.0
            w = smooth_max_w(c_vec, alpha)
            grad = c5_grad(best_h, n, w)
            
            # Find positions to decrease (positive gradient) and increase (negative gradient)
            pos = np.where((grad > 1e-10) & (best_h > 1e-6))[0]
            neg = np.where((grad < -1e-10) & (best_h < 1.0 - 1e-6))[0]
            
            if len(pos) == 0 or len(neg) == 0:
                break
            
            # Sort by gradient magnitude
            pos = pos[np.argsort(-grad[pos])]
            neg = neg[np.argsort(grad[neg])]
            
            # Move from top pos to top neg
            improved = False
            num_pairs = min(3, len(pos), len(neg))
            for pp in range(num_pairs):
                i = pos[pp]
                j = neg[pp]
                amt = min(step, best_h[i], 1.0 - best_h[j])
                if amt < 1e-10:
                    continue
                best_h[i] -= amt
                best_h[j] += amt
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    best_h[i] += amt
                    best_h[j] -= amt
                    best_h = project_feasible(best_h, n)
            
            if not improved:
                break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, num_iters=100000):
    """SA with continuous perturbations"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    curr_c = best_c
    
    T = 0.01  # initial temperature
    T_min = 1e-8
    
    for it in range(num_iters):
        if time.time() > deadline:
            break
        
        # Cool
        T = max(T_min, T * 0.9995)
        
        # Generate perturbation: pick 1-3 positions and perturb
        num_pert = rng.randint(1, 4)
        idx = rng.choice(n, size=num_pert, replace=False)
        pert = rng.uniform(-0.05, 0.05, size=num_pert)
        
        h_new = curr_h.copy()
        h_new[idx] += pert
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        delta = c_new - curr_c
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            curr_h = h_new
            curr_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    
    return best_h, best_c
```

c_k(h) = ∑ h[i+k'](1-h[i]) = ∑ h[i+k'] - ∑ h[i+k']h[i]

Linearized at h_old:
c_k_lin = ∑ h[i+k'] - ∑ (h[i+k']_old * h[i] + h[i]_old * h[i+k'] - h[i+k']_old * h[i]_old)
= ∑ h[i+k'] - ∑ h[i+k']_old * h[i] - ∑ h[i]_old * h[i+k'] + ∑ h[i+k']_old * h[i]_old

The last term is a constant. The other terms are linear in h.

∑ h[i+k'] - ∑ h[i+k']_old * h[i] - ∑ h[i]_old * h[i+k'] + const ≤ t

This is linear! And the LP is:
minimize t

            0 ≤ h[i] ≤ 1 for all i

```python
from scipy.optimize import linprog

def lp_step(h_old, n, top_k=50, c_max=None):
    """One step of the successive LP method"""
    if c_max is None:
        c_vec = compute_c5_vec(h_old, n)
        c_max = c_vec.max()
    else:
        c_vec = compute_c5_vec(h_old, n)
    
    # Get top-k active constraints
    top_indices = np.argsort(-c_vec)[:top_k]
    
    # Build A, b for LP: minimize t
    # Variables: [h(0),...,h(n-1), t]
    # Constraints: A @ x <= b
    # h >= 0: -h <= 0
    # h <= 1: h <= 1
    # sum(h) = n/2: -sum(h) <= -n/2 AND sum(h) <= n/2
    # c_k_lin <= t: coeff_k @ h - t <= -const_k → coeff_k @ h - t + const_k <= 0
    
    m = len(top_indices)
    # A has shape (m + 2n + 2, n+1)
    A = np.zeros((m + 2*n + 2, n + 1))
    b = np.zeros(m + 2*n + 2)
    
    row = 0
    # h >= 0 → -h <= 0
    A[row:row+n, 0:n] = -np.eye(n)
    b[row:row+n] = 0
    row += n
    
    # h <= 1
    A[row:row+n, 0:n] = np.eye(n)
    b[row:row+n] = 1.0
    row += n
    
    # sum(h) = n/2 → sum(h) <= n/2 and -sum(h) <= -n/2
    A[row, 0:n] = 1.0
    b[row] = n / 2.0
    row += 1
    A[row, 0:n] = -1.0
    b[row] = -n / 2.0
    row += 1
    
    # c_k_lin <= t for each active k
    for k in top_indices:
        lag = k - (n - 1)
        coeff = np.zeros(n)
        const = 0.0
        if lag >= 0:
            nv = n - lag
            if nv > 0:
                coeff[lag:n] += (1.0 - h_old[0:nv])
                coeff[0:nv] -= h_old[lag:n]
                const = h_old[lag:n] @ (1.0 - h_old[0:nv])
        else:
            kk = -lag
            if kk < n:
                coeff[kk:n] += (1.0 - h_old[0:n-kk])
                coeff[0:n-kk] -= h_old[kk:n]
                const = h_old[kk:n] @ (1.0 - h_old[0:n-kk])
        
        A[row, 0:n] = coeff
        A[row, n] = -1.0  # -t
        b[row] = -const
        row += 1
    
    # Objective: minimize t → c_vec = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0,1)]*n + [(0, None)], method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return h_new, res.x[n]
    return None, None
```

```python
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def c5_grad(h, n, w):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = c5_grad(x, n, w)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-30,
                               'gtol': 1e-20, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 100})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def linearize_correlation(h_old, n, k_idx):
    """Return (coeff, const) for linearized correlation at index k_idx"""
    lag = k_idx - (n - 1)
    coeff = np.zeros(n)
    if lag == 0:
        coeff[:] = 1.0 - 2.0 * h_old
        const = h_old @ h_old
    else:
        kk = abs(lag)
        if kk >= n:
            return coeff, 0.0
        nv = n - kk
        coeff[kk:] += (1.0 - h_old[:nv])
        coeff[:nv] -= h_old[kk:]
        const = h_old[kk:] @ (1.0 - h_old[:nv])
    return coeff, const

def lp_optimize_step(h_old, n, top_k=50, trust_delta=None):
    """One LP step: minimize t subject to linearized correlation constraints"""
    c_vec = compute_c5_vec(h_old, n)
    top_indices = np.argsort(-c_vec)[:top_k]
    
    m = len(top_indices)
    # Variables: h[0..n-1], t
    # LP: min t
    # s.t. coeff_k @ h + const_k <= t  for each active k
    #      0 <= h <= 1
    #      sum(h) = n/2
    #      |h - h_old|_inf <= trust_delta (if specified)
    
    num_constraints = m + 2 * n + 2
    if trust_delta is not None:
        num_constraints += 2 * n
    
    A = np.zeros((num_constraints, n + 1))
    b = np.zeros(num_constraints)
    
    row = 0
    # h >= 0
    A[row:row+n, 0:n] = -np.eye(n)
    b[row:row+n] = 0.0
    row += n
    
    # h <= 1
    A[row:row+n, 0:n] = np.eye(n)
    b[row:row+n] = 1.0
    row += n
    
    # sum(h) = n/2
    A[row, 0:n] = 1.0
    b[row] = n / 2.0
    row += 1
    A[row, 0:n] = -1.0
    b[row] = -n / 2.0
    row += 1
    
    # Trust region
    if trust_delta is not None:
        lo = np.maximum(0.0, h_old - trust_delta)
        hi = np.minimum(1.0, h_old + trust_delta)
        # h >= lo: -h <= -lo
        A[row:row+n, 0:n] = -np.eye(n)
        b[row:row+n] = -lo
        row += n
        # h <= hi: h <= hi
        A[row:row+n, 0:n] = np.eye(n)
        b[row:row+n] = hi
        row += n
    
    # Correlation constraints: coeff @ h - t <= -const
    for k_idx in top_indices:
        coeff, const = linearize_correlation(h_old, n, k_idx)
        A[row, 0:n] = coeff
        A[row, n] = -1.0
        b[row] = -const
        row += 1
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A[:row], b_ub=b[:row], bounds=bounds, method='highs')
        if res.success:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            return h_new
    except Exception:
        pass
    return None

def successive_lp(h_init, n, deadline, max_iters=50, top_k=50):
    """Successive LP optimization"""
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur, n)
    
    trust_delta = 0.5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        trust_delta = max(0.02, 0.5 * 0.93 ** it)
        
        h_new = lp_optimize_step(h_cur, n, top_k=top_k, trust_delta=trust_delta)
        if h_new is None:
            break
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_cur = h_new
        elif c_new < best_c + 1e-10:
            h_cur = h_new
        else:
            # Try with smaller trust region
            h_new2 = lp_optimize_step(h_cur, n, top_k=top_k, trust_delta=trust_delta * 0.5)
            if h_new2 is not None:
                c_new2 = compute_c5(h_new2, n)
                if c_new2 < best_c:
                    best_c = c_new2
                    best_h = h_new2.copy()
                    h_cur = h_new2
    
    return best_h, best_c

def multi_pair_descent(h, n, deadline, max_rounds=500, steps=None):
    """Coordinate descent with multi-pair moves"""
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in steps:
        if time.time() > deadline:
            break
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            c_vec = compute_c5_vec(best_h, n)
            alpha = 50000.0
            w = smooth_max_w(c_vec, alpha)
            grad = c5_grad(best_h, n, w)
            
            pos = np.where((grad > 1e-10) & (best_h > 1e-6))[0]
            neg = np.where((grad < -1e-10) & (best_h < 1.0 - 1e-6))[0]
            
            if len(pos) == 0 or len(neg) == 0:
                break
            
            pos = pos[np.argsort(-grad[pos])]
            neg = neg[np.argsort(grad[neg])]
            
            improved = False
            num_pairs = min(5, len(pos), len(neg))
            for pp in range(num_pairs):
                i = pos[pp]
                j = neg[pp]
                amt = min(step, best_h[i], 1.0 - best_h[j])
                if amt < 1e-10:
                    continue
                best_h[i] -= amt
                best_h[j] += amt
                best_h = project_feasible(best_h, n)
                c_new = compute_c5(best_h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                else:
                    best_h[i] += amt
                    best_h[j] -= amt
                    best_h = project_feasible(best_h, n)
            if not improved:
                break
    
    return best_h, best_c

def top_gradient_descent(h, n, deadline, num_iters=5000):
    """Focus on top few correlations with subgradient-like steps"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(num_iters):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        # Use top-5 correlations
        top5 = np.argsort(-c_vec)[:5]
        w = np.zeros(len(c_vec))
        for idx in top5:
            w[idx] = np.exp(50000 * (c_vec[idx] - c_vec.max()))
        w /= w.sum()
        
        grad = c5_grad(best_h, n, w)
        step = 0.002 / (1 + it * 0.0005)
        
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def alternating_refinement(h, n, deadline):
    """Alternate between LP steps and L-BFGS"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        # LP step
        t_lp = min(deadline - 30, time.time() + 60)
        h_lp, c_lp = successive_lp(best_h, n, t_lp, max_iters=5, top_k=30)
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp
        
        # L-BFGS step
        t_lbg = min(deadline - 10, time.time() + 30)
        alpha = max(50000, 50000 * (best_c / 0.38) ** 2)
        h_lb, c_lb = run_lbfgs(best_h, n, alpha=alpha, mu=0.005, maxit=2000)
        if c_lb < best_c:
            best_c = c_lb
            best_h = h_lb
        
        # Coord descent
        t_cd = min(deadline - 5, time.time() + 20)
        h_cd, c_cd = multi_pair_descent(best_h, n, t_cd, max_rounds=50,
                                        steps=[0.001, 0.0005, 0.0002, 0.0001])
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    h_parent = project_feasible(h_parent, n_parent)
    n = n_parent
    
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n)
    
    # Phase 1: Initial L-BFGS to get a good starting point (80s)
    t1 = min(t0 + 80, deadline - 200)
    h1, c1 = run_lbfgs(best_h, n, alpha=50000, mu=0.01, maxit=5000)
    if c1 < best_c:
        best_c, best_h = c1, h1
    h1b, c1b = run_lbfgs(best_h, n, alpha=200000, mu=0.005, maxit=3000)
    if c1b < best_c:
        best_c, best_h = c1b, h1b
    
    # Phase 2: LP-based optimization (200s)
    t2 = min(t0 + 280, deadline - 300)
    h2, c2 = successive_lp(best_h, n, t2, max_iters=40, top_k=50)
    if c2 < best_c:
        best_c, best_h = c2, h2
    
    # Phase 3: L-BFGS refinement after LP (100s)
    t3 = min(t0 + 380, deadline - 250)
    for alpha in [100000, 500000, 2000000]:
        if time.time() > t3 - 10:
            break
        h3, c3 = run_lbfgs(best_h, n, alpha=alpha, mu=0.003, maxit=3000)
        if c3 < best_c:
            best_c, best_h = c3, h3
    
    # Phase 4: Multi-pair coordinate descent (120s)
    t4 = min(t0 + 500, deadline - 200)
    h4, c4 = multi_pair_descent(best_h, n, t4, max_rounds=200,
                                steps=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    if c4 < best_c:
        best_c, best_h = c4, h4
    
    # Phase 5: Another round of LP (150s)
    t5 = min(t0 + 650, deadline - 180)
    h5, c5 = successive_lp(best_h, n, t5, max_iters=30, top_k=40)
    if c5 < best_c:
        best_c, best_h = c5, h5
    
    # Phase 6: Top-gradient descent (100s)
    t6 = min(t0 + 750, deadline - 130)
    h6, c6 = top_gradient_descent(best_h, n, t6, num_iters=3000)
    if c6 < best_c:
        best_c, best_h = c6, h6
    
    # Phase 7: Alternating refinement (200s)
    t7 = min(t0 + 950, deadline - 60)
    h7, c7 = alternating_refinement(best_h, n, t7)
    if c7 < best_c:
        best_c, best_h = c7, h7
    
    # Phase 8: Final coordinate descent (70s)
    t8 = min(deadline - 20, t0 + 1020)
    h8, c8 = multi_pair_descent(best_h, n, t8, max_rounds=300,
                                steps=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])
    if c8 < best_c:
        best_c, best_h = c8, h8
    
    # Phase 9: Final LP + L-BFGS polish
    if time.time() < deadline - 30:
        h9, c9 = successive_lp(best_h, n, deadline - 15, max_iters=10, top_k=30)
        if c9 < best_c:
            best_c, best_h = c9, h9
        if time.time() < deadline - 5:
            h9b, c9b = run_lbfgs(best_h, n, alpha=5000000, mu=0.001, maxit=2000)
            if c9b < best_c:
                best_c, best_h = c9b, h9b
    
    # Final: ensure exact constraint
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
```
