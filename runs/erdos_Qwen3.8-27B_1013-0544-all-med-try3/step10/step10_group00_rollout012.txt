import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())


def compute_corr_counts(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            cnt = int(np.ceil(abs(diff)))
            for idx in order[:cnt]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            cnt = int(np.ceil(abs(diff)))
            for idx in order[:cnt]:
                h_bin[ones[idx]] = 0.0
    return h_bin


def make_beatty(n, alpha_idx):

    target = n // 2

        pos = int(k * alpha * n) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    h = np.zeros(n)
    for p in positions:
        h[p] = 1.0
    if h.sum() != target:
        diff = target - h.sum()
        if diff > 0:
            zeros = np.where(h == 0)[0]

                if i < len(zeros):
                    h[zeros[i]] = 1.0
        else:
            ones = np.where(h == 1)[0]
            for i in range(int(-diff)):
                if i < len(ones):

def simulated_annealing_binary(h, n, deadline, rng, t_start=2.0, t_end=0.01, max_iters=200000):

    it = 0
    n_total = max_iters

        it += 1
        frac = it / n_total
        T = t_start * (t_end / t_start) ** frac

        i = 0
        j = 0
        while i == j:
            i = rng.randint(n)
            j = rng.randint(n)

        if h[i] == h[j]:
            continue

        # swap
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        delta = c_new - cur_c

        if delta < 0 or (T > 1e-6 and rng.rand() < np.exp(-delta / T)):
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()

            h[i], h[j] = h[j], h[i]

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    val = m + np.log(np.sum(np.exp(sh - smx)) + 1e-300) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val


def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g


def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h, jac=lbfgs_grad, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxit, 'ftol': 1e-25,
                                'gtol': 1e-16, 'maxfun': maxit * 10,
                                'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

        sub_dl = (d4 - time.time()) / len(starts)
        for s in starts:
            if time.time() > d4:
                break
            sd = time.time() + sub_dl
            hs, cs = gradient_binary_search(s, n, sd, rng, max_iters=50000)
            if cs < best_c:
                best_c = cs
                best_h = hs

    # Phase 5: More binary search from best (100s)
    d5 = min(t0 + 900, deadline - 100)
    if time.time() < d5:
        h5, c5 = gradient_binary_search(binarize(best_h, n), n, d5, rng, max_iters=100000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Continuous L-BFGS refinement (80s)
    d6 = min(t0 + 980, deadline - 50)
    if time.time() < d6:
        h6 = project_feasible(best_h.copy(), n)
        for alpha in [50000, 200000, 1000000]:
            if time.time() > d6 - 5:
                break
            h6, c6 = run_lbfgs(h6, n, d6, alpha=alpha, mu=0.001, maxit=500)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Fine coordinate descent (40s)
    d7 = min(t0 + 1020, deadline - 20)
    if time.time() < d7:
        h7, c7 = fine_coord_descent(best_h, n, d7,
                                    steps=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                    max_rounds=200)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final binary polish
    if time.time() < deadline - 5:
        h8, c8 = gradient_binary_search(binarize(best_h, n), n, deadline - 2, rng, max_iters=50000)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
