    m = c.max()

    m = c.max()

    mx = sh.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wi = w[mm]
        if wi < 1e-15:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas, mu, maxiter_list=None):
    best = project_feasible(h.copy(), n)
    best_c = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        if time.time() > deadline - 20:
            break
        if maxiter_list is not None and i < len(maxiter_list):
            maxit = maxiter_list[i]
        else:
            maxit = max(500, int((deadline - time.time()) * 4))
        try:
            res = minimize(sm_obj, best, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best = h_new
        except Exception:
            pass
    return best, best_c

                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:

def subgrad_refine(h, n, deadline, n_iter=10000):

def fourier_smooth(h, n, n_modes):
    """Fit a Fourier series to h and return the smoothed version."""
    x = np.linspace(0, 2, n, endpoint=False)
    # Use real FFT
    H = np.fft.rfft(h)
    # Keep only first n_modes + 1 frequencies
    H_smooth = np.zeros_like(H)
    keep = min(n_modes + 1, len(H))
    H_smooth[:keep] = H[:keep]
    h_smooth = np.fft.irfft(H_smooth, n=n)
    return h_smooth

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def slq_refine(h, n, deadline, alpha=1000000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return sm_obj(x, n, alpha, 0.0)
    def jac(x):
        return sm_grad(x, n, alpha, 0.0)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

        h4, c4 = run_lbfgs(h_start_h, n_high, p4_deadline,
                          alphas=[500, 5000, 50000, 200000, 1000000],
                          mu=0.003,
                          maxiter_list=[1500, 2000, 2500, 2500, 2500])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high

    # Phase 5: Active set refinement at best resolution
    p5_deadline = min(time.time() + 150, deadline - 100)
    if time.time() < p5_deadline:
        h5, c5 = run_active(best_h, best_n, p5_deadline, top_k=12,
                           alphas=[200000, 1000000, 5000000, 20000000])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Subgradient refinement
    p6_deadline = min(time.time() + 80, deadline - 60)
    if time.time() < p6_deadline:
        h6, c6 = subgrad_refine(best_h, best_n, p6_deadline, n_iter=5000)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Local pair search
    p7_deadline = min(time.time() + 100, deadline - 30)
    if time.time() < p7_deadline:
        h7, c7 = local_pair_search(best_h, best_n, p7_deadline, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final active set + pair
    p8_deadline = deadline - 5
    if time.time() < p8_deadline:
        h8, c8 = run_active(best_h, best_n, min(time.time() + 50, p8_deadline - 30),
                           top_k=6, alphas=[1000000, 5000000, 20000000])
        if c8 < best_c:
            best_c = c8
            best_h = h8
        if time.time() < p8_deadline - 5:
            h8b, c8b = local_pair_search(best_h, best_n, p8_deadline, radius=25)
            if c8b < best_c:
                best_c = c8b
                best_h = h8b

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
