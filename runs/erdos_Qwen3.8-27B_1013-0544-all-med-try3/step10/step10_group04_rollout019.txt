    # This returns the delta for each position in c_vec
    # c_vec has 2n-1 entries
    # c_vec[m] corresponds to shift s = m - (n-1)

    # Δpos[s] from i: -1 if (i+s<n and h[i+s]==0)
    #                 +1 if (i-s>=0 and h[i-s]==1)
    # Δpos[s] from j: +1 if (j+s<n and h[j+s]==0)
    #                 -1 if (j-s>=0 and h[j-s]==1)

    # Δneg[s] from i: -1 if (i-s>=0 and h[i-s]==0)

    # Δneg[s] from j: +1 if (j-s>=0 and h[j-s]==0)
    #                 -1 if (j+s<n and h[j+s]==1)

    # shift -s is at index (n-1)-s

    for s in range(1, n - i):
        if h[i+s] == 0:
            deltas[n-1+s] -= 1
    for s in range(1, i+1):
        if h[i-s] == 1:
            deltas[n-1+s] += 1

    for s in range(1, n - j):
        if h[j+s] == 0:
            deltas[n-1+s] += 1
    for s in range(1, j+1):
        if h[j-s] == 1:
            deltas[n-1+s] -= 1
    
    # i (1→0) effects on neg (shift -s, index n-1-s):

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return float(c.max())

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def to_binary(h, n):
    """Project h to binary with exactly n/2 ones."""
    # Sort by value, top n/2 become 1, rest become 0
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def swap_deltas(h, n, i, j):
    """Compute delta in raw correlate (before *dx) for swap i:1->0, j:0->1."""
    n2m1 = 2*n - 1

    # pos_count (shift +s, index center+s):
    #   i as first: h[i+s]==0 → delta -= 1 for s=1..n-1-i
    #   i as second: h[i-s]==1 → delta += 1 for s=1..i
    # neg_count (shift -s, index center-s):
    #   i as first: h[i-s]==0 → delta -= 1 for s=1..i
    #   i as second: h[i+s]==1 → delta += 1 for s=1..n-1-i
    
    for s in range(1, n - i):
        if h[i+s] == 0:

def to_binary(h, n):

    h_bin = np.zeros(n)

def smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val * (2.0 / n))


def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    return best_h, best_c, n_final
