def project_feasible(h, n):
    target = n / 2.0

            tot = room.sum()
            if tot < 1e-14:

            if tot < 1e-14:

def compute_swap_delta(h, n, i, j):
    """Compute delta in correlation when h[i]: 1->0 and h[j]: 0->1."""
    d = np.zeros(2*n-1)
    # h[i] 1->0:
    # -(1-h[i+m]) for m in [-i, n-i-1]
    lo1 = max(-(n-1), -i)
    hi1 = min(n-1, n-1-i)
    if lo1 <= hi1:
        idx1 = np.arange(lo1, hi1+1)
        d[n-1+idx1] -= (1.0 - h[i+idx1])
    # +h[i-m] for m in [i-(n-1), i]
    lo2 = max(-(n-1), -(n-1-i))  # i-m >= 0 => m <= i; i-m <= n-1 => m >= i-(n-1)
    hi2 = min(n-1, i)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2+1)
        d[n-1+idx2] += h[i-idx2]
    # h[j] 0->1:
    # +(1-h[j+m]) for m in [-j, n-j-1]
    lo3 = max(-(n-1), -j)
    hi3 = min(n-1, n-1-j)
    if lo3 <= hi3:
        idx3 = np.arange(lo3, hi3+1)
        d[n-1+idx3] += (1.0 - h[j+idx3])
    # -h[j-m] for m in [j-(n-1), j]
    lo4 = max(-(n-1), -(n-1-j))
    hi4 = min(n-1, j)
    if lo4 <= hi4:
        idx4 = np.arange(lo4, hi4+1)
        d[n-1+idx4] -= h[j-idx4]
    return d

def smooth_obj(x, n, alpha, mu):

    mx = sh.max()

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)


def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])

            nv = n - kk
            if nv > 0:

                grad[0:nv] += wm * (-x[kk:n])

    return grad

def slqp_polish(h, n, deadline, alpha=500000, maxiter=3000):

        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)

    def sq_fn(x, na):
        return float(x.sum() - na / 2.0)

    def sq_jac_fn(x, na):
        return np.ones(na)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fn, 'jac': sq_jac_fn, 'args': (n,)}]
    try:

        t_cont = time.time() + min(remaining * 0.4, 300)
        h_cont = best_h.copy()
        h_cont, c_cont = run_lbfgs(h_cont, best_n, t_cont,
                                    configs=[(200, 50.0), (500, 200.0), (1000, 500.0),
                                             (3000, 2000.0), (8000, 8000.0), (15000, 15000.0)])
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont.copy()

    # Phase 3: SLSQP polish
    remaining = deadline - time.time()
    if remaining > 50:
        t_slq = time.time() + min(remaining * 0.3, 200)
        h_slq, c_slq = slqp_polish(best_h, best_n, t_slq, alpha=2000000, maxiter=4000)
        if c_slq < best_c:
            best_c = c_slq
            best_h = h_slq.copy()

    # Phase 4: Re-binarize and do more local search
    remaining = deadline - time.time()
    if remaining > 40:
        h_ref_bin = binarize_keep_sum(best_h, best_n)
        t_re = time.time() + min(remaining * 0.3, 150)
        h_re, c_re = binary_hill_climb(h_ref_bin, best_n, t_re, rng,
                                        max_iters=2000000, patience=300,
                                        perturbation_size=2)
        if c_re < best_c:
            best_c = c_re
            best_h = h_re.copy()
            # Quick continuous relaxation after
            remaining = deadline - time.time()
            if remaining > 20:
                h_c2, c_c2 = run_lbfgs(h_re, best_n, time.time() + min(remaining * 0.5, 80),
                                        configs=[(1000, 500.0), (5000, 2000.0)])
                if c_c2 < best_c:
                    best_c = c_c2
                    best_h = h_c2.copy()

    # Phase 5: Final targeted refinement
    remaining = deadline - time.time()
    if remaining > 20:
        best_h, best_c = targeted_refine(best_h, best_n, time.time() + min(remaining * 0.5, 60))

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
