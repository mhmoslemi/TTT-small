import time

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def subgradient_method(h, n, deadline, n_iters=5000):
    """Subgradient method for the min-max problem."""
    dx = 2.0 / n
    h = project_feasible(h, n)
    lr = 0.1
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1.0 - h, mode='full') * dx
        max_g = g.max()
        
        # Subgradient: uniform over argmax positions
        max_idx = np.where(np.abs(g - max_g) < 1e-10)[0]
        n_max = len(max_idx)
        
        # Compute gradient of g[k] w.r.t. h
        # g[k] = sum_i h[i] * (1-h[i+k]) * dx for the appropriate range
        # dg[k]/dh[m] = (1-h[m+k])*dx - h[m-k]*dx (for valid indices)
        # Subgradient of max = average of gradients at max positions
        grad = np.zeros(n)
        for k_idx in max_idx:
            k = k_idx - (n - 1)
            # g[k_idx] = sum_i h[i] * (1-h[i+k]) for valid i
            # d/dh[m]: term where i=m: (1-h[m+k]), term where i+k=m i.e. i=m-k: -h[m-k]
            m = np.arange(n)
            mk = m + k
            mkn = m - k
            mask_f = (mk >= 0) & (mk < n)
            mask_b = (mkn >= 0) & (mkn < n)
            grad[mask_f] += (1.0 - h[mk[mask_f]]) * dx / n_max
            grad[mask_b] -= h[mkn[mask_b]] * dx / n_max
        
        # Update
        h = h - lr * grad
        h = project_feasible(h, n)
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Decrease learning rate
        lr *= 0.999
    
    return best_h, best_c

        h_work = slsqp_optimize(h_work, n0, a, 1200, p1_end)
        c = compute_c5(h_work, n0)
        if c < best_c:
            best_c = c
            best_h = h_work.copy()
            n_best = n0

    # Phase 2: Intensive single-swap on n0
    p2_end = t0 + 0.30 * (budget_s - 5)
    target_n = n0 // 2
    idx_bin = np.argsort(h_work)[::-1]
    h_bin = np.zeros(n0)
    h_bin[idx_bin[:target_n]] = 1.0
    corr_bin = np.correlate(h_bin, 1.0 - h_bin, mode='full').astype(float)
    c_bin = float(corr_bin.max()) * (2.0 / n0)

    h_bin, bh1, bc1, corr_bin = single_swap_search(h_bin, n0, corr_bin, c_bin, p2_end, max_sweeps=5000)
    if bc1 < best_c:
        best_c = bc1
        best_h = bh1.copy()
        n_best = n0

    # Phase 3: Block moves (size 2, 3) on n0
    p3_end = t0 + 0.45 * (budget_s - 5)
    h_blk, bh3, bc3 = block_swap_search(best_h, n_best, p3_end, block_size=2, max_sweeps=150, n_restarts=3, seed=seed + 10)
    if bc3 < best_c:
        best_c = bc3
        best_h = bh3.copy()
        n_best = n0

    h_blk2, bh3b, bc3b = multi_block_search(best_h, n_best, p3_end, sizes=[2, 3], max_sweeps_per_size=60, n_restarts=2, seed=seed + 20)
    if bc3b < best_c:
        best_c = bc3b
        best_h = bh3b.copy()
        n_best = n0

    # Phase 4: Simulated annealing on n0
    p4_end = t0 + 0.55 * (budget_s - 5)
    if time.time() < p4_end - 10:
        h_ann, bh4, bc4 = anneal_binary_search(best_h, n_best, p4_end, T_init=0.0015, T_min=1e-7, cooldown=0.997, max_sweeps=3000, seed=seed + 30)
        if bc4 < best_c:
            best_c = bc4
            best_h = bh4.copy()
            n_best = n_best

    # Phase 5: Post-anneal single swaps
    p5_end = t0 + 0.62 * (budget_s - 5)
    if time.time() < p5_end:
        corr_now = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
        c_now = float(corr_now.max()) * (2.0 / n_best)
        _, bh5, bc5, _ = single_swap_search(best_h, n_best, corr_now, c_now, p5_end, max_sweeps=2000)
        if bc5 < best_c:
            best_c = bc5
            best_h = bh5.copy()

    # Phase 6: Multi-resolution search
    p6_end = t0 + 0.85 * (budget_s - 5)
    resolutions = [240, 260, 280, 300, 320, 350]
    for nh in resolutions:
        if time.time() > p6_end - 30:
            break
        if nh == n_best:
            continue
        t_sub = min(p6_end - 5, time.time() + 80)
        try:
            hh, hc, hn = resolve_and_optimize(best_h, n_best, nh, t_sub, seed=seed + 100 + nh)
            if hc < best_c:
                best_c = hc
                best_h = hh.copy()
                n_best = nh
        except Exception:
            pass

    # Phase 7: Final intensive polish on best resolution
    p7_end = deadline - 5
    if time.time() < p7_end - 20:
        corr_final = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
        c_final = float(corr_final.max()) * (2.0 / n_best)
        _, bh7, bc7, _ = single_swap_search(best_h, n_best, corr_final, c_final, min(p7_end - 5, time.time() + 60), max_sweeps=2000)
        if bc7 < best_c:
            best_c = bc7
            best_h = bh7.copy()
            n_best = n_best

    if time.time() < p7_end - 10:
        h_b7, bh7b, bc7b = block_swap_search(best_h, n_best, min(p7_end - 3, time.time() + 30), block_size=2, max_sweeps=50, n_restarts=2, seed=seed + 200)
        if bc7b < best_c:
            best_c = bc7b
            best_h = bh7b.copy()
            n_best = n_best

    # Final SLSQP polish if time allows
    if time.time() < p7_end - 8:
        h_s = slsqp_optimize(best_h + 0.01 * rng_main.randn(n_best), n_best, 5000000, 2000, p7_end - 5)
        c_s = compute_c5(h_s, n_best)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s.copy()
            n_best = n_best
        # Snap to binary after SLSQP
        if time.time() < p7_end - 3:
            tgt = n_best // 2
            idx_s = np.argsort(h_s)[::-1]
            h_snap = np.zeros(n_best)
            h_snap[idx_s[:tgt]] = 1.0
            c_snap = compute_c5(h_snap, n_best)
            corr_s = np.correlate(h_snap, 1.0 - h_snap, mode='full').astype(float)
            _, bh_f, bc_f, _ = single_swap_search(h_snap, n_best, corr_s, c_snap, p7_end - 1, max_sweeps=500)
            if bc_f < best_c:
                best_c = bc_f
                best_h = bh_f.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)

    return best_h, best_c, n_best
