def guided_local_search(h, n, deadline):
    best_h = h.copy()
    c_vec = compute_c5_vec(best_h, n)
    best_c = c_vec.max()
    dx = 2.0 / n
    
    for iteration in range(1000000):
        if time.time() > deadline:
            break
        
        # Find active shifts (top few)
        active = np.argsort(-c_vec)[:5]
        
        # Generate candidate swaps from active shifts
        candidates = set()
        for k_idx in active:
            k = k_idx - (n - 1)  # actual shift
            if k > 0:
                # c[k_idx] = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])
                # To reduce: swap h[i+k]=1 with h[i]=0
                for i in range(n - k):
                    if best_h[i+k] > 0.5 and best_h[i] < 0.5:
                        candidates.add((i+k, i))  # swap these two
            elif k < 0:
                kk = -k
                # c[k_idx] = sum_{i=kk}^{n-1} h[i] * (1-h[i-kk])
                # To reduce: swap h[i]=1 with h[i-kk]=0
                for i in range(kk, n):
                    if best_h[i] > 0.5 and best_h[i-kk] < 0.5:
                        candidates.add((i, i-kk))
            # k == 0: c = 0 for binary, skip
        
        if not candidates:
            # Fallback: random swap
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            candidates.add((i, j))
        
        # Try each candidate
        best_gain = 0
        best_pair = None
        for (a, b) in candidates:
            if best_h[a] < 0.5 or best_h[b] > 0.5:
                continue  # already swapped or invalid
            best_h[a] = 1 - best_h[a]
            best_h[b] = 1 - best_h[b]
            c_new = compute_c5_vec(best_h, n).max()
            if c_new < best_c - best_gain:
                best_gain = best_c - c_new
                best_pair = (a, b)
                best_h[a] = 1 - best_h[a]  # undo for now
                best_h[b] = 1 - best_h[b]
            else:
                best_h[a] = 1 - best_h[a]
                best_h[b] = 1 - best_h[b]
        
        if best_pair is not None:
            a, b = best_pair
            best_h[a] = 1 - best_h[a]
            best_h[b] = 1 - best_h[b]
            c_vec = compute_c5_vec(best_h, n)
            best_c = c_vec.max()

            best_h[i] = 0
            best_h[j] = 1
            c_new = compute_c5_vec(best_h, n).max()
            if c_new > best_c:
                best_h[i] = 1
                best_h[j] = 0
    
    return best_h, best_c
