            if tot < 1e-14:

def binary_local_search(h, n, deadline):
    """Local search on binary version of h."""
    # Threshold to binary
    b = (h >= 0.5).astype(float)
    # Fix to exactly n/2 ones
    num_ones = int(b.sum())
    if num_ones > n/2:
        # Need to flip some 1s to 0s
        idx_ones = np.where(b == 1)[0]
        np.random.shuffle(idx_ones)
        for i in idx_ones[num_ones - n//2:]:
            b[i] = 0
    elif num_ones < n/2:
        idx_zeros = np.where(b == 0)[0]
        np.random.shuffle(idx_zeros)
        for i in idx_zeros[:n//2 - num_ones]:
            b[i] = 1
    
    best_c = compute_c5(b, n)
    np.random.seed(42)
    
    # Local search: try flipping pairs
    no_improve = 0
    while time.time() < deadline and no_improve < 500:
        i, j = np.random.randint(0, n, 2)
        if i == j or b[i] == b[j]:
            no_improve += 1
            continue
        # Flip: b[i] 1->0, b[j] 0->1
        b[i], b[j] = 1-b[i], 1-b[j]
        c = compute_c5(b, n)
        if c < best_c - 1e-15:
            best_c = c
            no_improve = 0
        else:
            b[i], b[j] = 1-b[i], 1-b[j]
            no_improve += 1
    
    # Convert back to continuous for further optimization
    h_cont = b.copy()
    return h_cont, best_c

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 500:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2

    m = c.max()

def smooth_grad(x, n, alpha, mu):

    m = c.max()

    smx = sh.max()
    w = np.exp(sh - smx)
