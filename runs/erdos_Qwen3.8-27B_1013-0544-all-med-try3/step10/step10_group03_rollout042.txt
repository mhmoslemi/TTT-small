    # For each shift k, the change is:
    # delta * [(1-h[i+k]) - h[i-k]] - delta * [(1-h[j+k]) - h[j-k]]
    # (with appropriate bounds checking)

    # c_vec has length 2n-1, indexed by correlation index
    # Shift k corresponds to index k + (n-1)

    vals = c[top_idx]
    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi < 1e-13:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def compute_delta_vec(h, n, i, j, shifts):
    """Compute delta vector for correlation when h[i]+=1, h[j]-=1."""
    dx = 2.0 / n
    n_corr = len(shifts)
    di = np.zeros(n_corr)
    valid1_1 = (i + shifts >= 0) & (i + shifts < n)
    di[valid1_1] += (1.0 - h[i + shifts[valid1_1]])
    valid1_2 = (i - shifts >= 0) & (i - shifts < n)
    di[valid1_2] += (-h[i - shifts[valid1_2]])
    dj = np.zeros(n_corr)
    valid2_1 = (j + shifts >= 0) & (j + shifts < n)
    dj[valid2_1] += (1.0 - h[j + shifts[valid2_1]])
    valid2_2 = (j - shifts >= 0) & (j - shifts < n)
    dj[valid2_2] += (-h[j - shifts[valid2_2]])
    return (di - dj) * dx

def fast_local_search(h, n, deadline, max_passes=200):
    """Fast local search using incremental correlation updates."""
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        
        max_val = c_vec.max()
        threshold = max_val * 0.998
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 30:
            active = active[np.argsort(-c_vec[active])[:30]]
        
        # Compute combined gradient over active shifts
        combined_grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            dx = 2.0 / n
            if k >= 0:
                nv = n - k
                combined_grad[0:nv] += (1.0 - h[k:n]) * dx
                combined_grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                combined_grad[kk:n] += (1.0 - h[0:nv]) * dx
                combined_grad[0:nv] += (-h[kk:n]) * dx
        combined_grad /= len(active)
        
        # Select top gradient positions for pair search
        top_positions = np.argsort(-np.abs(combined_grad))[:60]
        
        improved_this_pass = False
        for ii in range(len(top_positions)):
            if time.time() > deadline:
                return h, best_c, c_vec
            i = top_positions[ii]
            gi = combined_grad[i]
            for jj in range(len(top_positions)):
                if ii == jj:
                    continue
                j = top_positions[jj]
                gj = combined_grad[j]
                
                # Determine direction: move from higher grad to lower
                if gi > gj + 1e-10:
                    # h[i] decreases, h[j] increases
                    for delta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                        if h[i] - delta < 0 or h[j] + delta > 1:
                            continue
                        dv = compute_delta_vec(h, n, i, j, shifts)
                        new_c = c_vec - delta * dv
                        new_max = new_c.max()
                        if new_max < best_c - 1e-14:
                            h[i] -= delta
                            h[j] += delta
                            c_vec = new_c
                            best_c = new_max
                            improved_this_pass = True
                            break
                elif gj > gi + 1e-10:
                    # h[j] decreases, h[i] increases
                    for delta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                        if h[j] - delta < 0 or h[i] + delta > 1:
                            continue
                        dv = compute_delta_vec(h, n, j, i, shifts)
                        new_c = c_vec - delta * dv
                        new_max = new_c.max()
                        if new_max < best_c - 1e-14:
                            h[j] -= delta
                            h[i] += delta
                            c_vec = new_c
                            best_c = new_max
                            improved_this_pass = True
                            break
        if not improved_this_pass:
            # Try smaller steps
            improved = False
            for ii in range(min(30, len(top_positions))):
                if time.time() > deadline:
                    return h, best_c, c_vec
                i = top_positions[ii]
                gi = combined_grad[i]
                for jj in range(min(30, len(top_positions))):
                    if ii == jj:
                        continue
                    j = top_positions[jj]
                    gj = combined_grad[j]
                    if gi > gj + 1e-10:
                        for delta in [0.0001, 0.00005, 0.00002]:
                            if h[i] - delta < 0 or h[j] + delta > 1:
                                continue
                            dv = compute_delta_vec(h, n, i, j, shifts)
                            new_c = c_vec - delta * dv
                            new_max = new_c.max()
                            if new_max < best_c - 1e-14:
                                h[i] -= delta
                                h[j] += delta
                                c_vec = new_c
                                best_c = new_max
                                improved = True
                                break
                            if delta < 0.0001:
                                break
                    elif gj > gi + 1e-10:
                        for delta in [0.0001, 0.00005, 0.00002]:
                            if h[j] - delta < 0 or h[i] + delta > 1:
                                continue
                            dv = compute_delta_vec(h, n, j, i, shifts)
                            new_c = c_vec - delta * dv
                            new_max = new_c.max()
                            if new_max < best_c - 1e-14:
                                h[j] -= delta
                                h[i] += delta
                                c_vec = new_c
                                best_c = new_max
                                improved = True
                                break
            if not improved:
                break
    
    return h, best_c, c_vec

def global_pair_search(h, n, deadline, n_random=200, n_passes=50):
    """Global random pair search with incremental updates."""
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n_random):
            if time.time() > deadline:
                return h, best_c, c_vec
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            for direction in [1, -1]:
                delta = 0.005
                if direction == 1:
                    if h[i] + delta > 1 or h[j] - delta < 0:
                        continue
                    dv = compute_delta_vec(h, n, i, j, shifts)
                    new_c = c_vec + delta * dv
                    new_max = new_c.max()
                    if new_max < best_c - 1e-14:
                        h[i] += delta
                        h[j] -= delta
                        c_vec = new_c
                        best_c = new_max
                        improved = True
                else:
                    if h[i] - delta < 0 or h[j] + delta > 1:
                        continue
                    dv = compute_delta_vec(h, n, j, i, shifts)
                    new_c = c_vec + delta * dv
                    new_max = new_c.max()
                    if new_max < best_c - 1e-14:
                        h[i] -= delta
                        h[j] += delta
                        c_vec = new_c
                        best_c = new_max
                        improved = True
        if not improved:
            break
    return h, best_c, c_vec

def binary_projection_test(h, n):
    """Test if binary projection improves the solution."""
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[n // 2:]] = 1.0
    c_bin = compute_c5(h_bin, n)
    return h_bin, c_bin

def partial_binary_optimize(h, n, deadline, tol=0.05):
    """Fix points close to 0 or 1, optimize the rest."""
    h_work = h.copy()
    fixed_mask = (h_work < tol) | (h_work > 1.0 - tol)
    n_fixed = fixed_mask.sum()
    n_free = n - n_fixed
    
    if n_free < 10 or n_free > n - 10:
        return h, compute_c5(h, n)
    
    free_idx = np.where(~fixed_mask)[0]
    fixed_idx = np.where(fixed_mask)[0]
    
    # Project fixed values to exactly 0 or 1
    h_work[fixed_idx < tol if isinstance(fixed_idx, float) else (h_work[fixed_idx] < 0.5)] = 0.0
    h_work[fixed_idx[~(h_work[fixed_idx] < 0.5)]] = 1.0
    
    # Compute required sum for free variables
    target_sum = n / 2.0 - h_work[fixed_idx].sum()
    
    # Optimize only free variables
    h_free = h_work[free_idx].copy()
    h_free = np.clip(h_free, 0.0, 1.0)
    
    # Scale to meet target
    for _ in range(100):
        diff = target_sum - h_free.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h_free
            tot = room.sum()
            if tot < 1e-15:
                break
            h_free += (diff / tot) * room
        else:
            tot = h_free.sum()
            if tot < 1e-15:
                break
            h_free += (diff / tot) * h_free
    
    h_work[free_idx] = np.clip(h_free, 0.0, 1.0)
    h_work = project_feasible(h_work, n)
    c = compute_c5(h_work, n)
    
    if c < compute_c5(h, n):
        return h_work, c
    return h, compute_c5(h, n)

def multi_start_optimize(best_h, best_c, n, deadline, n_starts=3):
    """Try multiple perturbed starting points."""
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        t_phase = min(60, (deadline - time.time()) / (n_starts - s))
        if t_phase < 30:
            break
        
        # Create perturbation
        h_pert = best_h.copy()
        if s == 0:
            # Small random perturbation
            noise = np.random.normal(0, 0.01, n)
            h_pert += noise
        elif s == 1:
            # Swap a few pairs
            for _ in range(5):
                i, j = np.random.choice(n, 2, replace=False)
                val = h_pert[i]
                h_pert[i] = h_pert[j]
                h_pert[j] = val
        else:
            # Smooth and re-project
            kernel = np.array([0.1, 0.15, 0.5, 0.15, 0.1])
            h_pert = np.convolve(h_pert, kernel, mode='same')
        
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        # Quick L-BFGS
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lse_obj, h_pert, jac=lse_grad,
                          args=(n, 5000, 100.0), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': 20000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            
            # Quick local search
            h_new, c_new, _ = fast_local_search(h_new, n, time.time() + 10)
            
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_orig = 250
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick L-BFGS warm-up at original resolution (30s)
    t1 = min(t0 + 30, deadline - 800)
    if t1 > t0 + 5:
        bounds = [(0.0, 1.0)] * n_orig
        for alpha, mu in [(2000, 100.0), (10000, 1000.0)]:
            if time.time() > t1 - 3:
                break
            try:
                res = minimize(lse_obj, best_h, jac=lse_grad,
                              args=(n_orig, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-17,
                                       'gtol': 1e-14, 'maxfun': 20000})
                h_new = project_feasible(res.x, n_orig)
                c_new = compute_c5(h_new, n_orig)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 2: Upsample to n=500 if needed
    n = 500
    if n_orig != n:
        x_old = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n)
        c_up = compute_c5(h_up, n)
        best_h = h_up
        best_c = c_up
    n = len(best_h)
    
    # Phase 3: Fast local search with incremental updates (120s)
    t3 = min(t0 + 150, deadline - 600)
    best_h, best_c, _ = fast_local_search(best_h, n, t3, max_passes=300)
    
    # Phase 4: L-BFGS with top-k focus (200s)
    t4 = min(t0 + 350, deadline - 400)
    if time.time() < t4:
        bounds = [(0.0, 1.0)] * n
        c_vec = compute_c5_vec(best_h, n)
        for alpha in [5000, 20000, 100000]:
            if time.time() > t4 - 10:
                break
            # Find active shifts
            max_val = c_vec.max()
            threshold = max_val - max(1e-6, 0.002 * max_val)
            active = np.where(c_vec >= threshold)[0]
            if len(active) > 10:
                active = active[np.argsort(-c_vec[active])[:10]]
            if len(active) == 0:
                active = np.array([np.argmax(c_vec)])
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active, alpha, 10.0), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-17,
                                       'gtol': 1e-14, 'maxfun': 30000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    c_vec = compute_c5_vec(best_h, n)
            except Exception:
                pass
    
    # Phase 5: More fast local search (100s)
    t5 = min(t0 + 450, deadline - 300)
    best_h, best_c, _ = fast_local_search(best_h, n, t5, max_passes=400)
    
    # Phase 6: Global pair search (60s)
    t6 = min(t0 + 510, deadline - 250)
    best_h, best_c, _ = global_pair_search(best_h, n, t6, n_random=100, n_passes=40)
    
    # Phase 7: Another L-BFGS round with higher alpha (100s)
    t7 = min(t0 + 610, deadline - 200)
    if time.time() < t7:
        bounds = [(0.0, 1.0)] * n
        for alpha, mu in [(50000, 1000.0), (200000, 5000.0)]:
            if time.time() > t7 - 10:
                break
            try:
                res = minimize(lse_obj, best_h, jac=lse_grad,
                              args=(n, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': 50000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    
    # Phase 8: Fast local search round 3 (80s)
    t8 = min(t0 + 690, deadline - 150)
    best_h, best_c, _ = fast_local_search(best_h, n, t8, max_passes=300)
    
    # Phase 9: Multi-start (80s)
    t9 = min(t0 + 770, deadline - 100)
    if time.time() < t9:
        best_h, best_c = multi_start_optimize(best_h, best_c, n, t9, n_starts=2)
    
    # Phase 10: Final local search polish (remaining time)
    t10 = deadline - 5
    if time.time() < t10:
        best_h, best_c, _ = fast_local_search(best_h, n, t10, max_passes=500)
    
    # Final validation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
