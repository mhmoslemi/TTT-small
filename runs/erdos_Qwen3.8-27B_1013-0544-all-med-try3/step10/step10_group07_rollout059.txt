def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(-h if diff > 0 else h)
        for i in idx:
            if diff > 0:
                step = min(1.0 - h[i], diff)
                if step > 1e-14:

                    diff -= step

                step = min(h[i], -diff)
                if step > 1e-14:

                    diff += step

                    break
    return h

def multiplicative_weights(h, n, deadline, max_rounds=1000):
    dx = 2.0 / n
    n_lags = 2*n - 1
    h = h.copy()
    w = np.ones(n_lags) / n_lags
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        # Optimize weighted sum: min sum_k w_k * c_k(h)
        # subject to 0<=h<=1, sum(h)=n/2
        # c_k is quadratic in h, so this is a QP (if we keep it linear, it's LP)
        # Actually c_k = sum_i h[i](1-h[i+k]) which is quadratic
        # For simplicity, linearize: min sum_k w_k * sum_i (h[i] - h[i+k]) * ... 
        # Or just do gradient descent on the weighted sum
        
        c = np.correlate(h, 1-h, mode='full')
        obj = (w * c).sum()
        c_max = c.max() * dx
        
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        
        # Update weights (multiplicative weights)
        alpha = 0.1  # learning rate
        w = w * np.exp(alpha * c / c.max())  # emphasize large c_k
        w = w / w.sum()
        
        # Gradient step on weighted objective
        # d/dh[i] of sum_k w_k * c_k(h)
        # c_k(h) = sum_j h[j](1-h[j+k]) for valid j
        # = sum_j h[j] - sum_j h[j]*h[j+k]
        # d/dh[i]: appears as h[i] in first sum (coefficient 1 if i valid for lag k)
        #           and as -h[i]*h[i+k] or -h[i-k]*h[i] in second sum
        
        grad = np.zeros(n)
        for k_idx in range(n_lags):
            k = k_idx - (n-1)
            if k >= 0:
                valid = np.arange(0, n-k)
                # +h[j] term: +1 for j in valid
                grad[valid] += w[k_idx]
                # -h[j]*h[j+k] term: -h[j+k] for j, -h[j] for j+k
                grad[valid] += -w[k_idx] * h[valid + k]
                valid2 = np.arange(k, n)
                grad[valid2] += -w[k_idx] * h[valid2 - k]
            else:
                valid = np.arange(-k, n)
                grad[valid] += w[k_idx]
                grad[valid] += -w[k_idx] * h[valid + k]
                valid2 = np.arange(0, n+k)
                grad[valid2] += -w[k_idx] * h[valid2 - k]
        
        # Projected gradient step
        lr = 0.01
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        h = h_new
    
    return best_h, best_c

def subgradient_optimize(h, n, deadline, max_iters=5000):

    dx = 2.0 / n
    n_lags = 2*n - 1
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5 / np.sqrt(max_iters)

        c = np.correlate(h, 1-h, mode='full')
        # Find the max lag

        k = idx_max - (n-1)

        grad = np.zeros(n)
        if k >= 0:

            grad[valid1] = 1.0 - h[valid1 + k]

            grad[valid1] = 1.0 - h[valid1 + k]

def softmax_smooth_max(h, n, alpha, deadline, max_iters=500, rng=None):

    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()

        m = c.max()

        # d/dh[i] of sum_k c_k * p_k where p_k = exp(c_k/alpha) / sum(exp(c_j/alpha))
        # Actually, softmax(m) = m + (1/alpha) * log(sum exp(alpha*(c_k - m)))
        # Let's just use the gradient of sum_k w_k * c_k where w_k ∝ exp(c_k / alpha)
        exp_c = np.exp((c - m) / dx * 0.01)  # Normalize for numerical stability
        w = exp_c / exp_c.sum()

        grad = np.zeros(n)
        for idx in range(len(c)):
            k = idx - (n-1)
            wk = w[idx]
            if wk < 1e-6:

            if k >= 0:
                valid1 = np.arange(0, n-k)
                grad[valid1] += wk * (1.0 - h[valid1 + k])
                if k > 0:

                    grad[valid2] += wk * (-h[valid2 - k])

                grad[valid1] += wk * (1.0 - h[valid1 + k])
                valid2 = np.arange(0, n+k)
                grad[valid2] += wk * (-h[valid2 - k])

        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                step = min(1.0 - h[i], diff)
                if step > 1e-14:
                    h[i] += step
                    diff -= step
                if abs(diff) < 1e-13:
                    break
            else:
                step = min(h[i], -diff)
                if step > 1e-14:
                    h[i] -= step
                    diff += step
                if abs(diff) < 1e-13:
                    break
    return h

def compute_delta_single(h, n, ia, ic, m_range):
    n_lags = 2 * n - 1
    delta = np.zeros(n_lags)
    im = ia + m_range
    imn = ia - m_range
    cm = ic + m_range
    cmn = ic - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (cm >= 0) & (cm < n)
    delta[msk] -= 1.0 - h[cm[msk]]
    msk = (cmn >= 0) & (cmn < n)
    delta[msk] += h[cmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (ic - ia)
    delta[k1] += 1.0
    k2 = n - 1 + (ia - ic)
    delta[k2] += 1.0
    return delta

def binary_local_search(h, n, deadline, max_sweeps=20000, rng=None, perturb_every=80):
    if rng is None:
        rng = np.random.RandomState(0)
    n_lags = 2 * n - 1
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    m_range = np.arange(-(n - 1), n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones_list = np.where(h == 1)[0].tolist()
        zeros_list = np.where(h == 0)[0].tolist()
        if not ones_list or not zeros_list:
            break
        improved = False

        for ia in zeros_list:
            if time.time() > deadline:
                break
            im = ia + m_range
            imn = ia - m_range
            base = np.zeros(n_lags)
            msk = (im >= 0) & (im < n)
            if msk.any():
                base[msk] = 1.0 - h[im[msk]]
            msk = (imn >= 0) & (imn < n)
            if msk.any():
                base[msk] -= h[imn[msk]]

            for ic in ones_list:
                if time.time() > deadline:
                    break
                cm = ic + m_range
                cmn = ic - m_range
                delta = base.copy()
                msk = (cm >= 0) & (cm < n)
                if msk.any():
                    delta[msk] -= 1.0 - h[cm[msk]]
                msk = (cmn >= 0) & (cmn < n)
                if msk.any():
                    delta[msk] += h[cmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (ic - ia)
                delta[k1] += 1.0
                k2 = n - 1 + (ia - ic)
                delta[k2] += 1.0
                nc = (corr + delta).max()
                if nc < cur_c - 1e-12:
                    h[ia], h[ic] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    no_improve = 0
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve >= perturb_every:
                n_flips = rng.randint(2, 5)
                ones_arr = np.where(h == 1)[0]
                zeros_arr = np.where(h == 0)[0]
                if len(ones_arr) > n_flips and len(zeros_arr) > n_flips:
                    fa = rng.choice(zeros_arr, size=n_flips, replace=False)
                    fb = rng.choice(ones_arr, size=n_flips, replace=False)
                    h[fa] = 1.0
                    h[fb] = 0.0
                    corr = np.correlate(h, 1 - h, mode='full').astype(float)
                    cur_c = corr.max()
                no_improve = 0
            else:
                break

    return best_h, best_c * dx

def subgradient_optimize(h, n, deadline, max_iters=3000):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr_init = 5.0 / np.sqrt(max_iters)
    # Track average of last few subgradients for smoothing
    grad_avg = np.zeros(n)
    smooth_count = 0
    window = 20

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(c)
        k = idx_max - (n - 1)
        m = c[idx_max] * dx
        grad = np.zeros(n)
        if k >= 0:
            if k == 0:
                grad[:] = 0.0
            else:
                v1 = np.arange(0, n - k)
                grad[v1] = 1.0 - h[v1 + k]
                v2 = np.arange(k, n)
                grad[v2] -= h[v2 - k]
        else:
            k_abs = -k
            v1 = np.arange(k_abs, n)
            grad[v1] = 1.0 - h[v1 + k]
            v2 = np.arange(0, n - k_abs)
            grad[v2] -= h[v2 - k]

        grad_avg = grad_avg * (1.0 - 1.0/window) + grad * (1.0/window)
        smooth_count = min(smooth_count + 1, window)
        use_grad = grad_avg

        lr = lr_init * np.sqrt(max_iters) / (it + 1)
        lr = min(lr, 0.5)
        h_new = h - lr * use_grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c

def soft_max_optimize(h, n, deadline, alpha, max_iters=1000):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        amax = alpha * c
        amax_max = amax.max()
        exp_c = np.exp(amax - amax_max)
        w = exp_c / exp_c.sum()
        grad = np.zeros(n)
        n_lags = 2 * n - 1
        for idx in range(n_lags):
            wk = w[idx]
            if wk < 1e-8:
                continue
            k = idx - (n - 1)
            if k == 0:
                continue
            if k >= 0:
                v1 = np.arange(0, n - k)
                grad[v1] += wk * (1.0 - h[v1 + k])
                v2 = np.arange(k, n)
                grad[v2] += wk * (-h[v2 - k])
            else:
                ka = -k
                v1 = np.arange(ka, n)
                grad[v1] += wk * (1.0 - h[v1 + k])
                v2 = np.arange(0, n - ka)
                grad[v2] += wk * (-h[v2 - k])
        lr = 0.3 / (1.0 + 0.01 * it)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c

def fourier_init(n, p, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, p + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = project_feasible(h, n)
    bins = np.linspace(0, 1, n // 2 + 1)
    h_bin = np.digitize(h, bins) > n // 4
    if h_bin.sum() != n // 2:
        h = np.clip(h, 0, 1)
        target = n / 2.0
        diff = target - h.sum()
        if diff > 0:
            idx = np.argsort(1 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                s = min(1 - h[i], diff)
                h[i] += s
                diff -= s
            else:
                s = min(h[i], -diff)
                h[i] -= s
                diff += s
            if abs(diff) < 1e-10:
                break
        h = np.clip(h, 0, 1)
    return h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    # Get initial solution

        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)

    best_h = h0.copy()
    n = n0

    # Phase 1: Thorough binary local search on initial (~30% time)

    hb, cb = binary_local_search(best_h, n, p1, max_sweeps=50000, rng=np.random.RandomState(seed), perturb_every=60)
    if cb < best_c:
        best_c = cb
        best_h = hb

    # Phase 2: Subgradient continuous optimization (~15%)
    p2 = t0 + 0.45 * (budget_s - 5)
    if time.time() < p2:
        hs = best_h.astype(float).copy()
        hs = project_feasible(hs + 0.01 * rng.randn(n), n)
        hs, cs = subgradient_optimize(hs, n, p2, max_iters=5000)
        if cs < best_c:
            best_c = cs
            best_h = hs
        # Also try from the binary solution directly
        hs2 = best_h.astype(float).copy()
        if time.time() < p2 - 10:
            hs2, cs2 = subgradient_optimize(hs2, n, p2 - 5, max_iters=3000)
            if cs2 < best_c:
                best_c = cs2
                best_h = hs2

    # Phase 3: Soft-max optimization (~10%)
    p3 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p3:
        hc = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for alpha in [100.0, 500.0, 2000.0]:
            if time.time() > p3 - 5:
                break
            hc, cc = soft_max_optimize(hc, n, min(p3, time.time() + (p3 - time.time()) / 3), alpha, max_iters=500)
            if cc < best_c:
                best_c = cc
                best_h = hc

    # Phase 4: Binary local search from continuous solution (~20%)
    p4 = t0 + 0.75 * (budget_s - 5)
    # Binarize the continuous solution
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    if h_bin.sum() == n / 2.0:
        hb4, cb4 = binary_local_search(h_bin, n, p4, max_sweeps=50000, rng=np.random.RandomState(seed + 1), perturb_every=50)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4
    # Also try from the best binary
    hb4b, cb4b = binary_local_search(best_h.copy(), n, p4, max_sweeps=50000, rng=np.random.RandomState(seed + 2), perturb_every=40)
    if cb4b < best_c:
        best_c = cb4b
        best_h = hb4b

    # Phase 5: Multi-resolution exploration (~12%)
    p5 = t0 + 0.87 * (budget_s - 5)
    for nh in [200, 300, 320]:
        if time.time() > p5 - 30:

        if nh == n:
            continue

    p6 = dl - 3.0

            hb6 = project_feasible(hb6, n)
            if time.time() < p6 - 2:
                hb6, cb6 = binary_local_search(hb6, n, p6, max_sweeps=20000, rng=np.random.RandomState(seed + 100), perturb_every=30)
                if cb6 < best_c:
                    best_c = cb6
                    best_h = hb6

    best_h = project_feasible(best_h, n)
    assert abs(best_h.sum() - n / 2.0) < 1e-6, f"sum constraint violated: {best_h.sum()} vs {n/2.0}"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10), "box constraint violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
