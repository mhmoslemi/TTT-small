        rng = np.random.RandomState(0)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c = float(corr.max() * dx)
    argmax = int(corr.argmax())
    
    ones = set(np.where(best_h > 0.5)[0].tolist())
    zeros = set(np.where(best_h < 0.5)[0].tolist())
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones_list = list(ones)
        zeros_list = list(zeros)
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        
        for i0 in ones_list:
            if time.time() > deadline:
                break
            for j0 in zeros_list:
                # Compute delta
                s = np.arange(2*n-1) - (n-1)
                delta = np.zeros(2*n-1)
                
                # i0: 1→0
                m1 = (s >= -i0) & (s <= n-1-i0)
                if m1.any():

                    delta[m1] -= (1 - best_h[idx])

                if m2.any():
                    idx = i0 - s[m2]
                    delta[m2] += best_h[idx]
                
                # j0: 0→1
                m3 = (s >= -j0) & (s <= n-1-j0)
                if m3.any():
                    idx = j0 + s[m3]
                    delta[m3] += (1 - best_h[idx])
                m4 = (s >= j0-(n-1)) & (s <= j0)
                if m4.any():
                    idx = j0 - s[m4]
                    delta[m4] -= best_h[idx]
                
                # Check if this reduces the max
                new_corr_at_max = corr[argmax] + delta[argmax]  # approximately
                # But other positions might become the new max
                # For efficiency, just check a few key positions
                if new_corr_at_max < corr[argmax] - 1e-10:
                    # Likely improvement, verify
                    new_corr = corr + delta

def compute_corr(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_beatty(n, alpha_param, n_ones=None):
    h = np.zeros(n)
    if n_ones is None:
        n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        positions.add(pos)
        if len(positions) >= n_ones:
            break
    for pos in positions:
        h[pos] = 1.0
    # Fix any count issues

    if cur < n / 2.0:

        for z in zeros[:int(n/2.0 - cur)]:
            h[z] = 1.0
    elif cur > n / 2.0:
        ones = np.where(h == 1)[0]
        for o in ones[:int(cur - n/2.0)]:
            h[o] = 0.0

def make_thue_morse(n, n_ones=None):
    if n_ones is None:
        n_ones = n // 2
    # Thue-Morse: t(i) = popcount(i) mod 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Select n_ones positions where tm = 1 (or 0, whichever gives better balance)
    ones_pos = np.where(tm == 1)[0]
    zeros_pos = np.where(tm == 0)[0]
    # We need exactly n_ones positions
    if len(ones_pos) >= n_ones:
        selected = ones_pos[:n_ones]
    else:
        selected = np.concatenate([ones_pos, zeros_pos[:n_ones - len(ones_pos)]])
    h = np.zeros(n)
    h[selected] = 1.0
    return project_feasible(h, n)

def make_vander_corput(n, n_ones=None):
    if n_ones is None:
        n_ones = n // 2
    # Van der Corput sequence
    vdc = np.zeros(n)
    for i in range(n):
        x = 0.0
        d = 0.5
        k = i
        while k > 0:
            if k % 2 == 1:
                x += d
            k //= 2
            d /= 2
        vdc[i] = x
    # Sort and take the first n_ones
    order = np.argsort(vdc)
    h = np.zeros(n)
    h[order[:n_ones]] = 1.0
    return project_feasible(h, n)

For each j, the valid m's are:
