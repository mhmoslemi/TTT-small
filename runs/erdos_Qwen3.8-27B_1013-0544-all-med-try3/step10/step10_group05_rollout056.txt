from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix for numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Adjust the largest element
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] -= diff
    return np.clip(h, 0.0, 1.0)

def get_gradient_for_shift(k, h, n):
    """Compute gradient of overlap_k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        # overlap_k = sum_{i=0}^{n-k-1} h[i]*(1-h[i+k]) * dx
        # d/dh[i] = (1-h[i+k])*dx for i < n-k (first term)
        # plus -h[i-k]*dx for i >= k (second term, where j=i-k, h[j]*(1-h[j+k]), d/dh[j+k]=-h[j])
        grad[0:nv] += (1.0 - h[k:n]) * dx
        if k > 0:
            grad[k:n] += -h[0:nv] * dx
    else:
        kk = -k
        nv = n - kk
        # overlap_k = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk]) * dx
        # By symmetry, same formula with kk
        grad[kk:n] += (1.0 - h[0:nv]) * dx

    return grad

    best_c = compute_c5(best_h, n)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmax(h)) if diff < 0 else int(np.argmin(h))
        h[idx] -= diff
    return np.clip(h, 0.0, 1.0)

        h_pert = best_h + rng.uniform(-0.005, 0.005, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.003:
            c_vec = compute_c5_vec(h_pert, n)
            m = c_vec.max()
            active = np.where(c_vec >= m - 0.002 * m)[0]
            if len(active) > 15:
                active = np.array(np.argsort(-c_vec[active])[:15])
            rem = deadline - time.time() - 15
            if rem < 20:
                continue
            maxit = max(100, int(rem // (n_trials - t) * 1.5))
            maxit = min(maxit, 800)
            try:
                res = minimize(act_obj, h_pert, jac=act_grad,
                              args=(n, active, alpha, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n = 500

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)

    # Phase 1: Coordinate descent (fast exact 1D subproblem solving)
    p1_end = min(t0 + 80, deadline - 900)
    if p1_end > t0 + 10:
        h1, c1 = coordinate_descent(best_h, n, p1_end, max_sweeps=200)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: L-BFGS warmup with increasing alpha
    p2_end = min(time.time() + 120, deadline - 700)
    if p2_end > time.time() + 20:
        bounds = [(0.0, 1.0)] * n
        for alpha in [2000, 20000, 200000, 2000000]:
            if time.time() > p2_end - 10:
                break
            c_vec = compute_c5_vec(best_h, n)
            m = c_vec.max()
            active = np.where(c_vec >= m - 0.005 * m)[0]
            if len(active) > 20:
                active = np.array(np.argsort(-c_vec[active])[:20])
            h_new, c_new = run_active_round(best_h, n, p2_end - 8, len(active), alpha, 0.005)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 3: Aggressive active-set with high alpha (main optimization)
    p3_end = min(time.time() + 380, deadline - 450)
    if p3_end > time.time() + 30:
        configs = [
            (6, 50000, 0.002),
            (8, 100000, 0.001),
            (10, 300000, 0.001),
            (10, 1000000, 0.0005),
            (12, 3000000, 0.0005),
            (12, 10000000, 0.0003),
            (15, 30000000, 0.0003),
            (15, 100000000, 0.0002),
            (20, 300000000, 0.0002),
        ]
        for top_k, alpha, mu in configs:
            for _ in range(3):
                if time.time() > p3_end - 12:
                    break
                c = compute_c5_vec(best_h, n)
                m = c.max()
                tol = max(1e-10, 0.0003 * m)
                active = np.where(c >= m - tol)[0]
                if len(active) > top_k:
                    active = np.array(np.argsort(-c[active])[:top_k])
                h_new, c_new = run_active_round(best_h, n, p3_end - 8, len(active), alpha, mu)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                if time.time() > p3_end - 12:
                    break

    # Phase 4: Coordinate descent polish
    p4_end = min(time.time() + 120, deadline - 300)
    if p4_end > time.time() + 15:
        h4, c4 = coordinate_descent(best_h, n, p4_end, max_sweeps=300)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient-directed exchange
    p5_end = min(time.time() + 180, deadline - 200)
    if p5_end > time.time() + 15:
        h5, c5v = grad_directed_exchange(best_h, n, p5_end, rng, max_rounds=400)
        if c5v < best_c:
            best_c = c5v
            best_h = h5

    # Phase 6: Fine pair search
    p6_end = min(time.time() + 180, deadline - 120)
    if p6_end > time.time() + 15:
        h6, c6 = fine_pair_search(best_h, n, p6_end, radius=40,
            deltas=[0.002, 0.001, 0.0005, 0.0002, 0.0001])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Final active-set + perturbation
    p7_end = min(time.time() + 100, deadline - 50)
    if p7_end > time.time() + 15:
        h7, c7 = perturb_reopt(best_h, n, p7_end, rng, n_trials=5, alpha=5000000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Ultra-fine final polish
    p8_end = deadline - 2
    if p8_end > time.time() + 10:
        h8, c8 = fine_pair_search(best_h, n, p8_end, radius=25,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        if c8 < best_c:
            best_c = c8
            best_h = h8
        # One more coord descent pass
        if time.time() < p8_end - 5:
            h9, c9 = coordinate_descent(best_h, n, p8_end, max_sweeps=50)
            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
