def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:
                break

            h += scale * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 300:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain < 1e-14:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-17:
                best_c = c_new
                stall = 0
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
                stall += 1
                if stall > 500:
                    break
    return best_h, best_c

def fine_pair(h, n, deadline, radius=60, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] = min(1.0, best_h[i] + d)
                        best_h[j] = max(0.0, best_h[j] - d)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            for ii in range(n):
                                best_h[ii] = h_original[ii]
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] = max(0.0, best_h[i] - d)
                        best_h[j] = min(1.0, best_h[j] + d)
                        best_h = project_feasible(best_h, n)
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-17:
                            best_c = c
                            improved = True
                        else:
                            best_h = h_original.copy()
        h_original = best_h.copy()
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param) % n
        if pos not in positions:
            positions.add(pos)
            h[pos] = 1.0
            if len(positions) >= n_ones:
                break
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def multi_alpha_lbfgs(h, n, deadline, phase_alphas=None, mu_base=0.01):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if phase_alphas is None:
        phase_alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for i, alpha in enumerate(phase_alphas):
        if time.time() > deadline - 15:
            break
        mu = mu_base * max(0.01, 1.0 - i * 0.1)
        h_new = lbfgs_opt(best_h, n, deadline, sobj, sgrad, (n, alpha, mu),
                        maxit=max(300, int((deadline - time.time()) * 4)))
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
        elif c_new <= best_c + 1e-14:
            best_h = h_new
    return best_h, best_c

def active_set_optimize(h, n, deadline, n_top_list=None, alpha_list=None, rounds=40):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    if n_top_list is None:
        n_top_list = [5, 8, 12, 16, 20, 30, 50]
    if alpha_list is None:
        alpha_list = [100000, 500000, 2000000, 10000000, 50000000]
    for rnd in range(rounds):
        if time.time() > deadline - 20:
            break
        active, mx = identify_active(best_h, n, n_top=n_top, tol_frac=0.0005)
        if len(active) == 0:
            active = np.array([np.argmax(compute_c5_vec(best_h, n))])
        alpha = alpha_list[min(rnd // 2, len(alpha_list) - 1)]
        n_top_cur = n_top_list[min(rnd // 4, len(n_top_list) - 1)]
        active2, _ = identify_active(best_h, n, n_top=n_top_cur, tol_frac=0.001)
        if len(active2) > 0:
            active = active2
        h_new = lbfgs_opt(best_h, n, deadline, active_obj, active_grad,
                        (n, active, alpha, 0.0005),
                        maxit=max(200, int((deadline - time.time()) * 3)))
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-17:
            best_c = c_new
            best_h = h_new
        elif c_new <= best_c + 1e-15:
            best_h = h_new
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=5000000, maxiter=1000):
    def sqf(x):
        return x.sum() - n / 2.0
    def sqj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sqf, 'jac': sqj}]
    rem = max(10, int(deadline - time.time()) - 5)
    maxit2 = min(maxiter, max(100, rem * 5))
    try:
        res = minimize(sobj, h, args=(n, alpha, 0.0), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxit2, 'ftol': 1e-22})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def n_top_global(tol_frac=0.0005):
    return 10

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    best_n = n

    golden = (1.0 + np.sqrt(5.0)) / 2.0
    starts = [
        ('parent', best_h.copy()),
        ('beatty_g', make_beatty_start(n, golden / 2.0)),
        ('beatty_s2', make_beatty_start(n, np.sqrt(2.0) / 2.0)),
        ('beatty_pi', make_beatty_start(n, 2.0 * np.pi / 4.0)),
    ]
    for _ in range(4):
        starts.append(('rand', make_random_binary(n, rng)))

    for name, hs in starts:
        c_s = compute_c5(hs, n)
        if c_s < best_c:
            best_c = c_s
            best_h = hs.copy()

    # Phase 1: L-BFGS warmup (150s)
    p1_end = t0 + 150
    if best_c > 0.40:
        h1, c1 = multi_alpha_lbfgs(best_h, n, p1_end,
            phase_alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu_base=0.05)
    else:
        h1, c1 = multi_alpha_lbfgs(best_h, n, p1_end,
            phase_alphas=[100, 1000, 5000, 50000, 200000, 2000000],
            mu_base=0.01)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set refinement (250s)
    p2_end = t0 + 430
    h2, c2 = active_set_optimize(best_h, n, p2_end,
        n_top_list=[4, 6, 8, 10, 14, 20, 30],
        alpha_list=[50000, 200000, 1000000, 5000000, 20000000, 100000000],
        rounds=50)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Gradient descent steps (150s)
    p3_end = t0 + 590
    h3, c3 = grad_descent_steps(best_h, n, p3_end, n_steps_max=300000)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Active set again (120s)
    p4_end = t0 + 720
    h4, c4 = active_set_optimize(best_h, n, p4_end,
        n_top_list=[4, 6, 8, 12, 16],
        alpha_list=[500000, 2000000, 10000000, 50000000],
        rounds=30)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Fine pair search (180s)
    p5_end = t0 + 910
    h5, c5 = fine_pair(best_h, n, p5_end, radius=50,
        deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Gradient descent very fine (80s)
    p6_end = t0 + 995
    h6, c6 = grad_descent_steps(best_h, n, p6_end, n_steps_max=200000)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: SLSQP polish (50s)
    p7_end = deadline - 10
    if time.time() < p7_end - 20:
        h7, c7 = slqp_polish(best_h, n, p7_end, alpha=10000000, maxiter=500)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final very fine adjustments
    if time.time() < deadline - 5:
        h8, c8 = fine_pair(best_h, n, deadline - 2, radius=30,
            deltas=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
