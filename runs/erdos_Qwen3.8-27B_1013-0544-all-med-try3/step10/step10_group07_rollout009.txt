def swap_delta_fast(h, n, i, j, corr):
    """Compute delta in correlation when swapping h[i] (0) and h[j] (1).
    Returns the full delta array of size 2n-1.
    """
    dx_unused = 1  # We'll multiply by dx later
    # For binary h: h[i]=0, h[j]=1
    # corr[k] = sum_t h[t]*(1-h[t+k])
    # After swap: h[i]→1, h[j]→0
    
    delta = np.zeros(2 * n - 1)
    
    # Contribution from t=i (h[i] changes from 0 to 1):
    # Old: 0 * (1-h[i+k]) = 0
    # New: 1 * (1-h_new[i+k])
    # where h_new[i+k] = 1 if i+k==j else h[i+k]
    im = i  # start index for t=i
    # k ranges so that 0 <= i+k < n, i.e., -i <= k <= n-1-i
    k_lo = -i
    k_hi = n - 1 - i
    if k_lo <= k_hi:
        idx = n - 1 + np.arange(k_lo, k_hi + 1)
        positions = np.arange(im, im + k_hi - k_lo + 1)  # i+k for k in [k_lo, k_hi]
        # 1 - h_new[i+k]: normally 1-h[positions], but at position j it's 1-0=1
        vals = 1.0 - h[positions]
        vals[positions == j] = 1.0  # h_new[j] = 0, so 1-h_new[j] = 1
        delta[idx] += vals
    
    # Contribution from t=j (h[j] changes from 1 to 0):
    # Old: 1 * (1-h[j+k])
    # New: 0 * (1-h_new[j+k]) = 0
    # Delta: -(1-h_old[j+k]) = -(1-h[j+k])
    # BUT if j+k == i: h_old[i] = 0, so 1-h[i] = 1.
    # Actually old value: h[j+k] = h[i] = 0 (when j+k==i), so 1-h[j+k] = 1.
    # New value: 0 (since h_new[j]=0, the whole term is 0).
    # Delta = 0 - 1 = -1. And -(1-h[j+k]) = -(1-h[i]) = -(1-0) = -1. Same!
    # If j+k != i: h[j+k] is unchanged, so Delta = -(1-h[j+k]). Correct!
    jk_lo = -j
    jk_hi = n - 1 - j
    if jk_lo <= jk_hi:
        idx = n - 1 + np.arange(jk_lo, jk_hi + 1)
        positions = np.arange(j, j + jk_hi - jk_lo + 1)  # j+k
        vals = -(1.0 - h[positions])
        delta[idx] += vals
    
    # Contribution from t+k=i (h[t] * (1-h[i]) changes):
    # Old: h[t] * (1-0) = h[t] where t = i-k
    # New: h[t] * (1-1) = 0
    # Delta: -h[t] = -h[i-k]
    # BUT if i-k == j: h[j]=1, Delta = -1. And h_new[j]=0, but t=j so h_new[t]=0.
    #   Old: h[j]*(1-h[i]) = 1*1 = 1. New: h_new[j]*(1-h_new[i]) = 0*0 = 0. Delta = -1.
    #   -h[i-k] = -h[j] = -1. Same!
    # If i-k == i: k=0, h[i]=0, Delta = -0 = 0. Fine.
    ti_lo = i - n + 1
    ti_hi = i
    if ti_lo <= ti_hi:
        idx = n - 1 + np.arange(ti_lo, ti_hi + 1)
        positions = np.arange(i - ti_hi, i - ti_lo + 1)  # i-k for k in [ti_lo, ti_hi]
        # positions = i - k where k goes from ti_lo to ti_hi
        # when k = i: position = 0; when k = i-n+1: position = n-1
        # positions = i - arange(ti_lo, ti_hi+1) = arange(i - ti_hi, i - ti_lo + 1)
        positions = np.arange(i - ti_hi, i - ti_lo + 1)
        vals = -h[positions]
        delta[idx] += vals
    
    # Contribution from t+k=j (h[t] * (1-h[j]) changes):
    # Old: h[t] * (1-1) = 0
    # New: h[t] * (1-0) = h[t]
    # Delta: +h[t] = +h[j-k]
    # BUT if j-k == i: h[i]=0, Delta = 0. And h_new[i]=1.
    #   Old: h[i]*(1-h[j]) = 0*0 = 0. New: h_new[i]*(1-h_new[j]) = 1*1 = 1. Delta = 1.
    #   +h[j-k] = +h[i] = 0. WRONG! Should be +h_new[i] = 1.
    # So we need: +h_new[j-k]
    tj_lo = j - n + 1
    tj_hi = j
    if tj_lo <= tj_hi:
        idx = n - 1 + np.arange(tj_lo, tj_hi + 1)
        positions = np.arange(j - tj_hi, j - tj_lo + 1)  # j-k
        vals = h[positions].copy()
        vals[positions == i] = 1.0  # h_new[i] = 1
        delta[idx] += vals
    
    return delta
