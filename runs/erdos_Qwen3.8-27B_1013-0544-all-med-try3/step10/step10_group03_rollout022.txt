def compute_c5_vec_fast(h, n):

    # c(k) = Σ h[i](1-h[i+k]) for valid i, at each shift k
    # = Σ_{valid i} h[i] - autocorr(h)[k]
    
    # Linear autocorrelation via FFT
    N = 2 * n
    H = np.fft.rfft(h, N)
    autocorr = np.fft.irfft(H * np.conj(H), N)[:n]  # lags 0 to n-1
    
    # c(k) for k >= 0: sum of h[0:n-k] - autocorr[k]
    # c(k) for k < 0 (i.e., c at index (n-1)+k where k<0): 
    #   c(-kk) = sum of h[kk:n] - autocorr[kk]
    
    c_vec = np.zeros(2*n-1)
    cumsum_h = np.concatenate([[0], np.cumsum(h)])
    
    for k in range(n):
        # shift +k (index k in the positive part)
        # c(k) = sum(h[0:n-k]) - autocorr[k]
        # In the full correlate output, this is at index (n-1)+k
        c_vec[(n-1)+k] = (cumsum_h[n-k] - cumsum_h[0]) - autocorr[k]
        
        # shift -k (index in the negative part)
        # c(-k) = sum(h[k:n]) - autocorr[k]
        # In the full correlate output, this is at index (n-1)-k
        if k > 0:
            c_vec[(n-1)-k] = (cumsum_h[n] - cumsum_h[k]) - autocorr[k]
    
    return c_vec * dx

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5_vec_fft(h, n):
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    autocorr = np.fft.irfft(H * np.conj(H), N)[:n]
    cumsum_h = np.concatenate([[0.0], np.cumsum(h)])
    c_vec = np.zeros(2 * n - 1)
    for k in range(n):
        c_vec[(n - 1) + k] = cumsum_h[n - k] - autocorr[k]
        if k > 0:
            c_vec[(n - 1) - k] = cumsum_h[n] - cumsum_h[k] - autocorr[k]
    return c_vec * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_bisect(z, n):
    target = n / 2.0
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c_single(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        grad[:n - k] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:n - k]) * dx
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n - kk]) * dx
        grad[:n - kk] += (-h[kk:]) * dx
    return grad

def c5_at_shifts(h, n, shift_indices):
    dx = 2.0 / n
    vals = np.zeros(len(shift_indices))
    for i, si in enumerate(shift_indices):
        k = si - (n - 1)
        if k >= 0:
            vals[i] = float(np.sum(h[:n - k] * (1.0 - h[k:]))) * dx
        else:
            kk = -k
            vals[i] = float(np.sum(h[kk:] * (1.0 - h[:n - kk]))) * dx
    return vals

def fista_topk(h, n, deadline, top_k=5, alpha=500, block_size=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 1.0 / (4.0 * alpha)
    
    for block in range(200):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        top_idx = np.argsort(-c_vec)[:top_k]
        
        t_f = 1.0
        h_prev = h.copy()
        
        for it in range(block_size):
            if time.time() > deadline:
                return best_h, best_c
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_f * t_f)) / 2.0
            y = h + (t_f - 1.0) / t_new * (h - h_prev)
            y_clip = np.clip(y, 0.0, 1.0)
            
            c_top = c5_at_shifts(y_clip, n, top_idx)
            mx = float(c_top.max())
            sh = alpha * (c_top - mx)
            sh_mx = float(sh.max())
            w = np.exp(sh - sh_mx)
            w_sum = float(w.sum())
            if w_sum < 1e-30:
                w = np.ones(top_k) / top_k
            else:
                w /= w_sum
            
            grad = np.zeros(n)
            for wi, idx in zip(w, top_idx):
                if wi > 1e-12:
                    grad += wi * grad_c_single(y_clip, n, idx)
            
            h_cand = y_clip - step * grad
            h_new = project_bisect(h_cand, n)
            
            h_prev = h
            h = h_new
            t_f = t_new
            
            if (it + 1) % 200 == 0:
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
    
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha, top_idx=None, mu=0.0):
    c = compute_c5_vec(x, n)
    if top_idx is not None:
        vals = c[top_idx]
    else:
        vals = c
    mx = float(vals.max())
    sh = alpha * (vals - mx)
    smx = float(sh.max())
    if smx > 500:
        val = mx + (smx + float(np.log(np.sum(np.exp(sh - smx))))) / alpha
    else:
        val = mx + float(np.log(np.sum(np.exp(sh)))) / alpha
    val += mu * float((x.sum() - n / 2.0) ** 2)
    return float(val)

def smooth_grad(x, n, alpha, top_idx=None, mu=0.0):
    if top_idx is not None:
        c = c5_at_shifts(x, n, top_idx)
        num_shifts = len(top_idx)
    else:
        c = compute_c5_vec(x, n)
        top_idx = np.arange(len(c))
        num_shifts = len(c)
    mx = float(c.max())
    sh = alpha * (c - mx)
    smx = float(sh.max())
    w = np.exp(sh - smx)
    w /= w.sum()
    grad_o = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        if wi > 1e-14:
            grad_o += wi * grad_c_single(x, n, idx)
    if mu > 0:
        grad_o += 2.0 * mu * (x.sum() - n / 2.0)
    return grad_o

def run_lbfgs(h, n, deadline, configs=None, top_k=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:

                   (20000, 2000.0), (100000, 10000.0)]
    for alpha, mu in configs:

            break
        ti = None
        if top_k is not None:

            ti = np.argsort(-cv)[:top_k]
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, ti, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:

                h = h_new
        except Exception:
            pass

def lp_step(h0, n, active_shifts, deadline):
    dx = 2.0 / n
    c_vec = compute_c5_vec(h0, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for si in active_shifts:
        grad = grad_c_single(h0, n, si)
        row = np.zeros(n + 1)
        row[:n] = grad
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(float(c_vec[si] - grad @ h0))
    
    A_ub_arr = np.array(A_ub_list) if A_ub_list else None
    b_ub_arr = np.array(b_ub_list) if b_ub_list else None
    
    bounds_lp = [(0.0, 1.0)] * n + [(-1.0, 5.0)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ub_arr, b_ub=b_ub_arr,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                      method='highs')
        if res.success:
            h_new = project_feasible(res.x[:n], n)
            c_new = compute_c5(h_new, n)
            c_old = compute_c5(h0, n)
            return h_new, c_new
    except Exception:
        pass
    return h0, compute_c5(h0, n)

def pair_move_search(h, n, deadline, deltas=None):
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    radius = max(10, n // 10)
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
