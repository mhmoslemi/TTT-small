            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-8, 0.002 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        
        for alpha in [50000, 500000, 5000000, 50000000]:
            if time.time() > deadline - 2:
                break
            try:
                res = minimize(top_shifts_obj, best_h, jac=top_shifts_grad,
                              args=(n, active, alpha, 0.01), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxfun': 50000,
                                       'maxls': 100})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def peeling_local_search(h, n, deadline):
    """Targeted local search that peels off the max one shift at a time."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for iteration in range(500):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Find top shifts
        top_k = min(10, 2 * n - 1)
        top_idx = np.argsort(-c_vec)[:top_k]
        threshold = max_val - max(1e-8, 0.001 * max_val)
        active = top_idx[c_vec[top_idx] >= threshold]
        
        if len(active) == 0:
            active = top_idx[:3]
        
        # Compute combined gradient
        grad = np.zeros(n)
        for idx in active:
            grad += shift_grad_single(best_h, n, idx)
        grad /= len(active)
        
        # Project gradient to maintain sum constraint
        grad_mean = grad.mean()
        grad_proj = grad - grad_mean
        
        # Try multiple step sizes
        best_improvement = 0
        best_step = None
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = best_h - step * grad_proj
            h_trial = project_feasible(h_trial, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-15:
                improvement = best_c - c_trial
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_step = (step, h_trial)
        
        if best_step is not None:
            best_c = compute_c5(best_step[1], n)
            best_h = best_step[1]
        else:
            # Try pairwise moves based on gradient
            negative_indices = np.where(grad_proj < -0.01)[0]
            positive_indices = np.where(grad_proj > 0.01)[0]
            
            found = False
            for i in negative_indices[:30]:
                if time.time() > deadline:
                    return best_h, best_c
                for j in positive_indices[:30]:
                    if i == j:
                        continue
                    for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                        if best_h[i] - step >= 0 and best_h[j] + step <= 1:
                            best_h[i] -= step
                            best_h[j] += step
                            c_trial = compute_c5(best_h, n)
                            if c_trial < best_c - 1e-15:
                                best_c = c_trial
                                found = True
                                break
                            else:
                                best_h[i] += step
                                best_h[j] -= step
                        elif best_h[i] + step <= 1 and best_h[j] - step >= 0:
                            best_h[i] += step
                            best_h[j] -= step
                            c_trial = compute_c5(best_h, n)
                            if c_trial < best_c - 1e-15:
                                best_c = c_trial
                                found = True
                                break
                            else:
                                best_h[i] -= step
                                best_h[j] += step
                    if found:
                        best_h = project_feasible(best_h, n)
                        break
                if found:
                    break
            if not found:
                # Try random perturbations
                rng = np.random.RandomState(iteration)
                for _ in range(20):
                    if time.time() > deadline:
                        return best_h, best_c
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    for step in [0.01, 0.005, 0.002]:
                        if best_h[i] - step >= 0 and best_h[j] + step <= 1:
                            best_h[i] -= step
                            best_h[j] += step
                            c_trial = compute_c5(best_h, n)
                            if c_trial < best_c - 1e-15:
                                best_c = c_trial
                                found = True
                                break
                            else:
                                best_h[i] += step
                                best_h[j] -= step
                        elif best_h[i] + step <= 1 and best_h[j] - step >= 0:
                            best_h[i] += step
                            best_h[j] -= step
                            c_trial = compute_c5(best_h, n)
                            if c_trial < best_c - 1e-15:
                                best_c = c_trial
                                found = True
                                break
                            else:
                                best_h[i] -= step
                                best_h[j] += step
                    if found:
                        best_h = project_feasible(best_h, n)
                        break
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, nnodes=200):
    """Fine local search over a subset of indices."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Focus on indices that are in transition regions
    grad_total = np.zeros(n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    threshold = max_val - max(1e-6, 0.01 * max_val)
    active = np.where(c_vec >= threshold)[0]
    for idx in active[:10]:
        grad_total += shift_grad_single(best_h, n, idx)
    
    # Select indices with highest |gradient|
    idx_candidates = np.argsort(-np.abs(grad_total))[:nnodes]
    
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for ii in range(len(idx_candidates)):
                if time.time() > deadline:
                    return best_h, best_c
                i = idx_candidates[ii]
                for jj in range(len(idx_candidates)):
                    if ii == jj:
                        continue
                    j = idx_candidates[jj]
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def random_restart_search(h, n, deadline, n_restarts=50):
    """Try random perturbations around the current best."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    bounds = [(0.0, 1.0)] * n
    
    for r in range(n_restarts):
        if time.time() > deadline - 10:
            break
        # Create perturbed version
        h_pert = best_h.copy()
        n_changes = rng.randint(3, 15)
        indices = rng.choice(n, n_changes, replace=False)
        for idx in indices:
            delta = rng.uniform(-0.05, 0.05)
            h_pert[idx] += delta
        h_pert = project_feasible(h_pert, n)
        
        # Quick optimization from perturbed point
        t_sub = min(15, deadline - time.time() - 10)
        if t_sub < 5:
            continue
        try:
            res = minimize(smooth_obj, h_pert, jac=smooth_grad,
                          args=(n, 500000, 0.01), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-18,
                                   'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n0 = 200
    
    best_h = project_feasible(h_init, n0)
    best_c = compute_c5(best_h, n0)
    
    # Phase 1: Quick optimization at original resolution
    if time.time() < deadline - 900:
        t1 = min(time.time() + 80, deadline - 900)
        best_h, best_c = optimize_lbfgs(best_h, n0, t1,
            alphas=[500, 5000, 50000, 200000, 1000000])
    
    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n0)
    x_new = np.linspace(0, 2, n)
    h = np.interp(x_new, x_old, best_h)
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    best_h_n = h.copy()
    best_c_n = c
    
    # Phase 3: Aggressive L-BFGS at n=1000
    if time.time() < deadline - 700:
        t3 = min(time.time() + 250, deadline - 700)
        best_h_n, best_c_n = optimize_lbfgs(best_h_n, n, t3,
            alphas=[1000, 10000, 100000, 500000, 2000000, 10000000, 50000000])
    
    # Phase 4: Top-shifts focused optimization
    if time.time() < deadline - 400:
        t4 = min(time.time() + 200, deadline - 400)
        h4, c4 = optimize_top_shifts(best_h_n, n, t4, top_k=5)
        if c4 < best_c_n:
            best_c_n = c4
            best_h_n = h4
    
    # Phase 5: Peeling local search
    if time.time() < deadline - 200:
        t5 = min(time.time() + 120, deadline - 200)
        h5, c5 = peeling_local_search(best_h_n, n, t5)
        if c5 < best_c_n:
            best_c_n = c5
            best_h_n = h5
    
    # Phase 6: Fine pair search
    if time.time() < deadline - 80:
        t6 = min(time.time() + 60, deadline - 80)
        h6, c6 = fine_pair_search(best_h_n, n, t6, nnodes=150)
        if c6 < best_c_n:
            best_c_n = c6
            best_h_n = h6
    
    # Phase 7: Another top-shifts round with higher alpha
    if time.time() < deadline - 60:
        t7 = min(time.time() + 40, deadline - 60)
        h7, c7 = optimize_top_shifts(best_h_n, n, t7, top_k=3)
        if c7 < best_c_n:
            best_c_n = c7
            best_h_n = h7
    
    # Phase 8: Try n=2000 if time allows
    if time.time() < deadline - 100:
        n2 = 2000
        x_old2 = np.linspace(0, 2, n)
        x_new2 = np.linspace(0, 2, n2)
        h2 = np.interp(x_new2, x_old2, best_h_n)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)
        
        if time.time() < deadline - 50:
            h2, c2 = optimize_lbfgs(h2, n2, min(time.time() + 60, deadline - 30),
                alphas=[10000, 100000, 1000000, 10000000])
        
        if c2 < best_c_n:
            # Downsample back to n=1000 for return
            h2_down = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, n2), h2)
            h2_down = project_feasible(h2_down, n)
            c2_down = compute_c5(h2_down, n)
            if c2_down < best_c_n:
                best_c_n = c2_down
                best_h_n = h2_down
    
    # Phase 9: Final fine search
    if time.time() < deadline - 10:
        h9, c9 = fine_pair_search(best_h_n, n, deadline - 5, nnodes=100)
        if c9 < best_c_n:
            best_c_n = c9
            best_h_n = h9
    
    best_h = project_feasible(best_h_n, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
