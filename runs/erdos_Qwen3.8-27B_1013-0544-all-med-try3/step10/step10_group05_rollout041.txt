```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            if s < 1e-15:
                break
            h *= (target / s)
            h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_val(vals, alpha):
    m = vals.max()
    z = alpha * (vals - m)
    zmax = z.max()
    if zmax > 500:
        return float((m + zmax / alpha) + np.log(np.sum(np.exp(z - zmax))) / alpha)
    return float(m + np.log(np.sum(np.exp(z))) / alpha)

def lsmax_grad(x, n, active, alpha, mu):
    dx = 2.0 / n
    c = compute_c5_vec(x, n)
    vals = c[active]
    a = alpha * vals
    amax = a.max()
    w = np.exp(a - amax)
    w /= w.sum()
    center = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-16:
            continue
        k = int(idx) - center
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lsmax_val(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return logsumexp_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def optimize_lsmax(h, n, active, alpha, mu, maxit, bounds):
    res = minimize(lsmax_val, h, jac=lsmax_grad, method='L-BFGS-B',
                   args=(n, active, alpha, mu), bounds=bounds,
                   options={'maxiter': maxit, 'ftol': 1e-25,
                            'gtol': 1e-18, 'maxfun': maxit * 10,
                            'maxls': 50, 'maxcor': 50})
    return project_feasible(res.x, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def get_active(c_vec, tol_frac):
    max_val = c_vec.max()
    return np.where(c_vec >= max_val - tol_frac * max_val)[0]

def pair_descent(h, n, deadline, deltas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for delta in deltas:
        improved = True
        passes = 0
        while improved and time.time() < deadline - 5 and passes < 200:
            improved = False
            passes += 1
            c_vec = compute_c5_vec(best_h, n)
            c_max = c_vec.max()
            tol = max(1e-8, 0.002 * c_max)
            active = np.where(c_vec >= c_max - tol)[0]
            for idx in active[:80]:
                if time.time() > deadline - 5:
                    break
                k = int(idx) - (n - 1)
                if k == 0:
                    continue
                if k > 0:
                    stride = max(1, (n - k) // 300)
                    for i in range(0, n - k, stride):
                        if best_h[i] > delta and best_h[i + k] < 1.0 - delta:
                            best_h[i] -= delta
                            best_h[i + k] += delta
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                best_h[i] += delta
                                best_h[i + k] -= delta
                else:
                    kk = -k
                    stride = max(1, (n - kk) // 300)
                    for i in range(0, n - kk, stride):
                        if best_h[i + kk] > delta and best_h[i] < 1.0 - delta:
                            best_h[i + kk] -= delta
                            best_h[i] += delta
                            c_new = compute_c5(best_h, n)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                best_h[i + kk] += delta
                                best_h[i] -= delta
    return best_h, best_c

def full_pair_search(h, n, deadline, deltas, radius=80):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for delta in deltas:
        improved = True
        while improved and time.time() < deadline - 5:
            improved = False
            for i in range(n):
                if time.time() > deadline - 5:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] >= delta and best_h[j] <= 1.0 - delta:
                        best_h[i] -= delta
                        best_h[j] += delta
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += delta
                            best_h[j] -= delta
    return best_h, best_c

def perturb_reopt(h, n, deadline, rng, bounds, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        scale = 0.02 * (1.0 + 0.5 * t)
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.003:
            c_vec = compute_c5_vec(h_pert, n)
            active = get_active(c_vec, 0.005)
            try:
                h_new = optimize_lsmax(h_pert, n, active, 500000, 0.005, 600, bounds)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c
```
