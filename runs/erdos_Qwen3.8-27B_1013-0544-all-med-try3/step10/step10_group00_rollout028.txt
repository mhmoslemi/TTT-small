    """Compute gradient of weighted sum of overlaps w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(w)):

    c = compute_c5_vec(h, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n/2.0)**2

def swap_search(h, n, deadline, max_iters=500000, rng=None, alpha=100000.0):
    """Gradient-guided swap search for binary-like solutions"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = compute_c5(h, n)
    
    # Identify "ones" and "zeros" (for binary-like solutions)
    threshold = 0.5
    ones = set(np.where(h >= threshold)[0].tolist())
    zeros = set(np.where(h < threshold)[0].tolist())
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(h, n)
        w = smooth_max_w(c_vec, alpha)
        grad = h_grad(w, h, n)
        
        # For binary solution, the best move is to swap a 1 where grad is high with a 0 where grad is low
        best_gain = 0
        best_i = -1
        best_j = -1
        
        # Get top candidates
        one_grads = [(grad[i], i) for i in ones if grad[i] > 0]
        zero_grads = [(grad[j], j) for j in zeros if grad[j] < 0]
        
        if not one_grads or not zero_grads:
            # Try random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = list(ones)[rng.randint(len(ones))]
                j = list(zeros)[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h, n)
                if c > best_c:
                    h[i], h[j] = h[j], h[i]
            continue
        
        one_grads.sort(reverse=True)
        zero_grads.sort()
        
        for gi, i in one_grads[:20]:
            if best_gain > 0:
                break
            for gj, j in zero_grads[:20]:
                gain = gi - gj
                if gain > best_gain:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_gain = gain
                        best_i = i
                        best_j = j
                        ones.remove(i)
                        ones.add(j)
                        zeros.remove(j)
                        zeros.add(i)
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    val = m + np.log(np.sum(np.exp(sh))) / alpha
    return val

    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()

            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:] += wm * (1.0 - h[:nv])
                grad[:nv] += wm * (-h[kk:])
    return grad

def lbfgs_make_obj(n, alpha, mu):
    def obj(x):
        h = np.clip(x, 0, 1)
        c = compute_c5_vec(h, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val
    return obj

def lbfgs_make_grad(n, alpha, mu):
    def grad(x):
        h = np.clip(x, 0, 1)
        c = compute_c5_vec(h, n)
        w = smooth_max_w(c, alpha)
        g = h_grad(w, h, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        obj = lbfgs_make_obj(n, alpha, mu)
        grad = lbfgs_make_grad(n, alpha, mu)
        res = minimize(obj, h, jac=grad,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 12,
                               'maxls': 50, 'maxcor': 40})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        c_old = compute_c5(h, n)
        if c_new < c_old:
            return h_new, c_new
        return h, c_old
    except Exception:
        return h, compute_c5(h, n)

def binarize_fix(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:int(np.ceil(diff))]:
            if zeros[idx] is not None:
                h_bin[zeros[idx]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:int(np.ceil(-diff))]:
            if ones[idx] is not None:
                h_bin[ones[idx]] = 0.0
    return h_bin

def swap_search(h, n, deadline, max_iters=500000, rng=None, alpha=100000.0):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = compute_c5(h, n)
    ones_list = np.where(h >= 0.5)[0].tolist()
    zeros_list = np.where(h < 0.5)[0].tolist()
    best_c_val = best_c
    stuck_count = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(h, n)
        w = smooth_max_w(c_vec, alpha)
        grad = h_grad(w, h, n)

        one_candidates = [(grad[i], i) for i in ones_list if grad[i] > 1e-10]
        zero_candidates = [(grad[j], j) for j in zeros_list if grad[j] < -1e-10]

        found = False
        if one_candidates and zero_candidates:
            one_candidates.sort(reverse=True)
            zero_candidates.sort()
            top_k = min(30, len(one_candidates))
            top_m = min(30, len(zero_candidates))
            for gi, i in one_candidates[:top_k]:
                if found:
                    break
                for gj, j in zero_candidates[:top_m]:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c_val - 1e-14:
                        best_c_val = c
                        ones_list.remove(i)
                        ones_list.append(j)
                        zeros_list.remove(j)
                        zeros_list.append(i)
                        found = True
                        stuck_count = 0
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0

        if not found:
            stuck_count += 1
            if stuck_count % 20 == 0:
                for _ in range(5):
                    if time.time() > deadline:
                        break
                    if len(ones_list) == 0 or len(zeros_list) == 0:
                        break
                    i = ones_list[rng.randint(len(ones_list))]
                    j = zeros_list[rng.randint(len(zeros_list))]
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c_val - 1e-14:
                        best_c_val = c
                        ones_list.remove(i)
                        ones_list.append(j)
                        zeros_list.remove(j)
                        zeros_list.append(i)
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                stuck_count = 0
        else:
            stuck_count = 0

    return h, best_c_val

def fine_coord_descent(h, n, deadline, rng):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)

    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
             0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001]

    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0
        while improved_any and rounds < 80 and time.time() < deadline:
            improved_any = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            w = smooth_max_w(c_vec, 200000.0)
            grad = h_grad(w, best_h, n)

            pos_mask = (grad > 1e-12) & (best_h > 1e-10)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-10)

            if not pos_mask.any() or not neg_mask.any():
                break

            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]

            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:35]:
                for j in neg_idx[:35]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j

            if best_i < 0:
                break

            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-13:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)

    return best_h, best_c

def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def make_beatty(n, alpha_val):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 6 * n):
        pos = int(k * alpha_val) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] = min(diff, 1.0)
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] = max(diff, -1.0)
    return project_feasible(h, n)

def perturbed_start(h_base, n, rng, num_perturb=20):
    h = h_base.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(num_perturb):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Phase 1: L-BFGS at parent resolution (70s)
    p1_end = t0 + 70
    for alpha in [5000, 30000, 100000]:
        if time.time() > p1_end - 5:
            break
        sub_dl = min(time.time() + 22, p1_end)
        h1, c1 = run_lbfgs(best_h, best_n, sub_dl, alpha=alpha, mu=0.01, maxit=1500)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()

    # Phase 2: Binarize + swap search at parent resolution (130s)
    p2_end = t0 + 200
    h_bin = binarize_fix(best_h, best_n)
    h2, c2 = swap_search(h_bin, best_n, p2_end, max_iters=300000, rng=rng, alpha=100000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Try Beatty starts and pick best (40s)
    p3_end = t0 + 240
    beatty_alphas = [0.618033988749895, 0.763943322418861, 0.555555555555556,
                     0.701456979730288, 0.652703644666149]
    for ba in beatty_alphas:
        if time.time() > p3_end - 5:
            break
        h_b = make_beatty(best_n, ba)
        h_b, c_b = run_lbfgs(h_b, best_n, min(time.time() + 8, p3_end), alpha=50000, mu=0.01, maxit=500)
        if c_b < best_c:
            best_c = c_b
            best_h = h_b.copy()

    # Phase 4: Upsample and optimize at higher resolution (350s)
    p4_end = t0 + 590
    n_high = best_n * 2
    if n_high > 2000:
        n_high = 2000
    if n_high <= 1000:
        n_high = 1000

    h_high = upsample(best_h, best_n, n_high)
    c_high = compute_c5(h_high, n_high)

    for alpha in [10000, 50000, 150000, 500000]:
        if time.time() > p4_end - 20:
            break
        sub_dl = min(time.time() + 40, p4_end)
        h_high, c_high = run_lbfgs(h_high, n_high, sub_dl, alpha=alpha, mu=0.005, maxit=1500)

    # Phase 5: Swap search at higher resolution (180s)
    p5_end = t0 + 770
    h_high_bin = binarize_fix(h_high, n_high)
    h5, c5 = swap_search(h_high_bin, n_high, p5_end, max_iters=150000, rng=rng, alpha=100000)
    if c5 < c_high:
        c_high = c5
        h_high = h5.copy()

    # Phase 6: Fine coordinate descent at higher resolution (120s)
    p6_end = t0 + 890
    if time.time() < p6_end:
        h6, c6 = fine_coord_descent(h_high, n_high, p6_end, rng)
        if c6 < c_high:
            c_high = c6
            h_high = h6.copy()

    # Phase 7: Final L-BFGS at higher resolution (60s)
    p7_end = t0 + 950
    if time.time() < p7_end:
        for alpha in [200000, 1000000]:
            if time.time() > p7_end - 10:
                break
            h7, c7 = run_lbfgs(h_high, n_high, min(time.time() + 25, p7_end), alpha=alpha, mu=0.001, maxit=800)
            if c7 < c_high:
                c_high = c7
                h_high = h7.copy()

    # Phase 8: Final fine tune (40s)
    p8_end = deadline - 5
    if time.time() < p8_end:
        h8, c8 = fine_coord_descent(h_high, n_high, p8_end, rng)
        if c8 < c_high:
            c_high = c8
            h_high = h8.copy()

    # Decide which to return: best at original n or best at higher n
    # Compare on the same basis: downsample higher to original n
    if c_high < best_c:
        best_h = h_high
        best_c = c_high
        best_n = n_high
    else:
        # Try one more round of swap search at original n with remaining time
        if time.time() < deadline - 10:
            h_final_bin = binarize_fix(best_h, best_n)
            h_fin, c_fin = swap_search(h_final_bin, best_n, deadline - 5, max_iters=100000, rng=rng, alpha=200000)
            if c_fin < best_c:
                best_c = c_fin
                best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
