                    if time.time() > deadline:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if i == j:
                        continue
                    current_h[i] = 0.0
                    current_h[j] = 1.0
                    new_c = compute_c5(current_h, n)
                    if new_c < current_c - 1e-15:
                        current_c = new_c
                        if new_c < best_c:
                            best_c = new_c
                            best_h = current_h.copy()
                        no_improve_count = 0
                        found_improvement = True
                        break
                    else:
                        current_h[i] = 1.0
                        current_h[j] = 0.0
                
                if found_improvement:
                    continue
            
            # Diversification: random perturbation
            if no_improve_count > max_no_improve:
                n_perturb = rng.randint(2, 6)
                ones = np.where(current_h > 0.5)[0]
                zeros = np.where(current_h < 0.5)[0]
                for _ in range(n_perturb):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    current_h[i] = 0.0
                    current_h[j] = 1.0
                    ones = np.where(current_h > 0.5)[0]
                    zeros = np.where(current_h < 0.5)[0]
                new_c = compute_c5(current_h, n)
                if new_c < best_c:
                    best_c = new_c
                    best_h = current_h.copy()
                    current_c = new_c
                no_improve_count = 0
        
        if not tried_any and no_improve_count == 0:
            no_improve_count = 1
        
        # Early termination if stuck for too long
        if no_improve_count > 5000:
            break
    
    return best_h, best_c

def simulated_annealing_binary(h, n, deadline, rng, t_start=0.001, t_end=0.000001, max_iter=200000):
    """Simulated annealing on binary sequences."""
    current_h = h.copy()
    current_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = current_c
    
    ones = np.where(current_h > 0.5)[0].tolist()
    zeros = np.where(current_h < 0.5)[0].tolist()
    
    n_total = max_iter
    for it in range(n_total):
        if time.time() > deadline:
            break
        
        frac = it / max(n_total, 1)
        T = t_start * (t_end / t_start) ** frac
        
        if not ones or not zeros:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        current_h[i] = 0.0
        current_h[j] = 1.0
        ones[ones.index(i)] = j
        zeros[zeros.index(j)] = i
        
        new_c = compute_c5(current_h, n)
        delta = new_c - current_c
        
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = current_h.copy()
        else:
            current_h[i] = 1.0
            current_h[j] = 0.0
            ones[ones.index(j)] = i
            zeros[zeros.index(i)] = j
    
    return best_h, best_c

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def cont_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def cont_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def continuous_refine(h, n, deadline, alphas=None, maxit=None):
    """Continuous L-BFGS refinement."""
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxit is not None and i < len(maxit):
            mi = maxit[i]
        else:
            mi = max(300, int(remaining * 3))
        
        mu = 0.01 if i < len(alphas) // 2 else 0.001
        try:
            res = minimize(cont_obj, best_h, jac=cont_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': mi, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': mi * 8,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def fine_pair_adjust(h, n, deadline, radius=30, deltas=None):
    """Fine-tune continuous values using pair adjustments."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try increasing i, decreasing j
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Try decreasing i, increasing j
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def two_round_binary(h, n, deadline, rng):
    """Two rounds of binary local search with diversification in between."""
    h1, c1 = binary_local_search(h, n, min(time.time() + 120, deadline - 100), rng)
    
    if time.time() < deadline - 80:
        # Perturb and search again
        perturbed = h1.copy()
        ones = np.where(perturbed > 0.5)[0]
        zeros = np.where(perturbed < 0.5)[0]
        n_pert = min(5, len(ones) // 4)
        for _ in range(n_pert):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            perturbed[i] = 0.0
            perturbed[j] = 1.0
            ones = np.where(perturbed > 0.5)[0]
            zeros = np.where(perturbed < 0.5)[0]
        
        h2, c2 = binary_local_search(perturbed, n, deadline - 60, rng)
        if c2 < c1:
            return h2, c2
    
    return h1, c1

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 500
        h0 = np.full(n, 0.5)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Generate binary starting points
    starts = []
    
    # Start 1: round initial to binary
    starts.append(('init_binary', to_binary(h0, n)))
    
    # Start 2: Beatty with golden ratio
    golden = (1.0 + sqrt(5.0)) / 2.0
    starts.append(('beatty_golden', make_beatty_start(n, golden / 2.0)))
    
    # Start 3: Beatty with sqrt(2)
    starts.append(('beatty_sqrt2', make_beatty_start(n, sqrt(2.0) / 2.0)))
    
    # Start 4: Beatty with 1/phi^2
    starts.append(('beatty_phi2', make_beatty_start(n, 1.0 / (golden * golden))))
    
    # Start 5: Random
    h_rand = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h_rand[idx] = 1.0
    starts.append(('random', h_rand))
    
    # Evaluate all starts
    for name, hs in starts:
        c_s = compute_c5(hs, n)
        if c_s < best_c:
            best_c = c_s
            best_h = hs.copy()
    
    # Phase 1: Binary local search on best start (most of the time)
    p1_deadline = t0 + 400
    h_b, c_b = binary_local_search(best_h, n, p1_deadline, rng)
    if c_b < best_c:
        best_c = c_b
        best_h = h_b
    
    # Phase 2: Second round of binary search with perturbation
    if time.time() < deadline - 300:
        p2_deadline = t0 + 700
        perturbed = best_h.copy()
        ones = np.where(perturbed > 0.5)[0]
        zeros = np.where(perturbed < 0.5)[0]
        n_pert = min(3, len(ones) // 5)
        for _ in range(n_pert):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            perturbed[i] = 0.0
            perturbed[j] = 1.0
            ones = np.where(perturbed > 0.5)[0]
            zeros = np.where(perturbed < 0.5)[0]
        
        h_pc, c_pc = binary_local_search(perturbed, n, p2_deadline, rng)
        if c_pc < best_c:
            best_c = c_pc
            best_h = h_pc
    
    # Phase 3: Simulated annealing
    if time.time() < deadline - 250:
        p3_deadline = t0 + 850
        h_sa, c_sa = simulated_annealing_binary(best_h, n, p3_deadline, rng,
            t_start=0.0005, t_end=0.0000001, max_iter=150000)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 4: Another round of binary local search
    if time.time() < deadline - 150:
        p4_deadline = t0 + 950
        h_b2, c_b2 = binary_local_search(best_h, n, p4_deadline, rng)
        if c_b2 < best_c:
            best_c = c_b2
            best_h = h_b2
    
    # Phase 5: Continuous refinement
    if time.time() < deadline - 80:
        p5_deadline = t0 + budget_s - 60
        h_cont, c_cont = continuous_refine(best_h, n, p5_deadline,
            alphas=[200, 1000, 5000, 20000, 100000, 500000])
        if c_cont < best_c:
            best_c = c_cont
            best_h = h_cont
    
    # Phase 6: Fine pair adjustment
    if time.time() < deadline - 30:
        p6_deadline = deadline - 5
        h_fine, c_fine = fine_pair_adjust(best_h, n, p6_deadline,
            radius=20, deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00001])
        if c_fine < best_c:
            best_c = c_fine
            best_h = h_fine
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
