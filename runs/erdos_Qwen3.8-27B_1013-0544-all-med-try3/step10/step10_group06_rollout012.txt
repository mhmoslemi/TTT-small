            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_logsumexp(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    wc = smooth_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if wc[k] < 1e-10:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += wc[k] * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:n-L] += wc[k] * (1.0 - h[L:])
            grad[L:] += wc[k] * (-h[:n-L])
        else:
            M = -L
            grad[M:] += wc[k] * (1.0 - h[:n-M])
            grad[:n-M] += wc[k] * (-h[M:])
    return grad * dx

def grad_top_k(h, n, topK=5):
    """Subgradient using top K lags with equal weight."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    idx = np.argsort(c)[-topK:]
    w_val = 1.0 / topK
    grad = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        if L == 0:
            grad += w_val * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:n-L] += w_val * (1.0 - h[L:])
            grad[L:] += w_val * (-h[:n-L])
        else:
            M = -L
            grad[M:] += w_val * (1.0 - h[:n-M])
            grad[:n-M] += w_val * (-h[M:])
    return grad * dx

def adam_optimize(h, n, alpha, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, max_steps=50000):
    """Adam optimizer with projection."""
    h = h.copy().astype(float)
    n_params = n
    m = np.zeros(n_params)
    v = np.zeros(n_params)
    step_count = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        step_count += 1
        grad = grad_smooth_max(h, n, alpha)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)
        
        lr_t = lr * np.sqrt(1 - beta2 ** step_count) / (1 - beta1 ** step_count)
        actual_lr = lr_t * np.sqrt(1 - beta2 ** step_count) / (1 - beta1 ** step_count)
        # Simplified: just use lr / (sqrt(v_hat) + eps)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h + step
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate: reduce if no improvement for a while
        # (handled by outer loop restarts)
        
        if step_count % 500 == 0:
            c_check = compute_c5(h, n)
            if c_check < best_c:
                best_c = c_check
                best_h = h.copy()
    
    return best_h, best_c

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full')
        obj_val = smooth_logsumexp(c, alpha)
        diff = x.sum() - target_sum
        penalty = 1e8 * diff * diff
        obj_val += penalty
        grad = grad_smooth_max(x, n, alpha)
        grad += 2.0 * 1e8 * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def targeted_local_search(h, n, deadline, rng, topK=5, max_rounds=50000):
    """Direct local search targeting the worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rounds = 0
    
    while time.time() < deadline and rounds < max_rounds:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-topK:]
        k_star = top_idx[rng.randint(topK)]
        L = k_star - (n - 1)
        if L == 0:
            rounds += 1
            continue
        M = abs(L)
        if M >= n:
            rounds += 1
            continue
        
        # For lag L > 0: c = sum_{i=0}^{n-1-M} h[i](1-h[i+M])
        # To reduce: decrease h[i] where h[i+M] is high, or increase h[i+M] where h[i] is high
        # But need to maintain sum constraint: transfer from i to j (or vice versa)
        
        # Find the pair (i, i+M) with maximum contribution to this lag
        if L > 0:
            valid_i = np.arange(n - M)
            contributions = h[valid_i] * (1 - h[valid_i + M])
            # Also consider the reverse: (1-h[i]) * h[i+M] contributes to lag -M
            # For lag M, we want to reduce h[i]*(1-h[i+M])
            # Best move: find i where h[i] is high and h[i+M] is low, then
            # transfer mass from i to some j where h[j] is low and h[j+M] is high
            
            # Simple approach: find max contribution pair
            i_max = valid_i[np.argmax(contributions)]
            h_val = h[i_max]
            h_partner = h[i_max + M]
            
            if h_val > 0.1 and h_partner < 0.9:
                # Transfer mass from i_max to i_max + M
                d = min(0.05, h_val * 0.3)
                h[i_max] -= d
                h[i_max + M] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-12:
                    best_c = c_test
                    best_h = h.copy()
                    rounds += 1
                else:
                    h[i_max] += d
                    h[i_max + M] -= d
                    h = project_feasible(h, n)
            else:
                rounds += 1
        else:
            # L < 0, M = -L
            # c at lag -M = sum_{i=M}^{n-1} h[i](1-h[i-M])
            valid_i = np.arange(M, n)
            contributions = h[valid_i] * (1 - h[valid_i - M])
            i_max = valid_i[np.argmax(contributions)]
            h_val = h[i_max]
            h_partner = h[i_max - M]
            
            if h_val > 0.1 and h_partner < 0.9:
                d = min(0.05, h_val * 0.3)
                h[i_max] -= d
                h[i_max - M] += d
                h = project_feasible(h, n)
                c_test = compute_c5(h, n)
                if c_test < best_c - 1e-12:
                    best_c = c_test
                    best_h = h.copy()
                    rounds += 1
                else:
                    h[i_max] += d
                    h[i_max - M] -= d
                    h = project_feasible(h, n)
            else:
                rounds += 1
    
    return best_h, best_c

def pair_local_search(h, n, deadline, rng, topK=5, step_range=(0.01, 0.08)):
    """More general pair-based local search."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnation = 0
    
    while time.time() < deadline:
        if stagnation > 5000:
            break

        top_idx = np.argsort(c_all)[-topK:]
        k_star = top_idx[rng.randint(topK)]
        L = k_star - (n - 1)
        if L == 0:
            stagnation += 1
            continue
        M = abs(L)
        if M >= n:
            stagnation += 1
            continue
        
        if M < 3 or M > n // 2:
            # Skip very small or very large lags (harder to improve)
            # Actually, let's still try
            pass
        
        # Choose a random pair at distance M
        if L > 0:
            i = rng.randint(0, n - M)
            j = i + M
        else:
            i = rng.randint(M, n)
            j = i - M
        
        # Determine direction: reduce the larger of h[i]*(1-h[j]) or h[j]*(1-h[i])
        # For lag L (positive): contribution is h[i](1-h[j]) where i < j, i+M=j
        # Actually for correlation at lag L:
        # If L > 0: c = sum h[i](1-h[i+L]), so pair (i, i+L) contributes h[i](1-h[i+L])
        #   To reduce: decrease h[i] or increase h[i+L]
        #   Transfer from i to i+L (decrease h[i], increase h[i+L])
        
        if L > 0:
            if h[i] > h[j]:
                d = rng.uniform(step_range[0], step_range[1])
                d = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.3)
            else:
                d = -rng.uniform(step_range[0], step_range[1])
                d = max(d, -(h[j] - h[i]) * 0.3, -h[j] * 0.3)
        else:
            if h[i] > h[j]:
                d = rng.uniform(step_range[0], step_range[1])
                d = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.3)
            else:
                d = -rng.uniform(step_range[0], step_range[1])
                d = max(d, -(h[j] - h[i]) * 0.3, -h[j] * 0.3)

            stagnation += 1
            continue
        
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_test = compute_c5(h, n)
        if c_test < best_c - 1e-14:
            best_c = c_test
            best_h = h.copy()

        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)

    return best_h, best_c

        g += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)


def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def pair_local_search(h, n, deadline, rng, topK=5, step_lo=0.005, step_hi=0.05,
                      max_stagnation=15000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagnation = 0

    while time.time() < deadline and stagnation < max_stagnation:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-topK:]
        k_star = top_idx[rng.randint(topK)]
        L = k_star - (n - 1)
        if L == 0:
            stagnation += 1
            continue
        M = abs(L)
        if M >= n or M < 1:
            stagnation += 1
            continue

        if L > 0:
            i = rng.randint(0, n - M)
            j = i + M
        else:
            i = rng.randint(M, n)
            j = i - M

        if h[i] > h[j] + 1e-8:
            d = rng.uniform(step_lo, step_hi)
            d = min(d, (h[i] - h[j]) * 0.3, h[i] * 0.3)
        elif h[j] > h[i] + 1e-8:
            d = -rng.uniform(step_lo, step_hi)
            d = max(d, -(h[j] - h[i]) * 0.3, -h[j] * 0.3)
        else:
            stagnation += 1
            continue

        if abs(d) < 1e-10:
            stagnation += 1
            continue

        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        c_test = compute_c5(h, n)
        if c_test < best_c - 1e-14:
            best_c = c_test
            best_h = h.copy()
            stagnation = 0
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
            stagnation += 1

    return best_h, best_c


def multi_alpha_lbfgs(h, n, deadline, alpha_list, maxiter_list):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha, maxit in zip(alpha_list, maxiter_list):
        if time.time() > deadline - 10:
            break
        rem = deadline - time.time()
        if rem < 5:
            break
        h_opt = lbfgsb_optimize(best_h, n, alpha, min(time.time() + min(60, rem), deadline),
                                maxiter=maxit)
        c_opt = compute_c5(h_opt, n)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    return best_h, best_c


def subgradient_method(h, n, deadline, topK=3, lr_init=0.05, decay=0.98):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    step = 0
    obj_history = []

    while time.time() < deadline:
        step += 1
        c_all = compute_all_lags(h, n)
        obj_val = c_all.max()

        grad = grad_top_k(h, n, topK)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            obj_history.append(c_new)
        else:
            obj_history.append(obj_val)

        if len(obj_history) > 100 and c_new > obj_history[-50].mean() * 1.02:
            lr *= 0.7
            h = best_h.copy()
        else:
            h = h_new

    return best_h, best_c


def structured_perturb(h, n, rng, n_modes=4, amp=0.05):
    x = np.linspace(0, 2, n)
    h_new = h.copy()
    for k in range(1, n_modes + 1):
        freq = k * rng.randint(1, 5) * np.pi
        phase = rng.uniform(0, 2 * np.pi)
        h_new += amp / k * np.sin(freq * x + phase)
    h_new = project_feasible(h_new, n)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    t0 = time.time()
    rng = np.random.RandomState(seed)
    rng_global = np.random.RandomState(seed + 1000)
    deadline = t0 + budget_s - 3

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    n_cur = len(best_h)

    # Phase 1: L-BFGS-B at initial resolution (n=500) - quick warmup
    phase1_deadline = t0 + 50
    alphas_p1 = [500, 5000, 50000, 200000]
    maxits_p1 = [300, 400, 500, 600]
    for a, mi in zip(alphas_p1, maxits_p1):
        if time.time() > phase1_deadline - 5:
            break
        h_opt = lbfgsb_optimize(best_h, n_cur, a, min(time.time() + 15, phase1_deadline), maxiter=mi)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Subgradient method at n=500
    if time.time() < deadline - 350:
        sg_deadline = min(time.time() + 40, deadline - 350)
        h_opt, c_opt = subgradient_method(best_h, n_cur, sg_deadline, topK=3,
                                          lr_init=0.03, decay=0.99)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Second L-BFGS-B pass with structured perturbation
    if time.time() < deadline - 300:
        h_pert = structured_perturb(best_h, n_cur, rng, n_modes=5, amp=0.04)
        h_opt = lbfgsb_optimize(h_pert, n_cur, 100000, min(time.time() + 25, deadline - 300),
                                maxiter=800)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        h_opt = lbfgsb_optimize(best_h, n_cur, 500000, min(time.time() + 25, deadline - 300),
                                maxiter=1000)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Upscale to n=1000
    n_1000 = 1000
    if time.time() < deadline - 200:
        h_1000 = expand_resolution(best_h, len(best_h), n_1000)
        c_1000 = compute_c5(h_1000, n_1000)

        # Optimize at n=1000
        if time.time() < deadline - 150:
            alphas_p4 = [2000, 20000, 200000, 500000]
            maxits_p4 = [400, 500, 600, 700]
            for a, mi in zip(alphas_p4, maxits_p4):
                if time.time() > deadline - 100:
                    break
                rem = deadline - 100 - time.time()
                if rem < 8:
                    break
                h_opt = lbfgsb_optimize(h_1000, n_1000, a, min(time.time() + min(35, rem), deadline - 10),
                                        maxiter=mi)
                c_opt = compute_c5(h_opt, n_1000)
                if c_opt < c_1000:
                    c_1000 = c_opt
                    h_1000 = h_opt
                if c_1000 < best_c:
                    best_c = c_1000
                    best_h = h_1000
                    n_cur = n_1000

    # Phase 5: Adam optimization at best resolution
    if time.time() < deadline - 150:
        n_use = len(best_h)
        adam_deadline = min(time.time() + 80, deadline - 150)
        h_opt, c_opt = adam_optimize(best_h, n_use, 30000, adam_deadline, lr=0.004)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_cur = n_use

    # Phase 6: L-BFGS-B high-precision polish
    if time.time() < deadline - 100:
        n_use = len(best_h)
        alphas_p6 = [1000000, 5000000, 20000000]
        maxits_p6 = [2000, 2000, 2500]
        for a, mi in zip(alphas_p6, maxits_p6):
            if time.time() > deadline - 50:
                break
            rem = deadline - 50 - time.time()
            if rem < 8:
                break
            h_opt = lbfgsb_optimize(best_h, n_use, a, min(time.time() + min(30, rem), deadline - 20),
                                    maxiter=mi)
            c_opt = compute_c5(h_opt, n_use)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: Targeted pair local search
    if time.time() < deadline - 40:
        n_use = len(best_h)
        ls_deadline = min(time.time() + 80, deadline - 40)
        h_opt, c_opt = pair_local_search(best_h, n_use, ls_deadline, rng,
                                         topK=5, step_lo=0.003, step_hi=0.04,
                                         max_stagnation=20000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Another Adam pass with different parameters
    if time.time() < deadline - 50:
        n_use = len(best_h)
        adam_deadline2 = min(time.time() + 40, deadline - 50)
        h_opt, c_opt = adam_optimize(best_h, n_use, 100000, adam_deadline2, lr=0.002,
                                     max_steps=10000, reset_interval=2000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Final L-BFGS-B + local search
    if time.time() < deadline - 15:
        n_use = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_use, 50000000, min(time.time() + 20, deadline - 10),
                                maxiter=2000)
        c_opt = compute_c5(h_opt, n_use)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 3:
        n_use = len(best_h)
        h_opt, c_opt = pair_local_search(best_h, n_use, deadline - 1, rng,
                                         topK=3, step_lo=0.001, step_hi=0.01,
                                         max_stagnation=8000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation and cleanup
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
