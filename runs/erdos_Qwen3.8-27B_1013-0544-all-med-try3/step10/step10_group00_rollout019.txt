    dx = 2.0 / n

    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            cnt = int(np.ceil(diff))
            for idx in order[:cnt]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            cnt = int(np.ceil(-diff))
            for idx in order[:cnt]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def make_beatty_binary(n, rng):
    # Use golden ratio Beatty sequence
    phi = (1 + np.sqrt(5)) / 2
    h = np.zeros(n)
    positions = np.array([int(k * phi) % n for k in range(1, 4*n)])
    positions = positions[:n//2]
    # Handle duplicates
    seen = set()
    count = 0
    for p in positions:

            seen.add(p)
            h[p] = 1.0
            count += 1
            if count >= n//2:

    if count < n//2:
        zeros = np.where(h == 0)[0]
        extra = rng.choice(zeros, size=n//2-count, replace=False)
        h[extra] = 1.0
    return h

def simulated_annealing_binary(h, n, deadline, rng, n_iter=200000):
    """SA on binary configuration with swap moves."""
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    
    # SA schedule
    T0 = 0.02  # initial temperature
    Tf = 1e-7  # final temperature
    # Geometric cooling
    if n_iter > 1:
        cool_factor = (Tf / T0) ** (1.0 / n_iter)
    
    T = T0
    no_improve = 0
    plateau_limit = 5000
    
    start_time = time.time()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Periodic check (every 1000 iters)
        if it % 1000 == 0 and it > 0:
            elapsed = time.time() - start_time
            if elapsed > 0:
                actual_rate = it / elapsed
                remaining = n_iter - it
                if remaining / actual_rate > (deadline - time.time()) * 2:
                    break  # would take too long
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h, n)
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        # Update temperature
        T *= cool_factor
        
        # Restart if stuck
        if no_improve > plateau_limit:
            no_improve = 0
            T = max(T, 0.005)
            # Do some random moves
            for _ in range(50):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h[ii] = 0.0
                h[jj] = 1.0
                c2 = compute_c5(h, n)
                if c2 < cur_c:
                    cur_c = c2
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    ones.remove(ii)
                    ones.append(jj)
                    zeros.remove(jj)
                    zeros.append(ii)
                else:
                    h[ii] = 1.0
                    h[jj] = 0.0
    
    return best_h, best_c

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgs_obj, h.copy(), jac=lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        h_new = project_feasible(h, n)
        return h_new, compute_c5(h_new, n)

def greedy_binary_refine(h, n, deadline, rng, max_iters=100000):

                found = True
                break
            else:

    return h, best_c

                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-20, 'maxfun': maxit * 12,
                               'maxls': 50, 'maxcor': 60})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        h_new = project_feasible(h, n)
        return h_new, compute_c5(h_new, n)


def greedy_binary_refine(h, n, deadline, rng, max_iters=200000):
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    best_c = compute_c5(h, n)
    no_improve = 0
    stall_limit = 30000

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > stall_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

    return h, best_c


def target_local_search(h, n, deadline, rng, max_rounds=5000):
    """Focus on reducing the specific lags that achieve the maximum."""
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    best_c = compute_c5(h, n)

    for rd in range(max_rounds):
        if time.time() > deadline:
            break

        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        tol = min(0.002, 0.02 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]

        # For each active lag, find which positions contribute most
        contrib = np.zeros(n)
        for mm in active:
            k = mm - (n - 1)
            weight = c_vec[mm] / max(lay_sum(c_vec[active]), 1e-15)
            if k >= 0:
                nv = n - k
                contrib[0:nv] += weight * (1.0 - h[k:n]) * h[0:nv]
                contrib[k:n] += weight * h[0:nv] * (1.0 - h[0:nv])
            else:
                kk = -k
                nv = n - kk
                contrib[kk:n] += weight * (1.0 - h[0:nv]) * h[kk:n]
                contrib[0:nv] += weight * h[kk:n] * (1.0 - h[0:nv])

        # Try to reduce contribution from positions with high contrib
        high_pos = np.where(contrib > 0.5 * contrib.max())[0]
        found = False
        for _ in range(200):
            if time.time() > deadline:
                break
            if len(high_pos) > 2:
                i = high_pos[rng.randint(len(high_pos))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if h[i] > 0.5:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    found = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            elif h[j] < 0.5:
                continue

        if not found:
            # Random swap as escape
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h, n)
            if c < best_c - 1e-15:
                best_c = c
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0

    return h, best_c


def lay_sum(vals):
    return float(np.sum(vals)) if len(vals) > 0 else 1.0


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Phase 1: SA at parent resolution (180s)
    p1_deadline = t0 + 180
    h_bin_p = binarize(best_h, best_n)
    h1, c1 = simulated_annealing_binary(h_bin_p, best_n, p1_deadline, rng,
                                         n_iter=120000, T0=0.012, Tf=5e-8)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Quick continuous refinement (40s)
    p2_deadline = t0 + 220
    h2, c2 = run_lbfgs(best_h, best_n, p2_deadline, alpha=50000, mu=0.01, maxit=800)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: SA at n=1000 from upscaled init (180s)
    n_hires = 1000
    p3_deadline = t0 + 400
    h_fine = np.zeros(n_hires)
    for i in range(n_hires):
        fi = i * best_n / n_hires
        ci = int(fi)
        ci2 = min(ci + 1, best_n - 1)
        frac = fi - ci
        h_fine[i] = (1.0 - frac) * best_h[ci] + frac * best_h[ci2]
    h_fine = project_feasible(h_fine, n_hires)
    h_fine_bin = binarize(h_fine, n_hires)
    h3, c3 = simulated_annealing_binary(h_fine_bin, n_hires, p3_deadline, rng,
                                         n_iter=90000, T0=0.010, Tf=3e-8)

    # Also try Beatty init at n=1000
    h_beatty = make_beatty_binary(n_hires, rng)
    p3b_deadline = min(t0 + 480, deadline - 400)
    h3b, c3b = simulated_annealing_binary(h_beatty, n_hires, p3b_deadline, rng,
                                           n_iter=70000, T0=0.010, Tf=3e-8)
    if c3b < c3:
        h3, c3 = h3b, c3b

    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()
        best_n = n_hires

    # Phase 4: Multi-start SA (200s)
    p4_deadline = t0 + 680
    num_starts = 5
    for ms in range(num_starts):
        if time.time() > p4_deadline:
            break
        sub_dl = min(p4_deadline, time.time() + (p4_deadline - time.time()) / (num_starts - ms))
        if sub_dl - time.time() < 10:
            break
        h_ms = binarize(best_h, best_n)
        ones = np.where(h_ms > 0.5)[0]
        zeros = np.where(h_ms < 0.5)[0]
        nswaps = rng.randint(8, 25)
        for _ in range(nswaps):
            ii = ones[rng.randint(len(ones))]
            jj = zeros[rng.randint(len(zeros))]
            h_ms[ii], h_ms[jj] = h_ms[jj], h_ms[ii]
        h_ms_new, c_ms = simulated_annealing_binary(h_ms, best_n, sub_dl, rng,
                                                     n_iter=50000, T0=0.010, Tf=2e-8)
        if c_ms < best_c:
            best_c = c_ms
            best_h = h_ms_new.copy()

    # Phase 5: Target local search (80s)
    p5_deadline = t0 + 760
    h5, c5 = target_local_search(binarize(best_h, best_n), best_n, p5_deadline, rng,
                                  max_rounds=3000)
    if c5 < best_c:
        best_c = c5
        best_h = h5.copy()

    # Phase 6: Comprehensive continuous refinement (150s)
    p6_deadline = t0 + 910
    for alpha in [20000, 100000, 500000, 2000000]:
        if time.time() > p6_deadline - 3:
            break
        sub_dl = min(p6_deadline, time.time() + 30)
        h6, c6 = run_lbfgs(best_h, best_n, sub_dl, alpha=alpha, mu=0.0005, maxit=1200)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    # Phase 7: Greedy binary refine (80s)
    p7_deadline = t0 + 990
    h7, c7 = greedy_binary_refine(binarize(best_h, best_n), best_n, p7_deadline,
                                   rng, max_iters=300000)
    if c7 < best_c:
        best_c = c7
        best_h = h7.copy()

    # Phase 8: Final ultra-precise continuous polish (60s)
    p8_deadline = min(t0 + 1050, deadline - 20)
    for alpha in [1000000, 5000000, 20000000]:
        if time.time() > p8_deadline - 2:
            break
        sub_dl = min(p8_deadline, time.time() + 18)
        h8, c8 = run_lbfgs(best_h, best_n, sub_dl, alpha=alpha, mu=0.0001, maxit=800)
        if c8 < best_c:
            best_c = c8
            best_h = h8.copy()

    # Phase 9: One more SA pass if time allows (40s)
    if time.time() < deadline - 25:
        h9, c9 = simulated_annealing_binary(binarize(best_h, best_n), best_n,
                                             deadline - 25, rng,
                                             n_iter=60000, T0=0.006, Tf=1e-9)
        if c9 < best_c:
            best_c = c9
            best_h = h9.copy()

    # Final: ensure exact constraint and evaluate
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    # Verify constraint
    assert abs(best_h.sum() - best_n / 2.0) < 1e-10, "Constraint violation"
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15), "Bounds violation"
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
