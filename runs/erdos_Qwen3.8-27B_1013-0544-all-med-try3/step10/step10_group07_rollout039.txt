def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            s = room.sum()

            if s < 1e-14:

        m = g.max()

    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_sa(h_init, n, deadline, seed=42, T_start=5.0, T_end=0.2, directed_frac=0.3):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    h = make_binary(h_init, n)
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = T_start
    # We'll cool based on time remaining
    t_start = time.time()
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Adaptive temperature based on time
        t_frac = (time.time() - t_start) / max(deadline - t_start, 1)
        T = T_start * (T_end / T_start) ** t_frac
        
        # Choose move: directed or random
        if rng.random() < directed_frac:
            # Directed: find best candidate pair
            max_idx = np.argmax(corr)
            m_star = max_idx - (n - 1)
            
            # Score zeros for 'a': prefer positions where h[a+m*] and h[a-m*] are 1
            scores_a = np.zeros(len(zeros))
            for i, z in enumerate(zeros):
                s = 0
                if 0 <= z + m_star < n:
                    s += h[z + m_star]
                if 0 <= z - m_star < n:
                    s += h[z - m_star]
                scores_a[i] = s
            
            # Score ones for 'b': prefer positions where h[b+m*] and h[b-m*] are 0
            scores_b = np.zeros(len(ones))
            for i, o in enumerate(ones):
                s = 0
                if 0 <= o + m_star < n:
                    s += (1 - h[o + m_star])
                if 0 <= o - m_star < n:
                    s += (1 - h[o - m_star])
                scores_b[i] = s
            
            # Weighted random selection
            if scores_a.max() > 0:
                pa = scores_a / scores_a.sum()
                ai = rng.choice(len(zeros), p=pa)
            else:
                ai = rng.randint(len(zeros))
            if scores_b.max() > 0:
                pb = scores_b / scores_b.sum()
                bi = rng.choice(len(ones), p=pb)
            else:
                bi = rng.randint(len(ones))
            
            a = zeros[ai]
            b = ones[bi]
        else:
            # Random
            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
        
        # Compute delta
        delta = np.zeros(2 * n - 1)
        lo1 = n - 1 - b
        delta[lo1:lo1 + n] += h
        lo2 = n - 1 - a
        delta[lo2:lo2 + n] -= h
        delta[b:b + n] += h[::-1]
        delta[a:a + n] -= h[::-1]
        
        new_vals = corr + delta
        new_max = new_vals.max()
        
        # Acceptance
        if new_max <= cur_max:
            accept = True
        else:
            # Simulated annealing: accept with probability exp(-(new_max - cur_max)/T)
            if T < 0.01:
                accept = False
            else:
                accept = rng.random() < np.exp(-float(new_max - cur_max) / T)
        
        if accept:
            corr = new_vals
            h[a] = 1.0
            h[b] = 0.0
            # Update lists
            idx_b = np.where(ones == b)[0][0]
            ones = np.delete(ones, idx_b)
            ones = np.append(ones, a)
            idx_a = np.where(zeros == a)[0][0]
            zeros = np.delete(zeros, idx_a)

            cur_max = new_max
            c_new = cur_max * dx
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_init, n)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

    ones = list(np.where(h == 1)[0])

        if rng.random() < directed_frac:

def binary_greedy(h_init, n, deadline, seed=42, max_rounds=200):

    best_h = h.copy()

    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        max_idx = np.argmax(corr)
        m_star = max_idx - (n - 1)
        candidates = []
        for z in zeros:
            score_a = 0

            if 0 <= z - m_star < n:

                    score_b += (1.0 - h[o + m_star])

                    score_b += (1.0 - h[o - m_star])
                if score_a + score_b >= 2:
                    candidates.append((score_a + score_b, z, o))
        candidates.sort(reverse=True)
        found = False
        for score, a, b in candidates[:200]:
            if time.time() > deadline:
                break
            if h[a] != 0 or h[b] != 1:
                continue
            delta = np.zeros(2 * n - 1)
            lo1 = n - 1 - b
            delta[lo1:lo1 + n] += h
            lo2 = n - 1 - a
            delta[lo2:lo2 + n] -= h
            delta[b:b + n] += h[::-1]
            delta[a:a + n] -= h[::-1]
            new_vals = corr + delta
            new_max = new_vals.max()
            if new_max < cur_max:

def qr_start(n, p=None):
    """Generate a quadratic-residue-inspired starting point."""
    if p is None:
        primes = [127, 131, 251, 257, 509, 503]
        p = min(primes, key=lambda x: abs(x - n))
    rnd = np.random.RandomState(0)
    h = np.zeros(n)
    for i in range(n):
        r = i % p
        if r == 0:
            h[i] = 1
        else:
            if (r * r) % p < p // 2:
                h[i] = 1
    target = n // 2
    cur_sum = int(h.sum())
    if cur_sum > target:
        idx = np.where(h == 1)[0]
        to_remove = idx[rnd.permutation(cur_sum - target)][:cur_sum - target]
        h[to_remove] = 0
    elif cur_sum < target:
        idx = np.where(h == 0)[0]
        to_add = idx[rnd.permutation(target - cur_sum)][:target - cur_sum]
        h[to_add] = 1
    return h
