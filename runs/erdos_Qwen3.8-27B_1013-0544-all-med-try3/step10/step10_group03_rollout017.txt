            w /= w.sum()
            g = weighted_corr_grad(x, n, w, np.arange(len(c)))
            g += 2.0 * mu * (x.sum() - target)
            return g
        
        try:
            res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': maxiter * 10})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def top_shift_lbfgs(h, n, deadline, top_k=8, alphas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    if alphas is None:
        alphas = [50000, 200000, 1000000]
    
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        active, _ = get_active_set(h, n, tol=0.001, max_k=top_k)
        
        def obj(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            val = smooth_max_vals(vals, alpha)
            val += 0.1 * (x.sum() - target) ** 2
            return float(val)
        
        def grad(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            m = vals.max()
            sh = alpha * (vals - m)
            sh_max = sh.max()
            w = np.exp(sh - sh_max)
            w /= w.sum()
            g = np.zeros(n)
            for wi, idx in zip(w, active):
                if wi < 1e-14:
                    continue
                k = idx - (n - 1)
                wm = wi * (2.0 / n)
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            g += 0.2 * (x.sum() - target)
            return g
        
        try:
            res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-17,
                                   'gtol': 1e-14, 'maxfun': 50000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c

def binary_sa(h_init, n, deadline, n_rounds=200):
    rng = np.random.RandomState(42)
    h = (h_init > 0.5).astype(float)
    ones = n // 2
    n_ones = int(h.sum())
    if n_ones > ones:
        h[rng.choice(np.where(h == 1)[0], n_ones - ones, replace=False)] = 0
    elif n_ones < ones:
        h[rng.choice(np.where(h == 0)[0], ones - n_ones, replace=False)] = 1
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = 0.005
    n_total = 2 * n - 1
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        T *= 0.995
        for _ in range(5):
            if time.time() > deadline:
                break
            pos1 = np.where(h == 1)[0]
            pos0 = np.where(h == 0)[0]
            if len(pos1) == 0 or len(pos0) == 0:
                break
            i = pos1[rng.randint(len(pos1))]
            j = pos0[rng.randint(len(pos0))]
            
            c_old = compute_c5_vec(h, n)
            max_old = c_old.max()
            
            h[i] = 0
            h[j] = 1
            c_new = compute_c5_vec(h, n)
            max_new = c_new.max()
            delta = max_new - max_old
            
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                c_cur = compute_c5(h, n)
                if c_cur < best_c:
                    best_c = c_cur
                    best_h = h.copy()
            else:
                h[j] = 0
                h[i] = 1
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas=None):
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(max(0, i - 40), min(n, i + 41)):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
    return h, best_c

def multi_resolution(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for n_coarse in [n // 2, n // 4]:
        if n_coarse < 50 or time.time() > deadline - 20:
            continue
        x_fine = np.linspace(0, 2, n, endpoint=False)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
        h_c = np.interp(x_coarse, x_fine, h)
        h_c = project_feasible(h_c, n_coarse)
        
        t_c = min(30.0, (deadline - time.time()) * 0.2)
        if t_c > 10:
            h_c, c_c = full_lbfgs(h_c, n_coarse, time.time() + t_c,
                                   [(500, 10.0, 2000), (2000, 100.0, 5000), (5000, 1000.0, 10000)])
        
        h_f = np.interp(x_fine, x_coarse, h_c)
        h_f = project_feasible(h_f, n)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
    
    for n_fine in [int(n * 1.5), n * 2]:
        if n_fine > 2000 or time.time() > deadline - 30:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_fine, endpoint=False)
        h_u = np.interp(x_new, x_old, best_h)
        h_u = project_feasible(h_u, n_fine)
        
        t_u = min(30.0, (deadline - time.time()) * 0.2)
        if t_u > 10:
            h_u, c_u = full_lbfgs(h_u, n_fine, time.time() + t_u,
                                   [(200, 5.0, 1000), (1000, 50.0, 3000)])
        
        c_u = compute_c5(h_u, n_fine)
        if c_u < best_c:
            best_c = c_u
            best_h = h_u
    
    return best_h, best_c, len(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Block gradient descent (fast, many iterations)
    t1 = min(t0 + 80, deadline - 400)
    if time.time() < t1:
        h_gd, c_gd = block_gradient_descent(
            best_h, n,
            active=np.arange(2*n-1), alpha=2000,
            block_size=20, n_epochs=80,
            deadline=t1, lr_init=0.03, n_steps=6
        )
        if c_gd < best_c:
            best_c = c_gd
            best_h = h_gd
    
    # Phase 2: Block L-BFGS (more precise)
    t2 = min(t0 + 200, deadline - 300)
    if time.time() < t2:
        h_bl, c_bl = block_lbfgs(
            best_h, n,
            active=np.arange(2*n-1), alpha=5000,
            block_size=15, n_epochs=40,
            deadline=t2
        )
        if c_bl < best_c:
            best_c = c_bl
            best_h = h_bl
    
    # Phase 3: Full-space L-BFGS with multiple alphas
    t3 = min(t0 + 380, deadline - 250)
    if time.time() < t3:
        h_fl, c_fl = full_lbfgs(best_h, n, t3,
            configs=[
                (500, 5.0, 3000),
                (2000, 50.0, 5000),
                (5000, 500.0, 8000),
                (20000, 2000.0, 10000),
                (100000, 50000.0, 15000),
            ])
        if c_fl < best_c:
            best_c = c_fl
            best_h = h_fl
    
    # Phase 4: Top-shift focused optimization
    t4 = min(t0 + 500, deadline - 200)
    if time.time() < t4:
        h_ts, c_ts = top_shift_lbfgs(
            best_h, n, t4, top_k=8,
            alphas=[50000, 200000, 500000, 2000000]
        )
        if c_ts < best_c:
            best_c = c_ts
            best_h = h_ts
    
    # Phase 5: Binary SA for structural improvement (different approach)
    t5 = min(t0 + 620, deadline - 200)
    if time.time() < t5 and t5 - time.time() > 30:
        h_sa, c_sa = binary_sa(best_h, n, t5)
        h_sa_cont = project_feasible(h_sa, n)
        c_sa_cont = compute_c5(h_sa_cont, n)
        if c_sa_cont < best_c:
            best_c = c_sa_cont
            best_h = h_sa_cont
        # Also try refining the SA result continuously
        if time.time() < t5 + 30 and deadline - time.time() > 60:
            h_ref, c_ref = full_lbfgs(h_sa_cont, n, min(t5 + 30, deadline - 60),
                [(1000, 10.0, 3000), (5000, 100.0, 5000)])
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref
    
    # Phase 6: Multi-resolution
    t6 = min(t0 + 750, deadline - 100)
    if time.time() < t6 and t6 - time.time() > 40:
        h_mr, c_mr, n_mr = multi_resolution(best_h, n, t6)
        if c_mr < best_c:
            best_c = c_mr
            best_h = h_mr
            n = n_mr
    
    # Phase 7: Final top-shift polish
    t7 = min(t0 + 850, deadline - 60)
    if time.time() < t7:
        h_p, c_p = top_shift_lbfgs(
            best_h, n, t7, top_k=5,
            alphas=[500000, 2000000, 10000000]
        )
        if c_p < best_c:
            best_c = c_p
            best_h = h_p
    
    # Phase 8: Fine pair search
    t8 = min(t0 + 950, deadline - 20)
    if time.time() < t8:
        best_h, best_c = fine_pair_search(best_h, n, t8,
            deltas=[0.0005, 0.0002, 0.0001, 0.00005])
    
    # Phase 9: Final full L-BFGS polish
    if time.time() < deadline - 3:
        h_final, c_final = full_lbfgs(best_h, n, deadline - 2,
            [(50000, 10000.0, 10000), (500000, 100000.0, 5000)])
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
