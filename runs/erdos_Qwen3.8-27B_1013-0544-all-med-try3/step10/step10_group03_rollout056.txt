def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def shift_gradient(h, n, shift_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    k = shift_idx - (n - 1)
    if k >= 0:
        nv = n - k
        grad[0:nv] += dx * (1.0 - h[k:n])
        grad[k:n] += -dx * h[0:nv]
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += dx * (1.0 - h[0:nv])
        grad[0:nv] += -dx * h[kk:n]
    return grad

                        h[k] += step / 2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found = True
                            break
                        else:
                            h[i] += step
                            h[j] -= step / 2
                            h[k] -= step / 2
                    if h[i] + step <= 1 and h[j] - step/2 >= 0 and h[k] - step/2 >= 0:
                        h[i] += step
                        h[j] -= step / 2
                        h[k] -= step / 2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found = True
                            break
                        else:
                            h[i] -= step
                            h[j] += step / 2
                            h[k] += step / 2
        if not found:
            step *= 0.7
            if step < 1e-7:
                break
    return h, best_c

def flat_region_optimize(h, n, deadline):
    """Optimize regions where h is near 0 or 1 by small perturbations."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    eps = 0.001

    # Find indices near 0 or 1
    near_zero = np.where(h < 0.05)[0]
    near_one = np.where(h > 0.95)[0]

    # Try small perturbations to break ties
    for _ in range(100):
        if time.time() > deadline - 2:
            break
        if len(near_zero) < 2 or len(near_one) < 1:
            break
        i = near_zero[np.random.randint(0, len(near_zero))]
        j = near_one[np.random.randint(0, len(near_one))]
        h[i] += eps
        h[j] -= eps
        h[i] = min(1.0, max(0.0, h[i]))
        h[j] = min(1.0, max(0.0, h[j]))
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-16:
            best_c = c
            best_h = h.copy()
        else:
            h[i] -= eps
            h[j] += eps
            h = project_feasible(h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    h = project_feasible(h_init, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Active set L-BFGS (top 5)
    t1 = min(t0 + 250, deadline - 600)
    if time.time() < t1 - 5:
        h, c = active_lbfgs(h, n, t1, top_k=5)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h = h.copy()

    # Phase 2: Subgradient descent
    t2 = min(t0 + 420, deadline - 400)
    if time.time() < t2 - 5:
        h, c = subgradient_descent(h, n, t2, top_k=5, init_step=0.004)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h = h.copy()

    # Phase 3: Active set L-BFGS (top 3, higher alpha)
    t3 = min(t0 + 570, deadline - 300)
    if time.time() < t3 - 5:
        h, c = active_lbfgs(h, n, t3, top_k=3)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h = h.copy()

    # Phase 4: Targeted local search
    t4 = min(t0 + 750, deadline - 200)
    if time.time() < t4 - 5:
        h, c = targeted_local_search(h, n, t4, top_k=5, pool_size=80)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h = h.copy()

    # Phase 5: 3-point moves
    t5 = min(t0 + 870, deadline - 130)
    if time.time() < t5 - 5:
        h, c = try_ternary_move(h, n, t5, top_k=5, pool_size=50)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Another subgradient round from best
    t6 = min(t0 + 970, deadline - 80)
    if time.time() < t6 - 5:
        h = best_h.copy()
        h, c = subgradient_descent(h, n, t6, top_k=4, init_step=0.001)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: Final fine local search
    t7 = deadline - 3
    if time.time() < t7 - 3:
        h = best_h.copy()
        h, c = targeted_local_search(h, n, t7, top_k=3, pool_size=50)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Final: project and evaluate
    best_h = project_feasible(best_h, n)
    # Ensure exact sum
    target_sum = n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - best_h
            mask = room > 1e-14
            if mask.any():
                best_h[mask] += diff / mask.sum()
        else:
            mask = best_h > 1e-14
            if mask.any():
                best_h[mask] += diff / mask.sum()
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
