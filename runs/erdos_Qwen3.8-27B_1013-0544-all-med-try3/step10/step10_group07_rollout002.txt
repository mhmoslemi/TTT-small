def compute_c5(h, n):

def project_feasible(h, n):

            break

        else:

            if s < 1e-14:

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_swap_delta(h, n, i, j, m_range, corr):
    """Compute the change in correlation when h[i]: 0->1, h[j]: 1->0."""
    n_corr = 2 * n - 1
    delta = np.zeros(n_corr)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < n_corr:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < n_corr:
        delta[k2] += 1.0
    return delta

def make_binary(h_init, n):
    """Convert continuous h to binary with exactly n//2 ones."""
    target = n // 2
    h = np.array(h_init, dtype=float).copy()
    h_bin = (h >= 0.5).astype(float)
    cur = int(h_bin.sum())
    if cur > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h)[ones_idx]
        to_zero = ones_idx[order[:cur - target]]
        h_bin[to_zero] = 0.0
    elif cur < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h)[zeros_idx]
        to_one = zeros_idx[order[:target - cur]]
        h_bin[to_one] = 1.0
    return h_bin

            best_c = c_ls
            best_h = h_ls.copy()
    
    # Final SLSQP
    if time.time() < dl - 3:
        h_f = slsqp_smooth(best_h.copy(), n, 5000000, 1000, dl - 2)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
