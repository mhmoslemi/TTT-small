import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    w_thresh = 1e-12
    for mm in range(2 * n - 1):
        if w[mm] < w_thresh:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_smooth_opt(h, n, deadline, alphas, mu=0.01, maxiters=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiters is not None and i < len(maxiters):
            maxit = maxiters[i]
        else:
            maxit = max(500, int(remaining * 10))
        try:
            res = minimize(sm_obj, best_h, jac=sm_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': maxit * 12,
                                   'maxls': 100, 'maxcor': 80})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_active_set(h, n, deadline, top_k=10, alphas=None, rounds=15):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [5000, 20000, 100000, 500000, 2000000, 10000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.0008 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            order = np.argsort(-c_vec[active])
            active = active[order[:top_k]]
        if len(active) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 15:
                break
            maxit = max(400, int((deadline - time.time()) * 6))
            try:
                res = minimize(act_obj, best_h, jac=act_grad,
                              args=(n, active, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxfun': maxit * 12,
                                       'maxls': 80, 'maxcor': 60})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new <= best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run_slsqp(h, n, deadline, alpha=1000000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def subgrad_refine(h, n, deadline, n_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.008
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0005 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            eta = min(eta * 1.01, 0.04)
        else:
            eta *= 0.97
    return best_h, best_c

def gradient_guided_pairs(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        tol = max(1e-7, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx
                grad[k:n] += (-best_h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx
                grad[0:nv] += (-best_h[kk:n]) * dx
        grad /= len(active)
        pos = np.where(grad > 0.005)[0]
        neg = np.where(grad < -0.005)[0]
        if len(pos) == 0 or len(neg) == 0:
            continue
        n_try = min(30, len(pos) * len(neg))
        rng_local = np.random.RandomState(12345 + it)
        for _ in range(n_try):
            i = pos[rng_local.randint(len(pos))]
            j = neg[rng_local.randint(len(neg))]
            for step in [0.004, 0.002, 0.001, 0.0005]:
                if best_h[i] - step >= 0 and best_h[j] + step <= 1:
                    best_h[i] -= step
                    best_h[j] += step
                    best_h = project_feasible(best_h, n)
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        break
                    else:
                        best_h[i] += step
                        best_h[j] -= step
                        best_h = project_feasible(best_h, n)
    return best_h, best_c

def upsample(h_coarse, n_fine):
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return project_feasible(h_fine, n_fine)

def coarse_to_fine(h, n, deadline, n_coarse):
    """Downsample, optimize coarsely, upsample back, then fine optimize."""
    block = n // n_coarse
    if block < 2:
        return h, compute_c5(h, n)
    h_c = np.array([h[i*block:(i+1)*block].mean() for i in range(n_coarse)])
    h_c = project_feasible(h_c, n_coarse)
    c_bound = deadline - (deadline - time.time()) * 0.5
    h_c, c_c = run_smooth_opt(h_c, n_coarse, c_bound,
        alphas=[100, 1000, 10000, 100000], mu=0.01,
        maxiters=[2000, 3000, 3000, 3000])
    h_up = upsample(h_c, n)
    c_up = compute_c5(h_up, n)
    c_fine_bound = deadline - 30
    h_f, c_f = run_smooth_opt(h_up, n, c_fine_bound,
        alphas=[5000, 50000, 500000], mu=0.005,
        maxiters=[3000, 4000, 4000])
    return h_f, c_f

def make_binary_start(h, n):
    h_bin = np.zeros(n)
    n_ones = n // 2
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:n_ones]] = 1.0
    remaining = n / 2.0 - h_bin.sum()
    if abs(remaining) > 1e-10:
        next_idx = sorted_idx[n_ones]
        h_bin[next_idx] += remaining
    return project_feasible(h_bin, n)
