import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    nn = 2 * n - 1
    for mm in range(nn):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)

        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])

def top_grad(x, n, active, alpha, mu):

def run_lbfgs(h, n, deadline, configs):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in configs:

    current_c = best_c

        if h[i] + di < 0 or h[i] + di > 1:
            continue
        if h[j] + dj < 0 or h[j] + dj > 1:
            continue
        
        h[i] += di
        h[j] += dj
        c_new = compute_c5(h, n)
        
        if c_new < current_c:
            current_c = c_new
            no_improve = 0
            if c_new < best_c:
                best_c = c_new
                no_improve = 0
        else:
            # Annealing: accept with prob exp(-(delta)/T)
            delta = c_new - current_c
            if delta < T:
                if seed_rng.random() < np.exp(-delta / T):
                    current_c = c_new
                else:
                    h[i] -= di
                    h[j] -= dj
            else:
                h[i] -= di
                h[j] -= dj
        
        no_improve += 1
        T *= 0.9995
        if no_improve > 500:
            no_improve = 0
            if step_idx < len(step_sizes) - 1:
                step_idx += 1
    
    return h, best_c

def fine_local_search(h, n, deadline):
    """Very fine local pair search near critical points."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    deltas_descent = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas_descent:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline - 2:
                    return h, best_c
                # Try moving mass to/from nearby positions
                lo = max(0, i - 40)
                hi = min(n, i + 41)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    # h[i] -= d, h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def targeted_gradient_search(h, n, deadline):
    """Use gradient to identify best pair moves."""
    best_c = compute_c5(h, n)
    h = h.copy()
    alpha = 50000
    base_steps = [0.003, 0.001, 0.0003, 0.0001]
    step_i = 0
    
    while time.time() < deadline - 2:
        c_vec = compute_c5_vec(h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            break
        if len(active) > 12:
            active = active[np.argsort(-c_vec[active])[:12]]
        
        # Compute gradient for active shifts
        sh = alpha * (c_vec[active] - max_val)
        smx = sh.max()
        w = np.exp(sh - smx)
        w /= w.sum()
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for wi, idx in zip(w, active):
            k = idx - (n - 1)
            wm = wi * dx
            if k >= 0:
                nv = n - k
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
        
        # Find best direction: decrease grad (reduce correlation)
        # Variable with high positive grad should decrease, low grad should increase
        g_pos = np.where(grad > 0.01)[0]
        g_neg = np.where(grad < -0.01)[0]
        
        if len(g_pos) == 0 or len(g_neg) == 0:
            step_i = min(step_i + 1, len(base_steps) - 1)
            continue
        
        # Sort by magnitude
        g_pos = g_pos[np.argsort(-grad[g_pos])[:20]]
        g_neg = g_neg[np.argsort(grad[g_neg])[:20]]
        
        found = False
        for i in g_pos:
            if found:
                break
            for j in g_neg:
                for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    step = base_steps[step_i] * step_mult
                    if h[i] - step >= 0 and h[j] + step <= 1:
                        h[i] -= step; h[j] += step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found = True
                            break
                        else:
                            h[i] += step; h[j] -= step
                    if h[i] + step <= 1 and h[j] - step >= 0:
                        h[i] += step; h[j] -= step
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            found = True
                            break
                        else:
                            h[i] -= step; h[j] += step
        if not found:
            step_i = min(step_i + 1, len(base_steps) - 1)
            if step_i == len(base_steps) - 1 and base_steps[step_i] < 1e-5:
                break
    
    return h, best_c

def multiresolve(h_fine, n_fine, deadline):
    """Coarse-level optimization then upsample."""
    best_h = h_fine.copy()
    best_c = compute_c5(h_fine, n_fine)
    
    for ratio in [2, 4]:
        n_coarse = n_fine // ratio
        if n_coarse < 100:
            continue
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
        
        # Downsample
        h_c = np.zeros(n_coarse)
        for i in range(n_coarse):
            idxs = slice(i * ratio, (i + 1) * ratio)
            h_c[i] = h_fine[idxs].mean()
        h_c = project_feasible(h_c, n_coarse)
        
        t_avail = (deadline - time.time()) * 0.3
        if t_avail > 20:
            h_c, c_c = run_lbfgs(h_c, n_coarse, time.time() + t_avail,
                                  [(100, 10.0), (500, 100.0), (2000, 1000.0)])
        
        # Upsample
        h_up = np.interp(x_fine, x_coarse, h_c, period=2.0) if False else \
               np.interp(x_fine, x_coarse, h_c)
        h_up = project_feasible(h_up, n_fine)
        c_up = compute_c5(h_up, n_fine)
        
        if c_up < best_c:
            best_c = c_up
            best_h = h_up
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_orig = 500
    
    best_h = project_feasible(h_init, n_orig)
    best_c = compute_c5(best_h, n_orig)
    
    # Phase 1: Quick L-BFGS at original resolution
    t1 = min(t0 + 60, deadline - 600)
    if t1 > t0 + 10:
        best_h, best_c = run_lbfgs(best_h, n_orig, t1,
            [(100, 10.0), (500, 50.0), (2000, 500.0), (5000, 5000.0)])
    
    # Phase 2: Upsample to n=1000
    n = 1000
    x_old = np.linspace(0, 2, n_orig, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_new = np.interp(x_new, x_old, best_h)
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    if c_new <= best_c + 1e-6:
        best_h = h_new
        best_c = c_new
    else:
        best_h = h_new
        best_c = c_new
    
    # Phase 3: L-BFGS at n=1000 with various alpha
    t3 = min(t0 + 250, deadline - 400)
    if time.time() < t3 - 10:
        best_h, best_c = run_lbfgs(best_h, n, t3,
            [(200, 20.0), (1000, 100.0), (5000, 1000.0),
             (10000, 10000.0), (30000, 100000.0)])
    
    # Phase 4: Top-shifts focused optimization
    t4 = min(t0 + 450, deadline - 300)
    if time.time() < t4 - 10:
        best_h, best_c = run_top_opt(best_h, n, t4, top_k=8)
    
    # Phase 5: Stochastic pair exchange (new - simulated annealing)
    t5 = min(t0 + 700, deadline - 200)
    if time.time() < t5 - 10:
        h_sa, c_sa = stochastic_pair_exchange(best_h, n, t5, rng)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 6: More L-BFGS after stochastic phase
    t6 = min(t0 + 850, deadline - 150)
    if time.time() < t6 - 10:
        best_h, best_c = run_lbfgs(best_h, n, t6,
            [(5000, 500.0), (20000, 5000.0), (100000, 50000.0), (500000, 200000.0)])
    
    # Phase 7: Targeted gradient search
    t7 = min(t0 + 1000, deadline - 80)
    if time.time() < t7 - 10:
        best_h, best_c = targeted_gradient_search(best_h, n, t7)
    
    # Phase 8: Fine local search
    t8 = min(t0 + 1080, deadline - 20)
    if time.time() < t8 - 5:
        best_h, best_c = fine_local_search(best_h, n, t8)
    
    # Phase 9: Final L-BFGS polish
    if time.time() < deadline - 5:
        best_h, best_c = run_lbfgs(best_h, n, deadline - 2,
            [(20000, 2000.0), (100000, 20000.0)])
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    
    return best_h, best_c, n
