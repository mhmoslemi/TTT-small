def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def binarize(h, n):
    target = n // 2
    h_bin = np.zeros(n)
    # Sort by value, set top n/2 to 1
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    return h_bin

def energy_smoothmax(c, alpha):
    """Smooth max of correlation values (excluding lag 0)"""
    # c is the full correlation vector (length 2n-1)
    # Exclude the middle element (lag 0)
    n = len(c)
    mid = n // 2
    c_nonzero = np.concatenate([c[:mid], c[mid+1:]])
    m = c_nonzero.max()
    if m == 0:
        return 0.0
    sh = alpha * (c_nonzero - m)
    smx = sh.max()
    return m + np.log(np.sum(np.exp(sh - smx)) + 1e-300) / alpha

            h[zeros[:int(target - cnt)]] = 1.0
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h

def make_random_initial(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    positions = rng.choice(n, size=target, replace=False)
    h[positions] = 1.0
    return h

def make_parent_scaled(h_parent, n_parent, n_new, seed):
    """Scale parent solution to new resolution by interpolation and binarization."""
    x_old = np.linspace(0, 2, n_parent, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_parent)
    return binarize(h_interp, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)
    random.seed(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500
    
    # Binarize parent
    h_bin = binarize(h_parent, n_parent)
    best_c = compute_c5(h_bin, n_parent)
    best_h = h_bin.copy()
    best_n = n_parent
    
    # Phase 1: Try multiple resolutions with quick initialization (60s)
    p1_deadline = min(t0 + 60, deadline - 600)
    resolutions_to_try = [n_parent]
    for alt_n in [400, 600, 700]:
        if alt_n != n_parent and alt_n % 2 == 0:
            resolutions_to_try.append(alt_n)
    
    init_candidates = []
    for n_try in resolutions_to_try:
        if time.time() > p1_deadline:
            break
        if n_try == n_parent:
            h_init = h_bin.copy()
        else:
            h_init = make_parent_scaled(h_parent, n_parent, n_try, seed)
        
        # Quick local search (10s)
        sub_dl = min(time.time() + 10, p1_deadline)
        h_ls, c_ls = local_search_guided(h_init, n_try, sub_dl, rng, max_iters=20000)
        init_candidates.append((c_ls, h_ls, n_try))
        
        # Also try arithmetic initial
        h_arith = make_arithmetic_initial(n_try, seed)
        c_arith = compute_c5(h_arith, n_try)
        if c_arith < c_ls:
            sub_dl2 = min(time.time() + 5, p1_deadline)
            h_arith2, c_arith2 = local_search_guided(h_arith, n_try, sub_dl2, rng, max_iters=10000)
            init_candidates.append((c_arith2, h_arith2, n_try))
    
    # Sort by quality
    init_candidates.sort(key=lambda x: x[0])
    best_c = init_candidates[0][0]
    best_h = init_candidates[0][1].copy()
    best_n = init_candidates[0][2]
    
    # Phase 2: Main SA (600s)
    p2_deadline = min(t0 + 660, deadline - 200)
    if time.time() < p2_deadline:
        sa_dl = p2_deadline
        h_sa, c_sa = simulated_annealing_main(best_h, best_n, sa_dl, rng,
                                              alpha_start=3.0, alpha_end=80.0,
                                              t0=0.5, t1=0.005)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa
    
    # Phase 3: Local search refinement (150s)
    p3_deadline = min(t0 + 810, deadline - 130)
    if time.time() < p3_deadline:
        h_ls, c_ls = local_search_guided(best_h, best_n, p3_deadline, rng, max_iters=200000)
        if c_ls < best_c:
            best_c = c_ls
            best_h = h_ls
    
    # Phase 4: 2-swap search (100s)
    p4_deadline = min(t0 + 910, deadline - 80)
    if time.time() < p4_deadline:
        h_2s, c_2s = two_swap_search(best_h, best_n, p4_deadline, rng, max_iters=50000)
        if c_2s < best_c:
            best_c = c_2s
            best_h = h_2s
    
    # Phase 5: Another SA round from improved solution (100s)
    p5_deadline = min(t0 + 1010, deadline - 40)
    if time.time() < p5_deadline:
        rng2 = np.random.RandomState(seed + 1000)
        h_sa2, c_sa2 = simulated_annealing_main(best_h, best_n, p5_deadline, rng2,
                                                alpha_start=20.0, alpha_end=200.0,
                                                t0=0.1, t1=0.01)
        if c_sa2 < best_c:
            best_c = c_sa2
            best_h = h_sa2
    
    # Phase 6: Final local search (30s)
    if time.time() < deadline - 5:
        h_final, c_final = local_search_guided(best_h, best_n, deadline - 3, rng, max_iters=50000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Verify and return
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact constraint
    target_sum = best_n / 2.0
    current_sum = best_h.sum()
    if abs(current_sum - target_sum) > 1e-10:
        # Fix by adjusting
        if current_sum < target_sum:
            zeros = np.where(best_h < 0.5)[0]
            if len(zeros) > 0:
                best_h[zeros[0]] = 1.0
        else:
            ones = np.where(best_h > 0.5)[0]
            if len(ones) > 0:
                best_h[ones[0]] = 0.0
    
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
