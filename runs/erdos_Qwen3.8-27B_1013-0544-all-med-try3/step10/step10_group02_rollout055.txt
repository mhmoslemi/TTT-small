import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def sm_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sm_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 500:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])

def push_binary(h, n, strength=0.1):

    h = h.copy()

        for idx in mid:

                push_dir = 1.0
            else:
                push_dir = 0.0
            step = strength * (push_dir - h[idx])
            # Need to find a partner to maintain sum
            # Adjust all other values proportionally
            old_val = h[idx]
            h[idx] = np.clip(old_val + step, 0.0, 1.0)
            diff = target - h.sum()
            others = np.where(np.arange(n) != idx)[0]

                if tot > 1e-15:
                    h[others] += (diff / tot) * room_dn
            h = np.clip(h, 0.0, 1.0)
            changed = True
    return project_feasible(h, n)

            mu=0.01, maxiter_per=[1500, 2000, 2500, 3000, 3000, 3000])
        if c1 < best_c:
            best_c, best_h, best_n = c1, h1, best_n

    # Phase 2: Active-set at original resolution (60s)
    p2_end = t0 + 120
    if time.time() < p2_end:
        h2, c2 = active_phase(best_h, best_n, p2_end, top_k=4,
            alphas=[50000, 200000, 1000000, 5000000, 20000000])
        if c2 < best_c:
            best_c, best_h = c2, h2
        best_n = len(best_h)

    # Phase 3: Pair local search at original resolution (80s)
    p3_end = t0 + 200
    if time.time() < p3_end:
        h3, c3 = pair_local_search(best_h, best_n, p3_end, radius=min(best_n // 4, 150))
        if c3 < best_c:
            best_c, best_h = c3, h3
        best_n = len(best_h)

    # Phase 4: Upsample to n=1000 and optimize (250s)
    n_hi = 1000
    p4_end = t0 + 450
    if time.time() < p4_end - 30:
        h_hi = upsample_h(best_h, len(best_h), n_hi)
        c_hi = compute_c5(h_hi, n_hi)
        # L-BFGS at high res
        h4, c4 = lbfgs_phase(h_hi, n_hi, min(time.time() + 150, p4_end - 80),
            alphas=[10, 50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.005, maxiter_per=[1000, 1500, 1500, 2000, 2000, 2000, 2500, 2500])
        if c4 < best_c:
            best_c, best_h, best_n = c4, h4, n_hi
        # Active set at high res
        if time.time() < p4_end - 30:
            h4b, c4b = active_phase(best_h, best_n, p4_end - 10, top_k=4,
                alphas=[100000, 500000, 2000000, 10000000, 50000000])
            if c4b < best_c:
                best_c, best_h = c4b, h4b

    # Phase 5: Pair local search at high resolution (200s)
    p5_end = t0 + 650
    if time.time() < p5_end:
        h5, c5 = pair_local_search(best_h, best_n, p5_end, radius=min(best_n // 5, 100))
        if c5 < best_c:
            best_c, best_h = c5, h5

    # Phase 6: Block swap search (80s)
    p6_end = t0 + 730
    if time.time() < p6_end:
        h6, c6 = block_swap_search(best_h, best_n, p6_end, block_sizes=[2,3,4,5,6,8,12,16])
        if c6 < best_c:
            best_c, best_h = c6, h6

    # Phase 7: SA pair search (80s)
    p7_end = t0 + 810
    if time.time() < p7_end:
        h7, c7 = sa_pair_search(best_h, best_n, p7_end, T_init=0.003, T_final=0.000001)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Try even higher resolution n=1500 (100s)
    n_very_hi = 1500
    p8_end = t0 + 910
    if time.time() < p8_end - 20:
        h_vh = upsample_h(best_h, len(best_h), n_very_hi)
        c_vh = compute_c5(h_vh, n_very_hi)
        if c_vh >= best_c - 0.001:
            h8, c8 = lbfgs_phase(h_vh, n_very_hi, min(time.time() + 60, p8_end - 30),
                alphas=[10, 100, 1000, 10000, 100000, 1000000],
                mu=0.003, maxiter_per=[800, 1000, 1000, 1500, 1500, 1500])
            if c8 < best_c:
                best_c, best_h, best_n = c8, h8, n_very_hi

    # Phase 9: Final pair search (60s)
    p9_end = t0 + 970
    if time.time() < p9_end:
        h9, c9 = pair_local_search(best_h, best_n, p9_end, radius=min(best_n // 6, 80))
        if c9 < best_c:
            best_c, best_h = c9, h3 if len(h3) == len(best_h) else h9
        else:
            pass
        # Re-check
        h9 = project_feasible(h9, len(h9))
        c9 = compute_c5(h9, len(h9))
        if c9 < best_c:
            best_c, best_h = c9, h9

    # Phase 10: SLSQP polish (30s)
    p10_end = t0 + 1000
    if time.time() < p10_end:
        for alpha in [1000000, 5000000]:
            if time.time() > p10_end - 5:
                break
            h10, c10 = slqp_polish(best_h, best_n, p10_end - 2, alpha=alpha, maxiter=500)
            if c10 < best_c:
                best_c, best_h = c10, h10

    # Phase 11: Final tiny pair search (15s)
    if time.time() < deadline:
        h11, c11 = pair_local_search(best_h, best_n, deadline, radius=min(best_n // 8, 50))
        if c11 < best_c:
            best_c, best_h = c11, h11

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
