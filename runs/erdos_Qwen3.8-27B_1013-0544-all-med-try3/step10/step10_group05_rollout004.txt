    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def build_lp_for_active_lags(h, n, active_lags):
    """
    Build the LP for the linearized active-set minimax problem.
    
    Variables: [δh[0], ..., δh[n-1], t] (n+1 variables)
    Minimize: t
    Subject to:
      For each active lag m: c0[m] + sum_j g(j, m) * δh[j] ≤ t
      -h[j] ≤ δh[j] ≤ 1-h[j]
      sum δh[j] = 0 (maintain feasibility)
    """
    # Compute current c values
    c0 = compute_c5_vec(h, n)
    dx = 2.0 / n
    
    # For each active lag, compute the gradient of c[m] w.r.t. h
    # c[m] = sum_i h[i]*(1-h[i+m-n+1]) * dx (approximately, for the valid range)
    # Actually, let's compute the gradient directly
    n_full = 2 * n - 1
    grad = np.zeros((len(active_lags), n))
    
    for a_idx, m in enumerate(active_lags):
        k = m - (n - 1)
        if k >= 0:
            # c[m] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            nv = n - k
            # d/dh[i] = (1-h[i+k]) for i < nv
            # d/dh[i+k] = -h[i] for i+k → j, so d/dh[j] = -h[j-k]
            grad[a_idx, 0:nv] += (1.0 - h[k:n])
            grad[a_idx, k:n] += (-h[0:nv])
        else:
            kk = -k
            # c[m] = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk])
            nv = n - kk
            # d/dh[i] = (1-h[i-kk]) for i ≥ kk
            # d/dh[i-kk] = -h[i] → d/dh[j] = -h[j+kk] for j ≤ n-1-kk
            grad[a_idx, kk:n] += (1.0 - h[0:nv])
            grad[a_idx, 0:nv] += (-h[kk:n])
    
    grad *= dx  # Scale by dx since c includes the dx factor
    
    c_vals = c0[active_lags]
    
    # Build LP: minimize t
    # Variables: [δh(0), ..., δh(n-1), t]
    # c变量: t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag a: sum_j grad[a,j]*δh[j] - t ≤ -c_vals[a]
    # i.e., grad[a,:] @ δh - 1*t ≤ -c_vals[a]
    A_ub = np.zeros((len(active_lags), n + 1))
    b_ub = np.zeros(len(active_lags))
    for a in range(len(active_lags)):
        A_ub[a, 0:n] = grad[a, :]
        A_ub[a, n] = -1.0
        b_ub[a] = -c_vals[a]
    
    # Box constraints on δh: -h[j] ≤ δh[j] ≤ 1-h[j]
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((None, None))  # t is free (but will be bounded by constraints)
    
    # Equality constraint: sum δh[j] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    
    return A_ub, b_ub, A_eq, b_eq, bounds, c_obj

def solve_lp_minimax(h, n, active_lags):
    """Solve the LP and return the step direction."""
    A_ub, b_ub, A_eq, b_eq, bounds, c_obj = build_lp_for_active_lags(h, n, active_lags)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            delta_h = res.x[0:n]
            t_new = res.x[n]
            return delta_h, t_new
        return None, None
    except Exception:
        return None, None

def line_search(h, n, delta_h, max_iter=50):
    """Find best step size α in [0,1] to minimize max(c(h + α*δh))."""
    c0 = compute_c5(h, n)
    best_alpha = 0.0
    best_c = c0
    
    alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
    for alpha in alphas:
        h_test = h + alpha * delta_h
        h_test = np.clip(h_test, 0.0, 1.0)
        # Project to maintain sum constraint
        diff = n/2.0 - h_test.sum()
        if abs(diff) > 1e-10:
            # Simple projection
            room = 1.0 - h_test
            h_test += diff * room / room.sum() if diff > 0 else diff * h_test / h_test.sum()
            h_test = np.clip(h_test, 0.0, 1.0)
        c_test = compute_c5(h_test, n)
        if c_test < best_c:
            best_c = c_test
            best_alpha = alpha
    
    # Refine around best
    if best_alpha > 0:
        for frac in [0.5, 0.75, 1.25, 1.5]:
            a = best_alpha * frac
            if 0 < a <= 1:
                h_test = h + a * delta_h
                h_test = project_feasible(h_test, n)
                c_test = compute_c5(h_test, n)
                if c_test < best_c:
                    best_c = c_test
                    best_alpha = a
    
    best_alpha = min(best_alpha, 1.0)
    h_new = project_feasible(h + best_alpha * delta_h, n)
    return h_new, compute_c5(h_new, n), best_alpha

def lp_active_set_optimize(h, n, deadline, top_k=20, max_iters=200):
    """Main LP-based active set optimization."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Active set: lags within tolerance of max
        threshold = max_val - max(1e-9, 0.0001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        
        if len(active) > top_k:
            # Keep top k
            sorted_active = np.argsort(-c_vec[active])[:top_k]
            active = active[sorted_active]
        
        if len(active) == 0:
            active = [n - 1]  # At minimum, include k=0
        
        # Solve LP
        delta_h, t_new = solve_lp_minimax(best_h, n, active)
        if delta_h is None:
            break
        
        # Check if step is meaningful
        if np.abs(delta_h).max() < 1e-12:
            break
        
        # Line search
        h_new, c_new, alpha = line_search(best_h, n, delta_h)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
        else:
            # Try smaller active set or break
            break
    
    return best_h, best_c

def coordinate_descent_polish(h, n, deadline, max_rounds=100):
    """Polish using coordinate descent on individual points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for rd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        for i in range(n):
            if time.time() > deadline - 3:
                break
            # Try adjusting h[i] slightly
            for delta in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                if best_h[i] + delta < 0 or best_h[i] + delta > 1:
                    continue
                # To maintain sum, adjust another point
                # Simple: adjust h[(i+1)%n] in opposite direction
                j = (i + 1) % n
                if best_h[j] - delta < 0 or best_h[j] - delta > 1:
                    continue
                best_h[i] += delta
                best_h[j] -= delta
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                else:
                    best_h[i] -= delta
                    best_h[j] += delta
                if c < best_c + 1e-15:
                    break
        
        if not improved:
            break
    
    return best_h, best_c
