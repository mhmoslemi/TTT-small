def corr_fft(h):
    n = len(h)
    m = 1.0 - h
    N = 1
    while N < 2*n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    Mr = np.fft.rfft(m[::-1], N)
    return np.fft.irfft(H * Mr, N)[:2*n-1] * (2.0/n)

def c5_val(h):
    return float(corr_fft(h).max())

def project(h, n):

    for d in deltas:

        while improved and time.time() < deadline:
