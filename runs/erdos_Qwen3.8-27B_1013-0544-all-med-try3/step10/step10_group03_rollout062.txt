def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-15:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def coord_line_search(h, n, i, alpha, mu):

def coord_descent(h, n, alpha, mu, max_sweeps=20, deadline=None):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    omx = 1.0 - x
    for mm in range(2 * n - 1):
        wm = w[mm]
        if wm < 1e-14:
            continue
        k = mm - (n - 1)
        wm_dx = wm * dx
        if k >= 0:
            nv = n - k
            grad[:nv] -= wm_dx * x[k:]
            grad[k:] += wm_dx * omx[:nv]
        else:
            t = -k
            nv = n - t
            grad[:nv] += wm_dx * omx[t:]
            grad[t:] -= wm_dx * x[:nv]
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_grad(x, n, active_indices):
    dx = 2.0 / n
    omx = 1.0 - x
    grad = np.zeros(n)
    wavg = 1.0 / len(active_indices)
    for mm in active_indices:
        k = mm - (n - 1)
        wm_dx = wavg * dx
        if k >= 0:
            nv = n - k
            grad[:nv] -= wm_dx * x[k:]
            grad[k:] += wm_dx * omx[:nv]
        else:
            t = -k
            nv = n - t
            grad[:nv] += wm_dx * omx[t:]
            grad[t:] -= wm_dx * x[:nv]
    return grad

def find_active(c_vec, n_frac_width=0.005, max_active=20):
    c_max = c_vec.max()
    threshold = c_max - max(1e-8, n_frac_width * c_max)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > max_active:
        order = np.argsort(-c_vec[active])
        active = active[order[:max_active]]
    return active

def lbfgs_stage(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad,
                      args=(n, alpha, mu), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-22,
                               'gtol': 1e-16, 'maxfun': maxiter * 5, 'maxls': 80})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()

def coord_descent_fast(h, n, alpha, mu, deadline, max_sweeps=8):
    best_c = compute_c5(h, n)
    best_smooth = smooth_obj(h, n, alpha, mu)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            old_val = h[i]
            lo, hi = 0.0, 1.0
            best_t = old_val
            best_f = smooth_obj(h, n, alpha, mu)
            for _ in range(30):
                if hi - lo < 1e-11:
                    break
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h[i] = m1
                f1 = smooth_obj(h, n, alpha, mu)
                h[i] = m2
                f2 = smooth_obj(h, n, alpha, mu)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if f1 < best_f:
                    best_f = f1
                    best_t = m1
                if f2 < best_f:
                    best_f = f2
                    best_t = m2
            h[i] = best_t
            if best_f < best_smooth - 1e-14:
                any_change = True
            if abs(best_t - old_val) > 1e-9:
                any_change = True
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_smooth = smooth_obj(h, n, alpha, mu)
        if not any_change:
            break
    return h, best_c

def subgradient_step(h, n, step_size, deadline):
    c_vec = compute_c5_vec(h, n)
    active = find_active(c_vec, 0.002, 10)
    if len(active) == 0:
        return h
    grad = active_set_grad(h, n, active)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgradient_descent(h, n, deadline, max_iter=5000):
    best_c = compute_c5(h, n)
    step = 0.001
    n_no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h = subgradient_step(h, n, step, deadline)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            n_no_improve = 0
        else:
            n_no_improve += 1
            if n_no_improve > 200:
                step *= 0.7
                if step < 1e-8:
                    break
                n_no_improve = 0
    return h, best_c

def pair_fine_search(h, n, deadline, top_radius=40):
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(h, n)
    critical = np.argsort(-c_vec)[:20]
    centers = set()
    for k in critical:
        s = k - (n - 1)
        for i in range(n):
            if abs(i + s) < n:
                centers.add(i)
    centers = list(centers)
    if len(centers) > 200:
        centers = centers[:200]
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for i in centers:
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - top_radius)
                hi = min(n, i + top_radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c
