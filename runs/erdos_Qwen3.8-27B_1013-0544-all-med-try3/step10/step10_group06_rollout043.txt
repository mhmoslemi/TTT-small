```python
def compute_smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    # Compute correlation
    c = fft_correlate(h, 1-h)  # length 2n-1
    # Compute softmax weights
    shifted = alpha * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += w[k] * (1 - 2*h)
        elif lag > 0:
            grad[:n-lag] += w[k] * (1 - h[lag:])
            grad[lag:] += w[k] * (-h[:-lag])
        else:
            idx = -lag
            grad[idx:] += w[k] * (1 - h[:-idx])
            grad[:n-idx] += w[k] * (-h[idx:])
    return grad * dx
```

```python
def fft_correlate(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT for speed."""
    n = len(a)
    m = len(b)
    n_fft = n + m - 1
    n_pow = 1 << (n_fft - 1).bit_length()
    A = np.fft.rfft(a, n_pow)
    B = np.fft.rfft(b, n_pow)
    result = np.fft.irfft(A * np.conj(B), n_pow)[:n_fft]
    return result

def compute_all_lags_fft(h, n):
    """Compute all correlation lags using FFT. Returns array of length 2n-1."""
    c = fft_correlate(h, 1.0 - h)
    return c

def compute_c5(h, n):
    dx = 2.0 / n
    c = compute_all_lags_fft(h, n)
    return float(np.max(c) * dx)

def compute_smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = w / w.sum()
    return w

def compute_smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags_fft(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w[k] * (1.0 - h[lag:])
            grad[lag:] += w[k] * (-h[:-lag])
        else:
            idx = -lag
            grad[idx:] += w[k] * (1.0 - h[:-idx])
            grad[:n - idx] += w[k] * (-h[idx:])
    return grad * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def nesterov_optimize(h, n, alpha, deadline, maxiter=5000, beta=0.9, lr_init=0.02):
    """Nesterov accelerated gradient descent on smooth-max objective."""
    target_sum = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    y = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        # Nesterov lookahead
        y = h + beta * v
        
        # Compute gradient at y
        y_clipped = np.clip(y, 0.0, 1.0)
        grad = compute_smooth_max_grad(y_clipped, n, alpha)
        
        # Add gradient penalty for constraint violation
        y_for_sum = np.clip(y, 0.0, 1.0)
        grad += 50.0 * (y_for_sum.sum() - target_sum) * np.ones(n)
        
        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        lr = lr_init / (1.0 + 0.1 * it)
        
        # Update
        v = beta * v - lr * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags_fft(x, n)
        obj_val = compute_smooth_max(c, alpha)
        diff = x.sum() - target_sum
        obj_val += 1e6 * diff * diff
        grad = compute_smooth_max_grad(x, n, alpha)
        grad += 2.0e6 * diff * np.ones(n)
        return obj_val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 50,
                                'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def upscale(h_src, n_src, n_dst):
    """Upscale h from n_src to n_dst points using linear interpolation."""
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def lag_coord_descent(h, n, deadline, rng, n_lags=5, n_trials=2000):
    """Coordinate descent focusing on the worst lags."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    iters = 0
    
    while time.time() < deadline and iters < n_trials:
        iters += 1
        c_all = compute_all_lags_fft(h, n)
        c_max = c_all.max()
        
        if c_max * (2.0 / n) < best_c - 1e-12:
            best_c = float(c_max * (2.0 / n))
            best_h = h.copy()
        
        # Get top lags
        top_idx = np.argsort(c_all)[-n_lags:]
        k_star = top_idx[rng.randint(n_lags)]
        lag = k_star - (n - 1)
        if lag == 0:
            continue
        m = abs(lag)
        if m >= n:
            continue
        
        # Find best pair to swap
        if lag > 0:
            # overlap = sum h[i]*(1-h[i+m]) for i=0..n-m-1
            # To reduce: decrease h[i] where h[i+m] is small (less penalty for reducing h[i])
            # or increase h[i+m] where h[i] is small
            # Best move: find i where h[i] is high and h[i+m] is high (they contribute little to overlap)
            # Actually: to reduce sum h[i](1-h[i+m]), we want to decrease h[i]*(1-h[i+m])
            # Move mass from i to i+m if h[i](1-h[i+m]) > h[i+m](1-h[i])
            # i.e., h[i] - h[i]h[i+m] > h[i+m] - h[i]h[i+m], i.e., h[i] > h[i+m]
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-8:
                d = min(0.05, (h[i] - h[j]) * 0.3)
                h[i] -= d
                h[j] += d
                h = project_feasible(h, n)
                new_c = compute_c5(h, n)
                if new_c > best_c + 1e-6:
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, n)
            elif h[j] > h[i] + 1e-8:
                d = min(0.05, (h[j] - h[i]) * 0.3)
                h[i] += d
                h[j] -= d
                h = project_feasible(h, n)
                new_c = compute_c5(h, n)
                if new_c > best_c + 1e-6:
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, n)
        else:
            idx = -lag
            i = rng.randint(0, n - idx)
            j = i + idx
            if h[j] > h[i] + 1e-8:
                d = min(0.05, (h[j] - h[i]) * 0.3)
                h[j] -= d
                h[i] += d
                h = project_feasible(h, n)
                new_c = compute_c5(h, n)
                if new_c > best_c + 1e-6:
                    h[j] += d
                    h[i] -= d
                    h = project_feasible(h, n)
            elif h[i] > h[j] + 1e-8:
                d = min(0.05, (h[i] - h[j]) * 0.3)
                h[j] += d
                h[i] -= d
                h = project_feasible(h, n)
                new_c = compute_c5(h, n)
                if new_c > best_c + 1e-6:
                    h[j] -= d
                    h[i] += d
                    h = project_feasible(h, n)
    
    final_c = compute_c5(best_h, n)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c

def multi_lag_coord_descent(h, n, deadline, rng):
    """More aggressive coordinate descent trying multiple lags per iteration."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        c_all = compute_all_lags_fft(h, n)
        c_max = np.max(c_all)
        current_c = c_max * (2.0 / n)
        if current_c < best_c - 1e-14:
            best_c = current_c
            best_h = h.copy()
        
        # Focus on top 10 lags
        top_idx = np.argsort(c_all)[-10:]
        improved = False
        
        for k_star in top_idx:
            if time.time() > deadline:
                break
            lag = k_star - (n - 1)
            if lag == 0:
                continue
            m = abs(lag)
            if m >= n:
                continue
            
            # Try multiple random positions for this lag
            for _ in range(50):
                if time.time() > deadline:
                    break
                i = rng.randint(0, n - m)
                j = i + m
                if lag > 0:
                    if h[i] > h[j] + 1e-7:
                        d = min(0.01, (h[i] - h[j]) * 0.2)
                        if d < 1e-9:
                            continue
                        h[i] -= d
                        h[j] += d
                        h = project_feasible(h, n)
                        new_c = compute_c5(h, n)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
                            h = project_feasible(h, n)
                    elif h[j] > h[i] + 1e-7:
                        d = min(0.01, (h[j] - h[i]) * 0.2)
                        if d < 1e-9:
                            continue
                        h[i] += d
                        h[j] -= d
                        h = project_feasible(h, n)
                        new_c = compute_c5(h, n)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                            h = project_feasible(h, n)
                else:
                    idx = -lag
                    i2 = rng.randint(0, n - idx)
                    j2 = i2 + idx
                    if h[i2] > h[j2] + 1e-7:
                        d = min(0.01, (h[i2] - h[j2]) * 0.2)
                        if d < 1e-9:
                            continue
                        h[i2] -= d
                        h[j2] += d
                        h = project_feasible(h, n)
                        new_c = compute_c5(h, n)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i2] += d
                            h[j2] -= d
                            h = project_feasible(h, n)
                    elif h[j2] > h[i2] + 1e-7:
                        d = min(0.01, (h[j2] - h[i2]) * 0.2)
                        if d < 1e-9:
                            continue
                        h[i2] += d
                        h[j2] -= d
                        h = project_feasible(h, n)
                        new_c = compute_c5(h, n)
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i2] -= d
                            h[j2] += d
                            h = project_feasible(h, n)
        
        if not improved:
            # Restart with perturbation
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
    
    return best_h, best_c

def periodic_init(T, n, rng):
    """Create a periodic initial pattern."""
    base = np.zeros(T)
    # Create a random binary-like pattern
    fill = T / 2.0
    n_ones = int(fill)
    idx = rng.choice(T, size=n_ones, replace=False)
    base[idx] = 1.0
    if fill - n_ones > 0:
        base[idx[0]] += fill - n_ones
    # Tile
    h = np.zeros(n)
    for i in range(n):
        h[i] = base[i % T]
    return project_feasible(h, n)

def simulated_annealing_fast(h, n, deadline, rng, alpha=50000, t_init=0.005, t_min=0.00001, cool=0.9999):
    """Fast simulated annealing using FFT correlation."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    
    while time.time() < deadline and T > t_min:
        c_all = compute_all_lags_fft(h, n)
        e_old = compute_smooth_max(c_all, alpha)
        
        # Random move
        r = rng.random()
        if r < 0.7:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            max_step = min(0.08, 0.005 + T * 5)
            if h[i] > h[j]:
                d = rng.uniform(0.002, max_step)
                d = min(d, h[i] * 0.3, (1.0 - h[j]) * 0.3)
            elif h[j] > h[i]:
                d = -rng.uniform(0.002, max_step)
                d = max(d, -h[j] * 0.3, -(1.0 - h[i]) * 0.3)
            else:
                T *= cool
                continue
        else:
            # Structured move at worst lag
            k_star = int(np.argmax(c_all))
            lag = k_star - (n - 1)
            if lag == 0:
                T *= cool
                continue
            m = abs(lag)
            if m >= n:
                T *= cool
                continue
            i = rng.randint(0, n - m)
            j = i + m
            if h[i] > h[j] + 1e-6:
                d = rng.uniform(0.002, 0.03)
                d = min(d, (h[i] - h[j]) * 0.3)
            elif h[j] > h[i] + 1e-6:
                d = -rng.uniform(0.002, 0.03)
                d = max(d, -(h[j] - h[i]) * 0.3)
            else:
                T *= cool
                continue
        
        if abs(d) < 1e-10:
            T *= cool
            continue
        
        h[i] -= d
        h[j] += d
        h = project_feasible(h, n)
        
        c_new = compute_all_lags_fft(h, n)
        e_new = compute_smooth_max(c_new, alpha)
        max_new = float(c_new.max() * (2.0 / n))
        
        de = e_new - e_old
        accept = de < 0
        if not accept and T > t_min:
            ratio = -de / T
            if ratio < 50:
                accept = rng.random() < np.exp(-ratio)
        
        if accept:
            if max_new < best_c - 1e-14:
                best_c = max_new
                best_h = h.copy()
        else:
            h[i] += d
            h[j] -= d
            h = project_feasible(h, n)
        
        T *= cool
    
    return best_h, best_c

                                               deadline=deadline - 1, rng)

            best_c = c_opt
            best_h = h_opt
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    
    return best_h, best_c, len(best_h)
```
