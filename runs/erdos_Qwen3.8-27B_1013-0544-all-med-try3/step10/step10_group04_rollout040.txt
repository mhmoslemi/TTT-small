    target = n / 2.0

            if tot < 1e-14:

def fft_obj(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)

    if mx > 500:

    else:
        obj = m + np.log(np.sum(np.exp(sh))) / alpha
    obj += mu * (h.sum() - n / 2.0) ** 2
    return float(obj)

def fft_grad(h, n, alpha, mu):

    c = np.correlate(h, 1 - h, mode='full') * dx

    sh = alpha * (c - m)

    w_scaled = w * dx
    
    center = n - 1
    A_pos = w_scaled[center : 2*n - 1]  # length n, A_pos[k] for lag k>=0
    B_neg = np.zeros(n)
