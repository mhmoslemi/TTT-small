def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary sequence with exactly n/2 ones."""
    b = (h > 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(b.sum())
    target = n // 2
    if n_ones < target:
        # Need to add (target - n_ones) ones: pick zeros with highest h value
        zeros = np.where(b == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for i in range(target - n_ones):
            b[zeros[i]] = 1.0
    elif n_ones > target:
        # Need to remove (n_ones - target) ones: pick ones with lowest h value
        ones = np.where(b == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for i in range(n_ones - target):
            b[ones[i]] = 0.0
    return b

def binary_sa(h_binary, n, deadline, iterations=100000):
    best = h_binary.copy()
    best_c = compute_c5(best, n)
    current = h_binary.copy()
    current_c = best_c
    T = 0.01  # initial temperature
    
    for it in range(iterations):
        if time.time() > deadline:
            break
        
        # Pick a random swap: one 1 and one 0
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Make swap
        current[i], current[j] = current[j], current[i]
        new_c = compute_c5(current, n)
        
        if new_c < current_c or np.random.random() < np.exp((current_c - new_c) / T):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = current.copy()
        else:
            current[i], current[j] = current[i], current[j]  # revert
        
        # Cool
        T *= 0.99999
        if T < 0.001:
            T = 0.001
    
    return best, best_c

        # New: h[p+m] * (1-h[p]) = h[p+m] * 1 = h[p+m]
        # Change: +h[p+m]
        
        # Terms where i+m = q: i = q-m
        # Old: h[q] * (1-h[q-m]) = 0*(1-h[q-m]) = 0
        # New: h[q] * (1-h[q-m]) = 1*(1-h[q-m]) = 1-h[q-m]
        # Change: +(1-h[q-m])

        # Old: h[q+m] * (1-h[q]) = h[q+m] * 1 = h[q+m]
        # New: h[q+m] * (1-h[q]) = h[q+m] * 0 = 0
        # Change: -h[q+m]
        
        # Total change for c[n-1+m] (m>0, all indices valid):
        # = -(1-h[p-m]) + h[p+m] + (1-h[q-m]) - h[q+m]
        # = h[p+m] - 1 + h[p-m] + 1 - h[q-m] - h[q+m]
        # = h[p+m] + h[p-m] - h[q+m] - h[q-m]
        
        # Same for c[n-1-m] (m>0):
        # c[n-1-m] = sum_i h[i] * (1-h[i+m])
        # Terms where i = p: h[p]*(1-h[p+m]) changes from 1*(1-h[p+m]) to 0*(1-h[p+m])
        #   Change: -(1-h[p+m])
        # Terms where i+m = p: i = p-m: h[p-m]*(1-h[p]) changes from h[p-m]*0 to h[p-m]*1
        #   Change: +h[p-m]
        # Terms where i = q: h[q]*(1-h[q+m]) changes from 0*(1-h[q+m]) to 1*(1-h[q+m])
        #   Change: +(1-h[q+m])
        # Terms where i+m = q: i = q-m: h[q-m]*(1-h[q]) changes from h[q-m]*1 to h[q-m]*0
        #   Change: -h[q-m]
        # Total: -(1-h[p+m]) + h[p-m] + (1-h[q+m]) - h[q-m]
        # = -1 + h[p+m] + h[p-m] + 1 - h[q+m] - h[q-m]
        # = h[p+m] + h[p-m] - h[q+m] - h[q-m]
        
        # Same formula! Great.
        
        # For m = 0:
        # c[n-1] = sum_i h[i]*(1-h[i])
        # h[p] changes 1->0: term changes from 0 to 0 (1*0=0, 0*1=0)
        # h[q] changes 0->1: term changes from 0 to 0 (0*1=0, 1*0=0)
        # So c[n-1] doesn't change! Makes sense for binary.
        
        # Now compute the total change
        # We need to check validity for each m
        # For c[n-1+m], m ranges from 1 to n-1:
        #   Need: 0 <= p-m <= n-1-m (for i=p-m term) -> p-m >= 0 and p-m <= n-1-m -> p >= m and p <= n-1
        #   Need: 0 <= p+m <= n-1 (for i=p term) -> p+m <= n-1 -> p <= n-1-m
        #   Similarly for q.
        
        # This is getting complex. Let me just compute the change directly for each valid m.
        
        # Actually, for n=500, even computing the full correlation each time is only 500*500 = 250K mults.
        # In Python with numpy, this should take < 1ms.
        # So 100K iterations would take ~100s. That's feasible!

        h[p], h[q] = h[q], h[p]
        c_new = np.correlate(h, 1-h, mode='full')
        max_new = c_new.max()
        
        if max_new < best_c_max:
            best_c_max = max_new
            best_h = h.copy()
        
        if max_new <= c.max() or rng.random() < np.exp((c.max() - max_new) / T):
            c = c_new
        else:
            h[p], h[q] = h[q], h[p]  # revert
        
        T = max(0.001, T * cooling)
    
    return best_h, best_c_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = (h > 0.5).astype(float)
    n_ones = int(b.sum())
    target = n // 2
    if n_ones < target:
        zeros = np.where(b == 0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        for i in range(target - n_ones):
            b[zeros[i]] = 1.0
    elif n_ones > target:
        ones = np.where(b == 1)[0]
        ones = ones[np.argsort(h[ones])]
        for i in range(n_ones - target):
            b[ones[i]] = 0.0
    return b

def binary_sa(h_binary, n, deadline, rng_seed=42):
    h = h_binary.copy()
    c = np.correlate(h, 1-h, mode='full')
    best_c_max = c.max()
    best_h = h.copy()
    rng = np.random.default_rng(rng_seed)
    T = 0.03
    no_improve = 0
    iter_count = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[rng.integers(len(ones))]
        q = zeros[rng.integers(len(zeros))]
        if p == q:
            continue
        
        h[p], h[q] = 0.0, 1.0
        c_new = np.correlate(h, 1-h, mode='full')
        max_new = c_new.max()
        max_old = c.max()
        
        if max_new < best_c_max:
            best_c_max = max_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if max_new <= max_old or rng.random() < np.exp((max_old - max_new) / T):
            c = c_new
        else:
            h[p], h[q] = 1.0, 0.0
        
        T = max(0.002, T * 0.99999)
        iter_count += 1
        
        if no_improve > 5000:
            # Restart
            h = best_h.copy()
            c = np.correlate(h, 1-h, mode='full')
            T = 0.03
            no_improve = 0
    
    return best_h, best_c_max

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):

    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 5.0), (500, 50.0), (2000, 500.0),
                   (10000, 5000.0), (50000, 50000.0), (200000, 500000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 20000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c
