import time

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

            if tot < 1e-14:

            if tot < 1e-14:

            h += (diff / tot) * h

def coord_descend(h, n, deadline, n_sweeps=200, n_samples=32):
    """Fast coordinate descent using precomputed coefficients."""
    dx = 2.0 / n
    best_h = h.copy()
    c_vec = compute_c5_vec(best_h, n)
    best_c = float(c_vec.max())
    ns = 2 * n - 1  # length of correlation vector
    
    # Precompute t samples
    t_samples = np.linspace(0.0, 1.0, n_samples)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        any_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = best_h[i]
            
            # Compute coef vector: linear coefficient of change in c[m] per unit change in h[i]
            # For m != n-1: coef[m] = dx * [(1-h[i+k]) if role1 valid] + [-h[i-k] if role2 valid, k!=0]
            # For m == n-1: handled separately (quadratic)
            
            coef = np.zeros(ns)
            
            # Role 1: i is first factor, k = m-(n-1), valid when i+k in [0,n-1], k in [-i, n-1-i]
            m_start = max(0, n - 1 - i)
            m_end = min(ns - 1, 2 * n - 2 - i)
            if m_start <= m_end:
                for m in range(m_start, m_end + 1):
                    k = m - (n - 1)
                    partner = i + k  # this is i+k, must be in [0, n-1]
                    if 0 <= partner < n:
                        if k == 0:
                            pass  # handle separately
                        else:
                            coef[m] += (1.0 - best_h[partner]) * dx
            
            # Role 2: i is second factor, k = m-(n-1), valid when i-k in [0,n-1], k in [i-n+1, i]
            m_start2 = max(0, n - 1 + i - (n - 1))  # = i
            # Actually: k in [i-(n-1), i], m = k+(n-1) in [i, i+n-1]
            m_start2 = max(0, i)
            m_end2 = min(ns - 1, i + n - 1)
            if m_start2 <= m_end2:
                for m in range(m_start2, m_end2 + 1):
                    k = m - (n - 1)
                    partner = i - k  # this is i-k, must be in [0, n-1]
                    if 0 <= partner < n:
                        if k == 0:
                            pass  # handle separately
                        else:
                            coef[m] += (-best_h[partner]) * dx
            
            # Find best t
            best_t = old
            best_new_max = best_c
            
            for t in t_samples:
                if abs(t - old) < 1e-12:
                    continue
                # New max = max(c_vec + (t-old)*coef, with c[n-1] adjusted)
                # Efficient: compute c_vec + (t-old)*coef, then fix n-1
                new_c = c_vec + (t - old) * coef
                # Fix the k=0 term
                new_c[n-1] = c_vec[n-1] - old * (1.0 - old) * dx + t * (1.0 - t) * dx
                new_max = float(new_c.max())
                
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_t = t
            
            if best_t != old:
                best_h[i] = best_t
                c_vec = c_vec + (best_t - old) * coef
                c_vec[n-1] = c_vec[n-1] - old * (1.0 - old) * dx + best_t * (1.0 - best_t) * dx
                if best_new_max < best_c:
                    best_c = best_new_max
                any_improved = True
        
        if not any_improved:
            break
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

            best_t = old
            best_new_max = best_c
            for t in t_samples:
                if abs(t - old) < 1e-12:
                    continue
                new_max = float((c_vec + (t - old) * coef).max())
                c0_val = c_vec[n - 1] - old * (1.0 - old) * dx + t * (1.0 - t) * dx
                if c0_val > new_max:
                    new_max = c0_val
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_t = t

            if best_t != old:
                c_vec = c_vec + (best_t - old) * coef
                c_vec[n - 1] = c_vec[n - 1] - old * (1.0 - old) * dx + best_t * (1.0 - best_t) * dx
                best_h[i] = best_t
                best_c = best_new_max
                any_improved = True

        if not any_improved:
            break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def coord_descend_binary(h, n, deadline, n_sweeps=100):
    """Coordinate descent restricted to binary values {0,1}."""
    dx = 2.0 / n
    best_h = h.copy()
    c_vec = compute_c5_vec(best_h, n)
    best_c = float(c_vec.max())
    ns = 2 * n - 1

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        any_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = best_h[i]

            coef = np.zeros(ns)
            m1e = 2 * n - 2 - i
            m1s = n - 1 - i
            if m1s >= 0 and m1e < ns:
                coef[m1s:m1e + 1] += (1.0 - best_h[0:n - i]) * dx
                if m1s <= n - 1 <= m1e:
                    coef[n - 1] -= (1.0 - best_h[i]) * dx
            m2s = i
            m2e = min(i + n - 1, ns - 1)
            if m2s <= m2e:
                n_terms = m2e - m2s + 1
                partner_vals = best_h[n : n - n_terms : -1]
                coef[m2s:m2e + 1] += (-partner_vals) * dx
                if m2s <= n - 1 <= m2e:
                    coef[n - 1] -= (-best_h[i]) * dx

            best_t = old
            best_new_max = best_c
            for t in [0.0, 1.0]:
                if abs(t - old) < 1e-12:
                    continue
                new_max = float((c_vec + (t - old) * coef).max())
                c0_val = c_vec[n - 1] - old * (1.0 - old) * dx + t * (1.0 - t) * dx
                if c0_val > new_max:
                    new_max = c0_val
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_t = t

            if best_t != old:
                c_vec = c_vec + (best_t - old) * coef
                c_vec[n - 1] = c_vec[n - 1] - old * (1.0 - old) * dx + best_t * (1.0 - best_t) * dx
                best_h[i] = best_t
                best_c = best_new_max
                any_improved = True

        if not any_improved:
            break

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def lbfgs_smooth_max(h, n, deadline, alphas, mu=0.01, maxiter_per=None):
    dx = 2.0 / n

    def obj(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alphas[0] * (c - m)
        mx = sh.max()
        if mx > 500:
            val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alphas[0]
        else:
            val = m + np.log(np.sum(np.exp(sh))) / alphas[0]
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)

    def grad(x):
        c = compute_c5_vec(x, n)
        m = c.max()
        sh = alphas[0] * (c - m)
        mx = sh.max()
        w = np.exp(sh - mx)
        w /= w.sum()
        g = np.zeros(n)
        for mm in range(2 * n - 1):
            if w[mm] < 1e-14:
                continue
            k = mm - (n - 1)
            wm = w[mm] * dx
            if k >= 0:
                nv = n - k
                g[0:nv] += wm * (1.0 - x[k:n])
                g[k:n] += wm * (-x[0:nv])
            else:
                kk = -k
                nv = n - kk
                g[kk:n] += wm * (1.0 - x[0:nv])
                g[0:nv] += wm * (-x[kk:n])
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g

    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for idx, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 20:
            break
        if maxiter_per is not None and idx < len(maxiter_per):
            maxit = maxiter_per[idx]
        else:
            maxit = max(500, int(remaining * 3))

        def obj_a(x, a=alpha):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = a * (c - m)
            mx = sh.max()
            if mx > 500:
                val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / a
            else:
                val = m + np.log(np.sum(np.exp(sh))) / a
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return float(val)

        def grad_a(x, a=alpha):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = a * (c - m)
            mx = sh.max()
            w = np.exp(sh - mx)
            w /= w.sum()
            g = np.zeros(n)
            for mm in range(2 * n - 1):
                if w[mm] < 1e-14:
                    continue
                k = mm - (n - 1)
                wm = w[mm] * dx
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g

        try:
            res = minimize(obj_a, best_h, jac=grad_a, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 6,
                                   'maxls': 50, 'maxcor': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c

def sturmian_start(n, alpha=None):
    if alpha is None:
        alpha = (np.sqrt(5) - 1) / 2.0  # golden ratio conjugate ~0.618
    i = np.arange(n)
    h = (np.mod(i * alpha, 1.0) < 0.5).astype(float)
    # Adjust to get sum = n/2
    target = n // 2
    current = int(h.sum())
    if current < target:
        for j in range(current, n):
            h[j] = 1.0
            current += 1
            if current == target:
                break
    elif current > target:
        count = current - target
        for j in range(n - 1, -1, -1):
            if h[j] == 1.0 and count > 0:
                h[j] = 0.0
                count -= 1
    return project_feasible(h, n)

def random_binary_start(n, rng):
    positions = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return project_feasible(h, n)

def binary_from_continuous(h, n):
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    n_ones = n // 2
    h_bin[sorted_idx[:n_ones]] = 1.0
    return project_feasible(h_bin, n)

def add_noise(h, n, sigma, rng):
    h_noisy = h + rng.normal(0, sigma, size=n)
    return project_feasible(h_noisy, n)

def active_set_lbfgs(h, n, deadline, top_k=8, alphas=None):
    if alphas is None:
        alphas = [50000, 200000, 1000000]
    dx = 2.0 / n
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.001 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > top_k:
            active = np.array(np.argsort(-c_vec[active])[:top_k])
        if len(active) == 0:
            break

        def obj_act(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            mx = vals.max()
            sh = alpha * (vals - mx)
            smx = sh.max()
            if smx > 500:
                val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
            else:
                val = mx + np.log(np.sum(np.exp(sh))) / alpha
            val += 0.005 * (x.sum() - n / 2.0) ** 2
            return float(val)

        def grad_act(x):
            c = compute_c5_vec(x, n)
            vals = c[active]
            mx = vals.max()
            sh = alpha * (vals - mx)
            smx = sh.max()
            w = np.exp(sh - smx)
            w /= w.sum()
            g = np.zeros(n)
            for wi, idx in zip(w, active):
                k = idx - (n - 1)
                wm = wi * dx
                if k >= 0:
                    nv = n - k
                    g[0:nv] += wm * (1.0 - x[k:n])
                    g[k:n] += wm * (-x[0:nv])
                else:
                    kk = -k
                    nv = n - kk
                    g[kk:n] += wm * (1.0 - x[0:nv])
                    g[0:nv] += wm * (-x[kk:n])
            g += 2.0 * 0.005 * (x.sum() - n / 2.0)
            return g

        maxit = max(500, int(remaining * 2))
        try:
            res = minimize(obj_act, best_h, jac=grad_act, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-16, 'maxfun': maxit * 6})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: L-BFGS from parent (moderate)
    p1_end = t0 + 120
    h1, c1 = lbfgs_smooth_max(best_h, best_n, p1_end,
        alphas=[100, 1000, 10000, 100000, 500000, 2000000],
        mu=0.01, maxiter_per=[2000, 2000, 2000, 2000, 2000, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Coordinate descent (main innovation)
    if time.time() < deadline - 400:
        p2_end = min(time.time() + 250, deadline - 350)
        h2, c2 = coord_descend(best_h, best_n, p2_end, n_sweeps=300, n_samples=32)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: L-BFGS from CD result
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 100, deadline - 300)
        h3, c3 = lbfgs_smooth_max(best_h, best_n, p3_end,
            alphas=[50000, 200000, 1000000],
            mu=0.005, maxiter_per=[2000, 2000, 2000])
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: More coordinate descent
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 120, deadline - 200)
        h4, c4 = coord_descend(best_h, best_n, p4_end, n_sweeps=200, n_samples=40)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Binary coordinate descent from rounded solution
    if time.time() < deadline - 200:
        h_bin = binary_from_continuous(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        p5_end = min(time.time() + 80, deadline - 150)
        h5, c5 = coord_descend_binary(h_bin, best_n, p5_end, n_sweeps=50)
        if c5 < best_c:
            best_c = c5
            best_h = h5
            best_n = best_n
        # Refine binary solution with continuous CD
        if time.time() < deadline - 100:
            p5b_end = min(time.time() + 50, deadline - 80)
            h5b, c5b = coord_descend(h5, best_n, p5b_end, n_sweeps=100, n_samples=24)
            if c5b < best_c:
                best_c = c5b
                best_h = h5b

    # Phase 6: Active set L-BFGS
    if time.time() < deadline - 150:
        p6_end = min(time.time() + 60, deadline - 120)
        h6, c6 = active_set_lbfgs(best_h, best_n, p6_end, top_k=6,
            alphas=[100000, 500000, 2000000])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try n=1000 (upsample and optimize)
    if time.time() < deadline - 200:
        n_hi = 1000
        x_fine = np.linspace(0, 2, n_hi)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_hi)
        c_fine = compute_c5(h_fine, n_hi)

        p7_end = min(time.time() + 100, deadline - 120)
        h7, c7 = coord_descend(h_fine, n_hi, p7_end, n_sweeps=100, n_samples=24)
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = n_hi

    # Phase 8: Sturmian initialization at n=500
    if time.time() < deadline - 120:
        for alpha_try in [0.618, 0.707, 0.75, 0.65]:
            if time.time() > deadline - 80:
                break
            h_st = sturmian_start(best_n, alpha_try)
            c_st = compute_c5(h_st, best_n)
            if c_st < best_c:
                best_c = c_st
                best_h = h_st.copy()
            p8_end = min(time.time() + 30, deadline - 60)
            h8, c8 = coord_descend(h_st, best_n, p8_end, n_sweeps=50, n_samples=24)
            if c8 < best_c:
                best_c = c8
                best_h = h8

    # Phase 9: Random perturbation + CD
    if time.time() < deadline - 80:
        for sigma in [0.05, 0.1, 0.2]:
            if time.time() > deadline - 40:
                break
            h_pert = add_noise(best_h, best_n, sigma, rng)
            p9_end = min(time.time() + 25, deadline - 30)
            h9, c9 = coord_descend(h_pert, best_n, p9_end, n_sweeps=50, n_samples=24)
            if c9 < best_c:
                best_c = c9
                best_h = h9

    # Phase 10: Final coordinate descent polish
    if time.time() < deadline - 20:
        p10_end = deadline - 2
        h10, c10 = coord_descend(best_h, best_n, p10_end, n_sweeps=200, n_samples=40)
        if c10 < best_c:
            best_c = c10
            best_h = h10

    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
