def project_feasible(h, n):

            if tot < 1e-14:

        # Subgradient: average gradient of all lags at the max
        max_lags = np.where(np.abs(c_all - current_max) < 1e-8)[0]
        if len(max_lags) == 0:
            max_lags = np.array([np.argmax(c_all)])

            grad += lag_gradient(h, n, lag)
        grad /= len(max_lags)
        
        # Project gradient to maintain sum constraint
        grad -= grad.mean()
        
        # Momentum
        v = momentum * v - step * grad

        h_new = project_feasible(h + v, n)
        h = h_new
        
        # Decay step
        step *= 0.9995
    
    return best_h, best_c

                it_since_improve += 1
        
        # Check stuck
        if it_since_improve > 50000:
            h = best_h + rng.normal(0, 0.005, n)
            h = project_feasible(h, n)
            it_since_improve = 0
            T = max(T, 0.001)
        
        T *= cool
    
    return best_h, best_c

def expand_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
