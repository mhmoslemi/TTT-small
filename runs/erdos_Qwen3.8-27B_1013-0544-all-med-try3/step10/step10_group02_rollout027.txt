    target = 0.5*n
    h = np.clip(h,0,1)
    # if sum close, adjust exact
    if abs(h.sum()-target) < 1e-12:
       adjust_exact(h,target)
       return h
    # bisection for lambda
    lo = h.min() - 1.0
    hi = h.max()
    # We want sum(clip(h - lambda)) = target.
    # Since h in [0,1] before, lo min-1 <0, hi max.
    for _ in range(80):

       if s > target:
          lo = lam
       else:
          hi = lam
    lam = 0.5*(lo+hi)
    h2 = np.clip(h - lam,0,1)
    # residual adjust
    adjust_exact(h2,target)
    return h2

_FFT_LEN_CACHE = {}

def _fft_len(n):
    v = _FFT_LEN_CACHE.get(n)
    if v is None:
        m = 2 * int(n) - 1
        v = 1 << max(1, (m - 1).bit_length())
        _FFT_LEN_CACHE[int(n)] = int(v)
    return int(v)

def corr_vec_fft(h, n, N):
    h = np.ascontiguousarray(h, dtype=float)
    if N is None:
        N = _fft_len(n)
    yrev = 1.0 - h[::-1]
    a = np.fft.rfft(h, N)
    b = np.fft.rfft(yrev, N)
    c = np.fft.irfft(a * b, N)
    c = c[:2 * n - 1]
    return c * (2.0 / float(n))

def eval_fft(h, n):
    N = _fft_len(n)
    return float(np.max(corr_vec_fft(h, n, N)))

def eval_exact(h, n):
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)

    val = mx + logsum / alpha
    diff = float(np.sum(x) - 0.5 * n)
    if mu != 0.0:
        val += mu * diff * diff
    return float(val)

def fixed_smooth_gradient(x, n, active, alpha, mu, N):
    c = corr_vec_fft(x, n, N)
    vals = c[active]
    mx = float(np.max(vals))
    z = alpha * (vals - mx)
    z = np.where(z < 0.0, z, 0.0)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.full(len(active), 1.0 / max(1, len(active)))
    else:
        w = e / s

    grad = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    for t in range(len(active)):
        wi = float(w[t])
        if wi > 1.0e-14:
            add_lag_gradient(grad, x, int(active[t]), n, wi * dx)
    if mu != 0.0:
        diff = float(np.sum(x) - 0.5 * n)
        grad += 2.0 * mu * diff
    return grad

def run_active_lbfgs(h, n, deadline, alphas=None, top_k=16, mu=2.0, max_rounds=35):
    n = int(n)
    if alphas is None:
        alphas = [100.0, 300.0, 1.0e3, 3.0e3, 1.0e4, 3.0e4, 1.0e5, 3.0e5]
    N = _fft_len(n)
    bounds = [(0.0, 1.0)] * n

    work_h = project_feasible(h, n)
    work_fit = float(np.max(corr_vec_fft(work_h, n, N)))
    best_h = work_h.copy()
    if n <= 4096:
        best_c = float(eval_exact(work_h, n))
    else:
        best_c = work_fit

    exact_every = (n <= 2500)
    no_best = 0
    no_work = 0

    for rnd in range(int(max_rounds)):
        rem = float(deadline - time.time())
        if rem < 20.0:
            break

        csel = c_vec_selection(work_h, n, N)
        active = select_active_indices(csel, n, top_k)

        if rnd < len(alphas):
            alpha = float(alphas[rnd])
        else:
            alpha = float(alphas[-1])

        if n <= 700:
            base_iter = 520
        elif n <= 1400:
            base_iter = 340
        else:
            base_iter = 220

        mit = int(min(base_iter, max(80, int(rem))))

        try:
            res = minimize(
                fixed_smooth_objective,
                work_h,
                jac=fixed_smooth_gradient,
                args=(n, active, alpha, mu, N),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": mit,
                    "ftol": 1.0e-18,
                    "gtol": 1.0e-12,
                    "maxfun": max(200, 6 * mit),
                    "maxls": 30,
                    "maxcor": 30,
                }
            )
        except Exception:
            break

        x2 = getattr(res, "x", None)
        if x2 is None or not np.all(np.isfinite(x2)):
            no_work += 2
            if no_work > 5:
                break
            continue

        h2 = project_feasible(x2, n)
        c2fit = float(np.max(corr_vec_fft(h2, n, N)))
        updated_work = False
        updated_best = False

        if exact_every:
            c2ex = float(eval_exact(h2, n))
            if c2ex < best_c - 1.0e-16:
                best_h = h2.copy()
                best_c = c2ex
                no_best = 0
                updated_best = True
            else:
                no_best += 1
        else:
            if c2fit < best_c - 1.0e-16:
                best_h = h2.copy()
                best_c = c2fit
                no_best = 0
                updated_best = True
            else:
                no_best += 1

        if c2fit < work_fit - 1.0e-15 or updated_best:
            work_h = h2.copy()
            work_fit = c2fit
            no_work = 0
            updated_work = True
        else:
            no_work += 1

        if no_work > 5:
            break
        if no_best > 12 and rnd >= len(alphas) - 1:
            break

    if n <= 4096:
        cw = float(eval_exact(work_h, n))
        if cw < best_c:
            best_h = work_h.copy()
            best_c = cw
    return best_h, float(best_c)

def append_variant(cands, c, hv):
    cands.append((float(c), len(cands), hv))

def generate_variants(h, n, seed=1):
    cands = []
    h = np.asarray(h, dtype=float)
    n = int(n)

    hb = make_binary_start(h, n)
    app_var(cands, eval_exact(hb, n), hb)

    try:
        F = np.fft.rfft(h)
        keep = max(1, int(0.05 * len(F)))
        F[keep:] = 0.0
        hs = np.fft.irfft(F, n)
        hs = project_feasible(hs, n)
        app_var(cands, eval_exact(hs, n), hs)
        hb2 = make_binary_start(hs, n)
        app_var(cands, eval_exact(hb2, n), hb2)
    except Exception:
        pass

    try:
        rng = np.random.RandomState(int(seed))
        hg = project_feasible(h + 0.02 * rng.randn(n), n)
        app_var(cands, eval_exact(hg, n), hg)
        hb3 = make_binary_start(hg, n)
        app_var(cands, eval_exact(hb3, n), hb3)
    except Exception:
        pass

    for p in (4, 8, 16, 32, 64, 128, 20, 50, 10):
        if p % 2 == 0 and n % p == 0:
            hp = periodic_start(n, p, p // 2)
            app_var(cands, eval_exact(hp, n), hp)

    return cands

def app_var(cands, c, hv):
    append_variant(cands, c, hv)

def try_variant_polish(h, n, c_best, deadline, max_variants=2):
    if time.time() + 45.0 > float(deadline):
        return h, c_best

    cands = generate_variants(h, n, seed=7)
    cands.sort()
    tried = 0
    for c0, _, hv in cands:
        if tried >= int(max_variants):
            break
        if c0 > c_best + 0.015:
            break
        rem = float(deadline - time.time())
        if rem < 40.0:
            break
        dur = min(35.0, rem - 5.0)
        try:
            hp, cp = run_active_lbfgs(
                hv, n, time.time() + dur,
                alphas=[300.0, 1.0e3, 3.0e3, 1.0e4, 3.0e4, 1.0e5],
                top_k=16, mu=2.0, max_rounds=8
            )
            if cp < c_best:
                h = hp
                c_best = cp
        except Exception:
            pass
        tried += 1
    return h, float(c_best)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0:
        budget = 1100.0
    deadline = t0 + max(1.0, budget) - 2.0

    try:
        if "initial_h_values" in globals() and initial_h_values is not None:
            h0 = np.array(initial_h_values, dtype=float).copy()
        else:
            raise ValueError("missing")
    except Exception:
        h0 = periodic_start(500, 4, 2)

    if h0 is None or len(h0) == 0:
        h0 = periodic_start(500, 4, 2)

    n0 = int(len(h0))
    if n0 > 2500:
        h0 = resample_h(h0, 1024)
        n0 = int(len(h0))
    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_c = float(eval_exact(h0, n0))

    p1_budget = min(350.0, max(25.0, 0.30 * budget))
    reserve = min(0.50 * budget, max(70.0, 0.40 * budget))
    p1_end = min(t0 + p1_budget, deadline - reserve)
    if p1_end < t0 + 12.0:
        p1_end = min(deadline - 5.0, t0 + 12.0)

    if time.time() < p1_end - 5.0:
        topk0 = 24 if best_n <= 800 else 12
        h1, c1 = run_active_lbfgs(
            best_h, best_n, p1_end,
            alphas=[100.0, 300.0, 1.0e3, 3.0e3, 1.0e4, 3.0e4, 1.0e5, 3.0e5],
            top_k=topk0, mu=2.0, max_rounds=45
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    target_ns = []
    if best_n < 560:
        target_ns.append(512)
    if best_n < 1000:
        target_ns.append(1024)
    if best_n < 1900 and budget > 600.0:
        target_ns.append(2048)

    new_targets = []
    for tn in target_ns:
        if tn != best_n and tn not in new_targets:
            new_targets.append(int(tn))

    if new_targets and budget > 200.0:
        for phase_idx, tn in enumerate(new_targets):
            rem_phase = len(new_targets) - phase_idx
            now = time.time()
            if now >= deadline - 35.0:
                break
            avail = float(deadline - now)
            dur = min(330.0, max(25.0, (avail - 35.0) / max(1, rem_phase)))
            if dur < 22.0:
                break

            hup = resample_h(best_h, tn)
            c_up = float(eval_exact(hup, tn))
            hbin = make_binary_start(hup, tn)
            c_bin = float(eval_exact(hbin, tn))

            if c_up <= c_bin:
                start_h = hup
                start_c = c_up
            else:
                start_h = hbin
                start_c = c_bin

            try:
                F = np.fft.rfft(start_h)
                keep = max(1, int(0.05 * len(F)))
                F[keep:] = 0.0
                hlow = project_feasible(np.fft.irfft(F, tn), tn)
                clow = float(eval_exact(hlow, tn))
                if clow < start_c:
                    start_h = hlow
                    start_c = clow
            except Exception:
                pass

            target_c = start_c
            if start_c < best_c + 0.05:
                topk = 14 if tn > 1000 else 18
                hend, cend = run_active_lbfgs(
                    start_h, tn, time.time() + dur,
                    alphas=[200.0, 800.0, 3.0e3, 1.0e4, 5.0e4, 2.0e5, 5.0e5],
                    top_k=topk, mu=3.0, max_rounds=30
                )
                target_c = cend
                if cend < best_c:
                    best_h, best_c, best_n = hend, cend, tn

            if time.time() < deadline - 55.0 and target_c < best_c + 0.025:
                best_h, best_c = try_variant_polish(best_h, best_n, best_c, deadline, max_variants=1)

    if time.time() < deadline - 45.0:
        best_h, best_c = try_variant_polish(best_h, best_n, best_c, deadline, max_variants=2)

    if time.time() < deadline - 6.0:
        hf, cf = run_active_lbfgs(
            best_h, best_n, deadline - 2.0,
            alphas=[5.0e4, 1.5e5, 5.0e5],
            top_k=8, mu=5.0, max_rounds=8
        )
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, int(best_n))
    if best_n <= 4096:
        best_c = float(eval_exact(best_h, int(best_n)))
    else:
        best_c = float(eval_fft(best_h, int(best_n)))

    return best_h, best_c, int(best_n)
