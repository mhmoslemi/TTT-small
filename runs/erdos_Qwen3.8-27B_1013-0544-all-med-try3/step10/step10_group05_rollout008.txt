import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_k(h, n, k):
    """Gradient of c_k = sum_i h(i)*(1-h(i+k)) dx w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            # c_k = sum_{i=0}^{n-1-k} h(i)*(1-h(i+k)) * dx
            # d/dc h(i) += (1-h(i+k)) for i in [0, nv-1]
            # d/dc h(i+k) += -h(i) for i in [0, nv-1], i.e., for j in [k, n-1]
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            # c_k = sum_{i=0}^{n-1-kk} h(i+kk)*(1-h(i)) * dx
            # d/dh(i+kk) += (1-h(i))
            # d/dh(i) += -h(i+kk)
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, h, n):
    """Gradient of smooth max of c_k w.r.t. h"""
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        gk = grad_c_k(h, n, k)
        grad += w[mm] * gk
    return grad

def act_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (h.sum() - n / 2.0) ** 2

def act_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        grad += wi * grad_c_k(h, n, k)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def full_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    return smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2

def full_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    grad = smooth_max_grad(c, alpha, h, n)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_k(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[0:nv] += (1.0 - h[k:n]) * dx
            grad[k:n] += (-h[0:nv]) * dx
    else:
        kk = -k
        nv = n - kk
        if nv > 0:
            grad[kk:n] += (1.0 - h[0:nv]) * dx
            grad[0:nv] += (-h[kk:n]) * dx
    return grad

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad_full(c, alpha, h, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        grad += w[mm] * grad_c_k(h, n, k)
    return grad

def act_obj(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (h.sum() - n / 2.0) ** 2

def act_grad(h, n, active, alpha, mu):
    c = compute_c5_vec(h, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        grad += wi * grad_c_k(h, n, k)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def full_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    return smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2

def full_grad(h, n, alpha, mu):

    grad = smooth_max_grad_full(c, alpha, h, n)
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

def compute_subgradient(h, n, width_frac=0.005, sharpness=500):
    c_vec = compute_c5_vec(h, n)
    max_val = c_vec.max()
    tol = max(1e-10, width_frac * max_val)
    active = np.where(c_vec >= max_val - tol)[0]
    if len(active) == 0:

    weights = c_vec[active]
    w_exp = np.exp((weights - max_val) * sharpness)
    w_exp /= w_exp.sum()
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w_exp, active):
        k = int(idx) - idx_center

                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 8,
                                       'maxls': 40, 'maxcor': 40})
                h_new = project_feasible(res.x, best_n)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Active set optimization (250s)
    p2_end = min(time.time() + 250, deadline - 450)
    if p2_end > time.time() + 20:
        configs = [
            (4, 50000, 0.002),
            (5, 100000, 0.001),
            (6, 200000, 0.001),
            (8, 500000, 0.0005),
            (10, 1000000, 0.0005),
            (12, 5000000, 0.0003),
            (16, 10000000, 0.0003),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p2_end - 15:
                break
            for _ in range(2):
                if time.time() > p2_end - 10:
                    break
                h2, c2 = run_active_round(best_h, best_n, p2_end - 5, top_k, alpha, mu)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2

    # Phase 3: Gradient-directed pair search (300s) - KEY INNOVATION
    p3_end = min(time.time() + 300, deadline - 300)
    if p3_end > time.time() + 30:
        h3, c3 = grad_pair_search(best_h, best_n, p3_end, max_iters=3000000, top_candidates=35)
        if c3 < best_c:
            best_c = c3
            best_h = h3
        cur = len(best_h)
        h3b, c3b = grad_pair_search(best_h, cur, p3_end, max_iters=3000000, top_candidates=25)
        if c3b < best_c:
            best_c = c3b
            best_h = h3b

    # Phase 4: Stochastic gradient pair search (120s)
    p4_end = min(time.time() + 120, deadline - 200)
    if p4_end > time.time() + 20:
        h4, c4 = stochastic_grad_pair_search(best_h, best_n, p4_end, rng,
                                              max_iters=2000000, top_candidates=12)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Targeted worst-shift search (80s)
    p5_end = min(time.time() + 80, deadline - 150)
    if p5_end > time.time() + 15:
        h5, c5 = targeted_worst_shift_search(best_h, best_n, p5_end, rng)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: More active set rounds (100s)
    p6_end = min(time.time() + 100, deadline - 100)
    if p6_end > time.time() + 15:
        configs6 = [
            (6, 2000000, 0.0003),
            (8, 5000000, 0.0002),
            (10, 10000000, 0.0002),
            (14, 20000000, 0.0001),
        ]
        for top_k, alpha, mu in configs6:
            if time.time() > p6_end - 10:
                break
            h6, c6 = run_active_round(best_h, best_n, p6_end - 5, top_k, alpha, mu)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Final gradient pair search (80s)
    p7_end = min(time.time() + 80, deadline - 40)
    if p7_end > time.time() + 15:
        h7, c7 = grad_pair_search(best_h, best_n, p7_end, max_iters=5000000, top_candidates=50)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Perturbation and re-optimization (remaining)
    p8_end = deadline - 3
    if p8_end > time.time() + 15:
        h8, c8 = perturb_and_reopt(best_h, best_n, p8_end, rng, n_trials=4)
        if c8 < best_c:
            best_c = c8
            best_h = h8
        rem = p8_end - time.time()
        if rem > 10:
            h8b, c8b = grad_pair_search(best_h, best_n, p8_end, max_iters=1000000, top_candidates=40)
            if c8b < best_c:
                best_c = c8b
                best_h = h8b

    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
