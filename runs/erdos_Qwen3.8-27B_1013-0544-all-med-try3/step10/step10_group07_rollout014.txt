```python
def slsqp_optimize(h_init, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        if alpha < 1:
            return m
        sh = alpha * (g - m)
        s_max = sh.max()
        return m + (s_max + np.log(np.exp(sh - s_max).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        active = (g >= m - 1e-10)
        grad = np.zeros(n)
        for k_idx in np.where(active)[0]:
            # dg/dx for lag k_idx
            # g[k] = sum_i x[i]*(1-x[i+k-n+1]) * dx
            # dg/dx[i] = dx * [(1-x[i+k-n+1]) - x[i-k+n-1]] (for valid indices)
            k = k_idx - (n - 1)
            for i in range(n):
                ip = i + k
                im2 = i - k
                if 0 <= ip < n:
                    grad[i] += dx * (1 - x[ip]) * active[k_idx]
                if 0 <= im2 < n:
                    grad[i] -= dx * x[im2] * active[k_idx]
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

    zeros = list(np.where(h == 0)[0])

                continue

            for j_idx in o_perm:
                j = int(j_idx)

                delta = np.zeros(2*n-1)
                im = i + m_range

                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)

        if not improved:
```

This is simple but slow to converge for minimax problems. However, it can be combined with smoothing.

**Alternative: Alternating minimization**

Fix the set of active lags (those achieving the max), then minimize the average over those lags. Then update the active set.

This is essentially the "column generation" approach for minimax.

Let me try this:
1. Start with h
2. Compute all f_k, identify active set A = {k : f_k ≥ f_max - eps}
3. Minimize (1/|A|) sum_{k∈A} f_k(h) subject to constraints
4. Repeat

Step 3 is a quadratic (since f_k is bilinear in h). We can solve it with SLSQP or projected gradient.

This might converge faster than the full softplus approach because we're focusing on the critical lags.

```python
def active_lag_search(h, n, deadline, tol=1e-4, maxiter=100):
    """Alternating: identify active lags, optimize over them."""
    dx = 2.0 / n
    h = h.copy()
    for it in range(maxiter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1-h, mode='full') * dx
        f_max = corr.max()
        active = np.where(corr >= f_max - tol)[0]
        
        # Minimize mean over active lags using SLSQP
        def obj(x):
            g = np.correlate(x, 1-x, mode='full') * dx
            return g[active].mean()
        try:
            res = minimize(obj, h, method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0}],
                          options={'maxiter':200,'ftol':1e-14})
            h = project(res.x, n)
        except:
            break
    
    return h, compute_c5(h, n)
```
