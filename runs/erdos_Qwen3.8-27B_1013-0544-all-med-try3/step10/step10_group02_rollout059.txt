def active_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, active):
        if wi < 1e-15:
            continue
        k = mm - (n-1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def max_obj_slsqp(x, n):
    c = compute_c5_vec(x, n)
    return float(c.max())

def constraint_sum(x, n):
    return x.sum() - n/2.0

def constraint_jac(x, n):
    return np.ones(n)

def periodic_init(n, period, ones):
    h = np.zeros(n)
    for p in range(ones):
        h[p::period] = 1.0
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        for i in range(n):
            if h[i] < 1.0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
    elif diff < 0:
        for i in range(n-1, -1, -1):
            if h[i] > 0.0:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return project_feasible(h, n)

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def gradient_guided_search(h, n, deadline, n_iter=2000):
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    best_h = h_cur.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h_cur, n)
        c_max = c_vec.max()
        
        # Find active shifts
        tol = max(1e-8, 0.001 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        # Compute gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        for mm in active:
            k = mm - (n-1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - h_cur[k:n]) * dx
                grad[k:n] += (-h_cur[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - h_cur[0:nv]) * dx
                grad[0:nv] += (-h_cur[kk:n]) * dx
        grad /= len(active)
        
        # Move from high-gradient to low-gradient positions
        pos_mask = grad > 0
        neg_mask = grad < 0
        if not np.any(pos_mask) or not np.any(neg_mask):
            break
        
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        
        # Select top movers
        top_pos = pos_idx[np.argsort(-grad[pos_idx])[:min(5, len(pos_idx))]]
        top_neg = neg_idx[np.argsort(grad[neg_idx])[:min(5, len(neg_idx))]]
        
        improved = False
        for d in [0.01, 0.005, 0.002, 0.001]:
            for i in top_pos:
                for j in top_neg:
                    if i == j:
                        continue
                    if h_cur[i] - d >= 0.0 and h_cur[j] + d <= 1.0:
                        h_cur[i] -= d
                        h_cur[j] += d
                        c = compute_c5(h_cur, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_cur.copy()
                            improved = True
                        else:
                            h_cur[i] += d
                            h_cur[j] -= d
                    if h_cur[i] + d <= 1.0 and h_cur[j] - d >= 0.0:
                        h_cur[i] += d
                        h_cur[j] -= d
                        c = compute_c5(h_cur, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h_cur.copy()
                            improved = True
                        else:
                            h_cur[i] -= d
                            h_cur[j] += d
        if not improved:
            # Try smaller step
            for i in top_pos[:2]:
                for j in top_neg[:2]:
                    if i == j:
                        continue
                    for d in [0.0005, 0.0002]:
                        if h_cur[i] - d >= 0.0 and h_cur[j] + d <= 1.0:
                            h_cur[i] -= d
                            h_cur[j] += d
                            c = compute_c5(h_cur, n)
                            if c < best_c - 1e-16:
                                best_c = c
                                best_h = h_cur.copy()
                                improved = True
                            else:
                                h_cur[i] += d
                                h_cur[j] -= d
            if not improved:
                break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500
    
    h0 = project_feasible(h0, n0)
    best_h_coarse = h0.copy()
    best_c_coarse = compute_c5(h0, n0)
    best_h = h0.copy()
    best_c = best_c_coarse
    best_n = n0
    
    # === Phase A: Quick optimization at n0 ===
    nA = n0
    boundsA = [(0.0, 1.0)] * nA
    pA_end = t0 + 80
    hA = h0.copy()
    cA = best_c_coarse
    alphas_A = [50, 500, 5000, 50000]
    for alpha in alphas_A:
        if time.time() > pA_end:
            break
        maxit = max(500, int((pA_end - time.time()) * 10))
        try:
            res = minimize(soft_obj, hA, jac=soft_grad,
                        args=(nA, alpha, 0.01), method='L-BFGS-B',
                        bounds=boundsA,
                        options={'maxiter': maxit, 'ftol': 1e-18,
                                 'gtol': 1e-14, 'maxfun': maxit*5, 'maxls': 50})
            h_new = project_feasible(res.x, nA)
            c_new = compute_c5(h_new, nA)
            if c_new < cA:
                cA = c_new
                hA = h_new
        except Exception:
            pass
    if cA < best_c_coarse:
        best_c_coarse = cA
        best_h_coarse = hA
        best_c = cA
        best_h = hA.copy()
    
    # === Phase B: Upsample to n=1000 and optimize ===
    nB = 1000
    hB_start = upsample_h(best_h_coarse, len(best_h_coarse), nB)
    cB_start = compute_c5(hB_start, nB)
    boundsB = [(0.0, 1.0)] * nB
    
    # Also try periodic inits at n=1000
    inits_B = [hB_start]
    for (per, on) in [(6,3), (10,5), (12,6), (8,4)]:
        hp = periodic_init(nB, per, on)
        inits_B.append(hp)
    
    pB_end = t0 + 400
    
    for init_idx, hB in enumerate(inits_B):
        if time.time() > pB_end:
            break
        cB_init = compute_c5(hB, nB)
        
        # Soft-max L-BFGS
        alphas_B = [20, 200, 2000, 20000, 100000, 500000]
        h_cur = hB.copy()
        c_cur = min(cB_init, best_c)
        
        for alpha in alphas_B:
            if time.time() > pB_end - 20:
                break
            remaining = pB_end - time.time()
            if remaining < 10:
                break
            maxit = max(300, int(remaining * 6 / (len(alphas_B) - alphas_B.index(alpha))))
            try:
                res = minimize(soft_obj, h_cur, jac=soft_grad,
                            args=(nB, alpha, 0.01), method='L-BFGS-B',
                            bounds=boundsB,
                            options={'maxiter': maxit, 'ftol': 1e-18,
                                     'gtol': 1e-14, 'maxfun': maxit*5, 'maxls': 50})
                h_new = project_feasible(res.x, nB)
                c_new = compute_c5(h_new, nB)
                if c_new < c_cur:
                    c_cur = c_new
                    h_cur = h_new
            except Exception:
                pass
        
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur.copy()
            best_n = nB
    
    # === Phase C: Lp-norm equalization at best resolution ===
    nC = best_n
    boundsC = [(0.0, 1.0)] * nC
    pC_end = min(time.time() + 200, deadline - 150)
    
    for p in [10, 20, 50]:
        if time.time() > pC_end:
            break
        remaining = pC_end - time.time()
        if remaining < 15:
            break
        top_k = min(30, nC)
        maxit = max(500, int(remaining * 4 / (3 - [10,20,50].index(p))))
        try:
            res = minimize(lp_obj, best_h, jac=lp_grad,
                        args=(nC, p, 0.005, top_k), method='L-BFGS-B',
                        bounds=boundsC,
                        options={'maxiter': maxit, 'ftol': 1e-18,
                                 'gtol': 1e-14, 'maxfun': maxit*5, 'maxls': 40})
            h_new = project_feasible(res.x, nC)
            c_new = compute_c5(h_new, nC)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    # === Phase D: Active-set focused optimization ===
    nD = best_n
    boundsD = [(0.0, 1.0)] * nD
    pD_end = min(time.time() + 180, deadline - 100)
    
    for rnd in range(12):
        if time.time() > pD_end:
            break
        c_vec = compute_c5_vec(best_h, nD)
        max_val = c_vec.max()
        threshold = max_val - max(1e-7, 0.0008 * max_val)
        active = np.where(c_vec >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        if len(active) > 12:
            active = active[np.argsort(-c_vec[active])][:12]
        
        for alpha in [50000, 200000, 1000000]:
            if time.time() > pD_end:
                break
            remaining = pD_end - time.time()
            if remaining < 10:
                break
            maxit = max(300, int(remaining * 4))
            try:
                res = minimize(active_obj, best_h, jac=active_grad,
                            args=(nD, active, alpha, 0.003), method='L-BFGS-B',
                            bounds=boundsD,
                            options={'maxiter': maxit, 'ftol': 1e-18,
                                     'gtol': 1e-14, 'maxfun': maxit*5, 'maxls': 40})
                h_new = project_feasible(res.x, nD)
                c_new = compute_c5(h_new, nD)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # === Phase E: Try upsample to n=2000 if time allows ===
    if time.time() < deadline - 250:
        nE = 2000
        hE_start = upsample_h(best_h, best_n, nE)
        cE_start = compute_c5(hE_start, nE)
        boundsE = [(0.0, 1.0)] * nE
        
        pE_end = min(time.time() + 200, deadline - 120)
        hE = hE_start.copy()
        cE = cE_start
        
        alphas_E = [10, 100, 1000, 10000, 100000]
        for alpha in alphas_E:
            if time.time() > pE_end:
                break
            remaining = pE_end - time.time()
            if remaining < 15:
                break
            maxit = max(300, int(remaining * 5 / max(1, len(alphas_E) - alphas_E.index(alpha))))
            try:
                res = minimize(soft_obj, hE, jac=soft_grad,
                            args=(nE, alpha, 0.005), method='L-BFGS-B',
                            bounds=boundsE,
                            options={'maxiter': maxit, 'ftol': 1e-18,
                                     'gtol': 1e-14, 'maxfun': maxit*4, 'maxls': 40})
                h_new = project_feasible(res.x, nE)
                c_new = compute_c5(h_new, nE)
                if c_new < cE:
                    cE = c_new
                    hE = h_new
            except Exception:
                pass
        
        # Lp at n=2000
        if time.time() < deadline - 100:
            for p in [20, 50]:
                if time.time() > pE_end:
                    break
                remaining = pE_end - time.time()
                if remaining < 15:
                    break
                maxit = max(300, int(remaining * 3))
                try:
                    res = minimize(lp_obj, hE, jac=lp_grad,
                                args=(nE, p, 0.003, min(30, nE)), method='L-BFGS-B',
                                bounds=boundsE,
                                options={'maxiter': maxit, 'ftol': 1e-18,
                                         'gtol': 1e-14, 'maxfun': maxit*4, 'maxls': 40})
                    h_new = project_feasible(res.x, nE)
                    c_new = compute_c5(h_new, nE)
                    if c_new < cE:
                        cE = c_new
                        hE = h_new
                except Exception:
                    pass
        
        if cE < best_c:
            best_c = cE
            best_h = hE.copy()
            best_n = nE
    
    # === Phase F: SLSQP polish ===
    nF = best_n
    boundsF = [(0.0, 1.0)] * nF
    pF_end = min(time.time() + 60, deadline - 40)
    
    if time.time() < pF_end:
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_jac, 'args': (nF,)}]
        try:
            res = minimize(max_obj_slsqp, best_h, args=(nF,), method='SLSQP',
                        bounds=boundsF, constraints=cons,
                        options={'maxiter': 2000, 'ftol': 1e-20, 'iprint': 0})
            h_new = project_feasible(res.x, nF)
            c_new = compute_c5(h_new, nF)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    # === Phase G: Gradient-guided local search ===
    nG = best_n
    pG_end = deadline - 5
    if time.time() < pG_end - 5:
        hG, cG = gradient_guided_search(best_h, nG, pG_end, n_iter=3000)
        if cG < best_c:
            best_c = cG
            best_h = hG.copy()
            best_n = nG
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    
    return best_h, best_c, best_n
