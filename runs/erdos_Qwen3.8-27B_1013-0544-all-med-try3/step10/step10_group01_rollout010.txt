    c_vec = compute_c5_vec(best_h, n)

    nn = 2 * n - 1

        if time.time() > deadline:
            break
        improved = False

        zeros = list(np.where(best_h < 0.5)[0])

        rng.shuffle(zeros)

                return best_h, best_c
            for j in zeros:

                # Change h[i]: 1→0, h[j]: 0→1

                    delta[s + n - 1] += (h[i + s] - 1.0) * dx
                
                # Term h[i-s]*(1-h[i]): valid when 0<=i-s<n, i.e., i-n+1<s<=i
                s_lo = max(-(n-1), i-n+1)
                s_hi = min(n-1, i)
                if s_lo <= s_hi:
                    s = np.arange(s_lo, s_hi+1)
                    delta[s + n - 1] += h[i - s] * dx
                
                # h[j]: 0→1
