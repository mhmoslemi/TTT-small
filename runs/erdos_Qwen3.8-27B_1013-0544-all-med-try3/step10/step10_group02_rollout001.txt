import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w_sum = w.sum()
    w = w / w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for mm in range(n_corr):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_obj(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def active_grad(x, n, active_idx, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w = w / w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active_idx):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def penalty_obj(x, n, active_idx, penalty, mu):
    c = compute_c5_vec(x, n)
    vals = c[active_idx]
    m = vals.max()
    excess = np.maximum(vals - m, 0.0)
    val = m + penalty * np.sum(excess**2)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def sq_fun(x, n):
    return x.sum() - n / 2.0

def sq_jac(x, n):

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxiter_per is not None and i < len(maxiter_per):
            maxit = maxiter_per[i]
        else:
            maxit = max(500, int(remaining * 4))
        
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 60})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new

def run_slqp(h, n, deadline, alpha=1000000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sq_fun, 'jac': sq_jac, 'args': (n,)}]
    
    def slq_obj(x):
        return smooth_max_obj(x, n, alpha, 0.0)
    
    def slq_jac(x):
        return smooth_max_grad(x, n, alpha, 0.0)
    
    try:
        res = minimize(slq_obj, h, jac=slq_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def fine_local_search(h, n, deadline, window=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 2):
                if time.time() > deadline:
                    break
                lo = max(0, i - window)
                hi = min(n, i + window + 1)
                for j in range(lo, hi, 3):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d

    return best_h, best_c

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2)
    h_new = project_feasible(h_new, n_new)
    return h_new

def make_smooth_binary(n, duty_ratio=0.5):

    ones_per = max(2, int(period * duty_ratio))

    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick optimization at n_orig (150s)
    p1_end = t0 + 150
    alphas_p1 = [100, 500, 2000, 10000, 50000, 200000, 1000000]
    h1, c1 = run_lbfgs_phase(best_h, best_n, p1_end, alphas_p1,
                              mu=0.01,
                              maxiter_per=[1000, 1500, 2000, 2500, 3000, 3000, 4000])
    if c1 < best_c:
        best_c = c1
        best_h = h1
    
    # Phase 2: Upsample to n=1000 (300s)
    n_mid = 1000
    if time.time() < deadline - 400:
        h_mid = upsample(best_h, best_n, n_mid)
        c_mid = compute_c5(h_mid, n_mid)
        if c_mid < best_c:
            best_c = c_mid
            best_h = h_mid
            best_n = n_mid
        
        p2_end = min(time.time() + 250, deadline - 300)
        alphas_p2 = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
        h2, c2 = run_lbfgs_phase(h_mid, n_mid, p2_end, alphas_p2,
                                  mu=0.005,
                                  maxiter_per=[1000, 1500, 2000, 2500, 3000, 3000, 3500, 4000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = n_mid
    
    # Phase 3: Active set refinement at best_n (200s)
    if time.time() < deadline - 250:
        p3_end = min(time.time() + 180, deadline - 200)
        h3, c3 = run_active_refine(best_h, best_n, p3_end,
                                   top_k=8,
                                   alphas=[50000, 200000, 1000000, 5000000],
                                   rounds=20)
        if c3 < best_c:
            best_c = c3
            best_h = h3
    
    # Phase 4: Try n=2000 if time permits (150s)
    n_high = 2000
    if time.time() < deadline - 200:
        h_high = upsample(best_h, best_n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high < best_c:
            best_c = c_high
            best_h = h_high
            best_n = n_high
        
        p4_end = min(time.time() + 130, deadline - 120)
        alphas_p4 = [50, 500, 5000, 50000, 500000]
        h4, c4 = run_lbfgs_phase(h_high, n_high, p4_end, alphas_p4,
                                  mu=0.003,
                                  maxiter_per=[800, 1200, 1500, 2000, 2500])
        if c4 < best_c:
            best_c = c4
            best_h = h4
            best_n = n_high
        else:
            # Keep the upsampled version for further refinement
            if c_high < best_c:
                best_h = h_high
                best_n = n_high
            else:
                # Revert to previous best
                h_high = None
    
    # Phase 5: Active set at current best (100s)
    if time.time() < deadline - 100:
        p5_end = min(time.time() + 80, deadline - 60)
        h5, c5 = run_active_refine(best_h, best_n, p5_end,
                                   top_k=6,
                                   alphas=[100000, 500000, 2000000, 10000000],
                                   rounds=15)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    
    # Phase 6: SLSQP polish (80s)
    if time.time() < deadline - 50:
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 30:
                break
            p6_end = min(time.time() + 30, deadline - 15)
            h6, c6 = run_slqp(best_h, best_n, p6_end, alpha=alpha, maxiter=1000)
            if c6 < best_c:
                best_c = c6
                best_h = h6
    
    # Phase 7: Subgradient refinement (40s)
    if time.time() < deadline - 20:
        p7_end = min(time.time() + 30, deadline - 5)
        h7, c7 = subgrad_refine(best_h, best_n, p7_end, n_iter=5000, eta0=0.002)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Very fine local search (15s)
    if time.time() < deadline - 3:
        p8_end = deadline - 1
        h8, c8 = fine_local_search(best_h, best_n, p8_end, window=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final: verify and return at the resolution that gives best_c
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    # If we have a solution at n=500 that's competitive, prefer it for robustness
    if final_n != n_orig:
        try:
            h_500 = upsample(best_h, final_n, n_orig) if final_n > n_orig else upsample(best_h, final_n, n_orig)
        except:
            pass
    
    return best_h, best_c, final_n
