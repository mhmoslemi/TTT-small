The key observation: if most variables are 0 or 1, the "effective" search space is combinatorial. The fractional optimization (L-BFGS) can't help much because the optimum is at or near the boundary of the [0,1] box.

So the most effective approach is probably:
1. Get a good binary structure (250 ones, 250 zeros)
2. Do a very thorough local search on the binary structure
3. Then do a small fractional adjustment (maybe 1-2 variables slightly off 0/1) to squeeze out the last bit

For the binary local search:
- At each step, the worst lag k* is identified
- We want to swap a 1 at position i (where h[i+k*]=0) with a 0 at position j (where h[j+k*]=1)
- This reduces c_{k*} by 1 (in units of dx), but may increase other lags
- We pick the swap that minimizes the new max correlation

Efficient evaluation: when we swap positions i and j, the correlation at lag m changes by:
- Remove contribution of h[i]=1 at position i: affects lags where i is involved
- Add contribution of h[j]=1 at position j: affects lags where j is involved

For a given position p, it appears in correlation lag m when:
- m = p - q + (n-1) for some valid q (p is the first factor)
- m = q - p + (n-1) for some valid q (p is the second factor)

6. Pick the pair with the smallest estimated increase

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max_val(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs(h, n, deadline, alphas=None, mu=0.01, maxits=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    if maxits is None:
        maxits = [500, 800, 1000, 1200, 1500, 2000, 2500]
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        maxit = maxits[i] if i < len(maxits) else 2000
        try:
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': maxit * 10,
                                   'maxls': 50, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def snap_to_binary(h, n, threshold=0.4):

    h = h.copy()
    h[h > 1.0 - threshold] = 1.0
    h[h < threshold] = 0.0
    # Count ones

        zeros_idx = np.where(h == 0.0)[0]
        add = min(target - n_ones, len(zeros_idx))
        h[zeros_idx[:add]] = 1.0
    return h
