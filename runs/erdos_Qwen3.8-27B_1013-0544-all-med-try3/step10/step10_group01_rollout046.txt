        if w[mm] < 1e-12:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def upsample_parent(h_parent, n_new):
    n_old = len(h_parent)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_parent)

def make_beatty(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
        pos = int(k * alpha_param * n) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= n_ones:
                break
    for pos in positions:
        h[pos] = 1.0
    remaining = n / 2.0 - h.sum()
    if abs(remaining) > 1e-10:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] += remaining
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def make_smooth_init(n, freq1=1.0, freq2=2.0, amp1=0.3, amp2=0.1):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp1 * np.cos(2.0 * np.pi * freq1 * x / 2.0) + amp2 * np.sin(2.0 * np.pi * freq2 * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def run_lbfgs_smooth(x0, n, deadline, alphas, mu=0.005, per_alpha_time=30):
    best_h = project_feasible(x0, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        maxit = max(200, int(min(per_alpha_time, deadline - time.time()) * 5))
        try:
            res = minimize(full_smooth_obj, best_h, jac=full_smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c + 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def run_lbfgs_active(x0, n, deadline, top_k_list, alphas_list, mu=0.001, rounds=4):
    best_h = project_feasible(x0, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(rounds):
        if time.time() > deadline - 20:
            break
        c_vec = compute_c5_vec(best_h, n)
        for top_k, alpha in zip(top_k_list, alphas_list):
            if time.time() > deadline - 15:
                break
            active = get_active_set(c_vec, top_k, threshold_frac=0.003)
            if len(active) == 0:
                active = np.argsort(-c_vec)[:top_k]
            maxit = max(100, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(active_set_obj, best_h, jac=active_set_grad,
                              args=(n, active, alpha, mu), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-15, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new <= best_c + 1e-15:
                    best_c = c_new
                    best_h = h_new
                c_vec = compute_c5_vec(best_h, n)
            except Exception:
                pass
    return best_h, best_c

def coord_descent_swap(h, n, deadline, step_sizes=None, max_total_steps=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    total_steps = 0
    for step in step_sizes:
        while time.time() < deadline and total_steps < max_total_steps:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.003 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            if len(active) > 20:
                active = np.array(np.argsort(-c_vec[active])[:20])
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= len(active)
            pos_mask = (grad > 5e-10) & (best_h > 5e-7)
            neg_mask = (grad < -5e-10) & (best_h < 1.0 - 5e-7)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx[:30]:
                gi = grad[i]
                for j in neg_idx[:30]:
                    g = gi - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0 or best_gain < 1e-10:
                break
            amt = min(step, best_h[best_i] * 0.9, (1.0 - best_h[best_j]) * 0.9)
            if amt < 1e-10:
                break
            old_i = best_h[best_i]
            old_j = best_h[best_j]
            best_h[best_i] = old_i - amt
            best_h[best_j] = old_j + amt
            c_new = compute_c5(best_h, n)
            total_steps += 1
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] = old_i
                best_h[best_j] = old_j
        if step < 0.001:
            break
    return best_h, best_c

def fine_local_search(h, n, deadline, radius=None, deltas=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if radius is None:
        radius = max(20, n // 25)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    if improved:
                        break
                if improved:
                    best_h = project_feasible(best_h, n)
                    break
    return best_h, best_c

def pair_flip_search(h, n, deadline, max_rounds=30):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones_idx = np.where(best_h > 0.7)[0]
        zeros_idx = np.where(best_h < 0.3)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        improved = False
        rng_local = np.random.RandomState(42 + rnd)
        rng_local.shuffle(ones_idx)
        rng_local.shuffle(zeros_idx)
        for i in ones_idx:
            if time.time() > deadline:
                break
            for j in zeros_idx[:min(50, len(zeros_idx))]:
                old_i, old_j = best_h[i], best_h[j]
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_feasible(best_h, n)
                c = compute_c5(best_h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_feasible(best_h, n)
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12

    n = 1000
    rng = np.random.RandomState(seed)

    # Generate initializations
    inits = []
    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        h_up = upsample_parent(hp, n)
        h_up = project_feasible(h_up, n)
        inits.append(h_up)
    except Exception:
        pass
    inits.append(make_beatty(n, (1.0 + np.sqrt(5.0)) / 2.0 / 2.0))
    inits.append(make_beatty(n, np.sqrt(2.0) / 2.0))
    inits.append(make_beatty(n, 0.381966))
    inits.append(make_smooth_init(n, 1.0, 3.0, 0.25, 0.1))
    inits.append(make_smooth_init(n, 2.0, 5.0, 0.2, 0.15))
    inits.append(make_random_binary(n, rng))
    inits.append(make_random_binary(n, rng))

    # Pick best initial
    best_h = inits[0]
    best_c = compute_c5(best_h, n)
    for h_init in inits:
        c_init = compute_c5(h_init, n)
        if c_init < best_c:
            best_c = c_init
            best_h = h_init.copy()

    # Phase 1: L-BFGS full smooth max (warmup)
    p1_end = min(t0 + 180, deadline - 500)
    h1, c1 = run_lbfgs_smooth(best_h, n, p1_end,
        alphas=[20, 100, 500, 2000, 10000, 50000],
        mu=0.01, per_alpha_time=25)
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: L-BFGS active set (main optimization)
    p2_end = min(t0 + 480, deadline - 300)
    h2, c2 = run_lbfgs_active(best_h, n, p2_end,
        top_k_list=[4, 6, 8, 10, 12, 16, 20, 24, 30, 40],
        alphas_list=[100000, 500000, 2000000, 5000000, 10000000, 20000000, 50000000, 100000000, 200000000, 500000000],
        mu=0.0005, rounds=5)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Coordinate descent
    p3_end = min(t0 + 750, deadline - 200)
    h3, c3 = coord_descent_swap(best_h, n, p3_end,
        step_sizes=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
        max_total_steps=150000)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Phase 4: Active set refinement
    p4_end = min(t0 + 900, deadline - 120)
    h4, c4 = run_lbfgs_active(best_h, n, p4_end,
        top_k_list=[6, 10, 16, 24, 36, 50],
        alphas_list=[2000000, 10000000, 50000000, 100000000, 500000000, 1000000000],
        mu=0.0003, rounds=3)
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Fine local search
    p5_end = min(t0 + 1020, deadline - 40)
    h5, c5 = fine_local_search(best_h, n, p5_end,
        radius=30, deltas=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    if c5 < best_c:
        best_c = c5
        best_h = h5

    # Phase 6: Pair flip
    if time.time() < deadline - 20:
        h6, c6 = pair_flip_search(best_h, n, deadline - 10, max_rounds=15)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Final: very fine coordinate descent
    if time.time() < deadline - 3:
        h7, c7 = coord_descent_swap(best_h, n, deadline - 1,
            step_sizes=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
            max_total_steps=50000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
