import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def make_beatty(n, alpha):
    """Create a Beatty sequence based binary vector"""
    h = np.zeros(n)
    target = n // 2
    positions = sorted(set(int(k * alpha) % n for k in range(1, 4 * n)))
    for p in positions[:target]:
        h[p] = 1.0
    # Fix sum exactly
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] += min(diff, 1.0)
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] += max(diff, -1.0)
    return project_feasible(h, n)

def weighted_grad(h, n, lam):
    """Compute exact gradient of sum lam_k * c_k with respect to h.
    
    lam has length 2*n-1, indexed from -(n-1) to (n-1).
    lam[0] corresponds to shift -(n-1), lam[n-1] to shift 0, lam[2*n-2] to shift n-1.
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # For each shift k from -(n-1) to (n-1)
    for shift in range(-(n-1), n):
        idx = shift + (n - 1)  # index into lam
        ls = lam[idx]
        if ls < 1e-15:
            continue
        
        if shift >= 0:
            k = shift
            # ∂c_k/∂h[j] = dx * [1(j <= n-1-k) - 1(j <= n-1-k)*h[j+k] - 1(j >= k)*h[j-k]]
            # Term 1: dx * 1(j <= n-1-k)
            if n - 1 - k >= 0:
                grad[0:n-k] += ls * dx
                # Term 2: -dx * 1(j <= n-1-k) * h[j+k]
                grad[0:n-k] -= ls * dx * h[k:n]
                # Term 3: -dx * 1(j >= k) * h[j-k]
                if k > 0:
                    grad[k:n] -= ls * dx * h[0:n-k]
        else:
            m = -shift
            # ∂c_{-m}/∂h[j] = dx * [1(m <= j <= n-1) - 1(m <= j <= n-1)*h[j-m] - 1(0 <= j <= n-1-m)*h[j+m]]
            if n - 1 - m >= 0:
                # Term 1: dx * 1(m <= j)
                grad[m:n] += ls * dx
                # Term 2: -dx * 1(m <= j) * h[j-m]
                grad[m:n] -= ls * dx * h[0:n-m]
                # Term 3: -dx * 1(0 <= j <= n-1-m) * h[j+m]
                if m > 0:
                    grad[0:n-m] -= ls * dx * h[m:n]
    
    return grad


def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def run_lbfgs(h, n, deadline, alpha, mu, maxit, lam=None):
    """Run L-BFGS. If lam is provided, use weighted objective; else use smooth max."""
    bounds = [(0.0, 1.0)] * n
    
    if lam is not None:
        def obj(x):
            c = compute_c5_vec(x, n)
            val = np.dot(lam, c)
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return val
        def grad(x):
            c = compute_c5_vec(x, n)
            g = weighted_grad(x, n, lam)
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
    else:
        def obj(x):
            c = compute_c5_vec(x, n)
            val = smooth_max(c, alpha)
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return val
        def grad(x):
            c = compute_c5_vec(x, n)
            m = c.max()
            sh = alpha * (c - m)
            smx = sh.max()
            w = np.exp(sh - smx)
            w /= w.sum()
            g = weighted_grad(x, n, w)
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)


def alternating_minimax(h, n, deadline, max_rounds=50):
    """Alternating minimax: alternate between updating lam and optimizing h."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Soft-max weighting with decreasing temperature
        temp = max(0.001, 0.01 * (1 - round_idx / max_rounds))
        sh = (c_vec - max_val) / temp
        w = np.exp(sh)
        w /= w.sum()
        
        # Optimize h for the weighted objective
        h_new, c_new = run_lbfgs(best_h, n, min(deadline, time.time() + 20),
                                  alpha=10000, mu=0.01, maxit=500, lam=w)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    
    return best_h, best_c

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0

            order = np.argsort(h[ones])

def make_beatty(n, alpha):

    target = n // 2
    positions = sorted(set(int(k * alpha) % n for k in range(1, 4 * n)))
    for p in positions[:target]:
        h[p] = 1.0

def weighted_grad(h, n, lam):
    dx = 2.0 / n
    grad = np.zeros(n)
    for shift in range(-(n-1), n):
        idx = shift + (n - 1)
        ls = lam[idx]
        if ls < 1e-15:
            continue
        if shift >= 0:
            k = shift
            if n - 1 - k >= 0:
                grad[0:n-k] += ls * dx
                grad[0:n-k] -= ls * dx * h[k:n]
                if k > 0:
                    grad[k:n] -= ls * dx * h[0:n-k]
        else:
            m = -shift
            if n - 1 - m >= 0:
                grad[m:n] += ls * dx
                grad[m:n] -= ls * dx * h[0:n-m]
                if m > 0:
                    grad[0:n-m] -= ls * dx * h[m:n]
    return grad

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def make_beatty(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = sorted(set(int(k * alpha) % n for k in range(1, 4 * n)))
    for p in positions[:target]:
        h[p] = 1.0
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                h[zeros[0]] += min(diff, 1.0)
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                h[ones[0]] += max(diff, -1.0)
    return project_feasible(h, n)

def weighted_grad(h, n, lam):
    dx = 2.0 / n
    grad = np.zeros(n)
    for shift in range(-(n - 1), n):
        idx = shift + (n - 1)
        ls = lam[idx]
        if ls < 1e-15:
            continue
        if shift >= 0:
            k = shift
            if n - 1 - k >= 0:
                grad[0:n - k] += ls * dx
                grad[0:n - k] -= ls * dx * h[k:n]
                if k > 0:
                    grad[k:n] -= ls * dx * h[0:n - k]
        else:
            m = -shift
            if n - 1 - m >= 0:
                grad[m:n] += ls * dx
                grad[m:n] -= ls * dx * h[0:n - m]
                if m > 0:
                    grad[0:n - m] -= ls * dx * h[m:n]
    return grad

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def run_lbfgs(h, n, deadline, alpha, mu, maxit, lam=None):
    bounds = [(0.0, 1.0)] * n
    if lam is not None:
        lam_local = lam.copy()
        def obj(x):
            c = compute_c5_vec(x, n)
            val = np.dot(lam_local, c)
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return val
        def grad_fn(x):
            g = weighted_grad(x, n, lam_local)
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
    else:
        def obj(x):
            c = compute_c5_vec(x, n)
            val = smooth_max(c, alpha)
            if mu > 0:
                val += mu * (x.sum() - n / 2.0) ** 2
            return val
        def grad_fn(x):
            c = compute_c5_vec(x, n)
            w = smooth_max_w(c, alpha)
            g = weighted_grad(x, n, w)
            if mu > 0:
                g += 2.0 * mu * (x.sum() - n / 2.0)
            return g
    try:
        res = minimize(obj, h, jac=grad_fn, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def alternating_minimax(h, n, deadline, max_rounds=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        frac = round_idx / max_rounds
        temp = max(0.003, 0.04 * (1 - frac) ** 2)
        sh = (c_vec - max_val) / temp
        sh = sh - sh.max()
        w = np.exp(sh)
        w /= w.sum()
        time_per = max(5.0, remaining * 0.8 / max(1, max_rounds - round_idx))
        h_new, c_new = run_lbfgs(best_h, n, min(deadline, time.time() + time_per),
                                  alpha=10000, mu=0.01, maxit=400, lam=w)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

            break
        c_vec = compute_c5_vec(h, n)
        w = smooth_max_w(c_vec, 50000.0)
        grad = weighted_grad(h, n, w)
        cands_rm = [(grad[i], i) for i in ones if grad[i] > 1e-10]
        cands_add = [(grad[j], j) for j in zeros if grad[j] < -1e-10]
        improved = False
        if cands_rm and cands_add:
            cands_rm.sort(reverse=True, key=lambda x: x[0])
            cands_add.sort(key=lambda x: x[0])
            top_rm = cands_rm[:25]
            top_add = cands_add[:25]
            for gi, i in top_rm:
                if improved:
                    break
                for gj, j in top_add:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(i)
                        improved = True
                        stall = 0
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0

            stall += 1

                for _ in range(5):

                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0

                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        zeros.append(i)
                    else:

def fine_coord_descent(h, n, deadline, max_rounds=600):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0
        while improved_any and rounds < max_rounds and time.time() < deadline:
            improved_any = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            w = smooth_max_w(c_vec, 200000.0)
            grad = weighted_grad(best_h, n, w)
            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0.0
            best_i = -1
            best_j = -1
            for i in pos_idx[:40]:
                for j in neg_idx[:40]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
    return best_h, best_c
