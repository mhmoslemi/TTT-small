    return float(val)

def active_set_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def weighted_hinge_obj(x, n, weights, offset, mu):
    c = compute_c5_vec(x, n)
    vals = np.maximum(c - offset, 0)
    val = float(np.dot(weights, vals))
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def weighted_hinge_grad(x, n, weights, offset, mu):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(weights)):
        w = weights[i]
        if w < 1e-15 or c[i] <= offset:
            continue
        k = i - (n - 1)
        wm = w * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_hinge(h, n, deadline, beta=100.0, lr_init=0.02, max_iters=10000):
    """Weighted hinge loss optimization."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    lr = lr_init
    mu = 0.005
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        c_max = c.max()
        offset = c_max * 0.999  # slightly below max
        
        # Update weights
        shifted = beta * (c - c_max)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient
        grad = weighted_hinge_grad(best_h, n, weights, offset, mu)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Backtracking line search
        alpha = lr
        best_val = weighted_hinge_obj(best_h, n, weights, offset, mu)
        improved = False
        for _ in range(15):
            h_try = np.clip(best_h - alpha * grad, 0, 1)
            h_try = project_feasible(h_try, n)
            v = weighted_hinge_obj(h_try, n, weights, offset, mu)
            if v < best_val - 1e-15:
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    best_h = h_try
                    improved = True
                    break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        if improved:
            stall = 0
            lr = min(lr * 1.01, 0.1)
        else:
            stall += 1
            lr *= 0.9
            if stall > 100:
                # Perturb
                h_try = best_h + 0.005 * np.random.randn(n)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try
                stall = 0
                lr = lr_init * 0.5
    
    return best_h, best_c

def pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i >= j:
                        continue
                    # Transfer from j to i
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    # Transfer from i to j
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def periodic_init(n, period, pattern):
    """Create periodic initialization with given pattern."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern[i % period]
    # Adjust to meet sum constraint
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute evenly among non-saturated entries
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = mask.sum()
        if count > 0:
            h[mask] += diff / count
    return project_feasible(h, n)

            except Exception:
                pass
    return best_h, best_c

def hinge_grad(x, n, weights, offset, mu):
    c = compute_c5_vec(x, n)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(weights)):
        w = weights[i]
        if w < 1e-14 or c[i] <= offset:
            continue
        k = i - (n - 1)
        wm = w * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_hinge(h, n, deadline, beta=100.0, lr_init=0.02, max_iters=20000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    lr = lr_init
    mu = 0.005
    stall = 0
    n_dual = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c5_vec(best_h, n)
        c_max = c.max()
        offset = c_max * 0.9995
        shifted = beta * (c - c_max)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = hinge_grad(best_h, n, weights, offset, mu)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-13:
            break
        grad = grad / grad_norm * min(1.0, grad_norm)
        improved = False
        alpha = lr
        for _ in range(12):
            h_try = np.clip(best_h - alpha * grad, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            c_old = compute_c5_vec(best_h, n)
            rel_old = float(np.dot(weights, np.maximum(c_old - offset, 0)))
            rel_new = float(np.dot(weights, np.maximum(compute_c5_vec(h_try, n) - offset, 0)))
            if c_try < best_c - 1e-15 or (rel_new < rel_old - 1e-14 and c_try <= best_c + 1e-10):
                best_c = min(best_c, c_try)
                best_h = h_try
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        if improved:
            stall = 0
            lr = min(lr * 1.008, 0.08)
        else:
            stall += 1
            lr *= 0.95
            if stall > 200:
                h_try = best_h + 0.003 * np.random.randn(n)
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try
                lr = lr_init * 0.5
                stall = 0
    return best_h, best_c

def pair_search(h, n, deadline, radius=40):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i >= j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if d < 0.001 and best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

def block_search(h, n, deadline, block_len=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for block_len in [3, 5, 8, 12]:
        for step in [0.005, 0.002, 0.001, 0.0005]:
            improved = True
            while improved and time.time() < deadline:
                improved = False
                for a in range(0, n - block_len, 2):
                    if time.time() > deadline:
                        break
                    for b in range(0, n - block_len, 2):
                        if abs(a - b) < block_len:
                            continue
                        block_a = best_h[a:a+block_len].copy()
                        block_b = best_h[b:b+block_len].copy()
                        new_a = np.clip(block_a + step, 0, 1)
                        new_b = np.clip(block_b - step, 0, 1)
                        if np.allclose(new_a, block_a) or np.allclose(new_b, block_b):
                            continue
                        best_h[a:a+block_len] = new_a
                        best_h[b:b+block_len] = new_b
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                            break
                        else:
                            best_h[a:a+block_len] = block_a
                            best_h[b:b+block_len] = block_b
                    if improved:
                        break
    return best_h, best_c

def periodic_init(n, period, pattern):
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern[i % period]
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = mask.sum()
        if count > 0:
            h[mask] += diff / count
    return project_feasible(h, n)

def binary_snap(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(round(n / 2.0))
    current_ones = int(h_bin.sum())
    if current_ones > n_ones:
        idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h_bin[idx])
        for i in range(current_ones - n_ones):
            h_bin[idx[order[i]]] = 0.0
    elif current_ones < n_ones:
        idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[idx])
        for i in range(n_ones - current_ones):
            h_bin[idx[order[i]]] = 1.0
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_start = len(h0)
    except Exception:
        h0 = periodic_init(500, 8, [1, 1, 0, 0, 1, 1, 0, 0])
        n_start = 500

    h0 = project_feasible(h0, n_start)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_start)
    best_n = n_start

    # Phase 1: L-BFGS at starting resolution with alpha annealing
    p1_end = min(t0 + 320, deadline - 500)
    if p1_end > t0 + 30:
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000, 2000000],
            mu=0.01)
        if c1 < best_c:
            best_c, best_h = c1, h1

    # Phase 2: Active set refinement
    p2_end = min(time.time() + 180, deadline - 400)
    if p2_end > time.time() + 20:
        h2, c2 = run_active(best_h, best_n, p2_end, top_k=6,
            alphas=[10000, 50000, 200000, 1000000, 5000000])
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: Hinge loss optimization (different landscape)
    p3_end = min(time.time() + 180, deadline - 300)
    if p3_end > time.time() + 20:
        h3, c3 = run_hinge(best_h, best_n, p3_end, beta=80.0, lr_init=0.015)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: Try alternative initializations
    inits = [
        (n_start, 6, [1, 0, 1, 0, 1, 0]),
        (n_start, 6, [1, 1, 0, 1, 0, 0]),
        (n_start, 10, [1, 0, 1, 0, 0, 1, 0, 1, 0, 0]),
        (n_start, 8, [1, 0, 1, 1, 0, 1, 1, 0]),
        (n_start, 12, [1, 1, 0, 0, 1, 0, 1, 0, 1, 1, 0, 0]),
    ]
    for (n_p, period, pattern) in inits:
        if time.time() < deadline - 200:
            h_p = periodic_init(n_p, period, pattern)
            p_end = min(time.time() + 150, deadline - 150)
            if p_end < time.time() + 15:
                continue
            h_p, c_p = run_lbfgs(h_p, n_p, p_end,
                alphas=[100, 1000, 10000, 100000, 500000, 2000000],
                mu=0.01)
            if c_p < best_c:
                best_c, best_h, best_n = c_p, h_p, n_p

    # Phase 5: Binary snap + re-optimize
    if time.time() < deadline - 180:
        h_bin = binary_snap(best_h, best_n)
        c_bin = compute_c5(h_bin, best_n)
        p5_end = min(time.time() + 120, deadline - 130)
        if p5_end > time.time() + 15:
            h5, c5 = run_lbfgs(h_bin, best_n, p5_end,
                alphas=[200, 2000, 20000, 200000, 1000000],
                mu=0.01)
            if c5 < best_c:
                best_c, best_h = c5, h5

    # Phase 6: Upsample to n=1000
    if time.time() < deadline - 250:
        n_fine = 1000
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, best_n)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        p6_end = min(time.time() + 250, deadline - 150)
        if p6_end > time.time() + 30:
            h6, c6 = run_lbfgs(h_fine, n_fine, p6_end,
                alphas=[50, 500, 5000, 50000, 200000, 1000000],
                mu=0.005)
            if c6 < best_c:
                best_c, best_h, best_n = c6, h6, n_fine
            # Active set at fine resolution
            if time.time() < deadline - 100:
                p6b_end = min(time.time() + 80, deadline - 80)
                h6b, c6b = run_active(h6, n_fine, p6b_end, top_k=8,
                    alphas=[50000, 200000, 1000000])
                if c6b < best_c:
                    best_c, best_h, best_n = c6b, h6b, n_fine

    # Phase 7: Hinge at best resolution
    if time.time() < deadline - 100:
        p7_end = min(time.time() + 80, deadline - 70)
        h7, c7 = run_hinge(best_h, best_n, p7_end, beta=120.0, lr_init=0.012)
        if c7 < best_c:
            best_c, best_h = c7, h7

    # Phase 8: Block search
    if time.time() < deadline - 60:
        p8_end = min(time.time() + 50, deadline - 40)
        h8, c8 = block_search(best_h, best_n, p8_end)
        if c8 < best_c:
            best_c, best_h = c8, h8

    # Phase 9: Pair search (fine local optimization)
    if time.time() < deadline - 20:
        p9_end = deadline - 5
        h9, c9 = pair_search(best_h, best_n, p9_end, radius=35)
        if c9 < best_c:
            best_c, best_h = c9, h9

    # Phase 10: Final active set polish
    if time.time() < deadline - 3:
        p10_end = deadline - 2
        h10, c10 = run_active(best_h, best_n, p10_end, top_k=4,
            alphas=[200000, 1000000])
        if c10 < best_c:
            best_c, best_h = c10, h10

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
