def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    w = np.exp(shifted - mx)
    return w / w.sum()


def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += w[k] * (1.0 - h[0:n - lag])
            grad[0:n - lag] += w[k] * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += w[k] * (1.0 - h[idx:n])
            grad[idx:n] += w[k] * (-h[0:n - idx])
    return grad * dx


def lbfgsb_refine(h, n, alpha, deadline, maxiter=3000):
    target_sum = n / 2.0
    penalty_coeff = 1e6

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'maxls': 40, 'iprint': -1})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)


def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0 / n_dst
    # Step function interpolation: use the value at the left edge
    h_dst = np.zeros(n_dst)
    for i in range(n_dst):
        # Find the source bin that contains x_dst[i]
        src_pos = int(x_dst[i] / (2.0 / n_src))
        src_pos = min(src_pos, n_src - 1)
        h_dst[i] = h_src[src_pos]
    return to_binary(h_dst, n_dst)

            h_try = h_best.copy()
            n_flips = int(n * 0.05 * (start + 1))
            ones = np.where(h_try == 1)[0]
            zeros = np.where(h_try == 0)[0]
            for _ in range(n_flips):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_try[i], h_try[j] = h_try[j], h_try[i]
        
        alloc = max(30, (deadline - time.time()) / (n_starts - start))
        end_t = min(deadline, time.time() + alloc)
        h_opt, c_opt = pairwise_binary_search(h_try, n, end_t, rng)
        if c_opt < c_best:
            c_best = c_opt
            h_best = h_opt
    
    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 500

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        h_bin = to_binary(h_init, n_init)
        cinit = compute_c5(h_bin, n_init)
        h_cont = project_feasible(h_init, n_init)
        ccont = compute_c5(h_cont, n_init)
        if ccont < cinit:
            best_h = h_cont
            best_c = ccont
        else:
            best_h = h_bin
            best_c = cinit
    else:
        best_h = to_binary(thue_morse(n_init), n_init)
        best_c = compute_c5(best_h, n_init)

    n = len(best_h)
    print(f"Initial: n={n}, C5={best_c:.6f}")

    # Phase 1: Directed binary search at current resolution
    if time.time() < deadline - 400:
        end_t = min(deadline - 350, time.time() + 200)
        h_opt, c_opt = directed_binary_search(best_h, n, end_t, rng, top_k=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            print(f"After directed search (n={n}): C5={best_c:.6f}")

    # Phase 2: Pairwise binary search
    if time.time() < deadline - 300:
        end_t = min(deadline - 250, time.time() + 150)
        h_opt, c_opt = pairwise_binary_search(best_h, n, end_t, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            print(f"After pairwise search (n={n}): C5={best_c:.6f}")

    # Phase 3: Multi-start perturbation
    if time.time() < deadline - 250:
        end_t = deadline - 200
        best_h, best_c = multi_start_binary_search(best_h, best_c, n, end_t, rng, n_starts=3)
        print(f"After multi-start (n={n}): C5={best_c:.6f}")

    # Phase 4: Thue-Morse alternative
    if time.time() < deadline - 200:
        h_tm = thue_morse(n)
        c_tm = compute_c5(h_tm, n)
        if c_tm < best_c - 0.001:
            # Thue-Morse is significantly better, optimize it
            end_t = min(deadline - 150, time.time() + 100)
            h_opt, c_opt = directed_binary_search(h_tm, n, end_t, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                print(f"After Thue-Morse optimization: C5={best_c:.6f}")

    # Phase 5: Try higher resolution n=1000
    n_high = 1000
    if time.time() < deadline - 150:
        h_high = expand_resolution(best_h, n, n_high)
        c_high = compute_c5(h_high, n_high)
        if c_high > best_c - 0.005:
            # Optimize at higher resolution
            end_t = min(deadline - 100, time.time() + 120)
            h_opt, c_opt = directed_binary_search(h_high, n_high, end_t, rng, top_k=5)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                n = n_high
                print(f"After high-res search (n={n}): C5={best_c:.6f}")
            
            # More pairwise at high res
            if time.time() < deadline - 80:
                end_t2 = min(deadline - 60, time.time() + 80)
                h_opt, c_opt = pairwise_binary_search(best_h, n, end_t2, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    print(f"After high-res pairwise (n={n}): C5={best_c:.6f}")

    # Phase 6: Even higher resolution n=2000
    n_higher = 2000
    if time.time() < deadline - 100 and n < 1500:
        h_higher = expand_resolution(best_h, n, n_higher)
        c_higher = compute_c5(h_higher, n_higher)
        end_t = min(deadline - 50, time.time() + 80)
        h_opt, c_opt = directed_binary_search(h_higher, n_higher, end_t, rng, top_k=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n = n_higher
            print(f"After n=2000 search: C5={best_c:.6f}")

    # Phase 7: Continuous refinement
    if time.time() < deadline - 30:
        n_final = len(best_h)
        h_cont = project_feasible(best_h.astype(float), n_final)
        
        for alpha in [50000, 500000, 5000000]:
            if time.time() > deadline - 15:
                break
            h_opt = lbfgsb_refine(h_cont, n_final, alpha, min(deadline - 10, time.time() + 25), maxiter=1500)
            c_opt = compute_c5(h_opt, n_final)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h_cont = h_opt
                print(f"After continuous refinement (alpha={alpha}): C5={best_c:.6f}")

    # Phase 8: Final binary polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_bin_final = to_binary(best_h, n_final)
        c_bin_final = compute_c5(h_bin_final, n_final)
        if c_bin_final < best_c:
            best_c = c_bin_final
            best_h = h_bin_final
        
        # Quick local search
        end_t = deadline - 2
        h_opt, c_opt = directed_binary_search(best_h, n_final, end_t, rng, top_k=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    # Verify constraints
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10), "h out of bounds"
    assert abs(best_h.sum() - n_final / 2.0) < 1e-6, f"Sum constraint violated: {best_h.sum()} vs {n_final/2}"
    
    print(f"Final: n={n_final}, C5={best_c:.6f}")
    
    return best_h, best_c, n_final
