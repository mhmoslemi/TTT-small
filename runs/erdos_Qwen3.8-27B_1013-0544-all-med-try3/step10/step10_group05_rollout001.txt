import numpy as np
from scipy.optimize import minimize, line_search

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-18:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-18:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    if abs(h.sum() - target) > 1e-10:
        h = h / h.sum() * target
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return float(val)

def smooth_max_weights(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def grad_c5_smooth(h, n, alpha):
    c = compute_c5_vec(h, n)
    w = smooth_max_weights(c, alpha)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-16:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - h[0:nv])
                grad[0:nv] += wm * (-h[kk:n])
    return grad

    c = compute_c5_vec(h, n)
    val = smooth_max_val(c, alpha)
    penalty = mu * (h.sum() - n / 2.0) ** 2
    return val + penalty

def pair_search(h, n, deadline, delta_sequence, radius=None):
    """Fine pair search: move delta from i to j (or vice versa)."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    if radius is None:
        radius = n  # global
    
    for delta in delta_sequence:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            n_iter = 0
            while n_iter < 50000 and time.time() < deadline:
                n_iter += 1
                # Random pair
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                # Check bounds and try move
                if best_h[i] >= delta and best_h[j] <= 1.0 - delta:
                    best_h[i] -= delta
                    best_h[j] += delta
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-18:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] += delta
                        best_h[j] -= delta
                if best_h[i] <= 1.0 - delta and best_h[j] >= delta:
                    best_h[i] += delta
                    best_h[j] -= delta
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-18:
                        best_c = c
                        improved = True
                    else:
                        best_h[i] -= delta
                        best_h[j] += delta
        # After each delta, reproject
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c

    p3_end = min(time.time() + 180, deadline - 400)
    if p3_end > time.time() + 40 and n_up <= 4000:

                if c3 < best_c:
                    best_c = c3

                    break
                h3a, c3a = active_set_optimize(h_up, n_up, p3_end - 10, top_k, alpha, 0.001)
                if c3a < best_c:
                    best_c = c3a
                    best_h = h3a
                    h_up = h3a

    p4_end = min(time.time() + 120, deadline - 200)

        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: More active set rounds
    p5_end = min(time.time() + 100, deadline - 150)
    if p5_end > time.time() + 15:
        cur_n = len(best_h)
        for top_k, alpha, mu in [(5, 5000000, 0.0005), (10, 50000000, 0.0003),
                                  (15, 500000000, 0.0003), (20, 5000000000, 0.0002)]:
            if time.time() > p5_end - 10:
                break
            h5, c5 = active_set_optimize(best_h, cur_n, p5_end - 5, top_k, alpha, mu)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: Very fine pair search
    p6_end = min(time.time() + 100, deadline - 60)
    if p6_end > time.time() + 15:
        cur_n = len(best_h)
        h6, c6 = directed_pair_search(best_h, cur_n, p6_end, delta_start=0.002, delta_min=1e-8)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Perturbation + reopt
    p7_end = min(time.time() + 60, deadline - 30)
    if p7_end > time.time() + 20:
