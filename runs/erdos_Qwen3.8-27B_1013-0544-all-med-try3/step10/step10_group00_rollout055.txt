from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    cur = int(h_bin.sum())
    if cur > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[:cur-target]] = 0.0
    elif cur < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[:target-cur]] = 1.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def run_lbfgs(h, n, alpha, mu, maxit):
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val
    
    def grad(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha)
        g = h_grad_from_w(w, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, args=(),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-20,
                               'gtol': 1e-16, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def greedy_binary_descent(h, n, deadline, max_passes=200, rng=None):
    """Exhaustive greedy: try all single swaps, pick the best improvement"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    passes = 0
    while time.time() < deadline and passes < max_passes:
        c_vec = compute_c5_vec(h, n)
        cur_max = c_vec.max()
        
        best_gain = 0
        best_i = -1
        best_j = -1
        best_new_max = cur_max
        
        # Try all pairs
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_len = len(h)  # sanity
                new_c = np.correlate(h, 1.0-h, mode='full')
                new_max = new_c.max()  # without dx, since it's a constant factor
                gain = cur_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
                    best_new_max = new_max
                h[i] = 1.0
                h[j] = 0.0
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            cur_max = best_new_max
            best_c = min(best_c, cur_max * 2.0 / n)
            passes += 1
        else:
            break
    
    best_c = compute_c5(h, n)
    return h, best_c
