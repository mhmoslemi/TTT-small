```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
```

But c_k = ∫h(x)(1-h(x+k))dx is quadratic in h, so this is a QCQP (quadratic constrained QP), which is non-convex.

However, if I linearize: c_k ≈ c_k(h₀) + ∇c_k(h₀)·(h-h₀) (first-order approximation), then it becomes a linear program! I can use cvxpy with a linear solver.

This is essentially a **cutting plane** or **linearization** approach:
1. Start with h₀
2. Compute c_k(h₀) for all k
3. Identify active set A = {k: c_k(h₀) is in top-k}
4. Solve LP: min t s.t. c_k(h₀) + ∇c_k(h₀)·(h-h₀) ≤ t for k∈A, ∫h=1, 0≤h≤1
5. Update h₀, go to 2

This might converge faster than L-BFGS for this type of problem!

```python
import cvxpy as cp

def lp_refine(h0, n, deadline, k_active=20, n_iters=50):
    """LP-based cutting plane refinement."""
    h = h0.copy()
    best_c = compute_c5(h, n)
    dx = 2.0/n
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        # Get active set (top k shifts)
        active = np.argpartition(c, -k_active)[-k_active:]
        
        # Compute gradients for active shifts
        grads = np.zeros((len(active), n))
        center = n - 1
        for i, idx in enumerate(active):
            k = idx - center
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grads[i, 0:nv] += dx * (1.0 - h[k:])
                    grads[i, k:] += dx * (-h[0:nv])
            else:
                s = -k
                nv = n - s
                if nv > 0:
                    grads[i, 0:nv] += dx * (-h[s:])
                    grads[i, s:] += dx * (1.0 - h[0:nv])
        
        # LP: min t
        # s.t. c_active + grads @ (h_new - h) <= t  (linearized)
        #      sum(h_new) = n/2
        #      0 <= h_new <= 1
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n/2.0,
        ]
        
        for i in range(len(active)):
            c_lin = c[active[i]] + grads[i] @ (h_var - h)
            constraints.append(c_lin <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
            if h_var.value is not None:
                h_new = np.clip(h_var.value, 0, 1)
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    h = h_new.copy()
                else:
                    # Blend
                    for alpha in [0.5, 0.3, 0.1]:
                        h_blend = (1-alpha)*h + alpha*np.clip(h_var.value, 0, 1)
                        h_blend = project_feasible(h_blend, n)
                        c_blend = compute_c5(h_blend, n)
                        if c_blend < best_c:
                            best_c = c_blend
                            h = h_blend.copy()
                            break
        except Exception:
            pass
    
    return h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, k_top, mu):
    c = compute_c5_vec(x, n)
    if k_top >= len(c):
        val = float(c.max())
    else:
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        val = float(c[top_idx].mean())
    val += mu * (x.sum() - n/2.0)**2
    return val

def topk_grad(x, n, k_top, mu):
    c = compute_c5_vec(x, n)
    if k_top >= len(c):
        top_idx = np.array([np.argmax(c)])
    else:
        top_idx = np.argpartition(c, -k_top)[-k_top:]
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in top_idx:
        k = idx - center
        if k >= 0:
            nv = n - k
            if nv > 1:
                grad[0:nv] += dx * (1.0 - x[k:n])
                grad[k:n] += dx * (-x[0:nv])
        else:
            s = -k
            nv = n - s
            if nv > 1:
                grad[0:nv] += dx * (-x[s:n])
                grad[s:n] += dx * (1.0 - x[0:nv])
    grad /= len(top_idx)
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n/2.0)**2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 1:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            s = -k
            nv = n - s
            if nv > 1:
                grad[0:nv] += wm * (-x[s:n])
                grad[s:n] += wm * (1.0 - x[0:nv])
    grad += 2.0 * mu * (x.sum() - n/2.0)
    return grad

def run_lbfgs(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(10, 100.0), (20, 500.0), (50, 2000.0), (100, 5000.0)]
    for k_top, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(topk_obj, best_h, jac=topk_grad,
                          args=(n, k_top, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def lp_refine(h0, n, deadline, k_active=20, n_iters=100):
    """LP-based cutting plane refinement."""
    h = h0.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        c = compute_c5_vec(h, n)
        active = np.argpartition(c, -k_active)[-k_active:]
        
        # Compute gradients for active shifts
        center = n - 1
        conss = []
        for idx in active:
            k = idx - center
            if k >= 0:
                nv = n - k
                if nv > 1:
                    g = np.zeros(n)
                    g[0:nv] = dx * (1.0 - h[k:n])
                    g[k:n] = g[k:n] - dx * h[0:nv]
                    conss.append((c[idx], g))
            else:
                s = -k
                nv = n - s
                if nv > 1:
                    g = np.zeros(n)
                    g[0:nv] = -dx * h[s:n]
                    g[s:n] = g[s:n] + dx * (1.0 - h[0:nv])
                    conss.append((c[idx], g))
        
        if not conss:
            break
        
        # Solve LP
        try:
            h_var = cp.Variable(n)
            t_var = cp.Variable()
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n / 2.0,
            ]
            for c_val, g in conss:
                constraints.append(c_val + g @ h_var - g @ h <= t_var)
            
            obj = cp.Minimize(t_var)
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.GLOP)
            
            if h_var.value is not None and not np.any(np.isnan(h_var.value)):
                h_lp = np.clip(h_var.value, 0, 1)
                h_lp = project_feasible(h_lp, n)
                c_lp = compute_c5(h_lp, n)
                
                # Try full acceptance and blends
                candidates = [h_lp]
                for alpha in [0.7, 0.5, 0.3]:
                    candidates.append((1-alpha)*best_h + alpha*h_lp)
                
                for cand in candidates:
                    cand = project_feasible(cand, n)
                    c_cand = compute_c5(cand, n)
                    if c_cand < best_c:
                        best_c = c_cand
                        best_h = cand.copy()
                        h = cand.copy()
                        break
                else:
                    # If no improvement, take a small step
                    step = 0.3 * (h_lp - h)
                    h_new = project_feasible(h + step, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                        h = h_new.copy()
        except Exception:
            break
    
    return best_h, best_c

    x_new = np.linspace(0, 2, n_new)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topk_obj(x, n, k_top, mu):
    c = compute_c5_vec(x, n)
    if k_top >= len(c):
        val = float(c.max())
    else:
        top_idx = np.argpartition(c, -k_top)[-k_top:]
        val = float(c[top_idx].mean())
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def topk_grad(x, n, k_top, mu):
    c = compute_c5_vec(x, n)
    if k_top >= len(c):
        top_idx = np.array([int(np.argmax(c))])
    else:
        top_idx = np.argpartition(c, -k_top)[-k_top:]
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in top_idx:
        k = int(idx) - center
        if k >= 0:
            nv = n - k
            if nv > 1:
                grad[0:nv] += dx * (1.0 - x[k:n])
                grad[k:n] += dx * (-x[0:nv])
        else:
            s = -k
            nv = n - s
            if nv > 1:
                grad[0:nv] += dx * (-x[s:n])
                grad[s:n] += dx * (1.0 - x[0:nv])
    grad /= len(top_idx)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def softmax_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def softmax_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 1:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            s = -k
            nv = n - s
            if nv > 1:
                grad[0:nv] += wm * (-x[s:n])
                grad[s:n] += wm * (1.0 - x[0:nv])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def run_lbfgs_topk(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(5, 100.0), (10, 500.0), (20, 2000.0), (50, 5000.0), (100, 10000.0)]
    for k_top, mu in configs:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(topk_obj, best_h, jac=topk_grad,
                          args=(n, k_top, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 50000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def run_lbfgs_soft(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 20.0), (1000, 100.0), (5000, 500.0), (20000, 5000.0)]

        if time.time() > deadline - 10:
            break
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': 30000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def lp_refine(h0, n, deadline, k_active=20, n_iters=100):
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    center = n - 1

    for it in range(n_iters):
        if time.time() > deadline:
            break

        c = compute_c5_vec(h, n)
        if k_active < len(c):
            active = np.argpartition(c, -k_active)[-k_active:]
        else:
            active = np.array([int(np.argmax(c))])

        conss = []
        for idx in active:
            k = int(idx) - center
            if k >= 0:
                nv = n - k
                if nv > 1:
                    g = np.zeros(n)
                    g[0:nv] = dx * (1.0 - h[k:n])
                    g[k:n] = g[k:n] - dx * h[0:nv]
                    conss.append((float(c[idx]), g))
            else:
                s = -k
                nv = n - s
                if nv > 1:
                    g = np.zeros(n)
                    g[0:nv] = -dx * h[s:n]
                    g[s:n] = g[s:n] + dx * (1.0 - h[0:nv])
                    conss.append((float(c[idx]), g))

        if not conss:
            break

        try:
            h_var = cp.Variable(n)
            t_var = cp.Variable()
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n / 2.0,
            ]
            for c_val, g in conss:
                constraints.append(c_val + g @ h_var - g @ h <= t_var)

            obj = cp.Minimize(t_var)
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.GLOP)

            if h_var.value is not None and not np.any(np.isnan(h_var.value)):
                h_lp = np.clip(np.asarray(h_var.value).flatten(), 0, 1)
                h_lp = project_feasible(h_lp, n)

                candidates = [h_lp]
                for alpha in [0.7, 0.5, 0.3, 0.15]:
                    candidates.append((1 - alpha) * best_h + alpha * h_lp)

                found = False
                for cand in candidates:
                    cand = project_feasible(cand, n)
                    c_cand = compute_c5(cand, n)
                    if c_cand < best_c:
                        best_c = c_cand
                        best_h = cand.copy()
                        h = cand.copy()
                        found = True
                        break
                if not found:
                    h_new = project_feasible(0.3 * h + 0.7 * h_lp, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                        h = h_new.copy()
        except Exception:
            break

    return best_h, best_c

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def make_block_fn(n, n_blocks=20, pattern='alt'):
    h = np.zeros(n)
    block = n // n_blocks
    for b in range(0, n, block):
        if pattern == 'alt':
            if (b // block) % 2 == 0:
                h[b:b+block] = 1.0
        else:
            h[b:b+block] = 1.0
    return project_feasible(h, n)

def make_ramp_fn(n, frac=0.5):
    h = np.zeros(n)
    cutoff = int(n * frac)
    h[:cutoff] = 1.0
    return project_feasible(h, n)

def make_sinusoidal_fn(n, freq=3, phase=0):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)
```
