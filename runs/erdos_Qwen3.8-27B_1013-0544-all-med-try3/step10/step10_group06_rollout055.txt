def smooth_lse(c, alpha):
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(shifted - shifted.max())) + 1e-300) / alpha)

def smooth_weights(c, alpha):
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(shifted - shifted.max())
    s = w.sum()
    if s < 1e-300:
        return np.ones_like(c) / len(c)
    return w / s

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_weights(c_raw * dx, alpha)
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - nm1
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wk * (-h[lag:n])
        else:
            m = -lag
            grad[m:n] += wk * (1.0 - h[0:n - m])
            grad[0:n - m] += wk * (-h[m:n])
    return grad * dx

def lbfgsb_opt(h, n, alpha, maxiter=3000, ftol=1e-16):

    pen = 1e7

    def obj_grad(x):

        obj = smooth_lse(c * (2.0 / n), alpha)
        diff = x.sum() - target_sum
        obj += pen * diff * diff
        g = grad_smooth_max(x, n, alpha)
        g += 2.0 * pen * diff
        return obj, g

    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    # Get initial
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 500
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: L-BFGS-B at native resolution with sharp alpha
    alphas = [5000, 50000, 500000, 5000000, 50000000, 500000000]
    maxits = [1000, 2000, 3000, 3000, 3000, 3000]
    for a, mi in zip(alphas, maxits):
        if time.time() > deadline - 500:
            break
        h_new = lbfgsb_opt(h, n, a, maxiter=mi)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new

    # Phase 2: Targeted local search at native resolution
    if time.time() < deadline - 300:
        ls_dead = min(deadline - 200, time.time() + 200)
        h_new, c_new = targeted_ls(best_h, n, ls_dead, rng, top_k=5, max_iter=300000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 3: Fine coordinate search
    if time.time() < deadline - 200:
        fls_dead = min(deadline - 100, time.time() + 120)
        h_new, c_new = fine_coordinate_ls(best_h, n, fls_dead, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 4: L-BFGS-B polish after local search
    if time.time() < deadline - 150:
        for a in [500000000, 2000000000]:
            if time.time() > deadline - 100:
                break
            h_new = lbfgsb_opt(best_h, n, a, maxiter=5000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 5: Go to higher resolution n=1000
    n_high = 1000
    if time.time() < deadline - 120:
        h_high = interp_h(best_h, len(best_h), n_high)
        # Quick L-BFGS-B at high res
        for a in [20000, 200000, 2000000]:
            if time.time() > deadline - 60:
                break
            h_new = lbfgsb_opt(h_high, n_high, a, maxiter=2000)
            c_new = compute_c5(h_new, n_high)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                h_high = h_new

        # Local search at high res
        if time.time() < deadline - 30:
            ls_dead = min(deadline - 10, time.time() + 60)
            h_new, c_new = targeted_ls(best_h, n_high, ls_dead, rng, top_k=4, max_iter=100000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 6: Final fine polish at whatever resolution we have
    if time.time() < deadline - 10:
        n_final = len(best_h)
        fls_dead = min(deadline - 3, time.time() + 30)
        h_new, c_new = fine_coordinate_ls(best_h, n_final, fls_dead, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 7: Absolute final L-BFGS-B
    if time.time() < deadline - 3:
        n_final = len(best_h)
        h_new = lbfgsb_opt(best_h, n_final, 1000000000, maxiter=5000)
        c_new = compute_c5(h_new, n_final)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
