        elif it % 100 == 0:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                ones.remove(i1)
                ones.remove(i2)
                ones.extend([j1, j2])
                zeros.remove(j1)
                zeros.remove(j2)
                zeros.extend([i1, i2])
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
    return best_h, best_c

def fine_coord_descent(h, n, deadline, max_rounds=200):
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        rounds = 0
        while improved and rounds < max_rounds and time.time() < deadline:
            improved = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            w = softmax_weights(c_vec, 100000.0)
            grad = grad_smooth_max(w, best_h, n)
            pos_mask = (grad > 1e-10) & (best_h > 1e-6)
            neg_mask = (grad < -1e-10) & (best_h < 1.0 - 1e-6)
            if not pos_mask.any() or not neg_mask.any():
                break
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            best_gain = 0
            best_i, best_j = -1, -1
            for i in pos_idx[:30]:
                for j in neg_idx[:30]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i, best_j = i, j
            if best_i < 0:
                break
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-12:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                improved = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

def nemesis_subset(h, n, positions, deadline, maxit=3000):
    h_full = h.copy()
    x0 = h_full[positions].copy()
    try:
        def obj_small(x):
            h_tmp = h_full.copy()
            x_clipped = np.clip(x, 0.0, 1.0)
            h_tmp[positions] = x_clipped
            h_tmp = project_feasible(h_tmp, n)
            return compute_c5(h_tmp, n)
        res = minimize(obj_small, x0, method='Nelder-Mead',
                      options={'maxiter': maxit, 'xatol': 1e-12, 'fatol': 1e-14})
        h_new = h_full.copy()
        h_new[positions] = np.clip(res.x, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h_full, compute_c5(h_full, n)

def upsample_to_n(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = beatty_init(500, (np.sqrt(5.0) - 1.0) / 2.0)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    # Prepare multiple resolutions
    res_list = []
    if n_parent not in [500, 1000]:
        res_list.append(n_parent)
    res_list.append(500)
    res_list.append(1000)
    seen = set()
    res_list_dedup = []
    for r in res_list:
        if r not in seen:
            seen.add(r)
            res_list_dedup.append(r)
    res_list = res_list_dedup

    # Phase 1: L-BFGS at each resolution (60s total)
    for n_res in res_list:
        if time.time() > deadline - 400:
            break
        if n_res == n_parent:
            h_start = h_parent.copy()
        else:
            h_start = upsample_to_n(h_parent, n_parent, n_res)
        # Also try structured inits
        h_tm = thue_morse_n(n_res)
        h_beatty = beatty_init(n_res, (np.sqrt(5.0) - 1.0) / 2.0)

        phase_deadline = min(deadline - 300, time.time() + 30)
        for h0 in [h_start, h_tm, h_beatty]:
            if time.time() > phase_deadline:
                break
            h_new, c_new = run_lbfgs(h0, n_res, phase_deadline, alpha=20000, mu=0.01, maxit=1500)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                best_n = n_res

    # Phase 2: Binary SA search at best resolution (250s)
    p2_deadline = min(t0 + 350, deadline - 350)
    if time.time() < p2_deadline:
        h_bin = binarize(best_h, best_n)
        h2, c2 = binary_sa_search(h_bin, best_n, p2_deadline, rng, max_iters=200000, T_init=0.008, T_end=0.0005)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: 2-swap search (150s)
    p3_deadline = min(t0 + 500, deadline - 200)
    if time.time() < p3_deadline:
        h3, c3 = two_swap_search(best_h, best_n, p3_deadline, rng, max_iters=5000)
        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()

    # Phase 4: Another binary SA pass with different params (150s)
    p4_deadline = min(t0 + 650, deadline - 150)
    if time.time() < p4_deadline:
        h_bin2 = binarize(best_h, best_n)
        # Perturb starting point
        ones = np.where(h_bin2 > 0.5)[0]
        zeros = np.where(h_bin2 < 0.5)[0]
        for _ in range(rng.randint(10, 30)):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin2[i], h_bin2[j] = h_bin2[j], h_bin2[i]
        h4, c4 = binary_sa_search(h_bin2, best_n, p4_deadline, rng, max_iters=100000, T_init=0.005, T_end=0.0003)
        if c4 < best_c:
            best_c = c4
            best_h = h4.copy()

    # Phase 5: Nelder-Mead on sensitive positions (80s)
    p5_deadline = min(t0 + 730, deadline - 100)
    if time.time() < p5_deadline:
        c_vec = compute_c5_vec(best_h, best_n)
        w = softmax_weights(c_vec, 200000.0)
        grad = grad_smooth_max(w, best_h, best_n)
        sens = np.argsort(-np.abs(grad))[:20]
        h5, c5 = nemesis_subset(best_h, best_n, sens, p5_deadline, maxit=4000)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: L-BFGS refinement with high alpha (80s)
    p6_deadline = min(t0 + 810, deadline - 80)
    if time.time() < p6_deadline:
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p6_deadline - 5:
                break
            h6, c6 = run_lbfgs(best_h, best_n, min(p6_deadline, time.time() + 25),
                              alpha=alpha, mu=0.0005, maxit=1500)
            if c6 < best_c:
                best_c = c6
                best_h = h6

    # Phase 7: Fine coordinate descent (80s)
    p7_deadline = min(t0 + 890, deadline - 60)
    if time.time() < p7_deadline:
        h7, c7 = fine_coord_descent(best_h, best_n, p7_deadline, max_rounds=200)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Final binary polish (80s)
    p8_deadline = min(t0 + 970, deadline - 40)
    if time.time() < p8_deadline:
        h8, c8 = binary_sa_search(binarize(best_h, best_n), best_n, p8_deadline, rng,
                                  max_iters=80000, T_init=0.003, T_end=0.0001)
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Phase 9: Ultra-fine coord descent (40s)
    if time.time() < deadline - 5:
        h9, c9 = fine_coord_descent(best_h, best_n, deadline - 3, max_rounds=100)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
