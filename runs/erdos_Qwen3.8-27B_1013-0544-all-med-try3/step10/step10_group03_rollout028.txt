            sub = min(h[idx], -diff)
            h[idx] = h[idx] - sub

    return h


def corr_vec_exact(h, n):
    dx = 2.0 / n
    if n <= 1500:
        return np.correlate(h, 1.0 - h, mode="full") * dx
    return fast_corr_vec(h, n)


def fast_corr_vec(h, n):
    dx = 2.0 / n
    if not FAST_CORR_OK or n <= 300:
        return np.correlate(h, 1.0 - h, mode="full") * dx

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    fa = np.fft.rfft(h, size)
    fb = np.fft.rfft((1.0 - h)[::-1], size)
    conv = np.fft.irfft(fa * fb, size)
    return np.real(conv[:need][::-1]) * dx


def compute_c5_exact(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def eval_c(h, n):
    if n <= 1500:
        return compute_c5_exact(h, n)
    return float(np.max(fast_corr_vec(h, n)))


def select_active(c, max_count):
    cmax = float(c.max())
    tol = max(1e-6, 1e-4 * cmax)
    idx = np.where(c >= cmax - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)
    if idx.size > max_count:
        order = idx[np.argsort(-c[idx])][:max_count]
    else:
        order = idx
    return order.astype(int)


def softmax_weights(c, active, alpha):
    if active is None or len(active) == 0:
        return np.array([1.0])
    vals = c[active]
    mx = float(vals.max())
    sh = alpha * (vals - mx)
    w = np.exp(sh)
    s = w.sum()
    if s <= 0.0:
        w = np.zeros_like(w)
        w[int(np.argmax(vals))] = 1.0
        return w
    return w / s


def active_gradient_h(h, n, active, weights):
    dx = 2.0 / n
    pos = n - 1
    grad = np.zeros(n, dtype=float)

    for r, w in zip(active, weights):
        if abs(w) < 1e-16:
            continue
        m = int(r) - pos
        wm = float(w) * dx
        if m == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif m > 0:
            grad[0:n - m] += wm * (1.0 - h[m:n])
            grad[m:n] += -wm * h[0:n - m]
        else:
            p = -m
            grad[p:n] += wm * (1.0 - h[0:n - p])
            grad[0:n - p] += -wm * h[p:n]

    return grad


def smooth_obj_fast(x, n, active, alpha, mu):
    c = fast_corr_vec(x, n)
    if active is None or len(active) == 0:
        val = float(c.max())
    else:
        vals = c[active]
        mx = float(vals.max())
        sh = alpha * (vals - mx)
        logsum = float(np.log(np.sum(np.exp(sh))))
        val = float(mx + logsum / alpha)

    diff = x.sum() - 0.5 * n
    return float(val + mu * diff * diff)


def smooth_grad_fast(x, n, active, alpha, mu):
    c = fast_corr_vec(x, n)
    if active is None or len(active) == 0:
        w = np.array([1.0])
        active = np.array([int(np.argmax(c))], dtype=int)
    else:
        w = softmax_weights(c, active, alpha)

    grad = active_gradient_h(x, n, active, w)
    diff = x.sum() - 0.5 * n
    grad += 2.0 * mu * diff
    return grad


def lbfps_active_run(h, n, deadline):
    h = project_feasible(h, n).copy()
    c = corr_vec_exact(h, n)
    best = float(c.max())
    best_h = h.copy()

    if n < 8 or time.time() >= deadline:
        return best_h, best

    bounds = [(0.0, 1.0) for _ in range(n)]
    schedules = [
        (200.0, 20.0),
        (1000.0, 100.0),
        (5000.0, 500.0),
        (50000.0, 2000.0),
        (200000.0, 10000.0),
    ]

    active_count = max(16, min(64, n // 8))
    if n > 800:
        maxiter_list = [80, 80, 80, 60, 50]
        maxfun_list = [250, 250, 250, 200, 150]
    else:
        maxiter_list = [200, 200, 150, 120, 100]
        maxfun_list = [700, 700, 500, 400, 300]

    for idx, (alpha, mu) in enumerate(schedules):
        if time.time() > deadline - 8:
            break
        c = fast_corr_vec(h, n)
        active = select_active(c, active_count)
        try:
            res = minimize(
                smooth_obj_fast,
                h,
                jac=smooth_grad_fast,
                args=(n, active, alpha, mu),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxiter_list[idx],
                    "maxfun": maxfun_list[idx],
                    "ftol": 1e-18,
                    "gtol": 1e-14,
                    "maxls": 30,
                },
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            c2 = corr_vec_exact(h2, n)
            val2 = float(c2.max())
            if val2 < best - 1e-15:
                best = val2
                best_h = h2.copy()
                h = h2
            elif val2 < best + 0.002:
                h = h2  # keep a trajectory that is close under the exact objective

    h = project_feasible(h, n)
    c = corr_vec_exact(h, n)
    val = float(c.max())
    if val < best:
        best = val
        best_h = h.copy()
    return best_h, best


def pair_coeff(h, n, i):
    L = 2 * n - 1
    pos = n - 1
    dx = 2.0 / n
    a = np.zeros(L, dtype=float)

    fc = n - 1 - i
    sc = i

    if fc > 0:
        a[pos + 1:pos + 1 + fc] = 1.0 - h[i + 1:i + 1 + fc]
        a[pos - fc:pos] -= h[i + 1:i + 1 + fc][::-1]

    if sc > 0:
        a[pos + 1:pos + 1 + sc] -= h[i - 1::-1]
        a[pos - sc:pos] += 1.0 - h[:i]

    a *= dx
    a[pos] = 0.0
    return a


def pair_eval_delta(delta, c, A, scratch, pos, d, dx):
    np.multiply(A, delta, out=scratch)
    scratch += c
    d2 = delta * delta
    scratch[pos] += -2.0 * dx * d2
    if d > 0:
        scratch[pos + d] += dx * d2
        scratch[pos - d] += dx * d2
    return float(scratch.max())


def pair_grid_search(lo, hi, c, A, scratch, pos, d, dx, grid_fracs, cur_best):
    best_delta = 0.0
    best_val = float(cur_best)
    span = hi - lo
    if span <= 1e-13:
        return best_delta, best_val

    span = float(span)
    for t in grid_fracs:
        tf = float(t)
        if tf < 1e-15:
            continue
        delta = lo + tf * span
        val = pair_eval_delta(delta, c, A, scratch, pos, d, dx)
        if val < best_val:
            best_val = val
            best_delta = delta
    return best_delta, best_val


def pair_golden_refine(a, b, c, A, scratch, pos, d, dx, current_best):
    a = float(a)
    b = float(b)
    if not np.isfinite(a) or not np.isfinite(b):
        return 0.0, current_best
    if b <= a:
        b = a + max(1e-8, 0.01 * (a + 1.0))
    if b - a <= 1e-13:
        return 0.0, current_best

    best_delta = a
    best_val = current_best

    fa = pair_eval_delta(a, c, A, scratch, pos, d, dx)
    if fa < best_val:
        best_val = fa
        best_delta = a
    fb = pair_eval_delta(b, c, A, scratch, pos, d, dx)
    if fb < best_val:
        best_val = fb
        best_delta = b

    c1 = b - GOLDEN_RATIO * (b - a)
    c2 = a + GOLDEN_RATIO * (b - a)
    f1 = pair_eval_delta(c1, c, A, scratch, pos, d, dx)
    f2 = pair_eval_delta(c2, c, A, scratch, pos, d, dx)
    if f1 < best_val:
        best_val = f1
        best_delta = c1
    if f2 < best_val:
        best_val = f2
        best_delta = c2

    for _ in range(55):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - GOLDEN_RATIO * (b - a)
            f1 = pair_eval_delta(c1, c, A, scratch, pos, d, dx)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + GOLDEN_RATIO * (b - a)
            f2 = pair_eval_delta(c2, c, A, scratch, pos, d, dx)

        if f1 < best_val:
            best_val = f1
            best_delta = c1
        if f2 < best_val:
            best_val = f2
            best_delta = c2

    mid = 0.5 * (a + b)
    fm = pair_eval_delta(mid, c, A, scratch, pos, d, dx)
    if fm < best_val:
        best_val = fm
        best_delta = mid

    if best_val > current_best - 1e-14:
        return 0.0, current_best
    return best_delta, best_val


def make_binary_seed(h, n):
    b = np.zeros(n, dtype=float)
    k = n // 2
    if k <= 0:
        return b
    idx = np.argsort(h)[::-1][:k]
    b[idx] = 1.0
    # The sum is exactly k = n/2 for even n.
    return b


def make_sine_seed(n, phase=0.0):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.5 * np.cos(0.5 * np.pi * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def resample_h(h, old_n, new_n):
    if old_n == new_n:
        return h.copy()
    xo = np.linspace(0.0, 2.0, old_n)
    xn = np.linspace(0.0, 2.0, new_n)
    return np.interp(xn, xo, h)


def pair_descent(h, n, deadline, seed):
    h = project_feasible(h, n).copy()
    c = corr_vec_exact(h, n)
    cur = float(c.max())
    best_h = h.copy()
    best_val = cur

    if n < 16 or time.time() >= deadline:
        return best_h, best_val

    dx = 2.0 / n
    pos = n - 1
    scratch = np.empty(c.shape, dtype=float)

    if n > 850:
        base_grid = 22
        top_a = max(12, min(40, n // 18))
        top_p_base = 7
    else:
        base_grid = 30
        top_a = max(14, min(56, n // 12))
        top_p_base = 9

    base_fracs = np.linspace(0.0, 1.0, base_grid)
    extra_fracs = np.array([0.0, 0.005, 0.01, 0.02, 0.05, 0.2, 0.5, 0.8, 0.98, 1.0])
    grid_fracs = np.unique(np.concatenate((base_fracs, extra_fracs)))

    no_imp = 0
    round_no = 0
    accepts = 0

    while time.time() < deadline:
        round_no += 1

        c = corr_vec_exact(h, n)
        cur = float(c.max())
        if cur < best_val - 1e-13:
            best_val = cur
            best_h = h.copy()

        active = select_active(c, top_a)
        weights = softmax_weights(c, active, 120000.0)
        g = active_gradient_h(h, n, active, weights)

        P = top_p_base + min(10, no_imp // 3)
        P_use = min(P, max(2, n - 1))
        if P_use >= n:
            P_use = max(1, n - 1)

        low = np.argpartition(g, P_use)[:P_use]
        high = np.argpartition(-g, P_use)[:P_use]

        unique_list = np.unique(np.concatenate((low, high)))
        coeff_map = {}
        for idxv in unique_list:
            ki = int(idxv)
            coeff_map[ki] = pair_coeff(h, n, ki)

        best_pair = None
        best_pair_val = cur - 1e-12

        for ii in range(len(low)):
            i = int(low[ii])
            room_i = 1.0 - h[i]
            if room_i < 1e-12:
                continue
            gi = float(g[i])
            ci = coeff_map[i]

            for jj in range(len(high)):
                j = int(high[jj])
                if j == i:
                    continue
                gj = float(g[j])
                if gi >= gj:
                    continue
                hi = min(room_i, h[j])
                if hi < 1e-12:
                    continue

                A = ci - coeff_map[j]
                A[pos] = 2.0 * dx * (h[j] - h[i])
                d = abs(i - j)

                delta, val = pair_grid_search(0.0, hi, c, A, scratch, pos, d, dx, grid_fracs, cur)
                if delta > 1e-12 and val < best_pair_val:
                    best_pair_val = val
                    best_pair = (i, j, delta, d, hi)

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

        if time.time() > deadline:
            break

        nrand = 6 + (14 if no_imp > 3 else 0)
        if nrand > 0 and time.time() < deadline - 2:
            rng = np.random.default_rng(seed + round_no)
            for rr in range(nrand):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                if g[i] > g[j]:
                    i, j = j, i
                    i = int(i)
                    j = int(j)
                room_i = 1.0 - h[i]
                hi = min(room_i, h[j])
                if hi < 1e-12:
                    continue

                if i not in coeff_map:
                    coeff_map[i] = pair_coeff(h, n, i)
                if j not in coeff_map:
                    coeff_map[j] = pair_coeff(h, n, j)

                A = coeff_map[i] - coeff_map[j]
                A[pos] = 2.0 * dx * (h[j] - h[i])
                d = abs(i - j)

                delta, val = pair_grid_search(0.0, hi, c, A, scratch, pos, d, dx, grid_fracs, cur)
                if delta > 1e-12 and val < best_pair_val:
                    best_pair_val = val
                    best_pair = (i, j, delta, d, hi)

                if time.time() > deadline:
                    break

        if time.time() > deadline:
            break

        if best_pair is not None:
            i, j, delta, d, hi = best_pair
            A = coeff_map[i] - coeff_map[j]
            A[pos] = 2.0 * dx * (h[j] - h[i])

            step = hi / max(base_grid, 4)
            a = max(0.0, delta - step)
            b = min(hi, delta + step)
            if b - a > 1e-12:
                ndelta, nval = pair_golden_refine(a, b, c, A, scratch, pos, d, dx, best_pair_val)
                if ndelta > 1e-12 and nval < best_pair_val:
                    delta = ndelta
                    best_pair_val = nval

            if delta > 1e-12 and best_pair_val < cur - 1e-13:
                h[i] += delta
                h[j] -= delta

                if h[i] > 1.0:
                    h[i] = 1.0
                if h[i] < 0.0:
                    h[i] = 0.0
                if h[j] > 1.0:
                    h[j] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0

                accepts += 1
                no_imp = 0

                if accepts % 35 == 0:
                    h = project_feasible(h, n)
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 12 and time.time() < deadline - 25:
            rngp = np.random.default_rng(seed + 100000 + round_no)
            h2 = h.copy()
            cnt = max(1, int(0.25 * n))
            idxs = rngp.choice(n, size=cnt, replace=False)
            h2[idxs] += rngp.normal(0.0, 0.0025, size=cnt)
            h2 = project_feasible(h2, n)
            c2 = corr_vec_exact(h2, n)
            val2 = float(c2.max())
            if val2 < max(best_val, cur) + 0.005:
                h = h2
                if val2 < best_val:
                    best_val = val2
                    best_h = h.copy()
            no_imp = 0

    h = project_feasible(h, n)
    c = corr_vec_exact(h, n)
    val = float(c.max())
    if val < best_val:
        best_val = val
        best_h = h.copy()

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    global FAST_CORR_OK

    t0 = time.time()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 5.0

    # Read parent if available.
    try:
        raw_h = np.array(initial_h_values, dtype=float).copy()
        raw_n = int(len(raw_h))
    except Exception:
        raw_h = None
        raw_n = 0

    if raw_h is None or raw_n < 16 or not np.all(np.isfinite(raw_h)) or raw_n > 1500:
        base_n = 512
        x = np.linspace(0.0, 2.0, base_n)
        base_h = 0.5 + 0.5 * np.cos(0.5 * np.pi * x)
        base_h = project_feasible(base_h, base_n)
    else:
        base_n = raw_n
        base_h = project_feasible(raw_h, base_n)

    # Validate FFT correlation once.
    try:
        vt = np.random.default_rng(12345).random(400)
        c1 = fast_corr_vec(vt, 400)
        c2 = np.correlate(vt, 1.0 - vt, mode="full")
        if not np.allclose(c1, c2, atol=1e-7, rtol=1e-9):
            FAST_CORR_OK = False
    except Exception:
        FAST_CORR_OK = False

    # Build short list of resolutions to try.
    desired_ns = []
    for nn in (base_n, 512, 640, 768, 1024):
        nn = int(nn)
        if nn % 2 == 0 and 160 <= nn <= 1200 and nn not in desired_ns:
            desired_ns.append(nn)

    if not desired_ns:
        desired_ns = [512]

    candidates = []
    for n in desired_ns:
        if n == base_n:
            hh = base_h.copy()
        else:
            hh = resample_h(base_h, base_n, n)
        hh = project_feasible(hh, n)
        if not np.all(np.isfinite(hh)):
            continue
        cc = eval_c(hh, n)
        if np.isfinite(cc):
            candidates.append((cc, hh, n))

    if not candidates:
        hh = project_feasible(base_h, base_n)
        cc = eval_c(hh, base_n)
        if not np.isfinite(cc):
            hh = np.full(512, 0.5)
            cc = eval_c(hh, base_n)
        candidates.append((cc, hh, base_n))

    # Sort candidates by objective without using lambdas.
    ordered = []
    cand_copy = candidates[:]
    while cand_copy:
        min_idx = 0
        min_val = cand_copy[0][0]
        for ii in range(1, len(cand_copy)):
            if cand_copy[ii][0] < min_val:
                min_val = cand_copy[ii][0]
                min_idx = ii
        ordered.append(cand_copy.pop(min_idx))

    best_c = ordered[0][0]
    best_h = ordered[0][1].copy()
    best_n = ordered[0][2]

    # Add a couple of extra seeds if very cheap.
    extra_seeds = []
    if len(ordered) > 0:
        h0, n0 = ordered[0][1], ordered[0][2]
        hb = make_binary_seed(h0, n0)
        cb = eval_c(hb, n0)
        if np.isfinite(cb):
            extra_seeds.append((cb, hb, n0))

        if n0 == 512:
            hs1 = make_sine_seed(512, 0.35)
            cs1 = eval_c(hs1, 512)
            if np.isfinite(cs1):
                extra_seeds.append((cs1, hs1, 512))

    for cand in extra_seeds:
        candidates.append(cand)
        if cand[0] < best_c:
            best_c = cand[0]
            best_h = cand[1].copy()
            best_n = cand[2]

    # Re-sort merged candidates.
    ordered = []
    cand_copy = candidates[:]
    while cand_copy:
        min_idx = 0
        min_val = cand_copy[0][0]
        for ii in range(1, len(cand_copy)):
            if cand_copy[ii][0] < min_val:
                min_val = cand_copy[ii][0]
                min_idx = ii
        ordered.append(cand_copy.pop(min_idx))

    # Stage 1: short active-set L-BFGS for the most promising resolution(s).
    if deadline - time.time() > 220:
        top_lim = min(2, len(ordered))
        for rank in range(top_lim):
            if time.time() > deadline - 260:
                break
            c0, h0, n0 = ordered[rank]
            if c0 > best_c + 0.0025:
                break
            lb_deadline = min(t0 + 120.0 + 45.0 * rank, deadline - 260)
            if lb_deadline - time.time() > 12:
                h2, c2 = lbfps_active_run(h0, n0, lb_deadline)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = n0
                # update ordered entry in case this candidate improves
                ordered[rank] = (c2, h2, n0)

    # Stage 2: primary pair-descent polish on the current best construction.
    remaining = max(0.0, deadline - time.time())
    if remaining > 60:
        reserve = min(180.0, max(30.0, 0.28 * remaining))
        pd_deadline = deadline - reserve
        if pd_deadline - time.time() > 15:
            h2, c2 = pair_descent(best_h, best_n, pd_deadline, int(seed))
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = best_n

    # Stage 3: try a distinct promising candidate if time remains.
    second = None
    for cand in ordered:
        if cand[2] != best_n and cand[0] <= best_c + 0.003:
            second = cand
            break

    if second is not None and deadline - time.time() > 90:
        sec_end = min(deadline - 70.0, time.time() + min(80.0, 0.35 * (deadline - time.time())))
        if sec_end - time.time() > 15:
            h2, c2 = pair_descent(second[1], second[2], sec_end, int(seed) + 1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = second[2]

    # Stage 4: binary-flavored start from the current best, if close enough.
    if deadline - time.time() > 55:
        hb = make_binary_seed(best_h, best_n)
        cb = eval_c(hb, best_n)
        if np.isfinite(cb) and cb <= best_c + 0.009:
            bin_end = min(deadline - 25.0, time.time() + min(55.0, 0.3 * (deadline - time.time())))
            if bin_end - time.time() > 12:
                h2, c2 = pair_descent(hb, best_n, bin_end, int(seed) + 2000)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = best_n

    # Stage 5: one last active-set L-BFGS polish.
    if deadline - time.time() > 18:
        final_lb_deadline = deadline - 2.0
        h2, c2 = lbfps_active_run(best_h, best_n, final_lb_deadline)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = best_n

    # Final exact evaluation.
    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5_exact(best_h, best_n)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), int(best_n)
