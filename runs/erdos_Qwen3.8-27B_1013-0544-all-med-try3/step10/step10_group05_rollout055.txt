1. Find t where the "knee" is (where the active constraint changes)
2. The minimum of a convex PL function is where 0 ∈ subdifferential

Algorithm:
- f(t) = max_k(A_k + D_k*t)
- f(0) = max_k(A_k), achieved by k0 = argmax A_k
- f(1) = max_k(A_k + D_k), achieved by k1 = argmax(A_k + D_k)
- If f'(0) ≥ 0 (i.e., D_{k0} ≥ 0): min at t=0
- If f'(1) ≤ 0 (i.e., D_{k1} ≤ 0): min at t=1
- Otherwise: bisect to find t* where the active line's D changes sign

For bisection:
- lo=0, hi=1
- Midpoint tm = (lo+hi)/2
- Find k* = argmax_k(A_k + D_k*tm)
- If D_{k*} > 0: f'(tm) > 0, so min is to the left → hi = tm
- If D_{k*} < 0: f'(tm) < 0, so min is to the right → lo = tm
- If D_{k*} = 0: tm is optimal (or close enough)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Spread the difference over non-saturated entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()

def coord_descent(h, n, deadline, n_sweeps=None):
    """Efficient coordinate descent for minimizing max_k c_k."""
    h = h.copy()
    if n_sweeps is None:
        n_sweeps = 200
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        c_full = np.correlate(h, 1 - h, mode='full')  # without dx scaling
        
        for i in range(n):
            if time.time() > deadline:
                break
            # Compute D_k for all k
            # k ranges from -(n-1) to (n-1), stored at index 0 to 2n-2
            # For shift k: v1 = (i+k valid), v2 = (i-k valid)
            # D = v1*(1-h[i+k]) - v2*h[i-k]
            
            # Vectorized computation
            k_vals = np.arange(-(n-1), n)  # 2n-1 values
            idx1 = i + k_vals  # positions for h[i+k]
            idx2 = i - k_vals  # positions for h[i-k]
            
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            
            D = np.zeros(2*n-1)
            if v1.any():
                D[v1] = 1.0 - h[idx1[v1]]
            if v2.any():
                D[v2] -= h[idx2[v2]]
            
            A = c_full - D * h[i]
            
            # Minimize max(A + D*t) for t in [0,1] via bisection
            lo, hi_t = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi_t) / 2
                vals = A + D * mid
                k_star = np.argmax(vals)
                if D[k_star] > 1e-14:
                    hi_t = mid
                elif D[k_star] < -1e-14:
                    lo = mid
                else:
                    break
            
            new_val = (lo + hi_t) / 2
            delta = new_val - h[i]
            if abs(delta) > 1e-15:
                c_full += D * delta
                h[i] = new_val
        
        # Project to feasible set
        h = project_feasible(h, n)
    
    return h
