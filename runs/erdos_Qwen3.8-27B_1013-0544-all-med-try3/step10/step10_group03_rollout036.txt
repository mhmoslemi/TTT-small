def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:

    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:

import numpy as np

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 0:
                h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot > 0:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    threshold = 1e-12
    for mm in range(2 * n - 1):
        if w[mm] < threshold:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

    m = vals.max()
    sh = alpha * (vals - m)
    mx = sh.max()

    if time.time() < deadline - 200:
        t5 = min(time.time() + 120, deadline - 200)
        best_h, best_c = run_lbfgs(best_h, n, t5,
            alphas=[100000, 500000], maxiter_each=4000)

    # Phase 6: Random perturbations
    if time.time() < deadline - 120:
        t6 = min(time.time() + 80, deadline - 120)
        best_h, best_c = run_perturb_reopt(best_h, n, t6, n_tries=5, base_scale=0.01)

    # Phase 7: Subgradient with momentum
    if time.time() < deadline - 60:
        t7 = min(time.time() + 40, deadline - 60)
        h7, c7 = run_subgrad_momentum(best_h, n, t7, max_iter=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Fine local search
    if time.time() < deadline - 15:
        t8 = min(time.time() + 30, deadline - 15)
        best_h, best_c = fine_search_local(best_h, n, t8, radius=25, top_k=10)

    # Phase 9: Final high-alpha polish
    if time.time() < deadline - 5:
        h9, c9 = run_lbfgs(best_h, n, deadline - 3,
                           alphas=[1000000], maxiter_each=3000)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
