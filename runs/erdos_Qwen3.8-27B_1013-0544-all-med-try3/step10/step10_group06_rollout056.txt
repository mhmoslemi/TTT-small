def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

        grad = grad_c5_smooth(xp, n, alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        w = np.exp(shifted - mx)
    else:
        w = np.exp(shifted)
    return w / w.sum()

def grad_smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_lags(h, n)
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        lag = k - (n - 1)
        wi = w[k]
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:n] += wi * (1.0 - h[0:n - lag])
            grad[0:n - lag] += wi * (-h[lag:n])
        else:
            idx = -lag
            grad[0:n - idx] += wi * (1.0 - h[idx:n])
            grad[idx:n] += wi * (-h[0:n - idx])
    return grad * dx

def grad_single_lag(h, n, k_idx):
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:n] = (1.0 - h[0:n - lag]) * dx
        grad[0:n - lag] = (-h[lag:n]) * dx
    else:
        idx = -lag
        grad[0:n - idx] = (1.0 - h[idx:n]) * dx
        grad[idx:n] = (-h[0:n - idx]) * dx
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=5000):
    target_sum = n / 2.0
    penalty_coeff = 1e6

    def obj_and_grad(x):
        xp = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(xp, n)
        obj_val = smooth_max_value(c, alpha)
        diff = xp.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = grad_smooth_max(xp, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',

                                'iprint': -1})

                best_c = c_opt
                best_h = h_opt
                h1 = h_opt

    # Phase 4: Try n=2000 (150s)
    n2 = 2000
    if elapsed() < 480:
        h2 = expand_resolution(best_h, len(best_h), n2)
        for alpha in [50000, 500000, 5000000]:
            if elapsed() > 430:
                break
            rem = min(25, 430 - elapsed())
            if rem < 3:
                break
            h_opt = lbfgsb_optimize(h2, n2, alpha, t0 + elapsed() + rem, maxiter=1500)
            c_opt = compute_c5(h_opt, n2)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h2 = h_opt
            else:
                h2 = h_opt

        if elapsed() < 470:
            h_opt, c_opt = top_lag_local_search(h2, n2, t0 + 470, rng, m_top=8, max_rounds=60000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

        if elapsed() < 478:
            h_opt = lbfgsb_optimize(h2, n2, 100000000, t0 + 480, maxiter=2000)
            c_opt = compute_c5(h_opt, n2)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Alternative starting points at n=1000 (80s)
    if elapsed() < 570:
        for trial in range(4):
            if elapsed() > 560:
                break
            h_alt = boundary_construction(1000, rng)
            # Quick optimize
            h_opt = lbfgsb_optimize(h_alt, 1000, 100000, t0 + elapsed() + 10, maxiter=1000)
            c_opt = compute_c5(h_opt, 1000)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
            # If close, do more work
            if c_opt < best_c + 0.005:
                h_opt = lbfgsb_optimize(h_opt, 1000, 5000000, t0 + elapsed() + 10, maxiter=2000)
                c_opt = compute_c5(h_opt, 1000)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 6: Final intensive local search at best resolution (100s)
    if elapsed() < budget_s - 60:
        n_cur = len(best_h)
        h_opt, c_opt = top_lag_local_search(best_h, n_cur, t0 + elapsed() + 60, rng,
                                            m_top=12, max_rounds=200000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 7: Subgradient at best resolution
    if elapsed() < budget_s - 30:
        n_cur = len(best_h)
        h_opt, c_opt = subgradient_descent(best_h, n_cur, t0 + elapsed() + 25, rng, max_steps=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 8: Final L-BFGS-B polish
    if elapsed() < budget_s - 10:
        n_cur = len(best_h)
        h_opt = lbfgsb_optimize(best_h, n_cur, 200000000, t0 + budget_s - 12, maxiter=3000)
        c_opt = compute_c5(h_opt, n_cur)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 9: Very final local search
    if elapsed() < budget_s - 3:
        n_cur = len(best_h)
        h_opt, c_opt = top_lag_local_search(best_h, n_cur, t0 + budget_s - 2, rng,
                                            m_top=6, max_rounds=50000)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))

    return best_h, best_c, len(best_h)
