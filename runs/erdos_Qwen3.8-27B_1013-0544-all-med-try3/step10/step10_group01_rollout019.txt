            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_optimize(h, n, deadline, alphas, mu=0.01, maxit_list=None):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        if maxit_list is not None and i < len(maxit_list):
            maxit = maxit_list[i]
        else:
            maxit = max(300, int(remaining * 3))
        try:
            res = minimize(obj_lm, best_h, jac=grad_lm,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': maxit * 8,
                                   'maxls': 40, 'maxcor': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_optimize(h, n, deadline, top_k=5, alphas=None, rounds=20):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if alphas is None:
        alphas = [200000, 1000000, 5000000, 20000000, 100000000]
    for rnd in range(rounds):
        remaining = deadline - time.time()
        if remaining < 15:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        threshold = max_val - max(1e-9, 0.0005 * max_val)
        active_idx = np.where(c_vec >= threshold)[0]
        if len(active_idx) > top_k:
            active_idx = np.array(np.argsort(-c_vec[active_idx])[:top_k])
        if len(active_idx) == 0:
            break
        for alpha in alphas:
            if time.time() > deadline - 10:
                break
            maxit = max(200, int((deadline - time.time()) * 1.5))
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                              args=(n, active_idx, alpha, 0.001), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-15, 'maxfun': maxit * 6,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def binary_sa(h, n, deadline, iterations=200000, t_start=0.5, t_end=0.01):
    """Simulated annealing on binary structure."""
    best_h = np.where(h > 0.5, 1.0, 0.0).copy()
    # Ensure exactly n/2 ones
    ones_count = int(best_h.sum())
    target = n // 2
    if ones_count > target:
        ones_idx = np.where(best_h == 1)[0]
        extra = ones_count - target
        to_zero = rng_global.choice(ones_idx, size=extra, replace=False)
        best_h[to_zero] = 0.0
    elif ones_count < target:
        zeros_idx = np.where(best_h == 0)[0]
        need = target - ones_count
        to_one = rng_global.choice(zzeros_idx, size=need, replace=False)
        best_h[to_one] = 1.0
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    cur_h = best_h.copy()
    cur_c = best_c
    ones_set = np.where(cur_h == 1)[0]
    zeros_set = np.where(cur_h == 0)[0]
    n_iter = min(iterations, int((deadline - time.time()) * 800))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max(1, n_iter))
        a = ones_set[rng_global.randint(len(ones_set))]
        b = zeros_set[rng_global.randint(len(zeros_set))]
        if a == b:
            continue
        cur_h[a], cur_h[b] = 0.0, 1.0
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or rng_global.random() < np.exp(-delta / max(T, 1e-9)):
            cur_c = new_c
            ones_set = np.where(cur_h == 1)[0]
            zeros_set = np.where(cur_h == 0)[0]
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[a], cur_h[b] = 1.0, 0.0
    return best_h, best_c

def binary_greedy(h, n, deadline, max_rounds=200):
    """Greedy binary local search: always take the best swap."""
    best_h = np.where(h > 0.5, 1.0, 0.0).copy()
    ones_count = int(best_h.sum())
    target = n // 2
    if ones_count > target:
        ones_idx = np.where(best_h == 1)[0]
        to_zero = rng_global.choice(ones_idx, size=ones_count - target, replace=False)
        best_h[to_zero] = 0.0
    elif ones_count < target:
        zeros_idx = np.where(best_h == 0)[0]
        to_one = rng_global.choice(zeros_idx, size=target - ones_count, replace=False)
        best_h[to_one] = 1.0
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones_set = np.where(best_h == 1)[0]
        zeros_set = np.where(best_h == 0)[0]
        if len(ones_set) == 0 or len(zeros_set) == 0:
            break
        best_delta = 0
        best_a = -1
        best_b = -1
        # Sample a subset for speed
        if len(ones_set) > 50:
            sample_ones = rng_global.choice(ones_set, 50, replace=False)
        else:
            sample_ones = ones_set
        for a in sample_ones:
            if time.time() > deadline:
                break
            for b in zeros_set[:min(50, len(zeros_set))]:
                best_h[a], best_h[b] = 0.0, 1.0
                c = compute_c5(best_h, n)
                best_h[a], best_h[b] = 1.0, 0.0
                delta = c - best_c
                if delta < best_delta:
                    best_delta = delta
                    best_a = a
                    best_b = b
        if best_a >= 0:
            best_h[best_a], best_h[best_b] = 0.0, 1.0
            best_h = project_feasible(best_h, n)
            best_c = compute_c5(best_h, n)
        else:
            break
    return best_h, best_c

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    k = 1
    while len(positions) < n_ones and k < 10000:
        pos = int(k * alpha_param) % n
        positions.add(pos)
        k += 1
    for pos in positions:
        h[pos] = 1.0
    return project_feasible(h, n)

def make_period1_start(n, rng):
    """Create a period-1 symmetric starting point."""
    half = n // 2
    h_half = np.zeros(half)
    n_ones_half = half // 2
    idx = rng.choice(half, size=n_ones_half, replace=False)
    h_half[idx] = 1.0
    h = np.zeros(n)
    h[0:half] = h_half
    h[half:n] = h_half
    return project_feasible(h, n)

def interp_to_n(h, n_old, n_new):
    """Interpolate h from n_old points to n_new points."""
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def subgrad_active(h, n, deadline, max_iter=30000):
    """Subgradient on active shifts with momentum."""
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    momentum = np.zeros(n)
    target = n / 2.0
    lr = 0.02
    prev_h = best_h.copy()
    for t in range(max_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        tol = max(1e-8, 0.001 * max_val)
        active = np.where(c_vec >= max_val - tol)[0]
        dx_val = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                grad[k:n] += (-best_h[0:nv]) * dx_val
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                grad[0:nv] += (-best_h[kk:n]) * dx_val
        grad /= len(active)
        # Nesterov momentum
        beta = (1 + np.sqrt(1 + 4 * t * t)) / (2 * t + 2)
        momentum = beta * momentum - lr * grad
        new_h = best_h + momentum
        new_h = project_feasible(new_h, n)
        c_new = compute_c5(new_h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = new_h
            prev_h = new_h.copy()
        else:
            lr *= 0.99
            if lr < 0.0005:
                lr = 0.0005
            momentum = momentum * 0.5
    return best_h, best_c

def fine_coord_search(h, n, deadline, radius=30, deltas=None):
    """Fine coordinate pair search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, max(1, n // 200)):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = project_feasible(best_h, n)
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = project_feasible(best_h, n)
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
    return best_h, best_c

rng_global = np.random.RandomState(42)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    rng_global = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n0 = 500

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    # Phase 1: Quick L-BFGS polish on parent
    p1_end = t0 + 100
    h1, c1 = lbfgs_optimize(best_h, best_n, p1_end,
        alphas=[50, 500, 5000, 50000, 500000, 5000000],
        mu=0.01, maxit_list=[500, 800, 1000, 1200, 1500, 2000])
    if c1 < best_c:
        best_c = c1
        best_h = h1

    # Phase 2: Active set optimization
    p2_end = min(t0 + 250, deadline - 500)
    h2, c2 = active_set_optimize(best_h, best_n, p2_end, top_k=5,
        alphas=[500000, 5000000, 50000000], rounds=15)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: Binary SA for structure discovery
    p3_end = min(t0 + 450, deadline - 350)
    h3, c3 = binary_sa(best_h, best_n, p3_end, iterations=100000, t_start=0.3, t_end=0.005)
    if c3 < best_c:
        best_c = c3
        best_h = h3
        best_n = best_n

    # Phase 4: Continuous polish from best binary
    p4_end = min(t0 + 550, deadline - 250)
    h4, c4 = lbfgs_optimize(best_h, best_n, p4_end,
        alphas=[1000, 10000, 100000, 1000000, 10000000],
        mu=0.001, maxit_list=[500, 800, 1000, 1200, 1500])
    if c4 < best_c:
        best_c = c4
        best_h = h4

    # Phase 5: Try higher resolution n=600
    n_high = 600
    if n_high / 2.0 == int(n_high / 2.0) and time.time() < deadline - 250:
        p5_end = min(t0 + 750, deadline - 200)
        h_high = interp_to_n(best_h, best_n, n_high)
        hh, ch = lbfgs_optimize(h_high, n_high, p5_end,
            alphas=[50, 500, 5000, 50000, 500000],
            mu=0.01, maxit_list=[300, 500, 700, 900, 1000])
        if ch < best_c:
            best_c = ch
            best_h = hh
            best_n = n_high

    # Phase 6: Subgradient on active set
    p6_end = min(t0 + 850, deadline - 150)
    h6, c6 = subgrad_active(best_h, best_n, p6_end, max_iter=50000)
    if c6 < best_c:
        best_c = c6
        best_h = h6

    # Phase 7: Active set again
    p7_end = min(t0 + 950, deadline - 80)
    h7, c7 = active_set_optimize(best_h, best_n, p7_end, top_k=4,
        alphas=[1000000, 10000000, 100000000], rounds=12)
    if c7 < best_c:
        best_c = c7
        best_h = h7

    # Phase 8: Fine coordinate search
    p8_end = deadline - 5
    h8, c8 = fine_coord_search(best_h, best_n, p8_end, radius=20,
        deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    if c8 < best_c:
        best_c = c8
        best_h = h8

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
