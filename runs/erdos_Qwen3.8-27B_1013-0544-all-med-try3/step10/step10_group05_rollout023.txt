```python
def projected_subgradient(h, n, deadline, initial_lr=0.01, decay=0.99):
    dx = 2.0 / n
    lr = initial_lr
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        # Active set: lags within 1% of max
        threshold = c_max - 0.01 * c_max
        active = np.where(c_vec >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for m in active:
            s = m - (n - 1)
            # For each j:
            # contribution from h[j]: (1-h[j+s]) if valid
            # contribution from h[j] as second factor: -h[j-s] if valid
            if s >= 0:
                # h[j] appears as first factor for j in [0, n-s)
                # h[j] appears as second factor for j in [s, n) as -h[j-s]
                js = np.arange(0, n-s)
                g[js] += (1 - h[js + s])
                js2 = np.arange(s, n)
                g[js2] += (-h[js2 - s])
            else:
                ss = -s
                # h[j] appears as first factor for j in [0, n-ss) as (1-h[j+ss])
                # Wait, s is negative, so j+s = j-ss
                # h[j] as first factor: (1-h[j+s]) = (1-h[j-ss]) valid when 0 <= j-ss < n, i.e., ss <= j < n+ss
                # h[j] as second factor: -h[j-s] = -h[j+ss] valid when 0 <= j+ss < n, i.e., 0 <= j < n-ss
                js = np.arange(ss, n)
                g[js] += (1 - h[js - ss])
                js2 = np.arange(0, n - ss)
                g[js2] += (-h[js2 + ss])
        
        g *= dx / len(active)
        
        # Projected gradient step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < c_max:
            h = h_new
        else:
            lr *= decay
        
        lr = max(lr, 1e-8)
    return h, compute_c5(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        # Water-filling projection
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            adj = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            adj = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Fix any tiny sum error by distributing to elements with room
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            idx = np.argsort(-room)[:max(1, int(abs(diff)*2))]
        else:
            room = h.copy()
            idx = np.argsort(-room)[:max(1, int(abs(diff)*2))]
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(c_vec, h, n, active_frac=0.05):
    """Compute weighted subgradient of max c_k w.r.t. h"""
    dx = 2.0 / n
    c_max = c_vec.max()
    threshold = c_max - active_frac * c_max
    active = np.where(c_vec >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    weights = c_vec[active] - threshold
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for w, m in zip(weights, active):
        s = int(m) - (n - 1)
        if s >= 0:
            # h[j] as first factor: (1-h[j+s]) for j in [0, n-s)
            g[:n-s] += w * (1.0 - h[s:])
            # h[j] as second factor: -h[j-s] for j in [s, n)
            g[s:] += w * (-h[:n-s])
        else:
            ss = -s
            # h[j] as first factor: (1-h[j-ss]) for j in [ss, n)
            g[ss:] += w * (1.0 - h[:n-ss])
            # h[j] as second factor: -h[j+ss] for j in [0, n-ss)
            g[:n-ss] += w * (-h[ss:])
    
    return g * dx

def subgradient_method(h, n, deadline, lr0=0.05, momentum=0.9, active_frac=0.05):
    """Direct projected subgradient with momentum"""
    dx = 2.0 / n
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t = 0
    lr = lr0
    
    while time.time() < deadline:
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        
        g = compute_subgradient(c_vec, h, n, active_frac)
        
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        t += 1
        lr = lr0 / (1 + 0.01 * t * t**0.5)
        
        # Check convergence
        if t % 10000 == 0 and lr < 1e-10:
            break
    
    return best_h, best_c
```

```python
def compute_subgrad_fast(h, n, active):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in active:
        s = int(m) - (n - 1)
        if s >= 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] += -h[:n-s]
        else:
            k = -s
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] += -h[k:]
    return g * dx / len(active)
```

This is O(n * |active|) per computation. For n=500 and |active|=10, that's 5000 ops. Very fast.

```python
def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = np.where(h < 1.0)[0]
        else:
            room = np.where(h > 0.0)[0]
        if len(room) > 0:
            h[room[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            grad[:n-k] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:n-k])
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - x[:n-kk])
            grad[:n-kk] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def coord_descent(h, n, deadline):
    """Coordinate descent: adjust one coordinate at a time"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline - 5:
        improved = False
        for i in range(n):
            if time.time() > deadline - 10:
                break
            # Try increasing h[i]
            for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                new_val = best_h[i] + delta
                if 0.0 <= new_val <= 1.0:
                    best_h[i] = new_val
                    # Fix sum constraint
                    diff = n/2.0 - best_h.sum()
                    # Distribute to other coordinates
                    if abs(diff) > 1e-14:
                        mask = np.arange(n) != i
                        cnt = mask.sum()
                        best_h[mask] += diff / cnt
                        best_h = np.clip(best_h, 0.0, 1.0)
                        # Re-fix
                        best_h = project_feasible(best_h, n)
                    c = compute_c5(best_h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        break
                    else:
                        best_h[i] -= delta
        
        if not improved:
            break
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = np.where(h < 1.0 - 1e-15)[0]
        else:
            room = np.where(h > 1e-15)[0]
        if len(room) > 0:
            h[room[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, n, c_vec, active_idx):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in active_idx:
        s = int(m) - (n - 1)
        if s >= 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] += -h[:n-s]
        else:
            k = -s
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] += -h[k:]
    g *= dx / len(active_idx)
    return g

def get_active_idx(c_vec, frac=0.05, top_k=None):
    c_max = c_vec.max()
    threshold = c_max - frac * c_max
    active = np.where(c_vec >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c_vec))])
    if top_k is not None and len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    return active

def subgradient_opt(h, n, deadline, lr0=0.02, momentum=0.9, frac=0.05, top_k=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    t = 0
    stall = 0
    batch = 50
    while time.time() < deadline - 3:
        for _ in range(batch):
            if time.time() > deadline - 3:
                break
            c_vec = compute_c5_vec(h, n)
            c_max = c_vec.max()
            active_idx = get_active_idx(c_vec, frac, top_k)
            g = compute_subgrad(h, n, c_vec, active_idx)
            g -= g.mean()
            lr = lr0 / (1.0 + 0.0005 * np.sqrt(t))
            v = momentum * v - lr * g
            h = project_feasible(h + v, n)
            t += 1
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200 and t > 5000:
                rng = np.random.RandomState(t % 100000)
                h = project_feasible(h + rng.normal(0, 0.003 * t / 10000, n), n)
                v *= 0.5
                stall = 0
    return best_h, best_c

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()
    sh = alpha * (vals - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = int(idx) - idx_center
        wm = wi * dx
        if k >= 0:
            grad[:n-k] += wm * (1.0 - x[k:])
            grad[k:] += wm * (-x[:n-k])
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - x[:n-kk])
            grad[:n-kk] += wm * (-x[kk:])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

            best_h = hs.copy()

    # Phase 1: Subgradient optimization (fast, many iterations)
    p1_end = min(t0 + 200, deadline - 550)
    if p1_end > time.time() + 30:
        h1, c1 = subgradient_opt(best_h, best_n, p1_end, lr0=0.03, momentum=0.9, frac=0.05, top_k=None)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: L-BFGS active set refinement
    p2_end = min(time.time() + 130, deadline - 400)
    if p2_end > time.time() + 15:
        for (tk, al) in [(3, 20000), (5, 100000), (8, 500000), (10, 2000000)]:
            if time.time() > p2_end - 10:
                break
            h2, c2 = lbfgs_active(best_h, best_n, p2_end - 3, tk, al, mu=0.001, maxit=1000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Second round subgradient with finer settings
    p3_end = min(time.time() + 150, deadline - 300)
    if p3_end > time.time() + 20:
        h3, c3 = subgradient_opt(best_h, best_n, p3_end, lr0=0.008, momentum=0.93, frac=0.02, top_k=5)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: L-BFGS polish
    p4_end = min(time.time() + 80, deadline - 200)
    if p4_end > time.time() + 10:
        for (tk, al) in [(3, 500000), (5, 2000000), (7, 10000000)]:
            if time.time() > p4_end - 8:
                break
            h4, c4 = lbfgs_active(best_h, best_n, p4_end - 3, tk, al, mu=0.0005, maxit=800)
            if c4 < best_c:
                best_c = c4
                best_h = h4

    # Phase 5: Binary simulated annealing (different exploration)
    p5_end = min(time.time() + 100, deadline - 200)
    if p5_end > time.time() + 30:
        h5, c5 = binary_sa(best_h, best_n, p5_end, n_iter=300000)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        # Also try to use binary solution as start for continuous optimization
        h5b = (h5 > 0.5).astype(float)
        h5b = project_feasible(h5b, best_n)
        if time.time() < p5_end - 20:
            h5c, c5c = lbfgs_active(h5b, best_n, p5_end - 5, 5, 200000, 0.001, 500)
            if c5c < best_c:
                best_c = c5c
                best_h = h5c

    # Phase 6: Targeted local search
    p6_end = min(time.time() + 130, deadline - 150)
    if p6_end > time.time() + 20:
        h6, c6 = targeted_search(best_h, best_n, p6_end, top_lags=5,
                                  deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try higher resolution
    n_up = best_n * 2
    p7_end = min(time.time() + 120, deadline - 80)
    if p7_end > time.time() + 50 and n_up <= 1500:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.003:
            h7, c7 = subgradient_opt(h_up, n_up, p7_end - 30, lr0=0.015, momentum=0.9, frac=0.05)
            if c7 < best_c:
                best_c = c7
                best_h = h7
                best_n = n_up
            if time.time() < p7_end - 15:
                h7b, c7b = lbfgs_active(best_h, best_n, p7_end - 5, 5, 500000, 0.001, 600)
                if c7b < best_c:
                    best_c = c7b
                    best_h = h7b
                    best_n = n_up

    # Phase 8: Final targeted search with very small steps
    p8_end = deadline - 5
    if p8_end > time.time() + 15:
        h8, c8 = targeted_search(best_h, best_n, p8_end, top_lags=4,
                                  deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        if c8 < best_c:
            best_c = c8
            best_h = h8

    # Final: one more subgradient pass with very small lr
    if time.time() < deadline - 10:
        h9, c9 = subgradient_opt(best_h, best_n, deadline - 5, lr0=0.002, momentum=0.95, frac=0.01, top_k=3)
        if c9 < best_c:
            best_c = c9
            best_h = h9

    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)

    return best_h, best_c, final_n
```
