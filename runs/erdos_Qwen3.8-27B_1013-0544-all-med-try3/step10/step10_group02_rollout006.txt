def subgrad_refine(h, n, deadline, n_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.003
    best_eta = eta
    patience = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(h, n)
        c_max = c_vec.max()
        tol = max(1e-9, 0.0008 * c_max)
        active = np.where(c_vec >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c_vec)])
        
        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                if nv > 0:
                    grad[0:nv] += (1.0 - h[k:n]) * dx
                    grad[k:n] += (-h[0:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                if nv > 0:
                    grad[kk:n] += (1.0 - h[0:nv]) * dx
                    grad[0:nv] += (-h[kk:n]) * dx
        grad /= len(active)
        
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new.copy()
            if c_new < best_c + 1e-15:
                best_eta = min(eta * 1.01, 0.015)
                eta = best_eta
            patience = 0
        else:
            patience += 1
            if patience > 50:
                eta *= 0.95
                patience = 0
                if eta < 1e-5:
                    break
        
        h = h_new
    
    return best_h, best_c

def fine_pair_search(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    for d in deltas:
        if time.time() > deadline - 2:
            break
        improved = True
        max_rounds = 200
        rounds = 0
        while improved and rounds < max_rounds and time.time() < deadline - 2:
            improved = False
            rounds += 1
            for i in range(0, n, 1):
                if time.time() > deadline - 2:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    elif h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return best_h, best_c

def slqp_polish(h, n, deadline, alpha=1000000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, args=(n, alpha, 0.0),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h.copy(), n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Phase 1: Quick L-BFGS at original resolution
    p1_end = t0 + 40
    if time.time() < p1_end:
        h1, c1 = run_lbfgs(best_h, best_n, p1_end,
            alphas=[100, 1000, 10000], mu=0.01,
            maxiters=[1500, 2000, 2000])
        if c1 < best_c:
            best_c = c1
            best_h = h1
            best_n = n_orig
    
    # Phase 2: Upsample to n=1000 and optimize
    n_fine = 1000
    x_fine = np.linspace(0, 2, n_fine, endpoint=False)
    x_coarse = np.linspace(0, 2, best_n, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, best_h)
    h_fine = project_feasible(h_fine, n_fine)
    c_fine = compute_c5(h_fine, n_fine)
    
    # Decide which resolution to work at
    if c_fine < best_c:
        best_c = c_fine
        best_h = h_fine
        best_n = n_fine
    else:
        # Work at n_orig but use fine as separate track
        pass
    
    work_h = best_h.copy()
    work_n = best_n
    
    # If best is at n_orig, try to improve at n_fine separately and compare at end
    h_fine_track = h_fine.copy() if n_fine != best_n else best_h.copy()
    c_fine_track = compute_c5(h_fine_track, n_fine)
    
    # Phase 2a: Main L-BFGS at best resolution
    p2_end = t0 + 400
    if time.time() < p2_end:
        h2, c2 = run_lbfgs(work_h, work_n, p2_end,
            alphas=[50, 200, 1000, 5000, 20000, 100000, 500000],
            mu=0.008,
            maxiters=[2000, 2500, 3000, 3000, 3500, 4000, 4000])
        if c2 < best_c:
            best_c = c2
            best_h = h2
            best_n = work_n
    
    # Phase 2b: Also run L-BFGS at n_fine
    if time.time() < deadline - 500:
        p2b_end = min(time.time() + 180, deadline - 500)
        h2f, c2f = run_lbfgs(h_fine_track, n_fine, p2b_end,
            alphas=[50, 500, 5000, 50000, 500000],
            mu=0.008,
            maxiters=[2000, 3000, 3000, 3000, 3000])
        c2f = compute_c5(project_feasible(h2f, n_fine), n_fine)
        if c2f < c_fine_track:
            c_fine_track = c2f
            h_fine_track = project_feasible(h2f, n_fine)
    
    # Phase 3: Active-set optimization at best resolution
    if time.time() < deadline - 350:
        p3_end = min(time.time() + 180, deadline - 350)
        h3, c3 = run_active(best_h, best_n, p3_end, top_k=8,
            alphas=[50000, 200000, 1000000, 5000000])
        if c3 < best_c:
            best_c = c3
            best_h = h3
            best_n = best_n
    
    # Phase 3b: Active-set at n_fine
    if time.time() < deadline - 250:
        p3b_end = min(time.time() + 120, deadline - 250)
        h3f, c3f = run_active(h_fine_track, n_fine, p3b_end, top_k=8,
            alphas=[50000, 500000, 5000000])
        c3f = compute_c5(project_feasible(h3f, n_fine), n_fine)
        if c3f < c_fine_track:
            c_fine_track = c3f
            h_fine_track = project_feasible(h3f, n_fine)
    
    # Phase 4: Fourier perturbation + re-optimization (at best resolution)
    if time.time() < deadline - 250:
        p4_end = min(time.time() + 120, deadline - 250)
        for m_modes in [15, 30, 50]:
            for t_mix in [0.3, 0.5, 0.7]:
                if time.time() > p4_end - 10:
                    break
                h_fp, c_fp = fourier_perturb(best_h, best_n, m_modes, t_mix, p4_end)
                if c_fp < best_c:
                    best_c = c_fp
                    best_h = h_fp.copy()
                    # Quick re-optimize from this new point
                    if time.time() < p4_end - 20:
                        h4, c4 = run_lbfgs(best_h, best_n, p4_end - 5,
                            alphas=[5000, 50000, 500000], mu=0.005,
                            maxiters=[1500, 2000, 2000])
                        if c4 < best_c:
                            best_c = c4
                            best_h = h4
                # Also try Fourier on fine track
                if m_modes <= 30:
                    h_fp_fine, c_fp_fine = fourier_perturb(h_fine_track, n_fine, m_modes, t_mix, p4_end)
                    c_fp_fine = compute_c5(project_feasible(h_fp_fine, n_fine), n_fine)
                    if c_fp_fine < c_fine_track:
                        c_fine_track = c_fp_fine
                        h_fine_track = project_feasible(h_fp_fine, n_fine)
    
    # Phase 5: Multiple restarts with perturbation
    if time.time() < deadline - 200:
        p5_end = min(time.time() + 150, deadline - 200)
        for restart in range(6):
            if time.time() > p5_end - 15:
                break
            noise_scale = 0.02 / (restart + 1)
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_n)
            c_pert = compute_c5(h_pert, best_n)
            # Quick optimization from perturbed point
            p5_sub = min(time.time() + 20, p5_end)
            h5, c5 = run_lbfgs(h_pert, best_n, p5_sub,
                alphas=[10000, 100000], mu=0.005,
                maxiters=[1500, 2000])
            if c5 < best_c:
                best_c = c5
                best_h = h5
            # Also try from fine track
            if n_fine != best_n and time.time() < p5_end - 15:
                h_pert_f = h_fine_track + rng.randn(n_fine) * noise_scale
                h_pert_f = project_feasible(h_pert_f, n_fine)
                p5f_sub = min(time.time() + 15, p5_end)
                h5f, c5f = run_lbfgs(h_pert_f, n_fine, p5f_sub,
                    alphas=[10000, 100000], mu=0.005,
                    maxiters=[1000, 1500])
                c5f = compute_c5(project_feasible(h5f, n_fine), n_fine)
                if c5f < c_fine_track:
                    c_fine_track = c5f
                    h_fine_track = project_feasible(h5f, n_fine)
    
    # Choose the best between tracks
    if c_fine_track < best_c:
        best_c = c_fine_track
        best_h = h_fine_track.copy()
        best_n = n_fine
    else:
        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5(best_h, best_n)
    
    # Phase 6: Subgradient refinement
    if time.time() < deadline - 60:
        p6_end = min(time.time() + 80, deadline - 50)
        h6, c6 = subgrad_refine(best_h, best_n, p6_end, n_iter=8000)
        if c6 < best_c:
            best_c = c6
            best_h = h6
            best_n = best_n
    
    # Phase 7: Active-set final
    if time.time() < deadline - 50:
        p7_end = min(time.time() + 40, deadline - 45)
        h7, c7 = run_active(best_h, best_n, p7_end, top_k=4,
            alphas=[500000, 5000000, 50000000])
        if c7 < best_c:
            best_c = c7
            best_h = h7
            best_n = best_n
    
    # Phase 8: Fine pair search
    if time.time() < deadline - 20:
        p8_end = min(time.time() + 40, deadline - 15)
        h8, c8 = fine_pair_search(best_h, best_n, p8_end, radius=30)
        if c8 < best_c:
            best_c = c8
            best_h = h8
            best_n = best_n
    
    # Phase 9: SLSQP final polish
    if time.time() < deadline - 3:
        p9_end = deadline - 2
        h9, c9 = slqp_polish(best_h, best_n, p9_end, alpha=5000000, maxiter=800)
        if c9 < best_c:
            best_c = c9
            best_h = h9
            best_n = best_n
    
    # Final verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
