from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            frac = min(1.0, diff / tot)
            h = h + frac * room
        else:
            tot = h.sum()
            if tot < 1e-12:
                break
            frac = min(1.0, -diff / tot)
            h = h * (1.0 - frac)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:

            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:

            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

    """Generate a Sturmian-like binary sequence of length n with n/2 ones."""
    target = n // 2
    h = np.zeros(n)
    # Use the property: position i is 1 if frac(alpha * i) < alpha
    # But adjusted to get exactly n/2 ones
    vals = (np.arange(n) * slope) % 1.0
    # Sort by fractional part and take the largest n/2
    order = np.argsort(vals)
    h[order[:target]] = 1.0
    return h

def sturmian_beatty(n, slope):
    """Beatty sequence approach: mark positions floor(k*slope) mod n for k=1,..."""
    target = n // 2
    h = np.zeros(n)
    seen = np.zeros(n, dtype=bool)
    count = 0
    for k in range(1, 10 * n):
        pos = int(k * slope) % n
        if not seen[pos]:
            seen[pos] = True
            h[pos] = 1.0
            count += 1
            if count >= target:

    # Fix if needed
    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:int(diff)]:
            h[idx] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        for idx in ones[-int(-diff):]:
            h[idx] = 0.0
    return h

    h = h.copy()

        current_max = c_vec.max()

        old_c5 = current_max * dx

            if delta < 0:

            else:

        else:
            h[i] = 1.0
            h[j] = 0.0
        
        # Cool down

def lbfgs_round(h, n, deadline, alpha, mu, maxit):
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return float(val)

    def grad(x):

        if mu > 0:

def sturmian_binary(n, slope):

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(500, 0.5)
        n_parent = 500

    h_parent = project_feasible(h_parent, n_parent)

    best_h = h_parent.copy()
    best_c = compute_c5(h_parent, n_parent)
    best_n = n_parent

    resolutions = [498, 500, 502, 504, 506]
    if n_parent not in resolutions:
        resolutions.append(n_parent)
    resolutions = sorted(set(r for r in resolutions if 480 <= r <= 520))

    num_res = len(resolutions)
    time_per_res = (deadline - t0 - 60) / max(1, num_res)

    slope = 0.6180339887498949

    for ri, n in enumerate(resolutions):
        if time.time() > deadline - 30:
            break
        res_dl = min(deadline - 30, t0 + 60 + time_per_res * (ri + 1))

        inits = []
        if n == n_parent:
            inits.append(h_parent.copy())
        else:
            inits.append(resample_to(h_parent, n_parent, n))
        inits.append(sturmian_smooth(n, slope, mix=0.2))
        inits.append(sturmian_smooth(n, slope + 0.01, mix=0.3))

        for h_init in inits:
            if time.time() > res_dl - 5:
                break
            h_opt, c_opt = optimize_resolution(h_init, n, res_dl, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n

    # Phase 2: More thorough optimization on best resolution and neighbors
    if time.time() < deadline - 40:
        focus_n = best_n
        p2_dl = min(deadline - 20, time.time() + 300)
        h_work = best_h.copy()
        alphas2 = [10000, 50000, 200000, 500000, 2000000, 5000000]
        for alpha in alphas2:
            if time.time() > p2_dl - 5:
                break
            h_new, c_new = lbfgs_round(h_work, focus_n, min(p2_dl, time.time() + 45),
                                       alpha, 0.003, maxit=2000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h_work = h_new.copy()

    # Phase 3: Binary local search at best resolution
    if time.time() < deadline - 30:
        p3_dl = min(deadline - 15, time.time() + 120)
        h_bin, c_bin = binary_local_search(best_h, best_n, p3_dl, rng, max_iters=500000)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 4: Try neighboring resolutions with best solution resampled
    if time.time() < deadline - 25:
        for dn in [-2, 2, -4, 4]:
            if time.time() > deadline - 15:
                break
            n_try = best_n + dn
            if n_try < 480 or n_try > 520:
                continue
            h_res = resample_to(best_h, best_n, n_try)
            h_res2 = sturmian_smooth(n_try, slope, mix=0.15)
            for h_init in [h_res, h_res2]:
                if time.time() > deadline - 15:
                    break
                h_opt, c_opt = optimize_resolution(h_init, n_try, deadline - 10, rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
                    best_n = n_try

    # Phase 5: Final continuous refinement
    if time.time() < deadline - 8:
        h_final, c_final = lbfgs_round(best_h, best_n, deadline - 5, 1000000, 0.001, maxit=3000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    # Phase 6: Ultra-fine pair descent
    if time.time() < deadline - 3:
        h_final, c_final = cont_pair_search(best_h, best_n, deadline - 2, rng, max_iters=200000)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
