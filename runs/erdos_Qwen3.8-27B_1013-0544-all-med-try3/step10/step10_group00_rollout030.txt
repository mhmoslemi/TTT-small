import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(diff))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(-diff))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = compute_c5_vec(x, n)
        val = smooth_max(c, alpha)
        if mu > 0:
            val += mu * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = compute_c5_vec(x, n)
        w = smooth_max_w(c, alpha)
        g = h_grad_from_w(w, x, n)
        if mu > 0:
            g += 2.0 * mu * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, jac=grad, args=(),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def make_beatty(n, alpha):
    h = np.zeros(n)
    target = n // 2
    positions = set()
    for k in range(1, 10 * n):
        pos = int(k * alpha) % n
        if pos not in positions:
            positions.add(pos)
            if len(positions) >= target:
                break
    for p in positions:
        h[p] = 1.0
    return project_feasible(h, n)

def make_thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    h = np.zeros(n)
    for i in range(n):
        # Count bits in i
        cnt = 0
        x = i
        while x:
            cnt += x & 1
            x >>= 1
        h[i] = float(cnt % 2)
    return project_feasible(h, n)

def fine_coord_descent(h, n, deadline, steps=None, max_rounds=200):

            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:50]:
                for j in neg_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_gain:

                        best_i = i
                        best_j = j
            
            if best_i < 0:

            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:

                improved_any = True
            else:
                best_h[best_i] += amt

def compute_c5_vec(h, n):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def binarize(h, n):

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    diff = target - h_bin.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:int(np.ceil(abs(diff)))]:
                h_bin[zeros[idx]] = 1.0
        else:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:int(np.ceil(abs(diff)))]:
                h_bin[ones[idx]] = 0.0
    return h_bin

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        return m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    return w

def h_grad_from_w(w, h, n):
    dx = 2.0 / n
    n_full = len(w)
    grad = np.zeros(n)
    for mm in range(n_full):
        if w[mm] < 1e-13:
            continue
        k = mm - (n - 1)
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _lbfgs_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    val = smooth_max(c, alpha)
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return val

def _lbfgs_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    w = smooth_max_w(c, alpha)
    g = h_grad_from_w(w, x, n)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def run_lbfgs(h, n, deadline, alpha, mu, maxit):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbfgs_obj, h, jac=_lbfgs_grad, args=(n, alpha, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxit, 'ftol': 1e-24,
                               'gtol': 1e-18, 'maxfun': maxit * 10,
                               'maxls': 50, 'maxcor': 50})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return project_feasible(h, n), compute_c5(h, n)

def fine_coord_descent(h, n, deadline, steps=None, max_rounds=200):
    if steps is None:
        steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)

    for step in steps:
        if time.time() > deadline:
            break
        improved_any = True
        rounds = 0
        while improved_any and rounds < max_rounds and time.time() < deadline:
            improved_any = False
            rounds += 1
            c_vec = compute_c5_vec(best_h, n)
            w = smooth_max_w(c_vec, 200000.0)
            grad = h_grad_from_w(w, best_h, n)

            pos_mask = (grad > 1e-12) & (best_h > 1e-8)
            neg_mask = (grad < -1e-12) & (best_h < 1.0 - 1e-8)

            if not pos_mask.any() or not neg_mask.any():
                break

            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]

            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:50]:
                for j in neg_idx[:50]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j

            if best_i < 0:
                break

            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            best_h[best_i] -= amt
            best_h[best_j] += amt
            best_h = project_feasible(best_h, n)
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                improved_any = True
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
                best_h = project_feasible(best_h, n)
    return best_h, best_c

        best_c = c2
        best_h = h2.copy()

    # Phase 3: Best-improvement local search (100s)
    p3_deadline = t0 + 530
    h3, c3 = best_improve_local_search(best_h, n, p3_deadline, num_candidates=25)
    if c3 < best_c:
        best_c = c3
        best_h = h3.copy()

    # Phase 4: L-BFGS refinement with increasing smoothness (150s)
    p4_deadline = t0 + 680
    if time.time() < p4_deadline:
        for alpha in [5000, 20000, 100000, 500000]:
            if time.time() > p4_deadline - 10:
                break
            h4, c4 = run_lbfgs(best_h, n, min(p4_deadline, time.time() + 30),
                              alpha=alpha, mu=0.001, maxit=2000)
            if c4 < best_c:
                best_c = c4
                best_h = h4

    # Phase 5: Fine coordinate descent (100s)
    p5_deadline = t0 + 780
    if time.time() < p5_deadline:
        h5, c5 = fine_coord_descent(best_h, n, p5_deadline,
                                   steps=[0.001, 0.0005, 0.0002, 0.0001,
                                          0.00005, 0.00002, 0.00001, 0.000005],
                                   max_rounds=300)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Second tabu round from perturbed start (100s)
    p6_deadline = t0 + 880
    if time.time() < p6_deadline:
        h6_start = binarize(best_h, n)
        ones_p = np.where(h6_start > 0.5)[0]
        zeros_p = np.where(h6_start < 0.5)[0]
        for _ in range(8):
            ii = ones_p[rng.randint(len(ones_p))]
            jj = zeros_p[rng.randint(len(zeros_p))]
            h6_start[ii], h6_start[jj] = h6_start[jj], h6_start[ii]
        h6, c6 = tabu_search(h6_start, n, p6_deadline, rng, tabu_size=20,
                             max_iters=150000, candidates=12)
        if c6 < best_c:
            best_c = c6
            best_h = h6.copy()

    # Phase 7: Final L-BFGS + softmax (100s)
    p7_deadline = t0 + 980
    if time.time() < p7_deadline:
        h7, c7 = run_lbfgs(best_h, n, min(p7_deadline, time.time() + 35),
                          alpha=1000000, mu=0.0001, maxit=2000)
        if c7 < best_c:
            best_c = c7
            best_h = h7
        h7b, c7b = softmax_postprocess(best_h, n, p7_deadline, num_iters=30)
        if c7b < best_c:
            best_c = c7b
            best_h = h7b

    # Phase 8: Final tabu + fine tuning (85s)
    p8_deadline = t0 + 1065
    if time.time() < p8_deadline - 20:
        h8, c8 = tabu_search(binarize(best_h, n), n, p8_deadline - 20, rng,
                            tabu_size=15, max_iters=80000, candidates=15)
        if c8 < best_c:
            best_c = c8
            best_h = h8

        h8b, c8b = fine_coord_descent(best_h, n, p8_deadline,
                                     steps=[0.00002, 0.00001, 0.000005, 0.000002],
                                     max_rounds=150)
        if c8b < best_c:
            best_c = c8b
            best_h = h8b

    # Final: ensure exact constraint
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    return best_h, best_c, best_n
