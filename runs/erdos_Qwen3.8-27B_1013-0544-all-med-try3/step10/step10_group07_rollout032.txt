            if s < 1e-14:

def build_v_matrix(h, n):
    """Build v[p, li] where li = m + (n-1).
    v[p, li] = (1-h[p+m])*[0<=p+m<n] - h[p-m]*[0<=p-m<n] for m = li-(n-1)
    """
    n_lags = 2 * n - 1
    v = np.zeros((n, n_lags))
    for p in range(n):
        # First term: (1-h[p+m]) for m where 0<=p+m<n
        # m range: -p <= m < n-p, intersect with [-(n-1), n-1]

        for idx in perm:
            i = zeros[idx]
            base = corr + v[i, :]
            # Check all ones simultaneously
            diffs = base[None, :] - v[ones, :]
            maxes = diffs.max(axis=1)
            good = np.where(maxes < cur_max - 1e-12)[0]
            if len(good) > 0:
                best_idx = good[np.argmin(maxes[good])]
                j = ones[best_idx]
                # Perform swap
                corr = base - v[j, :]
                h[i] = 1.0
                h[j] = 0.0
                cur_max = corr.max()
                # Update v matrix
                for pp in range(n):
                    for mm in (i - pp, j - pp, pp - i, pp - j):
                        if -(n-1) <= mm <= n-1:
                            update_v_entry(v, h, pp, mm, n)
                found_swap = True
                stall = 0
                swap_count += 1
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break

        if not found_swap:
            stall += 1
            if stall > 2000:
                break
            # Perturbation: random swap
            if stall % 50 == 0:
                i = int(rng.choice(zeros))
                j = int(rng.choice(ones))
                old_max = cur_max
                new_corr = corr + v[i, :] - v[j, :]
                new_max = new_corr.max()
                # Accept with probability
                if new_max <= old_max:
                    accept = True
                else:
                    delta_w = (new_max - old_max) / n_lags
                    accept = rng.random() < np.exp(-delta_w * 5.0)
                if accept:
                    corr = new_corr
                    h[i] = 1.0
                    h[j] = 0.0
                    cur_max = new_max
                    for pp in range(n):
                        for mm in (i - pp, j - pp, pp - i, pp - j):
                            if -(n-1) <= mm <= n-1:
                                update_v_entry(v, h, pp, mm, n)
                    swap_count += 1

    return best_h, best_max * dx

def simulated_annealing(h_init, n, deadline, seed=42, T_start=0.5, T_end=0.05, cool_rate=0.9995):
    """Simulated annealing on binary solutions."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    h = h_init.copy()
    corr = corr_array(h, n).astype(float).copy()
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    v = build_v_matrix(h, n)
    rng = np.random.RandomState(seed)
    T = T_start
    step = 0

    while time.time() < deadline:
        step += 1
        if step % 1000 == 0:
            T = max(T_end, T * cool_rate)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))

        new_corr = corr + v[i, :] - v[j, :]
        new_max = new_corr.max()
        delta = new_max - cur_max

        if delta < 0 or rng.random() < np.exp(-delta * 10.0 / max(T, 0.01)):
            corr = new_corr
            h[i] = 1.0
            h[j] = 0.0
            cur_max = new_max
            for pp in range(n):
                for mm in (i - pp, j - pp, pp - i, pp - j):
                    if -(n-1) <= mm <= n-1:
                        update_v_entry(v, h, pp, mm, n)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    return best_h, best_max * dx

def slsqp_optimize(h_init, n, deadline, max_alpha=2e6, maxiter=3000):
    """SLSQP optimization for continuous h."""
    from scipy.optimize import minimize

    h = h_init.copy()
    dx = 2.0 / n
    n_lags = 2 * n - 1

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        # Smooth approximation of max
        alpha = 8000.0
        shifted = g - m
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm + 1e-15).sum())
        return m + logsum / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, n)
    except Exception:
        h_out = h

    return h_out

def multi_start_binary(n, deadline, seed=42, n_starts=5):
    """Try multiple random initial binary solutions and local search."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = np.inf

    for s in range(n_starts):
        if time.time() > deadline:
            break
        time_per = max(1.0, (deadline - time.time()) / (n_starts - s))
        # Random balanced binary
        idx = rng.permutation(n)
        h_rand = np.zeros(n)
        h_rand[idx[:n//2]] = 1.0
        # Local search
        dh, dc = local_search(h_rand, n, time.time() + time_per, max_swaps=50000, seed=seed+s)
        if dc < best_c:
            best_c = dc
            best_h = dh.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    # Binarize
    h_bin = make_binary(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Local search on binary (25% time)
    p1 = t0 + 0.25 * (budget_s - 3)
    dh, dc = local_search(h_bin, n0, p1, max_swaps=500000, seed=seed)
    if dc < best_c:
        best_c = dc
        best_h = dh.copy()
        n = n0

    # Phase 2: Simulated annealing (25% time)
    p2 = t0 + 0.50 * (budget_s - 3)
    if time.time() < p2:
        dh, dc = simulated_annealing(best_h, n, p2, seed=seed+10)
        if dc < best_c:
            best_c = dc
            best_h = dh.copy()
        # Follow up with local search
        if time.time() < p2:
            dh2, dc2 = local_search(best_h, n, p2, max_swaps=200000, seed=seed+20)
            if dc2 < best_c:
                best_c = dc2
                best_h = dh2.copy()

    # Phase 3: SLSQP continuous optimization (20% time)
    p3 = t0 + 0.70 * (budget_s - 3)
    if time.time() < p3:
        h_cont = best_h.astype(float).copy()
        time_left = p3 - time.time()
        iter_per = max(100, int(time_left / 5))
        h_cont = slsqp_optimize(h_cont, n, p3, maxiter=iter_per)
        cc = compute_c5(h_cont, n)
        if cc < best_c:
            best_c = cc
            best_h = h_cont.copy()

    # Phase 4: Try other resolutions (15% time)
    p4 = t0 + 0.85 * (budget_s - 3)
    other_ns = [200, 300, 350, 400]
    if time.time() < p4:
        time_per_res = max(5.0, (p4 - time.time()) / len(other_ns))
        for nh in other_ns:
            if time.time() > p4 - 2:
                break
            if nh == n:
                continue
            # Interpolate best_h to new resolution
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            h_interp = np.interp(xn, xo, best_h)
            h_interp = project_feasible(h_interp, nh)
            h_b = make_binary(h_interp, nh)
            # Local search
            rd = p4 - 1
            dh, dc = local_search(h_b, nh, min(time.time() + time_per_res, rd),
                                  max_swaps=100000, seed=seed+100+nh)
            if dc < best_c:
                best_c = dc
                best_h = dh.copy()
                n = nh

    # Phase 5: Final polish (remaining time)
    if time.time() < deadline - 2:
        # SLSQP from best
        h_polish = best_h.astype(float).copy()
        h_polish = slsqp_optimize(h_polish, n, deadline - 1, maxiter=5000)
        cp = compute_c5(h_polish, n)
        if cp < best_c:
            best_c = cp
            best_h = h_polish.copy()

        # Binary local search from polished
        if time.time() < deadline - 1:
            h_b2 = make_binary(best_h, n)
            dh, dc = local_search(h_b2, n, deadline - 0.5, max_swaps=200000, seed=seed+500)
            if dc < best_c:
                best_c = dc
                best_h = dh.copy()

    # Final: ensure feasibility and compute
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
