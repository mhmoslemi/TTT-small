def project_feasible(h,n):

   target=n/2.0
   # iterative proportional adjustment
   for _ in range(1000):
      diff=target-h.sum()
      if abs(diff) < 1e-13: break
      if diff>0:
         room=1-h
         tot=room.sum()
         if tot<1e-15: break
         h += (diff/tot)*room
         h = np.clip(h,0,1)
      else:
         tot=h.sum()
         if tot<1e-15: break
         h += (diff/tot)*h # diff negative reduces proportionally
         h = np.clip(h,0,1)
   # exact-ish residual
   diff=target-h.sum()
   if abs(diff) > 0:
      idx = int(np.argmax(1-h)) if diff>0 else int(np.argmax(h))
      h[idx] = float(np.clip(h[idx]+diff,0,1))
   # If still residual due to no room, distribute
   diff=target-h.sum()
   if abs(diff) > 1e-12:
      # adjust several
      order=np.arange(n)
      step=diff/n
      h += step
      h=np.clip(h,0,1)
      # redo iterative maybe

def project_feasible(h,n):
    h=np.array(h,float,copy=True)
    h=np.clip(h,0,1)
    target=n/2.0
    for _ in range(1000):
       diff=target-h.sum()
       if abs(diff) < 1e-14: break
       if diff > 0:
          room=1-h
          tot=room.sum()
          if tot <= 1e-15: break
          h += (diff/tot)*room
          np.clip(h,0,1,out=h)
       else:
          tot=h.sum()
          if tot <= 1e-15: break
          h += (diff/tot)*h
          np.clip(h,0,1,out=h)
    diff=target-h.sum()
    if abs(diff) > 1e-12:
       # adjust a few
       if diff>0:
          order=np.argsort(1-h)[::-1]
       else:
          order=np.argsort(h)
       for idx in order:
          room = (1-h[idx]) if diff>0 else h[idx]
          step = min(abs(diff), room)
          h[idx] += np.sign(diff)*step
          diff -= np.sign(diff)*step
          if abs(diff) <= 1e-14: break
    np.clip(h,0,1,out=h)
    return h

import time

from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0

    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
            np.clip(h, 0.0, 1.0, out=h)
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
            np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
            sign = 1.0
        else:
            order = np.argsort(h)
            sign = -1.0
        for idx in order:
            room = (1.0 - h[idx]) if sign > 0.0 else h[idx]
            step = min(abs(diff), room)
            if step <= 0.0:
                continue
            h[idx] += sign * step
            diff -= sign * step
            if abs(diff) <= 1e-14:
                break

    np.clip(h, 0.0, 1.0, out=h)
    return h


def eval_solution(h, n):
    h = np.asarray(h, dtype=float)
    s = np.correlate(h, 1.0 - h, mode="full")
    smax = float(s.max())
    c = smax * (2.0 / float(n))
    return s, smax, c


def resize_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return np.array(h, dtype=float, copy=True)
    if n_new % n_old == 0:
        return np.repeat(h, n_new // n_old)
    if n_old % n_new == 0:
        factor = n_old // n_new
        return h.reshape(n_new, factor).mean(axis=1).copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def make_mechanical(n, alpha):
    t = np.arange(n, dtype=float)
    vals = np.mod(t * alpha, 1.0)
    h = (vals < 0.5).astype(float)
    return project_feasible(h, n)


def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    return h


def build_update_indices(n):
    p = np.arange(n, dtype=np.int32).reshape(n, 1)
    q = np.arange(n, dtype=np.int32).reshape(1, n)
    c = np.int32(n - 1)
    first = c + p - q
    second = c + q - p
    return first.astype(np.int32, copy=False), second.astype(np.int32, copy=False)


def build_pair_arrays(n):
    ci = []
    cj = []
    center = n - 1
    for m in range(2 * n - 1):
        k = m - center
        start = max(0, k)
        stop = min(n, n + k)
        ii = np.arange(start, stop, dtype=np.int32)
        jj = ii - k
        ci.append(ii)
        cj.append(jj)
    return ci, cj


def top_k_indices(s, k):
    m = len(s)
    if k <= 0:
        return np.empty(0, dtype=np.intp)
    if k >= m:
        return np.arange(m, dtype=np.intp)
    part = np.argpartition(-s, k - 1)[:k]
    order = np.argsort(-s[part])
    return part[order]


def select_candidates(score, D, valid, top_count, rand_count, rng):
    idx = np.flatnonzero(valid)
    if idx.size == 0:
        return np.empty(0, dtype=np.intp)
    if idx.size <= top_count + rand_count:
        return idx
    sc = score[idx]
    k = min(top_count, idx.size)
    if k <= 0:
        part = np.empty(0, dtype=np.intp)
    else:
        part = np.argpartition(-sc, k - 1)[:k]
    top_idx = idx[part]
    rsize = min(rand_count, idx.size)
    if rsize > 0:
        rand_idx = idx[rng.randint(0, idx.size, size=rsize)]
        return np.unique(np.concatenate((top_idx, rand_idx)))
    return np.unique(top_idx)


def transfer_delta(tmp, h, inc, dec, d, idx_first, idx_second, n):
    tmp.fill(0.0)
    if d == 0.0:
        return
    dh = d * h
    domh = d * (1.0 - h)

    # inc = inc position is added, dec position is removed.
    # idx_first[p]: p is the h-factor, idx_second[p]: p is the 1-h-factor.
    tmp[idx_first[inc]] += domh
    tmp[idx_second[inc]] -= dh
    tmp[idx_first[dec]] -= domh
    tmp[idx_second[dec]] += dh

    cross = d * d
    center = n - 1
    tmp[center] -= 2.0 * cross
    tmp[center + inc - dec] += cross
    tmp[center + dec - inc] += cross


def best_trial_transfer(s, h, inc, dec, D, fracs, tmp, idx_first, idx_second, n, threshold):
    D = float(D)
    if D > 1.0:
        D = 1.0
    if D <= 1e-14:
        return None, threshold
    best_d = None
    best_mx = threshold
    for frac in fracs:
        d = float(frac) * D
        if d <= 1e-15:
            continue
        transfer_delta(tmp, h, inc, dec, d, idx_first, idx_second, n)
        s += tmp
        mx = float(s.max())
        if mx < best_mx - 1e-12:
            best_mx = mx
            best_d = d
        s -= tmp
    return best_d, best_mx


def targeted_search(h, n, pair_i, pair_j, idx_first, idx_second, tmp, rng, deadline,
                    active_k=30, top_cand=10, rand_cand=5, fracs=None):
    if fracs is None:
        fracs = (1.0, 0.5, 0.2, 0.05)

    h = np.array(h, dtype=float, copy=True)
    s = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(s.max())

    best_h = h.copy()
    best_s = s.copy()
    best_max = cur_max
    stall = 0
    max_stall = 6

    while time.time() < deadline and stall < max_stall:
        s = np.correlate(h, 1.0 - h, mode="full")
        cur_max = float(s.max())
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_h = h.copy()
            best_s = s.copy()

        active = top_k_indices(s, active_k)
        improved = False

        for m in active:
            if time.time() > deadline:
                break
            mi = int(m)
            ii = pair_i[mi]
            jj = pair_j[mi]
            if ii.size == 0:
                continue

            hi = h[ii]
            hj = h[jj]
            D = np.minimum(hi, 1.0 - hj)
            score = hi * (1.0 - hj)
            valid = (D > 1e-9) & (score > 1e-12)
            if not np.any(valid):
                continue

            cand = select_candidates(score, D, valid, top_cand, rand_cand, rng)
            if cand.size == 0:
                continue

            local_best_d = None
            local_best_mx = cur_max
            local_inc = -1
            local_dec = -1

            for pos in cand:
                inc = int(jj[pos])
                dec = int(ii[pos])
                Dval = float(min(h[dec], 1.0 - h[inc]))
                if Dval <= 1e-10:
                    continue
                best_d, best_mx = best_trial_transfer(
                    s, h, inc, dec, Dval, fracs, tmp,
                    idx_first, idx_second, n, local_best_mx
                )
                if best_d is not None:
                    local_best_d = best_d
                    local_best_mx = best_mx
                    local_inc = inc
                    local_dec = dec

            if local_best_d is not None and local_inc >= 0:
                transfer_delta(tmp, h, local_inc, local_dec, local_best_d,
                               idx_first, idx_second, n)
                s += tmp
                h[local_inc] += local_best_d
                h[local_dec] -= local_best_d
                if h[local_inc] > 1.0:
                    h[local_inc] = 1.0
                if h[local_dec] < 0.0:
                    h[local_dec] = 0.0

                s = np.correlate(h, 1.0 - h, mode="full")
                cur_max = float(s.max())
                if cur_max < best_max - 1e-12:
                    best_max = cur_max
                    best_h = h.copy()
                    best_s = s.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1

    return best_h, best_s, best_max


def random_search(h, n, idx_first, idx_second, tmp, rng, deadline,
                  temp_C=0.0005):
    h = np.array(h, dtype=float, copy=True)
    s = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(s.max())

    best_h = h.copy()
    best_s = s.copy()
    best_max = cur_max

    temp = float(temp_C) * (float(n) / 2.0)
    evals = 0
    accepts = 0

    while time.time() < deadline:
        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a == b:
            continue

        ha = float(h[a])
        hb = float(h[b])
        if ha <= hb:
            inc, dec = a, b
        else:
            inc, dec = b, a

        if rng.rand() < 0.18:
            inc, dec = dec, inc

        D = min(float(h[dec]), float(1.0 - h[inc]))
        if D > 1.0:
            D = 1.0
        if D <= 1e-10:
            continue

        u = rng.rand()
        if u < 0.35:
            d = D
        elif u < 0.65:
            d = 0.5 * D
        elif u < 0.85:
            d = 0.2 * D
        else:
            d = 0.05 * D

        if d <= 1e-15:
            continue

        transfer_delta(tmp, h, inc, dec, d, idx_first, idx_second, n)
        s += tmp
        mx = float(s.max())

        accept = False
        if mx < cur_max - 1e-11:
            accept = True
        elif temp > 0.0:
            diff = mx - cur_max
            if diff <= 0.0:
                accept = True
            else:
                u2 = rng.rand()
                if u2 > 1e-15 and diff <= temp * (-np.log(u2)):
                    accept = True

        if accept:
            h[inc] += d
            h[dec] -= d
            if h[inc] > 1.0:
                h[inc] = 1.0
            if h[dec] < 0.0:
                h[dec] = 0.0
            cur_max = mx
            accepts += 1

            if mx < best_max - 1e-11:
                s = np.correlate(h, 1.0 - h, mode="full")
                cur_max = float(s.max())
                best_h = h.copy()
                best_s = s.copy()
                best_max = cur_max
        else:
            s -= tmp

        evals += 1
        if accepts and accepts % 4000 == 0:
            s = np.correlate(h, 1.0 - h, mode="full")
            cur_max = float(s.max())
        elif evals % 20000 == 0:
            s = np.correlate(h, 1.0 - h, mode="full")
            cur_max = float(s.max())

    return best_h, best_s, best_max


def weighted_active_gradient(h, s, active, alpha, pair_i, pair_j):
    grad = np.zeros(len(h), dtype=float)
    if active is None or len(active) == 0:
        return grad

    vals = s[active]
    if vals.size == 0:
        return grad

    m = float(vals.max())
    sh = alpha * (vals - m)
    w = np.exp(sh)
    wsum = float(w.sum())
    if wsum <= 0.0:
        w = np.full(w.size, 1.0 / w.size)
    else:
        w = w / wsum

    for wi, mi in zip(w, active):
        if wi <= 1e-18:
            continue
        mio = int(mi)
        ii = pair_i[mio]
        jj = pair_j[mio]
        grad[ii] += wi * (1.0 - h[jj])
        grad[jj] -= wi * h[ii]
    return grad


def gradient_pair_search(h, n, pair_i, pair_j, idx_first, idx_second, tmp, rng, deadline,
                         active_k=35, alpha=5000.0, fracs=None):
    if fracs is None:
        fracs = (1.0, 0.5, 0.15, 0.05)

    h = np.array(h, dtype=float, copy=True)
    s = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(s.max())

    best_h = h.copy()
    best_s = s.copy()
    best_max = cur_max

    stall = 0
    max_stall = 10

    while time.time() < deadline and stall < max_stall:
        s = np.correlate(h, 1.0 - h, mode="full")
        cur_max = float(s.max())
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_h = h.copy()
            best_s = s.copy()

        active = top_k_indices(s, active_k)
        grad = weighted_active_gradient(h, s, active, alpha, pair_i, pair_j)
        desc = np.argsort(-grad)
        asc = np.argsort(grad)

        top_lim = min(20, n)
        desc = desc[:top_lim]
        asc = asc[:top_lim]

        found = False

        for sp in desc[:3]:
            if time.time() > deadline:
                break
            for tp in asc[:3]:
                if sp == tp:
                    continue
                dec = int(sp)
                inc = int(tp)
                D = min(float(h[dec]), float(1.0 - h[inc]))
                if D <= 1e-10:
                    continue
                best_d, _ = best_trial_transfer(
                    s, h, inc, dec, D, fracs, tmp,
                    idx_first, idx_second, n, cur_max
                )
                if best_d is not None:
                    transfer_delta(tmp, h, inc, dec, best_d, idx_first, idx_second, n)
                    s += tmp
                    h[inc] += best_d
                    h[dec] -= best_d
                    if h[inc] > 1.0:
                        h[inc] = 1.0
                    if h[dec] < 0.0:
                        h[dec] = 0.0
                    found = True
                    break
            if found:
                break

        if not found:
            tries = 24
            for _ in range(tries):
                if time.time() > deadline:
                    break
                dec = int(desc[rng.randint(0, len(desc))])
                inc = int(asc[rng.randint(0, len(asc))])
                if dec == inc:
                    continue
                D = min(float(h[dec]), float(1.0 - h[inc]))
                if D <= 1e-10:
                    continue
                best_d, _ = best_trial_transfer(
                    s, h, inc, dec, D, fracs, tmp,
                    idx_first, idx_second, n, cur_max
                )
                if best_d is not None:
                    transfer_delta(tmp, h, inc, dec, best_d, idx_first, idx_second, n)
                    s += tmp
                    h[inc] += best_d
                    h[dec] -= best_d
                    if h[inc] > 1.0:
                        h[inc] = 1.0
                    if h[dec] < 0.0:
                        h[dec] = 0.0
                    found = True
                    break

        if found:
            s = np.correlate(h, 1.0 - h, mode="full")
            cur_max = float(s.max())
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = h.copy()
                best_s = s.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_s, best_max


def smooth_max_val(vals, alpha):
    if vals.size == 0:
        return 0.0
    m = float(vals.max())
    sh = alpha * (vals - m)
    t = float(np.sum(np.exp(sh)))
    if t <= 0.0:
        t = 1.0
    return m + np.log(t) / alpha


def minmax_obj(x, n, active, alpha, mu, pair_i, pair_j):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")
    vals = c[active]
    val = smooth_max_val(vals, alpha)
    sm = float(x.sum()) - 0.5 * float(n)
    return val + mu * sm * sm


def minmax_jac(x, n, active, alpha, mu, pair_i, pair_j):
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode="full")
    vals = c[active]
    m = float(vals.max())
    sh = alpha * (vals - m)
    w = np.exp(sh)
    wsum = float(w.sum())
    if wsum <= 0.0:
        w = np.full(len(active), 1.0 / max(1, len(active)))
    else:
        w = w / wsum

    grad = np.zeros(n, dtype=float)
    for wi, mi in zip(w, active):
        if wi <= 1e-18:
            continue
        mio = int(mi)
        ii = pair_i[mio]
        jj = pair_j[mio]
        grad[ii] += wi * (1.0 - x[jj])
        grad[jj] -= wi * x[ii]

    if mu != 0.0:
        sm = float(x.sum()) - 0.5 * float(n)
        grad += 2.0 * mu * sm
    return grad


def gradient_refine_lbfgs(h, n, pair_i, pair_j, deadline,
                          active_k=40, alphas=(2000.0, 10000.0, 50000.0),
                          mu=20.0, max_rounds=2):
    h = project_feasible(np.array(h, dtype=float, copy=True), n)
    s = np.correlate(h, 1.0 - h, mode="full")
    best_max = float(s.max())
    best_h = h.copy()
    best_s = s.copy()

    bounds = [(0.0, 1.0)] * n

    for _ in range(max_rounds):
        if time.time() > deadline - 25.0:
            break
        s = np.correlate(h, 1.0 - h, mode="full")
        cur_max = float(s.max())
        active = top_k_indices(s, active_k)

        for alpha in alphas:
            if time.time() > deadline - 20.0:
                break
            rem = deadline - time.time() - 12.0
            if rem < 20.0:
                break
            maxit = max(40, int(rem * 2.0))
            if n <= 600:
                maxit = min(maxit, 1400)
            else:
                maxit = min(maxit, 700)

            try:
                res = minimize(
                    minmax_obj, h, jac=minmax_jac,
                    args=(n, active, alpha, mu, pair_i, pair_j),
                    method="L-BFGS-B", bounds=bounds,
                    options={"maxiter": maxit, "ftol": 1e-20,
                             "gtol": 1e-15, "maxfun": maxit * 10,
                             "maxls": 40}
                )
                h_new = project_feasible(res.x, n)
                s_new = np.correlate(h_new, 1.0 - h_new, mode="full")
                mx_new = float(s_new.max())

                if mx_new < cur_max - 1e-13:
                    h = h_new.copy()
                    s = s_new.copy()
                    cur_max = mx_new
                if mx_new < best_max - 1e-13:
                    best_max = mx_new
                    best_h = h_new.copy()
                    best_s = s_new.copy()
            except Exception:
                pass

    return best_h, best_s, best_max


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(int(seed))

    # Load or initialize parent construction.
    try:
        hp = np.asarray(initial_h_values, dtype=float).ravel()
        if hp.size < 20:
            raise ValueError("too small")
    except Exception:
        hp = np.full(500, 0.5)

    if hp.size <= 1000:
        n0 = int(hp.size)
        h0 = hp.astype(float, copy=True)
    else:
        n0 = 1000
        h0 = resize_h(hp, n0)

    if n0 < 20:
        n0 = 500
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)

    # Build several starts and track the two best without lambdas.
    starts = []
    starts.append(h0)
    for a in MECH_ALPHAS:
        starts.append(make_mechanical(n0, a))
    starts.append(make_random_binary(n0, rng))
    starts.append(make_random_binary(n0, rng))
    starts.append(1.0 - h0)

    best_start = None
    best_start_C = 1.0e100
    second_start = None
    second_start_C = 1.0e100

    for hs in starts:
        hs = project_feasible(hs, n0)
        _, _, cval = eval_solution(hs, n0)
        if cval < best_start_C:
            second_start = best_start
            second_start_C = best_start_C
            best_start = hs
            best_start_C = cval
        elif cval < second_start_C:
            second_start = hs
            second_start_C = cval

    global_best_h = np.array(best_start, dtype=float, copy=True)
    global_best_n = int(n0)
    global_best_C = float(best_start_C)

    current_h = np.array(best_start, dtype=float, copy=True)
    current_C = float(best_start_C)

    cache = {}
    if n0 not in cache:
        fi0, si0 = build_update_indices(n0)
        pi0, pj0 = build_pair_arrays(n0)
        cache[n0] = (fi0, si0, pi0, pj0, np.empty(2 * n0 - 1, dtype=float))
    fi0, si0, pi0, pj0, tmp0 = cache[n0]

    # Phase A: targeted active-set pair descent at native resolution.
    endA = min(t0 + 180.0, deadline - 550.0)
    if endA > time.time() + 20.0:
        try:
            bh, bs, bmax = targeted_search(
                current_h, n0, pi0, pj0, fi0, si0, tmp0, rng, endA,
                active_k=30, top_cand=10, rand_cand=5,
                fracs=(1.0, 0.5, 0.2, 0.05)
            )
            Cb = float(bmax) * (2.0 / float(n0))
            if Cb < global_best_C - 1e-15:
                global_best_C = Cb
                global_best_h = np.array(bh, dtype=float, copy=True)
                global_best_n = n0
            if Cb < current_C + 1e-15:
                current_h = np.array(bh, dtype=float, copy=True)
                current_C = Cb
        except Exception:
            pass

    # Optional: refine second-best start to avoid staying in one basin.
    if second_start is not None and second_start_C < global_best_C + 0.01:
        endA2 = min(time.time() + 60.0, deadline - 450.0)
        if endA2 > time.time() + 15.0:
            try:
                bh, bs, bmax = targeted_search(
                    second_start, n0, pi0, pj0, fi0, si0, tmp0, rng, endA2,
                    active_k=30, top_cand=10, rand_cand=5,
                    fracs=(1.0, 0.5, 0.2, 0.05)
                )
                Cb = float(bmax) * (2.0 / float(n0))
                if Cb < global_best_C - 1e-15:
                    global_best_C = Cb
                    global_best_h = np.array(bh, dtype=float, copy=True)
                    global_best_n = n0
                if Cb < current_C + 1e-15:
                    current_h = np.array(bh, dtype=float, copy=True)
                    current_C = Cb
            except Exception:
                pass

    # Phase B: combined active-gradient pair descent.
    endB = min(time.time() + 120.0, deadline - 400.0)
    if endB > time.time() + 20.0:
        try:
            bh, bs, bmax = gradient_pair_search(
                current_h, n0, pi0, pj0, fi0, si0, tmp0, rng, endB,
                active_k=35, alpha=5000.0,
                fracs=(1.0, 0.5, 0.15, 0.05)
            )
            Cb = float(bmax) * (2.0 / float(n0))
            if Cb < global_best_C - 1e-15:
                global_best_C = Cb
                global_best_h = np.array(bh, dtype=float, copy=True)
                global_best_n = n0
            if Cb < current_C + 1e-15:
                current_h = np.array(bh, dtype=float, copy=True)
                current_C = Cb
        except Exception:
            pass

    # Phase C: short randomized transfer search from the current best.
    endC = min(time.time() + 55.0, deadline - 300.0)
    if endC > time.time() + 15.0:
        try:
            bh, bs, bmax = random_search(
                current_h, n0, fi0, si0, tmp0, rng, endC, temp_C=0.0005
            )
            Cb = float(bmax) * (2.0 / float(n0))
            if Cb < global_best_C - 1e-15:
                global_best_C = Cb
                global_best_h = np.array(bh, dtype=float, copy=True)
                global_best_n = n0
            if Cb < current_C + 1e-15:
                current_h = np.array(bh, dtype=float, copy=True)
                current_C = Cb
        except Exception:
            pass

    # Phase D: L-BFGS smooth-max refinement.
    endD = min(time.time() + 130.0, deadline - 250.0)
    if endD > time.time() + 25.0:
        try:
            bh, bs, bmax = gradient_refine_lbfgs(
                current_h, n0, pi0, pj0, endD,
                active_k=40, alphas=(2000.0, 10000.0, 50000.0),
                mu=20.0, max_rounds=2
            )
            Cb = float(bmax) * (2.0 / float(n0))
            if Cb < global_best_C - 1e-15:
                global_best_C = Cb
                global_best_h = np.array(bh, dtype=float, copy=True)
                global_best_n = n0
            if Cb < current_C + 1e-15:
                current_h = np.array(bh, dtype=float, copy=True)
                current_C = Cb
        except Exception:
            pass

    # Phase E: finer targeted tuning at the same resolution.
    endE = min(time.time() + 70.0, deadline - 150.0)
    if endE > time.time() + 20.0:
        try:
            bh, bs, bmax = targeted_search(
                current_h, n0, pi0, pj0, fi0, si0, tmp0, rng, endE,
                active_k=40, top_cand=12, rand_cand=6,
                fracs=(0.2, 0.05, 0.02, 0.005)
            )
            Cb = float(bmax) * (2.0 / float(n0))
            if Cb < global_best_C - 1e-15:
                global_best_C = Cb
                global_best_h = np.array(bh, dtype=float, copy=True)
                global_best_n = n0
            if Cb < current_C + 1e-15:
                current_h = np.array(bh, dtype=float, copy=True)
                current_C = Cb
        except Exception:
            pass

    # Refine at a higher resolution if time permits.
    if time.time() < deadline - 280.0:
        n_base = int(global_best_n)
        n1 = n_base * 2
        if n1 > 1000:
            n1 = 1000 if n_base < 1000 else n_base
        if n1 > n_base and n1 <= 1000:
            h1 = project_feasible(resize_h(global_best_h, n1), n1)
            if n1 not in cache:
                fi1, si1 = build_update_indices(n1)
                pi1, pj1 = build_pair_arrays(n1)
                cache[n1] = (fi1, si1, pi1, pj1, np.empty(2 * n1 - 1, dtype=float))
            fi1, si1, pi1, pj1, tmp1 = cache[n1]

            endT1 = min(time.time() + 150.0, deadline - 150.0)
            if endT1 > time.time() + 25.0:
                try:
                    bh1, bs1, bmax1 = targeted_search(
                        h1, n1, pi1, pj1, fi1, si1, tmp1, rng, endT1,
                        active_k=35, top_cand=10, rand_cand=6,
                        fracs=(1.0, 0.5, 0.2, 0.05)
                    )
                    Cb1 = float(bmax1) * (2.0 / float(n1))
                    if Cb1 < global_best_C - 1e-15:
                        global_best_C = Cb1
                        global_best_h = np.array(bh1, dtype=float, copy=True)
                        global_best_n = n1
                        current_h = np.array(bh1, dtype=float, copy=True)
                        current_C = Cb1
                    else:
                        h1 = np.array(bh1, dtype=float, copy=True)
                except Exception:
                    pass

            endG1 = min(time.time() + 70.0, deadline - 100.0)
            if endG1 > time.time() + 15.0:
                try:
                    bh1, bs1, bmax1 = gradient_pair_search(
                        h1, n1, pi1, pj1, fi1, si1, tmp1, rng, endG1,
                        active_k=40, alpha=8000.0,
                        fracs=(1.0, 0.4, 0.12, 0.04)
                    )
                    Cb1 = float(bmax1) * (2.0 / float(n1))
                    if Cb1 < global_best_C - 1e-15:
                        global_best_C = Cb1
                        global_best_h = np.array(bh1, dtype=float, copy=True)
                        global_best_n = n1
                        current_h = np.array(bh1, dtype=float, copy=True)
                        current_C = Cb1
                    else:
                        h1 = np.array(bh1, dtype=float, copy=True)
                except Exception:
                    pass

            endR1 = min(time.time() + 30.0, deadline - 70.0)
            if endR1 > time.time() + 10.0:
                try:
                    bh1, bs1, bmax1 = random_search(
                        h1, n1, fi1, si1, tmp1, rng, endR1, temp_C=0.0004
                    )
                    Cb1 = float(bmax1) * (2.0 / float(n1))
                    if Cb1 < global_best_C - 1e-15:
                        global_best_C = Cb1
                        global_best_h = np.array(bh1, dtype=float, copy=True)
                        global_best_n = n1
                        current_h = np.array(bh1, dtype=float, copy=True)
                        current_C = Cb1
                except Exception:
                    pass

            endL1 = deadline - 45.0
            if endL1 - time.time() > 80.0:
                try:
                    bh1, bs1, bmax1 = gradient_refine_lbfgs(
                        h1, n1, pi1, pj1, endL1,
                        active_k=50, alphas=(10000.0, 40000.0),
                        mu=30.0, max_rounds=1
                    )
                    Cb1 = float(bmax1) * (2.0 / float(n1))
                    if Cb1 < global_best_C - 1e-15:
                        global_best_C = Cb1
                        global_best_h = np.array(bh1, dtype=float, copy=True)
                        global_best_n = n1
                        current_h = np.array(bh1, dtype=float, copy=True)
                        current_C = Cb1
                except Exception:
                    pass

    # Final ultra-fine polish at the global best resolution.
    nfin = int(global_best_n)
    if nfin not in cache:
        fif, sif = build_update_indices(nfin)
        pif, pjf = build_pair_arrays(nfin)
        cache[nfin] = (fif, sif, pif, pjf, np.empty(2 * nfin - 1, dtype=float))
    fif, sif, pif, pjf, tmpf = cache[nfin]

    if time.time() < deadline - 25.0:
        endF1 = min(deadline - 30.0, time.time() + 80.0)
        if endF1 > time.time() + 12.0:
            try:
                bh, bs, bmax = targeted_search(
                    global_best_h, nfin, pif, pjf, fif, sif, tmpf, rng, endF1,
                    active_k=50, top_cand=15, rand_cand=8,
                    fracs=(0.1, 0.03, 0.01, 0.003)
                )
                Cb = float(bmax) * (2.0 / float(nfin))
                if Cb < global_best_C - 1e-15:
                    global_best_C = Cb
                    global_best_h = np.array(bh, dtype=float, copy=True)
                    global_best_n = nfin
            except Exception:
                pass

    if time.time() < deadline - 20.0:
        endF2 = min(deadline - 5.0, time.time() + 35.0)
        if endF2 > time.time() + 10.0:
            try:
                bh, bs, bmax = gradient_pair_search(
                    global_best_h, nfin, pif, pjf, fif, sif, tmpf, rng, endF2,
                    active_k=55, alpha=20000.0,
                    fracs=(0.05, 0.02, 0.008, 0.002)
                )
                Cb = float(bmax) * (2.0 / float(nfin))
                if Cb < global_best_C - 1e-15:
                    global_best_C = Cb
                    global_best_h = np.array(bh, dtype=float, copy=True)
                    global_best_n = nfin
            except Exception:
                pass

    # Ensure feasibility and compute the returned bound.
    target_sum = 0.5 * float(global_best_n)
    h_final = np.array(global_best_h, dtype=float, copy=True)
    if abs(h_final.sum() - target_sum) > 1e-10:
        h_final = project_feasible(h_final, global_best_n)
    else:
        h_final = np.clip(h_final, 0.0, 1.0)

    _, smax_final, c_final = eval_solution(h_final, global_best_n)
    return h_final, float(c_final), int(global_best_n)
