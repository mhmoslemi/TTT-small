def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x) * n / 2.0  # undo dx scaling for consistency
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1
    for idx in range(n2m1):
        if w[idx] < 1e-13:
            continue
        k = idx - center
        wm = w[idx] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def binary_local_search(h_bin, n, deadline, max_iter=200000):
    h = h_bin.copy()
    c_vec = np.correlate(h, 1 - h, mode='full')
    best_int = c_vec.max()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng = np.random.RandomState(777)
    stall = 0
    max_stall = 5000

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue

        # Perform swap
        h[i] = 0
        h[j] = 1

        # Recompute correlation
        c_vec = np.correlate(h, 1 - h, mode='full')
        new_max = c_vec.max()

        if new_max < best_int:
            best_int = new_max
            # Update sets
            ones = np.setdiff1d(ones, [i])
            ones = np.append(ones, j)
            zeros = np.setdiff1d(zeros, [j])
            zeros = np.append(zeros, i)
            stall = 0
        else:
            # Undo
            h[i] = 1
            h[j] = 0
            stall += 1
            max_stall = min(stall + 5000, 20000)

    return h, best_int

def shift_equalize(h, n, deadline, n_rounds=200):
    """Water-filling: reduce top correlations by adjusting h."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    center = n - 1

    for r in range(n_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(best_h, 1 - best_h, mode='full') * dx
        m = c.max()
        # Focus on top 3 shifts
        top_idx = np.argsort(-c)[:min(5, len(c))]
        if top_idx[0] * dx < best_c * 0.95:
            break

        # Compute gradient that reduces top shifts
        grad = np.zeros(n)
        for idx in top_idx:
            if c[idx] < m * 0.9:
                continue
            k = idx - center
            if k >= 0:
                nv = n - k
                grad[0:nv] += (1.0 - best_h[k:n])
                grad[k:n] += (-best_h[0:nv])
            else:
                kk = -k
                nv = n - kk
                grad[kk:n] += (1.0 - best_h[0:nv])
                grad[0:nv] += (-best_h[kk:n])

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = 0.005 / gn
        h_new = best_h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs=None):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(100, 50.0), (500, 200.0), (2000, 1000.0),
                   (5000, 5000.0), (15000, 20000.0), (50000, 100000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 20:
            break
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 15000, 'ftol': 1e-22,
                                   'gtol': 1e-18, 'maxfun': 100000, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new.copy()
        except Exception:
            pass
    return h, best_c

def slqp_optimize(h, n, deadline, alpha=3000000, maxiter=5000):
    def obj_fn(x, na, al):
        c = np.correlate(x, 1 - x, mode='full') * (2.0 / na)
        m = c.max()
        sh = al * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / al)
        else:
            return float(m + np.log(np.sum(np.exp(sh))) / al)

    def eq_fun(x, na):
        return float(x.sum() - na / 2.0)

    def eq_jac(x, na):
        return np.ones(na)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    try:
        res = minimize(obj_fn, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res is not None and hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            return h_new, c_new
    except Exception:
        pass
    return h.copy(), compute_c5(h, n)

def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def perturb_restart(h, n, deadline, rng, n_trials=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in range(n_trials):
        if time.time() > deadline - 40:
            break
        mag = 0.03 * (1 + t * 0.5)
        h_p = h + mag * (rng.rand(n) - 0.5)
        h_p = project_feasible(h_p, n)
        try:
            res = minimize(smooth_obj, h_p, jac=smooth_grad,
                          args=(n, 5000, 5000.0), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 3000, 'ftol': 1e-18, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def final_refine(h, n, deadline):
    """Very fine step-size refinement on individual points."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_c5_vec(best_h)
    worst = np.where(c_vec >= best_c * 0.98)[0]

    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        c_vec = compute_c5_vec(best_h)
        for widx in worst[:3]:
            k = widx - (n - 1)
            if k >= 0:
                affected = list(range(0, n - k)) + list(range(k, n))
            else:
                kk = -k
                affected = list(range(0, n - kk)) + list(range(kk, n))

            for i in affected:
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002]:
                    nv = best_h[i] + delta
                    if nv < -1e-10 or nv > 1.0 + 1e-10:
                        continue
                    best_h[i] = nv
                    hc = compute_c5(best_h, n)
                    if hc < best_c - 1e-15:
                        best_c = hc
                        improved = True
                        break
                    else:
                        best_h[i] = nv - delta
                if improved:
                    break
            if improved:
                best_h = project_feasible(best_h, n)
        if not improved:
            # Try smaller steps
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for delta in [0.0001, -0.0001, 0.0002, -0.0002]:
                    nv = best_h[i] + delta
                    if nv < -1e-10 or nv > 1.0 + 1e-10:
                        continue
                    best_h[i] = nv
                    hc = compute_c5(best_h, n)
                    if hc < best_c - 1e-16:
                        best_c = hc
                        improved = True
                        break
                    else:
                        best_h[i] = nv - delta
                if improved:
                    best_h = project_feasible(best_h, n)
                    break
    return best_h, best_c

def block_init(n, pattern=None):
    """Create a block-structured initialization."""
    if pattern is None:
        # Alternating blocks of 1/4 and 3/4
        block_size = max(1, n // 16)
        h = np.zeros(n)
        for i in range(n):
            block = i // block_size
            if block % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
    else:
        h = pattern
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_base = 500

    best_h = project_feasible(h_init, n_base)
    best_c = compute_c5(best_h, n_base)

    # Phase 1: Binary local search at base resolution
    h_bin = make_binary(best_h, n_base)
    t1 = min(t0 + 40, deadline - 600)
    if t1 > time.time() + 5:
        h_bs, c_bs_int = binary_local_search(h_bin, n_base, t1, max_iter=100000)
        c_bs = compute_c5(h_bs, n_base)
        if c_bs < best_c:
            best_c = c_bs
            best_h = h_bs.copy()

    # Phase 2: L-BFGS from current best at base resolution
    t2 = min(t0 + 150, deadline - 500)
    if t2 > time.time() + 10:
        h2, c2 = lbfgs_optimize(best_h, n_base, t2,
            configs=[(200, 100.0), (1000, 500.0), (3000, 2000.0),
                     (10000, 10000.0), (30000, 50000.0)])
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Shift equalization
    t3 = min(t0 + 200, deadline - 450)
    if t3 > time.time() + 10:
        h3, c3 = shift_equalize(best_h, n_base, t3, n_rounds=300)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: SLSQP polish
    t4 = min(t0 + 280, deadline - 380)
    if t4 > time.time() + 10:
        h4, c4 = slqp_optimize(best_h, n_base, t4, alpha=5000000, maxiter=4000)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Upsample to n=1000
    n1 = 1000
    if time.time() < deadline - 300:
        h1000 = interpolate_to(best_h, n_base, n1)
        t5 = min(time.time() + 100, deadline - 200)
        if t5 > time.time() + 10:
            h1000, c1000 = lbfgs_optimize(h1000, n1, t5,
                configs=[(500, 200.0), (2000, 1000.0), (8000, 5000.0),
                         (20000, 20000.0), (50000, 80000.0)])
            if c1000 < best_c:
                best_c = c1000
                best_h = h1000

        # Shift equalize at n=1000
        t5b = min(time.time() + 40, deadline - 100)
        if t5b > time.time() + 5:
            h_eq, c_eq = shift_equalize(best_h, n1, t5b, n_rounds=200)
            if c_eq < best_c:
                best_c = c_eq
                best_h = h_eq

    # Phase 6: Binary local search at n=1000
    if time.time() < deadline - 200:
        h_bin2 = make_binary(best_h, n1)
        t6 = min(time.time() + 40, deadline - 150)
        if t6 > time.time() + 5:
            h_bs2, _ = binary_local_search(h_bin2, n1, t6, max_iter=50000)
            c_bs2 = compute_c5(h_bs2, n1)
            if c_bs2 < best_c:
                # Refine from binary
                t6b = min(time.time() + 60, deadline - 80)
                h_r, c_r = lbfgs_optimize(h_bs2, n1, t6b,
                    configs=[(1000, 500.0), (5000, 5000.0), (20000, 20000.0)])
                if c_r < best_c:
                    best_c = c_r
                    best_h = h_r

    # Phase 7: SLSQP at n=1000
    t7 = min(time.time() + 60, deadline - 80)
    if t7 > time.time() + 10:
        h7, c7 = slqp_optimize(best_h, n1, t7, alpha=8000000, maxiter=3000)
        if c7 < best_c:
            best_c = c7
            best_h = h7

    # Phase 8: Upsample to n=2000
    n2 = 2000
    if time.time() < deadline - 120:
        h2000 = interpolate_to(best_h, n1, n2)
        t8 = min(time.time() + 80, deadline - 50)
        if t8 > time.time() + 10:
            h2000, c2000 = lbfgs_optimize(h2000, n2, t8,
                configs=[(2000, 1000.0), (8000, 5000.0), (20000, 20000.0)])
            if c2000 < best_c:
                best_c = c2000
                best_h = h2000

        # Quick equalize
        t8b = min(time.time() + 30, deadline - 20)
        if t8b > time.time() + 5:
            h_eq2, c_eq2 = shift_equalize(best_h, n2, t8b, n_rounds=100)
            if c_eq2 < best_c:
                best_c = c_eq2
                best_h = h_eq2

    # Phase 9: Perturbation restart from best
    if time.time() < deadline - 40:
        t9 = min(time.time() + 50, deadline - 15)
        best_h, best_c = perturb_restart(best_h, len(best_h), t9, rng, n_trials=4)

    # Phase 10: Final targeted refinement
    if time.time() < deadline - 10:
        best_h, best_c = final_refine(best_h, len(best_h), deadline - 5)

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)

    return best_h, best_c, n_final
