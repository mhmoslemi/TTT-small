def alternating_lp(h, n, deadline, max_iters=50):
    """Alternating minimization: fix q=1-h, solve LP for h."""
    import cvxpy as cp
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    q = 1.0 - h
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        
        # For each shift mm, the coefficient of h[i] in c[mm]/dx
        # c[mm]/dx = sum_i h[i] * q[i + k] where k = mm - (n-1)
        for mm in range(2 * n - 1):
            k = mm - (n - 1)
            if k >= 0:
                # h[0:nv] * q[k:n] where nv = n - k
                nv = n - k
                coeff = q[k:n]  # coefficients for h[0:nv]
                # constraint: h[0:nv] @ coeff <= t/dx
                constraints.append(h_var[0:nv] @ coeff <= t_var / dx)
            else:
                kk = -k
                # h[kk:n] * q[0:nv] where nv = n - kk
                nv = n - kk
                coeff = q[0:nv]
                constraints.append(h_var[kk:n] @ coeff <= t_var / dx)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL if hasattr(cp, 'CLARABEL') else cp.SCS,
                      verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)

                break
        
        if h_var.value is None:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_nonzero = 0
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        n_nonzero += 1
        wm = w[mm] * dx
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def soft_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        if w[mm] < 1e-12:
            continue
        wm = w[mm] * dx
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def solve_lp_for_h(h_prev, n, deadline):
    """Solve the LP: min t s.t. sum_i h[i]*q[i+k] <= t/dx for all shifts."""
    dx = 2.0 / n
    q = 1.0 - h_prev
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for mm in range(2 * n - 1):
        k = mm - (n - 1)
        if k >= 0:
            nv = n - k
            coeff = q[k:n]
            constraints.append(h_var[0:nv] @ coeff <= t_var / dx)
        else:
            kk = -k
            nv = n - kk
            coeff = q[0:nv]
            constraints.append(h_var[kk:n] @ coeff <= t_var / dx)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solved = False
    for solver in [cp.ECOS, cp.SCS, cp.CVXOPT, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, verbose=False)
            solved = True
            break
        except Exception:
            continue
    if not solved or h_var.value is None:
        return None, None
    h_sol = np.clip(h_var.value, 0, 1)
    h_sol = project_feasible(h_sol, n)
    c_sol = compute_c5(h_sol, n)
    return h_sol, c_sol

def alternating_lp(h, n, deadline, max_iters=25):
    """Alternating minimization for the bilinear max-overlap problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_work = h.copy()
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_solve_start = time.time()
        h_sol, c_sol = solve_lp_for_h(h_work, n, deadline)
        if h_sol is None:
            break
        if c_sol < best_c - 1e-12:
            best_c = c_sol
            best_h = h_sol.copy()
            h_work = h_sol
            stagnant = 0
        else:
            # Blend to diversify
            h_work = np.clip(0.3 * h_sol + 0.7 * h_work, 0, 1)
            h_work = project_feasible(h_work, n)
            stagnant += 1
            if stagnant > 5:
                break
        if time.time() - t_solve_start > 30:
            # LP too slow, reduce remaining iters
            break
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, configs=None):
    """Run L-BFGS with multiple alpha/mu configs."""
    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    if configs is None:
        configs = [(200, 20.0), (2000, 200.0), (10000, 1000.0), (50000, 5000.0), (200000, 50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(soft_max_obj, best_h, jac=soft_max_grad,
                         args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                         options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-15, 'maxfun': 30000})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def local_refine(h, n, deadline, scale_init=0.04, max_rounds=500):
    """Random perturbation local search."""
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    scale = scale_init
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        found = False
        n_trials = 100
        for _ in range(n_trials):
            if time.time() > deadline:
                return h, best_c
            h_pert = h.copy()
            npert = rng.randint(2, 8)
            idx = rng.choice(n, npert, replace=False)
            deltas = rng.uniform(-scale, scale, npert)
            h_pert[idx] += deltas
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = project_feasible(h_pert, n)
            c = compute_c5(h_pert, n)
            if c < best_c - 1e-13:
                best_c = c
                h = h_pert
                found = True
        if not found:
            scale *= 0.65
            if scale < 0.0005:
                break
    return h, best_c

def pair_search_focused(h, n, deadline):
    """Pair exchange search on active (non-binary) coordinates."""
    best_c = compute_c5(h, n)
    active_mask = (h > 0.005) & (h < 0.995)
    active_idx = np.where(active_mask)[0]
    if len(active_idx) == 0:
        return h, best_c
    if len(active_idx) > 150:
        impact = np.abs(h[active_idx]) * (1 - h[active_idx])
        top_local = np.argsort(-impact)[:150]
        active_idx = active_idx[top_local]
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            for ii in range(len(active_idx)):
                if time.time() > deadline:
                    return h, best_c
                i = active_idx[ii]
                for jj in range(ii + 1, len(active_idx)):
                    j = active_idx[jj]
                    if h[i] - d >= -1e-10 and h[j] + d <= 1.0 + 1e-10:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
                    if h[i] + d <= 1.0 + 1e-10 and h[j] - d >= -1e-10:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
    return h, best_c
