    grad = np.zeros(n)

    threshold = w.max() * 1e-6
    active = np.where(w > threshold)[0]
    for idx in active:
        lag = idx - (n - 1)
        wi = w[idx]
        if lag >= 0:
            nv = n - lag
            if nv > 0:
                grad[0:nv] += wi * (1.0 - x[lag:n])
                grad[lag:n] += wi * (-x[0:nv])

            nv = n - ll
            if nv > 0:
                grad[ll:n] += wi * (1.0 - x[0:nv])
                grad[0:nv] += wi * (-x[ll:n])
    
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    w *= dx
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    center = n - 1

    active = np.where(w > threshold)[0]
    for idx in active:

        wm = w[idx]
        if k >= 0:

            if nv > 0:

                grad[k:n] += wm * (-x[0:nv])

            h_new = project_feasible(res.x, n)

def targeted_coordinate_descent(h, n, deadline, n_rounds=100):
    """Coordinate descent targeting the worst lags."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

        top_k = np.argsort(c_vec)[-5:]

                return best_h, best_c
            lag = int(t_idx) - (n - 1)
            
            # Find coordinates that most affect this lag

            h_up4, c_up4 = lbfgs_optimize(best_h, n4, t5b,
                alpha_list=[5000, 20000, 80000],
                mu_list=[50000.0, 200000.0, 500000.0],
                max_iter_list=[15000, 20000, 25000])
            if c_up4 < best_c:
                best_c = c_up4
                best_h = h_up4
        
        # SLSQP polish at high res
        t5c = min(time.time() + 40, t5)
        if time.time() < t5c:
            h5, c5 = slqp_optimize(best_h, n4, t5c, alpha=1000000, maxiter=1500)
            if c5 < best_c:
                best_c = c5
                best_h = h5
    
    # Phase 6: Multi-start at high resolution (120s)
    n_hi = len(best_h)
    t6 = min(t0 + 770, deadline - 150)
    if time.time() < t6 - 30 and n_hi >= 1000:
        h6, c6 = multistart_optimize(n_hi, t6, seed=seed + 500, n_starts=3)
        if h6 is not None and c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Targeted coordinate descent (100s)
    t7 = min(t0 + 870, deadline - 50)
    if time.time() < t7:
        best_h, best_c = targeted_coordinate_descent(best_h, len(best_h), t7, n_rounds=200)
    
    # Phase 8: Final SLSQP polish (80s)
    t8 = min(t0 + 950, deadline - 20)
    if time.time() < t8:
        n_fin = len(best_h)
        h8, c8 = slqp_optimize(best_h, n_fin, t8, alpha=2000000, maxiter=2000)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Final: one more targeted pass
    t9 = deadline - 5
    if time.time() < t9:
        best_h, best_c = targeted_coordinate_descent(best_h, len(best_h), t9, n_rounds=50)
    
    # Final feasibility and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    
    return best_h, best_c, n_final
