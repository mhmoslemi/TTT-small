def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    # Proportional rescaling
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Add/subtract from the element with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = np.argmax(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

def smooth_max_obj(x, n, alpha, mu):

    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    return val + mu * (x.sum() - n / 2.0) ** 2

    dx = 2.0 / n

                    best_h = h_new
            except Exception:
                pass

    # Phase 2: Binary swap search (main optimization)
    p2_end = min(time.time() + 550, deadline - 300)
    if p2_end > time.time() + 20:
        best_h, best_c = binary_swap_search(best_h, n, p2_end, rng)

    # Phase 3: Continuous refinement
    p3_end = min(time.time() + 250, deadline - 80)
    if p3_end > time.time() + 15:
        best_h, best_c = continuous_refine(best_h, n, p3_end)

    # Phase 4: More binary swaps after continuous refinement
    p4_end = min(time.time() + 120, deadline - 40)
    if p4_end > time.time() + 10:
        best_h, best_c = binary_swap_search(best_h, n, p4_end, rng)

    # Phase 5: Fine swap search
    p5_end = deadline - 3
    if p5_end > time.time() + 10:
        best_h, best_c = fine_swap_search(best_h, n, p5_end,
                                          [0.001, 0.0005, 0.0002, 0.0001])

    # Final projection and verification
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    return best_h, best_c, n
