import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_max_value(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -100, 500)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return float(val)

def smooth_max_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -100, 500)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    return w / w.sum()

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    w = smooth_max_weights(c, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[0:n-lag] += wk * (1.0 - h[lag:n]) * dx
            grad[lag:n] += wk * (-h[0:n-lag]) * dx
        else:
            idx = -lag
            grad[idx:n] += wk * (1.0 - h[0:n-idx]) * dx
            grad[0:n-idx] += wk * (-h[idx:n]) * dx
    return grad

def lbfgsb_optimize(h, n, alpha, deadline, maxiter=2000, penalty_coeff=1e8):
    target_sum = n / 2.0

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        c = compute_all_lags(x, n)
        obj_val = smooth_max_value(c, alpha)
        diff = x.sum() - target_sum
        obj_val += penalty_coeff * diff * diff
        grad = smooth_max_grad(x, n, alpha)
        grad += 2.0 * penalty_coeff * diff * np.ones(n)
        return obj_val, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B',
                       bounds=bounds, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 80,
                                'iprint': -1, 'gtol': 1e-12})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_feasible(h.copy(), n)

def gradient_descent_phase(h, n, deadline, alpha, lr_schedule=(0.02, 0.005), max_iter=5000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_schedule[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_all = compute_all_lags(h, n)
        cur_c5 = c_all.max()
        
        if cur_c5 < best_c - 1e-15:
            best_c = float(cur_c5)
            best_h = h.copy()
        
        grad = smooth_max_grad(h, n, alpha)
        
        # Projected gradient step
        step = h - lr * grad
        step = project_feasible(step, n)
        
        # Backtracking
        c_new = compute_c5(step, n)
        if c_new > best_c + 0.001:
            # Try smaller step
            for trial in range(5):
                lr = lr * 0.5
                step = h - lr * grad
                step = project_feasible(step, n)
                c_new = compute_c5(step, n)
                if c_new < cur_c5:
                    break
            else:
                lr = lr_schedule[1]
                break
        
        h = step
        # Slowly decay lr
        lr = max(lr * 0.999, lr_schedule[1])
    
    return best_h, best_c

def targeted_local_search(h, n, deadline, rng, max_rounds=100000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rounds = 0
    no_improve = 0
    K = 5  # number of top lags to target
    
    while time.time() < deadline and rounds < max_rounds and no_improve < 8000:
        c_all = compute_all_lags(h, n)
        top_idx = np.argsort(c_all)[-K:]
        improved = False
        
        for _ in range(20):
            if time.time() > deadline:
                break
            rounds += 1
            
            k_star = top_idx[rng.randint(K)]
            lag = k_star - (n - 1)
            if lag == 0 or abs(lag) >= n:
                continue
            m = abs(lag)
            
            # Find worst pair for this lag
            if lag > 0:
                # O = sum h[i]*(1-h[i+lag]) for i in [0, n-1-lag]
                contributions = h[0:n-m] * (1.0 - h[m:n])
                # Find pair with high h[i] and low h[i+m]
                idx_order = np.argsort(-contributions)
            else:
                idx2 = -lag
                contributions = h[idx2:n] * (1.0 - h[0:n-idx2])
                idx_order = np.argsort(-contributions)
            
            # Try a few of the worst pairs
            tried = set()
            for repick in range(5):
                pos = idx_order[rng.randint(min(30, len(idx_order)))]
                if lag > 0:
                    i = pos
                    j = pos + m
                else:
                    j = pos
                    i = pos + (-lag)
                
                if (i, j) in tried:
                    continue
                tried.add((i, j))
                
                if abs(h[i] - h[j]) < 1e-8:
                    continue
                
                # Move from higher to lower
                if h[i] > h[j]:
                    d = rng.uniform(0.002, 0.03)
                    d = min(d, (h[i] - h[j]) * 0.3)
                    h_try = h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                else:
                    d = rng.uniform(0.002, 0.03)
                    d = min(d, (h[j] - h[i]) * 0.3)
                    h_try = h.copy()
                    h_try[j] -= d
                    h_try[i] += d
                
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try, n)
                
                if c_try < best_c - 1e-15:
                    h = h_try
                    best_c = float(c_try)
                    best_h = h.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 2000:
                # Perturb and restart
                h = best_h + rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                no_improve = 0
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, rng, alpha_sa=100000, t_init=0.01, t_min=0.00002, cool=0.9999):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = t_init
    
    while time.time() < deadline and T > t_min:
        r = rng.random()
        if r < 0.7:
            # Random pair move
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                T *= cool
                continue
            base_dir = 1.0 if h[i] < h[j] else -1.0
            d = rng.uniform(0.002, 0.05) * base_dir
            h_try = h.copy()
            h_try[i] += d
            h_try[j] -= d
        else:
            # Structured move at worst lag
            c_all = compute_all_lags(h, n)
            top_k = np.argsort(c_all)[-3:]
            k_star = top_k[rng.randint(3)]
            lag = k_star - (n - 1)
            if lag == 0 or abs(lag) >= n:
                T *= cool
                continue
            m = abs(lag)
            i = rng.randint(0, n - m)
            j = i + m
            d = rng.uniform(0.003, 0.04)
            if h[i] < h[j]:
                h_try = h.copy()
                h_try[i] += d
                h_try[j] -= d
            else:
                h_try = h.copy()
                h_try[j] += d
                h_try[i] -= d
        
        h_try = project_feasible(h_try, n)
        c_new = compute_all_lags(h_try, n)
        e_new = smooth_max_value(c_new, alpha_sa)
        max_new = c_new.max()
        
        c_old_arr = compute_all_lags(h, n)
        e_old = smooth_max_value(c_old_arr, alpha_sa)
        
        de = e_new - e_old
        accept = de < 0
        if not accept and T > t_min:
            ratio = -de / T
            if ratio < 50:
                accept = rng.random() < np.exp(-ratio)
        
        if accept:
            h = h_try
            if max_new < best_c - 1e-15:
                best_c = float(max_new)
                best_h = h.copy()
        
        T *= cool
    
    return best_h, best_c

def expand_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def multi_start_polish(h, n, deadline, rng, n_restarts=5):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    time_per = (deadline - time.time()) / n_restarts
    
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        sub_deadline = min(deadline, time.time() + max(time_per, 20))
        
        # Different perturbation strategies
        strategy = r % 5
        if strategy == 0:
            h_pert = h + rng.normal(0, 0.01, n)
        elif strategy == 1:
            h_pert = h + rng.normal(0, 0.005, n)
        elif strategy == 2:
            # Binary-like perturbation
            mask = rng.random(n) > 0.95
            h_pert = h.copy()
            h_pert[mask] += rng.uniform(0.02, 0.08, mask.sum())
            h_pert[~mask] -= rng.uniform(0.02, 0.08, ((~mask)).sum())
        elif strategy == 3:
            # Shift perturbation
            shift = rng.randint(1, 5)
            h_pert = np.roll(h, shift)
            h_pert = 0.5 * h_pert + 0.5 * h
        else:
            # Scale perturbation
            h_pert = h * rng.uniform(0.95, 1.05)
        
        h_pert = project_feasible(h_pert, n)
        
        # Optimize the perturbed version
        alpha_use = 500000
        h_opt = lbfgsb_optimize(h_pert, n, alpha_use, sub_deadline, maxiter=1000)
        c_opt = compute_c5(h_opt, n)
        
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    return best_h, best_c

def gradient_polish(h, n, deadline, alpha=10000000, max_iter=2000):
    """Piecewise gradient descent with adaptive step size."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_all = compute_all_lags(h, n)
        cur_c5 = c_all.max()
        
        if cur_c5 < best_c - 1e-15:
            best_c = float(cur_c5)
            best_h = h.copy()
        
        grad = smooth_max_grad(h, n, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Normalized gradient direction
        direction = -grad / (grad_norm + 1e-15)
        
        # Try several step sizes
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = h + trial_lr * direction * grad_norm
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < cur_c5 - 1e-15:
                h = h_try
                lr = min(lr * 1.05, 0.05)
                improved = True
                break
        
        if not improved:
            lr *= 0.7
            if lr < 1e-5:
                break
    
    return best_h, best_c
