def correlate_fft(a, v):

from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def topK_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    if smx > 400:
        val = mx + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = mx + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def topK_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    mx = vals.max()
    sh = alpha * (vals - mx)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def active_set_optimize(h, n, deadline, configs=None):

            (5, 100000),

    best_h = project_feasible(h.copy(), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n

        if time.time() > deadline - 20:
            break
        for rnd in range(8):
            if time.time() > deadline - 10:

            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()

def coord_descent(h, n, deadline, step_sizes=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if step_sizes is None:
        step_sizes = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    for step in step_sizes:
        while time.time() < deadline:
            c_vec = compute_c5_vec(best_h, n)
            max_val = c_vec.max()
            tol = max(1e-9, 0.002 * max_val)
            active = np.where(c_vec >= max_val - tol)[0]
            dx_val = 2.0 / n
            
            # Compute gradient for active set
            grad = np.zeros(n)
            for idx in active:
                k = idx - (n - 1)
                if k >= 0:
                    nv = n - k
                    grad[0:nv] += (1.0 - best_h[k:n]) * dx_val
                    grad[k:n] += (-best_h[0:nv]) * dx_val
                else:
                    kk = -k
                    nv = n - kk
                    grad[kk:n] += (1.0 - best_h[0:nv]) * dx_val
                    grad[0:nv] += (-best_h[kk:n]) * dx_val
            grad /= max(1, len(active))
            
            # Find best pair: decrease h where grad>0, increase where grad<0
            pos_mask = (grad > 1e-8) & (best_h > 1e-8)
            neg_mask = (grad < -1e-8) & (best_h < 1.0 - 1e-8)
            
            if not pos_mask.any() or not neg_mask.any():
                break
            
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            
            # Greedy: find best (i, j) pair
            best_gain = 0
            best_i = -1
            best_j = -1
            for i in pos_idx[:15]:
                for j in neg_idx[:15]:
                    g = grad[i] - grad[j]
                    if g > best_gain:
                        best_gain = g
                        best_i = i
                        best_j = j
            
            if best_i < 0:
                break
            
            amt = min(step, best_h[best_i], 1.0 - best_h[best_j])
            if amt < 1e-10:
                break
            
            best_h[best_i] -= amt
            best_h[best_j] += amt
            c_new = compute_c5(best_h, n)
            
            if c_new < best_c - 1e-16:
                best_c = c_new
            else:
                best_h[best_i] += amt
                best_h[best_j] -= amt
        
        if step < 0.00002:
            break
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def make_beatty_start(n, alpha_param):
    h = np.zeros(n)
    n_ones = n // 2
    positions = set()
    for k in range(1, 4 * n):
