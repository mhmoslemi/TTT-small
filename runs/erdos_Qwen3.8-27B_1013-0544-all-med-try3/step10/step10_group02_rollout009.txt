import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_active(c_vec, top_k, rel_band=0.003):
    mx = c_vec.max()
    threshold = mx - max(1e-9, rel_band * mx)
    active = np.where(c_vec >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-c_vec[active])
        active = active[order[:top_k]]
    if len(active) == 0:
        active = np.array([np.argmax(c_vec)])
    return active

def obj_active(x, n, alpha, active, mu):
    c = compute_c5_vec(x, n)
    ca = c[active]
    m = ca.max()
    sh = alpha * (ca - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        d = x.sum() - n / 2.0
        val += mu * d * d
    return float(val)

def grad_active(x, n, alpha, active, mu):
    c = compute_c5_vec(x, n)
    ca = c[active]
    m = ca.max()
    sh = alpha * (ca - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, idx in zip(w, active):
        if wi < 1e-15:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[:nv] += wm * (1.0 - x[k:k+nv])
            grad[k:k+nv] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:kk+nv] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:kk+nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def obj_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    if mu > 0:
        d = x.sum() - n / 2.0
        val += mu * d * d
    return float(val)

def grad_full(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, mm in zip(w, range(len(w))):
        if wi < 1e-15:
            continue
        k = mm - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[:nv] += wm * (1.0 - x[k:k+nv])
            grad[k:k+nv] += wm * (-x[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:kk+nv] += wm * (1.0 - x[:nv])
            grad[:nv] += wm * (-x[kk:kk+nv])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def lbfgs_run(h, n, fun, grad, args, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=grad, args=args,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-14, 'maxfun': maxiter * 6,
                               'maxls': 50, 'maxcor': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_stage_optimize(h, n, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)

    stages = [
        (10, 60, 0.01, 1000),
        (50, 40, 0.01, 1500),
        (200, 30, 0.008, 2000),
        (1000, 20, 0.005, 2500),
        (5000, 15, 0.003, 3000),
        (20000, 12, 0.002, 3500),
        (100000, 10, 0.001, 4000),
        (500000, 8, 0.0008, 4500),
        (2000000, 6, 0.0005, 5000),
        (10000000, 5, 0.0003, 5000),
        (50000000, 4, 0.0002, 5000),
    ]

    for alpha, top_k, mu, maxit in stages:
        if time.time() > deadline - 20:
            break
        remaining = deadline - time.time()
        maxit_adj = min(maxit, max(100, int(remaining * 1.5)))

        c_vec = compute_c5_vec(best_h, n)
        active = get_active(c_vec, top_k, rel_band=0.002)

        h_new = lbfgs_run(best_h, n, obj_active, grad_active,
                         (n, alpha, active, mu), maxit_adj, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new
        elif c_new == best_c:
            best_h = h_new

    # Final full-correlation pass with high alpha
    if time.time() < deadline - 15:
        alpha_final = 50000000
        c_vec = compute_c5_vec(best_h, n)
        active = get_active(c_vec, 3, rel_band=0.001)
        h_new = lbfgs_run(best_h, n, obj_active, grad_active,
                         (n, alpha_final, active, 0.0001), 3000, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h_new

    return best_h, best_c

def subgrad_polish(h, n, deadline, n_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.002
    tol = 0.0005

    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = compute_c5_vec(best_h, n)
        c_max = c_vec.max()
        threshold = c_max - max(1e-10, tol * c_max)
        active = np.where(c_vec >= threshold)[0]
        if len(active) > 20:
            active = active[np.argsort(-c_vec[active])[:20]]

        dx = 2.0 / n
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                nv = n - k
                grad[:nv] += (1.0 - best_h[k:k+nv]) * dx
                grad[k:k+nv] += (-best_h[:nv]) * dx
            else:
                kk = -k
                nv = n - kk
                grad[kk:kk+nv] += (1.0 - best_h[:nv]) * dx
                grad[:nv] += (-best_h[kk:kk+nv]) * dx
        grad /= len(active)

        h_new = best_h - eta * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
            eta = min(eta * 1.01, 0.01)
            tol *= 0.995
        else:
            eta *= 0.97

    return best_h, best_c

def fine_swap(h, n, deadline, radius=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for d in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if best_h[i] + d <= 1.0 + 1e-12 and best_h[j] - d >= -1e-12:
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
                    if best_h[i] - d >= -1e-12 and best_h[j] + d <= 1.0 + 1e-12:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d

    return best_h, best_c

def make_periodic_init(n, period, pattern):
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern[i % period]
    return project_feasible(h, n)

def make_binary_random(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.permutation(n)
    h[idx[:ones]] = 1.0
    if n % 2 == 1:
        h[idx[ones]] = 1.0
        h = project_feasible(h, n)
    return project_feasible(h, n)
