import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    idx_center = n - 1
    for mm in range(len(c)):
        if w[mm] < 1e-14:
            continue
        k = mm - idx_center
        wm = w[mm] * dx
        if k >= 0:
            nv = n - k
            if nv > 0:
                grad[0:nv] += wm * (1.0 - x[k:n])
                grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            if nv > 0:
                grad[kk:n] += wm * (1.0 - x[0:nv])
                grad[0:nv] += wm * (-x[kk:n])
    return grad

def full_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    return smooth_max_val(c, alpha) + mu * (x.sum() - n / 2.0) ** 2

def full_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    grad = smooth_max_grad(c, alpha, x, n)
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def act_obj(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    return smooth_max_val(vals, alpha) + mu * (x.sum() - n / 2.0) ** 2

def act_grad(x, n, active, alpha, mu):
    c = compute_c5_vec(x, n)
    vals = c[active]
    m = vals.max()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(30000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    smx = sh.max()
    if smx > 400:
        val = m + (smx + np.log(np.sum(np.exp(sh - smx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def smooth_max_grad(c, alpha, x, n):

    sh = alpha * (c - m)
    smx = sh.max()
    w = np.exp(sh - smx)
    w /= w.sum()

def subgradient_loop(h, n, deadline, n_iters=2000, base_step=0.02):

    best_c = compute_c5(best_h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    tol = max(1e-6, 0.001 * max_val)
    active = np.where(c_vec >= max_val - tol)[0]
    
    momentum = np.zeros(n)
    beta = 0.7
    iterate_sum = np.zeros(n)
    
    for t in range(n_iters):
        if time.time() > deadline - 5:
            break
        step_size = base_step / (1.0 + t * 0.01)
        new_h = subgradient_step(best_h, n, active, step_size)
        momentum = beta * momentum + (1.0 - beta) * (new_h - best_h)
        cand = project_feasible(best_h + momentum, n)
        c_cand = compute_c5(cand, n)
        iterate_sum += cand
        if c_cand < best_c:

            best_h = cand
            c_vec = compute_c5_vec(best_h, n)

        try:
            res = minimize(full_obj, best_h, jac=full_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-24,
                                   'gtol': 1e-18, 'maxfun': maxit * 6,
                                   'maxls': 30, 'maxcor': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def coordinate_descent(h, n, deadline, top_k=8, n_sweeps=500):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    c_vec = compute_c5_vec(best_h, n)
    max_val = c_vec.max()
    tol = max(1e-7, 0.0005 * max_val)
    active = np.where(c_vec >= max_val - tol)[0]
    if len(active) > top_k:
        active = np.array(np.argsort(-c_vec[active])[:top_k])
    
    idx_center = n - 1
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):
        if time.time() > deadline - 5:
            break
        improved_any = False
        for idx in active:
            k = int(idx) - idx_center
            if k >= 0:
                for i in range(n - k):
                    if time.time() > deadline - 5:
                        break
                    c_val = c_vec[idx]
                    d_c_di = (1.0 - best_h[i + k]) * dx
                    d_c_dj = (-best_h[i]) * dx
                    if best_h[i] > 0.001 and best_h[i + k] < 0.999:
                        delta = min(0.01, best_h[i] * 0.1, (1.0 - best_h[i + k]) * 0.1)
                        best_h[i] -= delta
                        best_h[i + k] += delta
                        c_new = compute_c5_vec(best_h, n)
                        if c_new[idx] < c_val - 1e-14 and c_new.max() < best_c + 1e-14:
                            best_c = max(best_c, float(c_new.max()))
                            improved_any = True
                            c_vec = c_new
                        else:
                            best_h[i] += delta
                            best_h[i + k] -= delta
            else:
                kk = -k
                for i in range(n - kk):
                    if time.time() > deadline - 5:
                        break
                    a = i + kk
                    b = i
                    c_val = c_vec[idx]
                    if best_h[b] > 0.001 and best_h[a] < 0.999:
                        delta = min(0.01, best_h[b] * 0.1, (1.0 - best_h[a]) * 0.1)
                        best_h[b] -= delta
                        best_h[a] += delta
                        c_new = compute_c5_vec(best_h, n)
                        if c_new.max() < best_c + 1e-14:
                            best_c = max(best_c, float(c_new.max()))
                            improved_any = True
                            c_vec = c_new
                        else:
                            best_h[b] += delta
                            best_h[a] -= delta
        if not improved_any and sweep > 50:
            break
    return best_h, best_c

def perturb_reoptimize(h, n, deadline, rng, n_trials=8):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for t in range(n_trials):
        if time.time() > deadline - 30:
            break
        scale = [0.005, 0.01, 0.02, 0.01, 0.005, 0.008, 0.015, 0.003][t % 8]
        h_pert = best_h + rng.uniform(-scale, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c + 0.003:
            rem = (deadline - time.time() - 10) / (n_trials - t)
            maxit = min(500, max(50, int(rem * 2)))
            try:
                res = minimize(full_obj, h_pert, jac=full_grad,
                              args=(n, 500000, 0.003), method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-22,
                                       'gtol': 1e-16, 'maxfun': maxit * 5,
                                       'maxls': 30, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_orig = len(h0)
    except Exception:
        h0 = np.full(500, 0.5)
        n_orig = 500
    
    h0 = project_feasible(h0, n_orig)
    best_h = h0.copy()
    best_c = compute_c5(h0, n_orig)
    best_n = n_orig
    
    # Generate diverse starts and pick best
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alt_starts = [
        make_beatty(n_orig, golden / 2.0),
        make_beatty(n_orig, 1.0 / golden),
        make_beatty(n_orig, np.sqrt(2.0) - 1.0),
        make_beatty(n_orig, 0.4312),
        make_beatty(n_orig, 0.381966),
        make_beatty(n_orig, (np.sqrt(5.0) - 1.0) / 4.0 + 0.5),
        make_sturmian(n_orig, 1.0 / golden, 0.3),
        make_sturmian(n_orig, np.sqrt(2.0) / 2.0, 0.7),
        make_beatty(n_orig, rng.uniform(0.3, 0.55)),
        make_beatty(n_orig, rng.uniform(0.3, 0.55)),
    ]
    for h_alt in alt_starts:
        c_alt = compute_c5(h_alt, n_orig)
        if c_alt < best_c:
            best_c = c_alt
            best_h = h_alt.copy()
    
    # Phase 1: Quick multi-alpha L-BFGS warmup (80s)
    p1_end = min(t0 + 80, deadline - 800)
    if p1_end > t0 + 10:
        best_h, best_c = multi_alpha_lbfgs(best_h, best_n, p1_end,
            alphas=[500, 2000, 10000, 50000, 200000], mu=0.01, maxit_each=500)
    
    # Phase 2: Subgradient with momentum (100s)
    p2_end = min(time.time() + 100, deadline - 500)
    if p2_end > time.time() + 15:
        h_sg, c_sg = subgradient_loop(best_h, best_n, p2_end, n_iters=1500, base_step=0.03)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
            best_n = len(best_h)
        # Also try from the averaged solution
        h_sg2, c_sg2 = subgradient_loop(best_h, best_n, p2_end, n_iters=800, base_step=0.05)
        if c_sg2 < best_c:
            best_c = c_sg2
            best_h = h_sg2
    
    # Phase 3: Aggressive active-set L-BFGS (250s)
    p3_end = min(time.time() + 250, deadline - 400)
    if p3_end > time.time() + 20:
        configs = [
            (4, 50000, 0.002),
            (5, 100000, 0.001),
            (6, 200000, 0.001),
            (8, 500000, 0.0008),
            (10, 1000000, 0.0005),
            (12, 2000000, 0.0005),
            (15, 5000000, 0.0003),
            (20, 10000000, 0.0003),
            (25, 20000000, 0.0002),
        ]
        for top_k, alpha, mu in configs:
            if time.time() > p3_end - 20:
                break
            for _ in range(2):
                if time.time() > p3_end - 15:
                    break
                h3, c3 = run_active_round(best_h, best_n, p3_end - 10, top_k, alpha, mu, max_restarts=2)
                if c3 < best_c:
                    best_c = c3
                    best_h = h3
    
    # Phase 4: Coordinate descent on active lags (150s)
    p4_end = min(time.time() + 150, deadline - 250)
    if p4_end > time.time() + 15:
        h4, c4 = coordinate_descent(best_h, best_n, p4_end, top_k=8, n_sweeps=300)
        if c4 < best_c:
            best_c = c4
            best_h = h4
    
    # Phase 5: Higher resolution attempt (150s)
    n_up = best_n * 2
    p5_end = min(time.time() + 150, deadline - 200)
    if p5_end > time.time() + 40 and n_up <= 2000:
        h_up = upsample_h(best_h, best_n, n_up)
        c_up = compute_c5(h_up, n_up)
        if c_up <= best_c + 0.002:
            h_up, c_up = multi_alpha_lbfgs(h_up, n_up, min(p5_end - 100, time.time() + 80),
                alphas=[10000, 100000, 1000000], mu=0.003, maxit_each=400)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up
                best_n = n_up
            if time.time() < p5_end - 30 and best_n == n_up:
                h_up2, c_up2 = run_active_round(best_h, best_n, p5_end - 10, 10, 1000000, 0.0005, max_restarts=2)
                if c_up2 < best_c:
                    best_c = c_up2
                    best_h = h_up2
    
    # Phase 6: Targeted pair search on worst lags (180s)
    p6_end = min(time.time() + 180, deadline - 150)
    if p6_end > time.time() + 15:
        c_vec = compute_c5_vec(best_h, best_n)
        top_lags = np.argsort(-c_vec)[:8]
        h6, c6 = targeted_pair_search(best_h, best_n, p6_end, top_lags)
        if c6 < best_c:
            best_c = c6
            best_h = h6
    
    # Phase 7: Fine pair search (120s)
    p7_end = min(time.time() + 120, deadline - 80)
    if p7_end > time.time() + 15:
        h7, c7 = fine_pair_search(best_h, best_n, p7_end, radius=40)
        if c7 < best_c:
            best_c = c7
            best_h = h7
    
    # Phase 8: Perturbation + reoptimization (80s)
    p8_end = min(time.time() + 80, deadline - 40)
    if p8_end > time.time() + 15:
        h8, c8 = perturb_reoptimize(best_h, best_n, p8_end, rng, n_trials=6)
        if c8 < best_c:
            best_c = c8
            best_h = h8
    
    # Phase 9: Final ultra-fine search
    p9_end = deadline - 3
    if p9_end > time.time() + 10:
        h9, c9 = fine_pair_search(best_h, best_n, p9_end, radius=30)
        if c9 < best_c:
            best_c = c9
            best_h = h9
        # One more coordinate descent pass
        if time.time() < p9_end - 5:
            h9b, c9b = coordinate_descent(best_h, best_n, p9_end, top_k=5, n_sweeps=100)
            if c9b < best_c:
                best_c = c9b
                best_h = h9b
    
    # Final projection and verification
    final_n = len(best_h)
    best_h = project_feasible(best_h, final_n)
    best_c = compute_c5(best_h, final_n)
    
    return best_h, best_c, final_n
