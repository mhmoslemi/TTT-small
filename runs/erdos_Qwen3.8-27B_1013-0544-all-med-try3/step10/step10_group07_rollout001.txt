```python
def subgradient_step(h, n, alpha, step_size, top_m=10):
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    c = corr * dx
    c_max = c.max()
    # Find top m lags
    top_idx = np.argsort(c)[-top_m:]
    # Average gradient over top lags
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n-1)  # actual shift
        for j in range(n):
            # contribution from (1-h[j+k]) term
            jk = j + k
            if 0 <= jk < n:
                grad[j] += (1 - h[jk])
            # contribution from -h[j-k] term
            jmk = j - k
            if 0 <= jmk < n:
                grad[j] -= h[jmk]
    grad *= dx / len(top_idx)
    # Projected gradient descent
    h_new = h - step_size * grad
    # Project to feasible set
    h_new = project_feasible(h_new, n)
    return h_new, c_max
```

```python
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c.max()
        # Find active lags (within 1% of max)
        threshold = c_max - 0.01 * c_max
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient (average of active lags)
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            # ∂c_k/∂h[j] = (1-h[j+k]) - h[j-k] (with bounds)
            # Vectorized:
            if k >= 0:
                # (1-h[j+k]) for j+k < n
                js = np.arange(min(n, n - k))
                grad[js] += (1 - h[js + k])
                # -h[j-k] for j >= k, always valid since k >= 0
                js2 = np.arange(k, n)
                grad[js2] -= h[js2 - k]
            else:
                # k < 0, let k_neg = -k
                k_neg = -k
                # (1-h[j+k]) = (1-h[j-k_neg]) for j >= k_neg
                js = np.arange(k_neg, n)
                grad[js] += (1 - h[js - k_neg])
                # -h[j-k] = -h[j+k_neg] for j+k_neg < n
                js2 = np.arange(min(n, n - k_neg))
                grad[js2] -= h[js2 + k_neg]
        
        grad *= dx / len(active)
        
        # Projected gradient descent
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.7
            h = h_new.copy()  # still update to explore
    
    return best_h, best_c

def binary_search_multi_swap(h_init, n, deadline, max_sweeps=2000, seed=42):
    """Binary local search with steepest descent (best swap per sweep)."""
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try all zeros as candidates (we'll swap each zero with each one)
        # For efficiency, process in random order
        rng.shuffle(zeros)
        improved = False
        
        for i in zeros:
            if time.time() > deadline:
                break
            # For this zero position i, find the best one to swap with
            best_nc = cur_c
            best_j = -1
            for j in ones:
                if j == i:
                    continue
                im = i + m_range
                imn = i - m_range
                jm = j + m_range
                jmn = j - m_range
                
                delta = np.zeros(2 * n - 1)
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                
                new_corr = corr + delta
                nc = new_corr.max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_delta = delta.copy()
            
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + best_delta
                cur_c = best_nc
                improved = True
        
        if not improved:
            break
    
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    
    return best_h, best_c
```

Let me try a different approach: **iterative reweighted averaging**.

1. Start with h from initial_h_values
2. Compute all lag values c_k
3. Identify active lags A = {k : c_k > c_max - ε}
4. For the active lags, compute the "average" correlation: c_avg = (1/|A|) sum_{k∈A} c_k
5. Minimize c_avg (which is a smooth, though non-convex, quadratic) using L-BFGS-B
6. Update A and repeat

This is like alternating between "which lags matter" and "optimize for those lags".

For step 5, the objective is:
c_avg = (1/|A|) sum_{k∈A} sum_i h[i]*(1-h[i+k]) * dx

The gradient is:
∂c_avg/∂h[j] = (1/|A|) sum_{k∈A} [(1-h[j+k]) - h[j-k]] * dx

This can be computed efficiently.

Let me implement this as the main optimization loop.

Actually, I just realized: for the specific case where A = {k*} (single active lag), the objective is:
c_{k*} = sum_i h[i]*(1-h[i+k*]) * dx

And the gradient at j is: (1-h[j+k*]) - h[j-k*] (times dx)

This is a linear function of h! So minimizing c_{k*} with bounds and the sum constraint is a LINEAR program! We can solve it exactly with scipy.optimize.linprog or even analytically.

Wait: c_{k*} = sum_i h[i] - sum_i h[i]*h[i+k*]. The second term h[i]*h[i+k*] is NOT linear. So it's not an LP.

Hmm, right. The product h[i]*h[i+k*] makes it quadratic. So it is a QP after all.

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            s = mask.sum()
            if s == 0:
                break
            h[mask] += diff / s
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            s = mask.sum()
            if s == 0:
                break
            h[mask] += diff / s
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
