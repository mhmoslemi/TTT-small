```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_c5_vec(h, n).max())

def project_feasible_fast(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    n = len(h)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    # f(lambda) = sum(clip(h_i - lambda, 0, 1)) is continuous, non-increasing
    # f(-inf) = n, f(inf) = 0
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    result = np.clip(h - lam, 0, 1)
    # Fix any numerical issues
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0, 1)
    return result

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    if max_shifted > 500:
        val = m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(shifted))) / alpha
    if mu > 0:
        val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    # Use the property: d c_k / d h_i = dx * [(1-h_{i+k}) - h_{i-k}]
    # For each k with weight w_k:
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wm = w[idx] * dx
        if k >= 0:
            # c_k = dx * sum_{i=0}^{n-1-k} h_i (1-h_{i+k})
            # d c_k / d h_i = dx * (1-h_{i+k}) for i < n-k
            # d c_k / d h_i = dx * (-h_{i-k}) for i >= k
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    if mu > 0:
        grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def coordinate_descent(h, n, deadline, n_passes_max=200, mu_penalty=10.0):
    """Coordinate descent with O(n^2) per pass."""
    best_h = project_feasible_fast(h.copy(), n/2.0)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    target_sum = n / 2.0
    c_vec = compute_c5_vec(best_h, n)
    
    for pass_idx in range(n_passes_max):
        if time.time() > deadline:
            break
        
        improved = False
        # Randomize order for diversity
        order = np.arange(n)
        np.random.shuffle(order)
        
        for i in order:
            if time.time() > deadline + 50:  # small overage allowed
                break
            
            # Compute slope: how c_k changes when h_i changes by 1
            # d c_k / d h_i = dx * [(1-h_{i+k}) - h_{i-k}] for valid indices
            slope = np.zeros(2*n-1)
            k_start = n - 1  # index in slope for k=0
            
            # For k >= 0 (shift to the right):
            # d c_k / d h_i involves h_{i+k} if i+k < n
            for k in range(n):
                if i + k < n:
                    # Term: h_i * (1 - h_{i+k}), derivative wrt h_i: (1-h_{i+k})
                    idx = k_start + k
                    slope[idx] += dx * (1.0 - best_h[i+k])
                if i - k >= 0 and k > 0:
                    # Term: h_{i-k} * (1 - h_i), derivative wrt h_i: -h_{i-k}
                    idx = k_start - k
                    slope[idx] += dx * (-best_h[i-k])
            
            # Now c_k(t) = c_vec[k] + (t - best_h[i]) * slope[k] for t in [0,1]
            # We want to minimize max_k c_k(t) + mu * (sum + (t - best_h[i]) - target)^2
            # The penalty is: mu * (current_sum + t - best_h[i] - target)^2
            current_sum = best_h.sum()
            # penalty(t) = mu * (current_sum + t - best_h[i] - target)^2
            # = mu * (t - (target - current_sum + best_h[i]))^2
            # Let p0 = target - current_sum + best_h[i]
            # penalty(t) = mu * (t - p0)^2
            
            # Total: f(t) = max_k (c_vec[k] + (t-hi)*slope[k]) + mu*(t-p0)^2
            # This is convex in t. Find minimum in [0,1].
            
            hi = best_h[i]
            
            # Try a few candidate points
            candidates = [0.0, 1.0, hi]
            
            # The unconstrained min of the penalty alone is t = p0
            p0 = target_sum - current_sum + hi
            if 0 <= p0 <= 1:
                candidates.append(p0)
            
            # Try to equalize the top 2 active constraints
            # c_k(t) = c_vec[k] + (t-hi)*slope[k]
            # For the top 2: c_vec[i1] + (t-hi)*slope[i1] = c_vec[i2] + (t-hi)*slope[i2]
            # (t-hi) = (c_vec[i2] - c_vec[i1]) / (slope[i1] - slope[i2])
            top5 = np.argsort(-c_vec[:])[:5]  # top 5 shifts by current value
            for a in range(5):
                for b in range(a+1, 5):
                    ka, kb = top5[a], top5[b]
                    denom = slope[ka] - slope[kb]
                    if abs(denom) > 1e-15:
                        t_cross = hi + (c_vec[kb] - c_vec[ka]) / denom
                        if -0.1 <= t_cross <= 1.1:
                            candidates.append(np.clip(t_cross, 0, 1))
            
            # Also add some random candidates
            # candidates.append(np.random.uniform(0, 1))
            
            # Evaluate at candidates
            best_t = hi
            best_val = best_c + 1  # current value
            
            for t in candidates:
                t = float(np.clip(t, 0, 1))
                # Compute c_k(t) for all k
                c_new = c_vec + (t - hi) * slope
                # Add penalty
                pen = mu_penalty * (t - p0) ** 2
                val = c_new.max() + pen
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Update
            delta = best_t - hi
            if abs(delta) > 1e-10:
                best_h[i] = best_t
                c_vec = c_vec + delta * slope
                current_sum += delta
                improved = True
        
        # Project at end of pass
        best_h = project_feasible_fast(best_h, target_sum)
        c_vec = compute_c5_vec(best_h, n)
        c_now = c_vec.max()
        
        if c_now < best_c:
            best_c = c_now
        # (best_h is already updated)
    
    return best_h, best_c

def pair_flip_search(h, n, deadline, radius=20):
    """Local search: try flipping pairs of values."""
    best_h = project_feasible_fast(h.copy(), n/2.0)
    best_c = compute_c5(best_h, n)
    
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0001]
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(0, n, 3):  # step by 3 for speed
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try moving d from i to j
                    if best_h[i] - d >= 0 and best_h[j] + d <= 1:
                        best_h[i] -= d
                        best_h[j] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] += d
                            best_h[j] -= d
                    # Try moving d from j to i
                    if best_h[j] - d >= 0 and best_h[i] + d <= 1:
                        best_h[j] -= d
                        best_h[i] += d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                        else:
                            best_h[j] += d
                            best_h[i] -= d
    
    return best_h, best_c
```

```python
def coordinate_descent_v2(h, n, deadline, n_passes_max=200, mu=50.0):
    best_h = project_feasible_fast(h.copy(), n/2.0)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    target = n / 2.0
    c_vec = compute_c5_vec(best_h, n)
    
    for pass_idx in range(n_passes_max):
        if time.time() > deadline:
            break
        
        order = np.random.permutation(n)
        
        for i in order:
            hi = best_h[i]
            
            # Compute slope
            slope = np.zeros(2*n-1)
            k_min = max(-i, i-(n-1))
            k_max = min(n-1-i, i)
            if k_max >= k_min:
                ks = np.arange(k_min, k_max+1)
                slope[n-1+ks] = dx * (1.0 - best_h[i+ks]) - dx * best_h[i-ks]
            
            # c_k(t) = c_vec[k] + (t-hi) * slope[k]
            # Minimize max_k c_k(t) + mu*(sum + (t-hi) - target)^2
            # For the penalty, let offset = target - best_h.sum()
            # penalty(t) = mu * (offset + t - hi)^2
            
            offset = target - best_h.sum()
            # penalty(t) = mu * (offset + t - hi)^2 = mu * (t - (hi - offset))^2
            t_penalty_center = hi - offset
            
            # Find minimum of f(t) = max_k(c_vec[k] + (t-hi)*slope[k]) + mu*(t - t_pen)^2
            # over t in [0,1]
            
            # Strategy: evaluate at a set of candidates
            # 1. t = 0, t = 1
            # 2. t = t_penalty_center (if in [0,1])
            # 3. Intersections of top lines with the parabola
            # 4. Intersections between top lines
            
            candidates = [0.0, 1.0, hi]
            if 0 <= t_penalty_center <= 1:
                candidates.append(t_penalty_center)
            
            # Top 5 by current value
            top5 = np.argsort(-c_vec)[:5]
            for a in range(5):
                for b in range(a+1, 5):
                    ka, kb = top5[a], top5[b]
                    denom = slope[ka] - slope[kb]
                    if abs(denom) > 1e-15:
                        t_cross = hi + (c_vec[kb] - c_vec[ka]) / denom
                        if 0 <= t_cross <= 1:
                            candidates.append(t_cross)
            
            # Evaluate
            best_t = hi
            best_f = np.inf
            for t in candidates:
                t = max(0.0, min(1.0, t))
                c_at_t = c_vec + (t - hi) * slope
                pen = mu * (t - t_penalty_center) ** 2
                f_val = c_at_t.max() + pen
                if f_val < best_f:
                    best_f = f_val
                    best_t = t
            
            if abs(best_t - hi) > 1e-12:
                delta = best_t - hi
                best_h[i] = best_t
                c_vec = c_vec + delta * slope
        
        # Project at end of pass
        best_h = project_feasible_fast(best_h, target)
        c_vec = compute_c5_vec(best_h, n)
        c_now = c_vec.max()
        if c_now < best_c:
            best_c = c_now
    
    return best_h, best_c
```

```python
def linearized_lp(h, n, deadline, max_iters=50):
    """Iteratively solve the linearized problem using LP."""
    from scipy.optimize import linprog
    
    best_h = project_feasible_fast(h.copy(), n/2.0)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    target = n / 2.0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = compute_c5_vec(best_h, n)
        max_val = c_vec.max()
        
        # Linearize: c_k(h) ≈ c_vec[k] + grad_k^T (h - best_h)
        # For each k: grad_k[i] = dc_k/dh_i at h = best_h
        # = dx * [(1-best_h[i+k]) - best_h[i-k]] (same as slope in CD)
        
        # LP: minimize t
        # subject to: c_vec[k] + grad_k^T (h - best_h) ≤ t  for all k
        #             0 ≤ h_i ≤ 1
        #             sum(h_i) = n/2
        
        # Variables: [h_0, h_1, ..., h_{n-1}, t]
        # Constraints:
        # For each k: -grad_k^T h + t ≤ c_vec[k] - grad_k^T best_h
        # sum(h) = n/2
        # 0 ≤ h_i ≤ 1, t free (or t ≥ 0)
        
        # This is a large LP. Let's only use the top-K constraints.
        K = 30
        top_k = np.argsort(-c_vec)[:K]
        
        # Build constraint matrix
        # A_ub @ x ≤ b_ub where x = [h, t]
        A_ub = np.zeros((K, n+1))
        b_ub = np.zeros(K)
        
        for idx, k_idx in enumerate(top_k):
            k = k_idx - (n-1)
            # grad_k[i] = dx * [(1-h[i+k]) - h[i-k]]
            grad_k = np.zeros(n)
            k_min = max(0, -k)
            k_max = min(n, n-k)
            # For i in [max(0,-k), min(n-1, n-1-k)]:
            # Actually, let me just compute it directly
            for i in range(n):
                val = 0.0
                if 0 <= i+k < n:
                    val += (1 - best_h[i+k])
                if 0 <= i-k < n:
                    val -= best_h[i-k]
                grad_k[i] = dx * val
            
            # Constraint: c_vec[k_idx] + grad_k^T (h - best_h) ≤ t
            # grad_k^T h - t ≤ c_vec[k_idx] - grad_k^T best_h
            A_ub[idx, :n] = grad_k
            A_ub[idx, n] = -1.0
            b_ub[idx] = c_vec[k_idx] - grad_k @ best_h
        
        # c_eq: sum(h) = n/2
        c_eq = np.zeros((1, n+1))
        c_eq[0, :n] = 1.0
        c_eq[0, n] = 0.0
        b_eq = np.array([target])
        
        # Objective: minimize t
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0
        
        bounds = [(0, 1)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=c_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                h_new = project_feasible_fast(h_new, target)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
        except Exception:
            break
    
    return best_h, best_c
```
