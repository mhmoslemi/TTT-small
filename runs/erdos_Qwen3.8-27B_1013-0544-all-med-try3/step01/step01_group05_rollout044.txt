            stag += 1
            if stag > 40:
                ss *= 0.85
                if ss < 1e-8:
                    h = best_h + np.random.RandomState(int(best_c5 * 1e6) % 2**31).normal(0, 0.04, n)
                    h = project_feasible(h, target_sum)
                    ss = ss_start * 0.7
                    mom = np.zeros(n)
                    stag = 0
                    c5 = compute_c5(h, dx)
            elif stag % 15 == 0:
                ss *= 0.93
        h = h_new
        if stag == 0:
            c5 = c5_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return (start_time + budget_s - 5.0) - time.time()

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        init_h = None

    n_candidates = [55, 60, 70, 80, 90]
    best_h_global = None
    best_c5_global = 1.0
    best_n_global = 70

    # Phase 1: Multi-start across n_points
    for n in n_candidates:
        if tleft() < 50.0:
            break
        dx = 2.0 / n
        ts = n / 2.0
        inits = []

        if init_h is not None:
            if len(init_h) == n:
                inits.append(init_h.copy())
            else:
                xs = np.linspace(0, 1, len(init_h))
                inits.append(np.interp(np.linspace(0, 1, n), xs, init_h))

        tm = thue_morse(n).astype(float)
        inits.append(project_feasible(0.5 + 0.3 * (1 - 2 * tm), ts))

        h_alt = np.tile([0.75, 0.25], n // 2 + 1)[:n]
        inits.append(project_feasible(h_alt + rng.normal(0, 0.08, n), ts))

        inits.append(project_feasible(rng.uniform(0.2, 0.8, n), ts))

        h_s = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
        inits.append(project_feasible(h_s, ts))

        # Walsh-like: use Fourier modes
        freqs = [2, 3, 5, 7, 11]
        for f in freqs:
            h_w = 0.5 + 0.3 * np.cos(2 * np.pi * f * np.arange(n) / n)
            inits.append(project_feasible(h_w, ts))

        for h0 in inits:
            if tleft() < 20.0:
                break
            h_opt, c5_opt = optimize_local(h0, dx, ts, n, tleft, max_iter=1500, ss_start=0.004, tol=0.01)
            if c5_opt < best_c5_global:
                best_c5_global = c5_opt
                best_h_global = h_opt
                best_n_global = n

    # Phase 2: SLSQP on best
    if best_h_global is not None and tleft() > 120.0:
        n = best_n_global
        dx = 2.0 / n
        ts = n / 2.0
        try:
            res = minimize(slsqp_obj, best_h_global, args=(dx,), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (ts,)}],
                          options={'maxiter': 15000, 'ftol': 1e-16})
            if res.x is not None:
                h_ref = project_feasible(res.x, ts)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5_global:
                    best_c5_global = c5_ref
                    best_h_global = h_ref
        except Exception:
            pass

    # Phase 3: Fine gradient
    if best_h_global is not None and tleft() > 40.0:
        n = best_n_global
        dx = 2.0 / n
        ts = n / 2.0
        h = best_h_global.copy()
        ss = 0.0003
        mom = np.zeros(n)
        mu = 0.4
        c5 = best_c5_global
        for _ in range(50000):
            if tleft() < 10.0:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            active = get_active_set(corr, 5e-5, max_act=8)
            g = active_gradient(h, n, active)
            mom = mu * mom - ss * g
            h_new = project_feasible(h + mom, ts)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5_global:
                best_c5_global = c5_new
                best_h_global = h_new.copy()
            else:
                ss *= 0.93
                if ss < 1e-12:
                    break
            h = h_new

    # Phase 4: Perturbation search
    if best_h_global is not None and tleft() > 15.0:
        n = best_n_global
        dx = 2.0 / n
        ts = n / 2.0
        for scale in [0.005, 0.003, 0.002, 0.001]:
            for _ in range(400):
                if tleft() < 8.0:
                    break
                h_p = best_h_global + rng.normal(0, scale, n)
                h_p = project_feasible(h_p, ts)
                c5_p = compute_c5(h_p, dx)
                if c5_p < best_c5_global:
                    best_c5_global = c5_p
                    best_h_global = h_p.copy()
                    h = h_p.copy()
                    ss = 0.0001
                    mom = np.zeros(n)
                    for _ in range(500):
                        if tleft() < 8.0:
                            break
                        corr = np.correlate(h, 1.0 - h, mode='full')
                        act = get_active_set(corr, 1e-4, max_act=5)
                        g = active_gradient(h, n, act)
                        mom = 0.4 * mom - ss * g
                        h = project_feasible(h + mom, ts)
                        c5_t = compute_c5(h, dx)
                        if c5_t < best_c5_global:
                            best_c5_global = c5_t
                            best_h_global = h.copy()
                        else:
                            ss *= 0.9
                            if ss < 1e-13:
                                break

    # Phase 5: Second SLSQP pass if time allows
    if best_h_global is not None and tleft() > 80.0:
        n = best_n_global
        dx = 2.0 / n
        ts = n / 2.0
        try:
            res = minimize(slsqp_obj, best_h_global, args=(dx,), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (ts,)}],
                          options={'maxiter': 20000, 'ftol': 1e-16})
            if res.x is not None:
                h_ref = project_feasible(res.x, ts)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5_global:
                    best_c5_global = c5_ref
                    best_h_global = h_ref
        except Exception:
            pass

    return (best_h_global, best_c5_global, best_n_global)
