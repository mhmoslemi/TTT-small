def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, tau, mu):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    obj = (x_max + np.log(np.sum(np.exp(x - x_max)))) / tau * dx
    obj += mu * (h.sum() - target) ** 2
    return obj

def lse_grad(h, tau, mu):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= np.sum(w)
    conv_wh = np.convolve(w, h, mode='full')
    term1 = conv_wh[n-1:2*n-1]
    conv_rh = np.convolve(w[::-1], h, mode='full')
    term2 = conv_rh[n-1:2*n-1]
    grad = dx * (1.0 - term1 - term2)
    grad += 2.0 * mu * (h.sum() - target)
    return grad

def smooth_max_val(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def slsqp_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, tau)

def slsqp_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= np.sum(w)
    conv_wh = np.convolve(w, h, mode='full')
    term1 = conv_wh[n-1:2*n-1]
    conv_rh = np.convolve(w[::-1], h, mode='full')
    term2 = conv_rh[n-1:2*n-1]
    return 1.0 - term1 - term2

def optimize_lbfgsb(h_start, tau, mu, maxiter=400):
    n = len(h_start)
    bnds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(lse_obj, h_start, jac=lse_grad, args=(tau, mu),
                   method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30})
    return res.x

def optimize_slsqp(h_start, tau, maxiter=300):
    n = len(h_start)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,

                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

            x = optimize_lbfgsb(best_h, tau, mu, maxiter=400)
            x = fix_sum(np.clip(x, 0, 1), n0/2)
            c5 = evaluate_c5(x)
            if c5 < best_c5:
                best_h, best_c5 = x, c5
        except Exception:
            pass

    # Phase 2: SLSQP polish (exact constraint)
    for tau in [100, 50, 30]:
        if rem() < 20:
            break
        try:
            x = optimize_slsqp(best_h, tau, maxiter=300)
            x = fix_sum(np.clip(x, 0, 1), n0/2)
            c5 = evaluate_c5(x)
            if c5 < best_c5:
                best_h, best_c5 = x, c5
        except Exception:
            pass

    # Phase 3: Diverse restarts
    for i in range(8):
        if rem() < 40:
            break
        style = i % 5
        h_cand = make_perturbation(best_h, n0, rng, style)
        for tau in [500, 200, 80, 40, 25]:
            if rem() < 15:
                break
            try:
                x = optimize_lbfgsb(h_cand, tau, mu, maxiter=300)
                x = fix_sum(np.clip(x, 0, 1), n0/2)
                c5 = evaluate_c5(x)
                if c5 < best_c5:
                    best_h, best_c5 = x, c5
                h_cand = x
            except Exception:
                pass

    # Phase 4: Try different resolutions
    for tn in [80, 100, 140, 160]:
        if rem() < 40:
            break
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, tn/2)
        for tau in [500, 200, 80, 40, 25]:
            if rem() < 15:
                break
            try:
                x = optimize_lbfgsb(h_t, tau, mu, maxiter=300)
                x = fix_sum(np.clip(x, 0, 1), tn/2)
                c5 = evaluate_c5(x)
                if c5 < best_c5:
                    best_h, best_c5, tn = x, c5, tn
                h_t = x
            except Exception:
                pass
        # SLSQP polish at this resolution
        if rem() > 15:
            try:
                x = optimize_slsqp(h_t, 30, maxiter=200)
                x = fix_sum(np.clip(x, 0, 1), tn/2)
                c5 = evaluate_c5(x)
                if c5 < best_c5:
                    best_h, best_c5 = x, c5
            except Exception:
                pass

    # Phase 5: Final fine-tuning at best resolution
    hn = len(best_h)
    if rem() > 20:
        for tau in [20, 15, 10, 8]:
            if rem() < 12:
                break
            try:
                x = optimize_lbfgsb(best_h, tau, mu, maxiter=400)
                x = fix_sum(np.clip(x, 0, 1), hn/2)
                c5 = evaluate_c5(x)
                if c5 < best_c5:
                    best_h, best_c5 = x, c5
            except Exception:
                pass

    # Phase 6: Very fine SLSQP
    if rem() > 15:
        try:
            x = optimize_slsqp(best_h, 20, maxiter=400)
            x = fix_sum(np.clip(x, 0, 1), hn/2)
            c5 = evaluate_c5(x)
            if c5 < best_c5:
                best_h, best_c5 = x, c5
        except Exception:
            pass

    # Final constraint enforcement
    best_h = fix_sum(best_h, len(best_h)/2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
