def smooth_lse(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def obj_lam(h, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = smooth_lse(c, alpha)
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def obj_slsqp(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_lse(c, alpha)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B with penalty continuation (fast)
    lams = [500, 2000, 10000, 50000, 200000, 1000000]
    for lam in lams:
        if rem() < 8:
            break
        for alpha in [300, 1000]:
            try:
                res = minimize(obj_lam, best_h, args=(n, alpha, lam),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 2: SLSQP with increasing alpha (precise constrained optimization)
    for alpha in [500, 2000, 5000, 15000, 50000]:
        if rem() < 15:
            break
        try:
            res = minimize(obj_slsqp, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-18, 'maxls': 100})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 3: Aggressive multi-restart with L-BFGS-B + SLSQP polish
    attempt = 0
    while rem() > 40:
        attempt += 1
        scale = rng.uniform(0.005, 0.12) * (1.0 + 0.1 * np.sin(attempt))
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)

        try:
            res = minimize(obj_lam, h_pert, args=(n, 2000, 500000),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-13:
                best_h, best_c5 = h_try, c
                if rem() > 25:
                    try:
                        res2 = minimize(obj_slsqp, best_h, args=(n, 10000),
                                      method='SLSQP', bounds=bounds, constraints=cons,
                                      options={'maxiter': 3000, 'ftol': 1e-18})
                        h2 = project_feasible(res2.x, n)
                        c2 = compute_c5(h2, n)
                        if c2 < best_c5 - 1e-14:
                            best_h, best_c5 = h2, c2
                    except Exception:
                        pass
        except Exception:
            pass

    # Phase 4: Fine coordinate descent with very small steps
    if rem() > 15:
        deltas = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02]
        count = 0
        no_improve = 0
        while rem() > 15 and count < 100000 and no_improve < 10000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            improved = False
            for d in deltas:
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-15:
                            best_h = h_try
                            best_c5 = c
                            improved = True
                            break
                if improved:
                    no_improve = 0
                    break
            if not improved:
                no_improve += 1

    # Phase 5: Final SLSQP polish with very high alpha
    if rem() > 5:
        for alpha in [50000, 200000]:
            if rem() < 5:
                break
            try:
                res = minimize(obj_slsqp, best_h, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 8000, 'ftol': 1e-19, 'maxls': 200})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
