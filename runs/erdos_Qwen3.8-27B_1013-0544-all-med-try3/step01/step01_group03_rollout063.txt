    dx = 2.0 / n

def project_feasible(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Second pass
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(corr, tau):
    x = corr * tau
    xm = x.max()
    return (xm + np.log(np.sum(np.exp(x - xm)))) / tau

    """Returns (value, gradient) of smooth max of corr."""
    n = len(h)

    xm = x.max()
    w = np.exp(x - xm)
    w /= w.sum()
    val = (xm + np.log(w.sum() * np.exp(xm - xm))) / tau  # = (xm + 0)/tau since w sums to 1
    # Actually: logsumexp(x)/tau = (xm + log(sum(exp(x-xm))))/tau
    # sum(exp(x-xm)) = sum(w * sum(exp(x-xm))) / sum(w) ...

    """Nesterov accelerated projected gradient descent."""
    n = len(h_init)
    h_curr = h_init.copy()
    v = np.zeros(n)
    best_h = h_init.copy()
    best_val = evaluate_c5(h_init)
    t_start = time.time()
    lr = lr0
    
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        # Nesterov lookahead
        h_nest = h_curr + momentum * v
        h_nest = project_feasible(h_nest, target)
        
        val, grad = smooth_max_grad_objective(h_nest, tau)
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        h_new = h_curr + v
        h_new = project_feasible(h_new, target)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_val:
            best_val = c5
            best_h = h_new.copy()
            # Increase lr slightly
            lr = min(lr * 1.05, lr0 * 3)
        else:
            # Reduce lr
            lr *= 0.9
            if lr < lr0 * 0.01:
                lr = lr0
            # Backtrack
            v = momentum * v
            h_curr = project_feasible(h_curr + 0.5 * v, target)
            continue
            
        h_curr = h_new
    
    return best_h, best_val

def slsqp_optimize(h_init, target, tau, time_limit):
    """SLSQP optimization with smooth max objective."""
    n = len(h_init)
    bnds = Bounds(np.zeros(n), np.ones(n))

        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0/n)
        x_t = corr * tau
        xm = x_t.max()
        return (xm + np.log(np.sum(np.exp(x_t - xm)))) / tau

        val, grad = smooth_max_grad_objective(x, tau)
        return grad

            break
        hn = project_feasible(h_curr + momentum * v, target)
        val, grad = compute_grad(hn, tau)
        v = momentum * v - lr * grad
        h_new = project_feasible(h_curr + v, target)
        c5 = evaluate_c5(h_new)
        if c5 > best_c5 * 1.5:
            lr *= 0.7
            v *= 0.3
        elif c5 < best_c5:
            lr = min(lr * 1.01, 3.0)
        h_curr = h_new
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    return best_h, best_c5

def slsqp_fun(x, target, tau, n):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    xt = corr * tau
    xm = xt.max()
    return (xm + np.log(np.sum(np.exp(xt - xm)))) / tau

def slsqp_jac(x, target, tau, n):
    _, g = compute_grad(x, tau)
    return g

def slsqp_cfun(x, target, tau, n):
    return x.sum() - target

def slsqp_cjac(x, target, tau, n):
    return np.ones(n)

def slsqp_opt(h, target, tau, tlimit, maxiter=300):
    n = len(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    args = (target, tau, n)
    cons = [{'type': 'eq', 'fun': slsqp_cfun, 'jac': slsqp_cjac, 'args': args}]
    t_s = time.time()
    try:
        res = minimize(slsqp_fun, h, args=args, jac=slsqp_jac, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40})
        if time.time() - t_s > tlimit:
            return h, evaluate_c5(h)
        x = project_feasible(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h, evaluate_c5(h)

def make_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    for attempt in range(200):
        h = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
        c5 = evaluate_c5(h)
        if c5 < 0.42:
            return h
    h = np.zeros(n)
    h[:int(target)] = 1.0
    return h

def make_smooth_init(n, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    x = np.linspace(0, 1, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2*np.pi*x + rng.uniform(0, 2*np.pi))
    h += 0.1 * np.sin(4*np.pi*x + rng.uniform(0, 2*np.pi))
    return project_feasible(h, target)
