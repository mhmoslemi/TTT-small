```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, tau):
    x = tau * vals
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.array(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_gradient(h, target_sum, tau):
    """Numerical gradient of smooth max objective"""
    n = len(h)
    dx = 2.0 / n
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, hp[i] + eps)
        hp = project_feasible(hp, target_sum)
        cp = np.correlate(hp, 1.0 - hp, mode="full") * dx
        fp = smooth_max(cp, tau)
        
        hm = h.copy()
        hm[i] = max(0.0, hm[i] - eps)
        hm = project_feasible(hm, target_sum)
        cm = np.correlate(hm, 1.0 - hm, mode="full") * dx
        fm = smooth_max(cm, tau)
        
        grad[i] = (fp - fm) / (2.0 * eps)
    return grad

def projected_gradient_descent(h, target_sum, tau, max_iter=100, step_init=0.05):
    """Projected gradient descent with backtracking"""
    n = len(h)
    h = h.copy()
    step = step_init
    best_local = h.copy()
    best_local_c5 = evaluate_c5(h)
    
    for it in range(max_iter):
        if it % 50 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_local_c5:
                best_local = h.copy()
                best_local_c5 = c5
        
        grad = compute_gradient(h, target_sum, tau)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        # Backtracking line search
        alpha = step
        for _ in range(20):
            h_trial = project_feasible(h - alpha * grad, target_sum)
            f_trial = smooth_max(np.correlate(h_trial, 1.0 - h_trial, mode="full") * (2.0/n), tau)
            f_curr = smooth_max(np.correlate(h, 1.0 - h, mode="full") * (2.0/n), tau)
            if f_trial < f_curr - 1e-4 * alpha * grad_norm**2:
                break
            alpha *= 0.5
        
        h = project_feasible(h - alpha * grad, target_sum)
        step = alpha
    
    c5 = evaluate_c5(h)
    if c5 < best_local_c5:
        best_local = h.copy()
        best_local_c5 = c5
    return best_local, best_local_c5

def slsqp_optimize(h, n, tau, maxiter=300):
    """SLSQP optimization with smooth max"""
    target = n / 2.0
    dx = 2.0 / n
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        return smooth_max(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = project_feasible(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h, None
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(vals, tau):
    x = tau * vals
    x_max = x.max()
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    x = tau * corr
    x_max = x.max()
    w = np.exp(x - x_max)
    w_sum = w.sum()
    one_minus_h = 1.0 - h
    rev_omh = one_minus_h[::-1]
    T1 = np.array([np.dot(w[j:j+n], rev_omh) for j in range(n)])
    T2 = np.array([np.dot(w[n-1-j:2*n-1-j], h) for j in range(n)])
    return dx * (T1 - T2) / w_sum

def pgd_step(h, target_sum, tau, max_iter=100, step_init=0.1):
    n = len(h)
    h = h.copy()
    step = step_init
    for it in range(max_iter):
        grad = compute_grad(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Backtracking line search
        alpha = step
        f_curr = smooth_max_val(np.correlate(h, 1.0-h, mode="full") * (2.0/n), tau)
        for _ in range(30):
            h_trial = project_feasible(h - alpha * grad, target_sum)
            f_trial = smooth_max_val(np.correlate(h_trial, 1.0-h_trial, mode="full") * (2.0/n), tau)
            if f_trial <= f_curr - 1e-4 * alpha * gnorm**2:
                break
            alpha *= 0.5
        h = project_feasible(h - alpha * grad, target_sum)
        step = alpha
    return h

def slsqp_refine(h, n, tau, maxiter=200):
    target = n / 2.0
    dx = 2.0 / n
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smooth_max_val(np.correlate(x, 1.0-x, mode="full") * dx, tau)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        x = project_feasible(res.x, target)
        return x
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(120, 0.5)
    
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: PGD with annealing tau
    for tau in [3, 10, 30, 100, 300, 1000, 3000]:
        if rem() < 20:
            break
        h_new = pgd_step(best_h, target, tau, max_iter=100, step_init=0.1)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: SLSQP refinement
    for tau in [1000, 3000, 10000]:
        if rem() < 15:
            break
        h_new = slsqp_refine(best_h, n, tau, maxiter=300)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: More PGD at high tau
    for tau in [3000, 5000, 10000]:
        if rem() < 15:
            break
        h_new = pgd_step(best_h, target, tau, max_iter=50, step_init=0.05)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 4: Exploration - random perturbations + local optimization
    if rem() > 60:
        for trial in range(20):
            if rem() < 20:
                break
            scale = 0.03 + 0.07 * np.random.random()
            h_try = project_feasible(best_h + np.random.normal(0, scale, n), target)
            # Local optimize
            h_try = pgd_step(h_try, target, 300, max_iter=30, step_init=0.05)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                # Further refine
                h_try = slsqp_refine(best_h, n, 1000, maxiter=200)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
    
    # Phase 5: Try different resolutions
    if rem() > 40:
        for tn in [80, 100, 140, 160]:
            if rem() < 25:
                break
            t_target = tn / 2.0
            x_src = np.linspace(0, 1, n)
            x_dst = np.linspace(0, 1, tn)
            h_t = np.interp(x_dst, x_src, best_h)
            h_t = project_feasible(h_t, t_target)
            h_t = pgd_step(h_t, t_target, 100, max_iter=50, step_init=0.05)
            h_t = slsqp_refine(h_t, tn, 1000, maxiter=200)
            c5_t = evaluate_c5(h_t)
            if c5_t < best_c5:
                best_h = h_t
                best_c5 = c5_t
```

```python
        for i in range(n):

            # O = sum_j h[j]*(1-h[j+lag]) for valid j
            # Terms with h[i]: h[i]*(1-h[i+lag]) + h[i-lag]*(1-h[i])
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(vals, tau):
    x = tau * vals
    x_max = x.max()
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_grad(h, tau):
    """Analytical gradient of smooth max objective w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    x = tau * corr
    x_max = x.max()
    w = np.exp(x - x_max)
    w_sum = w.sum()
    omh = 1.0 - h

    T1 = np.array([np.dot(w[j:j+n], rev_omh) for j in range(n)])
    # T2[j] = sum_i w[n-1-j+i] * h[i]
    T2 = np.array([np.dot(w[n-1-j:2*n-1-j], h) for j in range(n)])
    return dx * (T1 - T2) / w_sum

def pgd_optimize(h, target_sum, tau, max_iter=100, step_init=0.1):

        alpha = step

            h_trial = project_feasible(h - alpha * grad, target_sum)
            f_trial = smooth_max_val(np.correlate(h_trial, 1.0 - h_trial, mode="full") * dx, tau)
            if f_trial <= f_curr - 1e-5 * alpha * gnorm * gnorm:
                h = h_trial
                step = alpha
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return h

def coord_descent_worst_lag(h, target_sum, max_iter=50):
    """Coordinate descent targeting the current worst lag"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for it in range(max_iter):
        # Find worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        # Coordinate descent on this lag
        for i in range(n):
            coeff = 1.0
            if 0 <= i + lag < n:
                coeff -= h[i + lag]
            if 0 <= i - lag < n:
                coeff -= h[i - lag]
            if coeff > 1e-10:
                h[i] = 0.0
            elif coeff < -1e-10:
                h[i] = 1.0
        h = project_feasible(h, target_sum)
        # Check convergence
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        if np.max(corr_new - corr) < 1e-10:
            break
        corr = corr_new
    return h

def slsqp_refine(h, n, tau, maxiter=300):
    target = n / 2.0
    dx = 2.0 / n

                      options={'maxiter': maxiter, 'ftol': 1e-14})

    except:
        return h
```
