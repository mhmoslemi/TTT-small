import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + (1.0/t) * np.log(np.sum(np.exp(t*(corr - mc))))) * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Redistribute proportionally
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_step(h, dx, t, target_sum, maxiter=300, penalty=1e4):
    n = len(h)
    def obj(x):
        x_clip = np.clip(x, 0.0, 1.0)
        s = smooth_max(x_clip, dx, t)
        pen = penalty * (np.sum(x_clip) - target_sum)**2
        return s + pen
    res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return fix_sum(res.x, target_sum)
