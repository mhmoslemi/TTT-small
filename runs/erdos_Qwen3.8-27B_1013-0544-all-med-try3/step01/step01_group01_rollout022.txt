    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def water_fill_project(y, target_sum):
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def subgradient_vec(h, s):
    n = len(h)
    if s == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if s > 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        t = -s
        g[t:] += (1.0 - h[:n - t])
        g[:n - t] -= h[t:]
    return g


def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def do_subgradient(h, n, target_sum, budget_end, top_m_init=3):
    step = 0.12
    top_m = top_m_init
    pat = 0
    c5 = compute_c5(h)
    it = 0
    while time.time() < budget_end:
        it += 1
        overlaps = compute_all_overlaps(h)
        if top_m >= len(overlaps):
            top_m = 3
        top_idx = np.argpartition(overlaps, -top_m)[-top_m:]
        g = np.zeros(n)
        for idx in top_idx:
            s = int(idx) - (n - 1)
            g += subgradient_vec(h, s)
        g /= float(top_m)
        g_norm = np.sqrt(np.sum(g * g))
        if g_norm < 1e-12:
            break
        g = g / g_norm
        h_new = water_fill_project(h - step * g_norm * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-13:
            h, c5 = h_new, c5_new
            pat = 0
            if step > 0.02:
                step *= 0.9985
        else:
            pat += 1
            if pat > 40:
                h = water_fill_project(h + np.random.randn(n) * 0.04, target_sum)
                c5 = compute_c5(h)
                step = 0.12
                pat = 0
                top_m = int(np.random.choice([2, 3, 5, 7]))
            elif pat % 15 == 0:
                step = max(0.02, step * 0.7)
    return h, c5


def do_slsqp_polish(h, n, target_sum, beta=150.0, maxiter=1500):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_func(h_in):
        corr = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max_val(corr, beta) * dx

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = water_fill_project(res.x, target_sum)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    global_best_h = None
    global_best_c5 = float('inf')
    global_best_n = 0

    def time_left():
        return budget_s - (time.time() - start_time)

    def update_best(h, c5, n):
        nonlocal global_best_h, global_best_c5, global_best_n
        if c5 < global_best_c5:
            global_best_h = h.copy()
            global_best_c5 = c5
            global_best_n = n

    def optimize_single(n_pts, h_init, time_alloc):
        n = n_pts
        target_sum = n / 2.0
        h = water_fill_project(h_init, target_sum)
        c5 = compute_c5(h)
        update_best(h, c5, n)
        deadline = start_time + time_alloc

        # Phase A: Subgradient descent
        sg_deadline = deadline - max(20, int(time_alloc * 0.3))
        h, c5 = do_subgradient(h, n, target_sum, sg_deadline)
        update_best(h, c5, n)

        # Phase B: SLSQP polish
        if time.time() < deadline - 10:
            h2, c52 = do_slsqp_polish(h, n, target_sum, beta=120, maxiter=1200)
            if c52 < c5:
                h, c5 = h2, c52
                update_best(h, c5, n)

        # Phase C: Another subgradient round (finer)
        if time.time() < deadline - 5:
            h, c5 = do_subgradient(h, n, target_sum, deadline - 5, top_m_init=2)
            update_best(h, c5, n)

        # Phase D: Final SLSQP
        if time.time() < deadline - 2:
            h2, c52 = do_slsqp_polish(h, n, target_sum, beta=200, maxiter=2000)
            if c52 < c5:
                h, c5 = h2, c52
                update_best(h, c5, n)

        return h, c5

    # Main run on initial n
    n_init = len(h0)
    t_main = int(budget_s * 0.55)
    optimize_single(n_init, h0, t_main)

    # Try alternative n values
    for n_try in [64, 72, 80, 90]:
        if time_left() < 50:
            break
        x_old = np.linspace(0, 2, n_init, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, h0, period=2.0)
        h_interp = water_fill_project(h_interp, n_try / 2.0)
        t_alloc = min(45, max(20, int(time_left() * 0.2)))
        if t_alloc < 18:
            break
        optimize_single(n_try, h_interp, t_alloc)

    # Final high-precision polish on global best
    if time_left() > 12 and global_best_h is not None:
        n = len(global_best_h)
        target_sum = n / 2.0
        h2, c52 = do_slsqp_polish(global_best_h, n, target_sum, beta=250, maxiter=3000)
        if c52 < global_best_c5:
            global_best_h, global_best_c5 = h2, c52

        # One more subgradient + SLSQP cycle
        if time_left() > 10:
            n = len(global_best_h)
            target_sum = n / 2.0
            global_best_h, c5_temp = do_subgradient(
                global_best_h, n, target_sum,
                time.time() + max(0, time_left() - 12), top_m_init=3)
            if c5_temp < global_best_c5:
                global_best_c5 = c5_temp
            h2, c52 = do_slsqp_polish(global_best_h, n, target_sum, beta=300, maxiter=2000)
            if c52 < global_best_c5:
                global_best_h, global_best_c5 = h2, c52

    return global_best_h, global_best_c5, global_best_n
