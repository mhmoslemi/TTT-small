def compute_c5(h, dx):

def project_box_sum(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(v)
    # Bisection on lambda
    lo, hi = -v.max() - 1, -v.min() + 1  # Actually: clip(v - lambda, 0, 1)
    # We want sum(clip(v - lambda, 0, 1)) = target_sum
    # At lambda = -inf: sum = n
    # At lambda = +inf: sum = 0

    hi = v.max() + 1.0

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

    lam = (lo + hi) / 2

def corr_gradient(h, k):
    """Compute gradient of the k-th correlation output w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    # For output index k (0-indexed in the full correlation of length 2n-1):
    # grad[j] = (1-h[j+k-(n-1)]) * I(0<=j+k-(n-1)<n) - h[j+(n-1)-k] * I(0<=j+(n-1)-k<n)

    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    
    # Find top_k worst shifts

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):

        shifted = t * (corr - max_c)
        return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) / dx
    def sum_fun(x):
        return np.sum(x) - target_sum
    try:
        res = minimize(smooth_max_obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def subgradient_loop(h, target_sum, n_iter, alpha_init=0.02, top_k=5):
    n = len(h)
    alpha = alpha_init
    momentum = 0.85
    v = np.zeros(n)
    for it in range(n_iter):
        grad, _ = smooth_max_gradient(h, t=top_k * 10.0, top_k=top_k)
        h_raw = h - alpha * grad
        h_new = project_box_sum(h_raw, target_sum)
        v = momentum * v + (h_new - h)
        h_next = project_box_sum(h + v, target_sum)
        h = h_next
        alpha *= 0.9995
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: FISTA with increasing smoothness
    for t_smooth in [10, 20, 40, 80, 150]:
        if elapsed() > budget_s - 350:
            break
        remaining = budget_s - 350 - elapsed()
        n_iter = max(200, min(4000, int(remaining * 15)))
        h_new = fista_optimize(best_h, target_sum, t_smooth, n_iter)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: SLSQP refinement
    for t in [30, 80, 200, 500]:
        if elapsed() > budget_s - 200:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=600)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Subgradient with diverse restarts
    for trial in range(40):
        if elapsed() > budget_s - 100:
            break
        h_curr = best_h.copy()
        if trial > 0:
            n_pert = np.random.randint(5, 30 + trial * 3)
            n_pert = min(n_pert, n)
            idx = np.random.choice(n, n_pert, replace=False)
            scale = 0.2 + 0.1 * (trial % 5)
            h_curr[idx] += np.random.uniform(-scale, scale, n_pert)
            h_curr = fix_sum(h_curr, target_sum)
        n_iter = max(200, int((budget_s - 100 - elapsed()) * 8))
        h_curr = subgradient_loop(h_curr, target_sum, n_iter, alpha_init=0.015, top_k=5)
        c5_trial = compute_c5(h_curr, dx)
        if c5_trial < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_trial

    # Phase 4: Final SLSQP polish with high t
    if elapsed() < budget_s - 30:
        for t in [200, 500, 1000]:
            if elapsed() > budget_s - 10:
                break
            h_final = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=1500)
            c5_final = compute_c5(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final

    # Phase 5: Very fine FISTA polish
    if elapsed() < budget_s - 15:
        remaining = budget_s - 15 - elapsed()
        n_iter = max(500, int(remaining * 20))
        h_final = fista_optimize(best_h, target_sum, 200, n_iter)
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
