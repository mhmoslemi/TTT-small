import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-16:
                break
            h = np.clip(h + (diff / t) * room, 0, 1)
        else:
            t = h.sum()
            if t < 1e-16:
                break
            h = np.clip(h + (diff / t) * h, 0, 1)
    return h


def obj_lse(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    return float(m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha)


def coord_descent(h, n, n_sweeps=30):
    h = h.copy()
    for _ in range(n_sweeps):
        for i in range(n):
            h[i] = 0.0
            c0 = np.correlate(h, 1 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1 - h, mode='full')
            a = c1 - c0
            b = c0
            vs = np.linspace(0, 1, 300)
            vals = b + np.outer(vs, a)
            f_vals = vals.max(axis=1)
            v_best = vs[np.argmin(f_vals)]
            lo = max(0.0, v_best - 0.01)
            hi = min(1.0, v_best + 0.01)
            vs2 = np.linspace(lo, hi, 80)
            vals2 = b + np.outer(vs2, a)
            f_vals2 = vals2.max(axis=1)
            h[i] = vs2[np.argmin(f_vals2)]

def slsqp_polish(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_try = project_feasible(res.x, n)
        return h_try
    except Exception:
        return project_feasible(h, n)


def upsample(h, n, n2):
    """Upsample by repeating each element (piecewise constant)."""
    ratio = n2 // n
    h2 = np.repeat(h, ratio)
    return project_feasible(h2, n2)
