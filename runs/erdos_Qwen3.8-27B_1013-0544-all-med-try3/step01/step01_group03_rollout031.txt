import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def compute_grad(h, tau):
    """Compute gradient of smooth_max(corr, tau) * dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    weights = exp_x / np.sum(exp_x)
    
    # lags
    L = np.arange(-(n-1), n)  # shape (2n-1,)
    J = np.arange(n)[:, None]  # shape (n, 1)
    Lm = L[None, :]  # shape (1, 2n-1)
    
    # j - l valid: 0 <= j-l < n
    valid1 = (J - Lm) >= 0
    # j + l valid: 0 <= j+l < n  
    valid2 = (J + Lm) >= 0
    
    term1 = np.where(valid1, 1.0 - h[J - Lm], 0.0)
    term2 = np.where(valid2, h[J + Lm], 0.0)
    
    # grad[j] = dx * sum_l weights[l] * (term1[j,l] - term2[j,l])
    grad = dx * (term1 - term2) @ weights
    return grad

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    """One subgradient step with projection."""
    n = len(h)
    grad = compute_grad(h, tau)
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n / 2.0)
    return h_new

Let me implement exact coordinate descent:

For coordinate j, the objective as a function of t = h[j] is:
f(t) = max_k g_k(t) * dx

where g_k(t) = corr_k with h[j] replaced by t.

Since each g_k is linear in t (for fixed other coordinates), f(t) is the max of linear functions of t, hence convex and piecewise linear.

The minimum of f(t) on [0,1] occurs at:
- t = 0 or t = 1 (boundaries), or
- A point where two (or more) of the linear functions cross

To find the crossing points, I need to find all pairs (k1, k2) such that g_{k1}(t) = g_{k2}(t) has a solution in [0,1], and evaluate f at those points.

This is O((2n-1)^2) per coordinate update, which for n=120 is about 57000 operations. With n=120 coordinates, that's ~7M operations per full pass. This is feasible!

But implementing this correctly is a bit involved. Let me simplify:

For coordinate j, each g_k as a function of t has the form a_k * t + b_k where:
- a_k = coefficient of t in g_k (depends on j and k)
- b_k = constant term

Then g_k(t) = a_k * t + b_k
f(t) = max_k (a_k * t + b_k)

The minimum of f on [0,1] is at a breakpoint of the upper envelope, which can be found by:
1. Evaluating f at t=0 and t=1
2. Finding all intersection points of pairs of lines in [0,1]
3. Evaluating f at each intersection point
4. Taking the minimum

For step 2, the intersection of line k1 and k2 is at t = (b_k2 - b_k1) / (a_k1 - a_k2) (if a_k1 != a_k2).

This is O(K^2) where K = 2n-1 is the number of correlation lags. For n=120, K=239, so K^2 ≈ 57000. This is fine.

But actually, I can make the evaluation much faster. Instead of recomputing the full correlation each time, I can compute it incrementally.

When we change h[j] from h[j] to t, the change in corr[k] is:
delta_corr[k] = (t - h[j]) * coeff(k, j)

where coeff(k, j) is the coefficient of h[j] in corr[k].

For k != 0 (in the lag sense):
coeff = (1-h[j+k])*[valid] - h[j-k]*[valid]

For k = 0:
The term is h[j]*(1-h[j]) = t*(1-t), and the base is h[j]*(1-h[j]) (with original h[j]).
So delta = t*(1-t) - h[j]*(1-h[j])

This is a constant shift for all k, plus a linear (or quadratic for k=0) change.

So I can precompute the base correlation and the coefficients, then evaluate f(t) in O(K) time without calling correlate.

    # 3. i=j and i+k=j (both, i.e., k=0): h[j]*(1-h[j]) = t*(1-t)

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt > 0:
        per = diff / cnt
        # If per would push some values out of [0,1], iterate
        h_new = h.copy()
        h_new[mask] += per
        # Clip and redistribute
        h_new = np.clip(h_new, 0.0, 1.0)
        # Final adjustment
        diff2 = target - h_new.sum()
        if abs(diff2) > 1e-10:
            if diff2 > 0:
                mask2 = h_new < 1.0
            else:
                mask2 = h_new > 0.0
            if mask2.sum() > 0:
                h_new[mask2] += diff2 / mask2.sum()
                h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    return np.clip(h, 0.0, 1.0)

def find_optimal_hj_vec(h, j):
    """Find t in [0,1] minimizing max_k corr with h[j]=t, using vectorized computation."""
    n = len(h)
    dx = 2.0 / n
    K = 2 * n - 1
    
    # Compute base correlation (h[j] set to 0)
    h_mod = h.copy()
    h_mod[j] = 0.0
    corr_base = np.correlate(h_mod, 1.0 - h_mod, mode="full")
    
    L = np.arange(-(n-1), n)

    validA = (j - L >= 0) & (j - L < n)
    validB = (j + L >= 0) & (j + L < n)

    consts = np.zeros(K)
    
    if np.any(validA):
        coeffs[validA] += (1.0 - h[j - L[validA]])
    if np.any(validB):
        coeffs[validB] -= h[j + L[validB]]
        consts[validB] += h[j + L[validB]]
    
    zero_idx = n - 1
    coeffs[zero_idx] = 0.0
    consts[zero_idx] = 0.0

        m2 = b - (b - a) / 3.0

        cv1[zero_idx] += m1 * (1.0 - m1)
        f1 = cv1.max()
        
        cv2 = corr_base + coeffs * m2 + consts
        cv2[zero_idx] += m2 * (1.0 - m2)
        f2 = cv2.max()
        
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    return (a + b) / 2.0

def compute_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    
    L = np.arange(-(n-1), n)[None, :]
    J = np.arange(n)[:, None]
    validA = (J - L >= 0)  # j-L valid (as first factor h[i] with i=j)
    validB = (J + L >= 0)  # j+L valid (as second factor (1-h[i]) with i=j)

    # Vectorized:
    termA = np.where(validA, 1.0 - h[J - L], 0.0)
    termB = np.where(validB, h[J + L], 0.0)
    
    grad = dx * (termA - termB) @ w
    return grad
