def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_free = free.sum()
        if n_free == 0:
            if diff > 0:
                # Need to increase; add to smallest values
                idx = np.argsort(h)[:max(1, int(abs(diff)))]
                h[idx] += diff / len(idx)
            else:
                idx = np.argsort(h)[-max(1, int(abs(diff))):]
                h[idx] += diff / len(idx)
        else:
            h[free] += diff / n_free
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        diff = target_sum - s
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def reduce_active_max(h, n_points, max_iter=100):
    """Try to reduce the maximum correlation by adjusting h."""
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n_points - 1)
        
        # Gradient of f_k w.r.t. h
        # f_k = dx * sum_{i valid} h[i] * (1-h[i+lag])
        # d f_k / d h[j] = dx * [(1-h[j+lag]) - h[j+lag] - h[j-lag] + h[j-lag]]
        # Wait, let me recompute.
        # f_k = dx * [sum_{i valid} h[i] - sum_{i valid} h[i]*h[i+lag]]
        # d/dh[j] of sum h[i] = dx * 1{j valid}
        # d/dh[j] of sum h[i]*h[i+lag] = dx * [1{j valid as i} * h[j+lag] + 1{j valid as i+lag} * h[j-lag]]
        
        grad = np.zeros(n_points)
        lo = max(0, -lag)
        hi = min(n_points, n_points - lag)
        for i in range(lo, hi):
            # d/dh[i] of h[i]*(1-h[i+lag]) = (1-h[i+lag])
            grad[i] += dx * (1 - h[i+lag])
            # d/dh[i+lag] of h[i]*(1-h[i+lag]) = -h[i]
            if 0 <= i+lag < n_points:
                grad[i+lag] -= dx * h[i]
        
        # Projected gradient descent step
        alpha = 0.1
        h_new = h - alpha * grad
        # Adjust to maintain sum
        h_new = h_new - (h_new.sum() - target_sum) / n_points
        h_new = np.clip(h_new, 0, 1)
        
        # Check if it improved the max
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        if np.max(corr_new) < np.max(corr) - 1e-15:
            h = h_new
        else:
            # Try smaller step
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
    return h

For k=0: ∫h(x)(1-h(x))dx = ∫h(x)dx - ∫h(x)²dx = 1 - ∫h²dx.
To minimize this, we want to maximize ∫h²dx, which means making h as "peaky" as possible (close to 0 or 1).

For k near 2: the overlap region is small, so the integral is small regardless.

For intermediate k: we want h(x) and h(x+k) to be "anti-correlated" (where h is high, h(·+k) is low).

This suggests that the optimal h might have a specific oscillatory pattern.

For a "two-level" function (h ∈ {0,1} a.e.), the problem becomes combinatorial: we want to arrange 1s and 0s in [0,2] (with half being 1s) such that for no shift k does the set of 1s overlap too much with the set of 0s (at shift k).

This is related to the concept of "difference sets" in combinatorics.

For the discrete version with n points, we want a binary (or near-binary) vector of length n with n/2 ones, such that the maximum cross-correlation between the vector and its complement is minimized.

This is similar to the "low autocorrelation" problem in signal processing!

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def obj_smooth(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx


def con_sum(h, target_sum):
    return np.sum(h) - target_sum


def con_sum_jac(h, target_sum):
    return np.ones(len(h))


def grad_lag(h, lag, n_points):
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    lo = max(0, -lag)
    hi = min(n_points, n_points - lag)
    i_idx = np.arange(lo, hi)
    j_idx = i_idx + lag
    grad[i_idx] += dx * (1.0 - h[j_idx])
    grad[j_idx] -= dx * h[i_idx]
    return grad


def active_reduction(h, n_points, max_iter=200):
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n_points - 1)
        grad = grad_lag(h, lag, n_points)
        alpha = 0.5
        improved = False
        for _ in range(10):
            h_try = h - alpha * grad
            h_try = h_try - (h_try.sum() - target_sum) / n_points
            h_try = np.clip(h_try, 0.0, 1.0)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            if np.max(corr_try) * dx < np.max(corr) * dx - 1e-14:
                h = h_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # Try all near-max lags
            thresh = np.max(corr) - 0.01
            active_lags = np.where(corr >= thresh)[0]
            any_improved = False
            for ak in active_lags:
                lag = ak - (n_points - 1)
                grad = grad_lag(h, lag, n_points)
                alpha = 0.3
                for _ in range(8):
                    h_try = h - alpha * grad
                    h_try = h_try - (h_try.sum() - target_sum) / n_points
                    h_try = np.clip(h_try, 0.0, 1.0)
                    corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
                    if np.max(corr_try) * dx < np.max(corr) * dx - 1e-14:
                        h = h_try
                        any_improved = True
                        break
                    alpha *= 0.5
                if any_improved:
                    break
            if not any_improved:
                break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(50, 0.5)

    best_h = h_init.copy()
    best_n = len(h_init)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    n_options = sorted(set([best_n, 35, 42, 49, 56, 63]))

    for n_points in n_options:
        if tl() < 40:
            break
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds_list = [(0.0, 1.0)] * n_points
        constraints = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac, 'args': (target_sum,)}]

        if n_points == best_n:
            h0 = best_h.copy()
        else:
            x_old = np.linspace(0, 1, best_n, endpoint=False)
            x_new = np.linspace(0, 1, n_points, endpoint=False)
            h0 = np.interp(x_new, x_old, best_h)
            h0 = project_h(h0, target_sum)

        # Anneal beta for SLSQP
        for beta in [20, 40, 80, 150, 300, 500]:
            if tl() < 15:
                break
            try:
                res = minimize(obj_smooth, h0, args=(beta, dx), method='SLSQP',
                               bounds=bounds_list, constraints=constraints,
                               options={'maxiter': 300, 'ftol': 1e-13})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_new.copy(), n_points
                    h0 = h_new.copy()
            except Exception:
                pass

        # L-BFGS-B pass
        if tl() > 20:
            try:
                res = minimize(obj_smooth, h0, args=(200, dx), method='L-BFGS-B',
                               bounds=bounds_list,
                               options={'maxiter': 500, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)

            except Exception:
                pass

        # Active-set gradient reduction
        if tl() > 25:

            c5 = compute_c5(h_red)
            if c5 < best_c5:

                h0 = h_red.copy()

        # Random restarts
        n_restarts = max(3, int(min(15, tl() / 20)))
        for r in range(n_restarts):
            if tl() < 12:
                break
            if best_n == n_points:
                h0 = best_h.copy()
            else:
                h0 = project_h(np.random.random(n_points), target_sum)
            strat = r % 4
            if strat == 0:
                h0 += np.random.randn(n_points) * 0.06
            elif strat == 1:

                h0[idx] = np.random.random(len(idx))
            elif strat == 2:
                blen = max(3, n_points // 6)
                bs = np.random.randint(0, max(1, n_points - blen))
                h0[bs:bs + blen] = 1.0 - h0[bs:bs + blen]
            else:
                h0 += np.random.randn(n_points) * 0.03
            h0 = project_h(h0, target_sum)
            try:
                res = minimize(obj_smooth, h0, args=(100, dx), method='SLSQP',
                               bounds=bounds_list, constraints=constraints,
                               options={'maxiter': 150, 'ftol': 1e-12})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_new.copy(), n_points
            except Exception:
                pass

        # Nelder-Mead on the max directly
        if tl() > 25 and best_n == n_points:
            try:
                res = minimize(compute_c5, best_h, method='Nelder-Mead',
                               options={'maxiter': 150, 'xatol': 1e-9, 'fatol': 1e-12})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_new.copy()
            except Exception:
                pass

    # Final precise polish
    if tl() > 20:
        n_points = best_n
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        bounds_list = [(0.0, 1.0)] * n_points
        constraints = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac, 'args': (target_sum,)}]
        for beta in [400, 700]:
            if tl() < 10:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(beta, dx), method='SLSQP',
                               bounds=bounds_list, constraints=constraints,
                               options={'maxiter': 1000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_new.copy()
            except Exception:
                pass
        # Final active reduction
        if tl() > 15:
            h_red = active_reduction(best_h, best_n, max_iter=500)
            c5 = compute_c5(h_red)
            if c5 < best_c5:
                best_c5, best_h = c5, h_red.copy()

    # Ensure feasibility
    target_sum = best_n / 2.0
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
