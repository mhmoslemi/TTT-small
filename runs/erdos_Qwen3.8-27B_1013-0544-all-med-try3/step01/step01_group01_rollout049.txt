def corr_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ex = np.exp(beta * (corr - m))
    weights = ex / np.sum(ex)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        # weight for this lag
        wm = weights[n - 1 + lag]
        if wm < 1e-12:
            continue
        # For this lag, d(corr)/dh[i] = (1-h[i-lag])_valid - h[i+lag]_valid
        # Vectorize:
        # Term 1: (1-h[i-lag]) for valid i-lag in [0,n-1]
        # i ranges over [max(0,lag), min(n, n+lag)]
        i_start = max(0, lag)
        i_end = min(n, n + lag)
        if i_end > i_start:
            idx = np.arange(i_start, i_end)
            grad[idx] += wm * (1.0 - h[idx - lag])
        # Term 2: -h[i+lag] for valid i+lag in [0,n-1]
        # i ranges over [max(0,-lag), min(n, n-lag)]
        j_start = max(0, -lag)
        j_end = min(n, n - lag)
        if j_end > j_start:
            jdx = np.arange(j_start, j_end)
            grad[jdx] -= wm * h[jdx + lag]
    grad *= dx
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def proj_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-10:
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            rsum = room.sum()
            if rsum > 1e-10:
                h = h + (diff / rsum) * room
        else:
            space = h.copy()
            ssum = space.sum()
            if ssum > 1e-10:
                h = h + (diff / ssum) * space
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-10:
        idx = (h > 0.01) & (h < 0.99)
        cnt = idx.sum()
        if cnt > 0:
            h[idx] += (target - s) / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def obj_and_grad(h, beta, dx, target_sum, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ex = np.exp(beta * (corr - m))
    ex_sum = np.sum(ex)
    weights = ex / ex_sum
    val = (m + np.log(ex_sum) / beta) * dx
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        wm = weights[n - 1 + lag]
        if wm < 1e-12:
            continue
        i_start = max(0, lag)
        i_end = min(n, n + lag)
        if i_end > i_start:
            idx = np.arange(i_start, i_end)
            grad[idx] += wm * (1.0 - h[idx - lag])
        j_start = max(0, -lag)
        j_end = min(n, n - lag)
        if j_end > j_start:
            jdx = np.arange(j_start, j_end)
            grad[jdx] -= wm * h[jdx + lag]
    grad *= dx
    if penalty > 0:
        s_diff = np.sum(h) - target_sum
        val += penalty * s_diff * s_diff
        grad += 2.0 * penalty * s_diff
    return val, grad


def make_obj_funcs(n_points, dx, target_sum, penalty):
    def ofun(h):
        return obj_and_grad(h, 100.0, dx, target_sum, penalty)[0]
    return ofun


def slsqp_solved(h0, beta, n_points, dx, target_sum, bounds, maxiter, ftol):
    def ofun(h):
        return obj_and_grad(h, beta, dx, target_sum, 0.0)[0]
    def ograd(h):
        return obj_and_grad(h, beta, dx, target_sum, 0.0)[1]
    def con_fun(h):
        return np.sum(h) - target_sum
    def con_jac(h):
        return np.ones(n_points)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(ofun, h0, jac=ograd, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return proj_h(res.x, target_sum)
    except Exception:
        return None


def lgbfgs_solve(h0, beta, n_points, dx, target_sum, bounds, maxiter, penalty):
    def ofun(h):
        return obj_and_grad(h, beta, dx, target_sum, penalty)[0]
    def ograd(h):
        return obj_and_grad(h, beta, dx, target_sum, penalty)[1]
    lo = np.zeros(n_points)
    hi = np.ones(n_points)
    try:
        res = minimize(ofun, h0, jac=ograd, method='L-BFGS-B',
                       bounds=(lo, hi), options={'maxiter': maxiter, 'gtol': 1e-10})
        return proj_h(res.x, target_sum)
    except Exception:
        return None


def subgradient_polish(h, n_points, dx, target_sum, n_iters, step0):
    best = h.copy()
    best_val = compute_c5(h)
    for t in range(n_iters):
        beta = 200.0
        val, grad = obj_and_grad(best, beta, dx, target_sum, 0.0)
        step = step0 / (1.0 + 0.01 * t)
        h_new = best - step * grad
        h_new = proj_h(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_val:
            best = h_new
            best_val = c5
    return best, best_val

        beta = np.random.choice([60, 100, 150, 250])
        h_new = slsqp_solved(h0, beta, n_points, dx, target_sum, bounds, 300, 1e-11)
        if h_new is not None:
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
                scale = max(0.3, scale * 0.92)
            else:
                if attempt % 8 == 0:
                    scale = max(0.3, scale * 0.85)

    # Phase 4: Subgradient polishing
    if time_left() > 15:
        best_h, best_c5 = subgradient_polish(best_h, n_points, dx, target_sum, 200, 0.02)

    # Phase 5: Final high-precision SLSQP polish
    for beta in [500, 800, 1200]:
        if time_left() < 10:
            break
        h_new = slsqp_solved(best_h, beta, n_points, dx, target_sum, bounds, 1000, 1e-14)
        if h_new is not None:
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Final subgradient pass
    if time_left() > 5:
        best_h, best_c5 = subgradient_polish(best_h, n_points, dx, target_sum, 100, 0.005)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
