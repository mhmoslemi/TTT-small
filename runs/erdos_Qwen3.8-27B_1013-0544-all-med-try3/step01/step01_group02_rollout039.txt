```python
def two_coord_optimize(h, i, j, n, top_m=30):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = len(c)  # 2*n-1
    
    # Compute d[k] = dc[k]/dh[i] - dc[k]/dh[j]
    # Using the formula: dc[p]/dh[a] = (1-h[a+j_p]) - h[a-j_p] where j_p = p-(n-1)
    n1 = n - 1
    d = np.zeros(m)
    for p in range(m):
        jp = p - n1
        # dc[p]/dh[i]:
        idx1 = i + jp
        val1 = (1 - h[idx1]) if 0 <= idx1 < n else 0.0
        idx2 = i - jp
        val2 = h[idx2] if 0 <= idx2 < n else 0.0
        dc_di = val1 - val2
        # dc[p]/dh[j]:
        idx1 = j + jp
        val1 = (1 - h[idx1]) if 0 <= idx1 < n else 0.0
        idx2 = j - jp
        val2 = h[idx2] if 0 <= idx2 < n else 0.0
        dc_dj = val1 - val2
        d[p] = dc_di - dc_dj
    
    # Now minimize max_k (c[k] + delta * d[k])
    # subject to delta in [lo, hi]
    lo = max(-h[i], h[j] - 1)
    hi = min(1 - h[i], h[j])
    if lo >= hi:
        return 0.0
    
    # Find the top M values of c to focus on
    # Actually, for correctness we should consider all, but for speed we use top M
    # Let's be exact: consider all m values
    # The optimal delta is where the upper envelope is minimized
    
    # Candidate deltas: 0, lo, hi, and all pairwise intersections
    candidates = [0.0, lo, hi]
    # For each pair (k1, k2), find where c[k1]+delta*d[k1] = c[k2]+delta*d[k2]
    # delta = (c[k2]-c[k1]) / (d[k1]-d[k2])
    
    # To keep it fast, only consider pairs where both have high c values
    # Sort by c descending
    order = np.argsort(-c)
    consider = order[:top_m]
    
    for a in range(len(consider)):
        for b in range(a+1, len(consider)):
            ka = consider[a]
            kb = consider[b]
            denom = d[ka] - d[kb]
            if abs(denom) > 1e-15:
                delta = (c[kb] - c[ka]) / denom
                if lo <= delta <= hi:
                    candidates.append(delta)
    
    # Evaluate at all candidates
    best_delta = 0.0
    best_val = c.max()
    for delta in candidates:
        if lo <= delta <= hi:
            val = (c + delta * d).max()
            if val < best_val:
                best_val = val
                best_delta = delta
    
    return best_delta
```

In 'full' mode, the output is indexed from 0 to 2N-2, corresponding to k = -(N-1) to N-1.

So c[0] = a[0] * v[N-1]

output[1] corresponds to k = -(N-2):
c[-(N-2)] = a[0]*v[N-2] + a[1]*v[N-1]

c[0] = Σ_n a[n] * conj(v[n]) = Σ_n a[n] * v[n]

output[2N-2] corresponds to k = N-1:
c[N-1] = a[N-1] * v[0]

```python
import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient_c5(h, n, alpha):
    """Compute gradient of the log-sum-exp surrogate."""
    dx = 2.0 / n
    n1 = n - 1
    m_total = 2 * n - 1
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    # Avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    
    # gradient[h[a]] = sum_p weights[p] * ((1-h[a-m_p]) - h[a+m_p])
    # where m_p = p - n1
    grad = np.zeros(n)
    for p in range(m_total):
        mp = p - n1
        w = weights[p]
        if w < 1e-15:
            continue
        for a in range(n):
            idx1 = a - mp
            idx2 = a + mp
            g = 0.0
            if 0 <= idx1 < n:
                g += (1 - h[idx1])
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[a] += w * g
    return grad

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -700, 700)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def two_coord_step(h, i, j, n, top_m=20):
    """Find optimal delta for 2-coordinate update h[i]+=delta, h[j]-=delta."""
    dx = 2.0 / n
    n1 = n - 1
    m_total = 2 * n - 1
    c = np.correlate(h, 1 - h, mode='full') * dx
    
    # Compute d[p] = dc[p]/dh[i] - dc[p]/dh[j]
    d = np.zeros(m_total)
    for p in range(m_total):
        mp = p - n1
        # dc[p]/dh[i]:
        val = 0.0
        idx = i - mp
        if 0 <= idx < n:
            val += (1 - h[idx])
        idx = i + mp
        if 0 <= idx < n:
            val -= h[idx]
        dc_di = val
        # dc[p]/dh[j]:
        val = 0.0
        idx = j - mp
        if 0 <= idx < n:
            val += (1 - h[idx])
        idx = j + mp
        if 0 <= idx < n:
            val -= h[idx]
        dc_dj = val
        d[p] = dc_di - dc_dj
    
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi - 1e-15:
        return 0.0, c.max()
    
    # Find candidates
    candidates = set()
    candidates.add(0.0)
    candidates.add(max(lo, -1e-10))
    candidates.add(min(hi, 1e10))
    
    # Top M lags
    order = np.argsort(-c)[:top_m]
    for a_idx in range(len(order)):
        for b_idx in range(a_idx + 1, len(order)):
            ka = order[a_idx]
            kb = order[b_idx]
            denom = d[ka] - d[kb]
            if abs(denom) > 1e-12:
                delta = (c[kb] - c[ka]) / denom
                if lo - 1e-12 <= delta <= hi + 1e-12:
                    candidates.add(max(lo, min(hi, delta)))
    
    best_delta = 0.0
    best_val = c.max()
    for delta in candidates:
        delta = max(lo, min(hi, delta))
        val = (c + delta * d).max()
        if val < best_val - 1e-15:
            best_val = val
            best_delta = delta
    
    return best_delta, best_val
```

```python
    s = alpha * (c - m)

                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

    n1 = n - 1

        j = rng.randint(n)

            if 0 <= idx < n: v -= h[idx]

            if 0 <= idx < n: v2 += 1.0 - h[idx]
            idx = j + mp
            if 0 <= idx < n: v2 -= h[idx]
```

```python
    lo_mu = y.min() - 1.0
    hi_mu = y.max() + 1.0
    for _ in range(100):
        mu = 0.5 * (lo_mu + hi_mu)
        h = np.clip(y - mu, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo_mu = mu
        else:
            hi_mu = mu
    mu = 0.5 * (lo_mu + hi_mu)
    return np.clip(y - mu, 0.0, 1.0)

def stochastic_2coord(h, n, rng, time_limit, top_m=20):
    dx = 2.0 / n
    n1 = n - 1
    m_total = 2 * n - 1
    t_start = time.time()
    
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = float(c.max())
    it = 0
    stall = 0
    
    while time.time() - t_start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi - 1e-13:
            continue
        
        # Vectorized d computation
        mps = np.arange(m_total) - n1
        ii1 = i - mps
        ii2 = i + mps
        m_a = (ii1 >= 0) & (ii1 < n)
        m_b = (ii2 >= 0) & (ii2 < n)
        di = np.zeros(m_total)
        if m_a.any():
            di[m_a] = 1.0 - h[ii1[m_a]]
        if m_b.any():
            di[m_b] -= h[ii2[m_b]]
        
        ji1 = j - mps
        ji2 = j + mps
        m_c = (ji1 >= 0) & (ji1 < n)
        m_d = (ji2 >= 0) & (ji2 < n)
        dj = np.zeros(m_total)
        if m_c.any():
            dj[m_c] = 1.0 - h[ji1[m_c]]
        if m_d.any():
            dj[m_d] -= h[ji2[m_d]]
        
        d = di - dj
        cmax = c.max()
        best_delta = 0.0
        best_val = cmax
        
        # Evaluate at boundaries
        for delta in (lo + 1e-10, hi - 1e-10):
            if lo <= delta <= hi:
                val = (c + delta * d).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = delta
        
        # Intersections of top m lines
        order = np.argsort(-c)[:top_m]
        for a in range(len(order)):
            ka = int(order[a])
            cka = c[ka]
            dka = d[ka]
            for b in range(a + 1, len(order)):
                kb = int(order[b])
                denom = dka - d[kb]
                if abs(denom) > 1e-12:
                    delta = (c[kb] - cka) / denom
                    if lo < delta < hi:
                        val = (c + delta * d).max()
                        if val < best_val - 1e-14:
                            best_val = val
                            best_delta = delta
        
        if abs(best_delta) > 1e-13:
            h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
            h[j] = np.clip(h[j] - best_delta, 0.0, 1.0)
            c = np.correlate(h, 1 - h, mode='full') * dx
            cv = c.max()
            if cv < best_c5:
                best_c5 = float(cv)
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        it += 1
    
    return h, best_c5

def projected_gradient(h, n, alpha, time_limit, lr_init=0.01):
    """Projected gradient descent on log-sum-exp surrogate."""
    t_start = time.time()
    n1 = n - 1
    m_total = 2 * n - 1
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    lr = lr_init
    n_epoch = 0
    
    while time.time() - t_start < time_limit:
        # Compute gradient
        s = alpha * (c - c.max())
        s = np.clip(s, -500, 500)
        w = np.exp(s)
        w = w / w.sum()
        
        s1 = np.zeros(n)
        s2 = np.zeros(n)
        for p in range(m_total):
            wm = w[p]
            if wm < 1e-16:
                continue
            m_p = p - n1
            idx1 = np.arange(n) - m_p
            mk = (idx1 >= 0) & (idx1 < n)
            if mk.any():
                s1[mk] += wm * h[idx1[mk]]
            idx2 = np.arange(n) + m_p
            mk2 = (idx2 >= 0) & (idx2 < n)
            if mk2.any():
                s2[mk2] += wm * h[idx2[mk2]]
        grad = 1.0 - s1 - s2
        
        # Gradient step + projection
        y = h - lr * grad
        h_new = proj_onto_feasible(y, n)
        
        c = np.correlate(h_new, 1 - h_new, mode='full') * (2.0 / n)
        cv = c.max()
        if cv < best_c5:
            best_c5 = float(cv)
            best_h = h_new.copy()
        h = h_new
        
        # Decay learning rate
        n_epoch += 1
        if n_epoch % 500 == 0:
            lr *= 0.7
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)
    
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - t0))
    
    # Phase 1: SLSQP continuation (use ~15% of budget)
    for alpha in [
        50, 300, 2000, 10000
    ]:
        if remaining() < 30:
            break
        h = do_slsqp(h, n, alpha, maxiter=2000)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5 = h.copy(), c
    
    # Phase 2: Projected gradient on LSE surrogate (~20% of budget)
    rem = remaining()
    if rem > 40:
        t_gp = min(rem * 0.3, budget_s * 0.2)
        h_pg, c_pg = projected_gradient(best_h, n, alpha=3000, time_limit=t_gp, lr_init=0.005)
        if c_pg < best_c5:
            best_h, best_c5 = h_pg, c_pg
    
    # Phase 3: Stochastic 2-coordinate descent (main phase, ~35% of budget)
    rem = remaining()
    if rem > 30:
        h2c, c2c = stochastic_2coord(best_h, n, rng, time_limit=rem * 0.7, top_m=20)
        if c2c < best_c5:
            best_h, best_c5 = h2c, c2c
    
    # Phase 4: SLSQP re-polish
    if remaining() > 25:
        for alpha in [10000, 50000]:
            if remaining() < 15:
                break
            h = do_slsqp(best_h, n, alpha, maxiter=5000, ftol=1e-20)
            c = compute_c5(h, n)
            if c < best_c5:
                best_h, best_c5 = h.copy(), c
    
    # Phase 5: More 2-coordinate with larger top_m
    rem = remaining()
    if rem > 20:
        h2c, c2c = stochastic_2coord(best_h, n, rng, time_limit=rem * 0.8, top_m=30)
        if c2c < best_c5:
            best_h, best_c5 = h2c, c2c
    
    # Phase 6: Final SLSQP + 2-coordinate
    if remaining() > 15:
        h = do_slsqp(best_h, n, 100000, maxiter=5000, ftol=1e-20)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5 = h.copy(), c
    
    rem = remaining()
    if rem > 10:
        h2c, c2c = stochastic_2coord(best_h, n, rng, time_limit=rem - 3, top_m=25)
        if c2c < best_c5:
            best_h, best_c5 = h2c, c2c
    
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    
    # Try n=160 upscaling if time permits
    rem = remaining()
    if rem > 120:
        n2 = 2 * n
        h2 = np.zeros(n2)
        h2[0::2] = best_h
        h2[1::2] = best_h
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2, n2)
        if c2 < best_c5:
            # Optimize at n2
            h2 = do_slsqp(h2, n2, 5000, maxiter=2000)
            c2 = compute_c5(h2, n2)
            if c2 < best_c5:
                rem2 = remaining()
                if rem2 > 30:
                    h2, c2 = stochastic_2coord(h2, n2, rng, time_limit=rem2 - 10, top_m=20)
                if c2 < best_c5:
                    best_h, best_c5, n = h2, c2, n2
    
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
