import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target):
    n = len(y)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj_and_grad(h, tau, mu=0.0):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    w_sum = exp_x.sum()
    val = (x_max + np.log(w_sum)) / tau
    
    # Gradient of smooth max w.r.t. h
    w = exp_x / w_sum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        wm = w[m]
        if wm < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        # h-side: j = n + d, n in [lo, hi]
        # grad[j] += w[m] * (1 - h[n]) = w[m] * omh[n]
        js = np.arange(lo + d, hi + 1 + d)
        ns = np.arange(lo, hi + 1)
        grad[js] += wm * omh[ns]
        # 1-h side: j = n, n in [lo, hi]
        # grad[j] += w[m] * (-h[n+d])
        js2 = np.arange(lo, hi + 1)
        ns2 = np.arange(lo + d, hi + 1 + d)
        grad[js2] += wm * (-h[ns2])
    
    # Penalty for sum constraint
    if mu > 0:
        s = h.sum()
        val += mu * (s - n / 2.0) ** 2
        grad += 2.0 * mu * (s - n / 2.0)
    
    return val, grad

        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)
    
    n = len(h_init)
    best_h = project_feasible(h_init, n / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Annealing with L-BFGS-B
    taus = [10, 30, 100, 300, 500, 1000, 2000, 3000, 5000, 8000, 12000]

            break
        h_cur = best_h.copy()

                break

            res = minimize(
                lambda x: smooth_max_obj_and_grad(x, tau, mu)[0],
                lambda x: smooth_max_obj_and_grad(x, tau, mu)[1],
                h_cur,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 200, 'ftol': 1e-14}
            )
            h_new = project_feasible(res.x, n / 2.0)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:

                best_c5 = c5_new
            h_cur = h_new

    for _ in range(30):

        hp = project_feasible(hp, n / 2.0)
        for tau in [500, 2000, 5000]:

            mu = n * tau * 0.1
            res = minimize(

                lambda x: smooth_max_obj_and_grad(x, tau, mu)[1],
                hp,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 150, 'ftol': 1e-14}
            )
            h_new = project_feasible(res.x, n / 2.0)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            hp = h_new
    
    # Phase 3: Fine-tune
    for tau in [10000, 20000, 50000]:
        if rem() < 10:

        mu = n * tau * 0.1
        res = minimize(
            lambda x: smooth_max_obj_and_grad(x, tau, mu)[0],
            lambda x: smooth_max_obj_and_grad(x, tau, mu)[1],
            best_h,
            method='L-BFGS-B',

    # Final projection

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n

from scipy.optimize import minimize
from functools import partial

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    w_sum = exp_x.sum()

        d = m - (n - 1)
        wm = w[m]

        hi = min(n - 1, n - 1 - d)
        grad[lo + d: hi + 1 + d] += wm * omh[lo: hi + 1]
        grad[lo: hi + 1] -= wm * h[lo + d: hi + 1 + d]
    return val, grad

    s = x.sum()
