    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

    h = np.clip(x, 0.0, 1.0)
    val = smooth_max(h, tau)
    s = np.sum(h) - target

            h = h_new

    # Phase 3: SLSQP t-variable polishing
    if time_left() > 100:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        h = h_best

    # Phase 4: More L-BFGS-B passes with very low temperatures
    for tau in [0.0003, 0.0001, 0.00005]:
        if time_left() < 50:
            break
        h_new, c5_new = optimize_lbfgs(h, tau, max_iter=3000, pen_w=1e10)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            h = h_new

    # Phase 5: Random restarts from best
    for i in range(12):
        if time_left() < 60:
            break
        scale = 0.03 + 0.015 * (i % 4)
        h_p = h_best + rng.randn(n) * scale
        h_p = project_feasible(h_p, n * 0.5)
        h_new, c5_new = optimize_lbfgs(h_p, 0.01, max_iter=1500, pen_w=1e8)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        h_new, c5_new = optimize_lbfgs(h_best, 0.003, max_iter=1500, pen_w=1e9)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 6: Increase to n=360
    if time_left() > 250:
        h_interp = interp_h(h_best, 360)
        h_cur = h_interp
        c5_cur = compute_c5(h_cur)
        for tau in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            if time_left() < 80:
                break
            h_new, c5_new = optimize_lbfgs(h_cur, tau, max_iter=2000, pen_w=1e9)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
                h_cur = h_new
        if time_left() > 80:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=400)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Phase 7: Final SLSQP polish at best resolution
    if time_left() > 40:
        n_best = len(h_best)
        h_new, c5_new = optimize_slsqp_smooth(h_best, 0.0005, max_iter=400)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        h_new, c5_new = optimize_t_var(h_best, max_iter=300)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    h_best = project_feasible(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
