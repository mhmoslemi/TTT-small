```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_values(h):
    return np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target_sum):
    h = np.array(h, dtype=float)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * len(h),
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

        ones.remove(i)

        zeros.remove(j)
```

```python
                best_n = n0
    
    # Phase 3: Try other n values
    for n_try in [n0 * 2, n0, n0 // 2, n0 * 3 // 2]:
        if elapsed() > budget_s - 100:
            break
        if n_try < 20 or n_try % 2 != 0 or n_try == n0:
            continue
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        
        if n_try > n0:
            x_old = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, n_try)
            h_try = np.interp(x_new, x_old, best_h)
        else:
            ratio = n0 // n_try
            h_try = best_h[:n_try * ratio].reshape(n_try, ratio).mean(axis=1)
        
        h_try = fix_sum(h_try, target_try)
        h_bin_try = make_binary_from_continuous(h_try, int(target_try))
        
        n_sa = min(80000, max(10000, int((budget_s - 100 - elapsed()) * 60)))
        h_bin_try, bm_try = sa_binary(h_bin_try, max_iter=n_sa, t0=2.0, t1=0.03,
                                       time_limit=budget_s - 80 - elapsed(), t_start=elapsed())
        bc5_try = bm_try * dx_try
        if bc5_try < best_c5:
            best_h = h_bin_try.copy()
            best_c5 = bc5_try
            best_n = n_try
        
        if elapsed() < budget_s - 30:
            h_pol = slsqp_optimize(h_bin_try, dx_try, target_try, t=300, maxiter=2000)
            c5_pol = compute_c5(h_pol, dx_try)
            if c5_pol < best_c5:
                best_h = h_pol
                best_c5 = c5_pol
                best_n = n_try
    
    # Phase 4: Final SLSQP polishing with high t
    if elapsed() < budget_s - 10:
        dx_best = 2.0 / best_n
        target_best = best_n / 2.0
        for t in [100, 500, 2000, 5000]:
            if elapsed() > budget_s - 5:
                break
            h_new = slsqp_optimize(best_h, dx_best, target_best, t=t, maxiter=5000)
            c5_new = compute_c5(h_new, dx_best)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
```
