import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n), corr

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(corr, alpha, h, n):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        wval = w[idx]
        if k == 0:
            grad += wval * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wval * (1.0 - h[:n - k])
            grad[:n - k] -= wval * h[k:]
        else:
            p = -k
            grad[:n - p] += wval * (1.0 - h[p:])
            grad[p:] -= wval * h[:n - p]
    return grad * (2.0 / n)

def single_lag_grad(h, k, n):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        p = -k
        grad[:n - p] += (1.0 - h[p:])
        grad[p:] -= h[:n - p]
    return grad * (2.0 / n)

def momentum_opt(h0, target_sum, deadline, rng,
                 alpha_init=100, beta=0.75, lr_init=0.003,
                 max_alpha=500, subgrad_prob=0.15,
                 accept_worse_prob=0.0, temp=0.001):

    h_prev = h0.copy()

    best_h = h0.copy()

    alpha = alpha_init
    lr = lr_init
    no_improve = 0
    stall_count = 0
    iter_count = 0

    while time.time() < deadline:
        c5_cur, corr_cur = compute_c5(h)

        if rng.random() < subgrad_prob:
            threshold = best_c5 / (2.0 / n) - 0.005 * n / 2.0
            active_lags = np.where(corr_cur > threshold)[0]
            if len(active_lags) == 0:
                active_lags = np.array([np.argmax(corr_cur)])
            k_idx = active_lags[rng.randint(len(active_lags))]
            k = k_idx - (n - 1)
            grad = single_lag_grad(h, k, n)
        else:
            grad = smooth_max_grad(corr_cur, alpha, h, n)

        y = h + beta * (h - h_prev)
        v = beta * v - lr * grad
        h_new = project_box_sum(y + v, target_sum)
        c5_new, _ = compute_c5(h_new)

                if stall_count > 12:

                continue

        h_prev = h
        h = h_new
        iter_count += 1

    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h_old)

def make_binary_init(n, rng):
    """Create a random binary-like init: n/2 ones and n/2 zeros, shuffled."""
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # Add small noise to break symmetry
    h += rng.randn(n) * 0.01
    return project_box_sum(h, n / 2.0)

def make_block_init(n, rng):
    """Create a block-structured init with random block widths."""
    h = np.zeros(n)
    pos = 0
    while pos < n:
        width = rng.randint(3, max(4, n // 8))
        fill = rng.random() < 0.5
        if fill:
            h[pos:pos + width] = 1.0
        pos += width + rng.randint(0, 3)
    h += rng.randn(n) * 0.02
    return project_box_sum(h, n / 2.0)
