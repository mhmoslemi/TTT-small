            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, bottleneck_k=0, bottleneck_mult=1.0):
    c = get_c_values(h, n)
    if bottleneck_k > 0 and bottleneck_mult > 1.0:
        top_idx = np.argsort(c)[::-1][:bottleneck_k]
        c = c.copy()
        c[top_idx] *= bottleneck_mult
    scaled = alpha * c
    mx = scaled.max()
    shifted = scaled - mx
    if np.max(shifted) > 700:
        val = mx + np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted))))
    else:
        val = np.log(np.sum(np.exp(scaled)))
    return float(val / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def compute_grads_c(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    nc = len(c)
    grad = np.zeros((nc, n))
    for k in range(nc):
        j = k - (n - 1)
        for i in range(n):
            ip = i + j
            if 0 <= ip < n:
                val = (1.0 - h[ip])
                for i2 in range(n):
                    if i2 + j == i:
                        val -= h[i2]
                grad[k, i] = dx * val
    return c, grad

def compute_grads_fast(h, n):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    nc = 2 * n - 1
    grad = np.zeros((nc, n))
    for k in range(nc):
        shift = k - (n - 1)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                grad[k, i] += dx * one_minus_h[j]
                m = j - shift
                if 0 <= m < n:
                    grad[k, i] -= dx * h[m]
    return np.correlate(h, one_minus_h, mode='full') * dx, grad

def linprog_mm(h, n, top_k=10):
    c_vals, grads = compute_grads_fast(h, n)
    m = c_vals.max()
    active = np.argsort(c_vals)[::-1][:top_k]
    A_rows = []
    b_rows = []
    for k in active:
        A_rows.append(grads[k, :])
        b_rows.append(m - c_vals[k] + 1e-10)
    A_mm = np.vstack(A_rows)
    b_vec = np.array(b_rows)
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    res = linprog(-np.zeros(n), A_ub=A_mm, b_ub=b_vec, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, 1)] * n, method='highs')
    if res.status == 0:
        delta = res.x
        return h + delta
    return h

def coord_descent(h, n, max_sweeps=50):
    h = h.copy()
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        improved = False
        perm = np.random.permutation(n - 1)
        for i in perm:
            best_val = h[i]
            best_c5 = compute_c5(h, n)
            candidate_vals = np.linspace(0, 1, 20)
            for v in candidate_vals:
                h[i] = v
                diff = h.sum() - n / 2.0
                if abs(diff) < 1e-12:
                    c = compute_c5(h, n)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_val = v
                h[i] = best_val
            if best_val != h[i]:
                h[i] = best_val
                improved = True
        if not improved:
            break
    return project_feasible(h, n)

def binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def binary_local_search(h, n, rng, max_rounds=200):
    best_c5 = compute_c5(h, n)
    h = h.copy()
    for _ in range(max_rounds):
        improved = False
        for _ in range(n * 3):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                continue
            h_orig = h.copy()
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            c = compute_c5(h, n)
            if c < best_c5 - 1e-12:
                best_c5 = c
                improved = True
                break
            else:
                h[:] = h_orig
        if not improved:
            break
    return h, best_c5

def slsqp_refine(h, n, alphas, time_limit, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    for alpha in alphas:
        if time_limit - (time.time() - t0_global) < 20:
            break
        for bk, bmult in [(0, 1.0), (5, 1.1), (10, 1.05), (3, 1.2)]:
            if time_limit - (time.time() - t0_global) < 15:
                break
            try:
                res = minimize(smooth_max_obj, best_h, args=(n, alpha, bk, bmult),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-18})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass
    return best_h, best_c5

t0_global = 0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0_global = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 80

    def remaining():
        return budget_s - (time.time() - t0_global)

    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 80
        h_init = np.full(n_init, 0.5)

    # Phase 1: Continuation SLSQP from initial
    h_cur = project_feasible(h_init, n_init)
    c_cur = compute_c5(h_cur, n_init)
    alphas = [20, 50, 100, 300, 1000, 3000, 10000]
    h_cur, c_cur = slsqp_refine(h_cur, n_init, alphas, budget_s, rng)
    best_h, best_c5, best_n = h_cur.copy(), c_cur, n_init

    # Phase 2: Try binary initializations
    if remaining() > 200:
        for attempt in range(5):
            if remaining() < 60:
                break
            h_bin = binary_init(n_init, rng)
            h_bin, c_bin = binary_local_search(h_bin, n_init, rng, max_rounds=100)
            if c_bin < best_c5:
                # Refine from this binary start
                h_ref, c_ref = slsqp_refine(h_bin, n_init, [300, 1000, 3000, 10000], budget_s, rng)
                if c_ref < best_c5:
                    best_h, best_c5 = h_ref, c_ref

    # Phase 3: Perturbation + SLSQP
    pert_count = 0
    while remaining() > 80 and pert_count < 40:
        pert_count += 1
        scale = rng.choice([0.01, 0.02, 0.03, 0.05, 0.1, 0.15, 0.25])
        h_pert = best_h + rng.normal(0, scale, size=n_init)
        h_pert = project_feasible(h_pert, n_init)
        h_try, c_try = slsqp_refine(h_pert, n_init, [500, 2000, 5000], budget_s, rng)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 4: Linearized min-max (LP-based)
    if remaining() > 30:
        h_lp = best_h.copy()
        for it in range(200):
            if remaining() < 10:
                break
            h_new = linprog_mm(h_lp, n_init, top_k=15)
            h_new = project_feasible(h_new, n_init)
            c_new = compute_c5(h_new, n_init)
            if c_new < compute_c5(h_lp, n_init) - 1e-12:
                h_lp = h_new
            else:
                # Try smaller step
                h_half = project_feasible(h_lp + 0.5 * (h_new - h_lp), n_init)
                if compute_c5(h_half, n_init) < compute_c5(h_lp, n_init):
                    h_lp = h_half
                else:
                    break
        c_lp = compute_c5(h_lp, n_init)
        if c_lp < best_c5:
            best_h, best_c5 = h_lp, c_lp

    # Phase 5: Fine coordinate descent
    if remaining() > 20:
        h_cd = coord_descent(best_h, n_init, max_sweeps=30)
        c_cd = compute_c5(h_cd, n_init)
        if c_cd < best_c5:
            best_h, best_c5 = h_cd, c_cd

    # Phase 6: Final SLSQP polish with very high alpha
    if remaining() > 10:
        alphas_final = [5000, 10000, 20000]
        best_h, best_c5 = slsqp_refine(best_h, n_init, alphas_final, budget_s, rng)

    # Final feasibility
    best_h = project_feasible(best_h, n_init)
    best_c5 = compute_c5(best_h, n_init)

    return best_h, best_c5, n_init
