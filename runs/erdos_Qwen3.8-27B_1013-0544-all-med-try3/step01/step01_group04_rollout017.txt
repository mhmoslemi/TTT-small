        h = np.clip(h, 0.0, 1.0)

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h_start, n, dx, target_sum, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(
            lambda x: compute_c5(x, dx),
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)
