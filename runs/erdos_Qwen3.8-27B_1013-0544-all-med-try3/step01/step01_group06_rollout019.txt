import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        grad += w[idx] * grad_corr_single(h, k)
    return grad * (2.0 / n)

def project_box_simplex(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def interp_resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float) + 0.5
    x_new = (np.arange(n_new, dtype=float) + 0.5) * n_old / n_new
    x_new = np.clip(x_new, 0.0, n_old - 1)
    return np.interp(x_new, x_old, h_old)

def pgd_optimize(h0, target_sum, deadline, rng, alpha_seq=None, lr_init=0.008):
    n = len(h0)
    h = project_box_simplex(h0.astype(float), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    if alpha_seq is None:
        alpha_seq = [20, 40, 80, 160, 320, 600]
    for alpha in alpha_seq:
        if time.time() > deadline:
            break
        lr = lr_init
        patience = 0
        stall = 0
        while time.time() < deadline:
            grad = compute_grad(h, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            h_new = h - lr * grad
            h_new = project_box_simplex(h_new, target_sum)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5 - 1e-15:
                h = h_new
                best_c5 = new_c5
                patience = 0
                if lr < 0.06:
                    lr *= 1.04
            else:
                patience += 1
                lr *= 0.90
                if patience > 45:

                    size = 0.012 + 0.008 * stall

                    if pc5 < best_c5:
