    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_corr_single(h, k)
    return grad * (2.0 / n)

def qr_sequence(p):
    """Build binary seq of length p-1 from quadratic residues mod p."""
    n = p - 1
    h = np.zeros(n)
    for i in range(1, p):
        if pow(i, (p - 1) // 2, p) == 1:
            h[i - 1] = 1.0
    return h

def local_search_binary(h, rng, deadline, max_evals=20000):
    n = len(h)

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    no_improve = 0

def interpolate(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    if n_new > n_old:
        x0 = np.linspace(0, 2, n_old, endpoint=False)
        x1 = np.linspace(0, 2, n_new, endpoint=False)
        h_ext = np.append(h_old, h_old[:max(1, n_old//10)])
        x_ext = np.append(x0, np.linspace(2.0, 2.0 + 2.0/n_old, max(1, n_old//10)))
        return np.interp(x1, x_ext, h_ext)
    else:
        # Downsample by averaging
        ratio = n_old / n_new
        idx = (np.arange(n_new) * ratio).astype(int)
        idx = np.minimum(idx, n_old - 1)
        return h_old[idx]

- When i is in the 1-block and i+k is in the 0-block: this happens for exactly k positions per period (the last k positions of the 1-block).

- When i is in the 1-block and i+k is also in the 1-block: h[i+k]=1, contributes 0.

For k = T/2 - 1: corr ≈ (T/2-1)*n/T ≈ n/2.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n-1)
        grad += w[idx] * grad_corr_single(h, k)
    return grad * (2.0/n)

def grad_optimize(h, target, deadline, rng, alpha=50, sa_temp=0.0):
    n = len(h)
    best_h = project_box_sum(h.astype(float).copy(), target)
    best_c5 = compute_c5(best_h)
    lr = 0.003
    patience = 0
    stalls = 0
    
    while time.time() < deadline:
        g = compute_grad(best_h, alpha)
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(5):
            h_new = project_box_sum(best_h - step * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
                patience = 0
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try SA acceptance
            h_new = project_box_sum(best_h - lr * g, target)
            c5_new = compute_c5(h_new)
            if sa_temp > 0 and rng.rand() < np.exp(-(c5_new - best_c5) / sa_temp):
                best_h = h_new
                best_c5 = c5_new  # temporarily worse
                patience = 0
                sa_temp *= 0.95
            else:
                patience += 1
                if patience > 30:

                        best_c5 = c5_p

                    lr = 0.003
                    patience = 0

                    if stalls > 10:
                        break
        else:
            if lr < 0.01:
                lr = min(lr * 1.02, 0.02)
    
    return best_h, best_c5
