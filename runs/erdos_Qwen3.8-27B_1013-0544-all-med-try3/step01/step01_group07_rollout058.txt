def pair_optimize(h, j, jp, dx):
    """Optimize h[j] and h[jp] together, maintaining their sum."""
    n = len(h)
    s = h[j] + h[jp]
    
    def f_of_x(x):
        # h[j] = x, h[jp] = s - x
        if x < 0 or x > 1 or s - x < 0 or s - x > 1:
            return np.inf
        h_temp = h.copy()
        h_temp[j] = x
        h_temp[jp] = s - x
        c = compute_corr(h_temp)
        return np.max(c) * dx
    
    # x ranges from max(0, s-1) to min(1, s)
    x_lo = max(0.0, s - 1.0)
    x_hi = min(1.0, s)
    
    if x_hi - x_lo < 1e-12:
        return h[j]
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(f_of_x, bounds=(x_lo, x_hi), method='bounded', options={'xatol': 1e-10})
    if res.success and res.fun < f_of_x(h[j]):
        h[j] = res.x
        h[jp] = s - res.x
    return h

    return np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    
    # Water-filling: find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    def proj_sum(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    
    # proj_sum is decreasing in lam
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if proj_sum(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def pair_optimize(h, j, jp, dx):
    """Optimize pair (j, jp) maintaining their sum. Returns new h[j]."""
    s = h[j] + h[jp]
    x_lo = max(0.0, s - 1.0)
    x_hi = min(1.0, s)
    if x_hi - x_lo < 1e-12:
        return h[j]
    
    def f_of_x(x):
        old_j, old_jp = h[j], h[jp]
        h[j] = x
        h[jp] = s - x
        c = compute_corr(h)
        val = np.max(c)
        h[j] = old_j
        h[jp] = old_jp
        return val
    
    current_val = f_of_x(h[j])
    res = minimize_scalar(f_of_x, bounds=(x_lo, x_hi), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 100})
    
    if res.success and res.fun < current_val - 1e-14:
        h[j] = res.x
        h[jp] = s - res.x
    return h[j]

    n = len(h)
    
    def obj(x):
        corr = compute_corr(x)
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        return (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx
    
    try:
        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = fix_sum(res.x, target_sum)
        return h_new
    except Exception:
        return h

import numpy as np

from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    def _proj(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if _proj(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pair_optimize(h, j, jp):
    s = h[j] + h[jp]
    x_lo = max(0.0, s - 1.0)
    x_hi = min(1.0, s)
    if x_hi - x_lo < 1e-14:
        return 0.0
    old_j, old_jp = h[j], h[jp]
    def f_of_x(x):
        h[j] = x
        h[jp] = s - x
        return np.max(compute_corr(h))
    current_max = f_of_x(old_j)
    h[j] = old_j
    h[jp] = old_jp
    try:
        res = minimize_scalar(f_of_x, bounds=(x_lo, x_hi), method='bounded',
                              options={'xatol': 1e-13, 'maxiter': 80})
        if res.success and res.fun < current_max - 1e-14:
            h[j] = res.x
            h[jp] = s - res.x
            return current_max - res.fun
    except Exception:
        pass
    return 0.0

def smooth_max_obj(x, t, dx):
    corr = compute_corr(x)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx
