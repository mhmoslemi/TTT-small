def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(x, target_sum):
    lo = float(x.min()) - 1.5
    hi = float(x.max()) + 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, L, n):
    omh = 1.0 - h
    grad = np.zeros(n)
    if L >= 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def smooth_max_grad(h, beta, n, w_thresh=1e-6):
    corr = compute_corr(h)

        w = exp_s[idx] / total
        if w < w_thresh:
            continue
        L = idx - n_half
        grad += w * grad_lag(h, L, n)
    return grad
