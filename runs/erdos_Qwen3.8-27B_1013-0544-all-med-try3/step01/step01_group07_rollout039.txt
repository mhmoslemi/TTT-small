from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(y, c):
    """Project y onto {x : 0 <= x <= 1, sum(x) = c}"""
    n = len(y)
    # Binary search for the threshold
    lo = y.min() - 2
    hi = y.max() + 2
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def smooth_max_and_grad(h, t):
    """Compute smooth max value and gradient of max_k f_k(h)."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    log_sum = max_c + (1.0 / t) * np.log(np.sum(exp_vals))
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient computation using padded h
    n_half = n - 1
    s = np.arange(corr.size) - n_half  # shifts
    j = np.arange(n)
    
    h_pad = np.zeros(corr.size + n - 1)

    grad = (p[None, :] * ((1 - h_minus) - h_plus)).sum(axis=1)

def subgradient_descent(h, dx, target_sum, t, n_iters, lr_init, momentum=0.9):

import numpy as np
import time
from scipy.optimize import minimize

_START_TIME = 0.0
_BUDGET_S = 1100.0

def elapsed():
    return time.time() - _START_TIME

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, c):
    lo, hi = float(y.min()) - 2.0, float(y.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0, 1)) > c:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0, 1)

def smax_obj(x, t, dx):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    sh = np.clip(t * (corr - mc), -700, 700)
    return (mc + np.log(np.sum(np.exp(sh))) / t) * dx

def smax_jac(x, t, dx):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    sh = np.clip(t * (corr - mc), -700, 700)
    ev = np.exp(sh)
    p = ev / ev.sum()
    nh = n - 1
    s = np.arange(corr.size) - nh
    j = np.arange(n)
    pad_len = corr.size + n - 1
    pad = np.zeros(pad_len)
    pad[nh:nh + n] = x
    im = np.clip(j[:, None] - s[None, :] + nh, 0, pad_len - 1)
    ip = np.clip(j[:, None] + s[None, :] + nh, 0, pad_len - 1)
    g = (p[None, :] * ((1.0 - pad[im]) - pad[ip])).sum(axis=1)
    return g * dx

def con_eq(x, target):
    return np.sum(x) - target

def slsqp(h, dx, ts, t, mi):
    try:
        r = minimize(smax_obj, h.copy(), jac=smax_jac, args=(t, dx),
                     method='SLSQP', bounds=[(0.0, 1.0)] * len(h),
                     constraints=[{'type': 'eq', 'fun': con_eq, 'args': (ts,)}],
                     options={'maxiter': mi, 'ftol': 1e-12})
        return project_feasible(r.x, ts)
    except Exception:
        return h

def sgrad_descent(h, dx, ts, t, niter, lr0, mom):
    x = h.copy()
    v = np.zeros(len(h))
    for it in range(niter):
        grad = smax_jac(x, t, 1.0)
        v = mom * v + (1.0 - mom) * grad
        lr = lr0 / (1.0 + 0.05 * it)
        x = project_feasible(x - lr * v, ts)
    return x

def block_obj(x, h_full, idx, t, dx):
    hf = np.array(h_full, dtype=float)
    hf[idx] = x
    return smax_obj(hf, t, dx)

def block_desc(h, dx, ts, bs, nr, t, seed):
    n = len(h)
    best = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    for _ in range(nr):
        i2 = rng.choice(n, min(bs, n), replace=False)
        fs = ts - (np.sum(h) - np.sum(h[i2]))
        try:
            r = minimize(block_obj, h[i2].copy(), args=(h, i2, t, dx),
                         method='SLSQP', bounds=[(0.0, 1.0)] * len(i2),
                         constraints=[{'type': 'eq', 'fun': con_eq, 'args': (fs,)}],
                         options={'maxiter': 80, 'ftol': 1e-10})
            hn = h.copy()
            hn[i2] = np.clip(r.x, 0, 1)
            c5 = compute_c5(hn, dx)
            if c5 < best:
                h, best = hn, c5
        except Exception:
            pass
    return h, best

def make_structured(n, pattern_len, ones_per_period, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    per_period = n // pattern_len
    offsets = rng.randint(0, pattern_len, size=min(n // per_period + 1, n))
    for i in range(n):
        block = i // per_period
        pos_in_block = (i + offsets[min(block, len(offsets)-1)]) % pattern_len
        if pos_in_block < ones_per_period:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Subgradient descent with increasing t (explore landscape)
    for t in [2, 3, 5, 8, 12, 20, 35]:
        if elapsed() > budget_s - 320:
            break
        h = sgrad_descent(best_h, dx, ts, t, 4000, 0.8 / np.sqrt(t), 0.95)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: SLSQP with analytic gradient, increasing t
    for t in [10, 20, 40, 80, 150, 300, 600]:
        if elapsed() > budget_s - 280:
            break
        h = slsqp(best_h, dx, ts, t, 1200)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Block coordinate descent with larger blocks
    if elapsed() < budget_s - 220:
        h, c5 = block_desc(best_h, dx, ts, 10, 120, 40, 777)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    if elapsed() < budget_s - 180:
        h, c5 = block_desc(best_h, dx, ts, 6, 80, 100, 999)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Multi-start with structured and random perturbations
    nt = 0
    while elapsed() < budget_s - 120 and nt < 60:
        nt += 1
        if nt % 10 == 1:
            # Structured restart
            h_trial = make_structured(n, np.random.choice([4, 5, 6, 7, 8]),
                                       np.random.choice([1, 2, 3]), seed + nt)
        else:
            # Perturbation of current best
            h_trial = best_h.copy()
            np_ = np.random.randint(3, 10)
            ix = np.random.choice(n, np_, replace=False)
            h_trial[ix] += np.random.uniform(-0.4, 0.4, np_)
            h_trial = project_feasible(h_trial, ts)

        h = slsqp(h_trial, dx, ts, 20, 400)
        h = slsqp(h, dx, ts, 80, 800)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Final high-precision polish
    if elapsed() < budget_s - 10:
        hf = slsqp(best_h, dx, ts, 800, 5000)
        c5f = compute_c5(hf, dx)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f
        hf = slsqp(best_h, dx, ts, 1500, 3000)
        c5f = compute_c5(hf, dx)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f

    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
