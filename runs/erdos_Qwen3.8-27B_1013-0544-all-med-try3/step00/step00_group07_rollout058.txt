def project_box_hyperplane(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    # The projection is: find λ such that sum(clamp(h - λ, 0, 1)) = target_sum
    # where clamp(x, 0, 1) = min(max(x, 0), 1)
    
    # Binary search on λ
    lo = h.min() - 1  # h - λ >= 0 for all means λ <= h.min()
    hi = h.max() + 1  # h - λ <= 1 for all means λ >= h.max() - 1
    
    def sum_clamped(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    # We need sum_clamped(λ) = target_sum
    # sum_clamped is decreasing in λ
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_clamped(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h - (lo + hi) / 2, 0, 1)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_grad(h):
    """Objective and gradient for minimize max_k correlate(h, 1-h)"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = float(np.max(c))
    tol = max(1e-10, 1e-9 * abs(max_val))

    grad = np.zeros(n)
    for k in active:
        lag = int(k) - n + 1
        # Term 1: (1-h[m+lag]) for valid m
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1.0 - h[lo1+lag:hi1+1+lag])
        # Term 2: -h[m-lag] for valid m
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:

        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try * 0.5)
        h_new = _slsqp(h_interp, n_try, 600)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
            n = n_try
            ts = n_try * 0.5

    # Phase 6: Final SA + SLSQP
    n = len(best_h)
    ts = n * 0.5
    if elapsed() < budget_s - 30:
        h_new, c_new = _sa(best_h, n, ts, iters=4000, T0=0.02, T1=0.002)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
        if elapsed() < budget_s - 15:
            h_new = _slsqp(best_h, n, 1000)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Phase 7: Final polish
    for i in range(15):
        if elapsed() > budget_s - 5:
            break
        h_pert = best_h + np.random.normal(0, 0.005, n)
        h_pert = proj(h_pert, ts)
        h_new = _slsqp(h_pert, n, 500)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Ensure final feasibility
    best_h = proj(best_h, ts)
    best_c = compute_c5(best_h)

    return best_h, best_c, len(best_h)
