```python
import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum} using bisection."""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_c5(h, dx):
    comp = np.correlate(h, 1.0 - h, mode="full")
    return float(comp.max() * dx)

def smooth_max_grad_step(h, T, alpha, dx, target_sum):
    """One step of projected gradient on smooth max."""
    n = len(h)
    n_comp = 2 * n - 1
    comp = np.correlate(h, 1.0 - h, mode="full")
    c = comp * dx
    
    # Softmax weights (stable)
    c_shifted = c - c.max()
    exp_c = np.exp(T * c_shifted)
    w = exp_c / exp_c.sum()
    
    # Gradient: sum_m w[m] * d(c[m])/d(h)
    # d(c[m])/d(h[j]) = dx * d(full[m])/d(h[j])
    # d(full[m])/d(h[j]) computed via the loop formula
    grad = np.zeros(n)
    for m in range(n_comp):
        if w[m] < 1e-10:
            continue
        lo = max(0, n - 1 - m)
        hi = min(n - 1, 2 * n - 2 - m)
        for nn in range(lo, hi + 1):
            j1 = nn + m - n + 1
            grad[j1] += w[m] * dx * (1.0 - h[nn])
            grad[nn] -= w[m] * dx * h[nn + m - n + 1]
    
    h_new = h - alpha * grad
    h_new = project_simplex(h_new, target_sum)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_c5(h, dx):
    comp = np.correlate(h, 1.0 - h, mode="full")
    return float(comp.max() * dx)

def smooth_grad_step(h, T, alpha, dx, target_sum):
    n = len(h)
    n_comp = 2 * n - 1
    comp = np.correlate(h, 1.0 - h, mode="full")
    c = comp * dx
    c_shifted = c - c.max()
    exp_c = np.exp(T * c_shifted)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for m in range(n_comp):
        if w[m] < 1e-12:
            continue
        lo = max(0, n - 1 - m)
        hi = min(n - 1, 2 * n - 2 - m)
        s_lo = max(0, m - n + 1)
        s_hi = min(n - 1, m)
        grad[s_lo:s_hi+1] += w[m] * dx * (1.0 - h[lo:hi+1])
        grad[lo:hi+1] -= w[m] * dx * h[s_lo:s_hi+1]
    h_new = h - alpha * grad
    return project_simplex(h_new, target_sum)

def optimize_gradient(h_init, dx, target_sum, T_init, alpha_init, max_iters, deadline):
    n = len(h_init)
    h = h_init.copy()
    T = T_init
    alpha = alpha_init

def slsqp_polish(h_init, dx, target_sum, n, max_iter=80):

        return float(comp.max() * dx)
    cons = [{'type': 'eq', 'fun': lambda h_in: np.sum(h_in) - target_sum,
             'jac': lambda h_in: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

        result = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if result.fun < compute_c5(h_init, dx):

            return h_res, compute_c5(h_res, dx)
    except Exception:
        pass
    return h_init, compute_c5(h_init, dx)
```
