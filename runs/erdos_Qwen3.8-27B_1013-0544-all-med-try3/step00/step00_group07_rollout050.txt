    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    """Project h onto feasible set: 0<=h<=1, sum(h)=n/2"""
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Iteratively fix the sum
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum; add to elements < 1
            room = 1.0 - h
            total_room = room.sum()
            if total_room < 1e-12:
                break
            h += diff * room / total_room
        else:
            # Need to decrease sum; subtract from elements > 0
            room = h
            total_room = room.sum()
            if total_room < 1e-12:

            h += diff * room / total_room

np.correlate(h, 1-h, 'full') at lag k gives:
∑_i h[i] * (1-h)[i-k+n-1] (for valid indices)

This is NOT the same as ∑_i h[i] - ∑_i h[i]*h[i-k+n-1] in general, because the (1-h) term is (1-h[j]) for specific j, not 1 - h[i].

Wait, actually:
∑_i h[i] * (1-h[i-k+n-1]) = ∑_i h[i] - ∑_i h[i]*h[i-k+n-1]

But the ranges of i are different! The first sum ∑_i h[i] is over all i where both i and i-k+n-1 are valid. The second sum is the same.
