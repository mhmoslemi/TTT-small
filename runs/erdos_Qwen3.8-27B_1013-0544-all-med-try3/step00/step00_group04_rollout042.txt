            hi = mid

Actually, the key issue is: the LP linearizes the objective around the starting point h. But the true objective is quadratic (bilinear). The LP solution might not be optimal for the true objective.

A better approach: iterate the LP. After getting h_new from the LP, re-solve the LP around h_new. Since the feasible region is convex and the objective is the max of quadratics (which is convex), iterating should converge to the optimal solution.

Wait, is the objective convex? The objective is max_k f_k(h) where f_k(h) = sum_i h[i]*(1-h[i+k-n+1]).

f_k is a quadratic form: f_k(h) = sum_i h[i] - sum_i h[i]*h[i+k-n+1]

The Hessian of f_k has -1s on the "diagonal" corresponding to the pairs (i, i+k-n+1). So f_k is not convex in general (it's indefinite).

So the max of these non-convex functions is not necessarily convex. The LP is just a local approximation.

However, in practice, the LP should still work well because:
1. Starting from a binary point, the perturbation is small
2. The LP gives a good direction to move
3. Iterating the LP (solving, evaluating, re-solving) should converge

import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(np.abs(h - 0.5))
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def solve_lp(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = -corr.astype(float).copy()
    
    for k in range(n_corr):
        A_ub[k, 0] = -1.0
        for m in range(n):
            coeff = 0.0
            idx1 = m - k + n - 1
            if 0 <= idx1 < n:
                coeff -= h[idx1]
            idx2 = m + k - n + 1
            if 0 <= idx2 < n:
                coeff += (1.0 - h[m])
            A_ub[k, m + 1] = coeff
    
    bounds = [(0.0, None)]
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return np.clip(h + result.x[1:], 0.0, 1.0)
    return h.copy()

    for _ in range(max_iters):

        h_new = project_feasible(h_new, n / 2.0)
        new_score = compute_c5(h_new, dx)
        if new_score >= current_score - 1e-12:
            break
        h = h_new
        current_score = new_score
    return h, current_score

def local_search_binary(h, dx, max_iters, rng):
    n = len(h)
    current = h.copy().astype(float)
    current_score = compute_c5(current, dx)
    best = current.copy()
    best_score = current_score
    ones = list(np.where(current >= 0.5)[0])
    zeros = list(np.where(current < 0.5)[0])
    stall = 0
    for _ in range(max_iters):
        if not ones or not zeros:
            break
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i, j = ones[i_idx], zeros[j_idx]
        current[i] = 0.0
        current[j] = 1.0
        new_score = compute_c5(current, dx)
        if new_score <= current_score:
            current_score = new_score
            ones[i_idx], zeros[j_idx] = j, i
            stall = 0
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            current[i] = 1.0
            current[j] = 0.0
            stall += 1
            if stall > 30000:
                break
    return best, best_score
