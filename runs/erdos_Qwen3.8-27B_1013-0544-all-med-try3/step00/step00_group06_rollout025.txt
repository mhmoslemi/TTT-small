def solve_1d(A, B, x_min=0.0, x_max=1.0):
    """Minimize max_k (A[k] + B[k]*x) over x in [x_min, x_max]."""
    # Ternary search
    for _ in range(80):
        m1 = x_min + (x_max - x_min) / 3
        m2 = x_max - (x_max - x_min) / 3
        f1 = np.max(A + B * m1)
        f2 = np.max(A + B * m2)
        if f1 < f2:
            x_max = m2
        else:
            x_min = m1
    x_opt = (x_min + x_max) / 2
    f_opt = np.max(A + B * x_opt)
    return x_opt, f_opt

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def fix_sum_inplace(h, target):
    """Adjust h in-place to have sum = target, maintaining [0,1] bounds."""
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Scale
    if s > 1e-12:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # If still not matching, adjust individual elements
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to increase; find elements < 1
            mask = h < 1.0
            if np.sum(mask) > 0:
                per = diff / np.sum(mask)
                h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            mask = h > 0.0
            if np.sum(mask) > 0:
                per = diff / np.sum(mask)
                h[mask] = np.maximum(h[mask] + per, 0.0)
    return h

def slsqp_h(h, n, tau, maxiter=300):
    """One round of SLSQP optimization."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        if tau < 1e-8:
            return float(np.max(corr))
        return float(tau * logsumexp(corr / tau))
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    res = minimize(
        obj, h, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return np.clip(res.x, 0.0, 1.0)

def local_search_sa(h, n, rng, time_limit):
    """Simulated annealing local search."""
    t0 = time.time()
    dx = 2.0 / n
    c5_cur = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    temp = 0.005
    no_improve = 0
    
    while time.time() - t0 < time_limit:
        # Move: transfer from j to k
        j = rng.randint(n)
        eps = rng.uniform(0.01, 0.05)
        # Try to find a valid k
        for _ in range(10):
            k = rng.randint(n)
            if k == j:
                continue
            h_new = h.copy()
            h_new[j] -= eps
            h_new[k] += eps
            if 0.0 <= h_new[j] <= 1.0 and 0.0 <= h_new[k] <= 1.0:
                break
        else:
            continue
        
        c5_new = compute_c5(h_new, n)
        delta = c5_new - c5_cur
        
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(temp, 1e-12)):
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Cool down
        if no_improve > 5000:
            temp *= 0.9
            no_improve = 0
        else:
            temp = max(temp * 0.9999, 0.0005)
    
    return best_h, best_c5

def interp_to_n(h, n_old, n_new):
    """Interpolate h from n_old to n_new points."""
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    return np.interp(x_new, x_old, h, period=2.0)

    best_c5 = compute_c5(h0, n)
