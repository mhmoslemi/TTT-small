import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    return float(max_c + np.log(np.sum(np.exp(T * (corr - max_c)))) / T)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def exhaustive_swap_search(h, time_limit, start_time, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    while time.time() - start_time < time_limit:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        for i in ones:
            for j in zeros:
                old_i, old_j = h[i], h[j]
                h[i], h[j] = old_j, old_i
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = old_i, old_j
            if time.time() - start_time > time_limit:
                return h, best_c5
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
            # Random kick to escape
            for _ in range(200):
                i, j = rng.choice(n, 2, replace=False)
                if h[i] > 0.5 and h[j] < 0.5:
                    old_i, old_j = h[i], h[j]
                    h[i], h[j] = old_j, old_i
                    c5 = compute_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        improved = True
                        no_improve = 0
                    else:
                        h[i], h[j] = old_i, old_j
        else:
            no_improve = 0
    
    return h, best_c5

def random_restart_search(h_best, best_c5, n, time_limit, start_time, rng):
    target = n / 2.0
    for _ in range(200):
        if time.time() - start_time > time_limit:
            break
        # Create random near-binary solution
        h_rand = np.zeros(n)
        num_ones = int(target)
        frac = target - num_ones
        idx = rng.choice(n, num_ones, replace=False)
        h_rand[idx] = 1.0
        if frac > 1e-10:
            remaining = np.setdiff1d(np.arange(n), idx)
            h_rand[rng.choice(remaining)] = frac
        c5 = compute_c5(h_rand)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h_rand.copy()
            # Do local search from here
            remaining_time = time_limit - (time.time() - start_time)
            if remaining_time > 5:
                h_best, best_c5 = exhaustive_swap_search(
                    h_best, min(remaining_time * 0.5, 30), time.time(), rng)
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    rng = np.random.RandomState(seed)
    
    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)

    best_c5 = compute_c5(best_h)
    
    # Phase 1: Gradient optimization with soft-max SLSQP
    for T in [30, 80, 200]:
        if elapsed() > budget_s * 0.25:
            break
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(soft_max_obj, best_h, args=(T,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12})
            if res.fun < best_c5 + 0.01:
                cand = make_feasible(res.x, n)
                c5 = compute_c5(cand)
                if c5 < best_c5:
                    best_h = cand
                    best_c5 = c5
        except Exception:
            pass
    
    # Phase 2: Exhaustive swap local search
    swap_limit = budget_s * 0.55
    if elapsed() < swap_limit:
        best_h, best_c5 = exhaustive_swap_search(
            best_h, swap_limit - elapsed(), start_time, rng)

        best_h, best_c5 = random_restart_search(
            best_h, best_c5, n, restart_limit, start_time, rng)
    
    # Phase 4: Final local search polish
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = exhaustive_swap_search(
            best_h, budget_s * 0.95 - elapsed(), start_time, rng)
    
    # Final feasibility check
    best_h = make_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, n
