- Find λ such that sum(clamp(h - λ, 0, 1)) = n/2
- clamp(x, 0, 1) = min(max(x, 0), 1)

This is a 1D problem that can be solved by bisection since the function is monotonically decreasing in λ.

def project_onto_feasible(h, target_sum):
    """Project h onto {0≤x≤1, sum(x)=target_sum}."""
    # If already feasible
    if np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - target_sum) < 1e-10:
        return h.copy()
    # Bisection on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):

        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def lag_gradient(h, L):
    """Gradient of correlation at lag L with respect to h."""
    n = len(h)
