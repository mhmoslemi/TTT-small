from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def shift_vec(v, k):
    n = len(v)
    result = np.zeros(n)
    if k >= 0:
        if k < n:
            result[:n-k] = v[k:]
    else:
        kk = -k
        if kk < n:
            result[kk:] = v[:n-kk]
    return result

def c5_grad(h, temperature=30.0):
    n = len(h)
    dx = 2.0 / n
    onemh = 1.0 - h
    corr = np.correlate(h, onemh, mode="full") * dx
    c_max = np.max(corr)
    exps = np.exp(temperature * (corr - c_max))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-8:
            continue
        k = idx - (n - 1)
        grad += w * dx * (shift_vec(onemh, k) - shift_vec(h, -k))
    return grad

        h_new = h - lr * grad

            if val < best_val - 1e-12:
                best_val = val
                best_h = h_new.copy()
                h = h_new
                stall = 0

    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def shift_vec(v, k):
    n = len(v)
    out = np.zeros(n)
    if k >= 0:
        if k < n:
            out[:n-k] = v[k:]
    else:
        kk = -k
        if kk < n:
            out[kk:] = v[:n-kk]
    return out

def c5_smooth_grad(h, temp=30.0):
    """Gradient of log-sum-exp smoothed max correlation"""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    cmax = np.max(corr)
    exps = np.exp(temp * (corr - cmax))
    ws = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = ws[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        grad += w * dx * (shift_vec(omh, k) - shift_vec(h, -k))
    return grad, cmax

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def shift_vec(v, k):
    n = len(v)
    out = np.zeros(n)
    if k >= 0:
        if k < n:
            out[:n - k] = v[k:]
    else:
        kk = -k
        if kk < n:
            out[kk:] = v[:n - kk]
    return out

def c5_smooth_grad(h, temp=30.0):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    cmax = np.max(corr)
    exps = np.exp(temp * (corr - cmax))
    ws = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = ws[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        grad += w * dx * (shift_vec(omh, k) - shift_vec(h, -k))
    return grad

def pgd(h0, max_steps=10000, lr=0.08, temp=30.0):
    n = len(h0)
    ts = n / 2.0
    h = project_feasible(h0.copy(), ts)
    best_h = h.copy()
    best_v = compute_c5(h)
    lr_cur = lr
    stall = 0
    for s in range(max_steps):
        g = c5_smooth_grad(h, temp)
        h_new = h - lr_cur * g
        h_new = project_feasible(h_new, ts)
        v = compute_c5(h_new)
        if v < best_v - 1e-13:
            best_v = v
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:

            if lr_cur < 1e-9 or stall > 500:

            h_new = h - lr_cur * g
            h_new = project_feasible(h_new, ts)

            if v < best_v - 1e-13:
                best_v = v
                best_h = h_new.copy()
                h = h_new
                stall = 0
    return best_h, best_v
