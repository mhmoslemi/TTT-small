import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(1000):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-15:
            break
        h = h + (target_sum - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def penalty_objective(x, n, dx, mu):
    h = x[:n]
    t = x[n]
    val = t
    val += mu * (np.sum(h) - n / 2.0) ** 2
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    v = np.maximum(0.0, corr - t)
    val += mu * np.sum(v * v)
    return val

def coordinate_descent_polish(h, n, dx, target_sum, n_steps=20):
    h = project_to_constraint(h, target_sum)
    best_c5 = compute_c5(h, dx)
    deltas = [-0.03, -0.015, -0.007, 0.007, 0.015, 0.03]
    for _ in range(n_steps):
        improved = False
        for i in range(n):
            for d in deltas:
                h_new = h.copy()
                h_new[i] += d
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_to_constraint(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-10:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

    start_time = time.time()

                       options={'maxiter': 500, 'ftol': 1e-16})
        h_cand = project_to_constraint(res.x[:n], target_sum)
