        return compute_c5(h, n)
    return obj

def make_constraint(n):
    target = n / 2.0
    def cf(h):
        return np.sum(h) - target
    def cj(h):
        return np.ones(n)
    return {'type': 'eq', 'fun': cf, 'jac': cj}

def slsqp_optimize(h_start, n, max_iter=400):
    def obj(h):
        return compute_c5(h, n)
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:
            return fix_sum(res.x, n)
    except Exception:
        pass
    return None

def local_search(h, n, rng, n_iter=3000):
    h = h.copy()
    cur_c5 = compute_c5(h, n)
    for _ in range(n_iter):
        i = rng.randint(n)
        delta = rng.choice([-0.2, -0.1, -0.05, 0.05, 0.1, 0.2])
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < cur_c5:
            h = h_new
            cur_c5 = c5_new
    return h, cur_c5

def round_and_adjust(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = int(n / 2)
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        order = np.argsort(h)[::-1]
        for idx in order:
            if h_bin[idx] == 1 and n_ones > target:
                h_bin[idx] = 0
                n_ones -= 1
    elif n_ones < target:
        order = np.argsort(h)[::-1]
        for idx in order:
            if h_bin[idx] == 0 and n_ones < target:
                h_bin[idx] = 1
                n_ones += 1
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n_init = len(h0)
    best_c5 = compute_c5(h0, n_init)
    best_h = h0.copy()
    best_n = n_init

    n_candidates = sorted(set([n_init, 64, 80, 96, 100, 120, 128, 160, 200]))

    for n in n_candidates:
        if time.time() - start_time > budget_s * 0.85:
            break

        if n == n_init:
            h_start = h0.copy()
        else:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n)
            h_start = np.interp(x_new, x_old, h0)

        h_start = fix_sum(h_start, n)
        n_attempts = 6 if n <= 100 else 3

        for attempt in range(n_attempts):
            if time.time() - start_time > budget_s * 0.90:
                break

            if attempt == 0:
                h_try = h_start.copy()
            elif attempt == 1:
                h_try = np.clip(h_start + rng.normal(0, 0.05, n), 0, 1)
            elif attempt == 2:
                h_try = np.clip(h_start + rng.normal(0, 0.15, n), 0, 1)
            elif attempt == 3:
                h_try = round_and_adjust(h_start, n, rng)
            elif attempt == 4:
                h_try = (rng.rand(n) > 0.5).astype(float)
            else:
                h_try = np.clip(0.5 * h_start + 0.5 * rng.rand(n), 0, 1)

            h_try = fix_sum(h_try, n)
            h_res = slsqp_optimize(h_try, n, max_iter=300)

            if h_res is not None:
                c5 = compute_c5(h_res, n)
                if abs(np.sum(h_res) - n / 2.0) < 1e-6 and c5 < best_c5:
                    best_c5 = c5
                    best_h = h_res.copy()
                    best_n = n
                    h_start = h_res.copy()

            # Local search from SLSQP result
            if h_res is not None and time.time() - start_time < budget_s * 0.88:
                h_ls, c5_ls = local_search(h_res, n, rng, n_iter=1500)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()
                    best_n = n

    # Final polish on best solution
    if time.time() - start_time < budget_s * 0.95:
        h_polish, c5_polish = local_search(best_h, best_n, rng, n_iter=5000)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish

    # Validate
    final_c5 = compute_c5(best_h, best_n)
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-6:
        best_h = fix_sum(best_h, best_n)
        final_c5 = compute_c5(best_h, best_n)

    return best_h, final_c5, best_n
