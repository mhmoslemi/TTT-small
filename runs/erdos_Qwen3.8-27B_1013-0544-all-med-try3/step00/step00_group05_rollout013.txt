import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h
    # Iterative repair
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / np.sum(mask), 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / np.sum(mask), 0.0)
    return h

def coordinate_ascent(h, n, max_iter=1000):

    best = evaluate(h)
    for it in range(max_iter):
        improved = False

                    best_val = val
                    best_hj = h[j]
                    improved = True

        if not improved and evaluate(h) == best:

Wait, it's NOT mod. It's the full correlation without wrapping. So:

For binary b, b[i]*(1-b[j]) = 1 if b[i]=1 and b[j]=0. So:
corr[k] = number of pairs (i, j) with j = i + (k - (n-1)), b[i]=1, b[j]=0

        h_ls, c5_ls = binary_sd(h_bi, n_try, 30000, rng, dl(0.75))
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_sa

    # Phase 5: Random restarts on higher res
    n_hi = 2 * n0 if 2 * n0 % 2 == 0 else 2 * n0 + 1
    if n_hi % 2 == 0 and n_hi <= 400:
        hs2, cs2 = multi_start_binary(n_hi, rng, dl(0.88), n_starts=15, max_sd=3000)
        if hs2 is not None and cs2 < best_c5:
            best_h, best_c5 = hs2, cs2

    # Phase 6: Final continuous refinement via Nelder-Mead
    if time.time() < dl(0.98):
        n = len(best_h)
        hb2 = to_binary(best_h, n)
        if hb2 is not None and objective_c5(hb2) <= best_c5 + 1e-12:
            best_h = hb2
            best_c5 = objective_c5(best_h)
        # SA polish
        h_pol, c5_pol = binary_sa(best_h, n, 100000, rng, dl(0.98))
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    return best_h, best_c5, len(best_h)
