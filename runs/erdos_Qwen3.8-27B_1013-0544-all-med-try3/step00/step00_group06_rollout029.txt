def project_to_constraint(h, target_sum):

Let me re-derive the gradient from scratch.

We have: corr[m] = np.correlate(h, 1-h, mode='full')[m]

From the numpy documentation, for `np.correlate(a, v, mode='full')`:
The output c has length len(a) + len(v) - 1 = 2n - 1.
c[k] = sum_n a[n+k] * conj(v[n]) for k = -(len(v)-1) ... len(a)-1

The indexing of the output array: output[m] corresponds to k = m - (len(v)-1) = m - (n-1).

So output[m] = sum_n h[n + m - n + 1] * (1 - h[n]) where the sum is over n such that:
- 0 ≤ n < n (i.e., 0 ≤ n ≤ n-1)
- 0 ≤ n + m - n + 1 < n (i.e., 0 ≤ n + k < n where k = m - n + 1)

Let me use k = m - (n-1) as the lag. Then:
output[m] = sum_n h[n+k] * (1-h[n]) for n such that 0 ≤ n ≤ n-1 and 0 ≤ n+k ≤ n-1.

The valid range for n: max(0, -k) ≤ n ≤ min(n-1, n-1-k).

Now, output[m] = sum_{n∈valid} h[n+k] - h[n+k]*h[n]

Define L(m) = sum_{n∈valid} h[n+k] (linear in h)
Define Q(m) = sum_{n∈valid} h[n+k]*h[n] (quadratic in h)

Then output[m] = L(m) - Q(m).

∂L(m)/∂h[i] = 1 if ∃ n ∈ valid with n+k = i, i.e., n = i-k ∈ valid, i.e., max(0,-k) ≤ i-k ≤ min(n-1, n-1-k).
This is equivalent to: max(k, 0) ≤ i ≤ min(n-1+k, n-1).

∂Q(m)/∂h[i]:
Q(m) = sum_{n∈valid} h[n+k]*h[n]
The derivative w.r.t. h[i] gets contributions from:
- Terms where n = i (and i ∈ valid): contribution is h[n+k] = h[i+k]
- Terms where n+k = i (and n ∈ valid): contribution is h[n] = h[i-k]

So ∂Q(m)/∂h[i] = (h[i+k] if i ∈ [n_lo, n_hi]) + (h[i-k] if i-k ∈ [n_lo, n_hi])

where n_lo = max(0,-k), n_hi = min(n-1, n-1-k).

The condition i ∈ [n_lo, n_hi] means the first term is present.
The condition i-k ∈ [n_lo, n_hi] means n_lo+k ≤ i ≤ n_hi+k, i.e., the second term is present.

So:
∂output[m]/∂h[i] = [max(k,0) ≤ i ≤ min(n-1+k, n-1)]
                   - [n_lo ≤ i ≤ n_hi] * h[i+k]
                   - [n_lo+k ≤ i ≤ n_hi+k] * h[i-k]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    weights = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        lin_lo = max(k, 0)
        lin_hi = min(n - 1 + k, n - 1)
        if lin_lo <= lin_hi:
            grad[lin_lo:lin_hi + 1] += w
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        if n_lo <= n_hi:
            h_idx1 = np.arange(n_lo, n_hi + 1) + k
            grad[n_lo:n_hi + 1] -= w * h[h_idx1]
            q2_lo = n_lo + k
            q2_hi = n_hi + k
            h_idx2 = np.arange(q2_lo, q2_hi + 1) - k
            grad[q2_lo:q2_hi + 1] -= w * h[h_idx2]
    return grad * dx

def project_constrained(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def interp_to_n(h_from, n_to):
    n_from = len(h_from)
    if n_from == n_to:
        return h_from.copy()
    x_from = np.linspace(0.0, 1.0, n_from, endpoint=False)
    x_to = np.linspace(0.0, 1.0, n_to, endpoint=False)
    h_to = np.interp(x_to, x_from, h_from)
    return project_constrained(h_to, n_to / 2.0)

def make_cosine_init(n, freq):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * x / n)
    h = np.clip(h, 0.0, 1.0)
    return project_constrained(h, n / 2.0)

def make_bimodal_init(n, split):
    h = np.zeros(n)
    n_left = int(n * split)
    h[:n_left] = 1.0
    h[n_left:] = 1.0
    total = h.sum()
    if total > 0:
        h = h * (n / 2.0) / total
    h = np.clip(h, 0.0, 1.0)
    return project_constrained(h, n / 2.0)

def optimize_h(h0, time_limit, rng, beta_start=50.0, lr_start=0.004):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = beta_start
    lr = lr_start
    start = time.time()
    no_improve = 0
    temperature = 0.005

    while time.time() - start < time_limit:
        grad = smooth_max_grad(h, beta)
        gnorm = np.linalg.norm(grad)

        c5 = compute_c5(h_new)

            best_h = h_new.copy()
            h = h_new
            no_improve = 0

                h = project_constrained(h, target_sum)
                lr = lr_start
                beta = beta_start
                no_improve = 0
                temperature = 0.005
