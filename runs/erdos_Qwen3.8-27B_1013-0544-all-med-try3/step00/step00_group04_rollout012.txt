The change in c[k] due to h[j]: 0→1:
- As first factor (t=j): The term was 0*(1-h[j-lag]) = 0. Now it's 1*(1-h[j-lag]) = 1-h[j-lag]. Change: +(1-h[j-lag]).

Δc[k] = -(1-h[i-lag]) + h[i+lag] + (1-h[j-lag]) - h[j+lag]

        delta[k] = dc

    t_start = time.time()

        if time.time() - t_start > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_new_c5 = best_c5 + 1e-12
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c5 < best_new_c5:
                    best_new_c5 = c5
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_c5 = best_new_c5
    return h, best_c5

1. Compute all d[k] values

3. Check if this ε is in [0, ε_max]
4. Evaluate max_k new_c[k] at each candidate ε
5. Also check ε = 0 and ε = ε_max
6. Pick the best

def compute_grad_c(h, n):
    """Compute ∂c[k]/∂h[i] for all k and i.
    
    c[k] = sum_t h[t]*(1-h[t-lag_k]) where lag_k = k-(n-1)
    
    ∂c[k]/∂h[i] = (1-h[i-lag_k]) [first factor, if valid] - h[i+lag_k] [second factor, if valid]
    """
    # grad[k, i] = ∂c[k]/∂h[i]
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n - 1)
        for i in range(n):
            g = 0.0
            # First factor: h[i] appears as h[t] with t=i, valid if 0 <= i-lag < n
            if 0 <= i - lag < n:
                g += (1.0 - h[i - lag])
            # Second factor: h[i] appears as h[t-lag] with t-lag=i, i.e., t=i+lag, valid if 0 <= i+lag < n
            if 0 <= i + lag < n:
                g -= h[i + lag]
            grad[k, i] = g
    return grad

def fractional_optimize(h, max_iter=1000, time_limit=30.0):
    """Coordinate descent allowing fractional values."""
    n = len(h)
    h = h.astype(np.float64).copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    
    for _ in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Compute gradient matrix
        grad = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lag = k - (n - 1)
            for i in range(n):
                g = 0.0
                if 0 <= i - lag < n:
                    g += (1.0 - h[i - lag])
                if 0 <= i + lag < n:
                    g -= h[i + lag]
                grad[k, i] = g
        
        # Try all pairs (i, j) and find the best transfer
        best_improvement = 1e-10
        best_i, best_j, best_eps = -1, -1, 0.0
        
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                # Transfer ε from i to j: h[i] -= ε, h[j] += ε
                eps_max = min(h[i], 1.0 - h[j])
                if eps_max <= 0:
                    continue
                eps_min = max(-h[j], -(1.0 - h[i]))
                if eps_min >= eps_max:
                    continue
                
                d = grad[:, j] - grad[:, i]
                new_corr = corr + eps_max * d  # at eps_max
                # Check if max decreased
                # For the optimal eps, we need to solve min max_k (corr[k] + eps*d[k])
                # This is a 1D convex optimization
                # Candidate eps values: 0, eps_max, eps_min, and breakpoints
                candidates = [0.0, eps_max]
                if eps_min < 0:
                    candidates.append(eps_min)
                
                # Breakpoints: corr[p] + eps*d[p] = corr[q] + eps*d[q]
                for p in range(2*n-1):
                    for q in range(p+1, 2*n-1):
                        dd = d[p] - d[q]
                        if abs(dd) > 1e-12:
                            eps_bp = (corr[q] - corr[p]) / dd
                            if eps_min <= eps_bp <= eps_max:
                                candidates.append(eps_bp)
                
                for eps in candidates:
                    if eps_min <= eps <= eps_max:
                        new_corr = corr + eps * d
                        c5 = np.max(new_corr) * (2.0/n)
                        if c5 < best_c5 - best_improvement:
                            best_improvement = best_c5 - c5
                            best_i, best_j, best_eps = i, j, eps
        
        if best_i < 0:
            break
        
        h[best_i] -= best_eps
        h[best_j] += best_eps
        # Clip to [0,1]
        h[best_i] = np.clip(h[best_i], 0, 1)
        h[best_j] = np.clip(h[best_j], 0, 1)
        best_c5 = compute_c5(h)
    
    return h, best_c5
