def subgradient(h, k_star, n):
    """Compute subgradient of C[k_star] w.r.t. h"""
    g = np.zeros(n)
    offset = n - 1
    # First term: (1-h[j + k_star - offset]) for valid indices
    idx1 = k_star - offset  # j + k_star - offset
    # j + idx1 in [0, n-1] means j in [-idx1, n-1-idx1]
    j_start1 = max(0, -idx1)

    if j_start1 < j_end1:
