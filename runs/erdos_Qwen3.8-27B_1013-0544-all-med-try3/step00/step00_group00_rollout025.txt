import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def soft_max_val(c, tau):
    return logsumexp(tau * c) / tau

def soft_max_grad(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(c)
    ew = np.exp(tau * c - tau * mc)
    ew /= ew.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        w = ew[ki]
        if w < 1e-14:
            continue
        k = ki - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += 1.0 - h[j - k]
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += w * g
    return tau * dx * grad

def obj_fn(x, dx, tau, lam, target):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return soft_max_val(c, tau) + lam * (np.sum(x) - target) ** 2

def grad_fn(x, dx, tau, lam, target):
    g = soft_max_grad(x, dx, tau)
    g += 2.0 * lam * (np.sum(x) - target)
    return g
