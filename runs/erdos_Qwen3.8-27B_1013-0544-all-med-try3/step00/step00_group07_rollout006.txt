                if n - L > 0:
                    grad[:n - L] += w * dx * (1.0 - h[L:])
                grad[L:] -= w * dx * h[:n - L]

    obj = lse_obj(h, alpha, n)
    return obj, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 1e-12:
                h += diff * room / total_room
        else:
            total = h.sum()
            if total > 1e-12:
                h *= target / total
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in idx:
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def optimize_from(h0, n, alpha, budget_start, budget_frac, rng):
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def obj_wrapper(x):
        return lse_obj(x, alpha, n)

    def grad_wrapper(x):
        return lse_grad(x, alpha, n)[1]

    best_local = h0.copy()
    best_local_c5 = compute_c5(h0)

    res = minimize(
        obj_wrapper, h0, jac=grad_wrapper,
        method='SLSQP', bounds=bounds, constraints=constraints,
        options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0}
    )

    if res.x is not None:
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_local_c5:
            best_local = h_new
            best_local_c5 = c5_new

    # Post-optimization: try a few small perturbations and re-optimize
    for _ in range(3):
        if time.time() - budget_start > budget_frac:
            break
        h_pert = best_local + rng.normal(0, 0.02, n)
        h_pert = project_feasible(h_pert, n)
        res2 = minimize(
            obj_wrapper, h_pert, jac=grad_wrapper,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0}
        )
        if res2.x is not None:
            h_new = project_feasible(res2.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_local_c5:
                best_local = h_new
                best_local_c5 = c5_new

    return best_local, best_local_c5

def try_binary_local_search(h0, n, rng, max_flips=5000):
    """Try to improve by flipping values toward 0 or 1."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0
    improved = True
    flips = 0
    while improved and flips < max_flips:
        improved = False
        flips += 1
        for i in range(n):
            # Try setting h[i] to 0
            if h[i] > 1e-6:
                h_new = h.copy()
                h_new[i] = 0.0
                new_sum = h_new.sum()
                if abs(new_sum - target) < 0.5 / n:
                    # Adjust one other element
                    pass
            # Try setting h[i] to 1
            if h[i] < 1.0 - 1e-6:
                h_new = h.copy()
                h_new[i] = 1.0
                new_sum = h_new.sum()
                if abs(new_sum - target) < 0.5 / n:
                    c5_try = compute_c5(h_new)
                    if c5_try < best_c5 - 1e-10:
                        h = h_new
                        best_c5 = c5_try
                        improved = True
                        break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    n = 80
    dx = 2.0 / n

    # Get initial values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP with increasing alpha on n=80
    alphas = [30, 60, 120, 250, 500]
    for alpha in alphas:
        if time.time() - t0 > budget_s * 0.5:
            break
        h_curr, c5_curr = optimize_from(best_h, n, alpha, t0, budget_s * 0.55, rng)
        if c5_curr < best_c5:
            best_h = h_curr
            best_c5 = c5_curr

    # Phase 2: Random restarts with perturbation
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.7:
            break
        sigma = 0.05 + 0.03 * r
        h_r = best_h + rng.normal(0, sigma, n)
        h_r = project_feasible(h_r, n)
        alpha_r = 200
        h_curr, c5_curr = optimize_from(h_r, n, alpha_r, t0, budget_s * 0.75, rng)
        if c5_curr < best_c5:
            best_h = h_curr
            best_c5 = c5_curr

    # Phase 3: Try larger n if we haven't beaten 0.3808 and have time remaining
    if best_c5 > 0.3808 and time.time() - t0 < budget_s * 0.75:
        for n_large in [160, 320]:
            if time.time() - t0 > budget_s * 0.9:
                break
            # Upsample best_h to n_large
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_large, endpoint=False) + 1.0 / n_large
            h_large = np.interp(x_new, x_old, best_h, period=2.0)
            h_large = project_feasible(h_large, n_large)
            c5_large = compute_c5(h_large)

            for alpha in [50, 150, 400]:
                if time.time() - t0 > budget_s * 0.92:
                    break
                h_curr, c5_curr = optimize_from(h_large, n_large, alpha, t0, budget_s * 0.93, rng)
                if c5_curr < c5_large:
                    h_large = h_curr
                    c5_large = c5_curr
                if c5_large < best_c5:
                    best_c5 = c5_large
                    best_h = h_large[:n]  # Will re-project below if needed
            # If n_large solution is better, keep it
            if c5_large < best_c5:
                best_c5 = c5_large
                best_h = h_large

    # Final: try one more refinement on the best solution
    if time.time() - t0 < budget_s * 0.95:
        n_final = len(best_h)
        alpha_final = 500
        h_final, c5_final = optimize_from(best_h, n_final, alpha_final, t0, budget_s * 0.97, rng)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Final projection and evaluation
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_out
