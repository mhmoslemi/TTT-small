import time

def project_onto_feasible(y, n):
    """Project y onto {h in [0,1]^n : sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Add/subtract from elements not at bounds
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h
