def pair_update(h, i, j, corr, n):
    """Find optimal δ for h[i]+=δ, h[j]-=δ that minimizes max corr"""
    target_lag = n - 1
    # Compute grad_m for this pair
    # grad_m = ∂corr[m]/∂h[i] - ∂corr[m]/∂h[j]
    # = (1-h[i-lag])*[valid] - h[i+lag]*[valid] - (1-h[j-lag])*[valid] + h[j+lag]*[valid]
    
    grads = np.zeros(2*n-1)
    for m in range(2*n-1):
        lag = m - target_lag
        g = 0.0
        idx = i - lag
        if 0 <= idx < n:
            g += (1.0 - h[idx])
        idx = i + lag
        if 0 <= idx < n:
            g -= h[idx]
        idx = j - lag
        if 0 <= idx < n:
            g -= (1.0 - h[idx])
        idx = j + lag
        if 0 <= idx < n:
            g += h[idx]
        grads[m] = g
    
    # New corr values: corr_new[m] = corr[m] + δ * grads[m]
    # Minimize max_m (corr[m] + δ * grads[m])
    # Subject to: max(-h[i], h[j]-1) <= δ <= min(1-h[i], h[j])
    
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_min >= delta_max:
        return 0.0
    
    # Find break points
    # At break point: corr[a] + δ*grads[a] = corr[b] + δ*grads[b]
    # δ = (corr[b] - corr[a]) / (grads[a] - grads[b])
    
    candidates = [delta_min, delta_max]
    for a in range(2*n-1):
        for b in range(a+1, 2*n-1):
            denom = grads[a] - grads[b]
            if abs(denom) > 1e-12:
                delta = (corr[b] - corr[a]) / denom
                if delta_min < delta < delta_max:
                    candidates.append(delta)
    
    # Also check where individual corr values are at their max
    # (the max of linear functions is minimized where two are equal, or at boundary)
    
    best_delta = 0.0
    best_val = np.max(corr)
    for d in candidates:
        val = np.max(corr + d * grads)
        if val < best_val:
            best_val = val
            best_delta = d
    
    return best_delta

Let me implement the pair-swap with the exact 1D minimization:

For a given pair (i, j), changing h[i] += δ, h[j] -= δ:
- New corr[m] = corr[m] + δ * g[m] where g[m] = ∂corr[m]/∂h[i] - ∂corr[m]/∂h[j]
- We want to minimize max_m (corr[m] + δ*g[m]) over δ ∈ [δmin, δmax]
- This is a 1D convex piecewise-linear minimization

The minimum is at:
- A breakpoint: δ* = (corr[b] - corr[a]) / (g[a] - g[b]) for some a, b
- Or a boundary: δmin or δmax

Algorithm:
1. Compute all breakpoints δ* in (δmin, δmax)
2. Evaluate max_m (corr[m] + δ* * g[m]) at each breakpoint and at boundaries
3. Return the δ that gives the minimum max value

For efficiency, I don't need to check all O(n²) breakpoints. I can use the fact that the max of linear functions changes its argmax as δ increases. The sequence of argmax values can be found in O(n log n) using a convex hull, but for n=89, even O(n²) = 8000 is fine.

def pair_step_exact(h, i, j, corr):
    n = len(h)
    # Compute g[m] = d(corr[m])/d(h[i]) - d(corr[m])/d(h[j])
    g = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        val = 0.0
        idx = i - lag
        if 0 <= idx < n:
            if idx != j:
                val += (1.0 - h[idx])
            else:
                # Cross term: (1-h[j]) + cross... 
                # Actually (1-h'[j]) where h'[j] = h[j] - δ
                # The term is h'[i] * (1-h'[j]) = (h[i]+δ)(1-h[j]+δ)
                # = h[i](1-h[j]) + δ(1-h[j]) + δ*h[i] + δ²
                # First order: δ(1-h[j]) + δ*h[i] = δ(1-h[j]+h[i])
                # But for the gradient (first order), we use the unsigrated values
                val += (1.0 - h[idx])  # Standard first-order
        idx = i + lag
        if 0 <= idx < n:
            val -= h[idx]
        idx = j - lag
        if 0 <= idx < n:
            val -= (1.0 - h[idx])
        idx = j + lag
        if 0 <= idx < n:
            val += h[idx]
        g[m] = val
    
    # Linear programming in 1D: minimize max_m (corr[m] + δ*g[m])
    # subject to δmin <= δ <= δmax
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_min >= delta_max:
        return 0.0, np.max(corr)
    
    # Find all breakpoints
    n_corr = 2 * n - 1
    breakpoints = [delta_min, delta_max]
    for a in range(n_corr):
        for b in range(a + 1, n_corr):
            denom = g[a] - g[b]
            if abs(denom) > 1e-14:
                db = (corr[b] - corr[a]) / denom
                if delta_min < db < delta_max:
                    breakpoints.append(db)
    
    best_delta = 0.0
    best_val = np.inf
    for d in breakpoints:
        val = np.max(corr + d * g)
        if val < best_val:
            best_val = val
            best_delta = d
    
    return best_delta, best_val

Let me optimize. Instead of checking all O(n²) pairs of correlation values for breakpoints, I can use a smarter approach:

The function f(δ) = max_m (corr[m] + δ*g[m]) is convex. Its minimum over [δmin, δmax] can be found by:
1. If f'(δmin) >= 0 (i.e., the max is achieved by a line with non-negative slope at δmin), the minimum is at δmin.
2. If f'(δmax) <= 0, the minimum is at δmax.
3. Otherwise, the minimum is at a kink point.

At a kink point, two lines a and b are tied: corr[a] + δ*g[a] = corr[b] + δ*g[b], i.e., δ = (corr[b]-corr[a])/(g[a]-g[b]).

But we don't need to check ALL pairs. We only need to check the pair that's active at the minimum. 

A practical approach: use bisection on the "slope" of f. The subgradient of f at δ is the set of slopes of the active (maximum) lines. f is minimized where 0 is in the subgradient, i.e., where there exist active lines with slopes on both sides of 0 (or one with slope 0).

Algorithm:
1. Evaluate f at δmin: find which m is active (max). Let its slope be g[active].
2. If g[active] >= 0, minimum is at δmin (f is increasing from δmin).
3. Evaluate f at δmax: find which m is active. If g[active] <= 0, minimum is at δmax.
4. Otherwise, the minimum is at a kink. Use binary search:
   - At the current δ, find the active m (the one with max corr[m] + δ*g[m])
   - If g[active] > 0, the minimum is to the left → move δ down
   - If g[active] < 0, the minimum is to the right → move δ up
   - If g[active] = 0, we're at the minimum (or a flat region)
   - But this doesn't work directly because the active m changes...

        if v < best_v:
            best_v = v
            best_d = d
    return best_d

def exact_pair_descent(h_init, target, t0, budget_s, frac, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deadline = t0 + budget_s * frac
    corr = get_corr(h)
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            for j in order:
                if i == j:
                    continue
                if i > j:
                    continue
                corr = get_corr(h)
                d = exact_pair_step(h, i, j, corr)
                if abs(d) < 1e-14:
                    continue
                new_c5 = compute_c5(h + d * (np.zeros(n)))
                h[i] += d
                h[j] -= d
                h = project_feasible(h, target)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    best_h = project_feasible(h0, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Smoothed gradient descent
    T_list_1 = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
    h1, c5_1 = smoothed_grad_desc(best_h, target, T_list_1, t0, budget_s, 0.35)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Exact pair descent
    h2, c5_2 = exact_pair_descent(best_h, target, t0, budget_s, 0.65, seed + 1)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Stochastic local search
    h3, c5_3 = pair_search(best_h, target, t0, budget_s, 0.80, seed + 2)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Final gradient polish
    T_list_2 = [0.005, 0.002, 0.001]
    h4, c5_4 = smoothed_grad_desc(best_h, target, T_list_2, t0, budget_s, 0.92)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

    # Phase 5: Final pair descent
    if time.time() - t0 < budget_s * 0.95:
        h5, c5_5 = exact_pair_descent(best_h, target, t0, budget_s, 0.98, seed + 3)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
