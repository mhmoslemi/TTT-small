import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h):
    """Subgradient of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m_star = np.argmax(corr)
    k = m_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        # h[j] in first factor: i = j - k
        i1 = j - k
        if 0 <= i1 < n:
            val += (1.0 - h[i1])
        # h[j] in (1-h[i]) factor: i = j
        j2 = j + k
        if 0 <= j2 < n:
            val += (-h[j2])
        grad[j] = val
    return dx * grad

def project_to_constraint(y, target_sum):
    """Project y onto {x: 0 <= x[i] <= 1, sum(x) = target_sum}."""
    n = len(y)
    # Bisection on lambda
    lo, hi = -1.0, 1.0 + max(y.max() - y.min(), 1.0)
    # Actually, lambda needs to be such that sum(clamp(y[i]-lambda, 0, 1)) = target_sum
    # When lambda -> -inf, clamp(y[i]-lambda, 0, 1) -> 1 for all i, sum -> n
    # When lambda -> +inf, clamp -> 0, sum -> 0
    # So we need sum = target_sum, find lambda
    
    def f(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0)) - target_sum
    
    # Find bracket
    lo_lam = y.min() - 1.0
    hi_lam = y.max() + 1.0
    for _ in range(100):
        mid = (lo_lam + hi_lam) / 2.0
        if f(mid) > 0:
            lo_lam = mid
        else:
            hi_lam = mid
    lam = (lo_lam + hi_lam) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m_star = np.argmax(corr)
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        j_s, j_e = k, n
    else:
        j_s, j_e = 0, n + k
    if j_s < j_e:
        grad[j_s:j_e] += (1.0 - h[j_s-k:j_e-k])
    if k >= 0:
        j_s2, j_e2 = 0, n - k
    else:
        j_s2, j_e2 = -k, n
    if j_s2 < j_e2:
        grad[j_s2:j_e2] += (-h[j_s2+k:j_e2+k])
    return dx * grad

def smooth_c5(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    shifted = corr - np.max(corr)
    weights = np.exp(beta * shifted)
    return dx * (np.max(corr) + np.log(np.sum(weights) / len(corr)) / beta)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_star = np.argmax(corr)
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        j_s, j_e = k, n
    else:
        j_s, j_e = 0, n + k
    if j_s < j_e:
        grad[j_s:j_e] += (1.0 - h[j_s - k:j_e - k])
    if k >= 0:
        j_s2, j_e2 = 0, n - k
    else:
        j_s2, j_e2 = -k, n
    if j_s2 < j_e2:
        grad[j_s2:j_e2] += (-h[j_s2 + k:j_e2 + k])
    return dx * grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    hi = y.max() + 1.5
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_descent(h, beta, target_sum, n_iters, step0, deadline):
    step = step0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        h = project_feasible(h - step * smooth_c5_grad(h, beta), target_sum)
        if it % 300 == 299:
            step *= 0.9
    return h

def subgrad_descent(h, target_sum, n_iters, step0, deadline):
    step = step0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        h = project_feasible(h - step * c5_subgradient(h), target_sum)
        if it % 1000 == 999:
            step *= 0.9
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Smooth gradient descent with increasing beta (annealing toward true max)
    phase_configs = [
        (15.0, 0.8, 10000),
        (40.0, 0.4, 10000),
        (80.0, 0.2, 10000),
        (150.0, 0.1, 10000),
        (300.0, 0.05, 10000),
    ]
    for beta, step0, n_iters in phase_configs:
        if time.time() > deadline:
            break
        h_cur = smooth_descent(best_h, beta, target_sum, n_iters, step0, deadline)
        c5v = compute_c5(h_cur)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cur.copy()

    # Phase 2: Exact subgradient descent
    h_cur = subgrad_descent(best_h, target_sum, 40000, 0.04, deadline)
    c5v = compute_c5(h_cur)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_cur.copy()

    # Phase 3: Random perturbations to escape local optima
    for trial in range(100):
        if time.time() > deadline:
            break
        scale = 0.015 + 0.004 * trial
        h_p = project_feasible(best_h + rng.normal(0, scale, n), target_sum)
        h_p = smooth_descent(h_p, 60.0, target_sum, 2000, 0.04, deadline)
        h_p = subgrad_descent(h_p, target_sum, 3000, 0.008, deadline)
        c5v = compute_c5(h_p)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_p.copy()

    # Phase 4: Final fine polish with small steps
    h_cur = subgrad_descent(best_h, target_sum, 30000, 0.003, deadline)
    c5v = compute_c5(h_cur)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_cur.copy()

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
