Let me also think about one more thing: the "full" correlation includes lags from -(N-1) to (N-1). But for the problem, do we care about all lags or just positive ones?

Looking at the problem statement: C₅ = max_k ∫ h(x)(1-h(x+k)) dx

The integral is over x such that both x and x+k are in [0,2]. So k ranges over all real values, and the integral is over the intersection [max(0,-k), min(2, 2-k)].

In the discrete setting, this corresponds to all lags in the full correlation. So yes, we need to consider all lags (positive and negative).

But by symmetry, overlap at lag k (positive) and overlap at lag -k (negative) are related:
overlap_pos(k) = #{i: h[i]=1, h[i+k]=0} (i from 0 to N-1-k)
overlap_neg(k) = #{i: h[i]=1, h[i-k]=0} (i from k to N-1) = #{j: h[j+k]=1, h[j]=0} (j from 0 to N-1-k)

These are different! overlap_pos counts 1→0 at distance k (forward), while overlap_neg counts 0→1 at distance k (forward).

In a "balanced" sequence, these should be similar but not necessarily equal.

The maximum over all lags includes both directions.

import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def local_search(h, dx, time_limit=5.0, strategy='best'):
    N = len(h)
    best_c5 = compute_c5(h, dx)
    start = time.time()
    while True:
        if time.time() - start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if strategy == 'best':
            best_swap = None
            best_new = best_c5
            for i in ones:
                for j in zeros:
                    h[i], h[j] = 0, 1
                    c5 = compute_c5(h, dx)
                    h[i], h[j] = 1, 0
                    if c5 < best_new - 1e-15:
                        best_new = c5
                        best_swap = (i, j)
            if best_swap is None:
                break
            i, j = best_swap
            h[i], h[j] = 0, 1
            best_c5 = best_new
        else:
            found = False
            rng_local = np.random.RandomState(np.random.randint(100000))
            order = list(range(len(ones)))
            rng_local.shuffle(order)
            for ii in order:
                if found:
                    break
                i = ones[ii]
                for j in zeros:
                    h[i], h[j] = 0, 1
                    c5 = compute_c5(h, dx)
                    h[i], h[j] = 1, 0
                    if c5 < best_c5 - 1e-15:
                        h[i], h[j] = 0, 1
                        best_c5 = c5
                        found = True
                        break
            if not found:
                break
    return h.copy(), best_c5

def sa_binary(h, dx, rng, time_limit, T0=0.04, T1=0.001):
    N = len(h)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    curr_c5 = best_c5
    start = time.time()
    while time.time() - start < time_limit:
        t_frac = (time.time() - start) / time_limit
        T = T0 * (T1 / T0) ** t_frac
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - curr_c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best_c5

def try_negate_and_swap(h, dx):
    """Try complementing and other transformations."""
    best = (h.copy(), compute_c5(h, dx))
    h_comp = 1.0 - h
    c5_comp = compute_c5(h_comp, dx)
    if c5_comp < best[1]:
        best = (h_comp.copy(), c5_comp)
    return best

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    init_h = None
    try:
        if initial_h_values is not None:
            init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        pass

    n_values = list(range(42, 102, 2))

    # Phase 1: Quick scan
    phase1_end = t0 + budget_s * 0.35
    for n in n_values:
        if time.time() > phase1_end:
            break
        dx = 2.0 / n
        time_per_n = 0.3
        n_starts = max(20, int(time_per_n / 0.01))

        for s in range(n_starts):
            if time.time() > phase1_end:
                break
            if s == 0 and init_h is not None and len(init_h) == n:
                h = np.zeros(n)
                h[np.argsort(init_h)[::-1][:n // 2]] = 1
            elif s == 1:
                h = np.zeros(n)
                h[:n // 2] = 1
            elif s == 2:
                h = np.zeros(n)
                h[::2] = 1
                if h.sum() > n // 2:
                    h[n - 1] = 0
                elif h.sum() < n // 2:
                    h[1::2] = 1
                    idx = np.where(h == 1)[0]
                    if len(idx) > n // 2:
                        h[idx[n // 2:]] = 0
            else:
                h = np.zeros(n)
                h[rng.choice(n, size=n // 2, replace=False)] = 1

            ls_time = 0.015
            h_ls, c5 = local_search(h, dx, time_limit=ls_time, strategy='first')
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ls.copy()
                best_n = n

    # Phase 2: Intensive local search on promising N values
    phase2_end = t0 + budget_s * 0.7
    # Try the best N and a few neighbors
    promising_n = sorted(set([best_n - 2, best_n, best_n + 2] + n_values[:8]))
    promising_n = [n for n in promising_n if n >= 40 and n <= 100 and n % 2 == 0]

    for n in promising_n:
        if time.time() > phase2_end:
            break
        dx = 2.0 / n
        while time.time() < phase2_end:
            h = np.zeros(n)
            h[rng.choice(n, size=n // 2, replace=False)] = 1
            ls_time = 0.03
            h_ls, c5 = local_search(h, dx, time_limit=ls_time, strategy='best')
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ls.copy()
                best_n = n

    # Phase 3: SA refinement
    if best_h is not None and time.time() < t0 + budget_s * 0.85:
        dx = 2.0 / best_n
        sa_time = min(t0 + budget_s * 0.85 - time.time(), 60.0)
        if sa_time > 1:
            h_sa, c5_sa = sa_binary(best_h, dx, rng, sa_time)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: Final local search polish
    if best_h is not None and time.time() < t0 + budget_s * 0.95:
        dx = 2.0 / best_n
        ls_time = t0 + budget_s * 0.95 - time.time()
        h_final, c5_final = local_search(best_h, dx, time_limit=max(1.0, ls_time), strategy='best')
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # If initial_h_values was non-binary and gave a good result, try to beat it
    if init_h is not None and time.time() < t0 + budget_s * 0.95:
        n_init = len(init_h)
        dx_init = 2.0 / n_init
        c5_init = compute_c5(init_h, dx_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = init_h.copy()
            best_n = n_init

    return best_h, best_c5, best_n
