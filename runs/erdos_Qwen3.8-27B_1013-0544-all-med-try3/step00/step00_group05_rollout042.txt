Let me derive the analytic gradient. The objective is f(h) = max_k c_k(h) * dx where c_k(h) = sum_{i} h[i+s] * (1-h[i]) for the appropriate shift.

The subgradient of f with respect to h[i] is:
∂f/∂h[i] = dx * (∂c_k*/∂h[i]) where k* is (one of) the maximizing indices.

For a specific k (corresponding to correlate output index idx):
c_idx(h) = sum over valid (p,q) with q-p = idx-(n-1) of h[q] * (1-h[p])

∂c_idx/∂h[i]:
- Terms where q = i: (1-h[p]) for each valid p where i-p = idx-(n-1), i.e., p = i - (idx-(n-1))
  If 0 ≤ p < n: contribution is (1-h[p])
- Terms where p = i: -h[q] for each valid q where q-i = idx-(n-1), i.e., q = i + (idx-(n-1))
  If 0 ≤ q < n: contribution is -h[q]

So ∂c_idx/∂h[i] = (1-h[p]) * [0≤p<n] - h[q] * [0≤q<n] where p = i - (idx-(n-1)) and q = i + (idx-(n-1)).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T, penalty_M):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    val = cmax + np.log(np.sum(np.exp(corr - cmax))) / T
    s = np.sum(h) - 0.5 * n
    val += penalty_M * s * s
    return val

def smooth_grad(h, T, penalty_M):
    n = len(h)
    dx = 2.0 / n
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    corr = raw_corr * dx
    cmax = np.max(corr)
    exp_vals = np.exp(T * (corr - cmax))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        s = k_idx - (n - 1)
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        if s == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wk * dx * (1.0 - h[:n - s])
            grad[:n - s] -= wk * dx * h[s:]
        else:
            j = -s
            grad[:n - j] += wk * dx * (1.0 - h[j:])
            grad[j:] -= wk * dx * h[:n - j]
    s_sum = np.sum(h) - 0.5 * n
    grad += 2.0 * penalty_M * s_sum
    return grad

def coordinate_step(h, n, inner_best_c5):
    dx = 2.0 / n
    improved = False
    for i in range(n):
        best_x = h[i]
        best_f = inner_best_c5
        for trial in [h[i] - 0.1, h[i] - 0.03, h[i] + 0.03, h[i] + 0.1]:
            trial = max(0.0, min(1.0, trial))
            if trial == h[i]:
                continue
            h[i] = trial
            diff = 0.5 * n - np.sum(h)
            if abs(diff) > 1e-12:
                h[i] += diff
                if h[i] < 0 or h[i] > 1:
                    h[i] = max(0.0, min(1.0, h[i]))
                    h += (0.5 * n - np.sum(h)) / n
                    h = np.clip(h, 0, 1)
            f = compute_c5(h)
            if f < best_f - 1e-12:
                best_f = f
                best_x = trial
        h[i] = best_x
        if best_x != h[i] - (best_x - best_x):
            if abs(best_x - h[i]) > 1e-15:
                improved = True
    if improved:
        h = make_feasible(h, n)
        if compute_c5(h) < inner_best_c5:
            return h, compute_c5(h)
    return h, inner_best_c5
