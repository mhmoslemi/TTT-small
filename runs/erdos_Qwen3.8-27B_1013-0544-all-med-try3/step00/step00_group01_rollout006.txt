        c = evaluate_c5(h_new, dx)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
            stuck = 0
        else:
            h = h_new
            stuck += 1
            if stuck > 500:
                h = best_h.copy()
                h += 0.02 * rng.randn(n)
                h = project_feasible(h, target_sum)
                stuck = 0
        
        lr *= 0.9995
        if lr < 0.003:
            lr = 0.02
        
        if time.time() - t0 > max_time * 0.8 and best_stuck == 0:
            best_stuck = stuck
    
    return best_h, best_c

def optimize_sa(h, n, rng, max_time):
    dx = 2.0 / n
    h = h.copy()
    current_c = evaluate_c5(h, dx)
    best_c = current_c
    best_h = h.copy()
    temp = 0.004
    t0 = time.time()
    iterations = 0
    
    while time.time() - t0 < max_time:
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        delta = 0.005 + 0.015 * rng.random()
        
        # Try i -> j
        if h[i] >= delta and h[j] + delta <= 1.0:
            h[i] -= delta
            h[j] += delta
            new_c = evaluate_c5(h, dx)
            accept = new_c < current_c or rng.random() < np.exp((current_c - new_c) / temp)
            if accept:
                current_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[i] += delta
                h[j] -= delta
        
        # Try j -> i
        if h[j] >= delta and h[i] + delta <= 1.0:
            h[j] -= delta
            h[i] += delta
            new_c = evaluate_c5(h, dx)
            accept = new_c < current_c or rng.random() < np.exp((current_c - new_c) / temp)
            if accept:
                current_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[j] += delta
                h[i] -= delta
        
        temp = max(0.0001, temp * 0.99995)
        if iterations % 5000 == 0 and temp < 0.0005:
            temp = 0.002
        iterations += 1
    
    return best_h, best_c

def optimize_systematic(h, n, rng, max_time):
    """Systematic sweep: try all pairs in order, accept improvements"""
    dx = 2.0 / n
    h = h.copy()
    current_c = evaluate_c5(h, dx)
    best_c = current_c
    best_h = h.copy()
    t0 = time.time()
    deltas = [0.01, 0.02, 0.05, 0.1]
    
    while time.time() - t0 < max_time:
        improved = False
        for delta in deltas:
            for i in range(n):
                for j in range(i + 1, n):
                    if h[i] >= delta and h[j] + delta <= 1.0:
                        h[i] -= delta
                        h[j] += delta
                        new_c = evaluate_c5(h, dx)
                        if new_c < current_c:
                            current_c = new_c
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
                    if h[j] >= delta and h[i] + delta <= 1.0:
                        h[j] -= delta
                        h[i] += delta
                        new_c = evaluate_c5(h, dx)
                        if new_c < current_c:
                            current_c = new_c
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
        if not improved:
            # Perturb to escape
            h = best_h + 0.03 * rng.randn(n)
            h = project_feasible(h, n * 0.5)
            current_c = evaluate_c5(h, dx)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(h0, 2.0 / n0)
    
    # Phase 1: FTL at n0
    t1 = time_left() * 0.35
    h1, c1 = optimize_ftl(h0, n0, rng, t1)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()
    
    # Phase 2: SA at n0
    if time_left() > 30:
        t2 = time_left() * 0.3
        h2, c2 = optimize_sa(best_h, n0, rng, t2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
    
    # Phase 3: Try other n values
    n_candidates = [48, 50, 60, 80, 100]
    for n_try in n_candidates:
        if time_left() < 80:
            break
        h_interp = interpolate_h(h0, n_try)
        c_interp = evaluate_c5(h_interp, 2.0 / n_try)
        t_alloc = min(time_left() * 0.25, 180)
        
        # FTL
        h_opt, c_opt = optimize_ftl(h_interp, n_try, rng, t_alloc * 0.6)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        
        # SA
        if time_left() > 40:
            h_opt2, c_opt2 = optimize_sa(h_opt, n_try, rng, min(time_left() * 0.4, 120))
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2
    
    # Phase 4: Systematic optimization at best n
    if time_left() > 60:
        n_best = len(best_h)
        h_sys, c_sys = optimize_systematic(best_h, n_best, rng, time_left() * 0.7)
        if c_sys < best_c:
            best_c = c_sys
            best_h = h_sys
    
    # Phase 5: Final SA polish
    if time_left() > 20:
        h_final, c_final = optimize_sa(best_h, len(best_h), rng, time_left() * 0.85)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
    
    # Ensure final feasibility
    best_h = project_feasible(best_h, len(best_h) * 0.5)
    best_c = evaluate_c5(best_h, 2.0 / len(best_h))
    
    return best_h, best_c, len(best_h)
