        else:
            hi = mu

    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jp = j + k_star
        jm = j - k_star
        if 0 <= jp < n:
            g += 1.0 - h[jp]
        if 0 <= jm < n:
            g -= h[jm]

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    if c_max > 0:

    else:
        log_w = np.zeros(len(c))
    # Softmax weights
    w = np.exp(log_w - log_w.max())
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            k_shift = k - (n - 1)
            grad += w[k] * compute_subgrad(h, k_shift)

            best_c, best_h = c, h

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c = evaluate_c5(best_h)[0]
    return best_h, best_c, len(best_h)
