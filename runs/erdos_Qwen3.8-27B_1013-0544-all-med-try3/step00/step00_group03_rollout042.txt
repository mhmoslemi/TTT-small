def incremental_swap(corr, h, p, q, n):
    """Update corr in-place after swapping h[p] (1->0) and h[q] (0->1).
    p must have h[p]=1, q must have h[q]=0.
    Returns the updated corr."""
    delta_p = -1.0  # h[p] changes by -1
    delta_q = 1.0   # h[q] changes by +1
    
    # For position p (delta_p = -1):
    # First term: L in [p-n+1, p], contribution = delta_p * (1-h[p-L])
    for L in range(max(p-n+1, -(n-1)), min(p, n-1)+1):
        corr[n-1+L] += delta_p * (1.0 - h[p-L])
    # Second term: L in [-p, n-1-p], contribution = -delta_p * h[p+L]
    for L in range(max(-p, -(n-1)), min(n-1-p, n-1)+1):
        corr[n-1+L] += -delta_p * h[p+L]
    
    # For position q (delta_q = +1):
    for L in range(max(q-n+1, -(n-1)), min(q, n-1)+1):
        corr[n-1+L] += delta_q * (1.0 - h[q-L])
    for L in range(max(-q, -(n-1)), min(n-1-q, n-1)+1):
        corr[n-1+L] += -delta_q * h[q+L]
    
    # Corrections for joint terms:
    # If p-q is a valid L: corr[n-1+(p-q)] += -delta_p*delta_q = 1
    L1 = p - q
    if -(n-1) <= L1 <= n-1:
        corr[n-1+L1] += 1.0  # -delta_p*delta_q = -(-1)(1) = 1
    # If q-p is a valid L: corr[n-1+(q-p)] += -delta_q*delta_p = 1
    L2 = q - p
    if -(n-1) <= L2 <= n-1:
        corr[n-1+L2] += 1.0  # -delta_q*delta_p = -(1)(-1) = 1
    
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_binary_feasible(h, target_sum, rng):
    """Convert h to binary with exactly target_sum ones."""
    h = np.round(h).astype(float)
    # Count
    diff = target_sum - h.sum()
    while diff > 0:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros)] = 1
        diff -= 1
    while diff < 0:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones)] = 0
        diff += 1
