    """Project v onto {x : 0 <= x_i <= 1, sum(x_i) = S}"""
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(v - t, 0.0, 1.0)

def c5_gradient(h, t_smooth=100.0):
    """Compute smooth gradient of the C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    vals = corr * dx
    
    # Log-sum-exp for stability
    max_val = np.max(vals)
    exp_vals = np.exp(t_smooth * (vals - max_val))
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] < 1e-10:
            continue
        s = m - (n - 1)
        for j in range(n):
            g = 0.0
            js = j - s
            if 0 <= js < n:
                g += (1.0 - h[js])
            jp = j + s
            if 0 <= jp < n:
                g += -h[jp]
            grad[j] += weights[m] * g * t_smooth * dx

Now the step size should be proportional to 1/L where L is the Lipschitz constant of the gradient. The gradient of f_t is bounded by... well, each ∂corr[m]/∂h[j] is bounded by 2 (since |1-h| ≤ 1 and |h| ≤ 1), and there are at most 2 terms per m. So |∂f_t/∂h[j]| ≤ dx * sum_m w[m] * 2 = 2*dx = 4/n.

The Lipschitz constant of the gradient (the Hessian) is bounded by... it's complicated because the Hessian involves second derivatives of the bilinear terms. Each second derivative is 0 or ±1, and we have the weight w[m]. So the Hessian is bounded by about dx * sum_m w[m] = dx = 2/n.

def compute_c5(h):

def project_capped_simplex(v, S):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid

    # Phase 5: Final local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15:
        h5, _ = local_search(best_h, min(remaining * 0.7, 100.0), rng)
        c5_5 = compute_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    return (best_h, best_c5, len(best_h))
