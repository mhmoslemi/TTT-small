**Coordinate swap algorithm:**
1. For each pair (i, j) with i < j:
   a. Compute a_k = (h[j-k]-h[i-k]) + (h[j+k]-h[i+k]) for all k (with h[m]=0 if m out of range)
   b. Find δ* = argmin_δ max_k (c_k + δ*a_k) subject to δ ∈ [max(-h[i], h[j]-1), min(1-h[i], h[j])]
   c. Update h[i] += δ*, h[j] -= δ*
2. Repeat until no improvement

For step (b), the function f(δ) = max_k (c_k + δ*a_k) is convex and piecewise linear. To find its minimum:
- If the subgradient at δ=0 contains 0 (i.e., there exist active k with a_k > 0 and active l with a_l < 0), we're at the minimum.
- Otherwise, move in the direction that decreases f, to the next breakpoint.

A practical approach: use scipy.optimize.minimize_scalar with method='bounded'.

Let me estimate the cost: for each (i,j) pair, minimize_scalar will take ~20 function evaluations, each O(n) = O(89). So per pair: ~1800 ops. For 990 pairs: ~1.8M ops per full pass. Very fast!

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return np.log(np.sum(np.exp(T * (c - m)))) / T + m

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    exp_c = np.exp(T * (c - m))
    p = exp_c / exp_c.sum()
    obj = np.log(exp_c.sum()) / T + m
    grad = np.zeros(n)
    k_vals = np.arange(-(n - 1), n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for idx in range(len(k_vals)):
            k = k_vals[idx]
            pk = p[idx]
            ik = i - k
            ip = i + k
            if 0 <= ik < n:
                s1 += pk * (1.0 - h[ik])
            if 0 <= ip < n:
                s2 += pk * h[ip]
        grad[i] = dx * (s1 - s2)
    return obj, grad

def swap_objective(delta, b, a):
    return np.max(b + delta * a)

def coordinate_swap_pass(h, n):
    target_sum = n / 2.0
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    b = c * dx
    best_c5 = np.max(b)
    improved = False
    h = h.copy()
    n_pairs = 0
    for i in range(n):
        for j in range(i + 1, n):
            delta_min = max(-h[i], h[j] - 1.0)
            delta_max = min(1.0 - h[i], h[j])
            if delta_max - delta_min < 1e-12:
                continue
            n_pairs += 1
            a = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = idx - (n - 1)
                val = 0.0
                jk = j - k
                ik = i - k
                jp = j + k
                ip = i + k
                if 0 <= jk < n:
                    val += h[jk]
                if 0 <= ik < n:
                    val -= h[ik]
                if 0 <= jp < n:
                    val += h[jp]
                if 0 <= ip < n:
                    val -= h[ip]
                a[idx] = val
            res = minimize_scalar(swap_objective, bounds=(delta_min, delta_max),
                                  args=(b, a), method='bounded', options={'xatol': 1e-11})
            if res.fun < best_c5 - 1e-12:
                h[i] += res.x
                h[j] -= res.x
                best_c5 = res.fun
                improved = True
                c = np.correlate(h, 1.0 - h, mode='full')
                b = c * dx
    return h, best_c5, improved

def slsqp_obj(x, T):
    return soft_max_obj(x, T)[0]

def slsqp_grad(x, T):
    return soft_max_obj(x, T)[1]

def sum_constraint(x, target_sum):
    return np.sum(x) - target_sum

def sum_constraint_jac(x, target_sum):
    return np.ones(len(x))

def make_h_from_seed(seed, n):
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    h = rng.uniform(0, 1, n)
    h += (target_sum - np.sum(h)) / n
    return np.clip(h, 0, 1)
