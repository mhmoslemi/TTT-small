from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_objective_and_grad(h, dx, t=50):
    """Smooth max approximation of C5 and its gradient"""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full") * dx
    max_c = corr.max()
    shifted = t * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_sum = np.exp(shifted).sum()
    weights = np.exp(shifted) / exp_sum
    c5 = max_c + np.log(exp_sum) / t
    
    grad = np.zeros(n)
    # Only process significant weights
    threshold = 1e-8
    for k in range(len(corr)):
        w = weights[k]
        if w < threshold:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1 - 2 * h)
        elif lag > 0:
            grad[:n-lag] += w * (1 - h[lag:])
            grad[lag:] += w * (-h[:n-lag])
        else:
            la = -lag
            grad[la:] += w * (1 - h[:n-la])
            grad[:n-la] += w * (-h[la:])
    grad *= dx
    return c5, grad

    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    lr = 0.1
    stagnation = 0

        c5, grad = smooth_objective_and_grad(h, dx, t)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_c5_grad(h, dx, t=50.0, top_k=30):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_c = corr.max()
    shifted = np.clip(t * (corr - max_c), -500.0, 0.0)
    ev = np.exp(shifted)
    es = ev.sum()
    c5 = max_c + np.log(es) / t
    ti = np.argsort(ev)[-top_k:]
    grad = np.zeros(n)
    for idx in ti:
        w = ev[idx] / es
        if w < 1e-10:
            continue
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n - lag])
        else:
            la = -lag
            grad[la:] += w * (1.0 - h[:n - la])
            grad[:n - la] += w * (-h[la:])
    grad *= dx
    return c5, grad
