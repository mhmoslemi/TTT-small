import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial solution
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(47, 0.5)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    n = len(h0)
    
    # Simulated annealing
    T = 0.1
    T_min = 0.0001
    cooling = 0.999
    
    def evaluate(h):
        return compute_c5(h)
    
    current_h = h0.copy()
    current_c5 = best_c5
    
    no_improve = 0
    max_no_improve = 50000
    
    while time.time() - t0 < budget_s - 10:
        if T < T_min:
            break
        if no_improve > max_no_improve:
            break
            
        # Generate move
        move_type = np.random.random()
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
            
        if move_type < 0.5:
            # Swap
            new_h = current_h.copy()
            new_h[i], new_h[j] = current_h[j], current_h[i]
        elif move_type < 0.8:
            # Transfer
            delta = np.random.uniform(0.01, 0.2)
            if current_h[i] - delta < 0 or current_h[j] + delta > 1:
                continue
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # Block swap
            block_len = np.random.randint(1, max(2, n//4))
            if i + block_len > n or j + block_len > n:
                continue
            new_h = current_h.copy()
            new_h[i:i+block_len], new_h[j:j+block_len] = current_h[j:j+block_len].copy(), current_h[i:i+block_len].copy()
        
        new_c5 = evaluate(new_h)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            
        T *= cooling
    
    return best_h, best_c5, len(best_h)

        diff = n / 2.0 - np.sum(new_h)
        if abs(diff) > 1e-10:
            if diff > 0:
                room = 1.0 - new_h
                order = np.argsort(-room)
            else:
                room = new_h.copy()
                order = np.argsort(-room)
            for idx in order:
                if diff > 0:
                    add = min(diff, room[idx])
                    new_h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, room[idx])
                    new_h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break

        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        h_work = new_h if new_c5 <= compute_c5(h_work) + 0.001 else h_work
        lr *= 0.999

    return best_h

def make_random_binary(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    ones_needed = int(target)
    frac = target - ones_needed
    idx = rng.permutation(n)
    h[idx[:ones_needed]] = 1.0
    if frac > 0:
        h[idx[ones_needed]] = frac
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = make_random_binary(47, rng)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SA from initial (40% budget)
    t1 = t0 + budget_s * 0.40
    h1 = sa_optimize(h_init, t1, rng, T_init=0.06, cooling=0.9995)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient refine (15% budget)
    t2 = t0 + budget_s * 0.55
    if time.time() < t2:
        h2 = gradient_refine(best_h, t2, rng)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: SA from random binary start (20% budget)
    t3 = t0 + budget_s * 0.75
    if time.time() < t3:
        n_rand = len(best_h)
        h_rand = make_random_binary(n_rand, rng)
        h3 = sa_optimize(h_rand, t3, rng, T_init=0.08, cooling=0.9994)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Upsample and optimize (20% budget)
    t4 = t0 + budget_s - 15
    if time.time() < t4 and len(best_h) < 200:
        h_up = upsample(best_h)
        h4 = sa_optimize(h_up, t4, rng, T_init=0.04, cooling=0.9996)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        else:
            h4d = downsample(h4)
            c5_4d = compute_c5(h4d)
            if c5_4d < best_c5:
                best_c5 = c5_4d
                best_h = h4d.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
