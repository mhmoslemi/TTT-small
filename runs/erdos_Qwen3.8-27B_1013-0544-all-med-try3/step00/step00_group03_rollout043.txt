def project_box_simplex(z, s):
    z = np.asarray(z, dtype=float)
    n = z.size
    if not np.all(np.isfinite(z)):
        return np.full(n, s / n)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(50):
        t = 0.5 * (lo + hi)

        if q > s:
            lo = t

def targeted_refine(h0, deadline, seed, max_evals_per_batch=80):
    n = h0.size; s = 0.5*n
    h = project_box_simplex(h0, s)
    f = eval_c5(h)
    best_h = h.copy(); best_f = f
    rng = np.random.default_rng(seed)
    stall = 0
    fracs = (1.0, 0.8, 0.6, 0.4, 0.2)
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        top = choose_top(c, 6)
        moves = []
        for ci in top:
            d = int(ci - (n - 1))
            if d >= 0:
                ia = np.arange(d, n)
            else:
                ia = np.arange(0, n + d)
            if ia.size == 0:
                continue
            ja = ia - d
            contrib = h[ia] * (1.0 - h[ja])
            cm = float(contrib.max())
            if cm <= 1e-8:
                continue
            if contrib.size <= 4:
                ord = np.arange(contrib.size)
            else:
                ord = np.argsort(contrib)[-4:]
            for p in ord:
                i = int(ia[p]); j = int(ja[p])
                for src, dst in ((i, j), (j, i)):
                    md = min(float(h[src]), float(1.0 - h[dst]))
                    if md > 1e-8:
                        for fr in fracs:
                            moves.append((src, dst, fr * md))
        # random moves
        while len(moves) < 30:
            i = int(rng.integers(n)); j = int(rng.integers(n))
            if i == j: continue
            for src, dst in ((i,j),(j,i)):
                md = min(float(h[src]), float(1.0 - h[dst]))
                if md > 1e-8:
                    moves.append((src, dst, float(rng.random()) * md))
        if not moves:
            break
        if len(moves) > max_evals_per_batch:
            moves = moves[:max_evals_per_batch]
        best_move_f = f
        best_move_h = h
        for src, dst, delta in moves:
            if delta <= 0.0: continue
            cand = h.copy()
            cand[dst] += delta
            cand[src] -= delta
            if cand[src] < -1e-12 or cand[dst] < -1e-12 or cand[src] > 1.0+1e-12 or cand[dst] > 1.0+1e-12:
                continue
            np.clip(cand, 0.0, 1.0, out=cand)
            cf = eval_c5(cand)
            if cf < best_move_f:
                best_move_f = cf; best_move_h = cand
        if best_move_f < f - 1e-11:
            h = best_move_h; f = best_move_f; stall = 0
            if f < best_f:
                best_f = f; best_h = h.copy()
        else:
            stall += 1
            if stall > 4:
                h = project_box_simplex(best_h + 0.04 * rng.normal(size=n), s)
                f = eval_c5(h)
                stall = 0
    return best_h, best_f

def binary_sa(n,start_x,end,seed,target_max=None):

import numpy as np
import scipy.optimize as so
import time
import math

def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def project_box_simplex(z, s):
    z = np.asarray(z, dtype=float)
    n = z.size
    if not np.all(np.isfinite(z)):
        return np.full(n, s / n)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(50):
        t = 0.5 * (lo + hi)
        q = float(np.sum(np.clip(z - t, 0.0, 1.0)))
        if q > s:
            lo = t
        else:
            hi = t
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = s - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def choose_top(c, k):
    m = float(np.max(c))
    idx = np.flatnonzero(c >= m - 0.05)
    if idx.size > 2 * k:
        idx = idx[np.argsort(c[idx])[-2 * k:]]
    if idx.size < k:
        idx = np.argsort(c)[-k:]
    return idx

def grad_active(h, idx):
    n = h.size
    g = np.zeros(n, dtype=float)
    for ci in idx:
        d = int(ci - (n - 1))
        if d >= 0:
            g[d:] += 1.0 - h[:n - d]
            g[:n - d] -= h[d:]
        else:
            r = -d
            g[:n - r] += 1.0 - h[r:]
            g[r:] -= h[:n - r]
    return g

def random_binary(n, seed):
    rng = np.random.default_rng(seed)
    x = np.zeros(n, dtype=np.int8)
    x[rng.choice(n, n // 2, replace=False)] = 1
    return x

def fractional_to_binary(h):
    n = h.size
    m = n // 2
    x = np.zeros(n, dtype=np.int8)
    x[np.argsort(-h)[:m]] = 1
    return x

def upsample(h, n):
    m = h.size
    if m == n:
        return h.copy()
    xo = (np.arange(m) + 0.5) * (2.0 / m)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xn, xo, h)

def target_max_for_n(n):
    return int(np.floor(0.38079 * (n / 2.0) - 1e-9))

def binary_sa(n, start_x, end, seed, target_max=None):
    x = np.asarray(start_x, dtype=np.int8).copy()
    m = n // 2
    cnt = int(x.sum())
    if cnt < m:
        x[np.flatnonzero(x == 0)[:m - cnt]] = 1
    elif cnt > m:
        x[np.flatnonzero(x == 1)[:cnt - m]] = 0
    base = np.arange(n, dtype=np.int64)
    idx_rev = n - 1 + base[:, None] - base[None, :]
    idx_fwd = n - 1 - base[:, None] + base[None, :]
    c = np.correlate(x, 1 - x, mode='full').astype(np.int64)
    cur = int(c.max())
    best = cur
    bx = x.copy()
    rng = np.random.default_rng(seed)
    d = np.zeros(2 * n - 1, dtype=np.int64)
    T = 1.0
    stall = 0
    it = 0
    while time.time() < end:
        i = int(rng.integers(n))
        while x[i] == 0:
            i = int(rng.integers(n))
        j = int(rng.integers(n))
        while x[j] == 1:
            j = int(rng.integers(n))
        d.fill(0)
        d[idx_rev[i]] += x - 1
        d[idx_rev[j]] += 1 - x
        d[n - 1 + j - i] += 1
        d[n - 1] -= 1
        d[idx_fwd[i]] += x
        d[n - 1 + j - i] += 1
        d[n - 1] -= 1
        d[idx_fwd[j]] -= x
        d[n - 1 + i - j] += 1
        d[n - 1 + j - i] -= 1
        c += d
        new = int(c.max())
        ok = (new <= cur) or (T > 1e-3 and rng.random() < math.exp(-(new - cur) / T))
        if ok:
            x[i] = 0
            x[j] = 1
            cur = new
            if new < best:
                best = new
                bx = x.copy()
                stall = 0
            else:
                stall += 1
            if target_max is not None and best <= target_max:
                break
        else:
            c -= d
            stall += 1
        it += 1
        if it % 5000 == 0:
            c = np.correlate(x, 1 - x, mode='full').astype(np.int64)
            cur = int(c.max())
            stall = 0
        if stall > 3000:
            x = bx.copy()
            c = np.correlate(x, 1 - x, mode='full').astype(np.int64)
            cur = int(c.max())
            best = cur
            stall = 0
            T = max(0.5, T * 1.5)
        if it % 1000 == 0:
            T = max(0.05, T * 0.95)
    return bx, best

def targeted_refine(h0, end, seed, batch=80):
    n = h0.size
    s = 0.5 * n
    h = project_box_simplex(h0, s)
    f = eval_c5(h)
    bh = h.copy()
    bf = f
    rng = np.random.default_rng(seed)
    stall = 0
    fr = (1.0, 0.8, 0.6, 0.4, 0.2)
    while time.time() < end:
        c = np.correlate(h, 1.0 - h, mode='full')
        top = choose_top(c, 6)
        moves = []
        for ci in top:
            d = int(ci - (n - 1))
            ia = np.arange(d, n) if d >= 0 else np.arange(0, n + d)
            if ia.size == 0:
                continue
            ja = ia - d
            cont = h[ia] * (1.0 - h[ja])
            if float(cont.max()) <= 1e-8:
                continue
            oi = np.argsort(cont)[-4:] if cont.size > 4 else np.arange(cont.size)
            for p in oi:
                i = int(ia[p])
                j = int(ja[p])
                for src, dst in ((i, j), (j, i)):
                    md = min(float(h[src]), float(1.0 - h[dst]))
                    if md > 1e-8:
                        for a in fr:
                            moves.append((src, dst, a * md))
                if h[i] > h[j]:
                    moves.append((i, j, float(h[i] - h[j])))
                elif h[j] > h[i]:
                    moves.append((j, i, float(h[j] - h[i])))
        while len(moves) < 30:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            for src, dst in ((i, j), (j, i)):
                md = min(float(h[src]), float(1.0 - h[dst]))
                if md > 1e-8:
                    moves.append((src, dst, float(rng.random()) * md))
            if h[i] > h[j]:
                moves.append((i, j, float(h[i] - h[j])))
            elif h[j] > h[i]:
                moves.append((j, i, float(h[j] - h[i])))
        if not moves:
            break
        if len(moves) > batch:
            moves = moves[:batch]
        loc_f = f
        loc_h = h
        for src, dst, delta in moves:
            if delta <= 0.0:
                continue
            cand = h.copy()
            cand[dst] += delta
            cand[src] -= delta
            if cand.min() < -1e-12 or cand.max() > 1.0 + 1e-12:
                continue
            np.clip(cand, 0.0, 1.0, out=cand)
            if abs(float(cand.sum()) - s) > 1e-8:
                cand = project_box_simplex(cand, s)
            cf = eval_c5(cand)
            if cf < loc_f:
                loc_f = cf
                loc_h = cand
        if loc_f < f - 1e-11:
            h = loc_h
            f = loc_f
            stall = 0
            if f < bf:
                bf = f
                bh = h.copy()
        else:
            stall += 1
            if stall > 4:
                h = project_box_simplex(bh + 0.04 * rng.normal(size=n), s)
                f = eval_c5(h)
                stall = 0
    return bh, bf

def gradient_refine(h0, end, seed, k=12, step0=0.04):
    n = h0.size
    s = 0.5 * n
    h = project_box_simplex(h0, s)
    f = eval_c5(h)
    bh = h.copy()
    bf = f
    rng = np.random.default_rng(seed)
    step = step0
    stall = 0
    while time.time() < end:
        c = np.correlate(h, 1.0 - h, mode='full')
        g = grad_active(h, choose_top(c, k))
        gm = float(np.max(np.abs(g)))
        if gm < 1e-12:
            break
        scale = step / gm
        loc_f = f
        loc_h = h
        for a in (1.0, 0.5, 0.25, 2.0):
            cand = project_box_simplex(h - a * scale * g, s)
            cf = eval_c5(cand)
            if cf < loc_f:
                loc_f = cf
                loc_h = cand
        if loc_f < f - 1e-11:
            h = loc_h
            f = loc_f
            stall = 0
            step = min(1.0, step * 1.15)
            if f < bf:
                bf = f
                bh = h.copy()
        else:
            stall += 1
            step *= 0.8
            if stall > 60:
                h = project_box_simplex(bh + 0.05 * rng.normal(size=n), s)
                f = eval_c5(h)
                step = step0
                stall = 0
    return bh, bf

SLQ = {}

def slq_cons(x):
    return float(np.sum(x) - SLQ['s'])

def slq_cons_grad(x):
    return np.ones_like(x)

def slq_comp(x):
    x = np.asarray(x, dtype=float)
    corr = np.correlate(x, 1.0 - x, mode='full')
    f = float(np.max(corr)) * (2.0 / x.size)
    g = grad_active(x, choose_top(corr, 10)) * (2.0 / x.size)
    SLQ['g'] = g
    SLQ['x'] = x.copy()
    return f

def slq_obj(x):
    return slq_comp(x)

def slq_jac(x):
    x = np.asarray(x, dtype=float)
    if 'x' in SLQ and np.allclose(x, SLQ['x'], rtol=1e-12, atol=1e-13):
        return SLQ['g']
    slq_comp(x)
    return SLQ['g']

def slsqp_refine(h0, end, seed, maxiter=20):
    n = h0.size
    SLQ['s'] = 0.5 * n
    h = project_box_simplex(h0, SLQ['s'])
    try:
        res = so.minimize(
            slq_obj, h, method='SLSQP', jac=slq_jac,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': slq_cons, 'grad': slq_cons_grad}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        h = project_box_simplex(res.x, SLQ['s'])
    except Exception:
        pass
    return h, eval_c5(h)
