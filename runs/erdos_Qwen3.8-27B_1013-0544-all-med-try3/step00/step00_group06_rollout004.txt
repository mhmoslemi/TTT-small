            rng.shuffle(in_set)
            rng.shuffle(out_set)
            for a in in_set:
                for b in out_set[:20]:
                    new_set = best_bin_set.copy()
                    new_set.discard(a)
                    new_set.add(b)
                    sc = bin_score(new_set)
                    if sc < cur - 1e-12:
                        best_bin_set = new_set
                        cur = sc
                        improved = True
                        if time_left() < 30:
                            break
                if time_left() < 30:
                    break
            best_bin_c5 = cur
            if best_bin_c5 < best_c5:
                best_c5 = best_bin_c5
                h_bin2 = np.zeros(n)
                h_bin2[list(best_bin_set)] = 1.0
                best_h = h_bin2.copy()

    # Phase 6: Final polish with Nelder-Mead
    if time_left() > 30:
        res = minimize(nm_obj, best_h, method='Nelder-Mead',
                       options={'maxiter': 30000, 'xatol': 1e-12, 'fatol': 1e-12})
        h_try = project_box_sum(res.x, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Ensure feasibility
    best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
