    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

    lo, hi = -5.0, 5.0
    for _ in range(60):

            lo = mid
        else:

    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx

    shifted = (corr - max_val) / tau
    exp_shifted = np.exp(shifted)

    weights = exp_shifted / np.sum(exp_shifted)

    h_part = np.zeros((n, 2 * n - 1))

import time

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    max_val = np.max(corr)
    shifted = (corr - max_val) / tau
    exp_sh = np.exp(shifted)
    logsum = max_val + tau * np.log(np.sum(exp_sh))
    weights = exp_sh / np.sum(exp_sh)
    j_idx = np.arange(n)[:, None]
    lags = (np.arange(2 * n - 1) - (n - 1))[None, :]
    idx_g = j_idx + lags
    idx_h = j_idx - lags
    valid_g = (idx_g >= 0) & (idx_g < n)
    valid_h = (idx_h >= 0) & (idx_h < n)
    g_part = np.zeros((n, 2 * n - 1))
    g_part[valid_g] = g[idx_g[valid_g]]
    h_part = np.zeros((n, 2 * n - 1))
    h_part[valid_h] = h[idx_h[valid_h]]
    grad = dx * (weights[None, :] * (g_part - h_part)).sum(axis=1)
    return logsum, grad

def projected_grad(h0, tau_start, tau_end, n_iter, deadline):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(h)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        frac = i / max(n_iter - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = 0.02 * (1.0 + frac * 0.5)

            _, grad = smooth_obj_and_grad(h, tau)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:

            h_new = h - lr * grad
            h_new = project_onto_set(h_new, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best = h_new.copy()
            h = h_new
    return best, best_c5

    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)
