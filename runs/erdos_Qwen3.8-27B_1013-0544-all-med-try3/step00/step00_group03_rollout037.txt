def compute_c5(h):

            c5_try = compute_c5(h_try)
            if c5_try < c5 - 1e-15:
                h = h_try
                c5 = c5_try
                improved = True
                break
            step *= 0.5

        if improved:
            no_improve = 0
            alpha = min(step * 1.05, 50.0)
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_local_h = h.copy()
        else:
            no_improve += 1
            alpha = max(alpha * 0.5, 0.05)
            if no_improve > 80:
                h = h + rng.normal(0.0, 0.08, n)
                h = project(h, tgt)
                c5 = compute_c5(h)
                no_improve = 0
                alpha = 3.0
            elif no_improve > 400:
                break

    # Try binary rounding
    if time.time() < deadline:
        h_bin = round_to_binary(h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_local_c5:
            best_local_c5 = c5_bin
            best_local_h = h_bin.copy()

    return best_local_h, best_local_c5


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    deadline = t_start + budget_s * 0.92
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = 1e9

    starts = []
    n0 = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        starts.append(h_init.copy())
        n0 = len(h_init)

    # Block start
    h_block = np.zeros(n0)
    h_block[:n0 // 2] = 1.0
    starts.append(h_block)

    # Random starts
    for _ in range(5):
        h_r = rng.uniform(0.15, 0.85, n0)
        starts.append(h_r)

    # Perturbed initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        for _ in range(3):
            h_p = h_init + rng.normal(0.0, 0.15, n0)
            starts.append(h_p)

    # Optimize from each start
    for idx, h0 in enumerate(starts):
        if time.time() > deadline:
            break
        n = len(h0)
        tgt = n / 2.0
        h_res, c5_res = optimize_from(h0, tgt, rng, deadline)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Final polish from best solution with fresh budget
    if time.time() < deadline and best_h is not None:
        h_final, c5_final = optimize_from(best_h, len(best_h) / 2.0, rng, deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
