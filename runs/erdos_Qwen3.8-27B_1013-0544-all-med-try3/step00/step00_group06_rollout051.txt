def project_box_sum(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    
    def sum_at_tau(tau):
        return np.sum(np.clip(y - tau, 0, 1))
    
    # Binary search for tau
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_at_tau(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2
    return np.clip(y - tau, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_k = int(np.argmax(corr))
    c5_val = corr[max_k] * dx
    lag = max_k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx1 = np.arange(lag, n)
        grad[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, n - lag)
        grad[idx2] += -h[idx2 + lag]

        idx1 = np.arange(k, n)
        grad[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        grad[idx2] += -h[idx2 + k]
    return c5_val, grad * dx

The log-sum-exp approximation:
f_β(h) = (1/β) log(Σ_k exp(β * c_k))

where c_k = corr_k * dx. As β → ∞, this approaches max(c_k).

The gradient:
∇f_β(h) = Σ_k p_k * ∇c_k where p_k = exp(β*c_k) / Σ_j exp(β*c_j)

This is a weighted average of subgradients, which is a proper gradient (since f_β is smooth).

The computation per gradient evaluation:
- Compute all c_k: O(n²) using np.correlate
- Compute weights p_k: O(n)
- Compute weighted sum of gradients: this is the expensive part

For the weighted sum, I need:
Σ_k p_k * ∇c_k

Where ∇c_k is the gradient of the k-th correlation value. For a given k (with lag l):
∇c_k[j] = (1-h[j-l]) * [l≤j≤n-1] + (-h[j+l]) * [0≤j≤n-1-l] (for l≥0)

So the weighted sum is:
Σ_k p_k * [(1-h[j-l_k]) * [l_k≤j≤n-1] + (-h[j+l_k]) * [0≤j≤n-1-l_k]]

def lse_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    weights = exps / np.sum(exps)
    center = n - 1
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        lag = k - center
        w = weights[k]
        if lag >= 0:
            idx1 = np.arange(lag, n)
            grad[idx1] += w * (1.0 - h[idx1 - lag])
            idx2 = np.arange(0, n - lag)
            grad[idx2] += w * (-h[idx2 + lag])

            grad[idx1] += w * (1.0 - h[idx1 - kk])
            idx2 = np.arange(0, n - kk)
            grad[idx2] += w * (-h[idx2 + kk])
    
    return grad * dx
