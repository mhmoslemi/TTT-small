    h = x
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    off = k_max - (n - 1)
    c5_val = corr[k_max] * dx
    integral = np.sum(h) * dx
    pen = penalty * (integral - 1.0) ** 2
    obj = c5_val + pen
    grad = np.zeros(n)
    if off >= 0:
        # First: grad[0:n-off] += (1 - h[off:n])
        m = n - off
        if m > 0:
            grad[:m] += (1.0 - h[off:off+m])
        # Second: if off>0, grad[off:n] += (-h[0:n-off])
        if off > 0:
            grad[off:] += (-h[:m])
        else:
            grad[:] += (-h[:])
    else:
        a = -off
        # First: grad[a:n] += (1 - h[0:n-a])
        grad[a:] += (1.0 - h[:n-a])
        # Second: grad[0:n-a] += (-h[a:n])
        m = n - a
        if m > 0:
            grad[:m] += (-h[a:])
    grad *= dx
    grad += 2.0 * penalty * (integral - 1.0) * dx
    return obj, grad

    remaining = budget_s - elapsed - 20

    for mult in [2, 3]:
        if remaining < 50:
            break
        n_new = n0 * mult
        dx_new = 2.0 / n_new
        x_old = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
        x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_constraint(h_new, n_new)

        phase_time = min(remaining * 0.5, 250)
        try:
            res = minimize(
                obj_func, h_new, jac=grad_func,
                args=(dx_new, 1e5 * mult, start_time + phase_time),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n_new,
                options={'maxiter': 3000, 'maxfun': 30000}
            )
            hc = project_constraint(np.clip(res.x, 0, 1), n_new)
            c5 = compute_c5(hc, dx_new)
            if c5 < best_c5:
                best_h = hc.copy()
                best_c5 = c5
        except Exception:
            pass
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 20

    # Phase 3: local refinement with perturbations
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 10
    n_best = len(best_h)
    dx_best = 2.0 / n_best

    for attempt in range(4):
        if remaining < 15:
            break
        h_try = best_h.copy()
        if attempt > 0:
            scale = 0.02 / attempt
            pert = rng.randn(n_best) * scale
            h_try = np.clip(h_try + pert, 0, 1)
            h_try = project_constraint(h_try, n_best)

        tlim = min(remaining * 0.35, 120)
        try:
            res = minimize(
                obj_func, h_try, jac=grad_func,
                args=(dx_best, 1e6, start_time + tlim),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n_best,
                options={'maxiter': 4000, 'maxfun': 25000}
            )
            hc = project_constraint(np.clip(res.x, 0, 1), n_best)
            c5 = compute_c5(hc, dx_best)
            if c5 < best_c5:
                best_h = hc.copy()
                best_c5 = c5
        except Exception:
            pass
        remaining = budget_s - (time.time() - start_time) - 10

    return best_h, best_c5, n_best
