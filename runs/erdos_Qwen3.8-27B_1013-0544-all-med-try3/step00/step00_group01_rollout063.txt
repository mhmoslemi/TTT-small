import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def evaluate_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def grad_hard_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    k = int(np.argmax(corr))
    m = k - n + 1
    g = np.zeros(n)
    if m >= 0:
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]
    return g * dx

def grad_smooth_max(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    w = np.exp(t * (corr - mc))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-8:
            continue
        m = k - n + 1
        if m >= 0:
            g[m:] += w[k] * (1.0 - h[:n - m])
            g[:n - m] -= w[k] * h[m:]
        else:
            p = -m
            g[:n - p] += w[k] * (1.0 - h[p:])
            g[p:] -= w[k] * h[:n - p]
    return g * dx

def tm_seq(n):
    a = np.zeros(n)
    for i in range(1, n):
        a[i] = a[i >> 1] ^ (i & 1)
    return a

def adam_optimize(h, n, time_limit, smooth_t):
    h = h.copy()
    bh, bc = h.copy(), evaluate_c5(h)
    v1 = np.zeros(n)
    v2 = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = 0.03
    te = time.time() + time_limit
    count = 0
    cur_t = smooth_t
    while time.time() < te:
        if cur_t > 0.5:
            g = grad_smooth_max(h, cur_t)
        else:
            g = grad_hard_max(h)
        v1 = b1 * v1 + (1.0 - b1) * g
        v2 = b2 * v2 + (1.0 - b2) * g * g
        h = np.clip(h - lr * v1 / (np.sqrt(v2) + eps), 0.0, 1.0)
        h = project_feasible(h, n)
        c = evaluate_c5(h)
        if c < bc:
            bh, bc = h.copy(), c
        count += 1
        if count % 1500 == 0:
            h = bh + np.random.randn(n) * 0.02
            h = project_feasible(h, n)
            cur_t = max(0.5, cur_t * 0.85)
    return bh, bc

def binary_hill_climb(h, n, time_limit):
    h = (h >= 0.5).astype(float)
    tgt = n // 2
    oi = np.where(h == 1)[0]
    zi = np.where(h == 0)[0]
    if len(oi) > tgt:
        h[np.random.choice(oi, len(oi) - tgt, replace=False)] = 0
    elif len(oi) < tgt:
        h[np.random.choice(zi, tgt - len(oi), replace=False)] = 1
    bh, bc = h.copy(), evaluate_c5(h)
    nw = 0
    te = time.time() + time_limit
    while time.time() < te:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        si = oi[np.random.randint(len(oi))]
        sj = zi[np.random.randint(len(zi))]
        h[si], h[sj] = h[sj], h[si]
        c = evaluate_c5(h)
        if c < bc - 1e-15:
            bh, bc = h.copy(), c
            nw = 0
        else:
            h[si], h[sj] = h[sj], h[si]
            nw += 1
            if nw > 4000:
                for _ in range(4):
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    s1 = np.random.choice(oi)
                    s2 = np.random.choice(zi)
                    h[s1], h[s2] = h[s2], h[s1]
                nw = 0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = tm_seq(48)
    n = len(h)
    h = project_feasible(h, n)
    best_h, best_c5 = h.copy(), evaluate_c5(h)

    # Phase 1: Continuous optimization on parent
    el = time.time() - t0
    rem = budget_s * 0.22 - el
    if rem > 5:
        r = adam_optimize(h, n, min(rem, 60), 5.0)
        if r[1] < best_c5:
            best_h, best_c5 = r

    # Phase 2: Binary solutions for even n
    n_list = [48, 50, 52, 54, 56, 60, 64, 72, 80]
    for nc in n_list:
        el = time.time() - t0
        rem = budget_s * 0.68 - el
        if rem < 4:
            break
        tl = min(rem / 12.0, 40.0)
        if tl < 3:
            break
        r = binary_hill_climb(tm_seq(nc), nc, tl)
        if r[1] < best_c5:
            best_h, best_c5 = r
        el = time.time() - t0
        rem = budget_s * 0.68 - el
        if rem > 4:
            r = binary_hill_climb(np.random.rand(nc), nc, tl)
            if r[1] < best_c5:
                best_h, best_c5 = r

    # Phase 3: Upscale best to larger n and refine
    el = time.time() - t0
    rem = budget_s * 0.12 - el
    if rem > 5:
        bn = len(best_h)
        new_n = bn * 2
        if new_n <= 200:
            old_idx = np.linspace(0, bn - 1, new_n)
            left = np.floor(old_idx).astype(int)
            right = np.minimum(left + 1, bn - 1)
            frac = old_idx - left
            h_up = best_h[left] * (1 - frac) + best_h[right] * frac
            h_up = project_feasible(h_up, new_n)
            r = adam_optimize(h_up, new_n, rem, 3.0)
            if r[1] < best_c5:
                best_h, best_c5 = r

    # Phase 4: Final refinement
    el = time.time() - t0
    rem = budget_s * 0.05 - el
    if rem > 3:
        r = adam_optimize(best_h, len(best_h), rem, 2.0)
        if r[1] < best_c5:
            best_h, best_c5 = r

    best_h = np.clip(best_h, 0.0, 1.0)
    bn = len(best_h)
    best_h = project_feasible(best_h, bn)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, bn
