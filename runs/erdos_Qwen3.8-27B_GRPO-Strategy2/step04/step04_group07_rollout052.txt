def compute_c5_indices(h, n):
    """Compute C5 value and the lag index where max is achieved."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = np.argmax(corr)
    return float(corr[k_max]), k_max, corr

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)

            if abs(d) < 1e-15:
                break

                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def subgrad_step(h, n, k_active, step, corr):
    """Take a subgradient step at the active lag."""
    dx = 2.0 / n
    lag = k_active - (n - 1)
    # Gradient of corr[k_active] * dx wrt h
    grad = np.zeros(n)
    # First term: (1-h[i-lag]) for valid i
    # h[i+lag] * (1-h[i]): 
    # ∂/∂h[j] of sum_i h[i+lag]*(1-h[i])
    # = (1-h[j-lag]) * I[j-lag in [0,n-1]) - h[j+lag] * I[j+lag in [0,n-1])
    
    # For j-lag in [0, n-1]: i = j-lag, term is (1-h[j-lag])
    # For j+lag in [0, n-1]: i = j, term is -h[j+lag]
    
    if lag >= 0:
        # j-lag in [0,n-1] means j in [lag, n-1+lag], intersect [0,n-1] = [lag, n-1]
        # j+lag in [0,n-1] means j in [-lag, n-1-lag], intersect [0,n-1] = [0, n-1-lag]
        for j in range(lag, n):
            grad[j] += (1.0 - h[j - lag]) * dx
        for j in range(0, n - lag):
            grad[j] += (-h[j + lag]) * dx
    else:
        # lag < 0, let l = -lag
        l = -lag
        # j-lag = j+l in [0,n-1] means j in [-l, n-1-l], intersect [0,n-1] = [0, n-1-l]
        # j+lag = j-l in [0,n-1] means j in [l, n-1+l], intersect [0,n-1] = [l, n-1]
        for j in range(0, n - l):
            grad[j] += (1.0 - h[j + l]) * dx
        for j in range(l, n):
            grad[j] += (-h[j - l]) * dx
    
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair(h_new, n)
    return h_new

def vectorized_grad(h, n, lag):
    """Compute gradient of correlation at given lag, vectorized."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        # h[i+lag]*(1-h[i]) for i in [0, n-1-lag]
        # ∂/∂h[j]: 
        #   as h[j] in first factor: (1-h[j-lag]) for j in [lag, n-1]
        #   as h[j] in second factor: -h[j+lag] for j in [0, n-1-lag]
        valid1 = slice(lag, n)
        valid2 = slice(0, n - lag)
        grad[valid1] += (1.0 - h[:n-lag]) * dx
        grad[valid2] += (-h[lag:]) * dx
    else:
        l = -lag
        # h[i-l]*(1-h[i]) for i in [l, n-1]
        # ∂/∂h[j]:
        #   as h[j] in first factor (j = i-l, i=j+l): (1-h[j+l]) for j in [0, n-1-l]
        #   as h[j] in second factor: -h[j-l] for j in [l, n-1]
        valid1 = slice(0, n - l)
        valid2 = slice(l, n)
        grad[valid1] += (1.0 - h[l:]) * dx
        grad[valid2] += (-h[:n-l]) * dx
    return grad

def soft_plus_approx(h, n, alpha):

- As (1-h[j]) = (1-h[i]) when j = i: contribution requires 0 ≤ i < n (true) and 0 ≤ i+l < n. So if 0 ≤ i+l < n: corr[l] includes h[i+l]*(1-h[i])

Let b_l = I[0←i+l<n] * h[i+l]

Then corr[l] = const_l + a_l * h[i] - b_l * h[i] = const_l + (a_l - b_l) * h[i]

The max over l of (const_l + slope_l * h[i]) is a piecewise linear, convex function of h[i]. Its minimum on [0,1] is at a point where the leftmost increasing line meets the rightmost decreasing line, or at a boundary.

def soft_obj(v, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_opt(h, n, alpha, maxiter=300):
    def obj(v):
        return soft_obj(v, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def grad_at_lag(h, n, l):
    dx = 2.0 / n
    grad = np.zeros(n)
    if l >= 0:
        grad[l:] += (1.0 - h[:n - l]) * dx
        grad[:n - l] += (-h[l:]) * dx
    else:
        m = -l
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] += (-h[:n - m]) * dx
    return grad

def coord_optimal(h, n, n_passes, t_end, rng):
    h = h.copy()
    best_c5, _ = compute_c5(h, n)
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            hi = h[i]
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = hi
            slope = corr1 - corr0
            intercept = corr0
            t_grid = np.linspace(0.0, 1.0, 256)
            vals = np.max(intercept + np.outer(t_grid, slope), axis=1)
            t_best = t_grid[np.argmin(vals)]
            lo = max(0.0, t_best - 0.004)
            hi_b = min(1.0, t_best + 0.004)
            t_fine = np.linspace(lo, hi_b, 128)
            vals_f = np.max(intercept + np.outer(t_fine, slope), axis=1)
            t_best = t_fine[np.argmin(vals_f)]
            if abs(t_best - hi) > 1e-10:
                h[i] = t_best
                c5_new, _ = compute_c5(h, n)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        h = repair(h, n)
        if not improved:
            break
    return h, best_c5

def subgrad_direct(h, n, max_iter, t_end, rng):
    h = h.copy()
    best_c5, _ = compute_c5(h, n)
    best_h = h.copy()
    step = 0.05
    patience = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c5_now, corr = compute_c5(h, n)
        thresh = c5_now - 1e-10
        active = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        count = 0
        for k_a in active[:4]:
            lag = k_a - (n - 1)
            grad += grad_at_lag(h, n, lag)
            count += 1
        grad /= count
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        c5_new, _ = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            step = min(step * 1.03, 0.3)
            patience = 0
        else:
            step *= 0.93
            patience += 1
            if step < 1e-9 or patience > 40:
                break
    return best_h, best_c5

def make_sine(n, freq, phase, amp):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(freq * np.pi * x + phase)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7777)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5, _ = compute_c5(init_h, init_n)
        best_n = init_n
    except NameError:
        pass

    n = best_n
    h_work = best_h.copy()

    # Phase 1: SLSQP with increasing alpha
    t_p1 = t0 + budget_s * 0.22
    for alpha in [3000, 15000, 60000, 250000]:
        if elapsed() > t_p1:
            break
        h_new = slsqp_opt(h_work.copy(), n, alpha, 400)
        c5_new, _ = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 2: Optimal coordinate descent (key innovation)
    t_p2 = t0 + budget_s * 0.48
    h_cd, c5_cd = coord_optimal(h_work, n, 500, t_p2, rng)
    if c5_cd < best_c5 - 1e-14:
        best_h = h_cd.copy()
        best_c5 = c5_cd
        h_work = h_cd

    # Phase 3: Direct subgradient on the max
    t_p3 = t0 + budget_s * 0.68
    h_sg, c5_sg = subgrad_direct(h_work, n, 5000, t_p3, rng)
    if c5_sg < best_c5 - 1e-14:
        best_h = h_sg.copy()
        best_c5 = c5_sg
        h_work = h_sg

    # Phase 4: More coordinate descent
    t_p4 = t0 + budget_s * 0.84
    h_cd2, c5_cd2 = coord_optimal(h_work, n, 500, t_p4, rng)
    if c5_cd2 < best_c5 - 1e-14:
        best_h = h_cd2.copy()
        best_c5 = c5_cd2
        h_work = h_cd2

    # Phase 5: Alternative n values
    t_p5 = t0 + budget_s * 0.94
    for n_alt in [48, 56, 80, 96, 128]:
        if elapsed() > t_p5:
            break
        x_old = np.linspace(0.0, 1.0, best_n)
        x_new = np.linspace(0.0, 1.0, n_alt)
        h_seed = np.interp(x_new, x_old, best_h)
        h_seed = repair(h_seed, n_alt)
        h_alt = slsqp_opt(h_seed, n_alt, 50000, 300)
        h_alt, c5_alt = coord_optimal(h_alt, n_alt, 80, t_p5, rng)
        if c5_alt < best_c5 - 1e-14:
            best_h = h_alt.copy()
            best_c5 = c5_alt
            best_n = n_alt

    # Phase 6: Final SLSQP + coord polish
    if elapsed() < budget_s * 0.99:
        h_fin = slsqp_opt(best_h.copy(), best_n, 800000, 400)
        c5_fin, _ = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_fin2, c5_fin2 = coord_optimal(best_h, best_n, 200, t0 + budget_s * 0.995, rng)
        if c5_fin2 < best_c5 - 1e-14:
            best_h = h_fin2.copy()
            best_c5 = c5_fin2

    if best_h is None:
        best_n = 64
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5, _ = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5, _ = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
