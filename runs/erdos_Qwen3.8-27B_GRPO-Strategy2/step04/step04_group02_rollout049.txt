    h = np.clip(h, 0.0, 1.0)

    if s < 1e-12:

                h[idx] += add
                diff -= add

        it += 1  # for cooling

from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def g_vec(h, i, n, k):
    ip = i + k
    im = i - k
    G = np.zeros(2 * n - 1)
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    G[vp] = h[ip[vp]]
    G[vm] += h[im[vm]]
    return G

def greedy_search(h, n, rng, deadline, scales, iters_per_scale):
    dx = 2.0 / n
    k = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    for scale in scales:
        improved = False
        for _ in range(iters_per_scale):
            if time.time() >= deadline:
                return best_h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            Gi = g_vec(h, i, n, k)
            Gj = g_vec(h, j, n, k)
            new_corr = corr + d * (Gj - Gi)
            nc5 = float(np.max(new_corr)) * dx
            if nc5 < best_c5 - 1e-15:
                h[i] = ni
                h[j] = nj
                corr = new_corr
                best_c5 = nc5
                best_h = h.copy()
                improved = True
        if not improved and scale < 0.003:
            break
    return best_h, best_c5

def sa_search(h, n, rng, deadline, T_init, T_end, cool, step_scale):
    dx = 2.0 / n
    k = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_init
    step = step_scale
    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * (2.0 * rng.random() - 1.0)
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        Gi = g_vec(h, i, n, k)
        Gj = g_vec(h, j, n, k)
        new_corr = corr + d * (Gj - Gi)
        nc5 = float(np.max(new_corr)) * dx
        dc = nc5 - cur_c5
        acc = dc < 0 or (T > 1e-8 and rng.random() < np.exp(-dc / T))
        if acc:
            h[i] = ni
            h[j] = nj
            corr = new_corr
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 3000:
            step *= 0.7
            stall = 0
        T = max(T_end, T * cool)
    return best_h, best_c5

def block_search(h, n, rng, deadline, n_iter):
    dx = 2.0 / n
    k = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    for _ in range(n_iter):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (0 <= h[i+di]+eps <= 1) or not (0 <= h[j+di]-eps <= 1):
                ok = False
                break
        if not ok:
            continue
        delta = np.zeros(2*n-1)
        for di in range(bs):
            Gi = g_vec(h, i+di, n, k)
            Gj = g_vec(h, j+di, n, k)
            delta += eps * (Gj - Gi)
        new_corr = corr + delta
        nc5 = float(np.max(new_corr)) * dx
        if nc5 < best_c5 - 1e-15:
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            corr = new_corr
            best_c5 = nc5
            best_h = h.copy()
    return best_h, best_c5

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n//8, 7*n//8))
    elif kind == 1:
        f = rng.randint(1, 7)
        ph = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.02, 0.10)
        hp += amp * np.sin(2*np.pi*f*np.arange(n)/n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.035, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 6)
        t = np.arange(n)/n
        hp += rng.uniform(0.03, 0.09)*np.sin(2*np.pi*f1*t)*np.sin(2*np.pi*f2*t+rng.uniform(0,6.28))
    else:
        hp *= rng.uniform(0.8, 1.2)
    return repair(hp, n)

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    dx = 2.0/n
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = np.max(c)
        s = np.clip(alpha*(c-m), -400, 400)
        return m + np.log(np.sum(np.exp(s)))/alpha + pen*(np.sum(x)-target)**2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0,1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        return repair(res.x, n), compute_c5(res.x, n)
    except Exception:
        return h, compute_c5(h, n)
