import time

    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def repair(h, n):

                diff -= add

- Where h[i] is large AND h[i+m] is large: this actually helps (product of h[i]*(1-h[i+m]) is small)

- If h[i]=1 and h[i+m]=0: contributes 1 (bad)
- If h[i]=0 and h[i+m]=1: contributes 0 (good)
- If h[i]=1 and h[i+m]=1: contributes 0 (good)
- If h[i]=0 and h[i+m]=0: contributes 0 (good)

So the "bad" configurations are where h[i] is high and h[i+m] is low.

    if cc < best_c5:
        best_h, best_c5 = hc.copy(), cc

    # Phase 3: Targeted simulated annealing - main innovation (350s)
    t_ph = time.time() + 350
    hA, cA = targeted_anneal(hc, n, dx, rng, time.time(), 300, T0=0.0004)
    if cA < best_c5:
        best_h, best_c5 = hA.copy(), cA

    # Phase 4: L-BFGS from annealed (80s)
    t_ph = time.time() + 80
    if time.time() < t_ph:
        for alpha in [5000, 30000, 150000]:
            if time.time() >= t_ph:
                break
            hh, cc2 = lbfgs_opt(hA, n, alpha, target, 8000, 400)
            if cc2 < cA:
                hA, cA = hh, cc2
        if cA < best_c5:
            best_h, best_c5 = hA.copy(), cA

    # Phase 5: SLSQP ultra-high alpha (70s)
    t_ph = time.time() + 70
    if time.time() < t_ph:
        for alpha in [10000, 100000, 500000]:
            if time.time() >= t_ph:
                break
            hh, cc2 = slsqp_opt(hA, n, alpha, target, 800)
            if cc2 < cA:
                hA, cA = hh, cc2
        if cA < best_c5:
            best_h, best_c5 = hA.copy(), cA

    # Phase 6: Second anneal from best with different seed (120s)
    t_rem = t_start + budget_s - time.time()
    if t_rem > 180:
        t_ph = time.time() + min(120, t_rem - 80)
        rng2 = np.random.RandomState(seed + 100)
        hB, cB = targeted_anneal(best_h, n, dx, rng2, time.time(), t_ph - time.time(), T0=0.00015)
        if cB < best_c5:
            best_h, best_c5 = hB.copy(), cB

    # Phase 7: Fine local search (remainder)
    t_rem = t_start + budget_s - time.time()
    if t_rem > 40:
        t_ph = time.time() + min(t_rem - 15, 200)
        rng3 = np.random.RandomState(seed + 200)
        hF, cF = fine_search(best_h, n, dx, rng3, t_ph)
        if cF < best_c5:
            best_h, best_c5 = hF.copy(), cF

    # Final: L-BFGS tiny polish
    t_rem = t_start + budget_s - time.time()
    if t_rem > 20:
        hh, cc2 = lbfgs_opt(best_h, n, 200000, target, 20000, 200)
        if cc2 < best_c5:
            best_h, best_c5 = hh.copy(), cc2

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
