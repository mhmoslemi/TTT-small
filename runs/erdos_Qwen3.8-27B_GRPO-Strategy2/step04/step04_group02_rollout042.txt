Valid n: 0 <= n < n0 (where n0 is length) AND 0 <= n+lag < n0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def grad_smooth_max(h, n, alpha=5000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    npts = len(h)
    for m_idx in range(2 * npts - 1):
        lag = m_idx - (npts - 1)
        wk = w[m_idx]
        if wk < 1e-14:
            continue
        lo_h = max(0, lag)
        hi_h = min(npts, npts + lag)
        if lo_h < hi_h:
            idx = np.arange(lo_h, hi_h)
            grad[lo_h:hi_h] += wk * (1.0 - h[idx - lag])
        lo_g = max(0, -lag)
        hi_g = min(npts, npts - lag)
        if lo_g < hi_g:
            idx = np.arange(lo_g, hi_g)
            grad[lo_g:hi_g] -= wk * h[idx + lag]
    return grad

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sum_con(x, target):
    return np.sum(x) - target

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def grad_descend(h, n, target, rng, deadline, n_iters):
    best_c5 = compute_c5(h, 2.0/n)
    dx = 2.0/n
    h = h.copy()
    step = 0.03
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        if it % 500 == 0:
            step *= 0.97
        g = grad_smooth_max(h, n, alpha=8000)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        direction = -g / gn
        h_try = h + step * direction
        h_try = repair(h_try, n)
        c_try = compute_c5(h_try, dx)
        if c_try < best_c5 - 1e-15:
            h = h_try
            best_c5 = c_try
        else:
            h_try2 = h + step * 0.2 * direction
            h_try2 = repair(h_try2, n)
            c_try2 = compute_c5(h_try2, dx)
            if c_try2 < best_c5 - 1e-15:
                h = h_try2
                best_c5 = c_try2
                step *= 0.6
            else:
                step *= 0.85
        if rng.random() < 0.25:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                d = rng.uniform(-0.015, 0.015)
                ni, nj = h[i]+d, h[j]-d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = compute_c5(h, dx)
                    if c < best_c5:
                        best_c5 = c
                    else:
                        h[i], h[j] = ni-d, nj+d
    return h, best_c5

def anneal(h, n, dx, rng, deadline, T0, T1):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    t0 = time.time()
    total = max(deadline - t0, 1.0)
    best_T = T0
    stagnant = 0
    while time.time() < deadline:
        frac = (time.time() - t0) / total
        T = T0 * (T1/T0) ** frac
        T = max(T, T1)
        r = rng.random()

            if i == j: continue
            d = rng.uniform(-0.06, 0.06)
            ni, nj = h[i]+d, h[j]-d
            if not (0<=ni<=1 and 0<=nj<=1): continue
            h[i], h[j] = ni, nj
            c = compute_c5(h, dx)
            dl = c - best_c5
            if dl < 0 or rng.random() < np.exp(-dl/max(T,1e-12)):
                if c < best_c5:
                    best_c5 = c; best_T = T; stagnant = 0
                else: stagnant += 1
            else:
                h[i], h[j] = ni-d, nj+d
        elif r < 0.65:
            bs = rng.randint(2, 6)
            i = rng.randint(0, n-bs)
            j = rng.randint(0, n-bs)
            if abs(i-j) < bs: continue
            eps = rng.uniform(-0.025, 0.025)
            if np.all((h[i:i+bs]+eps>=-1e-9)&(h[i:i+bs]+eps<=1+1e-9)&(h[j:j+bs]-eps>=-1e-9)&(h[j:j+bs]-eps<=1+1e-9)):
                h[i:i+bs]+=eps; h[j:j+bs]-=eps
                c = compute_c5(h, dx)
                dl = c - best_c5
                if dl < 0 or rng.random() < np.exp(-dl/max(T,1e-12)):
                    if c < best_c5: best_c5=c; stagnant=0
                    else: stagnant+=1
                else:
                    h[i:i+bs]-=eps; h[j:j+bs]+=eps
        else:
            g = grad_smooth_max(h, n, 3000)

            if gn > 1e-13:
                sp = rng.uniform(0.01, 0.06)

                dl = c - best_c5

                    h = ht
                    if c < best_c5: best_c5=c; stagnant=0
                    else: stagnant+=1
        if stagnant > 8000:
            T = best_T * 0.5
            stagnant = 0
    return h, best_c5

def fine_local(h, n, dx, rng, deadline):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for scale in [0.004, 0.0015, 0.0006, 0.0002, 0.00007]:

                return h, best_c5

            else:

def perturb_struct(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(n//8, 7*n//8))
    elif kind == 1:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 2*np.pi)
        hp += rng.uniform(0.02, 0.08)*np.sin(2*np.pi*f*np.arange(n)/n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.03, n)
    elif kind == 3:
        k = rng.randint(1, 5)
        hp[k:] += hp[:k]
        hp[:k] += hp[-k:]
    else:
        mask = rng.random(n) < 0.15
        hp[mask] += rng.normal(0, 0.06, mask.sum())
    return repair(hp, n)
