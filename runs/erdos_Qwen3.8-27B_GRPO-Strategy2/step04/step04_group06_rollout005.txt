import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort((-1.0 + h) if diff > 0 else h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _slsqp_obj(x, n, alpha, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, n)

    # Phase 1: SLSQP polish (up to 80s)
    dx = 2.0 / n
    s_end = t0 + min(80.0, budget_s * 0.12)
    for alpha in [50000, 200000]:
        if time.time() > s_end:
            break
        try:
            res = minimize(_slsqp_obj, h, args=(n, alpha, dx),
                           method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 200, 'ftol': 1e-18})
            h2 = _repair(res.x, n)
            c2 = _c5(h2, n)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2
                h = h2
        except Exception:
            pass

    # Phase 2: Simulated Annealing with directed moves (up to 70% of budget)
    sa_end = t0 + budget_s * 0.78
    T0 = 0.004
    T_floor = 5e-6
    step0 = 0.025
    c5_cur = best_c5
    it = 0
    # Precompute index arrays for directed moves
    n_half = n // 2

    while time.time() < sa_end:
        frac = min(1.0, (time.time() - t0) / (budget_s * 0.78))
        T = max(T0 * (1.0 - frac) ** 2.5 + T_floor, T_floor)
        step = step0 * (T / T0) ** 0.4

        # Choose move type: 60% random, 40% directed (lag-targeted)
        if rng.random() < 0.4 and it > 1000:
            # Find current max lag and target pairs
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_star = int(np.argmax(corr))
            lag = k_star - (n - 1)
            # Pick i uniformly, j = (i + lag) % n
            i = rng.randint(n)
            j = (i + lag) % n
            if i == j:
                j = (j + 1) % n
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue

        d = rng.uniform(-step, step)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:

        else:

        it += 1

                        h[a] += dk
                        h[b] -= dk

    # Phase 3: Deterministic fine search with adaptive deltas
    end_t = t0 + budget_s - 5

                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
            if not improved:
                break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 4: Final SLSQP on best
    if time.time() < end_t - 10:
        try:
            h_try = _repair(best_h + 0.005 * rng.randn(n), n)
            res = minimize(_slsqp_obj, h_try, args=(n, 500000, dx),
                           method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-18})
            h2 = _repair(res.x, n)
            c2 = _c5(h2, n)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
