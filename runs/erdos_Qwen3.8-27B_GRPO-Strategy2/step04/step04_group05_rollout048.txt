        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:

        g[:n - L] -= h[L:]
    else:

        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g
