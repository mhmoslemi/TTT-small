    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha

def _lbfgs_polish(h, target, dx, alpha, maxiter, t_end):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, target), h,
                     method='L-BFGS-B', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        hc = _repair(r.x, target)
        if time.time() < t_end:
            return hc, _c5(hc, dx)
    except Exception:
        pass
    return h, _c5(h, dx)

def _slsqp_polish(h, target, dx, alpha, maxiter):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, target), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _gen_structural(n, freq, rng):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / 2.0 + rng.uniform(0, 2 * np.pi))
    h += rng.normal(0, 0.05, n)
    return _repair(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
        except Exception:
            h = _gen_structural(140, 3, rng)
    n = len(h)
    dx = 2.0 / n
    tgt = n / 2.0
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: Subgradient at n (180s)
    te = t0 + 180
    best_h, best_c5 = _subgrad_loop(best_h, tgt, dx, te, 3000)

    # Phase 2: Targeted swaps (180s)
    te = t0 + 360
    best_h, best_c5 = _targeted_swap(best_h, tgt, dx, rng, te, 80000)

    # Phase 3: L-BFGS-B polish (120s)
    te = t0 + 480
    for alpha in [300, 1000, 3000]:
        if time.time() > te:
            break
        hc, c = _lbfgs_polish(best_h, tgt, dx, alpha, 400, te)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 4: SLSQP polish (100s)
    te = t0 + 580
    for alpha in [5000, 20000, 60000]:
        if time.time() > te:
            break
        hc, c = _slsqp_polish(best_h, tgt, dx, alpha, 500)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 5: Targeted swap again (80s)
    te = t0 + 660
    best_h, best_c5 = _targeted_swap(best_h, tgt, dx, rng, te, 100000)

    # Phase 6: Multi-restart (120s)
    te = t0 + 780
    for trial in range(6):
        if time.time() > te:
            break
        if trial == 0:
            hp = _gen_structural(n, 2 + trial + rng.randint(0, 3), rng)
        elif trial == 1:
            hp = _gen_structural(n, 4 + trial, rng)
        else:
            hp = best_h + 0.06 * (2.0 * rng.random(n) - 1.0)
            hp = _repair(hp, tgt)
        hp, c = _subgrad_loop(hp, tgt, dx, min(t0 + 780, time.time() + 15), 800)
        hp, c = _targeted_swap(hp, tgt, dx, rng, min(t0 + 780, time.time() + 20), 15000)
        if c < best_c5:
            best_h, best_c5, best_n = hp, c, n

    # Phase 7: Interpolate to higher resolution and polish (180s)
    n2 = 2 * best_n
    if n2 <= 560:
        dx2 = 2.0 / n2
        tgt2 = n2 / 2.0
        h2 = _repair(_interp(best_h, n2), tgt2)
        c2 = _c5(h2, dx2)
        te = t0 + 960
        for alpha in [500, 2000, 8000]:
            if time.time() > te:
                break
            h2, c2 = _lbfgs_polish(h2, tgt2, dx2, alpha, 500, te)
        if time.time() < te:
            h2, c2 = _slsqp_polish(h2, tgt2, dx2, 30000, 800)
        if time.time() < te:
            h2, c2 = _targeted_swap(h2, tgt2, dx2, rng, min(te, time.time() + 40), 30000)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 8: Final intensive targeted swap (100s)
    dxb = 2.0 / best_n
    te = t0 + budget_s - 20
    best_h, best_c5 = _targeted_swap(best_h, best_n / 2.0, dxb, rng, te, 200000)

    # Final repair and evaluation
    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
