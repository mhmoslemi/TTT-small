    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-14:
        return h
    if s > 1e-15:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-14:
        return h
    if s > 1e-15:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def gc(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

_G = {}

def so(x):
    n, a, l = _G['n'], _G['a'], _G['l']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / a + l * (x.sum() - n * 0.5) ** 2

    n, a, l = _G['n'], _G['a'], _G['l']

    m = c.max()
    s = np.clip(a * (c - m), -400, 400)

    g = np.zeros(n)

        if pw[k] > 1e-12:
            g += pw[k] * gc(x, k, n)

        except Exception:

    return bh, bc
