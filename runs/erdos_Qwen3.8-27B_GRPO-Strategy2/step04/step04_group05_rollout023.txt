import numpy as np
from scipy.optimize import minimize
import time

_GR = {'n': 0, 'a': 100.0, 'lam': 50.0}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def sm_obj(x):
    n, a, lam = _GR['n'], _GR['a'], _GR['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + lam * (x.sum() - n * 0.5) ** 2

def sm_jac(x):
    n, a, lam = _GR['n'], _GR['a'], _GR['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-13:
            g += pw[k] * grad1(x, k, n)
    g /= a
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def eq_fun(x):
    return x.sum() - _GR['n'] * 0.5

def eq_jac(x):
    return np.ones(_GR['n'])

def lbfgs_run(h, n, dx, t0, lim):
    _GR['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    for a, lam, mx in [(20, 30, 500), (100, 80, 400), (500, 50, 350),
                        (2000, 20, 350), (8000, 5, 300), (30000, 1, 300),
                        (100000, 0.1, 250), (500000, 0.01, 200),
                        (2000000, 0.001, 150)]:
        if time.time() - t0 > lim:
            break
        _GR['a'], _GR['lam'] = float(a), float(lam)
        try:
            r = minimize(sm_obj, bh.copy(), jac=sm_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': mx, 'ftol': 1e-22,
                                               'gtol': 1e-18, 'maxcor': 50})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def active_subgrad(h, n, dx, t0, lim):

    for ntop in [3, 5, 10, 20, 40, 80]:
        h = bh.copy()
        step = 0.06
        stall = 0
