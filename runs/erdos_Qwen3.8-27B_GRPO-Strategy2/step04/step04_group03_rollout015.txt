            if i1 < i2:
                g[i1:i2] += wk * (1.0 - h[i1 - lag:i2 - lag])
            i3, i4 = max(0, -lag), min(n, n - lag)
            if i3 < i4:
                g[i3:i4] -= wk * h[i3 + lag:i4 + lag]
        else:
            L = -lag
            i1, i2 = max(0, lag), min(n, n + lag)
            if i1 < i2:
                g[i1:i2] += wk * (1.0 - h[i1 - lag:i2 - lag])
            i3, i4 = max(0, -lag), min(n, n - lag)
            if i3 < i4:
                g[i3:i4] -= wk * h[i3 + lag:i4 + lag]
    return g * dx

def _c_sum(h, n):
    return np.sum(h) - n / 2.0

def _c_jac(h, n):
    return np.ones(n)

def _slsqp_run(h0, n, alpha, maxiter, t0, budget):
    if time.time() > t0 + budget - 30:
        return h0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_sum, 'jac': _c_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h0.copy(), jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h0

def _fine_search(h, n, rng, t_end):
    c5c = _eval(h)
    scale = 0.005
    no_imp = 0
    for _ in range(500000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.03, 0.02)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 500:
                scale *= 0.8
                no_imp = 0
                if scale < 2e-6:
                    break
    return h, c5c

def _top_lag_refine(h, n, rng, t0, budget, top_k=5):
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c_full)[-top_k:]
    h_cur = h.copy()
    c5c = _eval(h_cur)
    scale = 0.003
    no_imp = 0
    for _ in range(200000):
        if time.time() > t0 + budget - 20:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c2 = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        c5n = max(c2[idx] for idx in top_idx)
        if c5n < c5c - 1e-16:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.03, 0.015)
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 400:
                scale *= 0.8
                no_imp = 0
                if scale < 1e-6:
                    break
    return h_cur, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _tl():
        return budget_s - (time.time() - t0)

    h160 = _repair(np.array(initial_h_values, dtype=float), 160)
    c160 = _eval(h160)
    best_h, best_c5, best_n = h160.copy(), c160, 160

    n_list = [160, 120, 100, 140, 200]
    inits = {160: h160}
    for n in n_list:
        if n in inits:
            continue
        hc = _interp(h160, 160, n)
        inits[n] = _repair(hc, n)

    # Phase 1: SLSQP optimization at each resolution
    for n in n_list:
        if _tl() < 80:
            break
        h0 = inits[n]
        c0 = _eval(h0)
        if c0 > best_c5 + 0.03:
            continue
        for alpha in [500, 5000, 50000, 200000]:
            if _tl() < 60:
                break
            h_new = _slsqp_run(h0, n, alpha, 400, t0, budget_s)
            c_new = _eval(h_new)
            if c_new < c0 - 1e-14:
                c0 = c_new
                h0 = h_new
        if c0 < best_c5:
            best_c5, best_h, best_n = c0, h0.copy(), n

    # Phase 2: Multi-start perturbation from best
    if _tl() > 120:
        for pert in [0.02, 0.01, 0.005]:
            if _tl() < 100:
                break
            h_p = best_h + pert * rng.randn(best_n)
            h_p = _repair(h_p, best_n)
            c_p = _eval(h_p)
            if c_p < best_c5 + 0.002:
                for alpha in [50000, 200000]:
                    if _tl() < 60:
                        break
                    h_new = _slsqp_run(h_p, best_n, alpha, 300, t0, budget_s)
                    c_new = _eval(h_new)
                    if c_new < best_c5:
                        best_c5, best_h, best_n = c_new, h_new, best_n
                    elif c_new < c_p:
                        c_p, h_p = c_new, h_new

    # Phase 3: High-alpha SLSQP refinement
    if _tl() > 60:
        h_ref = best_h.copy()
        for alpha in [500000, 1000000, 2000000]:
            if _tl() < 40:
                break
            h_new = _slsqp_run(h_ref, best_n, alpha, 500, t0, budget_s)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new
                h_ref = h_new.copy()

    # Phase 4: Top-lag focused refinement
    if _tl() > 40:
        t_end = min(time.time() + 30, t0 + budget_s - 25)
        h_t, c_t = _top_lag_refine(best_h.copy(), best_n, rng, t0, budget_s, top_k=8)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 5: Fine local search
    if _tl() > 30:
        t_end = min(time.time() + min(25, _tl() - 10), t0 + budget_s - 15)
        h_f, c_f = _fine_search(best_h.copy(), best_n, rng, t_end)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    # Phase 6: Ultra-fine final polish
    if _tl() > 15:
        h_fin = best_h.copy()
        c_fin = _eval(h_fin)
        t_lim = t0 + budget_s - 3
        for _ in range(100000):
            if time.time() > t_lim:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-5 * rng.choice([1, 2, 3, 5, 8, 13])
            ni, nj = h_fin[i] + d, h_fin[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_fin[i], h_fin[j] = ni, nj
                c_new = _eval(h_fin)
                if c_new < c_fin - 1e-17:
                    c_fin = c_new
                else:
                    h_fin[i], h_fin[j] = ni - d, nj + d
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
