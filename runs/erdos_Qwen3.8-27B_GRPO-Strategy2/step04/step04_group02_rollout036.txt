import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def slsqp_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_step(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(slsqp_obj, h, args=(alpha, target), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sa_loop(h, n, dx, rng, t_end, T_init, T_reheat, stuck_thresh):
    c5 = compute_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    T = T_init
    T_floor = 1e-7
    no_improve = 0
    sigma_base = 0.008
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sigma = sigma_base * max(0.3, np.sqrt(T / T_init))
        d = sigma * (2.0 * rng.random() - 1.0)
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            half = d * 0.4
            ni = h[i] + half
            nj = h[j] - half
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            d = half
        h[i] = ni
        h[j] = nj
        c5_new = compute_c5(h, dx)
        delta = c5_new - c5
        if delta < 0 or (T > T_floor and rng.random() < np.exp(-delta / T)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            no_improve += 1
        T = max(T_floor, T * 0.999997)
        if no_improve > stuck_thresh:

            no_improve = 0

    return h, best_c5

    hp = h.copy()
    if kind == 0:
