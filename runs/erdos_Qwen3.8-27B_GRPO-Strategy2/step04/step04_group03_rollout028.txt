import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha, topm=1):
    c = np.correlate(x, 1.0 - x, mode='full')
    if topm >= len(c):
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    idx = np.argpartition(c, -topm)[-topm:]
    top = c[idx]
    m = top.max()
    s = np.clip(alpha * (top - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha, topm=1):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    ncorr = 2*n - 1
    if topm >= ncorr:
        idx = np.arange(ncorr)
    else:
        idx = np.argpartition(c, -topm)[-topm:]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        k = int(idx[i])
        wk = w[i]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _opt_slsqp(h, n, alpha, maxiter=500, topm=1):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha, topm),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_subgrad(h, n, rng, n_iter=500, t_end=None):
    """Projected subgradient using exact max."""
    c5c = _eval(h)
    lr = 0.01
    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        # Compute gradient at k_star
        g = np.zeros(n)
        lag = k_star - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= h[i2 + lag:j2 + lag]
        # Projected gradient step
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            h = h_new
            c5c = c5n
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.9
        if lr < 1e-8:
            break
    return h, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _fine_search(h, n, rng, t_end, scale_init=1e-4):
    """Very fine local search with small steps."""
    c5c = _eval(h)
    scale = scale_init
    no_imp = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.03, 0.05)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            no_imp += 1
            if no_imp > 200:
                scale *= 0.7
                no_imp = 0
                if scale < 1e-9:
                    break
    return h, c5c

def _block_search(h, n, rng, t_end, block_sizes=(2,3,4,5)):
    """Local search with block moves."""
    c5c = _eval(h)
    scale = 0.01
    no_imp = 0
    while time.time() < t_end:
        bs = rng.choice(block_sizes)
        start = rng.randint(n - bs + 1)
        idx = np.arange(start, start + bs)
        d = scale * rng.randn(bs)
        h_new = h.copy()
        h_new[idx] += d
        # Compensate with random other points
        for k in range(bs):
            j = rng.randint(n)
            while j in idx:
                j = rng.randint(n)
            h_new[j] -= d[k]
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.03, 0.03)
        else:
            no_imp += 1
            if no_imp > 300:
                scale *= 0.8
                no_imp = 0
                if scale < 1e-7:
                    break
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, topm):
    c = np.correlate(x, 1.0 - x, mode='full')
    nc = len(c)
    if topm >= nc:
        idx = np.arange(nc)
    else:
        idx = np.argpartition(c, -topm)[-topm:]
    top = c[idx]
    m = top.max()
    s = np.clip(alpha * (top - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad(x, alpha, topm):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    nc = len(c)
    if topm >= nc:
        idx = np.arange(nc)
    else:
        idx = np.argpartition(c, -topm)[-topm:]
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        k = int(idx[i])
        wk = w[i]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        # Correct gradient: g[j] += (1-h[j+lag]) for j+lag valid, g[j] -= h[j-lag] for j-lag valid
        # Term 1: (1-h[j+lag]) for j in [max(0,-lag), min(n,n-lag))
        a1 = max(0, -lag)
        b1 = min(n, n - lag)
        if a1 < b1:
            g[a1:b1] += wk * (1.0 - x[a1 + lag:b1 + lag])
        # Term 2: -h[j-lag] for j in [max(0,lag), min(n,n+lag))
        a2 = max(0, lag)
        b2 = min(n, n + lag)
        if a2 < b2:

    return np.sum(x) - n / 2.0

def _c_jac(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter, topm):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac, 'args': (n,)}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha, topm),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:

        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - tol:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.02, 0.05)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 400:
                scale *= cool
                no_imp = 0
                if scale < smin:
                    break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n0 = len(h0)
    bh, bc, bn = h0.copy(), _eval(h0), n0

    nlist = sorted(set([40, 50, 60, 70, 80, 90, 100, 120, 140, n0, 180]))
    inits = {}
    for n in nlist:
        if n == n0:
            inits[n] = h0.copy()
            continue
        hc = _repair(_interp_h(h0, n0, n), n)
        bc2, bh2 = _eval(hc), hc.copy()
        for f in [1, 2, 3, 4, 5]:
            for a in [0.3, 0.5, 0.7, 0.85]:
                hs = _repair(0.5 + a * np.sin(2.0 * np.pi * f * np.arange(n) / n), n)
                vs = _eval(hs)
                if vs < bc2:
                    bc2, bh2 = vs, hs.copy()
        inits[n] = bh2

    for n in nlist:
        if _tl() < 55:
            break
        h = inits[n].copy()
        cv = _eval(h)
        for al, tm, mi in [(50, 1, 400), (500, 1, 400), (5000, 2, 350),
                           (50000, 3, 300), (500000, 5, 250), (5e6, 8, 200)]:
            if _tl() < 65:
                break
            h2 = _slsqp(h, n, al, max(100, min(400, 500 - n // 3)), tm)
            c2 = _eval(h2)
            if c2 < cv - 1e-14:
                cv, h = c2, h2.copy()
        if _tl() > 35:
            t_lim = time.time() + min(22, _tl() - 15)
            while time.time() < t_lim:
                h, cv = _subgrad(h, n, 300, 0.02)
                h, cv2 = _subgrad(h, n, 200, 0.005)
                if cv2 >= cv:
                    break
                cv = cv2
        if _tl() > 20:
            h, cv = _search(h, n, rng, time.time() + min(15, _tl() - 10), 0.004)
        if _tl() > 12:
            h, cv = _search(h, n, rng, time.time() + min(10, _tl() - 6), 5e-5,
                            tol=1e-16, cool=0.6, smin=1e-10)
        if cv < bc:
            bc, bh, bn = cv, h.copy(), n

    if _tl() > 25:
        h = bh.copy()
        h = _slsqp(h, bn, 1e7, max(100, 350 - bn // 4), 8)
        c = _eval(h)
        if c < bc:
            bc, bh = c, h.copy()
    if _tl() > 15:
        h = bh.copy()
        h, c = _subgrad(h, bn, 500, 0.001)
        if c < bc:
            bc, bh = c, h.copy()
    if _tl() > 10:
        h2, c2 = _search(bh.copy(), bn, rng, time.time() + min(8, _tl() - 3),
                         3e-5, tol=1e-16, cool=0.55, smin=1e-11)
        if c2 < bc:
            bc, bh = c2, h2.copy()
    if _tl() > 4:
        h3 = bh.copy()
        c3 = _eval(h3)
        for _ in range(30000):
            if time.time() > t0 + budget_s - 1:
                break
            i, j = rng.randint(bn), rng.randint(bn)
            if i == j:
                continue
            d = rng.choice([1.0, -1.0]) * 1e-6 * rng.choice([1, 2, 5, 10])
            ni, nj = h3[i] + d, h3[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h3[i], h3[j] = ni, nj
                c = _eval(h3)
                if c < c3 - 1e-17:
                    c3 = c
                    if c3 < bc:
                        bc, bh = c3, h3.copy()
                else:
                    h3[i], h3[j] = ni - d, nj + d

    bh = _repair(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)
