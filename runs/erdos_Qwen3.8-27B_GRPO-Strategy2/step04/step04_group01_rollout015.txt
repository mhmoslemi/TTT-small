import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    s -= s.max()
    return mx + np.log(np.sum(np.exp(s))) / alpha + 1.0 / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[0:n - sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:n] -= wk * h[0:n - sh]
        else:
            sp = -sh
            g[sp:n] += wk * (1.0 - h[0:n - sp])
            g[0:n - sp] -= wk * h[sp:n]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cfun(x):
        return x.sum() - n * 0.5
    def _cjac(x):
        return np.ones(n)
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _compute_G(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        if sh >= 0:
            idx = np.arange(n - sh)
            G[k, idx] += (1.0 - h[idx + sh])
            if sh > 0:
                idx2 = np.arange(sh, n)
                G[k, idx2] -= h[idx2 - sh]
        else:
            sp = -sh
            idx = np.arange(sp, n)
            G[k, idx] += (1.0 - h[idx - sp])
            idx2 = np.arange(n - sp)
            G[k, idx2] -= h[idx2 + sp]
    return G

def _fast_local_search(h, n, budget_end, rng, max_scales=4):
    c5 = _eval(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _compute_G(h, n)
    scales = [0.008, 0.003, 0.0008, 0.0002]
    accepted_count = 0
    for si, scale in enumerate(scales[:max_scales]):
        nsteps = 800000 if scale > 0.001 else 400000
        for it in range(nsteps):
            if time.time() > budget_end:
                return h, c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                continue
            dc = d * (G[:, i] - G[:, j])
            new_max = (c + dc).max()
            if new_max < c5 - 1e-14:
                h[i] += d
                h[j] -= d
                c += dc
                c5 = new_max
                accepted_count += 1
        if time.time() > budget_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        G = _compute_G(h, n)
        c5 = _eval(h)
    return h, c5
