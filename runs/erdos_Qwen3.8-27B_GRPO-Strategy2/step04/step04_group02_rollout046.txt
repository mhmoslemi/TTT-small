                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sa_pair(h, n, dx, rng, T_start, T_end, deadline, base_scale):
    h = h.copy()
    curr_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = curr_c5
    prev_c5 = curr_c5
    it = 0
    stuck = 0
    total_budget = 400000
    while time.time() < deadline:
        it += 1
        progress = min(1.0, it / float(total_budget))
        T = max(T_start * (T_end / T_start) ** progress, 1e-10)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = base_scale * (0.3 + 0.7 * (1.0 - progress))
        d = scale * (2.0 * rng.random() - 1.0)
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i] = ni
        h[j] = nj
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - prev_c5
        if delta < 0 or (T > 1e-10 and np.log(max(rng.random(), 1e-300)) < -delta / T):
            prev_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i] -= d
            h[j] += d
            stuck += 1
        if stuck >= 8000:
            stuck = 0
            T_start = max(T_start * 3.0, 5e-4)
            for _ in range(15):
                p = rng.randint(n)
                q = rng.randint(n)
                if p != q:
                    dd = 0.015 * (2.0 * rng.random() - 1.0)
                    if 0.0 <= h[p] + dd <= 1.0 and 0.0 <= h[q] - dd <= 1.0:
                        h[p] += dd
                        h[q] -= dd
            prev_c5 = compute_c5(h, dx)
            if prev_c5 < best_c5:
                best_c5 = prev_c5
                best_h = h.copy()
    return best_h, best_c5

def sa_block(h, n, dx, rng, T_start, T_end, deadline):
    h = h.copy()
    curr_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = curr_c5
    prev_c5 = curr_c5
    it = 0
    stuck = 0
    total_budget = 150000
    while time.time() < deadline:
        it += 1
        progress = min(1.0, it / float(total_budget))
        T = max(T_start * (T_end / T_start) ** progress, 1e-10)
        bs = rng.randint(2, 5)
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        eps = 0.008 * (1.0 - 0.5 * progress) * (2.0 * rng.random() - 1.0)
        ok = True
        for di in range(bs):
            if not (0.0 - 1e-9 <= h[i+di] + eps <= 1.0 + 1e-9):
                ok = False
                break
            if not (0.0 - 1e-9 <= h[j+di] - eps <= 1.0 + 1e-9):
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - prev_c5
        if delta < 0 or (T > 1e-10 and np.log(max(rng.random(), 1e-300)) < -delta / T):
            prev_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
            stuck += 1
        if stuck >= 5000:
            stuck = 0
            T_start = max(T_start * 2.5, 3e-4)
            prev_c5 = compute_c5(h, dx)
    return best_h, best_c5

def local_search(h, n, dx, rng, deadline, fine):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    if fine:
        scales = [0.001, 0.0004, 0.00015, 0.00005]
        per_scale = 80000
    else:
        scales = [0.008, 0.003, 0.001, 0.0004]
        per_scale = 60000
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.003:
            break
    return h, best_c5

def directed_sa(h, n, dx, rng, T_start, T_end, deadline):
    """SA that biases moves toward the bottleneck lag."""
    h = h.copy()
    curr_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = curr_c5
    prev_c5 = curr_c5
    it = 0
    stuck = 0
    total_budget = 200000
    while time.time() < deadline:
        it += 1
        progress = min(1.0, it / float(total_budget))
        T = max(T_start * (T_end / T_start) ** progress, 1e-10)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Bias: prefer moves where i is near a high h and j+k_star is low h (or vice versa)
        scale = 0.006 * (0.4 + 0.6 * (1.0 - progress))
        d = scale * (2.0 * rng.random() - 1.0)
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i] = ni
        h[j] = nj
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - prev_c5
        if delta < 0 or (T > 1e-10 and np.log(max(rng.random(), 1e-300)) < -delta / T):
            prev_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i] -= d
            h[j] += d
            stuck += 1
        if stuck >= 6000:
            stuck = 0
            T_start = max(T_start * 2.0, 4e-4)
            prev_c5 = compute_c5(h, dx)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = repair(h0.copy(), n)
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5

    def remain():
        return budget_s - (time.time() - t_start)

    # Phase 1: L-BFGS from initial (25s)
    dl = t_start + 25
    for alpha in [800, 3000]:
        if time.time() >= dl or remain() < 900:
            break
        hh, cc = lbfgs_opt(h, n, alpha, target, 2000, 250)
        if cc < best_c5:
            best_h, best_c5 = hh.copy(), cc
    h = best_h

    # Phase 2: SA pair moves - main exploration (350s)
    dl = time.time() + min(350, remain() - 250)
    hh, cc = sa_pair(h, n, dx, rng, 4e-4, 1e-9, dl, 0.008)
    if cc < best_c5:
        best_h, best_c5 = hh.copy(), cc
    h = best_h

    # Phase 3: L-BFGS polish (50s)
    dl = time.time() + min(50, remain() - 200)
    for alpha in [2000, 8000, 30000]:
        if time.time() >= dl:
            break
        hh, cc = lbfgs_opt(h, n, alpha, target, 5000, 300)
        if cc < best_c5:
            best_h, best_c5 = hh.copy(), cc
    h = best_h

    # Phase 4: Directed SA (200s)
    dl = time.time() + min(200, remain() - 120)
    hh, cc = directed_sa(h, n, dx, rng, 3e-4, 1e-9, dl)
    if cc < best_c5:
        best_h, best_c5 = hh.copy(), cc
    h = best_h

    # Phase 5: Block SA (150s)
    dl = time.time() + min(150, remain() - 80)
    hh, cc = sa_block(h, n, dx, rng, 2e-4, 1e-9, dl)
    if cc < best_c5:
        best_h, best_c5 = hh.copy(), cc
    h = best_h

    # Phase 6: SLSQP polish (40s)
    if remain() > 70:
        dl = time.time() + min(40, remain() - 30)
        hh, cc = slsqp_opt(h, n, 10000, target, 400)
        if cc < best_c5:
            best_h, best_c5 = hh.copy(), cc
        h = best_h

    # Phase 7: Fine local search greedy (100s)
    if remain() > 50:
        dl = time.time() + min(100, remain() - 20)
        hh, cc = local_search(h, n, dx, rng, dl, fine=True)
        if cc < best_c5:
            best_h, best_c5 = hh.copy(), cc
        h = best_h

    # Phase 8: Final ultra-fine SA (remaining)
    if remain() > 20:
        dl = time.time() + max(5, min(60, remain() - 5))
        hh, cc = sa_pair(h, n, dx, rng, 1e-5, 1e-11, dl, 0.002)
        if cc < best_c5:
            best_h, best_c5 = hh.copy(), cc

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
