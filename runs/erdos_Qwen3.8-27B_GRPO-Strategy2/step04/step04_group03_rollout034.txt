def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _sobj(h, alpha, mu):
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(s))) / alpha
    n = len(h)
    return sm + mu * (h.sum() - n * 0.5) ** 2

def _sgrad(h, alpha, mu):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                g[:n-lag] -= wk * h[lag:]
                g[lag:] += wk * (1.0 - h[:n-lag])
        else:
            L = -lag
            if n - L > 0:
                g[:n-L] += wk * (1.0 - h[L:])
                g[L:] -= wk * h[:n-L]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _sobj(h, alpha, mu):
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(s))) / alpha
    n = len(h)
    return sm + mu * (h.sum() - n * 0.5) ** 2

def _sgrad(h, alpha, mu):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                g[:n-lag] -= wk * h[lag:]
                g[lag:] += wk * (1.0 - h[:n-lag])
        else:
            L = -lag
            if n - L > 0:
                g[:n-L] += wk * (1.0 - h[L:])
                g[L:] -= wk * h[:n-L]
    g += 2.0 * mu * (h.sum() - n * 0.5)

def _interp(h_s, n_s, n_d):
    if n_s == n_d:
        return h_s.copy()
    return np.interp(np.linspace(0, 2, n_d, endpoint=False),
                     np.linspace(0, 2, n_s, endpoint=False), h_s)

def _lgbf(h, n, alpha, mu, maxiter):
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, args=(alpha, mu),
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return _repair(h, n)

def _local(h, n, rng, t_end, max_it=80000):
    c5 = _eval(h)
    sc = 0.004
    ni = 0
    for _ in range(max_it):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1:
            continue
        h[i], h[j] = a, b
        v = _eval(h)
        if v < c5 - 1e-15:
            c5 = v
            ni = 0
            sc = min(sc * 1.02, 0.025)
        else:
            h[i], h[j] = a - d, b + d
            ni += 1
            if ni > 300:
                sc *= 0.82
                ni = 0

    return h, c5

def _perturb(h, n, rng, eps):
    p = h + eps * rng.randn(n)

def _block_opt(h, n, rng, t_end):
