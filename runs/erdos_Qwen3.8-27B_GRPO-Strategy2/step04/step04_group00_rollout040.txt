            cand_js = [j for j in order if j != i][:n_pairs]
            for j in cand_js:
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-12:
                    continue
                lo, hi = -max_d, max_d
                best_d, best_c = 0.0, c5c
                for _ in range(7):
                    mid = (lo + hi) * 0.5
                    h[i], h[j] = h[i] + mid, h[j] - mid
                    hc = _repair(h, n)
                    cv = _c5_val(hc, dx)
                    h[i], h[j] = hc[i], hc[j]
                    if cv < best_c:
                        best_c, best_d = cv, mid
                    if mid > 0:
                        lo = mid
                    else:
                        hi = mid
                if best_d != 0.0 and best_c < c5c - 1e-15:
                    h[i] = min(1.0, max(0.0, h[i] + best_d))
                    h[j] = min(1.0, max(0.0, h[j] - best_d))
                    h = _repair(h, n)
                    c5c = _c5_val(h, dx)
                    improved = True
                    if c5c < bc5:
                        bh, bc5 = h.copy(), c5c
        if not improved:
            break
    return bh, bc5

def _full_opt(h_init, n, dx, rng, tleft, budget):
    bounds = [(0.0, 1.0)] * n
    h = h_init.copy()
    c5v = _c5_val(h, dx)
    h, c5v = _run_lbfgs(h, n, dx, bounds, tleft, budget + 60)
    h, c5v = _run_subgrad(h, n, dx, tleft, budget + 30)
    h, c5v = _run_coord(h, n, dx, rng, tleft, budget)
    return h, c5v

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    best_h, best_c5 = h0.copy(), _c5_val(h0, dx0)

    # Phase A: full optimization at native n
    if tl() > 250:
        hr, cr = _full_opt(h0, n0, dx0, rng, tl, 220)
        if cr < best_c5:
            best_h, best_c5 = hr, cr

    # Phase B: try n=100
    if tl() > 200:
        n1 = 100
        dx1 = 2.0 / n1
        nb = len(best_h)
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n1, endpoint=False)
        h1 = _repair(np.interp(x_new, x_old, best_h), n1)
        hr, cr = _full_opt(h1, n1, dx1, rng, tl, 170)
        if cr < best_c5:
            best_h, best_c5 = hr, cr

    # Phase C: try n=80
    if tl() > 130:
        n2 = 80
        dx2 = 2.0 / n2
        nb = len(best_h)
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = _repair(np.interp(x_new, x_old, best_h), n2)
        hr, cr = _full_opt(h2, n2, dx2, rng, tl, 100)
        if cr < best_c5:
            best_h, best_c5 = hr, cr

    # Phase D: structured starts at n=100
    if tl() > 80:
        n3 = 100
        dx3 = 2.0 / n3
        xg = np.linspace(0, 2, n3, endpoint=False)
        for trial in range(4):
            if tl() < 40:
                break
            if trial == 0:
                ht = 0.6 + 0.4 * np.cos(np.pi * xg)
            elif trial == 1:
                ht = np.exp(-((xg - 1.0) ** 2) / 0.5)
            elif trial == 2:
                ht = 0.5 + 0.3 * np.sin(2 * np.pi * xg) + 0.1 * np.sin(4 * np.pi * xg)
            else:
                ht = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * xg)
            ht = _repair(ht, n3)
            hr, cr = _full_opt(ht, n3, dx3, rng, tl, 30)
            if cr < best_c5:
                best_h, best_c5 = hr, cr

    # Phase E: fine random polish on best
    if tl() > 5:
        nf = len(best_h)
        dxf = 2.0 / nf
        h = best_h.copy()
        c5c = _c5_val(h, dxf)
        nits = int(min(80000, (tl() - 2) * 8000))
        deltas = np.array([0.0005, 0.001, 0.002, 0.004, -0.0005, -0.001, -0.002, -0.004])
        for _ in range(nits):
            if tl() < 2:
                break
            i, j = rng.randint(nf), rng.randint(nf)
            if i == j:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5_val(h, dxf)
                if c < c5c:
                    c5c = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5_val(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
