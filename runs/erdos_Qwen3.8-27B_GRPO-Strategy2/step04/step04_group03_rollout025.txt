import time

def do_repair(h, n):

    if s < 1e-12:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:

        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def do_eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def do_objgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    e = np.exp(s - sm)
    e_sum = e.sum()
    w = e / e_sum
    val = m + (sm + np.log(e_sum)) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - h[lo1 + lag:hi1 + lag])
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * h[lo2 - lag:hi2 - lag]
    return val * (2.0 / n), g * (2.0 / n)

def do_penalty_objgrad(h, n, alpha, pen):
    val, g = do_objgrad(h, alpha)
    s = h.sum() - n / 2.0
    val += 0.5 * pen * s * s
    g += pen * s
    return val, g

def do_lbfgsb(h, n, alpha, pen, maxiter, t_end):
    bnds = [(0.0, 1.0)] * n
    def fun(x):
        return do_penalty_objgrad(x, n, alpha, pen)[0]
    def jac(x):
        return do_penalty_objgrad(x, n, alpha, pen)[1]
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return do_repair(res.x, n)
    except Exception:
        return do_repair(h, n)

        h_new = do_lbfgsb(h_cur, n, alpha, pen, maxiter, t_end - 25)
        c5_new = do_eval(h_new)
        if c5_new < c5_cur - 1e-15:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = do_repair(np.array(initial_h_values, dtype=float).copy(), 160)
    best_h = h_parent.copy()
    best_c5 = do_eval(h_parent)
    best_n = 160

    def time_left():
        return budget_s - (time.time() - t0)

    # Stage 1: Multi-resolution optimization
    n_list = [160, 128, 200, 100, 256]
    inits = {160: h_parent.copy()}

    for n in n_list:
        if time_left() < 80:
            break
        if n not in inits:
            h_i = do_repair(do_interp(h_parent, 160, n), n)
            cv = do_eval(h_i)
            # Also try perturbed versions
            best_c, best_hi = cv, h_i.copy()
            for _ in range(4):
                h_p = do_repair(np.clip(h_i + 0.1 * rng.randn(n), 0, 1), n)
                cp = do_eval(h_p)
                if cp < best_c:
                    best_c, best_hi = cp, h_p.copy()
            inits[n] = best_hi

    for n in n_list:
        if time_left() < 80:
            break
        h0 = inits[n]
        h_opt, c5_opt = do_multistage_opt(h0, n, [100, 1000, 10000, 100000], t0 + budget_s - 100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

    # Stage 2: Worst-lag targeted search
    if time_left() > 120:
        t_end = time.time() + min(80, time_left() - 60)
        h2, c5_2 = do_worst_lag_adjust(best_h.copy(), best_n, rng, t_end, n_iter=80000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Stage 3: Another round of L-BFGS-B with very high alpha
    if time_left() > 100:
        h3, c5_3 = do_multistage_opt(best_h, best_n, [50000, 200000, 500000], t0 + budget_s - 80)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Stage 4: Fine random local search
    if time_left() > 60:
        t_end = time.time() + min(50, time_left() - 30)
        h4, c5_4 = do_random_local(best_h.copy(), best_n, rng, t_end, n_iter=300000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Stage 5: Very fine adjustments
    if time_left() > 30:
        h5 = best_h.copy()
        c5c = best_c5
        for _ in range(80000):
            if time.time() > t0 + budget_s - 5:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.choice([1, 2, 3, 5]) * 1e-6
            ni, nj = h5[i] + d, h5[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                saved_i, saved_j = h5[i], h5[j]
                h5[i], h5[j] = ni, nj
                c5n = do_eval(h5)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h5.copy()
                else:
                    h5[i], h5[j] = saved_i, saved_j

    # Stage 6: Final worst-lag polish
    if time_left() > 20:
        t_end = time.time() + min(12, time_left() - 5)
        h6, c5_6 = do_worst_lag_adjust(best_h.copy(), best_n, rng, t_end, n_iter=30000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = do_repair(best_h, best_n)
    best_c5 = do_eval(best_h)
    return (best_h, best_c5, best_n)
