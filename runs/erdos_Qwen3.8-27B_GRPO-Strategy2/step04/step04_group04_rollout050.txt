    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _soft_obj(h, alpha, target, pen):
    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _targeted_sweep(h, dx, rng, t_end, max_iters=5000):

            if worst_k == 0:

def _slsqp_run(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, tgt, 0), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs_run(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best, bc = h.copy(), _c5(h, dx)
    for pen in [500, 5000, 50000]:
        try:
            r = minimize(lambda x: _soft_obj(x, alpha, tgt, pen), best,
                         method='L-BFGS-B', bounds=bds,
                         options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
            hc = _repair(r.x, tgt)
            c = _c5(hc, dx)
            if c < bc:
                best, bc = hc, c
        except Exception:
            pass
    return best, bc

                        no_improve = 0
        else:
            h[i] -= d; h[j] += d
    h = _repair(h, tgt)
    return h, _c5(h, dx)
