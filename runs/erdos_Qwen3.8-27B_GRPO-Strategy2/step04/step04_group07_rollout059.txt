        s = h.sum()
        if s < 1e-15:

        h = np.clip(h * target / s, 0.0, 1.0)
        s = h.sum()

        d = target - s
        if abs(d) < 1e-13:

        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)

            if abs(d) < 1e-13:

            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_max_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp(h, n, alpha, maxiter):
    try:
        res = minimize(soft_max_obj, h.copy(), args=(alpha,), method='SLSQP',
                       bounds=[(0,1)]*n, constraints=[{'type':'eq','fun':lambda v: v.sum()-n/2.0}],
                       options={'maxiter':maxiter,'ftol':1e-18})
        return repair(res.x, n)
    except:
        return h

def lbfgs(h, n, alpha, pen, maxiter):
    try:
        res = minimize(lambda v: soft_max_obj(v, alpha) + pen*(v.sum()-n/2.0)**2,
                       h.copy(), method='L-BFGS-B', bounds=[(0,1)]*n,
                       options={'maxiter':maxiter,'ftol':1e-20,'gtol':1e-12,'maxls':50})
        return repair(res.x, n)
    except:
        return h

def cyclic_cd(h, n, t_end, rng, n_passes=50):
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0/n
    # Precompute: which lags are "active" (near max)
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i+1, n):
                # Try small transfers
                hi, hj = h[i], h[j]
                if hi < 0.01 and hj < 0.01:
                    continue
                if hi > 0.99 and hj > 0.99:
                    continue
                for d in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
                    for s in (1,-1):
                        ni, nj = hi+s*d, hj-s*d

                        h[i], h[j] = ni, nj

                        if c < best - 1e-14:
                            best = c
                            improved = True
                            break
                        h[i], h[j] = hi, hj
                    if improved:
                        break
    return h, best

                best_h, best_c5, best_n, n, hw = hna, cna, na, na, hna

    # P6: Final polish
    if et() < budget_s * 0.995:
        tf = t0 + budget_s * 0.995
        hfin = go_slsqp(best_h.copy(), best_n, 8000000, 3000)
        cf = compute_c5(hfin, best_n)
        if cf < best_c5 - 1e-14:
            best_h, best_c5 = hfin, cf
        hf2, cf2 = cyc_cd(best_h, best_n, tf, dts_fine)
        if cf2 < best_c5 - 1e-14:
            best_h, best_c5 = hf2, cf2
        hs2, cs2 = swap_cd(best_h, best_n, tf)
        if cs2 < best_c5 - 1e-14:
            best_h, best_c5 = hs2, cs2

    best_h = safe_repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
