    best = _c5(h, dx)
    h = h.copy()
    step = 0.004
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 200 == 0:
            c = _corr(h)
            top_k = np.argsort(c)[-4:]
            for k_idx in top_k:
                shift = k_idx - (n - 1)
                if shift == 0:
                    continue
                if shift > 0:
                    ii = np.arange(n - shift)
                    jj = ii + shift
                else:
                    jj = np.arange(n + shift)
                    ii = jj - shift
                for _ in range(5):
                    if time.time() > t_end:
                        break
                    idx = rng.randint(len(ii))
                    i, j = int(ii[idx]), int(jj[idx])
                    if h[i] > 0.5:
                        d = rng.uniform(0.001, step)
                        if h[i] - d >= 0:
                            h[i] -= d
                            jj2 = j if (j % n) < n else j
                            if h[jj2] + d <= 1.0:
                                h[jj2] += d
                                c_new = _c5(h, dx)
                                if c_new < best:
                                    best = c_new
                                else:
                                    h[i] += d
                                    h[jj2] -= d
                            else:
                                h[i] += d
                    elif h[j] < 0.5:
                        d = rng.uniform(0.001, step)
                        if h[j] - d >= 0:
                            h[j] -= d
                            if h[i] + d <= 1.0:
                                h[i] += d
                                c_new = _c5(h, dx)
                                if c_new < best:
                                    best = c_new
                                else:
                                    h[j] += d
                                    h[i] -= d
                            else:
                                h[j] += d
                    else:
                        d = step * (2.0 * rng.random() - 1.0)
                        h[i] += d
                        h[j] -= d
                        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                            c_new = _c5(h, dx)
                            if c_new < best:
                                best = c_new
                            else:
                                h[i] -= d
                                h[j] += d
                        else:
                            h[i] -= d
                            h[j] += d
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = step * (2.0 * rng.random() - 1.0)
            h[i] += d
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c_new = _c5(h, dx)
                if c_new < best:
                    best = c_new
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
        if it % 5000 == 4999:
            step *= 0.85
    h = _repair(h, tgt)
    return h, _c5(h, dx)

def _random_local(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    per = max(30, n_iter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c = _c5(h, dx)
                if c < best:
                    best = c
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    h = _repair(h, tgt)
    best = _c5(h, dx)
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = _repair(h0, n0 / 2.0)
    best_h, best_c5, best_n = h0.copy(), _c5(h0, dx0), n0

    # Phase 1: L-BFGS-B at n0 with increasing sharpness
    t_end = t0 + 100
    for a in [100, 500, 2000, 8000]:
        if time.time() > t_end or rem() < 400:
            break
        h0, c = _lbfgs_run(h0, dx0, a, 800, t_end)
        if c < best_c5:
            best_h, best_c5 = h0, c

    # Phase 2: SLSQP polish
    t_end = t0 + 180
    for a in [2000, 10000]:
        if time.time() > t_end:
            break
        best_h, best_c5 = _slsqp_run(best_h, dx0, a, 800)

    # Phase 3: Multi-restart with structured perturbations
    t_end = t0 + 400
    for r in range(15):
        if time.time() > t_end or rem() < 450:
            break
        freq = 1.0 + r * 0.7
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n0)
        pert = 0.04 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        pert += 0.03 * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(best_h + pert, n0 / 2.0)
        hp, c = _lbfgs_run(hp, dx0, 3000, 400, min(t_end, t0 + budget_s - 400))
        hp, c = _slsqp_run(hp, dx0, 10000, 500)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 4: Higher resolution
    if rem() > 350:
        n1 = 2 * n0
        dx1 = 2.0 / n1
        h1 = _repair(_interp(best_h, n1), n1 / 2.0)
        c1 = _c5(h1, dx1)
        t_end = min(t0 + budget_s - 300, t0 + 650)
        for a in [1000, 5000]:
            if time.time() > t_end:
                break
            h1, c1 = _lbfgs_run(h1, dx1, a, 400, t_end)
        if time.time() < t_end - 10:
            h1, c1 = _slsqp_run(h1, dx1, 10000, 400)
        if c1 < best_c5:
            best_h, best_c5, best_n = h1, c1, n1
        else:
            # Try to rescue: optimize the n=140 version more
            pass

    # Phase 5: Surgical local search
    nb = best_n
    dxb = 2.0 / nb
    t_end = min(t0 + budget_s - 120, t0 + budget_s * 0.7)
    n_iter = max(30000, int((t_end - time.time()) * 120))
    best_h, best_c5 = _surgical_search(best_h, dxb, rng, t_end, n_iter)

    # Phase 6: SLSQP final polish
    if rem() > 60:
        for a in [20000, 50000]:
            if rem() < 30:
                break
            best_h, best_c5 = _slsqp_run(best_h, dxb, a, 600)

    # Phase 7: Random local search
    if rem() > 40:
        best_h, best_c5 = _random_local(best_h, dxb, rng, t0 + budget_s - 15, 300000)

    # Final verification
    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
