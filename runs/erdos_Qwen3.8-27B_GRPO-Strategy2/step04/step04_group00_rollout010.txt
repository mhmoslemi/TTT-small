import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_corr_L(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += (1.0 - h[M:])
        g[M:] -= h[:n-M]
    return g

def _active_subgrad(h, n, frac):
    c = _corr(h)
    cm, cn = np.max(c), np.min(c)
    spread = cm - cn if cm > cn else 1.0
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        g += _grad_corr_L(h, k - (n-1), n)
    g /= len(active)
    return g, c, active

def _softobj(x, alpha, n):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(x, alpha, n):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s -= np.max(s)
    ew = np.exp(s)
    w = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        L = k - (n-1)
        pk = w[k]
        if pk < 1e-15:
            continue
        g += pk * _grad_corr_L(x, L, n)
    return g / alpha

def _optimize_subgrad(h, n, dx, t0, budget_s, margin=200):
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    step = 0.025
    momentum = np.zeros(n)
    beta = 0.7
    stuck = 0
    decrease_frac = 0.03
    for it in range(60000):
        if time.time() - t0 > budget_s - margin:
            break
        g, c, active = _active_subgrad(h, n, decrease_frac)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-13:
            break
        momentum = beta * momentum + (1-beta) * (-g / ng)
        nmom = np.linalg.norm(momentum)
        if nmom < 1e-13:
            break
        best_s = 0
        best_v = c5c
        for s in [step, step*0.5, step*1.5, step*0.25]:
            h_try = _repair(h + s * momentum / nmom, n)
            v = _c5(h_try, dx)
            if v < best_v:
                best_v = v
                best_s = s
        if best_s == 0:
            step *= 0.65
            stuck += 1
            if stuck > 300:
                if decrease_frac > 0.01:
                    decrease_frac = max(0.01, decrease_frac * 0.8)
                    stuck = 0
                else:
                    break
        else:
            h = _repair(h + best_s * momentum / nmom, n)
            c5c = _c5(h, dx)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h.copy()
            step = min(step * 1.03, 0.06)
            stuck = 0
    return best_h, best_c5

def _local_search(h, n, dx, t0, budget_s, margin=80):
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    c = _corr(h)
    active = np.where(c >= np.max(c) - 0.04*(np.max(c)-np.min(c)))[0]
    crit = set()
    for k in active:
        L = k - (n-1)
        if L >= 0:
            crit.update(range(L, n))
            if L > 0:
                crit.update(range(n-L))
        else:
            M = -L
            crit.update(range(n-M))
            crit.update(range(M, n))
    clist = sorted(crit)
    if len(clist) > 80:
        clist = clist[:80]
    dvals = np.array([0.001, -0.001, 0.002, -0.002, 0.004, -0.004, 0.0005, -0.0005])
    for sweep in range(400):
        if time.time() - t0 > budget_s - margin:
            break
        improved = False
        for idx_i, i in enumerate(clist):
            for d in dvals:
                for j in clist[(idx_i+1) % len(clist): min(idx_i+15, len(clist))]:
                    if i == j:
                        continue
                    ni, nj = h[i]+d, h[j]-d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c_new = _c5(h, dx)
                        if c_new < c5c:
                            c5c = c_new
                            if c5c < best_c5:
                                best_c5 = c5c
                                best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[i], h[j] = ni-d, nj+d
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def _lbfgs_phase(h, n, dx, t0, budget_s, margin=150):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    for alpha in [300, 1500, 8000, 40000, 200000]:
        if time.time() - t0 > budget_s - margin:
            break
        try:
            res = minimize(_softobj, h.copy(), args=(alpha, n),
                          jac=lambda x: _softgrad(x, alpha, n),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 1500, 'ftol': 1e-17, 'gtol': 1e-13, 'maxls': 40})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                h = hc.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: L-BFGS with soft-max
    best_h, best_c5 = _lbfgs_phase(h0, n, dx, t0, budget_s, margin=250)

    # Phase 2: Nesterov subgradient
    best_h, best_c5 = _optimize_subgrad(best_h, n, dx, t0, budget_s, margin=150)

    # Phase 3: Local search on active correlations
    best_h, best_c5 = _local_search(best_h, n, dx, t0, budget_s, margin=80)

    # Phase 4: Try higher resolution
    if time.time() - t0 < budget_s - 120:
        nh = 200
        dxh = 2.0 / nh
        xs_fine = np.linspace(0, 2, nh, endpoint=False)
        xs_coarse = np.linspace(0, 2, n, endpoint=False)
        hh = np.interp(xs_fine, xs_coarse, best_h)
        hh = _repair(hh, nh)
        hh, c5h = _lbfgs_phase(hh, nh, dxh, t0, budget_s, margin=80)
        hh, c5h = _optimize_subgrad(hh, nh, dxh, t0, budget_s, margin=50)
        if c5h < best_c5:
            best_c5, best_h, n, dx = c5h, hh, nh, dxh

    # Phase 5: Final micro-perturbation
    if time.time() - t0 < budget_s - 3:
        h = best_h.copy()
        c5c = _c5(h, dx)
        remaining = budget_s - (time.time() - t0) - 2
        ntr = int(remaining * 1500)
        for _ in range(ntr):
            if time.time() - t0 > budget_s - 2:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.0015, 0.0015)
            ni, nj = h[i]+d, h[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                    if c < best_c5:
                        best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = ni-d, nj+d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
