    r, c = coord2d(bh, n, dx, rng, t0, lim - 1, 12000)
    if c < bc: bh, bc = r, c
    return bh, bc

def upsample(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(xn, xo, h, period=2.0)
    return repair(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=np.float64).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0

    # Phase 1: Full optimization at initial resolution
    r, c = refine(bh, n0, dx0, seed, t0, budget_s - 200)
    if c < bc:
        bh, bc, bn = r, c, n0

    # Phase 2: Multi-restart for diverse exploration
    rng = np.random.RandomState(seed)
    for restart in range(6):
        if time.time() - t0 > budget_s - 100:
            break
        if restart == 0:
            h_r = repair(bh + 0.1 * rng.randn(n0), n0)
        elif restart == 1:
            x = np.linspace(0, 2, n0, endpoint=False)
            h_r = repair(bh + 0.08 * np.sin(2 * np.pi * x) + 0.05 * rng.randn(n0), n0)
        elif restart == 2:
            h_r = repair(np.clip(bh + 0.2 * rng.randn(n0), 0.05, 0.95), n0)
        elif restart == 3:
            h_r = repair(bh + 0.06 * np.cos(4 * np.pi * x + rng.rand() * 3.14), n0)
        elif restart == 4:
            h_r = repair(np.where(bh > 0.7, 0.9, np.where(bh < 0.3, 0.1, bh)) + 0.03 * rng.randn(n0), n0)
        else:
            H = np.fft.rfft(bh)
            H[1:n0//2] *= 0.7 + 0.3 * rng.randn(n0//2 - 1)
            hr = np.fft.irfft(H, n0)
            h_r = repair(hr, n0)
        r, c = quick_refine(h_r, n0, dx0, seed + 200 + restart * 31, t0, budget_s - 90)
        if c < bc:
            bh, bc, bn = r, c, n0

    # Phase 3: Upsample to 2x
    n1 = n0 * 2
    if time.time() - t0 < budget_s - 140:
        h1 = upsample(bh, bn, n1)
        dx1 = 2.0 / n1
        r, c = refine(h1, n1, dx1, seed + 600, t0, budget_s - 25)
        if c < bc:
            bh, bc, bn = r, c, n1

    # Phase 4: Try 3x if under 1000 and time allows
    n2 = n0 * 3
    if n2 < 1000 and time.time() - t0 < budget_s - 40:
        h2 = upsample(bh, bn, n2)
        dx2 = 2.0 / n2
        rng2 = np.random.RandomState(seed + 900)
        r, c = quick_refine(h2, n2, dx2, seed + 900, t0, budget_s - 5)
        if c < bc:
            bh, bc, bn = r, c, n2

    # Final coordinate polish
    if time.time() - t0 < budget_s - 3:
        rng3 = np.random.RandomState(seed + 3000)
        r, c = coord2d(bh, bn, 2.0 / bn, rng3, t0, budget_s - 1, 15000)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
