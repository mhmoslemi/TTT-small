Term 2: h[p] appears in the sum as h[i+ℓ] when i+ℓ=p, i=p-ℓ (valid if p-ℓ ≥ 0 and p-ℓ ≤ n-1-ℓ, i.e., ℓ ≤ p ≤ n-1)
  → contributes h[p-ℓ]*(-1) = -h[p-ℓ]

So for ℓ ≥ 0:
∂c/∂h[p] = (1-h[p+ℓ])*1[0≤p≤n-1-ℓ] - h[p-ℓ]*1[ℓ≤p≤n-1]

For ℓ < 0, let m = -ℓ > 0:
c = ∑_{i=m}^{n-1} h[i]*(1-h[i-m])

∂c/∂h[p]:
Term 1: h[p] appears as h[i] when i=p (valid if m≤p≤n-1)
  → contributes (1-h[p-m])

Term 2: h[p] appears as h[i-m] when i-m=p, i=p+m (valid if m≤p+m≤n-1, i.e., 0≤p≤n-1-m)
  → contributes h[p+m]*(-1) = -h[p+m]

So for ℓ < 0, m = -ℓ:
∂c/∂h[p] = (1-h[p-m])*1[m≤p≤n-1] - h[p+m]*1[0≤p≤n-1-m]

Great, this is clean. Now for the full gradient of the soft-max:

∂F/∂h[p] = ∑_k w_k * ∂c[k]/∂h[p]

where w_k = softmax_k of (alpha * (c_k - max_c)).

```python
def _soft_max_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # ∂c/∂h[p] = (1-h[p+lag]) for p in [0, n-1-lag]
            #             -h[p-lag] for p in [lag, n-1]
            g[:n-lag] += wk * (1.0 - x[lag:n])
            g[lag:] -= wk * x[:n-lag]
        else:
            m2 = -lag
            # ∂c/∂h[p] = (1-h[p-m2]) for p in [m2, n-1]
            #             -h[p+m2] for p in [0, n-1-m2]
            g[m2:] += wk * (1.0 - x[:n-m2])
            g[:n-m2] -= wk * x[m2:]
    return g
```

This is O(n) per k, and we have 2n-1 k values, so O(n²) total. For n=160, that's about 50k operations per gradient, which is fast.

        else:
            order = np.argsort(h)

                h[idx] += amt
                diff -= amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt
                diff += amt
            if abs(diff) < 1e-12:
                break
    return h

            no_imp += 1
            if no_imp > 150:
                scale *= 0.75; no_imp = 0
                if scale < 3e-7:
                    break
    return h, c5c

def _local_3pt(h, n, rng, t_end, max_it=40000):
    c5c = _eval(h)
    scale = 0.004
    for _ in range(max_it):
        if time.time() > t_end:
            break
        idx = rng.choice(n, size=3, replace=False)
        i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return h, c5c

def _slsqp_constraint_c(x):
    return x.sum() - x.size / 2.0

def _slsqp_constraint_j(x):
    return np.ones(x.size)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    n_p = len(initial_h_values)
    h_p = _repair(np.array(initial_h_values, dtype=float).copy(), n_p)
    best_h = h_p.copy()
    best_c5 = _eval(h_p)
    best_n = n_p

    n_list = sorted(set([n_p, n_p - 20, n_p - 10, n_p + 10, n_p + 20, 100, 130, 200]))
    n_list = [n for n in n_list if 40 <= n <= 250]

    inits = {n_p: h_p.copy()}
    for n in n_list:
        if n not in inits:
            inits[n] = _repair(_interp(h_p, n_p, n), n)

    for n in sorted(n_list):
        if _tl() < 50:
            break
        h_cur = inits[n].copy()
        for alpha in [300, 2000, 10000, 50000, 150000]:
            if _tl() < 35:
                break
            h_cur = _grad_descent(h_cur, n, alpha, 1500, min(time.time() + 22, t0 + budget_s - 25))
            c5_c = _eval(h_cur)
            if c5_c < best_c5 - 1e-12:
                best_c5 = c5_c
                best_h = h_cur.copy()
                best_n = n
        inits[n] = h_cur.copy()

    if _tl() > 70:
        n = best_n
        h = best_h.copy()
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_constraint_c, 'jac': _slsqp_constraint_j}]
        for alpha in [80000, 300000, 800000]:
            if _tl() < 35:
                break
            try:
                res = minimize(_sm_val, h, jac=lambda x, a=alpha: _sm_grad(x, a),
                              method='SLSQP', bounds=bnds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-20})
                h = _repair(res.x, n)
                c5 = _eval(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except Exception:
                pass

    if _tl() > 35:
        t_end = time.time() + min(25, _tl() - 12)
        h2, c5_2 = _local_2pt(best_h.copy(), best_n, rng, t_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    if _tl() > 25:
        t_end = time.time() + min(15, _tl() - 10)
        h3, c5_3 = _local_3pt(best_h.copy(), best_n, rng, t_end)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    if _tl() > 20:
        h = best_h.copy()
        c5c = _eval(h)
        for _ in range(200000):
            if time.time() > t0 + budget_s - 3:
                break
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-6 * rng.choice([1, 2, 5, 10, 20, 50])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    if _tl() > 10:
        t_end = time.time() + min(8, _tl() - 4)
        h4, c5_4 = _local_2pt(best_h.copy(), best_n, rng, t_end, max_it=30000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
