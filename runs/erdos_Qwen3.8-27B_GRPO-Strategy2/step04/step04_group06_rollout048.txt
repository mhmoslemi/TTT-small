import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _grad_c5_smooth(x, n, alpha):
    """Gradient of smoothed max correlation."""
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(2 * n - 1)
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - n1
        if lag == 0:
            g[:] += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _pgd(h, n, alpha, budget_end, lr):
    h = h.copy()
    c5_cur = _c5(h, n)
    for it in range(100000):
        if time.time() > budget_end:

        g = _grad_c5_smooth(h, n, alpha)

        c5_new = _c5(h_new, n)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
        else:
            lr *= 0.7
    return h, c5_cur
