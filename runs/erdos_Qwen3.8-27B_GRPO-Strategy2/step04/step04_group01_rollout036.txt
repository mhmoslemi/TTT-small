Coordinate descent: for each coordinate i, find the optimal h[i] given all other h[j]. Since C₅ = max_k sum_j h[j](1-h[j+k]), changing h[i] only affects terms where i appears, which is O(n) terms. The optimization over h[i] is a 1D problem.

The terms involving h[i] in C(k) are:
- For shift where i appears: h[i](1-h[i+shift]) or h[i-shift](1-h[i]) or h[i](1-h[i])

- Linear: h[i] * (1-h[i+s]) = h[i] - h[i]h[i+s] → linear in h[i] (h[i+s] is fixed)
- Or: h[i-s] * (1-h[i]) = h[i-s] - h[i-s]*h[i] → linear in h[i]
- Or: h[i](1-h[i]) → quadratic

The max of these is piecewise linear or quadratic, which can be optimized in 1D.

   b. Determine how C(k) changes as a function of h[i]
   c. Find the h[i] that minimizes max_k C(k) (1D optimization)
   d. Update h[i]
2. Repeat for several passes

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj_to_sum(h, n / 2.0)

def _obj_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = n - 1 - k
        wk = w[k] * dx
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif 0 < sh < n:
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        elif -n < sh < 0:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _slsqp_obj(x, alpha):
    return _obj_smooth(x, alpha)

def _slsqp_jac(x, alpha):
    return _grad_smooth(x, alpha)

def _con_fun(x):
    return x.sum() - len(x) * 0.5

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter=800):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_slsqp_obj, h.copy(), jac=_slsqp_jac,
                       args=(alpha,), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pgd(h, alpha, nsteps, step0, mu=0.85):
    n = len(h)
    target = n / 2.0
    v = np.zeros(n)
    f = _eval(h)
    step = step0
    for t in range(nsteps):
        g = _grad_smooth(h, alpha)
        v = mu * v + step * g
        h_new = _proj_to_sum(h - v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-18:
            f = f_new
            h = h_new
            step = min(step * 1.01, step0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return h

    m = c.max()

def _grad_smooth(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
