import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, 0.0), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

                j = i + off

                    j = abs(j) % n
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            if i == j:

            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d

                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sa_phase(h, n, dx, rng, deadline, T_init, T_decay, target):
    best_c = compute_c5(h, dx)
    best_h = h.copy()
    h = h.copy()
    T = T_init
    full_len = 2 * n - 1
    while time.time() < deadline:
        T *= T_decay
        if T < 1e-8:
            T = 1e-8
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-8:]
        i = rng.randint(n)
        if rng.random() < 0.5:
            k_idx = top_k[rng.randint(len(top_k))]
            off = k_idx - (n - 1)
            j_candidates = []
            if off >= 0:
                j_candidates = [x for x in range(max(0, off), n) if x - off >= 0]
            else:
                j_candidates = [x for x in range(max(0, -off), n) if x + off >= 0]
            if j_candidates:
                i = j_candidates[rng.randint(len(j_candidates))]
                j = i + off
                if j < 0 or j >= n or j == i:
                    j = rng.randint(n)
            else:
                j = rng.randint(n)
        else:
            j = rng.randint(n)
            if j == i:
                continue
        d = rng.uniform(0.003, 0.02) * (1.0 if rng.random() < 0.5 else -1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i], h[j] = ni, nj
        c5n = compute_c5(h, dx)
        delta = c5n - best_c
        if delta < 0 or rng.random() < np.exp(-delta / T):
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def fine_search(h, n, dx, rng, deadline, n_passes, scale0):
    best_c = compute_c5(h, dx)
    h = h.copy()
    for p in range(n_passes):
        scale = scale0 * (0.65 ** p)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-10:]
        improved = False
        for _ in range(35000):
            if time.time() >= deadline:
                return h, best_c
            if rng.random() < 0.45:
                k_idx = top_k[rng.randint(len(top_k))]
                off = k_idx - (n - 1)
                i = rng.randint(max(1, n - abs(off)))
                j = i + off
                if j < 0 or j >= n or j == i:
                    j = rng.randint(n)
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if j == i:
                    continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c5n = compute_c5(h, dx)
            if c5n < best_c - 1e-15:
                best_c = c5n
                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not improved and scale < scale0 * 0.05:
            break
    return h, best_c

def perturb_struct(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(5, n - 5))
    elif kind == 1:
        f = rng.randint(1, 5)
        ph = rng.uniform(0, 6.2832)
        hp += rng.uniform(0.05, 0.15) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.05, n)
    elif kind == 3:
        i0 = rng.randint(0, n - 15)
        hp[i0:i0 + 15] += rng.uniform(0.04, 0.12)
        i1 = rng.randint(0, n - 15)
        hp[i1:i1 + 15] -= rng.uniform(0.04, 0.12)
    else:
        hp = (hp + np.roll(hp, 2)) / 2.0
        hp = (hp + np.roll(hp, 3)) / 2.0
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = repair(h_init.copy(), n)
    best_h, best_c = h.copy(), compute_c5(h, dx)

    def elapsed():
        return time.time() - t_start

    def left():
        return budget_s - elapsed()

    # Phase 1: Gradient optimization (50s)
    t_end = t_start + 50
    for alpha in [200, 1000, 5000, 20000]:
        if elapsed() > 50 or left() < 400:
            break
        for pen in [2000, 20000]:
            if elapsed() > 50:
                break
            hh, cc = lbfgs_opt(h, n, alpha, target, pen, 300)
            if cc < best_c:
                h, best_h, best_c = hh, hh.copy(), cc

    # Phase 2: SLSQP refinement (50s)
    t_end = t_start + 100
    for alpha in [5000, 20000, 100000]:
        if elapsed() > 100 or left() < 350:
            break
        hh, cc = slsqp_opt(h, n, alpha, target, 600)
        if cc < best_c:
            h, best_h, best_c = hh, hh.copy(), cc

    # Phase 3: SA-like exploration (250s)
    t_end = t_start + 350
    if left() > 300:
        hh, cc = sa_phase(h, n, dx, rng, t_end, 0.004, 0.999998, target)
        if cc < best_c:
            h, best_h, best_c = hh, cc
            h = best_h.copy()

    # Phase 4: Gradient re-opt (40s)
    t_end = t_start + 390
    for alpha in [20000, 80000, 300000]:
        if elapsed() > 390 or left() < 200:
            break
        hh, cc = lbfgs_opt(h, n, alpha, target, 5000, 400)
        if cc < best_c:
            h, best_h, best_c = hh, hh.copy(), cc

    # Phase 5: Multi-restart (100s)
    t_end = t_start + 490
    for r in range(8):
        if elapsed() > 490 or left() < 180:
            break
        hp = perturb_struct(h, n, rng, r % 5)
        hh, cc = lbfgs_opt(hp, n, 5000, target, 5000, 200)
        hh, cc = slsqp_opt(hh, n, 50000, target, 300)
        if cc < best_c:
            h, best_h, best_c = hh, hh.copy(), cc

    # Phase 6: Fine local search (200s)
    t_end = t_start + 690
    if left() > 150:
        h, best_c = fine_search(h, n, dx, rng, t_end, 8, 0.008)
        best_h = h.copy()

    # Phase 7: Ultra-high alpha gradient (50s)
    t_end = t_start + 740
    for alpha in [100000, 500000, 1000000]:
        if elapsed() > 740 or left() < 100:
            break
        hh, cc = lbfgs_opt(h, n, alpha, target, 10000, 500)
        if cc < best_c:
            h, best_h, best_c = hh, hh.copy(), cc

    # Phase 8: SLSQP final (30s)
    if left() > 80:
        hh, cc = slsqp_opt(h, n, 500000, target, 800)
        if cc < best_c:
            h, best_h, best_c = hh, hh.copy(), cc

    # Phase 9: Ultra-fine search (remaining - 20s)
    t_end = t_start + budget_s - 20
    if left() > 25:
        h, best_c = fine_search(h, n, dx, rng, t_end, 12, 0.002)
        best_h = h.copy()

    # Final: tiny search
    t_end = time.time() + max(2, left() - 3)
    if left() > 5:
        h, best_c = fine_search(h, n, dx, rng, t_end, 6, 0.0005)
        best_h = h.copy()

    best_h = repair(best_h, n)
    best_c = compute_c5(best_h, dx)
    return (best_h, best_c, n)
