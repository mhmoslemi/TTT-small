def _coord_step(h, n, step_bisect=25):
    """One coordinate descent sweep: for each i, find optimal h[i] given others fixed."""
    c = np.correlate(h, 1.0 - h, mode='full')  # O(n^2)
    n_total = 2 * n - 1
    
    for i in range(n):
        # Compute a_k = dc_k/dh[i] for all k
        a = np.zeros(n_total)
        for k in range(n_total):
            s = k - (n - 1)
            if s >= 0:
                # dc_k/dh[i] = (1-h[i-s]) if s <= i < n, minus h[i+s] if i <= n-1-s
                if s <= i:
                    a[k] += (1.0 - h[i - s])
                if i <= n - 1 - s:
                    a[k] -= h[i + s]
            else:
                t = -s
                # dc_k/dh[i] = (1-h[i+t]) if i <= n-1-t, minus h[i-t] if t <= i
                if i <= n - 1 - t:
                    a[k] += (1.0 - h[i + t])
                if t <= i:
                    a[k] -= h[i - t]
        # Minimize max_k (c[k] + a[k]*delta) over delta in [-h[i], 1-h[i]]
        lo = -h[i]
        hi = 1.0 - h[i]
        # Bisection on subgradient
        for _ in range(step_bisect):
            mid = (lo + hi) * 0.5
            # subgradient = a[argmax_k (c[k] + a[k]*mid)]
            vals = c + a * mid
            k_star = vals.argmax()
            if a[k_star] < 0:

            else:
                lo = mid  # need to go lower
        delta = (lo + hi) * 0.5
        h[i] += delta
        c += a * delta
        h[i] = np.clip(h[i], 0.0, 1.0)
    
    return h
