```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), 1.0 - h[i] if d > 0 else h[i])
            h[i] += a * np.sign(d)
            d -= a * np.sign(d)
            if abs(d) < 1e-14:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _project(v, target):
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _proj_grad(h, dx, alpha, t_end, max_iter=1000):
    n = len(h)
    target = n * 0.5
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.1
    stall = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        C = np.correlate(h, 1.0 - h, mode='full') * dx
        Cmax = C.max()
        ex = np.exp(np.clip(alpha * (C - Cmax), -600, 600))
        w = ex / ex.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(C)):
            if w[k] < 1e-14:
                continue
            s = k - (n - 1)
            wk = w[k] * dx
            if s == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:n-s] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:-s]
            else:
                s2 = -s
                grad[s2:] += wk * (1.0 - h[:-s2])
                grad[:n-s2] -= wk * h[s2:]
        
        # Gradient step with backtracking
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm  # normalize
        
        step = lr
        improved = False
        for _ in range(30):
            h_new = _project(h - step * grad, target)
            c_new = _c5(h_new, dx)
            if c_new < best:
                h = h_new
                best = c_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            stall += 1
            if stall > 50:
                # restart with perturbation
                h = _project(h + 0.01 * np.random.randn(n), target)
                stall = 0
            lr *= 0.8
        else:
            stall = 0
            if it % 200 == 0:
                alpha *= 1.5
            if it % 100 == 0:
                lr = min(lr * 1.2, 1.0)
    
    return h, best

def _obj_lbfgs(x, dx, alpha, pen):
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    Cmax = C.max()
    soft = Cmax + np.log(np.sum(np.exp(alpha * (C - Cmax)))) / alpha
    penalty = pen * (x.sum() - len(x) * 0.5) ** 2
    return soft + penalty

def _jac_lbfgs(x, dx, alpha, pen):
    n = len(x)
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    Cmax = C.max()
    ex = np.exp(np.clip(alpha * (C - Cmax), -700, 700))
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(len(C)):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif s > 0:
            grad[:n-s] += wk * (1.0 - x[s:])
            grad[s:] -= wk * x[:-s]
        else:
            s2 = -s
            grad[s2:] += wk * (1.0 - x[:n-s2])
            grad[:n-s2] -= wk * x[s2:]
    # Penalty gradient
    d = x.sum() - n * 0.5
    grad += 2 * pen * d * np.ones(n)
    return grad
```
