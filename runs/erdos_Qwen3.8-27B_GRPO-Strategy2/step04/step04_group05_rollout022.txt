import time

    s = h.sum()

            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

    return float(np.max(c)) * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_c(h, k, n):
    """Gradient of c_k w.r.t. h, where k is the index in the full correlation."""
    g = np.zeros(n)
    L = k - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def _pnorm(h):
    p = _P
    c = _corr(h)
    cp = np.maximum(c, 0.0)
    return (np.sum(cp ** p)) ** (1.0 / p)

    for p in p_list:

        h = bh.copy()

            cp = np.maximum(c, 0.0)
            val = (np.sum(cp ** p)) ** (1.0 / p)
            val += _PEN * (x.sum() - n * 0.5) ** 2

            cp = np.maximum(c, 0.0)

            g += 2.0 * _PEN * (x.sum() - n * 0.5) * np.ones(n)
            return g

            if c < bc:

        except:

    return bh, bc

def repair(h, n):

    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)

        if d > 0:

        else:

            d -= a

def flatten_phase(h, n, dx, t0, tlim):
    """Minimize variance of top-m correlations to flatten the peak."""
    global _gN
    _gN = n
    bh, bc = h.copy(), c5v(h, dx)
    for m in [5, 10, 20, 30]:
        h = bh.copy()
        step = 0.05
        stall = 0
        for _ in range(400):
            if time.time() - t0 > tlim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            idx = np.argsort(c)[::-1][:m]
            cm = c[idx]
            target = cm.mean()
            g = np.zeros(n)
            for ki in idx:
                g += grad_c(h, ki, n)
            g = (c[idx] - target)[:, None] * np.array([grad_c(h, ki, n) for ki in idx]).T
            g = (c[idx] - target) @ np.array([grad_c(h, ki, n) for ki in idx]).T
