def grad_max_corr(h, n, top_m=1):
    """Subgradient of max_k corr[k] w.r.t. h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)

            grad[j] += g
    grad /= len(indices)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_A(h, n):
    """Compute coefficient matrix A[m,j] for corr[m] as linear function of h[j]."""
    n_corr = 2 * n - 1
    lags = (np.arange(n_corr) - (n - 1))[:, None].astype(float)
    js = np.arange(n)[None, :].astype(float)
    idx1 = np.round(js - lags).astype(int)
    idx2 = np.round(js + lags).astype(int)
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(mask1, h[np.clip(idx1, 0, n-1)], 0.0)
    h2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    return (1.0 - h1) * mask1.astype(float) - h2 * mask2.astype(float)

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        return repair(res.x, n)
    except:
        return h

def pair_descent(h, n, deadline, top_k=12):
    """Pair descent with exact 1D minimization over top-k lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    best_c5 = float(np.max(corr) * dx)
    n_corr = len(corr)
    
    d_candidates = np.array([0.002, -0.002, 0.005, -0.005, 0.01, -0.01,
                             0.02, -0.02, 0.04, -0.04, 0.08, -0.08])
    
    n_stagnant = 0
    for it in range(5000):
        if time.time() >= deadline:
            break
        
        A = compute_A(h, n)
        top_idx = np.argsort(corr)[::-1][:top_k]
        A_top = A[top_idx, :]
        b_vec = corr[top_idx].copy()
        
        best_impr = 1e-14
        best_j, best_l, best_d = -1, -1, 0.0
        
        for j in range(n):
            e_all = A_top[:, j:j+1] - A_top[:, j+1:]  # (top_k, n-j-1)
            h_l = h[j+1:]
            d_min = np.maximum(-h[j], -(1.0 - h_l))
            d_max = np.minimum(1.0 - h[j], h_l)
            
            # Evaluate f(l, d) = max_k(b_k + d * e_k[l])
            # Shape: (n_j-1, len(d_candidates))
            if len(d_candidates) > 0:
                t = b_vec[:, None, None] + d_candidates[None, None, :] * e_all[:, :, None]
                f = np.max(t, axis=0)  # (n_j-1, n_d)
            else:
                f = np.zeros((n-j-1, 0))
            
            # Boundary values
            t_min = b_vec[:, None] + d_min[None, :] * e_all
            f_min = np.max(t_min, axis=0)
            t_max = b_vec[:, None] + d_max[None, :] * e_all
            f_max = np.max(t_max, axis=0)
            
            f_all = np.vstack([f, f_min[None, :], f_max[None, :]])
            col_best = np.min(f_all, axis=0)

                if row_best < len(d_candidates):

                    bd = d_min[l_local]

                if bd == 0:
                    continue
                best_impr = best_c5 / dx - col_best[l_local]
                best_j, best_l, best_d = j, j + 1 + l_local, bd

            # Try a random perturbation to escape

            best_c5 = float(np.max(corr) * dx)

        h[best_j] += best_d
        h[best_l] -= best_d

            h[best_l] += best_d

            if n_stagnant > 5:
                break
    
    return h, best_c5

def fine_local_search(h, n, rng, deadline):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    best_c5 = float(np.max(corr) * dx)
    
    scales = [0.003, 0.001, 0.0005, 0.0002, 0.00008]
    for scale in scales:
        improved = False
        n_per = 30000
        for _ in range(n_per):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:
                continue
            h[i], h[j] = ni, nj
            c5n = float(np.max(np.correlate(h, 1.0-h, 'full')) * dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def compute_A(h, n):
    n_corr = 2 * n - 1
    lags = (np.arange(n_corr) - (n - 1))[:, None]
    js = np.arange(n)[None, :]
    idx1 = js - lags
    idx2 = js + lags
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(mask1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(mask2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return (1.0 - h1) * mask1.astype(float) - h2 * mask2.astype(float)

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_step(h, n, alpha, target, pen, maxiter):
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
