import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    # Simple repair ensuring 0≤h≤1 and sum h = n/2
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = project_onto_box_hyperplane(h, n)
    return h

def project_onto_box_hyperplane(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr_shift(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        valid = np.arange(n - shift)
        grad[valid] += 1.0
        grad[valid] -= h[valid + shift]
        grad[valid + shift] -= h[valid]
    else:
        valid = np.arange(-shift, n)
        grad[valid] += 1.0
        grad[valid] -= h[valid + shift]
        grad[valid + shift] -= h[valid]
    return grad

def subgradient_nesterov(h, n, t_end, lr_init=0.01, momentum=0.9):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    v = np.zeros(n)  # momentum
    h_averaged = h.copy()
    count = 1
    it = 0
    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        grad = grad_corr_shift(h, n, shift) * dx
        
        # Nesterov step
        v = momentum * v + (1 - momentum) * grad
        lr = lr_init / (1 + it * 0.001)
        h = h - lr * v
        h = project_onto_box_hyperplane(h, n)
        
        h_averaged = h_averaged * (count / (count + 1)) + h * (1 / (count + 1))
        count += 1
        it += 1
        
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        c5_avg = compute_c5(h_averaged, n)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_averaged.copy()

        if s > target:

    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

    def obj(v):
        return soft_obj(v, alpha)
    bounds = [(0.0, 1.0)] * n
    # Add sum constraint via penalty
    pen = 1e6
    def obj_pen(v):
        return soft_obj(v, alpha) + pen * (v.sum() - n/2.0)**2

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

                        break

                    h[j] = hj
