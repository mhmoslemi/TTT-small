import numpy as np

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softgrad(h, n, beta):
    c = _corr(h)
    m = c.max()
    s = np.clip(beta * (c - m), -800, 800)
    s -= s.max()
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = pw[k]
        if pk < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            g += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += pk * (1.0 - h[:n - L])
            g[:n - L] -= pk * h[L:]
        else:
            m2 = -L
            g[:n - m2] += pk * (1.0 - h[m2:])
            g[m2:] -= pk * h[:n - m2]
    g -= g.mean()
    return g
