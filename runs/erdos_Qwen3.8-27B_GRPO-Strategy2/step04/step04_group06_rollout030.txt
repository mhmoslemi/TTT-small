    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _c5_eval(h, n, i, j, d):
    h[i] += d
    h[j] -= d
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    h[i] -= d
    h[j] += d
    return val

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
        hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dx

def _grad_step(h, n, alpha, lr, budget_end):

def _sa_optimize(h, n, t_end, rng, T_init=2e-4, reheat_T=5e-5, patience_max=50000):
    """Simulated annealing with optimal pair moves (ternary search for d)."""
    best_c5 = _c5(h, n)
    best_h = h.copy()
    c5_cur = best_c5
    T = T_init
    patience = 0
    no_best = 0

    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            T *= 0.99995
            continue

        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi:

        a, b = d_lo, d_hi

            c2 = _c5_eval(h, n, i, j, m2)
            if c1 < c2:

            else:

        c5_new = _c5_eval(h, n, i, j, d_opt)

            h[j] -= d_opt

                no_best = 0
            else:

                           constraints=[{'type': 'eq', 'fun': _con}],
                           options={'maxiter': 400, 'ftol': 1e-18})
            hc = _repair(res.x, best_n)
            c5c = _c5(hc, best_n)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc
        except Exception:
            pass

    # Phase 2: SA with optimal pair moves (65% of budget)
    t2 = t0 + budget_s * 0.77
    h_sa = best_h.copy()
    best_h, best_c5 = _sa_optimize(h_sa, best_n, t2, rng,
                                    T_init=2e-4, reheat_T=5e-5, patience_max=40000)

    # Phase 3: Fine polish (10% of budget)
    t3 = t0 + budget_s * 0.90
    best_h, best_c5 = _fine_polish(best_h, best_n, t3, rng)

    # Phase 4: Try alternative resolution (5% of budget)
    t4 = t0 + budget_s - 10
    if best_n < 200:
        n_alt = best_n + 40
        h_alt, c5_alt = _try_resolution(best_h, best_n, n_alt, t4 - 10, rng)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt
            best_n = n_alt
    # One more fine polish
    best_h, best_c5 = _fine_polish(best_h, best_n, t0 + budget_s - 5, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
