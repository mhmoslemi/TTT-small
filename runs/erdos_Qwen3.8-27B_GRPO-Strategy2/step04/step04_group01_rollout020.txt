def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

        for _bt in range(30):
            h_new = _proj_sum(h - lr * g, n * 0.5)
            f_new = _smooth_obj(h_new, alpha)
            if f_new < f - 1e-10:
                h = h_new
                f = f_new
                lr = min(lr * 1.1, lr0)
                break
            lr *= 0.5
        else:
            break
        if lr < 1e-9:
            break
    return h

def _lag_grad(h, k_idx):
    """Gradient of c[k_idx] specifically."""
    n = len(h)
    sh = (n - 1) - k_idx
    dx = 2.0 / n
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:sh+cnt])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _nesterov(h, alpha, nsteps, lr0):
    n = len(h)
    v = np.zeros(n)
    mu = 0.85
    lr = lr0
    f = _smooth_obj(h, alpha)
    t_val = 1.0
    got = True
    while got and nsteps > 0:
        got = False
        g = _smooth_grad(h, alpha)
        t_val += 1
        y = h + (t_val - 1) / t_val * (-v)
        y = np.clip(y, 0, 1)
        g_y = _smooth_grad(y, alpha)
        h_new = _proj_sum(h - lr * g_y, n * 0.5)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f - 1e-11:
            v = h_new - h
            h = h_new
            f = f_new
            nsteps -= 1
            got = True
            lr = min(lr * 1.02, lr0)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h
