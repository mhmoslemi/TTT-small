from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(np.where(diff > 0, 1.0 - h, h))
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_c(h, k, n):
    """Gradient of c_k with respect to h, for lag k (0-indexed full correlation)."""
    g = np.zeros(n)
    L = k - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

def _topk_obj(h, K):
    """Objective: max of top K correlations (smoothed slightly)."""
    c = _corr_full(h)
    # Use top-K with small epsilon smoothing
    cs = np.sort(c)[::-1]
    topk = cs[:K]
    # Soft max of top K
    m = topk.max()
    s = (topk - m) / 0.001  # small smoothing
    s -= s.max()
    return m + 0.001 * np.log(np.exp(s).sum()) + 1e-6 * (h.sum() - len(h)/2.0)**2

def _topk_grad(h, K):
    """Gradient of top-K smoothed objective."""
    c = _corr_full(h)
    n = len(h)
    cs = np.sort(c)[::-1]
    K_actual = min(K, 2*n-1)
    topk_idx = np.argsort(c)[::-1][:K_actual]
    topk = c[topk_idx]
    m = topk.max()
    eps = 0.001
    s = (topk - m) / eps
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, idx in enumerate(topk_idx):
        g += w[ii] * _grad_c(h, idx, n)
    g /= eps
    g += 2e-6 * (h.sum() - n/2.0) * np.ones(n)
    return g

def _active_grad(h, frac=0.05):
    """Average gradient over active constraints (top fraction of correlations)."""
    c = _corr_full(h)
    n = len(h)
    cm = c.max()
    spread = cm - c.min()
    if spread < 1e-12:
        spread = 1.0
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        g += _grad_c(h, k, n)
    g /= len(active)
    return g

Where c_k(h) = Σ_i h_i(1-h_{i+k}) (for the appropriate range).

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(np.where(diff > 0, 1.0 - h, h))
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_one(h, k, n):
    g = np.zeros(n)
    L = k - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g

def _active_grad(h, frac=0.04):
    c = _corr_full(h)
    n = len(h)
    cm = c.max()
    spread = cm - c.min()
    if spread < 1e-12:
        spread = 1.0
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        g += _grad_one(h, k, n)
    g /= len(active)
    return g

def _topk_obj(h, K):
    c = _corr_full(h)
    n = len(h)
    cm = c.max()
    idx = np.argsort(c)[::-1][:K]
    topk = c[idx]
    m = topk.max()
    eps = 0.0005
    s = (topk - m) / eps
    s -= s.max()
    return m + eps * np.log(np.sum(np.exp(s))) + 1e-8 * (h.sum() - n / 2.0) ** 2

def _topk_grad(h, K):
    c = _corr_full(h)
    n = len(h)
    cm = c.max()
    idx = np.argsort(c)[::-1][:K]
    topk = c[idx]
    m = topk.max()
    eps = 0.0005
    s = (topk - m) / eps
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        g += w[ii] * _grad_one(h, k, n)
    g /= eps
    g += 2e-8 * (h.sum() - n / 2.0)
    return g

def _lbfgs_polish(h, n, dx, tl):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    bounds = [(0.0, 1.0)] * n
    for K in [3, 5, 8, 12, 20]:
        if tl() < 10:
            break
        try:
            res = minimize(_topk_obj, bh.copy(), args=(K,),
                           jac=lambda x: _topk_grad(x, K),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-15})
            hc = _repair(res.x, n)
            cv = _c5(hc, dx)
            if cv < bc5:
                bc5, bh = cv, hc
        except Exception:
            pass
    return bh, bc5

def _grad_descent(h, n, dx, tl, max_iter=1500):
    bh = h.copy()
    bc5 = _c5(h, dx)
    v = np.zeros(n)
    step = 0.015
    mu = 0.85
    for _ in range(max_iter):
        if tl() < 5:
            break
        g = _active_grad(h)
        gc = g - g.mean()
        gn = np.linalg.norm(gc)
        if gn < 1e-13:
            break
        gc /= gn
        v = mu * v - step * gc
        h_new = _repair(h + v, n)
        cn = _c5(h_new, dx)
        if cn < _c5(h, dx) - 1e-16:
            h = h_new
            step = min(step * 1.04, 0.06)
            if cn < bc5:
                bc5, bh = cn, h.copy()
        else:
            v *= 0.3
            step *= 0.6
            if step < 1e-7:
                break
    return bh, bc5

def _pairwise(h, n, dx, tl, max_iter=4000):
    bh = h.copy()
    bc5 = _c5(h, dx)
    c5c = bc5
    steps = np.array([0.0005, 0.001, 0.003, 0.007, 0.014, 0.028,
                      -0.0005, -0.001, -0.003, -0.007, -0.014, -0.028])
    for _ in range(max_iter):
        if tl() < 5:
            break
        g = _active_grad(h)
        gc = g - g.mean()
        order = np.argsort(np.abs(gc))[::-1]
        improved = False
        for ii in range(min(25, n)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 12, n)):
                j = order[jj]
                bd, bv = 0.0, c5c
                for d in steps:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < bv:
                            bv, bd = c, d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0.0:
                    h[i] += bd; h[j] -= bd
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < bc5:
                        bc5, bh = c5c, h.copy()
                    break
        if not improved:
            for _ in range(40):
                i, j = np.random.randint(n, 2)
                if i == j:
                    continue
                d = np.random.uniform(-0.015, 0.015)
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c5c:
                        c5c = c
                        improved = True
                        if c5c < bc5:
                            bc5, bh = c5c, h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
            if not improved:
                break
    return bh, bc5

def _interp(h_src, ns, nt):
    xs = np.linspace(0, 2, ns, endpoint=False)
    xt = np.linspace(0, 2, nt, endpoint=False)
    return _repair(np.interp(xt, xs, h_src), nt)

def _optimize_block(h, n, dx, tl, label):
    h, c5 = _lbfgs_polish(h, n, dx, tl)
    if tl() > 15:
        h, c5 = _grad_descent(h, n, dx, tl, max_iter=1200)
    if tl() > 10:
        h, c5 = _pairwise(h, n, dx, tl, max_iter=3000)
    return h, c5
