    # Gradient (vectorized)
    lags = np.arange(2*n-1) - (n-1)
    j_idx = np.arange(n)[:, None]
    idx1 = j_idx - lags[None, :]
    idx2 = j_idx + lags[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n-1)
    idx2c = np.clip(idx2, 0, n-1)
    term1 = np.sum(w[None, :] * np.where(valid1, x[idx1c], 0.0), axis=1)
    term2 = np.sum(w[None, :] * np.where(valid2, x[idx2c], 0.0), axis=1)
    grad = (1.0 - term1 - term2) / alpha + 2.0 * pen * (np.sum(x) - target)
    return obj, grad

def lse_obj_wrapper(x, alpha, target, pen):
    return lse_obj_and_grad(x, alpha, target, pen)[0]

def lse_grad_wrapper(x, alpha, target, pen):
    return lse_obj_and_grad(x, alpha, target, pen)[1]

def gradient_descent_step(h, n, dx, alpha, target, pen, step_size, deadline):
    """One gradient descent step with line search."""
    obj, grad = lse_obj_and_grad(h, alpha, target, pen)
    # Remove mean component to preserve sum constraint approximately
    grad_centered = grad - np.mean(grad)
    # Try different step sizes
    best_h = h.copy()
    best_obj = obj
    for ss in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
        trial = h - ss * grad_centered
        trial = np.clip(trial, 0, 1)
        trial = repair(trial, n)
        trial_obj = lse_obj_wrapper(trial, alpha, target, pen)
        if trial_obj < best_obj:
            best_h = trial
            best_obj = trial_obj
    return best_h, best_obj

def gradient_local_search(h, n, dx, rng, deadline, n_iters=5000):
    """Gradient-guided local search with occasional random perturbations."""
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        
        c5_val, corr_arr, k_star = compute_c5_detail(best_h, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
        
        # Compute gradient of the actual max (subgradient)
        n_c = len(corr_arr)
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            ji = j - lag
            if 0 <= ji < n:
                g += (1.0 - best_h[ji])
            jj = j + lag
            if 0 <= jj < n:
                g -= best_h[jj]
            grad[j] = g
        
        # Also include gradients from near-max c_k (top-5)
        top_k = np.argsort(corr_arr)[-5:]
        for kk in top_k:
            ll = kk - (n-1)
            for j in range(n):
                g = 0.0
                ji = j - ll
                if 0 <= ji < n:
                    g += (1.0 - best_h[ji])
                jj = j + ll
                if 0 <= jj < n:
                    g -= best_h[jj]
                grad[j] += g * 0.3
        
        # Center gradient (preserve sum)
        grad -= np.mean(grad)
        
        # Take a step
        scale = 0.01 * (1.0 + 0.5 * np.random.randn())
        scale = max(0.001, min(0.05, scale))
        trial = best_h - scale * grad
        trial = np.clip(trial, 0, 1)
        trial = repair(trial, n)
        c5_trial = compute_c5(trial, dx)
        
        if c5_trial < best_c5 - 1e-15:
            best_h = trial
            best_c5 = c5_trial
        
        # Occasionally do a random perturbation
        if it % 500 == 0 and it > 0:
            pert = np.random.randn(n) * 0.02
            pert -= np.mean(pert)
            trial2 = best_h + pert
            trial2 = np.clip(trial2, 0, 1)
            trial2 = repair(trial2, n)
            c5_2 = compute_c5(trial2, dx)
            if c5_2 < best_c5:
                best_h = trial2
                best_c5 = c5_2
    
    return best_h, best_c5

def pair_local_search(h, n, dx, rng, deadline, fine=True):
    """Random pair swap local search."""
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    if fine:
        scales = [0.002, 0.0008, 0.0003, 0.0001]
        per_scale = 80000
    else:
        scales = [0.02, 0.008, 0.003, 0.001, 0.0004]
        per_scale = 40000
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0+1e-9 or nj < -1e-9 or nj > 1.0+1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.003:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_detail(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr, int(np.argmax(corr))

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj_params(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    exp_s = np.exp(s)
    se = np.sum(exp_s)
    obj = m + np.log(se) / alpha + pen * (np.sum(x) - target) ** 2
    w = exp_s / se
    n = len(x)
    lags = np.arange(2 * n - 1) - (n - 1)
    jx = np.arange(n)[:, None]
    idx1 = jx - lags[None, :]
    idx2 = jx + lags[None, :]
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    i1c = np.clip(idx1, 0, n - 1)
    i2c = np.clip(idx2, 0, n - 1)
    t1 = np.sum(w[None, :] * np.where(v1, x[i1c], 0.0), axis=1)
    t2 = np.sum(w[None, :] * np.where(v2, x[i2c], 0.0), axis=1)
    grad = (1.0 - t1 - t2) / alpha + 2.0 * pen * (np.sum(x) - target)
    return obj, grad

def lse_f(x, a, t, p):
    return lse_obj_params(x, a, t, p)[0]

def lse_g(x, a, t, p):
    return lse_obj_params(x, a, t, p)[1]

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9, 'maxls': 50})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def subgrad_step(h, n, dx, top_m, scale):
    c5v, corr, kstar = compute_c5_detail(h, dx)
    top_k = np.argsort(corr)[-top_m:]
    n_half = n - 1
    j = np.arange(n)
    grad = np.zeros(n)
    for kk in top_k:
        lag = kk - n_half
        ji = j - lag
        jj = j + lag
        vi = (ji >= 0) & (ji < n)
        vj = (jj >= 0) & (jj < n)
        grad[vi] += (1.0 - h[ji[vi]])
        grad[vj] -= h[jj[vj]]
    grad -= np.mean(grad)
    trial = h - scale * grad
    trial = np.clip(trial, 0.0, 1.0)
    trial = repair(trial, n)
    return trial, c5v

def subgrad_search(h, n, dx, rng, deadline, n_iters, top_m=5):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    scale = 0.008
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        trial, _ = subgrad_step(best_h, n, dx, top_m, scale)
        c5t = compute_c5(trial, dx)
        if c5t < best_c5 - 1e-15:
            best_h = trial
            best_c5 = c5t
            if it % 2000 == 0 and it > 0:
                scale = max(0.001, scale * 0.85)
        else:
            scale = max(0.0005, scale * 0.92)
        if it % 800 == 0 and it > 0:
            pert = rng.randn(n) * 0.015
            pert -= np.mean(pert)
            t2 = np.clip(best_h + pert, 0, 1)
            t2 = repair(t2, n)
            c2 = compute_c5(t2, dx)
            if c2 < best_c5:
                best_h = t2
                best_c5 = c2
                scale = 0.008
    return best_h, best_c5

def pair_search(h, n, dx, rng, deadline, fine):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    if fine:
        scales = [0.003, 0.0012, 0.0005, 0.00018]
        per_scale = 70000
    else:
        scales = [0.025, 0.01, 0.004, 0.0015, 0.0006]
        per_scale = 55000
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and scale < 0.004:
            break
    return h, best_c5

def targeted_swap(h, n, dx, rng, deadline, n_iters):
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    n_half = n - 1
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        c5v, corr, kstar = compute_c5_detail(best_h, dx)
        lag = kstar - n_half
        j = np.arange(n)
        contrib = np.zeros(n)
        ji = j - lag
        jj = j + lag
        vi = (ji >= 0) & (ji < n)
        vj = (jj >= 0) & (jj < n)
        contrib[vi] = best_h[ji[vi]] * (1.0 - best_h[jj[vi]])
        if np.sum(contrib) < 1e-15:
            break
        p_i = contrib / np.sum(contrib)
        i = rng.choice(n, p=p_i)
        best_j = (i + lag) % n
        pool = list(range(max(0, best_j - 3), min(n, best_j + 4)))
        j = rng.choice(pool)
        if i == j:
            continue
        d = 0.005 * (2.0 * rng.random() - 1.0)
        ni = best_h[i] + d
        nj = best_h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        best_h[i] = ni
        best_h[j] = nj
        c5n = compute_c5(best_h, dx)
        if c5n < best_c5 - 1e-14:
            best_c5 = c5n
        else:
            best_h[i] -= d
            best_h[j] += d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = repair(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS analytical gradient (120s)
    te = t0 + 120
    for alpha in [300, 1500, 6000, 25000]:
        if time.time() >= te:
            break
        for pen in [100, 1000, 10000]:
            if time.time() >= te:
                break
            hh, cc = lbfgs_opt(h, n, alpha, target, pen, 350)
            if cc < best_c5:
                best_h, best_c5 = hh, cc
            h = best_h.copy()

    # Phase 2: Subgradient search (180s)
    if rem() > 250:
        te = time.time() + min(180, rem() - 120)
        best_h, best_c5 = subgrad_search(best_h, n, dx, rng, te, 15000, top_m=5)
        h = best_h.copy()

    # Phase 3: L-BFGS again (60s)
    if rem() > 200:
        te = time.time() + min(60, rem() - 100)
        for alpha in [10000, 40000]:
            if time.time() >= te:
                break
            hh, cc = lbfgs_opt(h, n, alpha, target, 5000, 250)
            if cc < best_c5:
                best_h, best_c5 = hh, cc
            h = best_h.copy()

    # Phase 4: Pair search coarse (120s)
    if rem() > 180:
        te = time.time() + min(120, rem() - 80)
        best_h, best_c5 = pair_search(best_h, n, dx, rng, te, fine=False)
        h = best_h.copy()

    # Phase 5: Targeted swap (100s)
    if rem() > 150:
        te = time.time() + min(100, rem() - 70)
        best_h, best_c5 = targeted_swap(best_h, n, dx, rng, te, 20000)
        h = best_h.copy()

    # Phase 6: Subgradient fine (150s)
    if rem() > 120:
        te = time.time() + min(150, rem() - 60)
        best_h, best_c5 = subgrad_search(best_h, n, dx, rng, te, 20000, top_m=3)
        h = best_h.copy()

    # Phase 7: L-BFGS ultra-high (50s)
    if rem() > 60:
        te = time.time() + min(50, rem() - 30)
        for alpha in [80000, 200000]:
            if time.time() >= te:
                break
            hh, cc = lbfgs_opt(h, n, alpha, target, 10000, 300)
            if cc < best_c5:
                best_h, best_c5 = hh, cc
            h = best_h.copy()

    # Phase 8: Pair search fine (200s)
    if rem() > 60:
        te = time.time() + min(200, rem() - 25)
        best_h, best_c5 = pair_search(best_h, n, dx, rng, te, fine=True)
        h = best_h.copy()

    # Phase 9: Final subgradient polish (30s)
    if rem() > 20:
        te = time.time() + min(30, rem() - 5)
        best_h, best_c5 = subgrad_search(best_h, n, dx, rng, te, 5000, top_m=3)

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
