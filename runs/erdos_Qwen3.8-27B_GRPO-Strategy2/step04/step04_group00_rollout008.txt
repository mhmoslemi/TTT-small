def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _ovl_grad(h, k, n):
    L = k - (n - 1)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] = 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] = 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g

def _subgrad(h, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    idx = np.argsort(c)[::-1][:top_m]
    cm = c[idx[-1]]
    w = c[idx] - cm + 1e-10
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        g += wi * _ovl_grad(h, ki, n)
    return g

    c = np.correlate(x, 1.0 - x, mode='full')

    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)

            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]

    g += 2.0 * lam * (x.sum() - n / 2.0)
