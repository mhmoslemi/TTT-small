def repair(h, n):
    """Ensure h in [0,1], sum = n/2"""
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def project(h, n):
    """Project onto {h in [0,1]^n : sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def c5_of(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for k in idx:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[k]); h[k] += a; diff -= a
            else:
                a = min(-diff, h[k]); h[k] -= a; diff += a
    return h

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def lag_effect_A(h, n, d):
    A = np.zeros(n)
    if d == 0:
        A[:] = 1.0 - 2.0 * h
    elif d > 0:
        A[d:] += (1.0 - h[:n - d])
        if d < n:
            A[:n - d] -= h[d:]
    else:
        d2 = -d
        if d2 < n:
            A[:n - d2] += (1.0 - h[d2:])
            A[d2:] -= h[:n - d2]
    return A

            h = h_new
            c5_cur = c5_new
        else:
            lr *= 0.7
    return h, c5_cur

def slsqp_run(h_init, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def _jac(x):
        return smooth_grad(x, n, alpha)

    def _con(x):
        return float(np.sum(x) - target)

    def _con_j(x):
        return np.ones(n)

    res = minimize(_obj, h_init, jac=_jac, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_j}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res
