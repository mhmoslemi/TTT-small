def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _topk_idx(c, k):
    n = len(c)
    thresh = np.partition(c, n - k)[n - k]
    return np.where(c >= thresh)[0]

def _sf(x, alpha, k):
    c = _corr(x)
    if k < len(c):
        c = c[_topk_idx(c, k)]
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha, k):
    n = len(x)
    c_full = _corr(x)
    if k < len(c_full):
        idx = _topk_idx(c_full, k)
    else:
        idx = np.arange(len(c_full))
    vals = c_full[idx]
    m = vals.max()
    s = np.clip(alpha * (vals - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-14:
            continue
        lag = idx[ii] - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:

        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):

def _opt_slsqp(h, n, alpha, k, maxiter=500):
    bnds = [(0.0, 1.0)] * n

    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_step(h, n, step):
    c = _corr(h)
    kmax = np.argmax(c)
    lag = kmax - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]

    if gnorm < 1e-15:

            if h[ia:ia+bl].min() < 0 or h[ia:ia+bl].max() > 1:
                h[ia:ia+bl] = ba; h[ib:ib+bl] = bb; continue
            if h[ib:ib+bl].min() < 0 or h[ib:ib+bl].max() > 1:
                h[ia:ia+bl] = ba; h[ib:ib+bl] = bb; continue
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n; no_imp = 0
            else:
                h[ia:ia+bl] = ba; h[ib:ib+bl] = bb; no_imp += 1
        else:
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * 0.3 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n; no_imp = 0
            else:
                h[i], h[j] = ni - d, nj + d; no_imp += 1
        if no_imp > 150:
            scale *= 0.88; no_imp = 0
            if scale < 3e-6:
                break
    return h, c5c

def _sa_escape(h, n, rng, t_end):
    c5c = _eval(h)
    h_best = h.copy()
    T = 0.003
    while time.time() < t_end and T > 1e-9:
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = 0.008 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n - c5c < 0 or rng.random() < np.exp(-(c5n - c5c) / T):
            c5c = c5n
            if c5n < _eval(h_best):
                h_best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
        T *= 0.9998
    return h_best, _eval(h_best)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    h160 = _repair(np.array(initial_h_values, dtype=float).copy(), 160)
    c5_160 = _eval(h160)
    best_h = h160.copy()
    best_c5 = c5_160
    best_n = 160

    # Phase 1: SLSQP with top-k, increasing alpha
    if _tl() > 180:
        h_cur = h160.copy()
        c5_cur = c5_160
        for alpha, k in [(500, 8), (2000, 6), (10000, 5), (50000, 4), (200000, 3), (500000, 3)]:
            if _tl() < 50:
                break
            h_new = _opt_slsqp(h_cur, 160, alpha, k, maxiter=600)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new.copy()
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 2: Subgradient with line search
    if _tl() > 120:
        h_sg = best_h.copy()
        c5_sg = best_c5
        step = 0.008
        for _ in range(8000):
            if time.time() > t0 + budget_s - 100:
                break
            h_try = _subgrad_step(h_sg, best_n, step)
            c5_try = _eval(h_try)
            if c5_try < c5_sg - 1e-16:
                c5_sg = c5_try
                h_sg = h_try
                step = min(step * 1.02, 0.025)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()
            else:
                step *= 0.94
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Local search (aggressive)
    if _tl() > 80:
        t_end = time.time() + min(70, _tl() - 15)
        h_ls, c5_ls = _local_search(best_h.copy(), best_n, rng, t_end)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: SA escape + re-search
    if _tl() > 60:
        t_end = time.time() + min(25, _tl() - 20)
        h_sa, c5_sa = _sa_escape(best_h.copy(), best_n, rng, t_end)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        elif c5_sa < best_c5 + 0.002:
            t_end2 = time.time() + min(30, _tl() - 15)
            h_ls2, c5_ls2 = _local_search(h_sa.copy(), best_n, rng, t_end2)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()

    # Phase 5: Try n=200
    if _tl() > 80:
        h200 = _interp(best_h, best_n, 200)
        h200 = _repair(h200, 200)
        h200 = _opt_slsqp(h200, 200, 10000, 5, maxiter=400)
        h200 = _opt_slsqp(h200, 200, 100000, 3, maxiter=400)
        c5_200 = _eval(h200)
        if c5_200 < best_c5:
            best_c5 = c5_200
            best_h = h200.copy()
            best_n = 200
        if _tl() > 30:
            t_end3 = time.time() + min(20, _tl() - 10)
            h_ls3, c5_ls3 = _local_search(best_h.copy(), best_n, rng, t_end3)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()

    # Phase 6: Final SLSQP polish
    if _tl() > 30:
        h_fin = _opt_slsqp(best_h.copy(), best_n, 1000000, 3, maxiter=500)
        c5_fin = _eval(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Ultra-fine local search
    if _tl() > 15:
        t_end = time.time() + min(12, _tl() - 5)
        h_f, c5_f = _local_search(best_h.copy(), best_n, rng, t_end)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Tiny patience moves
    if _tl() > 8:
        h_t = best_h.copy()
        c5_t = best_c5
        for _ in range(30000):
            if time.time() > t0 + budget_s - 2:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 5e-6 * rng.choice([1, 2, 5])
            ni, nj = h_t[i] + d, h_t[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_t[i], h_t[j] = ni, nj
                c5n = _eval(h_t)
                if c5n < c5_t - 1e-17:
                    c5_t = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h_t.copy()
                else:
                    h_t[i], h_t[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
