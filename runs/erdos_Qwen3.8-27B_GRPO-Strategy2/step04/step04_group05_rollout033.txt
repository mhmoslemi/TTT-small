def repair(h, n):

    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr(h, ki, n):
    """Gradient of correlation at shift ki w.r.t. h"""
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

            d = (rng.rand() - 0.5) * 0.015
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = c5v(h, dx)
                if c < cc - 1e-15:
                    h = repair(h, n)
                    cc = c5v(h, dx)
                    imp = True
                    if cc < bc:
                        bc, bh = cc, h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 5:
            break
    return bh, bc

def polish_phase(h, n, dx, rng, t0, lim, cnt=30000):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.004
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def opt_full(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = slp_phase(bh, n, dx, t0, lim - 200)
    if c < bc:
        bh, bc = r, c
    r, c = pnorm_phase(bh, n, dx, t0, lim - 100)
    if c < bc:
        bh, bc = r, c
    r, c = act_phase(bh, n, dx, t0, lim - 60)
    if c < bc:
        bh, bc = r, c
    r, c = slp_phase(bh, n, dx, t0, lim - 40)
    if c < bc:
        bh, bc = r, c
    r, c = cd_phase(bh, n, dx, rng, t0, lim - 15)
    if c < bc:
        bh, bc = r, c
    r, c = polish_phase(bh, n, dx, rng, t0, lim - 2, cnt=20000)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    r, c = opt_full(h0, nl, dx0, seed, t0, budget_s - 350)
    if c < bc:
        bh, bc, bn = r, c, nl

    # Try 2x resolution
    n2 = nl * 2
    if n2 < 1000 and time.time() - t0 < budget_s - 200:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, h0, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = opt_full(h2, n2, d2, seed + 7, t0, budget_s - 100)
        if c < bc:
            bh, bc, bn = r, c, n2

    # Try 3x resolution
    n3 = nl * 3
    if n3 < 1000 and time.time() - t0 < budget_s - 80:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n3, endpoint=False)
        h3 = repair(np.interp(xn, xo, h0, period=2.0), n3)
        d3 = 2.0 / n3
        r, c = opt_full(h3, n3, d3, seed + 13, t0, budget_s - 30)
        if c < bc:
            bh, bc, bn = r, c, n3

    # Perturbed restart on best
    if time.time() - t0 < budget_s - 30:
        rng2 = np.random.RandomState(seed + 500)
        hp = repair(bh + 0.012 * rng2.randn(bn), bn)
        r, c = slp_phase(hp, bn, 2.0 / bn, t0, budget_s - 20)
        if c < bc:
            bh, bc = r, c
        r, c = pnorm_phase(bh, bn, 2.0 / bn, t0, budget_s - 5)
        if c < bc:
            bh, bc = r, c

    # Final polish
    if time.time() - t0 < budget_s - 3:
        rng3 = np.random.RandomState(seed + 999)
        r, c = polish_phase(bh, bn, 2.0 / bn, rng3, t0, budget_s - 1, cnt=15000)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
