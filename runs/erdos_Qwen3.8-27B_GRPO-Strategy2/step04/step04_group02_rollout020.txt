import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

               method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
               options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-8})
    return repair(r.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, tgt / n)
    h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_con(x, target):
    return np.sum(x) - target

def lbfgs_opt(h, n, alpha, target, pen, mi):
    r = minimize(lse_obj, h, args=(alpha, target, pen),
               method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
               options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-8})
    return repair(r.x, n)

def slsqp_opt(h, n, alpha, target, mi):
    r = minimize(lse_obj, h, args=(alpha, target, 0), method='SLSQP',
               bounds=[(0.0, 1.0)] * n,
               constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
               options={'maxiter': mi, 'ftol': 1e-15})
    return repair(r.x, n)

def coord_desc(h, n, dx, deadline):
    c5 = compute_c5(h, dx)
    uni = np.linspace(0.0, 1.0, 13)

            hi = min(1.0, ho + 0.15)
            local = np.linspace(lo, hi, 12)
            grid = np.unique(np.clip(np.concatenate([uni, local, [0.0, 1.0]]), 0, 1))
            bv, bc = ho, c5
            for t in grid:
                h[i] = t
                v = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                if v < bc - 1e-13:
                    bv, bc = t, v
            h[i] = bv
            if bv != ho:
                imp, c5 = True, bc
        h = repair(h, n)
        if not imp:
            break
    return h, c5

def swap_search(h, n, dx, rng, deadline, ns):
    c5 = compute_c5(h, dx)
    scales = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for sc in scales:
        if time.time() >= deadline:
            break
        imp_count = 0
        for _ in range(ns // len(scales)):
            if time.time() >= deadline:
                return h, c5

            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
            if c < c5 - 1e-14:
                c5, imp_count = c, imp_count + 1
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5

def block_search(h, n, dx, rng, deadline, ni):
    c5 = compute_c5(h, dx)
    for _ in range(ni):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 7)
        i, j = rng.randint(0, n - bs), rng.randint(0, n - bs)

            continue
        eps = rng.uniform(0.002, 0.02) * (1.0 if rng.random() < 0.5 else -1.0)
        if np.any(h[i:i+bs] + eps < -1e-9) or np.any(h[i:i+bs] + eps > 1 + 1e-9):
            continue
        if np.any(h[j:j+bs] - eps < -1e-9) or np.any(h[j:j+bs] - eps > 1 + 1e-9):
            continue
        h[i:i+bs] += eps; h[j:j+bs] -= eps
        c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
        if c < c5 - 1e-14:
            c5 = c
        else:
            h[i:i+bs] -= eps; h[j:j+bs] += eps
    return h, c5

    else:

    return repair(hp, n)
