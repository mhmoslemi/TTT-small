import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(20000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_obj(v, n, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    shifted = (corr - m) / alpha
    shifted = np.clip(shifted, -500.0, 500.0)
    return m + alpha * np.log(np.sum(np.exp(shifted))) / len(corr)

def slsqp_obj(v, n, alpha):
    return soft_obj(v, n, alpha)

def slsqp_eq(v, n):
    return v.sum() - n / 2.0

def slsqp_opt(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return h

def coord_desc(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.0008, 0.0003]
    no_improve = 0
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        n_iter = 8 * n
        for _ in range(n_iter):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                found = False
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hid + delta, hjd - delta
                    if ni < -1e-12 or ni > 1.0 + 1e-12:
                        continue
                    if nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-16:
                        best = c5
                        improved = True
                        found = True
                        break
                    h[i], h[j] = hid, hjd
                if found:
                    break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
    return h, best

def fourier_perturb(h, n, rng, n_modes, amp):
    H = np.fft.rfft(h)
    for k in range(1, min(n_modes + 1, len(H) - 1)):
        H[k] += amp * rng.randn(2)
    if len(H) > 1 and n % 2 == 0:
        H[-1] += amp * rng.randn()
    h_new = np.fft.irfft(H, n=n)
    return repair(h_new, n)

def resample(h, n_old, n_new):
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    h_new = np.interp(x_new, x_old, h, period=1.0)
    return repair(h_new, n_new)

def lfgs_obj(v, n, alpha, pen):
    return soft_obj(v, n, alpha) + pen * (v.sum() - n / 2.0) ** 2

def lfgs_opt(h, n, alpha, pen, maxiter):

                       bounds=bounds, options={'maxiter': maxiter,
                       'ftol': 1e-20, 'gtol': 1e-10})
        return repair(res.x, n)
    except Exception:
        return h
