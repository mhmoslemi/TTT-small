            # offset = k - (n-1)
            offset = k - (n - 1)
            if offset >= 0:
                # corr[k] = sum_i h[i+offset]*(1-h[i]) for i=0..n-1-offset
                # d/dh[j]: if j=i: -(h[j+offset]); if j=i+offset: (1-h[j-offset])
                for i in range(n - offset):
                    grad[i] -= h[i + offset]  # from (1-h[i]) term
                    grad[i + offset] += (1 - h[i])  # from h[i+offset] term
            else:
                # offset < 0, let o = -offset
                o = -offset
                # corr[k] = sum_i h[i]*(1-h[i+o]) for i=0..n-1-o
                for i in range(n - o):
                    grad[i] += (1 - h[i + o])
                    grad[i + o] -= h[i]
        # Normalize and apply
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm
            h_new = project(h - step * grad, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        step *= 0.995  # slow decay
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    d = target - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def project_onto feasible(h_raw, n):
    h = np.clip(h_raw, 0.0, 1.0)
    target = n / 2.0
    d = target - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def smax(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_phase(h, n, alpha, maxiter=2000):
    def obj(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        return smax(c, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-22})
        return repair(res.x, n)
    except Exception:
        return h

def subgrad_phase(h, n, n_iter, t_end):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    step = 0.015
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        active = np.where(corr > m - 0.005 * max(m, 1e-6))[0]
        grad = np.zeros(n)
        for m_idx in active:
            offset = m_idx - (n - 1)
            if offset >= 0:
                for i in range(n - offset):
                    grad[i] -= h[i + offset]
                    grad[i + offset] += (1.0 - h[i])
            else:
                o = -offset
                for i in range(n - o):
                    grad[i] += (1.0 - h[i + o])
                    grad[i + o] -= h[i]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = project_onto(h - step * grad / gnorm, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        step = max(step * 0.998, 0.0005)
    return best_h, best_c5

def coord_descent(h, n, t_end, rng, n_rounds=500):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(12 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni, nj = hid + sign * d, hjd - sign * d
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
                if improved:
                    break
        if not improved and r > 50:
            break
    return h, best

def pair_opt(h, n, t_end, n_rounds=100):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                if time.time() > t_end:
                    break
                hi, hj = h[i], h[j]
                h[i], h[j] = hj, hi
                c5 = compute_c5(h, n)
                if c5 < best - 1e-14:
                    best = c5
                    improved = True
                else:
                    h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best

def resample(h, n_from, n_to):
    x_old = np.linspace(0.0, 1.0, n_from)
    x_new = np.linspace(0.0, 1.0, n_to)
    h_new = np.interp(x_new, x_old, h)
    return repair(h_new, n_to)

    t0 = time.time()
    rng = np.random.RandomState(seed + 7)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()

    def elapsed():

    def tfrac(f):
        return t0 + budget_s * f

    n = best_n
    h_work = best_h.copy()

    # Phase 1: SLSQP with escalating alpha
    for alpha in [200000, 500000, 2000000, 5000000]:
        if elapsed() > tfrac(0.20):

        h_new = slsqp_phase(h_work.copy(), n, alpha, 1500)
        c5_new = compute_c5(h_new, n)

        elif c5_new < compute_c5(h_work, n) - 1e-14:

    # Phase 2: Subgradient projection

    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
