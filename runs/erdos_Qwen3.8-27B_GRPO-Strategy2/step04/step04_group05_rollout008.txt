            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    # Phase 1: DE exploration with multiple seeds
    _DE['n'], _DE['dx'] = nl, dx0
    bds = [(0.0, 1.0)] * nl
    de_budget = min(350, budget_s * 0.3)
    for s in [seed + 7, seed + 17, seed + 27, seed + 37]:
        if time.time() - t0 > de_budget:
            break
        try:
            result = differential_evolution(
                _de_obj, bds, seed=s, maxiter=100, popsize=22,
                tol=1e-9, mutation=(0.4, 1.0), crossover=0.8,
                polish=False, init='latinhypercube'
            )
            hc = repair(result.x, nl)
            c = c5v(hc, dx0)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass

    # Phase 2: Local refinement at n=140
    if time.time() - t0 < budget_s - 300:
        r, c = lbfgs_local(bh, nl, dx0, t0, t0 + budget_s * 0.45)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 250:
        r, c = active_refine(bh, nl, dx0, t0, t0 + budget_s * 0.6)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 200:
        r, c = spec_phase(bh, nl, dx0, t0, t0 + budget_s * 0.7)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 150:
        r, c = lbfgs_local(bh, nl, dx0, t0, t0 + budget_s * 0.8)
        if c < bc:
            bh, bc = r, c

    # Phase 3: Coordinate descent polish
    if time.time() - t0 < budget_s - 80:
        r, c = coord_refine(bh, nl, dx0, seed + 100, t0, t0 + budget_s * 0.85)
        if c < bc:
            bh, bc = r, c

    if time.time() - t0 < budget_s - 50:
        r, c = polish(bh, nl, dx0, seed + 200, 30000, t0, t0 + budget_s * 0.92)
        if c < bc:
            bh, bc = r, c

    # Phase 4: Try higher resolution n=280
    n2 = nl * 2
    if n2 < 1000 and time.time() - t0 < budget_s - 60:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, h0, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = lbfgs_local(h2, n2, d2, t0, t0 + budget_s * 0.95)
        if c < bc:
            bh, bc, bn = r, c, n2
            if time.time() - t0 < budget_s - 10:
                r, c = polish(bh, bn, d2, seed + 300, 15000, t0, t0 + budget_s * 0.99)
                if c < bc:
                    bh, bc = r, c

    # Final: multi-restart polish at current resolution
    if time.time() - t0 < budget_s - 5:
        rng2 = np.random.RandomState(seed + 500)
        hp = repair(bh + 0.01 * rng2.randn(bn), bn)
        r, c = coord_refine(hp, bn, 2.0 / bn, seed + 600, t0, t0 + budget_s * 0.99)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
