    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
        diff = target - h.sum()
    return h

def _sm(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
