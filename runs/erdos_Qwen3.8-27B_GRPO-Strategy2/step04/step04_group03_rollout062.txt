import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        # f_k = sum_i h[i] * (1 - h[i+lag]) for valid i
        # df_k/dh[j] = (1 - h[j+lag]) - h[j-lag] (with bounds)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * h[i2 + lag:j2 + lag]
    return g

def _project(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on tau: sum(clip(h - tau, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return _repair(proj, n)

def _pgd(h, n, alpha, t_end, max_iter=10000):
    """Projected gradient descent with backtracking"""
    c5c = _eval(h)
    step = 1.0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        # Backtracking line search
        for _ in range(30):
            h_new = _project(h - step * g, n)
            c_new = _smooth_max(_corr(h_new), alpha)
            c_cur = _smooth_max(_corr(h), alpha)
            if c_new <= c_cur - 1e-4 * step * np.linalg.norm(g)**2:
                break
            step *= 0.5
        h = h_new
        c5n = _eval(h)
        step = min(step * 1.05, 2.0)
    return h, _eval(h)

For the full correlation (not circular), at lag L (where L ranges from -(n-1) to n-1):
f_L = ∑_i h[i] * (1 - h[i+L]) where i and i+L are both in [0, n-1]

So for a given L:
- If L >= 0: i ranges from 0 to n-1-L, so f_L = ∑_{i=0}^{n-1-L} h[i](1-h[i+L])
- If L < 0: i ranges from |L| to n-1, so f_L = ∑_{i=|L|}^{n-1} h[i](1-h[i+L])

∂f_L/∂h[j]:
- If L >= 0:
  - j appears in h[j] factor: contributes (1-h[j+L]) for j in [0, n-1-L]
  - j appears in -h[j] factor (from -h[j-L] when i+L=j, i=j-L): contributes -h[j-L] for j in [L, n-1]
- If L < 0, let L' = -L > 0:

  - j appears in h[j] factor: contributes (1-h[j-L']) for j in [L', n-1]

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

Actually, one more thought: maybe I should try **subgradient projection** instead of smooth max. The subgradient of max_k f_k(h) at h is just the gradient of the active f_k (the one achieving the max). This is simpler and avoids the smooth max parameter altogether.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _sm(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sg(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14: continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1: g[i1:j1] += wk * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2: g[i2:j2] -= wk * h[i2 + lag:j2 + lag]
    return g

def _proj(h, n):
    t = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > t: lo = mid
        else: hi = mid
    return _repair(np.clip(h - (lo + hi) / 2, 0, 1), n)

def _pgd(h, n, alpha, t_end, max_it=50000, mom=0.85):
    v = np.zeros(n)
    sm = _sm(np.correlate(h, 1.0 - h, mode='full'), alpha)
    step = 0.01
    best_h, best_c5 = h.copy(), _eval(h)
    for _ in range(max_it):
        if time.time() > t_end: break
        g = _sg(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-13: break
        v = mom * v - step * g
        hn = _proj(h + v, n)
        cn = _sm(np.correlate(hn, 1.0 - hn, mode='full'), alpha)
        tr = 0
        while cn > sm + 1e-10 and tr < 12:
            step *= 0.6
            v = mom * v - step * g
            hn = _proj(h + v, n)
            cn = _sm(np.correlate(hn, 1.0 - hn, mode='full'), alpha)
            tr += 1
        h = hn; sm = cn
        step = min(step * 1.1, 0.15)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
    return best_h, best_c5

def _ls(h, n, rng, t_end):
    c5c = _eval(h); sc = 0.005; ni = 0
    for _ in range(500000):
        if time.time() > t_end: break
        i, j = rng.randint(n), rng.randint(n)

        d = sc * rng.randn()
        a, b = h[i] + d, h[j] - d
        if not (0 <= a <= 1 and 0 <= b <= 1): continue
        h[i], h[j] = a, b

        if c5n < c5c - 1e-15:
            c5c = c5n; ni = 0; sc = min(sc * 1.02, 0.012)
        else:
            h[i], h[j] = a - d, b + d
            ni += 1
            if ni > 400:
                sc *= 0.8; ni = 0
                if sc < 1e-8: break
    return h, c5c

def _interp(hs, ns, nd):
    return np.interp(np.linspace(0, 2, nd, endpoint=False),
                     np.linspace(0, 2, ns, endpoint=False), hs)

    nh = n // 2
    for sh in range(0, n, max(1, n // 8)):

def _cf(x, n): return float(np.sum(x) - n/2.0)
def _cj(x, n): return np.ones(n)

def _slsqp(h, n, alpha, t_end, maxiter=400):

                       method='SLSQP', bounds=[(0.,1.)]*n,
                       constraints=[{'type':'eq','fun':_cf,'jac':_cj,'args':(n,)}],
                       options={'maxiter':maxiter,'ftol':1e-20})
        r = _repair(res.x, n)
        return r if time.time() < t_end else r
    except: return _repair(h, n)

        hv[sh:sh + nh] = 1.0
        cands.append(_repair(hv, n))
    for f in [1, 2, 3, 4, 6, 8, 12]:
        for a in [0.3, 0.5, 0.7, 0.9]:
            hv = 0.5 + a * np.sin(2.0 * np.pi * f * np.arange(n) / n + rng.rand() * 6.28)
            cands.append(_repair(hv, n))
    for a in [0.05, 0.15, 0.3]:
        cands.append(_repair(_interp(ph, pn, n) + a * rng.randn(n), n))
    cands.append(_repair(rng.rand(n) * 1.2, n))
    return sorted([(_eval(c), c) for c in cands])

def _cf(x, n):
    return float(np.sum(x) - n / 2.0)

def _cj(x, n):
    return np.ones(n)

def _slsqp_obj(x, alpha):
    return _sm(np.correlate(x, 1.0 - x, mode='full'), alpha)

def _slsqp_jac(x, alpha):
    return _sg(x, alpha)

def _slsqp(h, n, alpha, t_end, maxiter=400):
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, args=(alpha,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    h_p = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    pn = len(h_p)
    bh, bc5, bn = h_p.copy(), _eval(h_p), pn

    n_list = [80, 120, 160, 200, 240, 320, 400]
    inits = {}
    for n in n_list:
        if _tl() < 15:
            break
        inits[n] = _inits(n, rng, h_p, pn)

    alphas = [200, 1000, 5000, 25000, 100000, 500000]
    for n in n_list:
        if n not in inits or _tl() < 20:
            continue
        for c5i, hi in inits[n][:3]:
            if _tl() < 12:
                break
            if c5i > bc5 + 0.04:
                continue
            hc, cc5 = hi.copy(), c5i
            for a in alphas:
                if _tl() < 8:
                    break
                dur = min(12.0, max(2.0, _tl() * 0.012))
                hc, cn = _pgd(hc, n, a, time.time() + dur)
                if cn < cc5 - 1e-12:
                    cc5, hc = cn, hc.copy()
            if cc5 < bc5:
                bc5, bh, bn = cc5, hc.copy(), n

    if _tl() > 30:
        h_s = bh.copy() if len(bh) == bn else _interp(bh, len(bh), bn)
        t_end_s = time.time() + min(25, _tl() - 15)
        for a_s in [200000, 500000]:
            if time.time() > t_end_s:
                break
            h_new = _slsqp(h_s, bn, a_s, t_end_s, maxiter=300)
            c5_new = _eval(h_new)
            if c5_new < bc5:
                bc5, bh = c5_new, h_new.copy()

    if _tl() > 20:
        h2, c2 = _ls(bh.copy(), bn, rng, time.time() + min(18, _tl() - 8))
        if c2 < bc5:
            bc5, bh = c2, h2.copy()

    if _tl() > 15:
        for a in [500000, 2000000]:
            if _tl() < 8:
                break
            h_new, cn = _pgd(bh.copy(), bn, a, time.time() + min(10, _tl() - 5))
            if cn < bc5 - 1e-12:
                bc5, bh = cn, h_new.copy()

    if _tl() > 10:
        h3, c3 = _ls(bh.copy(), bn, rng, time.time() + min(6, _tl() - 4))
        if c3 < bc5:
            bc5, bh = c3, h3.copy()

    bh = _repair(bh, bn)
    bc5 = _eval(bh)
    return (bh, bc5, bn)
