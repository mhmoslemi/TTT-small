def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _shift_grad(h, n, L):

    g = np.zeros(n)

    m = np.max(ca)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _shift_grad(h, n, L):
    g = np.zeros(n)
    if L == 0:

    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:

        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g
