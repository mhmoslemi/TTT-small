import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(_corr(h).max()) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14: break
            a = min((1 - h[i]) if d > 0 else h[i], abs(d))
            h[i] += a if d > 0 else -a
            d += a if d > 0 else -a
    return h

def _proj(h, n):
    t = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi2 = -5.0, 5.0
    for _ in range(100):
        m = (lo + hi2) * 0.5
        if np.sum(np.clip(h - m, 0.0, 1.0)) > t: lo = m
        else: hi2 = m
    hp = np.clip(h - (lo + hi2) * 0.5, 0.0, 1.0)
    d = t - hp.sum()
    if abs(d) > 1e-12:
        mk = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if mk.sum() > 0:
            hp[mk] += d / mk.sum()
            hp = np.clip(hp, 0.0, 1.0)
    d = t - hp.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - hp) if d > 0 else np.argsort(hp)
        for i in o:
            if abs(d) < 1e-12: break
            a = min((1 - hp[i]) if d > 0 else hp[i], abs(d))
            hp[i] += a if d > 0 else -a
            d += a if d > 0 else -a
    return hp

def _sobj(x, n, al):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(al * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(s))) / al

def _sgrad(x, n, al):
    dv = 2.0 / n
    c = _corr(x) * dv
    m = c.max()
    s = np.clip(al * (c - m), -350, 350)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14: continue
        lag = k - (n - 1)
        if lag == 0: g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l = -lag
            g[:n - l] += wk * (1.0 - x[l:])
            g[l:] -= wk * x[:n - l]
    return g * dv

_GN = 140
def _cf(x): return float(x.sum() - _GN / 2.0)
def _cj(x): return np.ones(_GN)

def _slsqp(h0, n, al, mx):
    global _GN
    _GN = n
    try:
        r = minimize(_sobj, h0, args=(n, al), jac=_sgrad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                     options={'maxiter': mx, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return h0

def _pgd(h, n, al, te, lr0):
    h = h.copy()
    lr = lr0
    cv = _c5(h, n)
    for _ in range(50000):
        if time.time() > te: break
        g = _sgrad(h, n, al)
        hn = _proj(h - lr * g, n)
        cn = _c5(hn, n)
        if cn < cv - 1e-18:
            h, cv = hn, cn
        else:
            lr *= 0.80
        if lr < 1e-13: break
    return h, _c5(h, n)

def _pair_search(h, n, te, rng):
    h = h.copy()
    cv = _c5(h, n)
    step_sets = [
        [0.02, -0.02, 0.006, -0.006, 0.002, -0.002],

        [0.001, -0.001, 0.0003, -0.0003, 0.0001, -0.0001, 0.0005, -0.0005],
        [0.0003, -0.0003, 0.0001, -0.0001, 0.00005, -0.00005],
        [0.0001, -0.0001, 0.00005, -0.00005, 0.00002, -0.00002],
    ]
    for ss in step_sets:
        no_imp = 0
        for rnd in range(200):
            if time.time() > te: break
            if no_imp >= 8: break
            imp = False
            idx = np.arange(n)
            rng.shuffle(idx)
            for ii in range(n):
                if time.time() > te: break
                i = idx[ii]
                hi = h[i]
                if hi < 3e-4 or hi > 1.0 - 3e-4: continue
                for jj in range(ii + 1, n):
                    if time.time() > te: break
                    j = idx[jj]
                    hj = h[j]
                    if hj < 3e-4 or hj > 1.0 - 3e-4: continue
                    for d in ss:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, n)
                        if c < cv - 1e-16:
                            cv = c
                            imp = True
                            hi, hj = ni, nj
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
            if not imp:
                no_imp += 1
            else:
                no_imp = 0
    return h, cv

def _triple_search(h, n, te, rng):
    """3-point moves: h[i]+=d, h[j]+=d2, h[k]-=(d+d2)"""
    h = h.copy()
    cv = _c5(h, n)
    no_imp = 0
    for rnd in range(500):
        if time.time() > te: break
        if no_imp >= 5: break
        imp = False
        i = rng.randint(n)
        j = rng.randint(n)
        k = rng.randint(n)
        if i == j or j == k or i == k:
            no_imp += 1; continue
        d1 = (rng.rand() - 0.5) * 0.02
        d2 = (rng.rand() - 0.5) * 0.02
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c = _c5(h, n)
            if c < cv - 1e-15:
                cv = c; imp = True
            else:
                h[i], h[j], h[k] = h[i]-d1, h[j]-d2, h[k]+d1+d2
        if not imp:
            no_imp += 1
        else:
            no_imp = 0
    return h, cv
