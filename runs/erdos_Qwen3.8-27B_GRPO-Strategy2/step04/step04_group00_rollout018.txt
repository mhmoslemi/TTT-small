import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lse_obj(h, alpha, lam):
    c = _corr(h); m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500); s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(h) - len(h)/2.0)**2

def _lse_grad(h, alpha, lam):
    c = _corr(h); m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500); s -= np.max(s)
    ew = np.exp(s); pw = ew / ew.sum()
    n = len(h); g = np.zeros(n)
    for k in range(2*n-1):
        L = k - (n-1); pk = pw[k]
        if pk < 1e-15: continue

def _subgrad(h, topm):

        else:
