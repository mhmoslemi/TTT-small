            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def spectral_refine(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    H0 = np.fft.rfft(bh)
    R0 = max(np.dot(bh, bh), 1e-10)
    tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
    tm[0] = abs(H0[0])
    for blend in [0.08, 0.15, 0.25, 0.4, 0.3, 0.2, 0.12, 0.06]:
        h = bh.copy()
        for _ in range(150):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            nm = (1 - blend) * mag + blend * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def active_set_opt(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [2, 3, 5, 8, 15]:
        h = bh.copy()
        step = 0.06
        stalled = 0
        for _ in range(400):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.004 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += grad1(h, ki, n)
            g /= len(idx)
            g -= g.mean()
            ok = False
            for _b in range(30):
                hn = repair(h - step * g, n)
                cn = c5v(hn, dx)
                if cn < c5v(h, dx) - 1e-18:
                    h = hn
                    ok = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.12, 1.5)
                    stalled = 0
                    break
                step *= 0.7
            if not ok:
                stalled += 1
                if stalled > 10:
                    break
    return bh, bc

def coord_desc(h, n, dx, rng, t0, lim, ni=60000):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    dt = np.array([0.004, 0.0015, 0.0006, -0.004, -0.0015, -0.0006, 0.012, -0.012])
    halts = 0
    for _ in range(ni):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = dt[rng.randint(len(dt))]
        ni2, nj2 = h[i] + d, h[j] - d
        if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
            h[i], h[j] = ni2, nj2
            c = c5v(h, dx)
            if c < cc - 1e-16:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
                halts = 0
            else:
                h[i], h[j] = ni2 - d, nj2 + d
                halts += 1
        if halts > 60:
            halts = 0
    return bh, bc

def de_global(n, dx, pop_size, max_iter, t0, lim, seed):
    def de_obj(x):
        return c5v(x, dx) + 5.0 * abs(x.sum() - n * 0.5)
    rng_d = np.random.RandomState(seed)
    pop = np.array([repair(rng_d.uniform(0, 1, n), n) for _ in range(pop_size)])
    # Add structured starts
    xg = np.linspace(0, 2, n, endpoint=False)
    for k in range(min(pop_size // 2, 4)):
        pop[k] = repair(np.clip(0.5 + 0.3 * np.sin((k + 1) * np.pi * xg), 0, 1), n)
    best_x = pop.copy()
    vals = np.array([de_obj(p) for p in pop])
    best_i = vals.argmin()
    best_v = vals[best_i]
    best_x = pop[best_i].copy()
    F, Cr = 0.75, 0.85
    for it in range(max_iter):
        if time.time() - t0 > lim:
            break
        for i in range(pop_size):
            a, b, c = rng_d.choice(pop_size, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            jrand = rng_d.randint(n)
            trial = pop[i].copy()
            mask = rng_d.rand(n) < Cr
            mask[jrand] = True
            trial[mask] = v[mask]
            trial = repair(trial, n)
            ft = de_obj(trial)
            if ft < vals[i]:
                pop[i] = trial
                vals[i] = ft
                if ft < best_v:
                    best_v = ft
                    best_x = trial.copy()
    return best_x, c5v(best_x, dx)

def make_structured(n, mode):
    x = np.linspace(0, 2, n, endpoint=False)
    if mode == 0:
        h = 0.5 + 0.3 * np.sin(np.pi * x)
    elif mode == 1:
        h = 0.5 + 0.3 * np.cos(np.pi * x)
    elif mode == 2:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x)
    elif mode == 3:
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
    elif mode == 4:
        h = 0.5 + 0.3 * np.sin(np.pi * x / 2)
    elif mode == 5:
        h = 0.5 + 0.2 * np.sin(3 * np.pi * x) + 0.2 * np.sin(np.pi * x)
    else:
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.15 * np.sin(3 * np.pi * x)
    return repair(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        n0 = len(initial_h_values)
        h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    except Exception:
        n0 = 140
        h0 = make_structured(n0, 0)
    dx0 = 2.0 / n0
    best_h, best_c, best_n = h0.copy(), c5v(h0, dx0), n0

    # Phase 1: Multi-start local opt (30%)
    t1 = t0 + 0.28 * budget_s
    starts = [h0.copy()]
    for m in range(6):
        starts.append(make_structured(n0, m))
    for m in range(4):
        starts.append(repair(h0 + 0.08 * rng.randn(n0), n0))
    for sh in starts:
        if time.time() > t1:
            break
        hh, cc = local_opt(sh, n0, dx0, t0, t1)
        if cc < best_c:
            best_h, best_c, best_n = hh, cc, n0

    # Phase 2: DE global at low res (22%)
    t2 = t0 + 0.50 * budget_s
    n_low = 45
    dx_low = 2.0 / n_low
    if time.time() < t2:
        try:
            de_h, de_c = de_global(n_low, dx_low, 18, 50, t0, t2, seed)
            xo = np.linspace(0, 2, n_low, endpoint=False)
            xn = np.linspace(0, 2, n0, endpoint=False)
            h_interp = repair(np.interp(xn, xo, de_h, period=2.0), n0)
            hh, cc = local_opt(h_interp, n0, dx0, t0, t2)
            if cc < best_c:
                best_h, best_c, best_n = hh, cc, n0
        except Exception:
            pass

    # Phase 3: Intensive local + active set (25%)
    t3 = t0 + 0.75 * budget_s
    for rnd in range(3):
        if time.time() > t3:
            break
        hh, cc = local_opt(best_h.copy(), best_n, 2.0 / best_n, t0, t3)
        if cc < best_c:
            best_h, best_c = hh, cc
        hh, cc = active_set_opt(best_h, best_n, 2.0 / best_n, t0, t3)
        if cc < best_c:
            best_h, best_c = hh, cc

    # Phase 4: Spectral + coord descent (20%)
    t4 = t0 + 0.95 * budget_s
    hh, cc = spectral_refine(best_h, best_n, 2.0 / best_n, t0, t4 - 25)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = local_opt(best_h.copy(), best_n, 2.0 / best_n, t0, t4 - 10)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = coord_desc(best_h, best_n, 2.0 / best_n, rng, t0, t4 - 3, ni=90000)
    if cc < best_c:
        best_h, best_c = hh, cc

    # Phase 5: Final polish + higher res (5%)
    tf = t0 + budget_s - 2
    hh, cc = spectral_refine(best_h, best_n, 2.0 / best_n, t0, tf - 5)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = coord_desc(best_h, best_n, 2.0 / best_n, rng, t0, tf, ni=30000)
    if cc < best_c:
        best_h, best_c = hh, cc

    # Try 2x resolution if time and size allow
    if time.time() < t0 + budget_s - 40 and best_n <= 200:
        n2 = min(best_n * 2, 280)
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, best_h, period=2.0), n2)
        d2 = 2.0 / n2
        hh, cc = local_opt(h2, n2, d2, t0, t0 + budget_s - 10)
        hh, cc = coord_desc(hh, n2, d2, rng, t0, t0 + budget_s - 2, ni=40000)
        if cc < best_c:
            best_h, best_c, best_n = hh, cc, n2

    best_h = repair(best_h, best_n)
    best_c = c5v(best_h, 2.0 / best_n)
    return (best_h, best_c, best_n)
