import numpy as np
import time

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _pgd(h, n, alpha, t_end, lr0, maxit):
    h = h.copy()
    lr = lr0
    c5c = _c5(h, n)
    for it in range(maxit):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, n, alpha)
        hn = _project(h - lr * g, n)
        c5n = _c5(hn, n)
        if c5n < c5c:
            h = hn
            c5c = c5n
        else:
            lr *= 0.7
        if it % 200 == 199:
            lr *= 0.85
    return h, c5c

def _line_search(h, n, i, j):
    """Golden section search for optimal transfer from j to i."""
    hi, hj = h[i], h[j]
    d_lo = max(-hi, hj - 1.0)
    d_hi = min(1.0 - hi, hj)
    if d_hi - d_lo < 1e-10:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = d_lo, d_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    h[i], h[j] = hi + c1, hj - c1
    f1 = _c5(h, n)
    h[i], h[j] = hi + c2, hj - c2
    f2 = _c5(h, n)
    for _ in range(14):
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            h[i], h[j] = hi + c1, hj - c1
            f1 = _c5(h, n)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            h[i], h[j] = hi + c2, hj - c2
            f2 = _c5(h, n)
    d = (a + b) * 0.5
    h[i], h[j] = hi + d, hj - d
    return d

def _guided_pairs(h, n, t_end, rng, n_rounds):
    """Iterative pair optimization guided by active correlations."""
    c5c = _c5(h, n)
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        cv = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(cv)
        active = np.where(cv >= mx * 0.92)[0]
        n_tries = min(80, n * 4)
        improved = 0
        for _ in range(n_tries):
            if time.time() > t_end:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[i], h[j]
            _line_search(h, n, i, j)
            c5n = _c5(h, n)
            if c5n < c5c - 1e-15:
                c5c = c5n
                improved += 1
            else:
                pass
        if improved == 0 and rnd > 3:
            # Try structured pairs for active lags
            for k_idx in active[:10]:
                lag = k_idx - (n - 1)
                if lag == 0:
                    continue
                for _ in range(10):
                    if time.time() > t_end:
                        break
                    i = rng.randint(n)
                    if lag > 0:
                        j2 = (i + lag) % n
                    else:
                        j2 = (i - lag) % n
                    if j2 == i:
                        continue
                    h[i], h[j2] = h[i], h[j2]
                    _line_search(h, n, i, j2)
                    c5n = _c5(h, n)
                    if c5n < c5c - 1e-15:
                        c5c = c5n
                        improved += 1
        if improved == 0 and rnd > 10:
            break
    return h, c5c

def _fine_tune(h, n, t_end, rng):
    c5c = _c5(h, n)
    deltas = np.array([1e-4, 3e-4, 1e-3, 3e-3, -1e-4, -3e-4, -1e-3, -3e-3])
    while time.time() < t_end:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5c:
                c5c = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c
