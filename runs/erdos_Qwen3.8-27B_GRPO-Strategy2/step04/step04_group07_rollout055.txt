import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            for idx in np.argsort(1.0 - h):
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
                if d < 1e-12:
                    break
        else:
            for idx in np.argsort(h):
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
                if -d < 1e-12:
                    break
    return h

def build_P(h, n):
    """Build the profile matrix P[i,s] = h[i+s] + h[i-s] for valid ranges.
    P has shape (n, 2n-1) where s index 0 corresponds to shift -(n-1)."""
    H_ext = np.zeros(3 * n - 1)
    H_ext[n-1:2*n-1] = h
    # P[i, k] where k = 0..2n-2 corresponds to shift s = k - (n-1)
    # h[i+s] = H_ext[i + k] (as derived earlier)
    # h[i-s] = H_ext[i + 2*(n-1) - k]
    # So P[i,k] = H_ext[i+k] + H_ext[i+2*(n-1)-k]
    idx_f = np.arange(n)[:, None] + np.arange(2*n-1)[None, :]  # (n, 2n-1)
    idx_b = np.arange(n)[:, None] + (2*(n-1) - np.arange(2*n-1))[None, :]  # (n, 2n-1)
    P = H_ext[idx_f] + H_ext[idx_b]
    return P

def grad_softmax(h, n, alpha):
    """Compute gradient of soft-max objective."""
    O = np.correlate(h, 1.0 - h, mode='full')
    m = O.max()
    w = np.exp(alpha * (O - m))
    w /= w.sum()
    P = build_P(h, n)
    grad = 1.0 - (P @ w)
    return grad, O

def projected_step(h, n, grad, lr):
    """Take a step and project onto constraints."""
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    # Project onto sum constraint
    s = h_new.sum()
    target = n / 2.0
    diff = s - target
    if abs(diff) > 1e-12:
        # Bisection for the right scaling
        if diff > 0:
            # Need to reduce sum: find threshold t such that sum(clip(h_new - t, 0, 1)) = target
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                proj = np.clip(h_new - mid, 0.0, 1.0)
                if proj.sum() > target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new - (lo+hi)/2, 0.0, 1.0)
        else:
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                proj = np.clip(h_new + mid, 0.0, 1.0)
                if proj.sum() < target:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(h_new + (lo+hi)/2, 0.0, 1.0)
    return h_new

def local_search(h, n, t_end):
    """Systematic local search with exact pair moves using profiles."""
    h = h.copy()
    dx = 2.0 / n
    two_nm1 = 2 * n - 1
    
    while time.time() < t_end:
        P = build_P(h, n)
        O = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = O.max() * dx
        improved = False
        
        for i in range(n):
            if time.time() > t_end:
                return h
            Pi = P[i]
            for j in range(n):
                if i == j:
                    continue
                g = P[j] - Pi  # = P[j] - P[i]
                
                dmax = min(1.0
