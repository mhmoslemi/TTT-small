    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    spread = cm - c.min()
    if spread < 1e-12: spread = 1.0
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= max(len(active), 1)
    return g

def _softobj(x, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14: continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

                        hpol[i], hpol[j] = ni - d, nj + d
                if bd != 0.0:
                    hpol[i] += bd; hpol[j] -= bd
                    hpol = _repair(hpol, npol)
                    cpol = _c5_val(hpol, dpol)
                    imp = True
                    if cpol < best_c5: best_c5, best_h = cpol, hpol.copy()
                    break
        if not imp:
            for _ in range(200):
                if time.time() > t0 + budget_s - 2: break
                i, j = rng.randint(npol), rng.randint(npol)
                if i == j: continue
                d = rng.uniform(-0.002, 0.002)
                ni, nj = hpol[i] + d, hpol[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    hpol[i], hpol[j] = ni, nj
                    c = _c5_val(hpol, dpol)
                    if c < cpol:
                        cpol = c
                        if cpol < best_c5: best_c5, best_h = cpol, hpol.copy()
                    else:
                        hpol[i], hpol[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_val(best_h, best_dx)
    return (best_h, best_c5, best_n)
