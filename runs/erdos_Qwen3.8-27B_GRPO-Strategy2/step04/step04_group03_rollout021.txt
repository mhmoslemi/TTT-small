To find the optimal t: we want to minimize max_k (c_k + (t-h0)*a_k) where a_k = [(1-h[i+lag_k])*I(valid) - h[i-lag_k]*I(valid)].

This is a 1D optimization: find t ∈ [0,1] minimizing the maximum of 2n-1 linear functions.

The optimal t is where two of the linear functions are equal and equal to the max, or at a boundary t=0 or t=1.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
            for ix in idx:
                if d > 0:
                    a = min(d, 1.0 - h[ix]); h[ix] += a; d -= a
                else:
                    a = min(-d, h[ix]); h[ix] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (x.sum() - n / 2.0) ** 2

def _grad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    g += 2.0 * lam * (x.sum() - n / 2.0)
    return g

def _lbfgs(h, n, alpha, lam, maxit):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha, lam, n),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-7, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _round01(h, n):
    target = n // 2
    h01 = np.zeros(n)
    h01[np.argsort(-h)[:target]] = 1.0
    return h01

def _local_swap(h, n, rng, t_end, max_iter=200000):
    c5c = _eval(h)
    c_vals = np.correlate(h, 1.0 - h, mode='full')
    scale = 0.004
    no_imp = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Random transfer
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n; no_imp = 0
            scale = min(scale * 1.02, 0.04)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 150:
                scale *= 0.75; no_imp = 0
                if scale < 5e-7:
                    break
    return h, c5c

def _coord_pass(h, n, rng, t_end, max_passes=30):
