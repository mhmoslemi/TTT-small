def repair(h, n):

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining
        if diff > 0:
            # Add to smallest values
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:

                sub = min(-diff, h[idx])

    corr = np.correlate(x, 1.0 - x, mode='full')

    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

    # Phase 6: L-BFGS ultra-high alpha (30s)
    dl6 = min(t0 + 830, t0 + budget_s - 40)
    if time.time() < dl6:
        for alpha in [50000, 150000]:
            if time.time() >= dl6:
                break
            hh, cc = lbfgs(best_h, n, alpha, target, 20000, 300)
            if cc < best_c5:
                best_c5, best_h = cc, hh.copy()

    # Phase 7: SLSQP final (20s)
    dl7 = min(t0 + 850, t0 + budget_s - 25)
    if time.time() < dl7:
        hh, cc = slsqp(best_h, n, 30000, target, 400)
        if cc < best_c5:
            best_c5, best_h = cc, hh.copy()

    # Phase 8: Fine local search (remaining)
    dl8 = t0 + budget_s - 5
    if time.time() < dl8:
        h2, c5_2 = fine_local(best_h, n, dx, rng, dl8)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
