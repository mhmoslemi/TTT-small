def _grad_ck(h, n, k):

    dx = 2.0 / n

    g = np.zeros(n)

        g[:n - lag] -= h[lag:] * dx
    else:
        l = -lag
        g[:n - l] += (1.0 - h[l:]) * dx
        g[l:] -= h[:n - l] * dx
    return g

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
