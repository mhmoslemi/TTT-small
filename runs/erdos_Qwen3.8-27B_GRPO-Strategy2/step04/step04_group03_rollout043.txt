import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf_focused(x, alpha, ktop):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    idx = np.argpartition(c, -ktop)[-ktop:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs_focused(x, alpha, ktop):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    idx = np.argpartition(c, -ktop)[-ktop:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _opt(h, n, alpha, ktop, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf_focused, h, jac=_gs_focused, args=(alpha, ktop),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _local_search(h, n, rng, t_end):
    c5c = _eval(h)
    scale = 0.004
    no_imp = 0
    for _ in range(400000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.02, 0.012)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 400:
                scale *= 0.82
                no_imp = 0
                if scale < 2e-6:
                    break
    return h, c5c

def _lag_targeted(h, n, rng, t_end):
    c5c = _eval(h)
    for _ in range(100000):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        km = np.argmax(c)
        lag = km - (n - 1)
        # Find pair that reduces this lag most
        if lag >= 0:
            i1, j1 = 0, n - lag
        else:
            i1, j1 = -lag, n
        best_improvement = 0
        best_i, best_j, best_dir = -1, -1, 0
        for _trial in range(50):
            ia = rng.randint(i1, j1)
            ja = rng.randint(n)
            if ja == ia:
                continue
            # Try increasing h[ia], decreasing h[ja]
            d = 0.02
            if 0 <= h[ia] + d <= 1 and 0 <= h[ja] - d <= 1:
                h[ia], h[ja] = h[ia] + d, h[ja] - d
                cn = _eval(h)
                h[ia], h[ja] = h[ia] - d, h[ja] + d
                if cn < c5c - best_improvement:
                    best_improvement = c5c - cn
                    best_i, best_j, best_dir = ia, ja, +d
            d = -0.02
            if 0 <= h[ia] + d <= 1 and 0 <= h[ja] - d <= 1:
                h[ia], h[ja] = h[ia] + d, h[ja] - d
                cn = _eval(h)
                h[ia], h[ja] = h[ia] - d, h[ja] + d
                if cn < c5c - best_improvement:
                    best_improvement = c5c - cn
                    best_i, best_j, best_dir = ia, ja, d
        if best_i >= 0:
            h[best_i] += best_dir
            h[best_j] -= best_dir
            h = _repair(h, n)
            c5c = _eval(h)
    return h, c5c

def _multi_start(h_best, n, rng, t_end):
    best_h, best_c5 = h_best.copy(), _eval(h_best)
    for _ in range(30):
        if time.time() > t_end:
            break
        freq = rng.randint(1, 12)
        amp = rng.uniform(0.08, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        h_try = h_best + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h_try = _repair(h_try, n)
        c5_try = _eval(h_try)
        if c5_try < best_c5 + 0.005:
            h_new = _opt(h_try, n, 100000, 10, maxiter=200)
            c5_new = _eval(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    return best_h, best_c5

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))

        h0 = h_init.copy() if n == n_init else _repair(_interp(h_init, n_init, n), n)

            cv = _eval(h_try)

        # SLSQP with increasing sharpness and decreasing ktop
