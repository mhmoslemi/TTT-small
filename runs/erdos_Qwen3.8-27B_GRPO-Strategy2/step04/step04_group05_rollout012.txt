def proj_subgrad(h, n, t0, tmax, niter, top_m=5):
    bh, bc = h.copy(), c5v(h)
    alpha0 = 0.15
    for t in range(niter):
        if time.time() - t0 > tmax:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = c.max()
