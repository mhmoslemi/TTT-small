    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_full(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

            best_obj = best_c

            best_c = c.max()

                    continue

                    best_v = v

                old = best_v

        h = _repair(h, target)
        c = _corr_dx(h, dx)
        best_c = c.max()
    return h, _c5(h, dx)

def _subgradient(h, dx, t_end, n_steps=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c = _c5(h, dx)
    best_h = h.copy()
    mu = 2.0  # initial step size
    for step in range(n_steps):
        if time.time() > t_end:
            break
        c = _corr_dx(h, dx)
        m = c.max()
        # Active set: top 10% of correlations
        thresh = np.percentile(c, 90)
        active = np.where(c >= thresh)[0]
        # Compute subgradient: average gradient of active lags
        grad = np.zeros(n)
        for m_idx in active:
            s = m_idx - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * h) * dx
            elif s > 0:
                if s < n:
                    grad[s:] += (1.0 - h[:-s]) * dx
                    grad[:n-s] -= h[s:] * dx
            else:
                sp = -s
                if sp < n:
                    grad[:n-sp] += (1.0 - h[sp:]) * dx
                    grad[sp:] -= h[:n-sp] * dx
        grad /= len(active)
        # Polyak step: mu = (m - best_c + epsilon) / ||grad||²
        gnorm2 = np.dot(grad, grad)
        if gnorm2 > 1e-20:
            mu = (m - best_c + 1e-4) / gnorm2
        h_new = h - mu * grad
        h_new = _repair(h_new, target)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_dx(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _coord_descent(h, dx, t_end, max_cycles=500):

    h = h.copy()
    target = n / 2.0
    c = _corr_dx(h, dx)
    best_c = c.max()
    grid = np.linspace(0, 1, 40)
    for cy in range(max_cycles):
        if time.time() > t_end:
            break
        improved_any = False
        for j in range(n):
            old = h[j]
            g = np.zeros(2 * n - 1)
            s_vals = np.arange(1, n)
            idx_pos = s_vals + n - 1
            m1 = s_vals <= j
            if m1.any():
                g[idx_pos[m1]] += (1.0 - h[j - s_vals[m1]])
            m2 = (j + s_vals) < n
            if m2.any():
                g[idx_pos[m2]] -= h[j + s_vals[m2]]
            sp_vals = np.arange(1, n)
            idx_neg = n - 1 - sp_vals
            m3 = (j + sp_vals) < n
            if m3.any():
