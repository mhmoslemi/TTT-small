    """Optimize h[i] and h[j] (with h[i]+h[j]=const) to max min autocorrelation."""
    s = n / 2.0 - (h.sum() - h[i] - h[j])
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        return h
    
    # Compute A_old(k) for all k
    # A(k) = sum_x h[x] * h[(x+k) mod n]
    # This is the circular autocorrelation
    h_ext = np.concatenate([h, h])
    A_old = np.array([np.dot(h, h_ext[k:k+n]) for k in range(n)])

    # Compute α_k and β_k
    idx_i_plus = (np.arange(n) + i) % n
    idx_i_minus = (np.arange(n) - i) % n
    idx_j_plus = (np.arange(n) + j) % n
    idx_j_minus = (np.arange(n) - j) % n

    A_new = p[:, None] * t_grid[None, :] + q[:, None]
    min_over_k = np.min(A_new, axis=0)  # shape (n_grid,)
    best_idx = np.argmax(min_over_k)
    best_t = t_grid[best_idx]
    
    h[i] = best_t
    h[j] = s - best_t
