import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-15:
                break
    return h

def _grad_obj(h, dx, alpha):
    n = len(h)
    c = _corr(h)
    ac = alpha * c * dx
    ac -= ac.max()
    w = np.exp(ac)
    w /= w.sum()
    grad = np.zeros(n)
    half = n - 1
    for ki in range(len(c)):
        wk = w[ki]
        if wk < 1e-14:
            continue
        k = ki - half
        if k >= 0:
            if k < n:
                grad[:n-k] += wk * (1.0 - h[k:])
                grad[k:] -= wk * h[:n-k]
        else:
            m = -k
            if m < n:
                grad[m:] += wk * (1.0 - h[:n-m])
                grad[:n-m] -= wk * h[m:]
    return grad * dx

def _pgd(h, dx, alpha, lr, n_iter, t_end):
    """Projected gradient descent"""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _grad_obj(h, dx, alpha)
        h_new = _project(h - lr * g, target)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
        h = h_new
    return best_h, best_c
