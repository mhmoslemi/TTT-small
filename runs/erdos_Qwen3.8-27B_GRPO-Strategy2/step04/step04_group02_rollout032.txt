import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b
                diff += b
            if abs(diff) < 1e-12:
                break
    return h

def grad_corr_at_shift(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        i = np.arange(n - shift)
        grad[i] += (1.0 - h[i + shift])
        grad[i + shift] += -h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        grad[i] += (1.0 - h[i - s])
        grad[i - s] += -h[i]
    return grad

def top_m_gradient(h, n, m):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-m:]
    grad = np.zeros(n)
    for k in top_idx:
        shift = int(k) - (n - 1)
        grad += grad_corr_at_shift(h, n, shift)
    grad /= m
    return grad, corr

def momentum_descent(h, n, dx, deadline, top_m=3, lr_init=0.5, max_iter=30000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    momentum = np.zeros(n)
    beta = 0.82
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        grad, corr = top_m_gradient(h, n, top_m)
        c5_old = float(np.max(corr) * dx)
        grad -= np.mean(grad)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = repair(h - lr * momentum, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if c5_new <= c5_old + 1e-15:
            h = h_new
            lr = min(lr * 1.03, 3.0)
        else:
            lr *= 0.55
            momentum *= 0.4
            if lr < 1e-7:
                lr = lr_init * 0.005
                momentum = np.zeros(n)
    return best_h, best_c5

def lbfgs_lse(h, n, alpha, target, maxiter=200):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha + 2000.0 * (np.sum(x) - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-7})
        return repair(res.x, n)
    except Exception:
        return h

def directed_search(h, n, dx, rng, deadline, n_iters=40000, n_active=15):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    offset = n - 1
    corr = compute_corr(h)
    active = np.argsort(corr)[-n_active:]
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        if it % 3000 == 0:
            corr = compute_corr(h)
            active = np.argsort(corr)[-n_active:]
        k = active[rng.randint(n_active)]
        shift = int(k) - offset
        if shift >= 0:
            i = rng.randint(0, n - shift)
        else:
            i = rng.randint(-shift, n)
        j = i + shift
        if j < 0 or j >= n:
            continue
        d = rng.uniform(0.002, 0.018)
        if h[i] >= 0.5:
            di, dj = -d, d
        else:
            di, dj = d, -d
        if h[i] + di < -1e-9 or h[i] + di > 1 + 1e-9:
            continue
        if h[j] + dj < -1e-9 or h[j] + dj > 1 + 1e-9:
            continue
        h[i] += di
        h[j] += dj
        c5n = compute_c5(h, dx)
        if c5n < best_c5:
            best_c5 = c5n
        else:
            h[i] -= di
            h[j] -= dj
    return repair(h, n), best_c5

def pair_search(h, n, dx, rng, deadline, n_iters=30000, scale=0.004):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
            continue
        h[i], h[j] = ni, nj
        c5n = compute_c5(h, dx)
        if c5n < best_c5:
            best_c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return repair(h, n), best_c5

def block_search(h, n, dx, rng, deadline, n_iters=8000):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.003, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        seg_i = h[i:i+bs]
        seg_j = h[j:j+bs]
        if (seg_i + eps).min() < -1e-9 or (seg_i + eps).max() > 1 + 1e-9:
            continue
        if (seg_j - eps).min() < -1e-9 or (seg_j - eps).max() > 1 + 1e-9:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = compute_c5(h, dx)
        if c5n < best_c5:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return repair(h, n), best_c5

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(3, n // 3))
    elif kind == 1:
        f = rng.randint(1, 8)
        ph = rng.uniform(0, 6.28)
        hp += rng.uniform(0.02, 0.08) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.025, n)
    elif kind == 3:
        bsz = rng.randint(3, 10)
        st = rng.randint(0, n - bsz)
        hp[st:st+bsz] = 1.0 - hp[st:st+bsz]
    else:
        hp *= rng.uniform(0.82, 1.18)
    return repair(hp, n)
