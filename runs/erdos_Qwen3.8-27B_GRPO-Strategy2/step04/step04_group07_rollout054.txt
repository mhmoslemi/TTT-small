So output[k] = sum_i a[i] * v[i + (k - (n-1))]

So for lag ≥ 0: i ranges from 0 to n-1-lag, and we sum h[i]*(1-h[i+lag])
For lag < 0: i ranges from -lag to n-1, and we sum h[i]*(1-h[i+lag]) = h[i]*(1-h[i-|lag|])

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def project_box_sum(v, target):
    """Project v onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    # Bisection on lambda: sum clip(v[j] - lam, 0, 1) = target
    lo, hi = -10.0, 10.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_direction(h, n, corr, K=5):
    """Compute averaged subgradient direction for top-K worst lags."""
    # corr has length 2n-1
    # C5 = dx * max(corr), so worst lags are top K entries of corr
    n_corr = len(corr)
    top_idx = np.argsort(corr)[-K:]
    
    n_half = n - 1
    grad = np.zeros(n)
    
    for m in top_idx:
        lag = m - n_half
        # d/dh[j] of sum_i h[i]*(1-h[i+lag])
        # = 1 - h[j+lag]*I[valid] - h[j-lag]*I[valid]
        g = np.ones(n)
        if lag > 0:
            # h[j+lag] valid when j < n-lag
            g[:n-lag] -= h[lag:]
            # h[j-lag] valid when j >= lag
            g[lag:] -= h[:n-lag]
        elif lag < 0:
            # h[j+lag] = h[j-|lag|] valid when j >= |lag|
            abs_lag = -lag
            g[abs_lag:] -= h[:n-abs_lag]
            # h[j-lag] = h[j+|lag|] valid when j < n-|lag|
            g[:n-abs_lag] -= h[abs_lag:]
        # lag == 0: g[j] = 1 - 2*h[j]
        else:
            g = 1.0 - 2.0 * h
        grad += g
    
    # Subtract mean to maintain sum constraint
    grad -= grad.mean()
    return grad

            if not improved:

    final_c5 = compute_c5(h, n)
    if final_c5 < best_c5:

    return best_h, best_c5

def coord_polish(h, n, t_end, rng):
    """Fine coordinate descent polish."""
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.001, 0.0005]
    for _ in range(20000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        hi, hj = h[i], h[j]
        improved = False
        for d in steps:
            for sign in (1, -1):
                delta = sign * d
                ni, nj = hi + delta, hj - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    best = c5
                    hi, hj = ni, nj
                    improved = True
                    break
                h[i], h[j] = hi, hj
            if improved:
                break
    return h, best

        g[0:n-lag] += (1.0 - h[lag:n])
        g[lag:n] -= h[0:n-lag]
    else:
        L = -lag
        # h[i]*(1-h[i-L]) for i=L..n-1
        g[L:n] += (1.0 - h[0:n-L])
        g[0:n-L] -= h[L:n]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def lag_gradient(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        g[0:n-lag] += (1.0 - h[lag:n])
        g[lag:n] -= h[0:n-lag]
    else:
        L = -lag
        g[L:n] += (1.0 - h[0:n-L])
        g[0:n-L] -= h[L:n]
    return g

def subgrad_optimize(h, n, max_iters=5000, K=3, t_end=None):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    n_half = n - 1
    for it in range(max_iters):
        if t_end is not None and time.time() > t_end:
            break
        corr = compute_all_corr(h, n)
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 200 == 0:
            top_idx = np.argsort(corr)[-K:]
            grad = np.zeros(n)
            for m in top_idx:
                lag = m - n_half
                grad += lag_gradient(h, n, lag)
            grad -= grad.mean()
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-12:
                break
            step_lr = 0.1 / g_norm
            improved = False
            for ls in range(25):
                h_trial = np.clip(h - step_lr * grad, 0.0, 1.0)
                h_trial = project_box_sum(h_trial, n / 2.0)
                c5_trial = compute_c5(h_trial, n)
                if c5_trial < c5 - 1e-14:
                    h = h_trial
                    improved = True
                    break
                step_lr *= 0.5
                if step_lr < 1e-12:
                    break
            if not improved:
                for ts in [0.005, 0.001, 0.0003]:
                    h_trial = np.clip(h - ts * grad / g_norm, 0.0, 1.0)
                    h_trial = project_box_sum(h_trial, n / 2.0)
                    c5_trial = compute_c5(h_trial, n)
                    if c5_trial < c5 - 1e-15:
                        h = h_trial
                        improved = True
                        break
                if not improved:
                    break
    final_c5 = compute_c5(h, n)
    if final_c5 < best_c5:
        best_h = h.copy()
        best_c5 = final_c5
    return best_h, best_c5

def smooth_obj(v, alpha, n):
    corr = compute_all_corr(v, n)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_optimize(h, n, alpha, maxiter=2000):
    def obj(v):
        return smooth_obj(v, alpha, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def coord_polish(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.0005]
    stalling = 0
    for _ in range(50000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        hi, hj = h[i], h[j]
        found = False
        for d in steps:
            for sign in (1, -1):
                delta = sign * d
                ni, nj = hi + delta, hj - delta
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, n)
                if c5 < best - 1e-15:
                    best = c5
                    hi, hj = ni, nj
                    found = True
                    break
                h[i], h[j] = hi, hj
            if found:
                break
        if found:
            stalling = 0
        else:
            stalling += 1
        if stalling > 2000:
            break
    return h, best

def make_init_ramp(n):
    x = np.linspace(0, 1, n)
    return repair(1.0 - x, n)

def make_init_sine(n, freq=2, phase=0.0):
    x = np.linspace(0, 2, n)
    return repair(0.5 + 0.35 * np.sin(np.pi * freq * x + phase), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = np.inf
    best_n = 64

    init_h = None
    init_n = 64
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n

    # Phase 1: Subgradient on initial solution
    if best_h is not None and elapsed() < budget_s * 0.25:
        t1 = t0 + budget_s * 0.25
        h_work = best_h.copy()
        for K in [1, 3, 5]:
            if elapsed() > t1:
                break
            h_new, c5_new = subgrad_optimize(h_work, best_n, 4000, K, t1)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new

    # Phase 2: Smooth SLSQP polish
    if elapsed() < budget_s * 0.40:
        t2 = t0 + budget_s * 0.40
        for alpha in [100000, 500000, 2000000]:
            if elapsed() > t2:
                break
            h_new = smooth_optimize(best_h.copy(), best_n, alpha, 800)
            c5_new = compute_c5(h_new, best_n)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new.copy()
                best_c5 = c5_new

    # Phase 3: Coordinate polish
    if elapsed() < budget_s * 0.55:
        h_pol, c5_pol = coord_polish(best_h, best_n, t0 + budget_s * 0.55, rng)
        if c5_pol < best_c5 - 1e-15:
            best_h = h_pol.copy()
            best_c5 = c5_pol

    # Phase 4: Subgradient again with refined best
    if elapsed() < budget_s * 0.70:
        h_new, c5_new = subgrad_optimize(best_h, best_n, 5000, 3, t0 + budget_s * 0.70)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 5: Try different n
    if elapsed() < budget_s * 0.85:
        t5 = t0 + budget_s * 0.85
        for n_alt in [48, 56, 72, 80, 96, 120]:
            if elapsed() > t5:
                break
            if n_alt == best_n:
                continue
            x_old = np.linspace(0, 1, best_n)
            x_new = np.linspace(0, 1, n_alt)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_try, c5_try = subgrad_optimize(h_seed, n_alt, 2000, 5,
                                             min(t0 + budget_s * 0.10, t5))
            h_try2, c5_try2 = coord_polish(h_try, n_alt, t5, rng)
            if c5_try2 < c5_try:
                h_try, c5_try = h_try2, c5_try2
            if c5_try < best_c5 - 1e-15:
                best_h = h_try.copy()
                best_c5 = c5_try
                best_n = n_alt

    # Phase 6: Multi-start with structured inits
    if elapsed() < budget_s * 0.93:
        t6 = t0 + budget_s * 0.93
        inits = [make_init_ramp(best_n), make_init_sine(best_n, 2, 0),
                 make_init_sine(best_n, 3, np.pi/3),
                 repair(best_h + 0.03 * rng.randn(best_n), best_n)]
        for h_init in inits:
            if elapsed() > t6:
                break
            h_try, c5_try = subgrad_optimize(h_init, best_n, 1500, 5,
                                             min(t0 + budget_s * 0.04, t6))
            if c5_try < best_c5 - 1e-15:
                best_h = h_try.copy()
                best_c5 = c5_try

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.99:
        t7 = t0 + budget_s * 0.99
        h_new, c5_new = subgrad_optimize(best_h, best_n, 4000, 3, t7)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
        h_pol, c5_pol = coord_polish(best_h, best_n, t7, rng)
        if c5_pol < best_c5 - 1e-15:
            best_h = h_pol.copy()
            best_c5 = c5_pol

    if best_h is None:
        best_n = 64
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
