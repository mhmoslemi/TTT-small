Let me think about the active set optimization more carefully:

The objective is C₅ = max_k c[k] where c = correlate(h, 1-h, 'full').
We want to minimize this subject to 0≤h≤1 and sum(h)=n/2.

The subgradient of max_k c[k] is the uniform average of gradients of all k achieving the max.
If only one k achieves the max, the gradient is just grad_overlap(h, k, n).
If multiple k achieve the max (or are near the max), we average their gradients.

The active set approach:
1. Identify the "active" constraints (k values where c[k] is near the max)
2. Compute the average gradient of those active constraints
3. Take a step in the negative gradient direction
4. Project back to the feasible set (repair)

This is essentially a subgradient method with projection. It can be made more effective by:
- Using a Polyak step size or backtracking
- Including more active constraints (not just the top few)

import time

    s = h.sum()

    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:

        else:
            o = np.argsort(h)

                break

                a = min(d, 1.0 - h[i])
            else:

            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_ov(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def soft_max_obj(x, n, a, lam):
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)
    s = np.clip(a*(c-m), -400, 400)
    ew = np.exp(s)
    return m + np.log(ew.sum())/a + lam*(x.sum()-n*0.5)**2

def soft_max_jac(x, n, a, lam):
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)
    s = np.clip(a*(c-m), -400, 400)
    ew = np.exp(s)
    pw = ew/ew.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if pw[k] > 1e-14:
            g += pw[k]*grad_ov(x, k, n)
    g /= a
    g += 2.0*lam*(x.sum()-n*0.5)
    return g

def lbfgs_opt(h, n, dx, a, lam, t0, tl, seed=0):
    bds = [(0.0, 1.0)]*n
    bh, bc = h.copy(), c5v(h, dx)
    try:
        r = minimize(soft_max_obj, bh.copy(), jac=lambda x: soft_max_jac(x, n, a, lam),
                     method='L-BFGS-B', bounds=bds,
                     options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-17})
        hc = repair(r.x, n)
        c = c5v(hc, dx)
        if c < bc:
            bh, bc = hc, c
    except Exception:

    return bh, bc

def binary_local(h, n, dx, rng, cnt, t0, tl):
    bh, bc = h.copy(), c5v(h, dx)
    # Snap to binary
    hb = (h >= 0.5).astype(float)
    # Ensure sum = n//2
    s = int(hb.sum())
    if s > n//2:
        # Remove excess ones

        order = np.argsort(hb[ones])  # ones closest to 0.5 first
        for i in order[:s-n//2]:
            hb[ones[i]] = 0
    elif s < n//2:
        zeros = np.where(hb==0)[0]
        order = np.argsort(-hb[zeros])  # zeros closest to 0.5 first
        for i in order[:n//2-s]:
            hb[zeros[i]] = 1
    bc_bin = c5v(hb, dx)
    cc = min(bc, bc_bin)
    best = hb if bc_bin < bc else bh.copy()
    
    for _ in range(cnt):
        if time.time() - t0 > tl:

        # Try swapping a 1 with a 0
        ones = np.where(best==1)[0]

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        best[i], best[j] = 0, 1
        c = c5v(best, dx)
        if c < cc:
            cc = c
            if cc < bc:

        else:

        rng_f = np.random.RandomState(seed + 5555)
        r, c = binary_search(bh, bn, 2.0 / bn, rng_f, 50000, t0, budget_s - 3)
        if c < bc:
            bh, bc = r, c
    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
