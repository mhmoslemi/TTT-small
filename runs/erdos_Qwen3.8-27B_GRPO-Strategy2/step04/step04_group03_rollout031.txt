    lo, hi = -1.0, 2.0

            lo = mid
        else:

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    return best, best_c5

def _localsearch(h, n, rng, t_end):

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _proj(h, S):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0, 1)
    S = n / 2.0
    if abs(h.sum() - S) < 1e-10:
        return h
    return _proj(h, S)

def _sf(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        s1, e1 = max(0, lag), min(n, n + lag)
        if s1 < e1:
            g[s1:e1] += wk * (1.0 - h[s1 - lag:e1 - lag])
        s2, e2 = max(0, -lag), min(n, n - lag)
        if s2 < e2:
            g[s2:e2] -= wk * h[s2 + lag:e2 + lag]
    return g

def _subgrad_top(h, n, m=4):
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[::-1][:m]
    g = np.zeros(n)
    for k in top:
        lag = int(k) - (n - 1)
        s1, e1 = max(0, lag), min(n, n + lag)
        if s1 < e1:
            g[s1:e1] += 1.0 - h[s1 - lag:e1 - lag]
        s2, e2 = max(0, -lag), min(n, n - lag)
        if s2 < e2:
            g[s2:e2] -= h[s2 + lag:e2 + lag]
    return g / m

def _pfun(x, n):
    return np.sum(x) - n / 2.0

def _pjac(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _pfun, 'jac': _pjac, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h0, n, t_end, m=4, lr_decay=0.02):
    S = n / 2.0
    x = h0.copy()
    x_prev = h0.copy()
    best = h0.copy()
    best_c5 = _eval(h0)
    tau = 1.0
    for it in range(100000):
        if time.time() > t_end:
            break
        y = x + (tau - 1) / (tau + 2) * (x - x_prev)
        yc = np.clip(y, 0, 1)
        g = _subgrad_top(yc, n, m)
        lr = 0.08 / (1 + lr_decay * it) ** 0.5
        x_new = _proj(y - lr * g, S)
        x_prev = x.copy()
        x = x_new
        tau = (1 + np.sqrt(1 + 4 * tau * tau)) / 2.0
        if it % 400 == 0:
            c5 = _eval(x)
            if c5 < best_c5:
                best_c5 = c5
                best = x.copy()
    return best, best_c5

def _localsearch(h, n, rng, t_end):
    c5c = _eval(h)
    scale = 0.004
    noimp = 0
    for _ in range(500000):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)

        h[i], h[j] = hi, hj

            noimp += 1
            if noimp > 150:
                scale *= 0.75
                noimp = 0
                if scale < 1e-7:
                    break
    return h, c5c

def _interp(h, n1, n2):
    return np.interp(np.linspace(0, 2, n2, endpoint=False),
                     np.linspace(0, 2, n1, endpoint=False), h)
