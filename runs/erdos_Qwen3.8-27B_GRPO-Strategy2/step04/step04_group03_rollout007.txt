def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_max(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)

def _grad_exact_max(h, num_top=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-num_top:]
    g = np.zeros(n)
    for k in top_idx:
        lag = k - (n-1)
        i1, j1 = max(0, lag), min(n, n+lag)
        if i1 < j1:
            g[i1:j1] += (1.0 - h[i1-lag:j1-lag])
        i2, j2 = max(0, -lag), min(n, n-lag)
        if i2 < j2:
            g[i2:j2] -= h[i2+lag:j2+lag]
    return g / num_top

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _acf(x, n):
    return np.sum(x) - n / 2.0

def _acj(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _acf, 'jac': _acj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj(h, n):
    target = n / 2.0
    c = h.copy()
    lo, hi = c.min() - 2.0, c.max() + 2.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        if np.clip(c - tau, 0.0, 1.0).sum() > target:
            lo = tau
        else:
            hi = tau
    hp = np.clip(c - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - hp[idx]); hp[idx] += a; diff -= a
            else:
                a = min(-diff, hp[idx]); hp[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return hp

def _pgd(h, n, t_end):

    return h, bv
