import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _corr_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)

        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _active_subgrad(h, n, top_k=3):
    """Subgradient using top-k active lags (weighted)"""
    c = _corr_vals(h, n)
    # Get top-k values
    top_idx = np.argsort(c)[-top_k:]
    top_vals = c[top_idx]
    # Weight by softmax over top values
    t = np.exp(100 * (top_vals - top_vals.max()))

def _nesterov_step(h, n, alpha, lr, momentum, budget_end):
    """Nesterov accelerated projected gradient"""
    h = h.copy()
    v = np.zeros(n)
    best_c5 = _c5(h, n)
    it = 0
    while time.time() < budget_end:
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        g = _smooth_grad(y, n, alpha)
        new_h = h - lr * g
        new_h = _project(new_h, n)
        new_v = new_h - h
        h = new_h
        v = momentum * v + (1 - momentum) * new_v
        c5 = _c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
        # Adaptive lr
        it += 1
        if it % 500 == 0:
            lr *= 0.95
    return h, best_c5

def _targeted_pair_search(h, n, budget_end, rng, top_lags=5):

    h = h.copy()
    c5_cur = _c5(h, n)
    deltas = np.array([0.001, 0.003, 0.01, 0.03,
                       -0.001, -0.003, -0.01, -0.03])
    max_pass = 200
    for p in range(max_pass):
        if time.time() > budget_end:
            break
        # Identify active lags
        c = _corr_vals(h, n)
        active_idx = np.argsort(c)[-top_lags:]
        active_lags = active_idx - (n - 1)
        # For each active lag, try targeted moves
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for lag in active_lags:
            if time.time() > budget_end:
                break
            alag = abs(lag)
            if alag == 0:
                continue
            # Pairs (i, i+alag) for this lag
            max_i = n - alag

                i = rng.randint(max_i)
                j = i + alag

                for d in deltas:

                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                        improved = True
                        hi, hj = ni, nj
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved and p > 20:

    return h, c5_cur

import numpy as np

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _corr_vals(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        m = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if m.sum() > 0:
            hp[m] += diff / m.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = _corr_vals(x, n)
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -350, 350)))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr_vals(x, n)
    m = c.max()
    e = np.exp(np.clip(alpha * (c - m), -350, 350))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _nesterov_pgd(h, n, alpha, budget_end, lr_init, momentum):
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    best = _c5(h, n)
    while time.time() < budget_end:
        y = np.clip(h + momentum * v, 0.0, 1.0)
        g = _smooth_grad(y, n, alpha)
        h_new = _project(h - lr * g, n)
        v = h_new - h
        c5 = _c5(h_new, n)
        if c5 < best:
            best = c5
            h = h_new
        else:
            lr *= 0.85
            v *= 0.5
    return h, best

def _try_swap(h, i, j, deltas, rng, c5_cur):

    for d in deltas:
