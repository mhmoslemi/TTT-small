import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target)**2

def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_con(x, target):
    return np.sum(x) - target

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h.copy(), compute_c5(h, dx)

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(_slsqp_obj, h, args=(alpha,), method='SLSQP',
                      bounds=[(0.0, 1.0)]*n,
                      constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h.copy(), compute_c5(h, dx)
