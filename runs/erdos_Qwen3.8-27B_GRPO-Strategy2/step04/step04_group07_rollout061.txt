    #   h[j] appears as second factor when i+m-n+1=j, i.e., i=j-m+n-1: contributes -h[j-m+n+1]

        #   h[j] as first factor (i=j): contributes (1-h[j+k]) if 0<=j+k<n
        #   h[j] as second factor (i=j-k, i+k=j): contributes -h[j-k] if 0<=j-k<n
        k = np.arange(-(n-1), n)
        slope = np.zeros(n_full)
        fk = (j + k >= 0) & (j + k < n)
        bs = (j - k >= 0) & (j - k < n)
        slope[fk] += 1.0 - h[j + k[fk]]
        slope[bs] -= h[j - k[bs]]
        
        # Current correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # C5(t) = dx * max_m(CORR[m] + (t - h[j]) * slope[m])
        # Find min over t in [0,1]
        # Check t=0, t=1, and crossing of top-2
        h_j = h[j]
        best_t = h_j
        best_val = np.inf
        
        for t in [0.0, 1.0, h_j]:
            val = np.max(corr + (t - h_j) * slope)
            if val < best_val:
                best_val = val
                best_t = t

        val_at_hj = corr  # at t=h_j, it's just corr
        top_idx = np.argsort(val_at_hj)[::-1]
        a, b = top_idx[0], top_idx[1]
        if abs(slope[a] - slope[b]) > 1e-15:
            t_cross = h_j + (corr[b] - corr[a]) / (slope[a] - slope[b])
            if 0.0 <= t_cross <= 1.0:
                val = np.max(corr + (t_cross - h_j) * slope)
                if val < best_val:
                    best_val = val
                    best_t = t_cross
        
        # Also check crossing of top 1 with 3rd, 4th (for safety)
        for b_idx in top_idx[1:4]:
            if abs(slope[a] - slope[b_idx]) > 1e-15:
                t_cross = h_j + (corr[b_idx] - corr[a]) / (slope[a] - slope[b_idx])
                if 0.0 <= t_cross <= 1.0:
                    val = np.max(corr + (t_cross - h_j) * slope)
                    if val < best_val:
                        best_val = val
                        best_t = t_cross
        
        h[j] = best_t
    
    # Repair after sweep
    h = repair(h, n)
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def cyclic_exact_cd(h, n, n_sweeps, t_end):
    """Cyclic coordinate descent with exact 1D minimization."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    n_full = 2 * n - 1
    k_vals = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            # Compute slope vector
            fk = (j + k_vals >= 0) & (j + k_vals < n)
            bs = (j - k_vals >= 0) & (j - k_vals < n)
            slope = np.zeros(n_full)
            slope[fk] += 1.0 - h[j + k_vals[fk]]
            slope[bs] -= h[j - k_vals[bs]]
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            h_j = h[j]
            
            # C5(t) ∝ max_m(corr[m] + (t-h_j)*slope[m])
            # Check boundaries and top crossings
            best_t = h_j
            best_v = np.inf
            for t in (0.0, 1.0):
                v = np.max(corr + (t - h_j) * slope)
                if v < best_v:
                    best_v = v
                    best_t = t
            
            top3 = np.argpartition(corr, -3)[::-1][:3]
            for ai in top3:
                for bi in top3:
                    if ai == bi:
                        continue
                    ds = slope[ai] - slope[bi]
                    if abs(ds) < 1e-15:
                        continue
                    tc = h_j + (corr[bi] - corr[ai]) / ds
                    if -0.5 <= tc <= 1.5:
                        tc = np.clip(tc, 0.0, 1.0)
                        v = np.max(corr + (tc - h_j) * slope)
                        if v < best_v:
                            best_v = v
                            best_t = tc
            
            if abs(best_t - h_j) > 1e-10:
                h[j] = best_t
                improved = True
        
        h = repair(h, n)
        c5_now = compute_c5(h, n)
        if c5_now < best_c5:
            best_c5 = c5_now
        elif not improved:
            break
    return h, best_c5

def pair_exact_cd(h, n, n_iter, t_end, rng):
    """Random pair moves with exact 1D line search."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    n_full = 2 * n - 1
    k_vals = np.arange(-(n-1), n)
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        
        fk_i = (i + k_vals >= 0) & (i + k_vals < n)
        bs_i = (i - k_vals >= 0) & (i - k_vals < n)
        sl_i = np.zeros(n_full)
        sl_i[fk_i] += 1.0 - h[i + k_vals[fk_i]]
        sl_i[bs_i] -= h[i - k_vals[bs_i]]
        
        fk_j = (j + k_vals >= 0) & (j + k_vals < n)
        bs_j = (j - k_vals >= 0) & (j - k_vals < n)
        sl_j = np.zeros(n_full)
        sl_j[fk_j] += 1.0 - h[j + k_vals[fk_j]]
        sl_j[bs_j] -= h[j - k_vals[bs_j]]
        
        # Move: h[i]+=δ, h[j]-=δ
        # Δcorr[m] = δ*(sl_i[m] - sl_j[m])
        dslope = sl_i - sl_j
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # δ range
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-14:
            continue
        
        best_d = 0.0
        best_v = np.inf
        for d in (d_lo, d_hi, 0.0):
            v = np.max(corr + d * dslope)
            if v < best_v:
                best_v = v
                best_d = d
        
        # Check crossings of top correlations
        top3 = np.argpartition(corr, -3)[::-1][:3]
        for ai in top3:
            for bi in top3:
                if ai == bi:
                    continue
                dd = dslope[ai] - dslope[bi]
                if abs(dd) < 1e-15:
                    continue
                tc = (corr[bi] - corr[ai]) / dd
                if d_lo <= tc <= d_hi:
                    v = np.max(corr + tc * dslope)
                    if v < best_v:
                        best_v = v
                        best_d = tc
        
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            h_new_c5 = compute_c5(h, n)
            if h_new_c5 < best_c5:
                best_c5 = h_new_c5
    
    h = repair(h, n)
    return h, compute_c5(h, n)

def slsqp_soft(h, n, alpha, maxiter=500):
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

So the approach is valid for finding the optimal δ, but for the NEXT iteration, I need to recompute sl_i and sl_j.

The key optimization: I don't need to recompute `np.correlate` every iteration. I just need to update it incrementally.

Let me restructure pair_descent:

```python
def pair_descent(h, n, n_iter, t_end, rng):

    n_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')

            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        sl_i = slopes_for_j(h, i, n)
        sl_j = slopes_for_j(h, j, n)
        dslope = sl_i - sl_j
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-14:
            continue
        d_new, _ = find_1d_min(corr, dslope, 0.0, d_lo, d_hi)
        if abs(d_new) > 1e-12:
            h[i] += d_new
            h[j] -= d_new
            np.clip(h, 0, 1, out=h)
            corr += d_new * dslope
            c5 = float(np.max(corr) * 2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
    
    h = repair(h, n)
    return h, compute_c5(h, n)
```

This avoids the O(n²) correlate computation at every step. The slopes_for_j is O(n) and the corr update is O(n). So each iteration is O(n) instead of O(n²). For n=64 and 25000 iterations, that's 25000*64 ≈ 1.6M operations. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(3000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def slopes_for_j(h, j, n):
    n_full = 2 * n - 1
    k = np.arange(-(n - 1), n)
    fk = (j + k >= 0) & (j + k < n)
    bs = (j - k >= 0) & (j - k < n)
    sl = np.zeros(n_full)
    sl[fk] += 1.0 - h[j + k[fk]]
    sl[bs] -= h[j - k[bs]]
    return sl

def find_1d_min(corr, slope, t0, lo=0.0, hi=1.0):
    best_t, best_v = t0, np.inf
    for t in (lo, hi, t0):
        v = np.max(corr + (t - t0) * slope)
        if v < best_v:
            best_v, best_t = v, t
    topk = np.argpartition(corr, -5)[::-1][:5]
    for ai in topk:
        sa = slope[ai]
        for bi in topk:
            if ai == bi:
                continue
            ds = sa - slope[bi]
            if abs(ds) < 1e-15:
                continue
            tc = t0 + (corr[bi] - corr[ai]) / ds
            if lo - 1e-9 <= tc <= hi + 1e-9:
                tc = np.clip(tc, lo, hi)
                v = np.max(corr + (tc - t0) * slope)
                if v < best_v:
                    best_v, best_t = v, tc
    return best_t, best_v

def cyclic_cd(h, n, n_sweeps, t_end):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        moved = False
        for j in range(n):
            sl = slopes_for_j(h, j, n)
            t_new, _ = find_1d_min(corr, sl, h[j])
            if abs(t_new - h[j]) > 1e-12:
                delta = t_new - h[j]
                h[j] = t_new
                corr += delta * sl
                moved = True
        h = repair(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
        if not moved:
            break
    return h, best_c5

def pair_descent(h, n, n_iter, t_end, rng):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    count = 0
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        sl_i = slopes_for_j(h, i, n)
        sl_j = slopes_for_j(h, j, n)
        dslope = sl_i - sl_j
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-14:
            continue
        d_new, _ = find_1d_min(corr, dslope, 0.0, d_lo, d_hi)
        if abs(d_new) > 1e-12:
            h[i] += d_new
            h[j] -= d_new
            np.clip(h, 0, 1, out=h)
            corr += d_new * dslope
            c5 = float(np.max(corr) * 2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                count = 0
            else:
                count += 1
        else:
            count += 1
        if count % 200 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
    h = repair(h, n)
    return h, compute_c5(h, n)

def slsqp_solve(h, n, alpha, maxiter=500):
    def obj(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(c)
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def block_cd(h, n, block_size, n_sweeps, t_end):
    """Coordinate descent updating blocks of consecutive points together."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for start in range(block_size):
            for j in range(start, n, block_size):
                idx = [j]
                if j + 1 < n and (j + 1 - start) % block_size == 0:
                    idx.append(j + 1)
                if len(idx) == 1:
                    sl = slopes_for_j(h, j, n)
                    t_new, _ = find_1d_min(
                        np.correlate(h, 1.0 - h, mode='full'), sl, h[j])
                    if abs(t_new - h[j]) > 1e-12:
                        h[j] = t_new
                        improved = True
        h = repair(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            improved = True
        if not improved:
            break
    return h, best_c5
