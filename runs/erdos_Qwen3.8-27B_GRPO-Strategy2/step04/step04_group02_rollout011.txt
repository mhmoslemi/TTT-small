def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def topk_obj(x, k, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    srt = np.sort(corr)[::-1]
    return np.mean(srt[:k]) + pen * (np.sum(x) - target) ** 2

        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)

        j = rng.randint(0, max(1, n - bs))
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if h[i+di]+eps < -1e-9 or h[i+di]+eps > 1.0+1e-9:
                ok = False; break
            if h[j+di]-eps < -1e-9 or h[j+di]-eps > 1.0+1e-9:
                ok = False; break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        omh[i:i+bs] = 1.0 - h[i:i+bs]
        omh[j:j+bs] = 1.0 - h[j:j+bs]
        corr = np.correlate(h, omh, mode='full')
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
            omh[i:i+bs] = 1.0 - h[i:i+bs]
            omh[j:j+bs] = 1.0 - h[j:j+bs]
    return h, best_c5

def fine_polish(h, n, dx, rng, deadline):
    h = h.copy()
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    best_c5 = float(np.max(corr) * dx)
    scales = [0.001, 0.0004, 0.00015, 0.00005, 0.00002]
    for scale in scales:
        guard = 500
        while time.time() < deadline and guard > 0:
            no_imp = 0
            while no_imp < 200000 and time.time() < deadline:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                hi = h[i] + d
                if hi < 0.0 or hi > 1.0:
                    no_imp += 1
                    continue
                hj = h[j] - d
                if hj < 0.0 or hj > 1.0:
                    no_imp += 1
                    continue
                h[i] = hi; h[j] = hj
                omh[i] = 1.0 - hi; omh[j] = 1.0 - hj
                corr = np.correlate(h, omh, mode='full')
                c5 = float(np.max(corr)) * dx
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    no_imp = 0
                else:
                    h[i] -= d; h[j] += d
                    omh[i] = 1.0 - h[i]; omh[j] = 1.0 - h[j]
                    no_imp += 1
            guard -= 1
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = repair(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Phase 1: L-BFGS-B quick polish (25s)
    t_end = t0 + 25
    for alpha in [300, 2000, 10000]:
        if time.time() >= t_end:
            break
        for pen in [1000, 10000]:
            if time.time() >= t_end:
                break
            try:
                res = minimize(lse_obj, h, args=(alpha, target, pen),
                             method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                             options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-9})
                hc = repair(res.x, n)
                c5c = compute_c5(hc, dx)
                if c5c < best_c5 - 1e-12:
                    h, best_c5, best_h = hc, c5c, hc.copy()
            except Exception:
                pass

    # Phase 2: Block search (60s)
    t_end = t0 + 85
    h, c5b = block_search(h, n, dx, rng, t_end)
    if c5b < best_c5 - 1e-12:
        best_c5, best_h = c5b, h.copy()

    # Phase 3: Main aggressive local search (700s)
    t_end = t0 + 785
    h, c5m = super_local_search(h, n, dx, rng, t_end)
    if c5m < best_c5 - 1e-12:
        best_c5, best_h = c5m, h.copy()

    # Phase 4: Another L-BFGS pass (30s)
    t_end = min(t0 + 815, time.time() + 30)
    for alpha in [5000, 30000]:
        if time.time() >= t_end:
            break
        try:
            res = minimize(lse_obj, h, args=(alpha, target, 5000),
                         method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-10})
            hc = repair(res.x, n)
            c5c = compute_c5(hc, dx)
            if c5c < best_c5 - 1e-12:
                h, best_c5, best_h = hc, c5c, hc.copy()
        except Exception:
            pass

    # Phase 5: Fine polish (remaining - 10s)
    t_end = t0 + budget_s - 10
    h, c5f = fine_polish(h, n, dx, rng, t_end)
    if c5f < best_c5 - 1e-12:
        best_c5, best_h = c5f, h.copy()

    # Final repair and evaluation
    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
