```python
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):

    """Returns (objective, gradient) for softplus-smoothed max of correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)

        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - h[ll:])
            g[ll:] -= wk * h[:n - ll]
    return logsum, g * dx

def _adam_step(h, g, m_v, v_v, t, lr, n, beta1=0.9, beta2=0.999, eps=1e-8):
    m_v = beta1 * m_v + (1 - beta1) * g
    v_v = beta2 * v_v + (1 - beta2) * g * g
    m_hat = m_v / (1 - beta1 ** t)
    v_hat = v_v / (1 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def _c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - h[ll:])
            g[ll:] -= wk * h[:n - ll]
    return obj, g * dx

def _adam_opt(h, n, budget_end, a_lo, a_hi, lr, max_it):
    h = h.copy()
    mv = np.zeros(n)
    vv = np.zeros(n)
    for t in range(1, max_it + 1):
        if time.time() > budget_end:
            break
        frac = (t - 1) / max(1, max_it - 1)
        alpha = a_lo * (a_hi / a_lo) ** frac
        _, g = _smooth_grad(h, n, alpha)
        mv = 0.9 * mv + 0.1 * g
        vv = 0.999 * vv + 0.001 * g * g
        mh = mv / (1 - 0.9 ** t)
        vh = vv / (1 - 0.999 ** t)
        h = _project(h - lr * mh / (np.sqrt(vh) + 1e-8), n)
    return h

def _pair_cd(h, n, budget_end, rng, passes=30, n_try=500):
    c5c = _c5_of(h, n)
    deltas = np.array([1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2,
                       -1e-4, -3e-4, -1e-3, -3e-3, -1e-2, -3e-2])
    for _p in range(passes):
        if time.time() > budget_end:
            break
        improved = False
        for _ in range(min(n_try, int((budget_end - time.time()) * 200))):
            if time.time() > budget_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            if hi < 0.005 and hj < 0.005:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5_of(h, n)
                if c < c5c - 1e-14:
                    c5c = c
                    hi, hj = ni, nj
                    improved = True
                else:
                    h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5c

def _anneal_pairs(h, n, budget_end, rng, T0=0.002, T1=0.0001):
    c5c = _c5_of(h, n)
    n_steps = 20000
    for s in range(n_steps):
        if time.time() > budget_end:
            break
        T = T0 * (T1 / T0) ** (s / n_steps)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        if hi < 0.005 and hj < 0.005:
            continue
        d = (rng.rand() - 0.5) * 0.06
        ni, nj = hi + d, hj - d
        if not (0 <= ni <= 1 and 0 <= nj <= 1):
            continue
        h[i], h[j] = ni, nj
        c = _c5_of(h, n)
        if c < c5c:
            c5c = c
            hi, hj = ni, nj
        elif rng.rand() < np.exp(-(c - c5c) / T):
            hi, hj = ni, nj
        else:
            h[i], h[j] = hi, hj
    return h, c5c

def _slsqp_polish(h, n, alpha, budget_end):
    nx = _GLOBAL_N
    bounds = [(0.0, 1.0)] * nx
    cons = [{'type': 'eq',
             'fun': lambda x: float(np.sum(x) - nx / 2.0),
             'jac': lambda x: np.ones(nx)}]
    def obj(x):
        return _smooth_grad(x, nx, alpha)[0]
    def jac(x):
        return _smooth_grad(x, nx, alpha)[1]
    try:
        res = minimize(obj, h.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-16})
        hc = _repair(res.x, nx)
        return hc, _c5_of(hc, nx)
    except Exception:
        return h.copy(), _c5_of(h, n)

_GLOBAL_N = 140

def run(seed=42, budget_s=1100, **kwargs):
    global _GLOBAL_N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n_init = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n_init)
    best_h = h0.copy()
    best_n = n_init
    best_c5 = _c5_of(best_h, best_n)
    _GLOBAL_N = best_n

    # Phase 1: Adam with alpha sweep
    p1_end = t0 + budget_s * 0.30
    h1 = _adam_opt(best_h, best_n, p1_end, 5000, 300000, 0.015, 15000)
    c1 = _c5_of(h1, best_n)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: SLSQP polish
    if time.time() < t0 + budget_s * 0.40:
        _GLOBAL_N = best_n
        hc, c5 = _slsqp_polish(best_h, best_n, 100000, t0 + budget_s * 0.40)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 3: Coordinate descent (many passes)
    p3_end = t0 + budget_s * 0.55
    best_h, best_c5 = _pair_cd(best_h, best_n, p3_end, rng, passes=40, n_try=800)

    # Phase 4: Diversification with re-optimization
    p4_end = t0 + budget_s * 0.82
    for r in range(12):
        if time.time() > p4_end:
            break
        # Different perturbation strategies
        strat = r % 4
        if strat == 0:
            hp = best_h + 0.02 * rng.randn(best_n)
        elif strat == 1:
            hp = best_h.copy()
            blk = 3 + rng.randint(5)
            st = rng.randint(0, best_n - blk)
            hp[st:st + blk] = hp[st:st + blk][::-1]
        elif strat == 2:
            hp = best_h.copy()
            for _ in range(3):
                st = rng.randint(0, best_n - 4)
                hp[st:st + 4], hp[st + 4:st + 8] = hp[st + 4:st + 8].copy(), hp[st:st + 4].copy()
        else:
            hp = best_h + 0.04 * rng.randn(best_n)
            hp = np.clip(hp, 0, 1)
        hp = _repair(hp, best_n)
        # Quick Adam
        seg_end = min(time.time() + 25, p4_end - 5)
        ho = _adam_opt(hp, best_n, seg_end, 30000, 200000, 0.008, 3000)
        co = _c5_of(ho, best_n)
        # Quick CD
        if time.time() < p4_end:
            ho, co = _pair_cd(ho, best_n, min(time.time() + 10, p4_end - 2), rng,
                              passes=8, n_try=400)
        if co < best_c5:
            best_c5, best_h = co, ho

    # Phase 5: Simulated annealing on pairs
    if time.time() < t0 + budget_s * 0.92:
        ha, ca = _anneal_pairs(best_h, best_n, t0 + budget_s * 0.92, rng,
                               T0=0.003, T1=0.00005)
        ha, ca = _pair_cd(ha, best_n, min(time.time() + 10, t0 + budget_s * 0.92),
                          rng, passes=10, n_try=500)
        if ca < best_c5:
            best_c5, best_h = ca, ha

    # Phase 6: Final fine-tuning
    p6_end = t0 + budget_s - 10
    if time.time() < p6_end:
        best_h, best_c5 = _pair_cd(best_h, best_n, p6_end, rng, passes=50, n_try=1200)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_of(best_h, best_n)
    return (best_h, best_c5, best_n)
```
