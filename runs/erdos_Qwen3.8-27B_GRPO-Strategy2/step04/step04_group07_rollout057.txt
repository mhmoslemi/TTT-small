import time

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_obj_and_grad(v, alpha, pen):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha + pen * (v.sum() - n / 2.0) ** 2
    
    # Gradient of soft-max part
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sft = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        j1 = max(0, sft)
        j2 = min(n, n + sft)
        if j1 < j2:
            grad[j1:j2] += wk * (1.0 - v[j1 - sft:j2 - sft])
        j3 = max(0, -sft)
        j4 = min(n, n - sft)
        if j3 < j4:
            grad[j3:j4] -= wk * v[j3 + sft:j4 + sft]
    
    # Gradient of penalty
    grad += 2 * pen * (v.sum() - n / 2.0)
    
    return obj, grad

def lbfgs_opt(h0, n, alpha, pen, maxiter, ftol, gtol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        fun=lambda v: soft_obj_and_grad(v, alpha, pen)[0],
        x0=h0,
        jac=lambda v: soft_obj_and_grad(v, alpha, pen)[1],
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                 'maxls': 50, 'maxfun': maxiter * 5}
    )
    return repair(res.x, n), res

    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_obj(v, alpha, pen):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (v.sum() - n / 2.0) ** 2

def soft_grad(v, alpha, pen):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sft = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        j1 = max(0, sft)
        j2 = min(n, n + sft)
        if j1 < j2:
            grad[j1:j2] += wk * (1.0 - v[j1 - sft:j2 - sft])
        j3 = max(0, -sft)
        j4 = min(n, n - sft)
        if j3 < j4:
            grad[j3:j4] -= wk * v[j3 + sft:j4 + sft]
    grad += 2.0 * pen * (v.sum() - n / 2.0)
    return grad

    bounds = [(0.0, 1.0)] * n
    res = minimize(soft_obj, h0, args=(alpha, pen), jac=soft_grad,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18,
                            'gtol': 1e-10, 'maxls': 60, 'maxfun': maxiter * 8})
    return repair(res.x, n)

    h = h0.copy()
    best = compute_c5(h, n)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > t_end:
            break
        grad = soft_grad(h, alpha, 0.0)
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best - 1e-15:
            best = c5_new
            h = h_new
            if it % 1000 == 0 and it > 100:
                lr *= 0.8
        else:
            lr *= 0.6
            if lr < 1e-8:
                break
    return h, best

def coord_descent(h0, n, t_end, rng, n_rounds=800):
    h = h0.copy()
    best = compute_c5(h, n)
    steps = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    stall = 0
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(8 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni = hid + sign * d
                    nj = hjd - sign * d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:

                        improved = True

    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_obj(v, alpha, pen):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (v.sum() - n / 2.0) ** 2

def soft_grad(v, alpha, pen):
    n = len(v)
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sft = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        j1 = max(0, sft)
        j2 = min(n, n + sft)
        if j1 < j2:
            grad[j1:j2] += wk * (1.0 - v[j1 - sft:j2 - sft])
        j3 = max(0, -sft)
        j4 = min(n, n - sft)
        if j3 < j4:
            grad[j3:j4] -= wk * v[j3 + sft:j4 + sft]
    grad += 2.0 * pen * (v.sum() - n / 2.0)
    return grad

def lbfgs_opt(h0, alpha, pen, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(soft_obj, h0, args=(alpha, pen), jac=soft_grad,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18,
                            'gtol': 1e-10, 'maxls': 60, 'maxfun': maxiter * 8})
    return repair(res.x, n)

def pgd_opt(h0, alpha, lr_init, t_end, max_iter):
    n = len(h0)
    h = h0.copy()
    best = compute_c5(h, n)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > t_end:
            break
        grad = soft_grad(h, alpha, 500.0)
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best - 1e-15:
            best = c5_new
            h = h_new
            if it > 300:
                lr *= 0.97
        else:
            lr *= 0.65
            if lr < 1e-10:
                break
    return h, best

def coord_descent(h0, n, t_end, rng, n_rounds):
    h = h0.copy()
    best = compute_c5(h, n)
    steps = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    stall = 0
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(8 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni = hid + sign * d
                    nj = hjd - sign * d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
                if improved:
                    break
        if not improved:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
    return h, best

def resample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return repair(np.interp(x_new, x_old, h_old), n_new)

    best_h = None

            elif c5_new < compute_c5(h_work, n) - 1e-14:

    # Phase 2: PGD refinement
