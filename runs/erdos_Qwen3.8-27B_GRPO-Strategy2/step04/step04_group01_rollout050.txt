def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad(h, tol=0.01):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    thr = mx * (1 - tol)
    act = np.where(c >= thr)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in act:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += 1.0 - h[sh:]
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += 1.0 - h[:n-sp]
            g[:n-sp] -= h[sp:]
    g /= len(act)
    return g * dx
