import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, n):
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):

        h_new = np.clip(h + delta, 0.0, 1.0)
        if h_new.sum() > target:
            hi = delta
        else:
            lo = delta
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for L in range(n):
        wL = w[n - 1 + L]
        if L > 0:
            wL += w[n - 1 - L]
        wL *= 2.0 / n
        if wL < 1e-14:
            continue
        nl = n - L
        g[:nl] += wL * (1.0 - h[L:])
        g[L:] += wL * (-h[:nl])
    return g

def _pgd(h, n, alpha, max_iter, step_init, t_end):
    h = h.copy()
    c5_cur = _eval(h)
    step = step_init
    no_imp = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h_new = h - step * g
        h_new = _project_sum(h_new, n)
        c5_new = _eval(h_new)
        if c5_new < c5_cur - 1e-14:
            h = h_new
            c5_cur = c5_new
            no_imp = 0
            step = min(step * 1.03, 0.5)
        else:
            no_imp += 1
            step *= 0.7
            if no_imp > 100:
                step = step_init * 0.5
                no_imp = 0
                if step < 1e-10:
                    break
    return h, c5_cur

def _local_search(h, n, rng, t_end, max_iter=500000):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, n):
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        delta = (lo + hi) / 2
        if np.clip(h + delta, 0, 1).sum() > target:
            hi = delta
        else:
            lo = delta
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for L in range(n):
        wL = w[n - 1 + L]
        if L > 0:
            wL += w[n - 1 - L]
        wL *= 2.0 / n
        if wL < 1e-14:
            continue
        nl = n - L
        g[:nl] += wL * (1.0 - h[L:])
        g[L:] += wL * (-h[:nl])
    return g

def _pgd(h, n, alpha, max_iter, step_init, t_end):
    h = h.copy()
    c5_cur = _eval(h)
    step = step_init
    no_imp = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h_new = _project_sum(h - step * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5_cur - 1e-14:
            h = h_new; c5_cur = c5_new; no_imp = 0
            step = min(step * 1.04, 0.5)
        else:
            no_imp += 1
            step *= 0.65
            if no_imp > 200:
                step = step_init * 0.4
                no_imp = 0
                if step < 1e-10:
                    break
    return h, c5_cur

def _local_search(h, n, rng, t_end, max_iter=400000):
    c5c = _eval(h)
    scale = 0.008
    no_imp = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n; no_imp = 0
            scale = min(scale * 1.03, 0.04)
        else:
            h[i], h[j] = h[i] - d, h[j] + d
            no_imp += 1
            if no_imp > 400:
                scale *= 0.8; no_imp = 0
                if scale < 1e-8:
                    break
    return h, c5c

def _fine_search(h, n, rng, t_end, max_iter=200000):
    c5c = _eval(h)
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.choice([1, -1]) * 1e-6 * rng.choice([1, 2, 5, 10, 20, 50])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _try_slsqp(h, n, alpha, maxiter, t_end):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def obj_grad(x):
        return _grad_smooth(x, alpha)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        if time.time() > t_end:
            return h, _eval(h)
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        hx = _repair(res.x, n)
        return hx, _eval(hx)
    except Exception:
        return h, _eval(h)

    t0 = time.time()

    def _tl():
