The problem min_h max_k f_k(h) can be viewed as a zero-sum game where:

- Update p to put more weight on the currently active lags

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad_lag(h, k, n):
    """Subgradient of the k-th correlation term w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _active_lags(h, top_k=10, tol=0.005):
    """Find lags that are within tol of the maximum"""
    c = _corr(h)
    mx = c.max()
    active = np.where(c >= mx - tol)[0]
    return active[:top_k], mx

def _momentum_subgrad(h, n, target_c, max_iter, alpha0, beta, top_k=15, tol=0.003):
    """Momentum subgradient method with lag equalization"""
    h = h.copy()
    v = np.zeros(n)
    c5 = _eval(h)
    for it in range(max_iter):
        c = _corr(h)
        active, mx = _active_lags(h, top_k, tol)
        # Weighted subgradient for active lags
        g = np.zeros(n)
        for k in active:
            g += _subgrad_lag(h, k, n)
        g /= len(active)
        # Nesterov momentum
        v = beta * v - alpha0 * g
        h_new = _proj_to_sum(h + v, n / 2.0)
        c5_new = _eval(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            # Decay step size
            alpha0 *= 0.995
        else:
            v *= 0.5
            alpha0 *= 0.9
            if alpha0 < 1e-10:
                break
    return h, c5
```

However, we can LINEARIZE around the current solution:

Then the problem becomes:
minimize C
subject to c_k(h₀) + ∇c_k(h₀)·(h - h₀) ≤ C for k in active set
0 ≤ h ≤ 1, sum(h) = n/2

This IS a linear program! And we can use scipy's linprog or cvxpy.

This is the **out-of-kilter** or **iterative Linearization** approach. Let me implement this:

```python
def _linearized_solve(h0, active_lags, n, max_iters=50):
    """Solve linearized subproblem iteratively"""
    from scipy.optimize import linprog
    c = _corr(h0)
    dx = 2.0 / n
    C = c.max() + 1e-6  # initial upper bound
    
    for it in range(max_iters):
        # Build LP: min C s.t. c_k_lin(h) <= C for k in active, 0<=h<=1, sum(h)=n/2
        # Variables: h[0..n-1], C
        # c_k_lin(h) = c_k(h0) + sum_i grad_k(h0)[i] * h[i] - (sum_i grad_k(h0)[i] * h0[i])
        #            = c_k(h0) - grad_k·h0 + grad_k·h
        # So constraint: grad_k·h + (c_k(h0) - grad_k·h0) <= C
        # Rearranged: grad_k·h - C <= grad_k·h0 - c_k(h0)
        
        m = 2 * n - 1  # number of correlation terms
        # Actually we only need constraints for active lags
        A_ub = []
        b_ub = []
        for k in active_lags:
            g = _subgrad_lag(h0, k, n)  # this already includes dx
            # g·h - C <= g·h0 - c_k(h0)
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(g @ h0 - c[k])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Objective: minimize C
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Bounds: 0 <= h[i] <= 1, C >= 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        # Constraint: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            C_new = res.x[n]
            # Check all lags
            c_all = _corr(h_new)
            if c_all.max() * dx <= C_new + 1e-4:
                # Feasible! But need to check other lags
                violating = np.where(c_all * dx > C_new + 1e-3)[0]
                if len(violating) == 0:
                    return h_new, C_new
                else:
                    active_lags = np.concatenate([active_lags, violating])
```

```python
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] = (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] = -h[i2 + sp]
    return g * dx

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_cutting(h0, n, max_rounds, tol=1e-5):
    dx = 2.0 / n
    c_raw = _corr(h0)
    top = np.argsort(c_raw)[-20:]
    active = set(top.tolist())
    best_h, best_c5 = h0.copy(), c_raw.max() * dx
    for rnd in range(max_rounds):
        cs = _corr(h0)
        A_rows, b_vals = [], []
        for k in active:
            g = _grad_lag(h0, k, n)
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(g @ h0 - cs[k] * dx)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0, 1)] * n + [(0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_new = _repair(res.x[:n], n)
        C_lp = max(res.x[n], 0.0)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        cs_all = _corr(h_new)
        viol = np.where(cs_all * dx > C_lp + tol)[0]
        top_new = np.argsort(cs_all)[-25:]
        active = set(viol.tolist()) | set(top_new.tolist())
        if len(active) > 300:
            active = set(np.argsort(cs_all)[-300:].tolist())
        h0 = h_new
        gap = c5_new - C_lp
        if gap < 1e-7 and len(viol) == 0:
            break
    return best_h, best_c5

def _subgrad_polish(h, n, n_iter, alpha0=0.02, beta=0.92):
    h = h.copy()
    v = np.zeros(n)
    alpha = alpha0
    c5 = _eval(h)
    for _ in range(n_iter):
        c = _corr(h)
        # Weighted top-5 subgradient
        top5 = np.argsort(c)[-5:]
        g = np.zeros(n)
        for k in top5:
            g += _grad_lag(h, k, n)
        g /= 5.0
        v = beta * v - alpha * g
        h_new = _proj_to_sum(h + v, n / 2.0)
        c5n = _eval(h_new)
        if c5n <= c5:
            h = h_new
            c5 = c5n
            alpha *= 0.998
        else:
            v *= 0.4
            alpha *= 0.9
            if alpha < 1e-13:
                break
    return h, c5

def _pairwise_search(h, n, n_trials, scale, rng):
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = hi - d, hj + d
    return h, c5

def _upsample(h, f):
    return np.repeat(h, f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: LP cutting plane at base resolution
    if _el() < budget_s - 300:
        h1, c1 = _lp_cutting(h, n0, max_rounds=200)
        if c1 < best_c5:
            best_c5, best_h, best_n = c1, h1.copy(), n0

    # Phase 2: Subgradient polish then LP again
    if _el() < budget_s - 250:
        h2, c2 = _subgrad_polish(best_h, best_n, 4000, alpha0=0.015)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        if _el() < budget_s - 150:
            h3, c3 = _lp_cutting(best_h, best_n, max_rounds=150)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()
            h4, c4 = _subgrad_polish(best_h, best_n, 2000, alpha0=0.008)
            if c4 < best_c5:
                best_c5, best_h = c4, h4.copy()

    # Phase 3: 2x resolution
    if _el() < budget_s - 400:
        n1 = best_n * 2
        h_up = _repair(_upsample(best_h, 2), n1)
        if _el() < budget_s - 350:
            h5, c5 = _lp_cutting(h_up, n1, max_rounds=120)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h5.copy(), n1
            if _el() < budget_s - 200:
                h6, c6 = _subgrad_polish(best_h, best_n, 1500, alpha0=0.006)
                if c6 < best_c5:
                    best_c5, best_h = c6, h6.copy()
                if _el() < budget_s - 100:
                    h7, c7 = _lp_cutting(best_h, best_n, max_rounds=80)
                    if c7 < best_c5:
                        best_c5, best_h = c7, h7.copy()

    # Phase 4: Structured random restarts
    n_cur = best_n
    if _el() < budget_s - 200:
        for r in range(10):
            if _el() > budget_s - 150:
                break
            hp = best_h.copy()
            mode = r % 4
            if mode == 0:
                hp += 0.02 * rng.randn(n_cur)
            elif mode == 1:
                hp = 1.0 - hp + 0.01 * rng.randn(n_cur)
            elif mode == 2:
                for _b in range(rng.randint(3, 7)):
                    bs = rng.randint(3, max(4, n_cur // 12))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
            else:
                freqs = rng.randint(1, n_cur // 3 + 1, size=4)
                t = np.arange(n_cur)
                for f in freqs:
                    hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * t / n_cur)
            hp = _repair(hp, n_cur)
            hr, cr = _lp_cutting(hp, n_cur, max_rounds=60)
            hr, cr = _subgrad_polish(hr, n_cur, 800, alpha0=0.01)
            if cr < best_c5:
                best_c5, best_h, best_n = cr, hr.copy(), n_cur

    # Phase 5: Final LP + pairwise at decreasing scales
    if _el() < budget_s - 60:
        h8, c8 = _lp_cutting(best_h, best_n, max_rounds=100)
        if c8 < best_c5:
            best_c5, best_h = c8, h8.copy()
        best_h, best_c5 = _pairwise_search(best_h, best_n, 20000, 0.003, rng)
        if _el() < budget_s - 20:
            best_h, best_c5 = _pairwise_search(best_h, best_n, 30000, 0.0005, rng)

    # Phase 6: Ultra-fine
    if _el() < budget_s - 3:
        best_h, best_c5 = _pairwise_search(best_h, best_n, 40000, 2e-5, rng)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```
