    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - len(x) / 2.0) ** 2

    n = len(x)

        else:
            M = -L

            g[M:] -= pk * x[:n - M]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - len(x) / 2.0) ** 2

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _subgrad(h, k_act=10):
    c = _corr(h)
    n = len(h)
    cm = np.max(c)
    thresh = cm - 0.04 * cm
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _lbgfb(h, n, bounds, alphas, t0, budget, reserve, lam):
    bh = h.copy()
    bc5 = _c5(h, 2.0 / n)
    dx_n = 2.0 / n
    for a in alphas:
        if time.time() - t0 > budget - reserve:
            break
        try:
            res = minimize(
                lambda x, _a=a, _l=lam: _softobj(x, _a, _l),
                bh.copy(),
                jac=lambda x, _a=a, _l=lam: _softgrad(x, _a, _l),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-16})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx_n)
            if c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass
    return bh, bc5

def _local_search(h, n, dx, rng, t0, budget, reserve):
    bh = h.copy()
    bc5 = _c5(h, dx)
    h = h.copy()
    c5c = bc5

    # Phase 1: Subgradient with adaptive step
    step = 0.025
    for _ in range(4000):
        if time.time() - t0 > budget - reserve:
            break
        g = _subgrad(h, k_act=8)
        g_c = g - g.mean()
        ng = np.linalg.norm(g_c)
        if ng < 1e-13:
            break
        g_c /= ng
        found = False
        for s_try in (step, step * 0.5, step * 0.25, step * 0.1):
            h_new = _repair(h - s_try * g_c, n)
            c_new = _c5(h_new, dx)
            if c_new < c5c - 1e-16:
                h = h_new
                c5c = c_new
                found = True
                if c5c < bc5:
                    bc5 = c5c
                    bh = h.copy()
                break
        if not found:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            step = min(step * 1.04, 0.06)

    # Phase 2: Gradient-guided 2-swap
    h = bh.copy()
    c5c = bc5
    for _sweep in range(500):
        if time.time() - t0 > budget - reserve:
            break
        g = _softgrad(h, 300000, 0.0)
        order = np.argsort(np.abs(g))[::-1][:40]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 25, len(order))):
                j = order[jj]
                gdiff = g[i] - g[j]
                spd = max(0.001, 0.005 * abs(gdiff) / (np.max(np.abs(g)) + 1e-10))
                dlist = (spd, spd * 3, spd * 8, 0.03, 0.08,
                        -spd, -spd * 3, -spd * 8, -0.03, -0.08)
                best_d, best_c = 0.0, c5c
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h.copy()
                    break
        if not improved:
            # Try larger random perturbation every few sweeps
            if _sweep % 20 == 15:
                for _rp in range(5):
                    i, j = rng.randint(n), rng.randint(n)
                    if i == j:
                        continue
                    d = rng.uniform(-0.05, 0.05)
                    h[i] += d
                    h[j] -= d
                    h = _repair(h, n)
                    c_new = _c5(h, dx)
                    if c_new < bc5 + 0.002:
                        if c_new < bc5:
                            bc5 = c_new
                            bh = h.copy()
                            c5c = c_new
                        break
            else:
                break

    # Phase 3: Fine random polish
    h = bh.copy()
    c5c = bc5
    remaining = max(0, budget - reserve - (time.time() - t0))
    nrand = int(remaining * 5000)
    for _ in range(nrand):
        if time.time() - t0 > budget - reserve:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.012, 0.012)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                bc5 = c
                bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc5
