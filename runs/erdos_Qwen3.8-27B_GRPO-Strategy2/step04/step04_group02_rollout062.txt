def compute_c5(h, dx):

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def ls_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

        hh, cc = lbfgs(hp, n, 5000, n / 2.0, 3000, 200)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
            h = hh.copy()

    # Phase 6: Block search + fine local search (remaining - 55s)
    t_end = deadline - 55
    if time.time() < t_end:
        t_mid = (time.time() + t_end) / 2
        best_h, best_c5 = block_search(best_h, n, dx, rng, t_mid, 30000)
        if time.time() < t_end:
            best_h, best_c5 = local_search(best_h, n, dx, rng, t_end, 100000)

    # Phase 7: Final polish (55s)
    if time.time() < deadline - 3:
        t_end2 = deadline - 25
        hh, cc = slsqp(best_h, n, 60000, n / 2.0, 300)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
        if time.time() < t_end2:
            best_h, best_c5 = local_search(best_h, n, dx, rng, t_end2, 60000)
        hh, cc = lbfgs(best_h, n, 100000, n / 2.0, 8000, 300)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
        if time.time() < deadline - 1:
            best_h, best_c5 = local_search(best_h, n, dx, rng, deadline - 0.5, 40000)

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
