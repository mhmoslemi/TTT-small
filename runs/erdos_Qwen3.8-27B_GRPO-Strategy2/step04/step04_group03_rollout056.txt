import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(x, alpha):
    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

    return np.sum(x) - n / 2.0

def _cf_jac(x, n):
    return np.ones(n)

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:

        else:

    return h, c5c

            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            no_imp = 0
        else:
            h[i] = 1.0; h[j] = 0.0
            no_imp += 1
            if no_imp % 2000 == 1999:
                for _ in range(30):
                    ii = int(rng.choice(ones)); jj = int(rng.choice(zeros))
                    h[ii] = 0.0; h[jj] = 1.0
                    c5t = _eval(h)
                    if c5t < c5:
                        c5 = c5t
                        ones.remove(ii); ones.append(jj)
                        zeros.remove(jj); zeros.append(ii)
                    else:
                        h[ii] = 1.0; h[jj] = 0.0
                no_imp = 0
    return h, c5

def _subgrad(h, n, t_end, lr0=0.03):
    c5c = _eval(h)
    lr = lr0
    for _ in range(2000000):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        max_k = int(np.argmax(c))
        lag = max_k - (n - 1)
        g = np.zeros(n)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= h[i2 + lag:j2 + lag]
        if np.all(g == 0):
            break
        h_new = _repair(h - lr * g, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            h = h_new; c5c = c5n
            lr = min(lr * 1.01, 0.15)
        else:
            lr *= 0.93
            if lr < 1e-6:
                break
    return h, c5c

def _multi_lag_grad(h, n, topk=5):
    c = _corr_full(h)
    m = c.max()
    idx = np.argsort(c)[::-1][:topk]
    w = np.zeros(2 * n - 1)
    for i in idx:
        w[i] = max(0.01, (m - 0.005) - c[i] + 0.005)
    w /= w.sum()
    g = np.zeros(n)
    for k in idx:
        wk = w[k]
        if wk < 1e-6:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * h[i2 + lag:j2 + lag]
    return g

def _multi_lag_descend(h, n, t_end, lr0=0.02):
    c5c = _eval(h)
    lr = lr0
    for _ in range(1000000):
        if time.time() > t_end:
            break
        g = _multi_lag_grad(h, n, topk=8)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        h_new = _repair(h - lr * g / gnorm, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            h = h_new; c5c = c5n
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.94
            if lr < 5e-6:
                break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n_init = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n_init)

    best_h = h_init.copy()
    best_c5 = _eval(h_init)
    best_n = n_init

    n_list = sorted(set([n_init, 40, 50, 60, 70, 80, 90, 100, 120]))
    inits = {n_init: h_init.copy()}
    for n in n_list:
        if n in inits:
            continue
        hc = _repair(_interp(h_init, n_init, n), n)
        inits[n] = hc

    for n in n_list:
        if time.time() - t0 > budget_s * 0.35:
            break
        h_cur = inits[n].copy()
        c5_cur = _eval(h_cur)
        for alpha in [5000, 50000, 200000, 1000000]:
            if time.time() - t0 > budget_s * 0.35:
                break
            h_new = _opt_slsqp(h_cur, n, alpha, maxiter=400)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
                h_cur = h_new.copy()
        if c5_cur < best_c5:
            best_c5, best_h, best_n = c5_cur, h_cur.copy(), n

    for n in [40, 50, 60, 70, 80, 90]:
        if time.time() - t0 > budget_s * 0.60:
            break
        h_bin = _make_binary(best_h, n, rng)
        h_b, c5_b = _binary_ls(h_bin.copy(), n, rng, time.time() + 12, 2500)
        if c5_b < best_c5:
            best_c5, best_h, best_n = c5_b, h_b.copy(), n
        for trial in range(4):
            h_r = np.zeros(n)
            h_r[rng.choice(n, n // 2, replace=False)] = 1.0
            h_r, c5_r = _binary_ls(h_r, n, rng, time.time() + 6, 1200)
            if c5_r < best_c5:
                best_c5, best_h, best_n = c5_r, h_r.copy(), n

    if time.time() - t0 < budget_s * 0.80:
        h = _opt_slsqp(best_h.copy(), best_n, 5000000, maxiter=800)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if time.time() - t0 < budget_s * 0.90:
        t_end = min(time.time() + 25, t0 + budget_s - 15)
        h, c5 = _multi_lag_descend(best_h.copy(), best_n, t_end)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if time.time() - t0 < budget_s * 0.96:
        t_end = min(time.time() + 15, t0 + budget_s - 5)
        h, c5 = _subgrad(best_h.copy(), best_n, t_end)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if time.time() - t0 < budget_s - 3:
        h = _opt_slsqp(best_h.copy(), best_n, 10000000, maxiter=400)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
