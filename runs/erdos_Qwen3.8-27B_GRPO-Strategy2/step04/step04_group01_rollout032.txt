    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            i2 = np.arange(n - sp)

Let me also think about another idea: **Fourier-based approach**.

The overlap at shift k is related to the autocorrelation of h. Specifically:
overlap(k) = ∫ h(x)(1-h(x+k)) dx = 1 - ∫ h(x)h(x+k) dx (for appropriate range)

The term ∫ h(x)h(x+k) dx is the autocorrelation of h at lag k. By Wiener-Khinchin, the autocorrelation is the inverse Fourier transform of |H(f)|².

So overlap(k) = 1 - IFFT(|FFT(h)|²)(k)

This gives an alternative way to compute the overlaps, but doesn't directly help with optimization.

However, it suggests that the problem is related to shaping the power spectrum of h. We want to maximize the minimum of the autocorrelation (since we want to minimize 1 - autocorrelation = overlap).

This is like designing a signal with a specific autocorrelation property. It's related to "minimum peak sidelobe" (MPSL) sequence design!

In radar signal design, MPSL sequences minimize the peak sidelobe of the autocorrelation. This is exactly analogous to our problem (minimizing the peak of the "overlap" which is related to 1 - autocorrelation).

Known techniques for MPSL include:
- FISTA (fast iterative shrinkage-thresholding)
- Proximal gradient methods
- Variable metric methods

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _tile_obj(x, T, n, alpha):
    h = np.tile(x, n // T)
    return _obj(h, alpha)

def _tile_grad(x, T, n, alpha):
    h = np.tile(x, n // T)
    gf = _grad(h, alpha)
    m = n // T
    g = np.zeros(T)
    for p in range(m):
        g += gf[p * T:(p + 1) * T]
    return g / m
