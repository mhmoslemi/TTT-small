    for a in [0.1, 0.2, 0.35, 0.5, 0.35, 0.2, 0.1, 0.05]:
        h = bh.copy()
        for _ in range(150):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-17:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def opt_pipeline(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 180)
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_opt(bh, n, dx, t0, lim - 120)
    if c < bc:
        bh, bc = r, c
    r, c = act_opt(bh, n, dx, t0, lim - 80)
    if c < bc:
        bh, bc = r, c
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 50)
    if c < bc:
        bh, bc = r, c
    r, c = spec_relax(bh, n, dx, t0, lim - 30)
    if c < bc:
        bh, bc = r, c
    r, c = act_opt(bh, n, dx, t0, lim - 15)
    if c < bc:
        bh, bc = r, c
    r, c = coord_polish(bh, n, dx, rng, t0, lim - 2, cnt=60000)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def make_diverse(h, n, seed):
    rng = np.random.RandomState(seed)
    cands = [h.copy()]
    for scale in [0.008, 0.015, 0.03, 0.06, 0.12, 0.2]:
        cands.append(repair(h + scale * rng.randn(n), n))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1.0, 1.5, 2.0, 3.0]:
        pert = 0.04 * np.sin(2 * np.pi * freq * x / 2.0 + rng.rand() * 6.28)
        cands.append(repair(h + pert, n))
    for sh in [-2, -1, 1, 2]:
        cands.append(repair(np.roll(h, sh), n))
    for s in [0.3, 0.5, 0.7]:
        cands.append(repair(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n, dtype=float), np.sort(h)), n))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0

    cands = make_diverse(h0, n0, seed)
    scored = sorted([(c5v(repair(c, n0), dx0), c) for c in cands], key=lambda t: t[0])

    t_end_main = budget_s - 120
    n_trials = min(len(scored), max(2, int((t_end_main - (time.time() - t0)) // 220)))

    for i in range(n_trials):
        if time.time() > t_end_main:
            break
        t_rem = t_end_main - (time.time() - t0)
        if t_rem < 180:
            break
        r, c = opt_pipeline(scored[i][1], n0, dx0, seed + i * 77, t0, t_end_main)
        if c < bc:
            bh, bc, bn = r, c, n0

    for n_new in [n0 - 20, n0 + 20, n0 + 40]:
        if n_new < 50 or n_new > 500:
            continue
        if time.time() > budget_s - 80:
            break
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, n_new, endpoint=False)
        h_new = repair(np.interp(xn, xo, bh, period=2.0), n_new)
        dx_new = 2.0 / n_new
        t_rem = budget_s - 10 - (time.time() - t0)
        if t_rem < 120:
            break
        r, c = opt_pipeline(h_new, n_new, dx_new, seed + 200 + n_new, t0, budget_s - 5)
        if c < bc:
            bh, bc, bn = r, c, n_new

    if time.time() < budget_s - 4:
        rng = np.random.RandomState(seed + 5000)
        r, c = coord_polish(bh, bn, 2.0 / bn, rng, t0, budget_s - 1, cnt=100000)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
