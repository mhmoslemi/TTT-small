def top_k_subgradient(h, n, K=5, temperature=0.01):
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    # Find top K
    # Weights: softmax of top K values
    top_k = np.argpartition(-corr, K)[:K]
    weights = np.exp((corr[top_k] - corr[top_k].max()) / temperature)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, k in zip(weights, top_k):
        lag = k - (n-1)
        for i in range(n):
            ip = i + lag
            if 0 <= ip < n:
                g[i] += w * dx * (1.0 - h[ip])
                g[ip] -= w * dx * h[i]
    return g, corr.max()

The Hessian is: -dx * (E[i,i+k] + E[i+k,i]) where E is the basis. This matrix is indefinite (it's not PSD in general for k≠0). So c_k is not convex.

The problem is non-convex. OK, so we can't use convex optimization. But the sub-gradient / local search approach should still work well.

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        m = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if m.any():
            hp[m] += diff / m.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _topk_grad(h, n, corr, K, temp):
    K = min(K, 2 * n - 1)
    top = np.argpartition(-corr, K - 1)[:K]
    vals = corr[top]
    w = np.exp((vals - vals.max()) / max(temp, 1e-10))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx, wi in zip(top, w):
        if wi < 1e-14:
            continue
        L = idx - (n - 1)
        if L >= 0:
            m = n - L
            if m > 0:
                g[:m] += wi * (1.0 - h[L:L + m])
                g[L:L + m] -= wi * h[:m]
        else:
            Lp = -L
            m = n - Lp
            if m > 0:
                g[Lp:Lp + m] += wi * (1.0 - h[:m])
                g[:m] -= wi * h[Lp:Lp + m]
    return g * dx

    c5c = _c5(h, n)
