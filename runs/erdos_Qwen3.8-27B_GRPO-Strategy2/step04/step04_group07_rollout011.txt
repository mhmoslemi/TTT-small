                d += a
    return h

def weighted_grad(h, n, w):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-16:
            continue
        if L >= 0:
            idx1 = np.arange(n - L)
            idx2 = idx1 + L
            grad[idx1] += wk * (1.0 - h[idx2])
            grad[idx2] += wk * (-h[idx1])
        else:
            Lp = -L
            idx2 = np.arange(Lp, n)
            idx1 = idx2 - Lp
            grad[idx1] += wk * (-h[idx2])
            grad[idx2] += wk * (1.0 - h[idx1])
    return grad

def slsqp_weighted(h, n, w, maxiter=200):

        return np.dot(w, corr)
    def obj_grad(v):
        return weighted_grad(v, n, w)
    bounds = [(0.0, 1.0)] * n

             'jac': lambda v: np.ones(n)}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h

def slsqp_softmax(h, n, alpha, maxiter=200):

                    best_h = h_new2.copy()
                    best = c5_new2
                    h_work = h_new2
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 999)
    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(init_h)
    except Exception:
        init_h = np.full(64, 0.5)
    init_h = repair(init_h, best_n)
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h, best_n)
    n = best_n

    def elapsed():
        return time.time() - t0

    # Phase 1: Iterative reweighting (30% budget)
    t1 = t0 + budget_s * 0.30
    best_h, best_c5 = reweight_loop(best_h, n, t1,
                                     [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005], 3)

    # Phase 2: Active-set SLSQP (10% budget)
    t2 = t0 + budget_s * 0.40
    h_work = best_h.copy()
    for m in [5, 8, 12, 16, 20]:
        if elapsed() > t2:
            break
        h_new = active_slsqp(h_work, n, m, maxiter=300)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 3: Softmax SLSQP (10%)
    t3 = t0 + budget_s * 0.50
    for alpha in [50000, 200000, 1000000]:
        if elapsed() > t3:
            break
        h_new = slsqp_softmax(h_work, n, alpha, maxiter=300)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new
        elif c5_new < compute_c5(h_work, n) - 1e-14:
            h_work = h_new

    # Phase 4: Targeted CD (20%)
    t4 = t0 + budget_s * 0.70
    if elapsed() < t4:
        h_cd, c5_cd = targeted_cd(best_h, n, t4, rng, 100000)
        if c5_cd < best_c5 - 1e-14:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Phase 5: Swap + Random CD (15%)
    t5 = t0 + budget_s * 0.85
    if elapsed() < t5:
        h_sw, c5_sw = swap_cd(best_h, n, min(t5, t0 + (t5 - t0) / 3), rng, 20000)
        if c5_sw < best_c5 - 1e-14:
            best_h = h_sw.copy()
            best_c5 = c5_sw
        h_r, c5_r = random_cd(best_h, n, t5, rng, 80000)
        if c5_r < best_c5 - 1e-14:
            best_h = h_r.copy()
            best_c5 = c5_r

    # Phase 6: Binary perturbation exploration (7%)
    t6 = t0 + budget_s * 0.92
    if elapsed() < t6:
        for sigma in [0.1, 0.2, 0.35]:
            if elapsed() > t6:
                break
            h_p = best_h + sigma * (rng.rand(n) - 0.5)
            h_p = repair(h_p, n)
            h_p = active_slsqp(h_p, n, 10, maxiter=200)
            h_p, c5_p = targeted_cd(h_p, n, t6 - 15, rng, 5000)
            if c5_p < best_c5 - 1e-14:
                best_h = h_p.copy()
                best_c5 = c5_p

    # Phase 7: Final reweighting + ultra-fine polish (8%)
    t_end = t0 + budget_s * 0.995
    if elapsed() < t_end:
        best_h, best_c5 = reweight_loop(best_h, n, t_end - 30,
                                         [0.0005, 0.0002, 0.0001], 2)
        h_fin, c5_fin = random_cd(best_h, n, t_end, rng, 50000)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin

    # Final repair and validation
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
