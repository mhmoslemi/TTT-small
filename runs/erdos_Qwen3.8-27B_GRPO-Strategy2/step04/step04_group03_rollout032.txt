                h_cur = h_new.copy()

    # Phase 2: Sharpen + re-optimize (push towards binary)
    if _tl() > 120:
        for pw in [2.5, 4.0, 6.0]:
            if _tl() < 50:
                break
            h_sh = _sharpen(best_h, best_n, pw)
            h_sh = _opt(h_sh, best_n, 100000, maxiter=400)
            c5_sh = _eval(h_sh)
            if c5_sh < best_c5:
                best_c5 = c5_sh
                best_h = h_sh.copy()

    # Phase 3: Try other resolutions
    if _tl() > 100:
        for n_try in [120, 140, 100, 180]:
            if _tl() < 40:
                break
            h_try = _repair(_interp(best_h, best_n, n_try), n_try)
            h_try = _opt(h_try, n_try, 50000, maxiter=300)
            h_try = _opt(h_try, n_try, 500000, maxiter=300)
            c5_try = _eval(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 4: Simulated annealing
    if _tl() > 50:
        t_end = time.time() + min(_tl() - 8, 30)
        h_a, c5_a = _anneal(best_h.copy(), best_n, rng, t_end)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 5: Block moves
    if _tl() > 35:
        t_end = time.time() + min(_tl() - 6, 20)
        h_b, c5_b = _block_search(best_h.copy(), best_n, rng, t_end)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: High-alpha SLSQP polish
    if _tl() > 30:
        h = _opt(best_h.copy(), best_n, 5000000, maxiter=600)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Fine local search
    if _tl() > 20:
        t_end = time.time() + min(_tl() - 4, 12)
        h_f, c5_f = _local_search(best_h.copy(), best_n, rng, t_end, init_scale=0.002)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Ultra-fine final touches
    if _tl() > 10:
        h = best_h.copy()
        c5c = _eval(h)
        for _it in range(200000):
            if time.time() > t0 + budget_s - 2:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-5 * rng.choice([1, 2, 5, 10, 20])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
