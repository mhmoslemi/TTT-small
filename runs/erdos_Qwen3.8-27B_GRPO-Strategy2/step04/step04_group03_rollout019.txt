    # Gradient of smooth max w.r.t. h
    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_max(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - h[i1-lag:j1-lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * h[i2+lag:j2+lag]
    return g

def _grad_single(h, k_idx):
    n = len(h)
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    i1, j1 = max(0, lag), min(n, n + lag)
    if i1 < j1:
        g[i1:j1] += (1.0 - h[i1-lag:j1-lag])
    i2, j2 = max(0, -lag), min(n, n - lag)
    if i2 < j2:
        g[i2:j2] -= h[i2+lag:j2+lag]
    return g

def _slsqp_opt(h, n, alpha, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_max, h, jac=_grad_smooth, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _gradient_search(h, n, rng, t_end):
    c5 = _eval(h)
    step = 0.015
    stall = 0
    for _ in range(100000):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(c)
        g = _grad_single(h, k_max)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g /= gn
        h_new = _repair(h - step * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new; c5 = c5_new; stall = 0
            step = min(step * 1.08, 0.06)
        else:
            stall += 1
            step *= 0.8
            if stall > 200:
                step = 0.01
                stall = 0
                if step < 1e-7:
                    break
    return h, c5

def _anneal_search(h, n, rng, t_end):
    c5 = _eval(h)
    T = 0.003
    T_min = 1e-7
    scale = 0.01
    for _ in range(200000):
        if time.time() > t_end:
            break
        T *= 0.9997
        if T < T_min:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_new = _eval(h)
        delta = c5_new - c5
        if delta < -1e-15:
            c5 = c5_new; scale = min(scale * 1.02, 0.04)
        else:
            if delta / max(T, 1e-12) < np.log(rng.random() + 1e-15):
                c5 = c5_new
            else:
                h[i], h[j] = ni - d, nj + d
            scale *= 0.995
            if scale < 3e-5:
                scale = 0.005
    return h, c5

def _fine_tune(h, n, rng, t_end):
    c5 = _eval(h)
    for _ in range(80000):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.choice([1, -1]) * rng.choice([0.3, 1, 3, 7]) * 1e-5
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5_new = _eval(h)
            if c5_new < c5 - 1e-17:
                c5 = c5_new
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5
