    """Newton's method for smooth max objective."""
    bds_lower = np.zeros(n)
    bds_upper = np.ones(n)
    
    for it in range(max_iter):
        c = full_corr(h)
        s = np.clip(a * (c - c.max()), -500, 500)
        ew = np.exp(s)
        p = ew / ew.sum()
        
        # Gradient
        g = np.zeros(n)
        for k in range(2*n-1):
            if p[k] > 1e-14:
                g += p[k] * grad_corr(h, k, n)
        g += 2*l*(h.sum() - n*0.5)
        
        # Check convergence
        if np.max(np.abs(g)) < 1e-12:
            break
        
        # Hessian (approximate: sum of active Hessians + outer product term)
        H = np.zeros((n, n))
        G_active = []
        for k in range(2*n-1):
            if p[k] > 1e-8:
                L = k - (n-1)
                H += p[k] * hess_for_L(L, n)
                G_active.append((p[k], grad_corr(h, k, n)))
        
        # Outer product term
        for pk, gk in G_active:
            for pl, gl in G_active:
                H += a * pk * pl * np.outer(gk, gl)

        H += 2*l * np.ones((n, n))

        # Solve H * d = -g with constraint 1^T d = 0
        # Use augmented system:
        # [H  1] [d]   [-g]
        # [1^T 0] [ν] = [ 0 ]
        
        n_aug = n + 1
        A = np.zeros((n_aug, n_aug))
        b = np.zeros(n_aug)
        A[:n, :n] = H

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def hess_for_L(L, n):
    H = np.zeros((n, n))
    if L == 0:
        np.fill_diagonal(H, -2.0)
    elif L > 0:
        for j in range(n - L):
            H[j, j] -= 1.0
            H[j+L, j+L] -= 1.0
            H[j, j+L] -= 1.0
            H[j+L, j] -= 1.0
    else:
        m = -L
        for j in range(n - m):
            H[j+m, j+m] -= 1.0
            H[j, j] -= 1.0
            H[j+m, j] -= 1.0
            H[j, j+m] -= 1.0
    return H

    c = full_corr(h)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    p = ew / ew.sum()
    obj = m + np.log(ew.sum()) / a + lam * (h.sum() - n * 0.5) ** 2
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] > 1e-14:
            g += p[k] * grad_corr(h, k, n)
    g /= a
    g += 2.0 * lam * (h.sum() - n * 0.5)
    return obj, g

    stages = [(10, 200), (50, 100), (200, 50), (800, 20), (3000, 8),
              (10000, 4), (50000, 2), (200000, 1), (1000000, 0.5)]
    for a, lam in stages:

        a, lam = float(a), float(lam)
        def f(x):
            return smooth_obj_jac(x, a, lam, n)[0]
        def j(x):
            return smooth_obj_jac(x, a, lam, n)[1]
        try:

            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:

        except Exception:
