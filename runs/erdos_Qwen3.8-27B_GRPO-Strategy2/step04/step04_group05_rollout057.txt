The spectral phase is interesting - it modifies the spectrum of h to reduce the max correlation. Let me add a simpler version.

The spectral idea: the correlation at all lags is related to the Fourier transform. Specifically, the autocorrelation of h is related to |F(h)|². By modifying the spectrum, we can shape the correlation.

But the objective is max_k correlate(h, 1-h)[k], which is not the same as autocorrelation. Let me think...

correlate(h, 1-h, mode='full')[k] = sum_i h[i+k-n+1] * (1-h[i])
= sum_i h[i+k-n+1] - sum_i h[i+k-n+1] * h[i]

The first term depends only on the sum of h (which is fixed). The second term is the cross-correlation of h with itself, which IS related to |F(h)|².

So maximizing the cross-correlation (to minimize 1-h cross-correlation) is what we want. The cross-correlation at all lags is the IFFT of |F(h)|².

So the spectral phase idea is: modify |F(h)| to make the IFFT more uniform (reducing the max). This is like a "flat spectrum" design.

            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            r0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(r0 / n))
            tm[0] = abs(H[0])
            nm = (1 - al) * mag + al * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            if c5v(hr, dx) < c5v(h, dx) - 1e-19:
                h = hr
                if c5v(h, dx) < bc:
                    bc, bh = c5v(h, dx), h.copy()
            else:
                break
    return bh, bc

def phase_dcd(h, n, dx, t0, lim, npass=4):
    bh, bc = h.copy(), c5v(h, dx)
    dts = np.array([3e-4, 8e-4, 2e-3, 5e-3, -3e-4, -8e-4, -2e-3, -5e-3])
    for _p in range(npass):
        if time.time() - t0 > lim:
            break
        for _sweep in range(6):
            any_imp = False
            if time.time() - t0 > lim:
                break
            for i in range(n):
                if time.time() - t0 > lim:
                    break
                base = c5v(h, dx)
                for j in range(n):
                    if i == j:
                        continue
                    bd, bc_ = 0.0, base
                    for d in dts:
                        ni_, nj_ = h[i] + d, h[j] - d
                        if 0 <= ni_ <= 1 and 0 <= nj_ <= 1:
                            h[i], h[j] = ni_, nj_
                            cv = c5v(h, dx)
                            h[i], h[j] = ni_ - d, nj_ + d
                            if cv < bc_ - 1e-18:
                                bc_, bd = cv, d
                    if bd != 0:
                        h[i] += bd
                        h[j] -= bd
                        h = repair(h, n)
                        any_imp = True
                        if c5v(h, dx) < bc:
                            bc, bh = c5v(h, dx), h.copy()
            if not any_imp:
                break
    return bh, bc

def phase_rand(h, n, dx, rng, t0, lim, ni=25000):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(ni):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.015
        ni_, nj_ = h[i] + d, h[j] - d
        if 0 <= ni_ <= 1 and 0 <= nj_ <= 1:
            h[i], h[j] = ni_, nj_
            c = c5v(h, dx)
            if c < cc - 1e-18:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni_ - d, nj_ + d
    return bh, bc

def full_opt(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = phase_lbfgs(bh, n, dx, t0, lim - 200)
    if c < bc:
        bh, bc = r, c
    r, c = phase_active(bh, n, dx, t0, lim - 130)
    if c < bc:
        bh, bc = r, c
    r, c = phase_spec(bh, n, dx, t0, lim - 90)
    if c < bc:
        bh, bc = r, c
    r, c = phase_lbfgs(bh, n, dx, t0, lim - 55)
    if c < bc:
        bh, bc = r, c
    r, c = phase_dcd(bh, n, dx, t0, lim - 12, npass=3)
    if c < bc:
        bh, bc = r, c
    r, c = phase_rand(bh, n, dx, rng, t0, lim - 2, 20000)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0
    r, c = full_opt(h0, n0, dx0, seed, t0, budget_s - 220)
    if c < bc:
        bh, bc, bn = r, c, n0
    for dm in [2, -2, 4, -4, -6, 6, 8, -8]:
        n2 = n0 + dm
        if n2 < 10 or n2 % 2 != 0 or time.time() - t0 > budget_s - 55:
            continue
        xo = np.linspace(0, 2, n0, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = full_opt(h2, n2, d2, seed + 300 + dm, t0, budget_s - 12)
        if c < bc:
            bh, bc, bn = r, c, n2
    if time.time() - t0 < budget_s - 25:
        rng = np.random.RandomState(seed + 999)
        hp = repair(bh + 0.012 * rng.randn(bn), bn)
        r, c = full_opt(hp, bn, 2.0 / bn, seed + 888, t0, budget_s - 3)
        if c < bc:
            bh, bc, bn = r, c, bn
    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
