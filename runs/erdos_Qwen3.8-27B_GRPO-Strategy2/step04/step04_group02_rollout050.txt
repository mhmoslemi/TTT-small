from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def subgrad_topk(h, n, k=3):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    topk = np.argsort(r)[-k:]
    grad = np.zeros(n)
    for m in topk:
        idx1 = np.arange(n) - m + n - 1
        v1 = (idx1 >= 0) & (idx1 < n)
        grad[v1] += (1.0 - h[idx1[v1]])
        idx2 = np.arange(n) + m - n + 1
        v2 = (idx2 >= 0) & (idx2 < n)
        grad[v2] -= h[idx2[v2]]
    grad *= dx / k
    val = float(np.max(r) * dx)
    return val, grad

def grad_descent(h, n, t_end):

    lr = 0.02

    stag = 0

        grad = grad - np.mean(grad)

        h = repair(h, n)

            vel *= 0.3
            stag = 0

            else:

        if not improved and scale < 0.003:
            break
    return h, best_c5

def upscale(h, n_new):
    n_old = len(h)
    ratio = n_new / n_old
    x_old = np.arange(n_old)

    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
