def compute_gradient(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exponents = alpha * (corr - m)
    exponents = np.clip(exponents, -500, 500)
    exps = np.exp(exponents)
    total = np.sum(exps)
    weights = exps / total  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    # L = k - (n-1) for k = 0..2n-2
    for k in range(len(corr)):
        wk = weights[k]
        if wk < 1e-12:
            continue
        L = k - (n - 1)
        if L >= 0:
            # First term: p from L to n-1, add wk*(1-h[p-L])
            if L < n:
                p = np.arange(L, n)
                grad[p] += wk * (1.0 - h[p - L])
            # Second term: p from 0 to n-L-1, subtract wk*h[p+L]
            if L < n:
                p = np.arange(0, n - L)
                grad[p] -= wk * h[p + L]
        else:
            M = -L  # M > 0
            # First term: p from 0 to n-M-1, add wk*(1-h[p+M])
            if M < n:
                p = np.arange(0, n - M)
                grad[p] += wk * (1.0 - h[p + M])
            # Second term: p from M to n-1, subtract wk*h[p-M]
            if M < n:
                p = np.arange(M, n)
                grad[p] -= wk * h[p - M]
    
    return grad

def adam_optimize(h, n, alpha, lr, n_steps, beta1=0.9, beta2=0.999, eps=1e-8, deadline=None):
    m = n // 2
    h = np.array(h, dtype=float)
    v1 = np.zeros(n)
    v2 = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/n)
    
    for step in range(n_steps):
        if deadline and time.time() >= deadline:
            break
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        grad = compute_gradient(h, n, alpha)
        
        # Adam update
        v1 = beta1 * v1 + (1 - beta1) * grad
        v2 = beta2 * v2 + (1 - beta2) * grad**2
        v1_hat = v1 / (1 - beta1**t)
        v2_hat = v2 / (1 - beta2**t)
        h_new = h - lr * v1_hat / (np.sqrt(v2_hat) + eps)
        
        # Project to constraints: 0<=h<=1, sum(h)=n/2
        h_new = project_to_feasible(h_new, m)
        
        c5 = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * (2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

            if L < n:
                p = np.arange(L, n)
                grad[p] += wk * (1.0 - h[p - L])
                p2 = np.arange(0, n - L)
                grad[p2] -= wk * h[p2 + L]
        else:
            M = -L
            if M < n:
                p = np.arange(0, n - M)
                grad[p] += wk * (1.0 - h[p + M])
                p2 = np.arange(M, n)
                grad[p2] -= wk * h[p2 - M]
    return grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def adam_step(h, n, alpha, lr, state, m):
    grad = compute_gradient(h, n, alpha)
    t = state['t'] + 1
    state['t'] = t
    state['v1'] = state['b1'] * state['v1'] + (1 - state['b1']) * grad
    state['v2'] = state['b2'] * state['v2'] + (1 - state['b2']) * grad * grad
    v1h = state['v1'] / (1.0 - state['b1']**t)
    v2h = state['v2'] / (1.0 - state['b2']**t)
    h_new = h - lr * v1h / (np.sqrt(v2h) + 1e-8)
    h_new = project_to_feasible(h_new, m)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(x, m):
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > m:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def exact_grad(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mval = np.max(corr)
    exponents = np.clip(alpha * (corr - mval), -500.0, 500.0)
    exps = np.exp(exponents)
    total = float(np.sum(exps))
    weights = exps / total
    grad = np.zeros(n)
    nn = n - 1
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-13:
            continue
        L = k - nn
        if L >= 0:
            if L < n:
                idx1 = np.arange(L, n)
                grad[idx1] += wk * (1.0 - h[idx1 - L])
                idx2 = np.arange(0, n - L)
                grad[idx2] -= wk * h[idx2 + L]
        else:
            M = -L
            if M < n:
                idx1 = np.arange(0, n - M)
                grad[idx1] += wk * (1.0 - h[idx1 + M])
                idx2 = np.arange(M, n)
                grad[idx2] -= wk * h[idx2 - M]
    return grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def adam_opt(h, n, alpha, lr, steps, deadline, m):
    h = np.array(h, dtype=float).copy()
    v1 = np.zeros(n)
    v2 = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / n)
    dx = 2.0 / n
    stale = 0
    for _ in range(steps):
        if time.time() >= deadline:
            break
        t += 1
        g = exact_grad(h, n, alpha)
        v1 = 0.9 * v1 + 0.1 * g
        v2 = 0.999 * v2 + 0.001 * g * g
        v1h = v1 / (1.0 - 0.9**t)
        v2h = v2 / (1.0 - 0.999**t)
        h_new = h - lr * v1h / (np.sqrt(v2h) + 1e-8)
        h_new = project_feasible(h_new, m)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        h = h_new
        if stale > 3000:
            break
    return best_h, best_c5

def slsqp_polish(h, n, alpha, maxiter, deadline):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

    def jac(x):
        return exact_grad(x, n, alpha)

    target = n / 2.0
    def con(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() < deadline:
            hc = repair(res.x, n)
            return hc, compute_c5(hc, 2.0 / n)
    except Exception:
        pass
    return h, compute_c5(h, 2.0 / n)

def local_search(h, n, dx, rng, deadline, max_iters, delay_buf=50):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    cur_c5 = best_c5
    buf = [cur_c5] * delay_buf
    bi = 0
    scales = [0.015, 0.006, 0.002, 0.0008, 0.0003, 0.0001]
    per_scale = max(500, max_iters // len(scales))
    for scale in scales:
        improved_any = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            ref = buf[bi % delay_buf]
            if c5n < ref - 1e-14:
                cur_c5 = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                improved_any = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
            buf[bi % delay_buf] = c5n
            bi += 1
        if not improved_any and scale < 0.003:
            break
    return h, best_c5

def gradient_ls(h, n, dx, alpha, rng, deadline, max_iters):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        g = exact_grad(h, n, alpha)
        sg = rng.choice(n, 20, replace=False)
        i = sg[np.argmax(g[sg])]
        j = sg[np.argmin(g[sg])]
        if i == j or abs(g[i] - g[j]) < 1e-10:
            continue
        d = 0.003 * (0.5 + 0.5 * rng.random())
        ni, nj = h[i] - d, h[j] + d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            d *= 0.3
            ni, nj = h[i] - d, h[j] + d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
        h[i], h[j] = ni, nj
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-14:
            best_c5 = c5n
        else:
            h[i], h[j] = ni + d, nj - d
    return h, best_c5

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(1, n))
    elif kind == 1:
        f = rng.randint(2, 10)
        ph = rng.uniform(0, 6.28)
        hp += rng.uniform(0.03, 0.12) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.04, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 6), rng.randint(2, 9)
        hp += rng.uniform(0.03, 0.1) * np.sin(2*np.pi*f1*np.arange(n)/n) * np.cos(2*np.pi*f2*np.arange(n)/n)
    else:
        idx = rng.choice(n, n//4, replace=False)
        hp[idx] += rng.normal(0, 0.06, len(idx))
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 140
    m = n // 2
    dx = 2.0 / n

    h = np.array(initial_h_values, dtype=float).copy()
    h = repair(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Adam annealing (low to high alpha), ~250s
    dl = t0 + 250
    for alpha in [30, 80, 200, 500, 1000, 3000]:
        if rem() < 200 or time.time() >= dl:
            break
        for lr in [0.4, 0.15, 0.05]:
            if time.time() >= dl:
                break
            h, c5 = adam_opt(h, n, alpha, lr, 4000, dl, m)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 2: SLSQP with exact gradient, ~150s
    dl = t0 + 400
    for alpha in [500, 2000, 8000, 30000]:
        if time.time() >= dl or rem() < 180:
            break
        hh, cc = slsqp_polish(best_h, n, alpha, 500, t0 + budget_s)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 3: Adam high-alpha, ~150s
    dl = t0 + 550
    for alpha in [5000, 15000, 50000]:
        if time.time() >= dl or rem() < 150:
            break
        for lr in [0.08, 0.03, 0.01]:
            if time.time() >= dl:
                break
            h, c5 = adam_opt(best_h, n, alpha, lr, 3000, dl, m)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 4: Multi-restart, ~200s
    dl = t0 + 750
    for r in range(25):
        if time.time() >= dl or rem() < 100:
            break
        hp = perturb(best_h, n, rng, r % 5)
        h, c5 = adam_opt(hp, n, 800, 0.15, 2000, dl, m)
        h, c5 = adam_opt(h, n, 5000, 0.03, 2000, dl, m)
        hh, cc = slsqp_polish(h, n, 20000, 300, dl)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 5: Gradient-guided local search, ~80s
    if rem() > 100:
        dl = t0 + budget_s - 80
        best_h, best_c5 = gradient_ls(best_h, n, dx, 10000, rng, dl, 30000)

    # Phase 6: Delayed-acceptance local search, ~70s
    if rem() > 80:
        dl = t0 + budget_s - 40
        best_h, best_c5 = local_search(best_h, n, dx, rng, dl, 80000, delay_buf=80)

    # Phase 7: Ultra-fine Adam + final local, ~35s
    if rem() > 45:
        dl = t0 + budget_s - 15
        for alpha in [100000, 500000]:
            if time.time() >= dl:
                break
            h, c5 = adam_opt(best_h, n, alpha, 0.005, 2000, dl, m)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
    if rem() > 12:
        dl = t0 + budget_s - 2
        best_h, best_c5 = local_search(best_h, n, dx, rng, dl, 20000, delay_buf=30)

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
