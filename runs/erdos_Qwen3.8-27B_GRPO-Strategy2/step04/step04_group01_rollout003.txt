import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    if abs(h.sum() - t) > 1e-10:
        h = _proj_sum(h, t)
    d = t - h.sum()
    if abs(d) > 1e-12:
        i = int(np.argmax(1.0 - h)) if d > 0 else int(np.argmin(h))
        h[i] = np.clip(h[i] + d, 0, 1)
    return h

def _smo_obj(h, alpha):
    cv = _corr(h)
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smo_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _topk_grad(h, topk):
    """Gradient of sum of top-k correlation values (subgradient)."""
    n = len(h)
    cv = _corr(h)
    idx = np.argsort(cv)[-topk:]
    w = np.zeros(len(cv))
    w[idx] = 1.0 / topk
    return _smo_grad_from_w(h, w)

def _smo_grad_from_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lbfgs_penalty(h, mu, alpha, maxiter=5000):
    n = len(h)
    def obj(x):
        cv = _corr(x)
        m = cv.max()
        s = np.clip(alpha * (cv - m), -500, 500)
        f = m + np.log(np.sum(np.exp(s))) / alpha
        f += mu * (x.sum() - n * 0.5) ** 2
        return f
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return _repair(res.x, n)

def _pgd(h, alpha, nsteps, step0, topk=5):
    """Projected gradient with adaptive top-k focus."""
    n = len(h)
    f = _smo_obj(h, alpha)
    step = step0
    for it in range(nsteps):
        if it % 50 == 0:
            tk = min(topk, max(2, topk - it // 100))
        else:
            tk = topk
        if tk <= 5:
            g = _topk_grad(h, tk)
        else:
            g = _smo_grad(h, alpha)
        improved = False
        for _bt in range(50):
            h_new = _proj_sum(h - step * g, n * 0.5)
            fn = _eval(h_new)
            if fn < f:
                h = h_new
                f = fn
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-10:
                break
        else:
            step = min(step * 1.1, step0)
    return h

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(_smo_obj, h, jac=_smo_grad, args=(alpha,),
                   method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return _repair(res.x, n)

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _local_search(h, n_iter, scale):
    n = len(h)
    c5c = _eval(h)
    rng = np.random.RandomState(42 + n)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _interpolate(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.clip(np.interp(x_new, x_old, h), 0, 1)

    t0 = time.time()

    def _el():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)

    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS-B with increasing penalty (fast)
    for mu, alpha, mi in [(10, 500, 3000), (100, 2000, 3000), (1000, 10000, 2000)]:
        if _el() > 120:
            break
        h = _lbfgs_penalty(best_h, mu, alpha, mi)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n0

    # Phase 2: SLSQP at original resolution
    for alpha in [5000, 50000, 200000, 1000000]:
        if _el() > 250:
            break
        h = _slsqp_opt(best_h, alpha, 2000, 1e-18)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n0

    # Phase 3: Projected gradient at original resolution
    if _el() < budget_s - 300:
        h = _pgd(best_h, 50000, 200, 0.005, topk=5)
        h = _repair(h, n0)
        c5 = _eval(h)
        if c5 < best_c5:

            best_n = n0
