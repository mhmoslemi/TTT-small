    target = n * 0.5
    s = h.sum()

    h *= target / s

                         bounds=bds, options={'maxiter': mi, 'ftol': 1e-20, 'gtol': 1e-18})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def slsqp_opt(h, n, dx, t0, lim):
    _G['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    for a in [2000, 10000, 50000]:
        if time.time() - t0 > lim:
            break
        _G['a'], _G['l'] = float(a), 0.0
        try:
            r = minimize(sm_obj, bh, jac=sm_jac, method='SLSQP', bounds=bds,
                         constraints=[{'type': 'eq', 'fun': se_fun, 'jac': se_jac}],
                         options={'maxiter': 400, 'ftol': 1e-20})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def active_phase(h, n, dx, t0, lim):
    """Minimize average of top-m correlation values."""
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [2, 3, 4, 6]:
        if time.time() - t0 > lim:
            break
        h2 = bh.copy()
        # Build objective: average of top-m c values (smoothly approximated)
        for a in [500, 5000, 50000]:
            if time.time() - t0 > lim:
                break
            _G['a'], _G['l'] = float(a), 50.0
            try:
                r = minimize(sm_obj, h2, jac=sm_jac, method='L-BFGS-B',
                             bounds=[(0.0,1.0)]*n, options={'maxiter': 200, 'ftol': 1e-20})
                hc = repair(r.x, n)
                c = c5v(hc, dx)
                if c < bc:
                    bc, bh = c, hc
                h2 = bh.copy()
            except Exception:
                pass
    return bh, bc

def de_global(n, h_init, dx, seed, t0, lim, popsize=25, maxiter=100):
    """Differential evolution global search."""
    def de_obj(x):
        hr = repair(x, n)
        return c5v(hr, dx)
    
    bds = [(0.0, 1.0)] * n
    rng = np.random.RandomState(seed)
    # Initial population: include h_init and random perturbations
    try:
        result = differential_evolution(
            de_obj, bds, seed=seed, popsize=popsize, maxiter=maxiter,
            tol=1e-12, mutation=(0.3, 1.0), recombination=0.8,
            init='halton', polish=False, updating='deferred',
            workers=1
        )
        hc = repair(result.x, n)
        return hc, c5v(hc, dx)
    except Exception:
        return h_init, c5v(h_init, dx)

def polish(h, n, dx, rng, cnt, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.002
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc - 1e-15:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def full_opt(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 300)
    if c < bc: bh, bc = r, c
    
    r, c = slsqp_opt(bh, n, dx, t0, lim - 200)
    if c < bc: bh, bc = r, c
    
    r, c = active_phase(bh, n, dx, t0, lim - 150)
    if c < bc: bh, bc = r, c
    
    r, c = lbfgs_opt(bh, n, dx, t0, lim - 100)
    if c < bc: bh, bc = r, c
    
    r, c = polish(bh, n, dx, rng, 30000, t0, lim - 10)
    if c < bc: bh, bc = r, c
    
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    # Start from initial_h_values
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    bh, bc, bn = h0.copy(), c5v(h0, dx0), n0
    
    # Main optimization at original resolution
    remain = budget_s - 60
    if time.time() - t0 < t0 + remain:
        r, c = full_opt(h0, n0, dx0, seed, t0, t0 + remain)
        if c < bc:
            bh, bc, bn = r, c, n0
    
    try:
        # Try upsampling to n=200
        n2 = 200
        if n2 != n0 and time.time() - t0 < budget_s - 200:
            xo = np.linspace(0, 2, n0, endpoint=False)
            xn = np.linspace(0, 2, n2, endpoint=False)
            h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
            dx2 = 2.0 / n2
            r, c = full_opt(h2, n2, dx2, seed+1, t0, budget_s - 80)
            if c < bc:
                bh, bc, bn = r, c, n2
    except Exception:
        pass
    
    try:
        # Try upsampling to n=280
        n3 = 280
        if n3 > n0 and time.time() - t0 < budget_s - 150:
            xo = np.linspace(0, 2, bn, endpoint=False)
            xn = np.linspace(0, 2, n3, endpoint=False)
            h3 = repair(np.interp(xn, xo, bh, period=2.0), n3)
            dx3 = 2.0 / n3
            r, c = full_opt(h3, n3, dx3, seed+2, t0, budget_s - 30)
            if c < bc:
                bh, bc, bn = r, c, n3
    except Exception:
        pass

                bh, bc = r, c
    except Exception:
        pass
    
    # Final polish
