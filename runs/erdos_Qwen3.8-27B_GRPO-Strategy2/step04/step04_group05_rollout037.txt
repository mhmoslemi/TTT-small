def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-14:
        return h
    # Proportional scaling + clip + redistribute
    if s > 0:
        h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_c(h, k, n):
    """Gradient of c_k w.r.t. h, where k is the full-correlate index."""
    g = np.zeros(n)
    L = k - (n - 1)  # lag
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def project(z, n):
    """Project z onto {x: 0<=x<=1, sum(x)=n/2}."""

        s = np.clip(z - mid, 0.0, 1.0).sum()

This is O(n_trials * O(n²)) for the correlate computation. For n=140, n²=19600, and 50000 trials gives ~10^9 operations. That might take 5-10 seconds. OK.

But wait, I can speed this up! When I change h[i] by d and h[j] by -d, the change in c_k is:
Δc_k = d * (∂c_k/∂h[i] - ∂c_k/∂h[j])

And ∂c_k/∂h[i] is mostly 0 (only nonzero for O(1) lags involving i). Specifically:
- ∂c_k/∂h[i] is nonzero only when the lag L satisfies: i-L is valid (i.e., the pair (i, i-L) is in the correlation) or i+L is valid.

For a given i, the lags where ∂c_k/∂h[i] ≠ 0 are:
- L = 0: always
- L = i-m for m ∈ [0, n-1] (h[i] as the "shifted" value): L ranges from i-(n-1) to i
- L = m-i for m ∈ [0, n-1] (h[i] as the "base" value): L ranges from -(i) to (n-1-i)

So there are O(n) lags affected by changing h[i]. For two indices, it's O(n) total lags.

But computing the full correlate is O(n²) and for n=140, it's fast enough. Let me just use the full correlate.

def project(z, n):
    target = n * 0.5
    lo, hi = float(z.min()) - 1.1, float(z.max()) + 1.1
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def sg(h, n, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    s = np.clip(a * (c - c.max()), -500, 500)
    ew = np.exp(s)
    w = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue

        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n-L])
            g[:n-L] -= wk * h[L:]
        else:
            m = -L
            g[:n-m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n-m]
    return g
