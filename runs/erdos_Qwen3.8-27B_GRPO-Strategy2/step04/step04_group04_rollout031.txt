def _c5v(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(_c5v(h).max() * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-16:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _interp(h, nn):
    return np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, len(h)), h)

def _grad_at(h, pos, s_arr):
    n = len(h)
    idx_p = pos + s_arr
    vm_p = (idx_p >= 0) & (idx_p < n)
    idx_m = pos - s_arr
    vm_m = (idx_m >= 0) & (idx_m < n)
    gp = np.where(vm_p, 1.0 - h[np.clip(idx_p, 0, n - 1)], 0.0)
    gm = np.where(vm_m, -h[np.clip(idx_m, 0, n - 1)], 0.0)
    return gp + gm

def _opt_pair_d(c, h, j, k, s_arr):
    n = len(h)
    dm = max(-h[j], h[k] - 1.0)
    dxp = min(1.0 - h[j], h[k])
    if dxp - dm < 1e-15:
        return 0.0
    g = _grad_at(h, j, s_arr) - _grad_at(h, k, s_arr)
    lo, hi = dm, dxp

    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n

        else:
