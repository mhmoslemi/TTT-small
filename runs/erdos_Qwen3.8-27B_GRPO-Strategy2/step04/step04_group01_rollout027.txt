            h_new = _proj_sum(h - step * g, n * 0.5)
            fn = _sm_val(h_new, alpha)
            if fn < f - 1e-10:
                h = h_new
                f = fn
                improved = True
                break
            step *= 0.5
        if not improved:
            break
        step = min(step * 1.1, step0)
    return h

def _topk_descent(h, t0, tlim, K=8):
    n = len(h)
    c5c = _eval(h)
    dx = 2.0 / n
    for _ in range(2000):
        if time.time() - t0 > tlim:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-K:]
        g = np.zeros(n)
        for k in idx:
            sh = k - (n - 1)
            if sh >= 0:
                nn = n - sh
                gg = np.zeros(n)
                gg[:nn] += (1.0 - h[sh:sh + nn])
                if sh > 0:
                    gg[sh:] -= h[:nn]
            else:
                sp = -sh
                nn = n - sp
                gg[sp:] += (1.0 - h[:nn])
                gg[:nn] -= h[sp:]
            g += gg
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        done = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                done = True
                break
        if not done:
            break
    return h, c5c

def _fine_pairwise(h, scale, niter, t0, tlim):
    n = len(h)
    rng = np.random.RandomState(12345)
    c5c = _eval(h)
    for _ in range(niter):
        if time.time() - t0 > tlim:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP sweep
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() - t0 > 120:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Projected gradient refinement
    for alpha in [50000, 100000, 500000]:
        if time.time() - t0 > 350:
            break
        h_new = _proj_grad(best_h, alpha, 8000, 0.005, t0, 350)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Top-K focused descent
    if time.time() - t0 < budget_s - 300:
        h_new, c5 = _topk_descent(best_h, t0, budget_s - 300, K=8)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 4: Multi-start with SLSQP
    if time.time() - t0 < budget_s - 250:
        for _r in range(20):
            if time.time() - t0 > budget_s - 250:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.015 * rng.randn(n0), n0)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(4, max(5, n0 // 8))
                    st = rng.randint(0, n0 - bs)
                    hp[st:st + bs] += 0.05 * rng.randn(bs)
                hp = _repair(hp, n0)
            elif strat == 3:
                hp = best_h.copy()
                freqs = rng.randint(1, n0 // 2 + 1, size=4)
                for f in freqs:
                    hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * np.arange(n0) / n0)
                hp = _repair(hp, n0)
            else:
                shift = rng.randint(1, n0 // 3)
                hp = _repair(np.roll(best_h, shift), n0)
                hp = _repair(hp + 0.01 * rng.randn(n0), n0)
            h_new = _slsqp(hp, 100000, 1200, 1e-19)
            h_new = _proj_grad(h_new, 500000, 3000, 0.003, t0, budget_s - 250)
            h_new = _repair(h_new, n0)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 5: Final Top-K descent
    if time.time() - t0 < budget_s - 100:
        h_new, c5 = _topk_descent(best_h, t0, budget_s - 100, K=12)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 6: SLSQP polish
    if time.time() - t0 < budget_s - 60:
        h_new = _slsqp(best_h, 1000000, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 7: Fine pairwise search (decreasing scales)
    if time.time() - t0 < budget_s - 30:
        for sc in [0.004, 0.001, 0.0003, 0.00005]:
            if time.time() - t0 > budget_s - 30:
                break
            h_new, c5 = _fine_pairwise(best_h.copy(), sc, 6000, t0, budget_s - 30)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 8: Ultra-fine touch
    if time.time() - t0 < budget_s - 5:
        h_new, c5 = _fine_pairwise(best_h.copy(), 1e-5, 20000, t0, budget_s - 5)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
