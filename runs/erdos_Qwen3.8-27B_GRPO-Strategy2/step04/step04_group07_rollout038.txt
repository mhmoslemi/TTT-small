import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(3000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def all_corrs(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def active_grad(h, n, corr, eps):
    cmax = np.max(corr)
    active = np.where(corr >= cmax - eps)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = k - (n - 1)
        if sh == 0:
            g += dx * (1.0 - 2.0 * h)
        elif sh > 0:
            g[0:n-sh] += dx * (1.0 - h[sh:])
            g[sh:] -= dx * h[0:n-sh]
        else:
            l = -sh
            g[l:] += dx * (1.0 - h[0:n-l])
            g[0:n-l] -= dx * h[l:]
    return g / len(active), cmax

def grad_descent(h, n, t_end, mom, lr0, lr_dec, base_eps):
    h = h.copy().astype(float)
    best_c5 = np.inf
    best_h = h.copy()
    vel = np.zeros(n)
    lr = lr0
    stall = 0
    while time.time() < t_end:
        corr = all_corrs(h, n)
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        eps = max(base_eps, 0.1 * c5)
        g, _ = active_grad(h, n, corr, eps)
        vel = mom * vel - lr * g
        h_new = np.clip(h + vel, 0.0, 1.0)
        h_new = repair(h_new, n)
        corr_new = all_corrs(h_new, n)
        c5_new = float(np.max(corr_new))
        if c5_new <= c5 + 1e-12:
            h = h_new
            lr *= 1.005
        else:
            lr *= lr_dec
            vel *= 0.3
        if lr < 5e-10 or stall > 5000:
            break
    return best_h, best_c5

def slsqp_polish(h, n, t_end):
    dx = 2.0 / n
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        m = np.max(corr)
        a = 50000
        s = np.clip(a * (corr - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / a
    def objg(v):
        corr = all_corrs(v, n)
        cmax = np.max(corr)
        ep = max(0.05 * cmax, 1e-6)
        g, _ = active_grad(v, n, corr, ep)
        return g
    def cons_fn(v):
        return v.sum() - n / 2.0
    def cons_jac(v):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def structured_perturb(h, n, rng):
    sp = h.copy()
    half = n // 2
    for _ in range(rng.randint(3, 10)):
        i = rng.randint(half)
        j = (i + half) % n
        d = rng.uniform(-0.3, 0.3)
        sp[i] += d
        sp[j] -= d
    return repair(sp, n)
