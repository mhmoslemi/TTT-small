    n = len(h)
    s = h.sum()

        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def lse_obj(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs(h, n, alpha, target, pen, nit):
    dx = 2.0 / n
    try:
        r = minimize(lse_obj, h, args=(alpha, target, pen), method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': nit, 'ftol': 1e-15, 'gtol': 1e-7})
        hc = repair(r.x, target)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

            if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9: continue
            h[i], h[j] = ni, nj
            c5n = compute_c5(h, dx)
            if c5n - cc5 <= 0 or rng.random() < np.exp(-min(500.0, (c5n-cc5)/T)):
                cc5 = c5n
                if c5n < bc5 - 1e-15: bc5, best = c5n, h.copy()
            else:

        hp = np.roll(hp, rng.randint(n // 4, 3 * n // 4))
    elif k == 1:
        f, ph = rng.randint(1, 5), rng.uniform(0, 6.28)
        hp += rng.uniform(0.02, 0.08) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif k == 2:
        hp += rng.normal(0, 0.03, n)
    else:
        f1, f2 = rng.randint(1, 4), rng.randint(1, 5)
        t = np.arange(n) / n
        hp += rng.uniform(0.02, 0.06) * np.sin(2 * np.pi * f1 * t) * np.sin(2 * np.pi * f2 * t + rng.uniform(0, 6.28))
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); rng = np.random.RandomState(seed)
    h0 = repair(np.array(initial_h_values, dtype=float), len(initial_h_values) / 2.0)
    n = len(h0); dx = 2.0 / n; tgt = n / 2.0
    h = h0.copy(); bc5 = compute_c5(h, dx); bh = h.copy()

    # P1: L-BFGS warm-up
    for a in [500, 2000]:
        if time.time() > t0 + 25: break
        hh, cc = lbfgs(h, n, a, tgt, 3000, 250)
        if cc < bc5: h, bc5, bh = hh, cc, hh.copy()

    # P2: Main simulated annealing
    h, bc5 = do_anneal(h, n, dx, tgt, rng, t0 + 25, t0 + 505, 0.003)
    bh = h.copy()

    # P3: L-BFGS refinement
    if time.time() < t0 + 535:
        for a in [2000, 10000]:
            if time.time() > t0 + 535: break
            hh, cc = lbfgs(h, n, a, tgt, 5000, 300)
            if cc < bc5: h, bc5, bh = hh, cc, hh.copy()

    # P4: Diversification
    for _ in range(4):
        if time.time() > t0 + 595: break
        hp = perturb(h, n, rng)
        hh, cc = lbfgs(hp, n, 3000, tgt, 5000, 300)
        if cc < bc5: h, bc5, bh = hh, cc, hh.copy()

    # P5: Second anneal at lower temperature
    h, bc5 = do_anneal(h, n, dx, tgt, rng, t0 + 595, t0 + 795, 0.0008)
    bh = h.copy()

    # P6: Final L-BFGS
    if time.time() < t0 + 820:
        hh, cc = lbfgs(h, n, 20000, tgt, 8000, 400)
        if cc < bc5: h, bc5, bh = hh, cc, hh.copy()

    # P7: Fine local search with auto-advancing scales
    h, bc5 = fine_ls(h, n, dx, tgt, rng, t0 + budget_s - 5,
                     [0.004, 0.0015, 0.0006, 0.0002, 8e-5, 3e-5, 1e-5])
    bh = h.copy()

    bh = repair(bh, tgt)
    return (bh, compute_c5(bh, dx), n)
