import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(50000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_corr_k(h, n, k):
    shift = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n, n - shift)
    if i_lo >= i_hi:
        return grad
    i = np.arange(i_lo, i_hi)
    grad[i] += (1.0 - h[i + shift])
    grad[i + shift] -= h[i]
    return grad

def grad_smoothed(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exponents = np.exp(np.clip(alpha * (corr - m), -500, 0))
    total = np.sum(exponents)
    if total < 1e-300:
        grad = grad_corr_k(h, n, np.argmax(corr))
        return grad
    weights = exponents / total
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-6:
            grad += weights[k] * grad_corr_k(h, n, k)
    return grad

def subgrad_line(h, n, t_end, n_max_iter=50000):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    alpha = 200.0
    no_improve = 0
    for t in range(1, n_max_iter + 1):
        if time.time() > t_end:
            break
        step = 0.08 / (1.0 + 0.005 * t)
        grad = grad_smoothed(h, n, alpha)
        improved = False
        for sf in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.005]:
            h_try = h - step * sf * grad
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = repair(h_try, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                best_h = h_try.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 200:
                alpha = max(alpha * 1.3, 1e6)
                if no_improve > 2000:
                    break
        if t % 500 == 0:
            alpha *= 1.1
    return best_h, best_c5

def coord_descent(h, n, t_end, rng, n_max_rounds=5000):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    stagnant = 0
    for r in range(n_max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(5 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            hid, hjd = h[i], h[j]
            found = False
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hid + delta, hjd - delta
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        found = True
                        break
                    h[i], h[j] = hid, hjd
                if found:
                    break
        if improved:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 50:
            break
    return h, best

def slsqp_opt(h, n, alpha, maxiter=500):
    def soft_obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(soft_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(res.x, n)
    except Exception:
        return h

def make_init(n, typ, rng):
    if typ == 0:
        h = np.zeros(n)
        blk = max(2, n // 8)
        for i in range(0, n, 2 * blk):
            h[i:i + blk] = 1.0
        return repair(h, n)
    if typ == 1:
        x = np.linspace(0, 2 * np.pi, n)
        h = 0.5 + 0.35 * np.sin(x + rng.uniform(0, 6.28)) + 0.1 * np.sin(3 * x + rng.uniform(0, 6.28))
        return repair(h, n)
    if typ == 2:
        h = np.zeros(n)
        for _ in range(rng.randint(2, 5)):
            c = rng.randint(n)
            w = rng.uniform(1.5, n / 4)
            idx = np.arange(n)
            h += np.exp(-0.5 * ((idx - c) / w) ** 2)
        if h.max() > 0:
            h /= h.max()
        return repair(h, n)
    if typ == 3:
        h = rng.uniform(0.1, 0.9, n)
        nk = max(1, n // 8)
        h = np.convolve(h, np.ones(nk) / nk, 'same')
        return repair(h, n)
    return repair(rng.uniform(0, 1, n), n)
