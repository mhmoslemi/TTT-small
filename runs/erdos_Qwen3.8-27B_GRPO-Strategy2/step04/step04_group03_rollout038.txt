import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / a

def _soft_grad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 0)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _cf(x, n):
    return np.sum(x) - n * 0.5

def _cj(x, n):
    return np.ones(n)

def _slsqp(h, n, a, mi):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        r = minimize(_soft_obj, h, jac=_soft_grad, args=(a,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-25})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _interp_h(src, ns, nd):
    return np.interp(np.linspace(0, 2, nd, False),
                     np.linspace(0, 2, ns, False), src)

def _sim_anneal(h, n, rng, te):
    c5c = _eval(h)
    tot = int(max(500, (te - time.time()) * 900))
    Ti, Tf = 0.005, 1e-7
    for it in range(tot):
        if time.time() > te:
            break
        T = Ti * (Tf / Ti) ** (it / tot)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.008 * rng.randn()
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1:
            continue
        h[i], h[j] = a, b
        v = _eval(h)
        if v < c5c or rng.rand() < np.exp(-(v - c5c) / max(T, 1e-15)):
            c5c = v
        else:
            h[i], h[j] = a - d, b + d
    return h, c5c

def _fine_tune(h, n, rng, te, sc0, mx):
    c5c = _eval(h)
    sc, no = sc0, 0
    for _ in range(mx):
        if time.time() > te:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0 <= a <= 1 and 0 <= b <= 1:
            h[i], h[j] = a, b
            v = _eval(h)
            if v < c5c - 1e-17:
                c5c, no = v, 0
                sc = min(sc * 1.015, 0.005)
            else:
                h[i], h[j] = a - d, b + d
                no += 1
                if no > 300:
                    sc *= 0.93
                    no = 0
                    if sc < 1e-10:
                        break
    return h, c5c

def _time_left(t0, bs):
    return bs - (time.time() - t0)
