    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        M = abs(L)
        if M >= n:
            return g
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g

    target = n / 2.0

                diff -= a

                diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _corr_grad(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        M = abs(L)
        if M >= n:
            return g
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _top_k_grad(h, K, n):
    c = _corr(h)
    K = min(K, len(c))
    top = np.argsort(c)[-K:]
    thresh = c[top[0]]
    active = np.where(c >= thresh - 1e-9)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        g += _corr_grad(h, L, n)
    g /= len(active)
    g -= g.mean()
    return g

def _nesterov_opt(h, n, dx, K0, Kmax, niter, tleft_fn):
    hc = h.copy()
    hp = h.copy()
    c5c = _c5(hc, dx)
    K = K0
    lr = 0.03
    mu = 0.5
    for it in range(niter):
        if tleft_fn() < 25:
            break
        hy = np.clip(hc + mu * (hc - hp), 0, 1)
        hy = _repair(hy, n)
        g = _top_k_grad(hy, K, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        found = False
        for s in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            hnew = _repair(hc - s * g, n)
            c5n = _c5(hnew, dx)
            if c5n < c5c - 1e-15:
                hp = hc.copy()
                hc = hnew
                c5c = c5n
                found = True
                mu = min(mu + 0.02, 0.92)
                break
        if not found:
            lr *= 0.75
            mu *= 0.9
            if lr < 1e-10:
                break
        if (it + 1) % 40 == 0:
            K = min(K * 2, Kmax)
            lr = min(lr * 1.15, 0.04)
    return hc, c5c

def _local_search(h, n, dx, tleft_fn, margin):
    hc = h.copy()
    c5c = _c5(hc, dx)
    dlist = np.array([0.001, 0.004, 0.01, 0.02, 0.04,
                      -0.001, -0.004, -0.01, -0.02, -0.04])
    cvals = _corr(hc)
    top_lags = np.argsort(cvals)[-15:]
    imp = np.zeros(n)
    for k in top_lags:
        imp += np.abs(_corr_grad(hc, k - (n - 1), n))
    for sweep in range(500):
        if tleft_fn() < margin:
            break
        if sweep % 25 == 0:
            cvals = _corr(hc)
            top_lags = np.argsort(cvals)[-15:]
            imp = np.zeros(n)
            for k in top_lags:
                imp += np.abs(_corr_grad(hc, k - (n - 1), n))
        order = np.argsort(imp)[::-1][:40]
        improved = False
        for ii in range(len(order)):
            if tleft_fn() < margin:
                break
            i = order[ii]
            for jj in range(ii + 1, min(ii + 12, len(order))):
                j = order[jj]
                for d in dlist:
                    ni, nj = hc[i] + d, hc[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        hc[i], hc[j] = ni, nj
                        c = _c5(hc, dx)
                        if c < c5c:
                            c5c = c
                            improved = True
                            break
                        hc[i], hc[j] = ni - d, nj + d
                if improved:
                    hc = _repair(hc, n)
                    c5c = _c5(hc, dx)
                    break
        if not improved:
            break
    return hc, c5c

def _thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        b = i
        cnt = 0
        while b > 0:
            cnt += b & 1
            b >>= 1
        tm[i] = 0.5 if cnt % 2 == 0 else 0.5
    arr = np.zeros(n)
    for i in range(n):
        b = i
        cnt = 0
        while b > 0:
            cnt += b & 1
            b >>= 1
        arr[i] = 1.0 if cnt % 2 == 0 else 0.0
    return _repair(arr, n)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def tleft():
        return budget_s - elapsed()

    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)

    # Phase 1: L-BFGS-B soft-max (quick)
    bounds = [(0.0, 1.0)] * n
    for alpha in [3000, 15000, 60000]:
        if tleft() < 400:
            break
        def obj(x, al=alpha):
            c = _corr(x)
            m = c.max()
            s = np.clip(al * (c - m), -400, 400)
            s -= s.max()
            return m + np.log(np.sum(np.exp(s))) / al + 50.0 * (np.sum(x) - n / 2.0) ** 2

        def jacobian(x, al=alpha):
            c = _corr(x)
            m = c.max()
            s = np.clip(al * (c - m), -400, 400)
            s -= s.max()
            ew = np.exp(s)
            pw = ew / ew.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                pk = pw[k]
                if pk < 1e-13:
                    continue
                g += pk * _corr_grad(x, k - (n - 1), n)
            g *= 1.0 / al
            g += 100.0 * (np.sum(x) - n / 2.0)
            return g

        try:
            res = minimize(obj, best_h.copy(), jac=jacobian, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 300, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            cv = _c5(hc, dx)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Nesterov active-set gradient
    h2, c5_2 = _nesterov_opt(best_h, n, dx, 5, 300, 600, lambda: tleft() - 280)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Guided local search
    h3, c5_3 = _local_search(best_h, n, dx, tleft, 200)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Try structured starting points at same n
    if tleft() > 200:
        h_tm = _thue_morse(n)
        h_tm, c5_tm = _nesterov_opt(h_tm, n, dx, 5, 200, 300, lambda: tleft() - 100)
        if c5_tm < best_c5:
            best_h, best_c5 = h_tm, c5_tm

    # Phase 5: Try smaller n
    for n_try in [100, 70]:
        if tleft() < 100:
            break
        dx_try = 2.0 / n_try
        h_try = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                          np.linspace(0, 2, len(best_h), endpoint=False), best_h)
        h_try = _repair(h_try, n_try)
        h_try, c5_try = _nesterov_opt(h_try, n_try, dx_try, 5, 150, 300,
                                       lambda: tleft() - 50)
        h_try, c5_try = _local_search(h_try, n_try, dx_try, tleft, 40)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 6: Final micro-polish
    h6 = best_h.copy()
    c5_6 = best_c5
    n6 = len(best_h)
    dx6 = 2.0 / n6
    tiny = np.array([0.0003, 0.0008, 0.002, 0.005, -0.0003, -0.0008, -0.002, -0.005])
    max_steps = int(max(0, tleft() - 3) * 2500)
    for _ in range(max_steps):
        if tleft() < 2:
            break
        i = rng.randint(n6)
        j = rng.randint(n6)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h6[i] + d, h6[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h6[i], h6[j] = ni, nj
            c = _c5(h6, dx6)
            if c < c5_6:
                c5_6 = c
                best_c5 = c
                best_h = h6.copy()
            else:
                h6[i], h6[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
