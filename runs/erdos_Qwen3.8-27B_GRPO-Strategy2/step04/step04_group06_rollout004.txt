def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    """Project onto {0<=h<=1, sum=h=n/2} using bisection on threshold."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_max(c, alpha):
    """Smoothed max of array c."""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -400, 400)))) / alpha

def _grad_smooth(h, n, alpha):
    """Gradient of smoothed max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -400, 400)))) / alpha

def _smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _two_coord_opt(h, n, i, j, n_samples=20):
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-14:
        return h, _c5(h)
    best_t = h[i]
    best_c5 = _c5(h)
    ts = np.linspace(lo, hi, n_samples)
    for t in ts:
        h[i] = t
        h[j] = S - t
        c = _c5(h)
        if c < best_c5:
            best_c5 = c
            best_t = t
    span = (hi - lo) / max(n_samples - 1, 1)
    for _ in range(5):
        a = max(lo, best_t - span)
        b = min(hi, best_t + span)
        ts2 = np.linspace(a, b, 12)
        for t in ts2:
            h[i] = t
            h[j] = S - t
            c = _c5(h)
            if c < best_c5:
                best_c5 = c
                best_t = t
        span *= 0.3
        if span < 1e-10:
            break
    h[i] = best_t
    h[j] = S - best_t
    return h, best_c5

def _pgd_phase(h, n, alpha, lr_init, end_t, max_it):
    h = h.copy()
    lr = lr_init
    c5_cur = _c5(h)
    for it in range(max_it):
        if time.time() > end_t:
            break
        g = _smooth_grad(h, n, alpha)
        h_new = _project(h - lr * g, n)
        c5_new = _c5(h_new)
        if c5_new < c5_cur - 1e-16:
            h = h_new
            c5_cur = c5_new
            lr = min(lr * 1.03, 0.4)
        else:
            lr *= 0.75
        if lr < 1e-8:
            break
    return h, c5_cur

def _coord_descent_pass(h, n, rng, end_t, n_samples=15):
    c5_cur = _c5(h)
    improved = False
    idx = np.arange(n)
    rng.shuffle(idx)
    for p in range(n - 1):
        if time.time() > end_t:
            break
        i, j = int(idx[p]), int(idx[p + 1])
        h, c5_new = _two_coord_opt(h, n, i, j, n_samples)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            improved = True
    return h, c5_cur, improved

    def elapsed():
        return time.time() - t0

    try:

        if elapsed() > end1:

        h_w, c5_w = _pgd_phase(best_h, best_n, alpha, 0.15, end1, 4000)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w.copy()

    # Phase 2: 2-coordinate descent (22-55%)

    # Phase 3: Simulated annealing with pair optimization (55-78%)

    h_w = best_h.copy()
    c5_w = _c5(h_w)

        T = max(T * 0.9997, 1e-7)
        i = rng.randint(best_n)

            continue
        S = h_w[i] + h_w[j]

        t_new = lo + rng.random() * (hi - lo)

            if c5_w < best_c5:

            h_w[i] = oi
            h_w[j] = oj

    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
