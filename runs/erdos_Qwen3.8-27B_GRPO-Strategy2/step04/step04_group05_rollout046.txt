def swap_refine(h, n, dx, rng, t0, lim, cnt=50000):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try moving epsilon from i to j
        for eps in (0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001):
            if 0 <= h[i] - eps <= 1 and 0 <= h[j] + eps <= 1:
                h[i] -= eps
                h[j] += eps

_GR = {'a': 100.0, 'l': 50.0, 'n': 70}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def _sm_obj(x):
