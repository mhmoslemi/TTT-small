    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr_full(h)) * dx)

    h = np.clip(h, 0.0, 1.0).copy()

        if abs(d) < 1e-15:

    return h

def _interp1d(h, n_new):

def _subgrad(h, frac):
    """Subgradient of max(corr) using top-fraction active shifts."""
    n = len(h)
    c = _corr_full(h)
    m = c.max()
    thresh = np.percentile(c, 100.0 * (1.0 - frac))
    active = np.where(c > thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for k in active:
        s = abs(k - (n - 1))
        if s == 0 or s >= n:
            continue
        grad[:n - s] += 1.0 - h[s:]
        grad[s:] -= h[:n - s]
    grad /= len(active)
    return grad, m

- If shift = -s < 0: dc/dx_j = -x[j+s] for j < n-s, and (1-x[j-s]) for j ≥ s

def _obj_and_grad(x, alpha, tgt, pen):
    n = len(x)
    c = _corr_full(x)
    # log-sum-exp for numerical stability
    mc = c.max()
    logsum = mc + np.log(np.exp(c - mc).sum())
    f = logsum / alpha + pen * (x.sum() - tgt) ** 2
    
    # Gradient
    p = np.exp(c - mc) / np.exp(c - mc).sum()  # softmax weights
    grad = np.zeros(n)
    center = n - 1
    # shift = 0
    grad += p[center] * (1.0 - 2.0 * x)
    for s in range(1, n):
        pw = p[center + s] if center + s < 2*n - 1 else 0.0
        nw = p[center - s] if center - s >= 0 else 0.0
        if pw == 0 and nw == 0:
            continue
        grad[:n-s] += pw * (1.0 - x[s:]) + nw * (-x[s:])
        grad[s:]   += pw * (-x[:n-s]) + nw * (1.0 - x[:n-s])
    grad /= alpha
    if pen > 0:
        grad += 2.0 * pen * (x.sum() - tgt)
    return f, grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp1d(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _obj_grad(x, alpha, pen):
    n = len(x)
    tgt = n / 2.0
    c = _corr_full(x)
    mc = c.max()
    ec = np.exp(c - mc)
    p = ec / ec.sum()
    f = mc + np.log(ec.sum()) / alpha + pen * (x.sum() - tgt) ** 2
    grad = np.zeros(n)
    ctr = n - 1
    grad += p[ctr] * (1.0 - 2.0 * x)
    for s in range(1, n):
        pw = p[ctr + s] if ctr + s < 2 * n - 1 else 0.0
        nw = p[ctr - s] if ctr - s >= 0 else 0.0
        if pw < 1e-15 and nw < 1e-15:
            continue
        a1 = pw * (1.0 - x[s:]) - nw * x[s:]
        a2 = -pw * x[:n - s] + nw * (1.0 - x[:n - s])
        grad[:n - s] += a1
        grad[s:] += a2
    grad /= alpha
    if pen > 0:
        grad += 2.0 * pen * (x.sum() - tgt)
    return f, grad

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return _obj_grad(x, alpha, 0.0)[0]
    def grad(x):
        return _obj_grad(x, alpha, 0.0)[1]
    try:
        r = minimize(obj, h, jac=grad, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
                                   'jac': lambda x: np.ones_like(x)}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs(h, dx, alpha, maxiter, pen):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return _obj_grad(x, alpha, pen)[0]
    def grad(x):
        return _obj_grad(x, alpha, pen)[1]
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        hc = _repair(r.x, n / 2.0)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _subgrad(h, frac):
    n = len(h)
    c = _corr_full(h)
    m = c.max()
    thresh = np.percentile(c, 100.0 * (1.0 - frac))
    active = np.where(c > thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    ctr = n - 1
    for k in active:
        shift = k - ctr
        s = abs(shift)
        if s == 0:
            grad += 1.0 - 2.0 * h
        elif s < n:
            if shift > 0:
                grad[:n - s] += 1.0 - h[s:]
                grad[s:] -= h[:n - s]
            else:
                grad[s:] += 1.0 - h[:n - s]
                grad[:n - s] -= h[s:]
    grad /= len(active)
    return grad, m

def _subgrad_opt(h, dx, t_end, lr0=0.003, frac=0.25, mom=0.8):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    v = np.zeros(n)
    step = 0
    stall = 0
    while time.time() < t_end:
        step += 1
        lr = lr0 / (1.0 + 0.005 * step)
        g, m = _subgrad(h, frac)
        v = mom * v - lr * g
        h_new = _repair(h + v, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best - 1e-12:
            best = c_new
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 800:
                mom *= 0.9
                lr0 *= 0.5
                v *= 0.5
                stall = 0
    return h, best

def _targeted(h, dx, rng, t_end, n_iter):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    nact = max(3, n // 8)
    ctr = n - 1
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        top = np.argsort(c)[::-1][:nact]
        k = top[rng.randint(nact)]
        s = abs(k - ctr)
        if s == 0 or s >= n:

        i = rng.randint(n - s)
        j = i + s
        for sign in ([1, -1] if rng.random() < 0.5 else [-1, 1]):

            h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                c_new = _c5(h, dx)
                if c_new < best:
                    best = c_new
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best
