import time

    target = n / 2.0

        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:

        for idx in order:
            if diff > 0:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs(h, n, alpha, pen, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h.copy(), args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except:
        return h, compute_c5(h, dx)

def sa_local(h, n, dx, rng, deadline, t_max, t_min, cool=0.9999):
    """Simulated annealing local search"""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = t_max
    target = n / 2.0
    
    while time.time() < deadline and T > t_min:
        T *= cool
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = cur_h[i] + d, cur_h[j] - d
        if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
            continue
        cur_h[i], cur_h[j] = ni, nj
        new_c5 = compute_c5(cur_h, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs(h, n, alpha, pen, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h.copy(), args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sa_search(h, n, dx, rng, deadline, t_init, t_final, cool):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    T = t_init
    while time.time() < deadline and T > t_final:
        T *= cool
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = max(T * 1.5, 0.001)
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = cur[i] + d, cur[j] - d
        if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
            continue
        cur[i], cur[j] = ni, nj
        nc = compute_c5(cur, dx)
        delta = nc - cur_c5
        if delta < -1e-15 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
    return best_h, best_c5
