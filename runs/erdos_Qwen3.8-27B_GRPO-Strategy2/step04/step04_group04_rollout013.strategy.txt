            d_values = np.sort(repair_x[direction * (repair_x < 0.5 * direction)] - 0.5 * direction)
            repair_x[repair_x < 0.5 * direction] += np.abs(d_values) * min(1, np.abs(d_values)[-1]/np.abs(d))
            repair_x[repair_x > 0.5 * direction] -= np.abs(d_values) * min(1, np.abs(d_values)[:1])
            sum_val = np.sum(repair_x)
    return repair_x, sum_val

def calculate object(h):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr)

def evaluate_erdos_objective(x):
    n_val = len(x)
    objective_val = calculate_object(x)
    return objective_val

def initial_candidate(parent_values=None, resolution=700):
    if parent_values is None:

        parent_values_repaired, _ = repair_constraint(parent_values)

    return parent_values

def local_search(options):
    if options.get('seed', None):
        np.random.seed(options['seed'])
    h_init = options['h_init']
    n_val = h_init.shape[0]
    lb = np.ones(n_val) * 0.0
    ub = np.ones(n_val) * 1.0
    cons = LinearConstraint(np.ones(n_val), [n_val/2], [n_val/2])
    bounds = list(zip(lb, ub))
    res = minimize(
        lambda x: evaluate_erdos_objective(x),
        np.clip(h_init, 0.0, 1.0),
        bounds=bounds,
        constraints=cons,
        method='trust-constr',
        options={'maxiter': options.get('max_iter', 200)}
    )
    return res, res.fun

def optimize_targeted(parent_h, target_scale=1.15, target_iterations=100):
    new_n = int(len(parent_h) * target_scale)
    repaired, _ = repair_constraint(parent_h)
    # Use Linear}' to perform optimization
    options = {'h_init': repaired_linear_resample(parent_h, new_n),
               'seed': int(time.time())}
    try:
        results = local_search(options)
        best_h = results.best.x
        # Final repair
        best_h_repaired, _ = repair_constraint(best_h)
        return best_h_repaired, results.best.fun
    except:
        return parent_h, calculate_object(parent_h)
