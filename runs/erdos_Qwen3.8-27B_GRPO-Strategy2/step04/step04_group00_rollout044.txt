One more idea: **DCA (Difference of Convex Algorithms)**. The objective max_k c_k(h) can be written as:
max_k (Σ h_i - Σ h_i h_{i+k})

The first term is linear (convex and concave). The second term Σ h_i h_{i+k} is not convex in general. But it can be decomposed as:
Σ h_i h_{i+k} = 0.5 * Σ [(h_i + h_{i+k})² - h_i² - h_{i+k}²]

Each term (h_i + h_{i+k})² is convex, and h_i² is convex. So:
Σ h_i h_{i+k} = 0.5 * Σ (h_i + h_{i+k})² - 0.5 * Σ h_i² - 0.5 * Σ h_{i+k}²

The first part is convex, the second part is -convex (i.e., concave). So c_k = Σ h_i - [convex - concave] = concave - convex + linear = difference of convex functions.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(h, n):
    target = n / 2.0
    lo, hi = -max(h.max(), 0.0) - 1.0, 1.0 - min(h.min(), 1.0) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - proj) if diff > 0 else np.argsort(proj)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - proj[idx])
                proj[idx] += a
                diff -= a
            else:
                a = min(-diff, proj[idx])
                proj[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return proj

def _subgrad_topk(h, n, top_k):
    c = _corr(h)
    idx = np.argsort(c)[::-1][:top_k]
    spread = c[idx[0]] - c[idx[-1]]
    if spread < 1e-12:
        w = np.ones(top_k) / top_k
    else:
        w = 2.0 ** (c[idx] - c[idx[0]])
        w /= w.sum()
    g = np.zeros(n)
    for w_k, k in zip(w, idx):
        L = k - (n - 1)
        if L == 0:
            g += w_k * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += w_k * (1.0 - h[:n - L])
            g[:n - L] -= w_k * h[L:]
        else:
            M = -L
            g[:n - M] += w_k * (1.0 - h[M:])
            g[M:] -= w_k * h[:n - M]
    return g

def _subgrad_optimize(h, n, dx, time_limit, top_ks=(3, 5, 8)):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h_c = h.copy()
    c_c = _c5(h_c, dx)
    step = 0.03
    for tk in top_ks:
        for it in range(600):
            if time.time() > time_limit:
                return best_h, best_c5
            g = _subgrad_topk(h_c, n, tk)
            g = g - g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-13:
                break
            g /= gn
            found = False
            for s in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
                h_try = _project(h_c - s * g, n)
                c_try = _c5(h_try, dx)
                if c_try < c_c - 1e-15:
                    h_c = h_try
                    c_c = c_try
                    found = True
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_c.copy()
                    break
            if found:
                step = min(step * 1.04, 0.06)
            else:
                step *= 0.6
                if step < 5e-7:
                    break
    return best_h, best_c5

def _coord_descent(h, n, dx, time_limit, max_sweeps=300):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h_c = h.copy()
    c_c = _c5(h_c, dx)
    rng_local = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        c = _corr(h_c)
        top_idx = np.argsort(c)[::-1][:6]
        g = np.zeros(n)
        for k in top_idx:
            L = k - (n - 1)
            if L == 0:
                g += 1.0 - 2.0 * h_c
            elif L > 0:
                g[L:] += 1.0 - h_c[:n - L]
                g[:n - L] -= h_c[L:]
            else:
                M = -L
                g[:n - M] += 1.0 - h_c[M:]
                g[M:] -= h_c[:n - M]
        g /= len(top_idx)
        order = np.argsort(np.abs(g))[::-1]
        n_try = min(60, n)
        improved = False
        for ai in range(n_try):
            i = order[ai]
            grads_j = np.abs(g)
            grads_j[i] = 0
            top_j = np.argsort(grads_j)[::-1][:15]
            for sign_d in [0.003, -0.003, 0.008, -0.008, 0.015, -0.015, 0.002, -0.002]:
                ni = h_c[i] + sign_d
                if ni < 0 or ni > 1:
                    continue
                for j in top_j:
                    if j == i:
                        continue
                    nj = h_c[j] - sign_d
                    if nj < 0 or nj > 1:
                        continue
                    h_c[i], h_c[j] = ni, nj
                    c_try = _c5(h_c, dx)
                    if c_try < c_c - 1e-15:
                        c_c = c_try
                        improved = True
                        if c_try < best_c5:
                            best_c5 = c_try
                            best_h = h_c.copy()
                        break
                    h_c[i], h_c[j] = ni - sign_d, nj + sign_d
                h_c[i], h_c[j] = ni - sign_d, nj + sign_d
                if improved:
                    h_c = _project(h_c, n)
                    c_c = _c5(h_c, dx)
                    break
            if improved:
                g = g * 0.5
        if not improved:
            break
    return best_h, best_c5

def _softmax_optimize(h, n, dx, time_limit):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    def obj(x, a):
        c = _corr(x)
        m = c.max()
        s = np.clip(a * (c - m), -500, 500)
        s -= s.max()
        return m + np.log(np.sum(np.exp(s))) / a + 80.0 * (x.sum() - target) ** 2

    def grad(x, a):
        c = _corr(x)
        m = c.max()
        s = np.clip(a * (c - m), -500, 500)
        s -= s.max()
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            L = k - (n - 1)
            pk = pw[k]
            if pk < 1e-14:
                continue
            if L == 0:
                g += pk * (1.0 - 2.0 * x)
            elif L > 0:
                g[L:] += pk * (1.0 - x[:n - L])
                g[:n - L] -= pk * x[L:]
            else:
                M = -L
                g[:n - M] += pk * (1.0 - x[M:])
                g[M:] -= pk * x[:n - M]
        g *= 1.0 / a
        g += 160.0 * (x.sum() - target)
        return g

    for a in [20000, 200000, 2000000, 20000000]:
        if time.time() > time_limit:
            break
        try:
            res = minimize(lambda x: obj(x, a), best_h.copy(),
                           jac=lambda x: grad(x, a),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-18})
            hc = _project(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def _random_refine(h, n, dx, time_limit, rng):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h_c = h.copy()
    c_c = _c5(h_c, dx)
    while time.time() < time_limit:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.004, 0.004)
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_c[i], h_c[j] = ni, nj
            c_try = _c5(h_c, dx)
            if c_try < c_c:
                c_c = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
    return best_h, best_c5
