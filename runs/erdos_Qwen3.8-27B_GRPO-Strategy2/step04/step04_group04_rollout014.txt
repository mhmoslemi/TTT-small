        j = np.arange(n)
        kj = k - j
        mask = (kj >= 0) & (kj < n)
        g[mask] += (1.0 - 2.0 * h[kj[mask]])
    g /= len(active)
    return g, c, m

def _project_feasible(h, target):
    """Project h to feasible set {sum(h)=target, 0<=h<=1}"""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not free.any():
            h = np.clip(h + d / len(h), 0, 1)
        else:
            h[free] = np.clip(h[free] + d / free.sum(), 0, 1)
    return _repair(h, target)

def _subgrad_descent(h, dx, t_end, n_active=5):
    """Projected subgradient descent on max correlation"""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr0 = 0.1
    t = 0
    while time.time() < t_end:
        t += 1
        g, c, m = _grad_max_corr(h, n_active)
        lr = lr0 / (1.0 + 0.01 * t)
        h_new = h - lr * dx * g
        h_new = _project_feasible(h_new, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
        else:
            # Try smaller step
            for _ in range(3):
                lr *= 0.5
                h_new = h - lr * dx * g
                h_new = _project_feasible(h_new, tgt)
                c_new = _c5(h_new, dx)
                if c_new < best:
                    best = c_new
                    h = h_new
                    break
        if t % 1000 == 0:
            # Periodic SLSQP micro-polish
            pass
    return h, best

def _soft_obj(h, alpha, target, pen):
    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2

def _slsqp_run(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(lambda x: _soft_obj(x, alpha, tgt, 0), h,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs_run(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best, bc = h.copy(), _c5(h, dx)

                         method='L-BFGS-B', bounds=bds,

            hc = _repair(r.x, tgt)
            c = _c5(hc, dx)
            if c < bc:
                best, bc = hc, c
        except Exception:
            pass
    return best, bc

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort((1.0 - h) if d > 0 else h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_max_corr(h, n_active=5):
    n = len(h)
    c = _corr_full(h)
    m = c.max()
    thresh = m - 1e-10
    active = np.where(c >= thresh)[0]
    if len(active) > n_active:
        active = np.argsort(c)[-n_active:]
    g = np.zeros(n)
    for k in active:
        j = np.arange(n)
        kj = k - j
        mask = (kj >= 0) & (kj < n)
        g[mask] += (1.0 - 2.0 * h[kj[mask]])
    g /= len(active)
    return g, c, m

        mm = cc.max()
        s = np.clip(alpha * (cc - mm), -500, 500)
        return mm + np.log(np.exp(s).sum()) / alpha

                     options={'maxiter': maxiter, 'ftol': 1e-16})

    h1 = _repair(_interp1d(best_h, n1), n1 / 2.0)
    t_end = min(t0 + 770, t0 + budget_s - 180)
    if time.time() < t_end:
        h1, c = _lbfgs_run(h1, dx1, 3000, 200)
        if c < best_c5:
            best_h, best_c5, best_n = h1, c, n1
    if time.time() < t_end:
        h1, c = _slsqp_run(h1, dx1, 30000, 400)
        if c < best_c5:
            best_h, best_c5, best_n = h1, c, n1
    if time.time() < t_end:
        h1, c = _peeling_local(h1, dx1, t_end, 80)
        if c < best_c5:
            best_h, best_c5, best_n = h1, c, n1

    # Phase 7: Final intensive local (130s)
    nb = best_n
    dxb = 2.0 / nb
    t_end = min(t0 + 900, t0 + budget_s - 60)
    best_h, best_c5 = _peeling_local(best_h, dxb, t_end, 120)
    if time.time() < t_end:
        best_h, best_c5 = _random_local(best_h, dxb, rng, t_end, 200000)

    # Phase 8: Final SLSQP + local
    t_end = t0 + budget_s - 10
    if time.time() < t_end:
        best_h, c = _slsqp_run(best_h, dxb, 200000, 400)
        if c < best_c5:
            best_c5 = c
    if time.time() < t_end - 3:
        best_h, best_c5 = _random_local(best_h, dxb, rng, t_end, 80000)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
