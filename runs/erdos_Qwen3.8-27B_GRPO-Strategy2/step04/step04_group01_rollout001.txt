def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0 / n)

def _proj(h, target):

        if s > target:

        else:

def _smooth_grad(h, alpha):

def _slsqp(h, alpha, maxiter, ftol):

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

def _targeted_pair(h, i, j):

        di = 0.0
        if 0 <= i+mk < n:

        if 0 <= i-mk < n:
            di -= h[i-mk]
        dj = 0.0
        if 0 <= j+mk < n:
            dj += (1.0 - h[j+mk])
        if 0 <= j-mk < n:
            dj -= h[j-mk]
