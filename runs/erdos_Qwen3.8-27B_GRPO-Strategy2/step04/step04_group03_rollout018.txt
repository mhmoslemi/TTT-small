import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_sm(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_sm(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i1 = max(0, lag)
        j1 = min(n, n + lag)
        if i1 < j1:
            src = i1 - lag
            g[i1:j1] += wk * (1.0 - h[src:src + (j1 - i1)])
        if lag != 0:
            i2 = max(0, -lag)
            j2 = min(n, n - lag)
            if i2 < j2:
                src = i2 + lag
                g[i2:j2] -= wk * h[src:src + (j2 - i2)]
    return g

def _adam_opt(h0, n, alpha, lr, max_iter, t_end):
    h = h0.copy()
    mb = np.zeros(n)
    vb = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = _eval(h)
    b1, b2, eps = 0.8, 0.999, 1e-8
    for it in range(max_iter):
        if time.time() > t_end:
            break
        t += 1
        g = _grad_sm(h, alpha)
        mb = b1 * mb + (1.0 - b1) * g
        vb = b2 * vb + (1.0 - b2) * g * g
        mh = mb / (1.0 - b1**t)
        vh = vb / (1.0 - b2**t)
        step = lr * mh / (np.sqrt(vh) + eps)
        h = np.clip(h - step, 0.0, 1.0)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 200 == 100:
            c = np.correlate(h, 1.0 - h, mode='full')
            true_max = c.max() * 2.0 / n
            sm_val = _obj_sm(h, alpha)
            gap = sm_val - true_max
            if gap > 0.001:
                alpha = min(alpha * 4.0, 1e8)
            elif gap < 2e-5 and lr > 1e-5:
                lr *= 0.75
    return best_h, best_c5

def _coord_descent(h0, n, t_end):
    h = h0.copy()
    c5c = _eval(h)
    for _pass in range(2000):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            best_x = h[i]
            best_v = c5c
            base = h[i]
            for x_try in np.linspace(0.0, 1.0, 21):
                dx = x_try - base
                if abs(dx) < 1e-10:
                    continue
                j = (i + n // 3) % n
                dy = -dx
                nj = h[j] + dy
                if nj < 0 or nj > 1:
                    continue
                h[i] = x_try
                h[j] = nj
                v = _eval(h)
                if v < best_v - 1e-14:
                    best_v = v
                    best_x = x_try
            h[i] = base
            if best_x != base:
                j = (i + n // 3) % n
                h[i] = best_x
                h[j] = h[j] - (best_x - base)
                h = _repair(h, n)
                c5c = _eval(h)
                improved = True
        if not improved:
            break
    return h, c5c

def _local_search(h0, n, rng, t_end):
    h = h0.copy()
    c5c = _eval(h)
    scale = 0.006
    no_imp = 0
    for _ in range(400000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            no_imp = 0
            scale = min(scale * 1.04, 0.015)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 150:
                scale *= 0.82
                no_imp = 0
                if scale < 5e-6:
                    break
    return h, c5c

def _interp_h(hs, ns, nd):
    xs = np.linspace(0.0, 2.0, ns, endpoint=False)
    xd = np.linspace(0.0, 2.0, nd, endpoint=False)
    return np.interp(xd, xs, hs)

def _binary_greedy(h0, n, t_end):
    h = (h0 > 0.5).astype(float)
    if abs(h.sum() - n / 2.0) > 1e-9:
        if h.sum() > n / 2.0:
            order = np.argsort(h0)[::-1]
            cnt = int(h.sum() - n / 2.0 + 0.5)
            for idx in order[:cnt]:
                h[idx] = 0
        else:
            order = np.argsort(h0)
            cnt = int(n / 2.0 - h.sum() + 0.5)
            for idx in order[:cnt]:
                h[idx] = 1.0
    best_c5 = _eval(h)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            h[i] = 1.0 - h[i]
            c5 = _eval(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
            else:
                h[i] = 1.0 - h[i]
    return h, best_c5

    def _tl():
