import numpy as np
from scipy.optimize import minimize
import time

_GN = 140

def _c5vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5val(h, n):
    return float(np.max(_c5vec(h, n)))

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx]); h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx]); h[idx] -= sub; diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = _c5vec(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _c5vec(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _con_f(x):
    return float(np.sum(x) - _GN / 2.0)

def _con_j(x):
    return np.ones(_GN)

def _do_pgd(h, n, alpha, t_end, lr0, niter):
    h = h.copy()
    lr = lr0
    c5c = _c5val(h, n)
    for it in range(niter):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, n, alpha)
        h2 = _project(h - lr * g, n)
        c52 = _c5val(h2, n)
        if c52 < c5c:
            h = h2
            c5c = c52
            if it % 150 == 149:
                lr *= 1.25
        else:
            lr *= 0.7
        if it % 600 == 599:
            lr *= 0.5
    return h, c5c

def _do_slsqp(h0, n, alpha, mx):
    global _GN
    _GN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_f, 'jac': _con_j}]
    res = minimize(_smooth_obj, h0, args=(n, alpha), jac=_smooth_grad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': mx, 'ftol': 1e-16})
    return _repair(res.x, n)

def _pair_cd(h, n, t_end, seed=0):
    c5c = _c5val(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    scales = [
        (np.array([0.02, 0.05, 0.1, -0.02, -0.05, -0.1]), 250),
        (np.array([0.005, 0.01, 0.03, -0.005, -0.01, -0.03]), 400),
        (np.array([0.001, 0.003, 0.008, -0.001, -0.003, -0.008]), 400),
        (np.array([0.0003, 0.001, 0.002, -0.0003, -0.001, -0.002]), 600),
    ]
    for deltas, npass in scales:
        for _p in range(npass):
            if time.time() > t_end:
                break
            imp = False
            idx = np.arange(n)
            rng.shuffle(idx)
            for ii in range(n):
                if time.time() > t_end:
                    break
                i = idx[ii]
                for jj in range(n):
                    if jj == ii:
                        continue
                    j = idx[jj]
                    for d in deltas:
                        ni = h[i] + d
                        nj = h[j] - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i] = ni
                        h[j] = nj
                        c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                        if c < c5c - 1e-16:
                            c5c = c
                            imp = True
                        else:
                            h[i] = ni - d
                            h[j] = nj + d
            if not imp:
                break
        if time.time() > t_end:
            break
    return h, c5c

def _perturb_reopt(h, n, c5_best, t_end, rng, max_restarts=5):
    cur_h = h.copy()
    cur_c5 = c5_best
    for _r in range(max_restarts):
        if time.time() > t_end:
            break
        npert = rng.randint(3, 8)
        pert_idx = rng.choice(n, size=npert, replace=False)
        h2 = cur_h.copy()
        for p in pert_idx:
            h2[p] += rng.uniform(-0.08, 0.08)
        h2 = _repair(h2, n)
        h2, c52 = _pair_cd(h2, n, min(t_end, time.time() + 60), seed=rng.randint(10000))
        c52 = _c5val(h2, n)
        if c52 < cur_c5 - 1e-8:
            cur_c5 = c52
            cur_h = h2.copy()
    return cur_h, cur_c5
