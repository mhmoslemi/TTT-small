        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            if d > 0:

                h[idx] += a
                d -= a
            else:

                h[idx] -= a
                d += a

    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def analytic_grad(h, alpha):
    """Gradient of soft-max C5 w.r.t. h"""
    n = len(h)
    corr = get_corr(h, n)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[n-1+lag]
        if wi < 1e-12:
            continue
        # d corr[n-1+lag]/dh[i] = (1-h[i+lag]) - h[i-lag] (with boundary checks)
        # First term: h[i] appears as first factor, need i+lag in [0,n-1]
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            g[idx] += wi * (1.0 - h[idx + lag])
        # Second term: h[i] appears as second factor h[j+lag] where j=i-lag, need i-lag in [0,n-1]
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            g[idx] -= wi * h[idx - lag]
    return g

    """Projected subgradient with adaptive step size"""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step_init
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = analytic_grad(h, alpha)
        # Project gradient onto the constraint sum(h)=n/2 by removing mean
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:

def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def analytic_grad(h, alpha):
    n = len(h)
    corr = get_corr(h, n)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    for lag in range(-(n - 1), n):
        wi = w[n - 1 + lag]
        if wi < 1e-12:
            continue
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            g[idx] += wi * (1.0 - h[idx + lag])
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            g[idx] -= wi * h[idx - lag]
    return g

            i = lo + rng.randint(hi - lo)
            j = i + lag
            d = steps[rng.randint(len(steps))] * (0.5 + rng.random())
            sgn = 1 if rng.random() < 0.5 else -1
            ni, nj = h_curr[i] + sgn * d, h_curr[j] - sgn * d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_curr[i], h_curr[j] = ni, nj
            c5_new = compute_c5(h_curr, n)
            delta = c5_new - c5_curr
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                c5_curr = c5_new
                if c5_curr < best:
                    best = c5_curr
                    h = h_curr.copy()
            else:
                h_curr[i], h_curr[j] -= sgn * d
                h_curr[j], h_curr[i] = h_curr[j], h_curr[i]
                h_curr[i], h_curr[j] = nj + sgn * d, h_curr[i] - sgn * d
                h_curr[i] = np.clip(hi - sgn * d + 2 * sgn * d, 0, 1)
                # Just revert
                h_curr[i] = h[i] if i == i else h_curr[i]
                # Simpler: just revert
    return h, best

def swap_finetune(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(3 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h, n)
            if c5 < best - 1e-14:
                best = c5
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved and r > 30:
            break
    return h, best

def resample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return repair(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 7)
    best_h = None
    best_c5 = np.inf
    best_n = 64
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except NameError:
        pass
    if best_h is None:
        best_n = 64
        x = np.linspace(0, 1, best_n)
        best_h = 0.5 + 0.3 * np.cos(2 * np.pi * x)
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient with increasing alpha (20% budget)
    h_work = best_h.copy()
    for alpha in [2000, 8000, 30000, 100000, 400000]:
        if elapsed() > budget_s * 0.20:
            break
        t_end = min(t0 + budget_s * 0.20, t0 + elapsed() + budget_s * 0.04)
        h_work, c5_w = subgrad_step(h_work, best_n, alpha, 400, t_end)
        if c5_w < best_c5 - 1e-13:
            best_h = h_work.copy()
            best_c5 = c5_w

    # Phase 2: Multi-lag targeted coord descent (25% budget)
    if elapsed() < budget_s * 0.45:
        t2 = t0 + budget_s * 0.45
        for K in [3, 5, 7]:
            if elapsed() > budget_s * 0.45:
                break
            h2, c5_2 = multi_lag_coord(best_h, best_n, 400, t2, rng, K)
            if c5_2 < best_c5 - 1e-13:
                best_h = h2.copy()
                best_c5 = c5_2

    # Phase 3: Fine random-pair coordinate descent (25% budget)
    if elapsed() < budget_s * 0.70:
        t3 = t0 + budget_s * 0.70
        h3, c5_3 = fine_coord(best_h, best_n, 3000, t3, rng)
        if c5_3 < best_c5 - 1e-13:
            best_h = h3.copy()
            best_c5 = c5_3

    # Phase 4: Swap finetuning (10% budget)
    if elapsed() < budget_s * 0.80:
        t4 = t0 + budget_s * 0.80
        h4, c5_4 = swap_finetune(best_h, best_n, 200, t4, rng)
        if c5_4 < best_c5 - 1e-13:
            best_h = h4.copy()
            best_c5 = c5_4
        # Another round of fine coord after swap
        h4b, c5_4b = fine_coord(best_h, best_n, 1000, t4, rng)
        if c5_4b < best_c5 - 1e-13:
            best_h = h4b.copy()
            best_c5 = c5_4b

    # Phase 5: Try other n values (10% budget)
    if elapsed() < budget_s * 0.90:
        t5 = t0 + budget_s * 0.90
        for n_alt in [60, 66, 68, 70, 72, 80, 96]:
            if elapsed() > budget_s * 0.90:
                break
            if n_alt == best_n:
                continue
            h_s = resample_h(best_h, best_n, n_alt)
            h_s, c5_s = subgrad_step(h_s, n_alt, 50000, 150, t5)
            h_s, c5_s = multi_lag_coord(h_s, n_alt, 200, t5, rng, 5)
            h_s, c5_s = fine_coord(h_s, n_alt, 500, t5, rng)
            if c5_s < best_c5 - 1e-13:
                best_h = h_s.copy()
                best_c5 = c5_s
                best_n = n_alt

    # Phase 6: Final ultra-fine polish (10% budget)
    if best_h is not None and elapsed() < budget_s * 0.99:
        t6 = t0 + budget_s * 0.995
        # Subgradient with very high alpha
        h6, c5_6 = subgrad_step(best_h, best_n, 500000, 200, t6)
        if c5_6 < best_c5 - 1e-13:
            best_h = h6.copy()
            best_c5 = c5_6
        # Ultra-fine coord
        h6b, c5_6b = fine_coord(best_h, best_n, 2000, t6, rng)
        if c5_6b < best_c5 - 1e-13:
            best_h = h6b.copy()
            best_c5 = c5_6b
        # One more multi-lag pass
        h6c, c5_6c = multi_lag_coord(best_h, best_n, 300, t6, rng, 3)
        if c5_6c < best_c5 - 1e-13:
            best_h = h6c.copy()
            best_c5 = c5_6c

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
