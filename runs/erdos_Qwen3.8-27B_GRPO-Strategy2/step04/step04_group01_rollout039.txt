    """Project h onto {0<=x<=1, sum(x)=target}"""
    lo, hi = -2.0, 2.0

    n = len(h)

        sh = j - (n-1)

            g[:n-sh] += wk * (1.0 - h[sh:n])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _f(h, a):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(a*(c-m), -500, 500); s -= s.max()
    return m + np.log(np.sum(np.exp(s)))/a

def _g(h, a):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(a*(c-m), -500, 500); s -= s.max()
    w = np.exp(s); ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2*n-1); w[c.argmax()] = 1.0
    else: w /= ws
    gr = np.zeros(n)
    for j in range(2*n-1):
        if w[j] < 1e-14: continue
        sh = j - (n-1); wk = w[j]
        if sh >= 0:

        if np.clip(h-mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    return np.clip(h-0.5*(lo+hi), 0, 1)

    except: return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _f(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / a

def _g(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[c.argmax()] = 1.0
    else:
        w /= ws
    gr = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        sh = j - (n - 1)
        wk = w[j]
        if sh >= 0:
            gr[:n - sh] += wk * (1.0 - h[sh:n])
            gr[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            gr[sp:] += wk * (1.0 - h[:n - sp])
            gr[:n - sp] -= wk * h[sp:]
    return gr

def _cf(x):
    return x.sum() - len(x) * 0.5

def _cj(x):
    return np.ones(len(x))

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, a, mi):
    n = len(h)
    bds = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        r = minimize(_f, h.copy(), jac=_g, args=(a,), method='SLSQP',
                     bounds=bds, constraints=cs,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _pgd(h, a, ns, lr0, t0, bud):
    n = len(h)
    tgt = n * 0.5
    fo = _f(h, a)
    lr = lr0
    for _ in range(ns):
        if time.time() - t0 > bud:
            break
        gr = _g(h, a)
        if np.linalg.norm(gr) < 1e-13:
            break
        for _bt in range(30):
            hn = _proj(h - lr * gr, tgt)
            fn = _f(hn, a)
            if fn < fo + 1e-14:
                h, fo = hn, fn
                break
            lr *= 0.5
        else:
            break
        lr = min(lr * 1.05, lr0)
    return _repair(h, n)

def _up(h, fac):
    n = len(h)
    r = np.zeros(n * fac)
    for i in range(n):
        r[i * fac:(i + 1) * fac] = h[i]
    return r

def _intv(n, m):
    h = np.zeros(n)
    tot = n // 2
    bt = tot // m
    rm = tot % m
    ws = [bt + (1 if i < rm else 0) for i in range(m)]
    gt = n - tot
    g0 = gt // (m + 1)
    rg = gt % (m + 1)
    p = 0
    for i in range(m):
        p += g0 + (1 if i < rg else 0)
        h[p:p + ws[i]] = 1.0
        p += ws[i]
    return _repair(h, n)

def _coord(h, rng, nit, sc, t0, bud):
    nc = len(h)
    c0 = _eval(h)
    for _ in range(nit):
        if time.time() - t0 > bud:
            break
        i, j = rng.randint(nc), rng.randint(nc)
        if i == j:

        d = sc * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < c0:
                c0 = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c0

def _sa_search(h, rng, nit, t0, bud):
    nc = len(h)
    c0 = _eval(h)
    T = 0.0005
    for _ in range(nit):

            c = _eval(h)

                c0 = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c0
