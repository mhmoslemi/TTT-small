def _pairwise(h, n, dx, rng, t0, budget, n_sweeps=200):
    """Greedy pairwise swap"""
    h = h.copy()
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    dlist = np.array([0.001, 0.003, 0.006, 0.01, 0.02, 0.04,
                      -0.001, -0.003, -0.006, -0.01, -0.02, -0.04])
    for sweep in range(n_sweeps):

            break
        g = _softgrad(h, 200000)
        order = np.argsort(np.abs(g))[::-1][:35]
        improved = False
        for ii in range(len(order)):

                j = order[jj]
                bd, bc = 0.0, c5c
                for d in dlist:

                    if 0 <= ni <= 1 and 0 <= nj <= 1:

                        cv = _c5(h, dx)
                        if cv < bc:
                            bc, bd = cv, d
                        h[i], h[j] = ni-d, nj+d
                if bd != 0:
                    h[i] += bd; h[j] -= bd
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < best_c5:
                        best_c5 = c5c; best_h = h.copy()
                    break
        if not improved:
            for _ in range(100):
                i, j = rng.randint(n), rng.randint(n)
                if i == j: continue

def _lfgs(h, n, t0, budget):

def _active_obj(x, peaks, lam):
    c = _corr(x)
    vals = np.array([c[p] for p in peaks])
    m = vals.max()
    # Smooth max: m + (1/alpha) * log(sum(exp(alpha*(vals-m))))
    # But for just a few peaks, we can use: m + (1/alpha)*log(sum(exp(alpha*vals - alpha*m)))
    alpha = 5000
    s = alpha * (vals - m)
    s -= s.max()
    smooth = m + np.log(np.sum(np.exp(s))) / alpha
    penalty = lam * (x.sum() - len(x)/2)**2
    return smooth + penalty

def _active_grad(x, peaks, n, lam):
    c = _corr(x)
    vals = np.array([c[p] for p in peaks])
    m = vals.max()
    alpha = 5000
    s = alpha * (vals - m)
    s -= s.max()
    ew = np.exp(s)
    w = ew / ew.sum()
    g = np.zeros(n)
    for wi, p in zip(w, peaks):
        L = p - (n-1)
        if L == 0:
            g += wi * (1-2*x)
        elif L > 0:
            g[L:] += wi * (1-x[:n-L])
            g[:n-L] -= x[L:]
        else:
            M = -L
            g[:n-M] += wi * (1-x[M:])
            g[M:] -= x[:n-M]
    g *= 1/alpha
    g += 2*lam*(x.sum()-n/2)
    return g

def _subgrad_opt(h, n, t0, budget, n_iter=500):
    dx = 2.0/n

    best_h, best_c5 = h.copy(), c5c
    step = 0.01
    mu = 0.3
    mom = np.zeros(n)
    for _ in range(n_iter):
        if time.time() - t0 > budget: break
        c = _corr(h)
        top = np.argsort(c)[::-1][:8]
        g = np.zeros(n)
        for p in top:
            g += _grad_shift(h, p - (n-1))
        g /= len(top)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-12: break
        g /= ng
        mom = mu * mom + (1-mu) * g
        found = False
        for s in [step, step*0.5, step*0.25, step*0.1]:

                break
        if not found:

        else:
            step = min(step * 1.1, 0.05)

                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                if abs(d) < 1e-14: break
        else:
            for i in np.argsort(h):
                a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-14: break
    return h

def _grad_shift(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    return g

def _top_grad(h, k=8):
    c = _corr(h)
    n = len(h)
    top = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for p in top:
        g += _grad_shift(h, p - (n - 1))
    return g / k

def _softobj(x, n, alpha, peaks):
    c = _corr(x)
    pa = np.asarray(peaks, dtype=int)
    v = c[pa]
    m = v.max()
    s = np.clip(alpha * (v - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha + 80.0 * (x.sum() - n / 2.0) ** 2

def _softgrad(x, n, alpha, peaks):
    c = _corr(x)
    pa = np.asarray(peaks, dtype=int)
    v = c[pa]
    m = v.max()
    s = np.clip(alpha * (v - m), -500, 500)
    s -= s.max()
    ew = np.exp(s)
    w = ew / ew.sum()
    g = np.zeros(n)
    for wi, p in zip(w, pa):
        g += wi * _grad_shift(x, p - (n - 1))
    g /= alpha
    g += 160.0 * (x.sum() - n / 2.0)
    return g

def _lfgs(h, n, t0, budget, top_k=10):
    bd = [(0.0, 1.0)] * n
    h = h.copy()
    bh, bc = h.copy(), _c5(h, 2.0 / n)
    for alpha in [200, 1000, 5000, 25000, 100000, 500000, 2000000]:
        if time.time() - t0 > budget:
            break
        pk = np.argsort(_corr(h))[::-1][:top_k].tolist()
        try:
            r = minimize(_softobj, h.copy(), args=(n, alpha, pk),
                         jac=_softgrad, method='L-BFGS-B', bounds=bd,
                         options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-15})
            hc = _repair(r.x, n)
            cv = _c5(hc, 2.0 / n)
            if cv < bc:
                bc, bh, h = cv, hc.copy(), hc
        except Exception:
            pass
    return bh, bc

def _subgrad(h, n, t0, budget, niter=500):
    dx = 2.0 / n
    h = h.copy()
    c5c = _c5(h, dx)
    bh, bc = h.copy(), c5c
    step, mu = 0.01, 0.3
    mom = np.zeros(n)
    for _ in range(niter):
        if time.time() - t0 > budget:
            break
        g = _top_grad(h, 8)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-12:
            break
        g /= ng
        mom = mu * mom + (1 - mu) * g
        ok = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            hn = _repair(h - s * mom, n)
            cv = _c5(hn, dx)
            if cv < c5c - 1e-15:
                h, c5c, ok = hn, cv, True
                if c5c < bc:
                    bc, bh = c5c, h.copy()
                break
        if not ok:
            step *= 0.7
            if step < 1e-8:
                break
        else:
            step = min(step * 1.1, 0.05)
    return bh, bc

def _pair(h, n, t0, budget, rng, nswp=200):

    s = alpha * (v - vm)
    ew = np.exp(s)
    w = ew / ew.sum()
    g = np.zeros(n)
    for wi, p in zip(w, pa):
        g += wi * _grad_shift(x, p - (n - 1))
    g += 160.0 * (x.sum() - n / 2.0)
    return g

def _lfgs(h, n, t0, budget, top_k):
    bd = [(0.0, 1.0)] * n
    h = h.copy()
    bh, bc = h.copy(), _c5(h, 2.0 / n)
    for a in [200, 1000, 5000, 25000, 100000, 500000, 2000000]:
        if time.time() - t0 > budget: break
        pk = np.argsort(_corr(h))[::-1][:top_k].tolist()
        try:
            r = minimize(_softobj, h.copy(), args=(n, a, pk),
                         jac=_softgrad, method='L-BFGS-B', bounds=bd,
                         options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-15})
            hc = _repair(r.x, n)
            cv = _c5(hc, 2.0 / n)
            if cv < bc: bc, bh, h = cv, hc.copy(), hc
        except Exception:
            pass
    return bh, bc

def _subgrad(h, n, t0, budget, niter):
    dx = 2.0 / n
    h = h.copy()
    c5c = _c5(h, dx)
    bh, bc = h.copy(), c5c
    step, mu = 0.01, 0.3
    mom = np.zeros(n)
    for _ in range(niter):
        if time.time() - t0 > budget: break
        g = _top_grad(h, 8)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-12: break
        g /= ng
        mom = mu * mom + (1.0 - mu) * g
        ok = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            hn = _repair(h - s * mom, n)
            cv = _c5(hn, dx)
            if cv < c5c - 1e-15:
                h, c5c, ok = hn, cv, True
                if c5c < bc: bc, bh = c5c, h.copy()
                break
        if not ok:
            step *= 0.7
            if step < 1e-8: break
        else:
            step = min(step * 1.1, 0.05)
    return bh, bc

def _pair(h, n, t0, budget, rng, nswp):
    dx = 2.0 / n
    h = h.copy()
    c5c = _c5(h, dx)
    bh, bc = h.copy(), c5c
    dl = np.array([0.001, 0.003, 0.006, 0.01, 0.02, 0.04,
                   -0.001, -0.003, -0.006, -0.01, -0.02, -0.04])
    for _ in range(nswp):
        if time.time() - t0 > budget: break
        g = _top_grad(h, 6)
        ord_ = np.argsort(np.abs(g))[::-1][:35]
        imp = False
        for ii in range(len(ord_)):
            i = ord_[ii]
            for jj in range(ii + 1, min(ii + 20, len(ord_))):
                j = ord_[jj]
                bd_, bc_ = 0.0, c5c
                for d in dl:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        cv = _c5(h, dx)
                        if cv < bc_: bc_, bd_ = cv, d
                        h[i], h[j] = ni - d, nj + d
                if bd_ != 0:
                    h[i] += bd_; h[j] -= bd_
                    h = _repair(h, n); c5c = _c5(h, dx)
                    imp = True
                    if c5c < bc: bc, bh = c5c, h.copy()
                    break
        if not imp:
            for _2 in range(100):
                i, j = rng.randint(n), rng.randint(n)
                if i == j: continue
                d = rng.uniform(-0.004, 0.004)
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    cv = _c5(h, dx)
                    if cv < c5c:
                        c5c = cv
                        if cv < bc: bc, bh = cv, h.copy()
                        imp = True
                    else:
                        h[i], h[j] = ni - d, nj + d
                if imp: break
            if not imp: break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    bh, bc = h0.copy(), _c5(h0, 2.0 / n0)

    h1, c1 = _lfgs(h0, n0, t0, budget_s * 0.25, 10)
    if c1 < bc: bc, bh = c1, h1.copy()

    h2, c2 = _subgrad(h1, n0, t0, budget_s * 0.42, 500)
    if c2 < bc: bc, bh = c2, h2.copy()

    h3, c3 = _pair(h2, n0, t0, budget_s * 0.58, rng, 250)
    if c3 < bc: bc, bh = c3, h3.copy()

    # restart from perturbed best
    if time.time() - t0 < budget_s * 0.65:
        hp = _repair(bh + rng.randn(len(bh)) * 0.02, len(bh))
        hp2, cp2 = _lfgs(hp, len(bh), t0, budget_s * 0.65, 10)
        if cp2 < bc: bc, bh = cp2, hp2.copy()

    # n=70
    if time.time() - t0 < budget_s * 0.70:
        n4 = 70
        x4 = _repair(np.interp(np.linspace(0, 2, n4, endpoint=False),
                               np.linspace(0, 2, len(bh), endpoint=False), bh), n4)
        h4, c4 = _lfgs(x4, n4, t0, budget_s * 0.76, 8)
        h4, c4 = _subgrad(h4, n4, t0, budget_s * 0.80, 300)
        h4, c4 = _pair(h4, n4, t0, budget_s * 0.84, rng, 150)
        if c4 < bc: bc, bh = c4, h4.copy()

    # n=100
    if time.time() - t0 < budget_s * 0.88:
        n5 = 100
        x5 = _repair(np.interp(np.linspace(0, 2, n5, endpoint=False),
                               np.linspace(0, 2, len(bh), endpoint=False), bh), n5)
        h5, c5v = _lfgs(x5, n5, t0, budget_s * 0.91, 10)
        h5, c5v = _lfgs(h5, n5, t0, budget_s * 0.94, 10)
        if c5v < bc: bc, bh = c5v, h5.copy()

    nb = len(bh)
    dxb = 2.0 / nb
    if time.time() - t0 < budget_s - 3:
        h6, c6 = _pair(bh, nb, t0, budget_s - 3, rng, 120)
        if c6 < bc: bc, bh = c6, h6.copy()

    if time.time() - t0 < budget_s - 1:
        h = bh.copy()
        c5c = _c5(h, dxb)
        for _ in range(10000):
            if time.time() - t0 > budget_s - 0.5: break
            i, j = rng.randint(nb), rng.randint(nb)
            if i == j: continue
            d = rng.uniform(-0.0005, 0.0005)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cv = _c5(h, dxb)
                if cv < c5c:
                    c5c = cv
                    if cv < bc: bc, bh = cv, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    bh = _repair(bh, nb)
    bc = _c5(bh, dxb)
    return (bh, bc, nb)
