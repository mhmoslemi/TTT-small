For a given lag k, c_k = sum_i h[i]*(1-h[i+k]). The gradient with respect to h[j] is:
- For the term where h[j] is the first factor: (1-h[j+k]) if j+k < n
- For the term where h[j] is the second factor: -h[j-k] if j-k >= 0

- Decrease h[j] where (1-h[j+k]) is large (i.e., h[j+k] is small)
- Increase h[j] where h[j-k] is large

1. Find the lag k* with max correlation
2. Compute the gradient direction for c_{k*}
3. Take a small step in that direction
4. Repeat

                h[i] += d
                h[j] -= d
                c = compute_c5(h, dx)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    imp = iters
                else:
                    h[i] -= d
                    h[j] += d
        if iters - imp > 200000:
            scale *= 0.75
            imp = iters
        if scale < 0.00005:
            scale = 0.00005
    return h, best_c5

def perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(1, n - 1))
    elif kind == 1:
        f = rng.randint(1, 8)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.04, 0.14)
        hp += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.04, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 6)
        t = np.arange(n) / n
        hp += rng.uniform(0.03, 0.10) * np.sin(2 * np.pi * f1 * t) * np.sin(2 * np.pi * f2 * t)
    else:
        hp *= rng.uniform(0.80, 1.20)
    return repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    dx0 = 2.0 / n0
    h0 = repair(h_init.copy(), n0)
    c5_0 = compute_c5(h0, dx0)
    best_h, best_c5, best_n = h0.copy(), c5_0, n0

    h_work = h0.copy()
    c5_work = c5_0

    # Phase 1: L-BFGS multi-alpha (35s)
    t_end = t_start + 35
    for alpha in [50, 200, 800, 3000, 10000]:
        if time.time() >= t_end:
            break
        for pen in [1000, 10000, 100000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_step(h_work, n0, alpha, n0 / 2.0, pen, 250)
            if cc < c5_work:
                h_work, c5_work = hh, cc
    if c5_work < best_c5:
        best_h, best_c5 = h_work.copy(), c5_work

    # Phase 2: Long local search (350s) - main workhorse
    if remaining() > 300:
        t_end = time.time() + min(350, remaining() - 120)
        h_work, c5_work = local_search(h_work, n0, dx0, rng, t_end, 0.05)
        if c5_work < best_c5:
            best_h, best_c5, best_n = h_work.copy(), c5_work, n0

    # Phase 3: L-BFGS polish (30s)
    if remaining() > 80:
        t_end = time.time() + min(30, remaining() - 40)
        for alpha in [5000, 20000, 100000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_step(h_work, n0, alpha, n0 / 2.0, 50000, 400)
            if cc < c5_work:
                h_work, c5_work = hh, cc
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    # Phase 4: Multi-restart (80s)
    if remaining() > 120:
        t_end = time.time() + min(80, remaining() - 60)
        for r in range(10):
            if time.time() >= t_end:
                break
            hp = perturb(h_work, n0, rng, r % 5)
            hh, cc = lbfgs_step(hp, n0, 2000, n0 / 2.0, 20000, 200)
            t_sub = min(time.time() + 8, t_end)
            hh, cc = local_search(hh, n0, dx0, rng, t_sub, 0.02)
            if cc < c5_work:
                h_work, c5_work = hh, cc
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    # Phase 5: SLSQP refinement (40s)
    if remaining() > 80:
        t_end = time.time() + min(40, remaining() - 40)
        for alpha in [1000, 5000, 30000]:
            if time.time() >= t_end:
                break
            hh, cc = slsqp_step(h_work, n0, alpha, n0 / 2.0, 350)
            if cc < c5_work:
                h_work, c5_work = hh, cc
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    # Phase 6: Fine local search (300s)
    if remaining() > 80:
        t_end = time.time() + min(300, remaining() - 40)
        h_work, c5_work = local_search(h_work, n0, dx0, rng, t_end, 0.008)
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    # Phase 7: Final L-BFGS + ultra-fine search (60s)
    if remaining() > 40:
        hh, cc = lbfgs_step(h_work, n0, 200000, n0 / 2.0, 100000, 500)
        if cc < c5_work:
            h_work, c5_work = hh, cc
        t_end = time.time() + max(5, min(40, remaining() - 10))
        h_work, c5_work = local_search(h_work, n0, dx0, rng, t_end, 0.001)
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    # Phase 8: Try n=2n upscale if time permits (120s)
    if remaining() > 100:
        n2 = 2 * n0
        dx2 = 2.0 / n2
        h2 = np.repeat(best_h, n2 // best_n).astype(float)
        h2 = repair(h2, n2)
        c5_2 = compute_c5(h2, dx2)
        t_end = time.time() + min(50, remaining() - 40)
        for alpha in [500, 2000, 8000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_step(h2, n2, alpha, n2 / 2.0, 20000, 200)
            if cc < c5_2:
                h2, c5_2 = hh, cc
        if remaining() > 30:
            t_end2 = time.time() + min(50, remaining() - 10)
            h2, c5_2 = local_search(h2, n2, dx2, rng, t_end2, 0.03)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2.copy(), c5_2, n2

    # Final
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
