def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def binarize(h, n):

    target = n // 2

    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)

The C5 value depends on dx = 2/n, so smaller n means larger dx, which means the correlation values are scaled up. But the integral is the same... wait no.

C5 = max_k corr[k] * dx = max_k (sum of products) * (2/n)

For binary h with n/2 ones:
- At lag 0: corr = 0 (since h*(1-h) = 0)
- At lag k: corr[k] = number of positions where h[i]=1 and h[i+k]=0 (approximately)

For a "random" binary sequence, the expected value at off-peak lags is about n/4 (since expected h[i]=0.5, h[i+k]=0.5, so h[i](1-h[i+k]) ≈ 0.25, sum over n positions ≈ n/4).

So C5 ≈ (n/4) * (2/n) = 1/2 for random. We need to do much better than that.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(2000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_local_search(hb, n, rng, t_end, n_max_iter=800000):
    best = compute_c5(hb, n)
    h = hb.copy()
    for it in range(n_max_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h, n)
        if c < best - 1e-15:
            best = c
        else:
            h[i], h[j] = h[j], h[i]
    return h, best

def binary_sa(hb, n, rng, t_end):
    current = compute_c5(hb, n)
    best = current
    h = hb.copy()
    t_start = time.time()
    total = max(t_end - t_start, 0.1)
    for it in range(2000000):
        if time.time() > t_end:
            break
        frac = min(1.0, (time.time() - t_start) / total)
        T = 0.08 * (1.0 - frac) ** 2 + 0.0005
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h, n)
        delta = c - current
        if delta < 0 or rng.rand() < np.exp(-max(delta, 0) / T):
            current = c
            if c < best:
                best = c
        else:
            h[i], h[j] = h[j], h[i]
    return h, best

def fine_tune(h, n, rng, t_end, n_max_iter=300000):
    h = h.copy()
    best = compute_c5(h, n)
    deltas = [0.08, 0.04, 0.02, 0.01, 0.005]
    for it in range(n_max_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        hi, hj = h[i], h[j]
        if hi < hj:
            i, j = j, i
            hi, hj = h[i], h[j]
        found = False
        for d in deltas:
            if d > hi or d > 1.0 - hj:
                continue
            h[i] -= d
            h[j] += d
            c = compute_c5(h, n)
            if c < best - 1e-15:
                best = c
                found = True
                break
            h[i] += d
            h[j] -= d
        if not found:
            h[i] += 0.0
            h[j] += 0.0
    return h, best

def soft_max_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def grad_opt(h, n, alpha, maxiter, t_end):
    def obj(v):
        return soft_max_obj(v, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() > t_end:
            return h
        return repair(res.x, n)
    except Exception:
        return h

def make_structured_init(n, kind, seed_val):
    r = np.random.RandomState(seed_val)
    if kind == 0:
        h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    elif kind == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif kind == 2:
        x = np.linspace(0, 2, n)
        h = (0.5 + 0.49 * np.sin(2 * np.pi * x)).clip(0, 1)
    elif kind == 3:
        x = np.linspace(0, 2, n)
        h = (0.5 + 0.49 * np.cos(2 * np.pi * 3 * x)).clip(0, 1)
    else:
        h = r.rand(n)
    return h

def resample(h, n_from, n_to):
    x0 = np.linspace(0, 1, n_from)
    x1 = np.linspace(0, 1, n_to)
    return np.interp(x1, x0, h)

    def el():
