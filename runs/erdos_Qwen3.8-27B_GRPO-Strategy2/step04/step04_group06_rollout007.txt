import numpy as np
from scipy.optimize import minimize
import time

_GN = 140

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-9) & (hp < 1.0 - 1e-9)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _sobj(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _con_fun(x):
    return float(np.sum(x) - _GN / 2.0)

def _con_jac(x):
    return np.ones(_GN)

def _pgd(h, n, alpha, t_end, lr_init):
    h = h.copy()
    best_c = _c5(h, n)
    lr = lr_init
    for _ in range(20000):
        if time.time() > t_end:
            break
        g = _sgrad(h, n, alpha)
        step = lr
        accepted = False
        for b in range(25):
            h_new = _project(h - step * g, n)
            c_new = _c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                accepted = True
                break
            step *= 0.5
        if accepted:
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.85
    return h, _c5(h, n)

def _slsqp(h_init, n, alpha, maxiter):
    global _GN
    _GN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_sobj, h_init, args=(n, alpha), jac=_sgrad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def _local_search(h, n, t_end, rng, alpha_ls=100000):
    h = h.copy()
    c_cur = _c5(h, n)
    h_best = h.copy()
    c_best = c_cur
    n_top = 15
    steps = np.array([0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002,
                      -0.008, -0.004, -0.002, -0.001, -0.0005, -0.0002])
    patience = 0
    while time.time() < t_end:
        g = _sgrad(h, n, alpha_ls)
        idx = np.argsort(np.abs(g))[-n_top:]
        rng.shuffle(idx)
        improved = False
        for i_idx in range(len(idx)):
            if time.time() > t_end:
                break
            i = idx[i_idx]
            hi = h[i]
            for j_idx in range(len(idx)):
                if time.time() > t_end:
                    break
                j = idx[j_idx]
                if j == i:
                    continue
                hj = h[j]
                for d in steps:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c_cur - 1e-15:
                        c_cur = c
                        hi, hj = ni, nj
                        improved = True
                        if c < c_best:
                            c_best = c
                            h_best = h.copy()
                    else:
                        h[i], h[j] = hi, hj

        if improved:
