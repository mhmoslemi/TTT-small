        stalled = 0
        for _ in range(1200):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.0005 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += grad1(h, ki, n)
            g /= len(idx)
            g -= g.mean()
            improved = False
            for _b in range(50):
                hn = repair(h - step * g, n)
                cn = c5v(hn, dx)
                ch = c5v(h, dx)
                if cn < ch - 1e-19:
                    h = hn
                    improved = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.18, 3.0)
                    stalled = 0
                    break
                step *= 0.68
            if not improved:
                stalled += 1
                if stalled > 12:
                    break
    return bh, bc

def spec_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.2, 0.35, 0.5, 0.35, 0.2, 0.12, 0.08]:
        h = bh.copy()
        for _ in range(300):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            target = np.full(n // 2 + 1, np.sqrt(R0 / n))
            target[0] = abs(H[0])
            nm = (1 - a) * mag + a * target
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            ch = c5v(h, dx)
            if cn < ch - 1e-19:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def coord_phase(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[::-1][:8]
    grads = np.array([grad1(h, ki, n) for ki in top_k])
    dt_vals = np.array([0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, -0.02, -0.01, -0.005, -0.002, -0.001, -5e-4])
    halts = 0
    for _ in range(800):
        if time.time() - t0 > lim:
            break
        if _ % 80 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
            top_k = np.argsort(c)[::-1][:8]
            grads = np.array([grad1(h, ki, n) for ki in top_k])
        imp = False
        for _ in range(400):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            gi = grads[:, i].mean()
            gj = grads[:, j].mean()
            if gi < gj:
                d = dt_vals[rng.randint(6)]
            else:
                d = dt_vals[6 + rng.randint(6)]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cv = c5v(h, dx)
                if cv < cc - 1e-17:
                    h = repair(h, n)
                    cc = c5v(h, dx)
                    imp = True
                    if cc < bc:
                        bc, bh = cc, h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 6:
            break
    return bh, bc

def polish(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    dt = np.array([1.5e-4, 3e-4, 6e-4, 1e-3, 2e-3, -1.5e-4, -3e-4, -6e-4, -1e-3, -2e-3])
    c = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(c)[::-1][:10]
    grads = np.array([grad1(h, ki, n) for ki in top_k])
    count = 0
    while time.time() - t0 < lim:
        if count % 300 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
            top_k = np.argsort(c)[::-1][:10]
            grads = np.array([grad1(h, ki, n) for ki in top_k])
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            count += 1
            continue
        d = dt[rng.randint(len(dt))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cv = c5v(h, dx)
            if cv < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        count += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    # Phase 1: L-BFGS-B warmup
    t1 = sum([100])
    r, c = lbgfs(bh, bn, dx0, t0, t0 + t1)
    if c < bc:
        bh, bc = r, c

    # Phase 2: Active set (main optimization)
    t2 = t1 + 250
    r, c = active_phase(bh, bn, dx0, t0, t0 + t2)
    if c < bc:
        bh, bc = r, c

    # Phase 3: Spectral
    t3 = t2 + 50
    r, c = spec_phase(bh, bn, dx0, t0, t0 + t3)
    if c < bc:
        bh, bc = r, c

    # Phase 4: More active set
    t4 = t3 + 150
    r, c = active_phase(bh, bn, dx0, t0, t0 + t4)
    if c < bc:
        bh, bc = r, c

    # Phase 5: Coordinate descent
    t5 = t4 + 150
    r, c = coord_phase(bh, bn, dx0, rng, t0, t0 + t5)
    if c < bc:
        bh, bc = r, c

    # Phase 6: Try resolution doubling if time and size permit
    if bn < 600 and t0 + t5 + 60 < t0 + budget_s:
        n2 = bn * 2
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        d2 = 2.0 / n2
        tb = min(t0 + budget_s - 30, t0 + t5 + 120)
        r, c = active_phase(h2, n2, d2, t0, tb)
        if c < bc:
            r, c = polish(r, n2, d2, rng, t0, t0 + budget_s - 20)
            if c < bc:
                bh, bc, bn = r, c, n2

    # Phase 7: Final polish with all remaining time
    r, c = polish(bh, bn, 2.0 / bn, rng, t0, t0 + budget_s - 5)
    if c < bc:
        bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
