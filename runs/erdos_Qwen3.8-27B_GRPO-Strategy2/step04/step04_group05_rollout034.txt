import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def smax_obj(x, n, a, lam):
    c = corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + lam * (x.sum() - n * 0.5) ** 2

def smax_jac(x, n, a, lam):
    c = corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 0.01:
            g += pw[k] * grad1(x, k, n)
    # NO division by a - fixed gradient!
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

    return x.sum() - n * 0.5

    return np.ones(n)

    bds = [(0.0, 1.0)] * n
    for a, lm in [(50, 100), (200, 100), (1000, 50), (5000, 20), (20000, 10), (100000, 5)]:

        f = lambda x: smax_obj(x, n, a, lm)
        j = lambda x: smax_jac(x, n, a, lm)

            c = c5v(hc, dx)
            if c < bc:

    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

_P = {'n': 0, 'a': 1.0, 'lam': 10.0, 'p': None}

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def obj_fn(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + lam * (x.sum() - n * 0.5) ** 2

def jac_fn(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 0.005:
            g += pw[k] * grad1(x, k, n)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def mm_obj(x):
    n = _P['n']
    p = _P['p']
    return float(np.dot(p, corr(x))) + _P['lam'] * (x.sum() - n * 0.5) ** 2

def mm_jac(x):
    n = _P['n']
    p = _P['p']
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] > 1e-8:
            g += p[k] * grad1(x, k, n)
    g += 2.0 * _P['lam'] * (x.sum() - n * 0.5)
    return g

def opt_smax(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    bds = [(0.0, 1.0)] * n
    _P['n'] = n
    for a, lm in [(50, 200), (200, 100), (1000, 50), (5000, 20), (20000, 10), (80000, 5)]:
        if time.time() - t0 > lim:
            break
        _P['a'], _P['lam'] = float(a), float(lm)
        try:
            r = minimize(obj_fn, bh.copy(), jac=jac_fn, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-15})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def minimax_alt(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    _P['n'] = n
    bds = [(0.0, 1.0)] * n
    for eps in [2.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:

        p = np.exp(c_arr / eps)

            try:

                             bounds=bds, options={'maxiter': 250, 'ftol': 1e-17, 'gtol': 1e-14})
                h = repair(r.x, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
                c_arr = corr(h)
