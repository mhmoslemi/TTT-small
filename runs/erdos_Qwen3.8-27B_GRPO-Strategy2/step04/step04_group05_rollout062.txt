    c = np.correlate(h, 1-h, mode='full')
    top = np.sort(c)[::-1][:M]
    return np.sum(top**p) / M

def topm_grad(h, M=5, p=2):
    c = np.correlate(h, 1-h, mode='full')
    top_idx = np.argsort(c)[::-1][:M]
    g = np.zeros(len(h))
    for k in top_idx:
        c_k = c[k]
        g += p * c_k**(p-1) * grad1(h, k, n) / M
    return g

import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 1:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def _sm_obj(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / a + lam * (x.sum() - n * 0.5) ** 2)

def _sm_grad(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-12:
            g += pw[k] * grad1(x, k, n)
    g /= a
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _tm_obj(x):
    n, M, p, lam = _P['n'], _P['M'], _P['p'], _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    top = np.sort(c)[::-1][:M]
    return float(np.sum(np.maximum(top, 0.0) ** p) / M + lam * (x.sum() - n * 0.5) ** 2)

def _tm_grad(x):
    n, M, p, lam = _P['n'], _P['M'], _P['p'], _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(c)[::-1][:M]
    g = np.zeros(n)
    for ki in idx:
        ck = max(c[ki], 0.0)
        w = p * ck ** (p - 1) / M if p > 1 else 1.0 / M
        g += w * grad1(x, ki, n)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def opt_lbfgs(h, n, dx, t0, lim, which='sm', max_iter=300):
    bh, bc = h.copy(), c5v(h, dx)
    bds = [(0.0, 1.0)] * n
    _P['n'] = n
    if which == 'sm':
        for a, lam in [(100, 50), (500, 100), (2000, 200), (10000, 500),
                       (50000, 1000), (200000, 2000),(500000, 3000)]:
            if time.time() - t0 > lim:
                break
            _P['a'], _P['lam'] = float(a), float(lam)
            try:
                r = minimize(_sm_obj, bh.copy(), jac=_sm_grad, method='L-BFGS-B',
                            bounds=bds, options={'maxiter': max_iter, 'ftol': 1e-19})
                hc = repair(r.x, n)
                c = c5v(hc, dx)
                if c < bc:
                    bc, bh = c, hc
            except Exception:
                pass
    elif which == 'tm':
        for M, p, lam in [(3, 8, 100), (5, 16, 200), (8, 8, 200), (5, 32, 500),
                          (10, 16, 500), (5, 64, 1000), (3, 64, 2000), (15, 32, 2000)]:
            if time.time() - t0 > lim:
                break
            _P['M'], _P['p'], _P['lam'] = M, p, lam
            try:
                r = minimize(_tm_obj, bh.copy(), jac=_tm_grad, method='L-BFGS-B',
                            bounds=bds, options={'maxiter': max_iter, 'ftol': 1e-19})
                hc = repair(r.x, n)
                c = c5v(hc, dx)
                if c < bc:
                    bc, bh = c, hc
            except Exception:
                pass
    return bh, bc

        cand = 0.5 + 0.35 * np.sin(3.14159 * k1 * x / 2 + ph1) + \

    for cand in cands:

def opt_cd(h, n, dx, rng, t0, lim, cnt):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.choice([0.001, 0.003, 0.008, 0.015, 0.03,
                        -0.001, -0.003, -0.008, -0.015, -0.03])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

            c = c5v(h, dx)

                cc = c5v(h, dx)
                if cc < bc:

            else:

def opt_activity(h, n, dx, t0, lim):

        t_var = cp.Variable()

import numpy as np
from scipy.optimize import minimize, linprog
import time

_P = {}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14: break
            a = min(d, 1.0-h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n-1)
    if L == 0:
        g[:] = 1.0 - 2.0*h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)
    s = np.clip(a*(c-m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s)))/a + lam*(x.sum()-n*0.5)**2)

def _sm_grad(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = np.correlate(x, 1.0-x, mode='full')
    m = np.max(c)

    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, linprog

_P = {}

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def _sm_obj(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / a + lam * (x.sum() - n * 0.5) ** 2)

def _sm_grad(x):
    n, a, lam = _P['n'], _P['a'], _P['lam']
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-12:
            g += pw[k] * grad1(x, k, n)
    g /= a
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _tm_obj(x):
    n, M, p, lam = _P['n'], _P['M'], _P['p'], _P['lam']

    top = np.sort(c)[::-1][:M]
    return float(np.sum(top ** p) / M + lam * (x.sum() - n * 0.5) ** 2)

    idx = np.argsort(c)[::-1][:M]

        ck = c[ki]
        w = p * ck ** (p - 1) / M if p > 1 else 1.0 / M
        g += w * grad1(x, ki, n)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g
