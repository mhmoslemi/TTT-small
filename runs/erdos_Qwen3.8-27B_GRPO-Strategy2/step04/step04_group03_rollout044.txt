```python
def _project(h, n):
    """Project onto {h: 0≤h_i≤1, sum(h)=n/2}."""
    target = n / 2.0
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    # Bisection for λ
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_at_lambda(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    # Clip and fix sum
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Proportional scaling
    if s > 1e-10:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix residual with distribution
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0, 1)

def _grad_max(h, n, alpha=0):
    """Gradient of soft-max of correlation values."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if alpha > 0:
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
    else:
        w = np.zeros(len(c))
        w[c >= m - 1e-6] = 1.0
        w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += w[k] * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= w[k] * h[i2 + lag:j2 + lag]
    return g

def _pgd_step(h, n, eta, alpha, rng, t_end):
    """One step of projected subgradient descent with backtracking."""
    g = _grad_max(h, n, alpha)
    c5_cur = _eval(h)
    # Backtracking line search
    for bt in range(15):
        h_new = _project(h - eta * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5_cur - 1e-15:
            return h_new, c5_new, True
        eta *= 0.5
    return h, c5_cur, False

def _pjd_opt(h, n, rng, t_end, alpha=5000):

def _gs(x, alpha):

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()

    w = np.exp(s - s.max())
    w /= w.sum()

        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
```

```python
            break
        if n2 == n:
            res_list.append((n, best_h.copy()))
            continue
        h2 = _interp(best_h, best_n, n2)
        h2 = _repair(h2, n2)
        # Also try from scratch for small n
        if n2 <= 120:
            h_alt = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n2) / n2 * 2)
            h_alt = _repair(h_alt, n2)
            if _eval(h_alt) < _eval(h2):
                h2 = h_alt
        h2 = _opt_slsqp(h2, n2, 50000, maxiter=250)
        h2 = _opt_slsqp(h2, n2, 200000, maxiter=200)
        res_list.append((n2, h2.copy()))

    # Upscale best from each resolution back to n
    for n2, h2 in res_list:
        if _tl() < 40:
            break
        if n2 != n:
            h_up = _interp(h2, n2, n)
            h_up = _repair(h_up, n)
        else:
            h_up = h2
        c5_up = _eval(h_up)
        if c5_up < best_c5 - 1e-15:
            best_c5 = c5_up
            best_h = h_up.copy()

    # Phase 4: Simulated annealing local search
    if _tl() > 60:
        t_e = min(time.time() + 40, t0 + budget_s * 0.75)
        h_sa, c5_sa = _sa_local(best_h.copy(), best_n, rng, t_e)
        if c5_sa < best_c5 - 1e-15:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: SLSQP polish at high alpha
    if _tl() > 30:
        h_new = _opt_slsqp(best_h, best_n, 1000000, maxiter=500)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Another PGD pass
    if _tl() > 30:
        t_e = min(time.time() + 20, t0 + budget_s * 0.92)
        h_p2, c5_p2 = _pgd(best_h.copy(), best_n, rng, t_e, alpha=100000)
        if c5_p2 < best_c5 - 1e-15:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    # Phase 7: Fine local search
    if _tl() > 15:
        t_e = min(time.time() + 12, t0 + budget_s - 5)
        h_f, c5_f = _fine_search(best_h.copy(), best_n, rng, t_e)
        if c5_f < best_c5 - 1e-15:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```
