        # dc/dh[j] = [0<=j<=n-1-s] * (1-h[j+s]) - [s<=j<=n-1] * h[j-s]
        for j in range(0, n - s):
            g[j] += 1.0 - h[j + s]
        for j in range(s, n):
            g[j] -= h[j - s]

    """Subgradient of max_k c_k."""
    c = _corr(h)
    mx = c.max()
    thresh = mx - 0.01 * mx - 1e-8
    active = np.where(c >= thresh)[0]
    if len(active) > n_active:
        # Take the n_active largest
        idx = np.argsort(c[active])[::-1][:n_active]
        active = active[idx]
    g = np.zeros(len(h))
    for k in active:
        g += _grad_c_k(h, k)
    g /= len(active)

    tgt = n / 2.0
    best = _c5(h0, dx)
    h = h0.copy()
    y = h0.copy()
    t_val = 1.0

        L = 2.0
    
    for it in range(n_iter):
        if time.time() > t_end:

        g, mx, c = _grad_max(y, n_active)
        c_val = mx * dx
        if c_val < best - 1e-14:
            best = c_val
            h = y.copy()
        
        # Step
        h_new = np.clip(y - g / L, 0.0, 1.0)
        h_new = _project_quick(h_new, tgt)
        
        # FISTA momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = h_new + (t_val - 1.0) / t_next * (h_new - h)
        y = np.clip(y, -1.0, 2.0)
        t_val = t_next
        h = h_new
    
    h = _repair(h, tgt)
    best = min(best, _c5(h, dx))
    return h, best

def _project_quick(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-12:
        return h
    # Single correction
    if d > 0:
        mask = h < 0.999
    else:
        mask = h > 0.001
    if mask.sum() > 0:
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += d / max(mask.sum(), 1)
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _grad_c_k(h, k_index):
    n = len(h)
    shift = k_index - (n - 1)
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        s = shift
        for j in range(s, n):
            g[j] += 1.0 - h[j - s]
        for j in range(0, n - s):
            g[j] -= h[j + s]
    else:
        s = -shift
        for j in range(0, n - s):
            g[j] += 1.0 - h[j + s]
        for j in range(s, n):
            g[j] -= h[j - s]
    return g

def _proj_simplex(h, target):
    """Project onto {0<=h<=1, sum=h=target} using bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.exp(s).sum()) / alpha
