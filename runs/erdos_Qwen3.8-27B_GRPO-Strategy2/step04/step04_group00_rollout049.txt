import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr_grad_single(h, k, n):
    L = k - (n - 1)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _smooth_og(h, alpha, vpen):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(h)
    obj = m + np.log(ew.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = pw[k]
        if pk < 1e-15:
            continue
        g += pk * _corr_grad_single(h, k, n)
    g *= 1.0 / alpha
    if vpen > 0:
        cmean = c.mean()
        obj += vpen * np.var(c)
        for k in range(2 * n - 1):
            w = 2.0 * vpen * (c[k] - cmean) / (2 * n - 1)
            if abs(w) > 1e-15:
                g += w * _corr_grad_single(h, k, n)
    return obj, g

def _opt_lbygbs(h, n, bounds, alpha_list, vpen_vals, t_limit):
    bh = h.copy()
    bc5 = None
    dx = 2.0 / n
    for alpha, vpen in zip(alpha_list, vpen_vals):
        if time.time() > t_limit:
            break
        try:
            def _o(x, al=alpha, vp=vpen):
                o, _ = _smooth_og(x, al, vp)
                return o
            def _g(x, al=alpha, vp=vpen):
                _, gg = _smooth_og(x, al, vp)
                return gg
            res = minimize(_o, bh.copy(), jac=_g, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-15})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if bc5 is None or c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass
    return bh, bc5

def _opt_nesterov(h, n, dx, t_limit, alphas):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    for alpha in alphas:
        h_cur = bh.copy()
        c5c = _c5(h_cur, dx)
        v = np.zeros(n)
        mu = 0.85
        step = 0.008
        for it in range(600):
            if time.time() > t_limit:
                break
            _, g = _smooth_og(h_cur, alpha, 0.0)
            g -= g.mean()
            ng = np.linalg.norm(g)
            if ng < 1e-14:
                break
            g /= ng
            y = np.clip(h_cur + mu * v, 0.0, 1.0)
            _, gy = _smooth_og(y, alpha, 0.0)
            gy -= gy.mean()
            v = mu * v + 0.15 * gy
            h_new = _repair(y - step * v, n)
            c_new = _c5(h_new, dx)
            if c_new < c5c - 1e-15:
                h_cur = h_new
                c5c = c_new
                if c5c < bc5:
                    bc5 = c5c
                    bh = h_cur.copy()
                step = min(step * 1.08, 0.04)
            else:
                step *= 0.6
                if step < 1e-9:
                    break
        mu = min(mu * 1.02, 0.95)
    return bh, bc5

def _opt_subgrad(h, n, dx, t_limit):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    h_cur = bh.copy()
    c5c = bc5
    step = 0.012
    for it in range(3000):
        if time.time() > t_limit:
            break
        c = _corr(h_cur)
        cm, mn = c.max(), c.min()
        spread = cm - mn + 1e-15
        thresh = mn + 0.65 * spread
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        g = np.zeros(n)
        for k in active:
            w = (c[k] - thresh + 0.05) ** 2
            g += w * _corr_grad_single(h_cur, k, n)
        g /= len(active)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-13:
            break
        g /= ng
        found = False
        for sf in [1.0, 0.5, 0.25, 0.12, 0.06]:
            h_new = _repair(h_cur - step * sf * g, n)
            c_new = _c5(h_new, dx)
            if c_new < c5c - 1e-16:
                h_cur = h_new
                c5c = c_new
                if c5c < bc5:
                    bc5 = c5c
                    bh = h_cur.copy()
                found = True
                break
        if found:
            step = min(step * 1.04, 0.05)
        else:
            step *= 0.45
            if step < 1e-9:
                break
    return bh, bc5

def _opt_local(h, n, dx, t_limit, rng):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    h_cur = bh.copy()
    c5c = bc5
    dlist = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                      -0.001, -0.003, -0.008, -0.02, -0.05])
    for sweep in range(400):
        if time.time() > t_limit:
            break
        c = _corr(h_cur)
        cm = c.max()
        spread = cm - c.min() + 1e-15
        top = np.where(c > cm - 0.05 * spread)[0]
        g = np.zeros(n)
        for k in top:
            g += _corr_grad_single(h_cur, k, n)
        g /= len(top)
        order = np.argsort(np.abs(g))[::-1][:60]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 12, len(order))):
                j = order[jj]
                best_d, best_c = 0.0, c5c
                for d in dlist:
                    ni, nj = h_cur[i] + d, h_cur[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h_cur[i], h_cur[j] = ni, nj
                        cc = _c5(h_cur, dx)
                        if cc < best_c:
                            best_c, best_d = cc, d
                        h_cur[i], h_cur[j] = ni - d, nj + d
                if best_d != 0:
                    h_cur[i] += best_d
                    h_cur[j] -= best_d
                    h_cur = _repair(h_cur, n)
                    c5c = _c5(h_cur, dx)
                    improved = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h_cur.copy()
                    break
            if improved:
                break
        if not improved:
            break
    # Random fine-tune
    h_cur = bh.copy()
    c5c = _c5(h_cur, dx)
    remaining = t_limit - time.time()
    n_rand = min(40000, int(max(0, remaining) * 5000))
    for _ in range(n_rand):
        if time.time() > t_limit:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.003, 0.003)
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_cur[i], h_cur[j] = ni, nj
            cc = _c5(h_cur, dx)
            if cc < c5c:
                c5c = cc
                if c5c < bc5:
                    bc5 = c5c
                    bh = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = ni - d, nj + d
    return bh, bc5
