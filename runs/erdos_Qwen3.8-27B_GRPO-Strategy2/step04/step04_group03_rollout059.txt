Actually, I want to make sure the gradient is correct. Let me re-derive:

We want to minimize f(h) = log(sum(exp(alpha * (c_k - m)))) / alpha + m
where m = max(c) and c_k are the correlation values.

The gradient of f w.r.t. h is:
df/dh = (1/alpha) * d/dh log(sum_k exp(alpha * c_k - alpha*m))
= (1/alpha) * (sum_k alpha * exp(alpha * c_k - alpha*m) * dc_k/dh) / (sum_k exp(alpha * c_k - alpha*m))
= sum_k w_k * dc_k/dh
where w_k = exp(alpha*(c_k - m)) / sum_j exp(alpha*(c_j - m))

And dc_L/dh[j] = (1-h[j+L]) * [0<=j+L<n] - h[j-L] * [0<=j-L<n] (as derived earlier)

So the gradient is a weighted sum of the individual lag gradients. The weights w_k form a probability distribution over lags.

For efficiency, I can pre-identify which lags have significant weight (w_k > 1e-10) and only iterate over those.

Let me code the gradient:

```python
def _gs(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        if w[k] < 1e-12:
            continue
        L = k - (n - 1)
        # First term: (1-h[j+L]) for j where 0<=j<n and 0<=j+L<n
        j1s = max(0, -L)
        j1e = min(n, n - L)
        if j1s < j1e:
            g[j1s:j1e] += w[k] * (1.0 - h[j1s+L:j1e+L])
        # Second term: -h[j-L] for j where 0<=j-L<n and 0<=j<n
        j2s = max(0, L)
        j2e = min(n, n + L)
        if j2s < j2e:
            g[j2s:j2e] -= w[k] * h[j2s-L:j2e-L]
    return g
```

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
                for idx in order:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                    if abs(diff) < 1e-12: break
            else:
                order = np.argsort(h)
                for idx in order:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                    if abs(diff) < 1e-12: break
    return h

    return np.ones(n)

def _slsqp(h, n, alpha, maxiter, t_end):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'disp': False})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cyc_cd(h, n, t_end, step_multipliers=None):
    c5c = _eval(h)
    if step_multipliers is None:
        step_multipliers = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    delta_list = []
    for sm in step_multipliers:
        delta_list.extend([sm, -sm])
    no_impr = 0
    for cycle in range(100000):
        if time.time() > t_end:
            break
        any_imp = False
        for i in range(n):
            j = (i + n // 2) % n
            if j == i:
                j = (i + 1) % n
            best_d = 0.0
            best_v = c5c
            for d in delta_list:
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                v = _eval(h)
                if v < best_v:
                    best_v = v; best_d = d
                h[i], h[j] = ni - d, nj + d
            if best_d != 0.0:
                h[i] += best_d; h[j] -= best_d
                c5c = best_v; any_imp = True
        if not any_imp:
            no_impr += 1
            if no_impr >= 2:
                break
    return h, c5c

def _rand_local(h, n, rng, t_end):
    c5c = _eval(h)
    scale = 0.015
    no_imp = 0
    for _ in range(5000000):
        if time.time() > t_end:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n; no_imp = 0
                scale = min(scale * 1.01, 0.04)
            else:
                h[i], h[j] = ni - d, nj + d
                no_imp += 1
                if no_imp > 800:
                    scale *= 0.8; no_imp = 0
                    if scale < 1e-7:
                        break
    return h, c5c

def _block_move(h, n, rng, t_end):
    c5c = _eval(h)
    no_imp = 0
    for _ in range(200000):
        if time.time() > t_end:
            break
        a = rng.randint(n); b = rng.randint(n)
        if abs(a - b) < 2:
            continue
        a, b = min(a, b), max(a, b)
        blk = np.arange(a, b + 1)
        d = 0.01 * rng.choice([1, -1]) * rng.choice([0.5, 1.0, 2.0])
        trial = h.copy()
        trial[blk] += d
        if trial.min() < -1e-12 or trial.max() > 1 + 1e-12:
            continue
        comp = -d * len(blk)
        others = np.setdiff1d(np.arange(n), blk)
        idx = others[rng.randint(len(others))]
        trial[idx] += comp
        if trial.min() < -1e-12 or trial.max() > 1 + 1e-12:
            continue
        c5n = _eval(trial)
        if c5n < c5c - 1e-16:
            h = trial; c5c = c5n; no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300:
                break
    return h, c5c

def _optimize_n(h_n, n, t_end):
    # Phase 1: SLSQP with smooth max
    if time.time() < t_end - 60:
        h_n = _slsqp(h_n, n, 50000, 300, time.time())
        h_n = _repair(h_n, n)
    # Phase 2: SLSQP with higher alpha
    if time.time() < t_end - 40:
        h_n = _slsqp(h_n, n, 300000, 200, time.time())
        h_n = _repair(h_n, n)
    # Phase 3: Cyclic CD
    if time.time() < t_end - 20:
        h_n, _ = _cyc_cd(h_n, n, min(time.time() + 25, t_end - 15))
    # Phase 4: Random local search
    if time.time() < t_end - 5:
        hat = np.random.RandomState(12345 + n)
        h_n, _ = _rand_local(h_n, n, hat, min(time.time() + 20, t_end - 5))
    # Phase 5: Fine CD
    if time.time() < t_end - 2:
        h_n, _ = _cyc_cd(h_n, n, t_end - 1)
    h_n = _repair(h_n, n)
    return h_n, _eval(h_n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = _eval(h_init)
    best_n = n_init

    n_list = [n_init]
    for cand in [80, 100, 120, 60]:
        if cand != n_init and cand not in n_list:
            n_list.append(cand)
    n_list.sort()

    per_n_budget = min(180, (budget_s - 120) // len(n_list))

    for n in n_list:
        elapsed = time.time() - t0
        if elapsed > budget_s - 80:
            break
        if n == n_init:
            h_n = h_init.copy()
        else:
            h_n = _repair(_interp(h_init, n_init, n), n)
        # Add small perturbation for diversity
        h_n = _repair(h_n + 0.02 * rng.randn(n), n)
        t_end_n = t0 + elapsed + per_n_budget
        h_n, c5n = _optimize_n(h_n, n, t_end_n)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_n.copy(); best_n = n
        # Also try without perturbation
        if n != n_init and time.time() - t0 < budget_s - 80:
            h_n2 = _repair(_interp(h_init, n_init, n), n)
            t_end_n2 = min(time.time() + per_n_budget * 0.6, t0 + budget_s - 80)
            h_n2, c5n2 = _optimize_n(h_n2, n, t_end_n2)
            if c5n2 < best_c5:
                best_c5 = c5n2; best_h = h_n2.copy(); best_n = n

    # Final intensive polish on best solution
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        hat = np.random.RandomState(seed + 999)
        t_end = time.time() + min(remaining - 10, 60)
        best_h, c5c = _rand_local(best_h, best_n, hat, t_end)
        if c5c < best_c5:
            best_c5 = c5c
        # Fine CD
        t_end = time.time() + min(remaining - 15, 20)
        best_h, c5c = _cyc_cd(best_h, best_n, t_end)
        if c5c < best_c5:
            best_c5 = c5c
    # Tiny steps
    if time.time() - t0 < budget_s - 5:
        best_h = best_h.copy()
        c5c = _eval(best_h)
        for _ in range(200000):
            if time.time() > t0 + budget_s - 2:
                break
            i = rng.randint(best_n); j = rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-5 * rng.choice([1, 2, 5, 10, 20])
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                c5n = _eval(best_h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                else:
                    best_h[i], best_h[j] = ni - d, nj + d
        best_c5 = min(best_c5, c5c)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
