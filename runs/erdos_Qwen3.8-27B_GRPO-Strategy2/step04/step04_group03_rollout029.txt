def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

**Better approach: Active-set identification + exact optimization on active set**

After getting a good approximate solution:
1. Identify the k values that achieve (or nearly achieve) the maximum
2. For those specific k values, the problem becomes: minimize max(c_{k1}, c_{k2}, ...) 
3. At the optimum, c_{k1} = c_{k2} = ... = c_max
4. This gives us a system of equations to solve

**Newton-style active set method:**
- Let S be the set of active k values (|S| = m)
- We want: c_{k1}(h) = c_{k2}(h) = ... = c_{km}(h) = t, minimize t
- This gives m equations in n+m-1 unknowns (h has n components with 1 equality constraint, plus t)
- We can use a generalized approach: minimize t subject to c_k(h) ≤ t for k in S

But c_k(h) is quadratic in h, so this is a QP-like problem (though non-convex due to the max).

Actually, for a fixed active set, we can do:
- Minimize t + epsilon * sum_{k in S} (c_k(h) - t)²
- This smooths the constraint and we can use gradient descent

The gradient of this w.r.t h:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp_to(hs, ns, nd):
    xs = np.linspace(0, 2, ns, endpoint=False)
    xd = np.linspace(0, 2, nd, endpoint=False)
    return np.interp(xd, xs, hs)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), 160)
    c5_p = _eval(h_parent)
    best_h, best_c5, best_n = h_parent.copy(), c5_p, 160

    def _tl():
        return budget_s - (time.time() - t0)

    # Phase 1: High-alpha SLSQP from parent (conservative, high alpha first)
    for alpha in [5e5, 2e6, 1e7]:
        if _tl() < 300:
            break
        for r in range(2):
            if _tl() < 200:
                break
            h_try = best_h.copy()
            if r == 1:
                h_try = h_try + 0.002 * rng.randn(160)
                h_try = _repair(h_try, 160)
            h_new = _slsqp(h_try, 160, alpha, maxiter=250, ftol=1e-22)
            c5_new = _eval(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5, best_h = c5_new, h_new

    # Phase 2: Free-parameter local search (only modify interior values)
    if _tl() > 80:
        t_end = min(time.time() + 50, t0 + budget_s - 30)
        h2, c5_2 = _free_ls(best_h.copy(), best_n, rng, t_end)
        if c5_2 < best_c5 - 1e-14:
            best_c5, best_h = c5_2, h2

    # Phase 3: Try n=80 and n=120 (different resolutions may find different optima)
    for n_try in [80, 120]:
        if _tl() < 160:
            break
        h_try = _repair(_interp_to(h_parent, 160, n_try), n_try)
        for alpha in [1e5, 1e6, 5e6]:
            if _tl() < 80:
                break
            h_new = _slsqp(h_try, n_try, alpha, maxiter=350, ftol=1e-22)
            c5_new = _eval(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5, best_h, best_n = c5_new, h_new, n_try

    # Phase 4: Re-converge at best_n with highest alpha
    if _tl() > 50:
        h_new = _slsqp(best_h.copy(), best_n, 5e7, maxiter=300, ftol=1e-22)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5, best_h = c5_new, h_new

    # Phase 5: Another free-parameter pass
    if _tl() > 50:
        t_end = min(time.time() + 35, t0 + budget_s - 20)
        h3, c5_3 = _free_ls(best_h.copy(), best_n, rng, t_end)
        if c5_3 < best_c5 - 1e-14:
            best_c5, best_h = c5_3, h3

    # Phase 6: Tiny perturbations (multi-scale)
    if _tl() > 20:
        for bs in [2e-5, 5e-6, 1e-6, 3e-7]:
            if _tl() < 8:
                break
            t_end = min(time.time() + 12, t0 + budget_s - 4)
            h4, c5_4 = _tiny_perturb(best_h.copy(), best_n, rng, t_end, bs)
            if c5_4 < best_c5 - 1e-18:
                best_c5, best_h = c5_4, h4

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
