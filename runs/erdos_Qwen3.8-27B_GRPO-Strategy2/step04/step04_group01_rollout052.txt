def _grad_col(h, j, n):
    """Gradient of all c[k] (full correlation) w.r.t. h[j]."""
    g = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):

            if tau <= j <= n - 1:  # h[j] appears as h[n+tau] with n=j-tau
                g[k] += 1.0 - h[j - tau]

                g[k] += 1.0 - h[j + s]
            if s <= j:  # h[j] appears as h[n] with n=j
                g[k] -= h[j - s]
    return g
