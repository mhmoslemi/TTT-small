    h = np.clip(h, 0.0, 1.0).copy()

from functools import partial
import time

_sum_n = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _correlate(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _correlate(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, ftol):
    global _sum_n
    _sum_n = len(h)
    n = _sum_n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(_sum_fun, n=n),
             'jac': partial(_sum_jac, n=n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _shift_grad(h, sh, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _worst_descent(h, nsteps, step0, topk):
    n = len(h)
    c5c = _eval(h)
    for _ in range(nsteps):
        c = _correlate(h)
        top = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for kw in top:
            sh = (n - 1) - kw
            g += _shift_grad(h, sh, n)
        g /= len(top)
        moved = False
        for step in [step0, step0*0.5, step0*0.25, step0*0.1, step0*0.05, step0*0.01]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                moved = True
                break
        if not moved:
            break
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i*factor:(i+1)*factor] = h[i]
    return h_hi

def _downsample(h, n_low):
    n = len(h)
    h_low = np.zeros(n_low)
    for i in range(n_low):
        idx = int(i * n / n_low)
        h_low[i] = h[idx]
    return _repair(h_low, n_low)

def _perturb(h, n, rng, mode):
    if mode == 0:
        return _repair(h + 0.02 * rng.randn(n), n)
    if mode == 1:
        return _repair(1.0 - h + 0.01 * rng.randn(n), n)
    if mode == 2:
        hp = h.copy()
        nb = rng.randint(3, 6)
        for _ in range(nb):
            bs = rng.randint(5, max(6, n // 8))
            st = rng.randint(0, n - bs)
            hp[st:st+bs] += 0.05 * rng.randn(bs)
        return _repair(hp, n)
    if mode == 3:
        hp = h.copy()
        b1 = rng.randint(0, n // 2)
        b2 = rng.randint(n // 2, max(n // 2 + 1, n - 5))
        blen = rng.randint(5, min(20, max(6, (b2 - b1) // 2)))
        tmp = hp[b1:b1+blen].copy()
        hp[b1:b1+blen] = hp[b2:b2+blen]
        hp[b2:b2+blen] = tmp
        return _repair(hp, n)
    hp = h.copy()
    mid = n // 2
    hp[mid:] = hp[mid:][::-1] + 0.01 * rng.randn(max(1, n - mid))
    return _repair(hp, n)

    # Phase 1: SLSQP with increasing alpha at original resolution

        h_new = _slsqp(best_h, alpha, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5 - 1e-15:

            best_h = h_new.copy()

    # Phase 2: Targeted descent on worst shifts
    if _el() < budget_s - 350:
        h, c5c = _worst_descent(best_h.copy(), 300, 0.02, 1)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()
        h, c5c = _worst_descent(best_h.copy(), 300, 0.01, 3)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()

    # Phase 3: Resolution change (down then up for cleaner structure)
    if _el() < budget_s - 300:
        for n_ds in [90, 70, 50]:
            if _el() > budget_s - 300:
                break
            h_ds = _downsample(best_h, n_ds)
            for alpha in [5000, 50000]:
                if _el() > budget_s - 300:
                    break
                h_ds = _slsqp(h_ds, alpha, 2000, 1e-19)
            factor = n0 // n_ds
            if factor >= 2:
                h_up = _upsample(h_ds, factor)
                h_up = _repair(h_up, n0)
                c5 = _eval(h_up)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_up.copy()

    # Phase 4: Higher resolution
    n_hi = n0 * 2
    if _el() < budget_s - 250:
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n_hi)
        for alpha in [2000, 20000, 100000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp(h_hi, alpha, 2000, 1e-19)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n_hi
        if _el() < budget_s - 200:
            h_hi, c5c = _worst_descent(best_h.copy(), 200, 0.01, 2)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_hi.copy()

    # Phase 5: Multi-start
    n_cur = best_n
    if _el() < budget_s - 150:
        n_restarts = 40
        for _r in range(n_restarts):
            if _el() > budget_s - 150:
                break
            mode = rng.randint(5)
            hp = _perturb(best_h, n_cur, rng, mode)
            h_new = _slsqp(hp, 50000, 1500, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    if _el() < budget_s - 100:
        h, c5c = _worst_descent(best_h.copy(), 500, 0.015, 1)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()
        h, c5c = _worst_descent(best_h.copy(), 300, 0.008, 2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()
        h, c5c = _worst_descent(best_h.copy(), 300, 0.005, 4)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()

    # Phase 7: Fine pairwise search
    if _el() < budget_s - 50:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.003, 0.001, 0.0003, 0.0001]:
            if _el() > budget_s - 50:
                break
            for _it in range(12000):
                if _el() > budget_s - 50:
                    break
                i1 = rng.randint(nc)
                i2 = rng.randint(nc)
                if i1 == i2:
                    continue
                d = scale * rng.randn()
                ni1, ni2 = h[i1] + d, h[i2] - d
                if 0 <= ni1 <= 1 and 0 <= ni2 <= 1:
                    h[i1], h[i2] = ni1, ni2
                    c5n = _eval(h)
                    if c5n < c5c - 1e-16:
                        c5c = c5n
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_h = h.copy()
                    else:
                        h[i1], h[i2] = ni1 - d, ni2 + d

    # Phase 8: Final SLSQP polish
    if _el() < budget_s - 15:
        for alpha in [200000, 500000]:
            if _el() > budget_s - 15:
                break
            h = _slsqp(best_h, alpha, 5000, 1e-20)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 9: Ultra-fine
    if _el() < budget_s - 5:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(25000):
            if _el() > budget_s - 5:
                break
            i1 = rng.randint(nc)
            i2 = rng.randint(nc)
            if i1 == i2:
                continue
            d = 1e-5 * rng.randn()
            ni1, ni2 = h[i1] + d, h[i2] - d
            if 0 <= ni1 <= 1 and 0 <= ni2 <= 1:
                h[i1], h[i2] = ni1, ni2
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i1], h[i2] = h[i1] - d, h[i2] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
