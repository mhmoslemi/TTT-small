def _project(h, n):
    """Project h onto {x ∈ [0,1]^n : sum(x) = n/2}."""
    target = n / 2.0
    # Bisection on tau: sum(clip(h - tau, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

    c = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(c.max()) * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _to_binary(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def _project(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def _grad_smooth_max(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    ww = np.exp(s - s.max())
    ww /= ww.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if ww[k] < 1e-14:
            continue
        lag = k - (n - 1)
        wk = ww[k]
        i1_lo = max(0, lag)
        i1_hi = min(n, n + lag)
        if i1_lo < i1_hi:
            idx = np.arange(i1_lo, i1_hi)
            g[idx] += wk * (1.0 - h[idx - lag])
        i2_lo = max(0, -lag)
        i2_hi = min(n, n - lag)
        if i2_lo < i2_hi:
            idx = np.arange(i2_lo, i2_hi)
            g[idx] -= wk * h[idx + lag]
    return g

def _obj_smooth_max(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha
