def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    g = np.zeros(n)

            if _time_left() > 80:
                t_end = time.time() + min(50, _time_left() - 40)
                h2, c5_2 = _direct_minmax(best_h.copy(), best_n, t_end)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()

    # Phase 5: Binary push
    if _time_left() > 60:
        t_end = time.time() + min(40, _time_left() - 30)
        h5, c5_5 = _binary_push(best_h.copy(), best_n, rng, t_end)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Local search
    if _time_left() > 40:
        t_end = time.time() + min(30, _time_left() - 20)
        h6, c5_6 = _local_search(best_h.copy(), best_n, rng, t_end)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final SLSQP polish
    if _time_left() > 30:
        h7 = _slsqp(best_h.copy(), best_n, 2000000, maxiter=400)
        c5_7 = _eval(h7)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Tiny perturbations
    if _time_left() > 8:
        h8 = best_h.copy()
        c5c = _eval(h8)
        cutoff = t0 + budget_s - 2
        for _ in range(50000):
            if time.time() > cutoff:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.choice([1, 2, 3, 5, 8]) * 4e-6
            ni, nj = h8[i] + d, h8[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h8.copy()
                else:
                    h8[i], h8[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
