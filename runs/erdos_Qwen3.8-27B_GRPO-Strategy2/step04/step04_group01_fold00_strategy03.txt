           target = N/2
           current_sum = s = np.sum(h)
           s_scale = target/current_sum
           h_adjusted = np.clip(h*s_scale, 0, 1)
           # Validate integral explicitly
           h_adjusted /= (np.sum(h_adjusted)/target)
           return h_adjusted
       return h.clip(0,1)
