**Augmented Lagrangian + active set**:
- Start with the current solution
- Identify the active set (top K lags)
- Solve a QP (or use Newton) to minimize the sum of squares of active c_k, subject to sum(h)=n/2, 0≤h≤1

But this requires handling the bilinear terms, which makes it non-convex. However, near a local optimum, the bilinear terms are approximately linear, so a fixed-point iteration might work.

Alternatively, **sequential quadratic programming (SQP)** idea:
- At the current h, linearize c_k around h
- Solve the resulting LP/QP: min T s.t. c_k^lin ≤ T, sum(h)=n/2, 0≤h≤1
- Update h
- Repeat

The linearization of c_k = ∑ h[i](1-h[i+k]) around h is:
c_k ≈ c_k(h) + ∑ (∂c_k/∂h[i])(δh[i]) = c_k(h) + ∑ g_k[i] δh[i]

This gives a linear constraint! So the subproblem is:
min T
s.t. c_k(h) + g_k·δ ≤ T for all active k
     ∑ h[i] + ∑ δ[i] = n/2
     0 ≤ h[i] + δ[i] ≤ 1

This is an LP! I can solve it with scipy.optimize.linprog.

from scipy.optimize import linprog

def _sqp_step(h, n, top_k=30, thresh_frac=0.1):
    """One SQP step: linearize and solve LP."""
    c = _corr(h)
    cm = np.max(c)
    # Active set: lags with c_k within thresh_frac of max
    thresh = cm - thresh_frac * (cm - np.min(c))
    active = np.where(c >= thresh)[0]
    if len(active) > top_k:
        # Keep only top_k
        active = np.argsort(c)[::-1][:top_k]
    
    # Build LP: variables are delta[0..n-1] and T
    # min T
    # s.t. g_k · delta <= T - c_k(h)  for k in active
    #      sum(delta) = 0  (to maintain constraint)
    #      -h[i] <= delta[i] <= 1-h[i]
    
    m_act = len(active)
    # Objective: minimize T -> c_obj = [0,...,0, 1] (T is last variable)
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    # Inequality constraints: g_k · delta - T <= -c_k(h)
    # i.e., [g_k, -1] · [delta, T] <= -c_k(h)
    A_ub = np.zeros((m_act, n+1))
    b_ub = np.zeros(m_act)
    for ii, k in enumerate(active):
        L = k - (n-1)
        A_ub[ii, :n] = _lag_grad(h, L, n)
        A_ub[ii, n] = -1.0
        b_ub[ii] = -c[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] <= delta[i] <= 1-h[i], T is unbounded (or 0 <= T)
    bounds_list = [(-h[i], 1.0-h[i]) for i in range(n)]
    bounds_list.append((0, None))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_list, method='highs')
    
    if res.status != 0:
        return None
    delta = res.x[:n]
    T = res.x[n]
    return delta, T

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lag_grad(h, L, n):
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

def _softobj(x, alpha, lam):
    c = _corr(x); m = np.max(c)
    s = np.clip(alpha*(c-m), -500, 500); s -= np.max(s)
    return m + np.log(np.sum(np.exp(s)))/alpha + lam*(np.sum(x)-len(x)/2.0)**2

def _softgrad(x, alpha, lam):
    c = _corr(x); m = np.max(c)
    s = np.clip(alpha*(c-m), -500, 500); s -= np.max(s)
    ew = np.exp(s); pw = ew/ew.sum(); n = len(x)
    g = np.zeros(n)

        g += pw[k]*_lag_grad(x, k-(n-1), n)
    g /= alpha
    g += 2.0*lam*(np.sum(x)-n/2.0)
    return g

        A_ub[ii, n] = -1.0
        b_ub[ii] = -c[k]
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bds = [(-h[i], 1.0-h[i]) for i in range(n)] + [(0, None)]
