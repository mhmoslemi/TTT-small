np.correlate(a, v, mode='full') where a and v have length n:
result[m] = sum_k a[k] * v[k+m-(n-1)] for k where indices are valid

So result[m] = sum_k a[k] * v[k+m-(n-1)]

In our case: a = h, v = 1-h
result[m] = sum_k h[k] * (1-h[k+m-(n-1)])

Let lag = m - (n-1). Then:
result[m] = sum_k h[k] * (1-h[k+lag])

where k and k+lag must both be in [0, n-1].

For lag > 0: k in [0, n-1-lag], so n-lag terms
For lag < 0: k in [-lag, n-1], so n+lag terms (= n-|lag| terms)
For lag = 0: all n terms

- h[j] appears as h[k] when k=j: term (1-h[j+lag])
- h[j] appears in (1-h[k+lag]) when k+lag=j, i.e., k=j-lag: term -h[j-lag]

For lag > 0:
- First term valid when j+lag < n, i.e., j < n-lag: contribution (1-h[j+lag])
- Second term valid when j-lag >= 0, i.e., j >= lag: contribution -h[j-lag]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        h = h * (target / max(s, 1e-15))
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def do_lbfgs(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        r = minimize(lse_obj, h, args=(alpha, target, pen), method='L-BFGS-B',
                    bounds=[(0, 1)] * n,
                    options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        hc = repair(r.x, n)
        c5, _ = compute_c5(hc, dx)
        return hc, c5
    except Exception:
        c5, _ = compute_c5(h, dx)
        return h, c5

def do_subgrad(h, n, deadline, top_m=5):
    dx = 2.0 / n
    h = h.copy()
    best_h, best_c5 = h.copy(), 1e9
    for it in range(1000000):
        if time.time() > deadline:
            break
        c5, corr = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        t = min(top_m, 2 * n - 1)
        top = np.argsort(corr)[-t:]
        lr = min(0.8, 0.5 / (1.0 + 0.0005 * it))
        grad = np.zeros(n)
        for ki in top:
            k = ki - (n - 1)
            w = 1.0 / t
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            elif k > 0:
                grad[:n - k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[:n - k]
            else:
                k2 = -k
                grad[k2:] += w * (1.0 - h[:n - k2])
                grad[:n - k2] -= w * h[k2:]
        h = repair(np.clip(h - lr * grad, 0.0, 1.0), n)
    return best_h, best_c5

def do_fine(h, n, dx, rng, deadline):
    h = h.copy()
    c5, _ = compute_c5(h, dx)
    best_h, best_c5 = h.copy(), c5
    no_impr = 0
    while time.time() < deadline:
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        s = 0.003 if no_impr < 5000 else (0.001 if no_impr < 20000 else 0.0004)
        d = s * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn, _ = compute_c5(h, dx)
        if cn < best_c5 - 1e-14:
            best_c5, best_h, no_impr = cn, h.copy(), 0
        else:
            h[i] -= d; h[j] += d; no_impr += 1
        if no_impr > 80000:
            break
    return best_h, best_c5

def do_block(h, n, dx, rng, deadline):
    h = h.copy()
    c5, _ = compute_c5(h, dx)
    best_h, best_c5 = h.copy(), c5
    no_impr = 0
    while time.time() < deadline:
        bs = rng.randint(2, 6)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(-0.012, 0.012)
        if np.any(h[i:i+bs] + eps < -1e-9) or np.any(h[i:i+bs] + eps > 1 + 1e-9):
            continue
        if np.any(h[j:j+bs] - eps < -1e-9) or np.any(h[j:j+bs] - eps > 1 + 1e-9):
            continue
        h[i:i+bs] += eps; h[j:j+bs] -= eps
        cn, _ = compute_c5(h, dx)
        if cn < best_c5 - 1e-14:
            best_c5, best_h, no_impr = cn, h.copy(), 0
        else:
            h[i:i+bs] -= eps; h[j:j+bs] += eps; no_impr += 1
        if no_impr > 40000:
            break
    return best_h, best_c5

def coord_sweep(h, n, dx, deadline):
    h = h.copy()
    c5, _ = compute_c5(h, dx)
    best_h, best_c5 = h.copy(), c5
    for _ in range(500):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            orig = h[i]
            for trial in [orig * 0.5, orig * 1.5, 0.0, 1.0, orig - 0.02, orig + 0.02]:
                trial = np.clip(trial, 0.0, 1.0)
                if trial == orig:
                    continue
                comp = (orig - trial)
                if abs(comp) < 1e-15:
                    continue
                j = rng_randint(n)
                h[i] = trial
                h[j] = np.clip(h[j] - comp, 0.0, 1.0)
                if h.sum() != n / 2.0:
                    h = repair(h, n)
                cn, _ = compute_c5(h, dx)
                if cn < best_c5 - 1e-14:
                    best_c5, best_h = cn, h.copy()
                    break
            h = repair(h, n)
    return best_h, best_c5

def rng_randint(n):
    return np.random randint(n) if False else int(np.random.randint(n))
