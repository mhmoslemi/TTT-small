        others = np.setdiff1d(np.arange(n), np.arange(start, start+L))

            no_imp += 1
            if no_imp > 500:
                amp *= 0.8
                no_imp = 0

        if c5n < c5c - 1e-16:
            h = h_tmp
            c5c = c5n
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300:
                amp *= 0.85
                no_imp = 0
                if amp < 1e-5:
                    break
    return h, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    h70 = _repair(np.array(initial_h_values, dtype=float).copy(), 160)
    n = 160
    c5_init = _eval(h70)

    best_h = h70.copy()
    best_c5 = c5_init
    best_n = n

    # Phase 1: Quick SLSQP at moderate alpha
    for alpha in [5000, 30000]:
        if _tl() < 30:
            break
        h_new = _slsqp(h70, n, alpha, maxiter=300)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h70 = h_new.copy()

    # Phase 2: L-BFGS-B (faster than SLSQP for n=160)
    for alpha in [50000, 200000]:
        if _tl() < 40:
            break
        h_new = _lbgfsb(best_h.copy(), n, alpha, maxiter=600)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: PGM refinement
    if _tl() > 60:
        h_pgd, c5_pgd = _pgd_refine(best_h.copy(), n, time.time() + min(60, _tl() - 30))
        if c5_pgd < best_c5:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()

    # Phase 4: High-alpha SLSQP polish
    for alpha in [200000, 1000000]:
        if _tl() < 40:
            break
        h_new = _slsqp(best_h.copy(), n, alpha, maxiter=250)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: L-BFGS-B at very high alpha
    if _tl() > 40:
        h_new = _lbgfsb(best_h.copy(), n, 500000, maxiter=800)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: PGM again (different trajectory from new starting point)
    if _tl() > 80:
        h_pgd, c5_pgd = _pgd_refine(best_h.copy(), n, time.time() + min(70, _tl() - 40))
        if c5_pgd < best_c5:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()

    # Phase 7: Block local search
    if _tl() > 50:
        t_end = time.time() + min(40, _tl() - 20)
        h_blk, c5_blk = _block_search(best_h.copy(), n, rng, t_end)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h_blk.copy()

    # Phase 8: Pairwise local search (very fine)
    if _tl() > 40:
        t_end = time.time() + min(30, _tl() - 15)
        h_pr, c5_pr = _pair_search(best_h.copy(), n, rng, t_end)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr.copy()

    # Phase 9: Try n=200
    if _tl() > 80:
        n2 = 200
        h200 = _repair(_interp(best_h, n, n2), n2)
        c5_200 = _eval(h200)
        h200 = _slsqp(h200, n2, 50000, maxiter=300)
        c5_200 = _eval(h200)
        if _tl() > 50:
            h200 = _lbgfsb(h200, n2, 200000, maxiter=500)
            c5_200 = _eval(h200)
        if _tl() > 30:
            h200, c5_200 = _pgd_refine(h200, n2, time.time() + min(30, _tl() - 15))
        if c5_200 < best_c5:
            best_c5 = c5_200
            best_h = h200.copy()
            best_n = n2

    # Phase 10: Final ultra-fine polish
    if _tl() > 30:
        h_new = _slsqp(best_h.copy(), best_n, 2000000, maxiter=400)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Tiny-step local search
    if _tl() > 15:
        h = best_h.copy()
        c5c = _eval(h)
        for _it in range(50000):
            if time.time() > t0 + budget_s - 3:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.choice([1, 2, 3, 5]) * 1e-6
            ni, NJ = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= NJ <= 1:
                h[i], h[j] = ni, NJ
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, NJ + d
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
