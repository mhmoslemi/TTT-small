def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        d = target - s
        if abs(d) < 1e-12:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

For a periodic binary sequence with period p, the autocorrelation is periodic. If we can find a sequence where the autocorrelation at all non-zero shifts is minimized, that would be ideal.

This is exactly the problem of finding a binary sequence with minimal aperiodic autocorrelation (or periodic autocorrelation if we consider the sequence circular).

For our problem, it's not quite circular (the "full" mode doesn't wrap around), but for large n, the boundary effects become less important.

Known results: For a binary sequence of length n with n/2 ones, the minimum possible maximum aperiodic autocorrelation relates to Little's conjecture and related results in sequence design.

= (number of 1s in h in the overlap region) - (number of positions where both are 1)

_positions where h is 1 AND h shifted by k is 0_

        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    n = best_n

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP refinement (20%)
    for alpha in [5000, 20000, 100000, 500000]:
        if elapsed() > budget_s * 0.20:
            break
        h_new = slsqp_opt(best_h.copy(), n, alpha, 1000)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5 = h_new.copy(), c5_new

    # Phase 2: Subgradient (10%)
    if elapsed() < budget_s * 0.30:
        h_sg, c5_sg = subgradient_opt(best_h, n, t0 + budget_s * 0.30, rng)
        if c5_sg < best_c5 - 1e-14:
            best_h, best_c5 = h_sg.copy(), c5_sg

    # Phase 3: Targeted SA - main search (45%)
    if elapsed() < budget_s * 0.75:
        t_sa = t0 + budget_s * 0.75
        h_sa, c5_sa = targeted_sa(best_h, n, t_sa, rng, max_block=6)
        if c5_sa < best_c5 - 1e-14:
            best_h, best_c5 = h_sa.copy(), c5_sa
        # Another SA round with different seed
        rng2 = np.random.RandomState(seed + 99999)
        h_sa2, c5_sa2 = targeted_sa(best_h, n, t_sa, rng2, max_block=5)
        if c5_sa2 < best_c5 - 1e-14:
            best_h, best_c5 = h_sa2.copy(), c5_sa2

    # Phase 4: SLSQP + fine CD polish (15%)
    if elapsed() < budget_s * 0.90:
        for alpha in [200000, 1000000]:
            if elapsed() > budget_s * 0.90:
                break
            h_p = slsqp_opt(best_h.copy(), n, alpha, 1200)
            c5_p = compute_c5(h_p, n)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_p.copy(), c5_p
        h_cd, c5_cd = fine_cd(best_h, n, t0 + budget_s * 0.90, rng)
        if c5_cd < best_c5 - 1e-14:
            best_h, best_c5 = h_cd.copy(), c5_cd

    # Phase 5: Try other n values (10%)
    if elapsed() < budget_s * 0.99:
        for n_alt in [48, 56, 80, 96, 128]:
            if elapsed() > budget_s * 0.99:
                break
            if n_alt == best_n:
                continue
            x_old = np.linspace(0.0, 1.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_alt, endpoint=False)
            h_seed = np.interp(x_new, x_old, best_h, period=None)
            h_seed = repair(h_seed, n_alt)
            h_alt = slsqp_opt(h_seed, n_alt, 100000, 600)
            c5_alt = compute_c5(h_alt, n_alt)
            if elapsed() < budget_s * 0.97:
                h_alt, c5_alt = targeted_sa(h_alt, n_alt, t0 + budget_s * 0.97,
                                            np.random.RandomState(seed + n_alt), max_block=5)
            if c5_alt < best_c5 - 1e-14:
                best_h, best_c5, best_n = h_alt.copy(), c5_alt, n_alt

    # Final repair and evaluation
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
