import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _project_sum(x, target):
    """Project x onto {0<=x<=1, sum(x)=target} using bisection on threshold."""
    x = np.clip(x, 0.0, 1.0).copy()
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)/2, 0.0, 1.0)
    # Fix residual
    d = target - y.sum()
    if abs(d) > 1e-12:
        # Distribute to non-saturated elements
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        if free.sum() > 0:
            y[free] += d / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def _soft_objective(h, alpha, target):
    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.exp(s).sum()) / alpha + 1e6 * (h.sum() - target) ** 2

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(lambda x: _soft_objective(x, alpha, tgt), h,
                     method='SLSQP', bounds=bds,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project_sum(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _worst_lag_descent(h, dx, max_iter, t_end):
    """Iteratively reduce the worst lags using projected gradient."""
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    m_top = 5  # number of worst lags to target
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        current_max = c.max()
        
        # Find worst lags
        top_idx = np.argsort(c)[-m_top:]
        
        # Compute combined negative gradient
        g = np.zeros(n)
        for k_abs in top_idx:
            # k_abs is index in the full correlation (length 2n-1)
            # Convert to actual lag
            k = k_abs - (n - 1)
            if k == 0:
                # c[0] = sum h[i](1-h[i]) - this gradient is 1-2h[i]
                g -= 1.0 - 2.0 * h
            elif k > 0:
                # c[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
                # grad: for j in [0, n-k): 1 - h[j+k]; for j in [k, n): minus h[j-k]
                g_grad = np.zeros(n)
                g_grad[:n-k] += 1.0
                g_grad[:n-k] -= h[k:]
                g_grad[k:] -= h[:n-k]
                g -= g_grad
            else:
                # k < 0, symmetric
                k2 = -k
                g_grad = np.zeros(n)
                g_grad[k2:] += 1.0
                g_grad[k2:] -= h[:n-k2]
                g_grad[:n-k2] -= h[k2:]
                g -= g_grad
        
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Projected line search
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002]:
            h_new = _project_sum(h + step * g, tgt)
            c_new = _c5(h_new, dx)
            if c_new < best:
                h = h_new
                best = c_new
                improved = True
                break
        if not improved:
            # Try smaller steps or break
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _project(x, target):
    x = np.clip(x, 0.0, 1.0).copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - y.sum()
    if abs(d) > 1e-12:
        free = (y > 1e-8) & (y < 1.0 - 1e-8)
        if free.sum() > 0:
            y[free] += d / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def _proj_grad(h, dx, alpha, max_iter, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        cm = c.max()
        s = np.clip(alpha * (c - cm), -300, 300)
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-6:
                continue
            k = m - (n - 1)
            if k == 0:
                g += w[m] * (1.0 - 2.0 * h)
            elif k > 0:
                g[:n-k] += w[m] * (1.0 - h[k:])
                g[k:] -= w[m] * h[:n-k]
            else:
                k2 = -k
                g[k2:] += w[m] * (1.0 - h[:n-k2])
                g[:n-k2] -= w[m] * h[k2:]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002]:
            h_new = _project(h + step * g, tgt)
            cn = _c5(h_new, dx)
            if cn < best - 1e-12:
                h, best = h_new, cn
                improved = True
                break
        if not improved:
            break
    return h, best

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.exp(s).sum()) / alpha + 1e5 * (x.sum() - tgt) ** 2
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _anneal(h, dx, rng, t_end, n_iter=6000):
    n = len(h)
    tgt = n / 2.0
    best_h, best_c = h.copy(), _c5(h, dx)
    h = h.copy()
    cur = _c5(h, dx)
    T0, T1 = 0.015, 0.0005
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (it / n_iter)
        c_arr = _corr_full(h)
        top_k = max(1, (2 * n - 1) // 8)
        idx = np.argsort(c_arr)[-top_k:]
        mi = rng.choice(idx)
        k = mi - (n - 1)
        if k == 0:
            i, j = rng.randint(n), rng.randint(n)
        else:
            ak = abs(k)
            mx = n - ak - 1
            if mx <= 0:
                i, j = rng.randint(n), rng.randint(n)
            else:
                i = rng.randint(0, mx)
                j = i + ak
        if i == j:
            continue
        d = rng.uniform(-0.012, 0.012)
        h[i] += d
        h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            nc = _c5(h, dx)
            if nc < cur or rng.random() < np.exp(-(nc - cur) / max(T, 1e-10)):
                cur = nc
                h = _project(h, tgt)
                if cur < best_c:
                    best_c, best_h = cur, h.copy()
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c

def _local(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    per = max(30, n_iter // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() > t_end:
                return h, best
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2 * rng.random() - 1)
            h[i] += d
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                c = _c5(h, dx)
                if c < best:
                    best = c
                    h = _project(h, tgt)
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return h, best

def _interp(h, nn):
    return np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, len(h)), h)

def _optimize_one(h, dx, t_end, rng, seed_val):
    n = len(h)
    best_h, best_c = h.copy(), _c5(h, dx)
    h_cur = h.copy()
    t1 = min(t_end - 120, time.time() + 20)
    for a in [3000, 15000]:

            break
        h_cur, c = _slsqp(h_cur, dx, a, 200)
        if c < best_c:
            best_h, best_c = h_cur, c
    t2 = min(t_end - 60, time.time() + 60)
    for a in [40, 100, 300]:
        if time.time() > t2:
            break
        h_cur, c = _proj_grad(h_cur, dx, a, 300, t2)
        if c < best_c:
            best_h, best_c = h_cur, c
    t3 = min(t_end - 20, time.time() + 40)
    best_h, c = _anneal(best_h, dx, rng, t3, 4000)
    if c < best_c:
        best_c = c
    t4 = t_end - 5
    best_h, c = _local(best_h, dx, rng, t4, 100000)
    if c < best_c:
        best_c = c
    return best_h, best_c, n
