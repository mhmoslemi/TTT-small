    n = len(h)
    fc = _eval(h)
    for _ in range(nsteps):
        c = np.correlate(h, 1.0 - h, mode='full')
        g = _corr_grad(c, h, alpha, n)
        step = step0
        improved = False
        for _ in range(30):
            hn = _proj_to_sum(h - step * g, n * 0.5)
            fn = _eval(hn)
            if fn < fc - 1e-16:
                h, fc = hn, fn
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, fc

def _upsample(h, f):
    return np.repeat(h, f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS at n0 with increasing sharpness
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if _el() > 180:
            break
        mu_seq = [100, 1000, 10000]
        for mu in mu_seq:
            if _el() > 180:
                break
            h = _lbg(best_h, alpha, mu, 400, 1e-18)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 2: Worst-shift descent at n0
    if _el() < budget_s - 350:
        for al in [50000, 200000]:
            if _el() > budget_s - 350:
                break
            h2, f2 = _ws_descent(best_h, al, 600, 0.008)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 3: Upsample to 2x
    if _el() < budget_s - 300:
        n1 = n0 * 2
        hh = _upsample(best_h, 2)
        hh = _repair(hh, n1)
        for alpha in [1000, 5000, 20000, 100000]:
            if _el() > budget_s - 200:
                break
            for mu in [1000, 10000]:
                if _el() > budget_s - 200:
                    break
                hh = _lbg(hh, alpha, mu, 300, 1e-18)
                c5 = _eval(hh)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, hh.copy(), n1
        if _el() < budget_s - 150:
            hh, f2 = _ws_descent(best_h, 200000, 400, 0.004)
            c5 = _eval(hh)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hh.copy(), n1

    # Phase 4: Multi-start at best resolution
    nc = best_n
    if _el() < budget_s - 180:
        for _r in range(25):
            if _el() > budget_s - 180:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.015 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                nb = rng.randint(3, max(4, nc // 7))
                st = rng.randint(0, nc - nb)
                hp[st:st+nb] += 0.05 * rng.randn(nb)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = best_h.copy()
                for fq in rng.randint(1, nc // 2 + 1, size=4):
                    amp = 0.01 * rng.randn()
                    hp += amp * np.sin(2 * np.pi * fq * np.arange(nc) / nc)
                hp = _repair(hp, nc)
            elif strat == 4:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    i = rng.randint(nc // 3)
                    j = nc - 1 - i
                    d = 0.02 * rng.randn()
                    hp[i] += d
                    hp[j] -= d
                hp = _repair(hp, nc)
            else:
                hp = _repair(best_h * rng.uniform(0.8, 1.2, size=nc), nc)
            h_new = _lbg(hp, 50000, 5000, 500, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase 5: Worst-shift refinement
    if _el() < budget_s - 80:
        for al in [100000, 500000]:
            if _el() > budget_s - 80:
                break
            h2, f2 = _ws_descent(best_h, al, 400, 0.003)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 6: Pairwise fine search
    if _el() < budget_s - 25:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.004, 0.001, 0.0003, 0.00008]:
            if _el() > budget_s - 25:
                break
            max_it = 12000 if scale > 0.001 else 15000
            for _it in range(max_it):
                if _el() > budget_s - 25:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h, best_n = c5n, h.copy(), nc
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 7: Final L-BFGS polish
    if _el() < budget_s - 5:
        h = _lbg(best_h, 500000, 50000, 800, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 8: Ultra-fine touch
    if _el() < budget_s - 2:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(20000):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h, best_n = c5n, h.copy(), nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
