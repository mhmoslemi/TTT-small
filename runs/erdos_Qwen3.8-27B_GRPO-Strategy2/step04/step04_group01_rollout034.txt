import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _obj_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    obj = m + s.max() / alpha + np.log(np.sum(np.exp(s))) / alpha
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        shift = (n - 1) - k
        wk = w[k] * dx
        if shift >= 0:
            g[shift:] += wk * (1.0 - h[:n - shift])
            g[:n - shift] -= wk * h[shift:]
        else:
            sp = -shift
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return obj, g

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _obj_grad(x, alpha)[0], h,
                       jac=lambda x: _obj_grad(x, alpha)[1],
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _crit_descent(h, niter, deadline):
    n = len(h)
    c5c = _eval(h)
    for _ in range(niter):
        if time.time() > deadline:
            break
        c = _corr(h)
        k = c.argmax()
        shift = (n - 1) - k
        g = np.zeros(n)
        if shift >= 0:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
        else:
            sp = -shift
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
        g *= 2.0 / n
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _pairwise(h, niter, deadline, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(niter):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        d = scale * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i], h[j] = h[i] + d, h[j] - d
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _adam_opt(h, alpha, niter, lr, deadline):
    n = len(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    for _ in range(niter):
        if time.time() > deadline:
            break
        _, g = _obj_grad(h, alpha)
        t += 1
        m_v = 0.9 * m_v + 0.1 * g
        v_v = 0.999 * v_v + 0.001 * g * g
        m_hat = m_v / (1 - 0.9**t)
        v_hat = v_v / (1 - 0.999**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = _proj(h - step, n * 0.5)
    return h

            best_c5, best_h, best_n = c5c, h.copy(), n0

    # Phase 3: 2x upsample + optimize
    if _el() < budget_s - 400:
        n1 = n0 * 2
        h1 = np.repeat(best_h, 2)
        h1 = _repair(h1, n1)
        for alpha in [5000, 30000, 100000, 500000]:
            if _el() > budget_s - 350:
                break
            h1 = _slsqp(h1, alpha, 2000, 1e-19)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h1.copy(), n1
        if _el() < budget_s - 300:
            h1, c5c = _crit_descent(best_h.copy(), 1000, t0 + budget_s - 300)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, h1.copy(), len(h1)

    # Phase 4: Multi-start SLSQP at best resolution
    nc = best_n
    for r in range(15):
        if _el() > budget_s - 250:
            break
        strat = r % 4
        if strat == 0:
            hp = best_h + 0.01 * (1 + 0.3 * (r // 4)) * rng.randn(nc)
        elif strat == 1:
            hp = 1.0 - best_h + 0.008 * rng.randn(nc)
        elif strat == 2:
            hp = best_h.copy()
            for _ in range(2):
                bs = rng.randint(5, max(6, nc // 8))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.03 * rng.randn(bs)
        else:
            hp = best_h.copy()
            for f in rng.randint(1, nc // 2 + 1, 3):
                hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * np.arange(nc) / nc)
        hp = _repair(hp, nc)
        h_new = _slsqp(hp, 100000, 800, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase 5: Adam optimization
    if _el() < budget_s - 200:
        for alpha in [5000, 50000, 200
000]:
            if _el() > budget_s - 180:
                break
            h = best_h.copy()
            h = _adam_opt(h, alpha, 400, 0.008, t0 + budget_s - 180)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 6: Critical lag descent again
    if _el() < budget_s - 150:
        h, c5c = _crit_descent(best_h.copy(), 3000, t0 + budget_s - 150)
        if c5c < best_c5:
            best_c5, best_h, best_n = c5c, h.copy(), len(h)

    # Phase 7: SLSQP polish
    if _el() < budget_s - 80:
        h = _slsqp(best_h, 500000, 3000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)
        h = _slsqp(best_h, 2000000, 2000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 8: Pairwise local search
    if _el() < budget_s - 30:
        for scale in [0.008, 0.003, 0.001, 0.0003, 0.00005]:
            if _el() > budget_s - 30:
                break
            h, c5c = _pairwise(best_h.copy(), 6000, t0 + budget_s - 30, scale, rng)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, h.copy(), len(h)

    # Phase 9: Final SLSQP + crit descent
    if _el() < budget_s - 10:
        h = _slsqp(best_h, 1000000, 2000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)
    if _el() < budget_s - 3:
        h, c5c = _crit_descent(best_h.copy(), 5000, t0 + budget_s - 3)
        if c5c < best_c5:
            best_c5, best_h, best_n = c5c, h.copy(), len(h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
