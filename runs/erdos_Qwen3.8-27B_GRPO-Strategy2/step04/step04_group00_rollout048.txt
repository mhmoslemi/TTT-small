from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h.copy() * target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _subgrad_active(h, frac):
    c = _corr(h)
    n = len(h)
    k = max(1, int(frac * (2 * n - 1)))
    thresh = np.partition(c, -k)[-k]
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for ki in active:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _nesterov_phase(h, n, dx, t0, budget, reserve):
    """Nesterov-accelerated subgradient descent."""
    v = np.zeros(n)
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    mu = 0.85
    lr = 0.03
    
    for it in range(5000):
        tl = budget - (time.time() - t0) - reserve
        if tl < 10:
            break
        # Nesterov extrapolation
        h_v = h_cur + mu * (h_cur - h_prev)
        h_v = np.clip(h_v, 0.0, 1.0)
        
        g = _subgrad_active(h_v, 0.08)
        g_c = g - g.mean()
        gn = np.linalg.norm(g_c)
        if gn < 1e-14:
            break
        g_c /= gn
        
        h_new = _repair(h_cur - lr * g_c, n)
        c_new = _c5(h_new, dx)
        
        if c_new < _c5(h_cur, dx):
            h_prev = h_cur
            h_cur = h_new
            lr = min(lr * 1.02, 0.06)
            if c_new < best_c5:

                best_h = h_new.copy()
        else:
            # Backtrack

                        best_h = h_try.copy()
                    break

                lr *= 0.7
                h_prev = h_cur.copy()
    
    return best_h, best_c5

def _pairwise_search(h, n, dx, t0, budget, reserve):

    dlist = np.array([0.0003, 0.0008, 0.002, 0.005, 0.01, 0.02,

                    h = _repair(h, n)
                    best_c5 = _c5(h, dx)
                    improved = True
                    if tl < 15:

        if not improved:
