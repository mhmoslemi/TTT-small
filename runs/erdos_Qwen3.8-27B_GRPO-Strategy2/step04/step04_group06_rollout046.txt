import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _grad_sub(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    tol = 1e-12
    active = c >= m - tol
    cnt = np.sum(active)
    if cnt == 0:
        active[np.argmax(c)] = True
        cnt = 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        wk = 1.0 / cnt
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _fw_vertex(g, n):
    """Solve min g·v s.t. sum(v)=n/2, 0<=v<=1."""
    # Sort by g ascending, pick n/2 smallest
    order = np.argsort(g)
    v = np.zeros(n)
    v[order[:n//2]] = 1.0
    return v

def _project_box_sum(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    # Binary search for the threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    hp = np.clip(h - lam, 0.0, 1.0)
    # Fix rounding
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-boundary elements
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

    s = np.clip(alpha * (c - m), -300, 300)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _obj_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _project_box_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _fw_optimize(h, n, budget_end):
    h = h.copy()
    c5_cur = _c5(h, n)
    for t in range(20000):
        if time.time() > budget_end:
            break
        g = _grad_sub(h, n)
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:n // 2]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = _project_box_sum(h_new, n)
        c = _c5(h_new, n)
        if c < c5_cur:
            c5_cur = c
            h = h_new
    return h, c5_cur

def _smooth_pgd(h, n, alpha, budget_end, lr_init):
    h = h.copy()
    lr = lr_init
    c5_cur = _c5(h, n)
    for it in range(50000):
        if time.time() > budget_end:
            break
        g = _grad_smooth(h, n, alpha)
        h_new = _project_box_sum(h - lr * g, n)
        c = _c5(h_new, n)
        if c < c5_cur:
            h = h_new
            c5_cur = c
        else:
            lr *= 0.7
        if it % 500 == 499:
            lr *= 0.85
    return h, c5_cur

def _pair_local_search(h, n, budget_end, rng):
    h = h.copy()
    c5_cur = _c5(h, n)
    c5_cur = float(c5_cur)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05,
                       -0.002, -0.005, -0.01, -0.02, -0.05])
    idx_all = np.arange(n)
    for _sweep in range(500):
        if time.time() > budget_end:
            break
        improved = False
        rng.shuffle(idx_all)
        for i in idx_all:
            if time.time() > budget_end:
                break
            hi = h[i]
            for j in idx_all:
                if j <= i:
                    continue
                hj = h[j]
                best_d = 0.0
                best_c = c5_cur
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < best_c:
                        best_c = c
                        best_d = d
                    h[i], h[j] = hi, hj
                if best_d != 0.0:
                    h[i] = hi + best_d
                    h[j] = hj - best_d
                    c5_cur = best_c
                    improved = True
        if not improved:
            break
    return h, c5_cur

def _multistart_quick(h0, n, alpha, budget_end, rng, n_start=5):
    best_h = h0.copy()
    best_c5 = _c5(h0, n)
    for s in range(n_start):
        if time.time() > budget_end:
            break
        scale = 0.01 * (s + 1)
        h_try = _repair(h0 + scale * rng.randn(n), n)
        h_try, c5_try = _smooth_pgd(h_try, n, alpha, time.time() + budget_end * 0.05, 0.03)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def _slsqp_refine(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    return minimize(
        _obj_smooth, h.copy(), args=(n, alpha),
        jac=lambda x: _grad_smooth(x, n, alpha),
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-16})

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: FW (fast, 15% time)
    t1_end = t0 + budget_s * 0.15
    h1, c1 = _fw_optimize(best_h, n, t1_end)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Smooth PGD with increasing alpha (25% time)
    t2_end = t0 + budget_s * 0.40
    for alpha in [2000, 10000, 50000, 150000, 300000]:
        if elapsed() > (t2_end - t0) * 0.8:
            break
        sub_end = t2_end - (t2_end - t0) * 0.05
        h2, c2 = _smooth_pgd(best_h, n, alpha, sub_end, lr_init=0.08)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: SLSQP polishing (15% time)
    t3_end = t0 + budget_s * 0.55
    for r in range(4):
        if elapsed() > t3_end - t0:
            break
        if r == 0:
            h_init = best_h.copy()
        else:
            h_init = _repair(best_h + 0.008 * rng.randn(n), n)
        try:
            res = _slsqp_refine(h_init, n, 200000, 500)
            hc = _repair(res.x, n)
            c5 = _c5(hc, n)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 4: Pair local search (25% time)
    t4_end = t0 + budget_s * 0.80
    best_h, best_c5 = _pair_local_search(best_h, n, t4_end, rng)

    # Phase 5: Multi-start with small perturbations (10% time)
    t5_end = t0 + budget_s * 0.92
    best_h, best_c5 = _multistart_quick(best_h, n, 100000, t5_end, rng, n_start=8)

    # Phase 6: Final very-fine pair search (8% time)
    if elapsed() < budget_s - 15:
        t6_end = t0 + budget_s - 5
        h_fine = best_h.copy()
        c_fine = best_c5
        tiny_deltas = np.array([0.0003, 0.001, 0.003, -0.0003, -0.001, -0.003])
        for _sweep in range(200):
            if time.time() > t6_end:
                break
            improved = False
            idx_all = np.arange(n)
            rng.shuffle(idx_all)
            for i in idx_all:
                if time.time() > t6_end:
                    break
                hi = h_fine[i]
                for j in idx_all:
                    if j <= i:
                        continue
                    hj = h_fine[j]
                    for d in tiny_deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h_fine[i], h_fine[j] = ni, nj
                        c = _c5(h_fine, n)
                        if c < c_fine - 1e-15:
                            c_fine = c
                            improved = True
                            break
                        h_fine[i], h_fine[j] = hi, hj
                    else:
                        h_fine[i], h_fine[j] = hi, hj
                        continue
                    h_fine[i] = ni
                    h_fine[j] = nj
            if not improved:
                break
        if c_fine < best_c5:
            best_c5, best_h = c_fine, h_fine.copy()

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
