            if time.time() >= t_end:

            hh, cc = lbfgs_opt(h, n, alpha, target, 3000, 300)
            if cc < c5_init:
                h, c5_init = hh, cc
        if c5_init < best_c5:
            best_h, best_c5 = h.copy(), c5_init

    # Phase 4: SA medium temp (200s)
    if time.time() < t_start + budget_s - 250:
        t_end = min(time.time() + 200, t_start + budget_s - 250)
        h, c5_init = sa_anneal(h, n, dx, rng, t_end, T0=0.001, T_rate=0.9998, n_phase=2)
        if c5_init < best_c5:
            best_h, best_c5 = h.copy(), c5_init

    # Phase 5: Pair local search (120s)
    if time.time() < t_start + budget_s - 130:
        t_end = min(time.time() + 120, t_start + budget_s - 130)
        h, c5_init = pair_local(h, n, dx, rng, t_end, 8)
        if c5_init < best_c5:
            best_h, best_c5 = h.copy(), c5_init

    # Phase 6: Multi-restart with L-BFGS (80s)
    if time.time() < t_start + budget_s - 100:
        t_end = min(time.time() + 80, t_start + budget_s - 100)
        for r in range(6):
            if time.time() >= t_end:
                break
            hp = perturb_restart(best_h, n, rng, r % 5)
            hh, cc = lbfgs_opt(hp, n, 2000, target, 3000, 150)
            hh, cc = slsqp_opt(hh, n, 5000, target, 200)
            if cc < best_c5:
                best_h, best_c5, h = hh.copy(), cc, hh
                c5_init = cc

    # Phase 7: SA low temp (80s)
    if time.time() < t_start + budget_s - 50:
        t_end = min(time.time() + 80, t_start + budget_s - 50)
        h_rep = repair(best_h.copy(), n)
        hh, cc = sa_anneal(h_rep, n, dx, rng, t_end, T0=0.0003, T_rate=0.9999, n_phase=3)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 8: Final local search + L-BFGS (40s)
    if time.time() < t_start + budget_s - 5:
        t_end = min(time.time() + 25, t_start + budget_s - 10)
        h_rep = repair(best_h.copy(), n)
        hh, cc = pair_local(h_rep, n, dx, rng, t_end, 5)
        if cc < best_c5:
            best_h, best_c5 = hh, cc
        if time.time() < t_start + budget_s - 5:
            hh, cc = lbfgs_opt(best_h, n, 20000, target, 5000, 300)
            if cc < best_c5:
                best_h, best_c5 = hh, cc

    best_h = repair(best_h, best_n if 'best_n' in dir() else n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
