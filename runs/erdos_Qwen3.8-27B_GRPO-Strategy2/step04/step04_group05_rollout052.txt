    else:

        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]

    v = np.zeros(n)
    mu = 0.0
    beta1, beta2, eps = 0.85, 0.95, 1e-8

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def sm_grad(h, n, a):
    """Gradient of smooth max of correlation wrt h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    sc = a * c
    sc -= sc.max()
    ew = np.exp(sc)
    pw = ew / ew.sum()
    g = np.zeros(n)
    g += pw[n-1] * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = pw[n-1+L]
        if wp > 1e-12:
            g[L:] += wp * (1.0 - h[:n-L])
            g[:n-L] -= wp * h[L:]
        wn = pw[n-1-L]
        if wn > 1e-12:
            g[:n-L] += wn * (1.0 - h[L:])
            g[L:] -= wn * h[:n-L]
    return g / a

    c = np.correlate(h, 1.0 - h, mode='full')
    sc = a * c
    sc -= sc.max()
    objs = sc.max() + np.log(np.exp(sc).sum()) / a
    return objs + lam * (h.sum() - n * 0.5) ** 2

    c = np.correlate(h, 1.0 - h, mode='full')
    sc = a * c
    sc -= sc.max()
    ew = np.exp(sc)
    pw = ew / ew.sum()
    g = np.zeros(n)
    g += pw[n - 1] * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = pw[n - 1 + L]
        if wp > 1e-13:
            g[L:] += wp * (1.0 - h[:n - L])
            g[:n - L] -= wp * h[L:]
        wn = pw[n - 1 - L]
        if wn > 1e-13:
            g[:n - L] += wn * (1.0 - h[L:])
            g[L:] -= wn * h[:n - L]
    return g / a

def sm_obj(x, n, a, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    sc = a * c
    sc -= sc.max()
    return (sc.max() + np.log(np.exp(sc).sum())) / a + lam * (x.sum() - n * 0.5) ** 2

def sm_jac(x, n, a, lam):
    g = sm_grad(x, n, a)
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def adam_opt(h, n, a, t0, lim, lr0=0.04, lam=5.0, iters=8000):
    m = n * 0.5
    best_h, best_c = h.copy(), c5v(h, 2.0 / n)
    h = h.copy()
    mu = np.zeros(n)
    v = np.zeros(n)
    lr = lr0
    for t in range(iters):
        if time.time() - t0 > lim:
            break
        g = sm_grad(h, n, a) + 2.0 * lam * (h.sum() - m)
        mu = 0.9 * mu + 0.1 * g
        v = 0.95 * v + 0.05 * g * g
        muh = mu / (1.0 - 0.9 ** (t + 1))
        vh = v / (1.0 - 0.95 ** (t + 1))
        step = lr * muh / (np.sqrt(vh) + 1e-8)
        hn = repair(h - step, n)
        cn = c5v(hn, 2.0 / n)
        ch = c5v(h, 2.0 / n)
        if cn < ch - 1e-16:
            h = hn
            if cn < best_c:
                best_c, best_h = cn, h.copy()
            lr = min(lr * 1.008, 0.25)
        else:
            lr *= 0.82
            if lr < 1e-8:
                break
    return best_h, best_c

def lbfgs_opt(h, n, a, t0, lim, lam=5.0):
    best_h, best_c = h.copy(), c5v(h, 2.0 / n)
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return sm_obj(x, n, a, lam)
    def jac(x):
        return sm_jac(x, n, a, lam)
    try:
        r = minimize(obj, best_h, jac=jac, method='L-BFGS-B',
                     bounds=bds, options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-15})
        hc = repair(r.x, n)
        c = c5v(hc, 2.0 / n)
        if c < best_c:
            best_c, best_h = c, hc
    except Exception:
        pass
    return best_h, best_c

def peak_desc(h, n, t0, lim):
    best_h, best_c = h.copy(), c5v(h, 2.0 / n)
    h = h.copy()
    step = 0.04
    stall = 0
    for _ in range(2000):
        if time.time() - t0 > lim:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:6]
        g = np.zeros(n)
        for ki in idx:
            L = ki - (n - 1)
            if L == 0:
                g += 1.0 - 2.0 * h
            elif L > 0:
                g[L:] += 1.0 - h[:n - L]
                g[:n - L] -= h[L:]
            else:
                m2 = -L
                g[:n - m2] += 1.0 - h[m2:]
                g[m2:] -= h[:n - m2]
        g /= len(idx)
        g -= g.mean()
        ok = False
        for _ in range(25):
            hn = repair(h - step * g, n)
            cn = c5v(hn, 2.0 / n)
            if cn < c5v(h, 2.0 / n) - 1e-16:
                h = hn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
                step = min(step * 1.1, 0.8)
                ok = True
                stall = 0
                break
            step *= 0.7
        if not ok:
            stall += 1
            if stall > 10:
                break
    return best_h, best_c

def spectral_refine(h, n, dx, t0, lim):
    best_h, best_c = h.copy(), c5v(h, dx)
    for alpha in [0.08, 0.15, 0.25, 0.4, 0.3, 0.2, 0.12, 0.06]:
        if time.time() - t0 > lim:
            break
        h = best_h.copy()
        for _ in range(300):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - alpha) * mag + alpha * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-16:
                h = hr
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                break
    return best_h, best_c

def coord_desc(h, n, dx, rng, t0, lim, cnt=60000):
    best_h, best_c = h.copy(), c5v(h, dx)
    h = h.copy()
    cc = best_c
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.choice([0.004, 0.0015, 0.0006, 0.015, -0.004, -0.0015, -0.0006, -0.015])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc - 1e-15:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < best_c:
                    best_c, best_h = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def binary_anneal(n, rng, t0, lim):
    h = np.zeros(n)
    ones = np.arange(n)
    rng.shuffle(ones)
    h[ones[:n // 2]] = 1.0
    best_h, best_c = h.copy(), c5v(h, 2.0 / n)
    T = 0.5
    dx = 2.0 / n
    cc = best_c
    for _ in range(200000):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = c5v(h, dx)
        if c_new < cc or rng.random() < np.exp((cc - c_new) / (T * dx)):
            cc = c_new
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= 0.9997
    return best_h, best_c

def full_opt(h, n, seed, t0, lim):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h, best_c = h.copy(), c5v(h, dx)
    for a in [5, 15, 50, 200, 800, 4000, 20000, 100000]:
        if time.time() - t0 > lim - 60:
            break
        r, c = adam_opt(best_h, n, a, t0, min(lim - 60, time.time() - t0 + 15))
        if c < best_c:
            best_h, best_c = r, c
    for a in [4000, 40000, 200000]:
        if time.time() - t0 > lim - 25:
            break
        r, c = lbfgs_opt(best_h, n, a, t0, lim - 25)
        if c < best_c:
            best_h, best_c = r, c
    r, c = peak_desc(best_h, n, t0, lim - 15)
    if c < best_c:
        best_h, best_c = r, c
    r, c = spectral_refine(best_h, n, dx, t0, lim - 8)
    if c < best_c:
        best_h, best_c = r, c
    r, c = coord_desc(best_h, n, dx, rng, t0, lim - 2, cnt=40000)
    if c < best_c:
        best_h, best_c = r, c
    if time.time() - t0 < lim - 1:
        r, c = adam_opt(best_h, n, 200000, t0, lim - 1, lr0=0.01)
        if c < best_c:
            best_h, best_c = r, c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    dx0 = 2.0 / n0
    best_h, best_c, best_n = h0.copy(), c5v(h0, dx0), n0

    # Phase A: Main optimization at native resolution
    r, c = full_opt(h0, n0, seed, t0, budget_s - 350)
    if c < best_c:
        best_h, best_c, best_n = r, c, n0

    # Phase B: Binary anneal + relax
    if time.time() - t0 < budget_s - 200:
        rng_b = np.random.RandomState(seed + 700)
        hb, cb = binary_anneal(n0, rng_b, t0, budget_s - 160)
        # Relax from best binary
        hr = repair(hb + 0.08 * rng_b.randn(n0), n0)
        r, c = full_opt(hr, n0, seed + 800, t0, budget_s - 100)
        if c < best_c:
            best_h, best_c, best_n = r, c, n0

    # Phase C: Multi-start perturbation
    if time.time() - t0 < budget_s - 100:
        rng_p = np.random.RandomState(seed + 1200)
        for pert in [0.015, 0.04]:
            if time.time() - t0 > budget_s - 80:
                break
            hp = repair(best_h + pert * rng_p.randn(n0), n0)
            r, c = full_opt(hp, n0, seed + int(pert * 10000), t0, budget_s - 70)
            if c < best_c:
                best_h, best_c, best_n = r, c, n0

    # Phase D: Lower resolution global search then upsample
    n_low = n0 // 2 if n0 >= 60 else n0
    if n_low >= 30 and time.time() - t0 < budget_s - 80:
        xo = np.linspace(0, 2, n0, endpoint=False)
        xl = np.linspace(0, 2, n_low, endpoint=False)
        hl = repair(np.interp(xl, xo, best_h, period=2.0), n_low)
        rng_l = np.random.RandomState(seed + 2000)
        hl_b, _ = binary_anneal(n_low, rng_l, t0, budget_s - 60)
        r, c = full_opt(hl_b, n_low, seed + 2100, t0, budget_s - 50)
        if c < best_c:
            best_h, best_c, best_n = r, c, n_low
        # Upsample
        xo_l = np.linspace(0, 2, n_low, endpoint=False)
        h_up = repair(np.interp(xo, xo_l, best_h[:n_low] if best_n == n_low else np.interp(xo_l, np.linspace(0,2,best_n,endpoint=False), best_h, period=2.0), period=2.0), n0)
        r, c = full_opt(h_up, n0, seed + 2200, t0, budget_s - 5)
        if c < best_c:
            best_h, best_c, best_n = r, c, n0

    # Phase E: Final polish
    if time.time() - t0 < budget_s - 5:
        rng_f = np.random.RandomState(seed + 3000)
        r, c = coord_desc(best_h, best_n, 2.0 / best_n, rng_f, t0, budget_s - 2, cnt=20000)
        if c < best_c:
            best_h, best_c, best_n = r, c, best_n

    best_h = repair(best_h, best_n)
    best_c = c5v(best_h, 2.0 / best_n)
    return (best_h, best_c, best_n)
