        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
    return np.clip(hp, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _slsqp(h0, n, alpha, maxit):
    def cf(x):
        return float(np.sum(x)) - n / 2.0
    def cj(x):
        return np.ones(n)
    res = minimize(_smooth_obj, h0, args=(n, alpha), jac=_smooth_grad,
                   method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                   options={'maxiter': maxit, 'ftol': 1e-18})
    return _repair(res.x, n)

def _pgd(h, n, alpha, t_end, lr0):
    h = h.copy()
    lr = lr0
    c5c = _c5(h, n)
    while time.time() < t_end:
        g = _smooth_grad(h, n, alpha)
        h2 = _project(h - lr * g, n)
        c52 = _c5(h2, n)
        if c52 < c5c:
            h, c5c = h2, c52
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.7
    return h, c5c

def _subgrad(h, n, t_end, topk=3):
    h = h.copy()
    lr = 0.02
    c5c = _c5(h, n)
    while time.time() < t_end:
        c = _corr(h) * (2.0 / n)
        m = c.max()
        idx = np.argsort(c)[-topk:]
        g = np.zeros(n)
        for ki in idx:
            lag = ki - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                l2 = -lag
                g[:n - l2] += (1.0 - h[l2:])
                g[l2:] -= h[:n - l2]
        g = g * (2.0 / n) / len(idx)
        h2 = _project(h - lr * g, n)
        c52 = _c5(h2, n)
        if c52 < c5c:
            h, c5c = h2, c52
            lr *= 1.01
        else:
            lr *= 0.85
    return h, c5c

def _sa_optimize(h, n, t_end, rng, T0=0.002, T1=1e-6):
    """Simulated annealing for pair moves."""
    h = h.copy()
    c5c = _c5(h, n)
    T = T0
    nsteps = 5000
    for it in range(nsteps):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (it / nsteps)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.05, 0.05)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c52 = _c5(h, n)
        dc = c52 - c5c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-10)):
            c5c = c52
        else:
            h[i], h[j] = ni - d, nj + d  # revert
    return h, c5c

def _pair_search(h, n, t_end, rng):
    c5c = _c5(h, n)
    deltas = np.array([0.0005, 0.001, 0.002, 0.005, 0.01, 0.02,
                       -0.0005, -0.001, -0.002, -0.005, -0.01, -0.02])
    for _ in range(1000):
        if time.time() > t_end:
            break
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for i in idx[:n]:  # all positions
            if time.time() > t_end:
                break
            hi = h[i]
            for j in idx[:n]:
                if j == i:
                    continue
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = np.clip(ni, 0, 1), np.clip(nj, 0, 1)
                    c = _c5(h, n)
                    if c < c5c - 1e-15:
                        c5c, improved = c, True
                        hi, hj = h[i], h[j]
                        break
                    else:
                        h[i], h[j] = hi, hj
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
    
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    else:
        n0 = 140
        h0 = _repair(np.full(n0, 0.5), n0)
    
    best_h, best_n, best_c5 = h0.copy(), n0, _c5(h0, n0)
    
    # Phase 1: PGD with increasing alpha
    for alpha in [5000, 20000, 100000, 300000]:
        t_end = t0 + 15
        h0, c5w = _pgd(h0, n0, alpha, t_end, 0.05)
        c5w = _c5(h0, n0)
        if c5w < best_c5:
            best_c5, best_h = c5w, h0.copy()
        if elapsed() > budget_s * 0.2:
            break
    
    # Phase 2: SLSQP
    for alpha in [50000, 200000, 500000]:
        if elapsed() > budget_s * 0.45:
            break
        for r in range(3):
            if elapsed() > budget_s * 0.45:
                break
            h_init = best_h + (0.01 * rng.randn(best_n) if r > 0 else 0)
            h_init = _repair(h_init, best_n)
            try:
                hc = _slsqp(h_init, best_n, alpha, 500)
                c5 = _c5(hc, best_n)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass
    
    # Phase 3: Subgradient
    if elapsed() < budget_s * 0.55:
        h0, c5w = _subgrad(best_h, best_n, t0 + budget_s * 0.55, topk=3)
        c5w = _c5(h0, best_n)
        if c5w < best_c5:
            best_c5, best_h = c5w, h0
    
    # Phase 4: Multi-resolution
    for n_try in [100, 120, 160, 200]:
        if elapsed() > budget_s * 0.65:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h, period=2.0)
        h_new = _repair(h_new, n_try)
        h_new, c5w = _pgd(h_new, n_try, 100000, t0 + budget_s * 0.68, 0.03)
