    w = np.exp(s)

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _soft_max_w(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    return m + np.log(w.sum()) / alpha, w

def _grad_corr(h, w, n):
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
        else:
            s2 = -s
            grad[s2:] += wk * (1.0 - h[:n - s2])
            grad[:n - s2] -= wk * h[s2:]
    return grad

def _project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    h = np.clip(h + d / len(h), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _proj_grad(h, dx, t_end, n_steps=3000):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    alphas = [200, 500, 1000, 3000, 8000, 20000, 60000]
    lrs = [0.03, 0.02, 0.015, 0.01, 0.006, 0.003, 0.001]
    spp = max(30, n_steps // len(alphas))
    for ai in range(len(alphas)):
        alpha = alphas[ai]
        lr = lrs[ai]
        for _ in range(spp):
            if time.time() > t_end:
                return best_h, best_c
            t += 1
            c = _corr_full(h)
            _, w = _soft_max_w(c, alpha)
            grad = _grad_corr(h, w, n)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            mh = m / (1 - beta1 ** t)
            vh = v / (1 - beta2 ** t)
            h_new = h - lr * mh / (np.sqrt(vh) + eps)
            h_new = _project_sum(h_new, tgt)
            c_new = _c5(h_new, dx)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.exp(s).sum()) / alpha
    def jac(x):
        c = _corr_full(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s)
        w /= w.sum()
        return _grad_corr(x, w, n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project_sum(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)
