import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_overlap(h, ki, n):
    """Gradient of overlap at lag ki (0-indexed in full correlation, n-1 is zero lag)."""
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def smooth_max(h, n, beta, mu):
    """Smooth max objective with integral penalty."""
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(beta * (c - cm), -500, 500)
    obj = cm + np.log(np.sum(np.exp(s))) / beta
    obj += mu * (h.sum() - n * 0.5) ** 2
    return obj

def smooth_max_grad(h, n, beta, mu):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(beta * (c - cm), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-12:
            g += pw[k] * grad_overlap(h, k, n)
    g /= beta
    g += 2.0 * mu * (h.sum() - n * 0.5)
    return g

def nesterov_phase(h, n, dx, t0, lim, beta=200, mu=100, max_iter=5000):
    """Nesterov-accelerated projected gradient."""
    x = h.copy()
    y = h.copy()
    t = 1.0
    bh, bc = h.copy(), c5v(h, dx)
    # Estimate L (Lipschitz constant of gradient)
    g0 = smooth_max_grad(y, n, beta, mu)
    L = max(np.linalg.norm(g0), 1.0) * 5.0 + 1.0
    
    for it in range(max_iter):
        if time.time() - t0 > lim:
            break
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = x + (t - 1) / t_new * (x - y)
        y = np.clip(y, 0, 1)
        
        g = smooth_max_grad(y, n, beta, mu)
        step = 1.0 / L
        x_new = repair(y - step * g, n)
        
        # Adaptive step
        fn = c5v(x_new, dx)
        if fn > c5v(x, dx) + 1e-12:
            step *= 0.5
            L *= 2
            x_new = repair(y - step * g, n)
        else:
            L = max(L * 0.8, 1.0)
        
        x = x_new
        t = t_new
        if c5v(x, dx) < bc:
            bc = c5v(x, dx)
            bh = x.copy()
    return bh, bc

def active_phase(h, n, dx, t0, lim):
    """Subgradient on top-m overlaps with Nesterov."""
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [1, 2, 3, 5, 8, 12, 20]:
        x = bh.copy()
        y = x.copy()
        t = 1.0
        step = 0.1
        stalled = 0
        for _ in range(400):
            if time.time() - t0 > lim:
                break
            c = np.correlate(x, 1.0 - x, mode='full')
            cm = c.max()
            thr = cm - 0.005 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = np.zeros(n)
            for ki in idx:
                g += grad_overlap(x, ki, n)
            g /= len(idx)
            g -= g.mean()  # preserve sum
            
            t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
            y = x + (t - 1) / t_new * (x - y)
            y = repair(y - step * g, n)
            
            cn = c5v(y, dx)
            if cn < c5v(x, dx) - 1e-16:
                x = y
                if cn < bc:
                    bc, bh = cn, x.copy()
                step = min(step * 1.05, 2.0)
                stalled = 0
            else:
                step *= 0.75
                if step < 1e-6:
                    break
                stalled += 1
                if stalled > 15:
                    break
            t = t_new
    return bh, bc

def spec_phase(h, n, dx, t0, lim):
    """Spectral equalization."""
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.1, 0.2, 0.35, 0.5, 0.4, 0.25, 0.15, 0.08]:
        x = bh.copy()
        for _ in range(200):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(x)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(x, x), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            xn = np.fft.irfft(nm * np.exp(1j * ph), n)
            xr = repair(xn, n)
            cn = c5v(xr, dx)
            if cn < c5v(x, dx) - 1e-16:
                x = xr
                if cn < bc:
                    bc, bh = cn, x.copy()
            else:
                break
    return bh, bc

def cd_phase(h, n, dx, rng, t0, lim):
    """Coordinate descent with 2D moves."""
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    for _ in range(150):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(800):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in (0.02, 0.005, 0.001, -0.02, -0.005, -0.001):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-15:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 5:
            break
    return bh, bc

def polish_phase(h, n, dx, rng, t0, lim):
    """Fine 2D polishing."""
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(30000):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 10 ** rng.uniform(-6.5, -3.5) * rng.choice([1, -1])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def lbfgs_phase(h, n, dx, t0, lim):
    """L-BFGS-B with increasing beta."""
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    for beta, mu, mi in [(50, 100, 300), (200, 40, 250), (800, 10, 200), (3000, 1, 150), (10000, 0.1, 100)]:
        if time.time() - t0 > lim:
            break
        try:
            r = minimize(smooth_max, bh.copy(), jac=lambda x: smooth_max_grad(x, n, beta, mu),
                        method='L-BFGS-B', bounds=bds,
                        options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def upscale(h, n_old, n_new):

    hn = np.interp((xn % 2.0), xo, h, period=2.0)

    return hn
