            g[a2:b2] -= wk * h[a2-lag:b2-lag]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n/2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _pen_obj(h, alpha, lam):
    f = _smooth_obj(h, alpha)
    cv = np.sum(h) - len(h)/2.0
    return f + lam*cv*cv

def _pen_grad(h, alpha, lam):
    g = _smooth_grad(h, alpha)
    cv = np.sum(h) - len(h)/2.0
    g += 2.0*lam*cv
    return g

def _slsqp(h, n, alpha, maxiter):
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: np.sum(x)-n/2.0,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _lbfgsb(h, n, alpha, lam, maxiter):
    bnds = [(0.0,1.0)]*n
    try:
        res = minimize(_pen_obj, h, jac=_pen_grad, args=(alpha,lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _grad_proj(h, n, alpha, lr, n_steps):
    """Gradient descent with projection onto feasible set"""
    for _ in range(n_steps):
        g = _smooth_grad(h, alpha)
        h = h - lr*g
        # Project onto {0<=h<=1, sum=n/2} via bisection on tau
        lo, hi = -1.0, 2.0
        for _ in range(80):
            mid = (lo+hi)/2
            proj = np.clip(h - mid, 0, 1)
            if proj.sum() > n/2.0:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo+hi)/2, 0, 1)
    return _repair(h, n)

def _local_search(h, n, rng, t_end):
    c5 = _eval(h)
    scale = 0.005
    no_imp = 0
    it = 0
    while time.time() < t_end and scale > 3e-6:
        it += 1
        if it % 1000 == 0:
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = scale * rng.randn()
            ni, nj = h[i]+d, h[j]-d
            if 0<=ni<=1 and 0<=nj<=1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5 - 1e-15:
                    c5 = c5n; no_imp = 0; scale = min(scale*1.03, 0.02)
                else:
                    h[i], h[j] = ni-d, nj+d; no_imp += 1
                    if no_imp > 200:
                        scale *= 0.9; no_imp = 0
        # Also try gradient-guided steps every so often
    return h, c5

def _interp(h, n_from, n_to):
    xs = np.linspace(0, 2, n_from, endpoint=False)
    xd = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xd, xs, h)
