def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _soft_obj(h, alpha, target, pen):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.exp(s).sum()) / alpha + pen * (h.sum() - target) ** 2)

def _slsqp(h, dx, alpha, maxiter, t_end):

    bds = [(0.0, 1.0)] * n

                     method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
