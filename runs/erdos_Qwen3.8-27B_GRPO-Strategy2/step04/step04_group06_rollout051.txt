def _repair(h, n):

    s = h.sum()
    if s < 1e-12:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):

    h = np.clip(h, 0.0, 1.0)

    for _ in range(70):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)

    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h

        else:

        # Periodically increase lr

    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - x[l2:])
            g[l2:] -= wk * x[:n - l2]
    return g * dxv

def _grad_worst_lag(h, n, top_k):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_k]
    g = np.zeros(n)
    cnt = 0
    for k in idx:
        lag = k - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += (1.0 - h[l2:])
            g[l2:] -= h[:n - l2]
        cnt += 1
    return g / max(cnt, 1)

def _pgd_momentum(h, n, alpha, budget_end, lr_init, mu, max_iters):
    h = h.copy().astype(float)
    m = np.zeros(n)
    lr = lr_init
    c5_cur = _c5(h, n)
    best_h, best_c5 = h.copy(), c5_cur
    for it in range(max_iters):
        if time.time() > budget_end:
            break
        y = np.clip(h - mu * m, 0.0, 1.0)
        g = _smooth_grad(y, n, alpha)
        new_m = mu * m - lr * g
        h_new = _project(h + new_m, n)
        c5_new = _c5(h_new, n)
        if c5_new < c5_cur - 1e-18:
            m = new_m
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
        else:
            lr *= 0.75
            if lr < 1e-8:
                lr = lr_init * 0.2
                m *= 0.3
        if it % 400 == 399:

            if it > 0 and it % 2000 == 1999:

                if _c5(h_try, n) < c5_cur:
                    h = h_try
                    c5_cur = _c5(h, n)
                    m *= 0.1
    return best_h, best_c5

def _pair_targeted(h, n, budget_end, rng):
    h = h.copy()
    c5_cur = _c5(h, n)
    deltas = np.array([0.0003, 0.001, 0.003, 0.01, 0.03, 0.08,
                       -0.0003, -0.001, -0.003, -0.01, -0.03, -0.08])
    for _pass in range(500):
        if time.time() > budget_end:
            break
        g = _grad_worst_lag(h, n, 5)
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for ii in range(n):
            if time.time() > budget_end:
                break
            i = idx[ii]
            hi = h[i]
            gi = g[i]
            # Pick candidate j's based on gradient sign
            if gi > 0:
                j_order = np.argsort(g)[:min(15, n)]
            else:
                j_order = np.argsort(-g)[:min(15, n)]
            for jj in j_order:
                if jj == i:
                    continue
                j = jj
                hj = h[j]
                for d in deltas:
                    if gi > 0 and d > 0:
                        continue
                    if gi < 0 and d < 0:
                        continue
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        improved = True
                        hi, hj = ni, nj
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5_cur

def _slsqp_run(h_init, n, alpha, maxiter, rng):
    bounds = [(0.0, 1.0)] * n
    def cf(x):
        return float(np.sum(x) - n / 2.0)
    def cj(x):
        return np.ones(n)
    try:
        res = minimize(_smooth_obj, h_init, args=(n, alpha),
                       jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return h_init
