    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.3, 0.5, 0.7, 0.5, 0.3, 0.15]:
        h = bh.copy()
        for _ in range(300):
            if time.time() - t0 > allotment:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-17:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def multi_refine(h, n, dx, seed, t0, allotment):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_phase(bh, n, dx, t0, allotment - 200)
    if c < bc:
        bh, bc = r, c
    r, c = active_phase(bh, n, dx, t0, allotment - 150)
    if c < bc:
        bh, bc = r, c
    r, c = lbfgs_phase(bh, n, dx, t0, allotment - 100)
    if c < bc:
        bh, bc = r, c
    r, c = active_phase(bh, n, dx, t0, allotment - 70)
    if c < bc:
        bh, bc = r, c
    r, c = spec_phase(bh, n, dx, t0, allotment - 50)
    if c < bc:
        bh, bc = r, c
    r, c = nd_phase(bh, n, dx, t0, allotment - 30)
    if c < bc:
        bh, bc = r, c
    r, c = coord_ds(bh, n, dx, rng, t0, allotment - 5)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def make_start(n, typ, rng):
    if typ == 0:
        h = np.full(n, 0.5)
        h += 0.25 * np.sin(2 * np.pi * rng.randint(2, 6) * np.arange(n) / n)
    elif typ == 1:
        nb = rng.randint(3, 8)
        h = np.zeros(n)
        seg = n // nb
        for i in range(nb):
            h[i*seg:(i+1)*seg] = rng.uniform(0.1, 0.9)
    elif typ == 2:
        h = np.clip(0.5 + rng.uniform(-0.4, 0.4), 0, 1)
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx), nl

    # Main refinement of parent
    alloc1 = budget_s * 0.35
    r, c = multi_refine(h0, nl, dx, seed, t0, t0 + alloc1)
    if c < bc:
        bh, bc, bn = r, c, nl

    # Try diverse starting points at same resolution
    rng = np.random.RandomState(seed + 7)
    alloc2 = budget_s * 0.20
    for typ in range(4):
        if time.time() - t0 > t0 + alloc1 + alloc2:
            break
        alloc_each = (t0 + alloc1 + alloc2 - time.time()) / (4 - typ)
        if alloc_each < 30:
            break
        hs = make_start(nl, typ, rng)
        r, c = multi_refine(hs, nl, dx, seed + 100 * (typ + 1), t0, t0 + alloc1 + alloc2)
        if c < bc:
            bh, bc, bn = r, c, nl

    # Try lower resolution then upscale
    if time.time() - t0 < t0 + budget_s * 0.85:
        n_low = max(50, nl // 2)
        rng2 = np.random.RandomState(seed + 200)
        for typ in range(3):
            h_l = make_start(n_low, typ, rng2)
            r, c = multi_refine(h_l, n_low, 2.0 / n_low, seed + 300 * (typ + 1), t0, t0 + int(budget_s * 0.85))
            if c < bc:
                xo = np.linspace(0, 2, n_low, endpoint=False)
                xn = np.linspace(0, 2, nl, endpoint=False)
                h_up = repair(np.interp(xn, xo, r, period=2.0), nl)
                r2, c2 = multi_refine(h_up, nl, dx, seed + 500 * (typ + 1), t0, t0 + int(budget_s * 0.92))
                if c2 < bc:
                    bh, bc, bn = r2, c2, nl
            if time.time() - t0 > t0 + budget_s * 0.85:
                break

    # Final polish with small perturbations
    if time.time() - t0 < t0 + budget_s - 10:
        rng3 = np.random.RandomState(seed + 999)
        hp = repair(bh + 0.01 * rng3.randn(bn), bn)
        alloc_final = t0 + budget_s - 5 - time.time()
        if alloc_final > 20:
            r, c = multi_refine(hp, bn, dx, seed + 777, t0, t0 + budget_s - 5)
            if c < bc:
                bh, bc, bn = r, c, bn

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
