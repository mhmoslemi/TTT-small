```python
def _pair_coord_opt(h, i, j, n):
    """Optimize h[i] with h[j] = c - h[i] to preserve sum."""
    c = h[i] + h[j]
    orig_i, orig_j = h[i], h[j]
    
    h[i] = 0.0; h[j] = c  # h[i] = 0, h[j] = c
    c0 = _corr(h)
    h[i] = 1.0; h[j] = c - 1.0  # h[i] = 1, h[j] = c-1
    c1 = _corr(h)
    h[i] = orig_i; h[j] = orig_j
    
    a = c1 - c0  # slope w.r.t. h[i]
    b = c0
    
    # But we need 0 ≤ h[i] ≤ 1 AND 0 ≤ h[j] = c - h[i] ≤ 1
    lo = max(0, c - 1)
    hi = min(1, c)
    if lo >= hi:
        return h  # infeasible
    
    # Ternary search on [lo, hi]
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    x = (lo + hi) / 2.0
    h[i] = x
    h[j] = c - x
    return h
```

```python
                else:
```

```python
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad(h, n, alpha=5000):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -800, 0)
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12: continue
        d = abs(k - (n - 1))
        L = n - d
        wk = w[k] * dx
        if k >= n - 1:
            g[d:] += wk * (1.0 - h[:L])
            g[:L] -= wk * h[d:]
        else:
            g[:L] += wk * (1.0 - h[d:])
            g[d:] -= wk * h[:L]
    return g

def _pair_opt(h, i, j):
    c = h[i] + h[j]
    oi, oj = h[i], h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-12:
        return
    h[i] = lo; h[j] = c - lo
    c0 = _corr(h)
    h[i] = hi; h[j] = c - hi
    c1 = _corr(h)
    a = c1 - c0; b = c0
    lo2, hi2 = lo, hi
    for _ in range(40):
        m1 = lo2 + (hi2 - lo2) / 3.0
        m2 = hi2 - (hi2 - lo2) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b): hi2 = m2
        else: lo2 = m1
    x = (lo2 + hi2) / 2.0
    h[i] = x; h[j] = c - x

def _struct_init(n, rng):
    t = np.arange(n) / n
    r = rng.randint(5)
    if r == 0:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * rng.randint(1, 6) * t + rng.uniform(0, 6.28))
    elif r == 1:
        p = rng.choice([3, 4, 5, 6, 7, 8])
        h = np.where(np.mod(t, 1.0 / p) < 0.5 / p, 0.85, 0.15)
    elif r == 2:
        f1, f2 = rng.randint(1, 5, size=2)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f1 * t) + 0.2 * np.cos(2 * np.pi * f2 * t)
    elif r == 3:
        h = 0.5 + 0.35 * np.cos(2 * np.pi * rng.randint(1, 10) * t)
    else:
        h = 0.5 + 0.3 * rng.randn() * np.sin(2 * np.pi * rng.randint(1, 12) * t)
    return _repair(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n0 = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_c5 = _eval(best_h)
    best_n = n0

    # Try structured inits at various resolutions
    for n_try in [n0, 100, 200, 280]:
        if n_try % 2 != 0 or _el() > budget_s * 0.04: continue
        for _ in range(12):
            hc = _struct_init(n_try, rng)
            c = _eval(hc)
            if c < best_c5:
                best_c5, best_h, best_n = c, hc.copy(), n_try

    # Phase 1: Subgradient descent with exact-max targeting
    h = best_h.copy(); n = best_n
    c5 = _eval(h); step = 0.04
    t_end1 = t0 + budget_s * 0.28
    while time.time() < t_end1:
        g = _subgrad(h, n, alpha=8000)
        h_new = _proj(h - step * g, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5 - 1e-15:
            h = h_new; c5 = c5n
            if c5 < best_c5: best_c5, best_h = c5, h.copy()
        else:
            step *= 0.96
        if step < 1e-6: step = 0.02
        if abs(h.sum() - n / 2) > 1e-6:
            h = _repair(h, n)

    # Phase 2: Paired coordinate descent (main workhorse)
    h = best_h.copy(); n = best_n
    c5 = _eval(h)
    t_end2 = t0 + budget_s * 0.68
    order = np.arange(n)
    sweeps = 0
    while time.time() < t_end2:
        rng.shuffle(order)
        for i in range(0, n - 1, 2):
            _pair_opt(h, order[i], order[i + 1])
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
            if c5 < best_c5: best_c5, best_h = c5, h.copy()
        sweeps += 1
        if sweeps % 500 == 0:
            g = _subgrad(h, n, alpha=20000)
            h = _proj(h - 0.008 * g, n * 0.5)
            h = _repair(h, n)

    # Phase 3: Combined subgrad + coord refinement
    h = best_h.copy(); n = best_n
    c5 = _eval(h)
    t_end3 = t0 + budget_s * 0.86
    while time.time() < t_end3:
        g = _subgrad(h, n, alpha=50000)
        h = _proj(h - 0.005 * g, n * 0.5)
        perm = rng.permutation(n)
        for i in range(0, n - 1, 2):
            _pair_opt(h, perm[i], perm[i + 1])
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
            if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 4: Fine pairwise perturbations
    h = best_h.copy(); n = best_n
    c5 = _eval(h)
    t_end4 = t0 + budget_s - 10
    for scale in [0.003, 0.001, 0.0003, 5e-5, 1e-6]:
        if time.time() > t_end4: break
        cnt = 30000 if scale >= 0.001 else 50000
        for _ in range(cnt):
            if time.time() > t_end4: break
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-16:
                c5 = c5n
                if c5 < best_c5: best_c5, best_h = c5, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    # Phase 5: Targeted multi-shift descent
    h = best_h.copy(); n = best_n
    c5 = _eval(h)
    t_end5 = t0 + budget_s - 3
    while time.time() < t_end5:
        c = _corr(h)
        top = np.argsort(c)[-3:][::-1]
        m = c.max()
        s = np.clip(10000 * (c[top] - m), -50, 0)
        w = np.exp(s); w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for idx, k in enumerate(top):
            d = abs(k - (n - 1))
            L = n - d
            wk = w[idx] * dx
            if k >= n - 1:
                g[d:] += wk * (1.0 - h[:L]); g[:L] -= wk * h[d:]
            else:
                g[:L] += wk * (1.0 - h[d:]); g[d:] -= wk * h[:L]
        h = _proj(h - 0.003 * g, n * 0.5)
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
            if c5 < best_c5: best_c5, best_h = c5, h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```
