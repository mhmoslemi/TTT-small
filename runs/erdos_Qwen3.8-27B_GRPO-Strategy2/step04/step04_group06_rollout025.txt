                h[idx] += add
                diff -= add

                diff += sub
    return h

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def _soft_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            l2 = -lag
            g[:n - l2] += wk * (1.0 - h[l2:])
            g[l2:] -= wk * h[:n - l2]
    return g * dx

def _pgd(h, n, alpha, end, lr, max_it):
    h = h.copy()
    c5 = _c5(h, n)
    for it in range(max_it):
        if time.time() > end:
            break
        g = _soft_grad(h, n, alpha)
        h_new = _project(h - lr * g, n)
        c5_new = _c5(h_new, n)
        if c5_new < c5:
            h = h_new; c5 = c5_new
        else:
            lr *= 0.8
        if it % 500 == 499:
            lr *= 0.75
    return h, c5

def _pair_opt(a, b, lo, hi):
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        mv = vals.max()
        ties = np.where(np.abs(vals - mv) < 1e-13)[0]
        if a[ties].min() > 1e-12:
            hi = mid
        elif a[ties].max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def _coord_descent(h, n, end, rng, max_rounds=2000):
    c5 = _c5(h, n)
    partner = (np.arange(n) + 1) % n
    rounds = 0
    for rnd in range(max_rounds):
        if time.time() > end:
            break
        rounds += 1
        improved = 0
        if rnd < 100:
            order = np.arange(n)
        else:
            order = np.arange(n)
            rng.shuffle(order)
        for j in order:
            i = partner[j]
            vi, vj = h[i], h[j]
            s_ij = vi + vj
            lo_x = max(0.0, s_ij - 1.0)
            hi_x = min(1.0, s_ij)
            if hi_x - lo_x < 1e-14:
                continue
            h[j] = lo_x; h[i] = s_ij - lo_x
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = hi_x; h[i] = s_ij - hi_x
            c2 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = vj; h[i] = vi
            a = c2 - c1
            b = c1 - a * lo_x
            x_opt = _pair_opt(a, b, lo_x, hi_x)
            h[j] = x_opt; h[i] = s_ij - x_opt
            c5_new = _c5(h, n)
            if c5_new < c5 - 1e-16:
                c5 = c5_new; improved += 1
            else:
                h[j] = vj; h[i] = vi
        if improved == 0:
            break
    return h, c5

def _slow_coord(h, n, end, rng):
    c5 = _c5(h, n)
    for _ in range(200000):
        if time.time() > end:
            break
        j = rng.randint(n)
        i = rng.randint(n)
        if i == j:
            continue
        vi, vj = h[i], h[j]
        s_ij = vi + vj
        lo_x = max(0.0, s_ij - 1.0)
        hi_x = min(1.0, s_ij)
        if hi_x - lo_x < 1e-14:
            continue
        h[j] = lo_x; h[i] = s_ij - lo_x
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[j] = hi_x; h[i] = s_ij - hi_x
        c2 = np.correlate(h, 1.0 - h, mode='full')
        h[j] = vj; h[i] = vi
        a = c2 - c1
        b = c1 - a * lo_x
        x_opt = _pair_opt(a, b, lo_x, hi_x)
        h[j] = x_opt; h[i] = s_ij - x_opt
        c5_new = _c5(h, n)
        if c5_new < c5 - 1e-16:
            c5 = c5_new
        else:
            h[j] = vj; h[i] = vi
    return h, c5

def _anneal(h, n, end, rng, T0=0.005):
    c5 = _c5(h, n)
    T = T0
    for it in range(300000):
        if time.time() > end:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.05, 0.05)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_new = _c5(h, n)
        delta = c5_new - c5
        if delta < 0 or (T > 1e-6 and rng.rand() < np.exp(-delta / T)):
            c5 = c5_new
        else:
            h[i], h[j] = h[i] - d, h[j] + d
        if it % 3000 == 2999:
            T *= 0.85
    return h, c5

def _multi_alpha_pgd(h, n, end, rng):
    c5 = _c5(h, n)
    best = h.copy()
    alphas = [500, 2000, 8000, 30000, 80000, 200000]
    per = max(1, int((end - time.time()) / len(alphas)))
    for al in alphas:
        if time.time() > end:
            break
        h_w, c5_w = _pgd(best, n, al, min(time.time() + per, end), 0.02, 3000)
        if c5_w < c5:
            c5 = c5_w
            best = h_w.copy()
    return best, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: PGD with multi-alpha (25%)
    t1 = t0 + budget_s * 0.25
    best_h, best_c5 = _multi_alpha_pgd(best_h, n, t1, rng)

    # Phase 2: Coordinate descent (25%)
    t2 = t0 + budget_s * 0.50
    h, c5 = _coord_descent(best_h, n, t2, rng, max_rounds=3000)
    if c5 < best_c5:
        best_c5 = c5; best_h = h

    # Phase 3: PGD again from improved point (15%)
    t3 = t0 + budget_s * 0.65
    for al in [50000, 150000, 400000]:
        if time.time() > t3:
            break
        h, c5 = _pgd(best_h, n, al, t3, 0.005, 2000)
        if c5 < best_c5:
            best_c5 = c5; best_h = h

    # Phase 4: Random coordinate descent (15%)
    t4 = t0 + budget_s * 0.80
    h, c5 = _slow_coord(best_h, n, t4, rng)
    if c5 < best_c5:
        best_c5 = c5; best_h = h

    # Phase 5: Simulated annealing (10%)
    t5 = t0 + budget_s * 0.90
    h, c5 = _anneal(best_h, n, t5, rng)
    if c5 < best_c5:
        best_c5 = c5; best_h = h

    # Phase 6: Final coord descent polish (8%)
    t6 = t0 + budget_s * 0.98
    h, c5 = _coord_descent(best_h, n, t6, rng, max_rounds=5000)
    if c5 < best_c5:
        best_c5 = c5; best_h = h

    # Phase 7: SLSQP polish if time remains
    if time.time() < t0 + budget_s - 15:
        bnds = [(0.0, 1.0)] * n
        cn = n / 2.0
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
            m = c.max()
            s = np.clip(100000.0 * (c - m), -400, 400)
            return m + np.log(np.sum(np.exp(s))) / 100000.0
        def _grad(x):
            return _soft_grad(x, n, 100000.0)
        def _cons_f(x):
            return x.sum() - cn
        def _cons_j(x):
            return np.ones(n)
        try:
            res = minimize(_obj, best_h, jac=_grad, method='SLSQP',
                           bounds=bnds,
                           constraints=[{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}],
                           options={'maxiter': 400, 'ftol': 1e-18})
            hc = _repair(res.x, n)
            c5v = _c5(hc, n)
            if c5v < best_c5:
                best_c5 = c5v; best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
