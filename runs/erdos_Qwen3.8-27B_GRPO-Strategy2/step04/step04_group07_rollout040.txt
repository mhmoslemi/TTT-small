    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)

                h[idx] += a
                d -= a

    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)

    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(10000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_obj(v, alpha):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_obj(v, alpha):
    return soft_obj(v, alpha)

def slsqp_con(v, n):
    return v.sum() - n / 2.0

def slsqp_opt(h, n, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - target}]
    try:
        res = minimize(soft_obj, h, args=(alpha,), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def subspace_con(u, target_free):
    return u.sum() - target_free

def subspace_opt(h, n, alpha, tol, maxiter=300):
    free = (h > tol) & (h < 1.0 - tol)
    nfree = int(np.sum(free))
    if nfree < 2 or nfree == n:
        return slsqp_opt(h, n, alpha, maxiter)
    fixed = ~free
    target_free = n / 2.0 - h[fixed].sum()
    bounds = [(0.0, 1.0)] * nfree
    cons = [{'type': 'eq', 'fun': lambda u: u.sum() - target_free}]
    def _sub_obj(u):
        v = h.copy()
        v[free] = u
        v = np.clip(v, 0.0, 1.0)
        return soft_obj(v, alpha)
    try:
        res = minimize(_sub_obj, h[free], method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        v = h.copy()
        v[free] = res.x
        return repair(v, n)
    except Exception:
        return h

def coord_descent(h, n, n_sweeps, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    for r in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        nsample = min(10 * n, 1500)
        for _ in range(nsample):
            if time.time() > t_end:
                break
            i = int(rng.randint(n))
            j = (i + 1 + int(rng.randint(n - 1))) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hid + delta, hjd - delta
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
                if improved:
                    break
        if not improved and r > 80:
            break
    return h, best

def targeted_cd(h, n, t_end, rng, n_top=5):
    """Focus coordinate descent on lags near the maximum."""
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0 / n
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for r in range(2000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        improved = False
        nsample = min(8 * n, 1200)
        for _ in range(nsample):
            if time.time() > t_end:
                break
            i = int(rng.randint(n))
            j = (i + 1 + int(rng.randint(n - 1))) % n
            hid, hjd = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hid + delta, hjd - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c2 = np.correlate(h, 1.0 - h, mode='full')
                    top_max = max(c2[k] for k in top_idx) * dx
                    if top_max < best - 1e-15:
                        best = top_max
                        improved = True
                        break
                    h[i], h[j] = hid, hjd
                if improved:
                    break
        if not improved and r > 100:
            break
    return h, best

def multi_start(h, n, alpha, rng, sigma, n_starts):
    best = h.copy()
    best_c5 = compute_c5(h, n)
    for _ in range(n_starts):
        if time.time() > 99999:
            break
        h_p = h + sigma * rng.randn(n)
        h_p = repair(h_p, n)
        h_new = slsqp_opt(h_p, n, alpha, 200)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best, best_c5 = h_new, c5_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 777)

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = np.inf
    best_n = 64

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(init_h)
        init_h = repair(init_h, best_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, best_n)
    except NameError:
        best_n = 64
        x = np.linspace(0, 2, best_n)
        best_h = repair(0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x), best_n)
        best_c5 = compute_c5(best_h, best_n)

    n = best_n
    h_work = best_h.copy()

    # Phase 1: Escalating SLSQP (18%)
    t1 = t0 + budget_s * 0.18
    for alpha in [3000, 10000, 30000, 100000, 300000, 1000000, 3000000, 10000000]:
        if time.time() > t1:
            break
        h_new = slsqp_opt(h_work.copy(), n, alpha, 600)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5, h_work = h_new.copy(), c5_new, h_new
        elif c5_new < compute_c5(h_work, n) - 1e-14:
            h_work = h_new

    # Phase 2: Subspace optimization (14%)
    t2 = t0 + budget_s * 0.32
    if time.time() < t2:
        for tol in [0.04, 0.015, 0.005]:
            if time.time() > t2:
                break
            for alpha in [200000, 1000000, 5000000]:
                if time.time() > t2:
                    break
                h_new = subspace_opt(h_work, n, alpha, tol, 300)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_c5 - 1e-14:
                    best_h, best_c5, h_work = h_new.copy(), c5_new, h_new

    # Phase 3: General coordinate descent (25%)
    t3 = t0 + budget_s * 0.57
    if time.time() < t3:
        h_cd, c5_cd = coord_descent(h_work, n, 5000, t3, rng)
        if c5_cd < best_c5 - 1e-14:
            best_h, best_c5, h_work = h_cd.copy(), c5_cd, h_cd

    # Phase 4: Targeted coordinate descent on top lags (13%)
    t4 = t0 + budget_s * 0.70
    if time.time() < t4:
        h_tgt, c5_tgt = targeted_cd(h_work, n, t4, rng, n_top=5)
        if c5_tgt < best_c5 - 1e-14:
            best_h, best_c5, h_work = h_tgt.copy(), c5_tgt, h_tgt

    # Phase 5: SLSQP refinement + multi-start (13%)
    t5 = t0 + budget_s * 0.83
    if time.time() < t5:
        for alpha in [3000000, 10000000, 30000000]:
            if time.time() > t5 * 0.7:
                break
            h_new = slsqp_opt(h_work.copy(), n, alpha, 400)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5 - 1e-14:
                best_h, best_c5, h_work = h_new.copy(), c5_new, h_new
        if time.time() < t5:
            for sigma in [0.015, 0.03]:
                if time.time() > t5:
                    break
                h_ms, c5_ms = multi_start(best_h, best_n, 3000000, rng, sigma, 3)
                if c5_ms < best_c5 - 1e-14:
                    best_h, best_c5, h_work = h_ms.copy(), c5_ms, h_ms

    # Phase 6: Try other n values (10%)
    t6 = t0 + budget_s * 0.93
    if time.time() < t6:
        for n_alt in [48, 56, 72, 80, 96]:
            if time.time() > t6:
                break
            if n_alt == best_n:
                continue
            x_old = np.linspace(0.0, 2.0, best_n)
            x_new = np.linspace(0.0, 2.0, n_alt)
            h_seed = np.interp(x_new, x_old, best_h)
            h_seed = repair(h_seed, n_alt)
            h_alt = slsqp_opt(h_seed, n_alt, 1000000, 400)
            h_alt = subspace_opt(h_alt, n_alt, 5000000, 0.01, 200)
            h_alt2, c5_alt2 = coord_descent(h_alt, n_alt, 300, t6, rng)
            if c5_alt2 < best_c5 - 1e-14:
                best_h, best_c5, best_n = h_alt2.copy(), c5_alt2, n_alt

    # Phase 7: Final ultra-fine polish (7%)
    t7 = t0 + budget_s * 0.995
    if best_h is not None and time.time() < t7:
        h_fin = slsqp_opt(best_h.copy(), best_n, 10000000, 500)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h, best_c5 = h_fin.copy(), c5_fin
        h_fin2, c5_fin2 = coord_descent(best_h, best_n, 2000, t7, rng)
        if c5_fin2 < best_c5 - 1e-14:
            best_h, best_c5 = h_fin2.copy(), c5_fin2
        h_fin3, c5_fin3 = targeted_cd(best_h, best_n, t7, rng, n_top=3)
        if c5_fin3 < best_c5 - 1e-14:
            best_h, best_c5 = h_fin3.copy(), c5_fin3

    if best_h is None:
        best_n = 64
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
