    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if pw[k] > 1e-14:
            g += pw[k] * grad1(x, k, n)
    g /= _G['a']
    g += 2.0 * _G['l'] * (x.sum() - n * 0.5)
    return g

def _ceq(x):
    return x.sum() - _G['n'] * 0.5

def _ceqj(x):
    return np.ones(_G['n'])

def lbfgs_run(h, n, dx, t0, lim):
    _G['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    for a, lm in [(20, 40), (80, 25), (300, 15), (1200, 8), (4000, 4), (15000, 2)]:
        if time.time() - t0 > lim:
            break
        _G['a'], _G['l'] = float(a), float(lm)
        try:
            r = minimize(_obj, bh.copy(), jac=_jac, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-18})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass
    return bh, bc

def slsqp_run(h, n, dx, t0, lim):
    _G['n'] = n
    bds = [(0.0, 1.0)] * n
    bh, bc = h.copy(), c5v(h, dx)
    for a in [2000, 10000, 50000, 200000]:
        if time.time() - t0 > lim:
            break
        _G['a'], _G['l'] = float(a), 0.0
        try:
            r = minimize(_obj, bh, method='SLSQP', bounds=bds,
                         constraints=[{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}],
                         options={'maxiter': 1000, 'ftol': 1e-20})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass
    return bh, bc

def act_mom(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for mt in [3, 5, 8, 15, 25]:
        h = bh.copy()
        v = np.zeros(n)
        step = 0.04
        mu = 0.65
        stalled = 0
        for _ in range(1000):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.0015 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            w = c[idx] - thr + 1e-7
            w /= w.sum()
            g = np.zeros(n)
            for ki, wi in zip(idx, w):
                g += wi * grad1(h, ki, n)
            g -= g.mean()
            v = mu * v + (1.0 - mu) * g
            ok = False
            for _b in range(50):
                hn = repair(h - step * v, n)
                cn = c5v(hn, dx)
                if cn < c5v(h, dx) - 1e-19:
                    h = hn
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.08, 0.8)
                    stalled = 0
                    ok = True
                    break
                step *= 0.72
            if not ok:
                stalled += 1
                v *= 0.5
                if stalled > 12:
                    break
    return bh, bc

def spec_eq(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.08, 0.15, 0.25, 0.4, 0.55, 0.7, 0.55, 0.4, 0.25, 0.15, 0.08, 0.04]:
        h = bh.copy()
        for _ in range(500):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-19:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def coord_desc(h, n, dx, rng, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    halts = 0
    for _ in range(400):
        if time.time() - t0 > lim:
            break
        imp = False
        for _ in range(1000):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in (0.006, 0.002, 0.0008, 0.0003, -0.006, -0.002, -0.0008, -0.0003):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = c5v(h, dx)
                    if c < cc - 1e-16:
                        h = repair(h, n)
                        cc = c5v(h, dx)
                        imp = True
                        if cc < bc:
                            bc, bh = cc, h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
        halts = halts + 1 if not imp else 0
        if halts >= 5:
            break
    return bh, bc

def polish(h, n, dx, rng, cnt, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    dt = np.array([1e-4, 3e-4, 7e-4, 1.5e-3, 3e-3, -1e-4, -3e-4, -7e-4, -1.5e-3, -3e-3])
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = dt[rng.randint(len(dt))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc - 1e-18:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def optimize(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_run(bh, n, dx, t0, lim - 280)
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_run(bh, n, dx, t0, lim - 200)
    if c < bc:
        bh, bc = r, c
    r, c = act_mom(bh, n, dx, t0, lim - 140)
    if c < bc:
        bh, bc = r, c
    r, c = spec_eq(bh, n, dx, t0, lim - 100)
    if c < bc:
        bh, bc = r, c
    r, c = lbfgs_run(bh, n, dx, t0, lim - 70)
    if c < bc:
        bh, bc = r, c
    r, c = act_mom(bh, n, dx, t0, lim - 45)
    if c < bc:
        bh, bc = r, c
    r, c = spec_eq(bh, n, dx, t0, lim - 25)
    if c < bc:
        bh, bc = r, c
    r, c = coord_desc(bh, n, dx, rng, t0, lim - 10)
    if c < bc:
        bh, bc = r, c
    r, c = polish(bh, n, dx, rng, 40000, t0, lim - 1)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    r, c = optimize(h0, nl, dx0, seed, t0, budget_s - 150)
    if c < bc:
        bh, bc, bn = r, c, nl

    if time.time() - t0 < budget_s - 50:
        rng2 = np.random.RandomState(seed + 77)
        hp = repair(bh + 0.008 * rng2.randn(bn), bn)
        r, c = optimize(hp, bn, 2.0 / bn, seed + 177, t0, budget_s - 10)
        if c < bc:
            bh, bc, bn = r, c, bn

    n2 = nl * 2
    if n2 < 800 and time.time() - t0 < budget_s - 30:
        xo = np.linspace(0, 2, nl, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = optimize(h2, n2, d2, seed + 300, t0, budget_s - 2)
        if c < bc:
            bh, bc, bn = r, c, n2

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
