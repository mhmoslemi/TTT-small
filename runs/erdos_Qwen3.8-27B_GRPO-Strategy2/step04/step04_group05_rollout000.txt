            r = minimize(_sm_obj, bh, method='SLSQP', bounds=bds,
                         constraints=[{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}],
                         options={'maxiter': 600, 'ftol': 1e-19})
            hc = repair(r.x, n)
            c = c5v(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass
    return bh, bc

def spectral_phase(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5v(h, dx)
    for a in [0.2, 0.35, 0.5, 0.3, 0.15, 0.08]:
        if time.time() - t0 > lim:
            break
        h = bh.copy()
        for _ in range(200):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H)
            ph = np.angle(H)
            R0 = max(np.dot(h, h), 1e-10)
            tm = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tm[0] = abs(H[0])
            nm = (1 - a) * mag + a * tm
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5v(hr, dx)
            if cn < c5v(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def coord_polish(h, n, dx, rng, t0, lim, cnt=40000):
    bh, bc = h.copy(), c5v(h, dx)
    cc = bc
    for _ in range(cnt):
        if time.time() - t0 > lim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.choice([0.002, 0.005, 0.001, -0.002, -0.005, -0.001, 0.008, -0.008])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5v(h, dx)
            if c < cc:
                h = repair(h, n)
                cc = c5v(h, dx)
                if cc < bc:
                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def opt_h(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5v(h, dx)
    r, c = lbfgs_phase(bh, n, dx, t0, lim - 350)
    if c < bc:
        bh, bc = r, c
    r, c = actset_phase(bh, n, dx, t0, lim - 220)
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_phase(bh, n, dx, t0, lim - 150)
    if c < bc:
        bh, bc = r, c
    r, c = spectral_phase(bh, n, dx, t0, lim - 110)
    if c < bc:
        bh, bc = r, c
    r, c = actset_phase(bh, n, dx, t0, lim - 60)
    if c < bc:
        bh, bc = r, c
    r, c = slsqp_phase(bh, n, dx, t0, lim - 30)
    if c < bc:
        bh, bc = r, c
    r, c = coord_polish(bh, n, dx, rng, t0, lim - 3, cnt=30000)
    if c < bc:
        bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    dx0 = 2.0 / nl
    bh, bc, bn = h0.copy(), c5v(h0, dx0), nl

    r, c = opt_h(h0, nl, dx0, seed, t0, budget_s - 150)
    if c < bc:
        bh, bc, bn = r, c, nl

    # Try a perturbed restart at same resolution
    if time.time() - t0 < budget_s - 120:
        rng2 = np.random.RandomState(seed + 300)
        hp = repair(bh + 0.02 * rng2.randn(bn), bn)
        r, c = opt_h(hp, bn, 2.0 / bn, seed + 700, t0, budget_s - 100)
        if c < bc:
            bh, bc, bn = r, c, bn

    # Resolution doubling
    n2 = nl * 2
    if n2 < 1000 and time.time() - t0 < budget_s - 100:
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        d2 = 2.0 / n2
        r, c = opt_h(h2, n2, d2, seed + 1, t0, budget_s - 12)
        if c < bc:
            bh, bc, bn = r, c, n2

    # Final intensive polish
    if time.time() - t0 < budget_s - 8:
        rng3 = np.random.RandomState(seed + 999)
        dxf = 2.0 / bn
        r, c = actset_phase(bh, bn, dxf, t0, budget_s - 5)
        if c < bc:
            bh, bc = r, c
        r, c = coord_polish(bh, bn, dxf, rng3, t0, budget_s - 2, cnt=20000)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
