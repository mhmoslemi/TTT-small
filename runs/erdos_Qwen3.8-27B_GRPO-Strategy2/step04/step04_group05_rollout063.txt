import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    # Clip to [0,1], scale to sum = n/2
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            if d > 0:
                amt = min(d, 1.0 - h[i])
            else:
                amt = max(d, -h[i])
            h[i] += amt
            d -= amt
    return h

def c5_eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def smooth_obj(x, n, temp, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(temp * (c - m), -500, 500)
    lse = m + np.log(np.sum(np.exp(s))) / temp
    return lse + lam * (x.sum() - n * 0.5) ** 2

def smooth_grad(x, n, temp, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(temp * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if pw[k] < 1e-12:
            continue
        w = pw[k] / temp
        L = k - (n - 1)
        if L == 0:
            g += w * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += w * (1.0 - x[:n-L])
            g[:n-L] -= w * x[L:]
        else:
            mm = -L
            g[:n-mm] += w * (1.0 - x[mm:])
            g[mm:] -= w * x[:n-mm]
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def active_grad(h, idx, n):
    g = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        if L == 0:
            g[:] += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:n-L]
            g[:n-L] -= h[L:]
        else:
            mm = -L
            g[:n-mm] += 1.0 - h[mm:]
            g[mm:] -= h[:n-mm]
    return g / len(idx)

def lbfgs_run(h, n, temp, lam, t0, lim, maxiter=300):
    bds = [(0.0, 1.0)] * n
    def _f(x):
        return smooth_obj(x, n, temp, lam)
    def _g(x):
        return smooth_grad(x, n, temp, lam)
    try:
        r = minimize(_f, h.copy(), jac=_g, method='L-BFGS-B',
                     bounds=bds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return repair(r.x, n)
    except:
        return h

def de_global(h_best, n, dx, seed, t0, lim):

    CR = 0.8

            a, b, c_ = rng.choice([j for j in range(pop_size) if j != i], 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c_])

Actually, I want to try something different from the previous solution. Let me use **projected gradient / Frank-Wolfe (conditional gradient)** method.

The Frank-Wolfe method works as follows:
1. At each iteration, compute the gradient g of the objective
2. Find the minimizer of the linear approximation over the feasible set (the "linear minimization oracle")
3. Take a step towards this minimizer

The feasible set here is: {h : 0 ≤ h[i] ≤ 1, ∑h[i] = n/2}

The linear minimization oracle: minimize ∑ g[i]*h[i] subject to 0 ≤ h[i] ≤ 1, ∑h[i] = n/2.

This is a transportation problem! The optimal solution is: set the top n/2 entries of -g to 1, the bottom n/2 to 0. (For n even, which it is for n=140.)

More precisely: sort indices by g[i], set h[i]=1 for the n/2 smallest g[i], h[i]=0 for the rest.

The Frank-Wolfe step: h_new = (1-t)*h + t*h_FW where t is the step size.

This is appealing because:
1. The oracle is very fast (just a sort, O(n log n))
2. No need for repair function (the feasible set is maintained exactly)
3. Convex in the relaxed sense (though the problem is non-convex)
4. Can be combined with the smooth max objective

But the problem is non-convex, so FW might not converge to the global optimum. Still, it can be a good complement to L-BFGS-B.

def repair(h, n):
    h = np.clip(h.astype(float), 0, 1)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d = t - h.sum()
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n - L]
        g[:n - L] -= h[L:]
    else:
        m = -L
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]

            r = minimize(so, bh.copy(), jac=sj, method='L-BFGS-B',
                         bounds=bds, options={'maxiter': mi, 'ftol': 1e-19,
                         'gtol': 1e-16})
            hc = repair(r.x, n)
            c = c5c(hc, 2.0 / n)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass
    return bh, bc

def ph_fw(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5c(h, dx)
    for a, l in [(50,100),(200,50),(1000,20),(5000,10)]:
        GR['n'], GR['a'], GR['l'] = n, float(a), float(l)
        h = bh.copy()
        for it in range(800):
            if time.time() - t0 > lim:
                break
            grad = sj(h)
            o = np.argsort(grad)
            hfw = np.zeros(n); hfw[o[:n // 2]] = 1.0
            cur = c5c(h, dx)
            for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
                hn = np.clip((1 - t) * h + t * hfw, 0, 1)
                hn = repair(hn, n)
                cn = c5c(hn, dx)
                if cn < cur - 1e-15:
                    h = hn
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    break
            else:
                if it > 50:
                    break
    return bh, bc

def ph_act(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5c(h, dx)
    for mt in [1, 2, 3, 5]:
        h = bh.copy()
        step = 0.06
        cur = c5c(h, dx)
        for _ in range(500):
            if time.time() - t0 > lim:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            cm = c.max()
            thr = cm - 0.003 * max(c.std(), 1e-10)
            idx = np.where(c >= thr)[0]
            if len(idx) > mt:
                idx = np.argsort(c)[::-1][:mt]
            g = sum(gk(h, k, n) for k in idx) / len(idx)
            g -= g.mean()
            ok = False
            for _ in range(30):
                hn = repair(h - step * g, n)
                cn = c5c(hn, dx)
                if cn < cur - 1e-18:
                    h = hn; cur = cn; ok = True
                    if cn < bc:
                        bc, bh = cn, h.copy()
                    step = min(step * 1.1, 1.5)
                    break
                step *= 0.7
            if not ok:
                break
    return bh, bc

def ph_spec(h, n, dx, t0, lim):
    bh, bc = h.copy(), c5c(h, dx)
    for alpha in [0.2, 0.35, 0.5, 0.3, 0.15, 0.1]:
        h = bh.copy()
        for _ in range(200):
            if time.time() - t0 > lim:
                break
            H = np.fft.rfft(h)
            mag = np.abs(H); ph = np.angle(H)
            R0 = max(float(np.dot(h, h)), 1e-10)
            tgt = np.full(n // 2 + 1, np.sqrt(R0 / n))
            tgt[0] = abs(H[0])
            nm = (1 - alpha) * mag + alpha * tgt
            hn = np.fft.irfft(nm * np.exp(1j * ph), n)
            hr = repair(hn, n)
            cn = c5c(hr, dx)
            if cn < c5c(h, dx) - 1e-18:
                h = hr
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                break
    return bh, bc

def ph_cd(h, n, dx, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy()
    cc = c5c(h, dx)
    while time.time() - t0 < lim:
        imp = False
        for _ in range(5000):
            if time.time() - t0 > lim:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice([0.002, 0.001, 0.0005, 0.0002,
                            -0.002, -0.001, -0.0005, -0.0002])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = c5c(h, dx)
                if c < cc - 1e-15:
                    h = repair(h, n); cc = c5c(h, dx); imp = True
                    if cc < bc:
                        bc, bh = cc, h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        if not imp:
            break
    return bh, bc

def local_opt(h, n, dx, seed, t0, lim):
    bh, bc = h.copy(), c5c(h, dx)
    r, c = ph_lbfgs(bh, n, t0, lim)
    if c < bc: bh, bc = r, c
    r, c = ph_fw(bh, n, dx, t0, lim)
    if c < bc: bh, bc = r, c
    r, c = ph_act(bh, n, dx, t0, lim)
    if c < bc: bh, bc = r, c
    r, c = ph_spec(bh, n, dx, t0, lim)
    if c < bc: bh, bc = r, c
    r, c = ph_lbfgs(bh, n, t0, lim)
    if c < bc: bh, bc = r, c
    r, c = ph_cd(bh, n, dx, seed, t0, lim)
    if c < bc: bh, bc = r, c
    return bh, bc
