    if _time_left(t0, budget_s) > 60:
        t_end = time.time() + min(35, _time_left(t0, budget_s) - 20)
        h2, c5_2 = _pair_swap_search(best_h.copy(), best_n, rng, t_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 4: SLSQP polish at high alpha
    if _time_left(t0, budget_s) > 40:
        h4 = _opt_slsqp(best_h.copy(), best_n, 2000000, maxiter=400, ftol=1e-22)
        c5_4 = _eval(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Micro-scale search
    if _time_left(t0, budget_s) > 30:
        t_end = time.time() + min(20, _time_left(t0, budget_s) - 10)
        h5, c5_5 = _micro_search(best_h.copy(), best_n, rng, t_end)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final tiny perturbations
    if _time_left(t0, budget_s) > 10:
        h = best_h.copy()
        c5c = _eval(h)
        t_end = time.time() + min(7, _time_left(t0, budget_s) - 3)
        for _ in range(300000):
            if time.time() > t_end:
                break
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * rng.choice([1, 3, 10]) * 1e-7
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
