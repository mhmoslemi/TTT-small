import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def logsumexp_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(logsumexp_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_con(x, target):
    return np.sum(x) - target

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(slsqp_obj, h, args=(alpha, target), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def greedy_local_search(h, n, dx, rng, deadline, iters):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for it in range(iters):
        if time.time() >= deadline:
            break
        # Try several random pair moves, accept the best improving one
        best_delta = 0
        best_move = None
        for trial in range(8):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.001, 0.04) * (1.0 if rng.random() < 0.5 else -1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            delta = best_c5 - c5n
            if delta > best_delta:
                best_delta = delta
                best_move = (i, j, d)
            h[i] -= d
            h[j] += d
        if best_move is not None:
            i, j, d = best_move
            h[i] += d
            h[j] -= d
            h = np.clip(h, 0.0, 1.0)
            best_c5 += -best_delta
        elif it % 500 == 0:

def sa_search(h, n, dx, rng, deadline, t0, t1):

        idx = rng.choice(n, n_pts, replace=False)

        deltas -= deltas.mean()
        h_try = h.copy()

        c5_try = compute_c5(h_try, dx)
        delta_c = c5_try - cur_c5

            cur_c5 = c5_try

                best_c5 = c5_try

            no_improve += 1
        T *= 0.99998
        if T < 0.0003:
            T = 0.0003
        if no_improve > 3000:

def block_search(h, n, dx, rng, deadline, iters):

def perturb_init(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        shift = rng.randint(n//4, 3*n//4)
        hp = np.roll(hp, shift)
    elif kind == 1:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 2*np.pi)
        hp = hp + rng.uniform(0.02, 0.08) * np.sin(2*np.pi*f*np.arange(n)/n + ph)
    elif kind == 2:
        hp = hp + rng.normal(0, 0.03, n)
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 6)
        t = np.arange(n)/n
        hp = hp + rng.uniform(0.02, 0.06)*np.sin(2*np.pi*f1*t)*np.cos(2*np.pi*f2*t)
    else:
        hp = hp * rng.uniform(0.8, 1.2)
    return repair(hp, n)
