```python
import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _subgrad_active(h, thresh_frac):
    c = _corr(h)
    n = len(h)
    cm = np.max(c)
    spread = cm - np.min(c)
    if spread < 1e-12:
        spread = 1.0
    thresh = cm - thresh_frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _sumfun(x, target):
    return np.sum(x) - target

def _sumjac(x, n):
    return np.ones(n)

def _nesterov_subgrad(h, n, dx, rng, t_start, t_end, init_step):
    h = h.copy()
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    v = np.zeros(n)
    step = init_step
    for it in range(5000):
        if time.time() - t_start > t_end:
            break
        g = _subgrad_active(h, 0.08)
        gc = g - g.mean()
        ng = np.linalg.norm(gc)
        if ng < 1e-12:
            break
        gc /= ng
        v = 0.5 * v + 0.5 * gc
        h_try = _repair(h - step * v, n)
        c_try = _c5(h_try, dx)

            c5c = c_try

                best_c5 = c5c

            step = min(step * 1.03, 0.05)
        else:
            v = 0.3 * v
            step *= 0.7
            if step < 1e-8:
                break
    return best_h, best_c5

def _block_search(h, n, dx, rng, t_start, t_end, max_iters):
    h = h.copy()
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    deltas = np.array([0.001, 0.003, 0.008, 0.02, -0.001, -0.003, -0.008, -0.02])
    for it in range(max_iters):
        if time.time() - t_start > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, min(n, 20))) % n
        k = (i + rng.randint(min(n,20), min(n, 50))) % n
        if i == j or i == k or j == k:
            continue
        d1 = deltas[rng.randint(len(deltas))]
        d2 = deltas[rng.randint(len(deltas))]
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-(d1+d2)
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c = _c5(h, dx)
            if c < c5c:
                c5c = c
                best_c5 = c
                best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+(d1+d2)
    return best_h, best_c5

def _pairwise_search(h, n, dx, rng, t_start, t_end, max_iters, dlist):

        d = dlist[rng.randint(len(dlist))]

def _sa_search(h, n, dx, rng, t_start, t_end, t0_sa, t1_sa, max_iters):

    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    t = t0_sa

        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _c5(h, dx)
            delta = c_new - c5c
            if delta < 0 or rng.uniform() < np.exp(-delta / t):

                    best_c5 = c5c
                    best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
        t = max(t1_sa, t * 0.997)
    return best_h, best_c5
```
