def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _obj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

        else:

            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _subgrad_dir(h, n, top_frac):
    c = _corr(h)
    cm = np.max(c)
    spread = cm - np.min(c)
    if spread < 1e-12:
        spread = 1.0
    thresh = cm - top_frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def _nlp_optimize(h0, n, dx, bounds, t0, tlim):
    best_h, best_c5 = h0.copy(), _c5(h0, dx)
    alphas = [1500, 8000, 40000, 200000]
    for a in alphas:
        if time.time() - t0 > tlim:
            break
        try:
            res = minimize(partial(_softobj, alpha=a, lam=40.0), best_h,
                          jac=partial(_softgrad, alpha=a, lam=40.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-20})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    return best_h, best_c5

def _subgrad_opt(h0, n, dx, t0, tlim):
    best_h, best_c5 = h0.copy(), _c5(h0, dx)
    step = 0.018
    for tf in [0.08, 0.12, 0.20, 0.35]:
        if time.time() - t0 > tlim:
            break
        for it in range(600):
            if time.time() - t0 > tlim:
                break
            g = _subgrad_dir(best_h, n, tf)
            g_c = g - g.mean()
            ng = np.linalg.norm(g_c)
            if ng < 1e-13:
                break
            g_c /= ng
            found = False
            for st in [step, step * 0.5, step * 0.25, step * 0.1]:
                h_new = _repair(best_h - st * g_c, n)
                c_new = _c5(h_new, dx)
                if c_new < best_c5 - 1e-15:
                    best_h, best_c5 = h_new, c_new
                    found = True
                    break
            if not found:
                step *= 0.5
                if step < 1e-8:
                    break
            else:
                step = min(step * 1.06, 0.045)
    return best_h, best_c5

    dlist = np.array([0.0003, 0.0008, 0.002, 0.005, 0.01, 0.02,

                        improved = True
                        break
                    else:
                        h0 = best_h.copy()
        if not improved:
            break
    return best_h, best_c5

def _interp_h(h_f, n_f, n_t):
    x_f = np.linspace(0, 2, n_f, endpoint=False)
    x_t = np.linspace(0, 2, n_t, endpoint=False)
    return np.interp(x_t, x_f, np.concatenate([h_f, h_f[:1]]), period=0)
