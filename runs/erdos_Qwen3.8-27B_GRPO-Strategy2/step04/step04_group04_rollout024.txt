```python
def _corr_grad(h, weights):
    """Compute gradient of sum_s weights[s] * C[s] w.r.t. h
    weights is indexed from -(n-1) to (n-1)
    C[s] = sum_i h[i+s]*(1-h[i]) (h=0 outside [0,n-1])
    """
    n = len(h)
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = weights[n-1+s]
        if abs(w) < 1e-15:
            continue
        if s >= 0:
            # C = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # d/dh[j] = (1-h[j-s]) for j in [s, n-1] minus h[j+s] for j in [0, n-1-s]
            grad[s:] += w * (1 - h[:n-s])  # (1-h[j-s]) for j=s..n-1, h[j-s] = h[0..n-1-s]
            grad[:n-s] -= w * h[s:]  # -h[j+s] for j=0..n-1-s
        else:
            ss = -s
            # C = sum_{i=ss}^{n-1} h[i-ss]*(1-h[i]) = sum_{j=0}^{n-1-ss} h[j]*(1-h[j+ss])

            grad[:n-ss] += w * (1 - h[ss:])  # (1-h[j+ss]) for j=0..n-1-ss
            grad[ss:] -= w * h[:n-ss]  # -h[j-ss] for j=ss..n-1
```

```python
def _project_simplex(h, target):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = target}"""
    h = np.clip(h, 0, 1).copy()
    n = len(h)
    # Need sum(h) = target = n/2
    # Use water-filling: find tau such that sum(clip(h - tau, 0, 1)) = target
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        h = np.clip(h + (target - s) / n, 0, 1)
    # Final fix
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1 - h)  # most room first
        else:
            idx = np.argsort(h)  # least value first
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h
```

```python
            else:
```

```python
    """Gradient of sum_s w[s]*C[s] at h"""
    n = len(h)
    grad = np.zeros(n)
    for si in range(2 * n - 1):
        wi = w[si]
        if wi < 1e-14:

            grad += wi * (1.0 - 2.0 * h)

        else:
            ss = -s
            grad[:n-ss] += wi * (1.0 - h[ss:])
            grad[ss:] -= wi * h[:n-ss]
    return grad

    cs = c - c.max()
    w = np.exp(alpha * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / ws

        if cc < best_c - 1e-15:

            best_h = h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _interp(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _wgrad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for si in range(2 * n - 1):
        wi = w[si]
        if wi < 1e-14:
            continue
        s = si - (n - 1)
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] -= wi * h[s:]
        else:
            ss = -s
            grad[:n - ss] += wi * (1.0 - h[ss:])
            grad[ss:] -= wi * h[:n - ss]
    return grad

def _pg_optimize(h, dx, t_end, alpha_start, alpha_end, lr):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    st_time = time.time()
    elapsed_init = st_time - t0_global
    total = max(1000, int((t_end - st_time) / 0.001))
    for it in range(total):
        if time.time() > t_end:
            break
        frac = min(1.0, it / total)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        c = _corr(h)
        cs = c - c.max()
        w = np.exp(alpha * cs)
        ws = w.sum()
        if ws > 1e-300:
            w = w / ws
        else:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        g = _wgrad(h, w)
        step = lr * (1.0 - 0.5 * frac)
        z = h - step * g
        h = _project(z, tgt)
        cc = _c5(h, dx)
        if cc < best_c:
            best_c = cc
            best_h = h.copy()
    return best_h, best_c

def _slsqp_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_polish(h, dx, t_end, alpha=20000):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n)}]
    best_h, best_c = h, _c5(h, dx)
    for a in [alpha]:
        if time.time() > t_end:
            break
        try:
            r = minimize(lambda x: _slsqp_obj(x, a), h, method='SLSQP',
                         bounds=bds, constraints=cons,
                         options={'maxiter': 600, 'ftol': 1e-18})
            hc = _project(r.x, tgt)
            cc = _c5(hc, dx)
            if cc < best_c:
                best_h, best_c = hc, cc
        except Exception:
            pass
    return best_h, best_c

def _lbfgs_polish(h, dx, t_end, alpha=5000):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    best_h, best_c = h, _c5(h, dx)
    for a in [alpha, alpha * 4]:
        if time.time() > t_end:
            break
        try:
            def obj_pen(x, pen=5000.0):
                return _slsqp_obj(x, a) + pen * (x.sum() - tgt) ** 2
            r = minimize(obj_pen, h, method='L-BFGS-B', bounds=bds,
                         options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-9})
            hc = _project(r.x, tgt)
            cc = _c5(hc, dx)
            if cc < best_c:
                best_h, best_c = hc, cc
        except Exception:
            pass
    return best_h, best_c

def _local_search(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per_scale = max(50, n_iter // len(scales))
    for sc in scales:
        for _ in range(per_scale):
            if time.time() > t_end:
                return h, best
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if -1e-10 <= h[i] <= 1.0 + 1e-10 and -1e-10 <= h[j] <= 1.0 + 1e-10:
                c = _c5(h, dx)
                if c < best:
                    best = c
                    h = _project(h, tgt)
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _informed_local(h, dx, rng, t_end, n_iter):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        mx = c.max()
        active = np.where(c > mx - 0.001 * mx)[0]
        k_active = rng.choice(active)
        s = k_active - (n - 1)
        if s == 0:
            i = rng.randint(n)
            j = (i + 1) % n
        elif s > 0:
            valid_i = np.arange(0, n - s)
            i = int(rng.choice(valid_i)) + s
            j = int(rng.choice(valid_i))
        else:
            ss = -s
            valid_j = np.arange(0, n - ss)
            j = int(rng.choice(valid_j))
            i = int(rng.choice(valid_j)) + ss
        d = rng.uniform(-0.003, 0.003)
        h[i] += d; h[j] -= d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            cc = _c5(h, dx)
            if cc < best:
                best = cc
                h = _project(h, tgt)
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return h, best
```
