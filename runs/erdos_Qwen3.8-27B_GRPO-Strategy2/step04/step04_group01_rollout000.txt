    h = np.clip(h, 0.0, 1.0).copy()

        return h

        if s > target:

        else:

        return repair(h, n)

    """Descend using average gradient over top n_active constraints."""
    n = len(h)
    f = eval_c5(h)

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _overlaps(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, kidx, n):
    dx = 2.0 / n
    sh = (n - 1) - kidx
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _smooth_obj(h, alpha):
    ov = _overlaps(h)
    m = ov.max()
    s = np.clip(alpha * (ov - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    ov = _overlaps(h)
    m = ov.max()
    s = np.clip(alpha * (ov - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            g += w[k] * _grad_k(h, k, n)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cons_fun(x, n):
    return x.sum() - n * 0.5

def _cons_jac(x, n):
    return np.ones(len(x))

def _dual_weights(h, n_active):
    ov = _overlaps(h)
    m = ov.max()
    threshold = m - max(1e-5, 0.005 * m)
    active = np.where(ov >= threshold)[0]
    if len(active) > n_active:
        top = np.argsort(ov[active])[-n_active:]
        active = active[top]
    vals = ov[active]
    w = np.exp(50.0 * (vals - vals.min()))
    w /= w.sum()
    return active, w

def _dual_descent(h, n_active, step0, n_iters):
    n = len(h)
    f = _eval(h)
    step = step0
    for _ in range(n_iters):
        active, w = _dual_weights(h, n_active)
        g = np.zeros(n)
        for k, wk in zip(active, w):
            g += wk * _grad_k(h, k, n)
        best_decrease = 0.0
        for scale in [1.0, 0.5, 0.25, 0.125, 0.06]:
            h_new = _repair(h - step * scale * g, n)
            f_new = _eval(h_new)
            if f_new < f - 1e-16:
                h = h_new
                f = f_new
                break
        else:
            step *= 0.5
            if step < 1e-9:
                break
        step = min(step * 1.08, step0)
    return h, f

def _upsample(h, factor):
    n = len(h)
    hi = np.zeros(n * factor)
    for i in range(n):
        hi[i * factor:(i + 1) * factor] = h[i]
    return hi

def _pairwise_local(h, n_iter, scale, rng):
    nc = len(h)
    f = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        fn = _eval(h)
        if fn < f - 1e-15:
            f = fn
        else:
            h[i], h[j] = ni - d, nj + d
    return h, f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP alpha sweep at base resolution
    for alpha in [500, 2000, 15000, 80000]:
        if rem() < 150:
            break
        h = _slsqp(best_h, alpha, 1200, 1e-18)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n0

    # Phase 2: Dual-weight descent at base resolution
    if rem() > 130:
        h, f = _dual_descent(best_h.copy(), 5, 0.008, 300)
        if f < best_c5:
            best_c5, best_h, best_n = f, h.copy(), n0

    # Phase 3: Upsample to 2x and optimize
    if rem() > 400:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [800, 5000, 50000, 200000]:
            if rem() < 300:
                break
            h_hi = _slsqp(h_hi, alpha, 900, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if rem() > 250:
            h_hi, f = _dual_descent(best_h.copy(), 7, 0.004, 250)
            if f < best_c5:
                best_c5, best_h, best_n = f, h_hi.copy(), n1

    # Phase 4: Upsample to 3x if time allows
    if rem() > 300 and best_n < n0 * 3:
        n2 = n0 * 3
        h_3x = _repair(_upsample(best_h, int(n2 / best_n)), n2)
        for alpha in [5000, 50000]:
            if rem() < 200:
                break
            h_3x = _slsqp(h_3x, alpha, 700, 1e-18)
            c5 = _eval(h_3x)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_3x.copy(), n2
        if rem() > 180:
            h_3x, f = _dual_descent(best_h.copy(), 7, 0.003, 200)
            if f < best_c5:
                best_c5, best_h, best_n = f, h_3x.copy(), n2

    # Phase 5: Multi-start randomization
    nc = best_n
    for r in range(12):
        if rem() < 220:
            break
        s = r % 5
        if s == 0:
            hp = best_h + 0.012 * rng.randn(nc)
        elif s == 1:
            hp = 1.0 - best_h + 0.008 * rng.randn(nc)
        elif s == 2:
            hp = best_h.copy()
            for _ in range(3):
                bs = rng.randint(4, max(5, nc // 7))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.035 * rng.randn(bs)
        elif s == 3:
            hp = best_h.copy()
            t = np.arange(nc)
            for fq in rng.randint(1, nc // 2 + 1, size=4):
                hp += 0.004 * rng.randn() * np.sin(2 * np.pi * fq * t / nc)
        else:
            hp = best_h.copy()
            hp[rng.randint(nc // 2):] += 0.02 * rng.randn(int(nc / 2))
        hp = _repair(hp, nc)
        h_new = _slsqp(hp, 80000, 700, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), nc

    # Phase 6: Dual descent polish
    if rem() > 100:
        h, f = _dual_descent(best_h.copy(), 10, 0.003, 300)
        if f < best_c5:
            best_c5, best_h, best_n = f, h.copy(), len(h)

    # Phase 7: Fine pairwise search
    if rem() > 50:
        h = best_h.copy()
        nc = len(h)
        for scale in [0.004, 0.001, 0.0003, 0.0001]:
            if rem() < 30:
                break
            h, f = _pairwise_local(h, 4000, scale, rng)
            if f < best_c5:
                best_c5, best_h, best_n = f, h.copy(), nc

    # Phase 8: Final SLSQP
    if rem() > 12:
        h = _slsqp(best_h, 500000, 1500, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), len(h)

    # Phase 9: Ultra-fine touch
    if rem() > 5:
        h = best_h.copy()
        h, f = _pairwise_local(h, 10000, 4e-5, rng)
        if f < best_c5:
            best_c5, best_h, best_n = f, h.copy(), len(h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
