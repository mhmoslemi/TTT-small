```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_single_grad(h, n, l):
    """Gradient of correlation at lag l (where l = idx - (n-1))."""
    grad = np.zeros(n)
    if l >= 0:
        grad[l:] += (1.0 - h[:n-l])
        grad[:n-l] -= h[l:]
    else:
        m = -l
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def weighted_corr_grad(h, n, c, top_k=10, temperature=0.01):
    """Compute weighted gradient of top correlations."""
    # Use softmax-like weighting
    c_norm = c - c.max()
    w = np.exp(c_norm / temperature)
    # Focus on top correlations
    top_idx = np.argsort(c)[-top_k:]
    top_w = w[top_idx]
    top_w /= top_w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(top_w, top_idx):
        if wi > 1e-10:
            grad += wi * corr_single_grad(h, n, idx - (n-1))
    return grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
```

```python
            break
        bs = rng.randint(2, 6)
        i = rng.randint(0, max(1, n - bs + 1))
        j = rng.randint(0, max(1, n - bs + 1))
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        for di in range(bs):
            if not (-1e-9 <= h[i+di] + eps <= 1.0 + 1e-9) or not (-1e-9 <= h[j+di] - eps <= 1.0 + 1e-9):
                ok = False
                break
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5n = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def perturb_h(h, n, rng, kind=0):
    hp = h.copy()
    if kind == 0:
        hp = np.roll(hp, rng.randint(1, n // 3))
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        hp += rng.uniform(0.04, 0.12) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        hp += rng.normal(0, 0.05, n)
    elif kind == 3:
        hp *= rng.uniform(0.82, 1.18)
    else:
        hp = np.concatenate([hp[n//5:], hp[:n//5]])
    return repair(hp, n)

def upscale_fn(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return repair(np.interp(x_new, x_old, np.concatenate([h, [h[0]]])), n_new)

def remaining_time(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    n0 = len(h_par)
    dx0 = 2.0 / n0
    h0 = repair(h_par.copy(), n0)
    c5_0 = compute_c5(h0, dx0)
    best_h, best_c5, best_n = h0.copy(), c5_0, n0

    # Phase 1: Subgradient on n0 (90s)
    t_end = t_start + 90
    h0, c5_0 = subgrad_opt(h0, n0, t_end, lr_init=0.06, temp_init=0.006)
    if c5_0 < best_c5:
        best_h, best_c5 = h0.copy(), c5_0

    # Phase 2: L-BFGS multi-param (80s)
    t_end = t_start + 170
    target = n0 / 2.0
    for alpha in [40, 150, 600, 3000]:
        if time.time() >= t_end or remaining_time(t_start, budget_s) < 280:
            break
        for pen in [1000, 10000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_opt(best_h, n0, alpha, target, pen, 250)
            if cc < best_c5:
                best_h, best_c5 = hh, cc

    # Phase 3: Fourier parameterization search (40s)
    t_end = t_start + 210
    if remaining_time(t_start, budget_s) > 200:
        best_h, best_c5 = nelder_mead_1d(best_h, n0, dx0, t_end)

    # Phase 4: Local search (120s)
    t_end = t_start + 330
    if remaining_time(t_start, budget_s) > 180:
        best_h, best_c5 = local_search(best_h, n0, dx0, rng, t_end, n_scales=5)

    # Phase 5: Multi-restart with subgrad+lbfgs (80s)
    t_end = t_start + 410
    for r in range(6):
        if time.time() >= t_end or remaining_time(t_start, budget_s) < 180:
            break
        hp = perturb_h(best_h, n0, rng, r % 5)
        hh, cc = subgrad_opt(hp, n0, time.time() + 6, lr_init=0.05, temp_init=0.01)
        hh, cc = lbfgs_opt(hh, n0, 1500, target, 8000, 150)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 6: Upscale to 2*n0
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h1 = upscale_fn(best_h, n0, n1)
    c5_1 = compute_c5(h1, dx1)

    # Phase 7: Subgradient on upscaled (70s)
    t_end = t_start + 530
    if remaining_time(t_start, budget_s) > 180:
        h1, c5_1 = subgrad_opt(h1, n1, t_end, lr_init=0.04, temp_init=0.005)

    # Phase 8: L-BFGS on upscaled (50s)
    t_end = t_start + 580
    target1 = n1 / 2.0
    if remaining_time(t_start, budget_s) > 130:
        for alpha in [80, 800, 4000]:
            if time.time() >= t_end:
                break
            hh, cc = lbfgs_opt(h1, n1, alpha, target1, 5000, 200)
            if cc < c5_1:
                h1, c5_1 = hh, cc

    # Phase 9: Local search on upscaled (80s)
    t_end = t_start + 660
    if remaining_time(t_start, budget_s) > 100:
        h1, c5_1 = local_search(h1, n1, dx1, rng, t_end, n_scales=4)

    # Phase 10: Block local search on better solution
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c5_1, n1

    # Phase 11: Final subgradient + local search
    if remaining_time(t_start, budget_s) > 50:
        t_end = time.time() + min(50, remaining_time(t_start, budget_s) - 20)
        dx_b = 2.0 / best_n
        best_h, best_c5 = subgrad_opt(best_h, best_n, t_end, lr_init=0.03, temp_init=0.003)

    if remaining_time(t_start, budget_s) > 25:
        t_end = time.time() + min(20, remaining_time(t_start, budget_s) - 5)
        dx_b = 2.0 / best_n
        best_h, best_c5 = local_search(best_h, best_n, dx_b, rng, t_end, n_scales=3)

    if remaining_time(t_start, budget_s) > 10:
        t_end = time.time() + min(8, remaining_time(t_start, budget_s) - 3)
        dx_b = 2.0 / best_n
        best_h, best_c5 = block_local(best_h, best_n, dx_b, rng, t_end, n_iters=3000)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
```
