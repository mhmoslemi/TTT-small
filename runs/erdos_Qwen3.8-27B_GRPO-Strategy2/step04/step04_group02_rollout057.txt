                h[idx] += add
                diff -= add

                h[idx] -= sub
                diff += sub

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def anneal(h, n, dx, rng, t_start, t_end):
    target = n / 2.0
    h = h.copy()
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    T0 = 3.0e-5
    T_min = 3e-8
    total = max(1.0, t_end - t_start)
    beta = np.log(T0 / T_min) / total
    stale = 0

    while time.time() < t_end:
        elapsed = time.time() - t_start
        T = max(T_min, T0 * np.exp(-beta * elapsed))
        r = rng.random()

        if r < 0.50:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            step = 0.0012 + T * 8.0
            d = rng.uniform(-step, step)
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            nc = compute_c5(h, dx)
            delta = nc - current_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-min(80.0, delta / T))):
                current_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                    stale = 0
                else:
                    stale += 1
            else:
                h[i] -= d
                h[j] += d
                stale += 1
        elif r < 0.78:
            bs = rng.randint(2, 6)
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            step = 0.002 + T * 5.0
            eps = rng.uniform(-step, step)
            seg_a = h[i:i + bs]
            seg_b = h[j:j + bs]
            if eps > 0:
                if seg_a.max() + eps > 1.0 or seg_b.min() - eps < 0.0:
                    continue
            else:
                if seg_a.min() + eps < 0.0 or seg_b.max() - eps > 1.0:
                    continue
            h[i:i + bs] += eps
            h[j:j + bs] -= eps
            nc = compute_c5(h, dx)
            delta = nc - current_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-min(80.0, delta / T))):
                current_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                    stale = 0
                else:
                    stale += 1
            else:
                h[i:i + bs] -= eps
                h[j:j + bs] += eps
                stale += 1
        else:

            kmax = int(np.argmax(corr))
            shift = kmax - (n - 1)
            if shift >= 0:
                vn = n - shift
                if vn <= 1:
                    continue
                prod = h[:vn] * (1.0 - h[shift:shift + vn])
                base = 0
            else:
                s2 = -shift
                vn = n - s2
                if vn <= 1:
                    continue
                prod = h[s2:s2 + vn] * (1.0 - h[:vn])
                base = s2
            tn = min(8, len(prod))
            top = np.argsort(prod)[-tn:]
            pos = int(top[rng.randint(tn)])
            ii = base + pos
            jj = rng.randint(n)
            if jj == ii:
                jj = (jj + 1) % n
            d = rng.uniform(0.003, 0.012)
            if h[ii] - d < 0 or h[jj] + d > 1:
                continue
            h[ii] -= d
            h[jj] += d
            nc = compute_c5(h, dx)
            delta = nc - current_c5
            Ti = max(T, 8e-5)
            if delta < 0 or rng.random() < np.exp(-min(80.0, delta / Ti)):
                current_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                    stale = 0
                else:
                    stale += 1
            else:
                h[ii] += d
                h[jj] -= d
                stale += 1

        if stale > 120000:
            T = max(T, 1.2e-4)
            stale = 0
            if rng.random() < 0.4:
                h = best_h + rng.normal(0, 0.012, n)
                h = repair(h, n)
                current_c5 = compute_c5(h, dx)

    return best_h, best_c5

def fine_search(h, n, dx, rng, deadline):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.0012, 0.0005, 0.0002, 0.00008]
    for si, scale in enumerate(scales):
        if time.time() >= deadline:
            break
        improved = False
        for _ in range(120000):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            c5n = compute_c5(h, dx)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                improved = True
            else:
                h[i] -= d
                h[j] += d
        if not improved and si >= 2:
            break
    return h, best_c5
