import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _soft_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(s).sum()) / alpha

def _obj_sm(h, alpha, target, pen):
    return _soft_max(_corr(h), alpha) + pen * (h.sum() - target) ** 2

def _obj_pen(h, T, dx, target, a):
    c = _corr(h) * dx
    v = np.clip(c - T, 0, None)
    return a * np.sum(v * v) + 0.01 * (h.sum() - target) ** 2

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _slsqp(h0, dx, alpha, maxiter):
    n = len(h0)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        r = minimize(lambda x: _obj_sm(x, alpha, tgt, 0), h0,
                     method='SLSQP', bounds=bds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except:
        return h0, _c5(h0, dx)
