    c = _corr_full(h)
    grad = np.zeros(n)

            for i in range(s, n):
                grad[i] -= h[i-s]
        else:
            s2 = -s
            for i in range(n - s2):
                grad[i+s2] += (1 - h[i]) ...

    h -= step_size * grad / (len(active_shifts) * max(1.0, np.linalg.norm(grad)))

                if i == j:
                    continue
            elif sh > 0:
                i = rng.randint(n - sh)
                j = i + sh
            else:
                j = rng.randint(n + sh)
                i = j - sh
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        h[i] += d
        h[j] -= d
        if -1e-9 < h[i] < 1.0 + 1e-9 and -1e-9 < h[j] < 1.0 + 1e-9:
            cn = _c5(h, dx)
            if cn < best:
                best = cn
                if rng.random() < 0.3:
                    d2 = d * rng.uniform(1.5, 2.5)
                    h[i] += d2
                    h[j] -= d2
                    if -1e-9 < h[i] < 1.0 + 1e-9 and -1e-9 < h[j] < 1.0 + 1e-9:
                        cn2 = _c5(h, dx)
                        if cn2 < best:
                            best = cn2
                        else:
                            h[i] -= d2
                            h[j] += d2
                    else:
                        h[i] -= d2
                        h[j] += d2
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h, best_c5, best_n = h0, _c5(h0, dx0), n0

    # Phase 1: L-BFGS-B polish (60s)
    t_end = t0 + 60
    h0, c = _lbfgs(h0, dx0, 3000, 400, [500, 5000, 50000])
    if c < best_c5:
        best_h, best_c5 = h0, c

    # Phase 2: Multi-restart perturbation + polish (120s)
    t_end = t0 + 180
    for r in range(10):
        if time.time() > t_end:
            break
        amp = 0.04 * (1.0 - r * 0.06)
        hp = best_h + amp * (2.0 * rng.random(n0) - 1.0)
        hp = _repair(hp, n0 / 2.0)
        hp, c = _lbfgs(hp, dx0, 2000, 200, [2000, 20000])
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 3: Coarse-to-fine local search at n0 (~500s)
    t_start = time.time()
    t_end = min(t0 + 680, t0 + budget_s - 350)
    best_h, best_c5 = _local(best_h, dx0, rng, t_start, t_end, 0.70, 0.005, 0.0004)

    # Phase 4: Try doubled resolution n=280 (~180s)
    if budget_s - (time.time() - t0) > 220:
        n1 = 2 * n0
        dx1 = 2.0 / n1
        h1 = _repair(_interp(best_h, n1), n1 / 2.0)
        c1 = _c5(h1, dx1)
        t_end1 = min(t0 + 870, t0 + budget_s - 120)
        h1, c1 = _lbfgs(h1, dx1, 5000, 300, [2000, 20000])
        h1, c1 = _local(h1, dx1, rng, time.time(), t_end1, 0.65, 0.004, 0.0002)
        if c1 < best_c5:
            best_h, best_c5, best_n = h1, c1, n1

    # Phase 5: Final fine local search (remaining ~200s)
    dxb = 2.0 / best_n
    t_start = time.time()
    t_end = t0 + budget_s - 5
    best_h, best_c5 = _local(best_h, dxb, rng, t_start, t_end, 0.75, 0.0015, 0.00003)

    # Final repair and evaluation
    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
