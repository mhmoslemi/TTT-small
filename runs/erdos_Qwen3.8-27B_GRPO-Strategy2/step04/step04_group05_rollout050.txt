import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad1(h, ki, n):
    g = np.zeros(n)
    L = ki - (n - 1)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        m = -L
        g[:n-m] += 1.0 - h[m:]
        g[m:] -= h[:n-m]
    return g

def hess_diags(L, n):
    if L == 0:
        return np.full(n, -2.0), 0
    L = abs(L)
    d = np.full(n - L, -1.0)
    return d, L

_GR = {'n': 0, 'a': 200.0, 'l': 100.0}

    xo = np.linspace(0, 2, n0, endpoint=False)
    xd = np.linspace(0, 2, n_de, endpoint=False)
    h_de = repair(np.interp(xd, xo, h0, period=2.0), n_de)
    r, c = de_global(h_de, n_de, t0, budget_s - 550, seed)
    # Interpolate back
    h_up = repair(np.interp(xo, xd, r, period=2.0), n0)
    c_up = c5v(h_up, 2.0 / n0)
    h_start = h_start if c_up < c5v(h0, 2.0 / n0) else h0
    h_start = h_up if c_up < c5v(h0, 2.0 / n0) else h0

    # Phase 2: Full local optimization on n0
    r, c = full_opt(h_start, n0, seed, budget_s - 250)
    if c < bc:
        bh, bc, bn = r, c, n0

    # Phase 3: Multi-start on n0 (structured)
    if time.time() - t0 < budget_s - 180:
        for shift in [n0 // 3, n0 // 5, 7]:
            if time.time() - t0 > budget_s - 160:
                break
            hs = repair(np.roll(bh, shift), n0)
            r2, c2 = lbfgs_phase(hs, n0, t0, budget_s - 150)
            if c2 < bc:
                r2, c2 = active_newton(r2, n0, t0, budget_s - 120)
                if c2 < bc:
                    bh, bc = r2, c2

    # Phase 4: Higher resolution
    n2 = min(n0 * 2, 560)
    if n2 < 1000 and time.time() - t0 < budget_s - 130:
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = repair(np.interp(xn, xo, bh, period=2.0), n2)
        r, c = full_opt(h2, n2, seed + 1, budget_s - 25)
        if c < bc:
            bh, bc, bn = r, c, n2

    # Phase 5: Final polish
    if time.time() - t0 < budget_s - 5:
        rng2 = np.random.RandomState(seed + 777)
        r, c = coord_desc(bh, bn, rng2, t0, budget_s - 2)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5v(bh, 2.0 / bn)
    return (bh, bc, bn)
