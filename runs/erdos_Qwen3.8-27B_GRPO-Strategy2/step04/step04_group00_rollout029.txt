    coef = np.zeros(2*n-1)
    for k in range(2*n-1):

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _coef_matrix(h, n):
    C = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        L = k - (n - 1)
        if L >= 0:
            idx1 = np.arange(0, n - L)
            C[idx1, k] += 1.0 - h[idx1 + L]
            idx2 = np.arange(L, n)
            C[idx2, k] -= h[idx2 - L]
        else:
            M = -L
            idx1 = np.arange(M, n)
            C[idx1, k] += 1.0 - h[idx1 - M]
            idx2 = np.arange(0, n - M)
            C[idx2, k] -= h[idx2 + M]
    return C

def _pair_line_search(h, n, i, j, c, C):
    delta = C[i, :] - C[j, :]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi - 1e-15:
        return 0.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c + delta * mid
        mx = np.max(vals)
        active = vals >= mx - 1e-13
        g_lo = np.min(delta[active])
        g_hi = np.max(delta[active])
        if g_lo <= 0.0 <= g_hi:
            break
        elif g_lo > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def _subgrad_descent(h, n, dx, max_iter=8000, tol=1e-12):
    h = h.copy()
    best_c5 = _c5(h, dx)
    c = _corr(h)
    C = _coef_matrix(h, n)
    for it in range(max_iter):
        cmax = np.max(c)
        cmin = np.min(c)
        spread = cmax - cmin
        if spread < 1e-12:
            spread = 1.0
        thresh = cmax - 0.05 * spread
        active = c >= thresh
        if not np.any(active):
            active = c >= cmax - 1e-12
        g = C[active].mean(axis=0)
        gc = g - g.mean()
        i_b = np.argmin(gc)
        j_b = np.argmax(gc)
        gap = gc[j_b] - gc[i_b]
        if gap < tol:
            # Try a few more pairs
            order_i = np.argsort(gc)[:5]
            order_j = np.argsort(gc)[::-1][:5]
            found_move = False
            for ii in order_i:
                for jj in order_j:
                    if ii == jj:
                        continue
                    d = _pair_line_search(h, n, ii, jj, c, C)
                    if d > 1e-12:

def _softobj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _smooth_opt(h, n, dx, bounds, time_fn, budget):
    bh = h.copy()
    bc5 = _c5(bh, dx)
    for a in [200, 1000, 5000, 20000, 100000, 500000]:

        try:
            res = minimize(
                lambda x, al=a: _softobj(x, al, 80.0, n), bh,

                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-14})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass
    return bh, bc5

    c = _corr(h)
    C = _coef_matrix(h, n)

            else:
                h[i] -= d
                h[j] += d
                h = _repair(h, n)
            c = _corr(h)

    return h, best

                h1s, c5_1s = h1r, c5_1s
        if c5_1s < best_c5:
            best_h, best_c5, n0, dx0 = h1s, c5_1s, n1, dx1

    # Phase 6: Try n=200
    n2 = 200
    dx2 = 2.0 / n2
    if budget_s - elapsed() > 100:
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n0, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        bounds2 = [(0.0, 1.0)] * n2
        if budget_s - elapsed() > 40:
            h2s, c5_2s = _smooth_opt(h2, n2, dx2, bounds2, elapsed, budget_s)
        else:
            h2s, c5_2s = h2, _c5(h2, dx2)
        if budget_s - elapsed() > 30:
            h2g, c5_2g = _subgrad_descent(h2s, n2, dx2, max_iter=6000)
            if c5_2g < c5_2s:
                h2s, c5_2s = h2g, c5_2g
        if budget_s - elapsed() > 10:
            h2r, c5_2r = _local_refine(h2s, n2, dx2, rng, elapsed, budget_s)
            if c5_2r < c5_2s:
                h2s, c5_2s = h2r, c5_2r
        if c5_2s < best_c5:
            best_h, best_c5, n0, dx0 = h2s, c5_2s, n2, dx2

    # Phase 7: Final subgradient + local on best
    if budget_s - elapsed() > 20:
        h_f, c5_f = _subgrad_descent(best_h, n0, dx0, max_iter=6000)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    if budget_s - elapsed() > 10:
        h_f, c5_f = _local_refine(best_h, n0, dx0, rng, elapsed, budget_s)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Final tiny adjustments
    if budget_s - elapsed() > 3:
        h = best_h.copy()
        c5c = _c5(h, dx0)
        for _ in range(30000):
            if elapsed() > budget_s - 2:
                break
            i, j = rng.randint(n0), rng.randint(n0)
            if i == j:
                continue
            d = rng.uniform(-0.002, 0.002)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                nc = _c5(h, dx0)
                if nc < c5c:
                    c5c = nc
                    best_c5 = nc
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n0)
    best_c5 = _c5(best_h, dx0)
    return (best_h, best_c5, n0)
