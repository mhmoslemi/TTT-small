    bc5 = _c5(bh, dx)
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()

    def tl():
        return time_limit - (time.time() - t_start)

    # Phase 1: L-BFGS-B softmax with increasing sharpness
    for a in [200, 1000, 5000, 30000, 150000, 800000, 4000000]:
        if tl() < 25:
            break
        try:
            res = minimize(_softobj, bh.copy(),
                           jac=lambda x: _softgrad(x, a, 80.0),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-16})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass

    # Phase 2: Adaptive active-set subgradient with backtracking
    h = bh.copy()
    c5c = _c5(h, dx)
    step = 0.012
    for _it in range(600):
        if tl() < 20:
            break
        c = _corr(h)
        cm = np.max(c)
        spread = max(cm - np.min(c), 1e-10)
        thresh = cm - 0.03 * spread
        active = np.where(c >= thresh)[0]
        g = np.zeros(n)
        for k in active:
            g += _grad_c(h, k - (n - 1))
        g /= len(active)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-12:
            break
        g /= ng
        found = False
        for s_try in [step, step * 0.6, step * 0.3, step * 0.1, step * 0.04]:
            h_new = _repair(h - s_try * g, n)
            c_new = _c5(h_new, dx)
            if c_new < c5c - 1e-16:
                h = h_new
                c5c = c_new
                found = True
                if c5c < bc5:
                    bc5 = c5c
                    bh = h.copy()
                break
        if not found:
            step *= 0.65
            if step < 1e-8:
                break
        else:
            step = min(step * 1.08, 0.035)

    # Phase 3: Gradient-guided coordinate descent with block moves
    h = bh.copy()
    c5c = _c5(h, dx)
    dlist = np.array([0.001, 0.004, 0.01, 0.02, 0.04,
                     -0.001, -0.004, -0.01, -0.02, -0.04])
    for _sweep in range(250):
        if tl() < 15:
            break
        c = _corr(h)
        cm = np.max(c)
        spread = max(cm - np.min(c), 1e-10)
        active = np.where(c >= cm - 0.06 * spread)[0]
        gmag = np.zeros(n)
        for k in active[:12]:
            gg = _grad_c(h, k - (n - 1))
            gmag += gg * gg
        gmag /= max(len(active[:12]), 1)
        order = np.argsort(gmag)[::-1][:40]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 8, len(order))):
                j = order[jj]
                best_d, best_c = 0.0, c5c
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        cval = _c5(h, dx)
                        if cval < best_c:
                            best_c = cval
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5c = _c5(h, dx)
                    improved = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h.copy()
                    break
            if improved:
                break
        # Block moves
        if not improved:
            for _bm in range(20):
                i0 = rng.randint(0, max(n - 3, 1))
                ln = rng.randint(2, min(5, n - 1))
                sh = rng.choice([-1, 1, -2, 2])
                j0 = i0 + sh
                if j0 < 0 or j0 + ln >= n:
                    continue
                h[:i0 + ln] = np.concatenate([h[:i0], h[i0 + ln:i0 + ln + ln]]) if i0 + sh > i0 else h
                h_new = h.copy()
                h_new[i0:i0 + ln] = h_new[j0:j0 + ln]
                h_new[j0:j0 + ln] = h_new[i0:i0 + ln] if False else h_new[i0:i0 + ln]
                # Simpler: just swap values
                h_new = h.copy()
                h_new[i0:i0 + ln], h_new[j0:j0 + ln] = h[j0:j0 + ln].copy(), h[i0:i0 + ln].copy()
                h_new = _repair(h_new, n)
                cval = _c5(h_new, dx)
                if cval < c5c:
                    h = h_new
                    c5c = cval
                    improved = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h.copy()
                    break
        if not improved:
            # Random pairs
            for _rp in range(40):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = dlist[rng.randint(len(dlist))]
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c5c:
                        c5c = c
                        improved = True
                        if c5c < bc5:
                            bc5 = c5c
                            bh = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
            if not improved:
                break

    # Phase 4: Nelder-Mead on top coordinates
    if tl() > 10:
        c = _corr(bh)
        cm = np.max(c)
        spread = max(cm - np.min(c), 1e-10)
        gmag = np.zeros(n)
        active = np.where(c >= cm - 0.08 * spread)[0]
        for k in active[:10]:
            gg = _grad_c(bh, k - (n - 1))
            gmag += gg * gg
        top_ids = np.argsort(gmag)[::-1][:20]
        top_vals = bh[top_ids].copy()
        h_base = bh.copy()

        def nm_obj(params, ids=top_ids, base=h_base):
            trial = base.copy()
            trial[ids] = np.clip(params, 0, 1)
            trial = _repair(trial, n)
            return _c5(trial, dx)

        try:
            res_nm = minimize(nm_obj, top_vals, method='Nelder-Mead',
                             options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-12})
            h_nm = h_base.copy()
            h_nm[top_ids] = np.clip(res_nm.x, 0, 1)
            h_nm = _repair(h_nm, n)
            c5_nm = _c5(h_nm, dx)
            if c5_nm < bc5:
                bc5 = c5_nm
                bh = h_nm
        except Exception:
            pass

    # Phase 5: Fine random polish
    if tl() > 8:
        h = bh.copy()
        c5c = _c5(h, dx)
        tiny = np.array([0.0002, 0.0005, 0.001, 0.002, -0.0002, -0.0005, -0.001, -0.002])
        n_iter = min(40000, int(tl() * 5000))
        for _ in range(n_iter):
            if tl() < 8:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                    bc5 = c
                    bh = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    # Primary: n=140 from initial
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h, best_c5 = optimize_h(h0, n, dx, budget_s - 120, rng)

    # Try 1-h symmetry (exploits h -> 1-h invariance to find new basin)
    if elapsed() < budget_s - 80:
        h_inv = _repair(1.0 - best_h, n)
        h_inv_pert = _repair(h_inv + 0.005 * rng.randn(n), n)
        hi, ci = optimize_h(h_inv_pert, n, dx, budget_s - elapsed() - 40, rng)
        if ci < best_c5:
            best_h, best_c5 = hi, ci

    # Try n=100
    if elapsed() < budget_s - 130:
        n2 = 100
        dx2 = 2.0 / n2
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        h2r, c5_2 = optimize_h(h2, n2, dx2, budget_s - elapsed() - 50, rng)
        if c5_2 < best_c5:
            best_h, best_c5, n, dx = h2r, c5_2, n2, dx2

    # Try n=200 for finer resolution
    if elapsed() < budget_s - 90:
        n3 = 200
        dx3 = 2.0 / n3
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = _repair(h3, n3)
        h3r, c5_3 = optimize_h(h3, n3, dx3, budget_s - elapsed() - 30, rng)
        if c5_3 < best_c5:
            best_h, best_c5, n, dx = h3r, c5_3, n3, dx3

    # Final ultra-fine polish
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5c = _c5(h, dx)
        tiny = np.array([0.0001, 0.0003, 0.0008, -0.0001, -0.0003, -0.0008])
        n_iter = min(80000, int((budget_s - elapsed() - 2) * 10000))
        for _ in range(n_iter):
            if elapsed() > budget_s - 2:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
