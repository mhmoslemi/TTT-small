                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 7: Final SLSQP
    if not _dl(15):
        h2 = _slsqp(best_h, 200000, 2500, best_n)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 8: Ultra-fine tweaks
    if not _dl(5):
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(10000):
            if _dl(5):
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 4e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
