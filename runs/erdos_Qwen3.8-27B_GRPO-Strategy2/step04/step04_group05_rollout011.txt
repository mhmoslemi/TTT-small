                    bc, bh = cc, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def de_quick(h0, n, t0, lim):
    _ST['n'] = n
    bounds = [(0.0, 1.0)] * n
    bh, bc = h0.copy(), c5val(h0)
    def _de_f(x):
        return c5val(repair(x, n))
    try:
        if time.time() - t0 < lim:
            r = differential_evolution(_de_f, bounds, seed=_ST['seed'],
                                       maxiter=30, popsize=20,
                                       mutation=(0.4, 0.9), recombination=0.8,
                                       polish=False, updating='deferred',
                                       tol=1e-8)
            if r is not None and r.x is not None:
                hx = repair(r.x, n)
                c = c5val(hx)
                if c < bc:
                    bh, bc = hx, c
    except Exception:
        pass
    return bh, bc

def refine(h, n, seed, t0, lim):
    rng = np.random.RandomState(seed)
    bh, bc = h.copy(), c5val(h)
    # Power-sum L-BFGS-B with increasing p
    for p, lam in [(8, 80.0), (16, 400.0), (32, 1500.0), (64, 5000.0)]:
        if time.time() - t0 > lim - 80:
            break
        r, c = lbfgs_psm(bh, n, p, lam, t0, lim - 80)
        if c < bc:
            bh, bc = r, c
    # Active set descent
    if time.time() - t0 < lim - 50:
        r, c = act_desc(bh, n, t0, lim - 50, topk=3, mxiter=500)
        if c < bc:
            bh, bc = r, c
    # More L-BFGS-B after active set
    if time.time() - t0 < lim - 30:
        for p, lam in [(32, 1500.0), (128, 20000.0)]:
            if time.time() - t0 > lim - 30:
                break
            r, c = lbfgs_psm(bh, n, p, lam, t0, lim - 30)
            if c < bc:
                bh, bc = r, c
    # 2D coordinate descent
    if time.time() - t0 < lim - 15:
        r, c = coord2d(bh, n, rng, t0, lim - 15, rounds=300)
        if c < bc:
            bh, bc = r, c
    # Active set again
    if time.time() - t0 < lim - 10:
        r, c = act_desc(bh, n, t0, lim - 10, topk=2, mxiter=300)
        if c < bc:
            bh, bc = r, c
    # Coord2d again
    if time.time() - t0 < lim - 5:
        r, c = coord2d(bh, n, rng, t0, lim - 5, rounds=200)
        if c < bc:
            bh, bc = r, c
    # Polish
    if time.time() - t0 < lim - 2:
        r, c = polish_tiny(bh, n, rng, t0, lim - 1, 40000)
        if c < bc:
            bh, bc = r, c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    _ST['seed'] = seed
    nl = len(initial_h_values)
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), nl)
    bh, bc, bn = h0.copy(), c5val(h0), nl

    # Phase A: DE for global diversity
    if time.time() - t0 < budget_s - 200:
        r, c = de_quick(h0, nl, t0, budget_s - 200)
        if c < bc:
            bh, bc, bn = r, c, nl

    # Phase B: Full refinement at N=140
    if time.time() - t0 < budget_s - 120:
        r, c = refine(bh, nl, seed, time.time(), budget_s - 120)
        if c < bc:
            bh, bc, bn = r, c, nl

    # Phase C: N=70 downsample
    n70 = 70
    if time.time() - t0 < budget_s - 180:
        xo = np.linspace(0.0, 2.0, bn, endpoint=False)
        xn = np.linspace(0.0, 2.0, n70, endpoint=False)
        h70 = repair(np.interp(xn, xo, bh, period=2.0), n70)
        r, c = refine(h70, n70, seed + 1, time.time(), budget_s - 180)
        if c < bc:
            bh, bc, bn = r, c, n70

    # Phase D: N=280 upsample
    n280 = 280
    if n280 < 1000 and time.time() - t0 < budget_s - 80:
        xo = np.linspace(0.0, 2.0, bn, endpoint=False)
        xn = np.linspace(0.0, 2.0, n280, endpoint=False)
        h280 = repair(np.interp(xn, xo, bh, period=2.0), n280)
        r, c = refine(h280, n280, seed + 2, time.time(), budget_s - 80)
        if c < bc:
            bh, bc, bn = r, c, n280

    # Phase E: Perturbation restart
    if time.time() - t0 < budget_s - 40:
        rng2 = np.random.RandomState(seed + 777)
        hp = repair(bh + 0.012 * rng2.randn(bn), bn)
        r, c = refine(hp, bn, seed + 444, time.time(), budget_s - 40)
        if c < bc:
            bh, bc, bn = r, c, bn

    # Phase F: Final coord2d + polish
    if time.time() - t0 < budget_s - 5:
        rng3 = np.random.RandomState(seed + 1234)
        r, c = coord2d(bh, bn, rng3, t0, budget_s - 5, rounds=200)
        if c < bc:
            bh, bc = r, c
    if time.time() - t0 < budget_s - 2:
        rng4 = np.random.RandomState(seed + 5678)
        r, c = polish_tiny(bh, bn, rng4, t0, budget_s - 1, 50000)
        if c < bc:
            bh, bc = r, c

    bh = repair(bh, bn)
    bc = c5val(bh)
    return (bh, bc, bn)
