def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

                diff -= add

                diff += sub

        best_h, best_c5 = block_local(best_h, n, dx, rng, bl_deadline, n_iters)

    # Phase 7: SA round 2 - cooler, finer (100s)
    if tleft() > 100:
        sa_deadline = time.time() + min(100, tleft() - 50)
        h2, c2 = sa_search(best_h, n, dx, rng, sa_deadline, 0.004, 0.00003)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 8: Multi-restart with local opt (80s)
    if tleft() > 80:
        t_end = time.time() + min(80, tleft() - 40)
        for r in range(8):
            if time.time() >= t_end:
                break
            hp = perturb_restart(best_h, n, rng, r % 5)
            hh, cc = lbfgs_lse(hp, n, 5000, target, 2000, 150)
            hh, cc = slsqp_exact(hh, n, target, 300)
            if cc < best_c5:
                best_h, best_c5 = hh, cc

    # Phase 9: Final fine local search (50s)
    if tleft() > 30:
        ls_deadline = time.time() + min(50, tleft() - 10)
        best_h, best_c5 = local_search(best_h, n, dx, rng, ls_deadline)

    # Phase 10: SLSQP final polish (15s)
    if tleft() > 15:
        hh, cc = slsqp_exact(best_h, n, target, 400)
        if cc < best_c5:
            best_h, best_c5 = hh, cc

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
