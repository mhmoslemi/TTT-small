def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Project onto sum constraint while maintaining [0,1]
    # Use bisection on the shift
    lo, hi = -max(h), max(1.0 - h)  # range of valid shifts
    for _ in range(60):
        mid = (lo + hi) / 2

        if h2.sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a

                break
    return h

    dx = 2.0 / n

    if tleft() > 20:
        h, c5 = _subgrad_polish(best_h, best_n, 80000, min(18, tleft() - 5))
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Final pairwise
    if tleft() > 10:
        h, c5 = _pairwise_search(best_h, best_n, rng, min(8, tleft() - 3))
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Tiny perturbation search
    if tleft() > 5:
        h = best_h.copy()
        c5c = best_c5
        for _ in range(80000):
            if time.time() > t0 + budget_s - 2:
                break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-5 * rng.choice([1, 2, 5, 10, 20])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
