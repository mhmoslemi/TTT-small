    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    # Water-filling: add 'diff' units distributed to have least impact
    if diff > 0:
        # Need to add diff, add to elements with most room
        order = np.argsort(1 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining < 1e-15:
                break
    else:
        # Need to remove -diff, remove from elements with most excess
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining < 1e-15:
                break
    return h

                    if c < c5c:
                        c5c = c
                        bh = h.copy()
                        improved = True
                    else:
                        h[i], h[j] = ni-d, nj+d
            if not improved:
                break
    return bh, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h0 = _project(np.array(initial_h_values, dtype=float), n)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # Phase 1: L-BFGS soft-max
    h1, c5_1 = _optimize_phase1(h0, n, dx, bounds, t0, budget_s)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Subgradient
    h2, c5_2 = _optimize_phase2(best_h, n, dx, t0, budget_s)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Coordinate descent
    h3, c5_3 = _optimize_phase3(best_h, n, dx, t0, budget_s, rng)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Try smaller n with interpolation
    for n_try in [100, 80, 60]:
        if time.time() - t0 > budget_s - 50:
            break
        dx_t = 2.0 / n_try
        bounds_t = [(0.0, 1.0)] * n_try
        h_t = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                        np.linspace(0, 2, n, endpoint=False), best_h)
        h_t = _project(h_t, n_try)
        # Quick optimization
        h_t1, c5_t1 = _optimize_phase1(h_t, n_try, dx_t, bounds_t, t0, budget_s)
        h_t2, c5_t2 = _optimize_phase2(h_t1, n_try, dx_t, t0, budget_s)
        h_t3, c5_t3 = _optimize_phase3(h_t2, n_try, dx_t, t0, budget_s, rng)
        c5_t = min(c5_t1, c5_t2, c5_t3)
        h_best_t = h_t1 if c5_t == c5_t1 else (h_t2 if c5_t == c5_t2 else h_t3)
        if c5_t < best_c5:
            best_h, best_c5, n, dx = h_best_t, c5_t, n_try, dx_t
            bounds = [(0.0, 1.0)] * n

    # Final fine-tuning phase
    h_f = best_h.copy()
    c5_f = _c5(h_f, dx)
    step_vals = np.array([0.0003, 0.0005, 0.001, 0.002, -0.0003, -0.0005, -0.001, -0.002])
    while time.time() - t0 < budget_s - 3:
        g = _softgrad(h_f, 800000)
        top_idx = np.argsort(np.abs(g))[::-1][:50]
        found = False
        for ii in range(len(top_idx)):
            i = top_idx[ii]
            for jj in range(ii+1, min(ii+15, len(top_idx))):
                j = top_idx[jj]
                best_d, best_c = 0.0, c5_f
                for d in step_vals:
                    ni, nj = h_f[i]+d, h_f[j]-d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h_f[i], h_f[j] = ni, nj
                        c = _c5(h_f, dx)
                        if c < best_c:
                            best_c, best_d = c, d
                        h_f[i], h_f[j] = ni-d, nj+d
                if best_d != 0:
                    h_f[i] += best_d
                    h_f[j] -= best_d
                    h_f = _project(h_f, n)
                    c5_f = _c5(h_f, dx)
                    if c5_f < best_c5:
                        best_c5, best_h = c5_f, h_f.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            # Random perturb
            for _ in range(50):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = rng.uniform(-0.0015, 0.0015)
                ni, nj = h_f[i]+d, h_f[j]-d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h_f[i], h_f[j] = ni, nj
                    c = _c5(h_f, dx)
                    if c < c5_f:
                        c5_f = c
                        if c5_f < best_c5:
                            best_c5, best_h = c5_f, h_f.copy()
                        found = True
                    else:
                        h_f[i], h_f[j] = ni-d, nj+d
            if not found:
                break

    best_h = _project(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
