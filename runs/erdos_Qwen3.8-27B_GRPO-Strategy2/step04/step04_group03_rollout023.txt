def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

    return h

def _subgrad_step(h, n, lr):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(c))
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        d = -lag
        g[d:] += (1.0 - h[:-d])
        g[:-d] -= h[d:]
    # Subtract mean to preserve sum
    g -= g.mean()
    h_new = h - lr * g
    return h_new, k_max
