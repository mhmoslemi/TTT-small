def lgse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def subgrad_step(h, n, step_size):
    """Take a projected subgradient step on the max correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    t = k_max - n + 1  # shift

    if t >= 0:
        grad[t:] += 1.0 - h[:n-t]
        grad[:n-t] -= h[t:]
    else:
        gt = -t
        grad[:n-gt] += 1.0 - h[gt:]
        grad[gt:] -= h[:n-gt]
    
    # Projected gradient step with sum constraint
    h_new = h - step_size * grad

grad = [1-a1, 1-a2-a0, -a1]

d(corr[1])/d(a0) = d/d(a0)[a0(1-a1) + a1(1-a2)] = 1-a1 ✓
d(corr[1])/d(a1) = d/d(a1)[a0(1-a1) + a1(1-a2)] = -a0 + (1-a2) = 1-a0-a2 ✓
d(corr[1])/d(a2) = d/d(a2)[a0(1-a1) + a1(1-a2)] = -a1 ✓

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lgse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def sum_eq(x, target):
    return np.sum(x) - target

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def grad_max_corr(h, k_max, n):
    t = k_max - n + 1
    grad = np.zeros(n)
    if t >= 0:
        grad[t:] += 1.0 - h[:n - t]
        grad[:n - t] -= h[t:]
    else:
        gt = -t
        grad[:n - gt] += 1.0 - h[gt:]
        grad[gt:] -= h[:n - gt]
    return grad

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lgse_obj, h, args=(alpha, target, pen),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def slsqp_opt(h, alpha, target, maxiter):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)
