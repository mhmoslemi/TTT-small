        x_new = np.linspace(0.0, 1.0, n_alt, endpoint=False)
        h_seed = np.interp(x_new, x_old, best_h)
        h_seed = repair(h_seed, n_alt)
        h_alt = slsqp_opt(h_seed, n_alt, 60000, 400)
        c5_alt = compute_c5(h_alt, n_alt)
        h_alt2, c5_alt2 = targeted_cd(h_alt, n_alt, 2000, t6, rng)
        if c5_alt2 < c5_alt:
            h_alt, c5_alt = h_alt2, c5_alt2
        if c5_alt < best_c5 - 1e-14:
            best_h, best_c5, best_n = h_alt.copy(), c5_alt, n_alt

    # Final polish
    if best_h is not None and time.time() < t0 + budget_s * 0.995:
        t_fin = t0 + budget_s * 0.995
        h_fin = slsqp_opt(best_h.copy(), best_n, 2000000, 600)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h, best_c5 = h_fin.copy(), c5_fin
        h_fin2, c5_fin2 = targeted_cd(best_h, best_n, 6000, t_fin, rng)
        if c5_fin2 < best_c5 - 1e-14:
            best_h, best_c5 = h_fin2.copy(), c5_fin2

    if best_h is None:
        best_n = 64
        best_h = repair(np.full(64, 0.5), 64)
        best_c5 = compute_c5(best_h, 64)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
