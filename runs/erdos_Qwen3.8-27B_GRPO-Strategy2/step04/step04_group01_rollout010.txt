import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    shift = k_idx - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        ns = n - shift
        g[:ns] += (1.0 - h[shift:ns + shift])
        if shift > 0:
            g[shift:] -= h[:ns]
    else:
        p = -shift
        g[p:] += (1.0 - h[:n - p])
        if p > 0:
            g[:n - p] -= h[p:]
    return g * dx

def _smooth_obj(h, alpha):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(h, k)
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, ftol, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() < t_end:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _topk_descent(h, n_top, max_iters, step0, t_end):
    n = len(h)
    v = np.zeros(n)
    f = _eval(h)
    step = step0
    mu = 0.85
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
        idx = np.argpartition(corr, -n_top)[-n_top:]
        f = corr[idx].max()
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, k)
        g /= n_top
        v = mu * v - step * g
        h_new = _proj_sum(h + v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 3)
        else:
            v *= 0.3
            step *= 0.97
            h = h_new
    return h

def _binary_swap(h, max_iters, t_end):
    n = len(h)
    b = (h >= 0.5).astype(float)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    f = _eval(b)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = ones[0]
        j = zeros[0]
        improved = False
        for i in range(len(ones)):
            for j in range(len(zeros)):
                b[ones[i]] = 0; b[zeros[j]] = 1
                f_new = _eval(b)
                if f_new < f - 1e-15:
                    f = f_new
                    ones[i], zeros[j] = zeros[j], ones[i]
                    improved = True
                    break
                else:
                    b[ones[i]] = 1; b[zeros[j]] = 0
            if improved:
                break
        if not improved:
            break
    return b

def _upsample(h, factor):
    n = len(h)
    h_hi = np.repeat(h, factor)
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: SLSQP at original resolution
    for alpha in [2000, 10000, 50000]:
        if _el() > 120:
            break
        h = _slsqp(best_h, alpha, 1500, 1e-19, t0 + budget_s - 500)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n0

    # Phase B: Upsample and optimize
    n1 = n0 * 2
    h_hi = _repair(_upsample(best_h, 2), n1)
    for alpha in [3000, 20000, 100000]:
        if _el() > budget_s - 400:
            break
        h_hi = _slsqp(h_hi, alpha, 2000, 1e-19, t0 + budget_s - 350)
        c5 = _eval(h_hi)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase C: Top-k gradient descent with momentum
    nc = best_n
    h_c = best_h.copy()
    h_c = _topk_descent(h_c, 5, 2000, 0.004, t0 + budget_s - 250)
    h_c = _repair(h_c, nc)
    c5 = _eval(h_c)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h_c.copy(), nc

    # Phase D: Binary round + swap + unround
    if _el() < budget_s - 200:
        b = _binary_swap(best_h, 500, t0 + budget_s - 150)
        c5b = _eval(b)
        if c5b < best_c5:
            h_un = b + 0.1 * rng.randn(nc)
            h_un = _repair(h_un, nc)
            h_un = _slsqp(h_un, 50000, 1000, 1e-19, t0 + budget_s - 150)
            c5u = _eval(h_un)
            if c5u < best_c5:
                best_c5, best_h, best_n = c5u, h_un.copy(), nc

    # Phase E: Multi-start with diverse initializations
    n_cur = best_n
    if _el() < budget_s - 150:

        inits.append(_repair(best_h + 0.02 * rng.randn(n_cur), n_cur))
        inits.append(_repair(1.0 - best_h, n_cur))
        t_arr = np.arange(n_cur)

        inits.append(_repair(0.5 + 0.4 * np.cos(2 * np.pi * 5 * t_arr / n_cur), n_cur))
        for hp in inits:
            if _el() > budget_s - 150:
                break
            h_new = _slsqp(hp, 30000, 800, 1e-19, t0 + budget_s - 150)
            h_new = _topk_descent(h_new, 5, 500, 0.003, t0 + budget_s - 150)
            h_new = _repair(h_new, n_cur)
            c5n = _eval(h_new)
            if c5n < best_c5:

    # Phase F: Fine pairwise search
    if _el() < budget_s - 40:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.003, 0.001, 3e-4, 1e-5]:

                break
            for _it in range(6000):
                if _el() > budget_s - 40:
                    break

                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5, best_h = c5c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase G: Final SLSQP polish
    if _el() < budget_s - 15:
        h = _slsqp(best_h, 200000, 2000, 1e-20, t0 + budget_s - 5)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase H: Ultra-fine touch
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(20000):
            if _el() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
