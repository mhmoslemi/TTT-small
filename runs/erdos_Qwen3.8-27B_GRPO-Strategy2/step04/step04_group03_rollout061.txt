import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_val(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += w[k] * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= w[k] * h[i2 + lag:j2 + lag]
    return (2.0 / n) * g

    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lbfgsb(h, n, alpha, maxiter=500, lt=10):
    """L-BFGS-B optimizing smooth_max + penalty for sum constraint."""
    lam = 500.0
    def obj(x):
        return _smooth_val(x, alpha) + lam * (x.sum() - n/2.0)**2
    def grad(x):
        return _smooth_grad(x, alpha) + 2.0*lam*(x.sum()-n/2.0)
    bnds = [(0.0, 1.0)]*n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return _repair(res.x, n)

def _fista(h, n, alpha, lr, n_iter, t_end=None):
    """FISTA with projection onto {0<=x<=1, sum=n/2}."""
    x = h.copy()
    y = h.copy()
    t = 1.0
    for it in range(n_iter):
        if t_end and time.time() > t_end:
            break
        g = _smooth_grad(y, alpha)
        x_new = _project_sum(y - lr * g, n/2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        y = x_new + (t-1.0)/t_new * (x_new - x)
        y = _project_sum(y, n/2.0)
        x = x_new
        t = t_new
    return x

    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_val(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _power_val(h, p):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    return float(np.sum(c ** p)) ** (1.0 / p)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += w[k] * (1.0 - h[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= w[k] * h[i2 + lag:j2 + lag]
    return (2.0 / n) * g

def _project_sum(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lbfgsb(h, n, alpha, maxiter=300):
    lam = 1000.0
    def obj(x):
        return _smooth_val(x, alpha) + lam * (x.sum() - n / 2.0) ** 2
    def grad(x):
        return _smooth_grad(x, alpha) + 2.0 * lam * (x.sum() - n / 2.0)
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return _repair(res.x, n)

def _fista(h, n, alpha, lr, n_iter, t_end=None):
    x = h.copy(); y = h.copy(); t = 1.0
    for it in range(n_iter):
        if t_end and time.time() > t_end:
            break
        g = _smooth_grad(y, alpha)
        x_new = _project_sum(y - lr * g, n / 2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = _project_sum(x_new + (t - 1.0) / t_new * (x_new - x), n / 2.0)
        x = x_new; t = t_new
    return x

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _local_search_active(h, n, rng, t_end, top_m=8):
    c5c = _eval(h); scale = 0.01; no_imp = 0
    for _ in range(80000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
        idx = np.argsort(corr)[-top_m:]
        k = idx[rng.randint(top_m)]
        lag = k - (n - 1)
        i_lo, i_hi = max(0, -lag), min(n - 1, n - 1 - lag)
        if i_lo > i_hi:
            continue
        i = i_lo + rng.randint(i_hi - i_lo + 1)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n; no_imp = 0; scale = min(scale * 1.03, 0.03)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 200:
                scale *= 0.75; no_imp = 0
                if scale < 1e-6:
                    break
    return h, c5c

def _fine_local(h, n, rng, t_end):
    c5c = _eval(h); scale = 3e-4

            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
        if c5c < 0.38089:
            scale = max(scale * 0.95, 1e-5)
    return h, c5c
