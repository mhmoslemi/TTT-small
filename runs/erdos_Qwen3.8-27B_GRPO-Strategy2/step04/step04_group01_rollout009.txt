                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
                break
            else:
                h[i] = ni - d
        if c5c == best_c5 and _it % 500 == 499:
            # Occasionally do pairwise with small step
            for _p in range(20):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = 0.003 * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-15:
                        c5c = c5n
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_h = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP warmup (short)
    for alpha in [5000, 50000, 200000]:
        if _el() > 60:
            break
        h = _slsqp_run(best_h, alpha, 800, 1e-16)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Active-set subgradient (longer)
    deadline2 = t0 + min(300, budget_s * 0.3)
    h, c5 = _active_subgrad(best_h, n0, 5000, t0, deadline2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Surgical 2D optimization (multiple rounds)
    deadline3 = t0 + min(500, budget_s * 0.45)
    for _rnd in range(20):
        if _el() > deadline3 - t0:
            break
        h, c5 = _surgical_2d(best_h, n0, t0 + max(0, budget_s * 0.45))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Small perturbation to escape local min
        h = best_h.copy()
        mask = rng.rand(n0) < 0.1
        h[mask] += 0.005 * rng.randn(mask.sum())
        h = _repair(h, n0)
        h, c5 = _active_subgrad(h, n0, 200, t0, t0 + max(1, budget_s * 0.45))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Pairwise descent with decreasing scale
    deadline4 = t0 + budget_s * 0.75
    for scale in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if _el() > deadline4 - t0:
            break
        h, c5 = _pairwise_descent(best_h, n0, scale, 300000, t0 + deadline4, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Coordinate descent with fine steps
    deadline5 = t0 + budget_s * 0.90
    h, c5 = _coordinate_descent(best_h, n0, t0 + deadline5, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Ultra-fine pairwise (final polish)
    deadline6 = t0 + budget_s - 5
    for scale in [5e-4, 1e-4, 5e-5, 1e-5]:
        if _el() > deadline6 - t0:
            break
        h, c5 = _pairwise_descent(best_h, n0, scale, 2000000, t0 + deadline6, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Final active-set polish
    if _el() < budget_s - 3:
        h, c5 = _active_subgrad(best_h, n0, 2000, t0, t0 + budget_s - 3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
