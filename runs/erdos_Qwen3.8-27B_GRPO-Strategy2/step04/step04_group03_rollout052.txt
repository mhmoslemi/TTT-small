        return np.clip(h, 0, 1)

        else:

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        m = -k
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def _grad_topk(h, k_top):
    n = len(h)
    g = np.zeros(n)
    for k in k_top:
        g += _grad_lag(h, k)
    return g * (2.0 / n)

def _softmax_obj(h, alpha):
    c = _get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softmax_grad(h, alpha):
    c = _get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    # grad of sum_k w_k * f_k
    g = np.zeros(len(h))
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        lag = k - (len(h) - 1)
        g += w[k] * _grad_lag(h, lag)
    return g * (2.0 / len(h))

    h = h.copy()

    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(iters):

            break
        c = _get_corr(h)
        idx = np.argsort(c)[-topk:]
        g = _grad_topk(h, idx)

        h_trial = _repair(h + v_new, n)
        # Simple step

            v *= 0.5

def _directed_search(h, n, rng, t_end):

    best_h = h.copy()

        if time.time() > t_end:

        c = _get_corr(h)

        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            contrib = h[lag:]* (1.0 - h[:n-lag])
        else:

                # To reduce: decrease h[i1] or increase h[i2]

            else:
                m = -lag
                i1, i2 = ti, ti + m

                # Undo
                for (idx, delta) in targets:
                    if 0 <= idx < n:
                        h[idx] -= delta

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        ht = np.clip(h - mid, 0.0, 1.0)
        if ht.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        m = -k
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _softmax_obj(h, alpha):
    c = _get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softmax_grad(h, alpha):
    c = _get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    e = np.exp(s - s.max())
    w = e / e.sum()
    n = len(h)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        lag = idx - (n - 1)
        g += w[idx] * _grad_lag(h, lag)
    return g * (2.0 / n)

def _topk_opt(h, n, topk, lr, iters, t_end):
    h = h.copy()
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for _ in range(iters):
        if time.time() > t_end:
            break
        c = _get_corr(h)
        idx_top = np.argsort(c)[-topk:]
        c_top = np.sort(c)[-topk:]
        w = np.exp((c_top - c_top.max()) * 15.0)
        w /= w.sum()
        g = np.zeros(n)
        for wk, ki in zip(w, idx_top):
            lag = ki - (n - 1)
            g += wk * _grad_lag(h, lag)
        g *= (2.0 / n)
        accepted = False
        for _t in range(12):
            h_t = _repair(h - lr * g, n)
            if _eval(h_t) < c5_cur - 1e-15:
                h = h_t
                c5_cur = _eval(h)
                accepted = True
                break
            lr *= 0.6
        if accepted:
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            lr = min(lr / 0.6 * 1.04, 0.08)
        else:
            lr *= 0.7
            if lr < 1e-9:
                lr = 1e-7
    return best_h, best_c5

def _directed_opt(h, n, rng, t_end):

    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur
    step = 0.012

        if time.time() > t_end:

        k_star = np.argmax(c)
        lag = k_star - (n - 1)
        if lag >= 0:
            contrib = h[lag:] * (1.0 - h[:n - lag])
        else:
            m = -lag
            contrib = h[:n - m] * (1.0 - h[m:])
        top_i = np.argsort(contrib)[-4:]
        improved = False
        for ti in top_i:
            if lag >= 0:
                i1, i2 = ti + lag, ti
            else:
                m = -lag
                i1, i2 = ti, ti + m
            for sign in [1, -1]:
                d = sign * step
                if not (0 <= i1 < n and 0 <= i2 < n):
                    continue
                h[i1] -= d
                h[i2] += d
                if h[i1] < 0 or h[i1] > 1 or h[i2] < 0 or h[i2] > 1:
                    h[i1] += d
                    h[i2] -= d
                    continue
                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < c5_cur - 1e-15:
                    c5_cur = c5n
                    improved = True
                    step = min(step * 1.03, 0.03)
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
                else:
                    h[i1] += d
                    h[i2] -= d
                    h = _repair(h, n)
        if not improved:
            step *= 0.9
            if step < 1e-7:
                break
    return best_h, best_c5

    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur

    no_imp = 0
    for _ in range(max_iter):

            break

        idxs = rng.choice(n, size=npts, replace=False)
        ds = scale * rng.randn(npts)
        ds -= ds.mean()
        h_t = h.copy()
        h_t[idxs] += ds
        if np.any(h_t < -0.1) or np.any(h_t > 1.1):
            continue
        h_t = _repair(h_t, n)
        c5n = _eval(h_t)
        if c5n < c5_cur - 1e-15:
            c5_cur = c5n
            h = h_t
            no_imp = 0
            scale = min(scale * 1.04, 0.025)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            no_imp += 1
            if no_imp > 250:
                scale *= 0.82
                no_imp = 0
                if scale < 5e-8:
                    break
    return best_h, best_c5

def _slsqp(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_softmax_obj, h, jac=_softmax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _interp_to(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)
