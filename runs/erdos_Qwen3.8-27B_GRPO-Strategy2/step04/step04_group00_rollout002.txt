                j = order[jj]
                best_d = 0.0
                best_c = best_c5
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _project(h, n)
                    best_c5 = _c5(h)
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _optimize_h(h0, n, bounds, tleft_fn):
    h = _project(h0.copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h)

    # Phase 1: L-BFGS-B with softmax
    for alpha in [2000, 10000, 50000, 200000, 800000]:
        if tleft_fn() < 15:
            break
        try:
            res = minimize(_softobj, best_h.copy(),
                           jac=lambda x, a=alpha: _softgrad(x, a, 80.0),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-20})
            hc = _project(res.x, n)
            c5v = _c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 2: Nesterov subsgradient
    h2, c5_2 = _nesterov_phase(best_h, n, tleft_fn)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: More L-BFGS-B at high alpha
    for alpha in [500000, 2000000]:
        if tleft_fn() < 10:
            break
        try:
            res = minimize(_softobj, best_h.copy(),
                           jac=lambda x, a=alpha: _softgrad(x, a, 200.0),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-20})
            hc = _project(res.x, n)
            c5v = _c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 4: Coordinate descent
    h4, c5_4 = _coord_phase2(best_h, n, tleft_fn, max_sweeps=80)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Simulated annealing
    rng = np.random.RandomState(12345)
    h5, c5_5 = _anneal_phase(best_h, n, rng, tleft_fn, n_iters=15000)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Fine random swaps
    h6, c5_6 = _fine_swaps(best_h, n, rng, tleft_fn)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    return best_h, best_c5

def _fine_swaps(h, n, rng, tleft_fn):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    n_swaps = min(40000, int(max(100, tleft_fn() * 3000)))
    for _ in range(n_swaps):
        if tleft_fn() < 2:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.choice([0.0005, 0.001, 0.002, -0.0005, -0.001, -0.002])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def tleft():
        return budget_s - elapsed()

    h_init = _project(np.array(initial_h_values, dtype=float), len(initial_h_values))
    n0 = len(h_init)
    bounds0 = [(0.0, 1.0)] * n0

    best_h, best_c5 = _optimize_h(h_init, n0, bounds0, lambda: tleft() - 200)

    # Try smaller n for potentially better bound
    for n_try in [100, 80]:
        if tleft() < 150:
            break
        h_interp = np.interp(
            np.linspace(0, 2, n_try, endpoint=False),
            np.linspace(0, 2, n0, endpoint=False), best_h)
        h_interp = _project(h_interp, n_try)
        bounds_try = [(0.0, 1.0)] * n_try
        h_try, c5_try = _optimize_h(h_interp, n_try, bounds_try, lambda: tleft() - 60)
        if c5_try < best_c5:
            best_c5, best_h, n0 = c5_try, h_try, n_try

    # Try n=120
    if tleft() > 100 and n0 != 120:
        h_interp = np.interp(
            np.linspace(0, 2, 120, endpoint=False),
            np.linspace(0, 2, n0, endpoint=False), best_h)
        h_interp = _project(h_interp, 120)
        h_try, c5_try = _optimize_h(h_interp, 120, [(0.0, 1.0)] * 120,
                                    lambda: tleft() - 20)
        if c5_try < best_c5:
            best_c5, best_h, n0 = c5_try, h_try, 120

    # Final micro-polish
    if tleft() > 5:
        rng = np.random.RandomState(seed + 7)
        h_fin, c5_fin = _fine_swaps(best_h, n0, rng, lambda: tleft() - 2)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin

    best_h = _project(best_h, n0)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, n0)
