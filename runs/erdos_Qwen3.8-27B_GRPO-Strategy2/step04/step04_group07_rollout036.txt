def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        d = target - s
        # Distribute proportionally
        if d > 0:
            # Add to entries not at 1
            mask = h < 1.0
            if mask.sum() == 0:
                break
            add = d / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + add)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            sub = (-d) / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - sub)
    h = np.clip(h, 0.0, 1.0)
    return h

def proj_box_hyperplane(h, target):
    """Project h onto {0<=h_i<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    f_val = np.max(corr)

What about a "sawtooth" or other non-sinusoidal shape? Or a two-level function?

**Two-level function**: h[i] ∈ {a, b} with (n1*a + (n-n1)*b)*dx = 1.
If we take b=0 and a determined by constraint: n1*a*2/n = 1, so a = n/(2*n1).
For a ∈ [0,1]: n1 ≥ n/2.

If h is 1 in n1 positions and 0 elsewhere: C₅ would be the maximum overlap of a 1-set with the complement of a shifted 1-set. This is the classic minimum overlap problem for sets!

For n1 = n/2: h is 1 in half the positions, 0 in the other half.

Actually for binary h: g_k = |{i : h[i]=1, h[i+k]=0}| + |{i : h[i]=0, h[i+k]=1}|... no.

g_k = sum_i h[i]*(1-h[i+k]) = number of i where h[i]=1 and h[i+k]=0 (for the range where both are defined).

For binary h with half 1s and half 0s, we want to minimize the maximum, over k, of the number of "1→0" transitions at lag k.

This is related to constructing "costas arrays" or "low cross-correlation" binary sequences.

The best binary sequences for this are related to "m-sequences" or "Barker sequences."

    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            add = min(d, (1.0 - h[mask]).min() * 0.5)
            h[mask] += add
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            sub = min(-d, h[mask].min() * 0.5)
            h[mask] -= sub
    return np.clip(h, 0.0, 1.0)

def proj_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def softmax_obj(v, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    m = np.max(corr)
    s = alpha * np.clip(corr - m, -500, 500)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(s)))

def slsqp_solve(h, n, alpha, maxiter=1000):
    dx = 2.0 / n
    def obj(v):
        return softmax_obj(v, n, alpha)
    def obj_grad(v):
        # Gradient of soft-max of corr
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        m = np.max(corr)
        s = alpha * np.clip(corr - m, -500, 500)
        e = np.exp(s)
        weights = e / np.sum(e)
        # d(corr[k])/d(v[i]) = (1-v[i+k]) - v[i-k] (with boundary conditions)
        grad = np.zeros(n)
        for k_idx in range(2*n - 1):
            if weights[k_idx] < 1e-15:
                continue
            k = k_idx - n + 1
            for i in range(n):
                g = 0.0
                if 0 <= i + k < n:
                    g += (1.0 - v[i + k])
                if 0 <= i - k < n:
                    g -= v[i - k]
                grad[i] += weights[k_idx] * g
        grad *= dx
        return grad, obj(v)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - n/2.0, 'jac': lambda v: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

Wait, the gradient computation in the inner loop is O(n * (2n-1)) = O(2n²) which for n=64 is about 8192 operations. That's fine. But I'm doing it in a Python loop which is slow. Let me vectorize.

For a given k (the lag), the gradient contribution is:
- For i in [max(0,-k), min(n-1, n-1-k)]: (1-v[i+k]) if i+k valid
- For i in [max(0,k), min(n-1, n-1+k)]: -v[i-k] if i-k valid

This can be vectorized as:
```python
def corr_gradient(v, n, weights):
    """Compute sum_k weights[k] * d(corr_k)/d(v)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        w = weights[k_idx]
        if abs(w) < 1e-15:
            continue
        k = k_idx - n + 1
        # (1-v[i+k]) term: valid for i in [0, n-1-k] if k>=0, or i in [k, n-1] if k<0
        if k >= 0:
            if n - 1 - k >= 0:
                grad[0:n-k] += w * (1.0 - v[k:n])
                grad[k:n] -= w * v[0:n-k]
        else:
            kp = -k
            if n - 1 - kp >= 0:
                grad[kp:n] += w * (1.0 - v[0:n-kp])
                grad[0:n-kp] -= w * v[kp:n]
    return grad * dx
```

This is O(n) per active lag, and at most all 2n-1 lags are active, so O(n²) total. For n=64, that's about 8000 ops, very fast.

Let me now think about the coordinate descent more carefully.

**Active-lag coordinate descent**:
- Maintain a set H of "hot" lags (those with g_k within 0.01 of the max)
- When changing h[i] and h[j], only recompute g_k for k in H (plus a few neighbors)
- If a new lag becomes active, add it to H
- Periodically recompute all lags to update H

When we change h[i] → h[i]+δ and h[j] → h[j]-δ:
The change in g_k is:
Δg_k = δ*(1-h[i+k]) - δ*h[i-k] - δ*(1-h[j+k]) + δ*h[j-k]
     (with appropriate boundary conditions)

Wait, this isn't quite right because h[i] also appears in g_k as the second factor for lag (k-i+j) or something. Let me be more careful.

g_k = sum_{valid i} h[i]*(1-h[i+k])

If we change h[i] by δ (and h[j] by -δ):
- In g_k, h[i] appears as first factor when i is valid: contributes δ*(1-h[i+k])
- h[i] appears as second factor (as h[i-k] for the term where the index is i-k): contributes -δ*h[i-k-k]... 

Hmm this is getting confusing. Let me just use the formula:
∂g_k/∂h[m] = (1-h[m+k])*[0≤m+k<n] - h[m-k]*[0≤m-k<n]

So changing h[i] by δ and h[j] by -δ changes g_k by:
δ * [(1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n]]
- δ * [(1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]]

This is O(1) per lag (just 4 lookups), so checking all 2n-1 lags is O(n). And we need to find the max, which is another O(n). So each coordinate descent step is O(n) = O(64) ops. Very fast!

But we also need to update the actual correlation values. We can maintain them incrementally:
corr[k] += δ * [(1-h[i+k]) - h[i-k]] - δ * [(1-h[j+k]) - h[j-k]] (with boundary conditions)

Then the max can be tracked with a heap or just re-scanned (O(n) for 127 values).

This makes coordinate descent very fast! Each step is O(n) instead of O(n²).

Wait, there's a bug in my incremental update. When I compute the change to corr[k_idx], I'm using the OLD values of h[i+k], h[i-k], h[j+k], h[j-k]. But h[i] and h[j] are about to change. However, the formula ∂g_k/∂h[m] uses the CURRENT values of h (before the change), so this is correct for a first-order update. For small δ, this is accurate.

But actually, the exact change is:
New g_k = sum h'[i](1-h'[i+k]) where h'[i] = h[i]+δ for i, h'[j] = h[j]-δ for j, else h.

The exact change involves products like (h[i]+δ)(1-h[i+k]) - h[i](1-h[i+k]) = δ(1-h[i+k]), which is what I have. Similarly for the second factor. So it's exact for the first factor. But for the second factor: if i+k = j, then the term is h[i](1-h[j]) and becomes (h[i]+δ)(1-h[j]+δ) = h[i](1-h[j]) + δ(1-h[j]) + δ*h[i] - δ². There's an extra -δ² term!

So the update is only first-order accurate when i+k = j or j+k = i. For small δ (≤0.05), the error is at most δ² = 0.0025 per affected term, which is small.

Actually, for the correctness of the comparison (new_max < best_c5), a small error in the update could lead to accepting a slightly worse move or rejecting a slightly better one. For δ ≤ 0.01, the error is ≤ 0.0001, which is negligible compared to the improvements we're looking for.

def repair(h, n):

        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:

    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    m = np.max(corr)
    s = alpha * np.clip(corr - m, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def corr_grad_w(v, n, weights):
    """Gradient of weighted sum of correlations. weights indexed 0..2n-2 for lag (k_idx - n + 1)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if abs(w) < 1e-14:
            continue
        k = k_idx - n + 1
        if k >= 0:
            if n - 1 - k >= 0:
                grad[:n-k] += w * (1.0 - v[k:])
                grad[k:] -= w * v[:n-k]
        else:
            kp = -k
            if n - 1 - kp >= 0:
                grad[kp:] += w * (1.0 - v[:n-kp])
                grad[:n-kp] -= w * v[kp:]
    return grad * dx

def slsqp_solve(h, n, alpha, maxiter=500):
    def obj(v):
        return soft_obj(v, n, alpha)
    def obj_jac(v):
        dx = 2.0 / n
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        m = np.max(corr)
        s = alpha * np.clip(corr - m, -500, 500)
        e = np.exp(s)
        sw = np.sum(e)
        weights = e / sw
        g = corr_grad_w(v, n, weights)
        return g, obj(v)
    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(400):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            h[m] = np.minimum(1.0, h[m] + d * 0.08)
        else:
            m = h > 1e-14
            if not m.any():
                break
            h[m] = np.maximum(0.0, h[m] + d * 0.08)
    return np.clip(h, 0.0, 1.0)

def proj_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def soft_obj(v, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    m = np.max(corr)
    s = alpha * np.clip(corr - m, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def corr_grad_w(v, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if abs(w) < 1e-14:
            continue
        k = k_idx - n + 1
        if k >= 0:
            if n - 1 - k >= 0:
                grad[:n-k] += w * (1.0 - v[k:])
                grad[k:] -= w * v[:n-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - v[:n-kp])
            grad[:n-kp] -= w * v[kp:]
    return grad * dx

def _slsqp_obj(v, n, alpha):
    return soft_obj(v, n, alpha)

def _slsqp_jac(v, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full') * dx
    m = np.max(corr)
    s = alpha * np.clip(corr - m, -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    g = corr_grad_w(v, n, w)
    return g, soft_obj(v, n, alpha)

    return v.sum() - _glb_n / 2.0

    return np.ones(len(v))

def slsqp_solve(h, n, alpha, maxiter=500):
    global _glb_n
    _glb_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_slsqp_obj, h, args=(n, alpha), jac=_slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def _lbfgs_obj(v, n, alpha, pen):

def subgrad_opt(h, n, n_iters, t_end, temp):
    h = h.copy()
    best = compute_c5(h, n)
    lr = 0.015
    for it in range(n_iters):

        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx

        w = np.exp((corr - f) / temp)
        w /= np.sum(w)
        grad = corr_grad_w(h, n, w)
        h_new = proj_box_sum(h - lr * grad, n / 2.0)
        f_new = compute_c5(h_new, n)
        if f_new < best:
            h = h_new
            best = f_new
            lr = min(lr * 1.05, 0.08)
        else:
            lr *= 0.8
