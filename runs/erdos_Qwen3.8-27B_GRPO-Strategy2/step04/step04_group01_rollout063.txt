def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_grad_contrib(h, k, n):
    """Add gradient contribution for correlation index k to a zero array, return it."""
    g = np.zeros(n)
    m = abs(k - (n - 1))
    if m < n:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    return g

def _active_obj_grad(h, active_k, alpha):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    c_a = c[active_k]
    m = c_a.max()
    s = np.clip(alpha * (c_a - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    obj = (m + np.log(w.sum()) / alpha) * dx
    
    g = np.zeros(n)
    for idx in range(len(active_k)):
        if w[idx] < 1e-14:
            continue
        k = active_k[idx]

        if sh >= n:
            continue
        g[sh:] += w[idx] * (1.0 - h[:n - sh])
        g[:n - sh] -= w[idx] * h[sh:]
    g *= dx
    return obj, g, c

    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    ca = c[ak]
    m = ca.max()
    s = np.clip(alpha * (ca - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()

            continue

        g[:n - sh] -= w[i] * h[sh:]
