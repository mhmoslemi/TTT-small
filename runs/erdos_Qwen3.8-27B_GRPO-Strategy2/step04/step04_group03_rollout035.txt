def project_feasible(v, n):
    """Project v onto {h : 0<=h<=1, sum(h)=n/2}"""
    # Bisection on lambda
    lo, hi = v.min() - 2, v.max() + 2
    target = n / 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

def compute_grad(h, n):

    # Compute all g_m values
    # Use the correlation
    c = np.correlate(h, 1.0 - h, mode='full')  # n = len

    # Let's use exact max for now
    k_max = np.argmax(c)
    m = k_max - (n - 1)  # the lag
    
    grad = np.zeros(n)
    if m == 0:
        grad = 1.0 - 2.0 * h
    elif m > 0:

                    hc[i], hc[j] = a - sc, b + sc
                if improved:
                    break
        if improved:
            fails = 0
        else:
            ok2 = False
            for _ in range(3000):
                if _tgl(t0, bud) < min_rem:
                    break
                ii, jj = rng.randint(n), rng.randint(n)
                if ii == jj:
                    continue
                dd = scale * rng.randn()
                aa, bb = hc[ii] + dd, hc[jj] - dd
                if aa < 0 or aa > 1 or bb < 0 or bb > 1:
                    continue
                hc[ii], hc[jj] = aa, bb
                cv = _eval(hc)
                if cv < cc - 1e-15:
                    cc, ok2 = cv, True
                    break
                hc[ii], hc[jj] = aa - dd, bb + dd
            if ok2:
                fails = 0
                scale = min(scale * 1.04, 0.04)
            else:
                fails += 1
                scale *= 0.85
                if fails > 25:
                    hc = _repair(hc + 0.03 * rng.randn(n), n)
                    cc, fails, scale = _eval(hc), 0, 0.012
    return hc, cc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    bH, bC = h0.copy(), _eval(h0)

    # Phase 1: Subgradient descent from initial
    h1, c1 = _subgrad_opt(h0, n0, t0, budget_s, 200, seed)
    if c1 < bC:
        bC, bH = c1, h1

    # Phase 2: Local search
    h2, c2 = _local_search(bH, len(bH), t0, budget_s, 80, seed + 1)
    if c2 < bC:
        bC, bH = c2, h2

    # Phase 3: Try other resolutions
    for nt in (80, 100, 120, 64, 200):
        if _tgl(t0, budget_s) < 60:
            break
        xd = np.linspace(0, 2, nt, endpoint=False)
        xs = np.linspace(0, 2, len(bH), endpoint=False)
        ht = _repair(np.interp(xd, xs, bH), nt)
        ct = _eval(ht)
        if ct > bC + 0.03:
            continue
        ho, co = _subgrad_opt(ht, nt, t0, budget_s, 40, seed + 2 + nt)
        if co < bC:
            bC, bH = co, ho
            ho2, co2 = _local_search(bH, nt, t0, budget_s, 30, seed + 3 + nt)
            if co2 < bC:
                bC, bH = co2, ho2

    # Phase 4: Final fine polish
    if _tgl(t0, budget_s) > 15:
        nn = len(bH)
        h4, c4 = bH.copy(), bC
        t_end = time.time() + min(10, _tgl(t0, budget_s) - 5)
        for _ in range(200000):
            if time.time() > t_end:
                break
            i, j = rng.randint(nn), rng.randint(nn)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 5e-5 * rng.choice([1, 2, 5, 10])
            ni, nj = h4[i] + d, h4[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h4[i], h4[j] = ni, nj
                cn = _eval(h4)
                if cn < c4 - 1e-16:
                    c4 = cn
                    if cn < bC:
                        bC, bH = cn, h4.copy()
                else:
                    h4[i], h4[j] = ni - d, nj + d

    bH = _repair(bH, len(bH))
    bC = _eval(bH)
    return (bH, bC, len(bH))
