def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr_gradient(h, alpha, lam):
    """Gradient of soft-max objective."""
    c = _corr(h)
    n = len(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += pk * (1.0 - h[:n - L])
            g[:n - L] -= pk * h[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - h[M:])
            g[M:] -= pk * h[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)

    c = _corr(h)
    n = len(h)
    cm = np.max(c)
    cmin = np.min(c)
    spread = cm - cmin
    if spread < 1e-12:
        spread = 1.0
    thresh = cm - active_frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    # Zero out mean component (to preserve sum constraint)
    g -= g.mean()
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smobj(h, alpha, lam):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(h) - len(h) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _smgrad(h, alpha, lam):
    c = _corr(h)
    n = len(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += pk * (1.0 - h[:n - L])
            g[:n - L] -= pk * h[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - h[M:])
            g[M:] -= pk * h[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(h) - n / 2.0)
    return g

def _subgrad(h, frac):
    c = _corr(h)
    n = len(h)
    cm = np.max(c)
    cmin = np.min(c)
    spread = max(cm - cmin, 1e-12)
    thresh = cm - frac * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    g -= g.mean()
    return g

def _lbfgs_phase(h, n, bounds, deadline):
    bh = h.copy()
    bc5 = _c5(bh, 2.0 / n)
    for a, lam in [(200,80),(1000,60),(5000,50),(20000,40),(100000,30),(500000,20)]:
        if time.time() > deadline - 15:
            break
        try:
            res = minimize(_smobj, bh.copy(), args=(a, lam),
                           jac=_smgrad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter':400, 'ftol':1e-18})
            hc = _repair(res.x, n)
            c5v = _c5(hc, 2.0 / n)
            if c5v < bc5:
                bc5 = c5v
                bh = hc
        except Exception:
            pass
    return bh, bc5

def _momentum_phase(h, n, deadline, max_it=3000):
    dx = 2.0 / n
    h = h.copy()
    c5c = _c5(h, dx)
    v = np.zeros(n)
    lr = 0.015
    beta = 0.85
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline - 10:
            break
        g = _subgrad(h, 0.07)
        ng = np.linalg.norm(g)
        if ng < 1e-13:
            break
        g /= ng
        v = beta * v - lr * g
        y = _repair(h + v, n)
        c_new = _c5(y, dx)
        if c_new < c5c - 1e-15:
            h = y
            c5c = c_new
            lr = min(lr * 1.01, 0.05)
            stall = 0
        else:
            stall += 1
            v *= 0.5
            lr *= 0.85
            if stall > 80 or lr < 1e-7:
                break
    return h, c5c

def _full_optimize(h0, n, deadline, rng):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h, c5v = _lbfgs_phase(h0, n, bounds, deadline)
    h, c5v2 = _momentum_phase(h, n, deadline, 2500)
    c5v = min(c5v, c5v2)
    h = _repair(h, n) if c5v2 < c5v else h
    c5v = _c5(h, dx)
    h, c5v3 = _coord_phase(h, n, deadline, rng)
    c5v = min(c5v, c5v3)
    return h, _c5(h, dx)
