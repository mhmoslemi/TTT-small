            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _subgrad_avg(h, n, lags_frac=0.15):
    c = _corr(h)
    n_lags = len(c)
    n_active = max(1, int(n_lags * lags_frac))
    top_lags = np.argsort(c)[::-1][:n_active]
    g = np.zeros(n)
    for k in top_lags:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= n_active
    return g, top_lags

def _binary_refine(h, n, dx, time_fn, deadline, max_rounds=200):
    """Snap to binary, do local search, return best binary h."""
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    c = _corr(h_bin)
    c5 = float(np.max(c)) * dx
    rv = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        if time_fn() > deadline:
            break
        k_max = int(np.argmax(c))
        L = k_max - (n - 1)
        improved = False
        
        # Targeted: find worst contributors to c[k_max]
        if L >= 0:
            valid = n - L
            # c[k_max] = sum_{i=0}^{valid-1} h[i+L]*(1-h[i])
            # Bad pairs: h[i+L]=1, h[i]=0
            bad_i = [i for i in range(valid) if h_bin[i+L]==1 and h_bin[i]==0]
            bad_i.sort(key=lambda i: -1)  # all contribute equally
        else:
            M = -L
            valid = n - M
            bad_i = [i for i in range(M, n) if h_bin[i]==1 and h_bin[i+L]==0]
        
        # Try flipping from the worst positions
        tried = 0
        for i in bad_i:
            if tried >= 50:

            # Flip h_bin at position p from 1→0, and q from 0→1
            if L >= 0:

                h_bin[q] = 1

                    c = c_new
                    c5 = c5_new
                    improved = True
                    tried += 1

                else:

                    h_bin[q] = 0

                    c = c_new
                    c5 = c5_new
                    break
                else:
                    h_bin[p] = 1
                    h_bin[q] = 0
    return h_bin, c5

def _pairwise_continuous(h, n, dx, time_fn, deadline, max_rounds=500):
    h = h.copy()
    c5 = _c5(h, dx)
    rv = np.random.RandomState(123)
    deltas = np.array([0.001, 0.003, 0.008, 0.02, -0.001, -0.003, -0.008, -0.02])

            if c_new < c5 - 1e-16:

    return h, c5

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _coord_deg(h, n, dx, t_fn, t_end, sweeps=10):
    """Coordinate descent: for each i, find best h[i] given others fixed."""
    h = h.copy()
    for sw in range(sweeps):
        if t_fn() > t_end:
            break
        res = _corr(h)  # current correlate values
        for i in range(n):
            if t_fn() > t_end:
                return h, _c5(h, dx)
            x0 = h[i]
            # Compute delta[k] for all k
            delta = np.zeros(2*n-1)
            for k in range(2*n-1):
                L = k - (n-1)
                d = 0.0
                if L >= 0:
                    if L <= i:
                        d += (1 - h[i-L])
                    if i+L < n:
                        d += (-h[i+L])
                else:
                    M = -L
                    if i < n-M:
                        d += (1 - h[i+M])
                    if i >= M:
                        d += (-h[i-M])
                delta[k] = d
            # Find x in [0,1] minimizing max_k (res[k] + (x-x0)*delta[k])
            best_x, best_m = x0, np.inf
            for x_trial in np.concatenate([[0, 1, 0.5], np.linspace(0.02, 0.98, 24)]):
                val = res + (x_trial - x0) * delta
                m = float(np.max(val))
                if m < best_m:
                    best_m = m
                    best_x = x_trial
            h[i] = best_x
        h = _repair(h, n)
    return h, _c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _coord_deg(h, n, dx, t_fn, t_end, sweeps=15, n_trials=50):
    h = h.copy()
    res = _corr(h).copy()
    xs = np.linspace(0.0, 1.0, n_trials)
    best_c5 = _c5(h, dx)
    for sw in range(sweeps):
        if t_fn() > t_end:
            break
        for i in range(n):
            x0 = h[i]
            delta = np.zeros(2 * n - 1)
            for L in range(-(n - 1), n):
                if L == 0:
                    continue
                k = L + (n - 1)
                d = 0.0
                if L > 0:
                    if L <= i:
                        d += (1.0 - h[i - L])
                    if i + L < n:
                        d += (-h[i + L])
                else:
                    M = -L
                    if i < n - M:
                        d += (1.0 - h[i + M])
                    if i >= M:
                        d += (-h[i - M])
                delta[k] = d
            best_x, best_m = x0, np.inf
            for x in xs:
                nr_max = np.max(res + (x - x0) * delta)
                center_val = res[n - 1] + x * (1.0 - x) - x0 * (1.0 - x0)
                m = max(float(nr_max), center_val)
                if m < best_m:
                    best_m = m
                    best_x = x
            h[i] = best_x
            res += (best_x - x0) * delta
            res[n - 1] += best_x * (1.0 - best_x) - x0 * (1.0 - x0)
        h = _repair(h, n)
        c_new = _c5(h, dx)
        if c_new < best_c5:
            best_c5 = c_new
        res = _corr(h).copy()
    return h, best_c5

def _bin_fn(h, n, dx, t_fn, t_end, rounds=300, seed=42):
    tgt = n // 2
    ord_ = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[ord_[:tgt]] = 1.0
    c = _corr(hb)
    c5 = float(np.max(c)) * dx
    rv = np.random.RandomState(seed)
    stall = 0
    for _ in range(rounds):
        if t_fn() > t_end:
            break
        if stall > 40:
            break
        km = int(np.argmax(c))
        L = km - (n - 1)
        ok = False
        bad = []
        if L >= 0:
            for i in range(n - L):
                if hb[i + L] == 1 and hb[i] == 0:
                    bad.append(i + L)
        else:
            M = -L
            for i in range(M, n):
                if hb[i] == 1 and hb[i - M] == 0:
                    bad.append(i)
        rv.shuffle(bad)
        for p in bad[:50]:
            if t_fn() > t_end:
                break
            zp = np.where(hb == 0)[0]
            rv.shuffle(zp)
            for q in zp[:30]:
                if q == p:
                    continue
                hb[p], hb[q] = 0, 1
                c2 = _corr(hb)
                c52 = float(np.max(c2)) * dx
                if c52 < c5 - 1e-15:
                    c, c5 = c2, c52
                    ok = True
                    break
                hb[p], hb[q] = 1, 0
            if ok:
                break
        if not ok:
            op = np.where(hb == 1)[0]
            zp = np.where(hb == 0)[0]
            for _ in range(100):
                p = op[rv.randint(len(op))]
                q = zp[rv.randint(len(zp))]
                hb[p], hb[q] = 0, 1
                c2 = _corr(hb)
                c52 = float(np.max(c2)) * dx
                if c52 < c5 - 1e-15:
                    c, c5 = c2, c52
                    ok = True
                    break
                hb[p], hb[q] = 1, 0
        stall = 0 if ok else stall + 1
    return hb, c5

def _sgd_fn(h, n, dx, t_fn, t_end, iters=300, frac=0.12):
    h = h.copy()
    c5 = _c5(h, dx)
    step = 0.012
    stale = 0
    for _ in range(iters):
        if t_fn() > t_end:
            break
        c = _corr(h)
        n_act = max(1, int(len(c) * frac))
        top = np.argsort(c)[::-1][:n_act]
        g = np.zeros(n)
        for k in top:
            L = k - (n - 1)
            if L == 0:
                g += (1.0 - 2.0 * h)
            elif L > 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                M = -L
                g[:n - M] += (1.0 - h[M:])
                g[M:] -= h[:n - M]
        g /= n_act
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-14:
            break
        g /= ng
        best_s, best_c, found = 0, c5, False
        for s in (step, step * 0.5, step * 0.25, step * 0.1):
            h_t = _repair(h - s * g, n)
            c_t = _c5(h_t, dx)
            if c_t < best_c:
                best_c, best_s, found = c_t, s, True
        if found:
            h = _repair(h - best_s * g, n)
            c5 = _c5(h, dx)
            stale = 0
            step = min(step * 1.08, 0.04)
        else:
            step *= 0.5
            stale += 1
            if step < 1e-8 or stale > 50:
                break
    return h, c5

def _tune(h, n, dx, t_fn, t_end, iters=5000, seed=7):
    h = h.copy()
    c5 = _c5(h, dx)
    rv = np.random.RandomState(seed)
    ds = np.array([0.001, 0.003, 0.008, 0.02, -0.001, -0.003, -0.008, -0.02])
    for _ in range(iters):
        if t_fn() > t_end:
            break
        i, j = rv.randint(n), rv.randint(n)
        if i == j:
            continue
        d = ds[rv.randint(len(ds))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5 - 1e-16:
                c5 = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def el():
        return time.time() - t0

    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
