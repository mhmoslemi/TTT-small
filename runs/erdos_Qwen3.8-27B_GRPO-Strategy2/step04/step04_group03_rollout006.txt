    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, n, lag):
    g = np.zeros(n)
    i1, j1 = max(0, lag), min(n, n + lag)
    if i1 < j1:
        g[i1:j1] += (1.0 - h[i1 - lag:j1 - lag])
    i2, j2 = max(0, -lag), min(n, n - lag)
    if i2 < j2:
        g[i2:j2] -= h[i2 + lag:j2 + lag]
    return g

def _topk_grad(h, n, k):
    c = _corr_lags(h)
    n_corr = len(c)
    if k >= n_corr:
        topk = np.arange(n_corr)
    else:
        topk = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for ki in topk:
        lag = ki - (n - 1)
        g += _grad_lag(h, n, lag)
    g /= len(topk)
    return g

def _subgrad_opt(h, n, t_end, k_start=20, k_end=3, steps_ratio=0.6):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval(h)
    step = 0.015
    vel = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    max_steps = 8000
    for it in range(max_steps):
        if time.time() > t_end:
            break
        frac = it / max_steps
        k = max(2, int(k_start * (1 - frac) + k_end * frac))
        g = _topk_grad(h, n, k)
        vel = momentum * vel - step * g
        h_new = _repair(h + vel, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-16:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            no_imp = 0
            step = min(step * 1.01, 0.04)
        else:
            step *= 0.95
            no_imp += 1
            if no_imp > 80:
                vel *= 0.5
                no_imp = 0
                if step < 2e-6:
                    break
    return best_h, best_c5

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _gs(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        wk = w[kk]
        if wk < 1e-14:
            continue
        lag = kk - (n - 1)
        i1, j1 = max(0, lag), min(n, n + lag)
        if i1 < j1:
            g[i1:j1] += wk * (1.0 - x[i1 - lag:j1 - lag])
        i2, j2 = max(0, -lag), min(n, n - lag)
        if i2 < j2:
            g[i2:j2] -= wk * x[i2 + lag:j2 + lag]
    return g

def _cf(x, n):
    return np.sum(x) - n / 2.0

def _cj(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(_sf, h, jac=_gs, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, n, rng, t_end, max_iters=300000):
    c5c = _eval(h)
    best_h = h.copy()
    scale = 0.006
    no_imp = 0
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            best_h = h.copy()
            no_imp = 0
            scale = min(scale * 1.02, 0.015)
        else:
            h[i] -= d
            h[j] += d
            h = _repair(h, n)
            no_imp += 1
            if no_imp > 150:
                scale *= 0.85
                no_imp = 0
                if scale < 3e-7:
                    break
    return best_h, c5c

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def _block_move_search(h, n, rng, t_end, max_iters=20000):
    c5c = _eval(h)
    best_h = h.copy()
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        blen = rng.randint(2, max(3, n // 10))
        i = rng.randint(n - blen + 1)
        j = rng.randint(n - blen + 1)
        if abs(i - j) < blen:
            continue
        h[i:i+blen], h[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            best_h = h.copy()
        else:
            h[i:i+blen], h[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
    return best_h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    hp = np.array(initial_h_values, dtype=float).copy()
    n_init = len(hp)
    h_init = _repair(hp, n_init)
    best_h = h_init.copy()
    best_c5 = _eval(h_init)
    best_n = n_init

    # Multi-resolution SLSQP
    n_list = sorted(set([n_init - 10, n_init - 20, n_init, n_init + 20, n_init + 40]))
    n_list = [n for n in n_list if 30 <= n <= 300]

    for n in n_list:
        if _tl() < 50:
            break
        if n == n_init:
            h0 = best_h.copy()
        else:
            h0 = _repair(_interp(h_init, n_init, n), n)
        c5_0 = _eval(h0)
        if c5_0 > best_c5 + 0.02:
            continue
        for alpha in [1000, 10000, 100000, 1000000]:
            if _tl() < 30:
                break
            mx = 500 if n <= 120 else 250
            h_new = _slsqp(h0, n, alpha, maxiter=mx)
            c5_new = _eval(h_new)
            if c5_new < c5_0 - 1e-15:
                c5_0 = c5_new
                h0 = h_new.copy()
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = n

    # Subgradient optimization (top-k)
    if _tl() > 100:
        t_end = time.time() + _tl() * 0.35
        h2, c5_2 = _subgrad_opt(best_h.copy(), best_n, t_end, k_start=25, k_end=3)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # SLSQP polish after subgradient
    if _tl() > 50:
        h3 = _slsqp(best_h.copy(), best_n, 2000000, maxiter=400)
        c5_3 = _eval(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Block move search
    if _tl() > 60:
        t_end = time.time() + min(30, _tl() * 0.4)
        h4, c5_4 = _block_move_search(best_h.copy(), best_n, rng, t_end, max_iters=50000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Intensive local search
    if _tl() > 40:
        t_end = time.time() + min(35, _tl() * 0.5)
        h5, c5_5 = _local_search(best_h.copy(), best_n, rng, t_end, max_iters=400000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final SLSQP with very high alpha
    if _tl() > 25:
        h6 = _slsqp(best_h.copy(), best_n, 5000000, maxiter=200)
        c5_6 = _eval(h6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final fine local search
    if _tl() > 12:
        t_end = time.time() + min(8, _tl() - 3)
        h7, c5_7 = _local_search(best_h.copy(), best_n, rng, t_end, max_iters=60000)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
