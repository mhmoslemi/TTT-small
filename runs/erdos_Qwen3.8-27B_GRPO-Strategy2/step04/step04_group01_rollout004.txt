def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _top_constraints(h, M=5):
    """Find the M shifts with the highest overlap values."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    idx = np.argsort(c)[::-1][:M]
    return idx, c

def _grad_single(h, k, n):
    """Gradient of the overlap at shift k."""
    dx = 2.0 / n
    g = np.zeros(n)
    shift = (n - 1) - k  # k is index in full correlation
    if shift >= 0:
        i1 = np.arange(n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        if shift > 0:
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:

        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

        else:

    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

    bnds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_set_descent(h, max_iters=500, n_top=3):
    """Descend on the top-k active constraints."""
    n = len(h)
    c5c = _eval(h)
    for _ in range(max_iters):
        idx, c = _top_constraints(h, n_top)
        # Weighted gradient: weight by how much each exceeds the mean of top-k
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            g += w * _grad_single(h, k, n)
        
        # Projected line search

            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                break
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, max_iters, n_top):

    dx = 2.0 / n
    c5c = _eval(h)

            h_new = _slsqp(hp, 300000, 800, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase F: Long active set + structured pairwise
    if _el() < budget_s - 150:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    if _el() < budget_s - 80:
        h_new, c5 = _structured_pair(best_h.copy(), 5000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase G: Multi-scale pairwise refinement
    if _el() < budget_s - 50:
        for scale in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]:
            if _el() > budget_s - 25:
                break
            h_new, c5 = _pairwise_refine(best_h.copy(), 4000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase H: Final SLSQP polish
    if _el() < budget_s - 15:
        h_new = _slsqp(best_h, 1000000, 5000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    # Phase I: Ultra-fine active set
    if _el() < budget_s - 5:
        h_new, c5 = _active_descent(best_h.copy(), 8000, 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
