import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    remaining = abs(diff)
    per = remaining / max(1, n)
    for idx in order:
        if diff > 0:
            a = min(per, 1.0 - h[idx])
            h[idx] += a
            remaining -= a
        else:
            a = min(per, h[idx])
            h[idx] -= a
            remaining -= a
        if remaining < 1e-14:
            break
    return h

def _sm_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(s.exp().sum()) / alpha

def _sm_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    ew = s.exp()
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    return g / alpha

def _top_grad(h, n, nn=40):
    c = _corr(h)
    thresh = c.max() - 0.1 * (c.max() - c.min())
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= max(1, len(active))
    return g

def _coord_descent(h, n, dx, rng, time_fn, budget,BudgetLeft):
    h = h.copy()
    c5c = _c5(h, dx)
    best_h = h.copy()
    best_c5 = c5c
    steps = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                      -0.001, -0.003, -0.008, -0.02, -0.05])
    no_improve = 0
    while time_fn() < budget:
        g = _top_grad(h, n)
        gc = g - g.mean()
        idx_order = np.argsort(np.abs(gc))[::-1]
        top = idx_order[:min(50, n)]
        improved = False
        cand = c5c
        for i_pos in range(len(top)):
            i = top[i_pos]
            for j_pos in range(len(top)):
                j = top[j_pos]
                if i == j:
                    continue
                for d in steps:
                    ni, nj = h[i] + d, h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    h[i], h[j] = ni - d, nj + d
                    if c < cand - 1e-15:

                        cand = c
                        improved = True
                        break
                if improved:
                    break
            if improved:

                cand = _c5(h, dx)

    f_prev = _sm_obj(h, 200000.0)
    step = 0.008
    alpha = 200000.0

            break
        alpha *= 1.3

        g = _sm_grad(h, alpha)

                f_prev = f_new
                accepted = True
                break
            step *= 0.5

            continue
