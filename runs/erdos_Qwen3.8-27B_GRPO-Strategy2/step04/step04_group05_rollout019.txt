def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
        else:
            o = np.argsort(h)
        for i in o:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = max(d, -h[i])
            h[i] += a
            d -= a
    return h

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_k(h, ki, n):
    g = np.zeros(n)
    j = ki - (n - 1)
    if j == 0:
        g[:] = 1.0 - 2.0 * h
    elif j > 0:
        g[j:] += 1.0 - h[:n - j]
        g[:n - j] -= h[j:]
    else:
        m = -j
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g
