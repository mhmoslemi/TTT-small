def repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                    break

def slsqp_active(h_init, n, target, top_k, maxiter, deadline):
    dx = 2.0 / n
    h0 = h_init.copy()
    
    def objective(x):
        h = x[:n]
        t = x[n]
        return t
    
    def constraint_eq(x):
        return np.sum(x[:n]) - target
    
    def make_ineq(k_idx):
        # corr[k_idx] - t <= 0
        def c(x):
            h = x[:n]
            corr = np.correlate(h, 1.0 - h, mode='full')
            return -corr[k_idx] + x[n]
        return c

    top_lags = np.argsort(-corr0)[:top_k]

    res = minimize(objective, np.concatenate([h0, [compute_c5(h0, dx)]]),

                   options={'maxiter': maxiter, 'ftol': 1e-15})
