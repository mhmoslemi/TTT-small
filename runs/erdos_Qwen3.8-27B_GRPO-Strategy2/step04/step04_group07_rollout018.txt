from scipy.optimize import minimize
import time

# Fourier basis evaluation
def fourier_H(x, coeffs):
    # coeffs = [a1, b1, a2, b2, ..., am, bm]
    m = len(coeffs) // 2
    H = np.zeros_like(x)
    for j in range(1, m+1):
        a, b = coeffs[2*(j-1)], coeffs[2*(j-1)+1]
        H += a * np.cos(j * np.pi * x) + b * np.sin(j * np.pi * x)
    return H

def fourier_c5_continuous(coeffs, n_k=200):
    # f(k) = 0.5 - sum r_j^2 cos(j*pi*k)
    m = len(coeffs) // 2
    ks = np.linspace(0, 2, n_k)
    r2 = np.array([coeffs[2*(j-1)]**2 + coeffs[2*(j-1)+1]**2 for j in range(1, m+1)])
    f = 0.5 - np.zeros(n_k)
    for j in range(1, m+1):
        f -= r2[j-1] * np.cos(j * np.pi * ks)
    return np.max(f)

def fourier_obj(coeffs, alpha, n_k=200, n_x=400):
    # Smooth max of f(k) + penalty for bound violation
    m = len(coeffs) // 2
    ks = np.linspace(0, 2, n_k)
    r2 = np.array([coeffs[2*(j-1)]**2 + coeffs[2*(j-1)+1]**2 for j in range(1, m+1)])
    f = 0.5 - np.zeros(n_k)
    for j in range(1, m+1):
        f -= r2[j-1] * np.cos(j * np.pi * ks)
    # Soft max
    f_clipped = np.clip(f, -1, 1)
    max_val = f.max()
    smax = max_val + np.log(np.sum(np.exp(alpha*(f_clipped - max_val)))) / alpha
    # Bound constraint
    x = np.linspace(0, 2, n_x)
    H = np.zeros(n_x)
    for j in range(1, m+1):
        a, b = coeffs[2*(j-1)], coeffs[2*(j-1)+1]
        H += a * np.cos(j * np.pi * x) + b * np.sin(j * np.pi * x)
    # Penalty: sum of violations of |H| > 0.5
    viol = np.maximum(0, np.abs(H) - 0.4999)
    pen = 100 * np.sum(viol**2)
    return smax + pen

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h
