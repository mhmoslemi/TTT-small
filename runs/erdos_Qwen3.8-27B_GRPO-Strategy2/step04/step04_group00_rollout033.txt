    L = np.arange(-(n-1), n)  # all lags

    valid_im = (i - L >= 0) & (i - L < n)
    # For index j:
    valid_jp = (j + L >= 0) & (j + L < n)
    valid_jm = (j - L >= 0) & (j - L < n)

    g[valid_jp] -= (1.0 - h[j + L[valid_jp]])
    g[valid_jm] += h[j - L[valid_jm]]
    return g

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _pair_grad_vec(h, i, j, n):
    L = np.arange(-(n-1), n)
    valid_ip = (i + L >= 0) & (i + L < n)
    valid_im = (i - L >= 0) & (i - L < n)
    valid_jp = (j + L >= 0) & (j + L < n)
    valid_jm = (j - L >= 0) & (j - L < n)
    g = np.zeros(2*n-1)
    g[valid_ip] += (1.0 - h[i + L[valid_ip]])
    g[valid_im] -= h[i - L[valid_im]]
    g[valid_jp] -= (1.0 - h[j + L[valid_jp]])
    g[valid_jm] += h[j - L[valid_jm]]
    return g

def _exact_pair_d(c, g, h, i, j):
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-14:
        return 0.0
    f0 = np.max(c)
    for _ in range(100):
        mid = (d_lo + d_hi) * 0.5
        vals = c + g * mid
        mv = np.max(vals)
        if mv > f0 + 1e-13:
            # Got worse, but that's ok for binary search
            pass
        active = np.where(vals >= mv - 1e-12)[0]
        ga = g[active]
        if np.min(ga) > 1e-10:
            d_hi = mid
        elif np.max(ga) < -1e-10:
            d_lo = mid
        else:
            break
    best_d = (d_lo + d_hi) * 0.5
    if np.max(c + g * best_d) < f0 - 1e-14:
        return best_d
    return 0.0

    pairs = list(range(n * (n-1) // 2)) if max_pairs is None else list(range(max_pairs))
    # Actually let's use random ordering
    idx = np.arange(n)
    for _ in range(n):  # n passes
        if not pairs and max_pairs is not None:
            break
        # For efficiency, iterate all pairs
        break  # Actually just do one pass per call

        for j in range(i+1, n):

            d = _exact_pair_d(c, g, h, i, j)

                h[j] -= d
                h = np.clip(h, 0, 1)
                c = _corr(h)
                new_c5 = float(np.max(c)) * dx
                if new_c5 < c5:
                    c5 = new_c5
                    improved += 1
                else:
                    h[i] -= d
                    h[j] += d
    return h, c5, improved

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _pair_grad_vec(h, i, j, n):
    L = np.arange(-(n - 1), n)
    vip = (i + L >= 0) & (i + L < n)
    vim = (i - L >= 0) & (i - L < n)
    vjp = (j + L >= 0) & (j + L < n)
    vjm = (j - L >= 0) & (j - L < n)
    g = np.zeros(2 * n - 1)
    g[vip] += (1.0 - h[i + L[vip]])
    g[vim] -= h[i - L[vim]]
    g[vjp] -= (1.0 - h[j + L[vjp]])
    g[vjm] += h[j - L[vjm]]
    return g

def _exact_pair_d(c, g, h, i, j):
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-14:
        return 0.0
    f0 = np.max(c)
    for _ in range(100):
        mid = (d_lo + d_hi) * 0.5
        vals = c + g * mid
        mv = np.max(vals)
        active = np.where(vals >= mv - 1e-12)[0]
        ga = g[active]
        if np.min(ga) > 1e-10:
            d_hi = mid
        elif np.max(ga) < -1e-10:
            d_lo = mid
        else:
            break
    best_d = (d_lo + d_hi) * 0.5
    if np.max(c + g * best_d) < f0 - 1e-14:
        return best_d
    return 0.0

def _cd_sweeps(h, n, dx, n_sweeps, time_limit, time_fn):
    h = h.copy()
    c = _corr(h)
    c5 = float(np.max(c)) * dx
    for _sw in range(n_sweeps):
        if time_fn() > time_limit:
            break
        improved = 0
        for i in range(n):
            if time_fn() > time_limit:
                break
            for j in range(i + 1, n):
                g = _pair_grad_vec(h, i, j, n)
                d = _exact_pair_d(c, g, h, i, j)
                if abs(d) > 1e-13:
                    h[i] += d
                    h[j] -= d
                    if h[i] < -1e-12 or h[i] > 1 + 1e-12 or h[j] < -1e-12 or h[j] > 1 + 1e-12:
                        h[i] -= d
                        h[j] += d
                        continue
                    c += g * d
                    new_c5 = float(np.max(c)) * dx
                    if new_c5 < c5:
                        c5 = new_c5
                        improved += 1
                    else:
                        h[i] -= d
                        h[j] += d
                        c -= g * d
        if improved == 0:
            break
    h = np.clip(h, 0, 1)
    return h, c5

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _polish(h, n, dx, rng, time_limit, time_fn):
    h = h.copy()
    c5 = _c5(h, dx)
    for _ in range(50000):
        if time_fn() > time_limit:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.003, 0.003)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5:
                c5 = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    # Phase 1: Exact pair CD on original n
    h1, c5_1 = _cd_sweeps(h0, n, dx, 100, time_left() - 120, lambda: elapsed() + t0 - t0)
    h1, c5_1 = _cd_sweeps(h1, n, dx, 60, budget_s - 120, elapsed)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Soft-max polish on n
    if time_left() > 80:
        bounds = [(0.0, 1.0)] * n
        for a in [50000, 200000, 1000000]:
            if time_left() < 30:
                break
            try:
                res = minimize(_softobj, best_h.copy(), jac=lambda x: _softgrad(x, a, 0.0),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-20})
                hc = _repair(res.x, n)
                cv = _c5(hc, dx)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass
        # CD after soft-max
        h2, c5_2 = _cd_sweeps(best_h, n, dx, 30, budget_s - 60, elapsed)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: Try n=100
    if time_left() > 60:
        n2 = 100
        dx2 = 2.0 / n2
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        h2r, c5_r = _cd_sweeps(h2, n2, dx2, 80, budget_s - 40, elapsed)
        if c5_r < best_c5:
            best_h, best_c5, n, dx = h2r, c5_r, n2, dx2

    # Phase 4: Try n=200
    if time_left() > 50:
        n3 = 200
        dx3 = 2.0 / n3
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = _repair(h3, n3)
        h3r, c5_3 = _cd_sweeps(h3, n3, dx3, 40, budget_s - 30, elapsed)
        if c5_3 < best_c5:
            best_h, best_c5, n, dx = h3r, c5_3, n3, dx3

    # Phase 5: Final polish
    if time_left() > 10:
        hp, cp = _polish(best_h, n, dx, rng, budget_s - 3, elapsed)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
