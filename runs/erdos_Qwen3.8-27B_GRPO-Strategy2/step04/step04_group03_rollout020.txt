import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _smooth_max(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth_max(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    # For each lag k (0 to 2n-2), lag_s = k - (n-1)
    # c[k] = sum over valid i of h[i]*(1-h[i+lag_s])
    # dc[k]/dh[j] = (1-h[j+lag_s]) - h[j-lag_s] (where valid)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        # Contribution to g[j] from this lag:
        # For j where j+lag is valid: add wk * (1 - h[j+lag])
        # For j where j-lag is valid: subtract wk * h[j-lag]
        if lag >= 0:
            # j+lag valid when j < n-lag, i.e., j in [0, n-lag-1]
            j1 = slice(0, n - lag)
            g[j1] += wk * (1.0 - h[lag:n])
            # j-lag valid when j >= lag, i.e., j in [lag, n-1]
            j2 = slice(lag, n)
            g[j2] -= wk * h[0:n - lag]
        else:

            ll = -lag
            # j+lag valid when j >= ll, i.e., j in [ll, n-1]

    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    f_prev = _smooth_max(h, alpha)
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = _grad_smooth_max(h, alpha)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = _repair(h_new, n)
        f_new = _smooth_max(h_new, alpha)
        if f_new < f_prev:
            h = h_new
            f_prev = f_new
            c_current = _eval(h)
            if c_current < best_c:
                best_c = c_current
                best_h = h.copy()
        else:
            v *= 0.5
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _smooth_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + (np.log(np.sum(np.exp(s))) + s.max()) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            nl = n - lag
            if nl > 0:
                g[:nl] += wk * (1.0 - x[lag:])
                g[lag:] -= wk * x[:nl]
        else:
            t = -lag
            nt = n - t
            if nt > 0:
                g[t:] += wk * (1.0 - x[:nt])
                g[:nt] -= wk * x[t:]
    return g

def _pg_grad_fn(x, alpha):
    g = _smooth_grad(x, alpha)
    return g - g.mean()

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _proj_descent(h, n, alpha, t_end):
    h = h.copy()
    lr = 5.0 / (alpha + 1.0)
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c = _eval(h)
    stagg = 0
    f_prev = _smooth_obj(h, alpha)
    for it in range(500000):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        g -= g.mean()
        v = beta * v + lr * g
        h_new = _repair(h - v, n)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f_prev - 1e-15:
            h = h_new
            f_prev = f_new
            stagg = 0
            lr = min(lr * 1.005, 50.0 / (alpha + 1.0))
            c_curr = _eval(h)
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
        else:
            stagg += 1
            lr *= 0.6
            v *= 0.5
            if stagg > 30 or lr < 1e-15:
                break
    return best_h, best_c

def _slsqp(h, n, alpha, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_pg_grad_fn, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)
