import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except:
        return h, compute_c5(h, dx)

def sum_con(x, target):
    return np.sum(x) - target

def slsqp_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_opt(h, n, alpha, target, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(slsqp_obj, h, args=(alpha, target), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con, 'args': (target,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except:
        return h, compute_c5(h, dx)

def simulated_annealing(h, n, dx, rng, deadline):
    T = 0.005  # initial temperature
    T_min = 1e-8
    cool = 0.999995
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    curr_c5 = best_c5
    target = n / 2.0
    stagnant = 0
    iters = 0
    
    while time.time() < deadline:
        iters += 1
        # Choose move type
        r = rng.random()
        if r < 0.6:
            # Single pair swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.02, 0.02)
            hi, hj = h[i] + d, h[j] - d
            if hi < -1e-9 or hi > 1.0 + 1e-9 or hj < -1e-9 or hj > 1.0 + 1e-9:
                continue
            h[i], h[j] = hi, hj
            new_c5 = compute_c5(h, dx)
            delta = new_c5 - curr_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                h[i] -= d
                h[j] += d
        elif r < 0.85:

            bs = rng.randint(2, 6)
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(-0.01, 0.01)
            ok = True
            for di in range(bs):
                if h[i+di] + eps < -1e-9 or h[i+di] + eps > 1.0 + 1e-9:
                    ok = False; break
                if h[j+di] - eps < -1e-9 or h[j+di] - eps > 1.0 + 1e-9:
                    ok = False; break
            if not ok:

            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            new_c5 = compute_c5(h, dx)
            delta = new_c5 - curr_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                h[i:i+bs] -= eps
                h[j:j+bs] += eps
        else:
            # Multi-pair
            for _ in range(3):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.uniform(-0.008, 0.008)
                hi, hj = h[i] + d, h[j] - d
                if hi < -1e-9 or hi > 1.0 + 1e-9 or hj < -1e-9 or hj > 1.0 + 1e-9:
                    h = repair(best_h, n)
                    curr_c5 = best_c5
                    break
                h[i], h[j] = hi, hj
            else:
                new_c5 = compute_c5(h, dx)
                delta = new_c5 - curr_c5
                if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):

                    if new_c5 < best_c5:

                        best_h = h.copy()
                        stagnant = 0
                    else:
                        stagnant += 1
                else:
                    h = best_h.copy()
                    curr_c5 = best_c5

        # Reheat if stagnant
        if stagnant > 20000:
            T = max(T, 0.002)
            stagnant = 0
            h = best_h.copy()
            curr_c5 = best_c5

The overlap at shift m is: O(m) = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])

We want to minimize max_m O(m).

Note that O(m) + O_rev(m) where O_rev(m) = sum h[i]*(1-h[i+m]) are related but not identical (different summation ranges).

Actually, for the "full" correlation:
- Positive lags: c[n-1+m] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
- Negative lags: c[n-1-m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m]) = sum_{j=0}^{n-1-m} h[j+m]*(1-h[j])

Wait, these are the same! c[n-1+m] = c[n-1-m]? No, let me recheck.

np.correlate(a, v, 'full')[k] = sum_{n} a[n+k]*v[n] (where the sum is over valid n)

For a=h, v=1-h, length n each:
- k=0: sum_{n=0}^{n-1} h[n]*v[n] = sum h[n]*(1-h[n])
- k=1: sum_{n=0}^{n-2} h[n+1]*v[n] = sum h[n+1]*(1-h[n])
- ...
- k=n-1: h[n-1]*v[0] = h[n-1]*(1-h[0])
- k=-1: sum_{n=1}^{n-1} h[n]*v[n-1] = sum h[n]*(1-h[n-1])
- ...

So c[n-1+k] for k=0,...,n-1: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
And c[n-1-k] for k=0,...,n-1: sum_{i=k}^{n-1} h[i]*(1-h[i-k])

These are NOT the same in general. c[n-1+k] uses h[i+k] and (1-h[i]) while c[n-1-k] uses h[i] and (1-h[i-k]).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def lse_obj(x, alpha, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def lbfgs_opt(h, n, alpha, target, pen, maxiter):
    dx = 2.0 / n
    try:
        res = minimize(lse_obj, h, args=(alpha, target, pen),
                      method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)

def sum_con(x, target):
    return np.sum(x) - target

def slsqp_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def slsqp_opt(h, n, alpha, target, maxiter):

                      options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = repair(res.x, n)
        return hc, compute_c5(hc, dx)
    except Exception:
        return h, compute_c5(h, dx)
