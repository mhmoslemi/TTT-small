def corr_and_grad(h, top_k=1):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    # Get top_k lags
    idx = np.argsort(c)[::-1][:top_k]
    g = np.zeros(n)
    for m in idx:
        lag = m - (n-1)  # range: -(n-1) to (n-1)
        if lag == 0:
            # c = sum h[i]*(1-h[i])
            g += 1.0 - 2.0*h
        elif lag > 0:
            # c = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # d/dh[j]: 
            #   j appears as h[i+lag] when i=j-lag (valid: 0<=j-lag<=n-1-lag, i.e., lag<=j<=n-1)
            #   j appears as (1-h[i]) when i=j (valid: 0<=j<=n-1-lag)
            g[lag:] += (1.0 - h[:-lag])  # term where j appears as h[i+lag], paired with h[i]=h[j-lag]
            g[:n-lag] -= h[lag:]  # term where j appears as h[i]=h[j], paired with h[i+lag]=h[j+lag]
        else:
            l = -lag
            # c = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l])
            # d/dh[j]:
            #   j appears as h[i] when i=j (valid: 0<=j<=n-1-l)
            #   j appears as (1-h[i+l]) when i+l=j, i=j-l (valid: 0<=j-l<=n-1-l, i.e., l<=j<=n-1)
            g[:n-l] += (1.0 - h[l:])  # term where j appears as h[i], paired with h[i+l]=h[j+l]
            g[l:] -= h[:n-l]  # term where j appears as h[i+l]=h[j], paired with h[i]=h[j-l]
    return c, g

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def smooth_max_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:-lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            l = -lag
            g[:n - l] += wk * (1.0 - x[l:])
            g[l:] -= wk * x[:n - l]
    return g * 2.0 / n

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

    h = h.copy()
    c5_cur = eval_c5(h)
    step_init = 0.02
    step = step_init
    cooldown = 0
    alpha_sg = 50.0
    for it in range(50000):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for k in idx:
            lag = k - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += (1.0 - h[:-lag])
                g[:n - lag] -= h[lag:]
            else:
                l = -lag
                g[:n - l] += (1.0 - h[l:])
                g[l:] -= h[:n - l]
        g *= 2.0 / n / top_k
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        h_new = h - step * g / gn
        h_new = repair(h_new, n)
        c5_new = eval_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            h = h_new
            c5_cur = c5_new
            step = min(step * 1.02, 0.05)
            cooldown = 0
        else:
            step *= 0.9
            cooldown += 1
            if cooldown > 50:
                h = repair(h + 0.01 * np.random.randn(n), n)
                cooldown = 0
                step = step_init
    return h, c5_cur

def local_search(h, n, rng, t_end):
    h = h.copy()
    c5c = eval_c5(h)
    scale = 0.005
    no_imp = 0
    while no_imp < 2000 and time.time() < t_end:

        if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            continue
        h[i], h[j] = ni, nj
        c5n = eval_c5(h)
        if c5n < c5c - 1e-15:

            no_imp = 0

            no_imp += 1
            if no_imp > 200:
                scale *= 0.8
                no_imp = 0
                if scale < 1e-6:
                    break
    return h, c5c
