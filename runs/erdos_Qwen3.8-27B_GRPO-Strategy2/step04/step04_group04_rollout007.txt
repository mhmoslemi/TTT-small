            else:

    c = _corr_full(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 500)
    exps = np.exp(s)
    w = exps / exps.sum()

    grad = np.zeros(n)

        else:

        d = 0.003 * (2.0 * rng.random() - 1.0)
        h[i] += d; h[j] -= d
        if -1e-9 <= h[i] <= 1.0 + 1e-9 and -1e-9 <= h[j] <= 1.0 + 1e-9:
            c = _c5_val(h, dx)
            if c < best:
                best = c
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n0 = len(h0)
    dx0 = 2.0 / n0
    best_h, best_c5 = h0.copy(), _c5_val(h0, dx0)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: PGD with momentum at n0 (100s)
    t_end = min(t0 + 100, t0 + budget_s - 800)
    for alpha in [200, 1000, 5000]:
        if time.time() > t_end:
            break
        h_pgd, _ = _pgd_momentum(best_h, alpha, 2000, 0.008, t_end)
        c = _c5_val(h_pgd, dx0)
        if c < best_c5:
            best_h, best_c5 = h_pgd, c

    # Phase 2: SLSQP polish (60s)
    t_end = min(t0 + 160, t0 + budget_s - 740)
    for a in [5000, 20000]:
        if time.time() > t_end:
            break
        best_h, best_c5 = _slsqp_run(best_h, dx0, a, 300)

    # Phase 3: Informed SA (250s)
    t_end = min(t0 + 410, t0 + budget_s - 500)
    best_h, best_c5 = _informed_sa(best_h, rng, t_end, 50000, 0.005)

    # Phase 4: Multi-start with structured perturbations (200s)
    t_end = min(t0 + 610, t0 + budget_s - 300)
    for r in range(6):
        if time.time() > t_end or rem() < 300:
            break
        hp = best_h.copy()
        # Sine perturbation
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        hp += 0.06 * np.sin(2 * np.pi * freq * np.arange(n0) / n0 + phase)
        hp = _repair(hp, n0 / 2.0)
        hp, c = _pgd_momentum(hp, 3000, 500, 0.005, t_end)
        hp, c = _slsqp_run(hp, dx0, 10000, 200)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 5: Interpolate to 2x and optimize (250s)
    n1 = 2 * n0
    dx1 = 2.0 / n1
    t_end = min(t0 + 860, t0 + budget_s - 150)
    h1 = _repair(_interp1d(best_h, n1), n1 / 2.0)
    c1 = _c5_val(h1, dx1)
    if c1 < best_c5:
        best_h, best_c5, n_b = h1, c1, n1
    else:
        n_b = n0
    dxb = 2.0 / n_b
    for alpha in [500, 3000]:
        if time.time() > t_end:
            break
        best_h, _ = _pgd_momentum(best_h, alpha, 1500, 0.004, t_end)
        c = _c5_val(best_h, dxb)
        if c < best_c5:
            best_c5 = c

    # Phase 6: Final SA + local search (remaining)
    t_end = min(t0 + budget_s - 60, t0 + 1000)
    best_h, best_c5 = _informed_sa(best_h, rng, t_end, 40000, 0.003)
    t_end = t0 + budget_s - 10
    best_h, best_c5 = _local_search_pair(best_h, dxb, rng, t_end, 200000)

    best_h = _repair(best_h, n_b / 2.0)
    best_c5 = _c5_val(best_h, dxb)
    return (best_h, best_c5, n_b)
