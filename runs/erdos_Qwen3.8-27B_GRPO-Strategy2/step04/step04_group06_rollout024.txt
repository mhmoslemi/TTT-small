def _c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def _c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, n):

        mid = (lo + hi) * 0.5

        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)

def _exact_subgrad(h, n):
    """Compute exact subgradient of C5 at h."""
    c = _c5_full(h, n)  # shape (2n-1,)
    cmax = np.max(c)
    # Find all indices where c is close to max
    tol = 1e-10 * cmax + 1e-15
    mask = c >= cmax - tol
    n_active = np.sum(mask)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if not mask[k]:
            continue
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            # c_k = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # d/dh[i] = (1-h[i+L]) for i < n-L
            # d/dh[j] = -h[j-L] for j >= L
            g[:n-L] += (1.0 - h[L:])

    hi, hj = h[i], h[j]

    hi_d = min(1.0 - hi, hj)  # h[i]+d <= 1, h[j]-d <= 1
    if hi_d - lo_d < 1e-12:
        return 0.0
    # Ternary search on [lo_d, hi_d]
    a, b = lo_d, hi_d
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h[i], h[j] = hi + m1, hj - m1
        f1 = _c5(h, n)
        h[i], h[j] = hi + m2, hj - m2
        f2 = _c5(h, n)

        else:

    else:

        best_c5, best_h = c5_w, h_w

    # Phase 3: PGD high alpha (42-52%)
    p3_end = t0 + budget_s * 0.52
    for alpha, lr_init in [(500000, 0.008), (1500000, 0.005)]:
        if elapsed() > p3_end:
            break
        h_w, c5_w = _pgd(best_h, best_n, alpha, p3_end, lr_init)
        if c5_w < best_c5:
            best_c5, best_h = c5_w, h_w

    # Phase 4: Subgradient again (52-62%)
    p4_end = t0 + budget_s * 0.62
    h_w, c5_w = _sgd(best_h, best_n, p4_end)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w

    # Phase 5: Gradient-guided pair CD (62-82%)
    p5_end = t0 + budget_s * 0.82
    h_w, c5_w = _pair_cd_grad(best_h, best_n, p5_end, rng)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w

    # Phase 6: Systematic local pair CD (82-92%)
    p6_end = t0 + budget_s * 0.92
    h_w, c5_w = _pair_cd_syst(best_h, best_n, p6_end)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w

    # Phase 7: Multi-resolution try (92-97%)
    p7_end = t0 + budget_s * 0.97
    for n_try in [200, 180]:
        if elapsed() > p7_end:
            break
        h_new = _interp_h(best_h, best_n, n_try)
        h_new = _project(h_new, n_try)
        h_w, c5_w = _pgd(h_new, n_try, 100000, min(p7_end, t0 + budget_s * 0.02), 0.02)
        if c5_w < best_c5 - 1e-6:
            h_w, c5_w = _sgd(h_w, n_try, p7_end)
            if c5_w < best_c5 - 1e-6:
                h_w, c5_w = _pair_cd_grad(h_w, n_try, p7_end, rng)
                if c5_w < best_c5:
                    best_c5, best_h, best_n = c5_w, h_w, n_try

    # Phase 8: Final squeeze (97-99.5%)
    p8_end = t0 + budget_s * 0.995
    h_w, c5_w = _pair_cd_grad(best_h, best_n, p8_end, rng)
    if c5_w < best_c5:
        best_c5, best_h = c5_w, h_w

    # Final repair and evaluation
    best_h = _project(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
