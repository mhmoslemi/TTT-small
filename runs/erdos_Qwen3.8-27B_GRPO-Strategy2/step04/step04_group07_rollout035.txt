from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    # Iterative repair
    for _ in range(10000):
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            # Need to add d total; add to smallest values first
            order = np.argsort(h)
            for idx in order:
                if d <= 0:
                    break
                add = min(d, 1.0 - h[idx])
                h[idx] += add
                d -= add
        else:
            order = np.argsort(-h)
            for idx in order:
                if d >= 0:
                    break
                sub = min(-d, h[idx])
                h[idx] -= sub
                d += sub
        s = h.sum()
    return h

def optimize_pair_1d(h, i, j, n):
    """Optimize h[i] and h[j] given h[i]+h[j]=s, using golden section on t=h[i]."""
    s = h[i] + h[j]
    lo, hi = max(0.0, s - 1.0), min(1.0, s)
    if hi - lo < 1e-14:
        return  # no freedom
    
    dx = 2.0 / n
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    def c5_at(t):
        h[i] = t
        h[j] = s - t
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx
    
    # Golden section search
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    fc = c5_at(c)
    fd = c5_at(d)
    
    for _ in range(60):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - inv_phi * (hi - lo)
            fc = c5_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + inv_phi * (hi - lo)
            fd = c5_at(d)
    
    t_opt = (lo + hi) / 2.0
    h[i] = t_opt
    h[j] = s - t_opt
    return
