```python
import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return np.max(compute_c5_vals(h))

def smooth_max(h, eps=0.003):
    vals = compute_c5_vals(h)
    m = np.max(vals)
    return m + eps * np.log(np.sum(np.exp((vals - m) / eps)))

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Binary search on threshold for water-filling

        return h
    h2 = h + diff / n

        if not np.any(mask):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return np.max(compute_c5_vals(h))

def smooth_max_c5(h, eps):
    vals = compute_c5_vals(h)
    m = np.max(vals)
    return m + eps * np.log(np.sum(np.exp((vals - m) / eps)))

def grad_smooth_max_c5(h, eps):
    n = len(h)
    dx = 2.0 / n
    vals = compute_c5_vals(h)
    shifted = vals - np.max(vals)
    w = np.exp(shifted / eps)
    wsum = np.sum(w)
    w /= wsum
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        m = k + (n-1)
        wk = w[m] * dx
        if wk < 1e-15:
            continue
        for j in range(n):
            val = 0.0
            jm = j - k
            jp = j + k
            if 0 <= jm < n:
                val += 1.0 - h[jm]
            if 0 <= jp < n:
                val -= h[jp]
            grad[j] += wk * val
    return grad

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(free)
        if cnt == 0:
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def gd_solve(h0, eps, max_iter, lr0):
    h = h0.copy()
    lr = lr0
    best_h = h.copy()
    best_val = compute_c5(h)
    prev_f = smooth_max_c5(h, eps)
    patience = 0
    
    for it in range(max_iter):
        g = grad_smooth_max_c5(h, eps)
        h_new = project_feasible(h - lr * g)
        f_new = smooth_max_c5(h_new, eps)
        
        if f_new < prev_f - 1e-15:
            h = h_new
            prev_f = f_new
            lr = min(lr * 1.02, 0.5)
            patience = 0
            c = compute_c5(h)
            if c < best_val:
                best_val = c
                best_h = h.copy()
        else:
            lr *= 0.7
            patience += 1
            if patience > 200 or lr < 1e-12:
                break
    return best_h, best_val
```

```python
def grad_smooth_max_c5_fast(h, eps):
    n = len(h)
    dx = 2.0 / n
    vals = compute_c5_vals(h)
    shifted = vals - np.max(vals)
    w = np.exp(shifted / eps)
    w /= np.sum(w)
    
    # For each shift k, the gradient contribution is:
    # grad[j] += w[k+n-1]*dx * [(1-h[j-k]) if 0<=j-k<n - h[j+k] if 0<=j+k<n]
    #
    # This can be written as:
    # grad = dx * [corr(w_shifted, 1-h) - corr(w_shifted, h)]  (roughly)
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        m = k + n - 1
        wk = w[m] * dx
        if wk < 1e-15:
            continue
        # (1-h)[j-k] for j in [k, n-1] (when k>0) or j in [0, n-1-k] (when k<0)
        # h[j+k] for j in [0, n-1-k] (when k>0) or j in [k, n-1] (when k<0)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n-k])  # (1-h)[j-k] for j=k..n-1
            grad[:n-k] -= wk * h[k:]  # -h[j+k] for j=0..n-1-k
        else:
            kj = -k
            grad[:n-kj] += wk * (1.0 - h[kj:])  # (1-h)[j-k] = (1-h)[j+kj] for j=0..n-1-kj
            grad[kj:] -= wk * h[:n-kj]  # -h[j+k] = -h[j-kj] for j=kj..n-1
    return grad
```

This is O(n) per shift, so O(n^2) total, but with numpy vectorization it's much faster than the Python loop.

        f_new = smooth_max_c5(h_new, eps)
        if f_new < prev_f:
            h = h_new
            prev_f = f_new
            lr = min(lr * 1.03, 0.8)
            stall = 0
            c = compute_c5(h)
            if c < best_val:
                best_val = c
                best_h = h.copy()
        else:
            lr *= 0.6
            stall += 1
            if stall > 300 or lr < 1e-13:
                break
    return best_h, best_val

def lbfgs_objective(x, eps):
    return smooth_max_c5(project_feasible(x), eps)
