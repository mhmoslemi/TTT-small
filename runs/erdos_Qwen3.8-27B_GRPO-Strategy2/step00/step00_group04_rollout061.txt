import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def c5_gradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr)
    # k_max is the index in the full correlation (0 to 2n-2)
    # The lag p = k_max - (n-1)
    p = k_max - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        # (1-h[i-p]) * I(0<=i-p<n) 
        if 0 <= i - p and i - p < n:
            grad[i] += (1 - h[i - p])
        # - h[i+p] * I(0<=i+p<n)
        if 0 <= i + p and i + p < n:
            grad[i] -= h[i + p]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def c5_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Handle ties by averaging gradients of all max terms
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for k_max in active:
        p = k_max - (n - 1)
        n_lo = max(0, -p)
        n_hi = min(n - 1, n - 1 - p)
        for n_idx in range(n_lo, n_hi + 1):
            ip = n_idx + p
            grad[ip] += (1 - h[n_idx])
            grad[n_idx] -= h[ip]
    grad /= len(active)
    return grad * dx

def slsqp_optimize(h, n, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2,
             'jac': lambda x: np.ones(n)}]
    res = minimize(compute_c5, h, jac=c5_subgradient, method='SLSQP',
                   bounds=bounds, constraints=cons, args=(n,),
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    # Repair
    x = res.x
    x = np.clip(x, 0, 1)
    diff = n / 2 - np.sum(x)
    x += diff / n
    x = np.clip(x, 0, 1)
    return x

    while np.sum(h_bin) > n / 2:
        idx = np.random.randint(n)
        if h_bin[idx] == 1:
            h_bin[idx] = 0
    while np.sum(h_bin) < n / 2:
        idx = np.random.randint(n)
        if h_bin[idx] == 0:

    best_val = compute_c5(h_bin, n)

    corr = np.correlate(x, 1 - x, mode="full")

        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            stall += 1
        if stall > 5000:
            break
    return best_h


def upscale(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = n_new / 2.0 - np.sum(h_new)
    h_new += diff / n_new
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(42, 0.5)

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP at original resolution
    rem = budget_s - (time.time() - t_start)
    if rem > 40:
        try:
            h1 = slsqp_opt(best_h, n, 500, 400)
            c5_1 = compute_c5(h1, n)
            if c5_1 < best_c5:
                best_h, best_c5 = h1, c5_1
        except Exception:
            pass
        # Second SLSQP pass with higher alpha
        rem = budget_s - (time.time() - t_start)
        if rem > 30:
            try:
                h1b = slsqp_opt(best_h, n, 2000, 300)
                c5_1b = compute_c5(h1b, n)
                if c5_1b < best_c5:
                    best_h, best_c5 = h1b, c5_1b
            except Exception:
                pass

    # Phase 2: Binary local search
    rem = budget_s - (time.time() - t_start)
    if rem > 40:
        h2 = binary_search(best_h, n, seed + 1, rem * 0.4)
        c5_2 = compute_c5(h2, n)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: SLSQP from binary best
    rem = budget_s - (time.time() - t_start)
    if rem > 30:
        try:
            h3 = slsqp_opt(best_h, n, 1000, 200)
            c5_3 = compute_c5(h3, n)
            if c5_3 < best_c5:
                best_h, best_c5 = h3, c5_3
        except Exception:
            pass

    # Phase 4: More binary search with different seed
    rem = budget_s - (time.time() - t_start)
    if rem > 30:
        h4 = binary_search(best_h, n, seed + 999, rem * 0.45)
        c5_4 = compute_c5(h4, n)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Try higher resolution
    rem = budget_s - (time.time() - t_start)
    for scale in [2, 4]:
        n_new = n * scale
        rem = budget_s - (time.time() - t_start)
        if rem < 80:
            break
        h_new = upscale(best_h, n, n_new)
        if rem > 50:
            try:
                h_new = slsqp_opt(h_new, n_new, 500, 150)
                c5_new = compute_c5(h_new, n_new)
                if c5_new < best_c5:
                    best_h, best_c5, n = h_new, c5_new, n_new
            except Exception:
                pass
        rem = budget_s - (time.time() - t_start)
        if rem > 25:
            h_b = binary_search(best_h, n, seed + 500 + n_new, rem * 0.5)
            c5_b = compute_c5(h_b, n)
            if c5_b < best_c5:
                best_h, best_c5 = h_b, c5_b

    # Phase 6: Final SLSQP polish
    rem = budget_s - (time.time() - t_start)
    if rem > 20:
        try:
            h_final = slsqp_opt(best_h, n, 3000, 100)
            c5_final = compute_c5(h_final, n)
            if c5_final < best_c5:
                best_h, best_c5 = h_final, c5_final
        except Exception:
            pass

    # Final cleanup and validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_n = len(best_h)
    target_sum = final_n / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        # Iteratively fix
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            best_h += diff / final_n
            best_h = np.clip(best_h, 0.0, 1.0)
            # Redistribute remaining error
            diff = target_sum - np.sum(best_h)
            free_idx = np.where((best_h > 1e-6) & (best_h < 1 - 1e-6))[0]
            if len(free_idx) > 0:
                best_h[free_idx] += diff / len(free_idx)
                best_h = np.clip(best_h, 0.0, 1.0)
                diff = target_sum - np.sum(best_h)

    final_c5 = compute_c5(best_h, final_n)
    return best_h.tolist(), final_c5, final_n
