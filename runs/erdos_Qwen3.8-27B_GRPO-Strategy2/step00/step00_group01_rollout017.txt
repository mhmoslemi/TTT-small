import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
        else:
            mask = h > 1e-7
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return make_feasible(h.copy(), n_new)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, float(n_old - 1), n_new)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    return make_feasible(h_new, n_new)

def erdos_obj(x, n):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(len(x))

def solve_slsqp(h_start, n, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(
            erdos_obj, h_start, args=(n,),
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        h_opt = make_feasible(res.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return None, np.inf

    if h_ref is not None:

        for sigma in [0.03, 0.08, 0.15, 0.3]:

    h_rand = rng.uniform(0.0, 1.0, n)
    starts.append(make_feasible(h_rand, n))

    starts.append(h_sym)

        for u in unique:
            if np.linalg.norm(s - u) < 1e-6:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique
