import time

def project_onto_constraint(z, target_sum):
    n = len(z)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    return np.max(corr)

def objective_and_grad(h_flat, n):
    h = h_flat.reshape(-1)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    max_idx = np.argmax(corr)
    val = corr[max_idx] * dx
    
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j] = (1-h[j-k]) if k<=j<=n-1, minus h[j+k] if 0<=j<=n-1-k
        mask_plus = np.zeros(n, dtype=bool)
        mask_plus[k:] = True
        mask_minus = np.zeros(n, dtype=bool)
        mask_minus[:n-k] = True
        grad[mask_plus] += (1.0 - h[mask_plus - k]) * dx  # h[j-k] where j in [k, n-1]
        grad[mask_minus] -= h[mask_minus + k] * dx  # h[j+k] where j in [0, n-1-k]
    else:
        kp = -k  # positive
        # sum_{i=kp}^{n-1} h[i-kp]*(1-h[i])
        # d/dh[j] = (1-h[j+kp]) if 0<=j<=n-1-kp, minus h[j-kp] if kp<=j<=n-1
        mask_plus = np.zeros(n, dtype=bool)
        mask_plus[:n-kp] = True
        mask_minus = np.zeros(n, dtype=bool)
        mask_minus[kp:] = True
        grad[mask_plus] += (1.0 - h[mask_plus + kp]) * dx
        grad[mask_minus] -= h[mask_minus - kp] * dx
    
    return val, grad

    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) / n_old  # in [0, 1)
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)

    h_new = project_onto_constraint(h_new, n_new / 2.0)
    return h_new

        {'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,

    ]

        objective_and_grad,
        h.copy(),
        args=(n,),
        method='SLSQP',
        jac=True,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-10}
    )
    # Project result to ensure feasibility
    x_opt = project_onto_constraint(result.x, n/2.0)
    return x_opt

def project_simplex(z, target):
    """Project z onto {x: sum(x)=target, 0<=x_i<=1}"""
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

    """Gradient of the argmax correlation term."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mi = int(np.argmax(corr))
    k = mi - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        p = np.arange(k, n)
        m = np.arange(0, n - k)
        g[p] += (1.0 - h[m]) * dx
        g[m] -= h[p] * dx
    else:

        m = np.arange(kp, n)
        g[p] += (1.0 - h[m]) * dx
        g[m] -= h[p] * dx

    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_simplex(np.interp(xn, xo, h_old), n_new * 0.5)

def slsqp_solve(h, n, max_it, tau):
    global _CONST_N
    _CONST_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_f, 'jac': eq_j}]
    res = minimize(
        smooth_f, h.copy(), args=(n, tau),
        method='SLSQP', jac=smooth_j,
        bounds=bnds, constraints=cons,
        options={'maxiter': max_it, 'ftol': 1e-12, 'disp': False}
    )
    return project_simplex(res.x, n * 0.5)

def pg_solve(h, n, deadline, max_iters=8000):
    best = h.copy()
    best_v = c5_val(h)
    alpha = 0.5
    stall = 0
    rng = np.random.RandomState(42)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = active_grad(h, n)
        h_new = project_simplex(h - alpha * g, n * 0.5)
        v = c5_val(h_new)
        if v < best_v:
            best_v = v
            best = h_new.copy()
            h = h_new
            stall = 0
            alpha = min(alpha * 1.02, 0.8)
        else:
            stall += 1
            if stall > 200:
                h = project_simplex(best + 0.08 * rng.randn(n), n * 0.5)
                alpha = 0.5
                stall = 0
            else:
                h = h_new
                alpha *= 0.92
        if alpha < 0.01:
            alpha = 0.4
            h = project_simplex(best + 0.05 * rng.randn(n), n * 0.5)
            stall = 0
        if it % 500 == 499:
            alpha = max(alpha, 0.3)
    return best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.94
    np.random.seed(seed)

    try:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_cur = np.full(45, 0.5)
    n0 = len(h_cur)
    h_cur = project_simplex(h_cur, n0 * 0.5)

    best_h = h_cur.copy()
    best_c5 = c5_val(h_cur)

    # SLSQP at increasing resolutions
    ns = sorted(set([max(n0, 50), 100, 200, 350]))
    for n in ns:
        if time.time() > deadline - 20:
            break
        h = interp_to(best_h, n) if n != len(best_h) else best_h.copy()
        rem = deadline - time.time()
        if rem < 15:
            break
        n_it = min(60, max(15, int(rem / 20)))
        h_opt = slsqp_solve(h, n, n_it, tau=12.0)
        c5 = c5_val(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Sharper max refinement
    if time.time() < deadline - 40:
        n_s = len(best_h)
        h2 = slsqp_solve(best_h, n_s, 30, tau=35.0)
        c5_2 = c5_val(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Projected gradient on current best
    if time.time() < deadline - 10:
        h_pg = pg_solve(best_h, len(best_h), deadline)
        c5_pg = c5_val(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Try higher resolution
    if time.time() < deadline - 15:
        n_hi = min(len(best_h) * 2, 800)
        if n_hi > len(best_h):
            h_hi = interp_to(best_h, n_hi)
            h_hi = pg_solve(h_hi, n_hi, deadline, max_iters=5000)
            c5_hi = c5_val(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()

    # Final SLSQP polish if time allows
    if time.time() < deadline - 5 and len(best_h) <= 400:
        h_pol = slsqp_solve(best_h, len(best_h), 20, tau=50.0)
        c5_pol = c5_val(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = project_simplex(best_h, len(best_h) * 0.5)
    best_c5 = c5_val(best_h)
    return best_h, best_c5, len(best_h)
