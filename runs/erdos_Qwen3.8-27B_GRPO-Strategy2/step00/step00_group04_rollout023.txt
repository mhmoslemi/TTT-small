        h = np.clip(h, 0.0, 1.0)

    if not stop():
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project(h_bin, best_n)
        h_b, c5_b = opt_pgd(h_bin, best_n, 3000, rng, stop, mu=0.008)
        c5_b = compute_c5(h_b, best_n)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b

    # Final validation
    best_h = project(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
