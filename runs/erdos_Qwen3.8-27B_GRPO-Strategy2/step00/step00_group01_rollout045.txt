import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max(), corr

def obj_direct(x, dx):
    c5, _ = compute_c5(np.clip(x, 0, 1), dx)
    return c5

def obj_smooth(x, dx, T):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    Tc = T * corr
    Tc = np.clip(Tc, -500, 500)
    m = Tc.max()
    if m > 500:
        return m
    return (m + np.log(np.sum(np.exp(Tc - m)))) / T

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()
