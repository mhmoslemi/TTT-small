import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.ndimage import gaussian_filter1d
import time
import math

def compute_c5(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def smooth_obj_grad(x, dx, target_sum, alpha, penalty_w):
    n = len(x)
    h = x
    corr = correlate(h, 1-h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    sw = w.sum()
    w = w / sw
    smax = m + np.log(sw) / alpha
    obj = smax * dx + penalty_w * (h.sum() - target_sum)**2
    
    grad_smax = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        s = k - (n-1)
        g = np.ones(n)
        if s >= 0:
            g[:n-s] -= h[s:]
            g[s:] -= h[:n-s]
        else:
            s2 = -s
            g[s2:] -= h[:n-s2]
            g[:n-s2] -= h[s2:]
        grad_smax += w[k] * g
    
    grad = grad_smax * dx + 2.0 * penalty_w * (h.sum() - target_sum)
    return obj, grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0 and abs(s - target_sum) > 1e-10:
        # Adjust: add (target_sum - s)/n to each, then clip and iterate
        for _ in range(20):
            h = h + (target_sum - h.sum()) / len(h)
            h = np.clip(h, 0.0, 1.0)
            if abs(h.sum() - target_sum) < 1e-12:
                break
    return h

def optimize_from(h0, dx, target_sum, budget_per=30.0, rng=None):
    n = len(h0)
    h0 = project_feasible(h0.copy(), target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    # Try different alpha values
    alphas = [20.0, 50.0, 100.0, 200.0]
    penalty_w = 500.0
    bounds = [(0.0, 1.0)] * n
    
    t0 = time.time()
    for a in alphas:
        if time.time() - t0 > budget_per:
            break
        res = minimize(
            lambda x: smooth_obj_grad(x, dx, target_sum, a, penalty_w)[0],
            h0.copy(),
            jac=lambda x: smooth_obj_grad(x, dx, target_sum, a, penalty_w)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 50000}
        )
        h_opt = project_feasible(res.x, target_sum)
        c5 = compute_c5(h_opt, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            h0 = h_opt.copy()
    
    return best_h, best_c5

    """Generate a random feasible h."""
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n/2.0)

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx


def smooth_obj(x, dx, target_sum, alpha, penalty_w):
    h = x
    corr = correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    sw = np.sum(np.exp(alpha * (corr - m)))
    smax = m + np.log(sw) / alpha
    return smax * dx + penalty_w * (h.sum() - target_sum) ** 2


def smooth_grad(x, dx, target_sum, alpha, penalty_w):
    n = len(x)
    h = x
    corr = correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    sw = w.sum()
    w = w / sw
    smax = m + np.log(sw) / alpha
    obj = smax * dx + penalty_w * (h.sum() - target_sum) ** 2

    grad_smax = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        g = np.ones(n)
        if s >= 0:
            if s > 0:
                g[:n - s] -= h[s:]
            g[s:] -= h[:n - s]
        else:
            s2 = -s
            g[s2:] -= h[:n - s2]
            g[:n - s2] -= h[s2:]
        grad_smax += w[k] * g

    grad = grad_smax * dx + 2.0 * penalty_w * (h.sum() - target_sum)
    return obj, grad


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def opt_one(h0, dx, target_sum, alpha, penalty_w, bounds, maxiter):
    n = len(h0)

    def f(x):
        return smooth_obj(x, dx, target_sum, alpha, penalty_w)

    def g(x):
        return smooth_grad(x, dx, target_sum, alpha, penalty_w)[1]

    res = minimize(f, h0.copy(), jac=g, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                           'gtol': 1e-10, 'maxfun': maxiter * 3})
    return res.x


def random_init(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n / 2.0)


def fourier_init(n, rng):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, 6):
        a = rng.uniform(-0.25, 0.25)
        b = rng.uniform(-0.25, 0.25)
        h += a * np.cos(2 * np.pi * k * x / 2.0) + b * np.sin(2 * np.pi * k * x / 2.0)
    return project_feasible(h, n / 2.0)


def block_init(n, rng):
    h = np.zeros(n)
    nb = rng.randint(3, 7)
    bs = max(1, n // nb)
    for kb in range(nb):
        lo = kb * bs
        hi = min(lo + bs, n)
        if rng.random() < 0.5:

        else:
            h[lo:hi] = rng.uniform(0.0, 0.4)
    return project_feasible(h, n / 2.0)
