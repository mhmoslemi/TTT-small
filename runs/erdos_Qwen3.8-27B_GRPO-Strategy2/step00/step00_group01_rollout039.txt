For the integral ∫₀² h(x)(1-h(x+k)) dx, with our discretization:

- x+k goes from k to 2+k

This is consistent with what np.correlate does - it only sums over valid indices.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def fix_sum(h, n):
    target = n / 2.0
    h += (target - h.sum()) / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)

def interp_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.astype(float).copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0.0, old_n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    fix_sum(h_new, new_n)
    return h_new

def corr_jac(h, k):
    n = len(h)
    lo = max(0, n - k - 1)
    hi = min(n - 1, 2 * n - k - 2)
    g = np.zeros(n)
    if lo > hi:
        return g
    m_lo = max(0, lo + k - n + 1)
    m_hi = min(n - 1, hi + k - n + 1)
    if m_lo <= m_hi:
        idx = np.arange(m_lo, m_hi + 1)
        g[idx] += 1.0 - h[idx - k + n - 1]
    idx2 = np.arange(lo, hi + 1)
    g[idx2] -= h[idx2 + k - n + 1]
    return g

class _CC:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx
    def fun(self, x):
        return x[self.n] - np.correlate(x[:self.n], 1.0 - x[:self.n], mode='full')[self.k] * self.dx
    def jac(self, x):
        gh = corr_jac(x[:self.n], self.k) * self.dx
        gt = np.zeros(self.n + 1)
        gt[:self.n] = -gh
        gt[self.n] = 1.0
        return gt

class _EC:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return x[:self.n].sum() - self.n / 2.0
    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def _obj(x):
    return x[-1]

def _obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def optimize_step(h_init, n, tlim, n_active=20):
    t0 = time.time()
    dx = 2.0 / n
    h = h_init.astype(float).copy()
    fix_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(40):
        if time.time() - t0 > tlim:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        max_c = c.max()
        active = np.argsort(c)[-n_active:]
        
        ec = _EC(n)
        cons = [{'type': 'eq', 'fun': ec.fun, 'jac': ec.jac}]
        for k in active:
            cc = _CC(int(k), n, dx)
            cons.append({'type': 'ineq', 'fun': cc.fun, 'jac': cc.jac})
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        x0 = np.concatenate([h, [max_c * 1.005 + 0.001]])
        
        try:
            res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-12})
        except Exception:
            break
        
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= max_c + 1e-9:
            break
    
    return best_h, best_c5

def random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0.0, 1.0, n)
    fix_sum(h, n)
    return h

def block_h(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def fix_sum(h, n):
    target = n / 2.0
    h += (target - h.sum()) / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)


def interp_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.astype(float).copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0.0, old_n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    fix_sum(h_new, new_n)
    return h_new


def corr_jac(h, m):
    n = len(h)
    j_lo = max(0, n - 1 - m)
    j_hi = min(n - 1, 2 * n - m - 2)
    g = np.zeros(n)
    if j_lo > j_hi:
        return g
    p1_lo = max(0, j_lo + m - n + 1)
    p1_hi = min(n - 1, j_hi + m - n + 1)
    if p1_lo <= p1_hi:
        idx = np.arange(p1_lo, p1_hi + 1)
        g[idx] += 1.0 - h[idx - m + n - 1]
    p2_lo = j_lo
    p2_hi = j_hi
    if p2_lo <= p2_hi:
        idx2 = np.arange(p2_lo, p2_hi + 1)
        g[idx2] -= h[idx2 + m - n + 1]
    return g


class _CC:
    def __init__(self, m, n, dx):
        self.m = m
        self.n = n
        self.dx = dx

    def fun(self, x):
        return x[self.n] - np.correlate(x[:self.n], 1.0 - x[:self.n], mode='full')[self.m] * self.dx

    def jac(self, x):
        gh = corr_jac(x[:self.n], self.m) * self.dx
        gt = np.zeros(self.n + 1)
        gt[:self.n] = -gh
        gt[self.n] = 1.0
        return gt


class _EC:
    def __init__(self, n):
        self.n = n

    def fun(self, x):
        return x[:self.n].sum() - self.n / 2.0

    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def _obj(x):
    return x[-1]


def _obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def optimize_step(h_init, n, tlim, n_active=20):
    t0 = time.time()
    dx = 2.0 / n
    h = h_init.astype(float).copy()
    fix_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for _ in range(50):
        if time.time() - t0 > tlim:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        max_c = c.max()
        active = np.argsort(c)[-n_active:]

        ec = _EC(n)
        cons = [{'type': 'eq', 'fun': ec.fun, 'jac': ec.jac}]
        for k in active:
            cc = _CC(int(k), n, dx)
            cons.append({'type': 'ineq', 'fun': cc.fun, 'jac': cc.jac})

        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        x0 = np.concatenate([h, [max_c * 1.005 + 0.002]])

        try:
            res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12})
        except Exception:
            break

        h_new = np.clip(res.x[:n], 0.0, 1.0)
        fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= max_c + 1e-9:
            break

    return best_h, best_c5


def random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0.0, 1.0, n)
    fix_sum(h, n)
    return h


def structured_h(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(np.pi * x + rng.uniform(0, 6.28))

    h = np.clip(h, 0.0, 1.0)
    fix_sum(h, n)
    return h
