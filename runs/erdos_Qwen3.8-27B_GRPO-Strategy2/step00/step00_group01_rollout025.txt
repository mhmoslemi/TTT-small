from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def enforce_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        m = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        c = np.sum(m)
        if c > 0:
            h[m] += diff / c
            h = np.clip(h, 0, 1)
    return h

def quantize_binary(h, target):
    n = len(h)
    order = np.argsort(-h)
    result = np.zeros(n)
    if abs(target - round(target)) < 1e-9:
        n1 = int(round(target))
        for i in range(min(n1, n)):
            result[order[i]] = 1.0
    else:
        nf = int(np.floor(target))
        frac = target - nf
        for i in range(min(nf, n)):
            result[order[i]] = 1.0
        if nf < n:
            result[order[nf]] = frac
    return result

def sa_swap(h, n, target, rng, budget_s, t0, max_iter=20000):

    T = 0.005
