        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': _constr, 'args': (tsum,)}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.zeros(45)
        h0[:22] = 1.0
        h0[22] = 0.5

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target_sum = n / 2.0

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Anneal lambda from large to small at original resolution
    for lam in [80.0, 50.0, 30.0, 20.0, 14.0, 10.0, 8.0, 6.0, 5.0, 4.0, 3.0, 2.5]:
        if time_left() < 15:
            break
        x_res = slsqp_optimize(best_h, lam, maxiter=200)
        x_res = normalize_h(x_res, target_sum)
        c5_try = compute_c5(x_res)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = x_res.copy()

    # Phase 2: Random perturbations from best
    n_pert = int(min(40, max(5, time_left() / 25)))
    for p in range(n_pert):
        if time_left() < 15:
            break
        scale = 0.06 + 0.04 * rng.rand()
        h_p = best_h + rng.randn(n) * scale
        h_p = normalize_h(h_p, target_sum)
        x_res = slsqp_optimize(h_p, 10.0, maxiter=100)
        x_res = normalize_h(x_res, target_sum)
        c5_try = compute_c5(x_res)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = x_res.copy()

    # Phase 3: Try structured perturbations (swap blocks)
    for p in range(min(20, max(3, int(time_left() / 20)))):
        if time_left() < 15:
            break
        h_s = best_h.copy()
        for _ in range(rng.randint(2, 6)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                h_s[i], h_s[j] = h_s[j], h_s[i]
        h_s = normalize_h(h_s, target_sum)
        x_res = slsqp_optimize(h_s, 8.0, maxiter=80)
        x_res = normalize_h(x_res, target_sum)
        c5_try = compute_c5(x_res)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = x_res.copy()

    # Phase 4: Higher resolution if time permits
    if time_left() > 200:
        for n2 in [90, 135, 180]:
            if time_left() < 150:
                break
            dx2 = 2.0 / n2
            ts2 = n2 / 2.0
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h2 = np.interp(x_new, x_old, np.append(best_h, best_h[0]))
            h2 = h2 * ts2 / max(h2.sum(), 1e-10)
            h2 = np.clip(h2, 0.0, 1.0)
            h2 = normalize_h(h2, ts2)

            for lam2 in [30.0, 15.0, 8.0]:
                if time_left() < 80:
                    break
                x_res = slsqp_optimize(h2, lam2, maxiter=120)
                x_res = normalize_h(x_res, ts2)
                c5_try = compute_c5(x_res)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = x_res.copy()
                    n = n2
                    target_sum = ts2

    # Phase 5: Final polish at current resolution
    if time_left() > 30:
        for lam3 in [5.0, 3.0, 2.0]:
            if time_left() < 15:
                break
            x_res = slsqp_optimize(best_h, lam3, maxiter=150)
            x_res = normalize_h(x_res, target_sum)
            c5_try = compute_c5(x_res)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = x_res.copy()

    # Ensure exact constraint satisfaction
    best_h = normalize_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
