    n = len(h)

        # i.e., 1 - 2h

import numpy as np
import time

def compute_c5(h, dx):
    """Return C5 = max_k correlate(h, 1-h, full)[k] * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_lag(h, k):
    """Gradient of the k-th lag correlation w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    kk = abs(k)
    if kk == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[kk:] += 1.0 - h[:n-kk]
        g[:n-kk] -= h[kk:]
    else:
        g[:n-kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n-kk]
    return g

def project_sum(h, target):
    """Clip to [0,1] and adjust sum to target."""
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            m = h < 1.0 - 1e-9
        else:
            m = h > 1e-9
        c = m.sum()
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
    return h

def bang_bang_search(h, dx, target, rng, time_limit, best_c5, best_h):
    """Local search on bang-bang solutions via random swaps."""
    n = len(h)
    h_bb = (h >= 0.5).astype(float)
    # Fix count
    diff = target - h_bb.sum()
    if diff > 0:
        zeros = np.where(h_bb == 0)[0]
        for i in rng.choice(len(zeros), int(diff), replace=False):
            h_bb[zeros[i]] = 1
    elif diff < 0:
        ones = np.where(h_bb == 1)[0]
        for i in rng.choice(len(ones), int(-diff), replace=False):
            h_bb[ones[i]] = 0
    
    current_c5 = compute_c5(h_bb, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_bb.copy()
    
    deadline = time.time() + time_limit
    patience = 0
    
    while time.time() < deadline:
        ones = np.where(h_bb == 1)[0]
        zeros = np.where(h_bb == 0)[0]
        found = False
        for _ in range(1000):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bb[i], h_bb[j] = 0.0, 1.0
            c5_try = compute_c5(h_bb, dx)
            if c5_try <= current_c5:
                current_c5 = c5_try
                found = True
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_bb.copy()
                patience = 0
                ones = np.where(h_bb == 1)[0]
                zeros = np.where(h_bb == 0)[0]
            else:
                h_bb[i], h_bb[j] = 1.0, 0.0
                patience += 1
        if not found:
            # Occasionally accept a worse move (simulated annealing)
            if rng.random() < 0.3:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bb[i], h_bb[j] = 0.0, 1.0
                current_c5 = compute_c5(h_bb, dx)
                ones = np.where(h_bb == 1)[0]
                zeros = np.where(h_bb == 0)[0]
            patience += 1000
            if patience > 50000:
                break
    
    return best_c5, best_h

def grad_descent(h, dx, target, rng, time_limit, best_c5, best_h):
    """Continuous gradient descent."""
    n = len(h)
    h = h.copy()
    lr = 0.01
    deadline = time.time() + time_limit
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Use top 5 lags for smoother gradient
        top_idx = np.argsort(corr)[-5:]
        g = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            g += grad_lag(h, lag)
        g *= dx
        
        h_new = h - lr * g
        h_new = project_sum(h_new, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.9999, 0.001)
    
    return best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    target = n / 2.0

    best_c5 = compute_c5(h, dx)
    
    # Phase 1: Bang-bang search (40% of budget)
    t1 = budget_s * 0.4
    best_c5, best_h = bang_bang_search(h, dx, target, rng, t1, best_c5, best_h)
    
    # Phase 2: Continuous gradient descent (30%)
    t2 = budget_s * 0.3
    best_c5, best_h = grad_descent(best_h, dx, target, rng, t2, best_c5, best_h)
    
    # Phase 3: More bang-bang search from new starting point (30%)
    remaining = max(0, budget_s - 5 - (time.time() - t0))
    best_c5, best_h = bang_bang_search(best_h, dx, target, rng, remaining, best_c5, best_h)
    
    return (best_h, best_c5, n)

np.correlate(a, v, mode='full') where a and v both have length n:
The result has length 2n-1.
result[i] = sum_j a[j] * v[j - i + (n-1)] for valid j

Equivalently, let lag = i - (n-1). Then result[(n-1)+lag] = sum_j a[j] * v[j - lag].

For a = h and v = 1-h:
result[(n-1)+lag] = sum_j h[j] * (1-h[j-lag])

where j-lag must be in [0, n-1] and j in [0, n-1].

For lag ≥ 0: j ranges from lag to n-1, so count = n - lag terms.
For lag < 0: j ranges from 0 to n-1+lag, so count = n + lag terms (where lag is negative, so n+lag < n).

C5 = max over all lag of result[(n-1)+lag] * dx.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    kk = abs(k)
    if kk == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[kk:] += 1.0 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    else:
        g[:n - kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n - kk]
    return g

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            m = h < 1.0 - 1e-9
        else:
            m = h > 1e-9
        c = m.sum()
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
    return h

def bang_bang_local(h, dx, target, rng, deadline, best_c5, best_h):
    n = len(h)
    h_bb = (h >= 0.5).astype(float)
    diff = int(target - h_bb.sum())
    if diff > 0:
        z = np.where(h_bb == 0)[0]
        h_bb[rng.choice(len(z), diff, replace=False)] = 1.0
    elif diff < 0:
        o = np.where(h_bb == 1)[0]
        h_bb[rng.choice(len(o), -diff, replace=False)] = 0.0

    cur = compute_c5(h_bb, dx)
    if cur < best_c5:
        best_c5, best_h = cur, h_bb.copy()

    fail_count = 0
    while time.time() < deadline:
        ones = np.where(h_bb == 1)[0]
        zeros = np.where(h_bb == 0)[0]
        batch_found = False
        for _ in range(500):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bb[i], h_bb[j] = 0.0, 1.0
            c5t = compute_c5(h_bb, dx)
            if c5t <= cur:
                cur = c5t
                batch_found = True
                if c5t < best_c5:
                    best_c5, best_h = c5t, h_bb.copy()
                ones = np.where(h_bb == 1)[0]
                zeros = np.where(h_bb == 0)[0]
            else:
                h_bb[i], h_bb[j] = 1.0, 0.0
                fail_count += 1
        if not batch_found:
            if fail_count > 8000:
                break
            if rng.random() < 0.4:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bb[i], h_bb[j] = 0.0, 1.0
                cur = compute_c5(h_bb, dx)
        else:
            fail_count = 0
    return best_c5, best_h

def full_sweep_bh(h_bb, dx, target, deadline):
    n = len(h_bb)
    cur = compute_c5(h_bb, dx)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h_bb == 1)[0]
        zeros = np.where(h_bb == 0)[0]
        for i in ones:
            if time.time() >= deadline:
                break
            for j in zeros:
                h_bb[i], h_bb[j] = 0.0, 1.0
                c5t = compute_c5(h_bb, dx)
                h_bb[i], h_bb[j] = 1.0, 0.0
                if c5t < cur - 1e-12:
                    h_bb[i], h_bb[j] = 0.0, 1.0
                    cur = c5t
                    improved = True
                    ones = np.where(h_bb == 1)[0]
                    zeros = np.where(h_bb == 0)[0]
                    break
    return h_bb, cur

def grad_desc(h, dx, target, deadline, best_c5, best_h):
    n = len(h)
    h = h.copy()
    lr = 0.008
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-4:]
        g = np.zeros(n)
        for idx in top:
            g += grad_lag(h, idx - (n - 1))
        g *= dx
        h_new = h - lr * g
        h_new = project_sum(h_new, target)
        c5n = compute_c5(h_new, dx)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()
        h = h_new
        lr = max(lr * 0.9997, 0.0005)
    return best_c5, best_h

    t0 = time.time()

        n = len(h0)

    dx = 2.0 / n
    target = n / 2.0
    h = project_sum(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    d1 = t0 + budget_s * 0.35
    best_c5, best_h = bang_bang_local(h, dx, target, rng, d1, best_c5, best_h)

    # Full sweep to polish
    h_s = (best_h >= 0.5).astype(float)
    d = int(target - h_s.sum())
    if d > 0:
        z = np.where(h_s == 0)[0]
        h_s[rng.choice(len(z), d, replace=False)] = 1.0
    elif d < 0:
        o = np.where(h_s == 1)[0]
        h_s[rng.choice(len(o), -d, replace=False)] = 0.0
    d2 = t0 + budget_s * 0.5
    h_s, cur_c5 = full_sweep_bh(h_s, dx, target, d2)
    if cur_c5 < best_c5:
        best_c5, best_h = cur_c5, h_s.copy()

    # Continue random local search
    d3 = t0 + budget_s * 0.7
    best_c5, best_h = bang_bang_local(best_h, dx, target, rng, d3, best_c5, best_h)

    # Continuous gradient descent
    d4 = t0 + budget_s * 0.85
    best_c5, best_h = grad_desc(best_h, dx, target, d4, best_c5, best_h)

    # Final bang-bang polish
    d5 = t0 + budget_s - 3
    best_c5, best_h = bang_bang_local(best_h, dx, target, rng, d5, best_c5, best_h)

    return (best_h, best_c5, n)
