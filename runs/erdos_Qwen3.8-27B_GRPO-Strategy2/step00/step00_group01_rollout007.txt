from scipy import optimize
import time

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    return np.max(corr) * dx

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    max_val = np.max(corr)
    active = (corr >= max_val - 1e-12)
    n_active = np.sum(active)
    
    # Affects: only lags t where |t| <= n-1 matter, but since h is 0 outside [0,n),
    # only shifts t in [-(n-1), n-1] can have non-zero correlation
    # For each active lag, compute gradient
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if not active[k_idx]:
            continue
        s = k_idx - (n - 1)  # shift
        t = abs(s)
        # grad[j] += (1-h[j+t]) * [j+t < n] - h[j-t] * [j >= t]
        if t == 0:
            # grad[j] = (1-h[j]) [j < n] - h[j] [j >= 0] = 1 - 2*h[j]
            grad[:n] += (1 - 2*h)
        else:
            # (1-h[j+t]) for j in [0, n-t-1]
            grad[:n-t] += (1 - h[t:n])
            # -h[j-t] for j in [t, n-1]
            grad[t:] -= h[:n-t]
    
    grad /= n_active
    return max_val * dx, grad * dx

def project_onto_feasible(y, target_sum):
    """Project y onto {h: 0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(y)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        lam = 0.5 * (lo + hi)
        proj = np.clip(y - lam, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(y - 0.5*(lo+hi), 0, 1)

def optimize_subgradient(h0, n_max_iter=5000, time_limit=60):
    n = len(h0)
    h = h0.copy()
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    alpha = 0.01
    start_time = time.time()
    
    for it in range(n_max_iter):
        if time.time() - start_time > time_limit:

        c5, grad = compute_subgradient(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient step + projection

        h_new = project_onto_feasible(h_new, target_sum)
        
        # Simple line search
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
        else:

            h_new = h - alpha * grad
            h_new = project_onto_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(y - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    # Exact fix
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Add diff to elements with room
        if diff > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - result[i])
            else:
                add = max(diff, -result[i])
            result[i] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return result

def subgrad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    act = corr >= mx - 1e-11
    cnt = max(act.sum(), 1)
    g = np.zeros(n)
    for k in np.where(act)[0]:
        s = abs(k - (n - 1))
        if s == 0:
            g += 1.0 - 2.0 * h
        else:
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
    return mx * dx, g * dx / cnt

def optimize_subgrad(h0, n, tl=30, a0=1.0):
    h = h0.copy()
    bh, bc = h.copy(), compute_c5(h)
    a = a0
    t0 = time.time()
    for _ in range(200000):
        if time.time() - t0 > tl:
            break
        c, g = subgrad_c5(h)
        if c < bc - 1e-13:
            bc, bh = c, h.copy()
        s = project_feasible(h - a * g, n)
        cs = compute_c5(s)
        if cs <= c:
            h = s
            a *= 1.01
        else:
            a *= 0.75
        if a < 1e-7:
            a = a0
            h = project_feasible(h + 0.05 * np.random.randn(n), n)
    return bh, bc
