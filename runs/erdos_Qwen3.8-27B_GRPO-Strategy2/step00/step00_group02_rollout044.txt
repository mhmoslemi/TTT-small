def project_onto_polytope(h, c):
    """Project h onto {x: 0 <= x <= 1, sum(x) = c} via bisection on offset."""
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr


def smooth_max(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def gradient_of_corr(h, n, s):
    """Gradient of correlation at shift s (signed) w.r.t. h."""
    grad = np.zeros(n)
    if s >= 0:
        for j in range(n - s):
            grad[j] += (1.0 - h[j + s])
        for j in range(s, n):
            grad[j] -= h[j - s]
    else:
        s2 = -s
        for j in range(s2, n):
            grad[j] += (1.0 - h[j - s2])
        for j in range(n - s2):
            grad[j] -= h[j + s2]
    return grad


def smooth_obj(h_flat, n, dx, beta):
    _, corr = compute_c5(h_flat, dx)
    return smooth_max(corr, beta)


def smooth_grad(h_flat, n, dx, beta):
    _, corr = compute_c5(h_flat, dx)
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        g = gradient_of_corr(h_flat, n, s)
        grad += w[k] * g
    return dx * beta * grad

    rng = np.random.RandomState(seed)

    n = 45
    dx = 2.0 / n
    c_target = n / 2.0

                      np.linspace(0.0, 2.0, len(h0)), h0)
    else:
        h = np.full(n, 0.5)

    h = project_onto_polytope(h, c_target)
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)

    # Phase 1: L-BFGS-B with smooth max
    beta = 40.0
    bounds = [(0.0, 1.0)] * n

    for restart in range(40):
        elapsed = time.time() - t0
        if budget_s - elapsed < 20:
            break

        if restart == 0:
            h_cur = best_h.copy()
        elif restart % 5 == 0:
            h_cur = project_onto_polytope(
                best_h + rng.normal(0, 0.25, n), c_target)
        else:
            h_cur = project_onto_polytope(
                best_h + rng.normal(0, 0.1, n), c_target)

        try:
            res = minimize(
                lambda x: smooth_obj(x, n, dx, beta),
                h_cur,
                jac=lambda x: smooth_grad(x, n, dx, beta),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            h_new = project_onto_polytope(res.x, c_target)
            c5_new, _ = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: Projected gradient descent fine-tuning
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        lr = 0.02
        h = best_h.copy()
        for it in range(5000):
            if time.time() - t0 > budget_s - 3:
                break
            c5, corr = compute_c5(h, dx)
            k_max = int(np.argmax(corr))
            s = k_max - (n - 1)
            grad = dx * gradient_of_corr(h, n, s)

            h_trial = project_onto_polytope(h - lr * grad, c_target)
            c5_trial, _ = compute_c5(h_trial, dx)
            if c5_trial < c5 - 1e-14:
                h = h_trial
                lr = min(lr * 1.05, 0.3)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
            else:
                lr *= 0.7

    # Phase 3: Try higher resolution (n=90) if time allows
    elapsed = time.time() - t0
    if budget_s - elapsed > 120:
        n2 = 90
        dx2 = 2.0 / n2
        c2 = n2 / 2.0
        h_hi = np.interp(np.linspace(0.0, 2.0, n2),
                         np.linspace(0.0, 2.0, n), best_h)
        h_hi = project_onto_polytope(h_hi, c2)
        best_c5_hi, _ = compute_c5(h_hi, dx2)

        bounds2 = [(0.0, 1.0)] * n2
        beta2 = 30.0

        for restart in range(15):
            if time.time() - t0 > budget_s - 30:
                break
            if restart == 0:
                h_cur = h_hi.copy()
            else:
                h_cur = project_onto_polytope(
                    h_hi + rng.normal(0, 0.12, n2), c2)
            try:
                res = minimize(
                    lambda x: smooth_obj(x, n2, dx2, beta2),
                    h_cur,
                    jac=lambda x: smooth_grad(x, n2, dx2, beta2),
                    method='L-BFGS-B',
                    bounds=bounds2,
                    options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-10}
                )
                h_new = project_onto_polytope(res.x, c2)
                c5_new, _ = compute_c5(h_new, dx2)
                if c5_new < best_c5_hi:
                    best_c5_hi = c5_new
                    h_hi = h_new.copy()
            except Exception:
                pass

        # Fine-tune at high res
        lr = 0.01
        h = h_hi.copy()
        for it in range(2000):
            if time.time() - t0 > budget_s - 5:
                break
            c5, corr = compute_c5(h, dx2)
            k_max = int(np.argmax(corr))
            s = k_max - (n2 - 1)
            grad = dx2 * gradient_of_corr(h, n2, s)
            h_trial = project_onto_polytope(h - lr * grad, c2)
            c5_trial, _ = compute_c5(h_trial, dx2)
            if c5_trial < c5 - 1e-14:
                h = h_trial
                lr = min(lr * 1.05, 0.2)
                if c5_trial < best_c5_hi:
                    best_c5_hi = c5_trial
                    h_hi = h.copy()
            else:
                lr *= 0.7

        if best_c5_hi < best_c5:
            best_c5 = best_c5_hi
            best_h = h_hi
            n = n2

    return best_h, best_c5, n
