import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return float(dx * np.max(corr))


def get_corr(h):
    dx = 2.0 / len(h)
    return correlate(h, 1 - h, mode='full') * dx


def corr_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    jac = np.zeros((n_corr, n))
    for m in range(n_corr):
        k = m - (n - 1)
        a1 = max(0, k)
        b1 = min(n - 1, n - 1 + k)
        if a1 <= b1:
            jac[m, a1:b1 + 1] += 1.0 - h[a1 - k:b1 - k + 1]
        a2 = max(0, -k)
        b2 = min(n - 1, n - 1 - k)
        if a2 <= b2:
            jac[m, a2:b2 + 1] -= h[a2 + k:b2 + k + 1]
    return jac * dx


class ErdosProblem:
    def __init__(self, n, alpha):
        self.n = n
        self.alpha = alpha

    def objective(self, x):
        c = get_corr(x)
        mx = np.max(c)
        if mx < 1e-15:
            return 0.0
        return mx + self.alpha * np.log(np.sum(np.exp((c - mx) / self.alpha)))

    def grad(self, x):
        c = get_corr(x)
        mx = np.max(c)
        w = np.exp((c - mx) / self.alpha)
        w /= w.sum()
        J = corr_jacobian(x)
        return w @ J

    def eq_con(self, x):
        return np.sum(x) - self.n / 2.0

    def eq_con_jac(self, x):
        return np.ones(self.n)


def enforce_sum(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = (target - np.sum(h)) / n
    h2 = np.clip(h + diff, 0, 1)
    for _ in range(5):
        d = (target - np.sum(h2)) / n
        h3 = np.clip(h2 + d, 0, 1)
        if np.sum(h3) == np.sum(h2):
            break
        h2 = h3
    return h2


def optimize_local(h0, n, t_limit):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    stages = [(0.05, 0.1), (0.02, 0.05), (0.008, 0.02), (0.003, 0.008), (0.001, 0.003)]
    for alpha, _ in stages:
        if time.time() > t_limit:
            break
        prob = ErdosProblem(n, alpha)
        cons = [{'type': 'eq', 'fun': prob.eq_con, 'jac': prob.eq_con_jac}]
        try:
            res = minimize(prob.objective, best_h, jac=prob.grad, method='SLSQP',
                           bounds=[(0, 1)] * n, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-12})
            h_new = np.clip(res.x, 0, 1)
            h_new = enforce_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5


def binary_search(h0, n, t_limit, seed):
    rng = np.random.RandomState(seed)
    h_bin = (h0 > 0.5).astype(float)
    target = int(n // 2)
    s = int(round(np.sum(h_bin)))
    if s > target:
        ones = np.where(h_bin == 1)[0]
        for _ in range(s - target):
            idx = rng.choice(ones)
            h_bin[idx] = 0
            ones = np.where(h_bin == 1)[0]
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        for _ in range(target - s):
            idx = rng.choice(zeros)
            h_bin[idx] = 1
            zeros = np.where(h_bin == 0)[0]
    best_c5 = compute_c5(h_bin)
    h_best = h_bin.copy()
    for it in range(500000):
        if time.time() > t_limit:
            break
        i = rng.randint(n)
        h_test = h_best.copy()
        h_test[i] = 1.0 - h_test[i]
        c5_test = compute_c5(h_test)
        if c5_test < best_c5:
            h_best = h_test
            best_c5 = c5_test
    return h_best, best_c5
