from scipy.interpolate import interp1d
import time

    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    target = n / 2.0
    if s > 1e-10:
        x = x * (target / s)
        x = np.clip(x, 0.0, 1.0)
    return x

def smooth_objective(x, dx, tau):
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -500, 500)
    return (m + tau * np.log(np.sum(np.exp(shifted)))) * dx
