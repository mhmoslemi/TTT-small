import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def fix_sum(h, target, rng):
    """Adjust sum of h to target while keeping values in [0,1]."""
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = (target - s) / len(h)
        h = np.clip(h + diff, 0.0, 1.0)
    return h

def make_binary_valid(h, n_target, rng):
    """Convert to exact binary with n_target ones."""
    h_b = (h >= 0.5).astype(float)
    n_ones = int(h_b.sum())
    while n_ones > n_target:
        ones = np.where(h_b == 1)[0]
        h_b[ones[rng.integers(len(ones))]] = 0
        n_ones -= 1
    while n_ones < n_target:
        zeros = np.where(h_b == 0)[0]
        h_b[zeros[rng.integers(len(zeros))]] = 1
        n_ones += 1
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(80)
        h0[::2] = 1.0
    n = len(h0)
    dx = 2.0 / n
    n_target = n / 2.0
    n_target_int = n // 2

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, n_target, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary simulated annealing (main phase)
    h_bin = make_binary_valid(best_h, n_target_int, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    h = h_bin.copy()
    c5_cur = compute_c5(h)
    if c5_cur < best_c5:
        best_h = h.copy()
        best_c5 = c5_cur

    T = 0.015
    n_steps = 0
    stall = 0
    max_stall = 8000

    while time.time() - t_start < budget_s * 0.55 and stall < max_stall:
        n_steps += 1
        r = rng.random()
        if r < 0.8:
            # Single swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            d = new_c5 - c5_cur
            if d < 0 or rng.random() < np.exp(-d / T):
                c5_cur = new_c5
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    stall = 0
                else:
                    stall += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                stall += 1
        else:
            # Double swap (4-point move)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= 2 and len(zeros) >= 2:

                h[i1] = 0; h[j1] = 1
                h[i2] = 0; h[j2] = 1

                    if new_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = new_c5
                        stall = 0
                    else:
                        stall += 1
                else:
                    h[i1] = 1; h[j1] = 0
                    h[i2] = 1; h[j2] = 0
                    stall += 1
        if n_steps % 500 == 0:

    # Phase 2: Continuous refinement around best

        hc = best_h.copy()
        c5_c = compute_c5(hc)
        step = 0.03
        for _ in range(200000):

            # Fix sum via another point
            diff = n_target - hc.sum()
            if abs(diff) > 1e-14:
                idx2 = int(rng.integers(n))
                hc[idx2] = np.clip(hc[idx2] - diff, 0.0, 1.0)
            new_c5 = compute_c5(hc)
            if new_c5 < c5_c:
                c5_c = new_c5
                if new_c5 < best_c5:
                    best_h = hc.copy()
                    best_c5 = new_c5
            else:
                hc[idx] = old
                hc = best_h.copy()
                c5_c = best_c5
            step *= 0.9998
            if step < 0.001:
                step = 0.03

    phase3_time = budget_s * 0.82

            break
        if restart == 0:
            h_r = make_binary_valid(best_h, n_target_int, rng)
        else:
            # Perturb best binary
            h_r = best_h.copy()
            h_r = np.where(h_r >= 0.5, 1.0, 0.0)
            for _ in range(rng.integers(3, 15)):
                ones = np.where(h_r == 1)[0]
                zeros = np.where(h_r == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                h_r[ones[rng.integers(len(ones))]] = 0
                h_r[zeros[rng.integers(len(zeros))]] = 1
            h_r = make_binary_valid(h_r, n_target_int, rng)

        c5_r = compute_c5(h_r)
        T_r = 0.008
        stall_r = 0
        for _ in range(30000):
            if time.time() - t_start > phase3_time:
                break
            if stall_r > 3000:
                break
            ones = np.where(h_r == 1)[0]
            zeros = np.where(h_r == 0)[0]
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            h_r[i] = 0; h_r[j] = 1
            new_c5 = compute_c5(h_r)
            d = new_c5 - c5_r
            if d < 0 or rng.random() < np.exp(-d / T_r):
                c5_r = new_c5
                if new_c5 < best_c5:

                    best_c5 = new_c5
                    stall_r = 0
                else:
                    stall_r += 1
            else:
                h_r[i] = 1; h_r[j] = 0
                stall_r += 1
            T_r = max(0.0003, T_r * 0.9997)

    # Phase 4: Final continuous polish
    if time.time() - t_start < budget_s * 0.95:
        hp = best_h.copy()
        c5_p = compute_c5(hp)
        for _ in range(100000):
            if time.time() - t_start > budget_s * 0.95:
                break
            idx = int(rng.integers(n))
            old = hp[idx]
            pert = rng.uniform(-0.02, 0.02)
            hp[idx] = np.clip(old + pert, 0.0, 1.0)
            diff = n_target - hp.sum()
            if abs(diff) > 1e-14:
                idx2 = int(rng.integers(n))
                hp[idx2] = np.clip(hp[idx2] - diff, 0.0, 1.0)
            new_c5 = compute_c5(hp)
            if new_c5 < c5_p:
                c5_p = new_c5
                if new_c5 < best_c5:
                    best_h = hp.copy()
                    best_c5 = new_c5
            else:
                hp[idx] = old

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = n_target - best_h.sum()
    if abs(diff) > 1e-10:
        best_h = best_h + diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
