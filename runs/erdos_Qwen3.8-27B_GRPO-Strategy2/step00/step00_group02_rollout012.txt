import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def smooth_max(values, alpha):
    vals = np.asarray(values, dtype=float)
    if alpha < 1:
        return np.max(vals)
    shifted = alpha * vals
    max_shifted = np.max(shifted)
    return max_shifted / alpha + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-6:
        mask = (h > 1e-4) & (h < 1.0 - 1e-4)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                idx = np.argsort(h)
                for i in idx:
                    if h[i] < 1.0:
                        add = min(diff, 1.0 - h[i])
                        h[i] += add
                        diff -= add
                        if diff < 1e-10:
                            break
            else:

                    if h[i] > 0.0:
                        rem = min(-diff, h[i])
                        h[i] -= rem
                        diff += rem
                        if diff > -1e-10:
                            break

    if abs(diff) > 1e-10 and diff != 0:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_max(values, alpha):
    vals = np.asarray(values, dtype=float)
    if alpha < 1.0:
        return float(np.max(vals))
    shifted = alpha * vals
    m = np.max(shifted)
    return float(m / alpha + np.log(np.sum(np.exp(shifted - m))) / alpha)

def project(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        best_h, best_c5 = local_refinement(best_h, best_c5, n, target_sum, rng, t0, budget_s)
        
        # One more SLSQP pass with high alpha
        if time.time() - t0 < budget_s - 10:
            bounds = [(0.0, 1.0)] * n
            constraints = [{'type': 'eq', 'fun': sum_constr}]
            hc = try_optimize(best_h, n, target_sum, bounds, constraints, 500, 500, t0, budget_s)
            if hc is not None:
                c5_c = compute_c5(hc)
                if c5_c < best_c5:
                    best_h = hc
                    best_c5 = c5_c
            best_h, best_c5 = local_refinement(best_h, best_c5, n, target_sum, rng, t0, budget_s)

    if best_h is None:
        best_h = make_initial(45, seed)
        best_c5 = compute_c5(best_h)
        best_n = 45

    return (best_h, best_c5, best_n)
