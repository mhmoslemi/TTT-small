def simulated_annealing(h, dx, target_sum, rng, T0=1.0, T_end=0.01, max_iters=50000):

    best = evaluate(h, dx)
    h_best = h.copy()

        T = T0 * (T_end/T0) ** (it/max_iters)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h[i] -= 1.0
        h[j] += 1.0
        new_val = evaluate(h, dx)
        
        delta = new_val - curr
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            curr = new_val
            if curr < best:
                best = curr
                h_best = h.copy()

            h[i] += 1.0
            h[j] -= 1.0

                h[idx] = 0.0

    """Simulated annealing on binary sequences via swaps."""
    n = len(h)
    h = ensure_binary_sum(h, n, rng)
    best = eval_c5(h, dx)
    h_best = h.copy()
    curr = best
    T = 0.005

        zeros = np.where(h == 0.0)[0]

        j = int(rng.choice(zeros))

    h = np.zeros(n)

    h[idx[:n // 2]] = 1.0
    return h


def thue_morse_seq(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = c & 1
    return h


def ensure_sum(h, n, rng):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h).clip(0.0, 1.0)
    target = n // 2
    curr = int(np.sum(h))
    if curr < target:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= target - curr:
            picks = rng.choice(zeros, size=target - curr, replace=False)
            h[picks] = 1.0
    elif curr > target:
        ones = np.where(h == 1)[0]
        if len(ones) >= curr - target:
            picks = rng.choice(ones, size=curr - target, replace=False)
            h[picks] = 0.0
    return h


def sa_search(h, dx, rng, time_limit):
    n = len(h)
    h = ensure_sum(h, n, rng).copy()
    best = eval_c5(h, dx)
    h_best = h.copy()
    curr = best
    T = 0.002
    t_start = time.time()
    stall = 0

    while time.time() - t_start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        new_val = eval_c5(h, dx)
        delta = new_val - curr
        if delta < -1e-12:
            curr = new_val
            stall = 0
            if curr < best:
                best = curr
                h_best = h.copy()
        elif delta < 0.05 and rng.rand() < np.exp(-delta / (T + 1e-15)):
            curr = new_val
            stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        if stall % 400 == 0:
            T = max(T * 0.96, 0.0002)
        if stall > 10000:
            T = 0.006
            stall = 0
    return h_best, best


def targeted_refine(h, dx, rng, n_tries=5000):
    n = len(h)
    best_val = eval_c5(h, dx)
    best_h = h.copy()
    for _ in range(n_tries):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        val = eval_c5(h, dx)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        h[i] = best_h[i]
        h[j] = best_h[j]
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 70

    def record(h_c, n_c):
        nonlocal best_h, best_c5, best_n
        h_c = ensure_sum(h_c, n_c, rng)
        dx_c = 2.0 / n_c
        c5_c = eval_c5(h_c, dx_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
            best_n = n_c

    # Build list of starting points
    starts = []

    # From initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_init = len(initial_h_values)
        starts.append((np.array(initial_h_values, dtype=float).copy(), n_init))

    # Thue-Morse at several sizes
    for n_tm in [64, 70, 96, 100, 128, 140]:
        if n_tm % 2 == 0:
            starts.append((thue_morse_seq(n_tm), n_tm))

    # Random starts
    for n_r in [70, 100, 140]:
        starts.append((make_binary(n_r, rng), n_r))

    # Mandelbrot-like / Fibonacci word
    def fib_word(n):
        a, b = np.zeros(1), np.ones(1)
        while len(a) < n:
            a = np.concatenate([a, b])
            a, b = b, a
        return a[:n]
    nw = 100
    starts.append((fib_word(nw), nw))

    # Deduplicate by (n, hash)
    seen = set()
    unique_starts = []
    for h_s, n_s in starts:
        key = (n_s,)
        if key not in seen:
            seen.add(key)
            unique_starts.append((h_s, n_s))
    starts = unique_starts

    # Phase 1: SA from each start
    time_per_start = max(15, (budget_s * 0.70) / len(starts))

    for h_s, n_s in starts:
        if time.time() - t0 > budget_s * 0.82:
            break
        record(h_s, n_s)
        dx_s = 2.0 / n_s
        remaining = budget_s * 0.82 - (time.time() - t0)
        t_alloc = min(time_per_start, max(5, remaining / 3))
        h_res, c5_res = sa_search(h_s, dx_s, rng, t_alloc)
        record(h_res, n_s)

    # Phase 2: Intensive refinement on best
    if best_h is not None:
        dx_best = 2.0 / best_n
        remaining = budget_s * 0.92 - (time.time() - t0)
        if remaining > 10:
            t_refine = min(remaining * 0.6, 200)
            h_res, c5_res = sa_search(best_h, dx_best, rng, t_refine)
            record(h_res, best_n)

    # Phase 3: Targeted greedy refinement
    if best_h is not None:
        dx_best = 2.0 / best_n
        remaining = budget_s * 0.95 - (time.time() - t0)
        if remaining > 5:
            n_tr = int(min(20000, remaining * 8000))
            h_res, c5_res = targeted_refine(best_h, dx_best, rng, max(100, n_tr))
            record(h_res, best_n)

    # Phase 4: One more SA burst if time remains
    if best_h is not None:
        dx_best = 2.0 / best_n
        remaining = budget_s - (time.time() - t0) - 5
        if remaining > 20:
            h_res, c5_res = sa_search(best_h, dx_best, rng, remaining * 0.8)
            record(h_res, best_n)

    # Fallback
    if best_h is None:
        h_fb = make_binary(70, rng)
        best_h = ensure_sum(h_fb, 70, rng)
        best_c5 = eval_c5(best_h, 2.0 / 70)
        best_n = 70

    return best_h, best_c5, best_n
