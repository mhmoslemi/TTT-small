    return float(np.max(corr) * dx)

def repair_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def try_slsqp(h, n, ts, maxiter=300):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts}]
        res = minimize(compute_c5, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            cand = repair_h(res.x, ts)
            return cand, compute_c5(cand)
    except Exception:
        pass
    return h, compute_c5(h)

def local_search(h, ts, n, max_passes=200, time_fn=None, time_limit=1100):
    best = h.copy()
    best_val = compute_c5(best)
    step_sizes = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for ss in step_sizes:
        for _ in range(max_passes):
            if time_fn and time_fn() > time_limit:
                return best, best_val
            improved = False
            order = np.random.permutation(n)
            for i in order:
                for delta in [ss, -ss]:
                    cand = best.copy()
                    cand[i] = np.clip(best[i] + delta, 0.0, 1.0)
                    # Repair: distribute difference to other coords
                    diff = ts - np.sum(cand)
                    if abs(diff) > 1e-14:
                        other = np.ones(n)
                        other[i] = 0
                        cnt = np.sum(other)

Actually, let me be more precise. For mode='full', the output has length len(a) + len(b) - 1. The k-th element (0-indexed) is:
c[k] = sum_i a[i] * b[i + k - (len(b) - 1)]

where the sum is over all valid i.

So for a = h and b = 1-h, with both of length n:
c[k] = sum_i h[i] * (1-h[i + k - (n-1)])

The center element is c[n-1] = sum_i h[i] * (1-h[i]), which is the "overlap" at lag 0.

For k = n-1 + m (m > 0), this is the correlation at lag m: sum_i h[i] * (1-h[i+m]) (for valid i).
For k = n-1 - m (m > 0), this is the correlation at lag -m: sum_i h[i] * (1-h[i-m]) (for valid i).

The max over all k gives the maximum "overlap" over all lags. This is what C5 is.

My compute_c5 function takes the max of the entire full correlation, which is correct.
