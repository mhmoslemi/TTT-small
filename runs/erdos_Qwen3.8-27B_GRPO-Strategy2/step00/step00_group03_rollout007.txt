- h'[i] = h[i] - δ, h'[j] = h[j] + δ
- Find the best δ (in [max(0, h[i]-1), min(1-h[j], h[i])]) that minimizes C5(h')
- Since C5 is the max of piecewise linear functions of δ, the minimum is at a "kink point" or at a boundary

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_h(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        d = target - float(np.sum(h))
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            cnt = int(np.count_nonzero(m))
            if cnt == 0:
                break
            step = min(d / cnt, 0.5)
            h[m] = np.minimum(h[m] + step, 1.0)
        else:
            m = h > 1e-14
            cnt = int(np.count_nonzero(m))
            if cnt == 0:
                break
            step = min(-d / cnt, 0.5)
            h[m] = np.maximum(h[m] - step, 0.0)
    d = target - float(np.sum(h))
    if abs(d) > 1e-14:
        if d > 0:
            idx = int(np.argmin(1.0 - h))
            h[idx] = float(min(h[idx] + d, 1.0))
        else:
            idx = int(np.argmax(h))
            h[idx] = float(max(h[idx] + d, 0.0))
    return h
