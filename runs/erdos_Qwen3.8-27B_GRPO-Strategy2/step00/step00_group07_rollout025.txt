    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_unitbox(h, S):
    """Project h onto {0<=x<=1, sum(x)=S} via bisection on shift."""
    lo, hi = h.min() - S, h.max()
    # Actually simpler bounds:
    # When shift = -big: all clipped to 1, sum = n
    # When shift = +big: all clipped to 0, sum = 0
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def smooth_max_obj_and_grad(h, tau, n):
    """Compute smooth max of correlation*dx and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx  # len 2n-1
    # Softmax
    c = corr - corr.max()
    exp_c = np.exp(c / tau)
    Z = exp_c.sum()
    obj = tau * np.log(Z)
    
    # Gradient
    w = exp_c / Z  # softmax weights, len 2n-1
    grad = np.zeros(n)
    for p in range(n):
        acc = 0.0
        for m in range(-(n-1), n):
            wm = w[m + n - 1]
            if wm < 1e-14:
                continue
            if 0 <= p + m < n:
                acc += wm * (1.0 - h[p + m])
            if 0 <= p - m < n:
                acc -= wm * h[p - m]
        grad[p] = acc * dx
    return obj, grad

from scipy.optimize import minimize

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

            hi = mid

    c = corr - corr.max()
