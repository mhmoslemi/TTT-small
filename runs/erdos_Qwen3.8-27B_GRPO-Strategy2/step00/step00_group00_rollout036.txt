def project_onto_feasible(h, target_sum):
    # Project h onto {0<=x<=1, sum(x)=target_sum}
    # Use bisection on tau
    lo, hi = -1, 2  # range for tau
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

from scipy.optimize import minimize
import time

def _eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_obj(h, beta, dx, lam, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    t = beta * (corr - m)
    e = np.exp(t)
    es = e.sum()
    obj = np.log(es) / beta + m
    diff = h.sum() - target_sum
    obj += lam * diff * diff
    return obj

def _smooth_grad(h, beta, dx, lam, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    t = beta * (corr - m)
    e = np.exp(t)
    es = e.sum()
    weights = e / es  # length 2n-1
    
    # Gradient of log-sum-exp part
    one_minus_h = 1.0 - h
    p = weights
    q = p[::-1]
    A = np.correlate(one_minus_h, p, mode='full')  # length 3n-2
    B = np.correlate(h, q, mode='full')  # length 3n-2
    idx = 3*n - 3 - np.arange(n)
    grad = A[idx] - B[idx]

import numpy as np
from scipy.optimize import minimize
import time


def _eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _smooth_obj(h, beta, dx, lam, target_sum):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    es = e.sum()
    obj = np.log(es) / beta + m
    diff = h.sum() - target_sum
    obj += lam * diff * diff
    return obj


def _smooth_grad(h, beta, dx, lam, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    es = e.sum()
    w = e / es
    p = w
    A = np.correlate(1.0 - h, p, mode='full')
    B = np.correlate(h, p[::-1], mode='full')
    idx = 3 * n - 3 - np.arange(n)
    grad = A[idx] - B[idx]
    grad += 2.0 * lam * (h.sum() - target_sum)
    return grad


def _project_feasible(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _interp_to_res(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=np.float64) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=np.float64) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)


def _opt_phase(h, n, beta, max_iter, dx, lam, target_sum):
    bounds = [(0.0, 1.0)] * n

        return _smooth_obj(x, beta, dx, lam, target_sum)

        return _smooth_grad(x, beta, dx, lam, target_sum)

    res = minimize(fun, h, jac=jac, bounds=bounds, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                            'maxfun': max_iter * 3})
    return res.x
