    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-14:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(300):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if diff > 0:
        mask = x < 1.0
    elif diff < 0:
        mask = x > 0.0
    else:
        return x
    cnt = np.sum(mask)
    if cnt > 0:
        x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def soft_max(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -200.0, 200.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_obj_factory(alpha, dx):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        return soft_max(c, alpha)
    return obj
