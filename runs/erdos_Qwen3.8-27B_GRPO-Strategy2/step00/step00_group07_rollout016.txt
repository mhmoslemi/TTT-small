import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def grad_corr_max(h, temp=0.01):
    """Compute gradient of soft-max of correlations wrt h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    corr_scaled = corr / temp
    max_c = np.max(corr_scaled)
    exp_c = np.exp(corr_scaled - max_c)
    w = exp_c / np.sum(exp_c)
    
    # Gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    # For efficiency, only use weights > threshold
    threshold = 1e-8
    active = np.where(w > threshold)[0]
    
    for k in active:
        wk = w[k]
        if wk < threshold:
            continue
        # First term
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            h_idx = np.arange(j_lo - k + n - 1, j_hi - k + n + 1)
            grad[j_lo:j_hi+1] += wk * (1.0 - h[h_idx])
        # Second term
        j_lo2 = max(0, n - 1 - k)
        j_hi2 = n - 1
        if j_lo2 <= j_hi2:
            h_idx2 = np.arange(j_lo2 + k - n + 1, j_hi2 + k - n + 2)
            grad[j_lo2:j_hi2+1] -= wk * h[h_idx2]
    
    return grad * temp  # d(soft_max)/dh = temp * sum_k w[k] * d(corr[k])/dh

def project_onto_box_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    # Bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_try = np.clip(h + mid, 0.0, 1.0)
        s_try = np.sum(h_try)
        if s_try > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    return h_proj

def binary_local_search(h_bin, max_sweeps=20):
    """Local search on binary h: swap 0 and 1 to reduce C5."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    best_c5 = compute_c5(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        improved = False
        # Try random swaps (or all swaps for small n)
        if n <= 200:
            pairs = [(i, j) for i in zeros for j in ones]
            np.random.shuffle(pairs)
        else:
            pairs = [(np.random.choice(zeros), np.random.choice(ones)) for _ in range(500)]
        
        for i, j in pairs:
            if h[i] != 0 or h[j] != 1:
                continue
            h[i] = 1
            h[j] = 0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i] = 0
                h[j] = 1
        
        if not improved:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s * 0.92  # Leave some margin

        h0 = h0[:n0]
    
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - start_time
    
    def time_left():
        return deadline - time.time()

        else:

            x_new = np.linspace(0, 2, n_target, endpoint=False)
            h = np.interp(x_new, x_old, h0)

            if it % 500 == 0 and it > 0:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_c5(h, temp=0.01, top_m=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m >= len(corr):
        top_idx = np.arange(len(corr))
        top_vals = corr.astype(float)
    else:
        top_idx = np.argsort(corr)[-top_m:]
        top_vals = corr[top_idx].astype(float)
    w = np.exp((top_vals - top_vals[-1]) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            h_idx = j_lo - k + n - 1
            grad[j_lo:j_hi+1] += wi * (1.0 - h[h_idx:h_idx + (j_hi - j_lo + 1)])
        j_lo2 = max(0, n - 1 - k)
        j_hi2 = n - 1
        if j_lo2 <= j_hi2:
            h_idx2 = j_lo2 + k - n + 1
            grad[j_lo2:j_hi2+1] -= wi * h[h_idx2:h_idx2 + (j_hi2 - j_lo2 + 1)]
    return grad

def project_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, target_sum):
    h_bin = (h >= 0.5).astype(float)
    excess = int(round(np.sum(h_bin) - target_sum))
    if excess > 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:excess]
        h_bin[ones[order]] = 0.0
    elif excess < 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:(-excess)]
        h_bin[zeros[order]] = 1.0
    return h_bin

def local_search_binary(h_bin, max_sweeps=200, time_limit=30.0):
    n = len(h_bin)
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        total = len(ones) * len(zeros)
        if total <= 20000:
            swaps = [(ii, jj) for ii in zeros for jj in ones]
            np.random.shuffle(swaps)
        else:
            n_try = 5000
            swaps = [(int(np.random.choice(zeros)), int(np.random.choice(ones))) for _ in range(n_try)]
        for i, j in swaps:

                continue
            h[i] = 1.0
            h[j] = 0.0
            c5_new = compute_c5(h)

                improved = True
            else:
                h[i] = 0.0
                h[j] = 1.0
        if not improved:

    return h, best_c5
