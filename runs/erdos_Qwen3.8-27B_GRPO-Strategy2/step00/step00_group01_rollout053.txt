def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def binarize(h):
    """Convert h to binary with correct sum."""
    n = len(h)
    target = n / 2
    # Round
    h_round = np.round(h)
    h_round = np.clip(h_round, 0, 1)

    if excess > 0:
        # Turn off some 1s
        idx = np.where(h_round == 1)[0]
        # Sort by h value (keep highest h values as 1)
        order = np.argsort(h[idx])[::-1]  # descending
        for k in range(int(np.ceil(excess))):

    elif excess < 0:
