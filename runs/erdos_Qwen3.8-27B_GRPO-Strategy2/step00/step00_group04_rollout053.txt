    if rng is None:

    # Ensure binary
    h = (h > 0.5).astype(float)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def make_binary(h, n, rng):
    """Convert to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, take top n/2
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1
    return h_bin

def pair_swap_optimize(h, n, max_iters, rng, improve_only=True):
    """Local search via pair swaps on binary vector."""
    h = make_binary(h, n, rng)
    best_val = compute_c5(h, n)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        i_idx = ones[i]
        j_idx = zeros[j]
        
        h[i_idx] = 0
        h[j_idx] = 1
        
        new_val = compute_c5(h, n)
        
        if new_val <= best_val:
            best_val = new_val
            ones[i] = j_idx
            zeros[j] = i_idx
        else:
            h[i_idx] = 1
            h[j_idx] = 0
    
    return h, best_val

def simulated_annealing(h, n, start_temp, end_temp, max_iters, rng):
    """SA on binary vector with pair swaps."""
    h = make_binary(h, n, rng)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    curr_val = best_val
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        temp = start_temp * (end_temp / start_temp) ** (it / max_iters)
        
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        i_idx = ones[i]
        j_idx = zeros[j]
        
        h[i_idx] = 0
        h[j_idx] = 1
        
        new_val = compute_c5(h, n)
        
        if new_val < curr_val or rng.random() < np.exp(-(new_val - curr_val) / max(temp, 1e-10)):
            curr_val = new_val
            ones[i] = j_idx
            zeros[j] = i_idx
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i_idx] = 1
            h[j_idx] = 0
    
    return best_h, best_val

def continuous_optimize(h0, n, max_iter=1000):
    """Use scipy to optimize continuous h with sum constraint."""
    def objective(x):
        c = np.correlate(x, 1 - x, mode="full")
        # Smoothed max
        tau = 0.01
        return np.log(np.sum(np.exp(c / tau))) * tau - np.log(len(c)) * tau
    
    def constraint_eq(x):
        return np.sum(x) - n / 2
    
    def constraint_eq_jac(x):
        return np.ones(n)
    
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]
    
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-8})
    return res.x, compute_c5(res.x, n)

def interpolate_f(h, n_old, n_new):
    """Interpolate from n_old to n_new points, preserving integral."""
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Add endpoint for proper interpolation
    x_old_ext = np.concatenate([x_old, [x_old[-1] + 2.0/n_old]])
    h_ext = np.concatenate([h, [h[0]]])  # periodic extension
    h_new = np.interp(x_new, x_old_ext, h_ext, period=2.0/n_old)
    # Fix integral
    h_new = np.clip(h_new, 0, 1)
    target_sum = n_new / 2
    h_new += (target_sum - np.sum(h_new)) / n_new
    h_new = np.clip(h_new, 0, 1)
    return h_new

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def make_binary(h, n, rng):
    h = np.array(h, dtype=float).copy()
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def greedy_hill_climb(h, n, max_sweeps=2000, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h = make_binary(h, n, rng)
    best_val = compute_c5(h, n)
    
    for sweep in range(max_sweeps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        best_delta = 1e-12
        best_oi, best_zj = -1, -1
        
        for oi in ones:
            for zj in zeros:
                h[oi] = 0
                h[zj] = 1
                val = compute_c5(h, n)
                h[oi] = 1
                h[zj] = 0
                delta = best_val - val
                if delta > best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zj = zj
        
        if best_oi >= 0:
            h[best_oi] = 0
            h[best_zj] = 1
            best_val = compute_c5(h, n)
            improved = True
        
        if not improved:
            break
    
    return h, best_val

def simulated_annealing(h, n, start_t=0.005, end_t=1e-7, max_iters=50000, rng=None):
    if rng is None:
        rng = np.random.default_rng(1)
    h = make_binary(h, n, rng)
    cur_val = compute_c5(h, n)
    best_h = h.copy()
    best_val = cur_val
    
    for it in range(max_iters):
        frac = it / max_iters
        temp = start_t * (end_t / start_t) ** frac
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        oi = ones[rng.integers(len(ones))]
        zj = zeros[rng.integers(len(zeros))]
        
        h[oi] = 0
        h[zj] = 1
        new_val = compute_c5(h, n)
        
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[oi] = 1
            h[zj] = 0
    
    return best_h, best_val

def continuous_refine(h0, n, max_iter=300):
    """SLSQP on smooth approximation of max correlation."""
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        # Smoothed max via log-sum-exp
        tau = 0.005
        m = np.max(c)
        return m + tau * np.log(np.sum(np.exp((c - m) / tau)))
    
    def obj_grad(x):
        # Numerical gradient (expensive but n is small)
        c = np.correlate(x, 1.0 - x, mode="full")
        tau = 0.005
        m = np.max(c)
        weights = np.exp((c - m) / tau) / np.sum(np.exp((c - m) / tau))
        # dc/dx for each component
        g = np.zeros(n)
        # ... this is complex, let's just use numerical
        eps = 1e-7
        for i in range(n):
            xp = x.copy()
            xp[i] += eps
            xm = x.copy()
            xm[i] -= eps
            # Need to maintain constraint... skip for now
            g[i] = (obj(xp) - obj(xm)) / (2 * eps)
        return g
    
    def constraint_eq(x):
        return np.sum(x) - n / 2
    
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    x = np.clip(res.x, 0, 1)
    # Project to constraint
    x += (n/2 - np.sum(x)) / n
    x = np.clip(x, 0, 1)
    return x, compute_c5(x, n)

def interpolate_resolution(h, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    # Periodic interpolation
    h_ext = np.concatenate([h[-10:], h, h[:10]])  # padding
    x_ext = np.concatenate([x_old[-10:] - 2.0/n_old, x_old, x_old[:10] + 2.0/n_old])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = np.clip(h_new, 0, 1)
    h_new += (n_new / 2 - np.sum(h_new)) / n_new
    h_new = np.clip(h_new, 0, 1)
    return h_new

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h
