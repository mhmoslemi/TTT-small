import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(c)

def smooth_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = np.max(c)
    return cm + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (c - cm))))

def smooth_gradient(h, alpha):
    n = len(h)
    eps = 1e-6
    grad = np.zeros(n)
    for j in range(n):
        hp = h.copy(); hp[j] += eps
        hm = h.copy(); hm[j] -= eps
        grad[j] = (smooth_objective(hp, alpha) - smooth_objective(hm, alpha)) / (2.0 * eps)
    return grad

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    result = np.clip(h - t, 0.0, 1.0)
    resid = target_sum - result.sum()
    if abs(resid) > 1e-12:
        free = (result > 1e-6) & (result < 1.0 - 1e-6)
        if free.sum() > 0:
            result[free] += resid / free.sum()
            result = np.clip(result, 0.0, 1.0)
        else:
            order = np.argsort(result) if resid > 0 else np.argsort(-result)
            for i in order:
                if resid > 0:
                    add = min(resid, 1.0 - result[i])
                    result[i] += add
                    resid -= add
                else:
                    sub = min(-resid, result[i])
                    result[i] -= sub
                    resid += sub
                if abs(resid) < 1e-12:
                    break
    return result

def pgd_step(h, alpha, lr, target_sum):
    g = smooth_gradient(h, alpha)
    h_new = h - lr * g
    h_new = project_box_sum(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values

    t0 = time.time()

    def elapsed():

    if h_init is None:

        h_init[::2] = 1.0
    h_init = project_box_sum(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Projected gradient descent, increasing alpha
    alphas = [5, 15, 30, 60, 120, 250, 500]
