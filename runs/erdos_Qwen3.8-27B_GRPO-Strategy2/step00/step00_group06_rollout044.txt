def compute_c5(h):

    return np.max(corr) * dx

def make_feasible(h, n):

    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

            if abs(diff) < 1e-12:

            idx = np.random.randint(n)
            h[idx] = np.clip(h[idx] + diff / n, 0.0, 1.0)
            diff = target - np.sum(h)
    return h

def objective_and_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    thresh = max_val - 1e-3
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    if len(active) == 0:

    for k in active:
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        j_range = i_range + n - 1 - k
        grad[i_range] += (1.0 - h[j_range])
        m_lo = max(0, i_lo + n - 1 - k)
        m_hi = min(n - 1, i_hi + n - 1 - k)
        if m_lo <= m_hi:
            m_range = np.arange(m_lo, m_hi + 1)
            i2 = m_range - n + 1 + k
            grad[m_range] -= h[i2]
    grad /= len(active)
    return max_val * dx, grad * dx

def slsqp_optimize(h0, n, maxiter=200, time_limit=60):

    target = n / 2.0
    t_start = time.time()
    best_x = h0.copy()

    def obj_and_grad(x):
        return objective_and_grad(x, len(x))

    def con_fun(x):
        return np.sum(x) - target

    constraints = [{'type': 'eq', 'fun': con_fun}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj_and_grad, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.fun < best_f:
            best_f = res.fun
            best_x = make_feasible(res.x, n)
            best_f = compute_c5(best_x)
    except Exception:
        pass
    return best_x, best_f

def coordinate_descent(h, n, steps=500, step_sizes=None, rng=None):
    if step_sizes is None:
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005]
    if rng is None:
        rng = np.random.RandomState(0)
    best_f = compute_c5(h)
    target = n / 2.0
    for step in step_sizes:
        for s in range(steps):
            idx = rng.randint(n)
            for direction in [1, -1]:

                h[idx] = np.clip(h_old + direction * step, 0.0, 1.0)
                # Adjust feasibility
