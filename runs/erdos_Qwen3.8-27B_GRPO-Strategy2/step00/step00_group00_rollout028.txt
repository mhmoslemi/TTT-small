import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1-h, mode='full') * dx))

def compute_corr(h, dx):
    return np.correlate(h, 1-h, mode='full') * dx

def smooth_max(corr, alpha):
    cmax = np.max(corr)
    return cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha

def compute_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')  # unscaled
    cmax = np.max(corr)
    exp_c = np.exp(alpha * (corr - cmax))
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    nm1 = n - 1
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - nm1
        # First term: (1-h[i-k]) for i where 0<=i-k<n
        is1 = max(0, k)
        ie1 = min(n, k + n)
        if is1 < ie1:
            grad[is1:ie1] += wk * (1.0 - h[is1-k:ie1-k])
        # Second term: -h[i+k] for i where 0<=i+k<n
        is2 = max(0, -k)
        ie2 = min(n, n - k)
        if is2 < ie2:
            grad[is2:ie2] -= wk * h[is2+k:ie2+k]
    
    return grad * dx  # Scale by dx since corr is multiplied by dx in C5

    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
