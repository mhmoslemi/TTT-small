def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def binary_sa(h_init, n, t_start, t_end, seed, T_init=0.05, T_final=0.0001):
    rng = np.random.RandomState(seed)
    cur = h_init.astype(float).copy()
    # Ensure exactly n//2 ones
    n_ones = int(np.round(np.sum(cur > 0.5)))
    cur[cur > 0.5] = 1.0
    cur[cur <= 0.5] = 0.0
    # Fix count if needed
    actual_ones = int(np.sum(cur))
    while actual_ones > n // 2:
        idx = rng.choice(np.where(cur == 1)[0])
        cur[idx] = 0
        actual_ones -= 1
    while actual_ones < n // 2:
        idx = rng.choice(np.where(cur == 0)[0])
        cur[idx] = 1
        actual_ones += 1
    
    cur_val = compute_c5(cur, n)
    best = cur.copy()
    best_val = cur_val
    
    while time.time() < t_end:
        frac = (time.time() - t_start) / max(t_end - t_start, 1e-6)
        frac = min(frac, 1.0)
        T = T_init * (T_final / T_init) ** frac
        
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        cur[i], cur[j] = 0.0, 1.0
        new_val = compute_c5(cur, n)
        delta = new_val - cur_val
        
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
    
    return best, best_val

    rng = np.random.RandomState(seed)
    x0 = h_init.astype(float).copy()
    
    def obj(x):
        h = np.clip(x, 0, 1)
        # Project to sum = n/2
        diff = n / 2 - np.sum(h)
        h = h + diff / n
        h = np.clip(h, 0, 1)
        # Final adjustment
        for _ in range(50):
            diff = n / 2 - np.sum(h)
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        return compute_c5(h, n)
    
    best_val = obj(x0)
    best_x = x0.copy()
    
    t_start = time.time()
    remaining = t_end - t_start
    
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                      options={'maxiter': int(remaining * 50), 'xatol': 1e-5, 'fatol': 1e-7,
                               'adaptive': True})
        new_val = obj(res.x)
        if new_val < best_val:
            best_val = new_val
            # Project best_x
            best_x = np.clip(res.x, 0, 1)
            diff = n / 2 - np.sum(best_x)
            best_x = best_x + diff / n
            best_x = np.clip(best_x, 0, 1)
    except:
        pass
    
    return best_x, best_val

def project_sum(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5

        s = np.sum(np.clip(h + mid, 0.0, 1.0))

def smooth_c5(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full') * dx
    m = np.max(corr)
    if T < 1e-10:
        return m
    shifted = (corr - m) / T
    max_sh = np.max(shifted)
    return m + T * (max_sh + np.log(np.sum(np.exp(shifted - max_sh))))

    return best, best_val

def nm_optimize(h_init, n, t_end, seed):
    def obj(x):
        h = project_sum(x, n)
        return smooth_c5(h, n, 0.04)
    x0 = h_init.astype(float).copy()
    best_val = compute_c5(project_sum(x0, n), n)
    best_x = project_sum(x0, n).copy()
    t_start = time.time()
    remaining = max(t_end - t_start, 1)
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': int(remaining * 80), 'xatol': 1e-5,
                                'fatol': 1e-7, 'adaptive': True})
        hv = project_sum(res.x, n)
        v = compute_c5(hv, n)
        if v < best_val:
            best_val = v
            best_x = hv.copy()
    except Exception:
        pass
    return best_x, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 4

    h_parent = np.clip(np.asarray(initial_h_values, dtype=float), 0.0, 1.0)
    n = len(h_parent)
    h_parent = project_sum(h_parent, n)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Nelder-Mead continuous refinement from parent
    if time.time() < deadline - 100:
        t1 = min(t0 + 120, deadline - 100)
        h1, v1 = nm_optimize(h_parent, n, t1, seed + 1)
        if v1 < best_c5:
            best_c5 = v1
            best_h = h1.copy()

    # Phase 2: Continuous SA from best
    if time.time() < deadline - 200:
        t2 = min(time.time() + 300, deadline - 200)
        h2, v2 = continuous_optimize(best_h, n, t2, seed + 10)
        if v2 < best_c5:
            best_c5 = v2
            best_h = h2.copy()

    # Phase 3: Binary SA from best's binary projection
    if time.time() < deadline - 150:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        t3 = min(time.time() + 250, deadline - 50)
        h3, v3 = binary_sa(h_bin, n, t3, seed + 20)
        if v3 < best_c5:
            best_c5 = v3
            best_h = h3.copy()

    # Phase 4: Binary SA from random start
    if time.time() < deadline - 200:
        rng_r = np.random.RandomState(seed + 99)
        h_rand = np.zeros(n)
        h_rand[:n // 2] = 1.0
        rng_r.shuffle(h_rand)
        t4 = min(time.time() + 250, deadline - 50)
        h4, v4 = binary_sa(h_rand, n, t4, seed + 100)
        if v4 < best_c5:
            best_c5 = v4
            best_h = h4.copy()

    # Phase 5: Final NM polish on best
    if time.time() < deadline - 10:
        t5 = deadline
        h5, v5 = nm_optimize(best_h, n, t5, seed + 200)
        if v5 < best_c5:
            best_c5 = v5
            best_h = h5.copy()

    # Phase 6: Brief continuous SA on final best
    if time.time() < deadline - 5:
        t6 = deadline
        h6, v6 = continuous_optimize(best_h, n, t6, seed + 300)
        if v6 < best_c5:
            best_c5 = v6
            best_h = h6.copy()

    return best_h, best_c5, len(best_h)
