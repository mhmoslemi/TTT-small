        h = h * (target / current)
        h = np.clip(h, 0, 1)

            h = h + diff / n
            h = np.clip(h, 0, 1)

def subgradient_c5(h, n):

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c = corr * dx

    mask = c == max_val
    if not np.any(mask):

    grad = np.zeros(n)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

def subgrad_obj(h, n):
    """Returns (objective, gradient) using softmax-smoothed max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c = corr * dx
    # Use the actual max for the objective
    obj = np.max(c)
    # Softmax weights for smooth gradient
    T = 0.01
    c_shifted = c - obj
    weights = np.exp(c_shifted / T)
    weights /= weights.sum()
    # Gradient of smoothed max
    # dc[j]/dh[k] where lag m = j - n + 1
    grad = np.zeros(n)
    for j in range(len(c)):
        w = weights[j]
        if w < 1e-10:
            continue
        m = j - n + 1
        # Vectorized computation of dc[m]/dh
        # dc[m]/dh[k] = (1-h[k+m])*[valid] - h[k-m]*[valid]
        idx_plus = np.arange(n) + m
        valid_plus = (idx_plus >= 0) & (idx_plus < n)
        idx_minus = np.arange(n) - m
        valid_minus = (idx_minus >= 0) & (idx_minus < n)
        term1 = np.zeros(n)
        term1[valid_plus] = 1 - h[idx_plus[valid_plus]]
        term2 = np.zeros(n)
        term2[valid_minus] = h[idx_minus[valid_minus]]
        grad += w * (term1 - term2)
    return obj, grad

def optimize_h(h_init, n, max_iters=5000, time_limit=300):
    """Subgradient descent with projection."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)[0]
    start = time.time()
    
    # Adaptive step size
    lr = 0.1 / n
    lr_decay = 0.999
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = subgrad_obj(h, n)
        h_new = h - lr * grad
        h_new = project(h_new, n)
        
        c5_new = evaluate_c5(h_new)[0]
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay
    
    return best_h, best_c5

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return h

def smooth_max_grad(h, n, T=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    obj = float(c.max())
    # Softmax weights
    w = np.exp((c - obj) / T)
    w /= w.sum()
    # Subgradient weighted by softmax
    grad = np.zeros(n)
    k_idx = np.arange(n)
    for j in range(len(c)):
        if w[j] < 1e-8:
            continue
        m = j - (n - 1)
        kp = k_idx + m
        km = k_idx - m
        t1 = np.where((kp >= 0) & (kp < n), 1.0 - h[np.clip(kp, 0, n - 1)], 0.0)
        t2 = np.where((km >= 0) & (km < n), h[np.clip(km, 0, n - 1)], 0.0)
        grad += w[j] * (t1 - t2)
    return obj, grad

def grad_desc(h0, n, time_limit, lr_init=0.5/n, max_iter=50000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0
    start = time.time()
    while t < max_iter and time.time() - start < time_limit:
        t += 1
        obj, grad = smooth_max_grad(h, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr = lr_init / (1 + 0.001 * t)
    return best_h, best_c5

            h = h_pert.copy()
            step = min(step * 1.05, 0.2)
        else:

            step = max(step, 0.001)

Let me optimize the gradient computation. Instead of looping over lags, I can use convolutions:

The gradient of c[j] w.r.t. h is:
- term_a: indicator of valid range for lag m
- term_b: h[k-m] for k in valid range
- term_c: h[k+m] for k+m in valid range

For term_a: it's a trapezoid shape (1 in the middle, 0 at edges). The weighted sum over all lags:
sum_j w[j] * term_a[j] = sum_j w[j] * 1_{max(0,m_j) ≤ k ≤ min(n-1, n-1+m_j)}

For each k: sum over j where k is in the valid range of lag m_j.
k is in [max(0,m), min(n-1, n-1+m)] iff:
- If m ≥ 0: m ≤ k ≤ n-1, i.e., m ≤ k and m ≤ n-1 (always true for m < n)
  So m ≤ k, i.e., j - (n-1) ≤ k, i.e., j ≤ k + n - 1
- If m < 0: 0 ≤ k ≤ n-1+m, i.e., k ≤ n-1+m, i.e., k ≤ n-1 + j - (n-1) = j, i.e., j ≥ k

So for a given k:
- Lags m ≥ 0: m ≤ k means j ≥ n-1 and j ≤ n-1+k. But j ranges from 0 to 2n-2.
  m = j-(n-1) ≥ 0 means j ≥ n-1. m ≤ k means j ≤ n-1+k.
  So j ∈ [n-1, min(2n-2, n-1+k)].
- Lags m < 0: k ≤ j-(n-1) means j ≥ k. m < 0 means j < n-1.
  So j ∈ [max(0, k), n-2].
  Wait: k ≤ n-1+m = n-1+j-(n-1) = j. So j ≥ k. And j < n-1 (for m<0).
  So j ∈ [max(0, k), n-2].

Combined: j ∈ [n-1, min(2n-2, n-1+k)] ∪ [max(0,k), n-2]

Hmm, this is getting complicated. For k in [0, n-1]:
- Positive m lags: j ∈ [n-1, n-1+k]
- Negative m lags: j ∈ [k, n-2]

So the total set of j for which k is in valid range:
{j: max(0,k) ≤ j ≤ n-2} ∪ {j: n-1 ≤ j ≤ min(2n-2, n-1+k)}
= [max(0,k), n-2] ∪ [n-1, n-1+k]

For k=0: [0, n-2] ∪ [n-1, n-1] = [0, n-1]
For k=n-1: [n-1, n-2] (empty) ∪ [n-1, 2n-2] = [n-1, 2n-2]

So term_a weighted sum at position k = sum_{j in valid set} w[j].

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return h

def subgrad(h, n):
    """Returns (C5 value, subgradient vector)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    obj = float(c.max())
    # Find active lags (within tolerance of max)
    tol = 1e-10
    active = np.where(c >= obj - tol)[0]
    # Use softmax for smoothness
    T = 0.005
    wa = np.exp((c - obj) / T)
    wa /= wa.sum()
    
    grad = np.zeros(n)
    k_idx = np.arange(n)
    for j in range(len(c)):
        wj = wa[j]
        if wj < 1e-8:
            continue
        m = j - (n - 1)
        i_lo = max(0, m)
        i_hi = min(n - 1, n - 1 + m)
        if i_lo > i_hi:
            continue
        # term_a: 1 in [i_lo, i_hi]
        term_a = np.zeros(n)
        term_a[i_lo:i_hi + 1] = 1.0
        # term_b: h[k-m] for k in [i_lo, i_hi]
        term_b = np.zeros(n)
        term_b[i_lo:i_hi + 1] = h[i_lo - m:i_hi - m + 1]
        # term_c: h[k+m] for k+m in [i_lo, i_hi], i.e., k in [i_lo-m, i_hi-m]
        c_lo = max(0, i_lo - m)
        c_hi = min(n - 1, i_hi - m)
        term_c = np.zeros(n)
        if c_lo <= c_hi:
            term_c[c_lo:c_hi + 1] = h[c_lo + m:c_hi + m + 1]
        grad += wj * (term_a - term_b - term_c)
    return obj, grad

def optimize(h0, n, time_limit, lr_scale=1.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    base_lr = lr_scale * 2.0 / n
    t = 0
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit:
        t += 1
        obj, grad = subgrad(h, n)
        m_t = b1 * m_t + (1 - b1) * grad
        v_t = b2 * v_t + (1 - b2) * grad * grad
        m_hat = m_t / (1 - b1**t)
        v_hat = v_t / (1 - b2**t)
        lr = base_lr / (1 + 0.005 * t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 5000:
            # Reshuffle
            h = h + 0.01 * np.random.randn(n)
            h = project(h, n)
            no_improve = 0
    return best_h, best_c5

def local_search(h0, n, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start = time.time()
    step = 0.03
    while time.time() - start < time_limit:
        # Perturb a random subset
        n_pert = max(1, n // 10)
        idx = np.random.choice(n, n_pert, replace=False)
        h_p = h.copy()
        h_p[idx] += step * np.random.randn(n_pert)
        h_p = project(h_p, n)
        c5_p = evaluate_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p
            h = h_p.copy()
            step *= 1.02
        else:
            step *= 0.97
        step = np.clip(step, 0.001, 0.3)
    return best_h, best_c5

def upscale(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_ext = np.concatenate([h, h[:1]])
    h_new = np.interp(x_new, x_old, h_ext)
    return project(h_new, new_n)
