                rem = budget_s * 0.93 - (time.time() - t0)
                max_it = int(min(rem * 400, 25000))
                if max_it > 100:
                    best_h, best_c5 = local_search(best_h, best_n, max_it, rng)

    # Phase 4: Final polish
    if time.time() - t0 < budget_s * 0.98:
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(obj_lse, best_h, args=(best_n, 1000, 1000.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 100, 'ftol': 1e-15})
            hf = repair_h(res.x, best_n)
            cvf = compute_c5(hf, best_n)
            if cvf < best_c5:
                best_h, best_c5 = hf, cvf
        except Exception:
            pass

    best_h = repair_h(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
