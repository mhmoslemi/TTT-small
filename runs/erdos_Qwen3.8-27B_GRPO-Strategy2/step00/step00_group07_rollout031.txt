import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def objective(h, dx):
    corr = compute_corr(h)
    return float(corr.max() * dx)

def subgrad_lag(h, d, n):
    g = np.ones(n)
    if d >= 0:
        g[:n-d] -= h[d:]
        g[d:] -= h[:n-d]
    else:
        dd = -d
        g[dd:] -= h[:n-dd]
        g[:dd] -= h[dd:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.zeros(80)
        h0[0::2] = 1.0

    best_c5 = objective(h0, 2.0/len(h0))

    target_ns.add(base_n)
    for m in [2, 3, 4, 6]:
        target_ns.add(base_n * m)
    target_ns = sorted(target_ns)
    
    cur_n = base_n
    cur_h = h0.copy()
    
    for n in target_ns:

            break
        
        if n == cur_n:

        else:

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(120):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _c5(h, dx):
    return float(_corr(h).max() * dx)

def _subg(h, d, n):
    g = np.ones(n)
    if d >= 0:
        if d < n:
            g[:n - d] -= h[d:]
            g[d:] -= h[:n - d]
    else:
        p = -d
        if p < n:
            g[p:] -= h[:n - p]
            g[:p] -= h[p:]
    return g

def _upsample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h, period=2.0)

def _optimize(h, n, dx, t0, budget_s, rs):
    corr = _corr(h)
    c5 = float(corr.max() * dx)
    best_h = h.copy()
    best_c5 = c5
    lr = 0.06
    muc = 0.9
    v = np.zeros(n)
    stale = 0
    T = 8.0
    while True:
        el = time.time() - t0
        if el > budget_s - 80:
            break
        corr = _corr(h)
        cmax = corr.max()
        w = np.exp(T * (corr - cmax))
        ws = w.sum()
        if ws < 1e-30:
            break
        top_k = np.argsort(corr)[-6:]
        g = np.zeros(n)
        for idx in top_k:
            d = idx - (n - 1)
            g += (w[idx] / ws) * _subg(h, d, n)
        v = muc * v - lr * g
        hn = _project(h + v, n)
        cn = _c5(hn, dx)
        if cn < best_c5 - 1e-10:
            h = hn
            best_c5 = cn
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale % 600 == 0:
                lr *= 0.93
                muc = min(0.96, muc + 0.01)
                T = max(3.0, T * 0.95)
            if stale > 4000:
                h = _project(h + rs.normal(0, 0.06, n), n)
                v = np.zeros(n)
                lr = 0.06
                muc = 0.9
                stale = 0
                T = 8.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rs = np.random.RandomState(seed)
    t0 = time.time()

    if initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.zeros(80)
        h0[0::2] = 1.0

    best_h = h0.copy()
    best_c5 = _c5(h0, 2.0 / len(h0))
    bn = len(h0)
    bh = h0.copy()

    for n in sorted({bn, 2 * bn, 4 * bn, 8 * bn}):
        if time.time() - t0 > budget_s - 60:
            break
        if n == bn:
            h = bh.copy()
        else:
            h = _upsample(bh, n)
            h = _project(h, n)
        dx = 2.0 / n
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h, c5_opt = _optimize(h, n, dx, t0, budget_s, rs)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h.copy()
        bh = h
        bn = n

    # Local refinement on best
    if time.time() - t0 < budget_s - 15:
        n = len(best_h)
        dx = 2.0 / n
        lb = best_c5
        step = 0.015
        for _ in range(5000):
            if time.time() - t0 > budget_s - 5:
                break
            d1 = rs.randint(n)
            d2 = rs.randint(n)
            s1 = rs.choice([-1, 1]) * step
            s2 = rs.choice([-1, 1]) * step
            hr = best_h.copy()
            hr[d1] = np.clip(hr[d1] + s1, 0, 1)
            hr[d2] = np.clip(hr[d2] + s2, 0, 1)
            hr = _project(hr, n)
            cr = _c5(hr, dx)
            if cr < lb:
                best_h = hr
                best_c5 = cr
                lb = cr
                step = max(0.003, step * 0.98)

    return best_h, best_c5, len(best_h)
