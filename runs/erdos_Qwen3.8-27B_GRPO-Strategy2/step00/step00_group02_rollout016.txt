def _smooth_obj(params, n, beta):
    h = np.clip(params, 0, 1)
    s = h.sum()
    target = n / 2.0
    # Soft projection of sum constraint
    h = h + (target - s) / n
    h = np.clip(h, 0, 1)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # Smooth max
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m))))

def _project(h, n):

    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        h = h + (target - s) / n
        h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_val(h, beta):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def _obj_main(x, n, beta, mu):
    h = np.clip(x, 0.0, 1.0)
    return _smooth_val(h, beta) + mu * (h.sum() - n / 2.0) ** 2

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    for _ in range(500):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        if not m.any():
            break
        h[m] = np.clip(h[m] + d / m.sum(), 0.0, 1.0)
    return h

def _cont_opt(h, n, beta, mu, maxiter, deadline):
    r = minimize(_obj_main, h, args=(n, beta, mu), method='L-BFGS-B',
                 bounds=[(0, 1)] * n, options={'maxiter': maxiter, 'ftol': 1e-10})
    if time.time() > deadline:
        return _proj(h, n)
    return _proj(r.x, n)
