from scipy.optimize import minimize
import time

_n_g = 42

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    rem = target - s
    if abs(rem) > 1e-11:
        order = np.argsort(h) if rem > 0 else np.argsort(-h)

            if abs(rem) < 1e-13:

            else:

def structured_start(n, rng, pi):

def upsample(h, n_old, n_new):
    h_new = np.zeros(n_new)
    for i in range(n_new):
        h_new[i] = h[min(i * n_old // n_new, n_old - 1)]
    return make_feasible(h_new, n_new)

    global _n_g
    _n_g = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_func, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        return make_feasible(res.x, n), float(eval_c5(make_feasible(res.x, n)))
