import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase some elements
            room = 1.0 - h
            mask = room > 1e-10
            if not mask.any():
                break
            # Distribute proportionally to room
            frac = diff / np.sum(room[mask])
            if frac <= 1.0:
                h[mask] += frac * room[mask]
            else:
                h[mask] = 1.0
        else:
            # Need to decrease some elements
            room = h
            mask = room > 1e-10
            if not mask.any():
                break
            frac = -diff / np.sum(room[mask])
            if frac <= 1.0:
                h[mask] -= frac * room[mask]
            else:
                h[mask] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    n = 47
    target_sum = n / 2.0  # 23.5
    dx = 2.0 / n
    
    # Get initial values
    h0 = np.array(initial_h_values, dtype=float)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_constraint(h0, target_sum)
    
    h_best = h0.copy()
    c5_best = compute_c5(h_best)
    
    elapsed = lambda: time.time() - start_time
    
    # Phase 1: Nelder-Mead local optimization
    def objective(h_flat):
        h = np.clip(h_flat, 0.0, 1.0)
        penalty = 100.0 * (np.sum(h) - target_sum) ** 2
        return compute_c5(h) + penalty
    
    if elapsed() < budget_s * 0.3:
        try:
            res = minimize(objective, h_best, method='Nelder-Mead',
                          options={'maxiter': 20000, 'xatol': 1e-10, 'fatol': 1e-12,
                                   'adaptive': True})
            h_cand = project_to_constraint(np.clip(res.x, 0, 1), target_sum)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                c5_best = c5_cand
                h_best = h_cand.copy()
        except Exception:
            pass
    
    # Phase 2: Powell method
    if elapsed() < budget_s * 0.5:
        try:
            res = minimize(objective, h_best, method='Powell',
                          options={'maxiter': 10000, 'xtol': 1e-10, 'ftol': 1e-12})
            h_cand = project_to_constraint(np.clip(res.x, 0, 1), target_sum)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                c5_best = c5_cand
                h_best = h_cand.copy()
        except Exception:
            pass
    
    # Phase 3: Differential evolution (global search)
    if elapsed() < budget_s * 0.75:
        bounds = [(0.0, 1.0)] * n
        try:
            res = differential_evolution(objective, bounds, seed=seed,
                                        maxiter=200, popsize=15,
                                        tol=1e-10, mutation=(0.3, 1.0),
                                        recombination=0.9,
                                        polish=True,
                                        updating='deferred')
            h_cand = project_to_constraint(np.clip(res.x, 0, 1), target_sum)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                c5_best = c5_cand
                h_best = h_cand.copy()
        except Exception:
            pass
    
    # Phase 4: Local search with adaptive step
    if elapsed() < budget_s * 0.9:
        step = 0.05
        for i in range(50000):
            if elapsed() > budget_s * 0.9:
                break
            h_pert = h_best + rng.normal(0, step, n)

            if c5_pert < c5_best:

                h_best = h_pert.copy()

    c5_best = compute_c5(h_best)

    h0 = np.clip(np.asarray(initial_h_values, dtype=float), 0.0, 1.0)
    h0 = enforce_constraint(h0, ts)
    h_best = h0.copy()
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Nelder-Mead on smooth objective
    if elapsed() < budget_s * 0.12:
        try:
            r = minimize(obj_smooth_fn, h_best, args=(ts, dx, 40.0),
                        method='Nelder-Mead',
                        options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
            hc = enforce_constraint(np.clip(r.x, 0, 1), ts)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc
        except Exception:
            pass

    # Phase 2: Nelder-Mead on exact objective
    if elapsed() < budget_s * 0.30:
        try:
            r = minimize(obj_fn, h_best, args=(ts, dx),
                        method='Nelder-Mead',
                        options={'maxiter': 15000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
            hc = enforce_constraint(np.clip(r.x, 0, 1), ts)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc
        except Exception:
            pass

    # Phase 3: Powell
    if elapsed() < budget_s * 0.45:
        try:
            r = minimize(obj_fn, h_best, args=(ts, dx),
                        method='Powell',
                        options={'maxiter': 2000, 'xtol': 1e-12, 'ftol': 1e-14})
            hc = enforce_constraint(np.clip(r.x, 0, 1), ts)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc
        except Exception:
            pass

    # Phase 4: Differential evolution
    if elapsed() < budget_s * 0.60:
        bounds = [(0.0, 1.0)] * n
        try:
            r = differential_evolution(obj_fn, bounds, args=(ts, dx), seed=seed,
                                       maxiter=100, popsize=15,
                                       tol=1e-8, mutation=(0.4, 1.0),
                                       recombination=0.8, initializing='latinhypercube',
                                       seed=seed)
            hc = enforce_constraint(np.clip(r.x, 0, 1), ts)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc
        except Exception:
            pass

    # Phase 5: Local stochastic search
    if elapsed() < budget_s * 0.82:
        step = 0.04
        no_improve = 0
        for i in range(500000):
            if elapsed() > budget_s * 0.82:
                break
            h_p = h_best + rng.normal(0, step, n)
            h_p = enforce_constraint(h_p, ts)
            c5p = compute_c5(h_p)
            if c5p < c5_best:
                c5_best = c5p
                h_best = h_p
                step = min(step * 1.02, 0.12)
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 3000:
                    step = max(step * 0.7, 0.003)
                    no_improve = 0

    # Phase 6: Coordinate descent
    if elapsed() < budget_s * 0.92:
        for round_num in range(20):
            if elapsed() > budget_s * 0.92:
                break
            improved = False
            for i in range(n):
                if elapsed() > budget_s * 0.92:
                    break
                h_try = h_best.copy()
                h_try[i] = rng.uniform(0, 1)
                h_try = enforce_constraint(h_try, ts)
                c5t = compute_c5(h_try)
                if c5t < c5_best:
                    c5_best = c5t
                    h_best = h_try
                    improved = True
                # Try a few more random values for this coordinate
                for _ in range(5):
                    h_try[i] = rng.uniform(0, 1)
                    h_try = enforce_constraint(h_try, ts)
                    c5t = compute_c5(h_try)
                    if c5t < c5_best:
                        c5_best = c5t
                        h_best = h_try
                        improved = True
            if not improved and round_num > 3:
                break

    # Phase 7: Final Nelder-Mead polish
    if elapsed() < budget_s * 0.97:
        try:
            r = minimize(obj_fn, h_best, args=(ts, dx),
                        method='Nelder-Mead',
                        options={'maxiter': 8000, 'xatol': 1e-14, 'fatol': 1e-16, 'adaptive': True})
            hc = enforce_constraint(np.clip(r.x, 0, 1), ts)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc
        except Exception:
            pass

    h_best = enforce_constraint(h_best, ts)
    c5_best = compute_c5(h_best)
    return (h_best, float(c5_best), n)
