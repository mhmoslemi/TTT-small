def repair_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    if h.shape[0] != n:
       # interpolate? but should not. If too short, pad; too long, truncate.
       if h.shape[0] < n:
           hp=np.zeros(n); hp[:h.shape[0]]=h; h=hp
       else: h=h[:n].copy()
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    diff=target-h.sum()
    if abs(diff)>1e-14:
       if diff>0:
          order=np.argsort(1.0-h)
          amount=diff
          for idx in order:
             room=1.0-h[idx]
             if room<=0: continue
             add=room if room<amount else amount
             h[idx]+=add; amount-=add
             if amount<=1e-14: break
       else:
          order=np.argsort(-h)
          amount=-diff
          for idx in order:
             rem=h[idx]
             if rem<=0: continue

import time
import math
import numpy as np
from scipy import optimize as spopt

_OPT_N = 0
_OPT_BETA = 1.0
_CACHE_H = None
_CACHE_VALS = None
_CACHE_N = 0
_BEST_H = None
_BEST_C = 1e100
_BEST_N = 0


def repair_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.shape[0] != n:
        if h.shape[0] < n:
            hp = np.zeros(n, dtype=float)
            hp[:h.shape[0]] = h
            h = hp
        else:
            h = h[:n].copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
            amount = diff
            for idx in order:
                room = 1.0 - h[idx]
                if room <= 0.0:
                    continue
                add = room if room < amount else amount
                h[idx] += add
                amount -= add
                if amount <= 1e-14:
                    break
        else:
            order = np.argsort(-h)
            amount = -diff
            for idx in order:
                rem = h[idx]
                if rem <= 0.0:
                    continue
                sub = rem if rem < amount else amount
                h[idx] -= sub
                amount -= sub
                if amount <= 1e-14:
                    break
    slack = target - h.sum()
    if abs(slack) > 1e-15:
        if slack > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = min(1.0, max(0.0, h[idx] + slack))
        slack = target - h.sum()
        if abs(slack) > 1e-15:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size:
                h[free] += slack / float(free.size)
                h = np.clip(h, 0.0, 1.0)
                slack = target - h.sum()
        if abs(slack) > 1e-12:
            h = np.full(n, target / float(n), dtype=float)
    return h


def lag_vals(h, n):
    vals = np.empty(2 * n - 1, dtype=float)
    om = 1.0 - h
    vals[0] = float(np.dot(h, om))
    idx = 1
    for s in range(1, n):
        vals[idx] = float(np.dot(h[:n - s], om[s:]))
        vals[idx + 1] = float(np.dot(h[s:], om[:n - s]))
        idx += 2
    return vals


def eval_c5_raw(h):
    n = len(h)
    if n == 0:
        return 0.0
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def update_best(h, n):
    global _BEST_H, _BEST_C, _BEST_N
    n = int(n)
    if n <= 0:
        return
    if n % 2 != 0:
        n -= 1
    if n <= 0:
        return
    h = repair_feasible(h, n)
    c = eval_c5_raw(h)
    if np.isfinite(c) and c < _BEST_C - 1e-13:
        _BEST_C = float(c)
        _BEST_H = h.copy()
        _BEST_N = n


def _smooth_common(h_in):
    global _CACHE_H, _CACHE_VALS, _CACHE_N
    n = _OPT_N
    beta = float(_OPT_BETA)
    h = np.asarray(h_in, dtype=float).copy()
    if _CACHE_N == n and _CACHE_H is not None and np.array_equal(_CACHE_H, h):
        vals = _CACHE_VALS
    else:
        vals = lag_vals(h, n)
        _CACHE_H = h.copy()
        _CACHE_VALS = vals
        _CACHE_N = n
    m = float(vals.max())
    e = np.exp(beta * (vals - m))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(vals)
        mask = vals >= (m - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt <= 0:
            p[0] = 1.0
            cnt = 1
        else:
            p[mask] = 1.0 / cnt
        f = m
    else:
        p = e / s
        f = m + math.log(s) / beta
    return h, p, f


def smooth_gradient(p, h, n):
    g = np.zeros(n, dtype=float)
    p0 = float(p[0])
    if p0 > 0.0:
        g += p0 * (1.0 - 2.0 * h)
    idx = 1
    for s in range(1, n):
        pA = float(p[idx])
        pB = float(p[idx + 1])
        idx += 2
        if pA > 0.0:
            g[:n - s] += pA * (1.0 - h[s:])
            g[s:] -= pA * h[:n - s]
        if pB > 0.0:
            g[s:] += pB * (1.0 - h[:n - s])
            g[:n - s] -= pB * h[s:]
    return g


def _opt_fun(h):
    return _smooth_common(h)[2]


def _opt_jac(h):
    h_arr, p, _ = _smooth_common(h)
    return smooth_gradient(p, h_arr, _OPT_N)

    return float(np.sum(np.asarray(h, dtype=float)) - 0.5 * _OPT_N)
