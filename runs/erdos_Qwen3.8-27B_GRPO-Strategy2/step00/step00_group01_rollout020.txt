    m = np.max(corr)

def soft_max_gradient(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

                best_c5, best_h = c5_r, h_r

    # Phase 3: SLSQP polish
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.5 and n0 <= 200:
        def obj_fn(x):
            return compute_c5(x)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target0}]
        bnds = [(0.0, 1.0)] * n0
        try:
            res = minimize(obj_fn, best_h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-10})
            if res.x is not None:
                h_s = project_feasible(res.x, target0)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5, best_h = c5_s, h_s
        except Exception:
            pass

    # Phase 4: Multi-resolution refinement
    current_n = len(best_h)
    for target_n in [94, 188, 376, 752]:
        if target_n < current_n or target_n > 800:
            continue
        if time.time() - t0 > budget_s * 0.85:
            break
        x_old = np.linspace(0, 2, current_n)
        x_new = np.linspace(0, 2, target_n)
        h_interp = np.interp(x_new, x_old, best_h)
        target_new = target_n / 2.0
        h_interp = project_feasible(h_interp, target_new)

        remaining = budget_s - (time.time() - t0)
        tl = max(remaining * 0.25, 10.0)
        max_it = max(500, int(30.0 * target_n / np.sqrt(target_n)))
        h_new, c5_new = pgd_optimize(h_interp, target_new, max_it, 0.03, 15.0, 120.0, tl, time.time())

        if c5_new < best_c5:
            best_c5, best_h, current_n = c5_new, h_new, target_n

    # Final PGD polish at best resolution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.95:
        n_fin = len(best_h)
        target_fin = n_fin / 2.0
        best_h, best_c5 = pgd_optimize(best_h, target_fin, 5000, 0.02, 30.0, 200.0,
                                        budget_s - (time.time() - t0) - 2.0, t0)

    # Ensure feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
