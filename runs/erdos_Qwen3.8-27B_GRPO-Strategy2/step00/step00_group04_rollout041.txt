import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    cores = np.correlate(h, 1.0 - h, mode="full")
    return np.max(cores) * dx

def get_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_soft_max(h, n, t=0.01):
    vals = get_vals(h, n)
    m = np.max(vals)
    w = np.exp((vals - m) / t)
    w /= np.sum(w)
    grad = np.zeros(n)
    kmax = n - 1
    for p in range(2 * n - 1):
        k = p - kmax
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            s = -k
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
        grad += w[p] * g
    return grad

def grad_max(h, n):
    vals = get_vals(h, n)
    idx = np.argmax(vals)
    k = idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        s = -k
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def make_initial(n, pattern="alternating"):
    if pattern == "alternating":
        h = np.zeros(n)
        h[::2] = 1.0
    elif pattern == "blocks":
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif pattern == "random":
        h = np.random.rand(n)
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def gradient_descent(h0, n, max_iter, t0, t1, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    t_curr = t0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        t_s = t_curr * (t1 / t0) ** (it / max(1, max_iter))
        g = grad_soft_max(h, n, max(t_s, 1e-4))
        # Adaptive step size
        lr = 0.1
        for _ in range(20):
            h_new = project_feasible(h - lr * g, n)
            v_new = compute_c5(h_new, n)
            if v_new < compute_c5(h, n):
                break
            lr *= 0.7
        h = project_feasible(h - lr * g, n)
        v = compute_c5(h, n)
        if v < best_val:
            best_val = v
            best_h = h.copy()
    return best_h, best_val

def slsqp_polish(h0, n, max_iter=300):
    h = h0.copy()
    def obj(p):
        return compute_c5(p, n)
    def constraint(p):
        return np.sum(p) - n / 2.0
    def cons_jac(p):
        return np.ones(n) / n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constraint, 'jac': cons_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or res.fun < obj(h):
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = make_initial(42, "alternating")
    n0 = len(h_init)
    h_init = project_feasible(h_init, n0)

    best_h = h_init.copy()
    best_val = compute_c5(best_h, n0)
    best_n = n0

    def update_best(h, n):
        nonlocal best_h, best_val, best_n
        v = compute_c5(h, n)
        if v < best_val:
            best_val = v
            best_h = h.copy()
            best_n = n
        return v

    # Phase 1: Gradient descent at n0 (first 25% time)
    t_phase1 = t_start + budget_s * 0.25
    h1, v1 = gradient_descent(h_init, n0, 3000, 0.05, 0.001, t_phase1)
    update_best(h1, n0)

    # Phase 2: SLSQP polish at n0 (next 10%)
    t_phase2 = t_start + budget_s * 0.35
    if time.time() < t_phase2:
        h2 = slsqp_polish(h1, n0, max_iter=200)
        update_best(h2, n0)
        h1 = h2.copy()

    # Phase 3: Upsample to 2*n0 and optimize (next 25%)
    t_phase3 = t_start + budget_s * 0.60
    if time.time() < t_phase3:
        n1 = n0 * 2
        h_up = upsample(h1, n0, n1)
        h3, v3 = gradient_descent(h_up, n1, 2000, 0.03, 0.001, t_phase3)
        update_best(h3, n1)

    # Phase 4: SLSQP at higher resolution (next 10%)
    t_phase4 = t_start + budget_s * 0.70
    if time.time() < t_phase4:
        h4 = slsqp_polish(h3, n1, max_iter=200)
        update_best(h4, n1)
        h3 = h4.copy()

    # Phase 5: Try alternative starting points (next 15%)
    t_phase5 = t_start + budget_s * 0.85
    if time.time() < t_phase5:
        for pattern in ["random", "blocks", "alternating"]:
            if time.time() >= t_phase5:
                break
            h_alt = make_initial(n0, pattern)
            if pattern == "random":
                h_alt = np.random.rand(n0)
            h_alt = project_feasible(h_alt, n0)
            h_a, v_a = gradient_descent(h_alt, n0, 1500, 0.04, 0.002, t_phase5)
            update_best(h_a, n0)
            h_a = slsqp_polish(h_a, n0, max_iter=100)
            update_best(h_a, n0)

    # Phase 6: Final polish on best solution (remaining time)
    if time.time() < t_end - 5:
        h_best = best_h.copy()
        # Downsample or use as-is
        n_best = len(h_best)
        h_best = project_feasible(h_best, n_best)
        h_f, v_f = gradient_descent(h_best, n_best, 1000, 0.02, 0.0005, t_end - 10)
        update_best(h_f, n_best)
        if time.time() < t_end - 5:
            h_f2 = slsqp_polish(h_f, n_best, max_iter=100)
            update_best(h_f2, n_best)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_val = compute_c5(best_h, best_n)
    # Ensure sum is exactly n/2
    s = np.sum(best_h)
    if abs(s - best_n / 2.0) > 1e-10:
        best_h = project_feasible(best_h, best_n)
        final_val = compute_c5(best_h, best_n)

    return (best_h.tolist(), final_val, best_n)
