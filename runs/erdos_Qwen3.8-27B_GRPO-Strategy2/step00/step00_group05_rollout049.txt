    return float(np.max(corr)) * dx

    return m + float(np.log(np.sum(np.exp(T * (vals - m)))) ) / T

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(vals, T):
    m = float(np.max(vals))
    return m + float(np.log(np.sum(np.exp(T * (vals - m))))) / T

def penalty_obj(h, n, T, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj_val = smooth_max(corr * dx, T)
    s = float(np.sum(h))
    return obj_val + lam * (s - n / 2.0) ** 2
