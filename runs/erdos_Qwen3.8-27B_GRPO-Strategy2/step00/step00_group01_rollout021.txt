from scipy.optimize import minimize


def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):

    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-14:

            for i in np.argsort(h)[::-1]:
                d = min(h[i], excess)
                h[i] -= d

        else:
            deficit = target - s
            for i in np.argsort(h):
                d = min(1.0 - h[i], deficit)
                h[i] += d
                deficit -= d
                if deficit < 1e-15:
                    break
    return h


def nm_optimize(h_start, n, maxiter=5000, fatol=1e-13):
    def obj(x):
        return compute_c5(project_feasible(x, n))
    res = minimize(obj, h_start, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10,
                            'fatol': fatol, 'adaptive': True})
    return project_feasible(res.x, n)


def make_binary_init(n, rng, num_ones):
    h = np.zeros(n)
    ones = rng.choice(n, size=num_ones, replace=False)
    h[ones] = 1.0
    return project_feasible(h, n)


def make_interval_init(n, m, rng):
    xs = (np.arange(n) + 0.5) * 2.0 / n
    h = np.zeros(n)
    total_w = 1.0
    widths = rng.dirichlet(np.ones(m)) * total_w
    positions = np.sort(rng.uniform(0.2, 1.8, m))
    for i in range(m):
        a = positions[i] - widths[i] / 2.0
        b = positions[i] + widths[i] / 2.0
        h[(xs >= max(a, 0)) & (xs < min(b, 2))] = 1.0
    return project_feasible(h, n)
