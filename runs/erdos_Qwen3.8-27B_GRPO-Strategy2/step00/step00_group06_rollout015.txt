def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def project_feasible(h, target_sum):

import numpy as np
import time
from scipy.optimize import minimize as sp_min


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())


def project_feasible(h, target):
    h = np.asarray(h, dtype=np.float64)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    mc = c.max()
    scaled = alpha * (c - mc)
    ms = scaled.max()
    exps = np.exp(scaled - ms)
    Z = exps.sum()
    w = exps / Z
    val = mc + (ms + np.log(Z)) / alpha
    grad = np.zeros(n)
    for j in range(n):
        k1lo = max(0, j - n + 1)
        k1hi = min(2 * n - 2, j)
        if k1lo <= k1hi:
            idx = j - np.arange(k1lo, k1hi + 1) + n - 1
            grad[j] += np.dot(w[k1lo:k1hi + 1], 1.0 - h[idx])
        k2lo = max(0, n - 1 - j)
        k2hi = min(2 * n - 2, 2 * n - 2 - j)
        if k2lo <= k2hi:
            idx = j + np.arange(k2lo, k2hi + 1) - n + 1
            grad[j] -= np.dot(w[k2lo:k2hi + 1], h[idx])
    grad *= dx
    return val, grad


def subgrad_opt(h0, n, n_iters, lr0, a0, a1, deadline):
    h = h0.copy()
    tgt = n / 2.0
    best = h.copy()

    stale = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, n_iters))
        alpha = a0 + (a1 - a0) * frac
        lr = lr0 / (1.0 + 0.05 * it)
        _, g = obj_grad(h, n, alpha)
        h = project_feasible(h - lr * g, tgt)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 3000 and it < n_iters - 50:
            h = project_feasible(h + 0.005 * np.random.randn(n), tgt)
            stale = 0
    return best, best_c5


def grid_project(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    hn = np.interp(xn, xo, h)
    return project_feasible(hn, n_new / 2.0)


def try_snap(h, n, levels):
    """Try snapping values to discrete levels, keeping feasibility"""
    tgt = n / 2.0
    h_snap = np.zeros(n)
    for i in range(n):
        h_snap[i] = levels[np.argmin(np.abs(levels - h[i]))]
    h_snap = project_feasible(h_snap, tgt)
    return h_snap


def polish_lbfgs(h0, n, deadline):
    """Extra polish using L-BFGS-B on smooth objective with projection"""
    tgt = n / 2.0
    alpha = 50.0

    def obj_wrapper(x):
        h = project_feasible(x, tgt)
        v, _ = obj_grad(h, n, alpha)
        return v

    x0 = h0.copy()

        h_new = project_feasible(res.x, tgt)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h0, compute_c5(h0)
