import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_limit = budget_s - 12.0

    # Initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(80)
        h0[:40] = 1.0

    n = len(h0)
    target_ones = n // 2

    # Binary initialization
    h = (h0 > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target_ones:
        idx = np.where(h == 1.0)[0]
        h[rng.choice(idx, n1 - target_ones, replace=False)] = 0.0
    elif n1 < target_ones:
        idx = np.where(h == 0.0)[0]
        h[rng.choice(idx, target_ones - n1, replace=False)] = 1.0

    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()

    # Phase 1: Simulated Annealing
    T = 0.05
    T_min = 1e-5
    dx = 2.0 / n
    p1_end = t_start + (t_limit - t_start) * 0.50
    no_improve = 0

    while time.time() < p1_end:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0

        if delta <= 0.0 or (T > T_min and rng.random() < np.exp(-delta / T)):

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

            h[j] = 0.0

                c5_t = c5_new
                if c5_t < best_c5:

                    best_h = h_tr.copy()
            else:

    h = best_h.copy()
