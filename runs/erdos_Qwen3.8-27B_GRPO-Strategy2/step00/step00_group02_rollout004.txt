    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)

    best_h = project_h(h_init, n)
    best_val = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': make_constraint(n)}]

    # Phase 1: SLSQP with multiple alphas and restarts
    alphas = [200.0, 100.0, 50.0, 25.0, 10.0]
    for alpha in alphas:
        if rem() < 40:
            break
        n_restarts = 4
        for r in range(n_restarts):
            if rem() < 15:
                break
            if r == 0:
                h0 = best_h.copy()
            else:
                scale = 0.015 + 0.01 * r
                h0 = best_h + rng.randn(n) * scale
                h0 = project_h(h0, n)
            res = minimize(
                lambda h, a=alpha: soft_max_obj(h, n, a),
                h0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 600, 'ftol': 1e-14}
            )
            h_new = project_h(res.x, n)
            val = compute_c5(h_new, n)
            if val < best_val - 1e-15:
                best_val = val
                best_h = h_new.copy()

    # Phase 2: High-alpha polish on best solution
    if rem() > 30:
        for alpha in [500.0, 2000.0]:
            if rem() < 15:
                break
            res = minimize(
                lambda h, a=alpha: soft_max_obj(h, n, a),
                best_h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 400, 'ftol': 1e-15}
            )
            h_new = project_h(res.x, n)
            val = compute_c5(h_new, n)
            if val < best_val - 1e-15:
                best_val = val
                best_h = h_new.copy()

    # Phase 3: Subgradient / coordinate refinement on active lags
    if rem() > 60:
        h_cur = best_h.copy()
        dx = 2.0 / n
        for it in range(200):
            if rem() < 15:
                break
            corr = np.correlate(h_cur, 1.0 - h, mode="full") * dx
            max_c = np.max(corr)
            active = corr > max_c - 1e-6
            # Subgradient step
            lag_shift = n - 1
            grad = np.zeros(n)
            for k in np.where(active)[0]:
                lag = k - lag_shift
                for j in range(n):
                    g = 0.0
                    i1 = j - lag
                    if 0 <= i1 < n:
                        g += (1.0 - h_cur[i1])
                    i2 = j + lag
                    if 0 <= i2 < n:
                        g += -h_cur[i2]
                    grad[j] += g
            grad *= dx / np.sum(active)
            lr = 1e-4 / (np.max(np.abs(grad)) + 1e-15)
            h_cur = h_cur - lr * grad
            h_cur = project_h(h_cur, n)
            val = compute_c5(h_cur, n)
            if val < best_val - 1e-15:
                best_val = val
                best_h = h_cur.copy()

    # Phase 4: Try higher resolution n*2
    n2 = n * 2
    if rem() > 150:
        h2 = np.repeat(best_h, 2)
        h2 = project_h(h2, n2)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': make_constraint(n2)}]
        for alpha in [50.0, 20.0, 10.0]:
            if rem() < 15:
                break
            res = minimize(
                lambda h, a=alpha: soft_max_obj(h, n2, a),
                h2, method='SLSQP', bounds=bounds2, constraints=cons2,
                options={'maxiter': 400, 'ftol': 1e-13}
            )
            h2_new = project_h(res.x, n2)
            val2 = compute_c5(h2_new, n2)
            if val2 < best_val - 1e-15:
                best_val = val2
                best_h = h2_new

    # Phase 5: Final SLSQP polish at whatever n we have
    n_final = len(best_h)
    if rem() > 20:
        bounds_f = [(0.0, 1.0)] * n_final
        cons_f = [{'type': 'eq', 'fun': make_constraint(n_final)}]
        res = minimize(
            lambda h: soft_max_obj(h, n_final, 1000.0),
            best_h, method='SLSQP', bounds=bounds_f, constraints=cons_f,
            options={'maxiter': 300, 'ftol': 1e-15}
        )
        h_new = project_h(res.x, n_final)
        val = compute_c5(h_new, n_final)
        if val < best_val - 1e-15:
            best_val = val
            best_h = h_new

    final_n = len(best_h)
    best_h = project_h(best_h, final_n)
    best_val = compute_c5(best_h, final_n)
    return best_h, best_val, final_n
