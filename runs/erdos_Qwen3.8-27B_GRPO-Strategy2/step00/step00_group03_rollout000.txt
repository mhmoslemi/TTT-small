        iters = per_iter_check
        h2, c52 = local_search_binary(h, n, iters, seed + int(time.time() * 1000) % 100000)
        elapsed += time.time() - t0
        if c52 < best_c5 - 1e-12:
            best_c5 = c52
            best_h = h2.copy()
    return best_h, best_c5

def resample_to_n(h_src, n, seed):
    if len(h_src) == n:
        return h_src.copy().astype(float)
    xs = np.linspace(0, 1, len(h_src))
    h = np.interp(np.linspace(0, 1, n), xs, h_src)
    h = np.clip(h, 0, 1)
    return enforce_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = 0

    # Check for initial_h_values
    init_avail = 'initial_h_values' in globals() and initial_h_values is not None
    init_h = np.array(initial_h_values, dtype=float) if init_avail else None

    n_list = [70, 90, 100, 110, 140, 160, 200, 250]

    for n in n_list:
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.82:
            break
        time_left = budget_s * 0.85 - elapsed
        if time_left < 15:
            break

        # Initialize
        if init_h is not None:
            h0 = resample_to_n(init_h, n, seed)
        else:
            h0 = make_balanced_binary(n, seed + n)

        # Continuous optimization (use most of time_left)
        cont_time = min(time_left * 0.6, 300)
        if cont_time > 10:
            h_c, c5_c = optimize_continuous(h0, n, cont_time, seed + 100 * n)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                best_n = n
            h0 = h_c
        else:
            h0 = enforce_sum(h0, n)

        # Binary local search
        time_left2 = budget_s * 0.85 - (time.time() - t_start)
        if time_left2 > 10:
            h_bin = (h0 > 0.5).astype(float)
            h_bin = enforce_sum(h_bin, n)
            # Quick local search first
            quick_iters = min(int(time_left2 * 30), 20000)
            h_ls, c5_ls = local_search_binary(h_bin, n, quick_iters, seed + 5000 + n)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n

            # Multi-start refinement if time
            time_left3 = budget_s * 0.85 - (time.time() - t_start)
            if time_left3 > 15:
                h_ms, c5_ms = multi_start_local(h_ls, n, min(time_left3, 120), seed + 9000 + n)
                if c5_ms < best_c5:
                    best_c5 = c5_ms
                    best_h = h_ms.copy()
                    best_n = n

    # Final: try one more pass on best_n with remaining time
    if best_h is not None:
        time_left = budget_s * 0.9 - (time.time() - t_start)
        if time_left > 10:
            n = best_n
            h_c, c5_c = optimize_continuous(best_h, n, min(time_left, 100), seed + 7777)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                best_n = n
    else:
        # Fallback
        n = 70
        best_h = make_balanced_binary(n, seed)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final enforcement and evaluation
    best_h = enforce_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
