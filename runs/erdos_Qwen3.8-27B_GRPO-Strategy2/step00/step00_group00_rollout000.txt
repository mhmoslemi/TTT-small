import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_obj(x, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    s = beta * corr
    s -= s.max()
    m = s.max()
    return (m + np.log(np.exp(s).sum())) / beta


def enforce_sum(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / N
            h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, N, beta=60.0, maxiter=400):
    dx = 2.0 / N
    target = N / 2.0
    h0 = enforce_sum(h0.copy(), N)

    def f(x):
        return smooth_max_obj(x, dx, beta)

    def g(x):
        return x.sum() - target

    def gj(x):
        return np.ones(N)

    res = minimize(f, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * N,
                   constraints=[{'type': 'eq', 'fun': g, 'jac': gj}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h = enforce_sum(res.x, N)
    return h, eval_c5(h, dx)


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.ones(60) * 0.5

    N_best = len(h_best)
    c5_best = eval_c5(h_best, 2.0 / N_best)

    # Phase 1: Optimize at initial resolution with multiple betas
    for beta in [40.0, 80.0, 120.0]:
        if time.time() - t0 > budget_s - 120:
            break
        try:
            h_new, c5_new = slsqp_optimize(h_best, N_best, beta=beta, maxiter=500)
            if c5_new < c5_best:
                h_best, c5_best = h_new, c5_new
        except Exception:
            pass

    # Phase 2: Try increasing resolution
    N_list = []
    n_try = 50
    while n_try <= 200:
        N_list.append(n_try)
        n_try += 10
    for N in N_list:
        if time.time() - t0 > budget_s - 60:
            break
        if N == N_best:
            continue
        x_old = np.linspace(0.0, 1.0, N_best)
        x_new = np.linspace(0.0, 1.0, N)
        h_start = np.interp(x_new, x_old, h_best)
        h_start = enforce_sum(h_start, N)
        try:
            h_new, c5_new = slsqp_optimize(h_start, N, beta=60.0, maxiter=400)
            if c5_new < c5_best:
                h_best, c5_best, N_best = h_new, c5_new, N
        except Exception:
            pass

    # Phase 3: Multiple random restarts at best N
    restarts = 0
    while time.time() - t0 < budget_s - 40 and restarts < 15:
        restarts += 1
        N = N_best
        # Random Dirichlet start
        w = rng.gamma(2.0, size=N)
        h_rand = w / w.sum() * N * 0.5
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = enforce_sum(h_rand, N)
        # Also try perturbation of best
        if restarts % 2 == 0:
            h_rand = h_best + rng.normal(0, 0.05, N)
            h_rand = enforce_sum(h_rand, N)
        try:
            h_new, c5_new = slsqp_optimize(h_rand, N, beta=80.0, maxiter=300)
            if c5_new < c5_best:

        except Exception:
            pass

    # Phase 4: Final polish with high beta
    if time.time() - t0 < budget_s - 20:

        except Exception:
            pass

    c5_final = eval_c5(h_best, 2.0 / N_final)
    return h_best, c5_final, N_final
