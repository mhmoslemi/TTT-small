import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    scaled = alpha * (vals - m)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    return m + np.log(np.sum(np.exp(scaled))) / alpha

def obj_smooth(x, n_pts, alpha):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * (2.0 / n_pts)
    return smooth_max(vals, alpha)

def obj_true(x, n_pts):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n_pts))

def make_valid(h, n_pts):
    """Project h onto [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n_pts / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    """Interpolate h from n_src points to n_dst points over [0,2]"""
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src, endpoint=False) + np.linspace(0, 2, n_src, endpoint=False)[0] / n_src  # midpoints
    # Actually, let's use a simpler approach
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    from scipy.interpolate import interp1d
    f = interp1d(x_src, h_src, kind='linear', fill_value='extrapolate')
    h_dst = f(x_dst)
    return make_valid(h_dst, n_dst)

def optimize_hlsqp(h0, n_pts, maxiter=300, alpha=40):
    """Optimize using SLSQP with smooth max objective"""
    target = n_pts / 2.0
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n_pts)}]
    
    def obj(x):
        return obj_smooth(x, n_pts, alpha)
    
    def obj_grad(x):
        # Numerical gradient
        eps = 1e-6
        g = np.zeros(n_pts)
        for i in range(n_pts):
            xp = x.copy()
            xp[i] += eps
            xm = x.copy()
            xm[i] -= eps
            g[i] = (obj(xp) - obj(xm)) / (2 * eps)
        return g
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_opt = make_valid(res.x, n_pts)
        return h_opt, obj_true(h_opt, n_pts)
    except Exception:
        return h0, obj_true(h0, n_pts)

def optimize_hlbfgs(h0, n_pts, maxiter=500, alpha=40):
    """Optimize using L-BFGS-B with penalty for sum constraint"""
    target = n_pts / 2.0
    bounds = [(0.0, 1.0)] * n_pts
    penalty = 100.0
    
    def obj(x):
        return obj_smooth(x, n_pts, alpha) + penalty * (np.sum(x) - target) ** 2
    
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = make_valid(res.x, n_pts)
        return h_opt, obj_true(h_opt, n_pts)
    except Exception:
        return h0, obj_true(h0, n_pts)

def random_init(n_pts, seed):
    rng = np.random.RandomState(seed)
    # Random with sum constraint
    h = rng.uniform(0, 1, n_pts)
    return make_valid(h, n_pts)

def block_init(n_pts, n_blocks, seed):
    """Create a block-structured initial guess"""
    rng = np.random.RandomState(seed)
    h = np.zeros(n_pts)
    block_size = n_pts // n_blocks
    for i in range(n_blocks):
        val = rng.choice([0.0, 1.0, 0.5])
        h[i*block_size:(i+1)*block_size] = val
    return make_valid(h, n_pts)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    # Get initial values

    try:

    except NameError:

    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Try multiple optimization strategies
    attempts = []
    
    # 1. Optimize at initial resolution
    n_list = [n_init]
    # Add a few other resolutions
    for n_try in [64, 96, 128, 160]:
        if n_try != n_init and n_try <= 300:
            n_list.append(n_try)

            h0 = h_init.copy()
        else:

        if c5 < best_c5:

            best_h = h_opt

        if time.time() - t0 > budget_s * 0.85:

        # Try L-BFGS-B
