    if diff > 0:
        idx = np.where(hb == 1)[0]
        rng.shuffle(idx)
        hb[idx[:diff]] = 0
    elif diff < 0:
        idx = np.where(hb == 0)[0]
        rng.shuffle(idx)
        hb[idx[:(-diff)]] = 1
    return hb

def binary_sa(h_bin, rng, t_max_start, cooling, n_iter, t_floor):
    n = len(h_bin)
    c5_cur = compute_c5(h_bin)
    best_local = c5_cur
    best_local_h = h_bin.copy()
    for it in range(n_iter):
        t = t_max_start * (cooling ** it) + t_floor
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h_bin[i] = 0
        h_bin[j] = 1
        c5_new = compute_c5(h_bin)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / t):
            c5_cur = c5_new
            if c5_cur < best_local:
                best_local = c5_cur
                best_local_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return best_local_h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        h_init = np.full(42, 0.5)

    best_h = h_init.copy().astype(np.float64)
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: COBYLA smooth optimization at multiple temperatures
    if time.time() - t0 < budget_s * 0.2:
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': sum_constraint}]
        for temp in [0.01, 0.004, 0.002]:
            if time.time() - t0 > budget_s * 0.2:
                break
            res = minimize(softmax_obj, best_h, args=(temp, n), method='COBYLA',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 1500, 'rhobeg': 0.06})
            h_new = normalize(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 2: Coordinate descent
    if time.time() - t0 < budget_s * 0.4:
        h = best_h.copy()
        cur_c5 = best_c5
        for sweep in range(400):
            if time.time() - t0 > budget_s * 0.4:
                break
            improved = False
            for i in range(n):
                old = h[i]
                h[i] = min(1.0, old + 0.25)
                c5_try = compute_c5(h)
                if c5_try < cur_c5:
                    cur_c5 = c5_try
                    improved = True
                else:
                    h[i] = max(0.0, old - 0.25)
                    c5_try = compute_c5(h)
                    if c5_try < cur_c5:
                        cur_c5 = c5_try
                        improved = True
                    else:
                        h[i] = old
            h = normalize(h, n)
            if not improved:
                break
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        best_h = normalize(best_h, n)
        best_c5 = compute_c5(best_h)

    # Phase 3: Binary SA from smoothed solution
    if time.time() - t0 < budget_s * 0.6:
        h_bin = make_binary(best_h, rng)
        h_new, c5_new = binary_sa(h_bin, rng, 0.4, 0.9997, 200000, 0.0003)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Random restart + SA
    if time.time() - t0 < budget_s * 0.75:
        for restart in range(3):
            if time.time() - t0 > budget_s * 0.75:
                break
            h_rand = rng.random(n)
            h_rand = normalize(h_rand, n)
            h_bin = make_binary(h_rand, rng)
            h_new, c5_new = binary_sa(h_bin, rng, 0.3, 0.9998, 100000, 0.0002)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: Multi-start binary SA from perturbed binary
    if time.time() - t0 < budget_s * 0.85:
        h_bin0 = (best_h > 0.5).astype(np.float64)
        for attempt in range(5):
            if time.time() - t0 > budget_s * 0.85:
                break
            h_pert = h_bin0.copy()
            for _ in range(rng.integers(3, 10)):
                oi = np.where(h_pert == 1)[0]
                zi = np.where(h_pert == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                h_pert[oi[rng.integers(len(oi))]] = 0
                h_pert[zi[rng.integers(len(zi))]] = 1
            h_new, c5_new = binary_sa(h_pert, rng, 0.5, 0.9996, 150000, 0.0004)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 6: Upsample x2 and COBYLA
    if time.time() - t0 < budget_s * 0.95:
        n2 = n * 2
        h_up = np.repeat(best_h, 2)
        bounds2 = [(0.0, 1.0)] * n2
        constraints2 = [{'type': 'eq', 'fun': sum_constraint}]
        res2 = minimize(softmax_obj, h_up, args=(0.004, n2), method='COBYLA',
                       bounds=bounds2, constraints=constraints2,
                       options={'maxiter': 800, 'rhobeg': 0.04})
        h_new = normalize(res2.x, n2)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final: polish with short SA on best binary
    if time.time() - t0 < budget_s * 0.99:
        h_fin = (best_h > 0.5).astype(np.float64)
        target = len(h_fin) // 2
        diff = int(np.sum(h_fin)) - target
        if diff > 0:
            idx = np.where(h_fin == 1)[0]
            rng.shuffle(idx)
            h_fin[idx[:diff]] = 0
        elif diff < 0:
            idx = np.where(h_fin == 0)[0]
            rng.shuffle(idx)
            h_fin[idx[:(-diff)]] = 1
        h_new, c5_new = binary_sa(h_fin, rng, 0.15, 0.9999, 80000, 0.0001)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_h = normalize(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
