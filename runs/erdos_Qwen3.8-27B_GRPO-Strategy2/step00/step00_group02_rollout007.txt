    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lse_obj(hv, n, beta):
    dx = 2.0 / n
    corr = np.correlate(hv, 1.0 - hv, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def sum_con(hv, n):
    return np.sum(hv) - n / 2.0

def do_optimize(h_init, n, beta, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    try:
        res = minimize(lse_obj, h_init, args=(n, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_init, n)

def sa_step(h_bin, n, rng):
    np_ones = n // 2
    i = rng.randint(n)
    if h_bin[i] > 0.5:
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) == 0:
            return None
        j = zeros[rng.randint(len(zeros))]
    else:
        ones = np.where(h_bin == 1)[0]
        if len(ones) == 0:
            return None
        j = ones[rng.randint(len(ones))]
    h_new = h_bin.copy()
    h_new[i], h_new[j] = 1.0 - h_new[i], 1.0 - h_new[j]
    return h_new

def time_left(start, budget):
    return budget - (time.time() - start)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.ones(44) / 2.0
    n0 = len(h0)

    best_h = project_feasible(h0, n0)
    best_c5 = eval_c5(best_h, n0)

    # Phase 1: Continuous optimization over multiple n
    n_list = sorted(set([n0, 30, 32, 44, 48, 56, 60, 64, 80, 96]))
    for n in n_list:
        if time_left(start_time, budget_s) < 20:
            break
        if n == n0:
            h_init = h0.copy()
        else:
            h_init = np.interp(
                np.linspace(0, 2, n, endpoint=False),
                np.linspace(0, 2, n0, endpoint=False), h0)
        h_init = project_feasible(h_init, n)

        for beta in [30, 100, 300]:
            if time_left(start_time, budget_s) < 10:
                break
            h_opt = do_optimize(h_init, n, beta, maxiter=200)
            c5v = eval_c5(h_opt, n)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt.copy()
            h_init = h_opt.copy()

        # One random restart per n
        if time_left(start_time, budget_s) < 10:
            break
        h_rand = project_feasible(rng.random(n) * 0.8 + 0.1, n)
        h_opt = do_optimize(h_rand, n, 80, maxiter=100)
        c5v = eval_c5(h_opt, n)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt.copy()

    # Phase 2: Binary simulated annealing for even n
    even_ns = []
    for n in [n0, 44, 48, 56, 60, 64]:
        if n % 2 == 0 and n not in even_ns:
            even_ns.append(n)
    even_ns = [n for n in even_ns if n <= 120]

    for n in even_ns:
        if time_left(start_time, budget_s) < 15:
            break
        # Initialize binary from best
        if len(best_h) == n:
            h_bin = (best_h > 0.5).astype(float)
        else:
            hi = np.interp(
                np.linspace(0, 2, n, endpoint=False),
                np.linspace(0, 2, len(best_h), endpoint=False), best_h)
            h_bin = (hi > 0.5).astype(float)
        # Fix count
        np_ones = n // 2
        while int(np.sum(h_bin)) > np_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.randint(len(idx))]] = 0
        while int(np.sum(h_bin)) < np_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.randint(len(idx))]] = 1

        c5_sa = eval_c5(h_bin, n)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_bin.copy()

        temp = 0.012
        sa_deadline = min(start_time + budget_s * 0.88, time.time() + 120)
        while time.time() < sa_deadline and temp > 0.0004:
            h_new = sa_step(h_bin, n, rng)
            if h_new is None:
                break
            c5_new = eval_c5(h_new, n)
            delta = c5_new - c5_sa
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-6)):
                h_bin = h_new
                c5_sa = c5_new
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_bin.copy()
            temp *= 0.9996

    # Phase 3: Final continuous polish on best solution
    if time_left(start_time, budget_s) > 15:
        n_f = len(best_h)
        for beta in [500, 2000]:
            if time_left(start_time, budget_s) < 8:
                break
            h_opt = do_optimize(best_h.copy(), n_f, beta, maxiter=400)
            c5v = eval_c5(h_opt, n_f)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = eval_c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
