def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(_corr(h)) * dx)


def _project(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / float(np.sum(free))
            h = np.clip(h, 0.0, 1.0)
    return h


def _sg_topk(h, n, top_k):
    c = _corr(h)
    ti = np.argsort(c)[::-1][:top_k]
    tv = c[ti]

            ip = i + k

    return g
