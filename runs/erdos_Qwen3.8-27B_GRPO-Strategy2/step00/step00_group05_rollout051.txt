from scipy.optimize import minimize
import time

_OPT_T = 50.0
_OPT_PEN = 1000.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1-h, mode='full')) * 2.0 / n)

def project_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    m = corr.max()
    soft = (m + np.log(np.sum(np.exp(corr - m))) / _OPT_T) * dx
    return soft + _OPT_PEN * (h.sum() - n/2.0)**2

def lbgfb(h, target, T, maxfun=3000):
    global _OPT_T, _OPT_PEN
    _OPT_T = T
    _OPT_PEN = 1000.0
    n = len(h)
    try:
        r = minimize(_obj, h.copy(), method='L-BFGS-B',
                     bounds=[(0.,1.)]*n,
                     options={'maxiter':500,'maxfun':maxfun,'ftol':1e-15})
        hp = project_feasible(r.x, target)
        return hp, compute_c5(hp)
    except:
        return h, compute_c5(h)

def coord_desc(h, target, iters=200):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    steps = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    for it in range(iters):
        imp = False
        for i in range(n):
            best, bc = h, c5
            for dv in steps:
                for sign in (1, -1):
                    v = h[i] + sign*dv
                    if v < 0 or v > 1: continue
                    cand = h.copy()
                    cand[i] = v
                    cand = project_feasible(cand, target)
                    c = compute_c5(cand)
                    if c < bc:
                        best, bc = cand, c
            if bc < c5:
                h, c5 = best, bc
                imp = True
        if not imp:
            break
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

_OPT_T = 50.0
_OPT_PEN = 1000.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)

def project_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    soft = (m + np.log(np.sum(np.exp(corr - m))) / _OPT_T) * dx
    return soft + _OPT_PEN * (h.sum() - n / 2.0) ** 2

def lbgfb(h, target, T, maxfun=3000):
    global _OPT_T, _OPT_PEN
    _OPT_T = T
    _OPT_PEN = 1000.0
    n = len(h)
    try:
        r = minimize(_obj, h.copy(), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 500, 'maxfun': maxfun, 'ftol': 1e-15})
        hp = project_feasible(r.x, target)
        return hp, compute_c5(hp)
    except Exception:
        return h, compute_c5(h)

def coord_desc(h, target, iters=100, steps=None):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    if steps is None:
        steps = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    for it in range(iters):
        imp = False
        for i in range(n):
            best, bc = h, c5
            for dv in steps:
                for sgn in (1.0, -1.0):
                    v = h[i] + sgn * dv
                    if v < -1e-12 or v > 1.0 + 1e-12:
                        continue
                    cand = h.copy()
                    cand[i] = v
                    cand = project_feasible(cand, target)
                    c = compute_c5(cand)
                    if c < bc:
                        best, bc = cand, c
            if bc < c5:
                h, c5 = best, bc
                imp = True
        if not imp:
            break
    return h, c5

def make_start(n, tgt, idx, seed):
    rng = np.random.RandomState(seed)
    if idx == 0:
        return np.full(n, 0.5)
    elif idx == 1:
        return np.full(n, 0.5) + rng.randn(n) * 0.3
    elif idx == 2:
        h = np.zeros(n)
        h[rng.choice(n, size=int(tgt), replace=False)] = 1.0
        return h
    elif idx == 3:
        x = np.arange(n, dtype=float)
        h = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2) + \
            np.exp(-0.5 * ((x - 0.7 * n) / (0.12 * n)) ** 2)
        return h
    elif idx == 4:
        x = np.arange(n, dtype=float)
        h = np.zeros(n)
        for c in [0.15, 0.45, 0.8] * n:
            h += np.exp(-0.5 * ((x - c) / (0.06 * n)) ** 2)
        return h
    else:
        h = np.zeros(n)
        for j in range(max(1, n // 15)):
            c = rng.uniform(0, n)
            w = rng.uniform(0.03, 0.15) * n
            h += np.exp(-0.5 * ((x - c) / w) ** 2)
        return h

    global _OPT_T, _OPT_PEN

    np.random.seed(seed)
    n = 70
    tgt = n / 2.0

    else:

    bc = compute_c5(bh)

    # Phase 1: Annealed L-BFGS-B on best solution
    for T in [5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 500.0, 1000.0]:
        if time.time() - t0 > budget_s * 0.35:
            break
        h, c = lbgfb(bh, tgt, T, maxfun=3000)
        if c < bc:
            bh, bc = h, c
        bh, c2 = coord_desc(bh, tgt, iters=30, steps=[0.05, 0.02, 0.01, 0.005])
        if c2 < bc:

    # Phase 2: Many diverse restarts
    for r in range(50):
        if time.time() - t0 > budget_s * 0.85:
            break
        h0 = make_start(n, tgt, r, seed + r * 997)
        h0 = project_feasible(h0, tgt)
        h, c = lbgfb(h0, tgt, 100.0, maxfun=1500)
        h, c = coord_desc(h, tgt, iters=80)
        if c < bc:

            # Extra refinement on new best
            h, c = lbgfb(bh, tgt, 300.0, maxfun=2000)

    bc = compute_c5(bh)
    return bh, bc, n
