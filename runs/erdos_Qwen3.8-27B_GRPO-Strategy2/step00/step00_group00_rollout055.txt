            else:
                stall += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
        if stall >= 6000:
            break
    return best_h, best_c5


def directed_search(h, n, rng, max_iter, dx):
    """Focus swaps on reducing the worst shift."""
    cur = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    stall = 0
    for _ in range(max_iter):
        if not ones or not zeros:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = int(np.argmax(corr))
        offset = n - 1
        # Find a swap that reduces the term at k_star
        improved = False
        for _tries in range(10):
            ai = int(rng.integers(len(ones)))
            bi = int(rng.integers(len(zeros)))
            a, b = ones[ai], zeros[bi]
            h[a] = 0.0
            h[b] = 1.0
            new_c5 = compute_c5(h, dx)
            if new_c5 <= cur - 1e-14:
                cur = new_c5
                ones[ai], zeros[bi] = b, a
                improved = True
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stall = 0
                break
            else:
                h[a] = 1.0
                h[b] = 0.0
        if improved:
            pass
        else:
            stall += 1
        if stall >= 2000:
            break
    return best_h, best_c5


def soft_obj(h, n, alpha=20.0, pen=15.0):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr * dx
    vm = float(v.max())
    sm = vm + np.log(float(np.sum(np.exp(v - vm)))) / alpha
    sv = float(np.sum(h)) - n / 2.0
    return sm + pen * sv * sv


def cont_opt(h0, n, max_iter=30):
    h0 = np.clip(h0, 0.01, 0.99)
    res = minimize(soft_obj, h0, args=(n,), method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-6, 'fatol': 1e-8})
    return np.clip(res.x, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    h_par = None
    try:
        h_par = np.asarray(initial_h_values, dtype=float)
        if h_par.size < 2:
            h_par = None
    except Exception:
        h_par = None

    n_list = []
    if h_par is not None:
        n_par = len(h_par)
        n_list.append(n_par)
    for nv in [60, 80, 100, 140, 200]:
        if nv not in n_list and nv % 2 == 0:
            n_list.append(nv)
    n_list.sort()

    for n in n_list:
        if time.time() > deadline - 50:
            break
        dx = 2.0 / n
        n_half = n // 2

        cands = []
        if h_par is not None:
            h_up = adjust_sum(upsample_h(h_par, n), n)
            cands.append(round_to_01(h_up, n))
        cands.append(random_01(n, rng))
        cands.append(random_01(n, rng))
        # Block structure
        if n >= 8:
            nblk = 4
            bs = n // nblk
            h_blk = np.zeros(n)
            for bi in range(0, n, 2 * bs):
                end = min(bi + bs, n)
                h_blk[bi:end] = 1.0
            cands.append(round_to_01(adjust_sum(h_blk, n), n))

        for hc in cands:
            if time.time() > deadline - 15:
                break
            remaining = deadline - time.time()
            iters = int(min(10000, max(2000, remaining * 30)))
            try:
                h_ls, c5_ls = local_search_01(hc.copy(), n, rng, iters, dx)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()
            except Exception:
                pass

            # Run SA from the local search result
            if time.time() < deadline - 20:
                sa_iters = int(min(5000, max(1000, (deadline - time.time()) * 15)))
                try:
                    h_sa, c5_sa = sa_local_search(h_ls.copy(), n, rng, sa_iters, dx)
                    if c5_sa < best_c5:
                        best_c5 = c5_sa
                        best_h = h_sa.copy()
                except Exception:
                    pass

        # If best is from this n, try directed search
        if best_h is not None and len(best_h) == n and time.time() < deadline - 20:
            try:
                h_dir, c5_dir = directed_search(best_h.copy(), n, rng, 5000, dx)
                if c5_dir < best_c5:
                    best_c5 = c5_dir
                    best_h = h_dir.copy()
            except Exception:
                pass

    # Final: try to refine with continuous + round
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        dx = 2.0 / n
        for _ in range(4):
            if time.time() > deadline - 5:
                break
            hp = best_h + rng.normal(0, 0.04, n)
            hp = np.clip(hp, 0.0, 1.0)
            hp = adjust_sum(hp, n)
            try:
                h_co = cont_opt(hp, n, max_iter=25)
                h_co = adjust_sum(h_co, n)
                c5_co = compute_c5(h_co, dx)
                if c5_co < best_c5:
                    best_c5 = c5_co
                    best_h = h_co.copy()
                h_rd = round_to_01(h_co, n)
                h_ls2, c5_ls2 = local_search_01(h_rd, n, rng, 5000, dx)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()
            except Exception:
                pass

    if best_h is None:
        best_h = random_01(100, rng)
        best_c5 = compute_c5(best_h, 0.02)

    best_h = adjust_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
