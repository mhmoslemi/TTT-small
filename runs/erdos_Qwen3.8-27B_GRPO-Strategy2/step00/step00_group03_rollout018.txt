def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_h(h, n_points):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Proportional adjustment on elements not at boundary
    if diff > 0:
        mask = h < 1.0 - 1e-6
        n_mask = np.sum(mask)
        if n_mask == 0:
            return h
        step = diff / n_mask
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    else:
        mask = h > 1e-6
        n_mask = np.sum(mask)
        if n_mask == 0:
            return h
        step = diff / n_mask
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

    if abs(diff) > 1e-14:

        else:
            idx = int(np.argmax(h))
