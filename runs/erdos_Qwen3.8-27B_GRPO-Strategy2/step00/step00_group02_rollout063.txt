Let me index it carefully. If d ranges from -(n-1) to n-1, and for index j, the contribution from shift d is h[j-d] (when valid), then:

This is a convolution (or cross-correlation) of h with w!

Similarly, sum_d weights[d] * h[j+d] = (h * w_flip)[j] where w_flip[d] = w[-d].

So the gradient is:

    c = corr - np.max(corr)
    weights = np.exp(T * c)
    weights /= weights.sum()

    # lag d = k - (n-1), so d ranges from -(n-1) to n-1
    # weights[k] corresponds to lag d = k-(n-1)
    
    # Gradient for lag d:
    # sg[j] = dx * [(1-h[j-d]) * 1_{valid} - h[j+d] * 1_{valid}]
    # 
    # Part 1: sum_d weights[d] * (1-h[j-d]) for valid j-d
    #   = (1-h) convolved with w (where w is indexed by d, flipped for convolution)
    #   
    # Part 2: sum_d weights[d] * h[j+d] for valid j+d
    #   = h convolved with w_flip
    
    # Build weight vector for convolution
    # np.convolve(a, b)[k] = sum_i a[i] * b[k-i]
    # We want: out[j] = sum_d w[d] * h[j-d] = sum_d w[d] * h[j+d'] where d' = -d
    # This is np.convolve(h, w_shifted) where w_shifted is set up properly
    
    # Let's use FFT for efficiency
    # Actually for n=45, direct computation is fast enough
    
    # Let me just do it directly
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        # Part 1: (1-h[j-d]) for valid j-d = 0,...,n-1
        # j ranges from max(0,d) to min(n-1, n-1+d)
        j_start = max(0, d)
        j_end = min(n, n + d)  # exclusive
        if j_start < j_end:
            grad[j_start:j_end] += w * (1.0 - h[j_start - d:j_end - d]) * dx
        # Part 2: -h[j+d] for valid j+d = 0,...,n-1
        # j ranges from max(0,-d) to min(n-1, n-1-d)
        j_start2 = max(0, -d)

    return c5_val, grad

import numpy as np
from scipy.optimize import minimize
import time

_N = 45
_T_sm = 100.0
_T0 = 0.0
_BT = 1100.0

def _rem():
    return _BT - (time.time() - _T0)

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _sog(h, T):
    """Smooth objective and gradient."""
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    s = T * (c - cm)
    e = np.exp(s)
    val = cm + np.log(e.sum()) / T
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1); wk = w[k]
        if wk < 1e-14: continue
        a, b = max(0, d), min(n, n + d)
        if a < b: g[a:b] += wk * (1.0 - h[a-d:b-d]) * dx
        a2, b2 = max(0, -d), min(n, n - d)
        if a2 < b2: g[a2:b2] -= wk * h[a2+d:b2+d] * dx
    return val, g

def _prj(h, n):
    t = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = t - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            m = h < 1.0 - 1e-14
            if not m.any(): break
            h[m] += diff / m.sum()
        else:
            m = h > 1e-14
            if not m.any(): break
            h[m] += diff / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _sf(x):
    return _sog(x, _T_sm)[0]

def _sg(x):

    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def _sog(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    s = T * (c - cm)
    e = np.exp(s)
    val = cm + np.log(e.sum()) / T
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        a = max(0, d)
        b = min(n, n + d)
        if a < b:
            g[a:b] += wk * (1.0 - h[a - d:b - d]) * dx
        a2 = max(0, -d)
        b2 = min(n, n - d)
