```python
import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] -= diff / nf
        h = np.clip(h, 0, 1)
    return h

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def subgradient(h, s):
    """Subgradient of shift s correlation w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        t1 = 1.0 - (h[i+s] if 0 <= i+s < n else 0.0)
        t2 = h[i-s] if 0 <= i-s < n else 0.0
        g[i] = t1 - t2
    return g

def gradient_step(h, s, alpha):
    g = subgradient(h, s)
    h_new = h - alpha * g
    return h_new

def local_search(h, n, time_limit, rng):
    """Optimize h on grid of size n"""
    best_h = h.copy()
    best_c5 = compute_c5_val(h)
    start = time.time()
    alpha = 0.1  # initial step size
    dx = 2.0 / n
    
    it = 0
    while time.time() - start < time_limit:
        corr = compute_corr(h)
        # Find worst shift (max correlation)
        worst_idx = np.argmax(corr)
        s = worst_idx - (n - 1)
        
        # Also consider: minimize by maximizing the negative
        # Take step
        g = subgradient(h, s)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5_val(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            # Increase alpha slightly
            alpha = min(alpha * 1.02, 0.5)
        else:
            h = h_new  # still move, since we're doing subgradient
            alpha = max(alpha * 0.95, 0.001)
        
        # Random perturbation occasionally
        if it % 50 == 0:
            pert = rng.normal(0, alpha * 0.5, size=n)
            h_pert = project_feasible(h + pert, n)
            c5_pert = compute_c5_val(h_pert)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
                h = h_pert
        
        it += 1
    
    return best_h, best_c5
```

```python
                best_h, best_c5, best_n = h_opt1, c5_opt1, n1

    # Phase 3: Final polish on best grid
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        t3 = min(remaining * 0.85, 200.0)
        if t3 > 5:
            h_opt2, c5_opt2 = gradient_descent(best_h, best_n, t3, rng)
            if c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2, c5_opt2, best_n

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
```
