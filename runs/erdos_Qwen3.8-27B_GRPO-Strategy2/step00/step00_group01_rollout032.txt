    xs2 = np.linspace(lo, hi, 201)
    lv2 = a[:, None] * xs2[None, :] + b[:, None]
    lv2[c0_idx, :] = -1e30
    lin_max2 = np.max(lv2, axis=0)
    quad2 = b[c0_idx] + xs2 * (1.0 - xs2)
    g2 = np.maximum(lin_max2, quad2)
    idx2 = int(np.argmin(g2))
    return float(xs2[idx2])


def coord_descent(h, max_sweeps=5000, time_limit=60.0):
    N = len(h)
    target_sum = N / 2.0
    h = np.array(h, dtype=float)
    h = project_sum(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    c0_idx = N - 1
    k_all = np.arange(2 * N - 1) - (N - 1)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        max_change = 0.0
        for j in range(N):
            old_val = h[j]
            h[j] = 0.0
            b = np.correlate(h, 1.0 - h, mode='full')
            valid_first = (j + k_all >= 0) & (j + k_all < N)
            valid_second = (j - k_all >= 0) & (j - k_all < N)
            a = np.zeros(2 * N - 1)
            a[valid_first] = 1.0 - h[j + k_all[valid_first]]
            a[valid_second] = a[valid_second] - h[j - k_all[valid_second]]
            a[c0_idx] = 0.0
            new_val = solve_1d_subproblem(a, b, c0_idx)
            h[j] = new_val
            max_change = max(max_change, abs(new_val - old_val))

        h = project_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if max_change < 1e-13:
            break

    return best_h, best_c5
