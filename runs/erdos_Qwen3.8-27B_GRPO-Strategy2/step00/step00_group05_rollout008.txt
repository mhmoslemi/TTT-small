import numpy as np

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def proj_sum(h, target, n):
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def smooth_max_obj(h_vec, n, T):
    dx = 2.0 / n
    corr = np.correlate(h_vec, 1.0 - h_vec, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def smooth_max_grad(h_vec, n, T):
    dx = 2.0 / n
    corr = np.correlate(h_vec, 1.0 - h_vec, mode='full') * dx
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / np.sum(e)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        shift = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-8:
            continue
        if shift >= 0:
            i1 = np.arange(n - shift)
            g[i1] += wi * (1.0 - h_vec[i1 + shift])
            g[i1 + shift] -= wi * h_vec[i1]
        else:
            s = -shift
            i1 = np.arange(s, n)
            g[i1 - s] += wi * (1.0 - h_vec[i1])
            g[i1] -= wi * h_vec[i1 - s]
    g -= np.mean(g)
    return g

def subgrad_step(h, n, eta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        i1 = np.arange(n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        g[i1 + shift] -= h[i1]
    else:
        s = -shift
        i1 = np.arange(s, n)
        g[i1 - s] += (1.0 - h[i1])
        g[i1] -= h[i1 - s]
    g -= np.mean(g)
    h_new = np.clip(h - eta * g, 0.0, 1.0)
    return proj_sum(h_new, n * 0.5, n)

def gradient_descent(h, n, T, lr, n_iter, deadline):
    h_curr = h.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad = smooth_max_grad(h_curr, n, T)
        h_curr = h_curr - lr * grad
        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr = proj_sum(h_curr, n * 0.5, n)
        lr = max(0.001, lr * 0.9998)
    return h_curr

def local_swap_search(h, n, max_iters, deadline):
