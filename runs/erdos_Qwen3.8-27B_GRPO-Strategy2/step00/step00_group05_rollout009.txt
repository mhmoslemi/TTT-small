def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_mc = max_corr(best_h)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP from parent (exact max objective)
    h_polished = slsqp_optimize(h0, n, 0.0, 5000)
    h_polished = project_feasible(h_polished, target)
    mc = max_corr(h_polished)
    if mc < best_mc:
        best_h, best_mc = h_polished, mc

    # Phase 2: L-BFGS-B with smooth objective
    h_smooth = lbgfb_optimize(best_h, n, 50.0, 2000)
    h_smooth = slsqp_optimize(h_smooth, n, 0.0, 3000)
    h_smooth = project_feasible(h_smooth, target)
    mc = max_corr(h_smooth)
    if mc < best_mc:
        best_h, best_mc = h_smooth, mc

    # Phase 3: Binary simulated annealing
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-best_h)
    h_bin[sorted_idx[:n // 2]] = 1.0
    h_bin, best_bmc = sa_binary(h_bin, n, 500000, t0, budget_s, 180, rng)
    if best_bmc < best_mc:
        best_h, best_mc = h_bin.copy(), best_bmc

    # Phase 4: SLSQP polish from best binary/continuous
    h_polished = slsqp_optimize(best_h, n, 0.0, 5000)
    h_polished = project_feasible(h_polished, target)
    mc = max_corr(h_polished)
    if mc < best_mc:
        best_h, best_mc = h_polished, mc

    # Phase 5: L-BFGS-B with tighter smoothing
    h_smooth = lbgfb_optimize(best_h, n, 200.0, 3000)
    h_smooth = slsqp_optimize(h_smooth, n, 0.0, 3000)
    h_smooth = project_feasible(h_smooth, target)
    mc = max_corr(h_smooth)
    if mc < best_mc:
        best_h, best_mc = h_smooth, mc

    # Phase 6: Random restarts with SLSQP
    while time.time() - t0 < budget_s - 60:
        scale = 0.05 + 0.1 * rng.random()
        h_init = best_h + rng.normal(0, scale, size=n)
        h_init = project_feasible(h_init, target)
        h_new = slsqp_optimize(h_init, n, 0.0, 2000)
        h_new = project_feasible(h_new, target)
        mc = max_corr(h_new)
        if mc < best_mc:
            best_h, best_mc = h_new, mc

    # Phase 7: Try interpolation to other n_points
    for n_try in [50, 80, 100, 140, 90]:
        if time.time() - t0 > budget_s - 40:
            break
        target_t = n_try / 2.0
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, target_t)
        h_try = slsqp_optimize(h_try, n_try, 0.0, 3000)
        h_try = project_feasible(h_try, target_t)
        mc_t = max_corr(h_try) * 2.0 / n_try
        if mc_t < best_mc * 2.0 / n:
            best_h, best_mc, n = h_try, max_corr(h_try), n_try
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Final projection and verification
    best_h = project_feasible(best_h, n / 2.0)
    best_mc = max_corr(best_h)
    best_c5 = best_mc * 2.0 / n
    return best_h, best_c5, n
