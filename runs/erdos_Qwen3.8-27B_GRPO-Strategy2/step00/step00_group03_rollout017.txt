    w = np.exp(z - z_max)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            # term1[j] = 1-h[j+s] for j=0..n-1-s
            if n - 1 - s >= 0:
                grad[:n-s] += wk * (1.0 - h[s:])
            # term2[j] = h[j-s] for j=s..n-1
            if s < n:
                grad[s:] -= wk * h[:n-s]
        else:
            t = -s
            # term1[j] = 1-h[j-t] for j=t..n-1
            if t < n:
                grad[t:] += wk * (1.0 - h[:n-t])
            # term2[j] = h[j+t] for j=0..n-1-t
            if n - 1 - t >= 0:
                grad[:n-t] -= wk * h[t:]
    
    grad *= dx
    return grad

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, target_sum / n)
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = h < 1.0 if diff > 0 else h > 0.0
        if not np.any(free):
            break
        h[free] += diff / np.count_nonzero(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_grad_num(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    z = alpha * corr * dx
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w /= np.sum(w)
    eps = 1e-7
    grad = np.empty(n)
    for j in range(n):
        h2 = h.copy()
        h2[j] += eps
        corr2 = np.correlate(h2, 1.0 - h2, mode="full")
        grad[j] = dx * np.dot(w, corr2 - corr) / eps
    return grad

def grad_step_h(h, dx, alpha, lr, target_sum, max_iter=3000, time_limit=None):
    best_local = h.copy()
    best_local_c5 = compute_c5(best_local, dx)
    for it in range(max_iter):
        if time_limit is not None and time.time() > time_limit:
            break
        g = lse_grad_num(h, dx, alpha)
        h_new = h - lr * g
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_local_c5:
            best_local_c5 = c5_new
            best_local = h_new.copy()
        h = h_new
        alpha *= 1.004
        lr *= 0.9995
    return best_local, best_local_c5

def binary_local_search(h_bin, dx, time_limit):
    n = len(h_bin)
    c5_cur = compute_c5(h_bin, dx)
    best = h_bin.copy()
    iterations = 0
    while time.time() < time_limit:
        iterations += 1
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        # Try all single swaps
        for i in ones:
            if time.time() > time_limit:
                return best, c5_cur
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c = compute_c5(h_bin, dx)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                if c < c5_cur - 1e-14:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    c5_cur = c
                    best = h_bin.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, c5_cur

def random_local_search(h, dx, target_sum, rng, time_limit, n_perturb=3):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    temp = 0.002
    while time.time() < time_limit:
        h_try = best.copy()
        k = int(rng.integers(1, n_perturb + 1))
        idx = rng.choice(n, size=k, replace=False)
        for ii in idx:
            h_try[ii] += rng.uniform(-0.5, 0.5)
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 or rng.random() < np.exp(-(c5_try - best_c5) / max(temp, 1e-10)):
            best = h_try
            best_c5 = c5_try
        temp *= 1.0 - 1.0 / (time_limit + 1000)
        if temp < 1e-9:
            temp = 1e-9
    return best, best_c5
