import numpy as np
import time

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return compute_correlations(h).max()

def subgradient(h, s_star):
    """Compute gradient of c_{s_star} w.r.t. h, where s_star = k - (n-1)."""
    n = len(h)
    dx = 2.0 / n
    s = s_star
    grad = np.zeros(n)
    if s == 0:
        grad = 1.0 - 2.0 * h
    elif s > 0:
        # ∂c_s/∂h[j] = (1-h[j+s]) for j=0..n-1-s, minus h[j-s] for j=s..n-1
        grad[0:n-s] += (1.0 - h[s:n])
        grad[s:n] -= h[0:n-s]
    else:
        t = -s
        grad[t:n] += (1.0 - h[0:n-t])
        grad[0:n-t] -= h[t:n]
    return grad * dx

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any(): break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, n, n_iters=100000):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.1
    for it in range(n_iters):
        c = compute_correlations(h)
        k_star = c.argmax()
        s_star = k_star - (n - 1)
        grad = subgradient(h, s_star)
        h_new = h - alpha * grad
        h_new = project_h(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        alpha *= 0.9999
    return best_h, best_c5

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def _smo(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = c.max()
    tc = tau * (c - M)
    ec = np.exp(tc)
    s = ec.sum()
    f = M + np.log(s) / tau
    p = ec / s
    grad = np.zeros(n)
    so = n - 1
    for k in range(len(c)):
        if p[k] < 1e-10:
            continue
        s_k = k - so
        if s_k == 0:
            grad += p[k] * (1.0 - 2.0 * h) * dx
        elif s_k > 0:
            grad[:n-s_k] += p[k] * (1.0 - h[s_k:]) * dx
            grad[s_k:] -= p[k] * h[:n-s_k] * dx
        else:
            t = -s_k
            grad[t:] += p[k] * (1.0 - h[:n-t]) * dx
            grad[:n-t] -= p[k] * h[t:] * dx

            if not m.any(): break

    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = f(c), f(d)
    while abs(b - a) > tol:

            c = b - gr * (b - a)
            fc = f(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = f(d)

            grad += pk * (1.0 - 2.0 * h) * dx
        elif s > 0:
            grad[:n - s] += pk * (1.0 - h[s:]) * dx
            grad[s:] -= pk * h[:n - s] * dx
        else:
            t = -s
            grad[t:] += pk * (1.0 - h[:n - t]) * dx
            grad[:n - t] -= pk * h[t:] * dx
    return f, grad

def _proj(h, n):
    tg = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = tg - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            if not m.any():
                break
            h[m] += d / m.sum()
        else:
            m = h > 0.0
            if not m.any():
                break
            h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _pair_grid(h, i, j, n_pts=7):
    lo, hi = -h[i], 1.0 - h[j]
    if hi <= lo + 1e-10:
        return 0.0
    best_d, best_v = 0.0, _c5(h)
    for d in np.linspace(lo, hi, n_pts):
        if abs(d) < 1e-12:
            continue
        h[i] += d
        h[j] -= d
        v = _c5(h)
        h[i] -= d
        h[j] += d
        if v < best_v:
            best_v, best_d = v, d
    return best_d

def _polish(h, rng, t_end, passes=40, pairs_per_pass=300):
    n = len(h)
    best = h.copy()
    best_c5 = _c5(h)
    for _ in range(passes):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(pairs_per_pass):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j:
                continue
            d = _pair_grid(h, i, j)
            if abs(d) > 1e-9:
                h[i] += d
                h[j] -= d
                improved = True
        c5 = _c5(h)
        if c5 < best_c5:
            best_c5, best = c5, h.copy()
        if not improved:
            break
    return best, best_c5

def _slsqp(h, n, tau, t_end, maxiter=600):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnd = [(0.0, 1.0)] * n
    try:
        res = minimize(_smo, h, args=(tau,), method='SLSQP',
                       bounds=bnd, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        x = _proj(np.asarray(res.x, dtype=float), n)
        c5 = _c5(x)
        if time.time() > t_end or c5 > _c5(h) + 1e-6:
            return h, _c5(h)
        return x, c5
    except Exception:
        return h, _c5(h)

def _rand_init(n, rng):
    h = rng.random(n)
    return _proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    dl = t0 + budget_s - 8

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = _rand_init(95, rng)
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5(h)

    # Phase 1: SLSQP with increasing tau
    for tau, mi in [(10, 400), (30, 600), (80, 800), (200, 1000), (500, 1200)]:
        if time.time() > dl - 250:
            break
        h, c5 = _slsqp(h, n, tau, dl - 200, maxiter=mi)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Polish
    t2 = min(t0 + 500, dl - 120)
    best_h, best_c5 = _polish(best_h, rng, t2, passes=30, pairs_per_pass=400)

    # Phase 3: Random restarts
    t3 = dl - 60
    n_restart = 0
    while time.time() < t3 and n_restart < 30:
        h_r = _rand_init(n, rng)
        h_r, c5_r = _slsqp(h_r, n, 50.0, t3, maxiter=300)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_h, best_c5 = _polish(best_h, rng, min(time.time() + 40, t3),
                                      passes=10, pairs_per_pass=200)
        n_restart += 1

    # Final polish
    best_h, best_c5 = _polish(best_h, rng, dl, passes=15, pairs_per_pass=500)

    return best_h, best_c5, n
