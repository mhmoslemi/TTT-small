def local_search_bottleneck(h, n, max_iters=10000, time_limit=100):
    t0 = time.time()
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        # Compute current correlate
        c = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(c)
        max_idx = np.argmax(c)  # This is the bottleneck lag
        lag = max_idx - (n - 1)  # Actual lag value
        
        # Find positions contributing to this lag
        # c[max_idx] = sum of h[i] * (1-h[i+lag]) for valid i
        contributions = []
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                if h[i] > 0.1 and h[j] < 0.9:
                    contributions.append(i)
        
        # Try swapping h[i] with h[i+lag] for each contributing i
        best_improve = 1e-12
        best_swap = None
        for i in contributions:
            j = i + lag
            if 0 <= j < n:
                # Swap
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                h[i], h[j] = h[j], h[i]  # Revert
                if new_c5 < best_c5 - best_improve:
                    best_improve = best_c5 - new_c5
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_c5 = compute_c5(h)
            h_best = h.copy()
        else:
            # Try random swap as escape
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[np.random.randint(len(ones))]

                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    h_best = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]

    h_bin = np.zeros(n)

    frac_part = target - int_part
    h_bin[idx[:int_part]] = 1.0

    x_new = np.arange(n_new) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])

def gradient_c5(h, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    active = np.where(c >= max_c - 1e-10)[0]
    
    grad = np.zeros(N)
    for n_idx in active:
        s = N - 1 - n_idx
        # ∂c[n]/∂h[j] = (1-h[j+s]) - h[j-s]
        t1 = np.zeros(N)  # (1-h[j+s]) for j+s in [0,N)
        t2 = np.zeros(N)  # h[j-s] for j-s in [0,N)
        if s >= 0:
            t1[:N-s] = 1.0 - h[s:]
            t2[s:] = h[:N-s]
        else:
            t1[-s:] = 1.0 - h[:N+s]
            t2[:N+s] = h[-s:]
        grad += t1 - t2
    
    grad /= len(active)
    return dx * grad
```

def projected_gradient_c5(h, N, lr=0.01, max_iter=5000, time_limit=60):
    t0 = time.time()
    target = N / 2.0
    best_c5 = compute_c5(h)
    h_best = h.copy()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        grad = gradient_c5(h, N)
        h_new = h - lr * grad
        
        # Project onto {0≤h≤1, sum(h)=N/2}
        h_new = np.clip(h_new, 0.0, 1.0)
        # Adjust sum
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Add/subtract evenly, then reclip
            h_new += diff / N
            h_new = np.clip(h_new, 0.0, 1.0)
            # Iterate projection
            for _ in range(10):
                diff = target - np.sum(h_new)
                if abs(diff) < 1e-10:
                    break
                if diff > 0:
                    mask = h_new < 1.0
                else:
                    mask = h_new > 0.0
                if mask.sum() == 0:
                    break
                h_new[mask] += diff / mask.sum()
                h_new = np.clip(h_new, 0.0, 1.0)
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()
        
        # Decrease learning rate
        lr *= 0.999
    
    return h_best, best_c5
```
