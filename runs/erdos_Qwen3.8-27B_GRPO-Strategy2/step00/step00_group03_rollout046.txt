import numpy as np
from scipy.optimize import minimize
import time

_N = 70

def compute_c5(h):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h):
    n = _N
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

    return np.sum(x) - 0.5 * _N

    return compute_c5(x)
