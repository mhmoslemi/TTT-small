    h = np.clip(h, 0.0, 1.0)

        return h

    if abs(diff) > 1e-12:

        h = h + diff / len(h)

                h[idx] = min(1.0, h[idx] + diff2)

                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff2)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def ensure_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) > 1e-10:
        if s > 0:
            h = h * (target_sum / s)
            h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def sa_optimize(h, target_sum, time_limit, rng, T_start=0.004, T_min=1e-10, eps=0.08):
    n = len(h)
    h = ensure_feasible(h.copy(), target_sum)
    c_best = compute_c5(h)
    T = T_start
    t_start = time.time()
    n_iter = 0
    while time.time() - t_start < time_limit:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = eps * (rng.random() * 2.0 - 1.0)
        ni = h[i] + delta
        nj = h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        c_new = compute_c5(h)
        if c_new <= c_best:
            c_best = c_new
        elif T > T_min:
            if rng.random() < np.exp(-(c_new - c_best) / T):
                pass  # accept worse
            else:
                h[i] = ni - delta
                h[j] = nj + delta
        else:
            h[i] = ni - delta
            h[j] = nj + delta
        T = max(T * 0.99995, T_min)
        n_iter += 1
    h = ensure_feasible(h, target_sum)
    c_final = compute_c5(h)
    return h, c_final

def binary_local_search(h, target_sum, max_passes=200):
    n = len(h)
    h = ensure_feasible(h.copy(), target_sum)
    c_best = compute_c5(h)
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        for a in ones:
            for b in zeros:
                if a == b:
                    continue
                old_a, old_b = h[a], h[b]
                h[a] = old_b
                h[b] = old_a
                c_new = compute_c5(h)
                if c_new < c_best - 1e-12:
                    c_best = c_new
                    improved = True
                else:
                    h[a] = old_a
                    h[b] = old_b
        # Refresh lists
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def ensure_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) > 1e-10:
        if s > 0:
            h = h * (target_sum / s)
            h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def sa_optimize(h, target_sum, time_limit, rng, T_start=0.004, T_min=1e-10, eps=0.08):
    n = len(h)
    h = ensure_feasible(h.copy(), target_sum)
    c_best = compute_c5(h)
    T = T_start
    t_start = time.time()
    while time.time() - t_start < time_limit:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = eps * (rng.random() * 2.0 - 1.0)
        ni = h[i] + delta
        nj = h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        c_new = compute_c5(h)
        if c_new <= c_best:
            c_best = c_new
        elif T > T_min and rng.random() < np.exp(-(c_new - c_best) / T):
            pass
        else:
            h[i] = ni - delta
            h[j] = nj + delta
        T = max(T * 0.99997, T_min)
    h = ensure_feasible(h, target_sum)
    return h, compute_c5(h)

def binary_hill_climb(h, target_sum, max_passes=500):

    h = (h > 0.5).astype(float)
    ones = np.sum(h)

    c_best = compute_c5(h)

        ones_list = np.where(h > 0.5)[0]
        zeros_list = np.where(h < 0.5)[0]
        rng2 = np.random.permutation(len(ones_list))
        rng3 = np.random.permutation(len(zeros_list))
        for ra in rng2:
            for rb in rng3:
                a = ones_list[ra]
                b = zeros_list[rb]
                h[a] = 0.0
                h[b] = 1.0
                c_new = compute_c5(h)
                if c_new < c_best - 1e-14:
                    c_best = c_new
                    improved = True
                else:
                    h[a] = 1.0
                    h[b] = 0.0
        if not improved:
            break
    return h, c_best
