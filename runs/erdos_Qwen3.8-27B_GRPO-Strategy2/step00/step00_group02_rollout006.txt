    h = h.copy()

        return h
    if diff > 0:

    else:

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def fix_integral(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    h = np.clip(h, 0, 1)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            mask2 = h < 1.0
        else:
            mask2 = h > 0.0
        cnt2 = np.sum(mask2)
        if cnt2 > 0:
            h[mask2] += diff2 / cnt2
        h = np.clip(h, 0, 1)
    return h


def smooth_max_obj(x, tau, dx):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip(tau * (corr - m), -60, 60)
    return m + np.log(np.sum(np.exp(shifted)) + 1e-30) / tau


def raw_max_obj(x, dx):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    return np.max(corr)


def integral_constraint(x, target):
    return np.sum(x) - target


def integral_jacobian(x, target):
    return np.ones(len(x))


def make_constraint(target):
    return {'type': 'eq', 'fun': integral_constraint,
            'jac': integral_jacobian, 'args': (target,)}


def try_optimize(x0, target, dx, n, tau, time_limit, start_t):
    if time.time() - start_t > time_limit:
        return None
    cons = make_constraint(target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=[cons], args=(tau, dx),
                       options={'maxiter': 3000, 'ftol': 1e-12, 'iprint': 0})
        h_new = np.clip(res.x, 0, 1)
        h_new = fix_integral(h_new, target)
        return h_new
    except Exception:
        return None


def try_optimize_raw(x0, target, dx, n, start_t):
    if time.time() - start_t > 30:
        return None
    cons = make_constraint(target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(raw_max_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=[cons], args=(dx,),
                       options={'maxiter': 5000, 'ftol': 1e-14, 'iprint': 0})
        h_new = np.clip(res.x, 0, 1)
        h_new = fix_integral(h_new, target)
        return h_new
    except Exception:
        return None


def random_binary_start(n, target, rng):
    n_ones = int(target)
    n_zeros = n - n_ones
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:n_ones]] = 1.0
    # Place the fractional part
    frac = target - n_ones
    if 0 < frac < 1:
        remaining = idx[n_ones:]
        if len(remaining) > 0:
            h[remaining[0]] = frac
    h = fix_integral(h, target)
    return h
