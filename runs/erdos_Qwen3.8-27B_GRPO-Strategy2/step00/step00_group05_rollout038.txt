    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    n_ones = n // 2
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:n_ones]] = 1.0
    return h_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.asarray(initial_h_values, dtype=float).copy()
    n = len(h0)
    h0 = repair_sum(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary Simulated Annealing from multiple starts
    h_b0 = to_binary(h0, n)
    c5_b0 = compute_c5(h_b0)
    if c5_b0 < best_c5:
        best_c5 = c5_b0
        best_h = h_b0.copy()

    starts = [h_b0.copy()]
    for _ in range(6):
        hr = np.zeros(n)
        hr[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hr)

    sa_end = t0 + budget_s * 0.65
    for si in range(len(starts)):
        if time.time() > sa_end:
            break
        remaining = sa_end - time.time()
        per_time = remaining / (len(starts) - si)
        phase_end = time.time() + min(per_time, 180.0)

        current = starts[si].copy()
        curr_c5 = compute_c5(current)
        T = 0.002
        T_min = 1e-7
        cooling = 0.99998
        no_improve = 0

        while time.time() < phase_end:
            ones = np.where(current == 1.0)[0]
            zeros = np.where(current == 0.0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            current[i] = 0.0
            current[j] = 1.0
            new_c5 = compute_c5(current)
            if new_c5 <= curr_c5:
                curr_c5 = new_c5
                no_improve = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current.copy()
            else:
                delta = new_c5 - curr_c5
                if delta < 50.0 * T:
                    if rng.random() < np.exp(-delta / max(T, 1e-12)):
                        curr_c5 = new_c5
                        no_improve = 0
                        continue
                current[i] = 1.0
                current[j] = 0.0
                no_improve += 1
                if no_improve > 30000:
                    for _p in range(5):
                        ip = rng.randint(n)
                        jp = rng.randint(n)
                        if current[ip] != current[jp]:
                            current[ip], current[jp] = current[jp], current[ip]
                            break
                    curr_c5 = compute_c5(current)
                    no_improve = 0
            T = max(T * cooling, T_min)

    # Phase 2: Best-neighbor binary local search
    h_ln = np.where(best_h > 0.5, 1.0, 0.0)
    if not (np.sum(h_ln) == n // 2):
        h_ln = to_binary(best_h, n)
    c5_ln = compute_c5(h_ln)
    if c5_ln > best_c5 + 1e-9:
        h_ln = best_h.copy()
        c5_ln = best_c5

    ln_end = t0 + budget_s * 0.80
    if (h_ln == 0.0).any() and (h_ln == 1.0).any():
        improved = True
        while improved and time.time() < ln_end:
            improved = False
            ones = np.where(h_ln == 1.0)[0]
            zeros = np.where(h_ln == 0.0)[0]
            best_gain = 1e-10
            best_pair = None
            for i in ones:
                h_ln[i] = 0.0
                for j in zeros:
                    h_ln[j] = 1.0
                    c5_try = compute_c5(h_ln)
                    gain = c5_ln - c5_try
                    if gain > best_gain:
                        best_gain = gain
                        best_pair = (i, j)
                    h_ln[j] = 0.0
                h_ln[i] = 1.0
            if best_pair is not None:
                i, j = best_pair
                h_ln[i] = 0.0
                h_ln[j] = 1.0
                c5_ln = compute_c5(h_ln)
                improved = True
                if c5_ln < best_c5:
                    best_c5 = c5_ln
                    best_h = h_ln.copy()

    # Phase 3: Relaxed random hill-climbing with small temperature
    rs_end = t0 + budget_s * 0.92
    h_r = best_h.copy()
    c5_r = best_c5
    T_rel = 0.0005
    while time.time() < rs_end:
        k = rng.randint(n)
        h_r[k] = np.clip(h_r[k] + rng.uniform(-0.5, 0.5), 0.0, 1.0)
        h_r = repair_sum(h_r, n)
        c5_try = compute_c5(h_r)
        if c5_try < c5_r or (c5_try - c5_r < 0.01 and rng.random() < np.exp(-(c5_try - c5_r) / T_rel)):
            c5_r = c5_try
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
        T_rel = max(T_rel * 0.9999, 1e-6)

    # Phase 4: Coordinate descent polish
    cd_end = t0 + budget_s - 5.0
    cp = best_h.copy()
    c5p = best_c5
    for _sweep in range(500):
        if time.time() > cd_end:
            break
        sw_improved = False
        for i in range(n):
            saved = cp.copy()
            for dlt in [0.1, -0.1, 0.05, -0.05, 0.02, -0.02]:
                cp = saved.copy()
                cp[i] = np.clip(saved[i] + dlt, 0.0, 1.0)
                cp = repair_sum(cp, n)
                c5_try = compute_c5(cp)
                if c5_try < c5p - 1e-10:
                    c5p = c5_try
                    sw_improved = True
                    break
            else:
                cp = saved
        if not sw_improved:
            break
    if c5p < best_c5:
        best_c5 = c5p
        best_h = cp.copy()

    # Final validation
    best_h = repair_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, float(best_c5), n)
