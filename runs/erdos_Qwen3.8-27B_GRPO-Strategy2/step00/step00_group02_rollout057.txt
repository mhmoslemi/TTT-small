```python
import numpy as np
from scipy.optimize import minimize
import time

_n_g = 45
_temp_g = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        room = (1.0 - h) if diff > 0 else h
        total = np.sum(room)
        if total > 1e-12:
            h = h + (diff / total) * room
    return np.clip(h, 0.0, 1.0)

def _smooth_max(vals, t):
    mx = np.max(vals)
    shifted = t * (vals - mx)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    val = mx + np.log(np.sum(exps)) / t
    return val, w

def _obj_val(h, n, temp):
    hc = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(hc, 1.0 - hc, mode="full") * dx
    val, weights = _smooth_max(corr, temp)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        m = k_idx - (n - 1)
        js1 = max(0, -m)
        je1 = min(n, n - m)
        if je1 > js1:
            grad[js1:je1] += w * dx * (1.0 - hc[js1 + m:je1 + m])
        js2 = max(0, m)
        je2 = n
        if je2 > js2:
            grad[js2:je2] += w * dx * (-hc[js2 - m:je2 - m])
    return val, grad

def _objective(x):
    return _obj_val(x, _n_g, _temp_g)[0]

def _gradient(x):
    return _obj_val(x, _n_g, _temp_g)[1]

def _con_fun(x):
    return np.sum(x) - _n_g / 2.0

def _con_jac(x):
    return np.ones(_n_g)

def _slsqp_step(h, n, temp):
    global _n_g, _temp_g
    _n_g, _temp_g = n, temp
    try:
        res = minimize(_objective, h, method='SLSQP', jac=_gradient,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       options={'maxiter': 80, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def _coord_descent(h, n, max_iter=200):
    best_c5, _ = compute_c5(h)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            for d in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005):
                h2 = h.copy()
                h2[i] += d
                h2 = project_feasible(h2, n)
                c5, _ = compute_c5(h2)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    h = h2
                    improved = True
                    break
    return h, best_c5

def _make_sin_start(n, freqs=None):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if freqs is None:
        h = 0.5 + 0.25 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    else:
        h = np.full(n, 0.5)
        for f, a in freqs:
            h += a * np.cos(2 * np.pi * f * x)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n = len(h0)
    best_h = project_feasible(h0, n).copy()
    best_c5, _ = compute_c5(best_h)

    starts = [best_h.copy()]
    for _ in range(4):
        starts.append(project_feasible(best_h + 0.12 * rng.randn(n), n))
    starts.append(_make_sin_start(n))
    starts.append(_make_sin_start(n, [(1, 0.3), (2, -0.2), (3, 0.1)]))

    temps = [80, 50, 35, 25, 18, 13, 10, 8, 6, 5, 4, 3]

    for h_start in starts:
        if time.time() - t0 > budget_s - 80:
            break
        h_curr = h_start.copy()
        for temp in temps:
            if time.time() - t0 > budget_s - 40:
                break
            h_new = _slsqp_step(h_curr, n, temp)
            c5, _ = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_curr = h_new

    # Coordinate descent polish
    if time.time() - t0 < budget_s - 20:
        polished, pc5 = _coord_descent(best_h, n, max_iter=150)
        if pc5 < best_c5:
            best_c5 = pc5
            best_h = polished

    # Try n=90: upsample, optimize briefly, downsample
    if time.time() - t0 < budget_s - 120:
        n2 = n * 2
        h_up = np.zeros(n2)
        idx = np.arange(n2) * n / n2
        lo = np.floor(idx).astype(int)
        hi = np.minimum(lo + 1, n - 1)
        frac = idx - lo
        for i in range(n2):
            h_up[i] = (1 - frac[i]) * best_h[lo[i]] + frac[i] * best_h[hi[i]]
        h_up = project_feasible(h_up, n2)
        for temp in [30, 20, 15, 12]:
            if time.time() - t0 > budget_s - 30:
                break
            h_up = _slsqp_step(h_up, n2, temp)
        # Downsample
        h_ds = np.zeros(n)
        for i in range(n):
            h_ds[i] = (h_up[2 * i] + h_up[2 * i + 1]) / 2.0
        h_ds = project_feasible(h_ds, n)
        c5_ds, _ = compute_c5(h_ds)
        if c5_ds < best_c5:
            best_c5 = c5_ds
            best_h = h_ds
        c5_90, _ = compute_c5(h_up)
        if c5_90 < best_c5:
            best_c5 = c5_90
            best_h = h_up
            n = n2

    # Final polish
    if time.time() - t0 < budget_s - 5:
        final_h, final_c5 = _coord_descent(best_h, len(best_h), max_iter=80)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    return (best_h.tolist(), float(best_c5), len(best_h))
```
