import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h, dx):
    return np.max(compute_corr(h)) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            tr = np.sum(room)
            if tr < 1e-12:
                break
            h += (diff / tr) * room
        else:
            fl = np.maximum(h, 0.0)
            tf = np.sum(fl)
            if tf < 1e-12:
                break
            h += (diff / tf) * fl
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, rng):
    target = n // 2
    order = h + rng.random(n) * 1e-8
    idx = np.argsort(-order)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def full_sweep_binary(h, n, dx, time_limit, rng):
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    start = time.time()
    improved = True
    kicks = 0
    while improved and time.time() - start < time_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-15
        best_move = None
        for i in ones:
            hi = h[i]
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = np.max(compute_corr(h)) * dx
                delta = best_c5 - c
                if delta > best_delta:
                    best_delta = delta
                    best_move = (int(i), int(j))
                h[i] = hi
                h[j] = 0.0
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            best_c5 -= best_delta
            improved = True
        else:
            # Try a kick: flip 2-3 random pairs
            if time.time() - start < time_limit - 5:
                kk = min(3, len(ones), len(zeros))
                for _ in range(kk):
                    ii = ones[rng.integers(len(ones))]
                    jj = zeros[rng.integers(len(zeros))]
                    h[ii], h[jj] = h[jj], h[ii]
                kicks += 1
                corr_k = compute_corr(h)
                best_c5 = float(np.max(corr_k)) * dx
                improved = True
                if kicks > 50:
                    break
    return h, best_c5

def gradient_optimize(h, n, dx, time_limit, rng):
    h = h.astype(float).copy()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h, dx)
    start = time.time()
    lr = 0.08
    no_imp = 0
    while time.time() - start < time_limit and lr > 1e-9:
        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        grad = np.zeros(n)
        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        if j_min < j_max:
            js = np.arange(j_min, j_max)
            grad[js + lag] += (1.0 - h[js])
            grad[js] -= h[js + lag]
        # Also consider second-worst lag to flatten
        corr_sorted = np.sort(corr)[::-1]
        if len(corr) > 1:
            # Add small penalty on second term
            k2_candidates = np.where(np.abs(corr - corr_sorted[1]) < 1e-12)[0]
            if len(k2_candidates) > 0:
                k_max2 = int(k2_candidates[0])
                lag2 = k_max2 - (n - 1)
                j_min2 = max(0, -lag2)
                j_max2 = min(n, n - lag2)
                if j_min2 < j_max2:
                    js2 = np.arange(j_min2, j_max2)
                    grad[js2 + lag2] += 0.3 * (1.0 - h[js2])
                    grad[js2] -= 0.3 * h[js2 + lag2]
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5 - 1e-14:
            h = h_new
            best_c5 = new_c5
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 150:
                lr *= 0.6
                no_imp = 0
    return h, best_c5

def project_grid(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
