Let me also think about a smarter local search. Instead of random pairs, I can do a "first improvement" or "steepest descent" style:

For each i with h[i]=1, for each j with h[j]=0:
  Compute C₅ after swapping i and j
  Track the best swap

Then apply the best swap. Repeat.

This is O(n²) evaluations per sweep, each O(n) to compute. So O(n³) per sweep = 80³ = 512000 operations. Very fast.

But wait, I can be smarter. When I swap h[i] (1→0) and h[j] (0→1), the correlation changes only at lags related to i and j. Specifically:

The correlation c(d) = sum_{valid i} h[i+d] * (1-h[i]) changes as:
- For the old h[i]=1: terms where i appears as the first index: (1-h[i]) changes from 0 to 1, so c(d) increases by h[i+d] for d where i+d is valid... 

This is getting complicated with the "full" mode. Let me just compute directly.

Actually, for n=80, computing the full C₅ takes about 1-2 microseconds (numpy.correlate of two length-80 arrays is extremely fast). So even O(n²) = 6400 evaluations per sweep takes only ~10ms. We can do thousands of sweeps.

```python
def binary_local_search(h, max_time, rng):

    c5_cur = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < max_time:
        best_delta = 0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                c5_test = compute_c5(h)
                delta = c5_test - c5_cur
                h[i], h[j] = 1, 0
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break  # Local optimum
        
        i, j = best_pair
        h[i], h[j] = 0, 1
        c5_cur += best_delta
        ones = list(set(ones) - {i} | {j})  # Update sets
        zeros = list(set(zeros) - {j} | {i})
    
    return h, c5_cur
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = np.where(h < 1.0)[0]
    else:
        free = np.where(h > 0.0)[0]
    if len(free) > 0:
        per = diff / len(free)
        h[free] += per
        h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            f2 = np.where(h < 1.0)[0]
        else:
            f2 = np.where(h > 0.0)[0]
        if len(f2) > 0:
            h[f2[0]] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target, rng):
    n = len(h)
    hbin = np.where(h >= 0.5, 1.0, 0.0)
    s = int(np.sum(hbin))
    if s > int(target):
        idx = np.where(hbin == 1)[0]
        rng.shuffle(idx)
        hbin[idx[int(target):]] = 0
    elif s < int(target):
        idx = np.where(hbin == 0)[0]
        rng.shuffle(idx)
        hbin[idx[:int(target - s)]] = 1
    return hbin

def steepest_binary(h, max_time_frac, t0, budget_s):
    n = len(h)
    c5_cur = compute_c5(h)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    deadline = t0 + max_time_frac * budget_s
    while time.time() < deadline and len(ones) > 0 and len(zeros) > 0:
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            hi = h[i]
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c5_test = compute_c5(h)
                delta = c5_test - c5_cur
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_i, best_j = i, j
                h[j] = 0.0
            h[i] = hi
        if best_i < 0:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5_cur += best_delta
        ones.remove(best_i)
        ones.append(best_j)
        zeros.remove(best_j)
        zeros.append(best_i)
    return h, c5_cur

def sa_binary(h, t0, budget_s, frac, rng, T0=0.004):
    n = len(h)
    c5_cur = compute_c5(h)
    deadline = t0 + frac * budget_s
    T = T0
    no_improve = 0
    best_local = c5_cur
    best_h_local = h.copy()
    while time.time() < deadline:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if h[i] == h[j]:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c5_new = compute_c5(h)
        d = c5_new - c5_cur
        if d < 0 or (T > 1e-8 and rng.random() < np.exp(-d / T)):
            c5_cur = c5_new
            no_improve += 1
            if c5_cur < best_local:
                best_local = c5_cur
                best_h_local = h.copy()
                no_improve = 0
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            no_improve += 1
        T *= 0.99995
        if no_improve > 8000:
            if T < 1e-5:
                h = best_h_local.copy()
                c5_cur = best_local
                for _ in range(3):
                    a = int(rng.integers(0, n))
                    b = int(rng.integers(0, n))
                    if h[a] != h[b]:
                        h[a], h[b] = 1.0 - h[a], 1.0 - h[b]
                c5_cur = compute_c5(h)
                T = 0.002
                no_improve = 0
            else:
                no_improve = 0
    return h, c5_cur

def continuous_refine(h, t0, budget_s, frac, rng):
    n = len(h)
    target = n / 2.0
    h_cur = fix_sum(h.copy(), target)
    c5_cur = compute_c5(h_cur)
    deadline = t0 + frac * budget_s
    step = 0.08
    for _ in range(200000):
        if time.time() > deadline:
            break
        k = int(rng.integers(2, 6))
        idx = rng.choice(n, size=k, replace=False)
        h_p = h_cur.copy()
        h_p[idx] += rng.uniform(-step, step, size=k)
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = fix_sum(h_p, target)
        c5_p = compute_c5(h_p)
        c5_c = compute_c5(h_cur)
        if c5_p < c5_c - 1e-12:
            h_cur = h_p
            c5_cur = c5_p
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.999
            if step < 0.003:
                step = 0.02
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, AttributeError):
        h_init = None

    if h_init is not None and len(h_init) > 0:
        h_start = h_init.copy()
    else:
        h_start = np.zeros(80)
        h_start[::2] = 1.0

    n = len(h_start)
    target = n / 2.0
    h_best = fix_sum(h_start.copy(), target)
    c5_best = compute_c5(h_best)

    # Phase 1: Steepest binary local search from initial
    h_bin = make_binary(h_start, target, rng)
    h_bin, c5_bin = steepest_binary(h_bin, 0.15, t0, budget_s)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()

    # Phase 2: SA from multiple starting points
    remaining_frac = 0.55
    n_restarts = 6
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.72:
            break
        frac = remaining_frac / (n_restarts - r)
        if r == 0:
            h_sa = h_bin.copy()
        elif r == 1:
            h_sa = make_binary(h_best, target, rng)
        elif r == 2:
            # Sorted-chunk pattern
            h_sa = np.zeros(n)
            chunk = n // 4
            h_sa[:chunk] = 1
            h_sa[3*chunk:] = 1
        elif r == 3:
            # Triangular wave
            x = np.linspace(0, 2, n, endpoint=False)
            h_sa = np.where((x * 4) % 2 < 1, 1.0, 0.0)
            h_sa = make_binary(h_sa, target, rng)
        elif r == 4:
            # Perturb best binary
            h_sa = make_binary(h_best, target, rng)
            for _ in range(10):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if h_sa[a] != h_sa[b]:
                    h_sa[a], h_sa[b] = 1.0 - h_sa[a], 1.0 - h_sa[b]
        else:
            # Quadratic pattern
            x = np.linspace(0, 1, n, endpoint=False)
            h_sa = np.where((x * 5) % 1 < 0.5, 1.0, 0.0)
            h_sa = make_binary(h_sa, target, rng)

        h_sa, c5_sa = sa_binary(h_sa, t0, budget_s, frac, rng)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa.copy()

        # Follow up with steepest descent
        if time.time() - t0 < budget_s * 0.70:
            h_sa2, c5_sa2 = steepest_binary(h_sa.copy(), 0.03, t0, budget_s)
            if c5_sa2 < c5_best:
                c5_best = c5_sa2
                h_best = h_sa2.copy()

    # Phase 3: Continuous refinement
    if time.time() - t0 < budget_s * 0.82:
        h_cont, c5_cont = continuous_refine(h_best, t0, budget_s, 0.10, rng)
        if c5_cont < c5_best:
            c5_best = c5_cont
            h_best = h_cont.copy()

    # Phase 4: Try another n
    if time.time() - t0 < budget_s * 0.92:
        n2 = n + 20
        h2 = np.zeros(n2)
        for i in range(n2):
            pos = 2.0 * i / n2 * (n - 1) / 2.0
            i0 = int(np.floor(pos))
            i1 = min(i0 + 1, n - 1)
            f = pos - i0
            h2[i] = (1 - f) * h_best[i0] + f * h_best[i1]
        h2 = fix_sum(h2, n2 / 2.0)
        c5_2 = compute_c5(h2)
        if c5_2 < c5_best:
            h2, c5_2 = continuous_refine(h2, t0, budget_s, 0.06, rng)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h2.copy()
                n = n2

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = compute_c5(h_best)

    return h_best, c5_final, len(h_best)
```
