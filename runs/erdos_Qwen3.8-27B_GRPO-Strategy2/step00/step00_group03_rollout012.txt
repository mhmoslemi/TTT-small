import math
import time

def compute_c5(h):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")

    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    return float(corr[max_idx]) * dx, max_idx

def make_feasible_binary(h):
    """Make h binary with exactly N/2 ones"""
    N = len(h)
    h = (h > 0.5).astype(float)
    n_ones = int(h.sum())
    target = N // 2
    if n_ones > target:

        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for i in range(n_ones - target):
            h[ones[i]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for i in range(target - n_ones):
            h[zeros[i]] = 1
    return h

    current_c5 = compute_c5(h)

    best_c5 = current_c5
    start = time.time()

def make_binary(h, rng):
    N = len(h)
    h = np.round(h).astype(float)
    target = N // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        for i in range(n_ones - target):
            h[idx[perm[i]]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        for i in range(target - n_ones):
            h[idx[perm[i]]] = 1
    return h

def optimize_binary(h, time_limit, rng):
    N = len(h)
    h = make_binary(h.copy(), rng)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()

    # Phase 1: Greedy steepest descent
    for _ in range(150):
        if time.time() - start > time_limit * 0.25:
            break
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        best_delta = 0.0
        best_o = -1
        best_z = -1
        for oi in ones_list:
            h[oi] = 0.0
            for zi in zeros_list:
                h[zi] = 1.0
                c5 = compute_c5(h)
                if current_c5 - c5 > best_delta:
                    best_delta = current_c5 - c5
                    best_o = oi
                    best_z = zi
                h[zi] = 0.0
            h[oi] = 1.0
        if best_o < 0:
            break
        h[best_o] = 0.0
        h[best_z] = 1.0
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Phase 2: Simulated annealing
    T_start = 0.05
    T_end = 0.001
    p2_start = time.time()
    total_p2 = max(1.0, time_limit * 0.75)
    stuck = 0

    while time.time() - start < time_limit:
        progress = min(1.0, (time.time() - p2_start) / total_p2)
        T = T_start * (T_end / T_start) ** progress

        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        o = int(ones_list[rng.integers(len(ones_list))])
        z = int(zeros_list[rng.integers(len(zeros_list))])

        h[o] = 0.0
        h[z] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5

        if delta < 0 or (T > 1e-8 and rng.random() < math.exp(-delta / T)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stuck = 0
        else:
            h[o] = 1.0
            h[z] = 0.0
            stuck += 1

        if stuck > 3000:
            for _ in range(4):
                o2 = int(ones_list[rng.integers(len(ones_list))])
                z2 = int(zeros_list[rng.integers(len(zeros_list))])
                h[o2] = 0.0
                h[z2] = 1.0
            current_c5 = compute_c5(h)
            stuck = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase A: N=70 from initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=float)
        N0 = len(h0)
        if N0 % 2 != 0:
            h0 = h0[:N0 - 1]
            N0 = len(h0)
        alloc = budget_s * 0.4
        h_opt, c5 = optimize_binary(h0, alloc, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = N0
    except Exception:
        pass

    # Phase B: Try other N values with different initializations
    n_candidates = [30, 40, 50, 60, 80, 100, 140, 200, 280, 350]
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 30
    if remaining > 40:
        t_each = remaining / len(n_candidates)
        for N in n_candidates:
            if time.time() - t0 > budget_s - 30:
                break
            # Use a spread-out initial pattern
            h_init = np.zeros(N)
            for i in range(N // 2):
                h_init[i * 2] = 1.0
            h_opt, c5 = optimize_binary(h_init, t_each, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = N

    # Phase C: Re-optimize the best N found with remaining time
    elapsed = time.time() - t0
    if best_h is not None and budget_s - elapsed > 30:
        alloc2 = budget_s - elapsed - 10
        best_h, best_c5 = optimize_binary(best_h, alloc2, rng)
        best_n = len(best_h)

    # Fallback
    if best_h is None:
        N = 70
        best_h = np.zeros(N)
        best_h[::2] = 1.0
        if best_h.sum() != N // 2:
            best_h = make_binary(best_h, rng)
        best_c5 = compute_c5(best_h)
        best_n = N

    return best_h, best_c5, best_n
