def subgradient_step(h, alpha, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    s = k_max - (n - 1)  # shift
    
    # Gradient of c_{k_max} with respect to h
    # c_s = sum_i h[i] * (1 - h[i+s]) for valid i
    # dc_s/dh[i] = (1 - h[i+s]) - h[i-s] (for valid indices)
    grad = np.zeros(n)
    for i in range(n):
        # Contribution from h[i] as first factor: (1 - h[i+s]) if i+s < n
        if 0 <= i + s < n:
            grad[i] += (1.0 - h[i + s])
        # Contribution from h[i] as second factor: -h[i-s] if i-s >= 0
        # Actually, c_s = sum_i h[i]*(1-h[i+s])
        # d/dh[i] of c_s:
        # If h[i] appears as first factor (position i): (1-h[i+s])
        # If h[i] appears as second factor (position i-s): -h[i] ... no
        # c_s = sum_j h[j]*(1-h[j+s])
        # d/dh[i]: 
        #   if i is j: (1-h[i+s]) [from the term where j=i]
        #   if i = j+s, i.e., j = i-s: -h[i-s] [from the term where j=i-s, second factor is h[i]]
        # So: grad[i] = (1-h[i+s]) * [i+s < n] - h[i-s] * [i-s >= 0 and i-s >= 0]
        if 0 <= i - s < n:
            grad[i] -= h[i - s]
    
    grad *= dx
    h_new = h - alpha * grad
    # Project
    h_new = np.clip(h_new, 0, 1)
    for _ in range(100):
        d = np.sum(h_new) - target_sum
        if abs(d) < 1e-10:
            break
        h_new = np.clip(h_new - d/n, 0, 1)

def gradient_c_s(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        # (1-h[j+s]) for j+s < n, i.e., j < n-s
        g[:n-s] += (1 - h[s:])
        # -h[j-s] for j-s >= 0, i.e., j >= s
        g[s:] -= h[:n-s]
    else:
        s = -s
        # (1-h[j-s]) for j-s < n, i.e., j < n (always) but j-s >= 0, i.e., j >= s
        g[s:] += (1 - h[:n-s])
        # -h[j+s] for j+s < n, i.e., j < n-s
        g[:n-s] -= h[s:]
    return g

            zeros = np.where(h_bin == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / max(temp, 1e-8)):
                cur_c5 = new_c5
                if cur_c5 < best_bin_c5:
                    best_bin_c5 = cur_c5
                    best_bin = h_bin.copy()
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
                    best_n = n
                    if best_c5 < target_c5:
                        break
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
            if it % 1000 == 0:
                temp *= 0.99
                h_bin = best_bin.copy()  # reset to best periodically
                cur_c5 = best_bin_c5

    # Phase 5: Fine subgradient polish at final best
    if remaining() > 20 and best_c5 >= target_c5:
        h = best_h.copy()
        alpha = 1e-4
        for it in range(200000):
            if remaining() < 5:
                break
            g, c_cur, _ = grad_max_corr(h, n)
            h_new = project_sum(h - alpha * g, target)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
                if best_c5 < target_c5:
                    break
            h = h_new
            if it % 2000 == 0:
                alpha *= 0.98

    # Final validation
    best_h = project_sum(best_h, float(best_n) / 2.0)
    best_c5 = compute_c5(best_h)
    # Assert constraints
    assert best_h.min() >= -1e-15, "h has negative values"
    assert best_h.max() <= 1.0 + 1e-15, "h exceeds 1"
    assert abs(best_h.sum() - best_n / 2.0) < 1e-8, "sum constraint violated"

    return best_h, float(best_c5), best_n
