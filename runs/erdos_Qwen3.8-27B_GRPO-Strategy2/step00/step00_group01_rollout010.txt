from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def c5_grad(h, dx, eps=1e-7):
    n = len(h)
    f0 = compute_c5(h, dx)
    g = np.zeros(n)
    for j in range(n):
        h2 = h.copy()
        old = h2[j]
        step = eps if old + eps <= 1.0 else -eps
        h2[j] = old + step
        g[j] = (compute_c5(h2, dx) - f0) / step
    return g
