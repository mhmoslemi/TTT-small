```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    p = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if p >= 0:
        grad[p:] += 1.0 - h[:n-p]
        grad[:n-p] -= h[p:]
    else:
        q = -p
        grad[:n-q] += 1.0 - h[q:]
        grad[q:] -= h[:n-q]
    return grad * dx

def project_onto_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    target = n / 2.0
    # Simple: scale and clip
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        h += (target - s) / n
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(45) / 45 * 2  # sum = 2, n=45, need sum = 22.5
        h_init = np.clip(h_init, 0, 1)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, len(best_h))
    
    # Try multiple resolutions
    resolutions = [len(h_init), 90, 135, 180, 225, 270, 360]
    
    for n in resolutions:
        if time.time() - t_start > budget_s * 0.85:
            break
        
        # Upsample initial
        h_start = np.interp(
            np.linspace(0, 1, n, endpoint=False),
            np.linspace(0, 1, len(h_init), endpoint=False),
            np.interp(
                np.linspace(0, 2, len(h_init), endpoint=False),
                np.linspace(0, 2, len(h_init), endpoint=False),
                h_init
            )
        )
        # Actually, simpler upsample
        h_start = np.interp(
            np.arange(n),
            np.arange(len(h_init)) * n / len(h_init),
            h_init
        )
        h_start = project_onto_feasible(h_start, n)
        
        # Multiple restarts
        n_restarts = max(1, int((budget_s * 0.8) / (n * 0.01)))
        n_restarts = min(n_restarts, 20)
        
        for r in range(n_restarts):
            if time.time() - t_start > budget_s * 0.9:
                break
            
            if r == 0:
                h_try = h_start.copy()
            else:
                h_try = h_start + rng.normal(0, 0.05, n)
                h_try = project_onto_feasible(h_try, n)
            
            # SLSQP optimization
            def obj(x):
                return compute_c5(x, n)
            
            def obj_grad(x):
                return compute_grad(x, n)
            
            constraints = [
                {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}
            ]
            bounds = [(0, 1)] * n
            
            try:
                res = minimize(obj, h_try, jac=obj_grad, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-8})
                if res.success or res.fun < best_c5:
                    h_new = res.x
                    h_new = project_onto_feasible(h_new, n)
                    c5_new = compute_c5(h_new, n)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
            except Exception:
                pass
    
    # Final evaluation
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
```

```python
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.full(45, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, len(best_h))
    n_init = len(h_init)

    resolutions = sorted(set([n_init, 90, 135, 180, 240, 300, 400, 500, 600]))
    resolutions = [n for n in resolutions if 20 <= n <= 700]

    for n in resolutions:
        if time.time() - t0 > budget_s * 0.85:
            break
        est_per = max(0.05, n * 0.003)
        n_restarts = max(2, min(12, int(budget_s * 0.06 / est_per)))

        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.90:
                break

            if r == 0:
                h_try = upsample(h_init, n)
            elif r == 1:
                h_try = upsample(best_h, n) + rng.normal(0.0, 0.04, n)
            elif r == 2:
                h_try = upsample(1.0 - h_init[::-1], n)
            elif r == 3:
                period = max(2, n // 8)
                pattern = (np.arange(n) % (2 * period) < period).astype(float)
                h_try = pattern + rng.normal(0.0, 0.2, n)
            else:
                h_try = 0.5 + rng.normal(0.0, 0.2, n)

            h_try = project_feasible(h_try, n)
            bounds = [(0.0, 1.0)] * n

            for method, opts in [
                ('SLSQP', {'maxiter': max(80, min(400, 120000 // n)), 'ftol': 1e-12}),
                ('Nelder-Mead', {'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-11}),
            ]:
                if time.time() - t0 > budget_s * 0.90:
                    break
                try:
                    if method == 'SLSQP':
                        res = minimize(penalty_obj, h_try, jac=penalty_grad,
                                       args=(n,), method='SLSQP',
                                       bounds=bounds, options=opts)
                    else:
                        res = minimize(penalty_obj, h_try, args=(n,),
                                       method='Nelder-Mead', bounds=bounds, options=opts)
                    h_opt = project_feasible(res.x, n)
                    c5_opt = compute_c5(h_opt, n)
                    if abs(np.sum(h_opt) - n / 2.0) < 1e-5 and c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                except Exception:
                    pass

    # Polish
    n_best = len(best_h)
    for _ in range(6):
        if time.time() - t0 > budget_s * 0.97:
            break
        h_try = project_feasible(best_h + rng.normal(0.0, 0.005, n_best), n_best)
        bounds = [(0.0, 1.0)] * n_best
        try:
            res = minimize(penalty_obj, h_try, jac=penalty_grad, args=(n_best,),
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-14, 'eps': 1e-11})
            h_opt = project_feasible(res.x, n_best)
            c5_opt = compute_c5(h_opt, n_best)
            if abs(np.sum(h_opt) - n_best / 2.0) < 1e-8 and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    try:
        verified = evaluate_erdos_solution(best_h)
        if verified is not None and verified < best_c5:
            best_c5 = verified
    except Exception:
        pass

    return best_h, best_c5, len(best_h)
```
