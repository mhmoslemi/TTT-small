def compute_c5(h, n_points=None):
    n = len(h)
    if n_points is None:
        n_points = n
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_integral(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

def binarize(h, n):
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(h_bin))
    if s > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(s - target):
            h_bin[ones_idx[order[i]]] = 0
    elif s < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target - s):
            h_bin[zeros_idx[order[i]]] = 1

def steepest_descent(h_init, n, max_time, rng):

    best = h.copy()
    best_val = compute_c5(h, n)
    current = h.copy()
    current_val = best_val
    t0 = time.time()
    stale = 0
    T = 0.02

    while time.time() - t0 < max_time:
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Try all 2-flip moves, find best
        best_move_val = current_val
        best_pi, best_pj = -1, -1
        for i in range(len(ones)):
            pi = ones[i]
            current[pi] = 0
            for j in range(len(zeros)):

                current[pj] = 1
                v = compute_c5(current, n)
