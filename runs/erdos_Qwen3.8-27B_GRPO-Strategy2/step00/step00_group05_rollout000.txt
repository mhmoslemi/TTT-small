    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlation_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_binary(h, n):
    """Project to binary with exactly n/2 ones"""
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        # Need more ones: flip zeros to ones
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(42)
        for i in rng.choice(zeros, target - n_ones, replace=False):

    elif n_ones > target:

        rng = np.random.RandomState(42)
        for i in rng.choice(ones, n_ones - target, replace=False):
            h[i] = 0
    return h

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def to_binary(h, n, seed=0):
    h = (h >= 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones < target:
        zeros = np.flatnonzero(h == 0)
        picks = rng.choice(zeros, target - n_ones, replace=False)
        h[picks] = 1.0
    elif n_ones > target:
        ones = np.flatnonzero(h == 1)
        picks = rng.choice(ones, n_ones - target, replace=False)
        h[picks] = 0.0
    return h

def local_search(h, n, max_iters=5000, seed=0, sa_temp=0.0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best = evaluate_c5(h)
    cur = best
    no_improve = 0
    for it in range(max_iters):
        i = rng.choice(np.flatnonzero(h == 1))
        j = rng.choice(np.flatnonzero(h == 0))
        h[i] = 0.0
        h[j] = 1.0
        new_val = evaluate_c5(h)
        if new_val < cur or (sa_temp > 0 and rng.rand() < np.exp(-(new_val - cur) / sa_temp)):
            cur = new_val
            if cur < best:
                best = cur
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve > 800:
            break
    return h, best

        )
        h_opt = np.clip(res.x, 0, 1)

    except Exception:
        return h0, evaluate_c5(h0)
