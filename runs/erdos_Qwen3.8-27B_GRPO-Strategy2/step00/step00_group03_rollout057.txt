import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if s < 1e-14:
        h[:] = target_sum / len(h)
        return h
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            h[:] = target_sum / len(h)
            return h
        h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
    else:
        mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            h[:] = target_sum / len(h)
            return h
        h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    s2 = float(np.sum(h))
    if abs(s2 - target_sum) > 1e-9:
        h = np.clip(h * (target_sum / s2), 0.0, 1.0)
    return h

        h[start:start + on_size] = 1.0
    return h
