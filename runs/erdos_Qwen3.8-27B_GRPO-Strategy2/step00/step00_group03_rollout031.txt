def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgradient(h, n):
    """Compute subgradient of max_k corr_k at h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    # corr[k] = sum_i h[i] * (1-h[i+k-n+1]) for valid i
    # d(corr[k])/d(h[j]) = (1-h[j+k-n+1]) - h[j-k+n-1] (with boundary handling)
    grad = np.zeros(n)
    offset = k_star - n + 1
    for j in range(n):
        # First term: i=j, need j+offset in [0, n-1]
        if 0 <= j + offset < n:
            grad[j] += 1.0 - h[j + offset]
        # Second term: i + offset = j, i = j - offset, need i in [0, n-1]
        if 0 <= j - offset < n:
            grad[j] -= h[j - offset]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    offset = k_star - n + 1
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j + offset < n:
            grad[j] += 1.0 - h[j + offset]
        if 0 <= j - offset < n:
            grad[j] -= h[j - offset]
    return grad * dx

def repair(h, n):
    h = np.array(h, dtype=float)
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s < 1e-15:
            h = np.full(n, 0.5)
            break
        h = h * (target / s)
        if abs(h.sum() - target) < 1e-12:
            break
    return h

def slsqp_optimize(h0, n, maxiter=250, ftol=1e-9):
    target = n / 2.0
    res = minimize(
        lambda x: compute_c5(x, n),
        h0.copy(),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return repair(res.x, n)

def grad_descent(h0, n, steps=300, lr=0.01):
    h = h0.copy()
    best_local = compute_c5(h, n)
    for step in range(steps):
        g = subgradient(h, n)
        h_new = h - lr * g
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_local:
            h = h_new
            best_local = c5_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr = max(lr * 0.92, 0.001)
    return h, best_local

def smooth_c5(h, n, T=15.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    smoothed = c_max + np.log(np.mean(np.exp(T * (corr - c_max)))) / T
    return smoothed * dx

def smooth_optimize(h0, n, maxiter=300):
    target = n / 2.0
    res = minimize(

        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
        options={'maxiter': maxiter, 'ftol': 1e-9}
    )
    return repair(res.x, n)
