import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} via bisection on shift."""
    h = h.astype(float)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction
    y += (target_sum - y.sum()) / len(y)
    return np.clip(y, 0.0, 1.0)

def c5_subgrad(h, n_points):
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)

            g[:n - s] -= h[s:]
            g[s:] -= h[:n - s]
        elif s < 0:
            s2 = -s
            g[s2:] -= h[:n - s2]
            g[:n - s2] -= h[s2:]
        else:
            g = 1.0 - 2.0 * h
        grad += g
    grad /= len(active)
    C5 = max_val * dx
    return C5, grad * dx

    best_c5 = compute_c5(h, n_points)
    cur_h = h.copy()
    cur_c5 = best_c5
    target_sum = n_points / 2.0
    n = n_points

        new_c5 = compute_c5(new_h, n)
        delta = new_c5 - cur_c5

            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = int(mask.sum())
        if n_free == 0:
            h += diff / len(h)
            break
        h[mask] += diff / n_free
    return np.clip(h, 0.0, 1.0)

def nm_objective(z, n_points, target_sum):
    h = project_feasible(z, target_sum)
    return compute_c5(h, n_points)

def sa_local_search(h, n_points, n_iter, rng, t_start, t_end):

    best_c5 = compute_c5(h, n_points)
    cur_h = h.copy()

    target_sum = n_points / 2.0
    n = n_points
    for i in range(n_iter):
        frac = i / max(n_iter - 1, 1)
        t = t_start * (t_end / t_start) ** frac
        step = 0.08 * (1.0 - frac) ** 0.7 + 0.005
        eps = rng.normal(0, step, n)
        new_h = project_feasible(cur_h + eps, target_sum)
        new_c5 = compute_c5(new_h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (t * best_c5 + 1e-15)):

            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
