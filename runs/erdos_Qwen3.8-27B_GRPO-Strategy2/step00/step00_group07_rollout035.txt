def c5_value(h):

    dx = 2.0 / n

    dx = 2.0 / n
    hf = h.astype(np.float64)

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        h_init = np.zeros(80)
        h_init[::2] = 1.0

    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n

    h_init = project_feasible(h_init, target, n)
    best_h = h_init.copy()
    best_c5 = c5_value(best_h)

    # Phase 1: Binary hill climbing
    h_b = np.where(h_init >= 0.5, 1.0, 0.0)
    h_b = project_feasible(h_b, target, n)
    h_b = np.where(h_b >= 0.5, 1.0, 0.0)
    h_work = h_b.copy()
    c5_w = c5_value(h_work)
    if c5_w < best_c5:
        best_c5 = c5_w
        best_h = h_work.copy()

    p1_end = budget_s * 0.22
    stall_count = 0
    while elapsed() < p1_end:
        ones = np.where(h_work == 1)[0]
        zeros = np.where(h_work == 0)[0]
        best_gain = 1e-15
        best_pair = None
        for i in ones:
            for j in zeros:
                h_work[i] = 0.0
                h_work[j] = 1.0
                c5_n = c5_value(h_work)
                h_work[i] = 1.0
                h_work[j] = 0.0
                g = c5_w - c5_n
                if g > best_gain:
                    best_gain = g
                    best_pair = (i, j)
        if best_pair is not None and best_gain > 1e-12:
            h_work[best_pair[0]] = 0.0
            h_work[best_pair[1]] = 1.0
            c5_w = c5_value(h_work)
            stall_count = 0
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_work.copy()
        else:
            stall_count += 1
            if stall_count > 3:
                break
            # Perturb to escape
            for _ in range(rng.randint(2, 6)):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_work[i] = 0.0
                h_work[j] = 1.0
            c5_w = c5_value(h_work)

    # Phase 2: Simulated annealing (binary)
    p2_end = budget_s * 0.45
    h_sa = h_work.copy()
    c5_sa = c5_value(h_sa)
    T = 0.06
    while elapsed() < p2_end:
        ones = np.where(h_sa == 1)[0]
        zeros = np.where(h_sa == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_sa[i] = 0.0
        h_sa[j] = 1.0
        c5_n = c5_value(h_sa)
        d = c5_n - c5_sa
        if d < 0 or rng.random() < np.exp(-d / T):
            c5_sa = c5_n
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
        else:
            h_sa[i] = 1.0
            h_sa[j] = 0.0
        T = max(T * 0.99992, 0.008)
        if T <= 0.01:
            T = 0.04

    # Phase 3: Continuous projected gradient
    p3_end = budget_s - 30
    starts = [best_h.copy()]
    # Add fractional blend of best binary and its complement region
    h_blend = best_h.copy().astype(np.float64)
    h_blend = 0.7 * h_blend + 0.3 * (1.0 - h_blend)
    h_blend = project_feasible(h_blend, target, n)
    starts.append(h_blend)

    for h_start in starts:
        if elapsed() > p3_end:
            break
        h_c = h_start.copy()
        c5_c = c5_value(h_c)
        lr = 0.005
        for step in range(100000):
            if elapsed() > p3_end:
                break
            c5_curr, grad = c5_and_grad(h_c)
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            h_c -= lr * grad
            h_c = project_feasible(h_c, target, n)
            c5_new = c5_value(h_c)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_c.copy()
            if c5_new > c5_c + 1e-10:
                lr = max(lr * 0.7, 1e-5)
            else:
                lr = min(lr * 1.03, 0.02)
            c5_c = c5_new

    # Phase 4: Final binary polish
    if elapsed() < budget_s - 10:
        h_fin = np.where(best_h >= 0.5, 1.0, 0.0)
        h_fin = project_feasible(h_fin, target, n)
        h_fin = np.where(h_fin >= 0.5, 1.0, 0.0)
        c5_fin = c5_value(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            h_work = h_fin.copy()
            c5_w = c5_fin
            p4_end = budget_s - 5
            while elapsed() < p4_end:
                ones = np.where(h_work == 1)[0]
                zeros = np.where(h_work == 0)[0]
                best_gain = 1e-15
                best_pair = None
                for i in ones:
                    for j in zeros:
                        h_work[i] = 0.0
                        h_work[j] = 1.0
                        c5_n = c5_value(h_work)
                        h_work[i] = 1.0
                        h_work[j] = 0.0
                        g = c5_w - c5_n
                        if g > best_gain:
                            best_gain = g
                            best_pair = (i, j)
                if best_pair is not None and best_gain > 1e-12:
                    h_work[best_pair[0]] = 0.0
                    h_work[best_pair[1]] = 1.0
                    c5_w = c5_value(h_work)
                    if c5_w < best_c5:
                        best_c5 = c5_w
                        best_h = h_work.copy()
                else:
                    break

    best_h = project_feasible(best_h, target, n)
    final_c5 = c5_value(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n
