import time

def _compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h):
    return float(np.max(_compute_corr(h)))

def _project(v, s):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    d = s - h.sum()
    if abs(d) > 1e-14:
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def _grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        g[:n-lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n-lag]
    else:
        m = -lag
        g[m:] += 1.0 - h[:n-m]
        g[:n-m] -= h[m:]
    return g
