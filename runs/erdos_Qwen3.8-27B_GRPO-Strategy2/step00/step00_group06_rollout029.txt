def corr_subgrad(h, k_star):
    """Compute subgradient of corr[k_star] w.r.t. h"""
    n = len(h)
    L = k_star - (n - 1)  # lag
    sg = np.zeros(n)
    # d/dh[j] = (1-h[j-L])*I(j-L valid) - h[j+L]*I(j valid)
    # valid m for first term: m=j-L, need max(0,-L) <= j-L <= min(n-1-L, n-1)
    # valid m for second term: m=j, need max(0,-L) <= j <= min(n-1-L, n-1)

        m1 = j - L
        if 0 <= m1 < n and max(0, -L) <= m1 <= min(n-1-L, n-1):

        m2 = j
        if 0 <= m2 < n and max(0, -L) <= m2 <= min(n-1-L, n-1):
            sg[j] -= h[m2 + L] if 0 <= m2 + L < n else 0
    return sg
