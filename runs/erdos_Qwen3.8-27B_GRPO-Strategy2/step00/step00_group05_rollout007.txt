import numpy as np
from scipy.optimize import minimize

_t0 = 0.0
_budget = 1100.0

def remaining_time():
    import time
    return _budget - (time.time() - _t0)

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() == 0:
            idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
            continue
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

class EO:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        self.target = n / 2.0
        self.bounds = [(0.0, 1.0)] * n
    def sum_eq(self, x):
        return x.sum() - self.target
    def cons(self):
        return [{'type': 'eq', 'fun': self.sum_eq}]
    def obj_smooth(self, x, T):
        c = np.correlate(x, 1.0 - x, mode="full") * self.dx
        m = c.max()
        return T * np.log(np.exp((c - m) / T).sum()) + m
    def obj_max(self, x):
        return compute_c5(x, self.dx)
    def optimize(self, h_init, n_temps=6):
        h = project_sum(h_init, self.target)
        best = h.copy()
        best_c5 = compute_c5(h, self.dx)
        temps = [0.4, 0.15, 0.06, 0.025, 0.01, 0.004][:n_temps]
        for T in temps:
            if remaining_time() < 12:
                break
            res = minimize(self.obj_smooth, h, args=(T,), method='SLSQP',
                         bounds=self.bounds, constraints=self.cons(),
                         options={'maxiter': 150, 'ftol': 1e-11})
            if res.x is not None:
                h_new = project_sum(res.x, self.target)
                c5_new = compute_c5(h_new, self.dx)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best = h_new.copy()
                    h = h_new
        if remaining_time() > 15:
            res = minimize(self.obj_max, h, method='SLSQP',
                         bounds=self.bounds, constraints=self.cons(),
                         options={'maxiter': 200, 'ftol': 1e-13})
            if res.x is not None:
                h_new = project_sum(res.x, self.target)
                c5_new = compute_c5(h_new, self.dx)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best = h_new.copy()
        return best, best_c5

def make_blocks(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_on = int(n / 2)
    k = rng.randint(2, min(6, max(2, n_on - 1)))
    sizes = np.floor(rng.dirichlet(np.ones(k)) * n_on).astype(int)
    sizes[int(rng.randint(k))] += n_on - sizes.sum()
    pos = np.sort(rng.choice(n, k, replace=False))
    for sz, p in zip(sizes, pos):
        if sz > 0:
            h[(p + np.arange(sz)) % n] = 1.0
    return project_sum(h, n / 2.0)

def make_random(n, seed):
    rng = np.random.RandomState(seed)
    return project_sum(rng.uniform(0, 1, n), n / 2.0)

def make_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h

def make_sine(n, seed):
    rng = np.random.RandomState(seed)
    f = rng.uniform(1, 4)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    return project_sum(h, n / 2.0)

def perturb(h, amount, seed):
    rng = np.random.RandomState(seed)
    h_new = h + rng.uniform(-amount, amount, len(h))
    return project_sum(h_new, len(h) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    import time
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        eo = EO(n0)
        h_opt, c5_opt = eo.optimize(h0, n_temps=8)
        best_h, best_c5, best_n = h_opt, c5_opt, n0

    n_list = [50, 60, 80, 100]
    if best_n in n_list:
        n_list.remove(best_n)

    for n in n_list:
        if remaining_time() < 50:
            break
        eo = EO(n)
        starts = [make_blocks(n, seed), make_blocks(n, seed + 137),
                  make_random(n, seed + 271), make_block(n), make_sine(n, seed)]
        for si, hs in enumerate(starts):
            if remaining_time() < 25:
                break
            h_opt, c5_opt = eo.optimize(hs)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
            if remaining_time() > 40 and c5_opt < best_c5 + 0.01:
                h_p = perturb(h_opt, 0.15, seed + si * 50 + 999)
                h_opt2, c5_opt2 = eo.optimize(h_p, n_temps=4)
                if c5_opt2 < best_c5:
                    best_h, best_c5, best_n = h_opt2, c5_opt, n

    if best_h is not None and remaining_time() > 25:
        eo = EO(best_n)
        h_final, c5_final = eo.optimize(best_h, n_temps=8)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
        if remaining_time() > 20:
            h_p = perturb(best_h, 0.08, seed + 7777)
            h_pert, c5_pert = eo.optimize(h_p, n_temps=4)
            if c5_pert < best_c5:
                best_h, best_c5 = h_pert, c5_pert

    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
