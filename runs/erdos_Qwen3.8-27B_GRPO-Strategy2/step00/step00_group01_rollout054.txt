            if time.time() - t0 > budget_s - 200:
                break
            h_r, v_r = perturb_optimize(h_start, n0, rng, 0.06, 25)
            if v_r < best_c5:
                best_c5, best_h = v_r, h_r.copy()
            h_start = h_r

    # Higher resolutions
    for mult in [2, 4, 8, 16]:
        n = n0 * mult
        if time.time() - t0 > budget_s - 200:
            break
        h_up = interp_up(best_h, n)
        v_up = compute_c5(h_up)
        if v_up < best_c5:
            best_c5, best_h = v_up, h_up.copy()

        if time.time() - t0 < budget_s - 150:
            h_up = slsqp_solve(h_up)
            v = compute_c5(h_up)
            if v < best_c5:
                best_c5, best_h = v, h_up.copy()

        if time.time() - t0 < budget_s - 120 and n <= 200:
            h_up = cobyla_solve(h_up, maxiter=2000, rhobeg=0.02)
            v = compute_c5(h_up)
            if v < best_c5:
                best_c5, best_h = v, h_up.copy()

        # Quick perturb at this resolution
        if time.time() - t0 < budget_s - 100 and n <= 100:
            h_p, v_p = perturb_optimize(h_up, n, rng, 0.02, 15)
            if v_p < best_c5:
                best_c5, best_h = v_p, h_p.copy()

    # Final polish at best resolution
    if time.time() - t0 < budget_s - 30:
        n_f = len(best_h)
        if n_f <= 100:
            h_f = cobyla_solve(best_h, maxiter=5000, rhobeg=0.01)
            v_f = compute_c5(h_f)
            if v_f < best_c5:
                best_c5, best_h = v_f, h_f.copy()
        h_f = slsqp_solve(best_h)
        v_f = compute_c5(h_f)
        if v_f < best_c5:
            best_c5, best_h = v_f, h_f.copy()

    # Binarize final and check
    h_fbin = binarize(best_h, len(best_h))
    v_fb = compute_c5(h_fbin)
    if v_fb < best_c5:
        best_c5, best_h = v_fb, h_fbin.copy()

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
