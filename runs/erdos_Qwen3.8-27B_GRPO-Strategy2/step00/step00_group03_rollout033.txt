```python
import numpy as np
from scipy.optimize import minimize
import time

_nm_n = 70

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h[int(np.argmin(np.minimum(h, 1.0 - h)))] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, rng):
    hb = (np.asarray(h, dtype=float) >= 0.5).astype(float)
    cnt = int(hb.sum())
    tgt = n // 2
    if cnt > tgt:
        idx = np.where(hb == 1)[0]
        hb[rng.choice(idx, cnt - tgt, replace=False)] = 0
    elif cnt < tgt:
        idx = np.where(hb == 0)[0]
        hb[rng.choice(idx, tgt - cnt, replace=False)] = 1
    return hb

def local_search(hb, n, rng, time_limit=60.0):
    t0 = time.time()
    c5 = compute_c5(hb, n)
    best_c5 = c5
    best_hb = hb.copy()
    stall = 0
    stall_limit = 1500
    while time.time() - t0 < time_limit:
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        i = int(rng.choice(zeros))
        j = int(rng.choice(ones))
        hb[i], hb[j] = 1.0, 0.0
        nc5 = compute_c5(hb, n)
        if nc5 <= c5:
            c5 = nc5
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_hb = hb.copy()
        else:
            hb[i], hb[j] = 0.0, 1.0
            stall += 1
        if stall > stall_limit:
            for _ in range(4):
                ones = np.where(hb == 1)[0]
                zeros = np.where(hb == 0)[0]
                i2 = int(rng.choice(zeros))
                j2 = int(rng.choice(ones))
                hb[i2], hb[j2] = 1.0, 0.0
            c5 = compute_c5(hb, n)
            stall = 0
    return best_hb, best_c5

def _nm_obj(x):
    return compute_c5(repair(x, _nm_n), _nm_n)

def run(seed=42, budget_s=1100, **kwargs):
    global _nm_n
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    best_c5 = np.inf
    best_h = None
    best_n = 70

    starts = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        starts.append((h0, len(h0)))

    for n in [70, 100, 140]:
        hr = (rng.rand(n) < 0.5).astype(float)
        starts.append((hr, n))
        phi = (1.0 + 5.0**0.5) / 2.0
        hg = ((np.arange(n) * phi) % 1.0 < 0.5).astype(float)
        starts.append((hg, n))

    for h_s, n_s in starts:
        if time.time() > deadline - 50:
            break
        alloc = min(150.0, max(20.0, (deadline - time.time()) * 0.35))
        hb = to_binary(h_s, n_s, rng)
        hb_opt, c5_opt = local_search(hb, n_s, rng, time_limit=alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = hb_opt.copy()
            best_n = n_s

    # Second round: refine best with more local search
    if best_h is not None and time.time() < deadline - 40:
        alloc = min(200.0, (deadline - time.time()) * 0.5)
        hb2 = to_binary(best_h, best_n, rng)
        hb_opt2, c5_opt2 = local_search(hb2, best_n, rng, time_limit=alloc)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = hb_opt2.copy()
            best_n = hb_opt2.shape[0]

    # Nelder-Mead continuous refinement
    if best_h is not None:
        for _ in range(6):
            if time.time() > deadline - 15:
                break
            _nm_n = best_n
            h_pert = best_h + rng.randn(best_n) * 0.06
            h_pert = repair(h_pert, best_n)
            try:
                res = minimize(_nm_obj, h_pert, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
                h_new = repair(res.x, best_n)
                c5_new = compute_c5(h_new, best_n)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Final check: ensure feasibility
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return (best_h, best_c5, best_n)
```
