from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = np.sum(np.exp(beta * (corr - m)))
    return m + np.log(s) / beta

def sum_con_fun(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)

            cands = np.where(h < 1.0)[0]
        else:

        if len(cands) > 0:

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    def obj(x):
        return smooth_c5(x, beta)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        best = h.copy()
        best_val = compute_c5(h)
        if res.fun < smooth_c5(h, beta):
            best = fix_sum(res.x, n)
        if compute_c5(best) < best_val:
            return best, compute_c5(best)
        return best, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def sum_con_fun(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            cands = np.where(h < 1.0)[0]
        else:
            cands = np.where(h > 0.0)[0]
        if len(cands) > 0:
            h[cands] += diff / len(cands)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, maxiter, ftol, beta):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    best = h.copy()
    best_val = compute_c5(h)
    if beta > 0:
        obj = smooth_c5.__code__ and None
        def obj_fn(x):
            return smooth_c5(x, beta)
    else:
        def obj_fn(x):
            return compute_c5(x)
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            cand = fix_sum(res.x, n)
            cv = compute_c5(cand)
            if cv < best_val:
                best, best_val = cand, cv
    except Exception:
        pass
    return best, best_val

def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

def binary_local_search(h, max_iters):
    n = len(h)
    h = (h >= 0.5).astype(float)
    target_ones = n // 2
    n_ones = int(np.sum(h))

        idx = np.where(h == 1)[0][target_ones:]
        for i in idx:
            h[i] = 0
    elif n_ones < target_ones:
        idx = np.where(h == 0)[0][:target_ones - n_ones]
        for i in idx:
            h[i] = 1
    h = fix_sum(h, n)
    best_val = compute_c5(h)
    for _ in range(max_iters):
        improved = False
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                v = compute_c5(h)

                    best_val = v
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0

    return h, best_val
