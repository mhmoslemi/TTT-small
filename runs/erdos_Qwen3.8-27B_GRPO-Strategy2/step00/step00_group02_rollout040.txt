from scipy.optimize import minimize

def compute_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def project_feasible(h, N):
    h = np.array(h, dtype=float).copy()
    target = N / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) == 0:
            h += diff / N
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / np.sum(free)
    return h

def obj_grad(h, tau):
    """Return (soft_max_value, gradient) for the smooth max objective."""
    N = len(h)
    vals = compute_corr(h)
    m = np.max(vals)
    shifted = tau * (vals - m)
    exps = np.exp(shifted - np.max(shifted))
    w = exps / np.sum(exps)
    fval = float(m + np.log(np.sum(exps)) / tau + (tau * np.max(shifted)) / tau)
    # Recompute properly

            s += w[i] * t
