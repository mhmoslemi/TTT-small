    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def shift_integrals(h, dx):
    """Compute f_k for all shifts k."""
    n = len(h)
    f = np.zeros(2*n - 1)
    for k_idx, k in enumerate(range(-(n-1), n)):
        i_min = max(0, -k)
        i_max = min(n, n - k)
        idx = np.arange(i_min, i_max)
        f[k_idx] = np.sum(h[idx] * (1 - h[idx + k])) * dx
    return f

def objective_and_grad(h, dx, temp):
    n = len(h)
    f = shift_integrals(h, dx)
    val = smooth_max(f, temp)
    # Softmax weights
    shifted = (f - np.max(f)) / temp
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    # Gradient
    grad = np.zeros(n)

                h = h_new
                lr = min(lr * 1.05, 0.1)
                stall = 0
            else:
                lr *= 0.75
                stall += 1
                if lr < 1e-7 or stall > 50:
                    break
            if it % 30 == 0:
                c = c5_of(h, dx)
                if c < best_c5:
                    best_c5 = c
                    best = h.copy()
    c = c5_of(h, dx)
    if c < best_c5:
        best_c5 = c
        best = h.copy()
    return best, best_c5

def random_init(n, rng):
    h = np.zeros(n)
    tgt = 0.5 * n
    idx = rng.permutation(n)
    whole = int(tgt)
    rem = tgt - whole
    if whole >= n:
        whole = n - 1
        rem = tgt - whole
    h[idx[:whole]] = 1.0
    if rem > 1e-6 and whole < n:
        h[idx[whole]] = rem
    return project(h, n)

def structured_init(n, rng):
    h = np.zeros(n)
    block = max(2, n // 5)
    for i in range(0, n, block * 2):
        end = min(i + block, n)
        h[i:end] = 1.0
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    else:
        h0 = random_init(45, rng)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = project(h0, n0)

    best_h = h0.copy()
    best_c5 = c5_of(h0, dx0)

    # Primary optimization on parent size
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 120
    if rem > 20:
        best_h, best_c5 = optimize_h(h0, dx0, n0, rem, rng)

    # Try multiple restarts on same size
    for restart in range(5):
        if time.time() - t_start > budget_s - 60:
            break
        h_r = h0 + rng.standard_normal(n0) * 0.15
        h_r = project(h_r, n0)
        rem_r = (budget_s - 60) - (time.time() - t_start)
        if rem_r < 5:
            break
        h_r, c5_r = optimize_h(h_r, dx0, n0, rem_r * 0.6, rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Try other sizes
    other_ns = [35, 55, 25, 65, 30, 50, 40]
    for n in other_ns:
        if time.time() - t_start > budget_s - 40:
            break
        dx = 2.0 / n
        # Two inits: random and structured
        for init_fn in [random_init, structured_init]:
            if time.time() - t_start > budget_s - 20:
                break
            h_n = init_fn(n, rng)
            rem_n = (budget_s - 20) - (time.time() - t_start)
            if rem_n < 5:
                continue
            h_n, c5_n = optimize_h(h_n, dx, n, rem_n * 0.7, rng)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n

    # Final: try local polish with very low temp
    if time.time() - t_start < budget_s - 10:
        nf = len(best_h)
        dxf = 2.0 / nf
        best_h, best_c5 = optimize_h(best_h, dxf, nf, budget_s - 10 - (time.time() - t_start), rng)

    best_h = project(best_h, len(best_h))
    best_c5 = c5_of(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
