import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        c = np.count_nonzero(mask)
        if c == 0:
            break
        h[mask] += diff / c
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def nm_optimize(h0, n, t0, t_end, rng):
    """Run Nelder-Mead from h0 until t_end, return best h and c5."""
    best_local = h0.copy()
    best_c5_local = compute_c5(best_local)
    
    def obj(h_flat):
        if time.time() - t0 > t_end:
            return 10.0
        return compute_c5(repair(h_flat, n))
    
    res = minimize(obj, h0.copy(), method='Nelder-Mead',
                   options={'maxiter': 300, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
    h_new = repair(res.x, n)
    c5_new = compute_c5(h_new)
    if c5_new < best_c5_local:
        best_local = h_new
        best_c5_local = c5_new
    return best_local, best_c5_local

        elif n_restarts < 40:

        elif n_restarts < 80:

            h0[idx] = rng.random(n_flip)
            h0 = repair(h0, n)
        elif n_restarts < 130:

        if c5_new < best_c5:
            best_h = h_new

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
