            hc_new += (target - np.sum(hc_new)) / n
            hc_new = np.clip(hc_new, 0, 1)
            nc = compute_c5(hc_new)
            if nc < hc_c5:
                hc = hc_new
                hc_c5 = nc
                lr = min(lr * 1.05, 0.4)
                if hc_c5 < best_c5:
                    best_c5 = hc_c5
                    best_h = hc.copy()
            else:
                lr *= 0.7

    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if abs(s - target) > 1e-8:
        best_h += (target - s) / n
        best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
