import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    return float(np.max(c) * dx)

def smooth_max_weights(c, alpha):
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    exp_s = np.exp(shifted - np.max(shifted))
    return exp_s / np.sum(exp_s)

def compute_grad(h, w):
    """Compute gradient of weighted sum of correlations.
    w has length 2n-1, w[m] corresponds to lag k = m-(n-1).
    grad[i] = sum_k w_k * (1-h[i-k])*I(0<=i-k<n) - sum_k w_k * h[i+k]*I(0<=i+k<n)
    """
    n = len(h)
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if w[m] == 0.0:
            continue
        k = m - n + 1
        wm = w[m]
        # Term 1: wm * (1-h[i-k]) for 0<=i-k<n -> k<=i<n+k
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += wm * (1 - h[lo1-k:hi1-k])
        # Term 2: -wm * h[i+k] for 0<=i+k<n -> -k<=i<n-k
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2+k:hi2+k]
    return grad

def project_simplex_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2, 0, 1)

def projected_gradient(h, alpha, lr, n_iters, time_limit):
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        c = np.correlate(h, 1-h, mode="full")
        w = smooth_max_weights(c, alpha)
        g = compute_grad(h, w)
        
        h_new = project_simplex_box(h - lr * g, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Decay learning rate
        lr *= 0.995
    return best_h, best_c5

def interpolate_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_simplex_box(h_new, n_new / 2.0)
    return h_new

def random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, size=n)
    return proj_box_sum(h, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    hv = globals().get('initial_h_values', None)
    if hv is not None and len(hv) > 0:
        h = np.asarray(hv, dtype=float).copy()
    else:
        h = np.full(45, 0.5)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    n0 = len(h)
    frac_time = 0.88

    mults = [1, 2, 4, 8, 16]
    alphas_low = np.array([5.0, 15.0, 50.0])
    alphas_hi = np.array([50.0, 200.0, 1000.0])

    for si, mult in enumerate(mults):
        n_r = n0 * mult
        if n_r > 1024:
            break
        t_lim = t0 + budget_s * frac_time
        if time.time() > t_lim:
            break

        if n_r == n0:
            h_cur = h.copy()
        else:
            h_cur = interp_h(best_h, n_r)

        lr_init = 0.6 * np.sqrt(n0 / float(n_r))
        n_it = max(200, int((t_lim - time.time()) * 80))
        n_it = min(n_it, 3000)

        h_cur, c5 = pg_optimize(h_cur, alphas_low, lr_init, n_it, t_lim)
        if c5 < best_c5:
            best_c5, best_h = c5, h_cur.copy()

        if time.time() < t_lim - 2:
            h_cur, c5 = pg_optimize(h_cur, alphas_hi, lr_init * 0.4, 800, t_lim)
            if c5 < best_c5:
                best_c5, best_h = c5, h_cur.copy()

    # Random restart at a mid resolution
    if time.time() < t0 + budget_s * 0.75:
        n_rand = max(n0 * 4, 72)
        n_rand = min(n_rand, 512)
        h_r = random_init(n_rand, seed + 1)
        t_lim_r = t0 + budget_s * 0.80
        lr_r = 0.5
        h_r, c5 = pg_optimize(h_r, alphas_low, lr_r, 1500, t_lim_r)
        if c5 < best_c5:
            best_c5, best_h = c5, h_r.copy()

    # Polish at best resolution
    if time.time() < t0 + budget_s * 0.97:
        t_fin = t0 + budget_s * 0.97
        alphas_polish = np.array([500.0, 2000.0, 5000.0])
        n_pol = len(best_h)
        lr_pol = 0.05
        best_h, best_c5 = pg_optimize(best_h, alphas_polish, lr_pol, 600, t_fin)

    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = proj_box_sum(best_h, 0.5 * len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
