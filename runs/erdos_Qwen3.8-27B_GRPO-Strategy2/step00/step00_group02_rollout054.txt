import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def local_search(h, t0, t_limit):
    n = len(h)
    best = eval_c5(h)
    while time.time() - t0 < t_limit:
        bvals = []
        for i in range(n):
            if h[i] < 0.02 or h[i] > 0.98:
                bvals.append((i, 1.0 if h[i] > 0.5 else 0.0))
        if len(bvals) < 2:
            break
        improved = False
        for a in range(len(bvals)):
            for b in range(a + 1, len(bvals)):
                i, vi = bvals[a]
                j, vj = bvals[b]
                h[i] = 1.0 - vi
                h[j] = 1.0 - vj
                c = eval_c5(h)
                if c < best - 1e-14:
                    best = c
                    improved = True
                    bvals[a] = (i, 1.0 - vi)
                    bvals[b] = (j, 1.0 - vj)
                else:
                    h[i] = vi
                    h[j] = vj
        if not improved:
            break
    return h, best

def sa_optimize(h, t0, t_limit, rng):
    n = len(h)
    best = eval_c5(h)
    best_h = h.copy()
    cur = best
    T = 0.025
    while time.time() - t0 < t_limit:
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        oi, oj = h[i], h[j]
        h[i] = 1.0 - oi
        h[j] = 1.0 - oj
        c = eval_c5(h)
        d = c - cur
        if d < 0 or rng.random() < np.exp(-d / T):
            cur = c
            if c < best:
                best = c
                best_h = h.copy()
        else:
            h[i] = oi
            h[j] = oj
        T = max(T * 0.99997, 0.003)
    return best_h, best
