    scaled = values * lam
    max_s = np.max(scaled)
    if max_s > 700:
        return np.max(values)
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / lam

import numpy as np
from scipy.optimize import minimize
import time

def _smooth_max(corr, beta):
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _c5_obj(z, target_sum, beta, n_pts):
    h = np.clip(z, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = _smooth_max(corr, beta) * dx
    pen = 1e6 * (h.sum() - target_sum) ** 2
    return sm + pen

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_sum(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    return h

def _binary_search(h_init, n, target_sum, rng, max_rounds, time_fn):
    hb = (h_init > 0.5).astype(float).copy()
    n_ones = int(round(hb.sum()))
    if n_ones != int(target_sum):
        idx = np.argsort(-h_init)
        hb = np.zeros(n)
        hb[idx[:int(target_sum)]] = 1.0
    c5_b = _compute_c5(hb)
    zeros = np.where(hb == 0)[0]
    ones_p = np.where(hb == 1)[0]
    stuck = 0
    for r in range(max_rounds):
        if time_fn():
            break
        imp = False
        for _ in range(30):
            if len(zeros) == 0 or len(ones_p) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones_p[rng.randint(len(ones_p))]
            hb[i], hb[j] = 1.0, 0.0
            c5_new = _compute_c5(hb)
            if c5_new < c5_b - 1e-15:
                c5_b = c5_new
                imp = True
                zeros = np.where(hb == 0)[0]
                ones_p = np.where(hb == 1)[0]
                stuck = 0
            else:
                hb[i], hb[j] = 0.0, 1.0
        if not imp:
            stuck += 1
            if stuck > 40:
                break
    return hb, c5_b

    t0 = time.time()

    def _tf(frac):
        return (time.time() - t0) > budget_s * frac
