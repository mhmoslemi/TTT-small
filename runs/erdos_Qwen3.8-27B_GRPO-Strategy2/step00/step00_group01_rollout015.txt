import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, n, T):
    cc = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(cc)
    return m + np.log(np.sum(np.exp(T * (cc - m)))) / T


def slsqp_step(h, n, t0, budget_s, T):
    def obj(x):
        return smooth_max_obj(x, n, T)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                   options={'maxiter': 600, 'ftol': 1e-14})
    return project_feasible(res.x, n)


def local_search_continuous(h, n, rng, t0, budget_s, n_iter):
    best = h.copy()
    best_c5 = compute_c5(best)
    idx = np.arange(n)
    for it in range(n_iter):
        if time.time() - t0 > budget_s:
            break
        i, j = rng.choice(idx, 2, replace=False)
        scale = 0.3 * rng.uniform(0.5, 1.5)
        d = (best[i] - best[j]) * scale
        hn = best.copy()
        hn[i] -= d
        hn[j] += d
        hn = project_feasible(hn, n)
        c5n = compute_c5(hn)
        if c5n < best_c5 - 1e-13:
            best, best_c5 = hn, c5n
    return best, best_c5


def local_search_binary(h, n, rng, t0, budget_s, n_iter):
    best = h.copy()
    best_c5 = compute_c5(best)
    ones = np.where(best == 1.0)[0]
    zeros = np.where(best == 0.0)[0]
    for it in range(n_iter):
        if time.time() - t0 > budget_s:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        best[i], best[j] = best[j], best[i]
        c5n = compute_c5(best)
        if c5n < best_c5 - 1e-13:
            best_c5 = c5n
            ones = np.where(best == 1.0)[0]
            zeros = np.where(best == 0.0)[0]
        else:
            best[i], best[j] = best[j], best[i]
            # rare acceptance for diversity

                best[i], best[j] = best[j], best[i]
                ones = np.where(best == 1.0)[0]
                zeros = np.where(best == 0.0)[0]
    return best, best_c5
