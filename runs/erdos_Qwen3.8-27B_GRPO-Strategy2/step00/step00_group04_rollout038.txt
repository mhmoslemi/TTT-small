def compute_c5(h, n):

    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx

    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)
    softmax = exp_c / np.sum(exp_c)
    
    # Compute gradient efficiently using the shift structure

    # Contribution to grad[j]:

    #   - dx * softmax[n-1+s] * h[j+s] if 0<=j+s<n

        if a2 < b2:

    return project_onto_polytope(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    def elapsed():
        return time.time() - t_start

    # Phase 1: work with initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
        c5_v, _ = compute_c5(h_init, n_init)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_init.copy()
            best_n = n_init
        rem = budget_s - elapsed()
        t1 = min(100, rem * 0.25)
        h_opt, c5_opt = optimize_continuous(h_init, n_init, t1, n_iter=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_init
        rem = budget_s - elapsed()
        t2 = min(80, rem * 0.2)
        h_bin, c5_bin = binary_local_search(h_opt, n_init, t2, max_rounds=300)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n_init
        # Re-optimize from binary
        rem = budget_s - elapsed()
        if rem > 30:
            t3 = min(80, rem * 0.2)
            h_re, c5_re = optimize_continuous(h_bin, n_init, t3, n_iter=3000)
            if c5_re < best_c5:
                best_c5 = c5_re
                best_h = h_re.copy()
                best_n = n_init
            h_b2, c5_b2 = binary_local_search(h_re, n_init, min(40, rem * 0.1), max_rounds=200)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n_init

    # Phase 2: try other sizes
    rem = budget_s - elapsed()
    sizes = [64, 96, 128]
    for n_try in sizes:
        rem = budget_s - elapsed()
        if rem < 40:
            break
        for mode in [0, 1, 2]:
            rem = budget_s - elapsed()
            if rem < 25:
                break
            h0 = make_initial_n(n_try, seed + mode * 7, mode)
            t_ct = min(45, rem * 0.3)
            h_r, c5_r = optimize_continuous(h0, n_try, t_ct, n_iter=2000)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
            rem = budget_s - elapsed()
            if rem < 15:
                break
            t_bn = min(30, rem * 0.3)
            h_bn, c5_bn = binary_local_search(h_r, n_try, t_bn, max_rounds=200)
            if c5_bn < best_c5:
                best_c5 = c5_bn
                best_h = h_bn.copy()
                best_n = n_try

    # Phase 3: refine best if time remains
    rem = budget_s - elapsed()
    if rem > 30 and best_h is not None:
        t_fin = min(80, rem * 0.5)
        h_f, c5_f = optimize_continuous(best_h, best_n, t_fin, n_iter=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        rem = budget_s - elapsed()
        if rem > 20:
            h_fb, c5_fb = binary_local_search(best_h, best_n, min(40, rem * 0.4), max_rounds=200)
            if c5_fb < best_c5:
                best_c5 = c5_fb
                best_h = h_fb.copy()

    # Final safety
    if best_h is None:
        best_h = np.full(42, 0.5)
        best_n = 42
    best_h = project_onto_polytope(best_h, best_n / 2.0)
    best_c5, _ = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
