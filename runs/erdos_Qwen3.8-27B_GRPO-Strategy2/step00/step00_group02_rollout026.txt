import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max(), corr, dx

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        diff = target_sum - h.sum()
    return h

def grad_shift(h, m):
    n = len(h)
    grad = np.zeros(n)
    # Term 1: (1-h[k-m]) for valid k
    lo = max(0, m)
    hi = min(n, n + m)
    if lo < hi:
        ks = np.arange(lo, hi)
        grad[ks] += (1.0 - h[ks - m])
    # Term 2: -h[k+m] for valid k
    lo = max(0, -m)
    hi = min(n, n - m)
    if lo < hi:
        ks = np.arange(lo, hi)
        grad[ks] -= h[ks + m]
    return grad

def focus_optimize(h, max_iters=500, lr=0.01, patience=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project(h, target)
    best_c5, _, _ = compute_c5(h)
    best_h = h.copy()
    
    stale = 0
    for it in range(max_iters):
        c5, corr, _ = compute_c5(h)
        idx = np.argmax(corr)
        m = idx - (n - 1)
        
        g = grad_shift(h, m)
        # Center to maintain sum constraint
        g = g - g.mean()
        
        # Line search
        for ls in range(10):
            step = lr * (0.5 ** ls)
            h_new = h - step * g
            h_new = project(h_new, target)
            c5_new, _, _ = compute_c5(h_new)
            if c5_new <= c5 + 1e-10:
                h = h_new
                break
            if ls == 9:
                h = h_new  # Accept anyway to escape
        
        c5, _, _ = compute_c5(h)
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        
        if stale >= patience:
            break
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max(), corr, dx

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    for _ in range(200):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        diff = target_sum - h.sum()
    return h

def grad_shift(h, m):
    n = len(h)
    grad = np.zeros(n)
    lo = max(0, m)
    hi = min(n, n + m)
    if lo < hi:
        ks = np.arange(lo, hi)
        grad[ks] += (1.0 - h[ks - m])
    lo = max(0, -m)
    hi = min(n, n - m)
    if lo < hi:
        ks = np.arange(lo, hi)
        grad[ks] -= h[ks + m]
    return grad

def focus_optimize(h, max_iters=300, lr=0.05, patience=40):
    n = len(h)
    target = n / 2.0
    h = project(h.copy(), target)
    best_c5, _, _ = compute_c5(h)
    best_h = h.copy()
    stale = 0
    for it in range(max_iters):
        c5, corr, _ = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale >= patience:
            break
        idx = np.argmax(corr)
        m = idx - (n - 1)
        g = grad_shift(h, m)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        g = g / gnorm
        improved = False
        for ls in range(8):
            step = lr * (0.5 ** ls)
            h_new = project(h - step * g, target)
            c5_new, _, _ = compute_c5(h_new)
            if c5_new <= c5 + 1e-12:
                h = h_new
                improved = True
                break
        if not improved:
            h = project(h - lr * 0.5**7 * g, target)
    return best_h, best_c5

def random_perturb(h, strength):
    n = len(h)
    target = n / 2.0
    noise = np.random.normal(0, strength, n)
    return project(h + noise, target)

def try_larger_n(h_small, n_large, seed_offset=0):
    np.random.seed(42 + seed_offset)
    n_small = len(h_small)
    n_target = n_large / 2.0
    x_small = (np.arange(n_small) + 0.5) * (2.0 / n_small)
    x_large = (np.arange(n_large) + 0.5) * (2.0 / n_large)
    h_interp = np.interp(x_large, x_small, h_small)
    h_interp = project(h_interp, n_target)
    return h_interp
