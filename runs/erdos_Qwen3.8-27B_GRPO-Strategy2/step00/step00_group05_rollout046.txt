    return np.max(corr) * dx

def make_feasible(h, n_points):
    """Project h to feasible set: [0,1] and sum = n/2"""
    desired = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.full(n_points, 0.5)
        return h
    # Simple scaling
    h = h * (desired / current_sum)

    diff = desired - np.sum(h)
    if abs(diff) > 1e-10:

        if diff > 0:

                if h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add

                    h[i] -= sub

Wait, sum_j h[j]*h[j-k] IS symmetric in the sense that it equals sum_j h[j+k]*h[j] = sum_j h[j]*h[j+k]. And the quadratic form is h^T A_k h where A_k has A_k[j, j+k] = 1/2 and A_k[j+k, j] = 1/2 (to make it symmetric). Actually no:

sum_j h[j]*h[j+k] = sum_j A_k[j,j+k]*h[j]*h[j+k] where A_k[j,j+k]=1.
As a quadratic form: h^T M_k h where M_k[j,j+k] = 1/2 and M_k[j+k,j] = 1/2.
M_k = (S^k + S^{-k})/2 where S is the shift matrix.
This is symmetric. Is it PSD? The eigenvalues of S^k are e^{2πikm/n}, so eigenvalues of M_k are cos(2πkm/n) for m=0,...,n-1. These range from -1 to 1, so M_k is NOT PSD for general k.

So the problem is genuinely non-convex. We'll use local optimization methods.

import numpy as np
from scipy.optimize import minimize
import time

def _corr_max(h, n):
    """Compute C5 = max_k dx * corr[k]"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _corr_max_grad(h, n):
    """Subgradient of C5 w.r.t. h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        # d/dh[j] of sum_i h[i]*(1-h[i-lag])
        # = (1-h[j-lag]) if j-lag valid
        #   - sum_i [i-lag=j] h[i] = -h[j+lag] if j+lag valid
        j1 = j - lag
        if 0 <= j1 < n:
            grad[j] += (1.0 - h[j1])
        j2 = j + lag
        if 0 <= j2 < n:
            grad[j] -= h[j2]
    return grad * dx

def _make_feasible(h, n):
    """Project to [0,1]^n with sum = n/2"""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Iterative projection to meet sum
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # Distribute among elements not at bounds
        if diff > 0:
            free = h < 1.0
            if not np.any(free):
                break
            per = diff / np.sum(free)
            h[free] += min(per, 1.0 - h[free]).max() if np.any(free) else 0
            # Actually simpler: just scale
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            free = h > 0.0
            if not np.any(free):
                break
            h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def _slsqp_optimize(h0, n, max_iter=500):
    """SLSQP optimization"""
    dx = 2.0 / n
    
    def obj(x):
        return _corr_max(x, n)
    
    def jac(x):
        return _corr_max_grad(x, n)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    result = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                      constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-10})
    return result.x, _corr_max(_make_feasible(result.x, n), n)

def _projected_gradient(h, n, lr, n_iter):
    """Projected gradient descent on the feasible set"""
    dx = 2.0 / n
    h = _make_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = _corr_max(h, n)
    
    for it in range(n_iter):
        g = _corr_max_grad(h, n)
        h_new = h - lr * g
        h_new = _make_feasible(h_new, n)
        c5 = _corr_max(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Adaptive learning rate
        if it % 100 == 0 and it > 0:
            lr *= 0.95
    
    return best_h, best_c5

def _coordinate_descent(h, n, n_passes=50):
    """Coordinate descent"""
    h = _make_feasible(h, n).copy()
    target = n / 2.0
    c5 = _corr_max(h, n)
    
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            old = h[i]
            best_val = old
            best_c5 = c5
            # Try several values
            for trial in np.linspace(0, 1, 20):
                h[i] = trial
                # Adjust others to maintain sum
                diff = target - np.sum(h)
                if abs(diff) < 1e-10:
                    pass
                else:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    free = np.where(mask)[0]
                    if len(free) == 0:
                        h[i] = old
                        continue
                    # Proportionally adjust
                    capacity = np.ones(len(free))
                    adj = diff / len(free)
                    for idx, j in enumerate(free):
                        h[j] += adj
                    h = np.clip(h, 0, 1)
                    # Re-fix
                if np.any(h < -1e-9) or np.any(h > 1+1e-9):
                    h[i] = old
                    continue
                new_c5 = _corr_max(h, n)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_val = trial
            h[i] = best_val
            c5 = best_c5
            if best_val != old:
                improved = True
        if not improved:
            break
    
    h = _make_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, n):
    dx = 2.0 / n
    return float(np.max(_corr_vec(h, n))) * dx

def _c5_grad(h, n):
    dx = 2.0 / n
    corr = _corr_vec(h, n)
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        j1 = j - lag
        if 0 <= j1 < n:
            grad[j] += (1.0 - h[j1])
        j2 = j + lag
        if 0 <= j2 < n:
            grad[j] -= h[j2]
    return grad * dx

def _project(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def _slsqp(h0, n, maxiter=500):
    def obj(x):
        return _c5_val(x, n)
    def jac(x):
        return _c5_grad(x, n)
    def cf(x):
        return np.sum(x) - n / 2.0
    def cj(x):
        return np.ones(n)
    res = minimize(obj, h0, method='SLSQP', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    hx = _project(res.x, n)
    return hx, _c5_val(hx, n)

def _pgd(h0, n, lr0=0.02, niter=8000):

                best_c = cv
                best_h = cand.copy()
                break
            hx = cand
            break
        hx = _project(hx - lr * g, n)

        if c_now < best_c:
            best_c = c_now
            best_h = hx.copy()
        if it % 500 == 499:
            lr *= 0.9
    return best_h, best_c

    best_c5 = _c5_val(best_h, n_final)
    return best_h, best_c5, n_final
