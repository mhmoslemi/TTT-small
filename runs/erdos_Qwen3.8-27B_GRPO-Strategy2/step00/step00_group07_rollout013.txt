def fix_sum(h, n):

    target = n / 2.0
    diff = target - np.sum(h)

import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum_cont(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)

def sa_bin(h, tl, rs):
    n = len(h)
    h = (h > 0.5).astype(float)
