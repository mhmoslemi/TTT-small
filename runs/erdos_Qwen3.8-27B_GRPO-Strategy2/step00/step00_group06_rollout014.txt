    return float(np.max(corr)) * dx

def _obj(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * _h_dx

def _cons_fun(h):
    return np.sum(h) - _h_target

def _cons_jac(h):
    return np.ones(len(h))

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _slsqp(h0, n, target, maxit=500, ftol=1e-10):
    global _h_target, _h_dx
    _h_target = target
    _h_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
        return _project(res.x, target)
    except Exception:
        return _project(h0, target)

def _local_swap(h, n, n_iters, rng):
    best = h.copy()
    cur_c5 = compute_c5(best)
    for _ in range(n_iters):
        i, j = rng.choice(n, 2, replace=False)
        oi, oj = h[i], h[j]
        h[i], h[j] = oj, oi
        c5 = compute_c5(h)
        if c5 < cur_c5:
            cur_c5 = c5
            best = h.copy()
        else:
            h[i], h[j] = oi, oj
        for delta in [0.05, 0.1, 0.2]:
            vi, vj = min(1.0, oi + delta), max(0.0, oj - delta)
            if vi != oi:
                h[i], h[j] = vi, vj
                c5 = compute_c5(h)
                if c5 < cur_c5:
                    cur_c5 = c5
                    best = h.copy()
                else:
                    h[i], h[j] = oi, oj
            vi2, vj2 = max(0.0, oi - delta), min(1.0, oj + delta)
            if vi2 != oi:
                h[i], h[j] = vi2, vj2
                c5 = compute_c5(h)
                if c5 < cur_c5:
                    cur_c5 = c5
                    best = h.copy()
                else:
                    h[i], h[j] = oi, oj
    return best, cur_c5

def _round_ternary(h, target):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for n_half in range(0, min(5, n + 1)):
        n1 = target - 0.5 * n_half
        if n1 < 0 or n1 > n:
            continue
        if abs(n1 - round(n1)) > 1e-9:
            continue
        n1 = int(round(n1))
        n0 = n - n1 - n_half
        if n0 < 0:
            continue
        sorted_idx = np.argsort(h)[::-1]
        h_new = np.zeros(n)
        h_new[sorted_idx[:n1]] = 1.0
        h_new[sorted_idx[n1:n1 + n_half]] = 0.5
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _h_target, _h_dx
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    def elapsed():
        return time.time() - t0

    try:
        parent = np.array(initial_h_values, dtype=float)
    except NameError:
        parent = None

    grid_sizes = [95, 151, 199]

    for n in grid_sizes:
        if elapsed() > budget_s * 0.7:
            break
        target = n / 2.0
        _h_target = target
        _h_dx = 2.0 / n
        starts = []

        if parent is not None:
            p = parent.copy()
            if len(p) != n:
                x_old = np.linspace(0, 1, len(p), endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                p = np.interp(x_new, x_old, p)
            starts.append(_project(p, target))

        for scale in [0.15, 0.4]:
            h_r = np.zeros(n)
            h_r[::2] = 1.0
            h_r += rng.randn(n) * scale
            starts.append(_project(h_r, target))

        for _ in range(2):
            h_r = np.zeros(n)
            nf = int(target)
            idx = rng.permutation(n)[:nf]
            h_r[idx] = 1.0
            fr = target - nf
            if fr > 1e-9:
                h_r[rng.randint(n)] += fr
            starts.append(_project(h_r, target))

        for h0 in starts:
            if elapsed() > budget_s * 0.8:
                break
            h_opt = _slsqp(h0, n, target, maxit=300, ftol=1e-9)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

        if best_h is not None and len(best_h) == n:
            h_ls, c5_ls = _local_swap(best_h, n, 10000, rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

        if best_h is not None and len(best_h) == n:
            h_rd, c5_rd = _round_ternary(best_h, target)
            if c5_rd < best_c5:
                best_c5 = c5_rd
                best_h = h_rd
            if elapsed() < budget_s * 0.92:
                h_opt = _slsqp(h_rd, n, target, maxit=400, ftol=1e-11)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt

        if best_h is not None and len(best_h) == n and elapsed() < budget_s * 0.95:
            h_opt = _slsqp(best_h, n, target, maxit=600, ftol=1e-12)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    if best_h is not None and elapsed() < budget_s - 10:
        n_b = len(best_h)
        h_opt = _slsqp(best_h, n_b, n_b / 2.0, maxit=400, ftol=1e-12)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        if elapsed() < budget_s - 5:
            h_ls2, c5_2 = _local_swap(best_h, n_b, 10000, rng)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_ls2

    if best_h is None:
        n = 95
        best_h = np.zeros(n)
        best_h[:48] = 1.0
        best_h = _project(best_h, 47.5)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
