            h_r, c5_r = optimize_smooth_full(
                h_rand, [2.0, 0.5, 0.1], maxiter=100,
                time_limit=10, t0=t_start)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
