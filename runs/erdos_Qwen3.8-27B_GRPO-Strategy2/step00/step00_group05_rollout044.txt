    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:

                if t > 0:

                t = np.sum(mass)

                    h[mask] += diff * mass / t

    h = project_feasible(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # smooth max
    s = alpha * corr
    ms = np.max(s)
    e = np.exp(s - ms)
    se = np.sum(e)

            mask = h < 1.0
            if np.any(mask):
                room = 1.0 - h[mask]
                t = np.sum(room)
                if t > 0:
                    h[mask] += diff * room / t
        else:
            mask = h > 0.0
            if np.any(mask):
                mass = h[mask]
                t = np.sum(mass)
                if t > 0:
                    h[mask] += diff * mass / t
    return np.clip(h, 0.0, 1.0)

def obj_fn(x):
    n = _st_n
    alpha = _st_alpha
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    s = alpha * corr
    ms = np.max(s)
    e = np.exp(s - ms)
    se = np.sum(e)
    return (ms + np.log(se)) / alpha * dx

def grad_fn(x):
    n = _st_n
    alpha = _st_alpha
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    s = alpha * corr
    ms = np.max(s)
    e = np.exp(s - ms)
    se = np.sum(e)
    p = e / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-8:
            continue
        lag = k - (n - 1)
        lo1 = max(0, -lag)
        hi1 = min(n - 1, n - 1 - lag)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += p[k] * (1.0 - x[lo1+lag:hi1+lag+1])
        lo2 = max(0, lag)
        hi2 = min(n - 1, n - 1 + lag)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= p[k] * x[lo2-lag:hi2-lag+1]
    return grad * dx

def _sum_fun(x):
    return float(np.sum(x) - _st_n / 2.0)

def _sum_jac(x):
    return np.ones(_st_n)

def _make_init(n, rng):
    h = np.zeros(n)
    for i in range(0, n, 2):
        h[i] = 1.0
    if np.sum(h) != n / 2:
        h[n // 2] = 1.0 - h[n // 2]
    h = project_feasible(h, n)
    h += rng.randn(n) * 0.1
    return project_feasible(h, n)

def _optimize(h0, n, alpha, maxiter, t_deadline):
    global _st_n, _st_alpha
    _st_n = n
    _st_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-9})
        h_out = project_feasible(res.x, n)
        return h_out
    except Exception:
        return project_feasible(h0, n)

def _pgd_refine(h, n, alpha, steps, t_deadline):
    global _st_n, _st_alpha
    _st_n = n
    _st_alpha = alpha
    best = h.copy()
    best_f = compute_c5(h)
    lr = 0.005
    for s in range(steps):
        if time.time() > t_deadline:
            break
        g = grad_fn(h)
        h_new = project_feasible(h - lr * g, n)
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best = h_new.copy()
            best_f = f_new
            h = h_new
        else:
            lr *= 0.7
        if lr < 1e-6:
            break
    return best

def run(seed=42, budget_s=1100, **kwargs):
    global _st_n, _st_alpha
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.92

    best_h = None
    best_c5 = float('inf')

    n_candidate = 70
    try:
        init = np.array(initial_h_values, dtype=float)
        n_candidate = len(init)
        best_h = project_feasible(init, n_candidate)
        best_c5 = compute_c5(best_h)
    except Exception:
        pass

    n_list = sorted(set([n_candidate, 50, 60, 70, 80, 100]))

    for n in n_list:
        if time.time() > deadline:
            break
        _st_n = n
        _st_alpha = 30.0

        if best_h is not None and len(best_h) == n:
            h0 = best_h.copy()
        else:
            h0 = _make_init(n, rng)

        h_opt = _optimize(h0, n, 30.0, 200, deadline)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        for r in range(4):
            if time.time() > deadline:
                break
            scale = 0.3 + 0.2 * r
            if len(best_h) == n:
                h_pert = project_feasible(best_h + rng.randn(n) * scale, n)
            else:
                h_pert = _make_init(n, rng)
            h_opt = _optimize(h_pert, n, 30.0, 150, deadline)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final refinement with increasing alpha
    if best_h is not None:
        n = len(best_h)
        for alpha in [50, 100, 200, 400]:
            if time.time() > deadline:
                break
            best_h = _optimize(best_h, n, alpha, 150, deadline)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5:
                best_c5 = c5_new
            else:
                break

        # PGD fine-tuning
        best_h = _pgd_refine(best_h, n, 200, 500, deadline)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new

    if best_h is None:
        h = np.ones(70) * 0.5
        best_h = h
        best_c5 = compute_c5(h)

    return best_h, best_c5, len(best_h)
