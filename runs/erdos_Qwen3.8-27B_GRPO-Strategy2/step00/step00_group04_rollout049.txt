import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def soft_obj(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + tau * np.log(np.sum(np.exp((corr - m) / tau))))


def repair_integral(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def optimize_tau(h0, n, tau, maxiter=200):
    def obj(h):
        return soft_obj(h, n, tau)
    def con(h):
        return np.sum(h) - n / 2.0
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
        h_new = repair_integral(res.x, n)
        return h_new
    return h0


def local_refine(h, n, iters=5000):
    """Hill-climbing: try small perturbations to individual coords."""
    best_val = compute_c5(h)
    dx = 2.0 / n
    for _ in range(iters):
        i = np.random.randint(n)
        delta = np.random.choice([-0.05, -0.02, -0.01, 0.01, 0.02, 0.05])
        h2 = h.copy()
        h2[i] = np.clip(h2[i] + delta, 0.0, 1.0)
        # Fix integral: adjust another element
        diff = h[i] - h2[i]
        if abs(diff) > 1e-12:
            # Spread correction over random element
            j = np.random.randint(n)
            h2[j] = np.clip(h2[j] - diff, 0.0, 1.0)
        if abs(np.sum(h2) - n / 2.0) > 1e-8:
            h2 = repair_integral(h2, n)
        v2 = compute_c5(h2)
        if v2 < best_val - 1e-12:
            h = h2
            best_val = v2
    return h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

    def remaining(safety=15):
        return budget_s - elapsed() - safety

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = np.ones(42) * 0.5
    n0 = len(h0)
    h0 = repair_integral(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0)

    # Phase 1: SLSQP with annealed tau at base resolution
    tau_seq = [0.5, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
    for tau in tau_seq:
        if remaining() < 10:
            break
        h1 = optimize_tau(best_h, best_n, tau, maxiter=150)
        v1 = compute_c5(h1)
        if v1 < best_val:
            best_val = v1
            best_h = h1

    # Phase 2: Local refinement
    if remaining() > 30:
        h2, v2 = local_refine(best_h, best_n, iters=8000)
        if v2 < best_val:
            best_val = v2
            best_h = h2

    # Phase 3: SLSQP again after local refine
    for tau in [0.02, 0.01, 0.005]:
        if remaining() < 10:
            break
        h3 = optimize_tau(best_h, best_n, tau, maxiter=100)
        v3 = compute_c5(h3)
        if v3 < best_val:
            best_val = v3
            best_h = h3

    # Phase 4: Try random restarts at same n
    if remaining() > 40:
        for trial in range(8):
            if remaining() < 20:
                break
            pert = best_h + np.random.randn(best_n) * 0.15
            pert = repair_integral(pert, best_n)
            hp = optimize_tau(pert, best_n, 0.05, maxiter=80)
            hp = optimize_tau(hp, best_n, 0.01, maxiter=60)
            vp = compute_c5(hp)
            if vp < best_val:
                best_val = vp
                best_h = hp

    # Phase 5: Local refine again
    if remaining() > 20:
        h5, v5 = local_refine(best_h, best_n, iters=5000)
        if v5 < best_val:
            best_val = v5
            best_h = h5

    # Phase 6: Scale up resolution and re-optimize
    if remaining() > 50:
        for scale in [2, 3]:
            if remaining() < 30:

            n_new = best_n * scale
            if n_new > 400:
                break
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
            # PCHIP-like interpolation using scipy
            from scipy.interpolate import PchipInterpolator
            # Ensure periodic for interpolation
            x_ext = np.concatenate([x_old, [2.0 + x_old[0]]])
            h_ext = np.concatenate([best_h, [best_h[0]]])
            try:
                pchip = PchipInterpolator(x_ext, h_ext)
                h_interp = pchip(x_new)
            except Exception:
                h_interp = np.interp(x_new, x_old,
                                     np.concatenate([best_h, best_h[0:1]]))
            h_interp = repair_integral(h_interp, n_new)
            v_interp = compute_c5(h_interp)

            # Optimize at higher resolution
            vh = optimize_tau(h_interp, n_new, 0.05, maxiter=100)
            vh = optimize_tau(vh, n_new, 0.01, maxiter=80)
            v_hi = compute_c5(vh)
            if v_hi < best_val:
                best_val = v_hi
                best_h = vh
                best_n = n_new

    # Final local polish
    if remaining() > 10:
        h6, v6 = local_refine(best_h, best_n, iters=3000)
        if v6 < best_val:
            best_val = v6
            best_h = h6

    # Final repair
    best_h = repair_integral(best_h, best_n)
    final_val = compute_c5(best_h)

    # Verify
    assert best_n == len(best_h)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, f"Integral: {np.sum(best_h)} vs {best_n/2}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)

    return (best_h.tolist(), final_val, best_n)
