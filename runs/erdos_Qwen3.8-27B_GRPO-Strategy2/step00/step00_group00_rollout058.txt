import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    shifted = corr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def normalize_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 0:
        h = h * (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def obj_wrapper(x, alpha):
    return smooth_max_obj(x, alpha)

def optimize_slsqp(h0, n, alpha=80.0, maxiter=300):
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj_wrapper, h0, method='SLSQP',
        bounds=bounds, constraints=cons,
        args=(alpha,),
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return normalize_h(res.x, n)

    return 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
