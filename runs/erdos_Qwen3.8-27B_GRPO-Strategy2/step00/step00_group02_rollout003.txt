                # term 1: h[j] as the "shifted" part
                if 0 <= ii < n:
                    grad[j] += wk * (1.0 - h[ii])
                # term 2: h[j] as the "base" part
                if 0 <= j + lag < n:
                    grad[j] += wk * (-h[j + lag])
    return grad

def make_random_h(n, rng):
    h = rng.random(n)
    return project_box_hyperplane(h, n / 2.0)

def make_block_h(n, nb, rng):
    h = np.zeros(n)
    bs = n // nb
    for b in range(nb):
        s = b * bs
        e = min(s + bs, n)
        h[s:e] = (n / 2.0 / nb) / max(e - s, 1)
    noise = rng.normal(0, 0.03, n)
    h = project_box_hyperplane(h + noise, n / 2.0)
    return h

def optimize_from(h0, n, max_time):
    global _n_global, _t_global
    _n_global = n
    h_curr = project_box_hyperplane(h0, n / 2.0).copy()
    c5_curr = compute_c5(h_curr, n)
    for t in [5.0, 15.0, 50.0, 150.0, 500.0]:
        _t_global = t
        if time.time() > max_time:
            break
        try:
            res = minimize(_obj_func, h_curr, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
            h_new = project_box_hyperplane(res.x, n / 2.0)
            c5_new = compute_c5(h_new, n)
            if c5_new < c5_curr:
                h_curr = h_new
                c5_curr = c5_new
        except Exception:
            pass
    return h_curr, c5_curr

def local_search(h0, n, rng, max_time, max_iters=500):
    h = project_box_hyperplane(h0, n / 2.0).copy()
    c5_curr = compute_c5(h, n)
    step = 0.08
    for it in range(max_iters):
        if time.time() > max_time:
            break
        k = max(2, n // 15)
        idx = rng.choice(n, size=k, replace=False)
        h_p = h.copy()
        h_p[idx] += step * rng.standard_normal(k)
        h_p = project_box_hyperplane(h_p, n / 2.0)
        c5_p = compute_c5(h_p, n)
        if c5_p < c5_curr:
            h = h_p
            c5_curr = c5_p
            step = min(step * 1.1, 0.15)
        else:
            step = max(step * 0.9, 0.01)
    return h, c5_curr

def run(seed=42, budget_s=1100, **kwargs):
    global _n_global, _t_global
    start_time = time.time()
    rng = np.random.default_rng(seed)
    deadline = start_time + budget_s - 5.0

    n_init = len(initial_h_values)
    best_h = np.array(initial_h_values, dtype=float)
    best_n = n_init
    best_c5 = compute_c5(best_h, best_n)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Optimize the initial construction at its native resolution
    if elapsed() < budget_s * 0.3:
        h_opt, c5_opt = optimize_from(best_h, n_init, start_time + budget_s * 0.3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_init
        h_ls, c5_ls = local_search(h_opt, n_init, rng, start_time + budget_s * 0.35, 300)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n_init

    # Phase 2: Try other resolutions
    n_list = sorted(set([n_init, 45, 50, 55, 60, 75, 90]))
    n_list = [n for n in n_list if 30 <= n <= 120]

    for n in n_list:
        if elapsed() > budget_s - 80:
            break

        # Build initial candidates
        candidates = []
        if n == best_n:
            candidates.append(best_h.copy())
        elif best_n > 0:
            x_old = np.linspace(0.0, 2.0, best_n, endpoint=False) + 1.0 / best_n
            x_new = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
            f = interp1d(x_old, best_h, kind='cubic', bounds_error=False, fill_value=0.5)
            h_interp = f(x_new)
            candidates.append(project_box_hyperplane(h_interp, n / 2.0))

        for nb in [3, 4, 5]:
            candidates.append(make_block_h(n, nb, rng))

        for _ in range(3):
            candidates.append(make_random_h(n, rng))

        for cand in candidates:
            if elapsed() > deadline:
                break
            time_left_frac = max(0.01, (deadline - time.time()) / max(budget_s - elapsed(), 1.0))
            t_budget = start_time + budget_s * 0.3 + max(2.0, (budget_s - 80) * time_left_frac * 0.08)
            h_opt, c5_opt = optimize_from(cand, n, min(t_budget, deadline))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            if elapsed() > deadline - 20:
                break
            h_ls, c5_ls = local_search(h_opt, n, rng, min(elapsed() + 15, deadline - 5), 200)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n

    # Phase 3: Final polish on the best solution found
    if elapsed() < deadline - 20:
        h_fin, c5_fin = optimize_from(best_h, best_n, deadline - 10)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = best_n
        h_ls, c5_ls = local_search(best_h, best_n, rng, deadline, 300)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = best_n

    # Final validation
    best_h = project_box_hyperplane(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
