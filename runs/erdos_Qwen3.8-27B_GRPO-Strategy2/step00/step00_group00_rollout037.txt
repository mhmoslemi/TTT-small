    dx = 2.0 / n

    tcorr_shifted = tcorr - np.max(tcorr)
    weights = np.exp(tcorr_shifted) / np.sum(np.exp(tcorr_shifted))  # shape (n_full,)

def grad_corr_s(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[0:n-s] += 1.0 - h[s:]
            g[s:] -= h[0:n-s]
    else:
        ss = -s
        if n - ss > 0:
            g[ss:] += 1.0 - h[0:n-ss]
            g[0:n-ss] -= h[ss:]
    return g

def obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    n_full = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode="full") * dx
    
    tcorr = tau * corr
    tmax = np.max(tcorr)
    exps = np.exp(tcorr - tmax)
    weights = exps / np.sum(exps)
    obj = (tmax + np.log(np.sum(exps))) / tau
    
    grad = np.zeros(n)
    for m in range(n_full):
        s = n - 1 - m
        grad += weights[m] * grad_corr_s(h, s)
    grad *= dx
    
    return obj, grad

            g[s:] -= h[0:n-s]

    res = minimize(_inner_obj, h0, method='SLSQP', jac=_inner_jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def _coord_min_scalar(c0, a):
    lo_v, hi_v = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1v = hi_v - gr * (hi_v - lo_v)
    c2v = lo_v + gr * (hi_v - lo_v)
    f1 = np.max(a * c1v + c0)
    f2 = np.max(a * c2v + c0)
    for _ in range(50):
        if hi_v - lo_v < 1e-12:
            break
        if f1 < f2:
            hi_v = c2v
            c2v = c1v
            f2 = f1
            c1v = hi_v - gr * (hi_v - lo_v)
            f1 = np.max(a * c1v + c0)
        else:
            lo_v = c1v
            c1v = c2v
            f1 = f2
            c2v = lo_v + gr * (hi_v - lo_v)
            f2 = np.max(a * c2v + c0)
    return (lo_v + hi_v) * 0.5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 0: try small perturbations
    for trial in range(10):
        if time.time() - t0 > budget_s * 0.05:
            break
        noise = np.random.randn(n) * 0.02
        hp = project_sum_box(h0 + noise, target)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    # Phase 1: SLSQP with increasing tau
    for tau in [30, 60, 100, 200, 500, 1000]:
        if time.time() - t0 > budget_s * 0.40:
            break
        try:
            x_new = _make_slsqp_call(best_h, n, tau, 500)
            x_new = project_sum_box(x_new, target)
            c5 = compute_c5(x_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_new
        except Exception:
            pass

    # Phase 2: projected subgradient
    if time.time() - t0 < budget_s * 0.65:
        h_curr = best_h.copy()
        for it in range(80000):
            if time.time() - t0 > budget_s * 0.65:
                break
            corr = compute_corr_vals(h_curr)
            c5 = np.max(corr)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_curr.copy()
            tol = max(1e-6, 2e-5 * (it + 1) * 0.01)
            active = np.where(np.abs(corr - c5) < tol)[0]
            g = np.zeros(n)
            for m in active:
                s_val = n - 1 - m
                g += grad_corr_s(h_curr, s_val)
            g /= len(active)
            g *= dx
            alpha = 0.08 / (1.0 + 0.002 * it)
            h_curr = project_sum_box(h_curr - alpha * g, target)
            if it % 5000 == 0:
                c5_chk = compute_c5(h_curr)
                if c5_chk < best_c5:
                    best_c5 = c5_chk
                    best_h = h_curr.copy()

    # Phase 3: coordinate descent
    if time.time() - t0 < budget_s * 0.90:
        h_curr = best_h.copy()
        n_full = 2 * n - 1
        for sweep in range(400):
            if time.time() - t0 > budget_s * 0.90:
                break
            for i in range(n):
                h_curr[i] = 0.0
                c0 = np.correlate(h_curr, 1 - h_curr, mode="full") * dx
                h_curr[i] = 1.0
                c1 = np.correlate(h_curr, 1 - h_curr, mode="full") * dx
                a = c1 - c0
                v_new = _coord_min_scalar(c0, a)
                h_curr[i] = v_new
            h_curr = project_sum_box(h_curr, target)
            c5 = compute_c5(h_curr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
                if c5 < 0.379:
                    break

    # Phase 4: final SLSQP polish on the best
    if time.time() - t0 < budget_s * 0.97:
        try:
            x_new = _make_slsqp_call(best_h, n, 2000, 300)
            x_new = project_sum_box(x_new, target)
            c5 = compute_c5(x_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_new
        except Exception:
            pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
