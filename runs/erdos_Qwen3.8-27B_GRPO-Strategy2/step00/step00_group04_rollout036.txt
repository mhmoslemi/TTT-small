        bh, bc = improve_solution(h_kick, n, seed + 2000 + kick_count, time_left)
        if bc < best_c5:
            best_h = bh.copy()
            best_c5 = bc
        kick_count += 1
    
    # Final greedy polish
    if time.time() - t_start < budget_s - 3:
        bh, bc = greedy_ls(best_h, n, 500)
        if bc < best_c5:
            best_h = bh.copy()
            best_c5 = bc
    
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
