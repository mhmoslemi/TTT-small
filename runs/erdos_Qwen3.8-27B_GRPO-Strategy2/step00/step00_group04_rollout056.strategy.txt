import numpy as np
from scipy.optimize import minimize

def compute_objective(h, n):
    dx = 2.0 / n
    cores = np.correlate(h, 1-h, mode="full")
    return np.max(cores) * dx

def repair_constraint(h, n, tol=1e-5):
    # Integral-conserving adjustment
    s = np.sum(h)
    target = n/2
    if abs(s - target) < tol:
        return h
    
    # Weighted adjustment
    diff = target - s
    avg_weight = 1/n
    adjusted = h + diff * avg_weight
    
    # Maintain bounds and integral
    bounded = np.clip(adjusted, 0, 1)
    if abs(np.sum(bounded) - target) < tol:
        return bounded
    
    # Linear projection to meet constraint
    scaling = (target - np.sum(bounded)) / n
    proj = bounded + scaling
    return np.clip(proj, 0, 1)

def combinatorial_search(init_h, n, max_iter, temp):
    best = init_h.copy()
    best_val = compute_objective(best, n)
    
    for _ in range(max_iter):
        # Stochastic perturbation
        idxs = np.random.choice(n, size=min(5, n), replace=False)
        deltas = np.random.uniform(-0.5, 0.5, size=len(idxs))
        
        next_h = init_h.copy()
        for i, d in zip(idxs, deltas):
            next_h[i] = max(0, min(1, next_h[i] + d))
            
        next_h = repair_constraint(next_h, n)
        new_val = compute_objective(next_h, n)
        
        # Simulated annealing
        if new_val < best_val or np.random.rand() < np.exp(-(new_val - best_val) / temp):
            best_val = new_val
            best = next_h.copy()
        
    return best, best_val

def smooth_optimization(h0, n, budget):
    # Setup objective and constraints
    def objective(h):
        return compute_objective(h, n)
    
    def constraint(h):
        return np.sum(h) - n/2
    
    # Gradient penalty (for constraint projection)
    def constraint_jacobian(h):
        return np.ones(len(h)) / n
    
    bounds = [(0,1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': constraint_jacobian}
    
    # First optimize using trust-constrained
    res = minimize(objective, h0, method='trust-constr', 
                   bounds=bounds, constraints=cons,
                   options={'gtol': 1e-5, 'max_iter': int(budget*0.9)})
    
    return res.x
