import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_beta = 40.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, beta):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def objective_fn(x):
    vals = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / _g_n
    vals = vals * dx
    return smooth_max(vals, _g_beta)

def constraint_fn(x):
    return x.sum() - _g_target

def enforce_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                adj = min(1.0 - h[i], diff)
                h[i] += adj
                diff -= adj
            else:
                adj = max(-h[i], diff)
                h[i] += adj
                diff -= adj
    return h

def do_slsqp(x0, beta, maxiter, ftol):
    global _g_beta
    _g_beta = beta
    constraints = [{'type': 'eq', 'fun': constraint_fn}]
    try:
        res = minimize(objective_fn, x0, method='SLSQP',
                       bounds=_g_bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res
    except Exception:
        return None

def jitter_search(best_h, best_c5, n, target_sum, steps, step_size):
    rng_j = np.random.default_rng(12345)
    for it in range(steps):
        if it % 500 == 499:
            step_size *= 0.7
        i, j = rng_j.integers(0, n, 2)
        if i == j:
            continue
        d = rng_j.uniform(-step_size, step_size)
        h_try = best_h.copy()
        h_try[i] += d
        h_try[j] -= d
        h_try = np.clip(h_try, 0, 1)
        if abs(h_try.sum() - target_sum) < 1e-9:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    return best_h, best_c5

def shift_focused(best_h, best_c5, n, target_sum, max_iter):
    rng_s = np.random.default_rng(999)
    for it in range(max_iter):
        corr_vals = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(corr_vals))
        shift = k_star - (n - 1)
        if shift == 0:
            i_candidates = np.arange(n)
        elif shift > 0:
            i_candidates = np.arange(0, n - shift)
        else:
            i_candidates = np.arange(-shift, n)
        if len(i_candidates) < 2:
            continue
        i = i_candidates[rng_s.integers(0, len(i_candidates))]
        j = i + shift
        if 0 <= j < n:
            d = rng_s.uniform(-0.08, 0.08)
            h_try = best_h.copy()
            if shift > 0:
                h_try[i] -= d
                h_try[j] += d
            else:
                h_try[i] += d
                h_try[j] -= d
            h_try = np.clip(h_try, 0, 1)
            if abs(h_try.sum() - target_sum) < 1e-9:
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _g_n, _g_target, _g_beta, _g_bounds
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

        return budget_s - (time.time() - t0)

    # Phase 1: Progressive beta SLSQP from initial
    for beta in [5, 10, 20, 40, 80, 150, 300]:
        if time_left() < 45:
            break
        res = do_slsqp(best_h, beta, 400, 1e-10)

            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Shift-focused local search
    if time_left() > 30:

    # Phase 3: Random restarts with SLSQP
    n_restarts = 35
    for r in range(n_restarts):
        if time_left() < 25:

        if r < 6:

        elif r < 15:

        x0 = enforce_feasible(x0, n)

                best_c5 = c5_new

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
