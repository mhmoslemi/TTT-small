```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def project(h, n):
    # Project to {0 ≤ h ≤ 1, sum(h) = n/2}
    h = np.clip(h, 0, 1)
    diff = n/2.0 - h.sum()
    if diff > 0:
        # Need to increase sum: increase zeros first, then small values
        for _ in range(100):
            order = np.argsort(-h)  # most room to grow
            mask = h[order] < 1.0
            if not mask.any():
                break
            idx = order[mask]
            room = 1.0 - h[idx]
            alloc = np.minimum(room, diff / len(idx))
            h[idx] += alloc
            diff -= alloc.sum()
            if diff <= 1e-10:
                break
    elif diff < 0:
        for _ in range(100):
            order = np.argsort(h)  # smallest values
            mask = h[order] > 0.0
            if not mask.any():
                break
            idx = order[mask]
            room = h[idx]
            alloc = np.minimum(room, -diff / len(idx))
            h[idx] -= alloc
            diff += alloc.sum()
            if diff <= 1e-10:
                break
    return h
```

```python
    for it in range(n_iter):

        k_star = np.argmax(corr)

        grad = np.zeros(n)

            h[j] += 1e-7

            h[j] -= 1e-7
            h[j] -= 1e-7
```

```python
def c5_and_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = corr.max() * dx
    k_star = corr.argmax()
    lag = k_star - (n - 1)
    
    idx = np.arange(n)
    grad = np.zeros(n)
    
    if lag >= 0:
        mask_first = idx <= n - 1 - lag  # j ≤ N-1-lag
        mask_second = idx >= lag  # j ≥ lag
        # First part: (1 - h[j-lag]) for j in [lag, N-1]
        j_range = np.arange(lag, n)
        grad[j_range] += (1.0 - h[j_range - lag])
        # Second part: -h[j+lag] for j in [0, N-1-lag]
        j_range2 = np.arange(0, n - lag)
        grad[j_range2] -= h[j_range2 + lag]
    else:
        m = -lag
        # (1 - h[j+m]) for j in [0, N-1-m]
        j_range = np.arange(0, n - m)
        grad[j_range] += (1.0 - h[j_range + m])
        # -h[j-m] for j in [m, N-1]
        j_range2 = np.arange(m, n)
        grad[j_range2] -= h[j_range2 - m]
    
    return c5, grad * dx
```

A better approach: use a smoothed max. Define:
f(h) = (1/α) * log(sum_k exp(α * corr[k] * dx))

This is a smooth approximation to max_k corr[k]*dx. As α → ∞, it approaches the max. For finite α, the gradient is a weighted average of the gradients of all components.

gradient of f: 
∂f/∂h[j] = (1/α) * d/dh[j] log(sum exp(α*c_k*dx))
= (1/α) * sum_k exp(α*c_k*dx) * α*dx * ∂c_k/∂h[j] / sum_l exp(α*c_l*dx)
= dx * sum_k w_k * ∂c_k/∂h[j]

where w_k = exp(α*c_k*dx) / sum_l exp(α*c_l*dx) (softmax weights)

This is the expected gradient under the softmax distribution. As α increases, it concentrates on the max.

This gives a smooth, differentiable objective. I can use L-BFGS-B on this (with the constraint handled by parameterization or penalty).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h, n):
    """Returns (c5_value, gradient_of_active_component * dx)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        j1 = np.arange(lag, n)
        grad[j1] += (1.0 - h[j1 - lag])
        j2 = np.arange(0, n - lag)
        grad[j2] -= h[j2 + lag]
    else:
        m = -lag
        j1 = np.arange(0, n - m)
        grad[j1] += (1.0 - h[j1 + m])
        j2 = np.arange(m, n)
        grad[j2] -= h[j2 - m]
    return corr[k_star] * dx, grad * dx

def project_feasible(h, n):
    h = h.copy().astype(float)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff * h / s
        np.clip(h, 0.0, 1.0, out=h)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def proj_grad_opt(h, n, n_iters=3000, lr=0.05):
    h = h.copy()
    for it in range(n_iters):
        _, grad = c5_gradient(h, n)
        grad -= grad.mean()  # Project to sum=0
        h -= lr * grad
        h = project_feasible(h, n)
    return h

def slsqp_opt(h0, n, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.fun < np.inf:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_obj_and_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    val = corr[idx] * dx
    lag = idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        j1 = np.arange(lag, n)
        grad[j1] += (1.0 - h[j1 - lag])
        j2 = np.arange(0, n - lag)
        grad[j2] -= h[j2 + lag]
    else:
        m = -lag
        j1 = np.arange(0, n - m)
        grad[j1] += (1.0 - h[j1 + m])
        j2 = np.arange(m, n)
        grad[j2] -= h[j2 - m]
    return val, grad * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14:
                break
            h += d * r / s
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += d * h / s
        np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()
    if abs(d) > 1e-15:
        i = int(np.argmax(1.0 - h)) if d > 0 else int(np.argmax(h))
        h[i] += d
        h[i] = float(np.clip(h[i], 0.0, 1.0))
    return h

def slsqp(h0, n, mi=200):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq',
          'fun': lambda x: x.sum() - n * 0.5,
          'jac': lambda x: np.ones(n)}]
    def f(x):
        return compute_c5(x)
    def g(x):
        _, gr = c5_obj_and_grad(x, n)
        return gr
    try:
        r = minimize(f, h0, jac=g, method='SLSQP',
                     bounds=b, constraints=c,
```
