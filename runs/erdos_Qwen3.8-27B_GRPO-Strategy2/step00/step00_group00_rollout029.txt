import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        d = -k
        if d < n:
            g[:n - d] += (1.0 - h[d:])
            g[d:] -= h[:n - d]
    return g

def _objective(h_flat, n):
    h = h_flat.reshape(n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def _gradient(h_flat, n):
    h = h_flat.reshape(n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = float(np.max(corr))
    tol = 0.03 * max_val + 1e-3
    active = np.where(corr >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    grad = np.zeros(n)
    for m in active:
        k = int(m) - (n - 1)
        grad += _grad_lag(h, k)
    grad *= dx / len(active)
    return grad

def _sum_con(h_flat, n):
    return float(np.sum(h_flat) - n / 2.0)

def _project(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _slsqp(h0, n, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        _objective, h0.copy(), args=(n,),
        method='SLSQP', jac=_gradient,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (n,)}],
        options={'maxiter': max_iter, 'ftol': ftol, 'disp': False}
    )
    return np.asarray(res.x, dtype=float)

def _binary_snap(h, n):
    if n % 2 != 0:
        return h
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin == 1)[0]
        order = idx[np.argsort(h[idx])]
        for i in range(n1 - target):
            h_bin[order[i]] = 0.0
    elif n1 < target:
        idx = np.where(h_bin == 0)[0]
        order = idx[np.argsort(-h[idx])]
        for i in range(target - n1):
            h_bin[order[i]] = 1.0
    return h_bin

def _local_refine(h, n, rng, steps=200):
    c5_cur = compute_c5(h)
    for _ in range(steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        eps = 0.05 * rng.rand()
        if h[i] < eps or h[j] > 1.0 - eps:
            continue
        h[i] -= eps
        h[j] += eps
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[i] += eps
            h[j] -= eps
    return h, c5_cur
