def smooth_max_grad(h, n, temperature=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Soft max
    shifted = corr * temperature
    shifted -= np.max(shifted)  # for numerical stability
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    f_smooth = np.log(np.sum(np.exp(shifted))) / temperature + np.max(corr)
    # Actually: soft_max = (1/t) * log(sum(exp(t*x_i)))
    # gradient_i = (1/t) * sum_j weights_j * grad_f_j
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        shift = k - (n-1)
        if shift >= 0:
            grad[:n-shift] += weights[k] * (1.0 - h[shift:shift+(n-shift)])
            grad[shift:] += weights[k] * (-h[:n-shift])
        else:
            s = -shift
            grad[s:] += weights[k] * (1.0 - h[:n-s])
            grad[:n-s] += weights[k] * (-h[s:])
    grad *= dx
    return f_smooth, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def compute_c5_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - shift] += (1.0 - h[shift:shift + n - shift])
        grad[shift:shift + n - shift] += -h[:n - shift]
    else:
        s = -shift
        grad[s:s + n - s] += (1.0 - h[:n - s])
        grad[:n - s] += -h[s:s + n - s]
    grad *= dx
    return float(corr[k_max]), grad

def project_box_sum(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - t, 0.0, 1.0))
        if s > target_sum:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    result = np.clip(h - t, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
        free = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(free) > 0:
            adj = np.clip(diff, -result[free[0]], 1.0 - result[free[0]])
            result[free[0]] += adj
    return result

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def pgd_optimize(h, n, max_iter=5000, t_start=None, budget_s=1100, stop_frac=0.95):
    target_sum = n / 2.0
    best_h = h.copy()
    best_f = compute_c5(h, n)
    lr = 2.0
    stall = 0

    for it in range(max_iter):
        if t_start is not None and time.time() - t_start > budget_s * stop_frac:
            break
        f, grad = compute_c5_subgrad(h, n)
        if f < best_f - 1e-12:
            best_f = f
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

        if np.linalg.norm(grad) < 1e-10:
            break

        alpha = lr
        improved = False
        for _ in range(30):
            h_new = project_box_sum(h - alpha * grad, target_sum)
            f_new = compute_c5(h_new, n)
            if f_new < f - 1e-14:
                improved = True
                break
            alpha *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break
        else:
            h = h_new
            if f_new < best_f - 1e-12:
                best_f = f_new
                best_h = h_new.copy()

    return best_h, best_f

def slsqp_optimize(h, n, t_start=None, budget_s=1100, stop_frac=0.95):
    dx = 2.0 / n
    target_sum = n / 2.0

    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full") * dx
        return np.max(corr)

    def obj_grad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full") * dx
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n - shift] += (1.0 - hv[shift:])
            grad[shift:] += -hv[:n - shift]
        else:
            s = -shift
            grad[s:] += (1.0 - hv[:n - s])
            grad[:n - s] += -hv[s:]
        return grad * dx

    cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target_sum,
             'jac': lambda hv: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h, method='SLSQP', jac=obj_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-9})
        h_opt = project_box_sum(result.x, target_sum)
        return h_opt, compute_c5(h_opt, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()

    best_h = None
    best_n = 0
    best_c5 = 1.0

    def try_update(h_cand, n_cand):
        nonlocal best_h, best_n, best_c5
        c5_cand = compute_c5(h_cand, n_cand)
        if c5_cand < best_c5:
            best_h = h_cand.copy()
            best_n = n_cand
            best_c5 = c5_cand
            return True
        return False

    # Starting points
    start_points = []
    if h_init is not None:
        n0 = len(h_init)
        h0 = project_box_sum(h_init, n0 / 2.0)
        start_points.append((h0, n0))
    else:
        start_points.append((np.full(45, 1.0 / 45.0 * 2.0), 45))

    # Two-block: h=1 on [0,1], h=0 on [1,2]
    n_blk = 50
    h_blk = np.zeros(n_blk)
    h_blk[:n_blk // 2] = 1.0
    start_points.append((h_blk, n_blk))

    # Random
    h_rnd = project_box_sum(rng.rand(50), 25.0)
    start_points.append((h_rnd, 50))

    # Phase 1: Optimize each starting point at its native resolution
    for h_sp, n_sp in start_points:
        if time.time() - t_start > budget_s * 0.1:
            break
        h_sp = project_box_sum(h_sp, n_sp / 2.0)
        try_update(h_sp, n_sp)
        h_opt, c5_opt = pgd_optimize(h_sp, n_sp, max_iter=5000,
                                     t_start=t_start, budget_s=budget_s, stop_frac=0.12)
        try_update(h_opt, n_sp)

    # Phase 2: Upsample best and optimize at higher resolutions
    if best_h is None:
        best_h = start_points[0][0].copy()
        best_n = start_points[0][1]
        best_c5 = compute_c5(best_h, best_n)

    target_resolutions = []
    base_n = best_n
    for r in [90, 120, 180, 200, 300, 400, 600]:
        if r > base_n:
            target_resolutions.append(r)

    for n_t in target_resolutions:
        if time.time() - t_start > budget_s * 0.70:
            break
        h_up = upsample(best_h, n_t)
        h_up = project_box_sum(h_up, n_t / 2.0)
        try_update(h_up, n_t)

        h_opt, c5_opt = pgd_optimize(h_up, n_t, max_iter=4000,
                                     t_start=t_start, budget_s=budget_s, stop_frac=0.70)
        try_update(h_opt, n_t)

        # SLSQP refinement
        if time.time() - t_start < budget_s * 0.70 and n_t <= 200:
            h_slq, c5_slq = slsqp_optimize(h_opt, n_t,
                                           t_start=t_start, budget_s=budget_s, stop_frac=0.70)
            try_update(h_slq, n_t)

    # Phase 3: Random restarts around best solution
    if time.time() - t_start < budget_s * 0.90:
        n_work = best_n
        perturb_scale = 0.15
        max_restarts = int((budget_s * 0.88 - (time.time() - t_start)) / 3)
        max_restarts = min(max_restarts, 30)

        for r in range(max_restarts):
            if time.time() - t_start > budget_s * 0.88:
                break
            h_pert = best_h + rng.normal(0, perturb_scale, n_work)
            h_pert = project_box_sum(h_pert, n_work / 2.0)
            h_opt, c5_opt = pgd_optimize(h_pert, n_work, max_iter=3000,
                                         t_start=t_start, budget_s=budget_s, stop_frac=0.88)
            try_update(h_opt, n_work)
            perturb_scale *= 0.85

    # Phase 4: Final SLSQP polish if time permits
    if time.time() - t_start < budget_s * 0.93 and best_n <= 150:
        h_slq, c5_slq = slsqp_optimize(best_h, best_n,
                                       t_start=t_start, budget_s=budget_s, stop_frac=0.95)
        try_update(h_slq, best_n)

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)

    return (best_h, best_c5, best_n)
