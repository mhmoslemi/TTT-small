import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_active(h, n, dx):
    g1 = 1.0 - h
    corr = np.correlate(h, g1, mode="full")
    mx = np.max(corr)
    tol = 1e-9
    active = np.where(corr >= mx - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n - lag] += g1[lag:]
            grad[lag:] -= h[:n - lag]
        else:
            m = -lag
            grad[m:] += g1[:n - m]
            grad[:n - m] -= h[m:]
    grad /= float(len(active))
    return grad * dx

def project_feasible(h, n):
    target = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def optimize_sgd(h0, n, dx, tlimit, seed=0):
    rng = np.random.RandomState(seed)
    h = project_feasible(h0.copy(), n)
    bh = h.copy()
    bc5 = compute_c5(h, dx)
    t0 = time.time()
    lr = 0.06
    stall = 0
    it = 0
    while time.time() - t0 < tlimit:
        g = grad_active(h, n, dx)
        h2 = project_feasible(h - lr * g, n)
        c5 = compute_c5(h2, dx)
        if c5 < bc5 - 1e-14:
            bh = h2.copy()
            bc5 = c5
            h = h2
            stall = 0
            lr = min(lr * 1.003, 0.12)
        else:
            h = h2
            stall += 1
            if stall > 400:
                h = project_feasible(h + 0.06 * rng.randn(n), n)
                lr = max(0.7 * lr, 0.003)
                stall = 0
        it += 1
        if time.time() - t0 > tlimit * 0.99:
            break
    return bh, bc5
