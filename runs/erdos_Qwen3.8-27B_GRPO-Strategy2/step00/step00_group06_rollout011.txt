import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def softmax_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if alpha < 1:
        return m
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400, 400)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + log_sum / alpha


def repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-12:
        diff = target - s2
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_random_binary(n, rng):
    n_ones = n // 2
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n_ones]] = 1.0
    return h


def make_step(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h


def do_slsqp(h_init, n, dx, max_iter, alpha, ftol):
    target = n / 2.0

    def objective(x):
        return softmax_obj(x, dx, alpha)

    def eq_con(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_con}]

    best_h = h_init.copy()
    best_val = compute_c5(h_init, dx)

    try:
        res = minimize(objective, h_init, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if hasattr(res, 'fun') and np.isfinite(res.fun):
            h_opt = repair(res.x, n)
            c5 = compute_c5(h_opt, dx)
            if c5 < best_val:
                best_h, best_val = h_opt, c5
    except Exception:
        pass
    return best_h, best_val


def do_refine(h_init, n, dx, max_iter, alphas):
    """Progressive alpha refinement"""
    best_h, best_val = h_init.copy(), compute_c5(h_init, dx)
    for a in alphas:
        h_new, v_new = do_slsqp(best_h, n, dx, max_iter, a, 1e-14)
        if v_new < best_val:
            best_h, best_val = h_new, v_new
    return best_h, best_val
