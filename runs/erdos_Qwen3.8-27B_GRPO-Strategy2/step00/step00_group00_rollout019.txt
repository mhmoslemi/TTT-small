from scipy.optimize import minimize
import time

def compute_corr(h):
    g = 1.0 - h
    return np.correlate(h, g, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def smooth_logsumexp(values, temp):
    m = np.max(values)
    if temp < 1e-10:
        return m
    return m + temp * np.log(np.sum(np.exp((values - m) / temp)))

def get_gradient(h, temp):
    """Gradient of smooth max of correlations * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # softmax weights
    m = np.max(corr)
    exps = np.exp((corr - m) / temp)
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        d = j - (n - 1)
        # corr at lag d: sum_i h[i]*(1-h[i+d]) for valid i
        # d(corr[d])/dh[m_idx] = (1-h[m_idx+d]) - h[m_idx-d] (with bounds checks)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        idx = np.arange(i_lo, i_hi)
        # First part: (1 - h[idx + d])
        grad[idx] += w[j] * (1.0 - h[idx + d])
        # Second part: -h[idx - d] for valid idx-d
        valid = (idx - d >= 0) & (idx - d < n)
        grad[idx[valid]] += w[j] * (-h[idx[valid] - d])
    
    return grad * dx

    """Project onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    return np.clip(h - t, 0.0, 1.0)

    """Optimize using SLSQP."""
    dx = 2.0 / n
    
    def objective(x):
        return smooth_logsumexp(compute_corr(x), temp) * dx
    
    def jac(x):
        return get_gradient(x, temp)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(objective, h, method='SLSQP', jac=jac,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iters, 'ftol': 1e-12})
    if result.success or result.x is not None:
        x = proj_feasible(result.x, n)
        return x
    return h

def get_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    exps = np.exp((corr - m) / temp)
    w = exps / exps.sum()
    a = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for m_idx in range(n):
        L = min(n, 2 * n - 1 - m_idx)
        if L > 0:
            grad[m_idx] = dx * np.dot(w[m_idx:m_idx + L], a[:L])
    return grad

def proj_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def slsqp_optimize(h, n, max_iters=300, temp=0.01):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_val(compute_corr(x), temp) * dx
    def jac(x):
        return get_gradient(x, temp)
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iters, 'ftol': 1e-14})
        x = res.x
    except:
        x = h
    return proj_feasible(x, n)

def proj_grad_optimize(h, n, budget_s, temp=0.01, max_iters=10000):
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    lr = 0.05
    for it in range(max_iters):
        if time.time() - start > budget_s:
            break
        grad = get_gradient(h, temp)
        h_new = h - lr * grad
        h_new = proj_feasible(h_new, n)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new
        else:
            # Reduce learning rate (simple backtracking)
            lr *= 0.7
            h = h  # keep current
        if it % 50 == 0:
            lr = max(lr, 0.01)  # don't let lr get too small too fast
            # Periodically boost lr
            if it % 200 == 0:
                lr = max(lr, 0.02)
    return best_h

def interp_to_res(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

    t_start = time.time()

    h_init = proj_feasible(h_init, n0)
    
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    
    # Phase 1: SLSQP at original resolution
    rem = budget_s - (time.time() - t_start)
    if rem > 20:
        h_opt = slsqp_optimize(h_init, n0, max_iters=200, temp=0.005)
        c5 = compute_c5(h_opt)
        if c5 < c5_best:
            h_best, c5_best = h_opt, c5
    
    # Phase 2: Higher resolutions
    resolutions = []
    for mult in [2, 3, 4, 6, 8, 12, 16]:

            resolutions.append(r)
    resolutions = sorted(set(resolutions))
    
    for n in resolutions:
        rem = budget_s - (time.time() - t_start)
        if rem < 30:
            break
        
        h = interp_to_res(h_best, n)
        h = proj_feasible(h, n)
        
        if n <= 200:
            alloc = min(rem * 0.5, 120)
            h_opt = slsqp_optimize(h, n, max_iters=200, temp=0.01)

            h_opt = slsqp_optimize(h, n, max_iters=100, temp=0.02)
        else:

            h_opt = proj_grad_optimize(h, n, budget_s=alloc, temp=0.02)
