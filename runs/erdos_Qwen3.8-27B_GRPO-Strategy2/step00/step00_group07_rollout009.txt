A better continuous approach: since the objective is the max of 2n-1 bilinear functions, I can use a "max dropping" strategy:
1. Compute all correlations, find the max at lag k*
2. Compute the gradient of corr(k*) with respect to h
   corr(k*) = sum_{i: valid} h[i](1-h[i+k*])
   ∂corr(k*)/∂h[m] = (1-h[m+k*]) if m is in the valid range for the "first" term
                          - h[m-k*] if m is in the valid range for the "second" term (as the "i+k*" position)
   Actually, for full correlation:
   corr_full[k] = sum_i a[i+k]*v[i] where a=h, v=1-h
   For k≥0: corr[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
   ∂corr[k]/∂h[m] = (1-h[m-k]) if k≤m≤n-1 (from the h[i+k] term with i=m-k)
                   - h[m+k] if 0≤m≤n-1-k (from the (1-h[i]) term with i=m)
   So: ∂corr[k]/∂h[m] = (1-h[m-k])*[k≤m≤n-1] - h[m+k]*[0≤m≤n-1-k]

3. To reduce the max correlation, we want to decrease corr(k*). So we move h in the direction -∂corr(k*)/∂h, but we need to maintain sum(h) = n/2.
   Projected gradient: g = ∂corr(k*)/∂h - mean(∂corr(k*)/∂h) (to maintain sum)
   Then h ← h - lr * g, clip to [0,1], re-project to sum = n/2.

This is a subgradient method for the max of differentiable functions. It should work, though slowly.

        k_star = np.argmax(c)  # Lag index in [0, 2n-2]
        k = k_star - (n - 1)  # Convert to actual lag [-n+1, n-1]
        
        # Compute gradient of corr at lag k
        grad = np.zeros(n)
        if k >= 0:

            # d/dh[m]: (1-h[m-k]) for m in [k, n-1] minus h[m+k] for m in [0, n-1-k]
            for m in range(k, n):

            for m in range(0, n - k):
                grad[m] -= h[m + k]

            for m in range(0, n - kk):
                grad[m] -= h[m + kk]
        
        # Project gradient to maintain sum constraint
        grad = grad - np.mean(grad)
        
        # Update

    t0 = time.time()
    stall = 0
    while time.time() - t0 < time_limit:
        c = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(c))
        k = k_idx - (n - 1)
        g = corr_grad(h, k)
        if np.all(g == 0):
            g = corr_grad(h, -k) if k != 0 else np.zeros(n)
        if np.all(g == 0):
            break
        g -= np.mean(g)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        for _ in range(50):
            h_new += (target - h_new.sum()) / n
            h_new = np.clip(h_new, 0.0, 1.0)
            if abs(h_new.sum() - target) < 1e-12:
                break
        new_val = compute_c5(h_new)
        if new_val < best_val - 1e-9:
            best_val = new_val
            best_h = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.05, 0.3)
            stall = 0
        else:
            h = h_new.copy()
            stall += 1
            if stall > 50:
                lr *= 0.7
                stall = 0
            if lr < 0.0001:
                break
    # Local perturbation polish
    lr2 = 0.02
    for _ in range(500):
        if time.time() - t0 > time_limit:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        h[i] += lr2; h[j] -= lr2
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            v = compute_c5(h)
            if v < best_val:
                best_val = v; best_h = h.copy()
            else:
                h[i] -= lr2; h[j] += lr2
        else:
            h[i] -= lr2; h[j] += lr2
    return best_h, best_val

def upsample(h, n_new):
    n_old = len(h)
    idx = (np.arange(n_new) * n_old / n_new).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    return h[idx].copy()

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 80

    n1 = 80
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        if len(h_init) != n1:
            h_init = upsample(h_init, n1)
    else:
        h_init = np.zeros(n1); h_init[::2] = 1

    h_init = ensure_binary(h_init, n1)
    c5_init = compute_c5(h_init)
    best_h = h_init.copy(); best_c5 = c5_init; best_n = n1

    # Quick multi-start evaluation
    starts = generate_starts(n1, 25)
    for s in starts:
        v = compute_c5(s)
        if v < best_c5:
            best_c5 = v; best_h = s.copy()

    # SA phase at n=80
    elapsed = time.time() - t_start
    sa_budget = min(budget_s * 0.35, max(budget_s - elapsed - 300, 120))
    h_sa, c5_sa = sa_binary(best_h, n1, sa_budget, seed)
    if c5_sa < best_c5:
        best_c5 = c5_sa; best_h = h_sa.copy()

    # Continuous refinement at n=80
    elapsed = time.time() - t_start
    cont_budget = min(budget_s * 0.15, max(budget_s - elapsed - 200, 60))
    h_cr, c5_cr = continuous_refine(best_h, n1, cont_budget, seed + 10)
    if c5_cr < best_c5:
        best_c5 = c5_cr; best_h = h_cr.copy()

    # SA again from refined continuous (binarized)
    elapsed = time.time() - t_start
    if budget_s - elapsed > 80:
        h_b = ensure_binary(best_h, n1)
        sa2_budget = min(budget_s * 0.1, max(budget_s - elapsed - 100, 40))
        h_sa2, c5_sa2 = sa_binary(h_b, n1, sa2_budget, seed + 20)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2; best_h = h_sa2.copy()
        # Re-refine
        if budget_s - (time.time() - t_start) > 30:
            h_cr2, c5_cr2 = continuous_refine(best_h, n1, 30, seed + 30)
            if c5_cr2 < best_c5:
                best_c5 = c5_cr2; best_h = h_cr2.copy()

    # Try n=160
    elapsed = time.time() - t_start
    if budget_s - elapsed > 150:
        n2 = 160
        h_up2 = upsample(best_h, n2)
        h_up2 = ensure_binary(h_up2, n2)
        sa2_budget2 = min(budget_s * 0.2, max(budget_s - elapsed - 100, 80))
        h_sa_n2, c5_n2 = sa_binary(h_up2, n2, sa2_budget2, seed + 40)
        if c5_n2 < best_c5:
            best_c5 = c5_n2; best_h = h_sa_n2.copy(); best_n = n2
        if budget_s - (time.time() - t_start) > 40 and best_n == n2:
            h_cr_n2, c5_cr_n2 = continuous_refine(best_h, n2, 40, seed + 50)
            if c5_cr_n2 < best_c5:
                best_c5 = c5_cr_n2; best_h = h_cr_n2.copy()

    # Try n=320
    elapsed = time.time() - t_start
    if budget_s - elapsed > 200:
        n3 = 320
        h_up3 = upsample(best_h, n3)
        h_up3 = ensure_binary(h_up3, n3)
        sa3_budget = min(budget_s * 0.2, max(budget_s - elapsed - 50, 100))
        h_sa_n3, c5_n3 = sa_binary(h_up3, n3, sa3_budget, seed + 60)
        if c5_n3 < best_c5:
            best_c5 = c5_n3; best_h = h_sa_n3.copy(); best_n = n3
        if budget_s - (time.time() - t_start) > 30 and best_n == n3:
            h_cr_n3, c5_cr_n3 = continuous_refine(best_h, n3, 30, seed + 70)
            if c5_cr_n3 < best_c5:
                best_c5 = c5_cr_n3; best_h = h_cr_n3.copy()

    # Final: ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = len(best_h) / 2.0
    best_h += (target_sum - best_h.sum()) / len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
