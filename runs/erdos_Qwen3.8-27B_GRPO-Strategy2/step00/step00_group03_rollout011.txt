```python
def project_feasible(h, ts):

    diff = ts - h.sum()

    if diff > 0:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def make_rand_initial(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def slsqp_solve(h, n, ts, beta, maxiter=800):
    def obj(x):
        return smooth_obj(x, n, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def binary_local_search(h, n, ts, rng, max_sweeps=200):
    best_c5 = compute_c5(h)
    t0_local = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0_local > 120:
            break
        improved = False
        for j in range(n):
            for t_val in (0.0, 1.0):
                if abs(t_val - h[j]) < 1e-8:
                    continue
                h[j] = t_val
                h = project_feasible(h, ts)
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    improved = True
                else:
                    h[j] = t_val  # keep whichever is better
                    h = project_feasible(h, ts)
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_rand_initial(70, seed)

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Candidate n values
    n_list = sorted(set([n_init] + [c for c in [60, 70, 72, 80, 84, 90, 96, 100, 108, 120, 140, 160] if c % 2 == 0]))

    for n in n_list:
        if remaining() < 80:
            break
        ts = n / 2.0

        # Create starting point for this n
        if n == n_init:
            h = h_init.copy()
        else:
            xo = np.linspace(0, 2, n_init)
            xn = np.linspace(0, 2, n)
            h = np.interp(xn, xo, h_init)
            h = project_feasible(h, ts)

        # SLSQP with annealed beta
        for beta in [3.0, 8.0, 20.0, 50.0, 150.0, 400.0]:
            if remaining() < 30:
                break
            h = slsqp_solve(h, n, ts, beta)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Random restart SLSQP
        if remaining() > 60:
            for _ in range(3):
                h_r = h_init.copy() if n == n_init else project_feasible(h + rng.normal(0, 0.2, n), ts)
                h_r = slsqp_solve(h_r, n, ts, 30.0, maxiter=400)
                h_r = slsqp_solve(h_r, n, ts, 200.0, maxiter=400)
                c5 = compute_c5(h_r)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_r.copy()

        # Binary local search
        if remaining() > 40:
            h_ls = best_h.copy() if len(best_h) == n else h
            h_ls, c5_ls = binary_local_search(h_ls, n, ts, rng, max_sweeps=100)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

        # Final SLSQP polish on best for this n
        if remaining() > 25 and len(best_h) == n:
            h_pol = slsqp_solve(best_h.copy(), n, ts, 300.0, maxiter=600)
            h_pol = slsqp_solve(h_pol, n, ts, 800.0, maxiter=400)
            c5_p = compute_c5(h_pol)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
