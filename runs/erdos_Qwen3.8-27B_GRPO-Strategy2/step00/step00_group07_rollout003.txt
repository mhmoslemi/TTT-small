def project(h, n):
    """Project to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    steps = 30

            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12

        if cnt == 0:
            break

        per = np.clip(per, -1.0, 1.0)
        h[mask] += per
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

    diff = target - h.sum()

        else:

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)


def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    ec = np.exp(beta * (c - c.max()))
    w = ec / ec.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-9:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[s:]
        else:
            t = -s
            m = n - t
            grad[:m] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:m]
    return grad


def opt_continuous(h, iters, beta, t_deadline):
    n = len(h)
    h = project(h.copy(), n)
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.5 / n
    v = np.zeros(n)
    mu = 0.85
    for it in range(iters):
        if time.time() >= t_deadline:
            break
        g = smooth_grad(h, beta)
        v = mu * v + (1 - mu) * g
        h_new = project(h - lr * v, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            h = h_new
            best_c5 = c5
            best = h.copy()
            lr = min(lr * 1.03, 3.0 / n)
        else:
            lr *= 0.85
            v = mu * v
            if lr < 1e-13:
                lr = 0.5 / n
                v = np.zeros(n)
    return best, best_c5


def bin_search(h, rng, iters, t_deadline):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = project(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    T = 0.003
    Tmin = 0.00005
    decay = 0.99995
    for it in range(iters):
        if time.time() >= t_deadline:
            break
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        delta = c5 - best_c5
        if delta < -1e-14 or rng.random() < np.exp(-max(delta, 0.0) / max(T, 1e-12)):
            if c5 < best_c5:
                best = h.copy()
                best_c5 = c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(Tmin, T * decay)
    return best, best_c5


def opt_cont(h, iters, beta, t_deadline):
    return opt_continuous(h, iters, beta, t_deadline)
