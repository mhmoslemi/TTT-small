    n = len(h)
    new_n = n * factor
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    # Linear interpolation
    x_old = np.arange(n) * dx_old
    x_new = np.arange(new_n) * dx_new
    h_new = np.interp(x_new, x_old, h)

    current_sum = np.sum(h_new)

        h_new += (target_sum - current_sum) / new_n
        h_new = np.clip(h_new, 0, 1)
        # Re-fix if clipping changed sum
        diff = target_sum - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Distribute among entries not at bounds
            free = (h_new > 1e-8) & (h_new < 1 - 1e-8)
            if np.sum(free) > 0:
                h_new[free] += diff / np.sum(free)
                h_new = np.clip(h_new, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor) * dx)


def compute_grads(h):
    n = len(h)
    L = 2 * n - 1
    o = 1.0 - h
    grads = np.zeros((L, n))
    for k in range(L):
        j = k - (n - 1)
        g = np.zeros(n)
        if j >= 0:
            if n - j > 0:
                g[:n - j] = o[j:]
                g[j:] -= h[:n - j]
        else:
            jj = -j
            if n - jj > 0:
                g[jj:] = o[:n - jj]
                g[:n - jj] -= h[jj:]
        grads[k] = g
    return grads


def fix_constraints(h, n):
