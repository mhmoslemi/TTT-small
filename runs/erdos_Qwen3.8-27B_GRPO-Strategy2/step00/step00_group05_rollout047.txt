def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    current = np.sum(h)
    if current < 1e-10:
        h[:] = 0.5
        return h
    # Scale and clip, iterate
    for _ in range(100):
        scale = target / current
        h_new = np.clip(h * scale, 0.0, 1.0)
        new_sum = np.sum(h_new)
        if abs(new_sum - target) < 1e-10:
            h = h_new
            break
        h = h_new
        current = new_sum
    # Final adjustment
    diff = target - np.sum(h)

The feasible set is {h : 0 ≤ h_i ≤ 1, sum(h_i) = n/2}. This is a convex set (intersection of a hyperplane and a box). The projection onto this set can be computed as follows:

1. Find λ such that sum(clip(h_i + λ, 0, 1)) = n/2
2. Then the projection is clip(h_i + λ, 0, 1)

This is because the projection onto a hyperplane is h + λ * 1 (adding a constant), and then we clip to [0,1]. By the intermediate value theorem, there exists a unique λ that satisfies the constraint.

- λ ∈ [-max(h), 1 - min(h)] (roughly)

            h_r = project_h(h_r)
        h_r = run_optimize_slqp(h_r, 50, 80)
        h_r = run_optimize_slqp(h_r, 200, 80)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        attempt += 1

    # Phase 3: Try different n_points (20% budget)
    limit3 = t0 + budget_s * 0.75
    h_parent = h0 if h0 is not None else best_h
    for n_try in [50, 100, 140]:
        if time.time() > limit3:
            break
        x_old = np.linspace(0, 2, len(h_parent), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, np.concatenate([h_parent, [h_parent[0]]]))
        h_try = project_h(h_interp)
        h_try = run_optimize_slqp(h_try, 50, 80)
        h_try = run_optimize_slqp(h_try, 200, 80)
        h_try = run_optimize_slqp(h_try, 500, 80)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Final refinement (25% budget)
    limit4 = t0 + budget_s - 15
    h_cur = best_h.copy()
    for tau in [300, 1000, 3000, 5000]:
        if time.time() > limit4:
            break
        h_new = run_optimize_slqp(h_cur, tau, 150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new.copy()

    # Phase 5: Local coordinate search
    if time.time() < t0 + budget_s - 10:
        best_h, best_c5 = local_search(best_h, rng, max_rounds=3)

    # Ensure feasibility and finalize
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
