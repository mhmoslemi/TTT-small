import numpy as np
from scipy.optimize import minimize
import time

_DX = [0.0]
_N = [0]

def evaluate_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    target = n / 2.0
    k = int(target)
    frac = target - k
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:k]] = 1.0
    result[idx[k]] = frac
    return result

def _obj(x):
    return evaluate_c5(x, _DX[0])

def _con(x):
    return np.sum(x) - _N[0] / 2.0

def local_search(h, n, dx, time_limit, t_start):
    best_c5 = evaluate_c5(h, dx)
    h = h.copy()
    while time.time() - t_start < time_limit:
        any_improved = False
        ones = list(np.where(h >= 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            for j in zeros:
                oi, oj = h[i], h[j]
                if abs(oi - oj) < 1e-12:
                    continue
                h[i], h[j] = oj, oi
                c5 = evaluate_c5(h, dx)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    any_improved = True
                else:
                    h[i], h[j] = oi, oj
        if not any_improved:
            break
    return h, best_c5
