def compute_c5(h):

    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def make_binary_move(h, rng, target_sum):
    """Flip a 1-ish to 0-ish and a 0-ish to 1-ish."""
    h_new = h.copy()
    # Find candidates
    high = np.where(h > 0.5)[0]
    low = np.where(h < 0.5)[0]
    if len(high) == 0 or len(low) == 0:
        return None
    i = rng.choice(high)
    j = rng.choice(low)
    if i == j:
        return None
    h_new[i] = 0.0
    h_new[j] = 1.0
    return project_to_constraint(h_new, target_sum)

def make_continuous_move(h, rng, target_sum, scale):
    """Perturb two positions."""
    h_new = h.copy()
    i, j = rng.choice(len(h), size=2, replace=False)
    d = rng.uniform(-scale, scale)
    h_new[i] += d
    h_new[j] -= d
    return project_to_constraint(h_new, target_sum)

    h_new = h.copy()

    delta = rng.uniform(-scale, scale, size=length)
    delta -= delta.mean()
    idx = (start + np.arange(length)) % n
    h_new[idx] += delta
    return project_to_constraint(h_new, target_sum)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, s):
    h = np.clip(h, 0.0, 1.0)
    d = s - h.sum()
    if d > 1e-14:
        m = h < 1.0
        if m.sum() > 0:
            h[m] += d / m.sum()
    elif d < -1e-14:
        m = h > 0.0
        if m.sum() > 0:
            h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def make_move(h, rng, ts, n, temp):
    r = rng.random()
    if r < 0.40:
        hi = np.where(h > 0.3)[0]
        lo = np.where(h < 0.7)[0]
        if len(hi) == 0 or len(lo) == 0:
            return None
        i = hi[rng.integers(len(hi))]
        j = lo[rng.integers(len(lo))]
        if i == j:
            return None
        h_new = h.copy()
        if temp > 0.005:
            h_new[i] = 0.0
            h_new[j] = 1.0
        else:
            h_new[i] = max(0.0, h_new[i] - 0.5)
            h_new[j] = min(1.0, h_new[j] + 0.5)
        return project(h_new, ts)
    elif r < 0.72:
        i, j = rng.choice(n, size=2, replace=False)
        scale = 0.12 if temp > 0.008 else 0.04
        d = rng.uniform(-scale, scale)
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        return project(h_new, ts)
    elif r < 0.88:
        L = int(rng.integers(2, min(8, n // 4) + 1))
        s = int(rng.integers(0, n - L + 1))
        idx = np.arange(s, s + L)

        h_new[idx] += delta

        h_new = h.copy()
        h_new[idx1[0]] += d1
        h_new[idx1[1]] -= d1
        h_new[idx2[0]] += d2
        h_new[idx2[1]] -= d2

def anneal_from(h_start, ts, n, rng, duration, init_temp):
    t0 = time.time()
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    T = init_temp
    alpha = 0.99975
    stale = 0
    step = 0
    while time.time() - t0 < duration:
        step += 1
        h_new = make_move(h, rng, ts, n, T)
        if h_new is None:

        new_c = compute_c5(h_new)
        dc = new_c - cur_c
