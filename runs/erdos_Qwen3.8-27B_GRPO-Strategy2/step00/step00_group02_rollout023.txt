def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
    return h

def smooth_max_obj(h, beta, penalty_w):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    v_max = np.max(vals)
    shifted = np.clip(beta * (vals - v_max), -500, 500)
    f = v_max + np.log(np.sum(np.exp(shifted))) / beta
    s = np.sum(h)
    f += penalty_w * (s - n / 2.0) ** 2
    return f

def smooth_max_grad(h, beta, penalty_w):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx

    # Gradient of smooth max: weighted sum of gradients of each corr_k * dx
    # d(corr_k)/d(h[i]) needs careful computation
    # Use numerical gradient

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s

    return h

            break

        l = k_star - (n - 1)

            c = np.correlate(x, 1.0 - x, mode="full")[k_star]
            return c * dx + 50.0 * (np.sum(x) - n / 2.0) ** 2
        
        def grad(x):
            gc = np.zeros(n)
            for p in range(n):
                v = 0.0
                if 0 <= p + l < n:
                    v += 1.0 - x[p + l]
                if 0 <= p - l < n:
                    v -= x[p - l]
                gc[p] = v
            return dx * gc + 100.0 * (np.sum(x) - n / 2.0)
        
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',

    g += 2.0 * penalty_w * (np.sum(x) - n / 2.0)
    return g

def smooth_max_obj(x, beta, penalty_w, n, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    vm = np.max(vals)
    sh = np.clip(beta * (vals - vm), -500, 500)
    f = vm + np.log(np.sum(np.exp(sh))) / beta
    f += penalty_w * (np.sum(x) - n / 2.0) ** 2
    return f

def smooth_max_optimize(h0, n, beta, time_budget=60, t0=0):
    dx = 2.0 / n
    pw = 50.0
    res = minimize(
        lambda x: smooth_max_obj(x, beta, pw, n, dx),
        h0,
        jac=lambda x: smooth_max_grad(x, beta, pw, n, dx),
        method='L-BFGS-B',
        bounds=[(0, 1)] * n,
        options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 8000}
    )
    h = np.clip(res.x, 0, 1)
    h = project_to_sum(h, n)
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_global = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(45)
        h0[::2] = 1.0
        h0[0] = 0.5
        h0 = project_to_sum(h0, 45)

    n = len(h0)
    h0 = np.clip(h0, 0, 1)
    h0 = project_to_sum(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Active-set from initial
    h1, c5_1 = active_set_optimize(h0, n, outer_iters=30, time_budget=120, t0=t_global)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Smooth max with increasing beta
    if time.time() - t_global < budget_s * 0.5:
        for beta in [10, 30, 100, 300, 1000]:
            if time.time() - t_global > budget_s * 0.55:
                break
            tb = min(50, budget_s * 0.06)
            h2, c5_2 = smooth_max_optimize(best_h, n, beta=beta, time_budget=tb, t0=time.time())
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()
            # Quick active-set polish after each smooth step
            h2b, c5_2b = active_set_optimize(best_h, n, outer_iters=10,
                                             time_budget=20, t0=time.time())
            if c5_2b < best_c5:
                best_c5, best_h = c5_2b, h2b.copy()

    # Phase 3: Random restarts
    if time.time() - t_global < budget_s * 0.65:
        for _ in range(10):
            if time.time() - t_global > budget_s * 0.8:
                break
            h_rand = best_h + np.random.randn(n) * 0.15
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = project_to_sum(h_rand, n)
            hr, cr = active_set_optimize(h_rand, n, outer_iters=15,
                                         time_budget=25, t0=time.time())
            if cr < best_c5:
                best_c5, best_h = cr, hr.copy()

    # Phase 4: Binary-like restarts
    if time.time() - t_global < budget_s * 0.75:
        for _ in range(5):
            if time.time() - t_global > budget_s * 0.85:
                break
            h_bin = (np.random.rand(n) < 0.5) * 1.0
            h_bin = project_to_sum(h_bin, n)
            hb, cb = active_set_optimize(h_bin, n, outer_iters=15,
                                         time_budget=25, t0=time.time())
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()

    # Phase 5: Try different n by upsampling
    if time.time() - t_global < budget_s * 0.75:
        n_cur = len(best_h)
        for n_try in [90, 60, 135]:
            if time.time() - t_global > budget_s * 0.88:
                break
            idx = np.minimum((np.arange(n_try) * n_cur) // n_try, n_cur - 1)
            h_try = best_h[idx].copy()
            h_try = project_to_sum(h_try, n_try)
            ht, ct = active_set_optimize(h_try, n_try, outer_iters=20,
                                         time_budget=50, t0=time.time())
            if ct < best_c5:
                best_c5, best_h = ct, ht.copy()

    # Final polish
    budget_left = budget_s - (time.time() - t_global)
    if budget_left > 10:
        n_final = len(best_h)
        hf, cf = active_set_optimize(best_h, n_final, outer_iters=20,
                                     time_budget=min(40, budget_left * 0.5),
                                     t0=time.time())
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = project_to_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
