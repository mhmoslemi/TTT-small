import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def smooth_max(vals, alpha):
    m = np.max(vals)
    if m < -100:
        return m
    shifted = (vals - m) / alpha
    emax = np.max(shifted)
    s = np.sum(np.exp(shifted - emax))
    return m + alpha * (emax + np.log(s))

def obj_func(args):
    h, alpha = args
    return smooth_max(compute_corr(h), alpha)

def obj_wrapped(h, alpha):
    return smooth_max(compute_corr(h), alpha)

def sum_con_fun(h):
    return np.sum(h) - len(h) / 2.0

def project_feasible(h):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        deficit = target - np.sum(h)
        if abs(deficit) < 1e-14:
            break
        if deficit > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += deficit / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += deficit / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run_slsqp(h0, alphas, maxiter, ftol, time_limit=None, t_start=None):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun}]
    h = project_feasible(h0.copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        res = minimize(obj_wrapped, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = project_feasible(res.x.copy())
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def perturb_random(h0, rng, scale):
    noise = rng.normal(0.0, scale, size=len(h0))
    return project_feasible(h0 + noise)

def perturb_swap(h0, rng):
    h = h0.copy()
    n = len(h)
    for _ in range(int(rng.integers(2, 7))):
        i = int(rng.integers(0, n - 1))
        d = rng.uniform(-0.08, 0.08)
        h[i] += d
        h[i + 1] -= d
    return project_feasible(h)

def perturb_block(h0, rng):
    h = h0.copy()
    n = len(h)
    i = int(rng.integers(0, max(1, n // 4)))
    j = int(rng.integers(n // 2, n))
    d = rng.uniform(-0.05, 0.05)
    h[i:i + 3] += d
    h[j:j + 3] -= d
    return project_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):

    # Phase 1: careful optimization from start
    alphas1 = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    best_h, best_c5 = run_slsqp(h_init, alphas1, maxiter=500, ftol=1e-14,
                                  time_limit=budget_s * 0.25, t_start=t0)
    elapsed = time.time() - t0
    if elapsed > budget_s * 0.5:
        return best_h, compute_c5(best_h), n

    # Phase 2: perturbations
    time_limit = budget_s * 0.88
    scales = [0.04, 0.02, 0.01, 0.005, 0.002]
    attempt = 0
    stagnation = 0

    while time.time() - t0 < time_limit:
        attempt += 1
        r = rng.random()
        if r < 0.4:
            sc = scales[int(rng.integers(0, len(scales)))]
            h_p = perturb_random(best_h, rng, sc)
        elif r < 0.75:
            h_p = perturb_swap(best_h, rng)
        else:
            h_p = perturb_block(best_h, rng)

        alphas2 = [0.08, 0.04, 0.02, 0.01, 0.005]
        h_try, c5_try = run_slsqp(h_p, alphas2, maxiter=350, ftol=1e-12,
                                   time_limit=budget_s * 0.85, t_start=t0)
        if c5_try < best_c5 - 1e-8:
            best_c5 = c5_try
            best_h = h_try
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 20:
                best_h, best_c5 = run_slsqp(h_init, alphas1, maxiter=300, ftol=1e-12,
                                             time_limit=budget_s * 0.8, t_start=t0)
                stagnation = 0

        alphas3 = [0.005, 0.002, 0.001, 0.0005]

                                     time_limit=budget_s * 0.97, t_start=t0)

    final_h = project_feasible(best_h.copy())
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n
