    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 70
    
    # Use parent construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float).copy()
        n_p = len(init)
        dx_p = 2.0 / n_p
        init = project_feasible(init, n_p, n_p / 2.0)
        c5_p = compute_c5(init, dx_p)
        best_h = init.copy()
        best_c5 = c5_p
        best_n = n_p
    
    # Build list of (n, [starts]) to optimize
    tasks = []
    
    if best_h is not None:
        tasks.append((best_n, [best_h.copy()]))
    
    n_options = [50, 70, 100, 140, 200]
    for n in n_options:
        if n == best_n and best_h is not None:
            continue
        starts = []
        starts.append(project_feasible(make_random_binary(n, rng), n, n / 2.0))
        starts.append(project_feasible(make_smooth_start(n, rng, [2]), n, n / 2.0))
        starts.append(project_feasible(make_smooth_start(n, rng, [3]), n, n / 2.0))
        if time_left() > 400:
            starts.append(project_feasible(make_smooth_start(n, rng, [4, 1]), n, n / 2.0))
        tasks.append((n, starts))
    
    # Process tasks, prioritizing ones likely to improve
    # Sort by n (prefer smaller n for speed)
    tasks.sort(key=lambda t: t[0])
    
    for n, starts in tasks:
        if time_left() < 25:
            break
        dx = 2.0 / n
        target = n / 2.0
        
        for h0 in starts:
            if time_left() < 15:
                break
            h_opt, c5_opt = optimize_one(h0, n, dx, target, rng, time_left)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                # Extend this solution with more optimization
                if time_left() > 60:
                    h_ext, c5_ext = optimize_one(best_h, n, dx, target, rng, time_left)
                    if c5_ext < best_c5:
                        best_c5 = c5_ext
                        best_h = h_ext.copy()
    
    # Final high-precision refinement
    if best_h is not None and time_left() > 10:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for tau in [800, 1500, 3000]:
            if time_left() < 3:
                break
            try:
                res = minimize(
                    smooth_obj, best_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    args=(n, dx, tau, target, 500.0),
                    options={'maxiter': 500, 'maxfun': 5000, 'ftol': 1e-13, 'disp': False}
                )
                h_new = project_feasible(res.x, n, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Final validation
    if best_h is None:
        n = 70
        h = np.zeros(n)
        h[:n // 2] = 1.0
        best_h = project_feasible(h, n, n / 2.0)
        best_n = n
    
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
