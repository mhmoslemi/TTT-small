import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    if m < -1e6:
        return -1e6
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def obj_func(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max_val(corr, alpha)

def normalize_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final fix: adjust median element
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        mid = n // 2
        h[mid] = np.clip(h[mid] + (target - s2), 0.0, 1.0)
    return h

def slsqp_solve(h_init, n, maxiter, alpha):
    def obj(h):
        return obj_func(h, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bnds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        if res.fun < 1e5:
            return normalize_h(res.x, n)
    except Exception:
        pass
    return h_init

def coord_descent(h, n, iters, rng):
    h = h.copy()
    best_local = compute_c5(h, n)
    for _ in range(iters):
        c5_cur = compute_c5(h, n)
        i, j = rng.choice(n, 2, replace=False)

        h_new = h.copy()

        h_new[j] = np.clip(h[j] + (h[i] - h_new[i]), 0, 1)
        s = np.sum(h_new)
        if abs(s - n / 2.0) > 1e-6:
            fix = n / 2.0 - s
            idxs = np.where((h_new < 1.0) & (h_new > 0.0))[0]
            if len(idxs) > 0:
                h_new[idxs[0]] = np.clip(h_new[idxs[0]] + fix, 0, 1)
        c5_new = compute_c5(h_new, n)
        if c5_new < c5_cur - 1e-10:
            h = h_new
            best_local = c5_new
    return h
