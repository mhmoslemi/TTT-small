    dx = 2.0 / n
    c = correlate(h, 1-h, mode="full")
    k_best = np.argmax(c)
    d = k_best - (n - 1)  # the actual shift
    
    # Gradient of c_k_best w.r.t. h
    # c_k = sum_i h[i] - sum_i h[i]*h[i+d]
    # grad = -sum_i h[i]*h[i+d] gradient
    # ∂/∂h[j] of sum_i h[i]*h[i+d] = h[j+d] + h[j-d] (with 0 padding)
    
    grad = np.zeros(n)
    # h[j+d] for valid j: j+d in [0,n-1] → j in [-d, n-1-d] → j in [max(0,-d), min(n-1, n-1-d)]
    if d >= 0:
        # h[j+d]: j from 0 to n-1-d
        grad[:n-d] += h[d:]
        # h[j-d]: j from d to n-1
        grad[d:] += h[:n-d]

        # h[j+d] = h[j-d']: j from d' to n-1

        grad[:n-d'] += h[d':]

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return correlate(h, 1 - h, mode="full").max() * dx

def c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1 - h, mode="full")
    k = int(np.argmax(c))
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            grad[:n - d] += (1.0 - h[d:])
            grad[d:] -= h[:n - d]
    else:
        dd = -d
        if n - dd > 0:
            grad[dd:] += (1.0 - h[:n - dd])
            grad[:n - dd] -= h[dd:]
    return c[k] * dx, grad * dx

def fix_sum(h, n):
    h = np.clip(h.astype(float), 0, 1)
    for _ in range(200):
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
            if not m.any():
                break
            step = min(diff / m.sum(), 1.0 - h[m]).max()
            h[m] += step
        else:
            m = h > 1e-12
            if not m.any():
                break
            step = min(-diff / m.sum(), h[m]).max()
            h[m] -= step
        h = np.clip(h, 0, 1)
    return h

def slsqp_opt(h0, n, max_iter=300):
    def obj(h):
        return compute_c5(h)
    def obj_grad(h):
        _, g = c5_grad(h)
        return g
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = fix_sum(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h0, compute_c5(h0)

def pgd_opt(h0, n, max_iter=20000):
    dx = 2.0 / n
    h = h0.copy()
    best_h, best_c = h.copy(), compute_c5(h)
    lr = 1.0 / (2.0 * dx * 1.5)
    st = 0
    for it in range(max_iter):
        c, g = c5_grad(h)
        if c < best_c - 1e-15:
            best_c, best_h = c, h.copy()
            st = 0
        else:
            st += 1
            if st > 2000:
                lr *= 0.9
                st = 0
        h_new = np.clip(h - lr * g, 0, 1)
        h_new = fix_sum(h_new, n)
        h = h_new
    return best_h, best_c

def upsample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=n)

    return fix_sum(h * (n / max(h.sum(), 1e-10)), n)
