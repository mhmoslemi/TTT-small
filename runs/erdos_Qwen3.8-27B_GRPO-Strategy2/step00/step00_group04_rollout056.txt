import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(h, target):
    """Project h to satisfy 0<=h<=1 and sum(h)=target"""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    # Iterate if needed
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_init(n, seed=0):
    """Create a random binary vector with n/2 ones"""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    indices = rng.choice(n, size=n//2, replace=False)
    h[indices] = 1.0
    return h

def local_search_binary(h, max_iters=10000, seed=0):
    """Local search: try swapping a 1 with a 0"""
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        # Pick a random 1 and a random 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def smooth_optimize(h0, n, maxiter=500):
    """SLSQP optimization"""
    target = n / 2.0
    
    def objective(x):
        return compute_c5(x)
    
    def eq_fun(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_onto_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def interpolate_to_resolution(h, n_old, n_new):
    """Interpolate h from n_old to n_new resolution"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h, period=2.0)
    # clip and project
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return project_onto_feasible(h_interp, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Start with initial values
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    best_h = project_onto_feasible(h0, n / 2.0)

    if elapsed > budget_s * 0.3:

    # Phase 2: Local binary search at original resolution
    h_binary = (best_h > 0.5).astype(float)
    h_binary = project_onto_feasible(h_binary, n / 2.0)
