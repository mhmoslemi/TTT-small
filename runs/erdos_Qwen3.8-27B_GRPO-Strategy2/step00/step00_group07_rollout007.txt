For mode='full', k ranges from -(M-1) to N-1 where M=len(v), N=len(a).

result[k - (n-1)] = c_k for k from -(n-1) to (n-1)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    n_ones = n // 2
    # Argsort and threshold
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:n_ones]] = 1.0
    return b

def local_search(h, dx, t_end, best_c5=None, seed=0):
    """Simulated annealing on binary swaps."""
    n = len(h)
    h = make_binary(h, n).copy()
    current_c5 = evaluate_c5(h, dx)
    if best_c5 is None:
        best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    temp = 0.003
    cooling = 0.9998
    rng = np.random.default_rng(seed)
    
    while time.time() < t_end and temp > 1e-9:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h, dx)
        
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_c5 - current_c5) / max(temp, 1e-15)):
                current_c5 = new_c5
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        temp *= cooling
    
    return best_h, best_c5

def continuous_refine(h, dx, t_end, best_c5=None):
    """Coordinate descent for continuous h in [0,1]."""
    n = len(h)
    h = h.copy().astype(float)
    h = np.clip(h, 0, 1)
    target_sum = n / 2.0
    
    # Fix sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-6:
        # Distribute evenly, then clip and fix
        h += diff / n
        h = np.clip(h, 0, 1)
        # Iterative fix
        for _ in range(10):
            diff = target_sum - np.sum(h)
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                room = 1.0 - h
                mask = room > 1e-6
                if np.sum(room[mask]) > 0:
                    h[mask] += diff / np.sum(room[mask])
                    h = np.clip(h, 0, 1)
            else:
                room = h
                mask = room > 1e-6
                if np.sum(room[mask]) > 0:
                    h[mask] += diff / np.sum(room[mask])
                    h = np.clip(h, 0, 1)
    
    current_c5 = evaluate_c5(h, dx)

        best_c5 = current_c5
    best_h = h.copy()
    
    rng = np.random.default_rng(42)
    step = 0.05
    shrink = 0.9
    
    while time.time() < t_end:
        improved = False
        order = rng.permutation(n)
        for idx in order:
            for direction in [1, -1]:
                new_val = h[idx] + direction * step
                if new_val < 0 or new_val > 1:
                    continue
                old_val = h[idx]
                h[idx] = new_val
                # Compensate on another element
                diff = target_sum - np.sum(h)
                if abs(diff) > 1e-10:
                    if diff > 0:
                        comp = int(np.argmin(h))
                    else:
                        comp = int(np.argmax(h))
                    if comp == idx:
                        comp = (idx + 1) % n
                    comp_old = h[comp]
                    h[comp] = np.clip(comp_old - diff, 0, 1)
                    # If can't fully compensate, revert
                    if abs(target_sum - np.sum(h)) > 1e-6:
                        h[idx] = old_val
                        h[comp] = comp_old
                        continue
                
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < current_c5 - 1e-12:
                    current_c5 = new_c5
                    improved = True
                    if current_c5 < best_c5 - 1e-12:
                        best_c5 = current_c5
                        best_h = h.copy()
                else:
                    h[idx] = old_val
                    h[comp] = comp_old
        
        if not improved:
            step *= shrink
            if step < 1e-4:
                break
    
    return best_h, best_c5
