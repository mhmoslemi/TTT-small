```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_T = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    products = corr * dx
    max_p = np.max(products)
    exps = np.exp(T * (products - max_p))
    s = np.sum(exps)
    return max_p + (1.0 / T) * np.log(s)

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    products = corr * dx
    max_p = np.max(products)
    exps = np.exp(T * (products - max_p))
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        offset = m - (n - 1)
        j_lo = max(0, offset)
        j_hi = min(n, n + offset)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo - offset:j_hi - offset])
        j_lo2 = max(0, -offset)
        j_hi2 = min(n, n - offset)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 + offset:j_hi2 + offset]
    grad *= dx
    val = max_p + (1.0 / T) * np.log(s)
    return val, grad

def _obj(h):
    return smooth_obj(h, _g_T)

def _grad(h):
    return smooth_grad(h, _g_T)[1]

def _cons_fun(h):
    return np.sum(h) - _g_n / 2.0

def _cons_jac(h):
    return np.ones(_g_n)

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        nm = np.sum(mask)
        if nm == 0:
            h[:] = target / n
            break
        h[mask] += diff / nm
        h[mask] = np.clip(h[mask], 0.0, 1.0)
    return h

def make_rand_binary(n, rng):
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def upsample(h, n_from, n_to):
    x0 = np.arange(n_from) * 2.0 / n_from
    x1 = np.arange(n_to) * 2.0 / n_to
    return np.interp(x1, x0, h)

    global _g_n, _g_T
    t_start = time.time()

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Build initial starting points at n0
    starts = [h0.copy()]
    for _ in range(5):
        starts.append(fix_sum(make_rand_binary(n0, rng), n0))
    
    # Resolutions
    res_list = [n0]
    while res_list[-1] * 2 <= 400:
        res_list.append(res_list[-1] * 2)
    
    T_list = [30, 60, 120, 250]
    
    for n in res_list:
        if time.time() - t_start > budget_s - 60:
            break
        _g_n = n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        
        # Starting points at this resolution
        n_starts = []
        if n == n0:
            n_starts = starts[:]
        else:
            h_up = fix_sum(upsample(best_h, len(best_h), n), n)
            n_starts.append(h_up)
            for _ in range(3):

        remaining = budget_s - (time.time() - t_start) - 30
        if remaining <= 0:
```
