import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 = max of correlation values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_all(h):
    """Compute all correlation values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def objective(h):
    """Objective: C5 value"""
    return compute_c5(h)

def gradient(h):
    """Gradient of C5 with respect to h (at the active lag)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # max_idx ranges from 0 to 2n-2
    # shift s = max_idx - (n-1)
    s = max_idx - (n - 1)
    
    # overlap_s = sum_i h[i] * (1-h[i-s]) for valid i
    # grad[j] = dx * (1-h[j-s]) * [0<=j-s<n] - dx * h[j+s] * [0<=j+s<n]
    
    grad = np.zeros(n)
    # First term: dx * (1-h[j-s]) for j where 0<=j-s<n
    valid_i = np.arange(max(0, s), min(n, n+s))
    grad[valid_i] += dx * (1 - h[valid_i - s])
    
    # Second term: -dx * h[j+s] for j where 0<=j+s<n
    valid_j = np.arange(max(0, -s), min(n, n-s))
    grad[valid_j] -= dx * h[valid_j + s]
    
    return grad

def constraints_eq(h):
    """Equality constraint: sum(h) = n/2"""
    n = len(h)
    return np.sum(h) - n / 2.0

def project(h, n):
    """Project h onto constraints: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    s = np.sum(h)
    if s > 0:
        # Scale
        h = h * (target / s)
        h = np.clip(h, 0, 1)
    return h

def upsample(h, new_n):
    """Upsample h from len(h) to new_n points"""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    # Interpolate
    new_h = np.interp(x_new, x_old, h, period=2)
    # Project
    return project(new_h, new_n)

                  constraints=constraints, method='SLSQP',

    if res.success and res.fun < best_c5:

    # Multi-resolution: double and re-optimize
    while True:

        remaining = budget_s - elapsed - 60  # leave 60s buffer
        
        new_n = n * 2
        if remaining < 30:

        # Estimate time needed (rough)

        constraints_new = [{'type': 'eq', 'fun': constraints_eq, 'jac': lambda x: np.ones(len(x))}]

                      options={'maxiter': 200, 'ftol': 1e-12})

        new_c5 = compute_c5(res.x) if res.x is not None else compute_c5(h_new)
        h_final = project(res.x if res.x is not None else h_new, new_n)
        new_c5 = compute_c5(h_final)

            n = new_n

        t = -s
        g[t:] += dx * (1.0 - x[:n - t])
        g[:n - t] -= dx * x[t:]
    return g


def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-8:
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h


def _upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n, dtype=float) * 2.0 / old_n
    x_new = np.arange(new_n, dtype=float) * 2.0 / new_n
    new_h = np.interp(x_new, x_old, h, period=2.0)
    return _project(new_h, new_n)


def _obj_pen(x, pen):
    n = len(x)
    val = _compute_c5(x)
    cv = np.sum(x) - n / 2.0
    return val + 0.5 * pen * cv * cv


def _grad_pen(x, pen):
    g = _grad(x).copy()
    n = len(x)
    cv = np.sum(x) - n / 2.0
    g += pen * cv
    return g


def _optimize_slsqp(h0, maxiter=400):
    n = len(h0)
    h0 = _project(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}
    try:
        res = minimize(_obj, h0, jac=_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.isfinite(res.fun):
            return _project(res.x, n)
    except Exception:
        pass
    return h0


def _optimize_lbfgsb(h0, maxiter=600):
    n = len(h0)
    h0 = _project(h0, n)
    bounds = [(0.0, 1.0)] * n
    pen = 50.0
    try:
        res = minimize(_obj_pen, h0, args=(pen,), jac=_grad_pen,
                       bounds=bounds, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if np.isfinite(res.fun):
            return _project(res.x, n)
    except Exception:
        pass
    return h0


def _try_optimize(h0, maxiter, use_lbfgsb):
    if use_lbfgsb:
        return _optimize_lbfgsb(h0, maxiter)
    return _optimize_slsqp(h0, maxiter)


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(45) * 0.5

    n = len(h)
    best_h = h.copy()
    best_c5 = _compute_c5(h)

    # Phase 1: Optimize at initial resolution with SLSQP
    h_opt = _optimize_slsqp(h, maxiter=500)
    c5 = _compute_c5(h_opt)
    if c5 < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5

    # Try L-BFGS-B as well at initial resolution
    h_opt2 = _optimize_lbfgsb(h, maxiter=800)
    c5_2 = _compute_c5(h_opt2)
    if c5_2 < best_c5:
        best_h = h_opt2.copy()
        best_c5 = c5_2

    # Phase 2: Multi-resolution
    for _ in range(7):
        elapsed = time.time() - t0
        if elapsed > budget_s - 150:
            break
        new_n = len(best_h) * 2
        if new_n > 999:
            break

        h_new = _upsample(best_h, new_n)
        use_lb = new_n >= 180
        h_opt = _try_optimize(h_new, 400, use_lb)
        c5 = _compute_c5(h_opt)

        elapsed = time.time() - t0
        if elapsed > budget_s - 80:
            break

        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
        else:
            # Retry with perturbation
            h_pert = h_new + np.random.randn(new_n) * 0.03
            h_opt2 = _try_optimize(h_pert, 250, use_lb)
            c5_2 = _compute_c5(h_opt2)
            if c5_2 < best_c5:
                best_h = h_opt2.copy()
                best_c5 = c5_2

    # Phase 3: Local refinements at best resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 40:
        n_best = len(best_h)
        for _ in range(15):
            elapsed = time.time() - t0
            if elapsed > budget_s - 20:
                break
            h_pert = best_h + np.random.randn(n_best) * 0.02
            h_opt = _optimize_slsqp(h_pert, maxiter=200)
            c5 = _compute_c5(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5

    # Phase 4: One more SLSQP polish at final resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 15:
        h_polish = _optimize_slsqp(best_h, maxiter=300)
        c5_polish = _compute_c5(h_polish)
        if c5_polish < best_c5:
            best_h = h_polish.copy()
            best_c5 = c5_polish

    n_final = len(best_h)
    best_h = _project(best_h, n_final)
    best_c5 = _compute_c5(best_h)

    return (best_h, best_c5, n_final)
