```python
import numpy as np
import time

def c5_max(h):
    """Compute max overlap (without dx factor)."""
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def make_feasible(h, target_sum):
    """Ensure 0<=h<=1 and sum(h)=target_sum."""
    h = np.clip(h, 0, 1).copy()
    for _ in range(100):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1 - 1e-12
            if m.any():
                h[m] += d / m.sum()
            else:
                break
        else:
            m = h > 1e-12
            if m.any():
                h[m] += d / m.sum()
            else:
                break
        np.clip(h, 0, 1, out=h)
    return h

def to_binary(h, target_sum):
    """Convert to binary with exactly target_sum ones."""
    vals, idx = np.sort(h), np.argsort(h)
    # Keep top target_sum
    binary = np.zeros_like(h)
    top_idx = idx[-int(target_sum):]
    binary[top_idx] = 1
    return binary

def resample(h, n):
    xi = np.linspace(0, 2, len(h), endpoint=False)
    xo = np.linspace(0, 2, n, endpoint=False)
    return np.interp(xo, xi, h)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    best_overall = None
    best_c5_overall = np.inf
    
    for n in [200, 400]:
        if time.time() - t0 > budget_s - 60:
            break
        dx = 2.0 / n
        target_sum = n // 2
        
        # Initialize
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h0 = resample(np.array(initial_h_values, dtype=float), n)
            h0 = make_feasible(h0, float(target_sum))
        else:
            h0 = to_binary(np.random.RandomState(seed).random(n), target_sum)
        
        # Simulated annealing on binary
        h_bin = to_binary(h0, target_sum)
        c5 = c5_max(h_bin) * dx
        
        # SA parameters
        T = 0.01 * dx
        T_min = 0.001 * dx
        cooling = 0.9999
        n_ones = n // 2
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        while time.time() - t0 < budget_s - 60 and T > T_min:
            # Pick random a from zeros, b from ones
            a = zeros[rng.integers(len(zeros))]
            b = ones[rng.integers(len(ones))]
            
            # Apply flip
            old_c5_max = c5_max(h_bin)
            h_bin[a] = 1
            h_bin[b] = 0
            new_c5_max = c5_max(h_bin)
            delta = (new_c5_max - old_c5_max) * dx
            
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
                # Accept
                c5 = new_c5_max * dx
                # Update ones/zeros
                ones = np.setdiff1d(ones, [a])
                ones = np.append(ones, [b])
                zeros = np.setdiff1d(zeros, [b])
                zeros = np.append(zeros, [a])
```

```python
    best_max = np.inf
```
