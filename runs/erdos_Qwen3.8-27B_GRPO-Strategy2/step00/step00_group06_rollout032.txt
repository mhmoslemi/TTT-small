        if n_try < 20 or n_try > 500:
            continue
        h_proj = project_grid(best_h, n_try)
        _cn = n_try
        try:
            bounds_t = [(0.0, 1.0)] * n_try
            cons_t = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
            res = minimize(compute_c5, h_proj.copy(), method='SLSQP',
                          bounds=bounds_t, constraints=[cons_t],
                          options={'maxiter': 500, 'ftol': 1e-13})
            if res.x is not None:
                h_new = sanitize(res.x, n_try)
                v = compute_c5(h_new)
                if v < best_c5:
                    best_c5, best_h, best_n = v, h_new.copy(), n_try
        except Exception:
            pass

    # Phase 5: Perturbation restarts
    if remaining() > 60:
        for it in range(2000):
            if remaining() < 30:
                break
            scale = 0.002 if it < 500 else 0.005
            h_try = best_h + np.random.randn(best_n) * scale
            h_try = sanitize(h_try, best_n)
            v_try = compute_c5(h_try)
            if v_try < best_c5:
                best_c5, best_h = v_try, h_try.copy()

    # Phase 6: Coordinate sweep (try 0, 1 for each non-binary coord)
    if remaining() > 40:
        for i in range(best_n):
            if remaining() < 20:
                break
            if 0.05 < best_h[i] < 0.95:
                for val in (0.0, 1.0):
                    h_try = best_h.copy()
                    h_try[i] = val
                    h_try = sanitize(h_try, best_n)
                    v_try = compute_c5(h_try)
                    if v_try < best_c5 - 1e-13:
                        best_c5, best_h = v_try, h_try.copy()

    # Phase 7: Final SLSQP polish
    if remaining() > 25:
        _cn = best_n
        try:
            bounds_f = [(0.0, 1.0)] * best_n
            cons_f = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
            res = minimize(compute_c5, best_h.copy(), method='SLSQP',
                          bounds=bounds_f, constraints=[cons_f],
                          options={'maxiter': 500, 'ftol': 1e-15})
            if res.x is not None:
                h_new = sanitize(res.x, best_n)
                v = compute_c5(h_new)
                if v < best_c5:
                    best_c5, best_h = v, h_new.copy()
        except Exception:
            pass

    best_h = sanitize(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
