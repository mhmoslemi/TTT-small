def repair(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def c5_subgrad(h, n, top_k=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)

        s = int(j - (n - 1))
        if s >= 0:
            # Term 1: +(1-h[i-s]) for i in [s, n-1]
            grad[s:] += (1.0 - h[:n-s])
            # Term 2: -h[i+s] for i in [0, n-1-s]
            if n - s > 0:

        else:

            m = -s
            # Term 1: +(1-h[i-s]) = +(1-h[i+m]) for i in [0, n-m-1]
            grad[:n-m] += (1.0 - h[m:])
            # Term 2: -h[i+s] = -h[i-m] for i in [m, n-1]

            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400 and step < max_steps - 200:
            h = best + rng.randn(n) * 0.12
            h = repair(h, n)
            no_improve = 0
    return best, best_c5

def smooth_max_obj(h, n, temp=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    shifted = temp * (corr - corr.max())
    return float(np.log(np.exp(shifted).sum()) / temp + corr.max()) * dx

def smooth_max_grad(h, n, temp=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    weights = np.exp(temp * (corr - corr.max()))
    weights /= weights.sum()
    grad = np.zeros(n)
    for j, w in enumerate(weights):
        if w < 1e-8:
            continue
        s = int(j - (n - 1))
        if s >= 0:
            if n - s > 0:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
        else:
            m = -s
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad * dx

def scipy_local(h0, n, t0, budget, seed):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(smooth_max_obj, h0.copy(), args=(n, 30.0),
                       method='COBYLA', bounds=bounds,
                       options={'maxiter': 500, 'rhofac': 0.5, 'catol': 1e-10})
        if time.time() - t0 < budget * 0.9:
            h_new = repair(res.x, n)
        else:
            return h0, compute_c5(h0, n)
    except Exception:
        return h0, compute_c5(h0, n)
    c5_new = compute_c5(h_new, n)
    c5_old = compute_c5(h0, n)
    if c5_new < c5_old:
        return h_new, c5_new
    return h0, c5_old

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = len(init_h)
    except NameError:
        n = 70
        init_h = np.full(n, 0.5)

    best_h = repair(init_h.copy(), n)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Subgradient descent from initial
    best_h, best_c5 = subgrad_optimize(best_h, n, 5000, 0.12, t0, budget_s, rng)

    # Phase 2: Try structured starts
    t = np.arange(n) / float(n)
    starts = []
    for f in [1, 2, 3, 4]:
        for amp in [0.1, 0.2, 0.3]:
            hs = 0.5 + amp * np.sin(2.0 * np.pi * f * t + rng.randn())
            starts.append(repair(hs, n))
    starts.append(repair(rng.uniform(0.1, 0.9, n), n))
    starts.append(repair(np.concatenate([np.ones(n//2), np.zeros(n - n//2)]), n))

    for hs in starts:
        if time.time() - t0 > budget_s * 0.75:
            break
        h_new, c5_new = subgrad_optimize(hs, n, 3000, 0.1, t0, budget_s, rng)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Scipy local refinement
    if time.time() - t0 < budget_s * 0.80:
        h_new, c5_new = scipy_local(best_h, n, t0, budget_s, seed)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Prolonged subgradient with smaller lr
    if time.time() - t0 < budget_s * 0.85:
        best_h, best_c5 = subgrad_optimize(best_h, n, 15000, 0.04, t0, budget_s, rng)

    # Phase 5: Smooth-max gradient polish
    if time.time() - t0 < budget_s * 0.90:
        h = best_h.copy()
        for temp in [100.0, 50.0, 20.0]:
            if time.time() - t0 > budget_s * 0.92:
                break
            for step in range(2000):
                grad = smooth_max_grad(h, n, temp)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-14:
                    break
                lr = 0.05 * (0.99 ** (step // 500))
                h = h - lr * grad / gnorm
                h = repair(h, n)
            c5_val = compute_c5(h, n)
            if c5_val < best_c5:
                best_h = h.copy()
                best_c5 = c5_val

    # Phase 6: Final random perturb + polish
    if time.time() - t0 < budget_s * 0.94:
        for trial in range(5):
            hp = best_h + rng.randn(n) * 0.05
            hp = repair(hp, n)
            h_new, c5_new = subgrad_optimize(hp, n, 2000, 0.03, t0, budget_s, rng)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return (best_h, best_c5, n)
