import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def _c5_grad(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    k = int(np.argmax(corr))
    n = len(h)
    grad = np.zeros(n)
    # Term 1: j in [max(0, k-n+1), min(n-1, k)]
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        m_lo = lo1 - k + n - 1
        m_hi = hi1 - k + n - 1
        grad[lo1:hi1+1] += (1 - h[m_lo:m_hi+1])
    # Term 2: j in [max(0, n-k-1), min(n-1, n+k-2)]
    lo2 = max(0, n - k - 1)
    hi2 = min(n - 1, n + k - 2)
    if lo2 <= hi2:
        m_lo = lo2 + k - n + 1
        m_hi = hi2 + k - n + 1
        grad[lo2:hi2+1] += -h[m_lo:m_hi+1]
    return grad * dx

def _project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    # Simple projection: clip then adjust
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Add diff spread across non-saturated elements
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if np.sum(mask) == 0:
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def _upsample(h_old, n_new):
    """Upsample h from n_old to n_new points"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return _project_to_feasible(h_new, n_new)

def _slsqp_optimize(h0, n, dx, time_limit=60):
    """Optimize using SLSQP"""
    start = time.time()
    target_sum = n / 2.0
    
    def objective(h):
        return _compute_c5(h, dx)
    
    def gradient(h):
        return _c5_grad(h, dx)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                    'jac': lambda h: np.ones(n)}]

    h_init = _project_to_feasible(h0.copy(), n)
    
    result = minimize(objective, h_init, jac=gradient, method='SLSQP',

    h_opt = _project_to_feasible(result.x, n)
    c5 = _compute_c5(h_opt, dx)
    return h_opt, c5

import numpy as np
from scipy.optimize import minimize
import time

def _c5_value(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _c5_grad(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        mlo = lo1 - k + n - 1
        mhi = hi1 - k + n - 1
        g[lo1:hi1 + 1] += (1.0 - h[mlo:mhi + 1])
    lo2 = max(0, n - k - 1)
    hi2 = min(n - 1, n + k - 2)
    if lo2 <= hi2:
        mlo = lo2 + k - n + 1
        mhi = hi2 + k - n + 1
        g[lo2:hi2 + 1] += -h[mlo:mhi + 1]
    return g * dx

def _feasible(h, n):
    t = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        m = (h < 1.0 - 1e-12)
    else:
        m = (h > 1e-12)
    if m.sum() == 0:
        return h
    h[m] += d / m.sum()
    return np.clip(h, 0.0, 1.0)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _feasible(h_new, n_new)

def _optimize(h0, n, maxiter=400):
    target = n / 2.0
    bounds = tuple((0.0, 1.0) for _ in range(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    h0 = _feasible(h0.copy(), n)
    res = minimize(_c5_value, h0, jac=_c5_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    h_opt = _feasible(res.x, n)
    return h_opt, _c5_value(h_opt)

    h = rng.random(n)
    return _feasible(h, n)
