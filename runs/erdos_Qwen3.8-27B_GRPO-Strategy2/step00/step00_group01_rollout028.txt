import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(h, n_points, tau):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full") * dx
    scaled = tau * corr
    m = np.max(scaled)
    if m > 500:
        return m / tau
    return (m + np.log(np.sum(np.exp(scaled - m)))) / tau

def fix_sum(h, n_pts):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n_pts / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n_pts
        h = np.clip(h, 0.0, 1.0)
        remaining = target - np.sum(h)
        if abs(remaining) > 1e-12:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if np.any(free):
                h[free] += remaining / np.sum(free)
                h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        h[(h > 0.1) & (h < 0.9)] += diff / max(1, np.sum((h > 0.1) & (h < 0.9)))
        h = np.clip(h, 0.0, 1.0)
    return h

def do_slsqp(h0, n_pts, tau, maxiter=300, ftol=1e-10, time_limit=None):
    bnds = [(0.0, 1.0)] * n_pts
    def obj(h):
        return smooth_max_val(h, n_pts, tau)
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_pts / 2.0,
             'jac': lambda h: np.ones(n_pts)}]
    try:
        result = minimize(obj, h0.copy(), method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return fix_sum(result.x, n_pts)
    except Exception:
        return fix_sum(h0, n_pts)

def local_search(h0, n_pts, rng, n_iter=500, step=0.15):
    best = h0.copy()
    best_val = compute_c5(h0, n_pts)
    cur = h0.copy()
    cur_val = best_val
    T = 0.02
    for it in range(n_iter):
        T = max(0.001, 0.02 * (0.995 ** it))
        idx = rng.randint(0, n_pts)
        cur[idx] += rng.uniform(-step, step)
        cur = np.clip(cur, 0.0, 1.0)
        cur = fix_sum(cur, n_pts)
        new_val = compute_c5(cur, n_pts)
        if new_val < cur_val or rng.rand() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[idx] -= (cur[idx] - h0[idx]) * 0.0
            # revert
            cur = best.copy()
            cur_val = best_val
    return best, best_val
