import numpy as np

import time

def compute_c5(h):
    """Compute the C5 objective value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max(x, alpha):
    """Smooth approximation of max(x)."""
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def fix_sum(h, n):
    """Adjust h to satisfy sum constraint while maintaining bounds."""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Spread adjustment, respecting bounds
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # If clipping broke the sum, do iterative correction
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask_pos = h < 1.0 if diff > 0 else h > 0.0
        if not np.any(mask_pos):
            break
        h[mask_pos] += diff / np.sum(mask_pos)
    return h

def local_search(h, n, time_limit, rng):

    best_h = h.copy()

                h[i] -= delta
