from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    # Simple adjustment
    diff = target_sum - s
    n = len(h)
    # Adjust non-bound coordinates
    mask = (h > 1e-8) & (h < 1 - 1e-8)
    if np.any(mask):
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
        # Final correction
        s2 = np.sum(h)
        if abs(s2 - target_sum) > 1e-10:
            h[mask] += (target_sum - s2) / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def local_search(h0, n, max_iter, seed=42):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    best = h0.copy()
    best_val = compute_c5(best)
    current = best.copy()
    current_val = best_val
    temp = 0.01
    
    for it in range(max_iter):
        temp = max(0.001, 0.01 * (1 - it / max_iter))
        # Perturb 1-3 random coordinates
        n_perturb = rng.randint(1, 4)
        idxs = rng.choice(n, size=n_perturb, replace=False)
        deltas = rng.uniform(-0.1, 0.1, size=n_perturb)
        
        new_h = current.copy()
        for idx, d in zip(idxs, deltas):
            new_h[idx] = np.clip(new_h[idx] + d, 0, 1)
        new_h = repair(new_h, target)
        new_val = compute_c5(new_h)
        
        if new_val < current_val or rng.rand() < np.exp(-(new_val - current_val) / temp):
            current = new_h
            current_val = new_val
            if new_val < best_val:
                best = new_h.copy()
                best_val = new_val
    
    return best, best_val

def upscale(h, n_new):
    """Upscale h from n_old to n_new preserving integral"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)  # Wait, interp doesn't do periodic

    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = repair(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Start from initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 42
        h = np.ones(n) * 0.5  # Default: integral = n*0.5*(2/n) = 1
    
    best_h = h.copy()
    best_n = len(h)
    best_val = compute_c5(best_h)
    
    # Stage 1: Local search at original resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 60:
        max_iter = int(remaining * 100)  # Adjust based on n
        best_h, best_val = local_search(h, best_n, min(max_iter, 50000), seed)
    
    # Stage 2: Try higher resolutions
    for n_new in [84, 126, 168, 210]:
        if time.time() - start_time > budget_s - 60:
            break
        h_up = upscale(best_h, n_new)
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 30:
            break
        max_iter = int(remaining * 50)
        h_up, val_up = local_search(h_up, n_new, min(max_iter, 30000), seed + n_new)
        if val_up < best_val:
            best_h = h_up
            best_n = n_new
            best_val = val_up
    
    # Final verification
    final_val = compute_c5(best_h)
    return best_h, final_val, best_n

        if n_new <= best_n:
            continue
        hu = upsample(best_h, n_new)
        dl_c = time.time() + min(70, budget_s - (time.time() - t0) - 50)
        if dl_c > time.time() + 5:
            hi, vi = sa_continuous(hu, dl_c, seed + n_new, step=0.08)
            if vi < best_val:
                best_h, best_val, best_n = hi, vi, n_new
        dl_b = time.time() + min(55, budget_s - (time.time() - t0) - 30)
        if dl_b > time.time() + 5:
            hi, vi = sa_binary(best_h, dl_b, seed + n_new + 7)
            if vi < best_val:
                best_h, best_val, best_n = hi, vi, n_new
        if time.time() > t0 + budget_s - 25:
            break

    # Final: ensure validity
    best_n = len(best_h)
    target = best_n / 2.0
    best_h = project(best_h, target)
    final_val = compute_c5(best_h)
    return best_h, final_val, best_n
