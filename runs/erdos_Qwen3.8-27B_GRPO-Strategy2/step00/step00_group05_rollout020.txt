import numpy as np
import time

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def proj_simplex(x, s):
    lo, hi = x.min() - s - 1.0, x.max() + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.clip(x - m, 0.0, 1.0).sum() > s:
            lo = m
        else:
            hi = m
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    tc = tau * corr
    mx = tc.max()
    ex = np.exp(tc - mx)
    sw = ex / ex.sum()
    val = (mx + np.log(ex.sum())) / tau
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if sw[k] < 1e-10:
            continue
        shift = k - (n - 1)
        w = sw[k]
        if shift >= 0:
            m = n - shift
            if m > 0:
                grad[shift:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[shift:]
        else:
            s = -shift
            m = n - s
            if m > 0:
                grad[:m] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:m]
    grad *= dx
    return val, grad

def pgd_optimize(h_init, n, deadline, max_iter=30000, lr_base=0.02):
    s_target = n / 2.0
    h = proj_simplex(h_init.copy(), s_target)
    best_local = h.copy()
    best_val = c5_of(best_local)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max(max_iter - 1, 1)
        tau = 1.0 + 99.0 * frac * frac
        lr = lr_base / (1.0 + 0.05 * it)
        _, g = smooth_max_and_grad(h, tau)
        h_new = h - lr * g
        h_new = proj_simplex(h_new, s_target)
        c5_new = c5_of(h_new)
        if c5_new < best_val:
            best_val = c5_new
            best_local = h_new.copy()
        h = h_new
    return best_local, best_val

    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h
