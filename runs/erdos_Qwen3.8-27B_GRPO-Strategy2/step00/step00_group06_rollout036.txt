            cur_h = new_h
            lr *= 0.997
            stall += 1
        else:
            lr *= 0.96
            stall += 1
        if stall > 60:
            stall = 0
            restart_count += 1
            if restart_count < 20:
                cur_h = best_h + rng.normal(0, 0.03, n)
            else:
                cur_h = best_h + rng.normal(0, 0.01, n)
            cur_h = project_feasible(cur_h, n)
            if compute_c5(cur_h) > best_c + 0.005:
                cur_h = best_h.copy()
            lr = max(lr0 * 0.3, 0.005)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: main subgradient on original grid
    d1 = t0 + (deadline - t0) * 0.35
    h1, c1 = multi_start_subgrad(h, n, 0.12, d1, rng)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n

    # Phase 2: perturbation search on original grid
    d2 = t0 + (deadline - t0) * 0.55
    if time.time() < d2:
        h2, c2 = perturb_search(best_h, best_n, rng, d2)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: second subgradient pass (finer lr)
    d3 = t0 + (deadline - t0) * 0.72
    if time.time() < d3:
        h3, c3 = multi_start_subgrad(best_h, best_n, 0.03, d3, rng)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: finer grid (2x)
    d4 = deadline - 15
    if time.time() < d4 - 40:
        n_fine = 2 * best_n
        hf = interp_to(best_h, n_fine)
        c_f = compute_c5(hf)
        if c_f < best_c5 or True:
            hf2, cf2 = multi_start_subgrad(hf, n_fine, 0.06, d4 - 20, rng)
            if cf2 < best_c5:
                best_h, best_c5, best_n = hf2, cf2, n_fine
            if time.time() < d4 - 3:
                hf3, cf3 = perturb_search(best_h, best_n, rng, d4)
                if cf3 < best_c5:
                    best_h, best_c5 = hf3, cf3

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
