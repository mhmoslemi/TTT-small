    target = n / 2

def compute_grad(h, n, weights):
    """Compute gradient of smooth max correlation"""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        w = weights[k]
        if d >= 0:
            # h[0:n-d] * (1-h[d:n])
            grad[:n-d] += w * (1 - h[d:])
            grad[d:] += w * (-h[:n-d])

            dp = -d
            # h[dp:n] * (1-h[0:n-dp])
            grad[dp:] += w * (1 - h[:n-dp])
            grad[:n-dp] += w * (-h[dp:])
    return grad

import numpy as np
import math
import time

def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            idx = np.argsort(-h)
            for i in idx:
                rem = min(h[i], -diff)
                h[i] -= rem
                diff += rem
                if diff >= -1e-14:
                    break
    return h

def to_binary(h, n):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    target = n // 2
    num = h.sum()
    if num > target:
        ones = np.where(h == 1)[0]
        h[ones[:int(num - target)]] = 0
    elif num < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:int(target - num)]] = 1
    return h

def binary_local_search(h0, n, time_limit=30, allow_stays=True):

    best = compute_c5(h, n)
    t0 = time.time()

        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])

            for j in ones:
                h[i], h[j] = 1.0, 0.0

                if c < best - 1e-15:

                    improved = True
                    ones = list(np.where(h == 1)[0])
                    zeros = list(np.where(h == 0)[0])
                    break
                else:

            if improved:

def smooth_max_obj(h, n, eps=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    s = corr / eps
    s -= s.max()
    return eps * (s.max() + np.log(np.exp(s).sum()))

def smooth_max_grad(h, n, eps=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    s = corr / eps
    s -= s.max()
    w = np.exp(s) / np.exp(s).sum()
    grad = np.zeros(n)
