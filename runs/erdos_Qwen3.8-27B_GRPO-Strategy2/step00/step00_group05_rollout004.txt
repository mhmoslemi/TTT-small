import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_smooth(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    v_max = np.max(f)
    sm = v_max + np.log(np.sum(np.exp(T * (f - v_max)))) / T
    pen_term = pen * (np.sum(h) - n / 2.0) ** 2
    return sm + pen_term

def fix_sum(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    h = h + (target - s) / n
    h = np.clip(h, 0.0, 1.0)
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-12:
        diff = target - s2
        for i in range(n):
            trial = h[i] + diff
            if 0.0 <= trial <= 1.0:
                h[i] = trial
                break
    return h

def make_initial(n, rng, parent=None):
    if parent is not None and len(parent) == n:
        return np.array(parent, dtype=float).copy()
    h = np.zeros(n)
    n_blocks = max(3, n // 12)
    bs = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            s = b * bs
            e = min(n, s + bs)
            h[s:e] = 1.0
    h = h + rng.uniform(-0.3, 0.3, n)
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, n)
    return h

def optimize_lbfgsb(h, n, time_budget):
    bounds = [(0.0, 1.0)] * n
    T_schedule = [1.0, 3.0, 8.0, 20.0, 50.0, 100.0, 200.0, 400.0]
    pen = 20.0
    t_start = time.time()
    total_time = time_budget
    n_stages = len(T_schedule)
    per_stage = total_time / n_stages
    for T in T_schedule:
        elapsed_stage = time.time() - t_start
        if elapsed_stage + 2 > total_time:
            break
        maxit = max(100, int(per_stage * 50))
        res = minimize(
            obj_smooth, h, args=(T, pen),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-13, 'maxls': 80}
        )
        h = np.clip(res.x, 0.0, 1.0)
        h = fix_sum(h, n)
    return h

def optimize_slsqp(h, n, time_budget):
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(
        compute_c5, h, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': 80, 'ftol': 1e-10}
    )
    h_out = np.clip(res.x, 0.0, 1.0)
    h_out = fix_sum(h_out, n)
    return h_out
