import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def gen_initial(n, rng, idx):
    target = n / 2.0
    if idx == 0:
        h = np.zeros(n)
        h[::2] = 1.0
        if h.sum() > target:
            h[h > 0.5][:(int(h.sum() - target))] = 0.0
        h += rng.normal(0, 0.08, n)
    elif idx == 1:
        block = max(2, n // 15)
        h = np.zeros(n)
        for s in range(0, n, 2 * block):
            h[s:s + block] = 1.0
        h += rng.normal(0, 0.05, n)
    elif idx == 2:
        h = rng.uniform(0, 1, n)
    elif idx == 3:
        h = np.zeros(n)
        n_peaks = max(3, n // 8)
        peaks = rng.choice(n, n_peaks, replace=False)
        w = max(1, n // 40)
        for p in peaks:
            h[max(0, p - w):min(n, p + w + 1)] = 1.0
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def slsqp_optimize(h0, n, alpha=12.0, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        if m < 1e-15:
            return 0.0
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    x0 = project_feasible(h0, n)
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return x0

def coord_descent(h, n, n_passes=50):
    h = h.copy()
    best = evaluate_c5(h)
    rng_local = np.random.RandomState(42)
    for _ in range(n_passes):
        improved = False
        for _ in range(8):
            i = rng_local.randint(n)
            for d in [0.05, -0.05, 0.02, -0.02]:
                h2 = h.copy()
                h2[i] += d
                h2 = project_feasible(h2, n)
                c5 = evaluate_c5(h2)
                if c5 < best - 1e-12:
                    h, best, improved = h2, c5, True
                    break
        if not improved:
            break
    return h

def binary_search(h, n, rng, n_swaps=500):
    """Try to reduce C5 via binary pair swaps."""
    h = h.copy()
    best = evaluate_c5(h)
    ones = set(np.where(h >= 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        h[i], h[j] = 0.0, 1.0
        c5 = evaluate_c5(h)
        if c5 > best + 1e-12:
            h[i], h[j] = 1.0, 0.0  # revert
        else:
            best = c5
            ones.discard(i); ones.add(j)
            zeros.discard(j); zeros.add(i)
    return h, best

_g_n = 70
_g_dx = 2.0 / 70
_g_alpha = 12.0

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    for _ in range(3000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full") * _g_dx
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    return m + np.log(np.sum(np.exp(_g_alpha * (corr - m)))) / _g_alpha

def _eq_con(x):
    return x.sum() - _g_target
