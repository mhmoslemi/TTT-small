import numpy as np
from scipy.optimize import minpack2
from scipy import stats

# Helper functions
def evaluate_objective(h):
    """Compute C_5 = max_k integral of h(x)(1-h(x+k))dx using provided correlation"""
    corr = np.correlate(h, 1-h, mode="full")
    return (2.0 / len(h)) * np.max(corr)

def project_to_feasible(h):
    """Enforce constraint ∑hᵢ = n/2. Projects closest valid point"""
    s = np.sum(h)
    n = len(h)
    if np.abs(s - n/2) < 1e-5:  # Already feasible
        return h
    # Evenly distribute deficit/surplus
    diff = (n/2 - s) / n
    h_projected = np.clip(h + diff, 0, 1)
    return h_projected

def calculate_gradient(h):
    """Approximate gradient of objective function using finite differences"""
    n = len(h)
    dx = 2.0 / n
    obj = evaluate_objective(h)
    
    # Finite difference approximation
    grad_approx = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] = min(1.0, max(0.0, h[i] + 1e-4))
        h_minus[i] = max(0.0, min(1.0, h[i] - 1e-4))
        
        obj_plus = evaluate_objective(h_plus)
        obj_minus = evaluate_objective(h_minus)
        grad_approx[i] = (obj_plus - obj_minus) * (1e-4) / dx
        
    return grad_approx

def optimize_local(h0, max_iter=50):

    Local optimization using SLSQP with projected gradient method
    Returns (h_opt, success)
    """

    dx = 2.0 / n

        return np.sum(h) - n/2
    
    def constraint_jac(h):
        return [1.0] * n
    
    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    try:
        options = {'maxiter': max_iter}
        # Define objective with constraints
        res = minpack2.minimize(evaluate_objective, h0, method='SLSQP',
                               bounds=bounds, constraints={'type': 'eq', 'fun': constraint_fun,
                                                         'jac': constraint_jac},
                               options=options)
        
        if res.success:
            return res.x, True
        return h0, False
    except:
        return project_to_feasible(h0), False
