    return float(np.max(correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def c5_grad_active(h):
    n = len(h)
    dx = 2.0 / n
    c_raw = correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(c_raw))
    lag = m - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        L = lag
        for j in range(n):
            if j >= L:
                g[j] += 1.0 - h[j - L]
            if j + L < n:
                g[j] -= h[j + L]
    else:
        L = -lag
        for j in range(n):
            if j + L < n:
                g[j] += 1.0 - h[j + L]
            if j >= L:
                g[j] -= h[j - L]
