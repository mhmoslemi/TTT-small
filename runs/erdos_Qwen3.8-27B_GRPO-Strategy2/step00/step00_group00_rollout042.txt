import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy().astype(float)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, n_new / 2.0)

def slsqp_solve(h0, max_iter=200, beta=30.0, ftol=1e-10):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr * dx
        mx = np.max(vals)
        shifted = vals - mx
        return mx + np.log(np.sum(np.exp(beta * shifted))) / beta

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_out = np.clip(res.x, 0.0, 1.0)
    h_out = fix_sum(h_out, target)
    return h_out, compute_c5(h_out)

def binary_local_search(h0, max_iters=2000):

        ones -= 1
    while ones < target:
        idx = np.where(h == 0)[0]
        h[idx[np.argmin(h[idx])] if len(idx)>0 else 0] = 1

    best_c5 = compute_c5(h)

        else:
