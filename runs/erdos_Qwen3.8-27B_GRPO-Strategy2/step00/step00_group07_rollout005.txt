            j2 = int(rng.integers(n))
            if h[j2] != 0:
                return None
        h[i1], h[j1] = h[j1], h[i1]
        h[i2], h[j2] = h[j2], h[i2]
        return ('double', i1, j1, i2, j2)

def revert_move(h, rev):
    if rev is None:
        return
    if rev[0] == 'swap':
        h[rev[1]], h[rev[2]] = h[rev[2]], h[rev[1]]
    elif rev[0] == 'block':
        h[rev[1]:rev[1]+rev[2]] = rev[3]
    elif rev[0] == 'double':
        h[rev[1]], h[rev[2]] = h[rev[2]], h[rev[1]]
        h[rev[3]], h[rev[4]] = h[rev[4]], h[rev[3]]

def sa_optimize(h, n, T_init, T_min, n_iters, rng, best_ref, t_limit):
    c5 = compute_c5(h)
    count = 0
    for it in range(n_iters):
        if it % 10000 == 0 and time.time() > t_limit:
            break
        T = T_init * (T_min / T_init) ** (it / max(n_iters - 1, 1))
        rev = make_move(h, n, rng)
        if rev is None:
            continue
        count += 1
        c5_new = compute_c5(h)
        d = c5_new - c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-9)):
            c5 = c5_new
            if c5 < best_ref[0]:
                best_ref[0] = c5
                best_ref[1] = h.copy()
        else:
            revert_move(h, rev)
    return count

def greedy_polish(h, n, rng, best_ref, t_limit, max_rounds=50):
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _try in range(200):
            if time.time() > t_limit:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if c5 < best_ref[0]:
                    best_ref[0] = c5
                    best_ref[1] = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h0 = (h0 > 0.5).astype(float)
    n_ones = int(h0.sum())
    if n_ones != n0 // 2:
        if n_ones > n0 // 2:
            oi = np.where(h0 == 1)[0]
            h0[rng.choice(oi, size=n_ones - n0//2, replace=False)] = 0
        else:
            zi = np.where(h0 == 0)[0]
            h0[rng.choice(zi, size=n0//2 - n_ones, replace=False)] = 1

    best_ref = [compute_c5(h0), h0.copy()]
    target = 0.3808

    # Phase 1: SA at n0
    t_lim1 = t0 + budget_s * 0.35
    for restart in range(8):
        if time.time() > t_lim1:
            break
        h_curr = best_ref[1].copy()
        # perturb
        for _ in range(4 + restart * 2):
            rev = make_move(h_curr, n0, rng)
        T_i = 0.006 * (0.85 ** restart)
        T_m = 0.0008 * (0.85 ** restart)
        sa_optimize(h_curr, n0, T_i, T_m, 500000, rng, best_ref, t_lim1)

    # Phase 2: higher resolutions
    t_lim2 = t0 + budget_s * 0.75
    for n_large in [160, 320, 640]:
        if time.time() > t_lim2 or best_ref[0] <= target:
            break
        h_large = resample_to_n(best_ref[1], n_large, rng)
        c5_l = compute_c5(h_large)
        if c5_l < best_ref[0]:
            best_ref[0] = c5_l
            best_ref[1] = h_large.copy()
        h_l = best_ref[1].copy()
        T_i = 0.004 * (160.0 / n_large)
        T_m = 0.0005 * (160.0 / n_large)
        n_it = int(300000 * (160.0 / n_large))
        sa_optimize(h_l, n_large, T_i, T_m, max(n_it, 50000), rng, best_ref, t_lim2)
        greedy_polish(best_ref[1], n_large, rng, best_ref, t_lim2)

    # Phase 3: polish at best resolution
    t_lim3 = t0 + budget_s * 0.95
    if time.time() < t_lim3 and best_ref[0] > target:
        n_b = len(best_ref[1])
        if n_b <= 320:
            h_p = best_ref[1].copy()
            sa_optimize(h_p, n_b, 0.0015, 0.0003, 200000, rng, best_ref, t_lim3)
        greedy_polish(best_ref[1], n_b, rng, best_ref, t_lim3)

    # Final check
    best_h = best_ref[1]
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
