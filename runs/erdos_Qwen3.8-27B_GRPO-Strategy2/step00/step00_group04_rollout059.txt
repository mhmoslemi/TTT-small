```python
    return np.max(corr) * dx

def corr_values(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def obj_and_grad(h, n):
    dx = 2.0 / n
    h = np.asarray(h)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr_raw))
    m = k_max - n + 1  # lag

        if 0 <= i - m < n:
            grad[i] += (1.0 - h[i - m])
        if 0 <= i + m < n:
            grad[i] -= h[i + m]
    grad *= dx
    return val, grad
```

```python
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr_raw)
    # Find lags near the max

    # d(c_m)/dh[i] = (1-h[i-m])*[valid] - h[i+m]*[valid]
    # To reduce c_m, increase h[i] where gradient is positive, decrease where negative
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_n_global = 42

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_grad(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr_raw))
    m = k_max - (n - 1)
    val = float(corr_raw[k_max] * dx)
    grad = np.zeros(n)
    if m >= 0:
        r = n - m
        if r > 0:
            grad[m:] += (1.0 - h[:r])
            grad[:r] -= h[m:]
    else:
        p = -m
        r = n - p
        if r > 0:
            grad[:r] += (1.0 - h[p:])
            grad[p:] -= h[:r]
    grad *= dx
    return val, grad

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h = h + (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    return obj_grad(x, _n_global)[0]

def _jac(x):
    return obj_grad(x, _n_global)[1]

def _con_fun(x):
    return np.sum(x) - _n_global * 0.5

def _con_jac(x):
    return np.ones(_n_global, dtype=float) / _n_global

def do_slsqp(h0, n, maxiter, ftol):
    global _n_global
    _n_global = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_jac,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair(res.x, n)
    except Exception:
        pass
    return h0

def do_trust(h0, n, maxiter):
    global _n_global
    _n_global = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0, method='trust-constr', bounds=bounds,
                       constraints=cons, jac=_jac,
                       options={'maxiter': maxiter, 'gtol': 1e-6})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair(res.x, n)
    except Exception:
        pass
    return h0

def binarize(h, n):
    target = int(n * 0.5)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = T * (corr - np.max(corr))
    exps = np.exp(shifted)
    val = np.max(corr) + np.log(np.sum(exps)) / T
    return val

def softmax_grad(h, n, T):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    shifted = T * (corr - np.max(corr))
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    max_val = np.max(corr) + np.log(np.sum(exps)) / T
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for k in range(n_full):
        if weights[k] < 1e-10:
            continue
        m = k - (n - 1)
        g_k = np.zeros(n)
        if m >= 0:
            r = n - m
            if r > 0:
                g_k[m:] += (1.0 - h[:r])
                g_k[:r] -= h[m:]
        else:
            p = -m
            r = n - p
            if r > 0:
                g_k[:r] += (1.0 - h[p:])
                g_k[p:] -= h[:r]
        grad += weights[k] * g_k
    grad *= dx
    return max_val, grad

def run(seed=42, budget_s=1100, **kwargs):
    global _n_global
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(42) * 0.5
    n = len(best_h)
    _n_global = n
    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP from initial
    h_opt = do_slsqp(best_h, n, 2000, 1e-15)
    c5_opt = compute_c5(h_opt, n)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Try binarize + re-optimize
    h_bin = binarize(best_h, n)
    c5_bin = compute_c5(h_bin, n)
    h_bin_opt = do_slsqp(h_bin, n, 1000, 1e-14)
    c5_bin_opt = compute_c5(h_bin_opt, n)
    if c5_bin_opt < best_c5:
        best_c5 = c5_bin_opt
        best_h = h_bin_opt.copy()

    # Phase 2: Many random restarts
    scales = [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]
    while time.time() - t0 < budget_s - 80:
        scale = float(np.random.choice(scales))
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = repair(h_pert, n)
        h_opt = do_slsqp(h_pert, n, 400, 1e-13)
        c5_opt = compute_c5(h_opt, n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Occasional trust-constr
        if int((time.time() - t0) * 10) % 7 == 0 and time.time() - t0 < budget_s - 100:
            h_tc = do_trust(h_pert, n, 200)
            c5_tc = compute_c5(h_tc, n)
            if c5_tc < best_c5:
                best_c5 = c5_tc
                best_h = h_tc.copy()

    # Phase 3: refine with soft-max (smooth) then SLSQP
    if time.time() - t0 < budget_s - 30:
        for T in [20.0, 50.0, 100.0]:
            if time.time() - t0 > budget_s - 20:
                break
            # A few gradient steps with soft-max
            h_sm = best_h.copy()
            for step in range(100):
                if time.time() - t0 > budget_s - 20:
                    break
                val, g = softmax_grad(h_sm, n, T)
                h_sm = h_sm - 0.01 * g
                h_sm = repair(h_sm, n)
            h_sm_opt = do_slsqp(h_sm, n, 500, 1e-14)
            c5_sm = compute_c5(h_sm_opt, n)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm_opt.copy()

    # Phase 4: Higher resolution
    if time.time() - t0 < budget_s - 40:
        for n_new in [int(n * 1.5), n * 2, int(n * 2.5)]:
            if time.time() - t0 > budget_s - 20:
                break
            if n_new < n or n_new > 252:
                continue
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = repair(h_interp, n_new)
            h_hi = do_slsqp(h_interp, n_new, 800, 1e-14)
            c5_hi = compute_c5(h_hi, n_new)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                n = n_new

    # Final repair and validation
    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
```
