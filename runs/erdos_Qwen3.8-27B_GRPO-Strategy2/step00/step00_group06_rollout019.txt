import numpy as np
from scipy.optimize import minimize
import time


def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def c5_eval(h, n):
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def smooth_max_obj(h, n, beta):
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    m = v.max()
    if beta < 1.0:
        return m
    return m + np.log(np.sum(np.exp(beta * (v - m)))) / beta


def slsqp_optimize(h0, n, beta, maxiter, ftol):
    h0 = repair(h0, n)
    bounds = [(0.0, 1.0)] * n

    def cons_eq(x):
        return x.sum() - n / 2.0

    def obj(x):
        return smooth_max_obj(x, n, beta)

    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair(res.x, n)
    except Exception:
        pass
    return h0


def local_perturb(h, n, rng, n_iters, step_size):
    best_c = c5_eval(h, n)
    best = h.copy()
    for _ in range(n_iters):
        cand = best.copy()
        k = rng.randint(1, 6)
        idx = rng.choice(n, size=k, replace=False)
        cand[idx] += rng.uniform(-step_size, step_size, size=k)
        cand = repair(cand, n)
        c = c5_eval(cand, n)
        if c < best_c - 1e-12:
            best_c = c
            best = cand
    return best, best_c


def proj_grad_step(h, n, lr, rng):
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    m = v.max()
    active = np.where(v > m - 0.05)[0]
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)
        if k >= 0:
            valid = slice(0, n - k)
            contrib = np.zeros(n)
            contrib[valid] = (1.0 - h[k:][valid])
            shift_h = np.zeros(n)
            shift_h[k:] = h[:n-k]
            contrib2 = -shift_h
        else:
            k2 = -k
            valid = slice(k2, n)
            contrib = -h[:n-k2]
            shift_h = np.zeros(n)
            shift_h[:n-k2] = h[k2:]
            contrib2 = np.zeros(n)
            contrib2[k2:] = (1.0 - h[:n-k2])
        grad += dx * (contrib + contrib2)
    h_new = h - lr * grad
    return repair(h_new, n)
