import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_and_grad(h, T, dx):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')  # length 2n-1
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    logsum = np.log(np.sum(exps))
    val = T * logsum * dx + cmax * dx  # = dx * (cmax + T*logsum)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = weights[k]
        if wk < 1e-12:
            continue
        for i in range(n):
            g = 0.0
            if 0 <= i + s < n:
                g += 1.0 - h[i + s]
            if 0 <= i - s < n:
                g -= h[i - s]
            grad[i] += wk * g
    grad *= dx
    return val, grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        step = min(diff / cnt, 1.0 - h[mask].min())
        h[mask] += step
        # If still not at target, distribute remaining
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            mask2 = h < 1.0
            if np.sum(mask2) > 0:
                h[mask2] += diff2 / np.sum(mask2)
                h = np.clip(h, 0, 1)
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        step = max(diff / cnt, -h[mask].min())
        h[mask] += step
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            mask2 = h > 0.0
            if np.sum(mask2) > 0:
                h[mask2] += diff2 / np.sum(mask2)
                h = np.clip(h, 0, 1)
    return h

def smooth_obj_grad(h, T, dx):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')  # length 2n-1
    cmax = np.max(corr)
    shifted = np.clip((corr - cmax) / T, -50, 50)
    exps = np.exp(shifted)
    val = dx * (cmax + T * np.log(np.sum(exps)))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = weights[t]
        if wt < 1e-10:
            continue
        s = t - (n - 1)

            g = 0.0
            if 0 <= i - s < n:
                g += 1.0 - h[i - s]
            if 0 <= i + s < n:
                g -= h[i + s]
            grad[i] += wt * g
    grad *= dx
    return val, grad

    s = np.sum(exps)
    val = dx * (cmax + T * np.log(s))
    weights = exps / s
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = weights[t]
        if wt < 1e-12:
            continue
        sh = t - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i - sh < n:
                g += 1.0 - h[i - sh]
            if 0 <= i + sh < n:
                g -= h[i + sh]
            grad[i] += wt * g
    grad *= dx
    return val, grad

def project_sum(h, target):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _f(x):
    x = np.clip(x, 0.0, 1.0)
    x = project_sum(x, _opt_p['target'])
    return smooth_obj_grad(x, _opt_p['T'], _opt_p['dx'])[0]

def _g(x):
    x = np.clip(x, 0.0, 1.0)
    x = project_sum(x, _opt_p['target'])
    return smooth_obj_grad(x, _opt_p['T'], _opt_p['dx'])[1]

def run_lbfgs(h, T, dx, target, maxiter, gtol):
    _opt_p['T'] = T
    _opt_p['dx'] = dx
    _opt_p['target'] = target
    n = len(h)
    try:
        res = minimize(_f, h, method='L-BFGS-B', jac=_g,
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'gtol': gtol})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()
