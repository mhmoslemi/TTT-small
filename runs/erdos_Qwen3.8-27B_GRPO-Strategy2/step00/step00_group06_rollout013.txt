import time

def project_to_constraint(h, target_sum):

        if s > target_sum:

        else:
            hi = mid

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def smooth_max_gradient(h, n, T=100.0):
    """Gradient of (1/T)*log(sum(exp(T*f_k)))"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    max_c = corr.max()
    weights = np.exp(T * (corr - max_c))
    weights /= weights.sum()
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        p_lo = max(0, k - n + 1)
        p_hi = min(n - 1, k)
        if p_lo > p_hi:
            continue
        idx = k - np.arange(p_lo, p_hi + 1)
        grad[p_lo:p_hi+1] += weights[k] * (1.0 - 2.0 * h[idx])
    
    return grad

def smooth_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = corr.max()
    w = np.exp(T * (corr - c_max))
    w /= w.sum()
    k_idx = np.arange(2 * n - 1)[:, None]
    p_idx = np.arange(n)[None, :]
    j_idx = k_idx - p_idx
    valid = (j_idx >= 0) & (j_idx < n)
    j_safe = np.clip(j_idx, 0, n - 1)
    vals = np.where(valid, 1.0 - 2.0 * h[j_safe], 0.0)
    return vals.T @ w


def optimize(h_init, n, time_limit, T_start=50.0, T_end=500.0, step0=0.1):
    target = n * 0.5
    h = project_to_constraint(h_init.astype(np.float64).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    step = step0
    momentum = np.zeros(n)
    beta = 0.5
    stall = 0

    for it in range(20000):
        if time.time() - t0 > time_limit:
            break
        frac = min(it / 5000.0, 1.0)
        T = T_start * (T_end / T_start) ** frac
        grad = smooth_grad(h, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - step * momentum
        h_new = project_to_constraint(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        step *= 0.9995
        if stall > 800:
            step *= 0.5
            stall = 0
        if step < 0.002:
            step = 0.002
    return best_h, best_c5


def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_to_constraint(h.copy(), n_new / 2.0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_constraint(h_new, n_new / 2.0)


def make_random_start(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    return project_to_constraint(h, n / 2.0)


def make_sine_start(n, seed):
    rng = np.random.RandomState(seed)
    freq = rng.uniform(1, 6)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_to_constraint(np.clip(h, 0, 1), n / 2.0)


def make_block_start(n, seed):
    rng = np.random.RandomState(seed)
    n_blocks = rng.randint(3, 9)
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        s = i * block_size
        e = s + block_size if i < n_blocks - 1 else n
        h[s:e] = rng.uniform(0.2, 0.8)
    return project_to_constraint(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    h_parent = None
    if 'initial_h_values' in globals():
        try:
            h_parent = np.array(initial_h_values, dtype=np.float64)
        except Exception:
            pass

    if h_parent is not None and len(h_parent) > 0:
        n_parent = len(h_parent)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_to_constraint(h_parent, n_parent / 2.0)
    else:
        n_parent = 95
        h_parent = np.full(n_parent, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    best_n = n_parent

    # Phase 1: optimize on parent grid
    t1 = min(350, int(0.32 * budget_s))
    if t1 > 30:
        h_opt, c5_opt = optimize(h_parent, n_parent, t1)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: alternative starts on parent grid
    elapsed = time.time() - t_start
    if elapsed < budget_s * 0.55:
        starts = [make_random_start(n_parent, seed + 1),
                  make_sine_start(n_parent, seed + 2),
                  make_block_start(n_parent, seed + 3)]
        for h_st in starts:
            if time.time() - t_start > budget_s * 0.55:
                break
            rem = int(budget_s * 0.55 - (time.time() - t_start))
            t2 = min(180, rem)
            if t2 < 20:
                break
            h_a, c5_a = optimize(h_st, n_parent, t2, T_start=30, T_end=300, step0=0.12)
            if c5_a < best_c5:
                best_h, best_c5 = h_a, c5_a

    # Phase 3: finer grids
    for n_fine in [127, 199, 255]:
        remaining = budget_s - (time.time() - t_start)
        if remaining < 80:
            break
        t3 = min(int(remaining * 0.45), 350)
        if t3 < 40:
            break
        h_fine = upsample(best_h, n_fine)
        h_f, c5_f = optimize(h_fine, n_fine, t3, T_start=80, T_end=800, step0=0.07)
        if c5_f < best_c5:
            best_h, best_c5, best_n = h_f, c5_f, n_fine

    # Phase 4: final polish on best grid
    remaining = int(budget_s - (time.time() - t_start) - 5)
    if remaining > 25:
        h_p, c5_p = optimize(best_h, best_n, remaining, T_start=300, T_end=3000, step0=0.02)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_constraint(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
