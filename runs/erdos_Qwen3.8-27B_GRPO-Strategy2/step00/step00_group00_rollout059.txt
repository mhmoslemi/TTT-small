import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GDX = 0.0
_GBETA = 30.0

def _obj(h):
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full') * _GDX
    m = np.max(corr)
    if m < 1e-15:
        return 0.0
    s = np.sum(np.exp(_GBETA * (corr - m)))
    return float(m + np.log(s) / _GBETA)

def _sum_con(h):
    return np.sum(h) - _GN / 2.0

def _sum_jac(h):
    return np.ones(_GN)

def compute_c5(h, dx):
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    return float(np.max(corr) * dx)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    h_new += (target - np.sum(h_new)) / n_new
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-8:
        mask = (h_new > 0.02) & (h_new < 0.98)
        nm = np.sum(mask)
        if nm > 0:
            h_new[mask] += diff / nm
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def do_slsqp(h0, n, maxiter, beta_start, beta_end):
