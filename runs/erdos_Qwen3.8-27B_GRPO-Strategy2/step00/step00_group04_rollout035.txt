import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def opt_coord(h, i):
    n = len(h)
    n_lags = 2 * n - 1
    h[i] = 0.0
    baseline = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    offsets = np.arange(n_lags) - (n - 1)
    idx1 = i + offsets
    idx2 = i - offsets
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    coeff = np.zeros(n_lags)
    if v1.any():
        coeff[v1] += (1.0 - h[idx1[v1]])
    if v2.any():
        coeff[v2] -= h[idx2[v2]]
    xs = np.linspace(0.0, 1.0, 65)
    vals = (np.outer(xs, coeff) + baseline).max(axis=1)
    bi = int(vals.argmin())
    bx = xs[bi]
    if 0 < bi < 64:
        lo, hi = xs[bi - 1], xs[bi + 1]
        xf = np.linspace(lo, hi, 65)
        vf = (np.outer(xf, coeff) + baseline).max(axis=1)
        bx = xf[int(vf.argmin())]
    h[i] = bx

def cd_optimize(h, max_sweeps, deadline):
    n = len(h)
    h = h.astype(np.float64).copy()
    best = compute_c5(h)
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        changed = False
        for i in range(n):
            old = h[i]
            opt_coord(h, i)
            if abs(h[i] - old) > 1e-11:
                c5 = compute_c5(h)
                if c5 < best - 1e-13:
                    best = c5
                    changed = True
        if not changed:
            break
    return h, best

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    d = h.sum() - tgt
    if abs(d) < 1e-9 * n:
        return h
    if d > 0:
        idx = np.argsort(h)[::-1]
        rem = d
        for k in idx:
            r = min(h[k], rem)
            h[k] -= r
            rem -= r
            if rem < 1e-12:
                break
    else:
        idx = np.argsort(h)
        rem = -d
        for k in idx:
            r = min(1.0 - h[k], rem)
            h[k] += r
            rem -= r
            if rem < 1e-12:
                break
    return h

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    s = h_new.sum()
    if s > 1e-15:
        h_new *= (n_new * 0.5) / s
    return fix_sum(h_new, n_new)

def random_h(n, rng):
    h = rng.uniform(0, 1, n)
    return fix_sum(h, n)

def pair_update(h, i, j):
    n = len(h)
    n_lags = 2 * n - 1
    h_i, h_j = h[i], h[j]
    h[i] = 0.0
    h[j] = 0.0
    baseline = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    offsets = np.arange(n_lags) - (n - 1)
    ci = np.zeros(n_lags)
    cj = np.zeros(n_lags)
    for idx in [i, j]:
        id1 = idx + offsets
        id2 = idx - offsets
        v1 = (id1 >= 0) & (id1 < n)
        v2 = (id2 >= 0) & (id2 < n)
        c = np.zeros(n_lags)
        if v1.any():
            c[v1] += (1.0 - h[id1[v1]])
        if v2.any():
            c[v2] -= h[id2[v2]]
        if idx == i:
            ci = c
        else:
            cj = c
    # result[k] = baseline[k] + ci[k]*x + cj[k]*y
    # Minimize max_k over x,y in [0,1]
    # Grid over x, solve 1D for y
    best_val = np.inf
    best_x, best_y = 0.0, 0.0
    for xs in np.linspace(0.0, 1.0, 33):
        resid = baseline + ci * xs
        yv = (np.linspace(0.0, 1.0, 33)[:, None] * cj[None, :] + resid[None, :]).max(axis=1)
        yi = int(yv.argmin())
        vy = yv[yi]
        if vy < best_val:
            best_val = vy
            best_x = xs
            best_y = np.linspace(0.0, 1.0, 33)[yi]
    h[i] = best_x
    h[j] = best_y
