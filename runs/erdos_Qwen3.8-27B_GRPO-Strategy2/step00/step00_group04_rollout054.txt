import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    cores = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cores) * dx)

def repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_grad(h, n, eps=1e-5):
    g = np.zeros(n)
    for i in range(n):
        h[i] += eps
        fp = compute_c5(h, n)
        h[i] -= 2.0 * eps
        fm = compute_c5(h, n)
        h[i] += eps
        g[i] = (fp - fm) / (2.0 * eps)
    return g

def integral_constraint(x):
    return np.sum(x) - len(x) / 2.0

def integral_jacobian(x):
    return np.ones(len(x)) / len(x)

def slsqp_optimize(h0, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': integral_constraint, 'jac': integral_jacobian}]
    res = minimize(compute_c5, h0, method='SLSQP', args=(n,),
                   jac=lambda x: c5_grad(x, n), bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res

def local_search(h0, n, time_limit, rng, init_step=0.03):
    best_h = h0.copy()
    best_val = compute_c5(best_h, n)
    start = time.time()
    step = init_step
    while time.time() - start < time_limit:
        n_pert = int(rng.integers(1, 5))
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_new = best_h.copy()
        h_new[idxs] += rng.uniform(-step, step, size=n_pert)
        h_new = repair(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_new.copy()
            step = max(0.002, step * 0.997)
        elif val < best_val + step * 0.01 and rng.random() < 0.05:
            best_h = h_new.copy()
        step = min(0.15, step * 1.003)
    return best_h, best_val

def coordinate_descent(h0, n, time_limit, rng):
    best_h = h0.copy()
    best_val = compute_c5(best_h, n)
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            h_try = best_h.copy()
            for delta in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                h_try[i] = np.clip(best_h[i] + delta, 0.0, 1.0)
                h_try = repair(h_try, n)
                val = compute_c5(h_try, n)
                if val < best_val - 1e-14:
                    best_val = val
                    best_h = h_try.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_val
