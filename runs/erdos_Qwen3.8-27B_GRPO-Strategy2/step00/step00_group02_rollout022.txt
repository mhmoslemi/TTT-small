def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        s = np.sum(room)
        if s < 1e-13:
            return h
        h = h + diff * room / s
    else:

        s = np.sum(room)
        if s < 1e-13:
            return h
        h = h + diff * room / s
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    obj = m + np.log(Z) / beta
    w = exp_shifted / Z
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-14:
            continue
        k = mi - (n - 1)
        wk = w[mi] * dx
        if k >= 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] += wk * (-h[:n - k])
        else:
            q = -k
            grad[q:] += wk * (1.0 - h[:n - q])
            grad[:n - q] += wk * (-h[q:])
    return obj, grad

def slsqp_step(h, n, beta, maxiter=400):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,

    res = minimize(
        fun=lambda x: smooth_max_and_grad(x, beta)[0],
        x0=h,
        jac=lambda x: smooth_max_and_grad(x, beta)[1],
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

            freq = rng.uniform(1.5, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            h_r = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(nr) / nr + phase)
            h_r += 0.1 * rng.randn(nr)
            h_r = project_feasible(h_r, nr)

        for beta in [10, 40, 100]:
            if tleft() < 10:
                break
            h_trial = lfbfgs_step(h_r, nr, beta, maxiter=500)
            c5 = compute_c5(h_trial)
            if c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5
            h_r = h_trial

    # Phase 5: Final polish
    if tleft() > 10:
        for beta in [200, 500, 1000]:
            if tleft() < 5:
                break
            h_new = slsqp_step(best_h, len(best_h), beta, maxiter=300)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
