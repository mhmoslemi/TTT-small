import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def repair_integral(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def interp_to_res(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    xs = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return np.interp(xt, xs, h_src, left=h_src[0], right=h_src[-1])

def nm_optimize(obj, x0, n, max_iter, t_limit):
    t_start = time.time()
    best = obj(x0)
    best_x = x0.copy()
    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-11,
                            'maxfev': max_iter * n * 2})
    t_elapsed = time.time() - t_start
    if time.time() - t_start > t_limit:
        return best_x, best
    return res.x, float(res.fun)

    t0 = time.time()
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.ones(47) * 0.5

    h0 = repair_integral(h0, n0)
    dx0 = 2.0 / n0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx0)

    def elapsed():
        return time.time() - t0

    n = n0
    dx = 2.0 / n

    def obj(x):
        return compute_c5(x, dx)

    remaining = budget_s - elapsed()
    if remaining > 30:
        mx = max(3000, int(min(remaining * 0.3, 300) * 10))
        r = nm_optimize(obj, h0, n, mx, min(remaining * 0.3, 300))
        h_new = repair_integral(r[0], n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # --- Phase 2: Powell refinement ---
    remaining = budget_s - elapsed()
    if remaining > 30:
        bounds = [(0.0, 1.0)] * n
        pw = minimize(obj, best_h, method='Powell', bounds=bounds,
                      options={'maxiter': 20000, 'xtol': 1e-10, 'ftol': 1e-12})
        h_pw = repair_integral(pw.x, n)
        c5_pw = compute_c5(h_pw, dx)

        c5_2 = compute_c5(h2, dx_hi)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

        remaining = budget_s - elapsed()
        if remaining > 60 and c5_2 < best_c5 + 0.001:
            bounds_hi = [(0.0, 1.0)] * n_hi
            pw2 = minimize(obj_hi, h2, method='Powell', bounds=bounds_hi,
                           options={'maxiter': 5000, 'xtol': 1e-9, 'ftol': 1e-11})
            h2b = repair_integral(pw2.x, n_hi)

    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Bounds violated"
