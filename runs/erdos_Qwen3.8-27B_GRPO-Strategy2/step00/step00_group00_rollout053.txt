            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(45, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, len(h_init))

    # Phase 1: SA on upsized solution
    for n in [150, 300, 500]:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 10
        if remaining < 40:
            break
        h_curr = build_upsample(best_h, n)
        sa_time = remaining * 0.55
        h_sa, c5_sa = sa_optimize(h_curr, n, sa_time, seed=seed + n)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa.copy(), c5_sa

    # Phase 2: SLSQP polish at best resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10
    if remaining > 30:
        n_polish = len(best_h)
        if n_polish > 400:
            n_polish = 300
            h_polish = build_upsample(best_h, n_polish)
            h_polish, c5p = sa_optimize(h_polish, n_polish, remaining * 0.3, seed=seed + 999)
        else:
            h_polish = best_h.copy()
        h_slq, c5_slq = slsqp_polish(h_polish, n_polish, maxiter=200)
        if c5_slq < best_c5:
            best_h, best_c5 = h_slq.copy(), c5_slq

    # Phase 3: Binary local search
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10
    if remaining > 20:
        for n_bin in [200, 400]:
            elapsed = time.time() - t0
            remaining = budget_s - elapsed - 5
            if remaining < 15:
                break
            h_seed = build_upsample(best_h, n_bin)
            h_b, c5_b = binary_local_search(h_seed, n_bin, remaining * 0.4, seed=seed + 500 + n_bin)
            if c5_b < best_c5:
                best_h, best_c5 = h_b.copy(), c5_b
            # Also try SA starting from binary
            h_sa2, c5_sa2 = sa_optimize(h_b, n_bin, remaining * 0.3, seed=seed + 700 + n_bin)
            if c5_sa2 < best_c5:
                best_h, best_c5 = h_sa2.copy(), c5_sa2

    # Phase 4: Random restarts
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 5
    if remaining > 30:
        for trial in range(3):
            elapsed = time.time() - t0
            remaining = budget_s - elapsed - 5
            if remaining < 15:
                break
            n = 200
            rng = np.random.RandomState(seed + 1000 + trial * 7)
            h_rand = rng.uniform(0, 1, n)
            h_rand = fix_constraint(h_rand, n)
            h_rr, c5_rr = sa_optimize(h_rand, n, remaining * 0.3, seed=seed + 2000 + trial)
            if c5_rr < best_c5:
                best_h, best_c5 = h_rr.copy(), c5_rr

    # Final polish
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 2
    if remaining > 10 and len(best_h) <= 400:
        h_slq, c5_slq = slsqp_polish(best_h, len(best_h), maxiter=100)
        if c5_slq < best_c5:
            best_h, best_c5 = h_slq.copy(), c5_slq

    n_best = len(best_h)
    best_h = fix_constraint(best_h, n_best)
    final_c5 = compute_c5(best_h, n_best)
    return best_h, final_c5, n_best
