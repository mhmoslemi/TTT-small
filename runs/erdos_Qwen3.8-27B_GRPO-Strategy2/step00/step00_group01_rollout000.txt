import numpy as np
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def obj(h):
    return compute_c5(h)


def sum_con_fun(h):
    return np.sum(h) - len(h) / 2.0


def sum_con_jac(h):
    return np.ones(len(h))


def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    deficit = n / 2.0 - np.sum(h)
    if abs(deficit) > 1e-12:
        if deficit > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h = h.copy()
            h[mask] += deficit / cnt

def make_cosine(n, freq=1.0, phase=0.0, amp=0.5):

    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    return project_feasible(h, n)


def make_random(n, rng, amp=0.3):
    h = 0.5 + amp * rng.standard_normal(n)
    return project_feasible(h, n)


def interp_up(h_from, n_to):

    x_old = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x_new = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return project_feasible(np.interp(x_new, x_old, h_from), n_to)
