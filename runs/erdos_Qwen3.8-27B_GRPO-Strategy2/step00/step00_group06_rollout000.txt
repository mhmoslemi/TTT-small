def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

    g = compute_correlation(h)
    return np.max(g) * dx

def gradient_max_overlap(h, n_points):
    """Compute gradient of max_k g_k where g = correlate(h, 1-h, 'full')"""
    n = len(h)
    g = compute_correlation(h)
    max_val = np.max(g)
    # Find active set (within tolerance of max)
    tol = 1e-6
    active = np.where(g > max_val - tol)[0]
    
    grad = np.zeros(n)
    for idx in active:
        shift = idx - (n - 1)
        # g_idx = sum_m h[m] * (1-h[m+shift]) for valid m, m+shift
        # dg/dh[j] = (1-h[j+shift]) if j+shift valid - h[j-shift] if j-shift valid
        # More carefully:
        # g_idx = sum over m where 0<=m<n and 0<=m+shift<n of h[m]*(1-h[m+shift])

        # contribution to dg/dh[j] from term m=j: (1-h[j+shift]) if 0<=j+shift<n
        # contribution to dg/dh[j] from term m+shift=j i.e. m=j-shift: -h[j-shift] if 0<=j-shift<n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(g) * dx)

def gradient_overlap(h, n_points, T=20.0):
    n = n_points
    g = np.correlate(h, 1.0 - h, mode='full')
    max_g = np.max(g)
    exp_g = np.exp(T * (g - max_g))
    wsum = np.sum(exp_g)
    grad = np.zeros(n)
    for idx in range(len(g)):
        w = exp_g[idx] / wsum
        if w < 1e-9:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n - shift]
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    return grad

def project_feasible(h, n_points):
