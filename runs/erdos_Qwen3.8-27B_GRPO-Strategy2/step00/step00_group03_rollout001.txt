def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def anneal(h, t_hi, t_lo, n_iter, rng):
    n = len(h)
    h = h.copy().astype(float)
    best_h = h.copy()
    cur = compute_c5(h)
    best = cur
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, cur
    lo = max(t_lo, 1e-18)
    hi = max(t_hi, 1e-18)
    log_r = np.log(lo / hi)
    for it in range(n_iter):
        frac = (it + 1) / n_iter
        T = hi * np.exp(log_r * frac)

        b = zeros[rng.randint(len(zeros))]

    h[pos[: n // 2]] = 1.0
    return h


def random_balanced(n, rng):
    h = np.zeros(n, dtype=float)

        end = min(start + bl, n)

    cnt = int(np.sum(h > 0.5))
    if cnt > n // 2:
        ones = np.where(h > 0.5)[0]

    elif cnt < n // 2:

        b = rng.randint(n)
        if a != b and h[a] != h[b]:
