import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def smooth_max_value(h, n, t):
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = t * (corr - m)
    return m + np.log(np.exp(shifted).sum()) / t


def smooth_max_grad(h, n, t):
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = t * (corr - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    n_full = 2 * n - 1
    grad = np.zeros(n)
    idx_range = range(n_full)
    for idx in idx_range:
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        jp = np.arange(n) + lag
        jm = np.arange(n) - lag
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        if np.any(vp):
            grad[vp] += w * dx * (1.0 - h[jp[vp]])
        if np.any(vm):
            grad[vm] -= w * dx * h[jm[vm]]
    return grad

        h = np.clip(h, 0.0, 1.0)
