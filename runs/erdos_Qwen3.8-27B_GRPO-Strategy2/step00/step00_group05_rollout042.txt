import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def make_binary(h, n):
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def weyl_binary(n, alpha=0.618033988749895):
    h = np.zeros(n)
    k = n // 2
    for i in range(k):
        pos = int((i + 0.5) * alpha * n) % n
        h[pos] = 1.0
    while np.sum(h) < k:
        empty = np.where(h == 0)[0]
        h[empty[0]] = 1.0
    return h

def block_binary(n, p):
    h = np.zeros(n)
    half = p // 2
    for i in range(n):
        if (i % p) < half:
            h[i] = 1.0
    while np.sum(h) < n // 2:
        e = np.where(h == 0)[0]
        h[e[0]] = 1.0
    while np.sum(h) > n // 2:
        o = np.where(h == 1)[0]
        h[o[0]] = 0.0
    return h

def sa_binary(h, max_iters, t0, cooling, seed):

    n = len(h)

    zi = 0

        t = t0 * (cooling ** it)
        i = ones[oi % len(ones)]
        j = zeros[zi % len(zeros)]
        oi += 1
        zi += 1
        h[i], h[j] = 0, 1
        new = compute_c5(h)
        d = new - cur
        if d < 0 or rng.random() < np.exp(-d / max(t, 1e-9)):
            cur = new
            ones = np.concatenate([ones[ones != i], [j]])
            zeros = np.concatenate([zeros[zeros != j], [i]])
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best_h, best

def gradient_refine(h, n, max_iters, seed):
    rng = np.random.RandomState(seed)
    h = project_feasible(h.copy(), n)
    cur_c = compute_c5(h)
    best = cur_c
    best_h = h.copy()
    for it in range(max_iters):
        lr = 0.08 * (0.9995 ** it)
        h_try = h + rng.randn(n) * lr
        h_try = project_feasible(h_try, n)
        c = compute_c5(h_try)
        if c < cur_c:
            h = h_try
            cur_c = c
        if c < best:
            best = c
            best_h = h_try.copy()
    return best_h, best
