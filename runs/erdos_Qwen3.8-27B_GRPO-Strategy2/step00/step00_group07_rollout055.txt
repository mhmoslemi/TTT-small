            h0 = np.array(initial_h_values, dtype=float)
        except Exception:
            h0 = None

    if h0 is not None and len(h0) == n:
        h = h0.copy()
    else:
        h = np.zeros(n, dtype=float)
        h[::2] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Binary SA (60% of budget)
    sa_end = t_start + budget_s * 0.55
    h_bin, c_bin = _binary_sa(h, n, target_sum, rng, sa_end)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()

    # Phase 1b: Second SA round from perturbed best (20% of budget)
    t2_end = t_start + budget_s * 0.75
    h_pert = best_h.copy()
    h_pert = (h_pert >= 0.5).astype(float)
    for _ in range(8):
        a = rng.integers(0, n); b = rng.integers(0, n)
        idx1 = np.where(h_pert == 1.0)[0]
        idx0 = np.where(h_pert == 0.0)[0]
        h_pert[rng.choice(idx1)] = 0.0
        h_pert[rng.choice(idx0)] = 1.0
    if time.time() < t2_end:
        h_bin2, c_bin2 = _binary_sa(h_pert, n, target_sum, rng, t2_end, T0=0.3, alpha=0.9996)
        if c_bin2 < best_c5:
            best_c5 = c_bin2
            best_h = h_bin2.copy()

    # Phase 2: Continuous optimization (15% of budget)
    cont_end = t_start + budget_s * 0.92
    best_h, best_c5 = _continuous_optimize(best_h, n, target_sum, cont_end, rng)

    # Phase 3: Fine local perturbations (8% of budget)
    final_end = t_start + budget_s * 0.98
    best_h, best_c5 = _local_perturb(best_h, n, target_sum, final_end, rng)

    # Ensure constraints
    best_h = _fix_sum(best_h, target_sum)

    return best_h, float(best_c5), n
