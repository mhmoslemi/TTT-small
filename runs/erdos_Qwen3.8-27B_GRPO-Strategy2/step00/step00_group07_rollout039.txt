```python
def project_box_sum(h, target_sum):

    h = h.copy()

        if abs(diff) < 1e-14:

            adj = -diff / mask.sum()
            h[mask] += min(adj, 1.0 - h[mask].max())
    h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def project_box_sum(h, target_sum):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] -= min(diff / mask.sum(), h[mask].min())
        else:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] += min(-diff / mask.sum(), 1.0 - h[mask].max())
    return np.clip(h, 0.0, 1.0)

def c5_obj_and_grad(h, target_sum, p=20.0, penalty_coef=50.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c = np.maximum(c, 1e-30)
    c_p = c ** p
    smooth_max = c_p.sum() ** (1.0 / p)
    alpha = smooth_max ** (1.0 - p)
    w = c ** (p - 1.0)
    A = 1.0 - h
    B = h
    T1 = np.convolve(A, w, mode='full')
    T2 = np.convolve(B, w[::-1], mode='full')
    c5_val = smooth_max * dx
    grad_c5 = dx * alpha * (T1[n-1:2*n-1] - T2[n-1:2*n-1])
    # Penalty for sum constraint
    sum_val = h.sum()
    penalty = penalty_coef * (sum_val - target_sum) ** 2
    grad_penalty = 2.0 * penalty_coef * (sum_val - target_sum) * np.ones(n)
    return c5_val + penalty, grad_c5 + grad_penalty

def optimize_at_resolution(h0, time_budget, p=20.0):
    n = len(h0)
    target_sum = n / 2.0
    start = time.time()
    h = h0.copy()
    
    best_obj = c5_obj_and_grad(h, target_sum, p)[0]
    best_h = h.copy()
    
    for _ in range(5):  # Multiple L-BFGS-B restarts
        elapsed = time.time() - start
        if elapsed > time_budget - 5:
            break
        try:
            res = minimize(
                lambda x: c5_obj_and_grad(x, target_sum, p)[0],
                h,
                jac=lambda x: c5_obj_and_grad(x, target_sum, p)[1],
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': int(time_budget / 10), 'ftol': 1e-12,
                         'gtol': 1e-10, 'maxls': 30}
            )
            h_new = project_box_sum(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(best_h):
                best_h = h_new
            h = h_new + np.random.randn(n) * 0.01  # Perturb for restart
            h = project_box_sum(h, target_sum)
        except Exception:
            break
    
    return project_box_sum(best_h, target_sum)

def upsample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(80, 0.5)
    n = len(h)
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phase 1: Optimize at initial resolution
    if remaining() > 60:
        h_opt = optimize_at_resolution(h, min(remaining(), 100))
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 2: Higher resolutions
    current_n = len(best_h)
    for target_n in [160, 320, 640]:
        if target_n <= current_n:
            continue
        if remaining() < 90:
            break
        h_new = upsample(best_h, target_n)
        h_opt = optimize_at_resolution(h_new, min(remaining() * 0.6, 180))
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5, current_n = h_opt, c5_opt, target_n
    
    # Phase 3: Random local search
    if remaining() > 30:
        h = best_h.copy()
        n = len(h)
        ts = n / 2.0
        step = 0.05
        for _ in range(int(remaining() * 3)):
            if remaining() < 5:
                break
            mask = np.random.random(n) < 0.2
            if mask.sum() == 0:
                continue
            pert = np.zeros(n)
            pert[mask] = np.random.randn(mask.sum()) * step
            h_p = project_box_sum(h + pert, ts)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                h = h_p
                best_h, best_c5 = h_p.copy(), c5_p
                step = min(step * 1.05, 0.15)
            else:
                step *= 0.97
                if step < 0.003:
                    break
    
    # Phase 4: Final L-BFGS-B polish
    if remaining() > 40:
        n = len(best_h)
        h_polish = optimize_at_resolution(best_h, min(remaining(), 80), p=30)
        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_h, best_c5 = h_polish, c5_polish
    
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    return best_h, compute_c5(best_h), len(best_h)
```

```python
        cnt = mask.sum()
        if cnt < 2:
            mask[np.random.choice(n, 3, replace=False)] = True
            cnt = 3
        pert = np.zeros(n)
        pert[mask] = np.random.randn(cnt) * step
        h_p = project_box_sum(h + pert, ts)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            h = h_p
            best_c5 = c5_p
            best_h = h.copy()
            step = min(step * 1.04, 0.18)
        else:
            step *= 0.98
            if step < 0.003:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _g_trunc
    t_start = time.time()
    np.random.seed(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Initialize from parent if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(80, 0.5)
    n_init = len(h)
    h = project_box_sum(h, n_init / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Optimize at initial resolution
    if remaining() > 50:
        h1 = optimize_round(h, min(remaining(), 90), p_val=20, n_iter_cap=200)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Try higher resolutions
    current_n = len(best_h)
    for target_n in [160, 320, 640]:
        if target_n <= current_n:
            continue
        if remaining() < 80:
            break
        h_new = upsample(best_h, target_n)
        c5_interp = compute_c5(h_new)
        h2 = optimize_round(h_new, min(remaining() * 0.65, 200), p_val=25,
                           n_iter_cap=max(50, 60000 // target_n))
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        elif c5_interp < best_c5:
            best_h, best_c5 = h_new, c5_interp

    # Phase 3: Random local search
    if remaining() > 25:
        h3, c5_3 = random_search(best_h, min(remaining() * 0.5, 200))
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Final L-BFGS-B polish with higher p
    if remaining() > 40:
        nf = len(best_h)
        h4 = optimize_round(best_h, min(remaining(), 80), p_val=50,
                           n_iter_cap=max(30, 40000 // nf))
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Second random search pass if time remains
    if remaining() > 20:
        h5, c5_5 = random_search(best_h, min(remaining() * 0.6, 100))
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
```
