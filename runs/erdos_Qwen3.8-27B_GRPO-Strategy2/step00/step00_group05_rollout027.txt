def binary_local_search(h, max_iters=10000, time_limit=100):
    n = len(h)

import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_continuous(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h = np.clip(h + adj * mask, 0.0, 1.0)
    return h

def binarize(h):
    # Convert to binary with exactly n/2 ones
    n = len(h)
    target = n // 2
    thresh = np.sort(h)[::-1][target - 1] if target > 0 else 1.0
    b = (h >= thresh).astype(float)
    # Adjust to get exactly target ones
    count = int(np.sum(b))
    if count > target:
        # Remove excess ones (those closest to threshold)
        ones_idx = np.where(b == 1)[0]
        vals = h[ones_idx]
        # Remove the smallest ones
        order = np.argsort(vals)
        for idx in order[:count - target]:
            b[idx] = 0
    elif count < target:
        zeros_idx = np.where(b == 0)[0]
        vals = h[zeros_idx]
        order = np.argsort(vals)[::-1]
        for idx in order[:target - count]:
            b[idx] = 1
    return b

def binary_steepest_descent(h, time_limit=30):
    n = len(h)
    h = h.copy().astype(float)
    target = n // 2
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_move = None
        best_val = current_c5
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                val = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if val < best_val - 1e-15:
                    best_val = val
                    best_move = (i, j)
        
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        current_c5 = best_val
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_sa(h, time_limit=60, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    target = n // 2
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    T = 0.01  # initial temperature (in units of C5)
    T_min = 1e-6
    cooling = 0.999
    
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cooling, T_min)
        if T <= T_min * 1.01:
            T = 0.005  # restart temperature
    
    return best_h, best_c5

def continuous_optimize(h, time_limit=60, seed=42):
    n = len(h)
    target = n * 0.5
    
    def objective(x):
        h = np.clip(x, 0.0, 1.0)
        # Soft max of correlation
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        tau = 0.01
        # shifted log-sum-exp for numerical stability
        m = np.max(corr)
        return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))

        base = objective(x)
        penalty = 100.0 * (np.sum(x) - target) ** 2 + 1000.0 * np.sum(np.maximum(0, x - 1)) ** 2 + 1000.0 * np.sum(np.maximum(0, -x)) ** 2
        return base + penalty
    
    bounds = [(0.0, 1.0)] * n
    result = minimize(constrained_objective, h, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 2000, 'ftol': 1e-10})
    
    h_opt = np.clip(result.x, 0.0, 1.0)
    h_opt = repair_continuous(h_opt, n)
    c5 = compute_c5(h_opt)
    return h_opt, c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def correlate_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_continuous(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h = np.clip(h + (diff / cnt) * mask, 0.0, 1.0)
    return h

def binarize(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b

def binary_steepest_descent(h, time_limit=60.0):
    n = len(h)
    h = h.copy().astype(float)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_move = None
        best_val = current_c5
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                val = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if val < best_val - 1e-15:
                    best_val = val
                    best_move = (i, j)
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        current_c5 = best_val
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return best_h, best_c5

def binary_sa(h, time_limit=90.0, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    T = 0.003
    T_min = 5e-6
    cooling = 0.9997
    stall = 0
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 0.01:
            break
        # Adaptive: do 50 moves per "batch"
        for _ in range(50):
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta <= 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 50
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T * cooling, T_min)
        if stall > 2000:
            T = 0.005
            stall = 0
    return best_h, best_c5

def continuous_refine(h, time_limit=60.0, seed=42):
    n = len(h)
    target = n * 0.5

    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        c = correlate_vals(hx)
        tau = 0.005
        m = np.max(c)
        softmax = m + tau * np.log(np.sum(np.exp((c - m) / tau)))
        pen = 500.0 * (np.sum(x) - target) ** 2
        return softmax + pen

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8})
    h_opt = repair_continuous(np.clip(res.x, 0.0, 1.0), n)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def structured_init(n, rng):
    h = np.zeros(n)
    # Spread ones roughly evenly with jitter
    positions = np.sort(rng.permutation(n)[:n // 2])
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 70
    target = n * 0.5

    # Initial from parent
    if 'initial_h_values' in dir() or True:
        try:
            h_init = np.array(initial_h_values, dtype=float)
        except NameError:
            h_init = np.full(n, 0.5)
    h_init = np.clip(h_init, 0.0, 1.0)
    if len(h_init) != n:
        h_init = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_init)), h_init)
        h_init = repair_continuous(h_init, n)
    else:
        h_init = repair_continuous(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    def elapsed():
        return time.time() - t0

    def budget_left():
        return budget_s - elapsed() - 5.0  # 5s safety

    # Phase 1: Binarize + steepest descent
    h_bin = binarize(h_init)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    tl = min(40.0, budget_left() * 0.15)
    if tl > 2:
        h_b, c5_b = binary_steepest_descent(h_bin, time_limit=tl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 2: SA from best
    tl = min(100.0, budget_left() * 0.2)
    if tl > 5:
        h_sa, c5_sa = binary_sa(best_h, time_limit=tl, seed=seed)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Continuous optimization
    tl = min(60.0, budget_left() * 0.12)
    if tl > 5:
        h_cont, c5_cont = continuous_refine(best_h, time_limit=tl, seed=seed)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 4: More SA with different temperature
    tl = min(120.0, budget_left() * 0.25)
    if tl > 10:
        h_sa2, c5_sa2 = binary_sa(best_h, time_limit=tl, seed=seed + 100)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Random restarts + local search
    restarts = 0
    while budget_left() > 20:
        tl = min(15.0, budget_left() * 0.1)
        h_r = random_binary(n, rng)
        h_r, c5_r = binary_steepest_descent(h_r, time_limit=tl)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        restarts += 1
        if restarts > 30:
            break

    # Phase 6: Final continuous refinement
    tl = min(60.0, budget_left() * 0.3)
    if tl > 5:
        h_c, c5_c = continuous_refine(best_h, time_limit=tl, seed=seed + 200)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 7: Final binary polish
    if budget_left() > 5:
        h_final_bin = binarize(best_h)
        c5_fb = compute_c5(h_final_bin)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_final_bin.copy()
        tl = min(20.0, budget_left() * 0.5)
        if tl > 2:
            h_pf, c5_pf = binary_steepest_descent(best_h, time_limit=tl)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()

    # Final validation
    best_h = repair_continuous(best_h, n)
    best_c5 = compute_c5(best_h)
    assert 0 <= np.min(best_h) and np.max(best_h) <= 1.0
    assert abs(np.sum(best_h) - target) < 1e-8
    assert np.all(np.isfinite(best_h))

    return best_h, best_c5, n
