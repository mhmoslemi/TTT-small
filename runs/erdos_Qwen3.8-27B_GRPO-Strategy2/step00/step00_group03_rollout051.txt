    if remaining() > 60:
        h_sg, c5_sg = subgrad_descent(h0, n, max_iter=3000, alpha0=0.08)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg
    if remaining() > 60:
        h_sg, c5_sg = subgrad_descent(best_h, n, max_iter=3000, alpha0=0.03)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg

    # Phase 2: Nelder-Mead from best
    if remaining() > 30:
        h_nm, c5_nm = nm_optimize(best_h, n, maxiter=2000)
        if c5_nm < best_c5:
            best_c5, best_h = c5_nm, h_nm

    # Phase 3: Random restarts with Nelder-Mead
    attempt = 0
    while remaining() > 40:
        r = attempt % 4
        if r == 0:
            z0 = best_h + 0.03 * rng.randn(n)
        elif r == 1:
            z0 = best_h + 0.1 * rng.randn(n)
        elif r == 2:
            z0 = best_h + 0.3 * rng.randn(n)
        else:
            z0 = np.clip(best_h * 0.7 + 0.3 * rng.rand(n), 0.0, 1.0)
            z0 = repair(z0, n)
        z0 = np.clip(z0, 0.0, 1.0)
        mx = min(1500, max(300, int(remaining() * 0.3 / 0.02)))
        h_nm, c5_nm = nm_optimize(z0, n, maxiter=mx)
        if c5_nm < best_c5:
            best_c5, best_h = c5_nm, h_nm
        attempt += 1
        if attempt % 5 == 0 and remaining() > 120:
            h_sg, c5_sg = subgrad_descent(best_h, n, max_iter=1000, alpha0=0.02)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg

    # Phase 4: Polish with SLSQP
    if remaining() > 20:
        def obj_s(z):
            return compute_c5(z, n)
        cons = [{'type': 'eq', 'fun': lambda z: z.sum() - n * 0.5}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_s, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_s = repair(res.x, n)
            c5_s = compute_c5(h_s, n)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s
        except Exception:
            pass

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return (best_h, best_c5, n)
