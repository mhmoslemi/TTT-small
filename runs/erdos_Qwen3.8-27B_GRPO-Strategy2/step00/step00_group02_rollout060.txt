        if cnt < 1e-12:

    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = float(np.sum(mask))
        if cnt < 1e-14:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def _lse(x, t):
    xs = np.asarray(x, dtype=float) / t
    m = np.max(xs)
    return float(t * (m + np.log(np.sum(np.exp(xs - m)))))


def _sm(h, t, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return _lse(c, t)

    s = np.sum(h) - n / 2.0
