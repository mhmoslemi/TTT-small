# Helper implementation functions
def _evaluate_correlation(h):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr) * (2.0/n))

def _upscale_linear(h, target_size):
    return np.interp(np.linspace(0,1,target_size), np.linspace(0,1,len(h)), h)

def _downscale_cosine(h, target_size, base_size):
    h_orig = np.interp(np.linspace(0,1,base_size), np.linspace(0,1,target_size), h)
    return np.interp(np.linspace(0,1,base_size), [i/base_size for i in range(base_size+1)], h_orig)
