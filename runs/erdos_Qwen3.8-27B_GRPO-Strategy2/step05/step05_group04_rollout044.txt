            h_new = _proj_to_sum(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new <= cur_c + 1e-15:
                h = h_new
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
        if (it + 1) % 200 == 0:
            mu = max(0.6, mu - 0.005)
    return best_h, best_c

def _slsqp_smooth(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-13:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _freq_perturb(h, rng, amp_scale):
    n = len(h)
    t = np.arange(n)
    hp = h.copy()
    for _f in range(rng.randint(1, 4)):
        f = rng.randint(1, n // 4 + 1)
        a = amp_scale * rng.randn()
        ph = rng.rand() * 2 * np.pi
        hp += a * np.sin(2 * np.pi * f * t / n + ph)
        hp += 0.5 * a * np.cos(2 * np.pi * f * t / n + ph)
    return _repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP polish at n0
    for alpha in [1000, 5000, 50000]:
        if _el() > 60:
            break
        hc = _slsqp_smooth(best_h, alpha, 2000)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: Active-set gradient (main workhorse) at n0
    if _el() < budget_s - 350:
        h2, c2 = _active_opt(best_h, 6000, 5, rng, mu=0.82)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        h2, c2 = _active_opt(best_h, 4000, 7, rng, mu=0.75)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Upsample to 2*n0 and optimize
    if _el() < budget_s - 280:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [1000, 10000, 80000]:
            if _el() > budget_s - 180:
                break
            hc = _slsqp_smooth(h_hi, alpha, 1500)
            c5 = _eval(hc)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hc.copy(), n1
        if _el() < budget_s - 120:
            h2, c2 = _active_opt(best_h, 3000, 5, rng, mu=0.8)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
            h2, c2 = _active_opt(best_h, 2000, 8, rng, mu=0.7)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 4: Multi-start with structured perturbations
    if _el() < budget_s - 100:
        nc = best_n
        n_starts = 25
        for r in range(n_starts):
            if _el() > budget_s - 100:
                break
            strat = r % 5
            if strat == 0:
                hp = _freq_perturb(best_h, rng, 0.04)
            elif strat == 1:
                hp = _freq_perturb(best_h, rng, 0.08)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(nc // 10, nc // 4)
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] = 1.0 - hp[st:st + bs]
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = _repair(best_h + 0.03 * rng.randn(nc), nc)
            else:
                hp = _freq_perturb(best_h, rng, 0.02)
            hc = _slsqp_smooth(hp, 20000, 800)
            c5 = _eval(hc)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hc.copy(), nc
            h2, c2 = _active_opt(hc, 1500, 5, rng)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 5: Focused active-set with small top_m
    if _el() < budget_s - 50:
        for tm in [3, 4, 6]:
            if _el() > budget_s - 50:
                break
            h2, c2 = _active_opt(best_h, 3000, tm, rng, mu=0.7)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 6: Fine coordinate descent
    if _el() < budget_s - 5:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        while _el() < budget_s - 3:
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 5e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
