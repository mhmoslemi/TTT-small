def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return obj

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs_scaled(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _softmax_obj(h, n, alpha, lam):
    c = _corrs_scaled(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha + lam * (h.sum() - n * 0.5) ** 2

def _grad_correct(h, n, w, lam):
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        coeff = w[k] * (2.0 / n)
        if s == 0:
            g += coeff * (1.0 - 2.0 * h)
        elif s > 0:
            g[s:] += coeff * (1.0 - h[:n - s])
            g[:n - s] -= coeff * h[s:]
        else:
            sp = -s
            g[:n - sp] += coeff * (1.0 - h[sp:])
            g[sp:] -= coeff * h[:n - sp]
    g += 2.0 * lam * (h.sum() - n * 0.5)
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        g = _grad_correct(h, n, w, 0.0)
        improved = False

            f_new = _corrs_scaled(h_new).max()
            if f_new < f_curr - 1e-14:
                h = h_new; f_curr = f_new; improved = True
                break
            step *= 0.4
        if not improved:
            break
        step = min(step * 1.1, step0)
    return h, f_curr

def _to_binary(h, n):
    idx = np.argsort(-h)[:n // 2]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _sa_binary(hb, n, t0, deadline, seed):

    ones = np.where(hb == 1)[0].tolist()

def _active_descent(h, n, topm, max_iter, t0, deadline):
    h = h.copy()
    f = _eval(h)
    step = 0.004
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top:
            s = k - (n - 1)
            if s == 0:
                g += (1.0 - 2.0 * h)
            elif s > 0:
                g[s:] += (1.0 - h[:n - s])
                g[:n - s] -= h[s:]
            else:

                g[:n - sp] += (1.0 - h[sp:])

        g *= (2.0 / (n * topm))

            step *= 0.5

        step = min(step * 1.06, 0.004)
    return h, f

def _slsqp_step(h, n, alpha, max_iter, ftol):
    lam = 50.0
    bnds = [(0.0, 1.0)] * n

    def _f(x):
        return _softmax_obj(x, n, alpha, lam)

    def _j(x):
        c = _corrs_scaled(x)
        s = np.clip(alpha * (c - c.max()), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        return _grad_correct(x, n, w, lam)

    try:
        res = minimize(_f, h, jac=_j, method='SLSQP', bounds=bnds,
                       options={'maxiter': max_iter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
