    if abs(s - target) > 1e-10:

        diff = target - h.sum()
        if abs(diff) > 1e-12:

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])

                    diff += a

    n = len(h)
    sh = (n - 1) - k_idx

        g[i1] += (1.0 - h[i1 - sp])

    step = step0

            g += _lag_gradient(h, ki)
