import time

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_c_vec(h, k, n):
    g = np.zeros(n)
    if k < n:
        shift = k
        if shift == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - shift] += (1.0 - h[shift:n])
            g[shift:] -= h[:n - shift]
    else:
        s = k - (n - 1)
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def _soft_grad(h, n, top_m, sa):
    corr = _corr(h)
    m = corr.max()
    idx = np.argsort(corr)[-top_m:]
    w = np.exp(sa * (corr[idx] - m))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(top_m); w[-1] = 1.0; w_sum = 1.0
    else:
        w /= w_sum
    g = np.zeros(n)
    for i in range(top_m):
        g += w[i] * _grad_c_vec(h, idx[i], n)
    return g, m * 2.0 / n

def _prox_step(h, g, alpha, n):
    v = h - alpha * g
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    target = 0.5 * n
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_c_vec(h, k, n):

            g[s:] -= h[:n - s]
    else:

        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def _soft_grad(h, n, top_m, sa):
    corr = _corr(h)
    m = float(corr.max())
    idx = np.argsort(corr)[-top_m:]

    if ws < 1e-30:

        g += w[i] * _grad_c_vec(h, idx[i], n)
    return g, m * 2.0 / n
