    tmp[i] += d
    tmp[j] -= d
    return float(np.max(np.correlate(tmp, 1.0 - tmp, mode='full')) * dx)

def _pair_opt(h, i, j, dx, iters=25):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-10: return 0.0
    gr = (np.sqrt(5) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    dd = a + gr * (b - a)
    fc = _c5_pair(h, i, j, c, dx)
    fd = _c5_pair(h, i, j, dd, dx)

        if fc < fd:
            b, dd, fd = dd, c, fc
            c = b - gr * (b - a)
            fc = _c5_pair(h, i, j, c, dx)
        else:
            a, c, fc = c, dd, fd
            dd = a + gr * (b - a)
            fd = _c5_pair(h, i, j, dd, dx)
    return (a + b) / 2.0

    n = len(h)
    best = _c5(h, dx)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        c = _corr(h)
        center = n - 1
        top_lags = np.argsort(c)[::-1][:4]
        # Build set of active positions
        active = set()

            if s == 0: continue
            for a in range(n - s):

                    active.add(a+s)

Actually, here's another idea: **alternating projection**. 
- The feasible set is: {h : 0<=h<=1, sum(h)=n/2}
- The "ideal" set (C5=0) is: {h : h(x)*(1-h(x+k))=0 for all x,k} which means h is a "code" where no two 1s are at any distance... but that's only possible for binary h with no two 1s at any distance, which means at most 1 is 1. Not feasible with sum=n/2.

So the problem is inherently non-zero. The question is what structure minimizes the max overlap.

One structural insight: if h is binary (0 or 1), then h(x)*(1-h(x+k)) = 1 iff h(x)=1 and h(x+k)=0. So the overlap at lag k is the number of positions where h has a 1 and position k away has a 0. To minimize this, we want the 1s to be "spread out" so that at every distance, about the same fraction of positions have 1 at x and 0 at x+k.

For a "random" binary h with half 1s and half 0s, the expected overlap at any lag is n/2 * 1/2 = n/4 (in units of dx, that's n/4 * 2/n = 1/2). But we can do much better with structured patterns.

The optimal binary pattern is related to "perfect difference sets" or "Bongarda networks" - patterns where the autocorrelation is as flat as possible.

But we don't need to be binary. The continuous version gives more degrees of freedom.

def _c5(h, dx):

def _repair(h, target):

    n = len(h)

        d = target - h.sum()
        if abs(d) < 1e-14:

def _grad_c(h, alpha):

    s = np.clip(alpha * (c - m), -300, 300)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return _obj_c(x, alpha)
    def grad(x):
        return _grad_c(x, alpha)
    def eq(x):
        return x.sum() - tgt
    def eqj(x):
        return ones
    try:
        r = minimize(obj, h.copy(), method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eqj}],
                     jac=grad, options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _pair_line(h, i, j, dx, iters=20):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    dd = a + gr * (b - a)
    tc = h.copy(); tc[i] += c; tc[j] -= c
    fc = float(np.max(np.correlate(tc, 1.0 - tc, mode='full')) * dx)
    td = h.copy(); td[i] += dd; td[j] -= dd
    fd = float(np.max(np.correlate(td, 1.0 - td, mode='full')) * dx)
    for _ in range(iters):
        if fc < fd:
            b = dd; dd = c; fd = fc
            c = b - gr * (b - a)
            tc[i] = h[i] + c; tc[j] = h[j] - c
            fc = float(np.max(np.correlate(tc, 1.0 - tc, mode='full')) * dx)
        else:
            a = c; c = dd; fc = fd
            dd = a + gr * (b - a)
            td[i] = h[i] + dd; td[j] = h[j] - dd
            fd = float(np.max(np.correlate(td, 1.0 - td, mode='full')) * dx)
    return (a + b) / 2.0

def _sgd_momentum(h, dx, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    vel = np.zeros(n)
    mu = 0.6
    lr = 0.004
    stall = 0

            h = h_new
            lr = min(lr * 1.005, 0.02)
