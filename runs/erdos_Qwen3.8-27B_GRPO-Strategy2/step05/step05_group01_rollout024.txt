```python
def _fw_step(h, grad):
    # Solve min <grad, g> s.t. 0<=g<=1, sum(g)=n/2
    n = len(h)
    target = n / 2.0
    sorted_g = np.sort(grad)
    # Find threshold
    cumsum = np.cumsum(sorted_g)
    # We need sum(g) = target
    # g_i = 0 if grad_i > threshold, g_i = 1 if grad_i < threshold
    # One g_i is fractional
    # Find k such that sum of smallest k-1 < target < sum of smallest k
    # Actually: g_i = 1 for i < k, g_k = target - k + 1, g_i = 0 for i > k
    # where sorted_g[k-1] is the threshold
    k = int(np.searchsorted(cumsum - sorted_g, target - 0.5))
    # Simpler: just set g = 0 or 1 based on grad, then fix sum
    threshold = np.median(grad)  # rough
    g = np.where(grad < threshold, 1.0, 0.0)
    # Fix
    diff = target - g.sum()
    if diff > 0:
        # Need to add diff more ones
        zero_idx = np.where(g == 0)[0]
        zero_idx = zero_idx[np.argsort(grad[zero_idx])[:int(np.ceil(diff))]]
        g[zero_idx] = 1.0
    elif diff < 0:
        one_idx = np.where(g == 1)[0]
        one_idx = one_idx[np.argsort(-grad[one_idx])[:int(np.ceil(-diff))]]
        g[one_idx] = 0.0
    # Handle fractional
    diff = target - g.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            zero_idx = np.where(g == 0)[0]
            if len(zero_idx) > 0:
                best = zero_idx[np.argmin(grad[zero_idx])]
                g[best] = diff
        else:
            one_idx = np.where(g == 1)[0]
            if len(one_idx) > 0:
                best = one_idx[np.argmax(grad[one_idx])]
                g[best] = 1.0 + diff  # = 1 - |diff|
    return g
```

Actually, the exact solution to the linear oracle is:
- Sort grad: grad_(1) ≤ grad_(2) ≤ ... ≤ grad_(n)
- Find k such that (k-1) < target < k, i.e., k = ceil(target) = n/2 (which is an integer since n is even!)
- So g has exactly n/2 ones and n/2 zeros
- g_i = 1 for the n/2 smallest grad_i, g_i = 0 otherwise

Since target = n/2 is an integer, there's no fractional element! The solution is simply:
g_i = 1 if grad_i is among the n/2 smallest, else 0.

This is O(n log n) per iteration.

Frank-Wolfe algorithm:
```
h = initial point
for t = 0, 1, 2, ...:
    grad = ∇f(h)
    g = argmin_{g∈P} <grad, g>  (linear oracle)
    gamma = 2/(t+2)  (or line search)
    h = (1-gamma)*h + gamma*g
```

With the 2/(t+2) step size, convergence is O(1/t). With line search, it can be faster.

```python
import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _topk_grad(h, K):
    n = len(h)
    c = _corr(h)
    topk = np.argsort(c[-K:]) + (2 * n - 1 - K)
    g = np.zeros(n)
    for k in topk:
        shift = (n - 1) - k
        if shift >= 0:
            j1 = np.arange(shift, n)
            g[j1] += (1.0 - h[j1 - shift])
            j2 = np.arange(0, n - shift)
            g[j2] -= h[j2 + shift]
        else:
            sp = -shift
            j1 = np.arange(sp, n)
            g[j1] += (1.0 - h[j1 - sp])
            j2 = np.arange(0, n - sp)
            g[j2] -= h[j2 + sp]
    g /= K
    return g, c.max()

def _proj_constraint(v, n):
    target = n / 2.0
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    s1_lo = max(i - n + 1, -(n - 1))
    s1_hi = min(i, n - 1)
    if s1_lo <= s1_hi:
        sv = np.arange(s1_lo, s1_hi + 1)
        dc[sv + n - 1] += di * (1.0 - h[i - sv])
    s2_lo = max(-i, -(n - 1))
    s2_hi = min(n - 1 - i, n - 1)
    if s2_lo <= s2_hi:
        sv2 = np.arange(s2_lo, s2_hi + 1)
        dc[sv2 + n - 1] -= di * h[i + sv2]
    return dc

def _nesterov_fast(h, t0, max_time, n_iter, K):
    n = len(h)
    h_cur = h.copy()
    y = h.copy()
    t_acc = 1.0
    step = 0.01
    best_c5 = _eval(h_cur)
    best_h = h_cur.copy()
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > max_time:
            break
        grad, _ = _topk_grad(y, K)
        h_new = _proj_constraint(y - step * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc)) * 0.5
        y = h_new + (t_acc - 1.0) / t_new * (h_new - h_cur)
        y = np.clip(y, 0.0, 1.0)
        h_cur = h_new
        t_acc = t_new
        c5 = _eval(h_cur)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_cur.copy()
            stall = 0
            step = min(step * 1.03, 0.025)
        else:
            stall += 1
            step *= 0.96
        if stall > 300:
            step = max(step, 0.005)
            stall = 0
    return best_h

def _incr_pair(h, c, rng, n_iter, scale, n, t0, max_time):
    dx = 2.0 / n
    c5c = c.max() * dx
    h = h.copy()
    c = c.copy()
    for _ in range(n_iter):
        if time.time() - t0 > max_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c5 = (c + dc).max() * dx
        if new_c5 < c5c - 1e-15:
            h[i] = ni
            h[j] = nj
            c = c + dc
            c5c = new_c5
    return h, c, c5c

def _grad_guided_pair(h, rng, n_iter, t0, max_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c = _corr(h)
    c5c = c.max() * dx
    K = 5
    for it in range(n_iter):
        if time.time() - t0 > max_time:
            break
        if it % 50 == 0:
            grad, _ = _topk_grad(h, K)
            neg_mask = grad < -1e-8
            pos_mask = grad > 1e-8
            if neg_mask.any() and pos_mask.any() and rng.random() < 0.75:
                w_neg = -grad[neg_mask]
                w_pos = grad[pos_mask]
                if w_neg.sum() > 0 and w_pos.sum() > 0:
                    w_neg /= w_neg.sum()
                    w_pos /= w_pos.sum()
                    ni_idx = np.where(neg_mask)[0]
                    pj_idx = np.where(pos_mask)[0]
                    i = ni_idx[rng.choice(len(ni_idx), p=w_neg)]
                    j = pj_idx[rng.choice(len(pj_idx), p=w_pos)]
                else:
                    i = rng.randint(n)
                    j = rng.randint(n)
            else:
                i = rng.randint(n)
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d = 0.008 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c5 = (c + dc).max() * dx
        if new_c5 < c5c - 1e-15:
            h[i] = ni
            h[j] = nj
            c = c + dc
            c5c = new_c5
    return h

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wv = w[k]
        if wv < 1e-14:
            continue
        shift = (n - 1) - k
        if shift >= 0:
            j1 = np.arange(shift, n)
            g[j1] += wv * (1.0 - h[j1 - shift])
            j2 = np.arange(0, n - shift)
            g[j2] -= wv * h[j2 + shift]
        else:
            sp = -shift
            j1 = np.arange(sp, n)
            g[j1] += wv * (1.0 - h[j1 - sp])
            j2 = np.arange(0, n - sp)
            g[j2] -= wv * h[j2 + sp]
    return g

def _slsqp_smooth(h, alpha, t0, max_time):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        return _smooth_grad(x, alpha)
    def _eq_fun(x):
        return x.sum() - target
    def _eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    from scipy.optimize import minimize
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-18})
        if time.time() - t0 < max_time:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)
```
