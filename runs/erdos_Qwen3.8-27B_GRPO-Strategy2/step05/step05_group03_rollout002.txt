                break

    c = _corr_vec(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s)))/alpha

    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    for ki in range(2*n-1):
        wi = w[ki]
        if wi < 1e-12:

        g[s_pos:] += wi*(1.0-h[:m_])
        g[:m_] -= wi*h[s_pos:]

    bds = [(0.0,1.0)]*n
    def obj(x):
        return _softobj(x, alpha)
    def eq_c(x):
        return [x.sum()-tgt]
    def jac(x):
        return _softgrad(x, alpha)
    try:

from scipy.optimize import minimize

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_vec(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _soft_obj_grad(h, alpha):
    c = _corr_vec(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    obj = m + np.log(np.sum(np.exp(s))) / alpha
    w = np.exp(s)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-13:
            continue
        sp = abs(ki - (n - 1))
        if sp == 0:
            continue
        m_ = n - sp
        g[sp:] += wi * (1.0 - h[:m_])
        g[:m_] -= wi * h[sp:]
    return obj, g
