def _lp_subgradient(h, n, solver='ECOS'):
    """Solve LP relaxation of the max-min problem"""
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    n_k = len(c)  # 2n - 1
    
    # Variables: h (n), t (scalar)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
    
    for k in range(n_k):
        # Gradient of c[k] * dx w.r.t. h
        sh = (n - 1) - k
        g = np.zeros(n)
        if sh >= 0:
            idx = np.arange(n - sh)
            g[idx] += (1.0 - h[idx + sh])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
        
        # Linearization: c[k]*dx + g^T * (h_var - h) <= t_var
        constraints.append(c[k] * dx + g @ (h_var - h) <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if h_var.value is not None and t_var.value is not None:
            return np.clip(h_var.value, 0, 1), t_var.value
    except Exception:
        pass
    return h, c.max() * dx

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad(h, k, n):
    """Gradient of the k-th correlation value w.r.t. h, without dx factor."""
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        idx = np.arange(n - sh)
        g[idx] += (1.0 - h[idx + sh])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _power_obj(h, p):
    c = _corr(h)
    return (np.sum(c.astype(float) ** p)) ** (1.0 / p)

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        g += wk * _lag_grad(h, k, n)
    return g

def _lp_refine(h, n, max_iters=50):
    """LP subgradient method for the max-min problem"""
    dx = 2.0 / n
    h_cur = h.copy()
    best = h_cur.copy()
    best_c = _eval(h_cur)
    
    for it in range(max_iters):
        c = _corr(h_cur)
        c_dx = c * dx
        c_max = c_dx.max()
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
        
        for k in range(2 * n - 1):
            g = _lag_grad(h_cur, k, n)
            cons.append(c_dx[k] + g @ (h_var - h_cur) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var + 1e-6 * cp.sum_squares(h_var - h_cur)), cons)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if h_var.value is not None:
                h_new = _repair(np.clip(h_var.value, 0, 1), n)
                c_new = _eval(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best = h_new.copy()
                # Damped update
                h_cur = 0.5 * h_cur + 0.5 * h_new
                h_cur = _repair(h_cur, n)
        except Exception:
            break
        
        if c_max < 0.38:
            break
    
    return best, best_c

def _alt_lag_grad(h, n, M=5, max_rounds=200):
    """Alternating top-M lag gradient descent with momentum"""
    dx = 2.0 / n
    momentum = np.zeros(n)
    beta = 0.5
    h_cur = h.copy()
    best = h_cur.copy()
    best_c = _eval(h_cur)
    
    for r in range(max_rounds):
        c = _corr(h_cur) * dx
        c_max = c.max()
        
        if c_max < best_c:
            best_c = c_max
            best = h_cur.copy()
        
        # Get top M worst lags
        top_k = np.argsort(c)[-M:][::-1]
        
        for k in top_k:
            g = _lag_grad(h_cur, k, n) * dx
            momentum = beta * momentum + (1 - beta) * g
            for step in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
                h_new = _proj_to_sum(h_cur - step * momentum, n / 2)
                c_new = _eval(h_new)
                if c_new < c_max - 1e-15:
                    h_cur = h_new
                    c_max = c_new
                    break
        
        if r % 20 == 0 and r > 0:
            beta *= 0.95
    
    return best, best_c

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sa_block(h, n, t0, t1, n_iter, rng):
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        blk = rng.randint(3, max(4, n // 10))
        st = rng.randint(0, n - blk)
        scale = 0.02 + 0.08 * t
        delta = rng.randn(blk) * scale
        h_new = h.copy()
        h_new[st:st + blk] += delta
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-8)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_to_sum(h, n * 0.5)
    return h

    n = len(h)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lag_grad(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += 1.0 - h[sh:]
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def _sobj(x, alpha, n, pw):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha + pw * (x.sum() - n * 0.5) ** 2

def _sgrad(x, alpha, n, pw):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        g += w[k] * dx * _lag_grad(x, (n - 1) - k, n)
    g += 2.0 * pw * (x.sum() - n * 0.5)
    return g

def _slsqp(h, alpha, mi, ft, pw=1.0e4):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, n, pw),
                     method='SLSQP', bounds=bnds,
                     options={'maxiter': mi, 'ftol': ft})
        if r.x is not None and not np.all(np.isnan(r.x)):
            return _repair(r.x, n)
    except Exception:
        pass
    return h

    dx, mom = 2.0 / n, np.zeros(n)

def _lp_ref(h, n, mi, td):
    h = h.copy()

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _proj_to_sum(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_to_sum(h, n * 0.5)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lag_grad(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += 1.0 - h[sh:]
        g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def _sobj(x, alpha, n, pw):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha + pw * (x.sum() - n * 0.5) ** 2

def _sgrad(x, alpha, n, pw):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        g += w[k] * dx * _lag_grad(x, (n - 1) - k, n)
    g += 2.0 * pw * (x.sum() - n * 0.5)
    return g

def _slsqp(h, alpha, mi, ft, pw=1.0e4):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, n, pw),
                     method='SLSQP', bounds=bnds,
                     options={'maxiter': mi, 'ftol': ft})
        if r.x is not None and not np.any(np.isnan(r.x)):
            return _repair(r.x, n)
    except Exception:
        pass
    return h

def _alt_lag(h, n, M, mr, td):
    h = h.copy()
    best, best_c = h.copy(), _eval(h)
    dx = 2.0 / n
    mom = np.zeros(n)
    for _ in range(mr):
        if time.time() > td:
            break
        c = _corr(h) * dx
        cm = c.max()
        mom *= 0.9
        for k in np.argsort(c)[-M:]:
            g = _lag_grad(h, (n - 1) - k, n) * dx
            mom += 0.3 * g
            for st in [0.05, 0.02, 0.01, 0.005, 0.002]:
                hn = _proj_to_sum(h - st * mom, n * 0.5)
                cn = _eval(hn)
                if cn < cm - 1e-15:
                    h = hn
                    cm = _corr(hn).max() * dx
                    if cm < best_c:
                        best, best_c = h.copy(), cm
                    break
    return best, best_c

def _lp_ref(h, n, mi, td):
    h = h.copy()
    best, best_c = h.copy(), _eval(h)
    dx = 2.0 / n
    for _ in range(mi):
        if time.time() > td:
            break
        c = _corr(h) * dx
        hv, tv = cp.Variable(n), cp.Variable()
        cs = [hv >= 0, hv <= 1, cp.sum(hv) == n * 0.5]
        for k in range(2 * n - 1):
            cs.append(c[k] + _lag_grad(h, (n - 1) - k, n) @ (hv - h) <= tv)
        p = cp.Problem(cp.Minimize(tv + 1e-4 * cp.sum_squares(hv - h)), cs)
        try:
            p.solve(solver=cp.ECOS, verbose=False)
            if hv.value is not None:
                hn = _repair(hv.value, n)
                cn = _eval(hn)
                if cn < best_c:
                    best, best_c = hn.copy(), cn
                h = _repair(0.5 * h + 0.5 * hn, n)
        except Exception:
            break
    return best, best_c

def _bin_sea(h, n, td, rng):
    hb = (h > 0.5).astype(float)
    no, tg = int(hb.sum()), n // 2
    if no > tg:
        oi = np.where(hb == 1)[0]
        hb[oi[np.argsort(h[oi])[:no - tg]]] = 0
    elif no < tg:
        zi = np.where(hb == 0)[0]
        hb[zi[np.argsort(-h[zi])[:tg - no]]] = 1
    best, bc = hb.copy(), _eval(hb)
    while time.time() < td:
        oi = np.where(hb == 1)[0]
        zi = np.where(hb == 0)[0]
        imp = False
        for _ in range(3000):
            if time.time() > td:
                break
            i = oi[rng.randint(len(oi))]
            j = zi[rng.randint(len(zi))]
            hb[i], hb[j] = 0.0, 1.0
            cn = _eval(hb)
            if cn < bc - 1e-14:
                bc, best, imp = cn, hb.copy(), True
                oi, zi = np.where(hb == 1)[0], np.where(hb == 0)[0]
            else:
                hb[i], hb[j] = 1.0, 0.0
        if not imp:
            break
    return best, bc

def _sa(h, n, ts, te, ni, rn, td):
    cc, bb, bc = _eval(h), h.copy(), _eval(h)
    for i in range(ni):
