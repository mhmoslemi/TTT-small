import time
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_objgrad(x, n, top_k, lam):
    """Objective: mean of top-k squared correlations + penalty for sum constraint.
    Returns (value, gradient)."""
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    c_scaled = c * dx
    top_idx = np.argsort(c_scaled)[::-1][:top_k]
    val = np.mean(c_scaled[top_idx] ** 2)
    s = x.sum() - n * 0.5
    val += lam * s * s
    g = np.zeros(n)
    for ki in top_idx:
        sh = ki - (n - 1)
        ck = c_scaled[ki]
        if sh >= 0:
            g[:n-sh] += ck * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= ck * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += ck * (1.0 - x[:n-sp])
            g[:n-sp] -= x[sp:] * ck
    g *= 2.0 * dx / top_k
    g += 2.0 * lam * s
    return val, g

def _refine_lbfgsb(h, n, t_end, top_k=10, lam=100.0, max_iter=5000):
    """Refine using L-BFGS-B on smooth top-k objective."""
    def fun(x):
        return _topk_objgrad(x, n, top_k, lam)[0]
    def jac(x):
        return _topk_objgrad(x, n, top_k, lam)[1]
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': max_iter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_sa(h, n, n_iter, t_end, rng, T0=10.0):
    """Simulated annealing on binary sequence."""
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = int(b.sum()) - target
    if diff > 0:
        idx = np.where(b == 1)[0]
        rng.shuffle(idx)
        b[idx[:diff]] = 0
    elif diff < 0:
        idx = np.where(b == 0)[0]
        rng.shuffle(idx)
        b[idx[:(-diff)]] = 1
    
    c5 = _eval(b) * n / 2.0
    best_b = b.copy()
    best_c5 = c5
    T = T0
    scale = 2.0 / n
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = max(T * 0.9997, 0.01)
        # Random swap of a 1 and a 0
        i = rng.randint(n)
        j = rng.randint(n)
        while b[i] == b[j] and i != j:
            j = rng.randint(n)
        if i == j:
            continue
        b[i], b[j] = 0.0, 1.0
        c5_new = _eval(b) * n / 2.0
        delta = c5_new - c5
        if delta < 0 or (T > 0.01 and rng.rand() < np.exp(-delta / T)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_b = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    
    return best_b, best_c5 * 2.0 / n

def _coord_descent(h, n, n_iter, t_end, rng, scale=0.003):
    """Coordinate descent with random pairwise moves."""
    best = h.copy()
    best_c5 = _eval(h)
    c5_c = best_c5
    curr = h.copy()
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = scale * rng.randn()
        ni, nj = curr[i] + d, curr[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        curr[i], curr[j] = ni, nj
        c5_new = _eval(curr)
        if c5_new < c5_c - 1e-15:
            c5_c = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = curr.copy()
        else:
            curr[i], curr[j] = curr[i] - d, curr[j] + d
            scale *= 0.999
            if scale < 1e-6:
                scale = 0.003
                curr = best + 0.005 * rng.randn(n)
                curr = _repair(curr, n)
                c5_c = _eval(curr)
    return best, best_c5

from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_val(x, n, top_k, lam):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[::-1][:top_k]
    s = x.sum() - n * 0.5
    return np.mean(c[idx] ** 2) + lam * s * s

def _topk_grad(x, n, top_k, lam):
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    c_sc = c_raw * dx
    idx = np.argsort(c_sc)[::-1][:top_k]
    s = x.sum() - n * 0.5
    g = np.zeros(n)
    for ki in idx:
        sh = ki - (n - 1)
        ck = c_sc[ki]
        if sh >= 0:
            g[:n - sh] += ck * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= ck * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += ck * (1.0 - x[:n - sp])
            g[:n - sp] -= x[sp:] * ck
    g *= 2.0 * dx / top_k
    g += 2.0 * lam * s
    return g

def _obj_wrap(x, n, top_k, lam):
    return _topk_val(x, n, top_k, lam)

def _jac_wrap(x, n, top_k, lam):
    return _topk_grad(x, n, top_k, lam)

                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_desc(h, n, n_iter, t_end, rng, scale0=0.003):

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = curr[i] + d, curr[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        curr[i], curr[j] = ni, nj
        c5_new = _eval(curr)

                c5_c = _eval(curr)

    best_c5 = _eval(h)
    curr = h.copy()
    c5_c = best_c5
    lr = lr_init
    rng2 = np.random.RandomState(0)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(curr, 1.0 - curr, mode='full')
        top_idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for ki in top_idx:
            sh = ki - (n - 1)
            if sh >= 0:
                g[:n - sh] += (1.0 - curr[sh:])
                if sh > 0:
                    g[sh:] -= curr[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - curr[:n - sp])
                g[:n - sp] -= curr[sp:]
        g *= (2.0 / (n * top_k))
        cand = _repair(curr - lr * g, n)
        c5_new = _eval(cand)
        if c5_new < c5_c * (1 - 1e-8):
            curr = cand
            c5_c = c5_new
            lr = min(lr * 1.03, 0.06)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = cand.copy()
        else:

                curr = best + 0.008 * rng2.randn(n)
                curr = _repair(curr, n)
                c5_c = _eval(curr)
                lr = lr_init * 0.5
