def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

                h_new += (target-s)/n
                h_new = np.clip(h_new, 0, 1)
            h, f = h_new, _smooth_obj(h_new, alpha)
        step = min(step*1.015, step0*5.0)
    return _repair(h, n)

def _topk_g(h, k):
    n = len(h)
    C = _corr(h)
    idx = np.argsort(C)[::-1][:k]
    dx = 2.0/n
    g = np.zeros(n)
    for ki in idx:
        sh = (n-1)-ki
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += (1.0-h[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2-sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0-h[i1-sp])
            i2 = np.arange(n-sp)
            g[i2] -= h[i2+sp]
    return g*dx/k

def _upsample(h, factor):
    return np.repeat(h, factor)

def _fourier_init(n, K, rng):
    t = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K+1):
        h += 0.12*rng.randn()*np.cos(np.pi*k*t)
        h += 0.12*rng.randn()*np.sin(np.pi*k*t)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    def _el():
        return time.time() - t0

    # Phase A: L-BFGS-B penalty (fresh optimizer vs parent SLSQP)
    for alpha in [3000, 15000, 60000]:
        if _el() > 100: break
        h_new = _lbfbs(best_h, alpha, 2500, 200.0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase B: SLSQP
    for alpha in [30000, 100000, 300000]:
        if _el() > 220: break
        h_new = _slsqp(best_h, alpha, 2500, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase C: Fast coordinate descent
    if _el() < budget_s-550:
        h_new = _coord_cd(best_h, 80, 20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase D: Mixed topk + smooth momentum gradient
    if _el() < budget_s-450:
        h_new = _topk_proj_grad(best_h, 20000, 5, 500, 0.003, 0.88)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase E: More coordinate descent
    if _el() < budget_s-350:
        h_new = _coord_cd(best_h, 100, 24)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase F: Upsample 2x and optimize
    if _el() < budget_s-300:
        n1 = n0*2
        h_hi = _repair(_upsample(best_h, 2), n1)
        h_hi = _lbfbs(h_hi, 10000, 2000, 200.0)
        h_hi = _slsqp(h_hi, 80000, 1500, 1e-19)
        h_hi = _coord_cd(h_hi, 60, 20)
        c5 = _eval(h_hi)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_hi, n1

    # Phase G: Fourier restarts
    if _el() < budget_s-200:
        for K in [4, 7, 10]:
            if _el() > budget_s-200: break
            h_f = _fourier_init(n0, K, rng)
            h_f = _lbfbs(h_f, 5000, 1500, 100.0)
            h_f = _slsqp(h_f, 30000, 800, 1e-18)
            h_f = _coord_cd(h_f, 40, 16)
            c5 = _eval(h_f)
            if c5 < best_c5:
                best_c5, best_h = c5, h_f

    # Phase H: Multi-start
    nc = best_n
    if _el() < budget_s-120:
        for _r in range(12):
            if _el() > budget_s-120: break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h+0.025*rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0-best_h+0.015*rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(4, nc//7)
                st = rng.randint(0, nc-bs)
                hp[st:st+bs] += 0.05*rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = _repair(np.roll(best_h, rng.randint(1, max(2,nc//12))), nc)
            elif strat == 4:
                t_arr = np.arange(nc)
                hp = best_h.copy()
                for f in rng.randint(1, nc//4+1, size=4):
                    hp += 0.009*rng.randn()*np.sin(2*np.pi*f*t_arr/nc)
                hp = _repair(hp, nc)
            else:
                hp = best_h.copy()
                mask = rng.rand(nc) < 0.25
                hp[mask] = np.clip(hp[mask]+0.04*rng.randn(mask.sum()), 0, 1)
                hp = _repair(hp, nc)
            al = [5000, 30000, 120000][rng.randint(3)]
            h_new = _slsqp(hp, al, 600, 1e-18)
            h_new = _coord_cd(h_new, 20, 16)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new

    # Phase I: Fine SA local search
    if _el() < budget_s-20:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.004, 0.0012, 0.0003]:
            if _el() > budget_s-20: break
            for _it in range(10000):
                if _el() > budget_s-20: break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j: continue
                dv = scale*rng.randn()
                ni, nj = h[i]+dv, h[j]-dv
                if ni<0 or ni>1 or nj<0 or nj>1: continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i]-dv, h[j]+dv

    # Phase J: Final coordinate descent
    if _el() < budget_s-3:
        h_new = _coord_cd(best_h, 120, 28)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
