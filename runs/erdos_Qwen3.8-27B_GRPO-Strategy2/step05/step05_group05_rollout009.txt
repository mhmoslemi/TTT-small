    c_obj[-1] = 1.0

from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _build_lp(h):
    n = len(h)
    dx = 2.0 / n
    n_k = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    A = np.zeros((n_k, n + 1))
    b = np.zeros(n_k)
    Ar = np.zeros(n)
    for k in range(n_k):
        sh = (n - 1) - k
        Ar[:] = 0.0
        if sh >= 0:
            Ar[:n - sh] += (1.0 - h[sh:]) * dx
            Ar[sh:] += -h[:n - sh] * dx
        else:
            sp = -sh
            Ar[sp:] += (1.0 - h[:n - sp]) * dx
            Ar[:n - sp] += -h[sp:] * dx
        A[k, :n] = Ar
        A[k, -1] = -1.0
        b[k] = c[k] - Ar @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(c.max() * 0.5, None)]
    return c_obj, A, b, A_eq, b_eq, bounds

def _cutting_plane(h, max_iters, t0, t_budget):
    n = len(h)
    c5c = _eval(h)
    for it in range(max_iters):
        if time.time() - t0 > t_budget:
            break
        c_obj, A, b, A_eq, b_eq, bounds = _build_lp(h)
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_lp = res.x[:n]
        improved = False
        for frac in [1.0, 0.6, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]:
            h_try = (1.0 - frac) * h + frac * h_lp
            h_try = _repair(h_try, n)
            c5_try = _eval(h_try)
            if c5_try < c5c - 1e-14:
                h = h_try
                c5c = c5_try
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _smooth_obj_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm), -400, 400)
    s_sh = s - s.max()
    exp_s = np.exp(s_sh)
    w = exp_s / exp_s.sum()
    obj = (cm + np.log(exp_s.sum()) / alpha) * (2.0 / n)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return obj, g

def _slsqp(h, alpha, maxiter, t0, t_budget):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - len(x) * 0.5
    def _cj(x):
        return np.ones(len(x))
    def _of(x):
        o, _ = _smooth_obj_grad(x, alpha)
        return o
    def _oj(x):
        _, g = _smooth_obj_grad(x, alpha)
        return g
    try:
        res = minimize(_of, h, jac=_oj, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, max_iters, n_top, t0, t_budget):

    dx = 2.0 / n
    target = n * 0.5

            best_h = h_new.copy()

    # Phase 6: Final SLSQP polish with very high alpha
    if _el() < budget_s - 30:
        h_new = _slsqp(best_h.copy(), 2000000, 3000, t0, budget_s - 10)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 7: Ultra-fine active descent
    if _el() < budget_s - 10:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 4, t0, budget_s - 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
