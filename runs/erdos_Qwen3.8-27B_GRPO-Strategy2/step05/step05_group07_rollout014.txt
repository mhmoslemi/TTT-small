    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n-sp]
    return g * (2.0 / n)

def _smooth_obj(x, alpha, mu):
    c = _corrs(x) * (2.0 / len(x))
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    smax = s.max()
    pen = mu * (x.sum() - len(x) * 0.5) ** 2
    return m + np.log(np.sum(np.exp(s - smax))) / alpha + pen

    n = len(x)
    c = _corrs(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(k, x)
    g += 2.0 * mu * (x.sum() - n * 0.5) * np.ones(n)
    return g

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    start = time.time()
    def _timeup():
        return (time.time() - start) > budget_rem or (time.time() - _t0) > _budget - 15
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxfun': 5000})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterog(h, iters, budget_rem, lr0=0.02):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    v = x.copy()
    best = _eval(x)
    best_x = x.copy()
    start = time.time()

        x = x_new
        f = _eval(x)
        if f < best:

            best_x = x.copy()
        x = best_x  # track best
    return best_x, best

    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
