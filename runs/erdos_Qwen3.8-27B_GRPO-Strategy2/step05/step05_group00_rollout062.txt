def _binary_best_swap(h, n):
    """Find the best single swap (i→1, j→0) that minimizes max overlap."""
    c_raw = np.correlate(h, 1-h, mode='full')  # length 2n-1
    c5_raw = c_raw.max()

    zeros = np.where(h == 0)[0]
    
    # Precompute up[p, k] for all p, k
    # up[p, k] = (1-h[p-s])*valid1 - h[p+s]*valid2 where s = k-(n-1)
    up = np.zeros((n, 2*n-1))
    for p in range(n):
        for k in range(2*n-1):
            s = k - (n-1)
            v1 = 0 <= p-s < n
            v2 = 0 <= p+s < n
            up[p, k] = 0
            if v1:
                up[p, k] += (1 - h[p-s])
            if v2:
                up[p, k] -= h[p+s]
    
    # For each zero i, find best one j
    best_c5 = c5_raw
    best_i, best_j = -1, -1
    for i in zeros:
        # new_c for swap (i→1, j→0): c + up[i] - up[j]
        # We want to minimize max over all j (in ones) of max_k (c[k] + up[i,k] - up[j,k])
        base = c_raw + up[i]  # shape (2n-1,)
        # For each j: max_k (base[k] - up[ones[j], k])
        D = base[None, :] - up[ones]  # shape (n_ones, 2n-1)
        maxes = D.max(axis=1)
        j_idx = maxes.argmin()
        if maxes[j_idx] < best_c5:
            best_c5 = maxes[j_idx]
            best_i = i
            best_j = ones[j_idx]
    
    return best_i, best_j, best_c5

                best_i = int(i)
                best_j = int(ones[ji])
        if best_i < 0:
            break
        h[best_i] = 1.0
        h[best_j] = 0.0
    c5 = _eval(h)
    return h, c5

def _binary_search_sa(h, n, time_limit, rng):
    t0 = time.time()
    h = h.copy()
    c5 = _eval(h)
    T = 0.05
    T_min = 0.001
    for _ in range(10000):
        if time.time() - t0 > time_limit:
            break
        if T <= T_min:
            T = T_min
        i = int(rng.choice(np.where(h == 0)[0]))
        j = int(rng.choice(np.where(h == 1)[0]))
        h[i], h[j] = 1.0, 0.0
        c5_new = _eval(h)
        delta = c5_new - c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5 = c5_new
            T *= 0.995
        else:
            h[i], h[j] = 0.0, 1.0
        T = max(T, T_min)
    # Final greedy
    h, c5 = _binary_search(h, n, min(time_limit, 10), rng)
    return h, c5

def _random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _structured_binary(n, rng, mode):
    h = np.zeros(n)
    half = n // 2
    if mode == 0:
        h[0::2] = 1.0
    elif mode == 1:
        h[:half] = 1.0
    elif mode == 2:
        p = 313
        for i in range(n):
            if i < p and pow(i, (p - 1) // 2, p) == 1:
                h[i] = 1.0
    elif mode == 3:
        for i in range(n):
            if (i * (i + 1) // 2) % n < half:
                h[i] = 1.0
    else:
        idx = rng.choice(n, half, replace=False)
        h[idx] = 1.0
    oc = int(h.sum())
    if oc > half:
        ones = np.where(h == 1)[0]
        h[ones[:oc - half]] = 0.0
    elif oc < half:
        zeros = np.where(h == 0)[0]
        h[zeros[:half - oc]] = 1.0
    return h

def _resample_binary(h_old, n_old, n_new):
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(xn, xo, h_old)
    return _to_binary(h_interp, n_new)

def _continuous_refine(h0, n, time_limit):
    t0 = time.time()
    h = h0.astype(float).copy()
    c5 = _eval(h)
    alpha = 5000.0
    dx = 2.0 / n
    for step_size in [0.1, 0.05, 0.01, 0.003, 0.001]:
        if time.time() - t0 > time_limit:
            break
        for _it in range(2000):
            if time.time() - t0 > time_limit:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            m = c.max()
            w = np.exp(alpha * (c - m))
            w /= w.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-14:
                    continue
                s = k - (n - 1)
                wk = w[k] * dx
                if s >= 0:
                    g[:n - s] += wk * (1.0 - h[s:])
                    g[s:] -= wk * h[:n - s]
                else:
                    t = -s
                    g[t:] += wk * (1.0 - h[:n - t])
                    g[:n - t] -= wk * h[t:]
            h_new = _repair(h - step_size * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
            else:
                h = _repair(h - step_size * 0.5 * g, n)
                c5_new2 = _eval(h)
                if c5_new2 < c5:
                    c5 = c5_new2
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Binary local search from initial
    h_bin = _to_binary(h_init, n0)
    h_b, c5_b = _binary_search(h_bin, n0, 200, rng)
    if c5_b < best_c5:
        best_c5, best_h, best_n = c5_b, h_b.copy(), n0

    # Phase 2: Multi-restart binary search
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 150
    n_restarts = max(3, int(remaining // 15))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 150:
            break
        mode = r % 6
        h_r = _structured_binary(n0, rng, mode)
        tl = min(20, (budget_s - 150 - (time.time() - t0)) / max(1, n_restarts - r))
        h_r, c5_r = _binary_search(h_r, n0, tl, rng)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n0

    # Phase 3: Simulated annealing from best
    if time.time() - t0 < budget_s - 120:
        h_sa = _to_binary(best_h, best_n)
        h_sa, c5_sa = _binary_search_sa(h_sa, best_n, 60, rng)
        if c5_sa < best_c5:
            best_c5, best_h, best_n = c5_sa, h_sa.copy(), best_n

    # Phase 4: 2-swap escape from best
    if time.time() - t0 < budget_s - 100:
        h2 = best_h.copy()
        n2 = best_n
        c5_2 = _eval(h2)
        for _esc in range(20):
            if time.time() - t0 > budget_s - 100:
                break
            ones = np.where(h2 == 1)[0]
            zeros = np.where(h2 == 0)[0]
            for _try in range(50):
                i1, i2 = rng.choice(zeros, 2, replace=False)
                j1, j2 = rng.choice(ones, 2, replace=False)
                h2[i1] = h2[i2] = 1.0
                h2[j1] = h2[j2] = 0.0
                c5_t = _eval(h2)
                if c5_t < best_c5 - 1e-12:
                    best_c5, best_h = c5_t, h2.copy()
                    break
                h2[i1] = h2[i2] = 0.0
                h2[j1] = h2[j2] = 1.0
            h2, c5_2 = _binary_search(h2, n2, 15, rng)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()

    # Phase 5: Continuous refinement
    if time.time() - t0 < budget_s - 40:
        h_cont = best_h.astype(float)
        h_cont, c5_c = _continuous_refine(h_cont, best_n, 50)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_cont

    # Phase 6: Try different n
    if time.time() - t0 < budget_s - 50:
        for n_try in [n0 - 10, n0 + 10, n0 - 20, n0 + 20, n0 - 4, n0 + 4]:
            if time.time() - t0 > budget_s - 50:
                break
            if n_try < 100 or n_try % 2 != 0:
                continue
            h_r = _resample_binary(best_h, best_n, n_try)
            tl = max(5, (budget_s - 50 - (time.time() - t0)) / 3)
            h_r, c5_r = _binary_search(h_r, n_try, tl, rng)
            if c5_r < best_c5:
                best_c5, best_h, best_n = c5_r, h_r.copy(), n_try
            h_r2 = _random_binary(n_try, rng)
            h_r2, c5_r2 = _binary_search(h_r2, n_try, tl / 2, rng)
            if c5_r2 < best_c5:
                best_c5, best_h, best_n = c5_r2, h_r2.copy(), n_try

    # Phase 7: Final binary polish + continuous
    if time.time() - t0 < budget_s - 10:
        h_fin = _to_binary(best_h, best_n)
        h_fin, c5_f = _binary_search(h_fin, best_n, 20, rng)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_fin
        if time.time() - t0 < budget_s - 3:
            h_c, c5_c = _continuous_refine(best_h, best_n, 5)
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
