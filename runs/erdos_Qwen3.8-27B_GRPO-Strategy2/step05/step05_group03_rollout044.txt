import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _smooth_max_val(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        return _smooth_max_val(c, alpha)
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(np.clip(r.x, 0, 1), tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _pair_adj(h, dx, rng, t_end, n_lags=6, n_pairs=6):
    n = len(h)
    tgt = n / 2.0
    best_c = _c5(h, dx)
    h = h.copy()
    step = 0.003
    stall = 0
    # Ensure sum is correct at start
    h = _repair(h, tgt)
    
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:n_lags]
        improved = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0 or s >= n:
                continue
            m = n - s
            # For the purposes of finding contributing pairs:
            # If ki >= n-1: pairs are (tp, tp+s), contribution h[tp+s]*(1-h[tp])
            # If ki < n-1: pairs are (tp, tp+s), contribution h[tp]*(1-h[tp+s])
            # In both cases, the pairs to consider are (tp, tp+s) for tp in [0, m-1]
            # and we want to reduce h[tp]*(1-h[tp+s]) or h[tp+s]*(1-h[tp])
            # Either way, making h[tp] and h[tp+s] more similar reduces the overlap
            
            contribs = h[s:s+m] * (1.0 - h[:m])  # one direction
            if ki < n - 1:
                contribs = h[:m] * (1.0 - h[s:s+m])  # other direction
            
            top_p = np.argsort(contribs)[::-1][:n_pairs]
            for tp in top_p:
                i1, i2 = int(tp), int(tp) + s
                old1, old2 = h[i1], h[i2]
                d = step * (2.0 * rng.random() - 1.0)
                h[i1] += d
                h[i2] -= d
                if h[i1] < 0.0 or h[i1] > 1.0 or h[i2] < 0.0 or h[i2] > 1.0:
                    h[i1], h[i2] = old1, old2
                    continue
                cn = _c5(h, dx)
                if cn < best_c:
                    best_c = cn
                    improved = True
                else:
                    h[i1], h[i2] = old1, old2
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 400:
                step *= 0.75
                stall = 0
                if step < 1e-7:
                    break
        else:
            if stall == 0:
                step = min(step * 1.01, 0.008)
            stall = 0
        # Periodic repair to prevent drift
        if stall % 200 == 0 and stall > 0:
            h = _repair(h, tgt)
    
    h = _repair(h, tgt)
    return h, _c5(h, dx)

def _fine_cd(h, dx, rng, t_end):

    tgt = n / 2.0
    best_c = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 0.0004, 0.00015, 0.00005, 0.00002, 0.000008]

            else:
