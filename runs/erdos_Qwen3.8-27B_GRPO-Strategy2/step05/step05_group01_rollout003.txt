            f_new = _smooth_obj(y_new, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = y_new + ((tk - 1.0) / t_new) * (y_new - h)
        y = _proj_sum(y, target)
        h = y_new
        f = f_new
        tk = t_new
        step = min(step * 1.05, step0)
    return _repair(h, n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _directed_search(h, c, rng, n_iter, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    top_lags = np.argsort(c)[-5:]
    accepted = 0
    for _ in range(n_iter):
        K = top_lags[rng.randint(len(top_lags))]
        sh = (n - 1) - K
        if sh >= 0:
            pairs = np.arange(sh, n)
        else:
            pairs = np.arange(0, n + sh)
        if len(pairs) == 0:
            continue
        idx = pairs[rng.randint(len(pairs))]
        if sh >= 0:
            i, j = idx, idx - sh
        else:
            i, j = idx, idx - sh
        d = 0.02 * rng.randn()
        if rng.rand() < 0.5:
            d = -d
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i], h[j] = ni, nj
            c = new_c
            c5c = new_c5
            accepted += 1
            if accepted % 100 == 0:
                top_lags = np.argsort(c)[-5:]
    return h, c, c5c

def _random_search(h, c, rng, n_iter, scale, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i], h[j] = ni, nj
            c = new_c
            c5c = new_c5
    return h, c, c5c

def _upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: FISTA at original resolution with increasing alpha
    for alpha in [5000, 20000, 100000, 300000]:
        if _el() > 120:
            break
        n_it = 800 if alpha < 50000 else 400
        step0 = 0.003 if alpha < 50000 else 0.001
        h_new = _fista(best_h, alpha, n_it, step0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: SLSQP refinement
    for alpha in [50000, 200000, 600000]:
        if _el() > budget_s - 600:
            break
        h_new = _slsqp(best_h, alpha, 4000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Upsample to 2x and optimize
    if _el() < budget_s - 500:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [10000, 100000, 500000]:
            if _el() > budget_s - 380:
                break
            h_hi = _slsqp(h_hi, alpha, 3000, 1e-19)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase 4: Directed local search (target top lags)
    if _el() < budget_s - 300:
        h_ds = best_h.copy()
        c_ds = _corr(h_ds)
        n_ds = len(h_ds)
        for _block in range(30):
            if _el() > budget_s - 300:
                break
            h_ds, c_ds, _ = _directed_search(h_ds, c_ds, rng, 5000, n_ds)
        h_ds = _repair(h_ds, n_ds)
        c5 = _eval(h_ds)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_ds.copy(), n_ds

    # Phase 5: Random pairwise search (coarse to fine)
    if _el() < budget_s - 250:
        h_rs = best_h.copy()
        c_rs = _corr(h_rs)
        n_rs = len(h_rs)
        for scale in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if _el() > budget_s - 250:
                break
            h_rs, c_rs, _ = _random_search(h_rs, c_rs, rng, 40000, scale, n_rs)
        h_rs = _repair(h_rs, n_rs)
        c5 = _eval(h_rs)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_rs.copy(), n_rs

    # Phase 6: SLSQP final polish
    if _el() < budget_s - 150:
        h_pol = _slsqp(best_h, 800000, 5000, 1e-20)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h = c5, h_pol.copy()

    # Phase 7: Ultra-fine search
    if _el() < budget_s - 100:
        h_fine = best_h.copy()
        c_fine = _corr(h_fine)
        n_fine = len(h_fine)
        for scale in [3e-4, 1e-4, 5e-5, 2e-5]:
            if _el() > budget_s - 100:
                break
            h_fine, c_fine, _ = _random_search(h_fine, c_fine, rng, 60000, scale, n_fine)
        h_fine = _repair(h_fine, n_fine)
        c5 = _eval(h_fine)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fine.copy(), n_fine

    # Phase 8: Multi-start exploration
    if _el() < budget_s - 180:
        nc = len(best_h)
        for _r in range(8):
            if _el() > budget_s - 180:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(4, max(5, nc // 6))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            else:
                # Rotate by random offset and mirror
                off = rng.randint(1, nc)
                hp = np.roll(best_h, off)
                if rng.rand() < 0.5:
                    hp = hp[::-1].copy()
                hp = _repair(hp, nc)
            h_ms = _fista(hp, 100000, 300, 0.002)
            h_ms = _slsqp(h_ms, 300000, 1500, 1e-19)
            c_ms = _corr(h_ms)
            h_ms, c_ms, _ = _random_search(h_ms, c_ms, rng, 20000, 0.003, nc)
            h_ms = _repair(h_ms, nc)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_ms.copy(), nc

    # Phase 9: Final directed + random polish
    if _el() < budget_s - 30:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        n_fin = len(h_fin)
        h_fin, c_fin, _ = _directed_search(h_fin, c_fin, rng, 20000, n_fin)
        for scale in [1e-4, 5e-5, 2e-5]:
            if _el() > budget_s - 30:
                break
            h_fin, c_fin, _ = _random_search(h_fin, c_fin, rng, 40000, scale, n_fin)
        h_fin = _repair(h_fin, n_fin)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin.copy(), n_fin

    # Phase 10: Tiny final nudge
    if _el() < budget_s - 2:
        h_t = best_h.copy()
        c_t = _corr(h_t)
        n_t = len(h_t)
        h_t, c_t, _ = _random_search(h_t, c_t, rng, 30000, 1e-5, n_t)
        h_t = _repair(h_t, n_t)
        c5 = _eval(h_t)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_t.copy(), n_t

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
