def _coord_derivative(h, p, n):
    d = np.zeros(2 * n - 1)
    mid = n - 1
    d[mid] = 1.0 - 2.0 * h[p]
    s = np.arange(1, n)
    
    # Positive shift s: index mid+s
    # h[p] as h[i+s] (first factor): i=p-s, need p>=s → contrib (1-h[p-s])

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cd_optimize(h, n, max_passes, seed, n_ternary=22):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = _eval(h)
    best_h = h.copy()
    mid = n - 1
    s_arr = np.arange(1, n)
    no_improve = 0
    
    for pit in range(max_passes):
        order = rng.permutation(n) if pit % 3 != 0 else np.arange(n)
        for p in order:
            hp = h[p]
            lo = 0.0 if hp < 1e-14 else -hp
            hi = 0.0 if hp > 1.0 - 1e-14 else 1.0 - hp
            if hi - lo < 1e-14:
                continue
            d = np.zeros(2 * n - 1)
            d[mid] = 1.0 - 2.0 * hp
            m1 = s_arr <= p
            d[mid + s_arr[m1]] += (1.0 - h[p - s_arr[m1]])
            m2 = (p + s_arr) < n
            d[mid + s_arr[m2]] -= h[p + s_arr[m2]]
            d[mid - s_arr[m2]] += (1.0 - h[p + s_arr[m2]])
            d[mid - s_arr[m1]] -= h[p - s_arr[m1]]
            a, b = lo, hi
            for _ in range(n_ternary):
                if b - a < 1e-12:
                    break
                t1 = a + (b - a) / 3.0
                t2 = b - (b - a) / 3.0
                v1 = c + t1 * d
                v1[mid] -= t1 * t1
                v2 = c + t2 * d
                v2[mid] -= t2 * t2
                if np.max(v1) < np.max(v2):
                    b = t2
                else:
                    a = t1
            delta = (a + b) * 0.5
            if abs(delta) < 1e-13:
                continue
            h[p] = np.clip(hp + delta, 0.0, 1.0)
            actual = h[p] - hp
            c += actual * d
            c[mid] -= actual * actual
        
        if (pit + 1) % 20 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
        if (pit + 1) % 30 == 0:
            h = _repair(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
        if (pit + 1) % 10 == 0:
            c5 = _eval(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 10
                if no_improve > 150:
                    break
    return best_h, best_c5

def _pair_hill_climb(h, c, n, rng, n_iter, scale):
    dx = 2.0 / n
    c5c = np.max(c) * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dv = scale * rng.randn()
        ni = h[i] + dv
        nj = h[j] - dv
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        mid = n - 1
        di = np.zeros(2 * n - 1)
        di[mid] = 1.0 - 2.0 * h[i]
        s = np.arange(1, n)
        m1 = s <= i
        di[mid + s[m1]] += 1.0 - h[i - s[m1]]
        m2 = (i + s) < n
        di[mid + s[m2]] -= h[i + s[m2]]
        di[mid - s[m2]] += 1.0 - h[i + s[m2]]
        di[mid - s[m1]] -= h[i - s[m1]]
        di *= dv
        dj = np.zeros(2 * n - 1)
        dj[mid] = 1.0 - 2.0 * h[j]
        m1j = s <= j
        dj[mid + s[m1j]] += 1.0 - h[j - s[m1j]]
        m2j = (j + s) < n
        dj[mid + s[m2j]] -= h[j + s[m2j]]
        dj[mid - s[m2j]] += 1.0 - h[j + s[m2j]]
        dj[mid - s[m1j]] -= h[j - s[m1j]]
        dj *= -dv
        nc = c + di + dj
        nc[mid] -= dv * dv
        if np.max(nc) * dx < c5c - 1e-15:
            h[i] = ni
            h[j] = nj
            c = nc
            c5c = np.max(nc) * dx
    return h, c, c5c

    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    h1, c5_1 = _cd_optimize(h0, n0, 500, seed + 1)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1, n0

    for n_try in [200, 350, 420, 560, 280]:
        if time.time() - t0 > budget_s - 350:
            break
        if n_try == best_n:
            continue
        pts_new = np.linspace(0, 2, n_try)
        pts_old = np.linspace(0, 2, best_n)
        h_init = np.interp(pts_new, pts_old, best_h)
        h_init = _repair(h_init, n_try)
        h_t, c5_t = _cd_optimize(h_init, n_try, 400, seed + n_try)
        if c5_t < best_c5:
            best_c5, best_h, best_n = c5_t, h_t, n_try

    for i in range(10):
        if time.time() - t0 > budget_s - 150:
            break
        strain = 0.01 + 0.015 * i
        h_ms = _repair(best_h + strain * rng.randn(best_n), best_n)
        h_m, c5_m = _cd_optimize(h_ms, best_n, 250, seed + 300 + i)
        if c5_m < best_c5:
            best_c5, best_h, best_n = c5_m, h_m, best_n

    if time.time() - t0 < budget_s - 100:
        h_ph = best_h.copy()
        c_ph = np.correlate(h_ph, 1.0 - h_ph, mode='full')
        for sc in [0.02, 0.01, 0.005, 0.002, 0.001]:
            if time.time() - t0 > budget_s - 100:
                break
            h_ph, c_ph, _ = _pair_hill_climb(h_ph, c_ph, best_n, rng, 20000, sc)
        h_ph = _repair(h_ph, best_n)
        c5_ph = _eval(h_ph)
        if c5_ph < best_c5:
            best_c5, best_h, best_n = c5_ph, h_ph, best_n

    if time.time() - t0 < budget_s - 20:
        h_f, c5_f = _cd_optimize(best_h, best_n, 400, seed + 999)
        if c5_f < best_c5:
            best_c5, best_h, best_n = c5_f, h_f, best_n

    if time.time() - t0 < budget_s - 5:
        h_fin = best_h.copy()
        c_fin = np.correlate(h_fin, 1.0 - h_fin, mode='full')
        for sc in [5e-4, 2e-4, 1e-4]:
            if time.time() - t0 > budget_s - 5:
                break
            h_fin, c_fin, _ = _pair_hill_climb(h_fin, c_fin, best_n, rng, 30000, sc)
        h_fin = _repair(h_fin, best_n)
        c5_fin = _eval(h_fin)
        if c5_fin < best_c5:
            best_c5, best_h, best_n = c5_fin, h_fin, best_n

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
