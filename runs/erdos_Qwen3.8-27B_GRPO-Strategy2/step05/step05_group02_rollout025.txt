def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_lag_grad(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_w = c.argmax()
    sh = (n - 1) - k_w
    dx = 2.0 / n
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx, k_w, c

def _proj_subgrad(h, n_steps, rng):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    best_local = h_cur.copy()
    best_c5 = c5_cur
    base_eta = 0.003
    for i in range(n_steps):
        frac = i / max(1, n_steps)
        eta = base_eta * (1.0 - 0.9 * frac)
        g, _, _ = _active_lag_grad(h_cur)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _repair(h_cur - eta * mult * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_local = h_cur.copy()
                break
        if not improved:
            if i % 200 == 199:
                h_cur = _repair(h_cur + 0.003 * rng.randn(n), n)
                c5_cur = _eval(h_cur)
    return best_local, best_c5

def _two_lag_subgrad(h, n_steps, rng):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    best_local = h_cur.copy()
    best_c5 = c5_cur
    for i in range(n_steps):
        frac = i / max(1, n_steps)
        eta = 0.002 * (1.0 - 0.8 * frac)
        g, _, c = _active_lag_grad(h_cur)
        k2 = np.argsort(c).min()
        if k2 != 0:
            g2, _, _ = _active_lag_grad(h_cur)
            c_sorted = np.sort(c)[::-1]
            if c_sorted[1] > 0.99 * c_sorted[0]:
                sh2 = (n - 1) - k2
                dx = 2.0 / n
                g2 = np.zeros(n)
                if sh2 >= 0:
                    i1 = np.arange(n - sh2)
                    g2[i1] += (1.0 - h_cur[i1 + sh2])
                    if sh2 > 0:
                        i2 = np.arange(sh2, n)
                        g2[i2] -= h_cur[i2 - sh2]
                else:
                    sp = -sh2
                    i1 = np.arange(sp, n)
                    g2[i1] += (1.0 - h_cur[i1 - sp])
                    i2 = np.arange(n - sp)
                    g2[i2] -= h_cur[i2 + sp]
                g = (g + g2 * dx) * 0.5
        h_new = _repair(h_cur - eta * g, n)
        c5_new = _eval(h_new)
        if c5_new < c5_cur - 1e-15:
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_new
                best_local = h_cur.copy()
        else:
            if i % 100 == 99:
                h_cur = _repair(h_cur + 0.002 * rng.randn(n), n)
                c5_cur = _eval(h_cur)
    return best_local, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP LSE at increasing alphas
    for alpha in [300, 1500, 8000, 30000, 80000]:
        if _tl() < 150:
            break
        h_new = _slsqp_opt(best_h, alpha, 2500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Projected subgradient on active lag
    if _tl() > 250:
        h_new, c5_new = _proj_subgrad(best_h, 6000, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: SLSQP polish after subgradient
    if _tl() > 200:
        for alpha in [50000, 200000]:
            if _tl() < 150:
                break
            h_new = _slsqp_opt(best_h, alpha, 2000, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Two-lag subgradient
    if _tl() > 200:
        h_new, c5_new = _two_lag_subgrad(best_h, 4000, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Multi-start with Fourier + structured perturbations
    nc = len(best_h)
    if _tl() > 150:
        for _r in range(35):
            if _tl() < 100:
                break
            strat = rng.randint(5)
            if strat == 0:
                f = rng.randint(1, nc // 3 + 1)
                amp = rng.uniform(0.02, 0.12) * (1 if rng.rand() < 0.5 else -1)
                phase = rng.rand() * 2 * np.pi
                t_arr = np.arange(nc)
                hp = _repair(best_h + amp * np.sin(2 * np.pi * f * t_arr / nc + phase), nc)
            elif strat == 1:
                hp = _repair(best_h + 0.03 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 5)):
                    bs = rng.randint(3, max(4, nc // 10))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 3:
                f1, f2 = rng.randint(1, nc//4+1), rng.randint(nc//4+1, nc//2+1)
                amp1, amp2 = rng.uniform(0.01, 0.06), rng.uniform(0.01, 0.06)
                t_arr = np.arange(nc)
                hp = _repair(best_h + amp1*np.sin(2*np.pi*f1*t_arr/nc) + amp2*np.cos(2*np.pi*f2*t_arr/nc), nc)
            else:
                hp = _repair(0.5 + 0.25 * np.sin(2*np.pi * rng.randint(1,5)*np.arange(nc)/nc + rng.rand()*6.28), nc)
                hp = _repair(hp + 0.02*(best_h - hp), nc)
            h_new = _slsqp_opt(hp, 30000, 1500, 1e-17)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Try 2x resolution
    if _tl() > 300:
        n1 = n0 * 2
        h_hi = np.zeros(n1)
        for i in range(n0):
            h_hi[i*2:(i+1)*2] = best_h[i]
        h_hi = _repair(h_hi, n1)
        for alpha in [5000, 30000]:
            if _tl() < 200:
                break
            h_hi = _slsqp_opt(h_hi, alpha, 2000, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _tl() > 150:
            h_hi = _repair(h_hi, n1)
            h_new, c5_new = _proj_subgrad(h_hi, 3000, rng)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n1

    # Phase 7: Fine pairwise search
    if _tl() > 30:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.002, 0.0007, 0.0002, 0.00007, 0.00003]:
            if _tl() < 25:
                break
            for _it in range(12000):
                if _tl() < 25:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase 8: Final SLSQP + subgradient polish
    if _tl() > 15:
        h = _slsqp_opt(best_h, 150000, 2000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if _tl() > 5:
        h_new, c5_new = _proj_subgrad(best_h, 5000, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
