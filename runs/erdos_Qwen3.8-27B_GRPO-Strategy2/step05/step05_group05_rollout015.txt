import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, sh, n):
    """Gradient of c[sh] w.r.t. h, where sh is the actual shift (-(n-1)..(n-1))."""
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _active_set_descent(h, max_iters, n_top, momentum_beta=0.9):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    v = np.zeros(n)  # momentum
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        # Top lags
        idx_top = np.argsort(c)[::-1][:n_top]
        cvals = c[idx_top]
        cmin = c.min()
        weights = np.maximum(cvals - cmin, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        # Gradient
        g = np.zeros(n)
        for w, k in zip(weights, idx_top):
            sh = (n - 1) - k
            g += w * _grad_lag(h, sh, n)
        g *= dx
        # Nesterov momentum
        v = momentum_beta * v - 0.5 * g  # step size 0.5
        h_new = _proj(h + v, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5c:
            h = h_new
            c5c = c5n
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                h_try = _proj(h - factor * g, n * 0.5)
                c5t = _eval(h_try)
                if c5t < c5c:
                    h = h_try
                    c5c = c5t
                    v = 0.0
                    break
            else:
                v *= 0.5
    return h, c5c

    return x.sum() - len(x) * 0.5

    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    dx_val = 2.0 / n
    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        mm = cc.max()
        ss = np.clip(alpha * (cc - mm), -500, 500)
        return (mm + np.log(np.sum(np.exp(ss))) / alpha) * dx_val
    def grad(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        mm = cc.max()
        ss = np.clip(alpha * (cc - mm), -500, 500)
        ww = np.exp(ss - ss.max())
        ww /= ww.sum()
        g = np.zeros(n)
        sh = np.arange(-(n-1), n)
        for wk, sk in zip(ww, sh):

            jp = np.arange(n) + sk
            jm = np.arange(n) - sk
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            idx_n = np.arange(n)
            g[idx_n[vp]] += wkv * (1.0 - x[jp[vp]])
            g[idx_n[vm]] -= wkv * x[jm[vm]]
        return g
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pairwise(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _proj(h, target):

    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _smooth_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    sh = np.arange(-(n - 1), n)
    idx = np.arange(n)
    for wk, sk in zip(w, sh):

            continue
        wv = wk * dx
        jp = idx + sk
        vp = (jp >= 0) & (jp < n)
        g[idx[vp]] += wv * (1.0 - x[jp[vp]])
        jm = idx - sk
        vm = (jm >= 0) & (jm < n)
        g[idx[vm]] -= wv * x[jm[vm]]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
