```python
                    diff += a

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]

    return x.sum() - len(x) * 0.5

    return np.ones(len(x))

        # g_j[k] = (1-h[j-sh])*[0<=j-sh<n] - h[j+sh]*[0<=j+sh<n]

        gj = 0.0
        if 0 <= i - sh < n:
            gi += 1.0 - h[i - sh]
        if 0 <= i + sh < n:
            gi -= h[i + sh]
        if 0 <= j - sh < n:
            gj += 1.0 - h[j - sh]
        if 0 <= j + sh < n:
            gj -= h[j + sh]
        Δg[k] = gi - gj
    return Δg
```

```python
def _grad_matrix(h, n):
    """Compute G[k, p] = dc[k]/dh[p] for all k, p."""
    G = np.zeros((2*n-1, n))
    for p in range(n):
        # G[k, p] where k = 0..2n-2, sh = n-1-k
        # G[k,p] = (1-h[p-sh])*[0≤p-sh<n] - h[p+sh]*[0≤p+sh<n]
        # sh = n-1-k
        # p-sh = p-(n-1-k) = p-n+1+k
        # p+sh = p+n-1-k
        # First term valid: 0 ≤ p-n+1+k < n → n-1-p ≤ k ≤ 2n-2-p
        # Second term valid: 0 ≤ p+n-1-k < n → p ≤ k ≤ p+n-1
        
        k1_start = max(0, n - 1 - p)
        k1_end = min(2*n-2, 2*n-2-p)
        if k1_start <= k1_end:
            idx = np.arange(k1_start, k1_end+1)
            # p-sh = p-n+1+k
            h_idx = p - n + 1 + idx
            G[k1_start:k1_end+1, p] += 1.0 - h[h_idx]
        
        k2_start = max(0, p)
        k2_end = min(2*n-2, p+n-1)
        if k2_start <= k2_end:
            idx = np.arange(k2_start, k2_end+1)
            # p+sh = p+n-1-k
            h_idx = p + n - 1 - idx
            G[k2_start:k2_end+1, p] -= h[h_idx]
    return G
```

The optimal d* is where the "crossing" happens. We can find it by:

```python
def _optimal_d(c, Δg, d_min, d_max):
    """Find d that minimizes max_k (c[k] + d*Δg[k]) for d in [d_min, d_max]."""
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    a, b = d_min, d_max
    c_val = b - gr * (b - a)
    d_val = a + gr * (b - a)
    fc = (c + c_val * Δg).max()
    fd = (c + d_val * Δg).max()
    for _ in range(60):
        if fc < fd:
            b, d_val, fd = d_val, c_val, fc
            c_val = b - gr * (b - a)
            fc = (c + c_val * Δg).max()
        else:
            a, c_val, fc = d_val, d_val, fd
            d_val = a + gr * (b - a)
            fd = (c + d_val * Δg).max()
    return (a + b) / 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _build_G(h, n):
    """Build gradient matrix G[k,p] = dc[k]/dh[p]."""
    G = np.zeros((2 * n - 1, n))
    for p in range(n):
        k1s = max(0, n - 1 - p)
        k1e = min(2 * n - 2, 2 * n - 2 - p)
        if k1s <= k1e:
            idx = np.arange(k1s, k1e + 1)
            G[k1s:k1e + 1, p] += 1.0 - h[p - n + 1 + idx]
        k2s = max(0, p)
        k2e = min(2 * n - 2, p + n - 1)
        if k2s <= k2e:
            idx = np.arange(k2s, k2e + 1)
            G[k2s:k2e + 1, p] -= h[p + n - 1 - idx]
    return G

def _update_G(G, h, i, j, d, n):
    """Update G after h[i]+=d, h[j]-=d."""
    for k in range(2 * n - 1):
        sh = n - 1 - k
        p1 = i + sh
        if 0 <= p1 < n:
```

```python
    # Evaluate at boundaries

    # Find top 3 lags
    idx = np.argpartition(c, -3)[-3:]

        for b_idx in range(a+1, 3):
            ka, kb = idx[a], idx[b_idx]
            denom = dg[kb] - dg[ka]
            if abs(denom) > 1e-15:
                d_cross = (c[ka] - c[kb]) / denom
                if dmin <= d_cross <= dmax:

    kmax = np.argmax(c)

    for kb in top10:

            if dmin <= d_cross <= dmax:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _build_G(h, n):
    G = np.zeros((2 * n - 1, n))
    for p in range(n):
        k1s = max(0, n - 1 - p); k1e = min(2 * n - 2, 2 * n - 2 - p)
        if k1s <= k1e:
            idx = np.arange(k1s, k1e + 1)
            G[k1s:k1e + 1, p] += 1.0 - h[p - n + 1 + idx]
        k2s = max(0, p); k2e = min(2 * n - 2, p + n - 1)
        if k2s <= k2e:
            idx = np.arange(k2s, k2e + 1)
            G[k2s:k2e + 1, p] -= h[p + n - 1 - idx]
    return G

def _update_G(G, i, j, d, n):

    sh = (n - 1) - kr
    p1 = i + sh; m1 = (p1 >= 0) & (p1 < n)
    if m1.any(): G[kr[m1], p1[m1]] -= d
    p2 = i - sh; m2 = (p2 >= 0) & (p2 < n)
    if m2.any(): G[kr[m2], p2[m2]] -= d
    p3 = j + sh; m3 = (p3 >= 0) & (p3 < n)
    if m3.any(): G[kr[m3], p3[m3]] += d
    p4 = j - sh; m4 = (p4 >= 0) & (p4 < n)
    if m4.any(): G[kr[m4], p4[m4]] += d

def _opt_d(c, dg, dmin, dmax):
    if dmin >= dmax or dmax - dmin < 1e-14:
        return 0.0
    a, b = dmin, dmax
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a); d1 = a + gr * (b - a)
    fc = (c + c1 * dg).max(); fd = (c + d1 * dg).max()
    for _ in range(18):
        if fc < fd:
            b, d1, fd = d1, c1, fc
            c1 = b - gr * (b - a); fc = (c + c1 * dg).max()
        else:
            a, c1, fc = d1, d1, fd
            d1 = a + gr * (b - a); fd = (c + d1 * dg).max()
    return (a + b) / 2.0

def _pair_random(h, c, G, rng, nmoves, n, deadline):
    cur_max = c.max()
    for _ in range(nmoves):
        if time.time() > deadline:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        dmin = max(-h[i], h[j] - 1.0); dmax = min(1.0 - h[i], h[j])
        if dmin >= dmax - 1e-14:
            continue
        dg = G[:, i] - G[:, j]
        d = _opt_d(c, dg, dmin, dmax)
        if abs(d) < 1e-14:
            continue
        c_new = c + d * dg
        nm = c_new.max()
        if nm < cur_max - 1e-14:
            h[i] += d; h[j] -= d
            c = c_new; _update_G(G, i, j, d, n); cur_max = nm
    return h, c, cur_max

def _pair_sweep(h, c, G, n, deadline):
    cur_max = c.max()
    for i in range(n):
        if time.time() > deadline:
            break
        for j in range(i + 1, n):
            dmin = max(-h[i], h[j] - 1.0); dmax = min(1.0 - h[i], h[j])
            if dmin >= dmax - 1e-14:
                continue
            dg = G[:, i] - G[:, j]
            d = _opt_d(c, dg, dmin, dmax)
            if abs(d) < 1e-14:
                continue
            c_new = c + d * dg
            nm = c_new.max()
            if nm < cur_max - 1e-14:
                h[i] += d; h[j] -= d
                c = c_new; _update_G(G, i, j, d, n); cur_max = nm
    return h, c, cur_max

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP
    for alpha in [2000, 20000, 200000, 800000]:
        if _el() > 70:
            break
        h_new = _slsqp(best_h, alpha, 3000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Random optimal-amplitude pairwise
    if _el() < budget_s - 200:
        h_ps = best_h.copy(); c_ps = _corr(h_ps); G_ps = _build_G(h_ps, best_n)
        dl = min(t0 + budget_s - 180, t0 + 150)
        h_ps, c_ps, _ = _pair_random(h_ps, c_ps, G_ps, rng, 500000, best_n, dl)
        h_ps = _repair(h_ps, best_n); c5 = _eval(h_ps)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ps.copy()

    # Phase 3: SLSQP polish
    if _el() < budget_s - 120:
        h_pol = _slsqp(best_h, 1500000, 5000, 1e-20)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h = c5, h_pol.copy()

    # Phase 4: Systematic sweeps
    if _el() < budget_s - 80:
        h_sw = best_h.copy(); c_sw = _corr(h_sw); G_sw = _build_G(h_sw, best_n)
        dl = min(t0 + budget_s - 70, t0 + 80)
        for _sw in range(15):
            if time.time() > dl:
                break
            h_sw, c_sw, _ = _pair_sweep(h_sw, c_sw, G_sw, best_n, dl)
        h_sw = _repair(h_sw, best_n); c5 = _eval(h_sw)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sw.copy()

    # Phase 5: More random pairwise
    if _el() < budget_s - 50:
        h_r = best_h.copy(); c_r = _corr(h_r); G_r = _build_G(h_r, best_n)
        dl = min(t0 + budget_s - 40, t0 + 80)
        h_r, c_r, _ = _pair_random(h_r, c_r, G_r, rng, 600000, best_n, dl)
        h_r = _repair(h_r, best_n); c5 = _eval(h_r)
        if c5 < best_c5:
            best_c5, best_h = c5, h_r.copy()

    # Phase 6: Final SLSQP
    if _el() < budget_s - 20:
        h_f = _slsqp(best_h, 3000000, 5000, 1e-20)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    # Phase 7: Final sweep
    if _el() < budget_s - 5:
        h_s2 = best_h.copy(); c_s2 = _corr(h_s2); G_s2 = _build_G(h_s2, best_n)
        for _ in range(5):
            if time.time() > t0 + budget_s - 3:
                break
            h_s2, c_s2, _ = _pair_sweep(h_s2, c_s2, G_s2, best_n, t0 + budget_s - 3)
        h_s2 = _repair(h_s2, best_n); c5 = _eval(h_s2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```
