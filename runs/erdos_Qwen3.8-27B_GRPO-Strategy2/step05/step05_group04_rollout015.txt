import time

    """Project to {x: 0<=x<=1, sum(x)=target}"""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:

            hi = mid

def _grad_w(h, w):
    """d/dh [sum_k w[k]*c[k]] where c = correlate(h, 1-h, full)"""
    n = len(h)

    for k in range(2*n-1):

            g[:r] += wk * (1.0 - x[sh:sh+r])

            g[sp:] += wk * (1.0 - x[:n-sp])

    m = c.max()

    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - n * 0.5) ** 2

    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g
