    dx = 2.0 / n
    g = np.zeros(n)
    for idx_k in range(K):
        k = topk[idx_k]
        if w[idx_k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[idx_k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _projected_grad(h, alpha, K, steps, step0, momentum):
    n = len(h)
    f = _smooth_obj_topk(h, alpha, K)
    v = np.zeros(n)
    step = step0
    for _s in range(steps):
        g = _smooth_grad_topk(h, alpha, K)
        h_prev = h.copy()
        f_prev = f
        v = momentum * v + (1 - momentum) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj_topk(h_new, alpha, K)
        if f_new < f_prev:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 3)
        else:
            step *= 0.5
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = f_prev
    return h

def _slsqp_smooth(h, alpha, K, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _c_fun(x):
        return x.sum() - n * 0.5
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_smooth_obj_topk, h, jac=_smooth_grad_topk, args=(alpha, K),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _sa_block(h, n, t0, t1, n_iter, rng):

    best_h = h.copy()
    best_c = c_cur
    for i in range(n_iter):

        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-8)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _worst_lag_descent(h, max_rounds, time_limit):

    dx = 2.0 / n
    c5_cur = _eval(h)
    best_c5 = c5_cur

            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]

                break

            # Try a small random perturbation to escape
            h[ rng.randint(n) ] += 0.01 * rng.randn()
            h = _repair(h, n)

    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a
