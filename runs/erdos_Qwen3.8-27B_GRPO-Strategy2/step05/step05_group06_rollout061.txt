def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 0.9999
            if not m.any():
                break
            h[m] += d / m.sum()
        else:
            m = h > 1e-4
            if not m.any():
                break
            h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sg_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _so_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _so_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)

        if w[k] < 1e-15:

        sh = (n - 1) - k

            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _downsample(h, factor):
    n = len(h)
    new_n = n // factor
    h_new = np.zeros(new_n)
    for i in range(new_n):
        h_new[i] = h[i*factor:(i+1)*factor].mean()
    return h_new

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i*factor:(i+1)*factor] = h[i]
    return h_hi
