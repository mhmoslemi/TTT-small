def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        if s > target:

        else:

    # correlation at index k uses shift sh = (n-1) - k

        i2 = np.arange(n - sp)

            if time.time() - t0 > budget_s - 250:
                break
            h_hi, c5 = _topk_descent(h_hi, n1, K, 250, t0 + 60)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if time.time() - t0 < budget_s - 250:
            h_hi, c5 = _topk_descent(best_h, best_n, 5, 300, t0 + budget_s - 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase 5: SLSQP at current best resolution
    if time.time() - t0 < budget_s - 200:
        h_new = _slsqp(best_h, 100000, 1500, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 6: More top-K rounds
    nc = best_n
    if time.time() - t0 < budget_s - 180:
        for K in [3, 5, 8, 10, 15]:
            h_new, c5 = _topk_descent(best_h, nc, K, 250, t0 + budget_s - 150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            if time.time() - t0 > budget_s - 150:
                break

    # Phase 7: Multi-start with structured perturbations
    if time.time() - t0 < budget_s - 120:
        for r in range(12):
            if time.time() - t0 > budget_s - 100:
                break
            strat = r % 4
            if strat == 0:
                hp = best_h + 0.012 * np.random.randn(nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = max(5, nc // 12)
                st = np.random.randint(0, nc - bs)
                hp[st:st + bs] += 0.03 * np.random.randn(bs)
            elif strat == 2:
                f = np.random.randint(1, nc // 4 + 1)
                ta = np.arange(nc)
                hp = best_h + 0.005 * np.sin(2 * np.pi * f * ta / nc + np.random.rand() * 6.28)
            else:
                idx = np.random.choice(nc, size=nc // 5, replace=False)
                hp = best_h.copy()
                hp[idx] = 1.0 - best_h[idx]
                hp += 0.008 * np.random.randn(nc)
            hp = _repair(hp, nc)
            h_new, c5 = _topk_descent(hp, nc, 5, 200, t0 + budget_s - 100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            if time.time() - t0 > budget_s - 100:
                break

    # Phase 8: Fine pairwise search (medium scale)
    if time.time() - t0 < budget_s - 60:
        best_h, best_c5 = _fine_search(best_h, best_n, 10000,
                                       [0.003, 0.001, 0.0003], t0 + budget_s - 55)

    # Phase 9: Simulated annealing fine search
    if time.time() - t0 < budget_s - 40:
        best_h, best_c5 = _fine_search(best_h, best_n, 15000,
                                       [0.0005, 0.0001], t0 + budget_s - 35, anneal=True)

    # Phase 10: Ultra-fine deterministic
    if time.time() - t0 < budget_s - 5:
        h = best_h.copy()
        nc = best_n
        c5c = _eval(h)
        tlim = t0 + budget_s - 2
        for _ in range(40000):
            if time.time() > tlim:
                break
            i = np.random.randint(nc)
            j = np.random.randint(nc)
            if i == j:
                continue
            d = 1.5e-5 * np.random.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 11: Final SLSQP polish if time remains
    if time.time() - t0 < budget_s - 8:
        h_new = _slsqp(best_h, 200000, 1000, best_n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
