import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _binary_sa(h_init, n, t_end, seed):
    """Simulated annealing binary search."""
    rng = np.random.RandomState(seed)
    h = (h_init > 0.5).astype(np.float64)
    n_half = n // 2
    # Fix sum
    excess = int(h.sum()) - n_half
    if excess > 0:
        one_ids = np.where(h == 1)[0]
        for idx in rng.permutation(one_ids)[:excess]:
            h[idx] = 0
    elif excess < 0:
        zero_ids = np.where(h == 0)[0]
        for idx in rng.permutation(zero_ids)[:(-excess)]:
            h[idx] = 1
    
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(c) * 2.0 / n)
    best_c5 = cur_c5
    best_h = h.copy()
    T = 0.02
    T_min = 1e-5
    cooling = 0.998
    
    while time.time() < t_end and T > T_min:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        # Swap: h[i] and h[j] have different values
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c_new = np.correlate(h, 1.0 - h, mode='full')
        c5_new = float(np.max(c_new) * 2.0 / n)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = c5_new
            c = c_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        T *= cooling
    
    return best_h, best_c5

def _slsqp(h_init, n, t_end, alpha=50000, maxiter=3000):
    """SLSQP with soft-max."""
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s))))/alpha
    
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0/n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k]*dx
            if sh >= 0:
                i1 = np.arange(n-sh)
                g[i1] += wk*(1.0-x[i1+sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk*x[i2-sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk*(1.0-x[i1-sp])
                i2 = np.arange(n-sp)
                g[i2] -= wk*x[i2+sp]
        return g
    
    def sum_eq(x):
        return x.sum() - n*0.5
    
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun':sum_eq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter':maxiter, 'ftol':1e-18})
        return _repair(res.x, n), _eval(_repair(res.x, n))
    except:
        return _repair(h_init, n), _eval(_repair(h_init, n))

    t0 = time.time()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                diff = target - h.sum()
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _fix_binary(h, n, rng):
    h = h.astype(float).copy()
    nh = n // 2
    while int(h.sum()) > nh:
        idx = rng.randint(n)
        if h[idx] == 1: h[idx] = 0
    while int(h.sum()) < nh:
        idx = rng.randint(n)
        if h[idx] == 0: h[idx] = 1
    return h

def _binary_search(h_init, n, t_end, seed, sa_tempo=True):
    rng = np.random.RandomState(seed)
    h = _fix_binary(h_init, n, rng)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur = float(np.max(c) * 2.0 / n)
    best = cur
    best_h = h.copy()
    T = 0.015 if sa_tempo else 0.0
    Tmin = 1e-6
    cool = 0.9995
    stall = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c_new = np.correlate(h, 1.0 - h, mode='full')
        c5_new = float(np.max(c_new) * 2.0 / n)
        d = c5_new - cur
        if d < -1e-15:
            cur = c5_new
            c = c_new
            if c5_new < best:
                best = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        elif sa_tempo and (d <= 0 or rng.random() < np.exp(-d / max(T, 1e-8))):
            cur = c5_new
            c = c_new
            stall += 1
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            stall += 1
        T *= cool
        if stall > 50000:
            T = max(T, 0.005)
            stall = 0
    return best_h, best

def _slsqp(h_init, n, t_end, alpha=50000, maxiter=5000):
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                l = np.arange(n - sh)
                g[l] += wk * (1.0 - x[l + sh])
                if sh > 0:
                    l2 = np.arange(sh, n)
                    g[l2] -= wk * x[l2 - sh]
            else:
                sp = -sh
                l1 = np.arange(sp, n)
                g[l1] += wk * (1.0 - x[l1 - sp])
                l2 = np.arange(n - sp)
                g[l2] -= wk * x[l2 + sp]
        return g

    def _se(x):
        return x.sum() - n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': lambda x: np.ones(n)}]

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        hr = _repair(res.x, n)
        return hr, _eval(hr)
    except Exception:
        hr = _repair(h_init, n)
        return hr, _eval(hr)
