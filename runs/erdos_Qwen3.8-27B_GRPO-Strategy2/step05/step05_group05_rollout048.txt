def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _coord_grad_p(h, p):

    n = len(h)

    # h[p] appears as h[i+sh] when i+sh=p, i=p-sh: contributes -h[p-sh] for valid p-sh

        if 0 <= p - sh < n:
            coeff -= h[p - sh]
        g[k] = coeff
    return g

            else:

        v = np.max(c0 + d * cd)
        if v < best_val:
            best_val = v
            best_delta = cd
    return best_delta, best_val

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha)

def _smooth_grad_fft(h, alpha):
    """Fast gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    
    # Gradient: ∂f/∂h[p] = sum_k w[k] * [(1-h[p+sh])*[valid] - h[p-sh]*[valid]]
    # where sh = k - (n-1)
    # = sum_k w[k]*(1-h[p+k-(n-1)]) - sum_k w[k]*h[p-k+(n-1)]
    # First term: cross-correlation of w (length 2n-1) with (1-h) (length n)
    # Second term: cross-correlation of w with h
    
    N = 2 * (2 * n)  # FFT size
    W = np.zeros(N)
    # w is indexed 0 to 2n-2, representing lags from -(n-1) to (n-1)
    W[:2*n-1] = w
    
    one_minus_h = 1.0 - h
    gh = np.zeros(N)
    gh[:n] = one_minus_h
    hh = np.zeros(N)
    hh[:n] = h
    
    # Correlation: corr(w, b)[p] = sum_k w[k] * b[p + (n-1) - k... ]
    # Actually, numpy.correlate(w, b, 'full')[p] = sum_k w[k] * b[k + p - (len(b)-1)]
    # We want: sum_k w[k] * (1-h[p+k-(n-1)])
    # Let j = p + k - (n-1), so k = j - p + (n-1)
    # = sum_j (1-h[j]) * w[j-p+n-1] where 0≤j<n and 0≤j-p+n-1<2n-1
    # This is the correlation of w and (1-h) at lag (p - (n-1))... 
    
    # Let me just use direct computation for clarity, it's fast enough for n=280
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _active_set_descent(h, max_iters, n_top, rng):
    """Active set gradient descent with exact 1D solves on top lags."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= wsum
        # Compute combined gradient
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = k - (n - 1)
            wk = w * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        # Projected line search
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_p(h, p, sh_arr, n):
    v1 = (p + sh_arr >= 0) & (p + sh_arr < n)
    v2 = (p - sh_arr >= 0) & (p - sh_arr < n)
    gp = np.zeros(2 * n - 1)
    if v1.any():
        gp[v1] += 1.0 - h[p + sh_arr[v1]]
    if v2.any():
        gp[v2] -= h[p - sh_arr[v2]]
    return gp

def _exact_pair(h, c0, p, q, sh_arr, n, dx, n_bisect=35):
    gp = _grad_p(h, p, sh_arr, n)
    gq = _grad_p(h, q, sh_arr, n)
    d = (gp - gq) * dx
    dlo = max(-h[p], h[q] - 1.0)
    dhi = min(1.0 - h[p], h[q])
    if dlo >= dhi - 1e-14:
        return 0.0, np.inf
    lo, hi = dlo, dhi
    for _ in range(n_bisect):
        mid = 0.5 * (lo + hi)
        vals = c0 + d * mid
        m = vals.max()
        active = vals >= m - 1e-13
        sl = d[active]
        if sl.max() <= 1e-11:
            hi = mid
        elif sl.min() >= -1e-11:
            lo = mid
        else:
            if sl.max() > abs(sl.min()):
                hi = mid
            else:
                lo = mid
        if hi - lo < 1e-12:
            break
    best_d = 0.5 * (lo + hi)
    best_v = (c0 + d * best_d).max()
    for cd in [dlo, dhi]:
        v = (c0 + d * cd).max()
        if v < best_v:
            best_v = v; best_d = cd
    return best_d, best_v

def _pairwise_round(h, rng, n_pairs, n_bisect=35):
    n = len(h)
    dx = 2.0 / n
    sh_arr = np.arange(-(n - 1), n)
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = c0.max()
    n_imp = 0
    for _ in range(n_pairs):
        p = rng.randint(n); q = rng.randint(n)
        if p == q:
            continue
        best_d, best_v = _exact_pair(h, c0, p, q, sh_arr, n, dx, n_bisect)
        if best_v < best_c5 - 1e-14 and abs(best_d) > 1e-10:
            h[p] += best_d; h[q] -= best_d
            h[p] = min(max(h[p], 0.0), 1.0)
            h[q] = min(max(h[q], 0.0), 1.0)
            c0 = np.correlate(h, 1.0 - h, mode='full') * dx
            best_c5 = c0.max()
            n_imp += 1
    return h, best_c5, n_imp

def _structured_round(h, rng, n_rounds, n_bisect=35):
    n = len(h)
    dx = 2.0 / n
    sh_arr = np.arange(-(n - 1), n)
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = c0.max()
    for _ in range(n_rounds):
        top_k = np.argsort(c0)[::-1][:3]
        for k_star in top_k:
            sh_star = k_star - (n - 1)
            if sh_star >= 0:
                valid_i = np.arange(n - sh_star)
            else:
                valid_i = np.arange(-sh_star, n)
            if len(valid_i) < 2:
                valid_i = np.arange(n)
            rng.shuffle(valid_i)
            for i in valid_i[:min(15, len(valid_i))]:
                if sh_star >= 0:
                    p, q = i, i + sh_star
                else:
                    p, q = i - sh_star, i
                if p == q:
                    continue
                best_d, best_v = _exact_pair(h, c0, p, q, sh_arr, n, dx, n_bisect)
                if best_v < c0.max() - 1e-13 and abs(best_d) > 1e-10:
                    h[p] += best_d; h[q] -= best_d
                    h[p] = min(max(h[p], 0.0), 1.0)
                    h[q] = min(max(h[q], 0.0), 1.0)
                    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
        for _ in range(200):
            p = rng.randint(n); q = rng.randint(n)
            if p == q:

            if best_v < c0.max() - 1e-13 and abs(best_d) > 1e-10:
                h[p] += best_d; h[q] -= best_d
                h[p] = min(max(h[p], 0.0), 1.0)
                h[q] = min(max(h[q], 0.0), 1.0)
                c0 = np.correlate(h, 1.0 - h, mode='full') * dx
        best_c5 = c0.max()
    return h, best_c5

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _cons_fun(x, n):
    return x.sum() - n * 0.5

def _cons_jac(x, n):
    return np.ones(n)

def _active_descent(h, max_iters, n_top):
    n = len(h); dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]; cmean = cvals.mean()
        weights = np.maximum(cvals - cmean, 0.0)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights /= ws
        g = np.zeros(n)
        for w, k in zip(weights, idx):
            sh = k - (n - 1); wk = w * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    g[sh:] -= wk * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n-sp])
                g[:n-sp] -= wk * h[sp:]
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            if _eval(h_new) < c5c - 1e-15:
                h = h_new; c5c = _eval(h); improved = True; break
        if not improved:
            break
    return h, c5c
