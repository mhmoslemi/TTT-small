def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        else:

    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

    c = np.correlate(h, 1.0 - h, mode='full')

    s = np.clip(alpha * (c - m), -400, 400)

    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * dx * _get_gradients(h, k)
    return g

def _slsqp(h, alpha, maxiter):

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

    n = len(h)

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

def _grad_one(h, k):

    else:

        g += w[k] * _grad_one(h, k)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except:
        return _repair(h, n)

                dg = dhi - dlo
                best_d, best_v = 0.0, c5c
                for f in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9]:
                    d = dlo + f * dg
                    v = np.max(c + d * a)
                    if v < best_v: best_v = v; best_d = d
                best_d = np.clip(best_d, dlo, dhi)
                if abs(best_d) > 1e-14 and best_v < c5c - 1e-15:
                    h[ii] = np.clip(h[ii]+best_d, 0, 1)
                    h[jj] = np.clip(h[jj]-best_d, 0, 1)
                    c += best_d * a; c5c = max(c5c, c.max())
    h = _repair(h, n)
    return h, _eval(h)

def _momentum(h, n_iter, beta, t_end):
    n = len(h); c5c = _eval(h); hp = h.copy(); alpha = 0.01
    for _ in range(n_iter):
        if time.time() > t_end: break
        c = np.correlate(h, 1.0-h, mode='full') * 2.0/n
        m = c.max()
        act = c >= m - 1e-9
        na = max(act.sum(), 1)
        g = np.zeros(n)
        for k in range(2*n-1):
            if act[k]: g += _grad_one(h, k) / na
        h_new = _proj_sum(h - alpha*g + beta*(h - hp), n*0.5)
        c5n = _eval(h_new)
        if c5n < c5c:
            hp = h; h = h_new; c5c = c5n; alpha *= 0.9995
        else:
            hp = h; alpha *= 0.85
        if alpha < 1e-7: break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(h); best_n = n0
    t_soft = budget_s - 60

    # Phase A: SLSQP at n0
    for alpha in [3000, 30000, 300000, 2000000]:
        if _el() > 150: break
        hn = _slsqp(best_h, alpha, 3000)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase B: Fast pairwise at n0 (long run)
    if _el() < t_soft:
        te = min(t_end_for(_el, budget_s, 120), t_soft)
        n_it = int((te - _el()) * 400)
        hn, c5 = _fast_pairwise(best_h.copy(), n_it, rng, te)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase C: SLSQP again
    if _el() < t_soft - 20:
        for alpha in [5000000, 20000000]:
            if _el() > t_soft - 30: break
            hn = _slsqp(best_h, alpha, 2000)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase D: Block pairwise
    if _el() < t_soft - 30:
        te = t_soft - 30
        n_it = int((te - _el()) * 80)
        hn, c5 = _block_pairwise(best_h.copy(), n_it, rng, te, 4)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase E: Momentum subgradient
    if _el() < t_soft - 40:
        hn, c5 = _momentum(best_h.copy(), 8000, 0.9, t_soft - 40)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase F: Multi-start
    if _el() < t_soft - 80:
        nc = best_n
        for _r in range(25):
            if _el() > t_soft - 70: break
            st = rng.randint(5)
            if st == 0: hp = _repair(best_h + 0.015*rng.randn(nc), nc)
            elif st == 1: hp = _repair(1.0-best_h + 0.01*rng.randn(nc), nc)
            elif st == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2,4)):
                    bs = rng.randint(3, max(4,nc//7)); s2 = rng.randint(0,nc-bs)
                    hp[s2:s2+bs] += 0.04*rng.randn(bs)
                hp = _repair(hp, nc)
            elif st == 3:
                hp = best_h.copy(); t = np.arange(nc)
                for _f in range(4):
                    f = rng.randint(1, nc//3+1)
                    hp += 0.004*rng.randn()*np.sin(2*np.pi*f*t/nc)
                hp = _repair(hp, nc)
            else:
                hp = _repair(np.clip(best_h + 0.04*(rng.rand(nc)-0.5), 0,1), nc)
            hn = _slsqp(hp, 3000000, 800)
            c5 = _eval(hn)
            if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase G: Final fast pairwise (long)
    if _el() < budget_s - 30:
        te = budget_s - 30
        n_it = int((te - _el()) * 400)
        hn, c5 = _fast_pairwise(best_h.copy(), n_it, rng, te)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase H: Final SLSQP polish
    if _el() < budget_s - 10:
        hn = _slsqp(best_h, 50000000, 3000)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    # Phase I: Ultra-fine pairwise
    if _el() < budget_s - 3:
        hn, c5 = _fast_pairwise(best_h.copy(), 5000, rng, budget_s - 1)
        if c5 < best_c5: best_c5 = c5; best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)

def t_end_for(el, budget, margin):
    return budget - margin
