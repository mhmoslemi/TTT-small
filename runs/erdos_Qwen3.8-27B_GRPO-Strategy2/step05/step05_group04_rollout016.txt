def smooth_max_and_weights(c, alpha):
    """Compute smooth max and softmax weights"""
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()

def pairwise_search(h, n_iter, rng, scale=5e-5):
    """Fine pairwise adjustments"""
    n = len(h)
    target = n * 0.5
    c5 = eval_c5(h)
    for _ in range(n_iter):

        j = rng.randint(n)

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = eval_c5(h)
            if c5n < c5 - 1e-16:
                c5 = c5n
            else:

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.5, 1.5
        for _ in range(70):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_ls(h, alpha):
    n = len(h)
    c = corr_raw(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] * dx < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def nesterov_opt(h, alpha, steps, lr0, mu=0.9):
    n = len(h)
    target = n * 0.5
    h_c = h.copy()
    v = np.zeros(n)
    f_c = float(np.log(np.exp(np.clip(alpha*(corr_raw(h_c)*(2.0/n)-corr_raw(h_c)*(2.0/n).max()),-500,500)).sum())/alpha+corr_raw(h_c)*(2.0/n).max())
    lr = lr0
    best = h_c.copy()
    best_f = f_c
    for s in range(steps):
        g = gradient_ls(h_c, alpha)
        h_save = h_c.copy()
        v = mu * v + (1.0 - mu) * g
        improved = False
        for _t in range(6):
            h_try = proj_sum(h_save - lr * v, target)
            c_try = corr_raw(h_try) * (2.0 / n)
            f_try = float(np.log(np.exp(np.clip(alpha*(c_try-c_try.max()),-500,500)).sum())/alpha+c_try.max())
            if f_try < f_c:
                h_c = h_try
                f_c = f_try
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
        if f_c < best_f:
            best_f = f_c
            best = h_c.copy()
        if s % 30 == 0:
            lr = min(lr * 1.08, lr0 * 4)
    return best

def worst_lag_descent(h, n_rounds):
    n = len(h)
    target = n * 0.5
    c5 = eval_c5(h)
    for r in range(n_rounds):
        c_r = corr_raw(h)
        order = np.argsort(c_r)[::-1]
        improved = False
        for k_w in order[:3]:
            if c_r[k_w] * (2.0/n) < c5 - 1e-10:
                break
            sh = (n - 1) - k_w
            g = np.zeros(n)
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
            g *= (2.0 / n)
            for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0003]:
                h_new = proj_sum(h - step * g, target)
                c5n = eval_c5(h_new)
                if c5n < c5 - 1e-15:
                    h = h_new
                    c5 = c5n
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def pairwise_search(h, n_iter, rng, scale=4e-5):
    n = len(h)
    target = n * 0.5
    c5 = eval_c5(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = eval_c5(h)
            if c5n < c5 - 1e-17:
                c5 = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5

def sa_targeted(h, n_iter, t0, t1, rng):
    n = len(h)
    target = n * 0.5
    c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        r = rng.rand()
        if r < 0.5:
            c_r = corr_raw(h)
            k_w = c_r.argmax()
            sh = (n - 1) - k_w
            if sh >= 0:
                lo, hi = max(0, sh-2), min(n, n-sh+3)
            else:
                lo, hi = max(0, -sh-2), min(n, n+sh+3)
            if hi <= lo:
                lo, hi = 0, n
            idx = rng.randint(lo, hi)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + (0.015+0.04*t)*rng.randn(), 0, 1)
        elif r < 0.8:
            blk = rng.randint(2, max(3, n // 12))
            st = rng.randint(0, n - blk)
            h_new = h.copy()
            h_new[st:st+blk] += (0.01+0.04*t) * rng.randn(blk)
            h_new = np.clip(h_new, 0, 1)
        else:
            idx = rng.randint(n)
            h_new = h.copy()
            h_new[idx] = 1.0 - h_new[idx]
        h_new = repair(h_new, n)
        c5n = eval_c5(h_new)
        if c5n < c5 or rng.rand() < np.exp(-(c5n - c5) / max(t, 1e-8)):
            h = h_new
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def slsqp_run(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        c = corr_raw(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return float(np.log(np.exp(s).sum()) / alpha + m)
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(obj, h, jac=lambda x: gradient_ls(x, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def binary_local_search(h_bin, n_iter, rng):
    n = len(h_bin)
    c5 = eval_c5(h_bin)
    ones = np.where(h_bin > 0.5)[0]

        j = rng.choice(zeros)

        if c5n < c5:

            h_bin[i], h_bin[j] = 1.0, 0.0
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin <= 0.5)[0]
    return h_bin, c5
