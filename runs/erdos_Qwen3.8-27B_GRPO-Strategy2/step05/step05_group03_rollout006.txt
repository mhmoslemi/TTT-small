    grad = np.zeros(n)
    n1 = n - 1
    for ki in range(len(c)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        lag = ki - n1
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            t = abs(lag)
            m = n - t
            if lag > 0:
                grad[t:] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[t:]
            else:
                grad[:m] += wk * (1.0 - h[t:])
                grad[t:] -= wk * h[:m]
    grad -= grad.mean()
    return grad

def _lgd(h, dx, alpha, t_end, lr0=0.008, mom=0.85):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    v = np.zeros(n)
    lr = lr0
    for _ in range(2000000):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        v = mom * v + g
        h_new = _repair(np.clip(h - lr * v, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new; h = h_new; lr = min(lr * 1.008, 0.04)
        else:
            lr *= 0.82; v *= 0.4
            if lr < 1e-9:
                break
    return h, best

def _frank_wolfe(h, dx, alpha, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    for it in range(50000):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        idx = np.argsort(g)
        s = np.zeros(n)
        k_int = int(tgt)
        s[idx[:k_int]] = 1.0
        if tgt != k_int:
            s[idx[k_int]] = tgt - k_int
        gamma = 2.0 / (it + 2.0)
        h_new = _repair((1 - gamma) * h + gamma * s, tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new; h = h_new
    return h, best

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    def eq_j(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_j}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _targeted(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:5]
        improved = False
        for ki in top:
            s = int(ki) - (n - 1)
            if s == 0:
                continue
            t = abs(s)
            m = n - t
            for _ in range(8):
                a = rng.randint(m)
                if s > 0:
                    i1, i2 = a + t, a
                else:
                    i1, i2 = a, a + t
                d = rng.uniform(0.002, 0.015)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; improved = True
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
        if not improved:
            for _ in range(6):
                i, j = rng.randint(n, size=2)
                if i == j:
                    continue
                d = rng.uniform(-0.01, 0.01)
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 4), replace=False)
        hp[idx] += 0.06 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.02 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    # Phase 1: Frank-Wolfe global exploration
    for a in [30, 100, 300]:
        if time.time() > t0 + 80:
            break
        h, c = _frank_wolfe(best_h, dx, a, t0 + 80)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: LGD with multiple alphas
    for a in [50, 200, 800, 3000]:
        if time.time() > t0 + 200:
            break
        h, c = _lgd(best_h, dx, a, t0 + 200, lr0=0.006, mom=0.88)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: SLSQP polishing
    for a in [2000, 10000, 50000]:
        if time.time() > t0 + 320:
            break
        h, c = _slsqp(best_h, dx, a, 350)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 4: Targeted local search
    best_h, best_c5 = _targeted(best_h, dx, rng, t0 + 450)

    # Phase 5: Multi-start diversification
    t_end5 = t0 + 620
    for r in range(16):
        if time.time() > t_end5 or budget_s - (time.time() - t0) < 200:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _lgd(hp, dx, 1000, min(t_end5 - 3, time.time() + 8), lr0=0.005)
        hp, c = _slsqp(hp, dx, 20000, 250)
        if c < best_c5:
            best_h, best_c5 = hp, c
            best_h, best_c5 = _targeted(best_h, dx, rng, min(t_end5 - 1, time.time() + 5))

    # Phase 6: Higher resolution
    t_end6 = t0 + 800
    if budget_s - (time.time() - t0) > 250:
        n2 = 2 * n
        dx2 = 2.0 / n2
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _lgd(h2, dx2, 500, min(t_end6 - 60, time.time() + 50), lr0=0.004)
        h2, c2 = _slsqp(h2, dx2, 10000, 300)
        h2, c2 = _targeted(h2, dx2, rng, t_end6)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 7: More SLSQP + LGD on best
    nb, dxb = best_n, 2.0 / best_n
    for a in [30000, 100000]:
        if time.time() > t0 + 920:
            break
        h, c = _slsqp(best_h, dxb, a, 300)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 8: Final targeted + fine search
    best_h, best_c5 = _targeted(best_h, dxb, rng, t0 + 1000)
    if budget_s - (time.time() - t0) > 60:
        best_h, best_c5 = _fine(best_h, dxb, rng, t0 + 1060)

    # Phase 9: Last SLSQP
    if budget_s - (time.time() - t0) > 20:
        h, c = _slsqp(best_h, dxb, 200000, 200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 10: Ultra-fine
    if budget_s - (time.time() - t0) > 5:
        best_h, best_c5 = _fine(best_h, dxb, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
