import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 5e-5:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _grad_topk(h, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    dx = 2.0 / n
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def _con_fun(x):
    return x.sum() - len(x) * 0.5

def _con_jac(x):
    return np.ones(len(x))

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smooth_obj, h0, jac=lambda x: _grad_smooth(x, alpha),
                       args=(alpha,), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _mom_pg(h, alpha, nsteps, step0, mu, topk_mode):
    n = len(h)
    f = _smooth_obj(h, alpha) if not topk_mode else _eval(h)
    v = np.zeros(n)

        improved = False
        for _ in range(20):
            ht = _proj_sum(h - lam * v, n * 0.5)
            fn = _smooth_obj(ht, alpha) if not topk_mode else _eval(ht)
            if fn < f:
                h, f = ht, fn
                improved = True
                break
            lam *= 0.5
        if not improved:
            h = _proj_sum(h - lam * v, n * 0.5)
            f = _smooth_obj(h, alpha) if not topk_mode else _eval(h)
        step = min(step * 1.02, step0 * 4.0)
        if it % 50 == 0 and step < step0 * 0.5:
            step = step0
    return h

def _greedy_pairs(h, n_pairs, n_sweeps, rng):
    n = len(h)
    c5c = _eval(h)
    for sw in range(n_sweeps):
        improved = False
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for delta in (-0.5, -0.1, -0.02, 0.02, 0.1, 0.5):
                ni, nj = h[i] + delta, h[j] - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < c5c:
                    c5c = cn
                    improved = True
                    break
                h[i], h[j] = h[i] - delta, h[j] + delta
        if not improved:
            break
    return h, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _structured_init(n, rng, kind):
    if kind == 0:
        return np.full(n, 0.5) + 0.05 * rng.randn(n)
    elif kind == 1:
        t = np.arange(n)
        return 0.5 + 0.3 * np.sin(2 * np.pi * rng.randint(2, n // 4 + 1) * t / n)
    elif kind == 2:
        h = (np.arange(n) % 2).astype(float) * 0.5
        return h + 0.05 * rng.randn(n)
    else:
        k = rng.randint(3, n // 6 + 1)
        centers = rng.choice(n, size=k, replace=False)
        sig = n / (2 * k) * 0.5
        h = np.zeros(n)
        for c in centers:
            d = np.minimum(np.abs(np.arange(n) - c), n - np.abs(np.arange(n) - c))
            h += np.exp(-d**2 / (2 * sig**2))
        return h / h.max() * 0.8 + 0.1
