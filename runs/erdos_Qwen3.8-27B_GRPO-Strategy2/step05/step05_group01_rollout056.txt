import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    km = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = km[k]
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            cnt = n - sp
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(cnt)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

        return h[i], h[j]

    I2i = ((i - km) >= 0) & ((i - km) < n)
    I1j = ((j + km) >= 0) & ((j + km) < n)
    I2j = ((j - km) >= 0) & ((j - km) < n)
    
    hi_km = np.zeros(2 * n - 1)
    if I1i.any():
        hi_km[I1i] = h[i + km[I1i]]
    hmi_km = np.zeros(2 * n - 1)
    if I2i.any():
        hmi_km[I2i] = h[i - km[I2i]]
    hj_km = np.zeros(2 * n - 1)
    if I1j.any():
        hj_km[I1j] = h[j + km[I1j]]
    hmj_km = np.zeros(2 * n - 1)
    if I2j.any():
        hmj_km[I2j] = h[j - km[I2j]]

    old_j = h[j] * (1 - hj_km) * I1j.astype(float) + hmj_km * (1 - h[j]) * I2j.astype(float)
    
    B = c - old_i - old_j + S * (1 - hj_km) * I1j.astype(float) + hmj_km * (1 - S) * I2j.astype(float)
    D = (1 - hi_km) * I1i.astype(float) - hmi_km * I2i.astype(float) - (1 - hj_km) * I1j.astype(float) + hmj_km * I2j.astype(float)
    
    t = (t_lo + t_hi) / 2.0

        if g > 1e-14:
            t_hi = t
        elif g < -1e-14:
            t_lo = t
        else:
            break
        t = (t_lo + t_hi) / 2.0
    return t, S - t
