import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha, mu, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    pen = mu * (h.sum() - n * 0.5) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + pen

def _smooth_grad(h, alpha, mu, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    g += 2.0 * mu * (h.sum() - n * 0.5)
    return g

    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, 0.0, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs(h, alpha, maxiter, ftol, mu):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, mu, n),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 40,
                                'm': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, iters, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(iters):

        if i == j:

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c:
                c5c = cn
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _anneal(h, iters, temp0, cool, rng):

    c5c = _eval(h)
    temp = temp0
    scale = 0.015
    for _ in range(iters):

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d

            cn = _eval(h)
            if cn < c5c or rng.rand() < np.exp(-(cn - c5c) / max(temp, 1e-12)):
                c5c = cn
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        temp *= cool
    return h, c5c

        if cn < c5b:

        else:
            hb[i], hb[j] = 1, 0
    return hb, c5b

def _mom_proj(h, alpha, nsteps, step0, mu, n):
    f = _smooth_obj(h, alpha, 0.0, n)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):

        lam = 1.0
