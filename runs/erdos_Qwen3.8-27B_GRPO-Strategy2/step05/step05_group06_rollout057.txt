def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _worst_grad(h, topk):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g / len(idx)

def _hc_worst(h, nsteps, step0, topk=6):
    n = len(h)
    c5 = _eval(h)
    step = step0
    for _ in range(nsteps):
        g = _worst_grad(h, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        ok = False
        for _ in range(30):
            hn = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5 - 1e-15:
                h, c5 = hn, c5n
                ok = True
                break
            step *= 0.5
        if not ok:
            step *= 0.5
            if step < 1e-14:
                break
        step = min(step * 1.03, step0)
    return h, c5

def _fine_pair(h, rng, ntr, scale):
    n = len(h)
    c5 = _eval(h)
    for _ in range(ntr):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5

def _anneal(h, rng, ntr, scale0):
    n = len(h)
    c5c = _eval(h)
    temp = scale0 * 0.15
    for _ in range(ntr):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        scale = scale0 * (0.3 + 0.7 * max(0, 1.0 - _ / ntr))
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            elif rng.random() < np.exp(-(c5n - c5c) / max(temp, 1e-12)):
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        temp *= 0.9998
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Progressive SLSQP at base resolution
    for alpha in [2000, 12000, 60000, 250000]:
        if _el() > 140:
            break
        h_new = _slsqp(best_h, alpha, 2500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Worst-lag targeted HC
    if _el() < budget_s - 500:
        h_new, c5 = _hc_worst(best_h, 2500, 0.005, 6)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Second SLSQP round with higher alpha
    if _el() < budget_s - 420:
        for alpha in [80000, 300000, 800000]:
            if _el() > budget_s - 380:
                break
            h_new = _slsqp(best_h, alpha, 2000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 4: Diverse multi-start
    nc = len(best_h)
    if _el() < budget_s - 200:
        n_starts = 45
        for r in range(n_starts):
            if _el() > budget_s - 200:
                break
            s = rng.randint(10)
            if s == 0:
                hp = _repair(best_h + 0.007 * rng.randn(nc), nc)
            elif s == 1:
                hp = _repair(best_h + 0.003 * rng.randn(nc), nc)
            elif s == 2:
                hp = best_h.copy()
                b = rng.randint(4, max(5, nc // 10))
                st = rng.randint(0, nc - b)
                hp[st:st+b] += 0.035 * rng.randn(b)
                hp = _repair(hp, nc)
            elif s == 3:
                hp = _repair(np.roll(best_h, rng.randint(1, 12)), nc)
            elif s == 4:
                hp = best_h.copy()
                t = np.arange(nc)
                for ff in rng.randint(2, nc // 4 + 1, size=3):
                    hp += 0.004 * rng.randn() * np.cos(2 * np.pi * ff * t / nc + rng.randn())
                hp = _repair(hp, nc)
            elif s == 5:
                hp = best_h.copy()
                for _ in range(6):
                    ii = rng.randint(nc - 1)
                    dd = 0.015 * rng.randn()
                    hp[ii] += dd
                    hp[ii + 1] -= dd
                hp = _repair(hp, nc)
            elif s == 6:
                hp = 0.75 * best_h + 0.25 * rng.rand(nc)
                hp = _repair(hp, nc)
            elif s == 7:
                hp = best_h.copy()
                m = rng.rand(nc) < 0.12
                hp[m] = 1.0 - hp[m]
                hp = _repair(hp, nc)
            elif s == 8:
                k = np.exp(-0.5 * (np.arange(-6, 7) / 2.5) ** 2)
                k /= k.sum()
                hp = np.convolve(best_h, k, 'same') + 0.005 * rng.randn(nc)
                hp = _repair(hp, nc)
            else:
                hp = best_h.copy()
                ii, jj = rng.randint(nc), rng.randint(nc)
                hp[ii], hp[jj] = hp[jj], hp[ii]
                hp = _repair(hp, nc)
            al = [30000, 100000, 400000][rng.randint(3)]
            h_new = _slsqp(hp, al, 500)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 5: Refined HC
    if _el() < budget_s - 120:
        h_new, c5 = _hc_worst(best_h, 3000, 0.0015, 5)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 50:
        h_new = _slsqp(best_h, 600000, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 7: Extended fineannealing - the key improvement phase
    if _el() < budget_s - 3:
        h = best_h.copy()
        c5c = _eval(h)
        scales = [1.5e-4, 6e-5, 2.5e-5, 1e-5, 5e-6]
        for sc in scales:
            if _el() > budget_s - 3:
                break
            time_per = max(30, (budget_s - 3 - _el()) / (len(scales) - scales.index(sc)))
            ntr = int(time_per / 0.00015)
            h, c5c = _anneal(h, rng, min(ntr, 80000), sc)
            if c5c < best_c5:
                best_c5, best_h = c5c, h.copy()
        # Pure greedy fine search
        if _el() < budget_s - 3:
            h = best_h.copy()
            for sc in [3e-5, 1e-5, 5e-6]:
                if _el() > budget_s - 3:
                    break
                h, c5c = _fine_pair(h, rng, 20000, sc)
                if c5c < best_c5:
                    best_c5, best_h = c5c, h.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
