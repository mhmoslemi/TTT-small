               (10000, 20, 50, 5000), (30000, 20, 50, 5000), (100000, 50, 100, 4000),
               (300000, 50, 100, 4000), (1000000, 100, 100, 3000)]
    for alpha, k, rho, mi in configs:
        if _el() > 280:
            break
        h_new = _lbfgs(best_h.copy(), alpha, k, rho, mi)
        c5 = _eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Active descent at n0
    if _el() < budget_s - 350:
        h_new, c5 = _active_descent(best_h.copy(), 4000, 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Multi-start exploration
    if _el() < budget_s - 200:
        n_cur = best_n
        for r in range(15):
            if _el() > budget_s - 150:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n_cur), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                for _ in range(rng.randint(2, 5)):
                    bs = rng.randint(3, max(4, n_cur // 8))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st+bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            elif strat == 3:
                hp = best_h.copy()
                t = np.arange(n_cur)
                for _ in range(4):
                    f = rng.randint(1, n_cur // 2 + 1)
                    hp += 0.005 * rng.randn() * np.sin(2 * np.pi * f * t / n_cur)
                hp = _repair(hp, n_cur)
            else:
                hp = _repair(np.clip(best_h + 0.06 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur)
            h_new = _lbfgs(hp, 500000, 20, 100, 3000)
            c5 = _eval_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur
            h_new, c5 = _active_descent(h_new, 500, 5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase 4: Upsample to 2x
    if _el() < budget_s - 300:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha, k, rho, mi in [(5000, 15, 50, 6000), (50000, 30, 100, 5000),
                                   (200000, 50, 100, 4000), (1000000, 100, 100, 3000)]:
            if _el() > budget_s - 250:
                break
            h_hi = _lbfgs(h_hi, alpha, k, rho, mi)
            c5 = _eval_c5(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _el() < budget_s - 100:
            h_new, c5 = _active_descent(best_h.copy(), 3000, 5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(best_h)

    # Phase 5: Pairwise refinement
    if _el() < budget_s - 80:
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if _el() > budget_s - 40:
                break
            h_new, c5 = _pairwise(best_h.copy(), 3000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Final L-BFGS-B polish
    if _el() < budget_s - 20:
        n_cur = best_n
        h_new = _lbfgs(best_h.copy(), 2000000, 10, 100, 5000)
        c5 = _eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = n_cur

    # Phase 7: Ultra-fine active descent
    if _el() < budget_s - 5:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 3, patience=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 8: Final micro pairwise
    if _el() < budget_s - 2:
        h_new, c5 = _pairwise(best_h.copy(), 2000, 5e-5, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, best_n)
