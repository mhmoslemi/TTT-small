def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a

                    diff += a

            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _structured_pair(h, max_rounds, rng):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_rounds):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        gi = _grad_component(h, 0, n)  # placeholder, we need per-shift
        # Compute a[k] = d(c_k)/d h_i - d(c_k)/d h_j for each k
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = 0.0
            cj = 0.0
            if sh >= 0:
                if 0 <= i < n - sh:
                    ci += (1.0 - h[i + sh])
                if 0 <= i - sh < n - sh:
                    ci -= h[i - sh]
                if 0 <= j < n - sh:
                    cj += (1.0 - h[j + sh])
                if 0 <= j - sh < n - sh:
                    cj -= h[j - sh]
            else:
                sp = -sh
                if sp <= i < n:
                    ci += (1.0 - h[i - sp])
                if 0 <= i + sp < n - sp:
                    ci -= h[i + sp]
                if sp <= j < n:
                    cj += (1.0 - h[j - sp])
                if 0 <= j + sp < n - sp:
                    cj -= h[j + sp]
            a[k] = ci - cj
        best_d = 0.0
        best_v = np.inf
        dgrid = np.linspace(dlo, dhi, 15)
        for d in dgrid:
            v = np.max(c0 + dx * d * a)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-12:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i] -= best_d
                h[j] += best_d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Reweighted optimization (iterative softmax weights)
    if _el() < budget_s - 500:
        h_new = _reweighted_opt(best_h.copy(), [5, 10, 20, 40, 80, 160], 1500, t0, budget_s - 450)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Active set descent (top-10, quadratic weights)
    if _el() < budget_s - 400:
        h_new, c5 = _active_descent(best_h.copy(), 4000, 10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: More reweighted with higher beta
    if _el() < budget_s - 350:
        h_new = _reweighted_opt(best_h.copy(), [80, 160, 320], 1000, t0, budget_s - 300)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 4: Structured pairwise
    if _el() < budget_s - 300:
        h_new, c5 = _structured_pair(best_h.copy(), 3000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 5: Active set with different n_top
    if _el() < budget_s - 280:
        for nt in [5, 15, 20]:
            h_new, c5 = _active_descent(best_h.copy(), 3000, nt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            if _el() > budget_s - 250:
                break

    # Phase 6: Multi-start exploration
    if _el() < budget_s - 250:
        for _r in range(12):
            if _el() > budget_s - 200:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(best_h + 0.04 * (rng.rand(n0) - 0.5), n0)
            elif strat == 2:
                hp = best_h.copy()
                t = np.arange(n0)
                for _f in range(2):
                    f = rng.randint(1, n0 // 2 + 1)
                    hp += 0.008 * rng.randn() * np.sin(2 * np.pi * f * t / n0)
                hp = _repair(hp, n0)
            elif strat == 3:
                hp = best_h.copy()
                for _b in range(3):
                    bs = rng.randint(4, max(5, n0 // 8))
                    st = rng.randint(0, n0 - bs)
                    hp[st:st + bs] += 0.03 * rng.randn(bs)
                hp = _repair(hp, n0)
            else:
                hp = _repair(0.5 + 0.3 * np.sin(2 * np.pi * rng.ranf() * np.arange(n0) / n0) + 0.01 * rng.randn(n0), n0)
            h_new = _reweighted_opt(hp, [40, 80], 800, t0, budget_s - 180)
            h_new, c5 = _active_descent(h_new, 1000, 8)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 7: Final reweighted polish
    if _el() < budget_s - 120:
        h_new = _reweighted_opt(best_h.copy(), [320, 640], 2000, t0, budget_s - 100)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 8: Structured pairwise final
    if _el() < budget_s - 80:
        h_new, c5 = _structured_pair(best_h.copy(), 3000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 9: Multi-scale pairwise
    if _el() < budget_s - 50:
        for scale in [0.003, 0.001, 0.0005, 0.0002, 1e-4, 5e-5, 2e-5, 1e-5]:
            if _el() > budget_s - 20:
                break
            h_new, c5 = _pairwise_refine(best_h.copy(), 3000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 10: Ultra-fine active set
    if _el() < budget_s - 5:
        h_new, c5 = _active_descent(best_h.copy(), 5000, 3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
