    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _fista(h0, alpha, n_iter, step0, deadline):
    n = len(h0)
    h = h0.copy(); h_prev = h0.copy(); t_cur = 1.0
    step = step0; best_val = _eval(h0); best_h = h0.copy()
    for i in range(n_iter):
        if time.time() > deadline: break
        t_new = (1 + np.sqrt(1 + 4 * t_cur * t_cur)) / 2
        y = h + ((t_cur - 1) / t_new) * (h - h_prev)
        y = np.clip(y, -0.5, 1.5)
        g = _grad_smooth(y, alpha)
        h_new = _proj(y - step * g, n * 0.5)
        f_new = _eval(h_new)
        if f_new < best_val:
            best_val = f_new; best_h = h_new.copy()
        h_prev = h; h = h_new; t_cur = t_new
        if f_new < best_val + 1e-12: step = min(step * 1.04, step0 * 3)
        else: step *= 0.6
    return best_h, best_val

def _slsqp(h, alpha, maxiter, ftol, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return x.sum() - n * 0.5
    def _cj(x): return np.ones(n)
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        return m + np.log(w.sum()) / alpha
    def _grad(x): return _grad_smooth(x, alpha)
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() < deadline: return _repair(res.x, n)
    except Exception: pass
    return _repair(h, n)

def _lag_targeted(h, n_rounds, deadline):
    n = len(h); c5 = _eval(h); dx = 2.0 / n
    for _ in range(n_rounds):
        if time.time() > deadline: break
        c = np.correlate(h, 1.0 - h, mode='full')
        g = np.zeros(n)
        for k_w in np.argsort(c)[-2:]:
            sh = (n - 1) - k_w
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0: g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= dx
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-14:
                h = h_new; c5 = c5n; improved = True; break
        if not improved: break
    return h, c5

def _block_cd(h, n_sweeps, deadline):
    """Block coordinate descent on pairs/triples."""
    n = len(h); c5 = _eval(h)
    for _ in range(n_sweeps):
        if time.time() > deadline: break
        for i in range(0, n - 1, 2):
            if time.time() > deadline: break
            # Optimize h[i], h[i+1] keeping sum fixed
            s = h[i] + h[i+1]
            t = np.linspace(0, s, 11)
            best_t = t[0]; best_c = _eval(h)
            for tv in t[1:]:
                if tv < 0 or tv > 1 or s - tv < 0 or s - tv > 1: continue
                h[i], h[i+1] = tv, s - tv
                cv = _eval(h)
                if cv < best_c: best_c = cv; best_t = tv
            h[i] = best_t; h[i+1] = s - best_t
            # Repair sum
            h = _repair(h, n)
            if _eval(h) < c5: c5 = _eval(h)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h)
    
    # Phase 1: FISTA at n0
    for alpha in [500, 2000, 10000, 50000]:
        dl = min(t0 + 100, t0 + budget_s * 0.12)
        hf, cf = _fista(best_h, alpha, 400, 0.005, dl)
        if cf < best_c5: best_c5 = cf; best_h = hf.copy()
    
    # Phase 2: SLSQP
    if time.time() < t0 + budget_s * 0.22:
        hs = _slsqp(best_h, 50000, 2000, 1e-18, t0 + budget_s * 0.22)
        cs = _eval(hs)
        if cs < best_c5: best_c5 = cs; best_h = hs.copy()
    
    # Phase 3: Lag-targeted
    if time.time() < t0 + budget_s * 0.32:
        hl, cl = _lag_targeted(best_h, 500, t0 + budget_s * 0.32)
        if cl < best_c5: best_c5 = cl; best_h = hl.copy()
    
    # Phase 4: Upsample 2x
    n1 = n0 * 2
    h_hi = _repair(np.repeat(best_h, 2), n1)
    for alpha in [1000, 5000, 30000]:
        dl = min(t0 + budget_s * 0.50, t0 + 400)
        if time.time() > dl: break
        hh, ch = _fista(h_hi, alpha, 200, 0.003, dl)
        if ch < best_c5: best_c5 = ch; best_h = hh.copy(); h_hi = hh
    if time.time() < t0 + budget_s * 0.55:
        hl1, cl1 = _lag_targeted(h_hi, 300, t0 + budget_s * 0.55)
        if cl1 < best_c5: best_c5 = cl1; best_h = hl1.copy()
    
    # Phase 5: Multi-start
    nc = len(best_h)
    dl5 = t0 + budget_s * 0.78
    for trial in range(50):
        if time.time() > dl5: break
