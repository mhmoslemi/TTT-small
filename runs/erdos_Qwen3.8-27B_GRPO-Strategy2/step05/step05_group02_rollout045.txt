def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break

    else:

        cnt = n - sp
        if cnt > 0:
            g[sp:] += (1.0 - h[:cnt])
            g[:cnt] -= h[sp:sp+cnt]

    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / n

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        if cnt > 0:
            g[:cnt] += (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= h[:cnt]
    else:
        sp = -sh
        cnt = n - sp
        if cnt > 0:
            g[sp:] += (1.0 - h[:cnt])
            g[:cnt] -= h[sp:sp+cnt]
    return g

def _lse_obj(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / n

def _lse_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        g += w[k] * _grad_lag(h, k, n)
    return g * (2.0 / n)

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad_opt(h, n, t0, budget, rng, top_m=5):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = _eval(best)
    for it in range(100000):
        if time.time() - t0 > budget:
            break
        c = _corr(h)
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, k, n)
        g = g * dx / top_m
        gn = float(np.dot(g, g))
        if gn < 1e-24:
            break
        c5 = float(c.max() * dx)
        improved = False
        for alpha in [min(c5 / gn, 1.0), 0.5, 0.1, 0.05, 0.01, 0.005, 0.001]:
            h_new = _proj_to_sum(h - alpha * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < best_c5 - 1e-16:
                h = h_new
                best = h.copy()
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            h = _repair(h + 0.005 * rng.randn(n), n)
    return best, best_c5

def _pair_descent(h, n, t0, budget, rng, n_iter, top_m=8):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = _eval(best)
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi - 1e-12:
            continue
        c = _corr(h) * dx
        top_idx = np.argsort(c)[-top_m:]
        D = np.zeros(top_m)
        for ii, k in enumerate(top_idx):
            sh = (n - 1) - k
            Dk = 0.0
            if 0 <= i - sh < n: Dk += h[i - sh]
            if 0 <= i + sh < n: Dk -= h[i + sh]
            if 0 <= j + sh < n: Dk -= h[j + sh]
            if 0 <= j - sh < n: Dk += h[j - sh]
            D[ii] = Dk
        c_top = c[top_idx]
        c_max = c.max()
        lo2, hi2 = lo, hi
        for _ in range(40):
            m1 = lo2 + (hi2 - lo2) / 3.0
            m2 = hi2 - (hi2 - lo2) / 3.0
            f1 = max(c_max, float(np.max(c_top + D * m1)))
            f2 = max(c_max, float(np.max(c_top + D * m2)))
            if f1 <= f2:
                hi2 = m2
            else:
                lo2 = m1
        delta = (lo2 + hi2)
