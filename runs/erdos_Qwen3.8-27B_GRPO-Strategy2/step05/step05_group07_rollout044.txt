def _grad_lag(k, h, n):
    # Gradient of c_k = sum_i h[i+k-shift] * (1-h[i]) with respect to h
    # where shift = k - (n-1)
    shift = k - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        # c_k = sum_{i=0}^{n-1-shift} h[i+shift] * (1-h[i])
        idx = np.arange(n - shift)
        g[idx + shift] += (1.0 - h[idx])
        g[idx] -= h[idx + shift]
    else:
        s = -shift
        # c_k = sum_{i=s}^{n-1} h[i] * (1-h[i-s])
        idx = np.arange(s, n)
        g[idx] += (1.0 - h[idx - s])
        g[idx - s] -= h[idx]
    return g

def _proj_sum(h, target):
    # Project h onto {0 <= x <= 1, sum(x) = target}
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(k, h, n):
    shift = k - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        if shift >= n:
            return g
        idx = np.arange(n - shift)
        g[idx + shift] += (1.0 - h[idx])
        g[idx] -= h[idx + shift]
    else:
        s = -shift
        if s >= n:
            return g
        idx = np.arange(s, n)
        g[idx] += (1.0 - h[idx - s])
        g[idx - s] -= h[idx]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_subgrad(h, n_iter, topm, step0, momentum, max_time):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    f = _eval(x)
    best_f = f
    best_x = x.copy()
    step = step0
    for it in range(n_iter):
        if time.time() - _t0 > max_time:
            break
        if it > 0 and momentum > 0:
            alpha = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum * it)) / 2.0
            beta = (alpha - 1.0) / (1.0 + momentum * it) if momentum * it > 0 else 0.0
            y = x + beta * (x - y)
        c = _corrs(y) * (2.0 / n)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_lag(k, y, n)
        g /= (topm * n * 0.5)
        found = False
        for _bt in range(30):
            x_new = _proj_sum(y - step * g, target)
            f_new = _eval(x_new)
            if f_new < f - 1e-10 * step * np.dot(g, x_new - y):
                x = x_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            step *= 0.5
            x = _proj_sum(y - step * g, target)
            f = _eval(x)
        step = min(step * 1.05, step0)
        if f < best_f:
            best_f = f
            best_x = x.copy()
    return best_x, best_f

def _round_robin(h, n_lags, steps_per_lag, max_time):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _ in range(n_lags):
        if time.time() - _t0 > max_time:
            break
        c = _corrs(h)
        k_star = int(c.argmax())
        for _s in range(steps_per_lag):
            g = _grad_lag(k_star, h, n) * (2.0 / n)
            found = False
            for step in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_new = _proj_sum(h - step * g, target)
                f_new = _eval(h_new)
                if f_new < f - 1e-14:
                    h = h_new
                    f = f_new
                    found = True
                    break
            if not found:
                break
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _pairwise(h, n_iter, scale, seed_val, max_time):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > max_time:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _slsqp_quick(h, alpha, maxiter, max_time):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        c = _corrs(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

    def _cjac(x):
        c = _corrs(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            g += w[k] * _grad_lag(k, x, n)
        g *= (2.0 / n)
        return g

    def _cfun(x):
        return x.sum() - n * 0.5

    _cj = np.ones(n)
    try:
        res = minimize(_obj, h, jac=_cjac, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': lambda x: _cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() - _t0 < max_time:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _el():
        return time.time() - _t0

    # Phase A: Quick SLSQP at n0 (short)
    if _el() < budget_s - 60:
        hn = _slsqp_quick(best_h, 30000, 600, budget_s - 55)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase B: Projected subgradient at n0
    if _el() < budget_s - 50:
        for topm, sm, mu in [(5, 0.006, 0.3), (3, 0.004, 0.4), (8, 0.005, 0.3)]:
            if _el() > budget_s - 45:
                break
            hn, fc = _proj_subgrad(best_h, 15000, topm, sm, mu, budget_s - 42)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase C: Upsample to 2*n0
    n1 = n0 * 2
    h2 = _upsample(best_h, 2)
    if _el() < budget_s - 100:
        best_h = h2.copy()
        best_n = n1
        best_c5 = _eval(best_h)

    # Phase D: Heavy projected subgradient at n1
    if _el() < budget_s - 80:
        for topm, sm, mu, niter in [(5, 0.005, 0.3, 50000), (3, 0.003, 0.4, 40000),
                                     (8, 0.004, 0.3, 30000), (12, 0.003, 0.2, 20000)]:
            if _el() > budget_s - 75:
                break
            hn, fc = _proj_subgrad(best_h, niter, topm, sm, mu, budget_s - 72)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase E: Round-robin on individual lags at n1
    if _el() < budget_s - 80:
        for _rr in range(6):
            if _el() > budget_s - 75:
                break
            hn, fc = _round_robin(best_h, 50, 10, budget_s - 72)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase F: Upsample to 4*n0 if time
    n2 = n0 * 4
    if _el() < budget_s - 60:
        h4 = _upsample(best_h, 2)
        hn, fc = _proj_subgrad(h4, 20000, 5, 0.004, 0.3, budget_s - 55)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()
            best_n = n2
        else:
            hn, fc = _proj_subgrad(h4, 15000, 3, 0.003, 0.4, budget_s - 50)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()
                best_n = n2

    # Phase G: SLSQP polish at best resolution
    nc = len(best_h)
    if _el() < budget_s - 60:
        hn = _slsqp_quick(best_h, 100000, 1000, budget_s - 55)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase H: More subgradient rounds
    if _el() < budget_s - 40:
        for topm, sm in [(5, 0.003), (3, 0.002), (10, 0.002)]:
            if _el() > budget_s - 35:
                break
            hn, fc = _proj_subgrad(best_h, 20000, topm, sm, 0.3, budget_s - 32)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase I: Round-robin + active descent polish
    if _el() < budget_s - 30:
        hn, fc = _round_robin(best_h, 60, 8, budget_s - 28)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()
        hn, fc = _proj_subgrad(best_h, 10000, 5, 0.002, 0.3, budget_s - 25)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase J: Multi-start escape
    nc = len(best_h)
    if _el() < budget_s - 30:
        for _r in range(6):
            if _el() > budget_s - 25:
                break
            hp = best_h + 0.005 * rng.randn(nc)
            hp = _repair(hp, nc)
            hn, fc = _proj_subgrad(hp, 8000, 5, 0.004, 0.3, budget_s - 22)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase K: Fine pairwise swaps
    for scale, nit in [(0.002, 12000), (0.0008, 15000), (0.0003, 15000), (0.0001, 10000)]:
        if _el() > budget_s - 8:
            break
        hn, fc = _pairwise(best_h, nit, scale, seed + int(scale * 1e7), budget_s - 6)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    # Phase L: Final round-robin polish
    if _el() < budget_s - 5:
        hn, fc = _round_robin(best_h, 30, 6, budget_s - 3)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
