In our case, a = h, b = 1-h, both of length n.
correlate(h, 1-h, 'full') has 2n-1 entries.
Entry k (0-indexed): sum_i h[i] * (1-h[i+k-(n-1)]) for valid i.

Let me define lag = k - (n-1). Then:
- If lag ≥ 0: sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
- If lag < 0: sum_{i=-lag}^{n-1} h[i] * (1-h[i+lag]) = sum_{i=0}^{n+lag-1} h[i-lag] * (1-h[i])

Let m = |lag|:
- If lag = m ≥ 0: sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
- If lag = -m < 0: sum_{i=m}^{n-1} h[i] * (1-h[i-m])

∂f_m/∂h[j] for lag = m ≥ 0:
= (1-h[j+m]) for j = 0,...,n-1-m  (from the h[j] term)
  - h[j-m] for j = m,...,n-1  (from the 1-h[j] term, when i+m = j, i.e., i = j-m)

∂f_{-m}/∂h[j] for lag = -m:
= (1-h[j-m]) for j = m,...,n-1  (from the h[j] term)

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    sm = s - s.max()
    w = np.exp(sm)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    h += diff / n
    return np.clip(h, 0.0, 1.0)

        v = momentum * v + (1 - momentum) * g

            h = h_new

    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lg_grad(x, alpha):
    return _grad(x, alpha)
```

But wait, I can precompute the A vectors more efficiently. For a given i:
- The "as first factor" contribution: A[k] += (1-h[i+lag]) where lag = k-(n-1), so i+lag = i+k-(n-1) must be in [0,n-1], i.e., k in [n-1-i, n-1-i+n-1] = [n-1-i, 2n-2-i].
  In this range, A[k] += 1 - h[i+k-(n-1)].

- The "as second factor" contribution: A[k] += -h[i-lag] = -h[i-k+(n-1)] where i-k+(n-1) must be in [0,n-1], i.e., k in [i, i+n-1].
  In this range, A[k] += -h[i-k+(n-1)].

I can compute this with numpy slicing:

```python
def _coord_A(h, i, n):
    A = np.zeros(2 * n - 1)
    # First factor: k from n-1-i to min(2n-2-i, 2n-2)
    k_start = n - 1 - i
    k_end = min(2 * n - 1 - i, 2 * n - 1)
    if k_start < k_end:
        idx = np.arange(k_start, k_end)
        j = i + idx - (n - 1)  # j = i + lag = i + k - (n-1)
        A[idx] += 1.0 - h[j]
    # Second factor: k from max(i, 0) to min(i+n-1, 2n-2)
    k_start2 = max(i, 0)
    k_end2 = min(i + n, 2 * n - 1)
    if k_start2 < k_end2:
        idx2 = np.arange(k_start2, k_end2)
        j2 = i - idx2 + (n - 1)  # j2 = i - lag = i - k + (n-1)
        A[idx2] += -h[j2]
    return A
```

This is O(n) per coordinate, vectorized.

```python
def _binary_local_search(h, rng, t_end, n_iter=100000):
    n = len(h)
    c5 = _eval(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for it in range(n_iter):
        if time.time() > t_end:
            break
        # Randomly pick a 1 and a 0, swap them
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5_new = _eval(h)
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            # Update ones and zeros
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i], h[j] = 1.0, 0.0
    return h, c5
```
