    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full')

def _block_optimize(h, p, B, n):
    """Optimize h[p:p+B] via LP, keeping sum fixed."""
    target_sum = h[p:p+B].sum()
    s = np.arange(-(n-1), n)
    full_c = _corr(h)
    positions = p + np.arange(B)
    
    idx1 = positions[None, :] + s[:, None]
    idx2 = positions[None, :] - s[:, None]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    safe_idx1 = np.clip(idx1, 0, n-1)
    safe_idx2 = np.clip(idx2, 0, n-1)
    
    M = np.where(valid1, 1.0 - h[safe_idx1], 0.0).astype(float)
    M += np.where(valid2, -h[safe_idx2], 0.0).astype(float)
    
    block_vals = h[p:p+B]
    A = full_c - M.dot(block_vals)
    
    # LP: min t s.t. M@x - t <= -A, sum(x)=target_sum, 0<=x<=1
    c_obj = np.zeros(B + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = -A
    A_eq = np.zeros((1, B+1))
    A_eq[0, :B] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * B + [(-1.0, 3.0*n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            h[p:p+B] = res.x[:B]
    except Exception:
        pass
    return h

def _pos_change(h, i, di, n):
    dc = np.zeros(2*n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _incr_pair(h, c, rng, n_iter, scale, n):

            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
    return h, c, c5c

def _smooth_obj(h, alpha):

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k

            g[sh:] -= w[k] * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5

        if s_val > target:

        else:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _block_lp(h, p, B, n):
    ts = h[p:p+B].sum()

    full_c = np.correlate(h, 1.0 - h, mode='full')
    pos = p + np.arange(B)
    i1 = pos[None, :] + s_arr[:, None]
    i2 = pos[None, :] - s_arr[:, None]
    v1 = (i1 >= 0) & (i1 < n)
    v2 = (i2 >= 0) & (i2 < n)
    M = np.zeros((2*n-1, B))
    if v1.any():
        M[v1] = 1.0 - h[np.clip(i1, 0, n-1)][v1]
    if v2.any():
        M[v2] += -h[np.clip(i2, 0, n-1)][v2]
    bv = h[p:p+B]
    A = full_c - M.dot(bv)
    c_obj = np.zeros(B+1); c_obj[-1] = 1
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = -A
    A_eq = np.zeros((1, B+1)); A_eq[0, :B] = 1

        if r.success:
            h[p:p+B] = np.clip(r.x[:B], 0, 1)
    except Exception:
        pass
    return h

    dc = np.zeros(2*n-1)

    dc[(n-1)-sh1] += di*(1.0-h[i:i+n-i])
    sh2 = np.arange(i+1)
    dc[(n-1)-sh2] -= di*h[i::-1][:i+1]
    sp = np.arange(1, n)
    k3 = (n-1)+sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di*(1.0-h[i-sp[m1]])
    m2 = sp < (n-i)
    if m2.any():
        dc[k3[m2]] -= di*h[i+sp[m2]]
    return dc

def _incr_pair(h, c, rng, n_iter, scale, n):

    c5c = c.max()*dx

        i = rng.randint(n); j = rng.randint(n)
