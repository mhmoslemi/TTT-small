    for _ in range(nsteps):

        v = mu * v - step * g
        lam = 1.0
        ok = False
        for _bt in range(25):
            h_try = _proj_sum(h - lam * v, n * 0.5)
            fn = _smooth_obj(h_try, alpha)
            if fn < f - 1e-6 * lam * np.dot(v, h_try - h):
                h, f = h_try, fn
                ok = True
                break
            lam *= 0.5
        if not ok:
            h = _proj_sum(h - lam * v, n * 0.5)
            f = _smooth_obj(h, alpha)
        step = min(step * 1.02, step0 * 4.0)
    return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase A: SLSQP continuation on base resolution
    for alpha in [2000, 20000, 200000, 2000000, 20000000]:
        if _el() > 200:
            break
        h_new = _slsqp(best_h, alpha, 3000, 1e-16)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase B: Momentum projected gradient
    if _el() < budget_s - 500:
        h_new = _mom_proj(best_h, 50000, 800, 0.002, 0.88, rng)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase C: Coordinate descent (many passes)
    if _el() < budget_s - 350:
        for _pass_block in range(5):
            if _el() > budget_s - 350:
                break
            best_h, _ = _coord_descent(best_h.copy(), 60, rng)
            c5 = _eval(best_h)
            if c5 < best_c5:
                best_c5 = c5

    # Phase D: SLSQP polish at very high alpha
    if _el() < budget_s - 250:
        h_new = _slsqp(best_h, 100000000, 3000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase E: Simulated annealing
    if _el() < budget_s - 200:
        h_sa, _ = _sim_anneal(best_h.copy(), 20000, rng, 0.003, 1e-7)
        h_sa = _repair(h_sa, n0)
        c5 = _eval(h_sa)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sa.copy()

    # Phase F: Upsample to 2x and optimize
    if _el() < budget_s - 300:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [5000, 50000, 500000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp(h_hi, alpha, 2000, 1e-16)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 200:
            for _pb in range(3):
                if _el() > budget_s - 200:
                    break
                best_h, _ = _coord_descent(best_h.copy(), 40, rng)
                c5 = _eval(best_h)
                if c5 < best_c5:
                    best_c5 = c5

    # Phase G: Multi-start diversification
    if _el() < budget_s - 150:
        nc = best_n
        for _r in range(15):
            if _el() > budget_s - 150:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, nc // 10))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] = np.clip(hp[st:st + bs] + 0.04 * rng.randn(bs), 0, 1)
                hp = _repair(hp, nc)
            elif strat == 2:
                shift = rng.randint(1, max(2, nc // 15))
                hp = _repair(np.roll(best_h, shift), nc)
            elif strat == 3:
                hp = best_h.copy()
                t_arr = np.arange(nc)
                for f in [2, 5, 11, 23]:
                    hp += 0.006 * rng.randn() * np.sin(2 * np.pi * f * t_arr / nc)
                hp = _repair(hp, nc)
            else:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(nc), nc)
            alpha_pick = [50000, 500000, 5000000][rng.randint(3)]
            h_new = _slsqp(hp, alpha_pick, 1500, 1e-16)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), nc
                # Quick coord descent restart
                if _el() < budget_s - 150:
                    best_h, _ = _coord_descent(best_h.copy(), 30, rng)
                    c5 = _eval(best_h)
                    if c5 < best_c5:
                        best_c5 = c5

    # Phase H: Final SLSQP polish (very tight)
    if _el() < budget_s - 15:
        nc = best_n
        h_new = _slsqp(best_h, 500000000, 3000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), nc
        # One more coord descent
        if _el() < budget_s - 5:
            best_h, _ = _coord_descent(best_h.copy(), 50, rng)
            c5 = _eval(best_h)
            if c5 < best_c5:
                best_c5 = c5

    # Phase I: Ultra-fine local search
    if _el() < budget_s - 2:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(20000):
            if _el() > budget_s - 2:
                break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 1e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
