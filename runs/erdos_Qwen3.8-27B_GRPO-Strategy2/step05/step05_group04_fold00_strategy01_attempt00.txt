                add_val = min(0.5, abs(delta)/len(to_fill))
                if (delta>0 and current_val+add_val<1.0) or (delta<0 and current_val-add_val>0):
                    h[i] = max(0.0, min(1.0, h[i] + add_val*(1 if delta>0 else -1)))
                    delta -= add_val if delta>0 else -add_val
                    if abs(delta) < 1e-10:
                        break
                        
    # Final clip and repair
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-5:
        # Gradient adjustment
        diff = target - h.sum()
        factor = diff / max(n, 1) * 1e2
        sort_idx = np.argsort(np.abs(h))
        h[sort_idx[:int(abs(diff))]] += factor * (-1 if diff<0 else 1)
        h = np.clip(h, 0.0, 1.0)
        h = repair(h, n, target)
        
    return h

def evaluate_solution(h):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    return np.max(c) * (2.0 / n)

def binary_anneal_step(h, n, alpha=0.1, delta=0.02):
    # Perturbation operators
    h_new = h.copy()
    n_lower, n_upper = 10, 30
    block_size = np.random.randint(n_lower, n_upper + 1)
    start_idx = np.random.randint(0, n - block_size)
    
    # More nuanced binary flip
    for i in range(block_size):
        h_new[start_idx+i] = 1.0 - h_new[start_idx+i]
        
    # Small continuous adjustments
    indices = np.random.choice(n, min(30, n), replace=False)
    for i in indices:
        h_new[i] += np.random.normal(0.0, delta)
        h_new[i] = np.clip(h_new[i], 0.0, 1.0)
        
    # Ensure repair
    h_new = repair(h_new, n, 0.5*n)
    return h_new, h_new

def calc_smooth_objective(h, alpha=1.0):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = np.max(c)
    
    # Stable penalty term
    if m < 1e-15:
        return m
    
    # Smoothed versions prevent local minima
    penalty = alpha * np.sum((c - m) ** 2)
    return m + penalty

def calc_smooth_gradient(h, alpha=1.0):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = np.max(c)
    
    if m < 1e-15:
        return np.zeros_like(h)
        
    # Get unique lags and weights
    lag_m = np.argmax(c)
    weights = np.zeros((2*n-1, n), dtype=float)
    
    # Construct gradient impact
    ws = np.diff(np.sign(c - m + 1e-5))  # Discontinuity points
    for k in range(2*n-1):
        if k < ns - 1:  # s is lag
            s = (n-1) - k
            if s > 0:
                w1 = np.exp(-(c[k] - m)**2 * alpha)
                pos_idx = s < n
                neg_idx = s >= n
                
                # Calculate gradient contributions
                grad_i = np.zeros_like(h)
                
                if np.any(pos_idx):
                    i1 = np.arange(n - s)
                    grad_i[i1] += w1[pos_idx] * (1.0 - h[i1+s])
                    
                if np.any(neg_idx):
                    i2 = np.arange(s, n)
                    grad_i[i2] -= w1[neg_idx] * h[i2-s]
                    
                return grad_i * 2.0/n
            else:
                # Small lags have minimal impact
                i_arr = np.arange(n - (-s))
                grad_i = (1.0 - h[i_arr]) * np.exp(-alpha * (c[k] - m)**2)
                return grad_i * 2.0/n
        
    return np.zeros_like(h)

def nesterov_step(h, alpha=1.0, max_iter=1000):
    n = len(h)
    beta = 0.95
    v = np.zeros_like(h)
    x = h.copy()
    f = lambda x: calc_smooth_objective(x, alpha)
    
    for it in range(max_iter):
        if it % 50 == 0 and it>0:
            print(f"Nesterov iter {it}: obj={f(x):.9f}")
            
        grad = calc_smooth_gradient(x, alpha)
        x_prev = x.copy()
        m_prev = f(x)
        grad_prev = grad
        
        # Adjust step size
        step = 0.9 / (1 + 0.2*it)
        
        # More adaptive projection
        x_new = x - step * grad
        x_new = repair(x_new, n, 0.5*n)
        
        f_new = f(x_new)
        if f_new < m_prev:
            x = x_new
        else:
            # Reduce learning rate
            step *= 0.5
            # Hybrid with exponential moving average
            x = (1 - beta) * x_new + beta * x
        
        if abs(f_new - m_prev) < 1e-10 or it > 1000:
            break
            
    return x

def binary_search_improved(h, n, target_sum=0.5*n, n_iter=500):
    h = np.array(h).copy()
    current_sum = h.sum()
    
    # Dynamic repair and transformation
    for _ in range(n_iter):
        # Local binary flipping
        if np.random.random() < 0.4:
            i = np.random.randint(n)
            h[i] = 1.0 - h[i]
            
        # Major structural changes
        elif np.random.random() < 0.3:
            # Flip random blocks
            block_size = np.random.randint(5, n//2 + 1)
            indices = np.arange(n)
            np.random.shuffle(indices)
            flip_points = indices[:block_size]
            h[flip_points] = 1.0 - h[flip_points]
            
        # Small additive changes
        else:
            indices = np.random.choice(n, min(10, n), replace=False)
            dx = 1.0 - 2.0 * np.random.random() * (current_sum != target_sum)
            h[indices] += dx
        
        # Repair to maintain constraints
        h = repair(h, n, target_sum)
            
        yield h.copy()

def gradient_driven_search(h, budget=300):
    n = len(h)
    best_h = h.copy()
    best_obj = calc_smooth_objective(h, 100.0)
    
    # Check all local worst lags
    lags = np.argmax(np.correlate(h, 1.0-h, mode='full'))
    
    # Targeted gradient descent on worst lags
    current_point = h.copy()
    eval_count = 0
    max_evals = int(budget / 0.05)
    
    for i in range(max_evals):
        if np.random.random() < 0.7:
            # Directional gradient descent
            cumulative_grad = np.zeros(n)
            
            # Prioritize worst lags
            if hasattr(best_h, 'from_gradient'):
                best_h = best_h.copy()
            grad = calc_smooth_gradient(best_h, 100.0)
            normalized_grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 1e-6 else grad
            
            # Create perturbed points
            candidate = current_point + 1e-4 * normalized_grad
            
        else:
            # Random sample around current point
            step = np.random.uniform(0.001, 0.01)
            candidate = current_point + step * (np.random.standard_normal(n) - 0.5)
            
        current_obj = calc_smooth_objective(candidate, 100.0)
        eval_count += 1
        
        if current_obj < best_obj - 1e-10 or np.random.random() < 0.7:
            best_h = candidate.copy()
            best_obj = current_obj
            
        if best_obj < 1e-5 * n or eval_count > max_evals/2:
            break
            
    return best_h, best_obj

def run_with_phases(seed=42, budget_s=1100, **kwargs):
    # Load parent if provided
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        h_init = np.array(kwargs['initial_h_values']).flatten()
        n_initial = len(h_init)
    else:
        n_initial = 280
        h_init = np.zeros(n_initial) + 0.5
        
    h = np.copy(h_init)
    best_h = h.copy()
    best_c5 = evaluate_solution(h)
    best_phase = 0
    
    start_time = time.time()
    remaining_time = lambda: budget_s - (time.time() - start_time)
    
    # Phase 0: Binary-repair and refinement (100 seconds)
    def repair_to_binary(h):
        h_bin = h.copy()
        sorted_vals = np.argsort(h)
        s = h_bin.sum()
        target = 0.5 * n_initial
        
        # Two-pass adjustment
        for idx in range(min(500, int(max(abs(s - target), 0)) + 1):
            if s < target:
                index = sorted_vals[idx]
                h_bin[index] = min(0.999, h_bin[index] + 0.5)
            else:
                index = sorted_vals[::-1][idx]
                h_bin[index] = max(0.001, h_bin[index] - 0.5)
                
        return h_bin
    
    # More effective repair mechanism
    h = repair(h, n_initial, 0.5*n_initial)
    best_h = repair(best_h, n_initial, 0.5*n_initial)
    best_c5 = evaluate_solution(best_h)
    
    # Phase 1: Binary annealing (300 seconds)
    for iter in range(100):
        if time.time() - start_time > budget_s - 5: break
        
        # SA parameters
        temperature = max(0.1, 0.9 - iter*0.005)
        h_new, _ = binary_anneal_step(h.copy(), n_initial, alpha=temperature)
        
        new_c5 = evaluate_solution(h_new)
        
        if new_c5 < best_c5 or np.random.random() < max(0.1, 1.0 - (iter/100)**0.5):
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = new_c5
            
        if iter % 10 == 0:
            print(f"Sparse Binary Phase: iter {iter}, best C5={best_c5:.6f}")
    
    # Phase 2: Continuous optimization (500 seconds)
    for alpha in [1000, 10000, 100000, 500000]:
        if time.time() - start_time > budget_s - 300: break
        
        try:
            # Optimized non-linear programming
            grad_func = lambda x: calc_smooth_objective(x, alpha)
            grad_jac = lambda x: calc_smooth_gradient(x, alpha)
            
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_initial/2})
            bounds = [(0, 1) for _ in range(n_initial)]
            
            x0 = h.copy()
            res = minimize(grad_func, x0, method='L-BFGS-B', jac=grad_jac,
                           bounds=bounds, constraints=cons, 
                           options={'maxcor': 10, 'disp': False})
            
            h = res.x
            best_h = repair(np.clip(h, 0, 1), n_initial, n_initial/2)
            best_c5 = evaluate_solution(h)
        except:
            best_h = nesterov_step(best_h, alpha, 500)
            best_c5 = evaluate_solution(best_h)
            
            # Cooldown
            for cooldown in range(100):
                if time.time() - start_time > budget_s - 30: break
                h_new = h.copy() + np.random.normal(0, 0.01, n_initial)
                h_new = repair(np.clip(h_new, 0, 1), n_initial, n_initial/2)
                new_c5 = evaluate_solution(h_new)
                if new_c5 < best_c5: best_h, best_c5 = h_new, new_c5
    
    # Phase 3: Multi-start local search (300 seconds)
    if time.time() - start_time < budget_s - 300:
        h_local = best_h.copy()
        obj_local = best_c5
        
        for start_no in range(20):
            if time.time() - start_time > budget_s - 60: break
            
            # Perturbation operation
            perturbation_type = np.random.randint(0, 5)
            if perturbation_type == 0:
                step_size = 1.0 / (1 + 10 * np.log(start_no+1))
                h_temp = h_local + step_size * (np.random.standard_normal(n_initial) - 0.5)
            elif perturbation_type == 1:
                h_temp = np.roll(h_local, np.random.randint(-10, 11), axis=-1)
                h_temp = repair(h_temp, n_initial, n_initial/2)
            elif perturbation_type == 2:
                h_temp = h_local.copy()
                block_idx = np.random.randint(5, n_initial//3 + 1)
                start = np.random.randint(0, n_initial - block_idx)
                h_temp[start:start+block_idx] = 1 - h_temp[start:start+block_idx]
                h_temp = repair(h_temp, n_initial, n_initial/2)
            elif perturbation_type == 3:
                h_temp = repair_to_binary(h_local)
                # Apply heavy regularization to binary representation
                s = h_temp.sum()
                delta = n_initial/2 - s
                indices = np.abs(np.sort(-h_temp))
                step = 0.01 * np.sign(delta)
                h_temp[indices[:abs(int(delta))]] += step if delta>0 else -step
                h_temp = repair(h_temp, n_initial, n_initial/2)
            else:
                h_temp = repair_to_binary(h_local)
                h_temp = h_temp.copy()
                s = h_temp.sum()
                indices = np.random.choice(n_initial, int(10)+1, replace=False)
                for i in sorted(indices):
                    h_temp[i] += 0.01 * np.random.randn()
                    h_temp[i] = np.clip(h_temp[i], 0, 1)
                h_temp = repair(h_temp, n_initial, n_initial/2)
                
            for iter in range(50):
                if time.time() - start_time > budget_s - 50: break
                h_temp_new = h_temp.copy() + 1e-2 * np.random.randn(n_initial)
                h_temp_new = repair(np.clip(h_temp_new, 0, 1), n_initial, n_initial/2)
                obj_temp = evaluate_solution(h_temp_new)
                if obj_temp < best_c5 or np.random.random() < np.exp((obj_temp - best_c5)/5):
                    h_temp = h_temp_new.copy()
                    obj_temp = obj_temp
                    
                if obj_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = obj_temp
            
            if evaluate_solution(h_temp) < best_c5:
                best_c5 = evaluate_solution(h_temp)
                best_h = h_temp.copy()
    
    # Phase 4: Gradient driven search for worst lags (200 seconds)
    if time.time() - start_time < budget_s - 200:
        for _ in range(2):
            cand, obj_val = gradient_driven_search(best_h.copy())
            
            if obj_val < best_c5:
                best_c5 = obj_val
                best_h = cand.copy()
            best_c5_new = evaluate_solution(best_h)
            
            if best_c5_new < best_c5:
                best_c5 = best_c5_new
                best_h = best_h.copy()
    
    # Phase 5: Final binary refinement (100 seconds)
    h_final = repair_to_binary(best_h)
    s_final = h_final.sum()
    if abs(s_final - n_initial/2) > 1e-4:
        delta = n_initial/2 - s_final
        indices = np.random.choice(n_initial, int(10)+1)
        step = 0.01 * np.sign(delta)
        h_final[indices[:abs(int(delta))]] += step
        h_final = repair(h_final, n_initial, n_initial/2)
        
    best_c5 = evaluate_solution(best_h)
    n_final = n_initial
    
    return (best_h / n_initial * 2.0 * (2.0/n_initial*best_c5)).clip(0, 1), best_c5, n_initial
