import time

def _repair(h, n):

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _proj(h, n):
    t = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t: lo = m
        else: hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)
