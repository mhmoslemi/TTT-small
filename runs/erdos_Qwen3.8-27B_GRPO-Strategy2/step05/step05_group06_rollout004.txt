                    diff -= a

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smgrad(x, alpha, n):
    h = np.clip(x, 0, 1)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            c0 = n - sh
            g[:c0] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:c0]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, tgt):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slq(h0, alpha, mi, n, te):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
          'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_smobj, h0, jac=_smgrad, args=(alpha, n),
                     method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h0, n)

def _topg(h, tk, n):
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:tk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            c0 = n - sh
            g[:c0] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:c0]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    return g * dx / tk

def _sg(h, n, ns, s0, tk, te):
    c5 = _eval(h)
    st = s0
    for _ in range(ns):
        if time.time() > te:
            break
        g = _topg(h, tk, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        ok = False
        for _ in range(30):
            hn = _proj(h - st * g, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5 - 1e-15:
                h, c5 = hn, c5n
                ok = True
                break
            st *= 0.5
        if not ok:
            st *= 0.3
            if st < 1e-14:
                break
        else:
            st = min(st * 1.03, s0)
    return h, c5

def _cd(h, n, ns, te, sd):
    rng = np.random.RandomState(sd)
    c5 = _eval(h)
    for _ in range(ns):
        if time.time() > te:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        for s in [0.01, 0.005, -0.005, -0.01, 0.02, -0.02, 0.003, -0.003]:
            hi, hj = h[i] + s, h[j] - s
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h[i], h[j] = hi, hj
                c5n = _eval(h)
                if c5n < c5:
                    c5 = c5n
                    break
                h[i], h[j] = hi - s, hj + s
    return h, c5

def _up(h, fac):
    n = len(h)
    hh = np.zeros(n * fac)
    for i in range(n):
        hh[i * fac:(i+1) * fac] = h[i]
    return hh

def _interp_up(h, fac):
    n = len(h)
    xo = np.arange(n, dtype=float)
    xn = np.arange(n * fac) / float(fac)
    return np.interp(xn, xo, h)
