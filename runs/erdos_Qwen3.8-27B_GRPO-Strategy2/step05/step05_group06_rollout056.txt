    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample_linear(h, factor):
    n = len(h)
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, n * factor)
    return np.interp(x_new, x_old, h)

def _mom_project(h, alpha, nsteps, step0, mu, deadline_fn):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        if deadline_fn():
            break
        g = _smooth_grad(h, alpha)
        v = mu * v - step * g
        lam = 1.0
        improved = False
        for _bt in range(20):
            h_try = _proj_to_sum(h - lam * v, n * 0.5)
            fn = _smooth_obj(h_try, alpha)
            if fn < f - 1e-5 * lam * np.dot(v, h_try - h):
                h, f = h_try, fn
                improved = True
                break
            lam *= 0.5
        if not improved:
            h = _proj_to_sum(h - lam * v, n * 0.5)
            f = _smooth_obj(h, alpha)
        step = min(step * 1.02, step0 * 4.0)
    return h

def _fine_search(h, nc, c5_best, rng, n_iter, max_scale, deadline_fn):
    c5c = _eval(h)
    for frac in [1.0, 0.4, 0.15, 0.05]:
        scale = max_scale * frac
        for _it in range(n_iter):
            if deadline_fn():
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        max_scale *= 0.15
    return h, c5c

def _periodic_init(n, seed):
    rng = np.random.RandomState(seed)
    t = np.arange(n)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * t / n) + 0.1 * np.sin(4 * np.pi * t / n + 0.5)
    h += 0.03 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _dl(limit):
        return _el() > budget_s - limit

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP sweep at base resolution
    for alpha in [2000, 10000, 50000, 200000, 500000, 1500000]:
        if _dl(550):
            break
        h_new = _slsqp_opt(best_h, alpha, 4000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Momentum projected gradient at base
    if not _dl(500):
        h_new = _mom_project(best_h, 80000, 800, 0.002, 0.88, lambda: _el() > budget_s - 500)
        h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 3: Multi-start with diverse inits at base res
    if not _dl(400):
        for r in range(8):
            if _dl(400):
                break
            if r == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n0), n0)
            elif r == 1:
                hp = _periodic_init(n0, seed + r * 7)
            elif r == 2:
                hp = best_h.copy()
                bs = max(4, n0 // 10)
                st = rng.randint(0, n0 - bs)
                hp[st:st + bs] += 0.06 * rng.randn(bs)
                hp = _repair(hp, n0)
            elif r == 3:
                hp = _repair(1.0 - best_h, n0)
            elif r == 4:
                freqs = rng.randint(1, n0 // 3 + 1, size=5)
                t_arr = np.arange(n0)
                hp = best_h.copy()
                for f in freqs:
                    hp += 0.01 * rng.randn() * np.sin(2 * np.pi * f * t_arr / n0)
                hp = _repair(hp, n0)
            else:
                shift = rng.randint(1, max(2, n0 // 15))
                hp = _repair(np.roll(best_h, shift), n0)
            al = [50000, 200000, 800000][r % 3]
            h_new = _slsqp_opt(hp, al, 1500, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 4: Upsample to 2x (linear) and optimize
    if not _dl(280):
        n1 = n0 * 2
        h1 = _repair(_upsample_linear(best_h, 2), n1)
        for alpha in [10000, 60000, 300000, 1000000]:
            if _dl(200):
                break
            h1 = _slsqp_opt(h1, alpha, 2500, 1e-19)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h1.copy(), n1
        if not _dl(160):
            h1 = _mom_project(best_h, 200000, 600, 0.001, 0.9, lambda: _el() > budget_s - 160)
            h1 = _repair(h1, n1)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h1.copy(), n1

    # Phase 5: Upsample to 4x
    if not _dl(180):
        n2 = n0 * 4
        h2 = _repair(_upsample_linear(best_h, 4), n2)
        for alpha in [20000, 100000, 500000]:
            if _dl(80):
                break
            h2 = _slsqp_opt(h2, alpha, 1500, 1e-19)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h2.copy(), n2

    # Phase 6: Fine random perturbation search
    if not _dl(40):
        nc = best_n
        h = best_h.copy()
        h, c5c = _fine_search(h, nc, best_c5, rng, 4000, 0.02, lambda: _dl(10))
        if c5c < best_c5:
            best_c5, best_h, best_n = c5c, h.copy(), nc

    # Phase 7: Final SLSQP polish
    if not _dl(5):
        h_final = _slsqp_opt(best_h, 2000000, 3000, 1e-20)
        c5 = _eval(h_final)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_final.copy(), len(h_final)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
