def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

            else:

def _build_lp_matrix(h0, n):

    dx = 2.0 / n
    A = np.zeros((2*n-1, n))

            # m as second index: 0 <= m <= n-1-p

    b = c0 - A @ h0
    return A, b, c0, dx

    if _el() < budget_s - 250:
        n1 = best_n * 2
        h_hi = _repair(np.repeat(best_h, 2), n1)
        h_new, c5 = _lp_solve(h_hi, n1, max_iter=250, damp=1.0)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new, n1
        if _el() < budget_s - 150:
            h_new, c5 = _lp_solve(best_h, best_n, max_iter=300, damp=0.6)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new, best_n

    # Phase 5: SLSQP with smooth max (different optimizer)
    for alpha in [10000, 50000, 200000]:
        if _el() > budget_s - 200:
            break
        h_new = _smooth_max_slsqp(best_h, alpha, 1500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new, best_n

    # Phase 6: Multi-start with LP
    nc = best_n
    if _el() < budget_s - 150:
        for trial in range(20):
            if _el() > budget_s - 100:
                break
            strat = trial % 5
            if strat == 0:
                hi = _repair(best_h + 0.03 * rng.randn(nc), nc)
            elif strat == 1:
                hi = _repair(1.0 - best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 2:
                hi = _step_init(nc, rng)
            elif strat == 3:
                hi = _repair(rng.rand(nc), nc)
            else:
                hi = best_h.copy()
                bs = rng.randint(10, max(11, nc // 5))
                st = rng.randint(0, nc - bs)
                hi[st:st + bs] += 0.1 * rng.randn(bs)
                hi = _repair(hi, nc)
            h_new, c5 = _lp_solve(hi, nc, max_iter=200, damp=1.0)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new, nc

    # Phase 7: Final subgradient + LP polish
    if _el() < budget_s - 30:
        h_new, c5 = _subgrad(best_h, 4000, 0.002, 3)
        h_new = _repair(h_new, best_n)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new, best_n

    if _el() < budget_s - 5:
        h_new, c5 = _lp_solve(best_h, best_n, max_iter=400, damp=1.0)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new, best_n

    # Final repair
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
