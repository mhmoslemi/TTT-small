                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_weighted(h, w):
    """Gradient of sum_k w_k * f_k(h) where f_k is the overlap at lag k."""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            i1 = np.arange(s, n)
            g[i1] += w[k] * (1.0 - h[i1 - s])
            if s > 0:
                i2 = np.arange(n - s)
                g[i2] -= w[k] * h[i2 + s]
        else:
            sp = -s
            i1 = np.arange(n - sp)
            g[i1] += w[k] * (1.0 - h[i1 + sp])
            i2 = np.arange(n - sp, n)
            g[i2] -= w[k] * h[i2 - sp]
    return g * (2.0 / n)

def _proj_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)
    c = _corrs(h)
    return np.sum(c**p)**(1.0/p) if p > 1 else c.max()

    n = len(h)

    #           = (sum c^p)^(1/p-1) * grad_weighted(h, c^(p-1))
    w = np.where(c > 0, c**(p-1), 0)
    scale = (np.sum(c**p))**(1.0/p - 1.0)
    return scale * _grad_weighted(h, w)
