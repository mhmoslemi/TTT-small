```python
def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())

            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _block_move_eval(h, block, new_vals, corr, n):
    """Evaluate change when setting h[block] = new_vals"""
    old_vals = h[block]
    delta = new_vals - old_vals
    if np.allclose(delta, 0):
        return corr, 0.0
    dc = np.zeros(2 * n - 1)
    for idx in range(len(block)):
        if abs(delta[idx]) > 1e-15:
            dc += _pos_change(h, block[idx], delta[idx], n)
    return corr + dc, dc.max()

def _sa_block(h, corr, rng, n_iters, T0, T1, block_size):
    """Simulated annealing with block moves"""
    n = len(h)
    dx = 2.0 / n
    c5c = corr.max() * dx
    T = T0
    best_local = h.copy()
    best_local_c5 = c5c
    best_local_corr = corr.copy()
    
    for it in range(n_iters):
        # Cool down
        T = T0 * (T1 / T0) ** (it / max(n_iters - 1, 1))
        
        # Pick a random block
        bs = rng.randint(1, block_size + 1)
        start = rng.randint(0, n - bs + 1)
        block = np.arange(start, start + bs)
        
        # Generate new values
        new_h = h.copy()
        perturbation = 0.1 * rng.randn(bs)
        new_h[block] = np.clip(h[block] + perturbation, 0, 1)
        
        # Evaluate
        new_corr = _corr(new_h)
        new_c5 = new_corr.max() * dx
        delta = new_c5 - c5c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = new_h
            corr = new_corr
            c5c = new_c5
            if c5c < best_local_c5:
                best_local = h.copy()
                best_local_c5 = c5c
                best_local_corr = corr.copy()
    
    return best_local, best_local_corr, best_local_c5

def _greedy_descend(h, corr, n, max_rounds=200):
    """Greedy coordinate descent: try each position, pick best single change"""
    dx = 2.0 / n
    c5c = corr.max() * dx
    target_sum = n * 0.5

def _proj_to_sum(h, target):
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    h[idx] += a
                    diff -= a
                else:

                    diff += a
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += w[k] * (1.0 - h[sh:sh + cnt])
            g[sh:] -= w[k] * h[:cnt]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n - sp])
            g[:n - sp] -= w[k] * h[sp:]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs_pen(h, alpha, penalty, maxiter):
    n = len(h)
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha + penalty * (x.sum() - n * 0.5) ** 2
    def grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += w[k] * (1.0 - x[sh:sh + cnt])
                g[sh:] -= w[k] * x[:cnt]
            else:
                sp = -sh
                g[sp:] += w[k] * (1.0 - x[:n - sp])
                g[:n - sp] -= w[k] * x[sp:]
        g += 2 * penalty * (x.sum() - n * 0.5)
        return g
    bnds = [(0.0, 1.0)] * n

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)

        return _repair(h, n)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    s1 = np.arange(0, n - i)
    k1 = (n - 1) - s1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    s2 = np.arange(0, i + 1)
    k2 = (n - 1) - s2

def _proj_to_sum(h, target):

            h_try = _repair(best_h + eps * rng.randn(n0), n0)
            h_try = _slsqp(h_try, 400000, 2000)
            c5_try = _eval(h_try)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

    # Phase 4: Simulated annealing (sum-preserving)
    if _el() < budget_s - 180:
        h_sa = best_h.copy()
        c_sa = _corr(h_sa)
        h_sa, c_sa, c5_sa = _sa_pair(h_sa, c_sa, rng, 20000, 0.008, 0.0004, 0.12, n0)
        h_sa = _repair(h_sa, n0)
        c5_try = _eval(h_sa)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_sa
        if _el() < budget_s - 180:
            h_sa2 = best_h.copy()
            c_sa2 = _corr(h_sa2)
            h_sa2, c_sa2, c5_sa2 = _sa_pair(h_sa2, c_sa2, rng, 15000, 0.003, 0.0002, 0.06, n0)
            h_sa2 = _repair(h_sa2, n0)
            c5_try = _eval(h_sa2)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_sa2

    # Phase 5: SLSQP after SA
    if _el() < budget_s - 120:
        h_try = _slsqp(best_h, 800000, 3000)
        c5_try = _eval(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 6: Projected gradient
    if _el() < budget_s - 80:
        h_try = _proj_grad(best_h, 300000, 200, 0.003)
        h_try = _repair(h_try, n0)
        c5_try = _eval(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 7: Incremental pairwise fine-tuning
    if _el() < budget_s - 50:
        h_ps = best_h.copy()
        c_ps = _corr(h_ps)
        for sc in [0.01, 0.004, 0.001, 0.0003, 0.0001]:
            if _el() > budget_s - 50:
                break
            h_ps, c_ps, _ = _incr_pair(h_ps, c_ps, rng, 35000, sc, n0)
        h_ps = _repair(h_ps, n0)
        c5_try = _eval(h_ps)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_ps

    # Phase 8: Final SLSQP polish
    if _el() < budget_s - 15:
        h_try = _slsqp(best_h, 1500000, 2500)
        c5_try = _eval(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 9: Ultra-fine incremental
    if _el() < budget_s - 3:
        h_f = best_h.copy()
        c_f = _corr(h_f)
        for sc in [5e-5, 2e-5, 8e-6]:
            if _el() > budget_s - 3:
                break
            h_f, c_f, _ = _incr_pair(h_f, c_f, rng, 25000, sc, n0)
        h_f = _repair(h_f, n0)
        c5_try = _eval(h_f)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_f

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
```
