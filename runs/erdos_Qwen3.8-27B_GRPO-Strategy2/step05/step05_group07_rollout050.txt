def _newton_smooth(h, alpha, n, target, max_iter=200):
    h = h.copy()
    f = _eval(h)
    
    for it in range(max_iter):
        c = _corrs_raw(h) * dx  # dx = 2/n
        # Smooth max
        c_shifted = c - c.max()
        w = np.exp(alpha * c_shifted)
        w /= w.sum()
        
        # Gradient
        g = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-15:
                continue
            g += w[k] * _grad_single(k, h) * dx
        
        # Check convergence
        if np.max(np.abs(g)) < 1e-10:
            break
        
        # Hessian (sparse part + rank corrections)
        # Sparse part: alpha * sum_k w_k * H_k
        # H_k for shift s has -1 on s-th off-diagonals
        H = np.zeros((n, n))
        for k in range(2*n - 1):
            if w[k] < 1e-10:
                continue
            s = k - (n - 1)
            if s == 0:
                H[np.arange(n), np.arange(n)] += alpha * w[k] * (-2.0)
            else:
                sp = abs(s)
                idx = np.arange(n - sp)
                H[idx, idx + sp] += alpha * w[k] * (-1.0)
                H[idx + sp, idx] += alpha * w[k] * (-1.0)
        
        # Rank correction: -alpha * sum_k (w_k - w_k^2) * g_k * g_k^T
        for k in range(2*n - 1):
            if w[k] < 1e-10:
                continue
            coef = -alpha * (w[k] - w[k]**2)
            if abs(coef) > 1e-15:
                gk = _grad_single(k, h) * dx
                H += coef * np.outer(gk, gk)
        
        # Newton step
        try:
            H += 1e-8 * np.eye(n)  # Regularization
            step = np.linalg.solve(H, -g)
        except:
            step = -g * 0.01
        
        # Line search with projection
        found = False
        step_scale = 1.0
        for _ in range(30):
            h_new = _proj_to_sum(h + step_scale * step, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-8 * step_scale * np.dot(g, step):
                # Check sufficient decrease
                h = h_new
                f = f_new
                found = True
                break
            step_scale *= 0.5
        
        if not found:
            break
    
    return h

import numpy as np
from scipy.interpolate import interp1d
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_single(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(min(s, n), n)
        g[i1] += (1.0 - h[i1 - s])
        if s > 0:
            i2 = np.arange(n - s)
            g[i2] -= h[i2 + s]
    else:
        sp = -s
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(n - sp, n)
        g[i2] -= h[i2 - sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corrs_raw(h) * dx
    c_shifted = c - c.max()
    w = np.exp(alpha * c_shifted)
    w_sum = w.sum()
    w /= w_sum
    obj = c.max() + np.log(w_sum) / alpha
    g = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_single(k, h) * dx
    return obj, g, w, c

def _hessian_smooth(h, w, alpha):
    n = len(h)
    dx = 2.0 / n
    H = np.zeros((n, n))
    # Sparse band part
    for k in range(2*n - 1):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        wk = alpha * w[k]
        if s == 0:
            H[np.arange(n), np.arange(n)] += wk * (-2.0)
        else:
            sp = abs(s)
            idx = np.arange(n - sp)
            H[idx, idx + sp] += wk * (-1.0)
            H[idx + sp, idx] += wk * (-1.0)
    # Rank correction
    for k in range(2*n - 1):
        if w[k] < 1e-10:
            continue
        coef = -alpha * (w[k] - w[k]*w[k]) * dx * dx
        if abs(coef) < 1e-15:
            continue
        gk = _grad_single(k, h)
        H += coef * np.outer(gk, gk)
    return H

def _newton_smooth(h, alpha, n, max_iter=300):
    target = n / 2.0
    h = h.copy()
    f = _eval(h)
    for it in range(max_iter):
        obj, g, w, c = _smooth_obj_and_grad(h, alpha)
        if np.max(np.abs(g)) < 1e-12:

        H = _hessian_smooth(h, w, alpha)
        # Regularize for PD

        if eig_min < 0:

        try:
            step = np.linalg.solve(H, -g)
        except:

                f = f_new
                found = True
                break
            scale *= 0.5
        if not found:
            break
    return h, f
