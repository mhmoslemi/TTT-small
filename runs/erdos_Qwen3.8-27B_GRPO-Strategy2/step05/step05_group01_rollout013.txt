    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, t0, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:

def _pairwise(h, c, c5, rng, n_iter, scale, n, deadline):
    dx = 2.0 / n
    accepted = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5 - 1e-15:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5 = new_c5
            accepted += 1
    return h, c, c5, accepted

def _sweep(h, n, n_grid, deadline):
    h = h.copy()
    c = _corr(h)
    c5 = c.max() * 2.0 / n
    xs = np.linspace(0.0, 1.0, n_grid)
    for _pass in range(3):
        for i in range(n):
            if time.time() > deadline:
                break
            best_x = h[i]
            best_val = c5
            for x in xs:
                di = x - h[i]
                if abs(di) < 1e-12:
                    continue
                dc = _pos_change(h, i, di, n)
                nv = (c + dc).max() * 2.0 / n
                if nv < best_val:
                    best_val = nv
                    best_x = x
            if best_x != h[i]:
                di = best_x - h[i]
                c = c + _pos_change(h, i, di, n)
                h[i] = best_x
                c5 = best_val
        h = _repair(h, n)
        c = _corr(h)
        c5 = c.max() * 2.0 / n
    return h, c5

    c5_ps = c_ps.max() * 2.0 / n0
    scales = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
    for sc in scales:
        if time.time() > deadline - 300:
            break
        n_it = max(20000, int(400000 / (1 + 100 * sc)))
        n_it = min(n_it, 500000)
        h_ps, c_ps, c5_ps, _ = _pairwise(h_ps, c_ps, c5_ps, rng, n_it, sc, n0, deadline - 300)
    h_ps = _repair(h_ps, n0)
    c5 = _eval(h_ps)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h_ps.copy(), n0

    # Phase 3: Sweep optimization at n0
    if time.time() < deadline - 280:
        h_sw, c5_sw = _sweep(best_h, n0, 41, deadline - 270)
        if c5_sw < best_c5:
            best_c5, best_h, best_n = c5_sw, h_sw.copy(), n0

    # Phase 4: More pairwise after sweep
    h_ps = best_h.copy()
    c_ps = _corr(h_ps)
    c5_ps = c_ps.max() * 2.0 / n0
    for sc in [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]:
        if time.time() > deadline - 250:
            break
        h_ps, c_ps, c5_ps, _ = _pairwise(h_ps, c_ps, c5_ps, rng, 200000, sc, n0, deadline - 250)
    h_ps = _repair(h_ps, n0)
    c5 = _eval(h_ps)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h_ps.copy(), n0

    # Phase 5: Upsample to 2x and optimize
    if time.time() < deadline - 150:
        n1 = n0 * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [50000, 200000, 500000]:
            if time.time() > deadline - 120:
                break
            h_hi = _slsqp(h_hi, alpha, 1000, t0, deadline - 120)
        h_hi, c5_hi = _sweep(h_hi, n1, 25, deadline - 80)
        h_hi = _repair(h_hi, n1)
        c5 = _eval(h_hi)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_hi, n1

    # Phase 6: Final fine pairwise at best resolution
    if time.time() < deadline - 30:
        nf = best_n
        h_f = best_h.copy()
        c_f = _corr(h_f)
        c5_f = c_f.max() * 2.0 / nf
        for sc in [0.001, 0.0003, 0.0001, 3e-5, 1e-5]:
            if time.time() > deadline - 30:
                break
            h_f, c_f, c5_f, _ = _pairwise(h_f, c_f, c5_f, rng, 150000, sc, nf, deadline - 30)
        h_f = _repair(h_f, nf)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    # Phase 7: Multi-start exploration
    if time.time() < deadline - 60:
        nc = best_n
        for _r in range(8):
            if time.time() > deadline - 60:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.03 * rng.randn(nc), nc)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 6)):
                    bs = rng.randint(3, max(4, nc // 6))
                    st = rng.randint(0, nc - bs)
                    hp[st:st + bs] += 0.08 * rng.randn(bs)
                hp = _repair(hp, nc)
            else:
                hp = _repair(0.5 + 0.3 * rng.randn(nc), nc)
            h_ms = _slsqp(hp, 300000, 800, t0, deadline - 55)
            c_ms = _corr(h_ms)
            c5_ms = c_ms.max() * 2.0 / nc
            h_ms, c_ms, c5_ms, _ = _pairwise(h_ms, c_ms, c5_ms, rng, 100000, 0.005, nc, deadline - 50)
            for sc in [0.001, 0.0001]:
                h_ms, c_ms, c5_ms, _ = _pairwise(h_ms, c_ms, c5_ms, rng, 100000, sc, nc, deadline - 45)
            h_ms = _repair(h_ms, nc)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_ms.copy(), nc

    # Final polish
    if time.time() < deadline - 3:
        nf = best_n
        h_f = best_h.copy()
        c_f = _corr(h_f)
        c5_f = c_f.max() * 2.0 / nf
        h_f, c_f, c5_f, _ = _pairwise(h_f, c_f, c5_f, rng, 200000, 1e-5, nf, deadline - 2)
        h_f = _repair(h_f, nf)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
