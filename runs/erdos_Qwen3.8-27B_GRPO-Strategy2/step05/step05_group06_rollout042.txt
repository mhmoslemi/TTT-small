def _rbf_eval(params, n, n_center, sigma):
    """params = [weights(n_center), centers(n_center), base]"""
    x = np.linspace(0, 2, n)
    base = params[-1]
    w = params[:n_center]
    mu = params[n_center:2*n_center]
    h = base + np.zeros(n)
    for i in range(n_center):

    return h

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lss_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lss_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] * alpha < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * alpha
        if sh >= 0:
            c1 = n - sh
            g[:c1] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:c1]
        else:
            sp = -sh
            c1 = n - sp
            g[sp:] += wk * (1.0 - h[:c1])
            g[:c1] -= wk * h[sp:]
    return g * dx

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_lss_obj, h, jac=_lss_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descent(h, n_passes, n500):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _p in range(n_passes):
        improved = False
        for j in range(n):
            if time.time() > n500:
                return h, c5
            c_full = np.correlate(h, 1.0 - h, mode='full') * dx
            slopes = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                sh = (n - 1) - k
                slope = 0.0
                if sh >= 0:
                    if j <= n - 1 - sh:
                        slope += 1.0 - h[j + sh]
                    if sh <= j:
                        slope -= h[j - sh]
                else:
                    sp = -sh
                    if sp <= j <= n - 1:
                        slope += 1.0 - h[j - sp]
                    if j <= n - 1 - sp:
                        slope -= h[j + sp]
                slopes[k] = slope * dx
            lo_d, hi_d = -h[j], 1.0 - h[j]
            if hi_d - lo_d < 1e-10:
                continue
            for _it in range(25):
                m1 = lo_d + (hi_d - lo_d) * 0.382
                m2 = lo_d + (hi_d - lo_d) * 0.618
                f1 = np.max(c_full + m1 * slopes)
                f2 = np.max(c_full + m2 * slopes)
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            new_v = h[j] + (lo_d + hi_d) * 0.5
            if abs(new_v - h[j]) > 1e-10:
                h[j] = new_v
                c_new = np.max(c_full + (new_v - h[j]) * slopes) if False else None
                c5_tmp = _eval(h)
                if c5_tmp < c5:
                    c5 = c5_tmp
                    improved = True
        if not improved:
            break
    return h, c5

def _rbf_h(params, n, nc):
    x = np.linspace(0.0, 2.0, n)
    mu = np.linspace(0.0, 2.0, nc)
    base = params[-1]
    w = params[:-1]
    sigma = 0.12
    h = np.full(n, base)
    for i in range(nc):
        h += w[i] * np.exp(-((x - mu[i]) ** 2) / (2.0 * sigma * sigma))
    return h

def _rbf_obj(params, n, nc):
    h = _rbf_h(params, n, nc)
    if h.min() < -0.3 or h.max() > 1.3:
        return 1.0
    h = _repair(h, n)
    return _eval(h)

def _upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_hi = np.zeros(n_new)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi
