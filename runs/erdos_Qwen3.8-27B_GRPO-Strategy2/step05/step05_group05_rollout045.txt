def _repair(h, n):

def _proj(h, target):

def _topk_grad(h, k):

    top = np.argsort(c)[::-1][:k]

def _pairwise(h, n_iter, scale, rng, deadline):

    c5c = _eval(h)

def _slsqp_quick(h, alpha, maxiter):

    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)

def _cons_j(x): return np.ones(len(x))

def _slsqp_call(h, alpha, maxiter, deadline):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':_cons_f,'jac':_cons_j}]

        return _slsqp_quick(x, alpha)[0]

        return _slsqp_quick(x, alpha)[1]
    try:

                       constraints=cons, options={'maxiter':maxiter,'ftol':1e-18})

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        else: hi = mid

    except:

            ht = _proj(h + fr * v, n * 0.5)
            ce = _eval(ht)
            if ce < bf:
                bf, bh = ce, ht
                break
        if bh is not None:
            h, c5c = bh, bf
            stall = 0
            lr = min(lr * 1.03, 0.08)
        else:
            stall += 1
            v *= 0.3
            if stall >= 40:
                lr *= 0.5
                stall = 0
                if lr < 1e-8:
                    break
    return h, c5c

def _block_pert(h, rng, amp):
    n = len(h)
    ht = h.copy()
    w = rng.randint(3, max(4, n // 6))
    s = rng.randint(0, n - w)
    ht[s:s + w] += amp * (rng.rand(w) - 0.5)
    t = np.arange(n)
    f = rng.randint(1, 8)
    ht += amp * 0.2 * np.sin(2 * np.pi * f * t / n + rng.rand() * 6.283)
    return _repair(ht, n)

def _block_swap(h, rng):
    n = len(h)
    ht = h.copy()
    w = rng.randint(4, max(5, n // 10))
    s1 = rng.randint(0, n - w)
    s2 = rng.randint(0, n - w)
    if abs(s1 - s2) < w:
        return ht
    ht[s1:s1 + w], ht[s2:s2 + w] = ht[s2:s2 + w].copy(), ht[s1:s1 + w].copy()
    return _repair(ht, n)

def _pairwise(h, n_iter, scale, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c5c - 1e-15:
            c5c = cn
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _slsqp_oj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    val = (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        if w[ki] < 1e-12:
            continue
        sh = (n - 1) - ki
        wk = w[ki] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            g[sh:n] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:n] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:n]
    return val, g

def _cf(x):
    return x.sum() - len(x) * 0.5

def _cj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    def f_obj(x):
        if time.time() > deadline:
            raise ValueError
        return _slsqp_oj(x, alpha)[0]
    def f_jac(x):
        if time.time() > deadline:
            raise ValueError
        return _slsqp_oj(x, alpha)[1]
    try:
        res = minimize(f_obj, h, jac=f_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP warmup to settle into basin
    for al in [2000, 20000]:
        if _el() > 70:
            break
        hn = _slsqp(best_h, al, 600, t0 + min(_el() + 45, 95))
        c5 = _eval(hn)
        if c5 < best_c5:
            best_h, best_c5 = hn.copy(), c5

    # Phase 2: Extended momentum top-k descent (main workhorse)
    dl2 = t0 + budget_s - 380
    schedules = [(5, 0.03, 0.92), (10, 0.02, 0.92), (15, 0.015, 0.93),
                 (8, 0.025, 0.91), (12, 0.02, 0.92), (6, 0.03, 0.9)]
    for k, lr, bt in schedules:
        if _el() > dl2:
            break
        hn, c5 = _mo_desc(best_h.copy(), 50000, k, lr, bt, dl2)
        if c5 < best_c5:
            best_h, best_c5 = hn.copy(), c5

    # Phase 3: Block perturb + re-optimize for exploration
    if _el() < t0 + budget_s - 300:
        for _r in range(60):
            if _el() > t0 + budget_s - 250:
                break
            strategy = rng.rand()
            if strategy < 0.4:
                ht = _block_pert(best_h, rng, [0.02, 0.04, 0.06, 0.03][rng.randint(4)])
            elif strategy < 0.7:
                ht = _block_swap(best_h, rng)
            else:
                ht = _repair(best_h + 0.015 * rng.randn(len(best_h)), len(best_h))
            hl = min(t0 + budget_s - 250, _el() + 50)
            hn, c5 = _mo_desc(ht, 12000, 10, 0.03, 0.9, hl)
            if c5 < best_c5:
                best_h, best_c5 = hn.copy(), c5

    # Phase 4: Upsample to 2x and refine
    if _el() < t0 + budget_s - 320:
        n1 = n0 * 2
        h_hi = _repair(np.repeat(best_h, 2), n1)
        for k in [10, 15]:
            if _el() > t0 + budget_s - 220:
                break
            hn, c5 = _mo_desc(h_hi, 20000, k, 0.02, 0.92, min(t0 + budget_s - 200, _el() + 100))
            if c5 < best_c5:
                best_h, best_c5, best_n = hn.copy(), c5, n1
        if _el() < t0 + budget_s - 180:
            hn = _slsqp(best_h, 50000, 400, t0 + budget_s - 150)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_h, best_c5, best_n = hn.copy(), c5, len(hn)
        if _el() < t0 + budget_s - 130:
            hn, c5 = _mo_desc(best_h.copy(), 10000, 8, 0.015, 0.9, t0 + budget_s - 120)
            if c5 < best_c5:
                best_h, best_c5, best_n = hn.copy(), c5, len(hn)

    # Phase 5: Multi-scale pairwise refinement
    if _el() < t0 + budget_s - 70:
        for sc in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4]:
            if _el() > t0 + budget_s - 35:
                break
            hn, c5 = _pairwise(best_h.copy(), 12000, sc, rng, min(t0 + budget_s - 35, _el() + 55))
            if c5 < best_c5:
                best_h, best_c5 = hn.copy(), c5

    # Phase 6: Final momentum polish with small k
    if _el() < t0 + budget_s - 12:
        hn, c5 = _mo_desc(best_h.copy(), 20000, 4, 0.01, 0.85, t0 + budget_s - 4)
        if c5 < best_c5:
            best_h, best_c5 = hn.copy(), c5

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
