def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, n):
    """Gradient of c[k] w.r.t. h[m] for all m."""
    # c[k] = sum_{valid i} h[i+k] * (1 - h[i])

        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_lag(h, k, n)
    g *= dx
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _c_fun(x):
        return x.sum() - n * 0.5
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_grad_smooth, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_push(h, n, strength):
    """Push h toward binary: h' = clip(h + strength*(0.5 - h) * |2h-1|, 0, 1)"""
    # Simple version: threshold with hysteresis
    h_push = h + strength * (h - 0.5) * np.abs(2*h - 1)
    # Actually simpler: just scale deviation from 0.5
    h_push = 0.5 + (h - 0.5) * (1 + strength)
    return _repair(h_push, n)

def _worst_lag_descent(h, max_iter, dx_factor=1.0):
    """Gradient descent targeting the worst lag."""
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur
    step = 0.01
    
    for it in range(max_iter):
        c = _corr(h)
        # Find top few lags
        top_k = np.argsort(c)[-5:]
        k_star = top_k[-1]  # worst
        # Subgradient: average over top lags weighted by proximity to max
        max_c = c[k_star]
        weights = np.zeros(len(c))
        for k in top_k:
            weights[k] = np.exp(-(max_c - c[k]) / (0.01 * dx + 1e-10))
        weights /= weights.sum()
        
        g = np.zeros(n)
        for k in top_k:
            if weights[k] > 1e-8:
                g += weights[k] * _grad_lag(h, k, n)
        g *= dx * dx_factor
        
        # Line search
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_new = _proj_box_sum(h - s * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                step = min(step * 1.2, 0.1)
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
    return best_h, best_c5

def _pairwise_descent(h, n_iter, rng):
    """Pairwise coordinate descent."""
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_new = _eval(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5

def _multi_pair_descent(h, n_rounds, rng, base_step):
    """Try multiple step sizes for pairwise."""
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(n_rounds):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        best_improvement = 1e-15
        best_d = 0
        for mult in [3, 1, 0.3, 0.1]:
            d = base_step * mult * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5_new = _eval(h)
            if c5_new < c5_cur - best_improvement:
                best_improvement = c5_cur - c5_new
                best_d = d
            h[i], h[j] = h[i] - d, h[j] + d
        if abs(best_d) > 0:
            h[i] += best_d
            h[j] -= best_d
            c5_cur = _eval(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    return best_h, best_c5
