def _coord_pair_line_search(h, i, j):
    n = len(h)
    dx = 2.0/n
    c0 = np.correlate(h, 1.0-h, mode='full')  # current correlation
    # Compute d_k = ∂c_k/∂h[i] - ∂c_k/∂h[j]
    d = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = (n-1)-k
        di = 0.0
        dj = 0.0
        if sh >= 0:
            if 0 <= i and i+sh < n:
                di += (1.0 - h[i+sh])
            if 0 <= i-sh and i-sh < n:
                di -= h[i-sh]
            if 0 <= j and j+sh < n:
                dj += (1.0 - h[j+sh])
            if 0 <= j-sh and j-sh < n:
                dj -= h[j-sh]
        else:
            sp = -sh
            if 0 <= i-sp and i-sp < n:
                di += (1.0 - h[i-sp])
            if 0 <= i+sp and i+sp < n:
                di -= h[i+sp]
            if 0 <= j-sp and j-sp < n:
                dj += (1.0 - h[j-sp])
            if 0 <= j+sp and j+sp < n:
                dj -= h[j+sp]
        d[k] = di - dj

    dhi = min(1.0-h[i], h[j])
    if dlo >= dhi:
        return h, _eval(h)
    
    # Find δ minimizing max_k(c0[k] + δ*d[k]) over [dlo, dhi]
    # This is a convex 1D optimization
    # Use ternary search
    a, b = dlo, dhi
    for _ in range(50):

        m2 = b - (b-a)/3
        f1 = np.max(c0 + m1*d)
        f2 = np.max(c0 + m2*d)
        if f1 < f2:
            b = m2
        else:
            a = m1
    best_d = (a+b)/2
    h[i] += best_d
    h[j] -= best_d
    return h, _eval(h)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smoother(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s - s.max()
    w = np.exp(sm)
    ws = w.sum()
    return (m + np.log(ws) / alpha), w, c, m

def _grad_correlation(x, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        wkw = wk * dx
        if sh >= 0:
            if sh == 0:
                g += wkw * (1.0 - 2.0 * x)
            else:
                i1 = np.arange(n - sh)
                g[i1] += wkw * (1.0 - x[i1 + sh])
                i2 = np.arange(sh, n)
                g[i2] -= wkw * x[i2 - sh]
        else:
            sp = -sh

            g[i1] += wkw * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wkw * x[i2 + sp]

        g += 1000.0 * (x.sum() - target)

    return d

def _coord_pass(h, n_pairs, rng):
    n = len(h)
    dx = 2.0 / n
    c0 = np.correlate(h, 1.0 - h, mode='full')
    best = np.max(c0) * dx
    improved = 0
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _compute_d(h, i, j, n)
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-14:
            continue
        a, b = dlo, dhi
        for _ in range(45):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            if np.max(c0 + m1 * d) < np.max(c0 + m2 * d):
                b = m2
            else:
                a = m1
        delta = (a + b) / 2.0
        if abs(delta) < 1e-12:
            continue
        f_new = np.max(c0 + delta * d)
        f_old = np.max(c0)
        if f_new < f_old - 1e-13:
            h[i] += delta
            h[j] -= delta
            c0 = np.correlate(h, 1.0 - h, mode='full')
            best = np.max(c0) * dx
            improved += 1
    return h, best, improved

def _deterministic_pass(h, n):
    dx = 2.0 / n
    c0 = np.correlate(h, 1.0 - h, mode='full')
    best = np.max(c0) * dx
    idx = np.argsort(-c0)
    for _pass in range(3):
        improved_any = False
        for i in range(n):
            for j in range(i + 1, n):
                d = _compute_d(h, i, j, n)
                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dlo >= dhi - 1e-14:
                    continue
                a, b = dlo, dhi
                for _ in range(35):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    if np.max(c0 + m1 * d) < np.max(c0 + m2 * d):
                        b = m2
                    else:
                        a = m1
                delta = (a + b) / 2.0
                if abs(delta) < 1e-13:
                    continue
                f_new = np.max(c0 + delta * d)
                if f_new < np.max(c0) - 1e-13:
                    h[i] += delta
                    h[j] -= delta
                    c0 = np.correlate(h, 1.0 - h, mode='full')
                    best = np.max(c0) * dx
                    improved_any = True
        if not improved_any:
            break
    return h, best

def _subgrad_step(h, eps, rng, n_top):
    n = len(h)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c0)
    top = np.argsort(-c0)[:n_top]
    active_rows = top
    d = np.zeros(n)
    for k in active_rows:
        sh = (n - 1) - k
        if sh >= 0:
            if sh == 0:
                d += (1.0 - 2.0 * h)
            else:
                i1 = np.arange(n - sh)
                d[i1] += (1.0 - h[i1 + sh])
                i2 = np.arange(sh, n)
                d[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            d[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            d[i2] -= h[i2 + sp]
    d /= len(active_rows)
    for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        h_new = _proj_sum(h - step * d, n * 0.5)
        c5n = _eval(h_new)
        if c5n < cmax * 2.0 / n - 1e-16:
            return h_new, c5n
    return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS-B with increasing smoothness
    for alpha in [1000, 5000, 50000, 500000]:
        if _el() > 120:
            break
        h_new = _lbfgs_opt(best_h, alpha, 400)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Coordinate-pair line search (random pairs)
    if _el() < budget_s - 200:
        for npr in [30000, 50000, 80000, 120000]:
            if _el() > budget_s - 150:
                break
            h_new, c5, _ = _coord_pass(best_h.copy(), npr, rng)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_new

    # Phase 3: Subgradient refinement
    if _el() < budget_s - 200:
        for _ in range(500):
            if _el() > budget_s - 150:
                break
            h_new, c5 = _subgrad_step(best_h, 0.001, rng, 3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 4: More L-BFGS-B polish
    if _el() < budget_s - 150:
        for alpha in [200000, 1000000, 5000000]:
            if _el() > budget_s - 120:
                break
            h_new = _lbfgs_opt(best_h, alpha, 500)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Heavy coordinate-pair passes
    if _el() < budget_s - 80:
        for _ in range(8):
            if _el() > budget_s - 50:
                break
            h_new, c5, _ = _coord_pass(best_h.copy(), 100000, rng)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_new

    # Phase 6: Deterministic full sweep (slower but thorough)
    if _el() < budget_s - 300 and best_n <= 300:
        h_new, c5 = _deterministic_pass(best_h.copy(), best_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 7: Try different n for alternative basin
    if _el() < budget_s - 250:
        for nt in [200, 250, 300, 280]:
            if _el() > budget_s - 200:
                break
            if nt == best_n:
                continue
            x_old = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, nt)
            hn = np.interp(x_new, x_old, best_h)
            hn = _repair(hn, nt)
            if _el() < budget_s - 180:
                hn = _lbfgs_opt(hn, 50000, 300)
            if _el() < budget_s - 160:
                hn, c5, _ = _coord_pass(hn, 30000, rng)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()
                best_n = nt

    # Phase 8: Final coordinate refinement
    if _el() < budget_s - 20:
        remaining = budget_s - _el() - 10
        npr = min(200000, max(10000, int(remaining * 80000)))
        h_new, c5, _ = _coord_pass(best_h.copy(), npr, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 9: Final deterministic polish if time allows
    if _el() < budget_s - 100 and len(best_h) <= 280:
        h_new, c5 = _deterministic_pass(best_h.copy(), len(best_h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
