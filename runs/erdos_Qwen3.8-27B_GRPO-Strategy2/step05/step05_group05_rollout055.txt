import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        s = np.clip(h-mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _shift_grad(h, k):
    n = len(h)
    dx = 2.0/n
    s = (n-1) - k
    g = np.zeros(n)
    if s >= 0:
        idx = np.arange(n-s)
        g[idx] += (1.0 - h[idx+s])
        if s > 0:
            idx2 = np.arange(s, n)
            g[idx2] -= h[idx2-s]
    else:
        p = -s
        idx = np.arange(p, n)
        g[idx] += (1.0 - h[idx-p])
        idx2 = np.arange(0, n-p)
        g[idx2] -= h[idx2+p]
    return g * dx

1. **Coordinate descent with exact evaluation**: For each coordinate, compute the optimal value exactly (since f is piecewise linear in each coordinate). This is O(n_top * n) per coordinate if we only check active shifts.

- This is a piecewise linear convex function
- Minimum is at a breakpoint or boundary
- Breakpoints: x = (b_{k'} - b_k) / (a_k - a_{k'}) for pairs (k, k')

    """Optimize h[i] and h[j] jointly (preserving sum) to minimize max of top shifts."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    active = np.argsort(c)[::-1][:n_top]

    # dc_k/dδ = dc_k/dh[i] - dc_k/dh[j]

    hj = h[j]
    a = np.zeros(n_top)  # dc_k/dδ
    b = np.zeros(n_top)  # c_k at δ=0
    for idx, k in enumerate(active):
        s = (n-1) - k
        ai = 0.0  # dc_k/dh[i]
        aj = 0.0  # dc_k/dh[j]
        if s >= 0:
            if 0 <= i <= n-1-s: ai += 1.0 - h[i+s]
            if s > 0 and i >= s: ai -= h[i-s]
            if 0 <= j <= n-1-s: aj += 1.0 - h[j+s]
            if s > 0 and j >= s: aj -= h[j-s]
        else:

            if i >= p: ai += 1.0 - h[i-p]
            if i <= n-1-p: ai -= h[i+p]
            if j >= p: aj += 1.0 - h[j-p]
            if j <= n-1-p: aj -= h[j+p]
        a[idx] = ai - aj
        b[idx] = c[k]  # c_k at current h (δ=0)
    # Minimize max_k (a_k * δ + b_k) for δ in [max(-hi, hj-1), min(1-hi, hj)]
    dlo = max(-hi, hj - 1.0)
    dhi = min(1.0 - hi, hj)
    if dlo >= dhi:
        return i, j, 0.0
    candidates = [dlo, dhi]
    for jj in range(n_top):
        for kk in range(jj+1, n_top):
            if abs(a[jj] - a[kk]) > 1e-14:
                xbr = (b[kk] - b[jj]) / (a[jj] - a[kk])
                if xbr > dlo and xbr < dhi:
                    candidates.append(xbr)
    best_d = 0.0
    best_f = np.max(b)  # f at δ=0
    for d in candidates:
        f = np.max(a * d + b)
        if f < best_f - 1e-15:
            best_f = f
            best_d = d
    return i, j, best_d

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _shift_grad_comp(h, k, idx):
    n = len(h)
    s = (n - 1) - k
    g = 0.0
    if s >= 0:
        if 0 <= idx <= n - 1 - s:
            g += 1.0 - h[idx + s]
        if s > 0 and idx >= s:
            g -= h[idx - s]
    else:
        p = -s
        if idx >= p:
            g += 1.0 - h[idx - p]
        if idx <= n - 1 - p:
            g -= h[idx + p]
    return g

def _full_shift_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.array([_shift_grad_comp(h, k, i) for i in range(n)])
    return g * dx

def _pair_coord_opt(h, i, j, n_top):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    active = np.argsort(c)[::-1][:n_top]
    hi, hj = h[i], h[j]
    a = np.zeros(n_top)
    b = np.zeros(n_top)
    for idx, k in enumerate(active):
        a[idx] = _shift_grad_comp(h, k, i) - _shift_grad_comp(h, k, j)
        b[idx] = c[k]
    dlo = max(-hi, hj - 1.0)
    dhi = min(1.0 - hi, hj)
    if dlo >= dhi:
        return 0.0
    candidates = [dlo, dhi, 0.0]
    for jj in range(n_top):
        for kk in range(jj + 1, n_top):
            if abs(a[jj] - a[kk]) > 1e-14:
                xbr = (b[kk] - b[jj]) / (a[jj] - a[kk])
                if dlo < xbr < dhi:
                    candidates.append(xbr)
    best_d = 0.0
    best_f = np.max(b)
    for d in candidates:
        f = np.max(a * d + b)
        if f < best_f - 1e-15:
            best_f = f
            best_d = d
    return best_d

def _pair_sweep(h, n_top, n_pairs, rng):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _pair_coord_opt(h, i, j, n_top)
        if abs(d) > 1e-14:
            h[i] += d
            h[j] -= d
            h = np.clip(h, 0, 1)
    h = _proj_sum(h, n * 0.5)
    return h, _eval(h)

def _subgrad_descent(h, n_iter, n_act_start, n_act_max, rng):
    n = len(h)
    c5 = _eval(h)
    stagnant = 0
    for it in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        n_act = min(n_act_start + it // 50, n_act_max, 2 * n - 1)
        top = np.argsort(c)[::-1][:n_act]
        cvals = c[top]
        cthresh = np.percentile(cvals, 40)
        w = np.maximum(cvals - cthresh, 0.0)
        if w.sum() < 1e-15:
            w = np.ones(n_act) / n_act
        else:
            w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, top):
            if wi > 1e-12:
                g += wi * _full_shift_grad(h, ki)
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                stagnant = 0
                improved = True
                break
        if not improved:
            stagnant += 1
            if stagnant > 30:
                break
    return h, c5

def _slsqp_smooth(h, alpha, maxiter):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for kk in range(2 * n - 1):
            if w[kk] < 1e-14:
                continue
            sh = (n - 1) - kk
            wk = w[kk] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n - sh]
            else:
                p = -sh
                g[p:] += wk * (1.0 - x[:n - p])
                g[:n - p] -= wk * x[p:]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, args=(), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _proj_sum(res.x, n * 0.5)
    except Exception:
        return h

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _pair_fine(h, n_iter, rng, n_top):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _pair_coord_opt(h, i, j, n_top)
        if abs(d) > 1e-14:
            h[i] += d
            h[j] -= d
            h = np.clip(h, 0, 1)
            c5_new = _eval(h)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
            else:
                h[i] -= d
                h[j] += d
    h = _proj_sum(h, n * 0.5)
    return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _proj_sum(np.array(initial_h_values, dtype=float).copy(), n0 * 0.5)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP smooth optimization at base resolution
    for alpha in [2000, 20000, 200000, 2000000]:
        if _el() > 180:
            break
        h_new = _slsqp_smooth(best_h, alpha, 1500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Subgradient descent with adaptive active set
    if _el() < budget_s - 450:
        h_new, c5 = _subgrad_descent(best_h.copy(), 2000, 8, 60, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Pairwise coordinate descent (larger active set)
    if _el() < budget_s - 350:
        for ntop in [15, 30, 50]:
            if _el() > budget_s - 340:
                break
            h_new, c5 = _pair_sweep(best_h.copy(), ntop, 8000, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Subgradient again with larger active set
    if _el() < budget_s - 300:
        h_new, c5 = _subgrad_descent(best_h.copy(), 3000, 20, 100, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 5: More pairwise with growing active set
    if _el() < budget_s - 250:
        for ntop in [30, 60, 100]:
            if _el() > budget_s - 240:
                break
            h_new, c5 = _pair_sweep(best_h.copy(), ntop, 6000, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Upsample to 2x and optimize
    if _el() < budget_s - 350:
        n1 = n0 * 2
        h_hi = _proj_sum(_upsample(best_h, 2), n1 * 0.5)
        for alpha in [5000, 50000, 500000]:
            if _el() > budget_s - 300:
                break
            h_hi = _slsqp_smooth(h_hi, alpha, 1000)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _el() < budget_s - 200:
            h_new, c5 = _subgrad_descent(best_h.copy(), 1000, 10, 80, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(best_h)
        if _el() < budget_s - 100:
            h_new, c5 = _pair_sweep(best_h.copy(), 40, 5000, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(best_h)

    # Phase 7: Multi-start exploration at best_n
    n_cur = best_n
    if _el() < budget_s - 300:
        for _r in range(15):
            if _el() > budget_s - 250:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _proj_sum(np.clip(best_h + 0.005 * rng.randn(n_cur), 0, 1), n_cur * 0.5)
            elif strat == 1:
                hp = best_h.copy()
                for _b in range(3):
                    bs = rng.randint(5, max(6, n_cur // 8))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st + bs] += 0.02 * rng.randn(bs)
                hp = _proj_sum(np.clip(hp, 0, 1), n_cur * 0.5)
            elif strat == 2:
                hp = _proj_sum(np.clip(best_h + 0.03 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur * 0.5)
            else:
                hp = best_h.copy()
                for _ in range(50):
                    ii = rng.randint(n_cur)
                    jj = rng.randint(n_cur)
                    if ii != jj:
                        dd = 0.01 * rng.randn()
                        if 0 <= hp[ii] + dd <= 1 and 0 <= hp[jj] - dd <= 1:
                            hp[ii] += dd
                            hp[jj] -= dd
                hp = _proj_sum(hp, n_cur * 0.5)
            h_new = _slsqp_smooth(hp, 200000, 600)
            h_new, c5 = _pair_sweep(h_new, 40, 3000, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase 8: Fine pairwise optimization (convergence polish)
    if _el() < budget_s - 80:
        for ntop in [50, 80, 120, 160]:
            if _el() > budget_s - 60:
                break
            h_new, c5 = _pair_fine(best_h.copy(), 5000, rng, ntop)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 9: Final subgradient + pairwise
    if _el() < budget_s - 30:
        h_new, c5 = _subgrad_descent(best_h.copy(), 1000, 30, 150, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    if _el() < budget_s - 10:
        h_new, c5 = _pair_fine(best_h.copy(), 8000, rng, 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Final repair and evaluation
    best_h = _proj_sum(best_h, best_n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
