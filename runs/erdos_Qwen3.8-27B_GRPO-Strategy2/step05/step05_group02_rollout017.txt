        idx = np.argpartition(c, -k)[-k:]
        m = c[idx].max()
        s = np.clip(alpha * (c[idx] - m), -100, 100)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def gr(x):
        return _topk_grad(x, alpha, k)
    def seq(x):
        return x.sum() - n * 0.5
    def sjac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=gr, method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':seq,'jac':sjac}],
                       options={'maxiter':maxit,'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pg_momentum(h, n_it, alpha, k, t0, t_end):
    n = len(h)
    h = _repair(h.copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)
    v = np.zeros(n)
    lr, mu = 0.02, 0.8
    stagn = 0
    for _ in range(n_it):
        if time.time() >= t_end:
            break
        g = _topk_grad(h, alpha, k)
        v = mu * v + (1-mu) * g
        h_new = _proj_sum(h - lr * v, n * 0.5)
        c5 = _eval(h_new)
        if c5 < best_c5 - 1e-16:
            best_c5, best_h = c5, h_new.copy()
            h = h_new
            stagn = 0
        elif c5 < _eval(h) + 1e-14:
            h = h_new
            stagn += 1
        else:
            lr *= 0.8
            v *= 0.5
            stagn += 1
        if stagn > 200:
            h = _repair(h + 0.005*np.random.randn(n), n)
            v = np.zeros(n)
            lr = 0.02
            stagn = 0
    return best_h, best_c5

def _local_search(h, n_it, scale, t0, t_end, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_it):
        if time.time() >= t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
            else:
                h[i], h[j] = ni-d, nj+d
    return h, c5

def _directed_search(h, n_it, t0, t_end, seed=7):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_it):
        if time.time() >= t_end:
            break
        c = _corr(h)
        top3 = np.argpartition(c, -3)[-3:]
        g = np.zeros(n)
        for ki in top3:
            sh = (n-1) - ki
            if sh >= 0:
                g[sh:] += 1.0 - h[:n-sh] if sh > 0 else 1.0 - h
                g[:n-sh] -= h[sh:] if sh > 0 else h
            else:
                sp = -sh
                g[:n-sp] += 1.0 - h[sp:]
                g[sp:] -= h[:n-sp]
        gi = rng.randint(n)
        gj = rng.randint(n)
        if gi == gj:
            continue
        d = 0.001 * rng.rand()
        if g[gi] > g[gj]:
            pass
        else:
            d = -d
        ni, nj = h[gi]+d, h[gj]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[gi], h[gj] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
            else:
                h[gi], h[gj] = ni-d, nj+d
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP at n0
    t_end = t0 + 50
    for alpha in [2000, 10000, 50000]:
        if time.time() >= t_end:
            break
        h_new = _slsqp(best_h, alpha, 5, 800, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Upsample to 2x, projected gradient
    n1 = n0 * 2
    h_hi = np.repeat(best_h, 2)
    h_hi = _repair(h_hi, n1)
    t_end = t0 + 280
    for alpha in [2000, 10000, 50000]:
        if time.time() >= t_end:
            break
        h_new, c5 = _pg_momentum(h_hi, 20000, alpha, 5, t0, t_end)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n1

    # Phase 3: SLSQP polish at n1
    t_end = t0 + 420
    for alpha in [20000, 100000]:
        if time.time() >= t_end:
            break
        h_new = _slsqp(best_h, alpha, 4, 1200, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 4: More PG with restarts + directed search
    t_end = t0 + 650
    h_cur = best_h.copy()
    nc = len(h_cur)
    for rst in range(8):
        if time.time() >= t_end:
            break
        h_new, c5 = _pg_momentum(h_cur, 10000, 80000, 4, t0, t_end)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        h_cur = h_new if c5 < best_c5 + 0.001 else _repair(best_h + 0.003*np.random.randn(nc), nc)
    if time.time() < t_end:
        h_new, c5 = _directed_search(best_h, 15000, t0, t_end)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: Fine pairwise local search
    t_end = t0 + 900
    for scale in [0.003, 0.001, 0.0003]:
        if time.time() >= t_end:
            break
        t_sub = min(time.time() + 60, t_end)
        h_new, c5 = _local_search(best_h, 50000, scale, t0, t_sub, seed=100+int(scale*1e6))
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 6: Ultra-fine
    t_end = t0 + budget_s - 5
    h_new, c5 = _local_search(best_h, 80000, 5e-5, t0, t_end, seed=999)
    if c5 < best_c5:
        best_c5, best_h = c5, h_new.copy()

    # Final: one more SLSQP if time permits
    if time.time() < t0 + budget_s - 3:
        h_new = _slsqp(best_h, 200000, 3, 500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
