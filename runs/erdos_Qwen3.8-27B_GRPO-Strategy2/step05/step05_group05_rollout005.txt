    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    # Focus on top-k with exponential weighting
    top_idx = np.argsort(c)[::-1][:k]
    vals = c[top_idx] - m
    w = np.exp(alpha * vals)
    w /= w.sum()
    
    g = np.zeros(n)
    for w_k, ki in zip(w, top_idx):
        sh = (n - 1) - ki
        wk = w_k * dx
        if sh >= 0:

            g[i1] += wk * (1.0 - h[i1 + sh])
