        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _worst_swap(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    it = 0
    while time.time() < t_end:
        it += 1
        c = _corr(h)
        ki = np.argmax(c)
        s = abs(int(ki - (n - 1)))
        if s == 0:
            break
        m = n - s
        contrib = h[s:n] * (1.0 - h[:m])
        for _ in range(10):
            total = contrib.sum()
            if total < 1e-12:
                break
            probs = contrib / total
            a = rng.choice(m, p=probs)
            i1, i2 = a + s, a
            scale = max(0.001, 0.02 * (1.0 - 0.0003 * it))
            d = rng.uniform(0.003, scale)
            h[i1] -= d; h[i2] += d
            if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i1] += d; h[i2] -= d
            else:
                h[i1] += d; h[i2] -= d
        for _ in range(4):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.012, 0.012)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _perturb_fourier(h, rng, amp=0.06):
    n = len(h)
    tgt = n / 2.0
    x = np.arange(n)
    f = rng.choice([2, 3, 4, 5, 6, 7, 8, 9, 10, 12])
    ph = rng.uniform(0, 2 * np.pi)
    hp = h + amp * np.sin(2 * np.pi * f * x / n + ph)
    return _repair(np.clip(hp, 0, 1), tgt)

def _perturb_block(h, rng, frac=0.25):
    n = len(h)
    tgt = n / 2.0
    bl = max(4, int(n * frac))
    st = rng.randint(0, max(1, n - bl))
    hp = h.copy()
    hp[st:st + bl] = 1.0 - hp[st:st + bl]
    return _repair(np.clip(hp, 0, 1), tgt)

def _perturb_random(h, rng, amp=0.05):
    n = len(h)
    tgt = n / 2.0
    hp = h + amp * (2.0 * rng.random(n) - 1.0)
    return _repair(np.clip(hp, 0, 1), tgt)

def _fine_polish(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 0.0004, 0.00015, 0.00006]
    for sc in scales:
        cnt = 0
        while time.time() < t_end and cnt < 30000:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            cnt += 1
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Wide-momentum subgradient descent
    h1, c1 = _nest_sg(best_h, dx, t0 + 140, n_top=50, lr0=0.004)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: SLSQP sweep
    for alpha in [400, 1500, 6000, 25000]:
        if rem() < 70:
            break
        hc, c = _slsqp_run(best_h, dx, alpha)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 3: Worst-lag targeted swaps
    h3, c3 = _worst_swap(best_h, dx, rng, t0 + 350)
    if c3 < best_c5:
        best_h, best_c5 = h3, c3

    # Phase 4: High-alpha SLSQP
    for alpha in [8000, 40000, 150000]:
        if rem() < 50:
            break
        hc, c = _slsqp_run(best_h, dx, alpha)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 5: Multi-start with structured perturbations
    t5 = t0 + 640
    for r in range(12):
        if time.time() > t5 or rem() < 220:
            break
        if r % 3 == 0:
            hp = _perturb_fourier(best_h, rng, 0.06)
        elif r % 3 == 1:
            hp = _perturb_block(best_h, rng, 0.2)
        else:
            hp = _perturb_random(best_h, rng, 0.045)
        hp, c = _nest_sg(hp, dx, min(t5 - 12, time.time() + 15), n_top=35, lr0=0.003)
        hc, c = _slsqp_run(hp, dx, 12000)
        if c < best_c5:
            best_h, best_c5 = hc, c
            hp, c = _worst_swap(hc, dx, rng, min(t5 - 3, time.time() + 7))
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 6: 2x resolution upsample
    t6 = t0 + 840
    if rem() > 280:
        n2 = 2 * n
        dx2 = 2.0 / n2
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _nest_sg(h2, dx2, min(t6 - 80, time.time() + 55), n_top=30, lr0=0.002)
        hc2, c2 = _slsqp_run(h2, dx2, 5000)
        if c2 > _c5(h2, dx2):
            hc2 = h2
        h2, c2 = _worst_swap(h2, dx2, rng, min(t6 - 15, time.time() + 55))
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # Phase 7: Final targeted + SLSQP
    nb, dxb = best_n, 2.0 / best_n
    best_h, best_c5 = _worst_swap(best_h, dxb, rng, min(t0 + 940, time.time() + 100))
    if rem() > 55:
        for alpha in [50000, 250000, 800000]:
            if rem() < 18:
                break
            hc, c = _slsqp_run(best_h, dxb, alpha, maxiter=200)
            if c < best_c5:
                best_h, best_c5 = hc, c

    # Phase 8: Fine polish
    if rem() > 15:
        best_h, best_c5 = _fine_polish(best_h, dxb, rng, time.time() + max(5, rem() - 5))

    # Final cleanup
    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
