        g[:n - sh] += 1.0 - h[sh:]
        if sh > 0:

        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    n = len(h)

    else:
        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-15:
            g += w[k] * dx * _grad_lag(h, n - 1 - k)
    return g

def _sum_con(x):
    return x.sum() - len(x) * 0.5

def _sum_con_j(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_j}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topm_descent(h, max_iter, rng, m_start, m_end):
    n = len(h)
    h = _repair(h, n).copy()
    c5 = _eval(h)
    for it in range(max_iter):
        m = max(1, int(m_start + (it / max(max_iter - 1, 1)) * (m_end - m_start)))
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-m:]
        dx = 2.0 / n
        g = np.zeros(n)
        for ki in idx:
            g += dx * _grad_lag(h, n - 1 - int(ki))
        if np.linalg.norm(g) < 1e-14:
            break
        ok = False
        base = 0.04 / (1.0 + 0.003 * it)
        for s in (base, base * 0.5, base * 0.25, 1e-4, 1e-5):
            hp = _proj_sum(h - s * g, n * 0.5)
            cn = _eval(hp)
            if cn < c5 - 1e-15:
                h, c5 = hp, cn
                ok = True
                break
        if not ok:
            h = _repair(h + 0.003 * rng.randn(n), n)
            c5 = _eval(h)
    return h, c5

def _pairwise(h, n_iter, scale, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
            h[i], h[j] = a, b
            cn = _eval(h)
            if cn < c5 - 1e-16:
                c5 = cn
            else:
                h[i], h[j] = a - d, b + d
    return h, c5

def _init_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _init_sine(n, rng):
    t = np.arange(n, dtype=float)
    f = rng.randint(2, 8)
    p = rng.rand() * 6.2832
    return 0.5 + 0.35 * np.sin(2 * np.pi * f * t / n + p) + 0.08 * rng.randn(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float), n0)
    bh, bc5 = h.copy(), _eval(h)

    for a in (200, 1000, 5000, 20000, 100000):
        if budget_s - (time.time() - t0) < 80:
            break
        hn = _slsqp(bh, a, 2500, 1e-19)
        c = _eval(hn)
        if c < bc5:
            bh, bc5 = hn.copy(), c

    if budget_s - (time.time() - t0) > 120:
        ha, ca = _topm_descent(bh, 4000, rng, 1, 5)
        if ca < bc5:
            bh, bc5 = ha.copy(), ca

    extras = [_repair(1.0 - bh, n0)]
    for _ in range(7):
        r = rng.rand()
        if r < 0.35:
            extras.append(_init_binary(n0, rng))
        elif r < 0.65:
            extras.append(_init_sine(n0, rng))
        else:
            extras.append(_repair(bh + 0.06 * rng.randn(n0), n0))
    for s in extras:
        if budget_s - (time.time() - t0) < 70:
            break
        hn = _slsqp(_repair(s, n0), 10000, 1500, 1e-19)
        c = _eval(hn)
        if c < bc5:
            bh, bc5 = hn.copy(), c

    if budget_s - (time.time() - t0) > 350:
        n1 = n0 * 2
        hh = _repair(np.repeat(bh, 2), n1)
        for a in (5000, 20000, 100000):
            if budget_s - (time.time() - t0) < 250:
                break
            hh = _slsqp(hh, a, 2500, 1e-19)
            c = _eval(hh)
            if c < bc5:
                bh, bc5 = hh.copy(), c
        if budget_s - (time.time() - t0) > 200:
            hd = _repair(np.interp(np.linspace(0, 1, n0),
                                    np.linspace(0, 1, len(bh)), bh), n0)
            hd = _slsqp(hd, 50000, 2500, 1e-19)
            c = _eval(hd)
            if c < bc5:
                bh, bc5 = hd.copy(), c

    if budget_s - (time.time() - t0) > 100:
        ha, ca = _topm_descent(bh, 5000, rng, 2, 5)
        if ca < bc5:
            bh, bc5 = ha.copy(), ca

    for sc in (0.005, 0.001, 0.0003):
        if budget_s - (time.time() - t0) < 50:
            break
        hp, cp = _pairwise(bh, 10000, sc, rng)
        if cp < bc5:
            bh, bc5 = hp.copy(), cp

    if budget_s - (time.time() - t0) > 40:
        hn = _slsqp(bh, 200000, 3000, 1e-20)
        c = _eval(hn)
        if c < bc5:
            bh, bc5 = hn.copy(), c

    if budget_s - (time.time() - t0) > 10:
        hp, cp = _pairwise(bh, 25000, 3e-5, rng)
        if cp < bc5:
            bh, bc5 = hp.copy(), cp

    bh = _repair(bh, len(bh))
    bc5 = _eval(bh)
    return (bh, bc5, len(bh))
