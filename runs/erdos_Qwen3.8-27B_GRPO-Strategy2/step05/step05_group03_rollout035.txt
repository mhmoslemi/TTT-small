import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _smooth_grad(h, alpha, top_only=30):
    n = len(h)
    c = _corr(h)
    ac = alpha * c
    ac_max = ac.max()
    w_raw = np.exp(np.clip(ac - ac_max, -500, 0))
    w_sum = w_raw.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w_raw / w_sum
    grad = np.zeros(n)
    n_full = len(c)
    thresh = 1e-10
    for ki in range(n_full):
        wi = w[ki]
        if wi < thresh:
            continue
        sh = ki - (n - 1)
        if sh == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif sh > 0:
            mm = n - sh
            grad[:mm] += wi * (1.0 - h[sh:sh+mm])
            grad[sh:] -= wi * h[:mm]
        else:
            sa = -sh
            mm = n - sa
            grad[sa:sa+mm] += wi * (1.0 - h[:mm])
            grad[:mm] -= wi * h[sa:sa+mm]
    return grad

def _smooth_obj(x, alpha):
    c = _corr(x)
    ac = alpha * c
    m = ac.max()
    return (m + float(np.log(np.sum(np.exp(ac - m))))) / alpha

def _slsqp(h, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    dx = 2.0 / n
    bds = [(0.0, 1.0)] * n
    def fun(x):
        return _smooth_obj(x, alpha)
    def jac(x):
        return _smooth_grad(x, alpha)
    def eq(x):
        return x.sum() - tgt
    def eq_jac(x):
        return np.ones(n)
    try:
        r = minimize(fun, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _proj_grad(h, alpha, t_end, lr0=0.01):
    n = len(h)
    tgt = n / 2.0
    dx = 2.0 / n
    best = _c5(h, dx)
    h = h.copy()
    lr = lr0
    for _ in range(500000):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        h_new = _repair(np.clip(h - lr * g, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h, best

def _local_search(h, rng, t_end, n_trials=50000):
    n = len(h)
    dx = 2.0 / n
    best = _c5(h, dx)
    h = h.copy()
    tgt = n / 2.0
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[::-1][:5]
        ki = top_idx[rng.randint(len(top_idx))]
        sh = abs(int(ki - (n - 1)))
        if sh == 0 or sh >= n:
            continue
        m = n - sh
        if rng.random() < 0.5:
            a = rng.randint(m)
            i1, i2 = a + sh, a
        else:
            a = rng.randint(m)
            i1, i2 = a, a + sh
        d = rng.uniform(0.001, 0.03)
        h[i1] -= d; h[i2] += d
        if 0 <= h[i1] and h[i1] <= 1 and 0 <= h[i2] and h[i2] <= 1:
            h = _repair(h, tgt)
            cn = _c5(h, dx)
            if cn < best:
                best = cn
            else:
                h[i1] += d; h[i2] -= d
                h = _repair(h, tgt)
        else:
            h[i1] += d; h[i2] -= d
            h = _repair(h, tgt)
    return h, best

def _fine_tune(h, rng, t_end):
    n = len(h)
    dx = 2.0 / n
    best = _c5(h, dx)
    h = h.copy()
    tgt = n / 2.0
    scales = [0.003, 0.001, 0.0005, 0.0002, 0.0001]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] and h[i] <= 1 and 0 <= h[j] and h[j] <= 1:
                h = _repair(h, tgt)
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
                    h = _repair(h, tgt)
            else:
                h[i] -= d; h[j] += d
                h = _repair(h, tgt)
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] += 0.06 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.02 * np.sin(2 * np.pi * rng.uniform(0.5, 4.0) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _interp_h(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)
