import time

def _repair(h, n):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    m = c.max()

        sh = (n - 1) - k

            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])

            g[i2] -= w[k] * h[i2 - sh]

            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += w[k] * (1.0 - h[:cnt])
            g[:cnt] -= w[k] * h[sp:]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if abs(di) < 1e-18:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _sa_pairwise(h, c, rng, n_iters, T0, T1, scale, deadline_fn, track_best):
    """Simulated annealing with pairwise moves."""
    n = len(h)
    dx = 2.0 / n
    c5c = c.max() * dx
    for step in range(n_iters):
        if deadline_fn() > 0:
            break
        T = T0 * (T1 / T0) ** (step / max(1, n_iters - 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        delta = new_c5 - c5c
        if delta < -1e-15 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            h[i] = ni
            h[j] = nj
            c = new_c
            c5c = new_c5
            track_best(c5c)
    return h, c, c5c

def _block_move(h, c, rng, n, blen, scale, deadline_fn, track_best):
    """Block move: change a contiguous block of positions."""
    dx = 2.0 / n
    c5c = c.max() * dx
    for step in range(100):
        if deadline_fn() > 0:
            break
        s = rng.randint(0, n - blen + 1)
        dvals = rng.randn(blen) * scale
        dvals -= dvals.mean()  # preserve sum
        # Check bounds
        for l in range(blen):
            if h[s+l] + dvals[l] < -0.01 or h[s+l] + dvals[l] > 1.01:
                break

            dc = np.zeros(2 * n - 1)

                dc += _pos_change(h, s+l, dvals[l], n)
            new_c = c + dc
            new_c5 = new_c.max() * dx
            if new_c5 < c5c:
                for l in range(blen):
                    h[s+l] += dvals[l]
                c = new_c
                c5c = new_c5
                track_best(c5c)
                break
    return h, c, c5c
