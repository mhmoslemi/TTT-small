import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    d = target - h.sum()
    for _ in range(200):
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
        d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_smoothed_max(h, alpha, n_top=10):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    top = np.argsort(w)[::-1]
    for ki in top:
        wi = w[ki]
        if wi < 1e-8:
            break
        s_off = int(ki) - (n - 1)
        sa = abs(s_off)
        mlen = n - sa

        if s_off >= 0:

            g[sa:n] -= h[:mlen]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(-d, h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-13:
                break
    return h

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return _repair(np.clip(h - 0.5 * (lo + hi), 0, 1), target)

def _grad_raw(h, alpha, top_k=12):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    order = np.argsort(w)[::-1]
    for ki in order[:top_k]:
        wi = w[ki]
        if wi < 1e-9:
            break
        ko = int(ki) - (n - 1)
        sa = abs(ko)
        ml = n - sa
        g = np.zeros(n)
        if ko >= 0:
            g[sa:n] += 1.0 - h[:ml]
            g[:ml] -= h[sa:n]
        else:
            g[:ml] += 1.0 - h[sa:n]
            g[sa:n] -= h[:ml]
        grad += wi * g
    return grad

def _grad_p(h, alpha):
    g = _grad_raw(h, alpha)
    g -= g.mean()
    return g

def _prox_descent(h, dx, alpha, t_end, lr0=0.008, beta=0.85):
    n = len(h)
    tgt = n / 2.0
    best, best_h = _c5(h, dx), h.copy()
    v = np.zeros(n)
    lr = lr0
    fail = 0
    while time.time() < t_end:
        g = _grad_p(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = beta * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break
        step = lr * v / vn
        for _ in range(40):
            ht = _project(h - step, tgt)
            cn = _c5(ht, dx)
            if cn < best:
                best, best_h = cn, ht
                h = ht
                lr = min(lr * 1.03, 0.04)
                fail = 0
                break
            lr *= 0.7
        else:
            fail += 1
            if fail > 30:
                break
    return best_h, best

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def objg(x):
        return _grad_raw(x, alpha, top_k=15)
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, jac=objg, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _exchange(h, dx, rng, t_end, n_trial=300):
    n = len(h)
    best, best_h = _c5(h, dx), h.copy()
    h = h.copy()
    c = _corr(h)
    top_lags = np.argsort(c)[::-1][:10]
    scales = [0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003]
    for sc in scales:
        for _ in range(n_trial):
            if time.time() > t_end:
                return best_h, best
            ki = top_lags[rng.randint(len(top_lags))]
            ko = int(ki) - (n - 1)
            sa = abs(ko)
            ml = n - sa
            if sa == 0:
                i, j = rng.randint(n), rng.randint(n)
                continue
            if ko >= 0:
                src = np.arange(sa, n)
                dst = np.arange(0, ml)
            else:
                src = np.arange(ml, n)
                dst = np.arange(0, sa)
            if len(src) < 2 or len(dst) < 2:
                i, j = rng.randint(n), rng.randint(n)
            else:
                i = src[rng.randint(len(src))]
                j = dst[rng.randint(len(dst))]
            if i == j:
                continue
            d = sc * (0.5 + rng.random()) * (1 if rng.random() < 0.5 else -1)
            h[i] += d
            h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                hc = _repair(h, n / 2.0)
                cn = _c5(hc, dx)
                if cn < best:
                    best, best_h = cn, hc.copy()
                    h = hc.copy()
                    c = _corr(h)
                    top_lags = np.argsort(c)[::-1][:10]
                else:
                    h[i] -= d
                    h[j] += d
            else:
                h[i] -= d
                h[j] += d
    return best_h, best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _perturb(h, rng, mode):
    n = len(h)
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] += 0.06 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.02 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), n / 2.0)
