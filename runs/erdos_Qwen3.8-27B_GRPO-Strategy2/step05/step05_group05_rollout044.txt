def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_s = c * dx
    m = c_s.max()
    s = np.clip(alpha * (c_s - m), -500, 0.0)
    w = np.exp(s)
    idx = np.argsort(w)[::-1][:top_k]
    w_sub = w[idx]
    w_sum = w_sub.sum()
    if w_sum < 1e-15:
        w_sub = np.ones(top_k) / top_k
    else:
        w_sub /= w_sum
    g = np.zeros(n)
    for wi, k in zip(w_sub, idx):
        sh = k - (n - 1)
        if sh >= 0:
            i1 = np.arange(sh, n)
            g[i1] += wi * (1.0 - h[i1 - sh])

                g[i2] -= wi * h[i2 + sh]
        else:

            g[i1] += wi * (1.0 - h[i1 + p])
            i2 = np.arange(p, n)
            g[i2] -= wi * h[i2 - p]

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_s = c * dx
    m = c_s.max()
    s_exp = np.clip(alpha * (c_s - m), -500, 0.0)
    w = np.exp(s_exp)
    idx = np.argsort(w)[::-1][:top_k]
    w_sub = w[idx]
    w_sum = w_sub.sum()
    if w_sum < 1e-15:
        w_sub = np.ones(top_k) / top_k
    else:
        w_sub /= w_sum
    g = np.zeros(n)
    for wi, k in zip(w_sub, idx):
        sh = k - (n - 1)
        if sh >= 0:
            i1 = np.arange(sh, n)
            g[i1] += wi * (1.0 - h[i1 - sh])
            if sh > 0:
                i2 = np.arange(n - sh)
                g[i2] -= wi * h[i2 + sh]
        else:
            p = -sh
            i1 = np.arange(n - p)
            g[i1] += wi * (1.0 - h[i1 + p])
            i2 = np.arange(p, n)
            g[i2] -= wi * h[i2 - p]
    return g * dx

def _fista(h0, t_end, alpha, top_k, lr):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_mom = 1.0
    best = h0.copy()
    best_c5 = _eval(h0)
    stale = 0
    for i in range(300000):
        if time.time() > t_end:
            break
        g = _sm_grad(y, alpha, top_k)
        h_new = _proj(y - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) * 0.5
        y = h_new + (t_mom / t_new - 1.0) * (h_new - h)
        h = h_new
        t_mom = t_new
        if i % 20 == 0:
            c5 = _eval(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best = h.copy()
                stale = 0
            else:
                stale += 1
            if stale >= 6:
                lr *= 0.6
                stale = 0
                t_mom = 1.0
    return best, best_c5

def _effect_vec(h, i, j):
    n = len(h)
    n_full = 2 * n - 1
    s_all = np.arange(n_full) - (n - 1)
    ef = np.zeros(n_full)
    mi1 = (i - s_all >= 0) & (i - s_all < n)
    mi2 = (i + s_all >= 0) & (i + s_all < n)
    ef[mi1] += (1.0 - h[i - s_all[mi1]])
    ef[mi2] -= h[i + s_all[mi2]]
    mj1 = (j - s_all >= 0) & (j - s_all < n)
    mj2 = (j + s_all >= 0) & (j + s_all < n)
    ef[mj1] -= (1.0 - h[j - s_all[mj1]])
    ef[mj2] += h[j + s_all[mj2]]
    return ef

def _pairwise(h, t_end, rng):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        ef = _effect_vec(h, i, j)
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-12:
            continue
        cands = [dlo, dhi, 0.0, (dlo + dhi) * 0.5]
        top = np.argsort(c_raw)[::-1][:10]
        for a in range(10):
            for b in range(a + 1, 10):
                denom = dx * (ef[top[b]] - ef[top[a]])
                if abs(denom) > 1e-14:
                    d = (c_raw[top[a]] - c_raw[top[b]]) / denom
                    if dlo + 1e-12 < d < dhi - 1e-12:
                        cands.append(d)
        best_d = 0.0
        best_v = np.max(c_raw)
        for d in cands:
            v = np.max(c_raw + dx * d * ef)
            if v < best_v - 1e-14:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-12:
            h[i] += best_d
            h[j] -= best_d
            np.clip(h, 0, 1, out=h)
            c5_new = _eval(h)
            if c5_new < c5:
                c5 = c5_new
    return h, c5

def _upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_hi = np.zeros(n_new)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _multi_start_h(n, rng):
    t = np.arange(n) / n
    strat = rng.randint(4)
    if strat == 0:
        h = 0.5 + 0.3 * np.sin(np.pi * t)
    elif strat == 1:
        h = 0.5 + 0.25 * np.cos(2 * np.pi * t)
    elif strat == 2:
        h = np.where(np.arange(n) % 2 == 0, 0.7, 0.3)
    else:
        h = np.full(n, 0.5)
        for _ in range(rng.randint(3, 7)):
            st = rng.randint(0, n // 2)
            ln = rng.randint(2, n // 4 + 1)
            h[st:st + ln] += 0.3 * rng.randn()
    return _repair(h, n)
