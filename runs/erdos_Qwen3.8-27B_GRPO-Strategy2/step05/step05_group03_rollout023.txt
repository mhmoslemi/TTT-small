    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:

        for i in order:

                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:

    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:n_top]
    w = np.exp(temp * (c[top] - c[top].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:

        m = n - s
        g = np.zeros(n)
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
        grad += wi * g
    grad -= grad.mean()
    return grad

def _proj_grad(h, dx, t_end, n_top=10):
    """Projective gradient with backtracking line search."""
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, dx)
    step = 0.02
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(500000):
        if time.time() > t_end:
            break
        g = _subgrad_fn(h, n_top)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        momentum = mu * momentum + (1 - mu) * g
        d = -momentum
        d -= d.mean()  # project onto sum=0
        
        # Backtracking line search
        found = False
        for attempt in range(15):
            t = step * (0.7 ** attempt)
            h_try = h + t * d
            h_try = _repair(h_try, tgt)
            c_try = _c5(h_try, dx)
            if c_try < best:
                h = h_try
                best = c_try
                step = min(step * 1.05, 0.1)
                found = True
                break
        if not found:
            step *= 0.5
            if step < 1e-7:
                break
    return h, best

def _slsqp_active(h, dx, alpha, K, maxiter=300):
    """SLSQP using only top-K lags in the softmax."""
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(x)
        idx = np.argsort(c)[::-1][:K]
        ck = c[idx]
        m = ck.max()
        s = np.clip(alpha * (ck - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    
    def eq(x):
        return x.sum() - tgt
    
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 4), replace=False)
        hp[idx] += 0.1 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1.5, 6.0)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        hp += amp * np.sin(2 * np.pi * f * x + ph)
    elif mode == 3:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    elif mode == 4:
        # Block flip
        bsz = rng.randint(5, n // 5)
        start = rng.randint(0, n - bsz)
        hp[start:start+bsz] = 1.0 - hp[start:start+bsz]
    else:
        # Random swap
        idx = rng.choice(n, 2, replace=False)
        hp[idx[0]], hp[idx[1]] = hp[idx[1]], hp[idx[0]]
    return _repair(np.clip(hp, 0, 1), tgt)

def _local_swap(h, dx, rng, t_end, n_iter=50000):
    """Targeted local search: swap moves focused on worst lags."""
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    c = _corr(h)
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 200 == 0:
            c = _corr(h)
        top = np.argsort(c)[::-1][:5]
        # Pick a random top lag
        ki = top[rng.randint(len(top))]
        s = abs(int(ki - (n - 1)))
        if s == 0:
            continue
        m = n - s
        # Try to reduce c_s by moving mass from high h[i] to low h[i+s]
        i = rng.randint(m)
        j = i + s
        d = rng.uniform(0.005, 0.03)
        # Try: decrease h[i], increase h[j] (if h[i] > h[j])
        if h[i] > h[j]:
            amt = min(d, h[i] - h[j])
            h[i] -= amt; h[j] += amt
            cn = _c5(h, dx)
            if cn < best:
                best = cn
            else:
                h[i] += amt; h[j] -= amt
        else:
            # Try: decrease h[j], increase h[i]
            amt = min(d, h[j] - h[i])
            h[j] -= amt; h[i] += amt
            cn = _c5(h, dx)
            if cn < best:
                best = cn
            else:
                h[j] += amt; h[i] -= amt
    
    # Also random pairwise swaps
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 200 == 0:
            c = _corr(h)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.02, 0.02)
        h[i] += d; h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            cn = _c5(h, dx)
            if cn < best:
                best = cn
            else:
                h[i] -= d; h[j] += d
        else:
            h[i] -= d; h[j] += d
    return h, best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()

            else:
                a = min(-d, h[i]); h[i] -= a; d += a

    n = len(h)
    g = np.zeros(n)
    s = abs(s)

    g[s:] -= h[:m]

        return g
