        g[sp:] += 1.0 - h[:nv]
        g[:nv] -= h[sp:]
    return g

def _wl(h, n, tm, mi, st, tl):
    c5 = _eval(h)
    h = h.copy()
    step = st
    stall = 0
    ts = time.time()
    for _ in range(mi):
        if time.time() - ts > tl:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(c, -tm)[-tm:]
        g = np.zeros(n)
        for k in idx:
            g += _lg(h, (n - 1) - k, n)
        g /= tm
        ok = False
        for m in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            hn = _repair(h - step * m * g, n)
            v = _eval(hn)
            if v < c5 - 1e-15:
                h = hn; c5 = v; stall = 0; ok = True; break
        if not ok:
            step *= 0.5; stall += 1
            if stall > 50 or step < 1e-13:
                break
    return h, c5

def _w1(h, n, mi, tl):
    c5 = _eval(h)
    h = h.copy()
    ts = time.time()
    for _ in range(mi):
        if time.time() - ts > tl:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k0 = c.argmax()
        sh = (n - 1) - k0
        g = _lg(h, sh, n)
        best_t = 0.0
        best_v = c5
        for t in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            hn = _repair(h - t * g, n)
            v = _eval(hn)
            if v < best_v:
                best_v = v; best_t = t
        if best_t > 0:
            h = _repair(h - best_t * g, n)
            c5 = best_v
        else:
            break
    return h, c5

def _so(hv):
    c = np.correlate(hv, 1.0 - hv, mode='full')
    m = c.max()
    s = np.clip(_B * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _B

def _sj(hv):
    n = _N; dx = 2.0 / n
    c = np.correlate(hv, 1.0 - hv, mode='full')
    m = c.max()
    s = np.clip(_B * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-14:
            g += w[k] * _lg(hv, (n - 1) - k, n)
    return dx * g

def _se(hv):
    return hv.sum() - _N * 0.5

def _ssj(hv):
    return np.ones(_N)

def _slq(h, a, mi, ft):
    global _B, _N
    _B = a; _N = len(h); n = _N
    try:
        r = minimize(_so, h, jac=_sj, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _se, 'jac': _ssj}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    bh = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bc = _eval(bh); bn = n0

    def rem():
        return budget_s - (time.time() - t0)

    # P1: Quick SLSQP
    if rem() > 300:
        h2 = _slq(bh, 5000, 600, 1e-18)
        c2 = _eval(h2)
        if c2 < bc: bh, bc = h2, c2

    # P2: Worst-lag descent (main)
    if rem() > 200:
        for tm, st in [(5, 0.1), (10, 0.05), (3, 0.05), (20, 0.02), (1, 0.03)]:
            if rem() < 100: break
            h2, c2 = _wl(bh, bn, tm, 6000, st, 70)
            if c2 < bc: bh, bc = h2, c2

    # P3: SLSQP high alpha
    if rem() > 160:
        h2 = _slq(bh, 50000, 1000, 1e-19)
        c2 = _eval(h2)
        if c2 < bc: bh, bc = h2, c2

    # P4: Single-lag refinement + WL
    if rem() > 120:
        h2, c2 = _w1(bh, bn, 3000, 40)
        if c2 < bc: bh, bc = h2, c2
        for tm, st in [(3, 0.02), (5, 0.01)]:
            if rem() < 60: break
            h2, c2 = _wl(bh, bn, tm, 5000, st, 45)
            if c2 < bc: bh, bc = h2, c2

    # P5: Pairwise search
    if rem() > 50:
        hh = bh.copy(); nc = bn; c5 = bc
        for sc in [0.005, 0.001, 0.0003, 0.0001]:
            if rem() < 12: break
            for _ in range(8000):
                if rem() < 12: break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j: continue
                d = sc * rng.randn()
                ni, nj = hh[i] + d, hh[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                hh[i], hh[j] = ni, nj
                c5n = _eval(hh)
                if c5n < c5 - 1e-16:
                    c5 = c5n
                    if c5n < bc: bc, bh = c5n, hh.copy()
                else:
                    hh[i], hh[j] = ni - d, nj + d

    # P6: Perturbation restarts
    if rem() > 40:
        for _ in range(8):
            if rem() < 20: break
            hp = _repair(bh + 0.005 * rng.randn(bn), bn)
            h2, c2 = _wl(hp, bn, 5, 3000, 0.03, 30)
            if c2 < bc: bh, bc = h2, c2

    # P7: Final SLSQP
    if rem() > 18:
        h2 = _slq(bh, 100000, 1200, 1e-20)
        c2 = _eval(h2)
        if c2 < bc: bh, bc = h2, c2

    # P8: Ultra-fine
    if rem() > 5:
        hh = bh.copy(); nc = bn; c5 = bc
        for _ in range(25000):
            if rem() < 3: break
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j: continue
            d = 1e-5 * rng.randn()
            ni, nj = hh[i] + d, hh[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                hh[i], hh[j] = ni, nj
                c5n = _eval(hh)
                if c5n < c5 - 1e-17:
                    c5 = c5n
                    if c5n < bc: bc, bh = c5n, hh.copy()
                else:
                    hh[i], hh[j] = hh[i] - d, hh[j] + d

    # P9: Final WL squeeze
    if rem() > 4:
        h2, c2 = _w1(bh, bn, 2000, rem() - 2)
        if c2 < bc: bh, bc = h2, c2

    bh = _repair(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)
