def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0, 1)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_grad(h, k, dx):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:sh+cnt])
        if sh > 0:
            g[sh:] -= h[:cnt]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _topk_grad(h, K, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:K]
    w = c[idx]
    w = w / max(w.sum(), 1e-30)
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * _lag_grad(h, idx[i], dx)
    return g

            best_c = c_new
            best_h = h_new.copy()
        if c_new < _eval(h):
            h = h_new
            lr = min(lr * 1.04, 0.08)
        else:
            lr *= 0.65
    return best_h, best_c

def _binary_push(h, n, strength):
    hp = h.copy()
    mask_lo = hp < 0.5
    mask_hi = ~mask_lo
    hp[mask_lo] *= (1.0 - strength)
    hp[mask_hi] = 1.0 - (1.0 - hp[mask_hi]) * (1.0 - strength)
    return _repair(hp, n)

def _fourier_perturb(h, n, rng, n_modes, amp):
    hq = h - h.mean()
    hf = np.fft.rfft(hq)
    upper = min(n_modes, len(hf) - 1)
    for _ in range(rng.randint(1, 4)):

        hf[j] += amp * complex(rng.randn(), rng.randn())
        if j < len(hf) - 1:
            hf[-j] = np.conj(hf[j])
    hq_new = np.fft.irfft(hf, n)
    return _repair(hq_new + h.mean(), n)
