    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[::-1][:topk]
    weights = np.exp(-np.arange(topk) * 0.3)
    weights /= weights.sum()
    g = np.zeros(n)
    for (k, w) in zip(idx, weights):
        shift = k - (n - 1)
        gs = np.zeros(n)
        if shift >= 0:
            for i in range(n - shift):
                gs[i] += (1.0 - h[i + shift])
                gs[i + shift] -= h[i]
        else:
            s = -shift
            for i in range(s, n):
                gs[i] += (1.0 - h[i - s])
                gs[i - s] -= h[i]
        g += w * gs
    return g * (2.0 / n), c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):

def _corrs(h):

def _proj(h, target):

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)

    def obj(x):

        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)

            if wk < 1e-14:
                continue
            shift = k - (n - 1)
            g += wk * _grad_lag(x, shift)
        return g * (2.0 / n)
    def eq_fun(x):
        return x.sum() - n * 0.5
    def eq_jac(x):
        return np.ones(n)
