import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    c5c = _eval(h)

def _pair_descent_eff(h, max_iters, grid_size, rng):
    """Efficient pair descent maintaining correlation array incrementally."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    c5c = c.max() * dx
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Compute delta_c[k] for all k due to h[i]+=d, h[j]-=d
        # For shift sh: contribution to c[k] where sh=(n-1)-k
        delta_c = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = 0.0
            cj = 0.0
            if sh >= 0:
                if i <= n - 1 - sh:
                    ci += (1.0 - h[i + sh])
                if 0 <= i - sh < n - sh:
                    ci -= h[i - sh]
                if j <= n - 1 - sh:
                    cj += (1.0 - h[j + sh])
                if 0 <= j - sh < n - sh:
                    cj -= h[j - sh]
            else:
                sp = -sh
                if sp <= i < n:
                    ci += (1.0 - h[i - sp])
                if 0 <= i + sp < n - sp:
                    ci -= h[i + sp]
                if sp <= j < n:
                    cj += (1.0 - h[j - sp])
                if 0 <= j + sp < n - sp:
                    cj -= h[j + sp]
            delta_c[k] = ci - cj
        # Grid search for best d
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi:
            continue
        dgrid = np.linspace(dlo, dhi, grid_size)
        best_d = 0.0
        best_v = np.inf
        cdx = c * dx
        for d in dgrid:
            v = np.max(cdx + dx * d * delta_c)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-12 and best_v < c5c - 1e-14:
            c += best_d * delta_c
            h[i] += best_d
            h[j] -= best_d
            c5c = best_v
    return h, c5c
