    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        h = np.clip(h, 0.0, 1.0)

            #   i=j: contributes (1-h[j+k-n+1]) if 0<=j+k-n+1<n → n-1-j<=k<=2n-2-j
            #   i+k-n+1=j → i=j-k+n-1: contributes -h[j-k+n-1] if 0<=j-k+n-1<n → j<=k<=j+n-1

                b[k1lo:k1hi + 1] = 1.0 - h[idx1]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

    if s < 1e-12:
        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

def _coord_descent(h, n, max_sweeps, t_start=None, deadline=None, seed=0):
    c = _recompute_c(h, n)
    k0 = n - 1
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = 0
        for j in order:
            hj = h[j]
            b = np.zeros(2 * n - 1)
            k1lo = max(0, n - 1 - j)
            k1hi = min(2 * n - 2, 2 * n - 2 - j)
            if k1lo <= k1hi:
                idx = j + np.arange(k1lo, k1hi + 1) - n + 1
                b[k1lo:k1hi + 1] = 1.0 - h[idx]
            k2lo = max(0, j)
            k2hi = min(2 * n - 2, j + n - 1)
            if k2lo <= k2hi:
                idx2 = j - np.arange(k2lo, k2hi + 1) + n - 1
                b[k2lo:k2hi + 1] -= h[idx2]
            D_s = np.empty(2 * n - 2)
            D_s[:k0] = c[:k0]
            D_s[k0:] = c[k0 + 1:]
            D_s -= (np.concatenate([b[:k0], b[k0 + 1:]])) * hj
            b_s = np.concatenate([b[:k0], b[k0 + 1:]])
            D_0 = c[k0] - hj * b[k0]
            t_new = _optimal_t(D_s, b_s, D_0, n)
            delta = t_new - hj
            if abs(delta) > 1e-14:
                h[j] = t_new
                c += b * delta
                improved += 1
        if improved == 0:
            break
        if sweep % 15 == 14:
            c = _recompute_c(h, n)
        if sweep % 10 == 9:
            rng_s = np.random.RandomState(sweep * 31 + seed)
            order = rng_s.permutation(n)
    return h, _eval(h)

def _perturb(best_h, n, rng, mode):
    h = best_h.copy()
    if mode == 0:
        h += 0.025 * rng.randn(n)
    elif mode == 1:
        h += 0.01 * rng.randn(n)
    elif mode == 2:
        bs = rng.randint(4, max(5, n // 5))
        st = rng.randint(0, n - bs)
        h[st:st + bs] += 0.06 * rng.randn(bs)
    elif mode == 3:
        h = 1.0 - best_h + 0.015 * rng.randn(n)
    elif mode == 4:
        freqs = rng.randint(1, n // 3 + 1, size=3)
        tt = np.arange(n)
        for f in freqs:
            h += 0.006 * rng.randn() * np.sin(2 * np.pi * f * tt / n)
    else:
        snap = (best_h > 0.5).astype(float)
        h = 0.25 * snap + 0.75 * best_h
        h += 0.02 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    h, c5 = _coord_descent(h, n, 400, deadline=deadline, seed=0)
    h = _repair(h, n)
    c5 = _eval(h)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    nrestarts = 0
    for r in range(50):
        if time.time() > deadline - 200:
            break
        mode = r % 6
        hp = _perturb(best_h, n, rng, mode)
        h, c5 = _coord_descent(hp, n, 130, deadline=deadline, seed=r + 100)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        nrestarts += 1

    if time.time() < deadline - 120:
        h = best_h.copy()
        for _it in range(3000):
            if time.time() > deadline - 120:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = 0.003 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c2 = _eval(h)
                if c2 < best_c5 - 1e-6:
                    best_h, best_c5 = h.copy(), c2
                elif c2 > _eval(best_h) + 0.005:
                    h[i], h[j] = h[i] - d, h[j] + d

    if time.time() < deadline - 30:
        h = best_h.copy()
        h, c5 = _coord_descent(h, n, 500, deadline=deadline, seed=999)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    if time.time() < deadline - 5:
        h = best_h.copy()
        nc = n
        c5c = _eval(h)
        for scale, nits in [(1e-4, 4000), (3e-5, 4000), (1e-5, 3000)]:
            if time.time() > deadline - 5:
                break
            for _it in range(nits):
                if time.time() > deadline - 5:
                    break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c3 = _eval(h)
                    if c3 < c5c:
                        c5c = c3
                        if c3 < best_c5:
                            best_h, best_c5 = h.copy(), c3
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
