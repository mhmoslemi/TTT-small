def _adam_optimize(h0, dx, t_end, alpha=5000, lr=0.01, n_top=30):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c = _c5(h, dx)
    best_h = h.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_count = 0
    
    while time.time() < t_end:
        t_count += 1
        g = _lse_grad(h, alpha, n_top)
        g -= g.mean()  # preserve sum
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t_count)
        v_hat = v / (1 - beta2**t_count)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _repair(h_new, target)
        
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

        if not free.any():
            break
        h[free] += d / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(d), max(0.0, 1.0 - h[i]) if d > 0 else h[i])
            h[i] += a * (1.0 if d > 0 else -1.0)
            d -= a * (1.0 if d > 0 else -1.0)
            if abs(d) < 1e-13:
                break

def _lse_obj_grad(h, alpha, n_top):
    n = len(h)
    c = _corr(h)
    cmax = c.max()
    top = np.argsort(c)[::-1][:n_top]

    e = np.exp(s)
    w = e / e.sum()

            grad[:m] += wi * (1.0 - h[lag:lag+m])

        else:
            t = -lag
            m = n - t
            grad[t:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[t:t+m]
    return obj, grad

    v_buf = np.zeros(n)

        obj, g = _lse_obj_grad(h, alpha, n_top)
        g -= g.mean()

        v_buf = b2 * v_buf + (1 - b2) * g * g

            best_h = h_new.copy()
            stalled = 0
        else:

        h = h_new
    return best_h, best_c
