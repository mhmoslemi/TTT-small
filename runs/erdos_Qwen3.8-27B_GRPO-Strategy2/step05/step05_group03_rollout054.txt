import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp_to(h, n_new):
    return np.interp(np.linspace(0.0, 1.0, n_new), np.linspace(0.0, 1.0, len(h)), h)

def _lp_optimize(h, dx, t_end, n_top=5, trust=None):
    """Iterative LP linearization for min max correlation."""
    n = len(h)
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    
    while time.time() < t_end:
        c = _corr(h)
        c5 = c.max() * dx
        # Get top lags
        top_idx = np.argsort(c)[::-1][:n_top]
        active = []
        for ki in top_idx:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            active.append((ki, s, c[ki]))
        
        if len(active) == 0:
            break
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        
        for ki, s, ck in active:
            m = n - s
            grad = np.zeros(n)
            grad[s:] += (1.0 - h[:m])
            grad[:m] -= h[s:]
            f_lin = ck + grad @ (h_var - h)
            cons.append(f_lin <= t_var / dx)
        
        if trust is not None:
            # Trust region: sum of squared changes
            cons.append(cp.norm(h_var - h) <= trust)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, cons)
        try:
            prob.solve(solver='CLARABEL', verbose=False)
        except:
            try:
                prob.solve(solver='ECOS', verbose=False)
            except:
                try:
                    prob.solve(solver='SCS', verbose=False)
                except:
                    break
        
        new_h = _repair(np.array(h_var.value).flatten(), n / 2.0)
        new_c = _c5(new_h, dx)
        
        if new_c < best_c:
            best_h = new_h.copy()
            best_c = new_c
            h = new_h
        else:
            # Partial step
            mid = 0.5 * h + 0.5 * new_h
            mid = _repair(mid, n / 2.0)
            mid_c = _c5(mid, dx)
            if mid_c < best_c:
                best_h = mid.copy()
                best_c = mid_c
                h = mid
            else:
                break
    
    return best_h, best_c

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:

                d -= a
            else:

def _lag_grad(h, k, n):
    """Gradient of c[k] = correlate(h, 1-h, 'full')[k] w.r.t. h."""
    s = k - (n - 1)  # signed shift
    abs_s = abs(s)
    if abs_s == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    m = n - abs_s
    if s > 0:
        # c[k] = sum_{i=0}^{m-1} h[i+abs_s](1-h[i])
        # grad[j] = (1-h[j-abs_s])*1_{j>=abs_s} - h[j+abs_s]*1_{j<m}
        grad[abs_s:] = (1.0 - h[:m])
        grad[:m] -= h[abs_s:]
    else:
        # c[k] = sum_{i=0}^{m-1} h[i](1-h[i+abs_s])
        # grad[j] = (1-h[j+abs_s])*1_{j<m} - h[j-abs_s]*1_{j>=abs_s}
        grad[:m] = (1.0 - h[abs_s:])
        grad[abs_s:] -= h[:m]
    return grad

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp_to(h, n_new):
    return np.interp(np.linspace(0.0, 1.0, n_new), np.linspace(0.0, 1.0, len(h)), h)

def _lag_grad(h, k, n):
    s = k - (n - 1)
    a = abs(s)
    if a == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    m = n - a
    if s > 0:
        grad[a:] = 1.0 - h[:m]
        grad[:m] -= h[a:]
    else:
        grad[:m] = 1.0 - h[a:]
        grad[a:] -= h[:m]
    return grad

def _lp_step(h, dx, n_top, solvers):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:n_top]
    active = []
    for ki in top:
        s = int(ki - (n - 1))
        if s == 0:
            continue
        g = _lag_grad(h, ki, n)
        active.append((g, c[ki]))
    if not active:
        return h, _c5(h, dx)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for g, ck in active:
        cons.append(ck + g @ (h_var - h) <= t_var / dx)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for sv in solvers:
        try:
            prob.solve(solver=sv, verbose=False)
            if h_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                nh = _repair(np.array(h_var.value).flatten(), n / 2.0)
                return nh, _c5(nh, dx)
        except Exception:
            continue
    return h, _c5(h, dx)

def _lp_loop(h, dx, t_end, n_top=5):
    n = len(h)
    best_h, best_c = h.copy(), _c5(h, dx)
    h = h.copy()
    stag = 0
    solvers = ['ECOS', 'GLPK', 'CBC', 'SCIP']
    while time.time() < t_end:
        nh, nc = _lp_step(h, dx, n_top, solvers)
        if nc < best_c - 1e-9:
            best_h, best_c, h = nh, nc, nh
            stag = 0
        elif nc < best_c:
            h = nh
            stag += 1
        else:
            stag += 1
            if stag >= 4:
                rng_s = np.random.RandomState(int(time.time() * 7919) % 100000)
                hp = _repair(h + 0.015 * (2.0 * rng_s.random(n) - 1.0), n / 2.0)
                nh2, nc2 = _lp_step(hp, dx, n_top, solvers)
                if nc2 < best_c:
                    best_h, best_c, h = nh2, nc2, nh2
                    stag = 0
                else:
                    for nt in [3, 8, 12]:
                        if nt == n_top:
                            continue
                        nh3, nc3 = _lp_step(h, dx, nt, solvers)
                        if nc3 < best_c:
                            best_h, best_c, h = nh3, nc3, nh3
                            stag = 0
                            break
                    else:
                        break
    return best_h, best_c

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj_f(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq_f(x):
        return x.sum() - tgt
    try:
        r = minimize(obj_f, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq_f}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _fine(h, dx, rng, t_end):

def _perturb(h, rng, mode):
    n = len(h)
    hp = h.copy()
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 5), replace=False)

    elif mode == 2:
        x = np.linspace(0, 2, n)

    else:
        idx = rng.choice(n, max(2, n // 8), replace=False)
        hp[idx] = 1.0 - hp[idx]
