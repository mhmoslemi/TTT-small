Let me add the analytic gradient for SLSQP. This should make it significantly faster and allow more iterations in the same time.

The gradient of the soft-max objective:
f(x) = m + (1/α) * log(∑ exp(α(c_k - m)))

where m = max_k c_k, c = correlate(x, 1-x, 'full')

df/dx = (1/α) * ∑_k exp(α(c_k - m))/∑_j exp(α(c_j - m)) * dc_k/dx
      = ∑_k w_k * dc_k/dx

where w_k = exp(α(c_k - m))/∑_j exp(α(c_j - m))

And dc_k/dx is the gradient of the k-th correlation, which I derived above.

Let me implement this:

```python
def _corr_grad(h, w):
    """Gradient of weighted sum of correlates: sum_k w[k] * corr_k"""
    n = len(h)
    grad = np.zeros(n)
    for ki in range(len(w)):
        wi = w[ki]
        if wi < 1e-15: continue
        s = ki - (n-1)
        if abs(s) >= n: continue
        if s == 0:
            grad += wi * (1.0 - 2.0*h)
        elif s > 0:
            m = n - s
            grad[s:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[s:]
        else:
            s2 = -s
            m = n - s2
            grad[:m] += wi * (1.0 - h[s2:])
            grad[s2:] -= wi * h[:m]
    return grad
```

This is O(n * number_of_nonzero_weights). With α=10000, most weights will be zero (underflow), so it's effectively O(n * n_active_lags) which is small.

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(_corr(h).max() * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _corr_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for ki in range(len(w)):
        wi = w[ki]
        if wi < 1e-14:
            continue
        s = ki - (n - 1)
        if abs(s) >= n:

        if s == 0:

        elif s > 0:

        else:
            s2 = -s
            m = n - s2
            grad[:m] += wi * (1.0 - h[s2:])

    return grad

    shifted = np.clip(alpha * (c - m), -500, 500)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    return m + np.log(exps.sum()) / alpha, w

    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n

        c = _corr(x)
        m = c.max()
        return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -500, 500)))) / alpha

        c = _corr(x)
        m = c.max()
        shifted = np.clip(alpha * (c - m), -500, 500)
        exps = np.exp(shifted)
        w = exps / exps.sum()
        return _corr_grad(x, w)

    def eq(x):
        return x.sum() - tgt

    try:
        r = minimize(obj, h, method='SLSQP', jac=obj_g, bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 40})
        return _repair(r.x, tgt)
    except Exception:
        return h

def _subgrad(h, n_lags=10):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:n_lags]
    cmax = c.max()
    w = np.exp(100.0 * (c[top] - cmax))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:
            continue
        m = n - s
        if ki >= n - 1:
            grad[s:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[s:]
        else:
            grad[:m] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:m]
    grad -= grad.mean()
    return grad
