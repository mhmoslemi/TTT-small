import numpy as np
import time
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lse(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g * dx

def _pgd(h, n, alpha, steps, step0, step_max, elapsed_fn, stop_el):
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    h = h.copy()
    f = _obj(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    stagnation = 0
    for _ in range(steps):
        if elapsed_fn() > stop_el:
            break
        g = _grad_lse(h, n, alpha)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj(h_new)
        if f_new < f - 1e-16:
            h = h_new
            f = f_new
            step = min(step * 1.04, step_max)
            stagnation = 0
        else:
            step *= 0.65
            if step < step0 * 0.1:
                step = step0
                v = 0.5 * v
                h_new = _proj_to_sum(h - step * v, n * 0.5)
                f_new = _obj(h_new)
                if f_new < f:
                    h = h_new
                    f = f_new
            stagnation += 1
            if stagnation > 200:
                break
    return h, f

def _water_fill(h, n, top_m, n_rounds, elapsed_fn, stop_el):
    dx = 2.0 / n
    h = h.copy()
    c5 = _eval(h)
    for _ in range(n_rounds):
        if elapsed_fn() > stop_el:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[::-1][:top_m]
        top_thresh = c[top_idx[min(top_m, len(c)-1)]-1]
        g = np.zeros(n)
        for k in range(len(c)):
            if c[k] < top_thresh:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                break
    return h, c5

def _pairwise_descent(h, n, n_tries, rng, elapsed_fn, stop_el):
    h = h.copy()
    c5 = _eval(h)
    steps = [1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
    for _ in range(n_tries):
        if elapsed_fn() > stop_el:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in steps:
            for sign in [1.0, -1.0]:
                dd = d * sign
                ni = h[i] + dd
                nj = h[j] - dd
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5 - 1e-15:
                        c5 = c5n
                    else:
                        h[i], h[j] = ni - dd, nj + dd
    return h, c5

def _slsqp_polish(h, n, alpha, maxiter, elapsed_fn, stop_el):
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        return _grad_lse(x, n, alpha)
    bnds = [(0.0, 1.0)] * n
    def _cons_fun(x):
        return x.sum() - n * 0.5
    def _cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _perturb(h, n, rng, mode):
    hp = h.copy()
    if mode == 0:
        hp += 0.02 * rng.randn(n)
    elif mode == 1:
        hp = np.roll(hp, rng.randint(1, n // 4))
    elif mode == 2:
        hp = 1.0 - hp
    elif mode == 3:
        t_arr = np.arange(n)
        f = rng.randint(1, n // 4 + 1)
        amp = 0.03
        hp += amp * np.sin(2 * np.pi * f * t_arr / n + rng.rand() * 2 * np.pi)
    elif mode == 4:
        bs = rng.randint(n // 6, n // 3)
        st = rng.randint(0, n - bs)
        hp[st:st+bs] = 1.0 - hp[st:st+bs]
    elif mode == 5:
        hp = np.maximum(hp * 0.9, 0.0)
    else:
        hp += 0.05 * rng.randn(n)
    return _repair(hp, n)
