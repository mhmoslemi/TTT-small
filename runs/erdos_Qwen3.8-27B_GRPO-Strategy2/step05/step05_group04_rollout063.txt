def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        ones_p = np.where(hb == 1)[0]
        zeros_p = np.where(hb == 0)[0]
        if rng.rand() < 0.75:
            i = ones_p[rng.randint(len(ones_p))]
            j = zeros_p[rng.randint(len(zeros_pos))] if False else zeros_p[rng.randint(len(zeros_p))]
            hb[i], hb[j] = 0, 1
            c_new = _eval(hb)
            if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(T, 1e-8)):
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_hb = hb.copy()
            else:
                hb[i], hb[j] = 1, 0
        else:
            idxs = rng.choice(n, 4, replace=False)
            vals = hb[idxs].copy()
            if vals.sum() < 1e-9 or vals.sum() > 2.0 + 1e-9:
                continue
            for perm in [(0,1,2,3),(0,2,1,3),(1,0,2,3),(1,2,0,3),(2,0,1,3),(2,1,0,3)]:
                new_vals = np.zeros(4)
                new_vals[[0,1]] = min(vals.sum(), 2.0)
                new_vals[[2,3]] = 2.0 - new_vals[[0,1]].sum() if new_vals[[0,1]].sum() <= 2.0 else 0
                break
            i1, j1 = idxs[np.where(vals==1)[0][:2]] if (vals==1).sum()>=2 else (idxs[0], idxs[1])
            i2, j2 = idxs[np.where(vals==0)[0][:2]] if (vals==0).sum()>=2 else (idxs[2], idxs[3])
            if i1 == i2 or i1 == j2 or j1 == i2:
                continue
            hb[i1], hb[j1], hb[i2], hb[j2] = 0, 0, 1, 1
            c_new = _eval(hb)
            if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(T, 1e-8)):
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_hb = hb.copy()
            else:
                hb[i1], hb[j1], hb[i2], hb[j2] = 1, 1, 0, 0
    return best_hb, best_c

def _peel_peaks(h, n, rng, max_iter):
    """Make poorly-placed 1s slightly fractional to reduce peak lags"""
    best_h = h.copy()
    best_c5 = _eval(h)
    for _it in range(max_iter):
        if time.time() - _TSTART > budget_s_global - 80:
            break
        c = _eval_vec(best_h)
        k_w = c.argmax()
        sh = (n - 1) - k_w
        # Find entries contributing most to this lag
        contrib = np.zeros(n)
        if sh >= 0:
            for i in range(n - sh):
                contrib[i] += h[i+sh] * (1 - h[i]) if False else (h[i+sh] * (1-h[i]))
                if sh > 0:
                    contrib[i+sh] += (1 - h[i+sh]) * h[i]
        else:
            sp = -sh
            for i in range(sp, n):
                contrib[i] += h[i-sp] * (1 - h[i])
                contrib[i-sp] += h[i-sp] * (1-h[i]) * 0
        top = np.argsort(contrib)[-min(5, n):]
        idx = top[rng.randint(len(top))]
        if best_h[idx] > 0.5:
            best_h[idx] = max(0, best_h[idx] - 0.05 * rng.rand())
        else:
            best_h[idx] = min(1, best_h[idx] + 0.05 * rng.rand())
        best_h = _repair(best_h, n)
        c5n = _eval(best_h)
        if c5n > best_c5 + 0.001:
            best_h = _repair(best_h, n)
    return best_h, best_c5

_TSTART = 0.0
budget_s_global = 1100.0

def run(seed=42, budget_s=1100, **kwargs):
    global _TSTART, budget_s_global
    _TSTART = time.time()
    budget_s_global = budget_s
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - _TSTART

    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-restart binary SA
    for restart in range(6):
        if _el() > 350:
            break
        if restart == 0:
            hb = _make_binary(best_h, n0, rng, 0.5)
        elif restart == 1:
            hb = _make_binary(best_h, n0, rng, 0.6)
        elif restart == 2:
            hb = _make_binary(best_h, n0, rng, 0.4)
        else:
            hb = (best_h + 0.05 * rng.randn(n0) > 0.5).astype(float)
            n_ones = int(hb.sum())
            tgt = n0 // 2
            if n_ones != tgt:
                ones_p = np.where(hb==1)[0]
                zeros_p = np.where(hb==0)[0]
                if n_ones > tgt:
                    for idx in rng.choice(ones_p, n_ones-tgt, replace=False):
                        hb[idx]=0
                else:
                    for idx in rng.choice(zeros_p, tgt-n_ones, replace=False):
                        hb[idx]=1
        c_b = _eval(hb)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()
        hb, c_b = _binary_sa(hb, n0, rng, 2500, 0.4 + 0.1*restart, 0.01)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()

    # Phase 2: SLSQP progressive alpha
    h = _repair(best_h, n0)
    for alpha in [5000, 50000, 300000]:
        if _el() > budget_s - 250:
            break
        h_cand = _slsqp(h, alpha, 2500, 1e-18)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()
            h = h_cand

    # Phase 3: Projected gradient descent with high alpha
    if _el() < budget_s - 150:
        h_cand = h.copy()
        step = 0.003
        f = _smooth_obj(h_cand, 200000)
        for _s in range(800):
            if _el() > budget_s - 150:
                break
            g = _smooth_grad(h_cand, 200000)
            h_new = _proj_to_sum(h_cand - step * g, n0 * 0.5)
            f_new = _smooth_obj(h_new, 200000)
            if f_new < f:
                h_cand = h_new
                f = f_new
                step = min(step * 1.05, 0.02)
            else:
                step *= 0.5
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 4: SLSQP high-alpha polish
    if _el() < budget_s - 100:
        h_cand = _slsqp(best_h, 1000000, 4000, 1e-20)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 5: Targeted worst-lag descent
    if _el() < budget_s - 50:
        h = best_h.copy()
        c5_cur = _eval(h)
        for _round in range(300):
            if _el() > budget_s - 50:
                break
            c = _eval_vec(h)
            k_w = c.argmax()
            sh = (n0 - 1) - k_w
            g = np.zeros(n0)
            if sh >= 0:
                i1 = np.arange(n0 - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n0)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n0)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n0 - sp)
                g[i2] -= h[i2 + sp]
            improved = False
            for st in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_new = _proj_to_sum(h - st * g, n0 * 0.5)
                c5n = _eval(h_new)
                if c5n < c5_cur - 1e-14:
                    h = h_new
                    c5_cur = c5n
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break

    # Phase 6: Ultra-fine pairwise search
    if _el() < budget_s - 5:
        h = best_h.copy()
        c5c = _eval(h)
        remaining = max(100, int((budget_s - 5 - _el()) / 0.004))
        for _it in range(remaining):
            if _el() > budget_s - 5:
                break
            i = rng.randint(n0)
            j = rng.randint(n0)
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:
                continue
            h[i], h[j] = max(0,min(1,ni)), max(0,min(1,nj))
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i] = ni - d
                h[j] = nj + d

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
