    """Gradient contribution from shift s (signed: positive = forward)"""
    g = np.zeros(n)

        if s > 0:
            g[s:] -= h[:n-s]
    else:

        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

    g = np.zeros(n)

        g += w * _grad_at_shift(h, s, n)
    return g
