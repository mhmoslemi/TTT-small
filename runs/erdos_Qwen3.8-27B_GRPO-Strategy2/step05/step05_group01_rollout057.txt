def _pos_change(h, i, di, n):
    """Compute change in correlation when h[i] changes by di."""

    if di == 0.0:
        return dc
    # h[i] increases by di
    # For shift sh >= 0 (index n-1-sh):
    #   if i < n-sh: c += di*(1-h[i+sh])
    #   if i >= sh: c -= di*h[i-sh]

def _pos_change_vec(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    # Positive shifts: k = n-1-sh, sh=0..n-1
    # dc[n-1-sh] contribution:
    # sh=0: di*(1-2*h[i])
    # sh>0, i < n-sh: di*(1-h[i+sh])
    # sh>0, i >= sh: -di*h[i-sh]
    dc[n-1] = di * (1.0 - 2.0 * h[i])
    # For sh=1..n-1-i: dc[n-1-sh] += di*(1-h[i+sh])
    if i < n - 1:
        shs = np.arange(1, n - i)
        dc[n - 1 - shs] += di * (1.0 - h[i + shs])
    # For sh=1..i: dc[n-1-sh] -= di*h[i-sh]
    if i > 0:
        shs = np.arange(1, i + 1)
        dc[n - 1 - shs] -= di * h[i - shs]
    # Negative shifts: k = n-1+sh, sh=1..n-1
    # dc[n-1+sh]:
    # i >= sh: di*(1-h[i-sh])
    # i+sh < n: -di*h[i+sh]
    if i > 0:
        shs = np.arange(1, i + 1)
        dc[n - 1 + shs] += di * (1.0 - h[i - shs])
    if i < n - 1:
        shs = np.arange(1, n - i)
        dc[n - 1 + shs] -= di * h[i + shs]
    return dc

    lo, hi = -2.0, 2.0

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(h, 0.0, 1.0).copy()

    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
