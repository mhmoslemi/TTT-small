    h *= target / s

        return h

    dc[(n-1) - np.arange(a)] += di * (1.0 - h[i:i+a])

    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    for alpha in alphas:

            h[j] = nj_val
            c = new_c

One more idea: **coordinate descent with exact 1D optimization**. Fix all variables except h[i], and find the optimal h[i] that minimizes the max correlation. This is a 1D problem:
- Changing h[i] by δ changes c by δ * g_i (where g_i = _pos_change(h,i,1,n))
- We want to minimize max_k (c[k] + δ * g_i[k])
- This is a piecewise linear function of δ (the max of linear functions)
- The minimum is at a "kink" where two linear functions cross
- We can binary search for the optimal δ!

This is O(n log n) per coordinate update (O(n) to compute g_i, O(n) to find the kink). For n=280, that's about 280*9 = 2520 ops per coordinate. For 280 coordinates, one full pass is 700k ops ≈ 0.07s. We can do 1000 passes in 70s!

def _coord_opt(h, c, n, i, t0, deadline):
    """Find optimal delta for h[i] minimizing max(c + delta * g_i)"""
    g_i = _pos_change(h, i, 1.0, n)
    # c_new(d) = c + d * g_i
    # Want to minimize max(c_new(d)) subject to 0 <= h[i]+d <= 1
    # c_new(d) is max of linear functions: c[k] + d*g_i[k] for each k
    # The minimum of max of linear functions is at a vertex (intersection of two lines)
    # Binary search on d:
    # f(d) = max_k (c[k] + d*g_i[k])
    # f(d) is convex in d, so we can binary search for where f'(d) = 0
    
    # f'(d) = g_i[k*] where k* = argmax_k (c[k] + d*g_i[k])
    # At optimum, either h[i]=0 or h[i]=1 (boundary) or g_i[k*] = 0 (flat)
    
    # Simpler: ternary search or golden section on d in [min_d, max_d]
    hi_d = 1.0 - h[i]
    lo_d = -h[i]
    
    # f is convex, use golden section
    invphi = (np.sqrt(5) - 1) / 2
    a, b = lo_d, hi_d
    c_pt = b - invphi * (b - a)
    d_pt = a + invphi * (b - a)
    fc = (c + c_pt * g_i).max()
    fd = (c + d_pt * g_i).max()
    
    for _ in range(60):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - invphi * (b - a)
            fc = (c + c_pt * g_i).max()
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + invphi * (b - a)
            fd = (c + d_pt * g_i).max()
        if time.time() - t0 > deadline:
            break
    
    best_d = (a + b) / 2
    best_c5 = (c + best_d * g_i).max()
    hi_c5 = c.max()
    if best_c5 < hi_c5 - 1e-15:
        h[i] = h[i] + best_d
        c = c + best_d * g_i
    return h, c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-14:
            break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    a = n - i
    dc[(n-1) - np.arange(a)] += di * (1.0 - h[i:i+a])
    b = i + 1
    dc[(n-1) - np.arange(b)] -= di * h[i::-1][:b]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < a
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    def _obj(x):
        cx = _corr(x)
        mx = cx.max()
        sx = np.clip(alpha * (cx - mx), -500, 500)
        return mx + np.log(np.sum(np.exp(sx))) / alpha
    def _grad(x):
        cx = _corr(x)
        mx = cx.max()
        sx = np.clip(alpha * (cx - mx), -500, 500)
        ww = np.exp(sx - sx.max())
        ww /= ww.sum()
        g = np.zeros(n)
        for sv in range(n):
            wk = ww[n - 1 + sv]
            if wk < 1e-15: continue
            g[:n-sv] += wk * (1.0 - x[sv:])
            g[sv:] -= wk * x[:n-sv]
        for sv in range(1, n):
            wk = ww[n - 1 - sv]
            if wk < 1e-15: continue
            g[sv:] += wk * (1.0 - x[:n-sv])
            g[:n-sv] -= wk * x[sv:]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, c, rng, n, n_iter, scale, t0, deadline):
    dx = 2.0 / n
    c5 = c.max() * dx
    for _ in range(n_iter):
        if time.time() - t0 > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-8 or ni > 1.0 + 1e-8 or nj < -1e-8 or nj > 1.0 + 1e-8:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5 - 1e-16:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5 = new_c5
    return h, c, c5

def _coord_step(h, c, n, i, t0, deadline):
    g_i = _pos_change(h, i, 1.0, n)
    lo_d = -h[i]
    hi_d = 1.0 - h[i]
    if hi_d - lo_d < 1e-14:
        return h, c
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_d, hi_d
    c_pt = b - invphi * (b - a)
    d_pt = a + invphi * (b - a)
    fc = (c + c_pt * g_i).max()
    fd = (c + d_pt * g_i).max()
    for _ in range(40):
        if time.time() - t0 > deadline:

        if fc < fd:
            b = d_pt; d_pt = c_pt; fd = fc
            c_pt = b - invphi * (b - a)
            fc = (c + c_pt * g_i).max()
        else:
            a = c_pt; c_pt = d_pt; fc = fd
            d_pt = a + invphi * (b - a)
            fd = (c + d_pt * g_i).max()
    best_d = (a + b) * 0.5
    best_val = (c + best_d * g_i).max()
    if best_val < c.max() - 1e-15:
        h[i] = h[i] + best_d
        c = c + best_d * g_i
    return h, c
