def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:

    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _gm(h, sh, n, dx):
    g = np.zeros(n)
    if sh > 0:
        c = n - sh
        g[:c] += dx * (1.0 - h[sh:sh+c])
        g[sh:] -= dx * h[:c]
    elif sh < 0:
        c = n + sh
        g[-sh:] += dx * (1.0 - h[:c])
        g[:c] -= dx * h[-sh:]
    return g

def _topk_desc(h, n, k, lr0, tlim):
    dx = 2.0 / n
    h = h.copy()
    c5 = _eval(h)
    lr = lr0
    t0 = time.time()
    stall = 0
    while time.time() - t0 < tlim:
        cf = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(cf)[-k:]

            g += _gm(h, (n-1)-idx, n, dx)

        g -= g.mean()

Another insight: for a binary sequence with n/2 ones, the overlap at shift k is:
O(k) = |{i : h[i]=1, h[i+k]=0}| for k ≥ 0
O(-k) = |{i : h[i]=0, h[i+k]=1}| for k ≥ 0

And O(k) + O(-k) = |{i : h[i] ≠ h[i+k]}| (the number of disagreements at distance k).

The total number of (i,j) pairs with i<j and h[i]≠h[j] is (n/2)(n/2) = n²/4.

So sum_k (O(k) + O(-k)) over k=1 to n-1 = n²/4 (each pair (i,j) with h[i]≠h[j] is counted exactly once for k=j-i).

This means: (1/(2(n-1))) * sum_k (O(k)+O(-k)) = n/(4(n-1)) ≈ 1/4.

        return np.full(n, 0.5)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _eval(h):

        g[:c] += dx * (1.0 - h[sh:sh+c])

    elif sh < 0:
