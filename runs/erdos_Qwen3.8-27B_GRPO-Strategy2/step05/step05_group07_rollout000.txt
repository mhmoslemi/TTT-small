    g = np.zeros(n)
    for k, w in zip(lags, weights):
        g += w * _grad_lag(k, h)
    return g * (2.0 / n)

def _momentum_grad(h, topm, max_iters, step0, mom):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    v = np.zeros(n)
    f_prev = _eval(h)
    step = step0
    stall = 0
    for it in range(max_iters):
        if time.time() - _t0 > _budget - 12:
            break
        c = _corrs(h)
        idx = np.argsort(c)[::-1][:max(topm, 3)]
        c_max = c[idx[0]]
        c_flo = c[idx[-1]]
        span = c_max - c_flo
        if span < 1e-10:
            wts = np.ones(len(idx))
        else:
            wts = 0.3 + 0.7 * (c[idx] - c_flo) / span
        wts /= wts.sum()
        g = _grad_weighted(h, idx, wts)
        v = mom * v + (1.0 - mom) * g
        found = False
        for _ in range(35):
            h_new = _proj_sum(h - step * v, target)
            f_new = _eval(h_new)
            if f_new < f_prev - 1e-14:
                h = h_new
                f_prev = f_new
                found = True
                stall = 0
                break
            step *= 0.5
        if not found:
            stall += 1
            v *= 0.5
            if stall > 15:
                break
        else:
            step = min(step * 1.03, step0)
    return h, f_prev

def _micro_descent(h, topm, max_iters, step0):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    for _ in range(max_iters):
        if time.time() - _t0 > _budget - 8:
            break
        c = _corrs(h)
        idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(k, h)
        g *= (2.0 / (n * topm))
        found = False
        for step in [step0, step0*0.5, step0*0.25, step0*0.1, step0*0.04]:
            h_new = _proj_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-15:
                h = h_new; f = f_new; found = True
                break
        if not found:
            break
    return h, f

def _upsample_spline(h, factor):
    n = len(h)
    x = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    fi = interp1d(x, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(fi(x_new), 0, 1), n * factor)

def _slsqp_smooth(h, alpha, maxiter):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
        m = c.max()
        s = np.clip(alpha*(c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
        m = c.max()
        s = np.clip(alpha*(c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-14: continue
            sh = (n-1) - k
            wk = w[k] * (2.0/n)
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1+sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2-sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1-sp])
                i2 = np.arange(n-sp)
                g[i2] -= wk * x[i2+sp]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - _t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Momentum gradient at original resolution
    for topm, mom, iters, step in [(5, 0.6, 4000, 0.006), (10, 0.65, 3000, 0.004), (20, 0.7, 2500, 0.003)]:
        if _el() > budget_s * 0.20:
            break
        hn, fc = _momentum_grad(best_h, topm, iters, step, mom)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase 2: Micro-descent on top few lags
    if _el() < budget_s * 0.30:
        for topm, iters, st in [(3, 5000, 0.002), (5, 4000, 0.001), (3, 6000, 0.0005)]:
            if _el() > budget_s * 0.28:
                break
            hn, fc = _micro_descent(best_h, topm, iters, st)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase 3: SLSQP at original resolution
    if _el() < budget_s * 0.42:
        for alpha in [10000, 80000]:
            if _el() > budget_s * 0.40:
                break
            hn = _slsqp_smooth(best_h, alpha, 1200)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy()

    # Phase 4: Upsample to 2x and optimize
    n1 = n0 * 2
    if _el() < budget_s * 0.45:
        h2 = _upsample_spline(best_h, 2)
        c5_2 = _eval(h2)
        # Quick SLSQP at higher res
        if _el() < budget_s * 0.60:
            hn = _slsqp_smooth(h2, 30000, 800)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy(); best_n = n1
        # Momentum gradient at higher res
        for topm, iters, st in [(5, 3000, 0.003), (10, 2500, 0.002)]:
            if _el() > budget_s * 0.72:
                break
            hn, fc = _momentum_grad(best_h, topm, iters, st, 0.65)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy(); best_n = n1
        # Micro at higher res
        if _el() < budget_s * 0.75:
            hn, fc = _micro_descent(best_h, 3, 5000, 0.001)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy(); best_n = n1

    # Phase 5: Try 4x upsampling if time permits
    if _el() < budget_s * 0.55 and best_n == n0:
        n4 = n0 * 4
        h4 = _upsample_spline(best_h, 4)
        if _el() < budget_s * 0.70:
            hn, fc = _momentum_grad(h4, 5, 2000, 0.002, 0.6)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy(); best_n = n4
            hn, fc = _micro_descent(best_h, 3, 3000, 0.0005)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy(); best_n = n4

    # Phase 6: Multi-start (structured)
    nc = len(best_h)
    if _el() < budget_s * 0.82:
        # Phase shifts
        for sh in [1, 2, 3, -1, -2]:
            if _el() > budget_s * 0.80:
                break
            hp = np.roll(best_h, sh)
            hp = _repair(hp, nc)
            hnp, fcp = _momentum_grad(hp, 8, 1500, 0.003, 0.6)
            if fcp < best_c5:
                best_c5 = fcp; best_h = hnp.copy(); best_n = nc
        # Random perturbations
        for amp in [0.015, 0.008, 0.004]:
            if _el() > budget_s * 0.80:
                break
            for _ in range(4):
                if _el() > budget_s * 0.80:
                    break
                hp = best_h + amp * rng.randn(nc)
                hp = _repair(hp, nc)
                hnp, fcp = _momentum_grad(hp, 8, 1200, 0.003, 0.6)
                if fcp < best_c5:
                    best_c5 = fcp; best_h = hnp.copy(); best_n = nc

    # Phase 7: Final micro-polish
    if _el() < budget_s * 0.93:
        for topm, iters, st in [(3, 8000, 0.0008), (5, 6000, 0.0004), (3, 10000, 0.0003)]:
            if _el() > budget_s * 0.92:
                break
            hn, fc = _micro_descent(best_h, topm, iters, st)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase 8: Ultra-fine final steps
    if _el() < budget_s * 0.97:
        hn, fc = _micro_descent(best_h, 2, 10000, 0.0001)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
        hn, fc = _micro_descent(best_h, 3, 8000, 0.0001)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
