import numpy as np

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) * 0.5, 0, 1)
        # Fine adjustment
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

    """Find best di for moving h[i]+=di, h[j]-=di"""
    n = len(h)
    # Feasible range
    di_lo = max(-h[i], h[j] - 1.0) - 1e-12
    di_hi = min(1.0 - h[i], h[j]) + 1e-12
    if di_lo >= di_hi:
        return 0.0
    
    # For top active lags, compute d_k = grad_k[i] - grad_k[j]
    # We want to minimize max(c[k] + di * d_k)
    # Try: di that levels the top-2 active lags
    c_ak = c[active[:top_k]]
    # Find the two highest
    idx_sorted = np.argsort(c_ak)[::-1]
    k1 = idx_sorted[0]
    k2 = idx_sorted[1] if top_k > 1 else k1

    s1 = active[k1] - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        m1 = np.arange(s, n)
        g[m1] += (1.0 - h[m1 - s])
        m2 = np.arange(0, n - s)
        g[m2] -= h[m2 + s]
    else:
        p = -s
        m1 = np.arange(0, n - p)
        g[m1] += (1.0 - h[m1 + p])
        m2 = np.arange(p, n)
        g[m2] -= h[m2 - p]
    return g

def _topk_obj(h, active, k, alpha):
    c = _corr(h)
    ak = c[active[:k]]
    m = ak.max()
    sv = np.clip(alpha * (ak - m), -500, 500)
    return m + np.log(np.sum(np.exp(sv))) / alpha

def _topk_grad(h, active, k, alpha):
    n = len(h)
    c = _corr(h)
    ak = c[active[:k]]
    m = ak.max()
    sv = np.clip(alpha * (ak - m), -500, 500)
    w = np.exp(sv - sv.max())
    w /= w.sum()
    g = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-15:
            continue
        sv_ = active[i] - (n - 1)
        g += w[i] * _grad_lag(h, sv_, n)
    return g

def _sum_fun(h):
    return h.sum() - len(h) * 0.5

def _sum_jac(h):
    return np.ones(len(h))

def _slsqk_topk(h, active, k, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_topk_obj, h, jac=_topk_grad, args=(active, k, alpha),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = _repair(res.x, n)
    except Exception:
        h_out = _repair(h.copy(), n)
    return h_out

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    s_lo1 = max(i - (n - 1), -(n - 1))
    s_hi1 = min(i, n - 1)
    if s_lo1 <= s_hi1:
        sv = np.arange(s_lo1, s_hi1 + 1)
        dc[(n - 1) + sv] += di * (1.0 - h[i - sv])
    s_lo2 = max(-i, -(n - 1))
    s_hi2 = min(n - 1 - i, n - 1)
    if s_lo2 <= s_hi2:
        sv = np.arange(s_lo2, s_hi2 + 1)
        dc[(n - 1) + sv] -= di * h[i + sv]
    return dc

def _incr(h, c, rng, n_iter, scale, n):
    cm = c.max()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        nm = (c + dc).max()
        if nm < cm - 1e-14:
            h[i] = ni
            h[j] = nj
            c = c + dc
            cm = nm
    return h, c, cm

def _active_set(h):
    c = _corr(h)
    n = len(h)
    return np.argsort(c)[::-1]

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def _pg_step(h, alpha, n_act, step0, n_steps, rng):
    n = len(h)
    c = _corr(h)
    step = step0
    for _ in range(n_steps):
        act = _active_set(h)[:n_act]
        ak = c[act]
        m = ak.max()
        sv = np.clip(alpha * (ak - m), -500, 500)
        w = np.exp(sv - sv.max())
        w /= w.sum()
        g = np.zeros(n)
        for i in range(n_act):
            if w[i] < 1e-14:
                continue
            sv_ = act[i] - (n - 1)
            g += w[i] * _grad_lag(h, sv_, n)
        f0 = m + np.log(np.sum(np.exp(sv))) / alpha
        for _bt in range(40):
            h_new = _proj_sum(h - step * g, n * 0.5)
            c_new = _corr(h_new)
            ak_new = c_new[act]
            m_new = ak_new.max()
            sv_new = np.clip(alpha * (ak_new - m_new), -500, 500)
            f_new = m_new + np.log(np.sum(np.exp(sv_new))) / alpha
            if f_new < f0 - 1e-7 * step * np.dot(g, h_new - h):
                h = h_new
                c = c_new
                f0 = f_new
                break
            step *= 0.5
        else:
            break
        step = min(step * 1.05, step0)
    return h
