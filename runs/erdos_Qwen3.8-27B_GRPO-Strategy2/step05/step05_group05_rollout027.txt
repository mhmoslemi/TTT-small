def _one_dim_minmax(d_lo, d_hi, c0, g, n_k):
    """Find d in [d_lo, d_hi] minimizing max_k (c0[k] + g[k]*d)"""
    # Start at d=0 (clamped to [d_lo, d_hi])
    d = np.clip(0.0, d_lo, d_hi)
    
    for _ in range(n_k * 2 + 10):  # max iterations
        vals = c0 + g * d
        active = np.where(vals >= vals.max() - 1e-12)[0]
        if len(active) == 0:
            active = np.argmax(vals: np.array([np.argmax(vals)])
        g_active = g[active]
        if g_active.max() < -1e-12:
            # All active lines decreasing: increase d
            # Find where the next line crosses the current max
            current_max = vals.max()
            # For inactive lines, find where they cross current_max
            inactive = np.setdiff1d(np.arange(n_k), active)
            # Line k crosses current_max at d_k = (current_max - c0[k]) / g[k] (if g[k] > 0)
            # We want the smallest such d_k > d
            cross_points = []
            for ik in inactive:
                if g[ik] > 1e-12:
                    d_k = (current_max - c0[ik]) / g[ik]
                    if d > d_k >= d_lo:
                        cross_points.append(d_k)
            if cross_points:
                d = max(d + 1e-15, min(cross_points))
            else:
                d = d_hi
        elif g_active.min() > 1e-12:
            # All active lines increasing: decrease d
            current_max = vals.max()
            inactive = np.setdiff1d(np.arange(n_k), active)

            else:
                d = d_lo
        else:

            break
    
    d = np.clip(d, d_lo, d_hi)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    n = len(h)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s.max()
    return (m + np.log(np.sum(np.exp(s - sm)) + 1e-300) / alpha + sm / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    sh = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * (2.0 / n)
        ib = np.arange(n) - sh[k]
        v1 = (ib >= 0) & (ib < n)
        ib2 = np.arange(n) + sh[k]
        v2 = (ib2 >= 0) & (ib2 < n)
        g[v1] += wk * (1.0 - h[ib[v1]])
        g[v2] -= wk * h[np.arange(n)[v2]]
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return x.sum() - n * 0.5
    def _cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

    lo, hi = -2.0, 2.0

        s = np.clip(h - mid, 0, 1).sum()
        if s > target: lo = mid

        y = h_new + (beta - 1.0) / beta_new * (h_new - h)
        h = h_new
        beta = beta_new
    return _repair(h, n)

    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
