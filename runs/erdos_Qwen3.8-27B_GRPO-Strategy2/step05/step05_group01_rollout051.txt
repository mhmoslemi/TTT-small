import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-8:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) > 1e-10:
            idx = np.argsort(-(h if d < 0 else 1.0 - h))
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a
                    d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a
                    d += a
                if abs(d) < 1e-10:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _scf(x):
    return x.sum() - len(x) * 0.5

def _scj(x):
    return np.ones(len(x))

def _smo(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / a

def _smg(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, a, mi, ft):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scf, 'jac': _scj}]
    try:
        r = minimize(lambda x: _smo(x, a), h, jac=lambda x: _smg(x, a),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': ft, 'disp': False})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _pos_chg(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    if i >= 1:
        shs = np.arange(0, i + 1)
        dc[n - 1 - shs] += di * (1.0 - h[i - shs])
    shs2 = np.arange(0, n - i)
    dc[n - 1 - shs2] -= di * h[i + shs2]
    if i <= n - 2:
        sps = np.arange(1, n - i)
        dc[n - 1 + sps] += di * (1.0 - h[i + sps])
    if i >= 1:
        sps2 = np.arange(1, i + 1)
        dc[n - 1 + sps2] -= di * h[i - sps2]
    return dc

def _coord_descent(h, c, rng, n_iter, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        hi = h[i]
        hj = h[j]
        if i == j:
            continue
        candidates = []
        dm = min(hj, 1.0 - hi)
        dn = min(hi, 1.0 - hj)
        if dm > 1e-10:
            candidates.extend([dm, dm * 0.5, dm * 0.1, 0.01, 0.005, 0.001])
        if dn > 1e-10:
            candidates.extend([-dn, -dn * 0.5, -dn * 0.1, -0.01, -0.005, -0.001])
        for d in candidates:
            ni = hi + d
            nj = hj - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            dc = _pos_chg(h, i, d, n) + _pos_chg(h, j, -d, n)
            new_c5 = (c + dc).max() * dx
            if new_c5 < c5c - 1e-14:
                h[i] = ni
                h[j] = nj
                c = c + dc
                c5c = new_c5
                break
    return h, c, c5c

def _binary_round(h, n):
    sorted_idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[sorted_idx[:n // 2]] = 1.0
    return hb
