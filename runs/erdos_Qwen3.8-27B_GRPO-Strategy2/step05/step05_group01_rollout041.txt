import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smo_obj(h, a, tl):
    c = _corr(h)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    v = m + np.log(np.sum(np.exp(s))) / a
    if tl is not None and len(tl) > 0:
        v += 0.5 * float(np.sum(np.maximum(c[tl] - m + 1e-3, 0.0) ** 2))
    return v

def _smo_grad(h, a, tl):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    active = set(tl.tolist()) if tl is not None else set()
    for k in range(2 * n - 1):
        if w[k] < 1e-14 and k not in active:
            continue
        coef = w[k]
        if k in active:
            coef += np.maximum(c[k] - m + 1e-3, 0.0)
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += coef * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= coef * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += coef * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= coef * h[i2 + sp]
    return g

def _lbfgs(h, a, tl, maxit):

        res = minimize(_smo_obj, h, jac=_smo_grad, args=(a, tl),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 40, 'maxcor': 25})
        return _repair(res.x, n)
    except Exception:
        return h

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _incr_pair(h, c, rng, n_iter, scale, n):
    dx = 2.0 / n
    c5c = c.max() * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5c - 1e-15:
            h[i] = ni; h[j] = nj
            c = new_c; c5c = new_c5
    return h, c, c5c

def _binary_flip(h, rng, n, group_size):
    c5c = _eval(h)
    c = _corr(h)
    start = rng.randint(0, n - group_size + 1)
    idx = np.arange(start, start + group_size)
    flip = (h[idx] > 0.5).astype(float)
    new_vals = 1.0 - flip
    diff = new_vals - h[idx]
    dc = np.zeros(2 * n - 1)
    for ii in idx:
        dc += _pos_change(h, int(ii), diff[list(idx).index(ii)], n)
    new_c = c + dc
    new_c5 = new_c.max() * (2.0 / n)
    if new_c5 < c5c - 1e-12:
        h[idx] = new_vals
        return h
    return h

    n_c = len(c)

    order = np.argsort(c)[::-1]
