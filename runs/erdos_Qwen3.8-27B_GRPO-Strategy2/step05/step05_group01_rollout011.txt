    best_c5 = _eval_c5(best_h)
    best_n = n0

    def elapsed():
        return time.time() - t_start

    # Phase 1: SLSQP polish
    for alpha in [3000, 30000, 150000]:
        if elapsed() > 45:
            break
        h_new = _slsqp(best_h, alpha, 2000, 1e-18)
        c5 = _eval_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Main SA with wide exploration
    if elapsed() < budget_s - 400:
        h_sa = best_h.copy()
        c_sa = _corr_full(h_sa)
        h_sa, c_sa = _sa_phase(h_sa, c_sa, rng, t_start,
                               t_start + budget_s - 250, 0.8, 0.0003, 0.06)
        h_sa = _repair(h_sa, n0)
        c5 = _eval_c5(h_sa)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sa.copy()

    # Phase 3: Multi-scale greedy pairwise
    if elapsed() < budget_s - 250:
        h_ps = best_h.copy()
        c_ps = _corr_full(h_ps)
        scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]
        dl = t_start + budget_s - 200
        for sc in scales:
            if time.time() > dl:
                break
            h_ps, c_ps, _ = _incr_pair(h_ps, c_ps, rng, 45000, sc, n0, dl)
        h_ps = _repair(h_ps, n0)
        c5 = _eval_c5(h_ps)
        if c5 < best_c5:
            best_c5, best_h = c5, h_ps.copy()

    # Phase 4: Cooled SA + FFT perturbations for basin escaping
    if elapsed() < budget_s - 180:
        h_sa2 = best_h.copy()
        c_sa2 = _corr_full(h_sa2)
        dl2 = t_start + budget_s - 120
        cool_iters = 0
        while time.time() < dl2 and cool_iters < 300000:
            cool_iters += 1
            h_sa2, c_sa2 = _sa_phase(h_sa2, c_sa2, rng, t_start,
                                      min(time.time() + 15, dl2),
                                      0.15, 0.0001, 0.02)
            # Occasional FFT kick
            if cool_iters % 80000 == 0:
                h_fft = _fft_perturb(h_sa2, rng, 0.02)
                h_fft = _repair(h_fft, n0)
                c_fft = _corr_full(h_fft)
                if float(c_fft.max()) < float(c_sa2.max()):
                    c_sa2 = c_fft
                    h_sa2 = h_fft
                else:
                    h_sa2 = _repair(best_h + 0.005 * rng.randn(n0), n0)
                    c_sa2 = _corr_full(h_sa2)
        h_sa2 = _repair(h_sa2, n0)
        c5 = _eval_c5(h_sa2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_sa2.copy()

    # Phase 5: SLSQP high-precision polish
    if elapsed() < budget_s - 80:
        h_pol = _slsqp(best_h, 800000, 3000, 1e-20)
        c5 = _eval_c5(h_pol)
        if c5 < best_c5:
            best_c5, best_h = c5, h_pol.copy()

    # Phase 6: Ultra-fine pairwise
    if elapsed() < budget_s - 30:
        h_fine = best_h.copy()
        c_fine = _corr_full(h_fine)
        dl6 = t_start + budget_s - 25
        for sc in [3e-5, 1e-5, 5e-6, 2e-6]:
            if time.time() > dl6:
                break
            h_fine, c_fine, _ = _incr_pair(h_fine, c_fine, rng, 60000, sc, n0, dl6)
        h_fine = _repair(h_fine, n0)
        c5 = _eval_c5(h_fine)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fine.copy()

    # Phase 7: Multi-start exploration
    if elapsed() < budget_s - 90:
        dl7 = t_start + budget_s - 80
        for _r in range(5):
            if time.time() > dl7:
                break
            hp = _repair(best_h + 0.012 * rng.randn(n0), n0)
            h_ms = _slsqp(hp, 400000, 1500, 1e-18)
            c_ms = _corr_full(h_ms)
            h_ms, c_ms, _ = _incr_pair(h_ms, c_ms, rng, 35000, 0.004, n0, dl7)
            h_ms = _repair(h_ms, n0)
            c5 = _eval_c5(h_ms)
            if c5 < best_c5:
                best_c5, best_h = c5, h_ms.copy()

    # Final
    best_h = _repair(best_h, best_n)
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, best_n)
