        corr0 = np.correlate(h, 1.0 - h, mode='full')
        h[p] = 1.0
        corr1 = np.correlate(h, 1.0 - h, mode='full')
        h[p] = h[p]  # keep
        diffs = corr1 - corr0
        Tc = np.linspace(0.0, 1.0, n_grid_coarse)
        vals_c = corr0[np.newaxis, :] + Tc[:, np.newaxis] * diffs[np.newaxis, :]
        vals_c[:, n-1] += Tc * (1.0 - Tc)
        fc = vals_c.max(axis=1)
        ci = fc.argmin()
        lo = max(0.0, Tc[max(0, ci-2)] - window)
        hi = min(1.0, Tc[min(len(Tc)-1, ci+2)] + window)
        Tf = np.linspace(lo, hi, n_grid_fine) if hi > lo else np.array([lo])
        vals_f = corr0[np.newaxis, :] + Tf[:, np.newaxis] * diffs[np.newaxis, :]
        vals_f[:, n-1] += Tf * (1.0 - Tf)
        ff = vals_f.max(axis=1)
        fi = ff.argmin()
        h[p] = Tf[fi]
    return h

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _downsample(h, factor):
    n = len(h)
    n_new = n // factor
    h_new = np.zeros(n_new)
    for i in range(n_new):
        h_new[i] = h[i*factor:(i+1)*factor].mean()
    return h_new

def _upsample(h, factor):
    n = len(h)
    h_hi = np.repeat(h, factor)
    return h_hi

def _mom_proj(h, alpha, nsteps, step0, mu):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    step = step0
    for _ in range(nsteps):
        g = _smooth_grad(h, alpha)
        v = mu * v - step * g
        lam = 1.0
        improved = False
        for _bt in range(20):
            h_try = _proj_to_sum(h - lam * v, n * 0.5)
            fn = _smooth_obj(h_try, alpha)
            if fn < f - 1e-5 * lam * np.dot(v, h_try - h):
                h, f = h_try, fn
                improved = True
                break
            lam *= 0.5
        if not improved:
            h = _proj_to_sum(h - lam * v, n * 0.5)
            f = _smooth_obj(h, alpha)
        step = min(step * 1.02, step0 * 4.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)
    best_n = n0

    # Phase 1: Coordinate sweeps (fast local opt)
    for _s in range(80):
        if _el() > 120:
            break
        h = _coord_sweep(h, 300)
        h = _repair(h, n0)
    c5 = _eval(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: SLSQP with increasing alpha
    for alpha in [300, 2000, 10000, 60000]:
        if _el() > budget_s - 500:
            break
        h_new = _slsqp_opt(best_h, alpha, 1200, 1e-16)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        # Coordinate polish after each SLSQP
        for _s in range(30):
            if _el() > budget_s - 500:
                break
            h_new = _coord_sweep(h_new, 300)
            h_new = _repair(h_new, n0)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Multi-resolution coarse-to-fine
    if _el() < budget_s - 400:
        for n_coarse in [35, 70, 140]:
            if _el() > budget_s - 250:
                break
            factor = n0 // n_coarse
            h_c = _downsample(best_h, factor)
            h_c = _repair(h_c, n_coarse)
            # Optimize at coarse level
            for _s in range(40):
                if _el() > budget_s - 250:
                    break
                h_c = _coord_sweep(h_c, 200)
                h_c = _repair(h_c, n_coarse)
            h_c = _slsqp_opt(h_c, 5000, 600, 1e-14)
            h_c = _repair(h_c, n_coarse)
            for _s in range(30):
                if _el() > budget_s - 250:
                    break
                h_c = _coord_sweep(h_c, 200)
                h_c = _repair(h_c, n_coarse)
            # Upsample back
            h_up = _upsample(h_c, factor)
            h_up = _repair(h_up, n0)
            c5 = _eval(h_up)
            if c5 < best_c5:
                best_c5, best_h = c5, h_up.copy()

    # Phase 4: More coordinate sweeps + momentum
    if _el() < budget_s - 200:
        for _s in range(60):
            if _el() > budget_s - 200:
                break
            best_h = _coord_sweep(best_h, 300)
            best_h = _repair(best_h, n0)
        c5 = _eval(best_h)
        best_c5 = c5
        if _el() < budget_s - 150:
            h_m = _mom_proj(best_h, 20000, 300, 0.002, 0.85)
            h_m = _repair(h_m, n0)
            for _s in range(40):
                if _el() > budget_s - 150:
                    break
                h_m = _coord_sweep(h_m, 300)
                h_m = _repair(h_m, n0)
            c5m = _eval(h_m)
            if c5m < best_c5:
                best_c5, best_h = c5m, h_m.copy()

    # Phase 5: Multi-start perturbations
    if _el() < budget_s - 180:
        for _r in range(15):
            if _el() > budget_s - 100:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(best_h + 0.01 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(best_h + 0.03 * rng.randn(n0), n0)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, n0 // 6))
                st = rng.randint(0, n0 - bs)
                hp[st:st+bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n0)
            elif strat == 3:
                shift = rng.randint(2, max(3, n0 // 10))
                hp = _repair(np.roll(best_h, shift), n0)
            else:
                hp = best_h.copy()
                freqs = rng.randint(1, n0//3 + 1, size=3)
                t_arr = np.arange(n0)
                for f in freqs:
                    hp += 0.006 * rng.randn() * np.sin(2*np.pi*f*t_arr/n0)
                hp = _repair(hp, n0)
            h_new = _slsqp_opt(hp, 20000, 600, 1e-15)
            for _s in range(30):
                if _el() > budget_s - 100:
                    break
                h_new = _coord_sweep(h_new, 250)
                h_new = _repair(h_new, n0)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 40:
        h_new = _slsqp_opt(best_h, 100000, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 7: Refined coordinate sweeps
    if _el() < budget_s - 10:
        for _s in range(80):
            if _el() > budget_s - 5:
                break
            best_h = _coord_sweep_refined(best_h, 200, 800, 0.03)
            best_h = _repair(best_h, n0)
        c5 = _eval(best_h)
        if c5 < best_c5:
            best_c5 = c5

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
