        bb = 0.5
        stuck = 0
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            if g_p is not None:
                s_v = h_c - h_p
                y_v = g_c - g_p
                sy = np.dot(s_v, y_v)
                if sy > 1e-14:
                    bb = np.clip(np.dot(s_v, s_v) / sy, 0.005, 5.0)
            h_u = h_c - bb * g_c
            h_n = _proj_sum(h_u, n / 2.0)
            f_n = _eval(h_n)
            if f_n > f_c * (1 - 1e-15):
                for _ in range(12):
                    bb *= 0.5
                    h_n = _proj_sum(h_c - bb * g_c, n / 2.0)
                    f_n = _eval(h_n)
                    if f_n < f_c:
                        break
                if f_n >= f_c:
                    stuck += 1
                    if stuck > 30:
                        break
                    continue
            h_p, g_p = h_c.copy(), g_c.copy()
            h_c = h_n
            f_c = f_n
            _, g_c = _obj_grad(h_c, alpha)
            stuck = 0
            if f_c < best_f:
                best_f = f_c
                best_h = h_c.copy()
    return best_h, best_f

def _coord_descent(h, n_iter, deadline, rng):
    n = len(h)
    c_f = _eval(h)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.01 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c_n = _eval(h)
        if c_n < c_f - 1e-15:
            c_f = c_n
        else:
            h[i], h[j] = ni - d, nj + d
    h = _proj_sum(h, n / 2.0)
    return h, _eval(h)

def _slsqp_polish(h, alpha, maxiter, deadline):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def _grad(x):
        _, g = _obj_grad(x, alpha)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        if time.time() > deadline:
            return h
        return _proj_sum(res.x, n / 2.0)
    except Exception:
        return h

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_o= h)

def _interpolate(h, n_new):
    n_old = len(h)
    xo = np.arange(n_old) * 2.0 / n_old
    xn = np.arange(n_new) * 2.0 / n_new
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _proj_sum(np.array(initial_h_values, dtype=float).copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: BB optimize at original resolution with increasing alpha
    alphas_p1 = [50, 200, 1000, 5000, 20000, 80000, 300000]
    best_h, best_c5 = _bb_optimize(best_h, alphas_p1, 3000, deadline - 280)
    best_h = _proj_sum(best_h, best_n / 2.0)
    best_c5 = _eval(best_h)

    # Phase 2: Complement
    if time.time() < deadline - 250:
        h_c = _proj_sum(1.0 - best_h, best_n / 2.0)
        h_c, c_c = _bb_optimize(h_c, [200, 2000, 20000, 100000], 2000, deadline - 240)
        if c_c < best_c5:
            best_c5 = c_c
            best_h = h_c.copy()

    # Phase 3: Multiple random starts
    if time.time() < deadline - 200:
        for rq in [3, 5, 7, 10]:
            if time.time() > deadline - 200:
                break
            half = best_n // 2
            if half % rq != 0:
                continue
            w = half // rq
            starts = np.array([i * (best_n // rq) for i in range(rq)])
            h_r = np.zeros(best_n)
            for s in starts:
                h_r[s:s + w] = 1.0
            h_r += 0.05 * rng.randn(best_n) * (1.0 - 2.0 * h_r)
            h_r = _proj_sum(h_r, best_n / 2.0)
            h_r, c_r = _bb_optimize(h_r, [100, 1000, 10000, 50000], 1500, deadline - 195)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()

    # Phase 4: Upsample to 2x and optimize
    if time.time() < deadline - 150:
        n2 = best_n * 2
        h_up = _interpolate(best_h, n2)
        h_up = _proj_sum(h_up, n2 / 2.0)
        h_up, c_up = _bb_optimize(h_up, [100, 1000, 10000, 50000, 200000], 2000, deadline - 140)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()
            best_n = n2

    # Phase 5: SLSQP polish (fewer iterations, higher alpha)
    if time.time() < deadline - 60:
        h_s = _slsqp_polish(best_h, 200000, 3000, deadline - 55)
        c_s = _eval(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 6: Coordinate descent fine-tuning
    if time.time() < deadline - 10:
        n_cd = 20000 if best_n < 500 else 8000
        best_h, best_c5 = _coord_descent(best_h, n_cd, deadline - 8, rng)

    # Phase 7: Final BB with very high alpha
    if time.time() < deadline - 3:
        best_h, c_f = _bb_optimize(best_h, [500000, 2000000], 2000, deadline - 2)
        if c_f < best_c5:
            best_c5 = c_f

    best_h = _proj_sum(best_h, best_n / 2.0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
