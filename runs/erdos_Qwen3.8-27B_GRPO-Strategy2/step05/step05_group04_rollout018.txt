            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * beta * (0.5 - h)
    return g

def _obj_smooth(h, alpha, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + beta * np.sum(h * (1.0 - h))

def _nesterov(h, alpha, beta, steps, step0, t_limit):
    n = len(h)
    f = _obj_smooth(h, alpha, beta)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    h_cur = h.copy()
    for s in range(steps):
        if time.time() > t_limit:
            break
        g = _grad_smooth(h_cur, alpha, beta)
        h_prev = h_cur.copy()
        mu_s = mu * (0.999 ** s)
        v = mu_s * v + (1.0 - mu_s) * g
        h_new = _proj_to_sum(h_cur - step * v, n * 0.5)
        f_new = _obj_smooth(h_new, alpha, beta)
        if f_new < f:
            h_cur = h_new
            f = f_new
            step = min(step * 1.04, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-9:
                step = step0
                v = np.zeros(n)
            h_cur = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smooth(h_cur, alpha, beta)
    return h_cur

def _lag_grad(h, k_w):
    n = len(h)
    sh = (n - 1) - k_w
    g = np.zeros(n)
    dx = 2.0 / n
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _targeted_descent(h, t_limit, max_rounds=300):
    n = len(h)
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > t_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_w = c.argmax()
        g = _lag_grad(h, k_w)
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _de_evolve(h_best, n, t_limit, rng, pop_size=12, max_gen=200):
    best = h_best.copy()
    best_c = _eval(best)
    pop = [best.copy()]
    for i in range(pop_size - 1):
        pert = np.zeros(n)
        nb = rng.randint(3, 10)
        for _ in range(nb):
            s = rng.randint(n - 1)
            d = 0.04 * rng.randn()
            pert[s:s + rng.randint(2, 6)] += d
        hc = _repair(best + pert, n)
        pop.append(hc)
    c_pop = [_eval(p) for p in pop]
    for gen in range(max_gen):
        if time.time() > t_limit:
            break
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            F = rng.uniform(0.4, 1.0)
            CR = rng.uniform(0.6, 0.95)
            trial = pop[i].copy()
            j0 = rng.randint(n)
            for j in range(n):
                if j == j0 or rng.rand() < CR:
                    trial[j] = pop[r1][j] + F * (pop[r2][j] - pop[r3][j])
            trial = _repair(np.clip(trial, 0, 1), n)
            ct = _eval(trial)
            if ct < c_pop[i]:
                pop[i] = trial
                c_pop[i] = ct
                if ct < best_c:
                    best_c = ct
                    best = trial.copy()
    return best, best_c

def _pair_polish(h, t_limit, rng):
    n = len(h)
    c5c = _eval(h)
    start = time.time()
    it = 0
    while time.time() < t_limit:
        it += 1
        if it % 500 == 0:
            if (time.time() - start) / 500.0 > 0.01:
                break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            d2 = 5e-5 * rng.randn()
            ni, nj = h[i] + d2, h[j] - d2
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            d = d2
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-18:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _block_flip(h, n, rng):
    hp = h.copy()
    bs = rng.randint(n // 6, n // 2)
    st = rng.randint(0, n - bs)
    hp[st:st + bs] = 1.0 - hp[st:st + bs]
    return _repair(hp, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov with binary bias at increasing alpha
    for alpha, beta in [(500, 0.01), (2000, 0.005), (8000, 0.002), (30000, 0.001), (100000, 0.0)]:
        if _el() > 150:
            break
        h_c = _nesterov(best_h, alpha, beta, 500, 0.003, t0 + 150)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
            best_n = n0

    # Phase 2: Targeted worst-lag descent
    if _el() < budget_s - 350:
        h_c, c5_t = _targeted_descent(best_h, t0 + budget_s - 350, 500)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_c.copy()

    # Phase 3: DE global search
    if _el() < budget_s - 250:
        de_t = t0 + budget_s - 250
        h_de, c5_de = _de_evolve(best_h, best_n, de_t, rng, pop_size=10, max_gen=150)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()

    # Phase 4: Nesterov again with higher alpha after DE
    if _el() < budget_s - 200:
        for alpha in [50000, 200000, 500000]:
            if _el() > budget_s - 200:
                break
            h_c = _nesterov(best_h, alpha, 0.0, 400, 0.002, t0 + budget_s - 200)
            c5 = _eval(h_c)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c.copy()

    # Phase 5: Multi-start with block flips + Nesterov
    if _el() < budget_s - 150:
        for _r in range(15):
            if _el() > budget_s - 150:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _block_flip(best_h, best_n, rng)
            elif strat == 1:
                hp = _repair(best_h + 0.02 * rng.randn(best_n), best_n)
            elif strat == 2:
                hp = best_h.copy()
                fb = 2
                hp = _upsample(hp, fb)
                hp = _repair(hp, best_n * fb)
            else:
                hp = best_h.copy()
                for _f in range(3):
                    f = rng.randint(1, best_n // 3 + 1)
                    amp = 0.008 * rng.randn()
                    idx = np.arange(best_n)
                    hp += amp * np.sin(2 * np.pi * f * idx / best_n + rng.rand() * 6.28)
                hp = _repair(hp, best_n)
            h_new = _nesterov(hp, 50000, 0.0, 300, 0.002, t0 + budget_s - 150)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Final targeted descent + Nesterov polish
    if _el() < budget_s - 80:
        h_c, c5_t = _targeted_descent(best_h, t0 + budget_s - 80, 200)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_c.copy()

    if _el() < budget_s - 30:
        h_c = _nesterov(best_h, 1000000, 0.0, 300, 0.001, t0 + budget_s - 30)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 7: Pair polish
    if _el() < budget_s - 5:
        h_c, c5_p = _pair_polish(best_h, t0 + budget_s - 5, rng)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_c.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
