import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lse_obj(x, T):
    c = _corr(x)
    m = c.max()
    s = np.clip(T * (c - m), -700, 700)
    return (m + np.log(np.sum(np.exp(s))) / T) * (2.0 / len(x))

def _lse_grad(x, T):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(T * (c - m), -700, 700)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh == 0:
            g[:] += wk * (1.0 - 2.0 * x)
        elif sh > 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:])
            g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - x[:cnt])
            g[:cnt] -= wk * x[sp:]
    return g

def _grad_k(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh == 0:
        g[:] += (1.0 - 2.0 * h)
    elif sh > 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:])
        g[sh:] -= h[:cnt]
    else:
        sp = -sh
        cnt = n - sp
        g[sp:] += (1.0 - h[:cnt])
        g[:cnt] -= h[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, T, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(T,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, t_deadline, top_m=5):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    step = 0.005
    while time.time() < t_deadline:
        c = _corr(h)
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(h, k, n)
        g /= (top_m * dx)
        h_new = _repair(h - step * g, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            step = min(step * 1.01, 0.05)
        else:
            step *= 0.75
            if step < 1e-8:
                break
    return h, c5c

def _pairwise_cd(h, n_iters, scale, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    c = _corr(h)
    for _ in range(n_iters):

        j = rng.randint(n)

        if c5n < c5c - 1e-15:

        else:

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    m = c.max()
    s = np.clip(T * (c - m), -700, 700)
    return (m + np.log(np.sum(np.exp(s))) / T) * (2.0 / len(x))

            g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - x[:cnt])
            g[:cnt] -= wk * x[sp:]
    return g

def _grad_k(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        cnt = n - sh
        g[:cnt] += 1.0 - h[sh:]
        g[sh:] -= h[:cnt]
    else:
        sp = -sh
        cnt = n - sp
        g[sp:] += 1.0 - h[:cnt]
        g[:cnt] -= h[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, T, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(T,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, t_end, top_m):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    step = 0.005
    while time.time() < t_end:
        c = _corr(h)
        idx = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(h, k, n)
        g /= top_m * dx
        h_new = _repair(h - step * g, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.7
            if step < 1e-8:
                break
    return h, c5c

def _pairwise_cd(h, n_iters, scale, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
