import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=300, top_k=8):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        idx = np.argsort(c)[::-1][:top_k]
        vals = c[idx]
        m = vals.max()
        s = np.exp(alpha * (vals - m))
        return m + float(np.log(s.sum() / len(idx)) / alpha)
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _sgd(h, dx, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    lr = 0.004
    for _ in range(200000):
        if time.time() > t_end:
            break
        c = _corr(h)
        top = np.argsort(c)[::-1][:5]
        w = np.exp(100.0 * (c[top] - c[top].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            g = np.zeros(n)
            if ki >= n - 1:
                g[s:n] += 1.0 - h[:m]
                g[:m] -= h[s:n]
            else:
                g[:m] += 1.0 - h[s:n]
                g[s:n] -= h[:m]
            grad += wi * g
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        h_new = _repair(np.clip(h - lr * grad / gn, 0, 1), tgt)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
            lr = min(lr * 1.004, 0.025)
        else:
            lr *= 0.91
            if lr < 1e-6:
                break
    return h, best

def _greedy_local(h, dx, rng, t_end, step=0.008, n_trials=15000):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    stall = 0
    while time.time() < t_end:
        c = _corr(h)
        top_lags = np.argsort(c)[::-1][:8]
        improved = False
        idxs = rng.randint(n * n, size=n_trials)
        for idx_val in idxs:
            if time.time() > t_end:
                break
            i = int(idx_val) // n
            j = int(idx_val) % n
            if i == j:
                continue
            for d in (step, -step):
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i] = ni
                h[j] = nj
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
        # Also try targeted moves on worst lags
        for ki in top_lags[:4]:
            s = int(ki - (n - 1))
            if s == 0:
                continue
            sa = abs(s)
            m = n - sa
            for _ in range(12):
                a = rng.randint(m)
                if s > 0:
                    i1, i2 = a + sa, a
                else:
                    i1, i2 = a, a - sa
                d = step * (2.0 * rng.random() - 1.0)
                if abs(d) < 1e-7:
                    continue
                ni, nj = h[i1] + d, h[i2] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i1] = ni
                h[i2] = nj
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                    improved = True
                else:
                    h[i1] -= d
                    h[i2] += d
        if not improved:
            stall += 1
            if stall > 8:
                break
            step *= 0.7
        else:
            stall = 0
    return h, best

def _fine_tune(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    for sc in [0.003, 0.001, 3e-4, 1e-4, 5e-5]:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i] = ni
            h[j] = nj
            cn = _c5(h, dx)
            if cn < best:
                best = cn
            else:
                h[i] -= d
                h[j] += d
    return h, best

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5, best_n = h.copy(), _c5(h, dx), n

    def elapsed():

    # P1: SLSQP warm-up
    t1 = t0 + 50

    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # P3: SLSQP
    t3 = t0 + 190
    h2, c2 = _slsqp(best_h, dx, 8000, 350)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # P4: Greedy local search (main workhorse)
    t4 = t0 + 500
    h2, c2 = _greedy_local(best_h, dx, rng, t4, step=0.008, n_trials=15000)
    if c2 < best_c5:

    # P5: SLSQP refinement
    t5 = t0 + 560
    h2, c2 = _slsqp(best_h, dx, 20000, 300)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # P6: More greedy local
    t6 = t0 + 820
    h2, c2 = _greedy_local(best_h, dx, rng, t6, step=0.005, n_trials=20000)

    # P7: Higher resolution attempt
    if budget_s - elapsed() > 260:
        t7 = t0 + 1050
        n2 = 2 * n
        dx2 = 2.0 / n2
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        h2, c2 = _sgd(h2, dx2, time.time() + 55)
        h2, c2 = _slsqp(h2, dx2, 5000, 200)

        if c2 < best_c5:

    # P8: Fine tune on best
    nb, dxb = best_n, 2.0 / best_n
    t8 = t0 + budget_s - 40
    h2, c2 = _fine_tune(best_h, dxb, rng, t8)

        if c2 < best_c5:

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
