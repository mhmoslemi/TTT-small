```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h
```

```python
    for i in range(niter):

        hn = h.copy()
        if r < 0.35:

            hn[s:s+b] += rng.randn(b)*(0.03+0.1*t)
        elif r < 0.6:
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-10:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        diff = tgt - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lwgrad(h, idx, wts, dx):
    n = len(h)
    g = np.zeros(n)
    for ki in range(len(idx)):
        w = wts[ki]
        if w < 1e-14:
            continue
        s = (n - 1) - idx[ki]
        if s >= 0:
            m = n - s
            g[:m] -= w * h[s:s + m]
            if s > 0:
                g[s:] += w * (1.0 - h[:m])
        else:
            sp = -s
            m = n - sp
            g[:m] += w * (1.0 - h[sp:sp + m])
            g[sp:] -= w * h[:m]
    g *= dx
    return g

def _lse_obj(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    x = np.clip(a * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(x))) / a

def _lse_grad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    x = np.clip(a * (c - m), -500, 0)
    e = np.exp(x)
    w = e / e.sum()
    idx = np.where(w > 1e-10)[0]
    return _lwgrad(h, idx, w[idx], 2.0 / n)

def _proj(h, tgt):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _gd(h, a, steps, step0, t0, dl):
    n = len(h)
    f = _lse_obj(h, a)
    v = np.zeros(n)
    step = step0
    for _ in range(steps):
        if time.time() - t0 > dl:
            break
        g = _lse_grad(h, a)
        v = 0.85 * v + 0.15 * g
        h_old = h.copy()
        for _t in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            f_new = _lse_obj(h_new, a)
            if f_new < f - 1e-15:
                h, f = h_new, f_new
                step = min(step * 1.04, step0 * 3)
                break
            step *= 0.6
        else:
            h, step = h_old, step * 0.4
    return h

def _active(h, top_m, rounds, t0, dl):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _r in range(rounds):
        if time.time() - t0 > dl:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:top_m]
        wts = c[idx] / (c[idx].sum() + 1e-30)
        g = _lwgrad(h, idx, wts, dx)
        ok = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
            h_new = _proj(h - st * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h, c5c, ok = h_new, c5n, True
                break
        if not ok:
            break
    return h, c5c

def _sa(h, n, tmax, tmin, niter, rng, t0, dl):
    cc = _eval(h)
    bh, bc = h.copy(), cc
    for i in range(niter):
        if time.time() - t0 > dl:
            break
        t = tmax * (tmin / tmax) ** (i / max(1, niter - 1))
        r = rng.rand()
        hn = h.copy()
        if r < 0.30:
            b = rng.randint(1, max(2, n // 8))
            s = rng.randint(0, n - b)
            hn[s:s + b] += rng.randn(b) * (0.03 + 0.1 * t)
        elif r < 0.55:
            s = rng.randint(0, n // 4)
            l = rng.randint(max(2, n // 12), n // 3)
            hn[s:s + l] = 1.0 - hn[s:s + l]
        elif r < 0.72:
            i1 = rng.randint(n)
            hn[i1] = 1.0 - hn[i1]
        elif r < 0.88:
            i1, i2 = rng.choice(n, 2, replace=False)
            hn[i1], hn[i2] = hn[i2], hn[i1]
        else:
            i1 = rng.randint(n // 4)
            l = rng.randint(2, n // 4)
            cen = i1 + l // 2
            for j in range(i1, i1 + l):
                hn[j] += 0.05 * (0.5 - abs(j - cen) / l) * rng.randn()
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < cc or rng.rand() < np.exp(-(cn - cc) / max(t, 1e-10)):
            h, cc = hn, cn
            if cc < bc:
                bc, bh = cc, h.copy()
    return bh, bc

def _fine(h, ntries, rng, t0, dl):
    n = len(h)
    cc = _eval(h)
    bh, bc = h.copy(), cc
    scale = 280.0 / n
    for _ in range(ntries):
        if time.time() - t0 > dl:
            break
        if rng.randint(3) < 2:
            i, j = rng.choice(n, 2, replace=False)
            d = 5e-5 * scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < cc:
                    cc = cn
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        else:
            i = rng.randint(n - 1)
            for d in [2e-5, -2e-5, 1e-5, -1e-5, 5e-6, -5e-6]:
                ni, nj = h[i] + d, h[i + 1] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[i + 1] = ni, nj
                    cn = _eval(h)
                    if cn < cc:
                        cc = cn
                        break
                    h[i], h[i + 1] = ni - d, nj + d
        if cc < bc:
            bc, bh = cc, h.copy()
    return bh, bc

def _tern(h):
    n = len(h)
    out = np.full(n, 0.5)
    out[h > 2.0 / 3.0] = 1.0
    out[h < 1.0 / 3.0] = 0.0
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n0 = len(h)
    bh = h.copy()
    bc5 = _eval(bh)
    bn = n0

    # Phase 1: LSE gradient descent at original resolution
    for a in [500, 5000, 50000]:
        dl = t0 + 100
        if time.time() - t0 > dl:
            break
        h1 = _gd(bh, a, 200, 0.003, t0, dl)
        c5 = _eval(h1)
        if c5 < bc5:
            bc5, bh = c5, h1.copy()

    # Phase 2: Active-set descent at original resolution
    dl = t0 + 170
    if time.time() - t0 < dl:
        h2, c52 = _active(bh, 5, 300, t0, dl)
        if c52 < bc5:
            bc5, bh = c52, h2.copy()

    # Phase 3: Upsample to 2x, LSE GD + active-set
    dl = t0 + budget_s - 350
    if time.time() - t0 < dl:
        n1 = n0 * 2
        h_hi = _repair(np.repeat(bh, 2), n1)
        for a in [5000, 50000]:
            dl2 = t0 + budget_s - 250
            if time.time() - t0 > dl2:
                break
            h_hi = _gd(h_hi, a, 150, 0.002, t0, dl2)
            c5 = _eval(h_hi)
            if c5 < bc5:
                bc5, bh, bn = c5, h_hi.copy(), n1
        dl3 = t0 + budget_s - 200
        if time.time() - t0 < dl3:
            h3, c53 = _active(bh, 8, 200, t0, dl3)
            if c53 < bc5:
                bc5, bh, bn = c53, h3.copy(), len(h3)

    # Phase 4: Simulated annealing with diverse moves
    nc = len(bh)
    dl = t0 + budget_s - 200
    if time.time() - t0 < dl:
        ns = min(int((dl - (time.time() - t0)) / 0.04), 12000)
        h4, c54 = _sa(bh, nc, 0.005, 1e-5, ns, rng, t0, dl)
        if c54 < bc5:
            bc5, bh, bn = c54, h4.copy(), len(h4)

    # Phase 5: Multi-start from perturbed / ternary versions
    nc = len(bh)
    dl = t0 + budget_s - 100
    if time.time() - t0 < dl:
        for _r in range(12):
            if time.time() - t0 > dl:
                break
            st = rng.randint(4)
            if st == 0:
                hp = _repair(bh + 0.02 * rng.randn(nc), nc)
            elif st == 1:
                hp = bh.copy()
                l = rng.randint(nc // 8, max(nc // 8 + 1, nc // 3))
                s = rng.randint(0, max(1, nc - l))
                hp[s:s + l] = 1.0 - hp[s:s + l]
                hp = _repair(hp, nc)
            elif st == 2:
                hp = _repair(bh + 0.06 * rng.randn(nc), nc)
            else:
                hp = _repair(_tern(bh) + 0.01 * rng.randn(nc), nc)
            hn = _gd(hp, 20000, 100, 0.002, t0, dl)
            hn, c5n = _active(hn, 5, 100, t0, dl)
            if c5n < bc5:
                bc5, bh, bn = c5n, hn.copy(), len(hn)

    # Phase 6: Fine pairwise / adjacent local search
    nc = len(bh)
    dl = t0 + budget_s - 15
    if time.time() - t0 < dl:
        ns = min(int((dl - (time.time() - t0)) / 0.004), 25000)
        h6, c56 = _fine(bh, ns, rng, t0, dl)
        if c56 < bc5:
            bc5, bh, bn = c56, h6.copy(), len(h6)

    # Phase 7: Final active-set polish with wider top set
    dl = t0 + budget_s - 3
    if time.time() - t0 < dl:
        h7, c57 = _active(bh, 10, 200, t0, dl)
        if c57 < bc5:
            bc5, bh, bn = c57, h7.copy(), len(h7)

    bh = _repair(bh, len(bh))
    bc5 = _eval(bh)
    return (bh, bc5, len(bh))
```
