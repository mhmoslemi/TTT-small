    """One step of subgradient descent on the max lag."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_best = c.argmax()
    s = (n - 1) - k_best  # shift
    
    # Gradient of c[k_best] with respect to h
    g = np.zeros(n)
    if s >= 0:
        # c = sum_{i=0}^{n-1-s} h[i] * (1 - h[i+s])
        # d/dh[i] for i in [0, n-1-s]: (1 - h[i+s])
        # d/dh[i] for i in [s, n-1]: -h[i-s]
        idx1 = np.arange(n - s)
        g[idx1] += (1.0 - h[idx1 + s])
        if s > 0:
            idx2 = np.arange(s, n)
            g[idx2] -= h[idx2 - s]
    else:
        sp = -s
        idx1 = np.arange(sp, n)
        g[idx1] += (1.0 - h[idx1 - sp])
        idx2 = np.arange(n - sp)
        g[idx2] -= h[idx2 + sp]
    
    # Projected gradient step with sum constraint
    h_new = h - step * g
    h_new = _project_sum(h_new, n * 0.5)
    return h_new, c[k_best] * 2.0 / n

        if time.time() > t0 + budget_s * 0.8:
            break
        st = rng.randint(5)
        if st == 0:
            hp = _fourier_pert(best_h, nc, rng, 0.03)
        elif st == 1:
            hp = _binarize(best_h, nc, rng, 0.25)
        elif st == 2:
            hp = _repair(best_h + 0.012 * rng.randn(nc), nc)
        elif st == 3:
            hp = best_h.copy()
            bs = rng.randint(8, nc // 3)
            s0 = rng.randint(0, nc - bs)
            hs = rng.randint(1, nc // 2)
            hp[s0:s0 + bs] = np.roll(best_h, hs)[s0:s0 + bs]
            hp = _repair(hp, nc)
        else:
            hp = _repair(0.5 + 0.25 * np.sin(2 * np.pi * rng.rand() * np.arange(nc) / nc + rng.rand() * 6.28)
                         + 0.05 * rng.randn(nc), nc)
        h2, c52 = _subgrad(hp, nc, 2500, t0 + budget_s * 0.8, 5)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 5: Coordinate descent (multiple scales)
    for scale, iters in [(0.04, 60000), (0.01, 80000), (0.003, 100000), (0.0005, 120000)]:
        if time.time() > t0 + budget_s - 60:
            break
        h2, c52 = _coord_desc(best_h, nc, iters, scale, seed + scale * 1e6, t0 + budget_s - 60)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 6: Final SLSQP polish
    if time.time() < t0 + budget_s - 8:
        h2 = _slsqp_lse(best_h, 100000, 4000, 1e-20)
        c52 = _eval(h2)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # Phase 7: Ultra-fine coord
    if time.time() < t0 + budget_s - 2:
        h2, c52 = _coord_desc(best_h, nc, 80000, 5e-5, 99999, t0 + budget_s - 1)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
