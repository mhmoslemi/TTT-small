    """Gradient of f_k (correlate value at index k) w.r.t. h, times dx"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        # ∂f/∂h[j] = (1-h[j+sh]) for j < n-sh, minus h[j-sh] for j >= sh
        for j in range(n - sh):

        if sh > 0:
            for j in range(sh, n):
                g[j] -= h[j - sh] * dx
    else:

        for j in range(sp, n):
            g[j] += (1.0 - h[j - sp]) * dx
        for j in range(n - sp):
            g[j] -= h[j + sp] * dx
    return g
