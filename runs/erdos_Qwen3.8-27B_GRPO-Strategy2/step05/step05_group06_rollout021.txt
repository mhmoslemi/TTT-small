import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w /= ws
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol, time_limit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    t_start = time.time()
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def obj_grad(x, which):
        if which == 'f':
            return _smooth_obj(x, alpha)
        return _smooth_grad(x, alpha)
    try:
        # Use a callback to check time
        state = {'count': 0}
        def callback(xk):
            state['count'] += 1
            if time.time() - t_start > time_limit:
                return True
            return None
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return _repair(res.x, n)
    except Exception:

def _interp_upsample(h, factor):
    n = len(h)
    n_new = n * factor
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, n_new)
    return np.interp(x_new, x_old, h)

def _topk_local_search(h, n_iter, step_size, topk):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        c = _corr(h)
        idx = np.argsort(c)[::-1][:topk]
        # Compute gradient from top-k only
        dx = 2.0 / n
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= dx / topk
        # Step

            h_new = _repair(h - lam * g, n)
            c5n = _eval(h_new)
            if c5n < c5c:
                h, c5c = h_new, c5n
                break
    return h, c5c
