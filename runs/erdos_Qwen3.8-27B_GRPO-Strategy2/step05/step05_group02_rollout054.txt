import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    if abs(h.sum() - t) < 1e-10:
        return h
    return _proj(h, t)

def _obj_lse(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        wk *= dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _obj_viol(h, tau):
    c = np.correlate(h, 1.0 - h, mode='full')
    exc = np.maximum(c - tau, 0.0)
    return float(np.sum(exc * exc))

def _grad_viol(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if c[k] <= tau:
            continue
        w = 2.0 * (c[k] - tau) * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g

def _sf(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _slsqp_lse(h, alpha, maxiter=3000, ftol=1e-19):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj}]
    try:
        res = minimize(_obj_lse, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp_viol(h, tau, maxiter=3000, ftol=1e-19):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj}]
    try:
        res = minimize(_obj_viol, h, jac=_grad_viol, args=(tau,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _topT_descent(h, T, n_iter, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        thresh = m - 0.02 * m
        active = np.where(c >= thresh)[0]
        active = active[:T]
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= dx / len(active)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            h = _repair(h + 0.0005 * rng.randn(n), n)
            c5c = _eval(h)
    return h, c5c

def _coord_search(h, n_iter, scale, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c
