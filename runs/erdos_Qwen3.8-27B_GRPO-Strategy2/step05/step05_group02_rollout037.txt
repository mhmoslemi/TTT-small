    N = 2*n  # FFT size
    
    H = np.fft.rfft(h, N)
    Bf = np.fft.rfft(b, N)
    Af = np.fft.rfft(A, N)
    
    # Term 1: sum_sh A[sh]*b[i+sh] = corr(b, A) = ifft(conj(fft(A)) * fft(b))
    # Actually: (b ⊛ A)[i] = sum_j b[i+j]*A[j]  -- this is cross-correlation
    # Using FFT: ifft(conj(fft(A, N)) * fft(b, N))
    t1 = np.fft.irfft(np.conj(Af) * Bf)[:n]
    
    # Term 2: sum_sh A[sh]*h[i-sh] = conv(h, A) = ifft(fft(h) * fft(A))
    t2 = np.fft.irfft(H * Af)[:n]
    
    # Term 3: sum_sp B[sp]*b[i-sp] = conv(b, B) -- B has length n-1
    Bf2 = np.fft.rfft(B, N)
    t3 = np.fft.irfft(Bf2 * Bf)[:n]
    
    # Term 4: sum_sp B[sp]*h[i+sp] = corr(h, B) = ifft(conj(fft(B)) * fft(h))
    t4 = np.fft.irfft(np.conj(Bf2) * H)[:n]
    
    g = dx * (t1 - t2 + t3 - t4)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0, 1)
        d = tgt - h.sum()
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in o:
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_corr(h, w):
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lse_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_jac(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _grad_corr(x, w)

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        r = minimize(_lse_obj, h, jac=_lse_jac, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _grad_desc(h, alpha0, nsteps, lr0, alpha_growth):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    alpha = alpha0
    c0 = _eval(h)
    for it in range(nsteps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = _grad_corr(h, w)
        lr = lr0
        for _ in range(40):
            hn = _proj(h - lr * g, tgt)
            c1 = _eval(hn)
            if c1 < c0 - 1e-17:
                h = hn
                c0 = c1
                break
            lr *= 0.5
        if it % 500 == 499:
            alpha *= alpha_growth
    return h

def _pairwise(h, rng, niter, scale):
    n = len(h)
    c0 = _eval(h)
    for _ in range(niter):
        i = rng.randint(n)
        j = rng.randint(n)

        h[i], h[j] = ni, nj
        c1 = _eval(h)
        if c1 < c0 - 1e-17:
            c0 = c1
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c0
