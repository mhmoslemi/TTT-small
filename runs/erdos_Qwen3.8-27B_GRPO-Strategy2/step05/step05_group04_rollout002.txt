import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), h[idx] if diff < 0 else 1.0 - h[idx])
            h[idx] += a if diff > 0 else -a
            diff += a if diff < 0 else -a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha, n_top):
    n = len(h)
    c = _corr(h)
    if n_top > 0 and n_top < len(c):
        idx = np.argsort(c)[-n_top:]
        c_f = c[idx]
        m = c_f.max()
        s = np.clip(alpha * (c_f - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        w_full = np.zeros(len(c))
        w_full[idx] = w
    else:
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w_full = np.exp(s - s.max())
        w_full /= w_full.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w_full[k] < 1e-14:
            continue
        wk = w_full[k] * dx
        sh = k - (n - 1)
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _adam(h, alpha, steps, lr, n_top, decay=0.999):
    n = len(h)
    target = n * 0.5
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.85, 0.95, 1e-8
    cur_lr = lr
    for t in range(1, steps + 1):
        g = _grad_smooth(h, alpha, n_top)
        m_v = b1 * m_v + (1.0 - b1) * g
        v_v = b2 * v_v + (1.0 - b2) * g * g
        m_hat = m_v / (1.0 - b1**t)
        v_hat = v_v / (1.0 - b2**t)
        h = _proj(h - cur_lr * m_hat / (np.sqrt(v_hat) + eps), target)
        cur_lr *= decay
    return h

def _worst_lag_descent(h, max_rounds=50):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    steps_list = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for _ in range(max_rounds):
        c = _corr(h)
        k_w = c.argmax()
        sh = k_w - (n - 1)
        g = np.zeros(n)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
        g *= dx
        c_cur = c[k_w]
        improved = False
        for step in steps_list:
            h_new = _proj(h - step * g, target)
            c_new = _corr(h_new)
            if c_new[k_w] < c_cur - 1e-13:
                h = h_new
                improved = True
                break
        if not improved:
            break
    return h

def _sa_block(h, n_iter, rng, t0=0.003, t1=1e-5):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    c_cur = best_c
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        blk = rng.randint(2, max(3, n // 8))
        st = rng.randint(0, n - blk)
        delta = rng.randn(blk) * (0.005 + 0.04 * t)
        h_new = h.copy()
        h_new[st:st + blk] += delta
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c

def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max())) - (len(c) - 1)) / alpha
    def _grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-14:
                continue
            wk = w[k] * dx
            sh = k - (n - 1)
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    def _c_fun(x):
        return x.sum() - target
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _multi_restart(h, n, rng, best_c5, n_starts=6):
    best_h = h.copy()
    best_c = best_c5
    for _ in range(n_starts):
        strat = rng.randint(4)
        if strat == 0:
            hp = _repair(best_h + 0.02 * rng.randn(n), n)
        elif strat == 1:
            hp = best_h.copy()
            bs = rng.randint(n // 6, n // 2)
            st = rng.randint(0, n - bs)
            hp[st:st + bs] = 1.0 - hp[st:st + bs]
            hp = _repair(hp, n)
        elif strat == 2:
            hp = (best_h > 0.5).astype(float)
            hp = _repair(hp, n)
        else:
            hp = best_h.copy()
            for _b in range(rng.randint(5, 15)):
                i = rng.randint(n - 1)
                d = 0.05 * rng.randn()
                ni, nj = hp[i] + d, hp[i + 1] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    hp[i], hp[i + 1] = ni, nj
            hp = _repair(hp, n)
        h_new = _adam(hp, 50000, 400, 0.008, 5)
        h_new = _worst_lag_descent(h_new, 20)
        h_new = _repair(h_new, n)
        c5 = _eval(h_new)
        if c5 < best_c:
            best_c = c5
            best_h = h_new.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
