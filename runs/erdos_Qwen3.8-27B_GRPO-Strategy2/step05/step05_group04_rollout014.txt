def _smooth_grad(h, alpha, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(np.clip(s, -50.0, 0.0))
    w /= w.sum()
    g = np.zeros(n)
    # Only compute for significant weights
    idx = np.where(w > 1e-6)[0]
    for k in idx:
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            p = -sh
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g

def _slsqp(h, alpha, n, maxiter, t0, budget_s):
    def _c_fun(x):
        return x.sum() - n * 0.5
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, alpha, n),
                       args=(alpha,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
