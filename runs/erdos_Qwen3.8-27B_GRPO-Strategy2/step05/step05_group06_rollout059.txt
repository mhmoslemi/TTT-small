    if n == 0:

    idx = np.argsort(a)
    a_s = a[idx]
    b_s = b[idx]

    # Each line (a_i, b_i): on the upper envelope, it's the topmost at some range of t

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            f_vals = np.maximum(lin_vals, q_val)

                hi_t = t_grid[j_best + 1]
                for _ in range(30):

                    m2 = lo_t + (hi_t - lo_t) * 0.618

                else:
