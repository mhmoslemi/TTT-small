    from scipy.optimize import linprog
    
    nx = len(h)
    dx = 2.0 / nx
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Get top m lags
    top_idx = np.argsort(c)[-top_m:][::-1]
    
    # Compute gradients for these lags
    # c_k = dx * sum_i h[i]*(1-h[i+k]) for k in full correlate indexing
    # In full mode, index k corresponds to shift sh = (n-1) - k
    # For sh >= 0: c_sh = dx * sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])
    # For sh < 0 (i.e., sh = -sp): c_{-sp} = dx * sum_{i=sp}^{n-1} h[i]*(1-h[i-sp])
    
    # Build objective: minimize t
    # Variables: h_0, ..., h_{n-1}, t  (n+1 variables)
    # Constraints:
    #   For each active lag: g_k · h <= t - c_k(h*) + g_k · h*  (linearized)
    #   0 <= h_i <= 1
    #   sum(h_i) = n/2
    
    n_vars = nx + 1  # h[0..n-1], t
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k in top_idx:
        sh = (nx - 1) - k
        g = np.zeros(nx)
        if sh >= 0:
            for i in range(nx - sh):
                g[i] += dx * (1.0 - h[i + sh])
                if i + sh < nx:
                    g[i + sh] -= dx * h[i]
        else:
            sp = -sh
            for i in range(sp, nx):
                g[i] += dx * (1.0 - h[i - sp])
                if i - sp >= 0:
                    g[i - sp] -= dx * h[i]
        
        # Linearized: c_k(h*) + g·(h - h*) <= t
        # => g·h - t <= -c_k(h*) + g·h*
        # => g·h - t <= g·h* - c_k(h*)
        row = np.zeros(n_vars)
        row[:nx] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(g.dot(h) - c[k])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :nx] = 1.0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _proj_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_step(h, n, top_m, t_cap):
    c = _corr(h)
    top_idx = np.argsort(c)[-top_m:][::-1]
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((top_m, n_vars))
    b_ub = np.zeros(top_m)
    for i, k in enumerate(top_idx):
        g = _grad_lag(h, k, n)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g.dot(h) - c[k]
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(c.min() - 0.01, t_cap + 0.01)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for kk in range(2 * n - 1):
        if w[kk] < 1e-15:
            continue
        g += w[kk] * _grad_lag(x, kk, n)
    return g
