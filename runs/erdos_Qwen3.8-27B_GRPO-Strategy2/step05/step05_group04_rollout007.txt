                hp = _repair(hp, bn)
            elif st == 2:
                hp = _repair(bh + 0.03 * rng.randn(bn), bn)
            elif st == 3:
                hp = bh.copy()
                for _ in range(8):
                    ii = rng.randint(bn - 1)
                    dd = 0.015 * rng.randn()
                    if 0 <= hp[ii] + dd <= 1 and 0 <= hp[ii + 1] - dd <= 1:
                        hp[ii] += dd
                        hp[ii + 1] -= dd
                hp = _repair(hp, bn)
            elif st == 4:
                hp = bh.copy()
                for _ in range(4):
                    f = rng.randint(1, bn // 2 + 1)
                    amp = 0.006 * rng.randn()
                    t2 = np.arange(bn)
                    hp += amp * np.sin(2 * np.pi * f * t2 / bn + rng.rand() * 2 * np.pi)
                hp = _repair(hp, bn)
            else:
                hp = 1.0 - bh
                hp = _repair(hp, bn)
            hn2 = _slsqp(hp, 400000, 2000)
            c = _eval(hn2)[0]
            if c < bc:
                bc, bh = c, hn2.copy()

    # Phase 8: Final SLSQP with very high alpha
    if el() < budget_s - 20:
        hc = _slsqp(bh, 3000000, 5000)
        c = _eval(hc)[0]
        if c < bc:
            bc, bh = c, hc.copy()

    # Phase 9: Fine local search
    if el() < budget_s - 3:
        bnl = len(bh)
        cc, _ = _eval(bh)
        ss = np.array([0.004, 0.002, 0.001, 0.0005, 0.0002])
        nt = min(int((budget_s - 3 - el()) / 0.003), 50000)
        for _ in range(nt):
            if el() > budget_s - 3:
                break
            i = rng.randint(bnl)
            j = rng.randint(bnl)
            if i == j:
                continue
            d = rng.choice(ss) * rng.randn()
            a, b2 = bh[i] + d, bh[j] - d
            if 0 <= a <= 1 and 0 <= b2 <= 1:
                bh[i], bh[j] = a, b2
                cn, _ = _eval(bh)
                if cn < cc - 1e-16:
                    cc = cn
                    if cn < bc:
                        bc = cn
                else:
                    bh[i], bh[j] = a - d, b2 + d

    bh = _repair(bh, bn)
    bc = _eval(bh)[0]
    return (bh, bc, bn)
