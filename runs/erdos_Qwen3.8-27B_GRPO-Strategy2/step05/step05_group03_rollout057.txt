import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                amt = min(d, 1.0 - h[i]); h[i] += amt; d -= amt
            else:
                amt = min(-d, h[i]); h[i] -= amt; d += amt
            if abs(d) < 1e-13: break
    return h

def _interp(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _coord_descent(h, dx, t_end, max_sweeps=3000):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    C = _corr(h)
    best_c = float(np.max(C)) * dx
    nl = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        for j in range(n):
            old = h[j]
            a = np.zeros(nl)
            rk = n - 1 - j
            if rk > 0:
                a[n:n+rk] += 1.0 - h[j+1:n]
            if j > 0:
                a[n:n+j] -= h[j-1::-1]
            if j > 0:
                a[n-2:n-2-j:-1] += 1.0 - h[j-1::-1]
            if rk > 0:
                a[n-2:n-2-rk:-1] -= h[j+1:n]
            # Golden section on [0,1]
            lo, hi = 0.0, 1.0
            x1 = 1.0 - gr
            x2 = gr
            def fv(v):
                Cv = C + (v - old) * a
                Cv[n-1] = C[n-1] + v*(1.0-v) - old*(1.0-old)
                return float(np.max(Cv))
            f1, f2 = fv(x1), fv(x2)
            for _ in range(30):
                if f1 < f2:
                    hi = x2; x2 = x1; f2 = f1
                    x1 = lo + (1.0-gr)*(hi-lo); f1 = fv(x1)
                else:
                    lo = x1; x1 = x2; f1 = f2
                    x2 = lo + gr*(hi-lo); f2 = fv(x2)
            v_opt = (lo+hi)*0.5
            f0, f1b = fv(0.0), fv(1.0)
            fbest = min(f0, f1b, fv(v_opt))
            if fbest == f0: v_opt = 0.0
            elif fbest == f1b: v_opt = 1.0
            if abs(v_opt - old) > 1e-10:
                h[j] = v_opt
                C += (v_opt - old) * a
                C[n-1] += v_opt*(1.0-v_opt) - old*(1.0-old)
        h = _repair(h, tgt)
        C = _corr(h)
        c_val = float(np.max(C)) * dx
        if c_val < best_c - 1e-13:
            best_c = c_val; stall = 0
        else:
            stall += 1

def _slsqp(h, dx, alpha, maxiter=350):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha*(c-m), -250, 250)
        return m + float(np.log(np.sum(np.exp(s))))/alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type':'eq','fun':eq}],
                     options={'maxiter':maxiter,'ftol':1e-12})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _targeted(h, dx, rng, t_end):

        top = np.argsort(c)[::-1][:5]

                    if cn < best: best = cn
                    else: h[i1]+=d; h[i2]-=d
                else: h[i1]+=d; h[i2]-=d
        for _ in range(3):
            i,j = rng.randint(n), rng.randint(n)
            if i==j: continue
            d = rng.uniform(-0.01, 0.01)
            h[i]+=d; h[j]-=d
            if 0<=h[i]<=1 and 0<=h[j]<=1:
                cn = _c5(h, dx)
                if cn < best: best = cn
                else: h[i]-=d; h[j]+=d
            else: h[i]-=d; h[j]+=d
        h = _repair(h, tgt)
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n/2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05*(2.0*rng.random(n)-1.0)
    elif mode == 1:
        idx = rng.choice(n, max(4, n//4), replace=False)
        hp[idx] += 0.1*(2.0*rng.random(len(idx))-1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04*np.sin(2*np.pi*rng.uniform(1,6)*x + rng.uniform(0,6.28))
    else:
        idx = rng.choice(n, max(3, n//6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _fine(h, dx, rng, t_end):
    n = len(h)
    tgt = n/2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    for sc in scales:
        while time.time() < t_end:
            i,j = rng.randint(n), rng.randint(n)
            if i==j: continue
            d = sc*(2.0*rng.random()-1.0)
            h[i]+=d; h[j]-=d
            if 0<=h[i]<=1 and 0<=h[j]<=1:
                cn = _c5(h, dx)
                if cn < best: best = cn
                else: h[i]-=d; h[j]+=d
            else: h[i]-=d; h[j]+=d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0/n
    best_h = _repair(h0.copy(), n/2.0)
    best_c5 = _c5(best_h, dx)

    # Phase 1: Coordinate descent (main novel approach)
    h, c = _coord_descent(best_h, dx, t0 + 350)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 2: SLSQP refinement
    t2 = t0 + 430
    for a in [2000, 8000]:
        if time.time() > t2: break
        hc, cc = _slsqp(best_h, dx, a, 300)
        if cc < best_c5:
            best_h, best_c5 = hc, cc

    # Phase 3: More coordinate descent
    h, c = _coord_descent(best_h, dx, t0 + 600)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 4: Multi-start
    t4 = t0 + 750
    for r in range(6):
        if time.time() > t4: break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _coord_descent(hp, dx, min(t4-10, time.time()+60))
        hc, cc = _slsqp(hp, dx, 5000, 250)
        if cc < best_c5:
            best_h, best_c5 = hp, cc
            hp, c = _targeted(hp, dx, rng, min(t4-2, time.time()+15))
            if c < best_c5: best_h, best_c5 = hp, c

    # Phase 5: 2x resolution
    if budget_s - (time.time()-t0) > 180:
        n2 = 2*n; dx2 = 1.0/n
        h2 = _repair(_interp(best_h, n2), n2/2.0)
        h2, c2 = _coord_descent(h2, dx2, min(t0+900, time.time()+80))
        h2, c2 = _slsqp(h2, dx2, 5000, 200)
        if c2 < best_c5:
            best_h, best_c5, n, dx = h2, c2, n2, dx2

    # Phase 6: Final targeted + SLSQP
    nb = len(best_h); dxb = 2.0/nb
    best_h, best_c5 = _targeted(best_h, dxb, rng, t0 + budget_s - 80)
    if budget_s - (time.time()-t0) > 50:
        for a in [5000, 20000]:
            if time.time() > t0 + budget_s - 30: break
            hc, cc = _slsqp(best_h, dxb, a, 200)
            if cc < best_c5: best_h, best_c5 = hc, cc

    # Phase 7: Fine search
    if budget_s - (time.time()-t0) > 20:
        best_h, best_c5 = _fine(best_h, dxb, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, len(best_h)/2.0)
    best_c5 = _c5(best_h, 2.0/len(best_h))
    return (best_h, best_c5, len(best_h))
