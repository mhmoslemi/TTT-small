    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        for _ in range(80):

                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h

def _lse(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha*(c-m), -500, 500)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n-1) - k
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += wk * (1-h[i1+sh])
            if sh > 0:
                g[i1+sh] -= wk * h[i1]
        else:
            sp = -sh
            i1 = np.arange(n-sp)
            g[i1+sp] += wk * (1-h[i1])
            g[i1] -= wk * h[i1+sp]

    return x.sum() - len(x)*0.5

def _sum_j(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter=2000, ftol=1e-18):
    n = len(h)
    try:
        res = minimize(_lse, h, jac=_lse_grad, args=(alpha,),
                      method='SLSQP', bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':_sum_f,'jac':_sum_j}],
                      options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except:

    return _repair(h - lr*g, n)

                g[i1] += (1-h_work[i1+sh])

                g[i1] -= h_work[i1+sp]

            c5_work = c5_try

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -500, 500)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * h[i1]
        else:
            sp = -sh
            i1 = np.arange(n - sp)
            g[i1 + sp] += wk * (1 - h[i1])
            g[i1] -= wk * h[i1 + sp]
    return g

def _viol_obj(h, tau):

    exc = np.maximum(c - tau, 0)
    return float(np.sum(exc * exc))

def _viol_grad(h, tau):

    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):

        if cv <= tau:
            continue
        exc = 2.0 * (cv - tau) * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += exc * (1 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= exc * h[i1]
        else:
            sp = -sh
            i1 = np.arange(n - sp)
            g[i1 + sp] += exc * (1 - h[i1])
            g[i1] -= exc * h[i1 + sp]
    return g

def _sf(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _slsqp(h, obj, jac, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(obj, h, jac=jac, args=(alpha,), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cycle_topm(h, m, rounds, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _eval(h)
    hw, cw = h.copy(), best_c5
    for _ in range(rounds):
        if time.time() > deadline:
            break
        c = np.correlate(hw, 1.0 - hw, mode='full')
        idx = np.argpartition(c, -m)[-m:]
        for k in idx:
            g = np.zeros(n)
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1 - hw[i1 + sh])
                if sh > 0:
                    g[i1 + sh] -= hw[i1]
            else:

                i1 = np.arange(n - sp)
                g[i1 + sp] += (1 - hw[i1])
                g[i1] -= hw[i1 + sp]
            g *= dx
            for lr in [0.3, 0.15, 0.07, 0.03, 0.01, 0.004]:
                ht = _repair(hw - lr * g, n)
                ct = _eval(ht)
                if ct < cw:
                    hw, cw = ht, ct
                    if cw < best_c5:
                        best_c5, best_h = cw, hw.copy()
                    break
    return best_h, best_c5

def _block_search(h, bs, niter, scale, rng, t0, deadline):

        elif rng.rand() < 0.005 and ct < cw + 0.002:

    return best_h, best_c5

def _pairwise(h, scale, niter, rng, t0, deadline):

                if ct < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -500, 500)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * h[i1]
        else:
            sp = -sh
            i1 = np.arange(n - sp)
            g[i1 + sp] += wk * (1 - h[i1])
            g[i1] -= wk * h[i1 + sp]
    return g

def _viol_obj(h, tau):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    exc = np.maximum(c - tau, 0)
    return float(np.sum(exc * exc))

def _viol_grad(h, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        cv = c[k] * dx
        if cv <= tau:
            continue
        exc = 2.0 * (cv - tau) * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += exc * (1 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= exc * h[i1]
        else:
            sp = -sh
            i1 = np.arange(n - sp)
            g[i1 + sp] += exc * (1 - h[i1])
            g[i1] -= exc * h[i1 + sp]
    return g

def _sf(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _slsqp(h, obj, jac, param, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(obj, h, jac=jac, args=(param,), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _cycle_topm(h, m, rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _eval(h)
    hw, cw = h.copy(), best_c5
    for _ in range(rounds):
        if time.time() > deadline:
            break
        c = np.correlate(hw, 1.0 - hw, mode='full')
        idx = np.argpartition(c, -m)[-m:]
        for k in idx:
            g = np.zeros(n)
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1 - hw[i1 + sh])
                if sh > 0:
                    g[i1 + sh] -= hw[i1]
            else:
                sp = -sh
                i1 = np.arange(n - sp)
                g[i1 + sp] += (1 - hw[i1])
                g[i1] -= hw[i1 + sp]
            g *= dx
            for lr in [0.3, 0.15, 0.07, 0.03, 0.01, 0.004]:
                ht = _repair(hw - lr * g, n)
                ct = _eval(ht)
                if ct < cw:
                    hw, cw = ht, ct
                    if cw < best_c5:
                        best_c5, best_h = cw, hw.copy()
                    break
    return best_h, best_c5

def _block_search(h, bs, niter, scale, rng, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    hw, cw = h.copy(), best_c5
    for _ in range(niter):
        if time.time() > deadline:
            break
        st = rng.randint(0, max(1, n - bs + 1))
        en = min(st + bs, n)
        bsz = en - st
        d = scale * rng.randn(bsz)
        d -= d.mean()
        htry = hw.copy()
        htry[st:en] += d
        htry = _repair(htry, n)
        ct = _eval(htry)
        if ct < cw - 1e-15:
            hw, cw = htry, ct
            if cw < best_c5:
                best_c5, best_h = cw, hw.copy()
        elif rng.rand() < 0.004 and ct < cw + 0.002:
            hw, cw = htry, ct
    return best_h, best_c5

def _pairwise(h, scale, niter, rng, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    hw, cw = h.copy(), best_c5
    for _ in range(niter):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = hw[i] + d, hw[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            hw[i], hw[j] = ni, nj
            ct = _eval(hw)
            if ct < cw - 1e-16:
                cw = ct
                if ct < best_c5:
                    best_c5, best_h = ct, hw.copy()
            else:
                hw[i], hw[j] = ni - d, nj + d
    return best_h, best_c5
