def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, idx, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    j = np.arange(n)
    k1 = (n - 1) + idx - j
    dc[k1] += di * (1.0 - h)
    lag_lo = -idx
    lag_hi = n - 1 - idx
    lags = np.arange(lag_lo, lag_hi + 1)
    k2 = (n - 1) + lags
    dc[k2] -= di * h[idx + lags]
    return dc

def _directed_move(h, c, rng, n, scale):
    """Return (i, j, d) for a directed or random pair move."""
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    if rng.rand() < 0.55 and lag != 0:
        if lag > 0:
            prods = h[lag:] * (1.0 - h[:n - lag])
            m = n - lag
            if prods.sum() < 1e-12:
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    return None
                return (i, j, scale * rng.randn())
            p = prods / prods.sum()
            mi = rng.choice(m, p=p)
            i, j = int(mi), int(mi) + lag
            d = abs(scale) * (0.3 + 0.7 * rng.rand())
        else:
            sp = -lag
            prods = h[:n - sp] * (1.0 - h[sp:])
            m = n - sp
            if prods.sum() < 1e-12:
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    return None
                return (i, j, scale * rng.randn())
            p = prods / prods.sum()
            mi = rng.choice(m, p=p)
            i, j = int(mi), int(mi) + sp
            d = -abs(scale) * (0.3 + 0.7 * rng.rand())
            i, j = j, i
        return (i, j, d)
    elif rng.rand() < 0.3 and lag == 0:
        # Push h[mi] toward 0 or 1
        mi = rng.randint(n)
        mj = rng.randint(n)
        if mi == mj:
            return None
        direction = 1.0 if h[mi] < 0.5 else -1.0
        d = direction * abs(scale) * (0.3 + 0.7 * rng.rand())
        return (mi, mj, d)
    else:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            return None
        return (i, j, scale * rng.randn())

def _incr_search(h, c, rng, n_iter, scale, n, dx, directed=True):
    c5 = c.max() * dx
    for _ in range(n_iter):
        mv = _directed_move(h, c, rng, n, scale) if directed else None
        if mv is None:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
        else:
            i, j, d = mv
        if i < 0 or i >= n or j < 0 or j >= n or i == j:
            continue
        if abs(d) < 1e-15:
            continue
        ni = h[i] + d
        nj = h[j] - d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5 - 1e-15:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5 = new_c5
    return h, c, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Directed incremental search at n0
    h_w = best_h.copy()
    c_w = _corr(h_w)
    n_w = len(h_w)
    dx_w = 2.0 / n_w
    c5_w = c_w.max() * dx_w

    scales1 = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for sc in scales1:
        if _el() > budget_s - 450:
            break
        h_w, c_w, c5_w = _incr_search(h_w, c_w, rng, 50000, sc, n_w, dx_w, True)

    h_w = _repair(h_w, n_w)
    c5_chk = _eval(h_w)
    if c5_chk < best_c5:
        best_c5, best_h, best_n = c5_chk, h_w.copy(), n_w

    # Phase 2: Upsample to 2x resolution
    if _el() < budget_s - 380:
        n1 = n_w * 2
        h_up = np.repeat(best_h, 2)
        h_up = _repair(h_up, n1)
        c_up = _corr(h_up)
        dx_up = 2.0 / n1
        c5_up = c_up.max() * dx_up

        scales2 = [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003, 0.0001]
        for sc in scales2:
            if _el() > budget_s - 220:
                break
            h_up, c_up, c5_up = _incr_search(h_up, c_up, rng, 30000, sc, n1, dx_up, True)

        h_up = _repair(h_up, n1)
        c5_chk = _eval(h_up)
        if c5_chk < best_c5:
            best_c5, best_h, best_n = c5_chk, h_up.copy(), n1

    # Phase 3: Secondary resolution (4x if time permits)
    if _el() < budget_s - 250:
        n2 = best_n * 2
        h_q = np.repeat(best_h, 2)
        h_q = _repair(h_q, n2)
        c_q = _corr(h_q)
        dx_q = 2.0 / n2
        c5_q = c_q.max() * dx_q

        scales3 = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
        for sc in scales3:
            if _el() > budget_s - 120:
                break
            h_q, c_q, c5_q = _incr_search(h_q, c_q, rng, 20000, sc, n2, dx_q, True)

        h_q = _repair(h_q, n2)
        c5_chk = _eval(h_q)
        if c5_chk < best_c5:
            best_c5, best_h, best_n = c5_chk, h_q.copy(), n2

    # Phase 4: Final ultra-fine polish on best solution
    if _el() < budget_s - 20:
        h_f = best_h.copy()
        c_f = _corr(h_f)
        n_f = len(h_f)
        dx_f = 2.0 / n_f
        c5_f = c_f.max() * dx_f
        for sc in [2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6]:
            if _el() > budget_s - 5:
                break
            h_f, c_f, c5_f = _incr_search(h_f, c_f, rng, 35000, sc, n_f, dx_f, True)
        h_f = _repair(h_f, n_f)
        c5_chk = _eval(h_f)
        if c5_chk < best_c5:
            best_c5, best_h, best_n = c5_chk, h_f.copy(), n_f

    # Phase 5: Asymmetric random restart + short search (diversity)
    if _el() < budget_s - 60:
        nc = len(best_h)
        for _r in range(3):
            if _el() > budget_s - 60:
                break
            hp = best_h + 0.015 * rng.randn(nc)
            hp = _repair(hp, nc)
            cp = _corr(hp)
            ncp = len(hp)
            dxp = 2.0 / ncp
            for sc in [0.008, 0.004, 0.002, 0.001]:
                if _el() > budget_s - 60:
                    break
                hp, cp, _ = _incr_search(hp, cp, rng, 15000, sc, ncp, dxp, True)
            hp = _repair(hp, ncp)
            c5_chk = _eval(hp)
            if c5_chk < best_c5:
                best_c5, best_h, best_n = c5_chk, hp.copy(), ncp

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
