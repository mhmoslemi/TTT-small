def _grad_single(k, h):
    n = len(h)

    g = np.zeros(n)

        i1 = np.arange(s, n)

            g[i2] -= h[i2 + s]

        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(n - sp, n)
        g[i2] -= h[i2 - sp]

    """For each bad shift, try to fix it by moving mass."""
    n = len(h)
    target = n * 0.5

        if s >= 0:

import numpy as np
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(52):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_single(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        if s > 0:
            i2 = np.arange(n - s)
            g[i2] -= h[i2 + s]
    else:
        sp = -s
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(n - sp, n)
        g[i2] -= h[i2 - sp]
    return g

def _smooth_grad_full(h, alpha):
    n = len(h)
    c = _corrs_raw(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _fista_projected(h0, alpha, n_iter, target):
    n = len(h0)
    x = _proj_to_sum(h0, target)
    x_prev = x.copy()
    y = x.copy()
    t_k = 1.0
    step = 0.02
    best_h = x.copy()
    best_f = _eval(x)
    for it in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        g = _smooth_grad_full(y, alpha)
        for _bt in range(20):
            x_new = _proj_to_sum(y - step * g, target)
            f_new = _eval(x_new)
            if f_new < best_f + 1e-10 or step < 1e-9:
                break
            step *= 0.5
        if it > 0:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = x + ((t_k - 1.0) / t_new) * (x - x_prev)

            t_k = t_new

            y = x_new.copy()
        x_prev = x
        x = x_new
        step = min(step * 1.08, 0.08)
        if it % 100 == 0 or it == n_iter - 1:
            f = _eval(x)
            if f < best_f:
                best_f = f
                best_h = x.copy()
    return best_h, best_f

def _active_descent(h, topm, max_iter, step0=0.005):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    step = step0
    stall = 0
    for _ in range(max_iter):
        if time.time() - _t0 > _budget - 12:
            break
        c = _corrs_raw(h)
        top_idx = np.argsort(c)[::-1][:topm]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_single(k, h)
        g *= (2.0 / (n * topm))
        found = False
        for _bt in range(38):
            h_new = _proj_to_sum(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 5e-8:
                h = h_new
                f = f_new
                found = True
                stall = 0
                break
            step *= 0.5
        if not found:
            stall += 1
            if stall > 30:
                break
            step = min(step * 1.15, step0)
        else:
            step = min(step * 1.04, step0)
    return h, f

def _shift_fix(h, n_rounds):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(12345)
    for _ in range(n_rounds):

            break

        k_star = c.argmax()
        s = k_star - (n - 1)
        if s >= 0:
            if s == 0:
                break
            valid = np.arange(s, n)
        else:
            if s == -(n - 1) and s != 1:
                valid = np.arange(-(s + 1), -(s + 1) + 2)
            sp = -s
            valid = np.arange(n - sp)
        if len(valid) < 2:
            break
        contrib = np.zeros(n)
        if s >= 0:
            contrib[valid] = h[valid] * (1.0 - h[valid - s])
        else:
            sp = -s
            contrib[valid] = h[valid] * (1.0 - h[valid + sp])
        top = valid[np.argsort(contrib)[::-1]][:15]
        for idx in top:
            if s >= 0:
                j = idx - s
            else:
                j = idx + (-s)
            if j < 0 or j >= n:
                continue
            if h[idx] < h[j]:
                idx, j = j, idx
            for dm in [0.02, 0.05, 0.01]:

                h[j] += dm
                h = _proj_to_sum(h, target)
                fn = _eval(h)
                if fn < f - 1e-14:
                    f = fn
                    break
                h[idx] += dm
                h[j] -= dm
                h = _proj_to_sum(h, target)
    return h, f

def _pairwise(h, n_iter, scale, seed_val):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() - _t0 > _budget - 8:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    fi = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(fi(x_new), 0, 1), n * factor)
