    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _make_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    s = int(h.sum()); target = n // 2
    if s > target:
        ones = np.where(h == 1.0)[0]
        flip = rng.choice(ones, s - target, replace=False); h[flip] = 0.0
    elif s < target:
        zeros = np.where(h == 0.0)[0]
        flip = rng.choice(zeros, target - s, replace=False); h[flip] = 1.0
    return h

def _obj_soft(x, alpha, n, t0, timeout):
    if time.time() - t0 > timeout:
        raise Exception("timeout")
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha + 1e6 * (x.sum() - n * 0.5) ** 2

def _lfbfgs(h, alpha, maxiter, timeout):
    n = len(h)
    t0 = time.time()
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_soft, h, args=(alpha, n, t0, timeout),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_search(h, rng, max_time, informed=True):
    n = len(h)
    h = _make_binary(h, n, rng)
    c5 = _eval(h)
    best = h.copy(); best_c5 = c5
    t_s = time.time()
    while time.time() - t_s < max_time:
        C = _corr(h)
        m = C.max()
        if informed:
            topk = np.argsort(C)[-3:]
            k_best = topk[-1]
            shift = k_best - (n - 1)
            i1 = 0; i2 = n - 1
            if shift >= 0:
                i1 = max(0, shift); i2 = min(n, n - shift)
            else:
                i1 = 0; i2 = min(n, n + shift)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            good_a = [a for a in ones if a >= i1 and a < i2 and (a + shift) >= 0 and (a + shift) < n and h[a + shift] == 0] if shift >= 0 else \
                     [a for a in ones if a + shift >= 0 and a + shift < n and h[a + shift] == 0]
            good_b = [b for b in zeros if b >= i1 and b < i2 and (b + shift) >= 0 and (b + shift) < n and h[b + shift] == 1] if shift >= 0 else \
                     [b for b in zeros if b + shift >= 0 and b + shift < n and h[b + shift] == 1]
            if good_a and good_b:
                a = good_a[rng.randint(len(good_a))]
                b = good_b[rng.randint(len(good_b))]
            else:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
        else:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0; h[b] = 1.0
        c5_new = _eval(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new; best = h.copy()
        else:
            h[a] = 1.0; h[b] = 0.0
    return best, best_c5

def _sim_anneal(h, rng, max_time, T0=0.01, T1=1e-5):
    n = len(h)
    h = _make_binary(h, n, rng)
    c5 = _eval(h)
    best = h.copy(); best_c5 = c5
    t_s = time.time(); total_t = max_time
    while time.time() - t_s < max_time:
        frac = (time.time() - t_s) / total_t
        T = T0 * (T1 / T0) ** frac
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0; h[b] = 1.0
        c5_new = _eval(h)
        if c5_new < c5 or rng.rand() < np.exp(-(c5_new - c5) / (T + 1e-15)):
            c5 = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new; best = h.copy()
        else:
            h[a] = 1.0; h[b] = 0.0
    return best, best_c5

def _periodic_opt(h_init, n, T, alpha, maxiter, timeout):
    if n % T != 0:
        return h_init.copy(), _eval(h_init)
    period = h_init[:T].copy()
    m = n // T
    def _pobj(x, al, nv, t0, to):
        if time.time() - t0 > to:
            raise Exception("timeout")
        hh = np.tile(x, nv)
        c = np.correlate(hh, 1.0 - hh, mode='full')
        mx = c.max()
        ss = np.clip(al * (c - mx), -500, 500)
        return mx + np.log(float(np.sum(np.exp(ss)))) / al + 1e6 * (x.sum() - T * 0.5) ** 2
    bnds = [(0.0, 1.0)] * T
    t0 = time.time()
    try:
        res = minimize(_pobj, period, args=(alpha, m, t0, timeout),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        res_x = _repair(res.x, T)
        full = np.tile(res_x, m)
        full = _repair(full, n)
        return full, _eval(full)
    except Exception:
        return h_init.copy(), _eval(h_init)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best_h = h0.copy(); best_c5 = _eval(best_h); best_n = n0

    # Phase 1: L-BFGS-B continuous polish
    for alpha in [5000, 50000, 200000]:
        if time.time() - t0 > budget_s - 400:
            break
        h1 = _lfbfgs(best_h, alpha, 2000, 25)
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5 = c5; best_h = h1.copy()

    # Phase 2: Informed binary local search
    if time.time() - t0 < budget_s - 350:
        tb = min(200.0, budget_s - t0 - 350)
        h2, c52 = _binary_search(best_h, rng, tb, informed=True)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2.copy()

    # Phase 3: Simulated annealing
    if time.time() - t0 < budget_s - 200:
        ta = min(180.0, budget_s - t0 - 200)
        h3, c53 = _sim_anneal(best_h, rng, ta)
        if c53 < best_c5:
            best_c5 = c53; best_h = h3.copy()

    # Phase 4: More binary search from current best
    if time.time() - t0 < budget_s - 150:
        tb2 = min(120.0, budget_s - t0 - 150)
        h4, c54 = _binary_search(best_h, rng, tb2, informed=True)
        if c54 < best_c5:
            best_c5 = c54; best_h = h4.copy()

    # Phase 5: Periodic optimization at current n
    if time.time() - t0 < budget_s - 100:
        divisors = [d for d in range(2, best_n // 2 + 1) if best_n % d == 0]
        for T in divisors[:15]:
            if time.time() - t0 > budget_s - 100:
                break
            for alpha in [20000, 100000]:
                if time.time() - t0 > budget_s - 100:
                    break
                h5, c55 = _periodic_opt(best_h, best_n, T, alpha, 2000, 15)
                if c55 < best_c5:
                    best_c5 = c55; best_h = h5.copy()

    # Phase 6: Try nearby resolutions
    if time.time() - t0 < budget_s - 80:
        for dn in [-10, 10, -20, 20, 30, -30]:
            if time.time() - t0 > budget_s - 80:
                break
            nt = best_n + dn
            if nt < 50 or nt % 2 != 0:
                continue
            hr = _repair(_resample(best_h, nt), nt)
            hr = _lfbfgs(hr, 50000, 1000, 20)
            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5 = c5r; best_h = hr.copy(); best_n = nt

    # Phase 7: Final binary search
    if time.time() - t0 < budget_s - 20:
        tb3 = max(5.0, budget_s - t0 - 20)
        h7, c57 = _binary_search(best_h, rng, tb3, informed=True)
        if c57 < best_c5:
            best_c5 = c57; best_h = h7.copy()

    # Phase 8: Final L-BFGS-B polish
    if time.time() - t0 < budget_s - 2:
        h8 = _lfbfgs(best_h, 500000, 2000, max(5, budget_s - t0 - 2))
        c58 = _eval(h8)
        if c58 < best_c5:
            best_c5 = c58; best_h = h8.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
