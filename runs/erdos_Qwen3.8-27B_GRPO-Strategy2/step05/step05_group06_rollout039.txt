def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsoring(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_of_corr_lag(h, k, n):
    """Gradient of corr[k] w.r.t. h"""
    dx = 2.0 / n

        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

    dx = 2.0 / n

    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])

            g[np.arange(sh, n)] -= h[np.arange(sh, n) - sh]

        g[np.arange(sp, n)] += (1.0 - h[np.arange(sp, n) - sp])
        g[np.arange(n - sp)] -= h[np.arange(n - sp) + sp]
    return g * dx

    return g

    f = _sobj(h, a)
    for _ in range(ns):
        g = _sgrad(y, a)
        y = h + (t_ - 1) / (t_ + 2) * (y - h)
        hn = _psum(y - st * g, n * .5)
        fn = _sobj(hn, a)
        if fn > f + 1e-8: y, t_ = h.copy(), 1.
        else: h, f = hn, fn
        t_ = .5 * (1 + np.sqrt(1 + 4 * t_ ** 2))
    return _repair(h, n)

def _bins(h, ns, rg):
    n = len(h); hb = (h > .5).astype(float)
    on, tg = int(hb.sum()), n // 2
    if on > tg:
        ix = np.where(hb == 1)[0]; rg.shuffle(ix); hb[ix[:on - tg]] = 0
    elif on < tg:
        ix = np.where(hb == 0)[0]; rg.shuffle(ix); hb[ix[:tg - on]] = 1
    c5 = _eval(hb)
    for _ in range(ns):
        i, j = rg.randint(n), rg.randint(n)
        if i == j or hb[i] == hb[j]: continue
        hb[i], hb[j] = hb[j], hb[i]
        cn = _eval(hb)
        if cn < c5: c5 = cn
        else: hb[i], hb[j] = hb[j], hb[i]
    return hb, c5

def _pso(h0, np_, ni, rg, t0, tl):
    n = len(h0)
    P = np.array([_repair(h0 + .05 * rg.randn(n), n) for _ in range(np_)])
    V = .05 * rg.randn(np_, n)
    PB = P.copy(); PC = np.array([_eval(p) for p in P])
    gi = np.argmin(PC); GB = P[gi].copy(); GC = PC[gi]
    w, c1, c2 = .7, 1.5, 1.5
    for it in range(ni):
        if time.time() - t0 > tl: break
        for i in range(np_):
            V[i] = w * V[i] + c1 * rg.rand() * (PB[i] - P[i]) + c2 * rg.rand() * (GB - P[i])
            hn = _repair(P[i] + V[i], n)
            c5 = _eval(hn)
            if c5 < PC[i]:
                PB[i], PC[i] = hn, c5
                if c5 < GC: GB, GC = hn.copy(), c5
            P[i] = hn
        w *= .998
    return GB, GC

def _down(h, nc):
    n = len(h)
    return np.array([h[i * n // nc:(i + 1) * n // nc].mean() for i in range(nc)])

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rg = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, float), n0)
    BH, BC, BN = h.copy(), _eval(h), n0

    # P1: Coarse-to-fine
    if time.time() - t0 < budget_s - 350:
        nc = max(28, n0 // 4)
        hc = _repair(_down(h, nc), nc)
        for a in [2000, 20000, 150000]:
            if time.time() - t0 > budget_s - 320: break
            hc = _psl(hc, a, 800, 1e-18)
            hu = _repair(np.repeat(hc, n0 // len(hc)), n0)
            c5 = _eval(hu)
            if c5 < BC: BC, BH, BN = c5, hu.copy(), n0
        hu = _repair(np.repeat(hc, n0 // len(hc)), n0)
        for a in [10000, 80000]:
            if time.time() - t0 > budget_s - 270: break
            hu = _psl(hu, a, 1000, 1e-18); c5 = _eval(hu)
            if c5 < BC: BC, BH, BN = c5, hu.copy(), n0

    # P2: SLSQP increasing alpha
    for a in [3000, 20000, 100000, 400000]:
        if time.time() - t0 > budget_s - 250: break
        h2 = _psl(BH, a, 1500, 1e-18); c5 = _eval(h2)
        if c5 < BC: BC, BH, BN = c5, h2.copy(), BN

    # P3: FISTA
    if time.time() - t0 < budget_s - 200:
        for a, ns, st in [(20000, 1200, .008), (100000, 600, .003), (300000, 300, .001)]:
            if time.time() - t0 > budget_s - 160: break
            h2 = _fista(BH, a, ns, st); c5 = _eval(h2)
            if c5 < BC: BC, BH, BN = c5, h2.copy(), BN

    # P4: Binary-continuous alternation
    if time.time() - t0 < budget_s - 120:
        for _ in range(3):
            if time.time() - t0 > budget_s - 80: break
            hb, _ = _bins(BH, 6000, rg)
            c5b = _eval(hb)
            if c5b < BC: BC, BH, BN = c5b, hb.copy(), BN
            for a in [5000, 50000]:
                if time.time() - t0 > budget_s - 60: break
                h2 = _psl(hb, a, 600, 1e-18); c5 = _eval(h2)
                if c5 < BC: BC, BH, BN = c5, h2.copy(), BN

    # P5: PSO
    if time.time() - t0 < budget_s - 100:
        h2, c5 = _pso(BH, 20, 100, rg, t0, budget_s - 60)
        if c5 < BC: BC, BH, BN = c5, h2.copy(), BN

    # P6: Multi-start SLSQP
    if time.time() - t0 < budget_s - 60:
        for r in range(10):
            if time.time() - t0 > budget_s - 40: break
            nc = BN; s = rg.randint(5)
            if s == 0: hp = _repair(BH + .03 * rg.randn(nc), nc)
            elif s == 1: hp = _repair(1. - BH, nc)
            elif s == 2: hp = _repair(np.roll(BH, rg.randint(1, nc // 4)), nc)
            elif s == 3:
                hp = BH.copy(); f = rg.randint(2, nc // 2)
                hp += .02 * rg.randn() * np.sin(2 * np.pi * f * np.arange(nc) / nc)
                hp = _repair(hp, nc)
            else:
                hp = BH.copy(); l = nc // 8
                b1, b2 = sorted(rg.randint(0, nc - l, 2))
                hp[b1:b1 + l], hp[b2:b2 + l] = hp[b2:b2 + l].copy(), hp[b1:b1 + l].copy()
                hp = _repair(hp, nc)
            h2 = _psl(hp, [5000, 50000, 200000][rg.randint(3)], 500, 1e-18)
            c5 = _eval(h2)
            if c5 < BC: BC, BH, BN = c5, h2.copy(), nc

    # P7: 2x resolution
    if time.time() - t0 < budget_s - 130:
        n1 = BN * 2
        hh = _repair(np.repeat(BH, 2), n1)
        for a in [10000, 100000]:
            if time.time() - t0 > budget_s - 65: break
            hh = _psl(hh, a, 600, 1e-18); c5 = _eval(hh)
            if c5 < BC: BC, BH, BN = c5, hh.copy(), n1

    # P8: Fine coordinate descent with adaptive step
    if time.time() - t0 < budget_s - 5:
        h = BH.copy(); nc = len(h); c5c = _eval(h)
        for _ in range(40000):
            if time.time() - t0 > budget_s - 2: break
            i, j = rg.randint(nc), rg.randint(nc)
            if i == j: continue
            for d in [3e-5, -3e-5, 1e-5, -1e-5, 8e-5, -8e-5]:
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj; c5n = _eval(h)
                    if c5n < c5c:
                        c5c = c5n
                        if c5n < BC: BC, BH, BN = c5n, h.copy(), nc
                        break
                    else: h[i], h[j] = ni - d, nj + d

    # P9: Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        h2 = _psl(BH, 500000, 800, 1e-19)
        c5 = _eval(h2)
        if c5 < BC: BC, BH, BN = c5, h2.copy(), len(h2)

    BH = _repair(BH, BN); BC = _eval(BH)
    return (BH, BC, BN)
