from scipy.optimize import minimize
import time

_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_vec(h, top_idx):
    """Vectorized gradient for top-m active lags."""
    n = len(h)
    g = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            sp = -s
            g[:n-sp] += (1.0 - h[sp:])
            g[n-sp:] -= h[:n-sp]
    return g / len(top_idx) * (2.0 / n)

def _pgd(h, topm, max_iter, step0, time_limit):
    """Projected gradient descent with backtracking."""
    n = len(h)
    target = n * 0.5
    h = h.copy().astype(float)
    f = _eval(h)
    step = step0
    stuck = 0
    for it in range(max_iter):
        if time.time() - _t0 > time_limit:
            break
        c = np.correlate(h, 1.0 - h, 'full')
        top_idx = np.argsort(c)[-topm:]
        g = _grad_vec(h, top_idx)
        improved = False
        for _ in range(40):
            h_new = _proj(h - step * g, target)
            f_new = _eval(h_new)
            if f_new < f - 1e-7:
                h = h_new
                f = f_new
                improved = True
                break
            step *= 0.5
        if not improved:
            stuck += 1
            if stuck > 50:
                break
            # Try a random perturbation
            h = _repair(h + 0.001 * np.random.randn(n), n)
            f = _eval(h)
            step = step0
        else:
            stuck = 0
            step = min(step * 1.1, step0)
    return h, f

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a; d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a; d += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, tgt):

        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
