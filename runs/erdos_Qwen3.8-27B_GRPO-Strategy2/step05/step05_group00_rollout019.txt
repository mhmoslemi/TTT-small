def _slsqp_refine(h, alpha, maxiter, ftol, time_limit):

    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)

                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g

        h_sd, c5_sd = _subgrad_descent(best_h, n, sd_end, 5000, tk)
        if c5_sd < best_c5:
            best_c5 = c5_sd
            best_h = h_sd.copy()

    # Phase 4: SLSQP polish (15% budget)
    slp_end = t0 + budget_s * 0.80
    for alpha in [5000, 50000, 500000, 2000000]:
        if time.time() > slp_end:
            break
        h_s = _slsqp_refine(best_h, alpha, 3000, 1e-20)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s.copy()

    # Phase 5: SA from refined continuous solution (10% budget)
    sa3_end = t0 + budget_s * 0.92
    h_bin3 = np.clip(np.round(best_h), 0, 1)
    h_bin3 = _repair(h_bin3, n)
    h_s3, c5_s3 = _sa_binary(h_bin3, n, 0.003, 1e-9, 4000000, rng, sa3_end)
    if c5_s3 < best_c5:
        best_c5 = c5_s3
        best_h = h_s3.copy()

    # Phase 6: Final targeted + subgradient
    if time.time() < t0 + budget_s * 0.97:
        h_tg2, c5_tg2 = _targeted_sa(best_h, n, rng, t0 + budget_s * 0.97, 20)
        if c5_tg2 < best_c5:
            best_c5 = c5_tg2
            best_h = h_tg2.copy()

    # Phase 7: Try nearby n
    if time.time() < t0 + budget_s * 0.99:
        for dn in [2, -2, 4, -4, 6, -6]:
            if time.time() > t0 + budget_s * 0.99:
                break
            n_try = n + dn
            if n_try < 50 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            h_r = _repair(h_r, n_try)
            h_r_s, c5_r = _sa_binary(h_r, n_try, 0.003, 1e-8, 1500000, rng,
                                     t0 + budget_s * 0.99)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_s.copy()
                n = n_try

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
