import numpy as np
from scipy.optimize import linprog, minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        for m in range(n):
            if m < k:
                if m + k < n:
                    g[m] = -h[m + k]
            elif m + k < n:
                g[m] = 1.0 - h[m - k] - h[m + k]
            else:
                g[m] = 1.0 - h[m - k]
    else:
        s = -k
        for m in range(n):
            if m + s >= n:
                g[m] = 0  # shouldn't happen
            elif m < s:
                g[m] = 1.0 - h[m + s]
            elif m + s < n:
                g[m] = 1.0 - h[m + s] - h[m - s]
            else:
                g[m] = -h[m - s]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Adjust to meet sum constraint
    # Use bisection on shift
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_step(h, h_cur, top_k, lam):
    """One cutting-plane LP step."""
    n = len(h)
    lags = _all_lags(h_cur)
    # Get top_k lags
    idx = np.argsort(lags)[::-1][:top_k]
    val_cur = _eval(h_cur)
    
    # Build LP
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub_rows = []
    b_ub_vals = []
    for i in idx:
        k = i - (n - 1)
        g = _lag_grad(h_cur, k)
        val_k = lags[i]
        # g @ h - t <= g @ h_cur - val_k
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        rhs = g @ h_cur - val_k
        A_ub_rows.append(row)
        b_ub_vals.append(rhs)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return h.copy()
    h_new = res.x[:n]
    # Blend with current for stability
    h_blend = (1 - lam) * h + lam * h_new
    return _repair(h_blend, n)

def _all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _pairwise_descent(h, n_steps, rng):
    n = len(h)
    c_cur = _eval(h)
    for _ in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        c_new = _eval(h)
        if c_new < c_cur:
            c_cur = c_new
        else:
            h[i] -= d
            h[j] += d
    return h

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        for m in range(n):
            if m < k:
                if m + k < n:
                    g[m] = -h[m + k]
            elif m + k < n:
                g[m] = 1.0 - h[m - k] - h[m + k]
            else:
                g[m] = 1.0 - h[m - k]
    else:
        s = -k
        for m in range(n):
            if m < s:
                if m + s < n:
                    g[m] = 1.0 - h[m + s]
            elif m + s < n:
                g[m] = 1.0 - h[m + s] - h[m - s]
            else:
                g[m] = -h[m - s]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_step(h_cur, max_lags=80):
    n = len(h_cur)
    lags = _all_lags(h_cur)
    idx = np.argsort(lags)[::-1][:max_lags]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub_rows = []
    b_ub_vals = []
    for i in idx:
        k = i - (n - 1)
        g = _lag_grad(h_cur, k)
        val_k = lags[i]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        rhs = float(g @ h_cur) - val_k
        A_ub_rows.append(row)
        b_ub_vals.append(rhs)
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return h_cur.copy()
    return _repair(res.x[:n], n)

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _pairwise_descent(h, n_steps, rng, scale=1e-4):
    n = len(h)
    c_cur = _eval(h)
    for _ in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        h[i] += d
        h[j] -= d
        if not (0 <= h[i] <= 1 and 0 <= h[j] <= 1):
            h[i] -= d
            h[j] += d
            continue
        c_new = _eval(h)
        if c_new < c_cur:
            c_cur = c_new
        else:
            h[i] -= d
            h[j] += d
    return h
