def _optimal_delta(c, dc, h_i, h_j, n):
    # Find optimal delta for pair (i,j): h[i]+delta, h[j]-delta
    # c: current correlation (length 2n-1)
    # dc: change in correlation per unit delta (length 2n-1)
    # Returns optimal delta
    
    # Bounds on delta
    d_lo = max(-h_i, h_j - 1.0)
    d_hi = min(1.0 - h_i, h_j)
    if d_lo >= d_hi:
        return 0.0
    
    # f(d) = max_k (c[k] + d * dc[k])
    # Find minimum over d in [d_lo, d_hi]
    
    # Evaluate at boundaries
    f_lo = np.max(c + d_lo * dc)
    f_hi = np.max(c + d_hi * dc)
    
    # Check if d=0 is already a local min
    # Subgradient at 0: dc[k] for k in argmax(c)
    m0 = c.max()
    active = np.where(c >= m0 - 1e-14)[0]
    if len(active) > 0:
        if dc[active].min() <= 0 <= dc[active].max():
            # 0 is in subgradient, so d=0 is optimal (locally)
            # But check boundaries too
            return 0.0
    
    # Vertex walking
    # At any point d, the max is achieved by some set of k's
    # Walk from d_lo to d_hi
    
    best_d = d_lo
    best_f = f_lo
    
    # Try a grid of points first
    n_grid = 200
    ds = np.linspace(d_lo, d_hi, n_grid)
    # Vectorized: compute f at all grid points
    # f(d) = max_k (c[k] + d*dc[k])
    # = max_k (c[k] + dc[k]*d) for each k, then max over k
    # Shape: (n_grid, 2n-1)
    vals = c[None, :] + ds[:, None] * dc[None, :]
    f_vals = vals.max(axis=1)
    best_idx = f_vals.argmin()
    best_d = ds[best_idx]
    best_f = f_vals[best_idx]
    
    # Refine around best_d
    if 0 < best_idx < n_grid - 1:
        d1, d2 = ds[best_idx-1], ds[best_idx+1]
        ds2 = np.linspace(d1, d2, 200)
        vals2 = c[None, :] + ds2[:, None] * dc[None, :]
        f_vals2 = vals2.max(axis=1)
        best_idx2 = f_vals2.argmin()
        best_d = ds2[best_idx2]
        best_f = f_vals2[best_idx2]
    
    return best_d

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, n):
    """Change in full correlation when h[i] increases by 1."""
    dc = np.zeros(2 * n - 1)
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= h[i + sp[m2]]
    return dc

def _optimal_delta(c, dc, hi, hj, n_grid=200):
    """Find(delta) that minimizes max_k(c[k] + delta*dc[k])
    subject to 0 <= hi+delta <= 1 and 0 <= hj-delta <= 1."""
    d_lo = max(-hi, hj - 1.0)
    d_hi = min(1.0 - hi, hj)
    if d_lo >= d_hi:
        return 0.0
    if d_lo > 0:
        d_lo = 0.0  # Allow staying
    if d_hi < 0:
        d_hi = 0.0
    
    # Actually, we should allow delta in [d_lo, d_hi] which already includes 0
    # (since hi >= 0, hj >= 0, hi <= 1, hj <= 1, we have -hi <= 0 <= 1-hi and hj-1 <= 0 <= hj)
    
    grid = np.linspace(d_lo, d_hi, n_grid)
    vals = c[None, :] + grid[:, None] * dc[None, :]
    f_vals = vals.max(axis=1)
    best_idx = f_vals.argmin()
    best_d = grid[best_idx]
    best_f = f_vals[best_idx]
    
    # Refine
    if 0 < best_idx < n_grid - 1:
        d1, d2 = grid[best_idx - 1], grid[best_idx + 1]
        grid2 = np.linspace(d1, d2, 100)
        vals2 = c[None, :] + grid2[:, None] * dc[None, :]
        f_vals2 = vals2.max(axis=1)
        best_idx2 = f_vals2.argmin()
        best_d = grid2[best_idx2]
    
    return best_d

        grid2 = np.linspace(d1, d2, 100)
        vals2 = c[None, :] + grid2[:, None] * dc[None, :]
        f_vals2 = vals2.max(axis=1)
        best_d = grid2[f_vals2.argmin()]
    return best_d

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g
