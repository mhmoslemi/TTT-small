```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, n):
    """Gradient of c[k] = correlate(h, 1-h)[k] with respect to h"""
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _topk_grad(h, topk=10):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(h, k, n)
    return g / topk * (2.0 / n)

def _adam_opt(h, nsteps, lr0, topk, decay_every=100):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    best_c5 = _eval(h)
    h_best = h.copy()
    
    for step in range(nsteps):
        # Recompute top-k indices periodically
        if step % decay_every == 0:
            c = _corr(h)
            topk_idx = np.argsort(c)[::-1][:topk]
        
        # Compute gradient for top-k lags
        g = np.zeros(n)
        for k in topk_idx:
            g += _grad_lag(h, k, n)
        g *= (2.0 / n) / topk
        
        # ADAM update
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj_sum(h_new, n * 0.5)
        
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h_new.copy()
        h = h_new
        
        # Decay learning rate
        if step % (decay_every * 5) == 0 and step > 0:
            lr *= 0.7
    
    return h_best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        g[:n - sh] += (1.0 - h[sh:])
        g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(h, k, n)
    return g * (2.0 / n) / topk

def _adam_opt(h, nsteps, lr0, topk):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    best_c5 = _eval(h)
    h_best = h.copy()
    for step in range(nsteps):
        if step % 50 == 0:
            c = _corr(h)
            topk_idx = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for k in topk_idx:
            g += _grad_lag(h, k, n)
        g *= (2.0 / n) / topk
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj_sum(h_new, n * 0.5)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h_new.copy()
        h = h_new
        if step > 0 and step % 250 == 0:
            lr *= 0.85
    return h_best, best_c5

def _lp_step(h, n):
    c = _corr(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        G[k] = _grad_lag(h, k, n)
    # rhs_k = t - c[k] + G[k] @ h
    # We want: G[k] @ x <= t - c[k] + G[k] @ h for all k
    # which is: G[k] @ x - t <= -c[k] + G[k] @ h
    rhs = -c + G @ h
    x = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [x >= 0, x <= 1, cp.sum(x) == n / 2]
    A_mat = np.hstack([G, -np.ones((n_lags, 1))])
    constraints.append(A_mat @ cp.concatenate([x, t_var]) <= rhs)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CLP)
    except:
        try:
            prob.solve(solver=cp.GLPK)
        except:
            return h.copy()
    if x.value is not None:
        return _repair(x.value, n)
    return h.copy()

def _lp_phase(h, max_steps, time_limit):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    t_start = time.time()
    for i in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        h_new = _lp_step(h_cur, n)
        c5_new = _eval(h_new)
        if c5_new < c5_cur:
            h_cur, c5_cur = h_new, c5_new
        else:
            # Line search: try alpha * LP solution
            for alpha in [0.5, 0.25, 0.1]:
                h_ls = (1 - alpha) * h_cur + alpha * h_new
                h_ls = _repair(h_ls, n)
                if _eval(h_ls) < c5_cur:
                    h_cur = h_ls
                    c5_cur = _eval(h_cur)
                    break
    return h_cur, c5_cur

def _local_search(h, nsteps, rng, step_sizes):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for ss in step_sizes:
        for _ in range(nsteps // len(step_sizes)):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = ss * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        g[:n - sh] += (1.0 - h[sh:])
        g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam_opt(h, nsteps, lr0, topk):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    c = _corr(h)
    topk_idx = np.argsort(c)[::-1][:topk]
    best_c5 = _eval(h)
    h_best = h.copy()
    for step in range(nsteps):
        if step % 30 == 0:
            c = _corr(h)
            topk_idx = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for k in topk_idx:
            g += _grad_lag(h, k, n)
        g *= (2.0 / n) / topk
        t += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj_sum(h_new, n * 0.5)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h_new.copy()
        h = h_new
        if step > 0 and step % 300 == 0:
            lr *= 0.82
    return h_best, best_c5

def _lp_step(h, n):
    c = _corr(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        G[k] = _grad_lag(h, k, n)
    rhs = -c + G @ h
    x = cp.Variable(n)
    tv = cp.Variable()
    constraints = [x >= 0, x <= 1, cp.sum(x) == n / 2]
    cat = cp.vstack([x, tv])
    A = np.vstack([G, -np.ones((1, n_lags)) if False else np.zeros((0, n_lags))])
    A_full = np.hstack([G, -np.ones((n_lags, 1))])
    constraints.append(A_full @ cat <= rhs)
    prob = cp.Problem(cp.Minimize(tv), constraints)
    try:
        prob.solve(solver=cp.CLP, max_iters=10000)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            return h.copy()
    if x.value is None:
        return h.copy()
    return _repair(x.value, n)

def _lp_phase(h, max_steps, time_limit):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    t_start = time.time()
    for i in range(max_steps):
        if time.time() - t_start > time_limit:
            break
        h_new = _lp_step(h_cur, n)
        c5_new = _eval(h_new)
        if c5_new <= c5_cur + 1e-10:
            h_cur, c5_cur = h_new, c5_new
        else:
            improved = False
            for alpha in (0.5, 0.25, 0.1, 0.05):
                h_ls = _repair((1.0 - alpha) * h_cur + alpha * h_new, n)
                if _eval(h_ls) < c5_cur:
                    h_cur = h_ls
                    c5_cur = _eval(h_cur)
                    improved = True
                    break
            if not improved:
                continue
    return h_cur, c5_cur

def _local_search(h, nsteps, rng, step_sizes):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    per_size = nsteps // len(step_sizes)
    for ss in step_sizes:
        for _ in range(per_size):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = ss * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5
```
