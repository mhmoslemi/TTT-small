    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    exc = np.maximum(c-tau, 0)
    return np.sum(exc*exc) + mu*(h.sum()-n/2.0)**2

def _viol_penal_grad(h, tau, mu):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    g = np.zeros(n)
    for k in range(2*n-1):
        if c[k] <= tau: continue
        g += 2.0*(c[k]-tau) * _grad_one_lag(h, (n-1)-k, dx)
    g += 2*mu*(h.sum()-n/2.0)
    return g

        if s > target: lo = mid
        else: hi = mid

def _nesterov(h, n, max_iter, t0, deadline):

            g += _grad_one_lag(y, (n-1)-k, dx)

# Helper functions (top-level, no closures)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        i1 = np.arange(n-sp)
        g[i1] += dx * (1.0 - h[i1+sp])
        i2 = np.arange(sp, n)
        g[i2] -= dx * h[i2-sp]
    return g

def _lse_p_obj(h, alpha, mu):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    lse = m + np.log(np.sum(np.exp(s)))/alpha
    return lse + mu*(h.sum()-n/2.0)**2

def _lse_p_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s-s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        g += w[k] * _grad_lag(h, (n-1)-k, dx)
    g += 2.0*mu*(h.sum()-n/2.0)
    return g

def _viol_p_obj(h, tau, mu):
    n = len(h)
    c = _cvec(h)
    exc = np.maximum(c-tau, 0.0)
    return np.sum(exc*exc) + mu*(h.sum()-n/2.0)**2

def _viol_p_grad(h, tau, mu):
    n = len(h)
    dx = 2.0/n
    c = _cvec(h)
    g = np.zeros(n)
    for k in range(2*n-1):
        if c[k] <= tau: continue
        g += 2.0*(c[k]-tau) * _grad_lag(h, (n-1)-k, dx)
    g += 2.0*mu*(h.sum()-n/2.0)
    return g

def _proj_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _lbfgs(h, obj, grad, alpha_or_tau, mu, maxiter, deadline, t0):
    n = len(h)
    try:
        res = minimize(obj, h.copy(), jac=grad, args=(alpha_or_tau, mu),
                      method='L-BFGS-B', bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time()-t0 < deadline:
            return _repair(res.x, n)
    except:
        pass
    return _repair(h, n)

def _slsqp_sum(h, obj, grad, alpha_or_tau, maxiter, deadline, t0):
    n = len(h)
    try:
        res = minimize(obj, h, jac=grad, args=(alpha_or_tau,),
                      method='SLSQP', bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun': lambda x: x.sum()-n/2.0,
                                    'jac': lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _sum_fun(x, n): return x.sum()-n/2.0
def _sum_jac(x, n): return np.ones(n)

def _upsample(h, f):
    return np.repeat(h, f)
