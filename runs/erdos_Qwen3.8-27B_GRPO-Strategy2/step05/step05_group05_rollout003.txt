import numpy as np
from scipy.signal import correlate as _scorr
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = _scorr(h, 1.0 - h, mode='full', method='fft')
    return float(np.max(c) * 2.0 / n)

def _eval_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(h, s):
    """Gradient of c_s (correlation at shift s) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:n - s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _active_grad(h, k_list):
    """Weighted gradient of top-k correlations."""
    n = len(h)
    c = _scorr(h, 1.0 - h, mode='full', method='fft')
    # Get top k
    idx = np.argsort(c)[::-1][:len(k_list)]
    max_k = idx[0]
    # Weights: exponential of excess
    excess = c[idx] - c[idx].min()
    w = np.exp(0.5 * excess)
    w /= w.sum()
    g = np.zeros(n)
    for wi, k in zip(w, idx):
        s = (n - 1) - k
        g += wi * _grad_shift(h, s)
    return g * (2.0 / n), c

def _proj_grad_step(h, n, eta, k_top, max_iter):
    """Projected gradient descent with top-k active set."""
    c5 = _eval(h)
    v = np.zeros(n)
    beta = 0.7  # momentum
    for it in range(max_iter):
        g, c = _active_grad(h, k_top)
        # Backtracking line search
        improved = False
        for _ in range(10):
            v = beta * v - eta * g
            h_new = _proj(h + v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-14:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            eta *= 0.5
            if eta < 1e-8:
                break
        # Decay momentum
        beta = max(0.5, beta * 0.999)
    return h, c5

def _pairwise(h, max_iter, scale, rng):
    """Random pairwise moves with sum preservation."""
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(max_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * (rng.rand() - 0.5) * 2
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i] = ni
        h[j] = nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
        else:
            h[i] = ni - d
            h[j] = nj + d
    return h, c5

def _structured_pair(h, max_rounds, rng):
    """Optimize 2-coordinate moves by grid search."""
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    c0 = _scorr(h, 1.0 - h, mode='full', method='fft')
    for _ in range(max_rounds):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Compute effect of changing h[i] by d, h[j] by -d
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            s = (n - 1) - k
            gi = 0.0
            gj = 0.0
            if s >= 0:
                if s <= i < n:
                    gi += (1.0 - h[i + s]) if i + s < n else 0.0
                if 0 <= i - s < n - s:
                    gi -= h[i - s]
                if s <= j < n:
                    gj += (1.0 - h[j + s]) if j + s < n else 0.0
                if 0 <= j - s < n - s:
                    gj -= h[j - s]
            else:
                sp = -s
                if sp <= i < n:
                    gi += (1.0 - h[i - sp])
                if 0 <= i + sp < n - sp:
                    gi -= h[i + sp]
                if sp <= j < n:
                    gj += (1.0 - h[j - sp])
                if 0 <= j + sp < n - sp:
                    gj -= h[j + sp]
            a[k] = gi - gj
        # Find best d
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi:
            continue
        dgrid = np.linspace(dlo, dhi, 15)
        best_d = 0.0
        best_v = np.max(c0)
        for d in dgrid:
            v = np.max(c0 + dx * d * a)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-10:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                c0 = _scorr(h, 1.0 - h, mode='full', method='fft')
            else:
                h[i] -= best_d
                h[j] += best_d
    return h, c5

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0. 0, 1.0)
        diff = target - h.sum()
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
                if abs(diff) < 1e-12:
                    break
