    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_single(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        if s > 0:
            i2 = np.arange(n - s)
            g[i2] -= h[i2 + s]
    else:
        sp = -s
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp])
        i2 = np.arange(n - sp, n)
        g[i2] -= h[i2 - sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5

        else:
            hi = mid
