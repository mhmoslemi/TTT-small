def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _subgrad(h, n_top=8, sharp=80.0):
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:n_top]
    w = np.exp(sharp * (c[top] - c[top][0]))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top):
        s = abs(int(ki - (n - 1)))
        if s == 0:
            continue
        m = n - s
        g = np.zeros(n)
        g[s:n] += 1.0 - h[:m]
        g[:m] -= h[s:n]
        grad += wi * g
    grad -= grad.mean()
    return grad
