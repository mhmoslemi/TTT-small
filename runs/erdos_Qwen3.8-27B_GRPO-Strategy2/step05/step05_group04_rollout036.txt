import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    if abs(h.sum() - t) < 1e-10:
        return h
    return _proj(h, t)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pmean_grad(h, p):
    n = len(h)
    c = _corr(h)
    cp = np.maximum(c, 0.0)
    cp1 = cp ** (p - 1.0)
    S = np.sum(cp ** p)
    if S < 1e-30:
        return np.zeros(n)
    pref = S ** (1.0 / p - 1.0)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = cp1[k] * pref * dx
        if w < 1e-16:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            g[s:s + m] += w * (1.0 - h[:m])
            g[:m] -= w * h[s:s + m]
        else:
            t = -s
            g[:n - t] += w * (1.0 - h[t:])
            g[t:] -= w * h[:n - t]
    return g

def _fista(h, p, niter, step0, t_end):
    n = len(h)
    target = n * 0.5
    c = h.copy()
    y = h.copy()
    tk = 1.0
    step = step0
    f_best = _c5(h)
    h_best = h.copy()
    for it in range(niter):
        if time.time() > t_end:
            break
        g = _pmean_grad(y, p)
        c = _proj(y - step * g, target)
        L = np.dot(c - h, g)
        if L > 1e-20:
            step = max(0.5 * step, 0.7 * L / (np.dot(g, g) + 1e-30))
        else:
            step = min(step * 1.05, step0 * 3)
        h_old = c.copy()
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = c + ((tk - 1.0) / tk_new) * (c - h)
        h = c
        tk = tk_new
        c5v = _c5(h)
        if c5v < f_best - 1e-15:
            f_best = c5v
            h_best = h.copy()
    return h_best, f_best

def _slsqp(h, alpha, maxiter):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n

    def obj(x):
        cc = _corr(x)
        mm = cc.max()
        ss = np.clip(1e4 * (cc - mm), -400, 400)
        e = np.exp(ss - ss.max())
        return mm + np.log(np.sum(e)) / 1e4 + 1e6 * (x.sum() - n * 0.5) ** 2

    def grad(x):
        cc = _corr(x)
        mm = cc.max()
        ss = np.clip(1e4 * (cc - mm), -400, 400)
        ee = np.exp(ss - ss.max())
        ww = ee / ee.sum()
        gp = np.zeros(n)
        for k in range(2 * n - 1):
            if ww[k] < 1e-14:
                continue
            wk = ww[k] * dx / 1e4
            sh = (n - 1) - k
            if sh >= 0:
                mi = n - sh
                gp[sh:sh + mi] += wk * (1.0 - x[:mi])
                gp[:mi] -= wk * x[sh:sh + mi]
            else:
                sp = -sh
                gp[:n - sp] += wk * (1.0 - x[sp:])
                gp[sp:] -= wk * x[:n - sp]
        gp += 2e6 * (x.sum() - n * 0.5)
        return gp

    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0, 1)] * n, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _active_descent(h, nrounds, t_end):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    c5c = _c5(h)
    for _ in range(nrounds):
        if time.time() > t_end:
            break
        c = _corr(h)
        k_w = c.argmax()
        s = k_w - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            mi = n - s
            g[s:s + mi] += (1.0 - h[:mi])
            g[:mi] -= h[s:s + mi]
        else:
            sp = -s
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
        g *= dx
        improved = False
        for st in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h - st * g, target)
            c5n = _c5(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _local_search(h, n_trials, rng, t_end):
    n = len(h)
    target = n * 0.5
    c5c = _c5(h)
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _c5(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _block_sa(h, n_iter, rng, t_end):
    n = len(h)
    target = n * 0.5
    c_cur = _c5(h)
    best_h = h.copy()
    best_c = c_cur
    t0, t1 = 0.005, 1e-5
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        blk = max(2, rng.randint(2, n // 10))
        st = rng.randint(0, n - blk)
        h_new = h.copy()
        h_new[st:st + blk] += rng.randn(blk) * (0.005 + 0.03 * t)
        h_new = _repair(h_new, n)
        c_new = _c5(h_new)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            h = h_new
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c
