                       args=(), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nelder_mead_free(h, n, threshold, max_iter):
    free_mask = (h > threshold) & (h < 1.0 - threshold)
    free_idx = np.where(free_mask)[0]
    n_free = len(free_idx)
    if n_free == 0:
        return h, _eval(h)
    h_base = h.copy()
    
    def _obj(x):
        ht = h_base.copy()
        ht[free_idx] = np.clip(x, 0, 1)
        ht = _repair(ht, n)
        return _eval(ht)
    
    x0 = h[free_idx].copy()
    res = minimize(_obj, x0, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-12, 'fatol': 1e-14, 'adapt': True})
    h_new = h_base.copy()
    h_new[free_idx] = np.clip(res.x, 0, 1)
    h_new = _repair(h_new, n)
    return h_new, _eval(h_new)

def _worst_lag_descent(h, n, max_rounds, step_sizes):
    c5_cur = _eval(h)
    dx = 2.0 / n
    for _ in range(max_rounds):
        c = _corr(h)
        k_w = int(c.argmax())
        sh = (n - 1) - k_w
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
        g *= dx
        improved = False
        for step in step_sizes:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _binary_snap_smooth(h, n, snap_alpha, smooth_sigma):
    h_snapped = np.where(h > 0.5, 1.0, 0.0).copy()
    h_interp = (1 - snap_alpha) * h + snap_alpha * h_snapped
    if smooth_sigma > 0:
        kernel_size = int(2 * smooth_sigma * n / 2) | 1
        if kernel_size >= 3:
            x = np.arange(kernel_size) - kernel_size // 2
            kernel = np.exp(-0.5 * (x / smooth_sigma) ** 2)
            kernel /= kernel.sum()
            h_interp = np.convolve(h_interp, kernel, mode='same')
    return _repair(h_interp, n)

def _coord_descent_free(h, n, max_iters, thresholds):
    c5_cur = _eval(h)
    for _it in range(max_iters):
        improved = False
        for thr in thresholds:
            free_idx = np.where((h > thr) & (h < 1.0 - thr))[0]
            for idx in free_idx:
                for delta in [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002]:
                    h_new = h.copy()
                    h_new[idx] = np.clip(h[idx] + delta, 0, 1)
                    h_new = _repair(h_new, n)
                    c5n = _eval(h_new)
                    if c5n < c5_cur - 1e-16:
                        h = h_new
                        c5_cur = c5n
                        improved = True
                        break
                    else:
                        h_new[idx] = h[idx]
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5_cur

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t_start

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: Progressive SLSQP with top-k smoothing
    for alpha, topk, mi in [(500, 20, 2000), (5000, 15, 3000), (50000, 10, 4000), (500000, 8, 5000)]:
        if _el() > 100:
            break
        h_c = _slsqp(best_h, alpha, topk, mi, 1e-18)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
            best_n = n0

    # Phase B: Nelder-Mead on free variables
    if _el() < budget_s - 500:
        for thr in [0.08, 0.05, 0.03]:
            if _el() > 180:
                break
            h_nm, c5_nm = _nelder_mead_free(best_h, best_n, thr, 8000)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm.copy()

    # Phase C: Worst-lag gradient descent + Nelder-Mead
    if _el() < budget_s - 400:
        h_cl = best_h.copy()
        h_cl, c5_cl = _worst_lag_descent(h_cl, best_n, 100, [0.05, 0.02, 0.01, 0.005, 0.002])
        if c5_cl < best_c5:
            best_c5 = c5_cl
            best_h = h_cl.copy()
        h_nm, c5_nm = _nelder_mead_free(best_h, best_n, 0.05, 10000)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()

    # Phase D: Binary snap + re-optimize (escape local optima)
    if _el() < budget_s - 300:
        for snap_a in [0.3, 0.5, 0.7]:
            if _el() > budget_s - 250:
                break
            h_snap = _binary_snap_smooth(best_h, best_n, snap_a, 0.5)
            h_snap = _slsqp(h_snap, 10000, 10, 3000, 1e-18)
            h_snap, c5_s = _nelder_mead_free(h_snap, best_n, 0.05, 5000)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_snap.copy()

    # Phase E: Multi-start with diverse perturbations
    if _el() < budget_s - 200:
        for _r in range(30):
            if _el() > budget_s - 180:
                break
            strat = rng.randint(6)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(best_n), best_n)
            elif strat == 1:
                hp = _repair(best_h + 0.05 * rng.randn(best_n), best_n)
            elif strat == 2:
                hp = best_h.copy()
                for _f in range(rng.randint(2, 6)):
                    f = rng.randint(1, best_n // 2 + 1)
                    amp = 0.015 * rng.randn()
                    t_arr = np.arange(best_n)
                    hp += amp * np.sin(2 * np.pi * f * t_arr / best_n + rng.rand() * 2 * np.pi)
                hp = _repair(hp, best_n)
            elif strat == 3:
                hp = best_h.copy()
                bs = rng.randint(best_n // 10, best_n // 4)
                st = rng.randint(0, best_n - bs)
                hp[st:st+bs] = 1.0 - hp[st:st+bs]
                hp = _repair(hp, best_n)
            elif strat == 4:
                hp = best_h.copy()
                n_swap = rng.randint(3, 10)
                for _b in range(n_swap):
                    i = rng.randint(best_n - 1)
                    d = 0.03 * rng.randn()
                    ni, nj = hp[i] + d, hp[i+1] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        hp[i], hp[i+1] = ni, nj
                hp = _repair(hp, best_n)
            else:
                hp = _binary_snap_smooth(best_h, best_n, 0.4 + 0.3 * rng.rand(), 1.0)
            h_new = _slsqp(hp, 20000, 10, 2500, 1e-18)
            h_new, c5_new = _nelder_mead_free(h_new, best_n, 0.05, 4000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase F: Upsample to 2x and refine
    if _el() < budget_s - 150:
        n1 = best_n * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha, topk in [(2000, 15), (20000, 10), (200000, 8)]:
            if _el() > budget_s - 80:
                break
            h_hi = _slsqp(h_hi, alpha, topk, 2500, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _el() < budget_s - 60:
            h_hi, c5_nm = _nelder_mead_free(best_h, best_n, 0.05, 6000)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm  # oops, fix below
                best_h = h_hi.copy()
                best_n = best_n

    # Phase G: Coordinate descent fine-tuning
    if _el() < budget_s - 30:
        h_cd, c5_cd = _coord_descent_free(best_h, best_n, 500, [0.01, 0.05])
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase H: Final SLSQP + Nelder-Mead polish
    if _el() < budget_s - 10:
        h_fin = _slsqp(best_h, 1000000, 6, 5000, 1e-20)
        h_fin, c5_fin = _nelder_mead_free(h_fin, len(h_fin), 0.03, 10000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = len(h_fin)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
