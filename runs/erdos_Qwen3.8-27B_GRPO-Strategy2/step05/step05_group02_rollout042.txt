    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _proj(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):

    if abs(h.sum() - n/2.0) < 1e-9:
        return h
    return _proj(h, n)

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha*(c-m), -500, 500)))) / alpha

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)

            g[:n-sp] -= wk*h[sp:]

    try:

                      method='SLSQP', bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0,
                                    'jac':lambda x: np.ones(n)}],
                      options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except:

The objective is f(h) = max_k c_k * dx where c_k = (correlate(h, 1-h, 'full'))_k.

The subgradient of f at h (when c_{k*} is the unique max) is:
∇f(h) = dx * ∇_{h} c_{k*}

For c_k with sh = (n-1) - k:
If sh >= 0: c_k = Σ_{i=0}^{n-1-sh} h_i(1-h_{i+sh})
∂c_k/∂h_j = (1-h_{j+sh}) * 1[j≤n-1-sh] - h_{j-sh} * 1[j≥sh]

If sh < 0, sp = -sh: c_k = Σ_{i=0}^{n-1-sp} h_{i+sp}(1-h_i)
∂c_k/∂h_j = (1-h_{j-sp}) * 1[j≥sp] - h_{j+sp} * 1[j≤n-1-sp]

For the top-k version, I use a softmax weighting:
∇f_approx = dx * Σ_{k in top-k} w_k * ∂c_k/∂h

This is a smoothed approximation to the subgradient that considers near-max lags too.

One more thought: for the "coordinated local search" (Phase 4), instead of full subgradient, I can do a smarter thing:
- Identify the single lag k* that achieves the max
- The gradient at k* tells us exactly which h[i] to increase/decrease
- Do a greedy coordinate descent: find the best single coordinate to move, move it a bit
- This is O(n) per step (after computing the gradient, which is O(n) for a single lag)

For a single lag k* with sh:
∂c_{k*}/∂h_j = (1-h_{j+sh})*1[j≤n-1-sh] - h_{j-sh}*1[j≥sh] (for sh≥0)

The "best" coordinate to increase is the one where this derivative is most negative (increasing h_j decreases c_{k*} the most).
The "best" coordinate to decrease is where the derivative is most positive.

But we need to maintain the sum constraint. So we increase one and decrease another.

Move: h[i] += d, h[j] -= d.
Change in c_{k*}: d * (∂c_{k*}/∂h_i - ∂c_{k*}/∂h_j)
We want this negative: ∂c_{k*}/∂h_i < ∂c_{k*}/∂h_j
So i should have the smallest gradient and j the largest.

This is like a "steepest descent" in the pairwise swap metric.

Let me implement this. For each iteration:
1. Compute c and find k* (or top-k)
2. Compute the gradient g (O(n) for single lag, O(n*k) for top-k)
3. Find i = argmin(g), j = argmax(g) (with boundary checks: h[i] < 1, h[j] > 0)
4. Determine max possible d: d_max = min(1-h[i], h[j])
5. Binary search for optimal d in [0, d_max]
6. Update h

The issue: after updating h, the gradient changes, and a different k* might be the max. So this is really just one step of subgradient descent with an exact line search.

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _proj(h, n):
    """Project to {0≤h≤1, sum(h)=n/2}"""
    tgt = n*0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -0.5, 1.5
    for _ in range(70):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0.0, 1.0)

def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha*(c-m),-500,500))))/alpha

def _lse_jac(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        sh = (n-1)-k
        wk = w[k]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-x[sh:])
            if sh: g[sh:] -= wk*x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-x[:n-sp])
            g[:n-sp] -= wk*x[sp:]
    return g

def _sum_jac(x): return np.ones(len(x))
