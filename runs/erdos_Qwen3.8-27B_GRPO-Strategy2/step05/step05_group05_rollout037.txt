def _pair_delta(h, i, j, n):
    """Compute b[k] = ∂c[k]/∂h[i] - ∂c[k]/∂h[j] for all k"""
    b = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = (n-1) - k
        # ∂c[k]/∂h[p]
        def dg(p):
            val = 0.0
            if 0 <= p + sh < n:
                val += (1.0 - h[p+sh])
            if 0 <= p - sh < n:
                val -= h[p-sh]
            return val
        b[k] = dg(i) - dg(j)
    return b

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _sobj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] >= 1e-14:
            g += w[k] * _grad_k(h, k, n)
    return g * (2.0 / n)

def _cfun(x):
    return x.sum() - len(x) * 0.5

def _cjac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, t_limit):
    try:
        res = minimize(_sobj, h, _sgrad, args=(alpha,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * len(h),
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        if time.time() > t_limit:
            return _repair(h, len(h))
        return _repair(res.x, len(res.x))
    except Exception:
        return _repair(h, len(h))

def _pair_delta(h, i, j, n):
    b = np.zeros(2 * n - 1)
    sh = np.arange(n - 1, -1, -1)
    v = sh <= n - 1 - i
    if v.any():
        b[v] += 1.0 - h[i + sh[v]]
    v2 = sh <= i
    if v2.any():
        b[v2] -= h[i - sh[v2]]
    v = sh <= n - 1 - j
    if v.any():
        b[v] -= 1.0 - h[j + sh[v]]
    v2 = sh <= j
    if v2.any():
        b[v2] += h[j - sh[v2]]
    sp = np.arange(1, n)
    v3 = sp <= i
    if v3.any():
        nv = np.where(v3)[0] + n
        b[nv] += 1.0 - h[i - sp[v3]]
    v4 = sp < n - i
    if v4.any():
        nv = np.where(v4)[0] + n
        b[nv] -= h[i + sp[v4]]
    v3 = sp <= j
    if v3.any():
        nv = np.where(v3)[0] + n
        b[nv] -= 1.0 - h[j - sp[v3]]
    v4 = sp < n - j
    if v4.any():
        nv = np.where(v4)[0] + n
        b[nv] += h[j + sp[v4]]
    return b

def _golden(c, b, lo, hi, iters=60):
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    a, d = lo, hi
    x1 = a + (1.0 - gr) * (d - a)
    x2 = a + gr * (d - a)
    f1 = np.max(c + x1 * b)
    f2 = np.max(c + x2 * b)
    for _ in range(iters):
        if f1 < f2:
            d, x2, f2 = x2, x1, f1
            x1 = a + (1.0 - gr) * (d - a)
            f1 = np.max(c + x1 * b)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = a + gr * (d - a)
            f2 = np.max(c + x2 * b)
    return (a + d) * 0.5

def _exact_pairs(h, n, rng, n_pairs, t_end, c5c):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    acc = 0
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0) + 1e-12
        dhi = min(1.0 - h[i], h[j]) - 1e-12
        if dlo >= dhi:
            continue
        b = _pair_delta(h, i, j, n)
        d = _golden(c, b, dlo, dhi)
        if abs(d) < 1e-13:
            continue
        c5n = float(np.max(c + d * b) * dx)
        if c5n < c5c - 1e-15:
            h[i] += d
            h[j] -= d
            c += d * b
            c5c = c5n
            acc += 1
            if acc % 500 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
    return h, c5c

def _topk_descent(h, n, max_iters, n_top, t_end):
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5c = c.max()
        idx = np.argsort(c)[::-1][:n_top]

        w = np.maximum(cvals - cvals.mean(), 0.0)
        if w.sum() < 1e-15:
            w = np.ones(n_top) / n_top
        else:
            w /= w.sum()
        g = np.zeros(n)
        for wt, k in zip(w, idx):

        g *= dx
        improved = False

                c5c = c5n
