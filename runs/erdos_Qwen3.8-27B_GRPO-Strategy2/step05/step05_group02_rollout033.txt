            j = rng.randint(n)
            if i == j:
                continue
            delta = _golden_pair(h, c0, G, i, j, n)
            if abs(delta) > 1e-13:
                h[i] += delta
                h[j] -= delta
                c0 += delta * (G[i] - G[j])
                G = _grad_matrix(h, n)
                c5n = float(np.max(c0) * 2.0 / n)
                if c5n < c5c - 1e-14:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5c, h.copy()
                if it % 300 == 0:
                    c0 = np.correlate(h, 1.0 - h, mode='full')
                    G = _grad_matrix(h, n)

    # Phase 5: Upsample to 2x and optimize if time permits
    if _tl(200):
        n2 = n * 2
        h2 = np.zeros(n2)
        for i in range(n):
            h2[i:i+2] = best_h[i]
        h2 = _repair(h2, n2)
        c5_2 = _eval(h2)
        if c5_2 <= best_c5 + 1e-6:
            h2 = _slsqp(h2, 50000, 1500, 1e-19)
            c5_2 = _eval(h2)
            if c5_2 < best_c5:
                best_c5, best_h, n = c5_2, h2.copy(), n2
                c0 = np.correlate(best_h, 1.0 - best_h, mode='full')
                G = _grad_matrix(best_h, n)
                for it in range(200000):
                    if not _tl(40):
                        break
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    delta = _golden_pair(best_h, c0, G, i, j, n)
                    if abs(delta) > 1e-13:
                        best_h[i] += delta
                        best_h[j] -= delta
                        c0 += delta * (G[i] - G[j])
                        G = _grad_matrix(best_h, n)
                        c5n = float(np.max(c0) * 2.0 / n)
                        if c5n < best_c5:
                            best_c5 = c5n
                            if it % 400 == 0:
                                c0 = np.correlate(best_h, 1.0 - best_h, mode='full')
                                G = _grad_matrix(best_h, n)

    # Phase 6: Final SLSQP polish
    if _tl(10):
        h_new = _slsqp(best_h, 500000, 2500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
