Since f(d) is the max of linear functions of d, it's convex and piecewise linear. The minimum is at a point where the derivative changes sign, i.e., at an intersection of two of the lines c_k(d).

To find the minimum:
1. Compute all slopes s_k = a_k(i) - a_k(j)
2. The active constraint at d=0 is the set of k achieving max c_k(0)
3. The optimal d is where the "upper envelope" of the lines is minimized
4. This is a 1D convex minimization - binary search works!

Actually, for a max of linear functions, the minimum can be found by:
- Computing the subgradient at d=0: it's the set of slopes of the lines achieving the max at d=0
- If 0 is in the subgradient, d=0 is optimal
- Otherwise, the optimal d is at a kink point

1. Evaluating f at the endpoints and at all kink points (intersections of pairs of lines)
2. The kink points are O(n_lags²) in worst case, but in practice we can binary search

    # For index i:
    p1 = i + sh  # shape (2n-1,)

    p2 = i - sh
    v2 = (p2 >= 0) & (p2 < n)

    up_i[v1] = 1.0 - h[p1[v1]]
    dn_i = np.zeros(2*n-1)

    p1j = j + sh
    v1j = (p1j >= 0) & (p1j < n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slope_vec(h, i, j, n):
    sh = np.arange(n - 1, -(n - 1), -1)
    p1 = i + sh
    v1 = (p1 >= 0) & (p1 < n)
    p2 = i - sh
    v2 = (p2 >= 0) & (p2 < n)
    up_i = np.where(v1, 1.0 - h[np.clip(p1, 0, n - 1)], 0.0)
    dn_i = np.where(v2, h[np.clip(p2, 0, n - 1)], 0.0)
    p1j = j + sh
    v1j = (p1j >= 0) & (p1j < n)
    p2j = j - sh
    v2j = (p2j >= 0) & (p2j < n)
    up_j = np.where(v1j, 1.0 - h[np.clip(p1j, 0, n - 1)], 0.0)
    dn_j = np.where(v2j, h[np.clip(p2j, 0, n - 1)], 0.0)
    return (up_i - dn_i) - (up_j - dn_j)

def _grid_2coord(h, i, j, c0, n, n_grid=30):
    dx = 2.0 / n
    slopes = _slope_vec(h, i, j, n)
    dlo = max(-h[i], h[j] - 1.0)
    dhi = min(1.0 - h[i], h[j])
    if dlo >= dhi:
        return 0.0, np.inf
    dgrid = np.linspace(dlo, dhi, n_grid)
    best_d = 0.0
    best_v = np.inf
    for d in dgrid:
        v = (c0 + d * slopes).max()
        if v < best_v:
            best_v = v
            best_d = d
    return best_d, best_v * dx

def _smooth_obj(x, alpha, penalty):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n) + penalty * (x.sum() - n * 0.5) ** 2

def _smooth_grad(x, alpha, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += w[k] * (1.0 - x[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= w[k] * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - x[:n - sp])
            g[:n - sp] -= w[k] * x[sp:]
    g *= dx
    g += 2.0 * penalty * (x.sum() - n * 0.5)
    return g

def _lbfgs(h, alpha, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, penalty),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_desc(h, max_iters, n_lags):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:n_lags]
        cv = c[idx]
        alpha = 200.0
        s = np.clip(alpha * (cv - cv.max()), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, idx):

            if sh >= 0:
                cnt = n - sh

            else:
                sp = -sh
                g[sp:] += wi * (1.0 - h[:n - sp])
                g[:n - sp] -= wi * h[sp:]
        g *= dx

                c5c = c5n
                improved = True

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slope_vec(h, i, j, n):
    sh = np.arange(n - 1, -(n - 1), -1)
    p1 = i + sh
    v1 = (p1 >= 0) & (p1 < n)
    p2 = i - sh
    v2 = (p2 >= 0) & (p2 < n)
    up_i = np.where(v1, 1.0 - h[np.clip(p1, 0, n - 1)], 0.0)
    dn_i = np.where(v2, h[np.clip(p2, 0, n - 1)], 0.0)
    p1j = j + sh
    v1j = (p1j >= 0) & (p1j < n)
    p2j = j - sh
    v2j = (p2j >= 0) & (p2j < n)
    up_j = np.where(v1j, 1.0 - h[np.clip(p1j, 0, n - 1)], 0.0)
    dn_j = np.where(v2j, h[np.clip(p2j, 0, n - 1)], 0.0)
    return (up_i - dn_i) - (up_j - dn_j)

def _grid_2coord(h, i, j, c0, n, n_grid=30):
    slopes = _slope_vec(h, i, j, n)
    dlo = max(-h[i], h[j] - 1.0)
    dhi = min(1.0 - h[i], h[j])
    if dlo >= dhi:
        return 0.0, np.inf
    dgrid = np.linspace(dlo, dhi, n_grid)
    best_d = 0.0
    best_v = np.inf
    for d in dgrid:
        v = (c0 + d * slopes).max()
        if v < best_v:
            best_v = v
            best_d = d
    return best_d, best_v

def _smooth_obj(x, alpha, penalty):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n) + penalty * (x.sum() - n * 0.5) ** 2

def _smooth_grad(x, alpha, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += w[k] * (1.0 - x[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= w[k] * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - x[:n - sp])
            g[:n - sp] -= w[k] * x[sp:]
    g *= dx
    g += 2.0 * penalty * (x.sum() - n * 0.5)
    return g

def _lbfgs(h, alpha, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, penalty),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_desc(h, max_iters, n_lags):

    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):

        idx = np.argsort(c)[::-1][:n_lags]
        cv = c[idx]
        alpha = 200.0
        s = np.clip(alpha * (cv - cv.max()), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, idx):
            sh = (n - 1) - k
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wi * (1.0 - h[sh:sh + cnt])
                if sh > 0:
                    g[sh:] -= wi * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wi * (1.0 - h[:n - sp])
                g[:n - sp] -= wi * h[sp:]
        g *= dx
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
            h_new = _proj_sum(h - step * g, n * 0.5)

            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _two_coord_phase(h, budget_till, n_pairs, n_grid, seed_rng):

    dx = 2.0 / n
    c5c = _eval(h)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(n_pairs):
        if time.time() > budget_till:
            break
        i = seed_rng.randint(n)
        j = seed_rng.randint(n)
        if i == j:
            continue
        d, raw_v = _grid_2coord(h, i, j, c0, n, n_grid)
        if abs(d) > 1e-10 and raw_v * dx < c5c - 1e-12:
            h[i] += d
            h[j] -= d
            h = np.clip(h, 0, 1)
            h = _repair(h, n)
            c5c = raw_v * dx
            c0 = np.correlate(h, 1.0 - h, mode='full')
    return h, c5c
