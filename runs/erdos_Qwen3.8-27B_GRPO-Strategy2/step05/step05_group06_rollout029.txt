    if sh >= 0:
        idx = np.arange(n-sh)
        g[idx] += (1.0 - h[idx+sh])
        if sh > 0:
            idx2 = np.arange(sh, n)
            g[idx2] -= h[idx2-sh]
    else:
        sp = -sh
        idx = np.arange(sp, n)
        g[idx] += (1.0 - h[idx-sp])
        idx2 = np.arange(n-sp)
        g[idx2] -= h[idx2+sp]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        s = np.clip(h-mid, 0, 1).sum()
        lo, hi = (mid, hi) if s > target else (lo, mid)
    return np.clip(h-(lo+hi)*0.5, 0, 1)

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s-s.max()))) / alpha + s.max()/alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    sm = s.max()
    w = np.exp(s-sm)
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        sh = (n-1)-k
        wk = w[k]*dx
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += wk*(1.0-h[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk*h[i2-sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk*(1.0-h[i1-sp])
            i2 = np.arange(n-sp)
            g[i2] -= wk*h[i2+sp]
    return g

def _slsqp(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun': lambda x: x.sum()-n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h0, jac=_smooth_grad, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return _repair(h0, n)

def _topgrad_descent(h, nsteps, step0, topk=1):
    """Exact subgradient descent on the top-k lags."""
    n = len(h)
    c5 = _eval(h)
    step = step0
    stalled = 0
    for i in range(nsteps):
        c = _corr(h)
        top = np.argsort(c)[::-1][:topk]
        g = np.zeros(n)
        for k in top:
            g += _lag_grad(h, k)
        g /= topk
        g *= 2.0/n  # dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        # Backtracking line search
        for t in [step, step*0.5, step*0.25, step*0.1, step*0.01]:
            h_new = _proj(h - t*g, n*0.5)
            c5n = _eval(h_new)
            if c5n < c5:
                h, c5 = h_new, c5n
                stalled = 0
                break
        else:
            stalled += 1
            step *= 0.5
            if step < 1e-10:
                break
        if stalled > 100:
            break
    return h, c5

def _local_search(h, n_iters, scale, rng):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iters):

            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = ni-d, nj+d
    return h, c5

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])

                    diff += a
