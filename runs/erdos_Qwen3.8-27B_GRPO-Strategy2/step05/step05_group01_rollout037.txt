```python
def _sa_search(h, c, c5, rng, budget_fn, n_steps_per_T, T0, cooling):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c5
    T = T0
    step = 0.01  # base step size for perturbations
    
    total_steps = 0
    while budget_fn() and T > 1e-8:
        for _ in range(n_steps_per_T):
            move_type = rng.rand()
            if move_type < 0.5:
                # Pair move
                i, j = rng.randint(n), rng.randint(n)
                if i == j: continue
                d = step * rng.randn()

                if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    continue

                delta = new_c5 - c5
                if delta < 0 or rng.rand() < np.exp(-delta / T):
                    h[i], h[j] = ni, nj

                    if c5 < best_c5:
```

```python
                dc = _pos_change(h2, i, di, n) + _pos_change(h2, j, dj, n)
                nc = c2 + dc
                nc5 = nc.max() * dx
                dl = nc5 - c5c

                    if c5c < bc5: bc5, bh = c5c, h2.copy()
        T *= cool
        sc = max(sc * 0.996, 1e-6)
    return bh, _corr(bh), bc5

def _cdesc(h, c, c5, rng, t_end, nvals=16):
    """Coordinate descent: for each position, find best value by grid search."""
    n = len(h)
    dx = 2.0 / n
    bh, bc5 = h.copy(), c5
    idx = np.arange(n)
    grid = np.linspace(0, 1, nvals)
    while time.time() < t_end:
        rng.shuffle(idx)
        for i in idx:
            old = h[i]
            bv, bn5 = old, c5
            for v in grid:
                d = v - old
                if abs(d) < 1e-12: continue
                dc = _pos_change(h, i, d, n)
                nc5 = (c + dc).max() * dx
                if nc5 < bn5: bn5, bv = nc5, v
            if bv != old:
                d = bv - old
                dc = _pos_change(h, i, d, n)
                h[i] = bv
                c += dc
                c5 = bn5
                if c5 < bc5: bc5, bh = c5, h.copy()
    return bh, _corr(bh), bc5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _sobj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:n - sh]
        else:
            p = -sh
            g[p:] += w[k] * (1.0 - h[:n - p])
            g[:n - p] -= w[k] * h[p:]
    return g

def _sfun(x):
    return x.sum() - len(x) * 0.5

def _sjac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_sobj, h.copy(), jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sfun, 'jac': _sjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

                if nc5 < c5c or rng.random() < np.exp(-(nc5 - c5c) / max(T, 1e-15)):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _sobj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:n - sh]
        else:
            p = -sh
            g[p:] += w[k] * (1.0 - h[:n - p])
            g[:n - p] -= w[k] * h[p:]
    return g

def _sfun(x):
    return x.sum() - len(x) * 0.5

def _sjac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(_sobj, h.copy(), jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sfun, 'jac': _sjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
```

```python
def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[k] * (1.0 - h[sh:])
            g[sh:] -= w[k] * h[:n - sh]
        else:
            p = -sh
            g[p:] += w[k] * (1.0 - h[:n - p])
            g[:n - p] -= w[k] * h[p:]
    return g
```
