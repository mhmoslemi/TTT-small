        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase C: More max-gradient refinement
    if _el() < budget_s - 200:
        dx_c = 2.0 / best_n
        h_c, c_c = _max_grad_loop(best_h, best_n, dx_c, 5000, budget_s - 200 - _el())
        if c_c < best_c5:
            best_c5, best_h, best_n = c_c, h_c.copy(), best_n

    # Phase D: Multi-start local search with diverse perturbations
    nc = best_n
    if _el() < budget_s - 180:
        for _r in range(30):
            if _el() > budget_s - 150:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = rng.randint(nc // 10, nc // 4)
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] = 1.0 - hp[st:st + bs]
                hp = _repair(hp, nc)
            elif strat == 2:
                hp = best_h.copy()
                for _f in range(rng.randint(1, 4)):
                    f = rng.randint(1, nc // 2)
                    amp = 0.02 * rng.randn()
                    ta = np.arange(nc)
                    hp += amp * np.sin(2 * np.pi * f * ta / nc + rng.rand() * 2 * np.pi)
                hp = _repair(hp, nc)
            else:
                hp = _repair(best_h + 0.005 * rng.randn(nc), nc)
                for _b in range(rng.randint(2, 6)):
                    i = rng.randint(nc - 1)
                    d = 0.01 * rng.randn()
                    if 0 <= hp[i] + d <= 1 and 0 <= hp[i+1] - d <= 1:
                        hp[i] += d
                        hp[i+1] -= d
                hp = _repair(hp, nc)
            hx = _slsqp(hp, 100000, 1000, 1e-18, nc)
            dx_t = 2.0 / nc
            hx, cx = _max_grad_loop(hx, nc, dx_t, 300, 3.0)
            c5 = _eval(hx)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hx.copy(), nc

    # Phase E: Try at double resolution
    if _el() < budget_s - 120:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        dx1 = 2.0 / n1
        h_hi, c_h = _max_grad_loop(h_hi, n1, dx1, 2000, budget_s - 100 - _el())
        if _el() < budget_s - 60:
            h_hi = _slsqp(h_hi, 200000, 1500, 1e-18, n1)
            c_h = _eval(h_hi)
        h_hi, c_h = _max_grad_loop(h_hi, n1, dx1, 2000, max(0, budget_s - 50 - _el()))
        if c_h < best_c5:
            best_c5, best_h, best_n = c_h, h_hi.copy(), n1

    # Phase F: Final SLSQP with very high alpha
    if _el() < budget_s - 30:
        hf = _slsqp(best_h, 2000000, 5000, 1e-20, best_n)
        c5f = _eval(hf)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    # Phase G: Ultra-fine pairwise local search
    if _el() < budget_s - 3:
        h_g = best_h.copy()
        ng = len(h_g)
        c5g = _eval(h_g)
        n_tries = min(int((budget_s - 3 - _el()) / 0.002), 100000)
        for _it in range(n_tries):
            if _el() > budget_s - 3:
                break
            i = rng.randint(ng)
            j = rng.randint(ng)
            if i == j:
                continue
            d = 2e-5 * rng.randn()
            ni, nj = h_g[i] + d, h_g[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_g[i], h_g[j] = ni, nj
            c5n = _eval(h_g)
            if c5n < c5g - 1e-17:
                c5g = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h_g.copy()
            else:
                h_g[i], h_g[j] = ni, nj
                h_g[i] -= 2 * d
                h_g[j] += 2 * d

    # Final max-gradient polish
    if _el() < budget_s - 1:
        dx_f = 2.0 / best_n
        h_f, c_f = _max_grad_loop(best_h, best_n, dx_f, 500, 0.5)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
