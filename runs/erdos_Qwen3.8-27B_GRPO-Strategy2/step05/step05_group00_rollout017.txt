import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_grad(x, alpha, dt):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dt
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    obj = m + (s.max() + np.log(e.sum())) / alpha
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return obj, g * dt

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    dt = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        return _obj_grad(x, alpha, dt)[0]
    def _g(x):
        return _obj_grad(x, alpha, dt)[1]
    def _c(x):
        return x.sum() - n / 2.0
    def _cj(x):
        return np.ones(n)
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _c, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs(h, alpha, maxiter, gtol, n_fresh):
    n = len(h)
    dt = 2.0 / n
    pen = 1e5
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        o, g = _obj_grad(x, alpha, dt)
        s = x.sum() - n / 2.0
        return o + pen * s * s
    def _g(x):
        o, g = _obj_grad(x, alpha, dt)
        s = x.sum() - n / 2.0
        g = g + 2.0 * pen * s
        return g
    try:
        res = minimize(_f, h, jac=_g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': gtol,
                                'maxls': 30, 'maxfun': maxiter * 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

                c5 = c2
                improved = True

    return h, c5

def _topk_descent(h, topk, n_steps, eta0):

    c5 = _eval(h)
    h_prev = h.copy()
