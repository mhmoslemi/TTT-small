import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _power_obj(h, p):
    c = _corr(h)
    cpos = np.maximum(c, 0)
    return np.sum(cpos ** p) ** (1.0 / p)

def _power_grad(h, p):
    n = len(h)
    c = _corr(h)
    cpos = np.maximum(c, 0)
    S = np.sum(cpos ** p)
    if S < 1e-30:
        return np.zeros(n)
    w = p * cpos ** (p - 1) / S ** (1.0 - 1.0/p)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

                if d > 0:
                    a = min(d, 1.0 - h[j]); h[j] += a; d -= a
                else:
                    a = min(-d, h[j]); h[j] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _wgrad(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            s2 = -sh
            g[s2:] += wk * (1.0 - h[:n-s2])
            g[:n-s2] -= wk * h[s2:]
    return g

def _pgrad(h, p):
    n = len(h)
    c = _corr(h)
    cx = np.maximum(c, 0.0)
    S = np.sum(cx**p)
    if S < 1e-30: return np.zeros(n)
    w = p * cx**(p-1) / (S**(1.0 - 1.0/p))
    return _wgrad(h, w, n)

def _pwval(h, p):
    c = np.maximum(_corr(h), 0.0)
    return float(np.sum(c**p)**(1.0/p))

def _fw(h, p, tl):
    n = len(h)
    h = h.astype(float).copy()
    for _ in range(500000):
        if time.time() > tl: break
        g = _pgrad(h, p)
        v = np.zeros(n)
        v[np.argsort(g)[:n//2]] = 1.0
        if np.max(np.abs(v - h)) < 1e-13: break
        fk = _pwval(h, p)
        a = 1.0
        for _ in range(12):
            h2 = (1.0 - a) * h + a * v
            if _pwval(h2, p) <= fk + 1e-14: break
            a *= 0.5
        h = (1.0 - a) * h + a * v
    return _repair(h, n)

def _obj_sm(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _jac_sm(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    return _wgrad(x, w, n)

def _csum(x):
    return x.sum() - len(x) * 0.5

def _csumj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, tl):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _csum, 'jac': _csumj}

                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': 3000, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _proj(h, tg):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tg: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    if abs(s - tg) > 1e-9:
        h *= tg / s
        h = np.clip(h, 0.0, 1.0)

        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)

                else:
                    a = min(-d, h[j]); h[j] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))

def _wgrad(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n - sh]
        else:
            s2 = -sh
            g[s2:] += wk * (1.0 - h[:n - s2])
            g[:n - s2] -= wk * h[s2:]
    return g

def _pgrad(h, p):
    n = len(h)
    c = _corr(h)
    cx = np.maximum(c, 0.0)
    S = np.sum(cx ** p)
    if S < 1e-30: return np.zeros(n)
    w = p * cx ** (p - 1) / (S ** (1.0 - 1.0 / p))
    return _wgrad(h, w, n)

def _pwval(h, p):
    c = np.maximum(_corr(h), 0.0)
    return float(np.sum(c ** p) ** (1.0 / p))

def _fw(h, p, tl):
    n = len(h)
    h = h.astype(float).copy()
    for _ in range(500000):
        if time.time() > tl: break
        g = _pgrad(h, p)
        v = np.zeros(n)
        v[np.argsort(g)[:n // 2]] = 1.0
        if np.max(np.abs(v - h)) < 1e-13: break
        fk = _pwval(h, p)
        a = 1.0
        for _ in range(12):
            h2 = (1.0 - a) * h + a * v
            if _pwval(h2, p) <= fk + 1e-14: break
            a *= 0.5
        h = (1.0 - a) * h + a * v
    return _repair(h, n)

def _obj_sm(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _jac_sm(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    return _wgrad(x, w, n)

def _csum(x):
    return x.sum() - len(x) * 0.5

def _csumj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, tl):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _csum, 'jac': _csumj}
    try:
        r = minimize(_obj_sm, h, jac=_jac_sm, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': 2500, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _proj(h, tg):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tg: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_fix(h, tl, nsteps, K, step0):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    step = step0
    for _ in range(nsteps):
        if time.time() > tl: break
        c = _corr(h)
        idx = np.argsort(c)[::-1][:K]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        g = _wgrad(h, w, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        g /= gn
        improved = False
        for _ in range(25):
            h2 = _proj(h - step * g, n * 0.5)
            c5n = _eval(h2)
            if c5n < c5 - 1e-15:
                h, c5 = h2, c5n
                step = min(step * 1.04, 0.2)
                improved = True
                break
            step *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-14: break
    return h, c5

def _mom_grad(h, p, tl, nsteps, lr0, mu):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = lr0
    for _ in range(nsteps):
        if time.time() > tl: break
        g = _pgrad(h, p)
        v = mu * v - lr * g
        h = _proj(h - v, n * 0.5)
        lr = min(lr * 1.02, lr0 * 8.0)
    return _repair(h, n)

def _grad_polish(h, alpha, tl, nsteps, lr0):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    lr = lr0
    for _ in range(nsteps):
        if time.time() > tl: break
        g = _jac_sm(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        g /= gn
        h2 = _proj(h - lr * g, n * 0.5)
        c5n = _eval(h2)
        if c5n < c5:
            h, c5 = h2, c5n
            lr = min(lr * 1.02, 1e-3)
        else:
            lr *= 0.5
            if lr < 1e-13: break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh = h0.copy()
    bc5 = _eval(h0)

    # Phase 1: Frank-Wolfe with power-sum
    tl1 = t0 + min(180.0, budget_s * 0.3)
    for p in [8, 24, 64]:
        h_new = _fw(bh, p, tl1)
        c5 = _eval(h_new)
        if c5 < bc5: bh, bc5 = h_new, c5

    # Phase 2: SLSQP polish with increasing smoothness
    tl2 = min(t0 + 400, t0 + budget_s * 0.55)
    for al in [2000, 20000, 150000]:
        if time.time() > tl2: break
        h_new = _slsqp(bh, al, tl2)
        c5 = _eval(h_new)
        if c5 < bc5: bh, bc5 = h_new, c5

    # Phase 3: Top-K peak gradient descent
    tl3 = min(t0 + 550, t0 + budget_s * 0.72)
    if time.time() < tl3:
        h_new, c5n = _topk_fix(bh, tl3, 80000, 4, 0.02)
        if c5n < bc5: bh, bc5 = h_new, c5n

    # Phase 4: Momentum with high power
    tl4 = min(t0 + 660, t0 + budget_s * 0.8)
    if time.time() < tl4:
        h_new = _mom_grad(bh, 48, tl4, 200000, 0.003, 0.9)
        c5 = _eval(h_new)
        if c5 < bc5: bh, bc5 = h_new, c5

    # Phase 5: Binary structure search
    tl5 = min(t0 + 760, t0 + budget_s * 0.9)
    if time.time() < tl5:
        hb = np.where(bh > 0.5, 1.0, 0.0)
        hb = _repair(hb, n)
        c5b = _eval(hb)
        tb = np.random.RandomState(42)
        for _ in range(300000):
            if time.time() > tl5: break
            i, j = tb.randint(n), tb.randint(n)
            if i == j or hb[i] == hb[j]: continue
            hb[i], hb[j] = hb[j], hb[i]
            c5n = _eval(hb)
            if c5n < c5b: c5b = c5n
            else: hb[i], hb[j] = hb[j], hb[i]
        if c5b < bc5: bh, bc5 = hb.copy(), c5b
        if time.time() < tl5:
            hr, cr = _grad_polish(bh, 50000, tl5, 30000, 1e-4)
            if cr < bc5: bh, bc5 = hr.copy(), cr

    # Phase 6: Fine gradient polish
    tl6 = t0 + budget_s - 25
    if time.time() < tl6:
        h_new = _slsqp(bh, 400000, tl6)
        c5 = _eval(h_new)
        if c5 < bc5: bh, bc5 = h_new, c5

    # Phase 7: Ultra-fine pair moves
    tl7 = t0 + budget_s - 2
    if time.time() < tl7:
        h = bh.copy()
        c5 = _eval(h)
        for _ in range(400000):
            if time.time() > tl7: break
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = 3e-5 * rng.randn()
            if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                h[i] += d; h[j] -= d
                c5n = _eval(h)
                if c5n < c5: c5 = c5n
                else: h[i] -= d; h[j] += d
        if c5 < bc5: bh, bc5 = h.copy(), c5

    # Phase 8: Final gradient step
    if time.time() < t0 + budget_s - 1:
        h, c5 = _grad_polish(bh, 500000, t0 + budget_s - 1, 20000, 5e-6)
        if c5 < bc5: bh, bc5 = h.copy(), c5

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
