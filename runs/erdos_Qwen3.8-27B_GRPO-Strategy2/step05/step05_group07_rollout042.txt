        m = max(m_init, int(m_init + (m_max - m_init) * prog))
        g = _grad_topm(h, m)
        v = mu * v - lr * g
        h_new = _proj(h + v, target)
        f_new = _eval(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
            stalling = 0
        else:
            stalling += 1
        h = h_new
        if stalling < 10:
            lr = min(lr * 1.01, 0.35)
        elif stalling > 50:
            lr = max(lr * 0.92, 0.005)
            stalling = 25
            mu = max(mu * 0.99, 0.5)
    return best_h, best_f

def _greedy_sweep(h, n, target, t0, t_end, rounds=3):
    global _BUDGET
    h = h.copy().astype(float)
    best_f = _eval(h)
    for rd in range(rounds * 50):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_m = 15
        top = np.argpartition(c, -top_m)[-top_m:]
        g = np.zeros(n)
        for k in top:
            g += _grad_lag(h, k, n)
        g /= top_m
        order = np.argsort(np.abs(g))[::-1]
        improved = False
        for i in order[:40]:
            if time.time() > t_end:
                break
            for delta in [0.05, 0.02, 0.01, 0.005, -0.05, -0.02, -0.01, -0.005]:
                hn = h.copy()
                hn[i] = np.clip(hn[i] + delta, 0, 1)
                hn = _proj(hn, target)
                fn = _eval(hn)
                if fn < best_f - 1e-10:
                    h = hn
                    best_f = fn
                    improved = True
                    break
        if not improved and rd > 5:
            break
    return h, best_f

def _binary_push(h, n):
    tgt = n / 2.0
    b = (h >= 0.5).astype(float)
    ones = b.sum()
    if ones > tgt:
        excess = int(ones - tgt)
        zeros_below = np.where(b == 1)[0]
        fracs = h[zeros_below]
        order = np.argsort(fracs)[:excess]
        b[zeros_below[order]] = 0
    elif ones < tgt:
        deficit = int(tgt - ones)
        ones_above = np.where(b == 0)[0]
        fracs = 1.0 - h[ones_above]
        order = np.argsort(fracs)[:deficit]
        b[ones_above[order]] = 1
    b = _repair(b, n)
    return b

def _upsample(h, factor, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    idx = np.searchsorted(x_old, x_new)
    idx = np.clip(idx, 1, n_old - 1)
    t = (x_new - x_old[idx - 1]) / (x_old[idx] - x_old[idx - 1] + 1e-30)
    t = np.clip(t, 0, 1)
    h_new = (1 - t) * h[idx - 1] + t * h[idx]
    return _repair(h_new, n_new)

def _lsq_smooth(h, n, target, alpha, t0, t_end):
    global _BUDGET
    h = h.copy().astype(float)
    best_h = h.copy()
    best_f = _eval(h)
    lr = 0.05
    for it in range(200000):
        if time.time() > t_end:
            break
        c = _corr(h)
        m_val = c.max()
        s = np.clip(alpha * (c - m_val), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            g += w[k] * _grad_lag(h, k, n)
        h = _proj(h - lr * g, target)
        f = _eval(h)
        if f < best_f:
            best_f = f
            best_h = h.copy()
        lr = min(lr * 1.005, 0.15)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Quick PG polish at parent resolution
    t1 = _T0 + 40
    hn, fn = _pg_momentum(best_h, best_n, best_n // 2, _T0, t1, m_init=3, m_max=25, lr0=0.12, mu=0.9)
    if fn < best_c5:
        best_c5, best_h = fn, hn
    bn = _binary_push(best_h, best_n)
    bf = _eval(bn)
    if bf < best_c5:
        best_c5, best_h = bf, bn

    # Phase 2: Upsample to 2x and optimize
    n2 = min(best_n * 2, 600)
    h2 = _upsample(best_h, 2, n2)
    t2 = _T0 + 200
    hn2, fn2 = _pg_momentum(h2, n2, n2 // 2, _T0, t2, m_init=3, m_max=30, lr0=0.10, mu=0.88)
    if fn2 < best_c5:
        best_c5, best_h, best_n = fn2, hn2, n2

    # Phase 3: Long PG with increasing m at best resolution
    tn = len(best_h)
    t3 = _T0 + 450
    hn3, fn3 = _pg_momentum(best_h, tn, tn // 2, _T0, t3, m_init=5, m_max=50, lr0=0.08, mu=0.82)
    if fn3 < best_c5:
        best_c5, best_h = fn3, hn3
    bn3 = _binary_push(best_h, tn)
    bf3 = _eval(bn3)
    if bf3 < best_c5:
        best_c5, best_h = bf3, bn3

    # Phase 4: Smooth (log-sum-exp) optimization
    t4 = _T0 + 600
    for alpha in [3000, 8000, 15000]:
        if time.time() > t4:
            break
        hna, fna = _lsq_smooth(best_h, best_n, best_n // 2, alpha, _T0, t4)
        if fna < best_c5:
            best_c5, best_h = fna, hna

    # Phase 5: Greedy coordinate sweep
    t5 = _T0 + 800
    hn5, fn5 = _greedy_sweep(best_h, best_n, best_n // 2, _T0, t5, rounds=3)
    if fn5 < best_c5:
        best_c5, best_h = fn5, hn5

    # Phase 6: Fine PG polish with small m
    t6 = _T0 + 950
    hn6, fn6 = _pg_momentum(best_h, best_n, best_n // 2, _T0, t6, m_init=1, m_max=5, lr0=0.03, mu=0.7)
    if fn6 < best_c5:
        best_c5, best_h = fn6, hn6

    # Phase 7: Final greedy + binary check
    t7 = _T0 + budget_s - 10
    hn7, fn7 = _greedy_sweep(best_h, best_n, best_n // 2, _T0, t7, rounds=2)
    if fn7 < best_c5:
        best_c5, best_h = fn7, hn7
    bn7 = _binary_push(best_h, best_n)
    bf7 = _eval(bn7)
    if bf7 < best_c5:
        best_c5, best_h = bf7, bn7

    # Phase 8: Try a random restart if time remains
    if time.time() < _T0 + budget_s - 30:
        hp = best_h + 0.03 * rng.randn(best_n)
        hp = _repair(hp, best_n)
        t8 = _T0 + budget_s - 10
        hn8, fn8 = _pg_momentum(hp, best_n, best_n // 2, _T0, t8, m_init=2, m_max=15, lr0=0.10, mu=0.85)
        if fn8 < best_c5:
            best_c5, best_h = fn8, hn8

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
