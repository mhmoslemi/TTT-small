from scipy.optimize import minimize
import time

# Helper functions (top-level)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_full(h, alpha):
    """Compute softmax-weighted gradient of smoothed max correlation."""
    c = _corr(h)
    n = len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    tot = e.sum()
    w = e / tot
    grad = np.zeros(n)
    # Process all lags
    # ki = 0..2n-2, lag = |ki - (n-1)|
    # For efficiency, group by absolute lag
    for ki in range(2 * n - 1):
        if w[ki] < 1e-18:
            continue
        lab = abs(int(ki - (n - 1)))
        if lab == 0:
            grad += w[ki] * (1.0 - 2.0 * h)
        elif ki > n - 1:  # positive lag
            m2 = n - lab
            grad[lab:] += w[ki] * (1.0 - h[:m2])
            grad[:m2] -= w[ki] * h[lab:]
        else:  # negative lag
            m2 = n - lab
            grad[:m2] += w[ki] * (1.0 - h[lab:])
            grad[lab:] -= w[ki] * h[:m2]
    return grad

def _slsqp_exact(h, dx, alpha, maxiter=500):
    """SLSQP with exact gradient for log-sum-exp smoothed objective."""
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        e = np.exp(s)
        return m + float(np.log(e.sum())) / alpha
    
    def jac(x):
        return _grad_full(x, alpha)
    
    def eq(x):
        return x.sum() - tgt
    
    def eq_jac(x):
        return np.ones(n)
    
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return _repair(h, tgt), _c5(_repair(h, tgt), dx)

Let me think about this more carefully. For a fixed lag s:
c_s(h) = sum_i h[i+s]*(1-h[i])

As a function of h[j] (with all else fixed):
- If j >= s: c_s includes term h[j]*(1-h[j-s]), so c_s = A + h[j]*(1-h[j-s]) where A is the rest
  → c_s is linear in h[j]!
- If j < s: c_s includes term -h[j]*h[j+s], so c_s = A - h[j]*h[j+s]
  → c_s is also linear in h[j]!

Wait, that's because c_s = sum h[i+s] - sum h[i+s]*h[i]. The first sum is linear, the second is bilinear. But fixing all variables except h[j], the sum h[i+s]*h[i] is:
- If j >= s: includes h[j]*h[j-s], which is linear in h[j]
- If j < s and j+s < n: includes h[j+s]*h[j], which is linear in h[j]

So for each individual lag, c_k is LINEAR in h[j] when all other variables are fixed!

This means the problem is convex in each coordinate! And we can do coordinate descent:
- For each j, find the h[j] that minimizes max_k c_k, given all other h fixed
- Since each c_k is linear in h[j], max of linear functions is convex (piecewise linear)
- So we can find the optimum by checking the "breakpoints"

The breakpoints for h[j] in c_k:
- c_k(h[j]) = a_k * h[j] + b_k where a_k is the coefficient

The max of these piecewise linear functions changes slope at the points where two c_k's are equal. Since all are linear in h[j], the max changes at most O(n) times.

But actually, since c_k is linear in h[j] for each k, the max is the upper envelope of lines, which has at most O(n) vertices. We can find the minimum of this upper envelope by:
1. Evaluate max_k c_k at h[j]=0 and h[j]=1
2. The minimum is either at 0, 1, or where two lines cross

Actually, for a set of lines, the minimum of the maximum is at a point where two (or more) lines are the maximum and equal. This can be found by sorting the lines by slope and finding the "lower hull of the upper envelope."

But this is getting complex. Let me just use a simpler approach: for each j, do a ternary search or binary search on h[j] to minimize max_k c_k.

Actually, since max of linear functions is convex, I can use binary search on the subgradient (which is one of the active lines' slopes).

Or even simpler: since it's 1D convex, I can just evaluate at a few points and pick the best. Or use scipy.optimize.minimize_scalar.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_sm(h, alpha):
    c = _corr(h)
    n = len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-16:
            continue
        lab = abs(int(ki - (n - 1)))
        if lab == 0:
            g += wk * (1.0 - 2.0 * h)
        elif ki > n - 1:
            m2 = n - lab
            g[lab:] += wk * (1.0 - h[:m2])
            g[:m2] -= wk * h[lab:]
        else:
            m2 = n - lab
            g[:m2] += wk * (1.0 - h[lab:])
            g[lab:] -= wk * h[:m2]
    return g

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def jac(x):
        return _grad_sm(x, alpha)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
                                   'jac': lambda: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-20})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        h2 = _repair(h, tgt)
        return h2, _c5(h2, dx)

import time

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

        if abs(d) < 1e-14:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_sm(h, alpha):
    c = _corr(h)
    n = len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)

        wk = w[ki]

            m2 = n - lab
            g[lab:] += wk * (1.0 - h[:m2])

        else:
            m2 = n - lab
            g[:m2] += wk * (1.0 - h[lab:])
            g[lab:] -= wk * h[:m2]
    return g

def _slsqp(h, dx, alpha, maxiter=400):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def jac(x):
        return _grad_sm(x, alpha)
    def cfun(x):
        return x.sum() - tgt
    def cjac(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        h2 = _repair(h, tgt)
        return h2, _c5(h2, dx)

def _alpha_sweep(h, dx, t_end, alphas):
    best_h, best_c = h, _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        hc, c = _slsqp(best_h, dx, a, 350)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.05 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(3, n // 4), replace=False)
        hp[idx] += 0.09 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.04 * np.sin(2 * np.pi * rng.uniform(1, 6) * x + rng.uniform(0, 6.28))
    elif mode == 3:
        idx = rng.choice(n, max(2, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    else:
        b = rng.randint(4, n // 3)
        idx = rng.choice(n, b, replace=False)
        hp[idx] += 0.03 * (2.0 * rng.random(b) - 1.0)
    return _repair(np.clip(hp, 0, 1), tgt)

def _block_search(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 5e-04, 2e-04]
    for sc in scales:
        while time.time() < t_end:
            bsz = rng.randint(2, 6)
            i0 = rng.randint(0, max(1, n - bsz))
            blk = h[i0:i0 + bsz].copy()
            shift = sc * (2.0 * rng.random() - 1.0)
            h[i0:i0 + bsz] = blk + shift
            if np.any(h[i0:i0 + bsz] < -0.01) or np.any(h[i0:i0 + bsz] > 1.01):
                h[i0:i0 + bsz] = blk
                continue
            h = np.clip(h, 0, 1)
            h = _repair(h, tgt)
            cn = _c5(h, dx)
            if cn < best:
                best = cn
            else:
                h[i0:i0 + bsz] = blk
                h = _repair(h, tgt)
    return h, best

def _xcorr(h1, h2, rng):
    """Crossover between two solutions."""
    n = len(h1)
    tgt = n / 2.0
    t = rng.uniform(0.2, 0.8)
    hc = t * h1 + (1 - t) * h2
    return _repair(np.clip(hc, 0, 1), tgt)
