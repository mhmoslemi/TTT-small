import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _sum_fun(x):
        return x.sum() - n * 0.5
    def _sum_jac(x):
        return np.ones(n)
    
    def _smooth_obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def _smooth_grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w[k] * (1.0 - x[i1 + sh])
                i2 = np.arange(sh, n)
                g[i2] -= w[k] * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w[k] * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[k] * x[i2 + sp]
        return g
    
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    best_c5 = c.max() * 2.0 / n
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        for i in range(n):
            for j in range(i + 1, n):
                s = h[i] + h[j]
                ti_lo = max(0.0, s - 1.0)
                ti_hi = min(1.0, s)
                if ti_hi - ti_lo < 1e-14:
                    continue
                # Compute p[m] = coeff_i[m] - coeff_j[m]
                ci = np.zeros(2*n-1)
                v1 = (i + shifts >= 0) & (i + shifts < n)
                ci[v1] += 1.0 - h[i + shifts[v1]]
                v2 = (i - shifts >= 0) & (i - shifts < n)
                ci[v2] += -h[i - shifts[v2]]
                
                cj = np.zeros(2*n-1)
                v1 = (j + shifts >= 0) & (j + shifts < n)
                cj[v1] += 1.0 - h[j + shifts[v1]]
                v2 = (j - shifts >= 0) & (j - shifts < n)
                cj[v2] += -h[j - shifts[v2]]
                
                p = ci - cj
                if np.all(np.abs(p) < 1e-15):
                    continue
                
                # Minimize max_m(c[m] + p[m]*d) for d in [ti_lo-h[i], ti_hi-h[i]]

    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += w[k] * (1.0 - x[sh:])
            g[sh:] -= w[k] * x[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += w[k] * (1.0 - x[:cnt])
            g[:cnt] -= w[k] * x[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _sf(x):
        return x.sum() - n * 0.5
    def _sj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _precompute(n):
    sh_raw = np.arange(-(n - 1), n)
    idx_p, idx_m, mask_p, mask_m = [], [], [], []
    for i in range(n):
        mp = (i + sh_raw >= 0) & (i + sh_raw < n)
        mm = (i - sh_raw >= 0) & (i - sh_raw < n)
        mask_p.append(mp)
        mask_m.append(mm)
        idx_p.append(i + sh_raw[mp])
        idx_m.append(i - sh_raw[mm])
    return idx_p, idx_m, mask_p, mask_m

def _pair_descent(h, c, n, idx_p, idx_m, mask_p, mask_m, max_sweeps, t0, deadline):
    best_c5 = c.max() * 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(n):
            for j in range(i + 1, n):
                s = h[i] + h[j]
                ti_lo = max(0.0, s - 1.0)
                ti_hi = min(1.0, s)
                if ti_hi - ti_lo < 1e-13:
                    continue
                ci = np.zeros(2 * n - 1)
                ci[mask_p[i]] += 1.0 - h[idx_p[i]]
                ci[mask_m[i]] += -h[idx_m[i]]
                cj = np.zeros(2 * n - 1)
                cj[mask_p[j]] += 1.0 - h[idx_p[j]]
                cj[mask_m[j]] += -h[idx_m[j]]
                p = ci - cj
                if np.abs(p).max() < 1e-14:
                    continue
                vmax0 = c.max()
                km0 = np.where(np.abs(c - vmax0) < 1e-12)[0]
                if len(km0) == 0:
                    km0 = np.array([c.argmax()])
                if p[km0].min() <= 1e-12 and p[km0].max() >= -1e-12:
                    continue
                d_lo = ti_lo - h[i]
                d_hi = ti_hi - h[i]
                for _ in range(45):
                    d_mid = 0.5 * (d_lo + d_hi)
                    vals = c + p * d_mid
                    vm = vals.max()
                    ks = np.where(np.abs(vals - vm) < 1e-13)[0]
                    if len(ks) == 0:
                        ks = np.array([vals.argmax()])
                    if p[ks].min() > 1e-13:
                        d_hi = d_mid
                    elif p[ks].max() < -1e-13:
                        d_lo = d_mid
                    else:
                        break
                d_opt = 0.5 * (d_lo + d_hi)
                vn = (c + p * d_opt).max()
                if vn < vmax0 - 1e-13:
                    c += p * d_opt
                    h[i] += d_opt
                    h[j] -= d_opt
                    any_change = True
        c5 = c.max() * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
        if not any_change:
            break
    return h, c, best_c5

def _incr(h, c, rng, n_iter, scale, n, deadline):
    c5c = c.max() * 2.0 / n
    sh_raw = np.arange(-(n - 1), n)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        if h[i] + d < -1e-10 or h[i] + d > 1.0 + 1e-10:
            continue
        if h[j] - d < -1e-10 or h[j] - d > 1.0 + 1e-10:
            continue
        dc = np.zeros(2 * n - 1)
        v1 = (i + sh_raw >= 0) & (i + sh_raw < n)
        dc[v1] += d * (1.0 - h[i + sh_raw[v1]])
        v2 = (i - sh_raw >= 0) & (i - sh_raw < n)
        dc[v2] += d * (-h[i - sh_raw[v2]])
        v1 = (j + sh_raw >= 0) & (j + sh_raw < n)
        dc[v1] += -d * (1.0 - h[j + sh_raw[v1]])
        v2 = (j - sh_raw >= 0) & (j - sh_raw < n)
        dc[v2] += -d * (-h[j - sh_raw[v2]])
        nc = c + dc
        if nc.max() * 2.0 / n < c5c - 1e-15:
            h[i] += d
            h[j] -= d
            c = nc
            c5c = nc.max() * 2.0 / n
    return h, c, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    idx_p, idx_m, mask_p, mask_m = _precompute(n0)

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP warm start
    if _el() < 50:
        h1 = _slsqp(h, 50000, 1500, 1e-18)
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5, best_h = c5, h1

    # Phase 2: Exact pairwise coordinate descent
    hp = best_h.copy()
    cp = _corr(hp)
    dl = t0 + min(350, budget_s - 300)
    hp, cp, c5p = _pair_descent(hp, cp, n0, idx_p, idx_m, mask_p, mask_m, 200, t0, dl)
    if c5p < best_c5:
        best_c5, best_h = c5p, hp.copy()

    # Phase 3: SLSQP to escape 2-wise local optimum
    if _el() < budget_s - 250:
        h2 = _slsqp(best_h, 200000, 2000, 1e-19)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 4: More pairwise descent
    hp = best_h.copy()
    cp = _corr(hp)
    dl = t0 + min(budget_s - 150, t0 - t0 + 450)
    hl = t0 + min(budget_s - 150, max(hp.size, 0) + 450)
    hp, cp, c5p = _pair_descent(hp, cp, n0, idx_p, idx_m, mask_p, mask_m, 200, t0, hl)
    if c5p < best_c5:
        best_c5, best_h = c5p, hp.copy()

    # Phase 5: Another SLSQP polish
    if _el() < budget_s - 80:
        h3 = _slsqp(best_h, 400000, 1500, 1e-20)
        c5 = _eval(h3)
        if c5 < best_c5:
            best_c5, best_h = c5, h3

    # Phase 6: Final pairwise + incremental
    hp = best_h.copy()
    cp = _corr(hp)
    hl = t0 + budget_s - 30
    hp, cp, c5p = _pair_descent(hp, cp, n0, idx_p, idx_m, mask_p, mask_m, 100, t0, hl)
    if c5p < best_c5:
        best_c5, best_h = c5p, hp.copy()

    if _el() < budget_s - 10:
        hf = best_h.copy()
        cf = _corr(hf)
        for sc in [0.002, 0.0005, 0.0001, 5e-5]:
            if _el() > budget_s - 10:
                break
            hf, cf, _ = _incr(hf, cf, rng, 40000, sc, n0, t0 + budget_s - 10)
        hf = _repair(hf, n0)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5, best_h = c5, hf

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
