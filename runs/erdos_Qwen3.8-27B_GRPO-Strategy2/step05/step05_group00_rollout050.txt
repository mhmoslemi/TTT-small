import numpy as np
from scipy.optimize import minimize, minimize_scalar, differential_evolution
import time

def _repair(h, n):
    """Repair h to satisfy constraints: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        # Fix sum by adjusting free variables
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            free = np.where((h > 0) & (h < 1))[0]
            if len(free) == 0:
                free = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in free:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _softmax_obj(x, alpha, top_k=0):
    n = len(x)
    c = _correlate(x)
    if top_k > 0:
        idx = np.argsort(c)[-top_k:]
        c_act = c[idx]
        m = c_act.max()
        s = np.clip(alpha * (c_act - m), -100, 100)
        return m + np.log(c_act.size * np.mean(np.exp(s - s.max()))) / alpha
    else:
        m = c.max()
        s = np.clip(alpha * (c - m), -100, 100)
        return m + np.log(c.size * np.mean(np.exp(s - s.max()))) / alpha

def _softmax_grad(x, alpha, top_k=0):
    n = len(x)
    c = _correlate(x)
    dx = 2.0 / n
    if top_k > 0:
        idx = np.argsort(c)[-top_k:]
        c_act = c[idx]
        m = c_act.max()
        s = np.clip(alpha * (c_act - m), -100, 100)
        w_act = np.exp(s - s.max())
        w_act /= w_act.sum()
        w = np.zeros(len(c))
        w[idx] = w_act
    else:
        m = c.max()
        s = np.clip(alpha * (c - m), -100, 100)
        w = np.exp(s - s.max())
        w /= w.sum()
    
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _lbfgs(h, alpha, maxiter=5000, top_k=0):

def _step_h(positions, widths, n):
    """Create step function from block positions and widths"""
    h = np.zeros(n)
    for pos, w in zip(positions, widths):
        i0 = int(round(pos))
        i1 = int(round(pos + w))
        i0 = max(0, min(i0, n-1))
        i1 = max(0, min(i1, n))
        h[i0:i1] = 1.0
    h = _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(_corr(h).max() * 2.0 / n)

def _sm_obj(x, alpha, tk):
    n = len(x)
    c = _corr(x)
    if tk > 0 and tk < len(c):
        idx = np.argsort(c)[-tk:]
        ca = c[idx]
        m = ca.max()
        s = np.clip(alpha*(ca - m), -80, 80)
        return m + np.log(tk * np.mean(np.exp(s - s.max())))/alpha
    m = c.max()
    s = np.clip(alpha*(c - m), -80, 80)
    return m + np.log(len(c) * np.mean(np.exp(s - s.max())))/alpha

def _sm_grad(x, alpha, tk):
    n = len(x)
    c = _corr(x)
    dx = 2.0 / n
    if tk > 0 and tk < len(c):
        idx = np.argsort(c)[-tk:]
        ca = c[idx]
        m = ca.max()
        s = np.clip(alpha*(ca - m), -80, 80)
        w = np.zeros(len(c))
        wa = np.exp(s - s.max()); wa /= wa.sum()
        w[idx] = wa
    else:
        m = c.max()
        s = np.clip(alpha*(c - m), -80, 80)
        w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk*x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0 - x[:n-sp])
            g[:n-sp] -= wk*x[sp:]
    return g

    n = len(h)
    bnds = [(0.0, 1.0)]*n
    try:

                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _slsqp(h, alpha, maxiter=2000, tk=0):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    def _feq(x): return x.sum() - n*0.5
    def _feqj(x): return np.ones(n)
    cons = [{'type':'eq','fun':_feq,'jac':_feqj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, tk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
