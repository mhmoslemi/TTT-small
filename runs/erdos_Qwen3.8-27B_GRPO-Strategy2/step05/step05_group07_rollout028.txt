    # Clip and enforce sum constraint
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_t0 = 0.0
_budget = 1100.0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    c = _corrs(h)
    m = c.max()
    z = np.clip(alpha * (c - m), -500, 500)
    z -= z.max()
    return m + np.log(np.exp(z).sum()) / alpha

def _smooth_grad(h, alpha, topm):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    z = np.clip(alpha * (c - m), -500, 500)
    z -= z.max()
    w = np.exp(z)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1); w[c.argmax()] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                g += wk * (1.0 - 2.0 * h)
            else:

                g[i1] += wk * (1.0 - h[i1 + s])
                i2 = np.arange(s, n)
                g[i2] -= wk * h[i2 - s]
        else:
            sp = -s
            i1 = np.arange(n - sp)
            g[i1] += wk * (1.0 - h[i1 + sp])
            i2 = np.arange(n - sp, n)
            g[i2] -= wk * h[i2 - sp]
    return g * (2.0 / n)

def _nesterov_pg(h_init, alpha, n_iter, lr0):
    n = len(h_init)
    target = n * 0.5
    x = h_init.copy()
    y = h_init.copy()
    f_curr = _eval(h_init)
    for i in range(n_iter):
        if time.time() - _t0 > _budget - 15:
            break
        t_i = (1.0 + np.sqrt(1.0 + 4.0 * (i + 1) ** 2)) / 2.0
        beta = (t_i - 1.0) / t_i
        y = x + beta * (x - y)
        y = np.clip(y, 0, 1)
        g = _smooth_grad(y, alpha, 50)
        lr = lr0 / (1.0 + 0.05 * i)
        x_new = x.copy()

            f_new = _eval(x_new)
            if f_new <= f_curr + 1e-4 * lr * np.dot(g, x_new - y) + 1e-16:
                break
            lr *= 0.5
        x = x_new
        f_curr = _eval(x)
    return x, f_curr

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, alpha, 100),
                       args=(alpha,), method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _con_fun(x, n):
    return x.sum() - n * 0.5

def _con_jac(x, n):
    return np.ones(len(x))

def _pairwise(h, n_iter, scale, sv):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(sv)
    for _ in range(n_iter):

        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-16:
                f = fn
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, f

                h = h_new; f = f_new; found = True
                break
            step *= 0.5

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
