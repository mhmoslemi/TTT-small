from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _delta_c(h, j, jp, k):
    n = len(h)
    kk = k - (n - 1)
    delta = 0.0
    jm = j - kk
    if 0 <= jm < n:
        delta += 1.0 - h[jm]
    jp2 = j + kk
    if 0 <= jp2 < n:
        delta -= h[jp2]
    jpm = jp - kk
    if 0 <= jpm < n:
        delta -= (1.0 - h[jpm])
    jpp = jp + kk
    if 0 <= jpp < n:
        delta += h[jpp]
    return delta

def _slsqp(h, alpha, maxiter):

    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - target
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, target)

        return h
