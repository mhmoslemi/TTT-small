import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_soft(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for sh in range(-(n - 1), n):
        k = n - 1 - sh
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _grad_topk(h, kt):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-kt:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / kt
    dx = 2.0 / n
    g = np.zeros(n)
    for sh in range(-(n - 1), n):
        k = n - 1 - sh
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _bb_loop(h, alpha_fn, steps, deadline):
    n = len(h)
    x = h.copy()
    g = alpha_fn(x)
    fn = _eval(x)
    prev_x = x.copy()
    prev_g = g.copy()
    step = 1.0 / (np.dot(g, g) + 1e-12)
    best_x = x.copy()
    best_fn = fn
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        x_new = _repair(x - step * g, n)
        fn_new = _eval(x_new)
        if fn_new < fn - 1e-14:
            y = alpha_fn(x_new) - g
            s = x_new - x
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-16 and sy > 1e-16:
                step = sy / yy
            step = np.clip(step, 1e-8, 5.0)
            prev_x, prev_g = x.copy(), g.copy()
            x, g, fn = x_new, alpha_fn(x_new), fn_new
            stall = 0
            if fn < best_fn:
                best_fn = fn
                best_x = x.copy()
        else:
            step *= 0.4
            stall += 1
            if step < 1e-10 or stall > 30:
                step = 1e-3
                stall = 0
                x_new = _repair(x - step * g, n)
                fn_new = _eval(x_new)
                if fn_new < fn - 1e-14:
                    x, fn = x_new, fn_new
                    if fn < best_fn:
                        best_fn = fn
                        best_x = x.copy()
                g = alpha_fn(x)
    return best_x, best_fn

def _coord_descent(h, rng, deadline):
    n = len(h)
    x = h.copy()
    fn = _eval(x)
    for scale in [0.05, 0.02, 0.008, 0.003, 0.001]:
        count = 0
        while time.time() < deadline and count < 15000:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = x[i] + d, x[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            x[i], x[j] = ni, nj
            xn = _repair(x, n)
            fn_new = _eval(xn)
            if fn_new < fn - 1e-15:
                x = xn
                fn = fn_new
            else:
                x[i], x[j] = ni - d, nj + d
            count += 1
    return x, fn

def _binary_refine(h, rng, deadline):
    n = len(h)
    x = h.copy()
    fn = _eval(x)
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        n_try = min(4 * n, 3000)
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            old = x[i]
            x[i] = 0.0 if old > 0.4 else 1.0
            xn = _repair(x, n)
            fn_new = _eval(xn)
            if fn_new < fn - 1e-13:
                fn = fn_new
                x = xn.copy()
                improved = True
            else:
                x[i] = old
        if not improved:
            break
    return x, fn

def _resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
