import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    w = np.exp(s)
    ws = w.sum()
    return m + np.log(ws) / alpha

def _smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    w = np.exp(s)
    ws = w.sum()
    w /= ws
    obj = m + np.log(ws) / alpha
    g = np.zeros(n)
    Nc = 2 * n - 1
    for k in range(Nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    g *= dx
    return obj, g

def _lbgfbsb_step(u, alpha, maxiter):
    n = len(u)
    bnds = [(-0.5, 0.5)] * n
    def _of(x):
        return _smooth_max(0.5 + x, alpha)
    def _gf(x):
        _, g = _smooth_max_grad(0.5 + x, alpha)
        g -= np.mean(g)
        return g
    try:
        res = minimize(_of, u, jac=_gf, args=None,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 30})
        return res.x
    except Exception:
        return u

def _nesterov(h, alpha, steps, step0, t_limit, t0):
    n = len(h)
    f_cur = _smooth_max(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _s in range(steps):
        if time.time() > t_limit:
            break
        _, g = _smooth_max_grad(h, alpha)
        v = mu * v + (1.0 - mu) * g
        h_try = _proj(h - step * v, n)
        f_try = _smooth_max(h_try, alpha)
        if f_try < f_cur:
            h = h_try
            f_cur = f_try
            step = min(step * 1.04, step0 * 4)
        else:
            step *= 0.5
    return h

def _upsample_interp(h, factor):
    n = len(h)
    nnew = n * factor
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, nnew)
    return np.interp(x_new, x_old, h)

def _fourier_perturb(h, rng, n_harm, amp):
    n = len(h)
    h_new = h.copy()
    t_arr = np.arange(n)
    for _ in range(rng.randint(1, n_harm + 1)):
        f = rng.randint(1, max(2, n // 4))
        ph = rng.rand() * 2 * np.pi
        a = amp * rng.randn()
        sign = rng.rand()
        if sign < 0.5:
            h_new += a * np.cos(2 * np.pi * f * t_arr / n + ph)
        else:
            h_new += a * np.sin(2 * np.pi * f * t_arr / n + ph)
    return _proj(h_new, n)
