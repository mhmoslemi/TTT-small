        hn, fc = _pairwise(best_h, 12000, 0.00002, seed + 7777)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
