def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort((1.0 - h) if d > 0 else h)
        for i in order:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-12:
                break
    return h


def _sobj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g


def _cfun(x):
    return x.sum() - len(x) * 0.5


def _cjac(x):
    return np.ones(len(x))


def _slsqp(h0, alpha, mi):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_sobj, h0, jac=_sgrad, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h0, n)

import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _repair(h, n):
    h = np.nan_to_num(h)
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort((1.0 - h) if d > 0 else h)
        for i in order:
            a = min(abs(d), (1.0 - h[i]) if d > 0 else h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-12:
                break
    return h


def _sobj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g


def _cfun(x):
    return x.sum() - len(x) * 0.5


def _cjac(x):
    return np.ones(len(x))


def _slsqp(h0, alpha, mi):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_sobj, h0, jac=_sgrad, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h0, n)


def _proj(h, tgt):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)


def _mom(h0, alpha, ns, lr, mu):
    n = len(h0)
    h = h0.copy()
    f = _sobj(h, alpha)
    v = np.zeros(n)
    for _ in range(ns):
        g = _sgrad(h, alpha)
        v = mu * v - lr * g
        h2 = _proj(h - v, n * 0.5)
        f2 = _sobj(h2, alpha)
        if f2 < f:
            h, f = h2, f2
        else:
            v *= 0.5
    return h


def _pair(h0, rng, ni, sc, tp):
    n = len(h0)
    h = h0.copy()
    c5 = _eval(h)
    for _ in range(ni):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = sc * rng.randn()
        a, b = h[i] + d, h[j] - d
        if a < 0 or a > 1 or b < 0 or b > 1:
            continue
        h[i], h[j] = a, b
        c5n = _eval(h)
        if c5n <= c5:
            c5 = c5n
        elif rng.rand() < np.exp(-((c5n - c5) / max(tp, 1e-10))):
            c5 = c5n
        else:
            h[i], h[j] = a - d, b + d
        tp *= 0.9995
    return h, c5


def _elapsed(ts):
    return time.time() - ts
