    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    deadline = t0 + budget_s

    # Phase 1: SLSQP continuation at n0
    for alpha in [200, 2000, 20000, 200000]:
        if time.time() > deadline - 350:
            break
        h_new = _slsqp(best_h, alpha, 2500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Upsample to 2x, SLSQP
    if time.time() < deadline - 300:
        n1 = best_n * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [500, 5000, 50000, 500000]:
            if time.time() > deadline - 180:
                break
            h_hi = _slsqp(h_hi, alpha, 2500, 1e-18)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase 3: Targeted top-K gradient descent
    if time.time() < deadline - 140:
        for K in [3, 5, 8]:
            if time.time() > deadline - 120:
                break
            h_g, c5_g = _pg_search(best_h, 30.0, K, 4000, deadline - 120)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 4: SLSQP polish at current resolution
    if time.time() < deadline - 100:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 80:
                break
            h_new = _slsqp(best_h, alpha, 4000, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Multi-start at current resolution
    nc = len(best_h)
    rng = np.random.RandomState(seed + 7)
    if time.time() < deadline - 80:
        for _r in range(20):
            if time.time() > deadline - 60:
                break
            s = rng.randint(5)
            if s == 0:
                hp = _repair(best_h + 0.008 * rng.randn(nc), nc)
            elif s == 1:
                hp = _repair(best_h * rng.uniform(0.92, 1.08), nc)
                hp += 0.005 * rng.randn(nc)
                hp = _repair(hp, nc)
            elif s == 2:
                hp = best_h.copy()
                bs = rng.randint(4, max(5, nc // 8))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.015 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif s == 3:
                f = rng.randint(1, nc // 4 + 1)
                amp = 0.004 * rng.rand()
                ta = np.arange(nc)
                hp = _repair(best_h + amp * np.sin(2 * np.pi * f * ta / nc + rng.rand() * 6.28), nc)
            else:
                hp = best_h.copy()
                i = rng.randint(nc // 4)
                hp[i:i + nc // 4] = 1.0 - hp[i:i + nc // 4]
                hp = _repair(hp, nc)
            h_new = _slsqp(hp, 100000, 2500, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Fine gradient descent
    if time.time() < deadline - 40:
        h_g, c5_g = _pg_search(best_h, 100.0, 3, 10000, deadline - 40)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 7: Pairwise coordinate search
    if time.time() < deadline - 25:
        for scale in [0.001, 0.0003, 0.0001, 3e-5, 1e-5]:
            if time.time() > deadline - 15:
                break
            h_p, c5_p = _pairwise(best_h, 20000, scale, deadline - 15)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 8: Final SLSQP
    if time.time() < deadline - 5:
        h_new = _slsqp(best_h, 2000000, 5000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 9: Ultra-fine pairwise
    if time.time() < deadline - 1:
        h_p, c5_p = _pairwise(best_h, 50000, 5e-6, deadline - 0.5)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
