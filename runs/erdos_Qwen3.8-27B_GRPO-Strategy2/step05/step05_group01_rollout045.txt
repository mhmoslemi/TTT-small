    if time.time() - t0 < budget_s - 150:
        h_ps = best_h.copy()
        c_ps = _corr(h_ps)
        scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for sc in scales:
            if time.time() - t0 > budget_s - 150:
                break
            n_it = 40000 if sc >= 0.001 else 50000
            h_ps, c_ps, _ = _pair_search(h_ps, c_ps, rng, n_it, sc, n)
            h_ps = _repair(h_ps, n)
            c_ps = _corr(h_ps)
            c5 = c_ps.max() * 2.0 / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ps.copy()

    # Phase 5: More coordinate descent on best
    if time.time() - t0 < budget_s - 100:
        h_c2 = best_h.copy()
        c_c2 = _corr(h_c2)
        for cyc in range(500):
            if time.time() - t0 > budget_s - 100:
                break
            h_c2, c_c2 = _coord_cycle(h_c2, c_c2, rng.permutation(n), n, 25)
            if cyc % 15 == 14:
                h_c2 = _repair(h_c2, n)
                c_c2 = _corr(h_c2)
            c5 = c_c2.max() * 2.0 / n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c2.copy()

    # Phase 6: Simulated annealing to escape local optima
    if time.time() - t0 < budget_s - 60:
        h_sa = best_h.copy()
        c_sa = _corr(h_sa)
        h_sa, c_sa, _ = _sa_search(h_sa, c_sa, rng, 40, 0.004, 0.00005, n, 15000)
        h_sa = _repair(h_sa, n)
        c5 = _eval(h_sa)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sa.copy()
            # Quick coord descent after SA
            h_c3 = best_h.copy()
            c_c3 = _corr(h_c3)
            for _ in range(100):
                if time.time() - t0 > budget_s - 60:
                    break
                h_c3, c_c3 = _coord_cycle(h_c3, c_c3, rng.permutation(n), n, 20)
                h_c3 = _repair(h_c3, n)
                c_c3 = _corr(h_c3)
            c5b = c_c3.max() * 2.0 / n
            if c5b < best_c5:
                best_c5 = c5b
                best_h = h_c3.copy()

    # Phase 7: Final ultra-fine pairwise
    if time.time() - t0 < budget_s - 5:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        for sc in [5e-5, 2e-5, 1e-5, 5e-6]:
            if time.time() - t0 > budget_s - 5:
                break
            h_fin, c_fin, _ = _pair_search(h_fin, c_fin, rng, 40000, sc, n)
        h_fin = _repair(h_fin, n)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin.copy()

    # Final coord polish
    if time.time() - t0 < budget_s - 2:
        h_fp = best_h.copy()
        c_fp = _corr(h_fp)
        for _ in range(50):
            if time.time() - t0 > budget_s - 2:
                break
            h_fp, c_fp = _pair_cycle(h_fp, c_fp, rng.permutation(n), n, 20)
        h_fp = _repair(h_fp, n)
        c5 = _eval(h_fp)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fp.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
