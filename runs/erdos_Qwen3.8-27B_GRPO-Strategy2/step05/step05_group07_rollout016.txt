        if _elapsed() < budget_s - 50:
            hn, fc = _descent(best_h, 5, 4000, 0.004, 0.002)
            if fc < best_c5:
                best_c5 = fc; best_h = hn.copy()

    # Phase 6: Multi-start at best resolution
    nc = len(best_h)
    if _elapsed() < budget_s - 80:
        for r in range(6):
            if _elapsed() > budget_s - 60:
                break
            hp = best_h + 0.01 * rng.randn(nc)
            hp = _repair(hp, nc)
            hn = _slsqp_run(hp, 50000, 800, 1e-15)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5; best_h = hn.copy()
                hn, fc = _descent(hn, 5, 2000, 0.004, 0.002)
                if fc < best_c5:
                    best_c5 = fc; best_h = hn.copy()

    # Phase 7: Fine pairwise
    for sc, ni in [(0.003, 8000), (0.001, 12000), (0.0003, 15000), (0.0001, 12000)]:
        if _elapsed() > budget_s - 15:
            break
        hn, fc = _pairwise(best_h, ni, sc, seed + int(sc * 1e7))
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()

    # Phase 8: Final active descent polish
    if _elapsed() < budget_s - 8:
        hn, fc = _descent(best_h, 4, 4000, 0.003, 0.001)
        if fc < best_c5:
            best_c5 = fc; best_h = hn.copy()
    if _elapsed() < budget_s - 3:
        hn = _slsqp_run(best_h, 300000, 800, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
