import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grads(h, n):
    """Compute d(c[k])/d(h[j]) for all j,k. Returns (n, 2n-1) array."""
    g = np.zeros((n, 2 * n - 1))
    for lag in range(n):
        k = n - 1 + lag
        g[lag:, k] += (1.0 - h[:n - lag])
        g[:n - lag, k] -= h[lag:]
    for p in range(1, n):
        k = n - 1 - p
        g[:n - p, k] += (1.0 - h[p:])
        g[p:, k] -= h[:n - p]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, alpha, n),
                       args=(), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _all_pairs_cd(h, n, n_passes=10, verbose=False):
    """All-pairs coordinate descent with analytical gradients."""
    dx = 2.0 / n
    best_c5 = _eval(h)
    c = _corr(h)
    for _pass in range(n_passes):
        gr = _grads(h, n)
        c = _corr(h)
        moved = 0
        for i in range(n):
            for j in range(i + 1, n):
                d_vec = gr[i] - gr[j]
                dm = max(-h[i], h[j] - 1.0)
                dM = min(1.0 - h[i], h[j])
                if dm >= dM - 1e-14:
                    continue
                k1 = np.argmax(c)
                srt = np.argsort(c)
                k2 = srt[-2]
                cands = [0.0, dm, dM]
                if abs(d_vec[k1] - d_vec[k2]) > 1e-15:
                    dc = (c[k2] - c[k1]) / (d_vec[k1] - d_vec[k2])
                    dc = max(dm, min(dM, dc))
                    cands.append(dc)
                best_f = c[k1]
                best_d = 0.0
                for dc in cands:
                    f = np.max(c + dc * d_vec)
                    if f < best_f - 1e-14:
                        best_f = f
                        best_d = dc
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    c = c + best_d * d_vec
                    moved += 1
        c5 = c.max() * dx
        if c5 < best_c5:
            best_c5 = c5
        if moved == 0:
            break
        if verbose:
            print(f"  pass {_pass}: {moved} moves, c5={c5:.6f}")
    return h, best_c5

def _structured_search(h, n, rng, n_iter=20000, budget_fn=None):
    """Structured local search: swaps, reversals, segment moves."""
    c = _corr(h)
    dx = 2.0 / n
    c5c = c.max() * dx
    for _it in range(n_iter):
        if budget_fn is not None and budget_fn() < 10:
            break
        op = rng.randint(4)
        if op < 2:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c_new = _corr(h)
            if c_new.max() * dx < c5c - 1e-14:
                c = c_new
                c5c = c_new.max() * dx
            else:
                h[i], h[j] = h[j], h[i]
        else:
            if op == 2:
                i = rng.randint(n - 1)
                j = i + rng.randint(2, min(20, n - i))
                h[i:j] = h[i:j][::-1]
                c_new = _corr(h)
                if c_new.max() * dx < c5c - 1e-14:
                    c = c_new
                    c5c = c_new.max() * dx
                else:
                    h[i:j] = h[i:j][::-1]
            else:
                i = rng.randint(n - 1)
                ln = rng.randint(2, min(10, n - i))
                j = rng.randint(n - ln)
                if j == i:
                    continue
                seg = h[i:i + ln].copy()
                if j > i:
                    h[i:i + ln + 1] = np.concatenate([h[i + ln:i + ln], seg, h[i + ln + 1:] if i + ln + 1 < n else np.array([])])
                else:
                    h[j:j + ln + 1] = np.concatenate([seg, h[j:i] if j < i else np.array([])])
                c_new = _corr(h)
                if c_new.max() * dx < c5c - 1e-14:
                    c = c_new
                    c5c = c_new.max() * dx
                else:
                    h[i:i + ln] = seg
                    h = _repair(h, n)
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
