import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_max(h, M):
    """Gradient of average of top-M correlation values, times dx"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    topM = np.argpartition(-c, M - 1)[:M]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in topM:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g * dx / M

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha*(c-m), -400, 400)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(np.clip(alpha*(c-m), -400, 400))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12: continue
        sh = (n-1) - k
        if sh >= 0:
            g[:n-sh] += w[k]*(1.0-h[sh:n])
            if sh > 0: g[sh:] -= w[k]*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w[k]*(1.0-h[:n-sp])
            g[:n-sp] -= w[k]*h[sp:]
    return g

def _sf(x): return x.sum() - len(x)*0.5
def _sj(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    try:
        r = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                     method='SLSQP', bounds=[(0.,1.)]*n,
                     constraints=[{'type':'eq','fun':_sf,'jac':_sj}],
                     options={'maxiter':maxiter,'ftol':1e-15})
        return _repair(r.x, n)
    except: return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    lo, hi = -1.5, 1.5

        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

            hi2 = min(n, n - sh)
            g[lo2:hi2] -= h[lo2 + sh:hi2 + sh]
    return g * dx / M

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -400, 400)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(np.clip(alpha * (c - m), -400, 400))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = k - (n - 1)
        if sh == 0:
            g += w[k] * (1.0 - 2.0 * h)
        else:
            lo1 = max(0, sh)
            hi1 = min(n, n + sh)
            g[lo1:hi1] += w[k] * (1.0 - h[lo1 - sh:hi1 - sh])
            lo2 = max(0, -sh)
            hi2 = min(n, n - sh)
            g[lo2:hi2] -= w[k] * h[lo2 + sh:hi2 + sh]
    return g

def _sf(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    try:
        r = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topM(h, M):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    topM = np.argpartition(-c, M - 1)[:M]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in topM:
        sh = k - (n - 1)
        if sh == 0:
            g += 1.0 - 2.0 * h
        else:
            lo1 = max(0, sh); hi1 = min(n, n + sh)
            g[lo1:hi1] += 1.0 - h[lo1 - sh:hi1 - sh]
            lo2 = max(0, -sh); hi2 = min(n, n - sh)
            g[lo2:hi2] -= h[lo2 + sh:hi2 + sh]
    return g * dx / M

def _lse_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -400, 400)))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(np.clip(alpha * (c - m), -400, 400))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12: continue
        sh = k - (n - 1)
        if sh == 0:
            g += w[k] * (1.0 - 2.0 * h)
        else:
            lo1 = max(0, sh); hi1 = min(n, n + sh)
            g[lo1:hi1] += w[k] * (1.0 - h[lo1 - sh:hi1 - sh])
            lo2 = max(0, -sh); hi2 = min(n, n - sh)
            g[lo2:hi2] -= w[k] * h[lo2 + sh:hi2 + sh]
    return g

def _sf(x): return x.sum() - len(x) * 0.5
def _sj(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    try:
        r = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sf, 'jac': _sj}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP LSE at increasing alphas
    for a in [200, 500, 2000, 10000]:
        if time.time() - t0 > 180: break
        hn = _slsqp(best_h, a, 2500)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5, best_h = c5, hn.copy()

    # Phase 2: Active-set projected gradient with momentum
    h, c5 = best_h.copy(), best_c5
    vel = np.zeros(n)
    M_seq = [1, 3, 5, 3, 1, 5, 8, 3, 1, 2]
    max_rd = 6000
    for _rd in range(max_rd):
        if time.time() - t0 > budget_s - 350: break
        M = M_seq[_rd % len(M_seq)]
        g = _grad_topM(h, M)
        vel = 0.75 * vel + 0.25 * g
        sf = max(0.15, 1.0 - _rd / max_rd)
        improved = False
        for base in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            step = base * sf
            hn = _proj(h - step * vel, n * 0.5)
            c5n = _eval(hn)
            if c5n < c5 - 1e-14:
                h, c5 = hn, c5n; improved = True
                if c5 < best_c5: best_c5, best_h = c5, h.copy()
                break
        if not improved:
            h = _repair(h + 0.003 * rng.randn(n), n)
            c5 = _eval(h)
            vel *= 0.3

    # Phase 3: Extended pairwise search with adaptive scale
    h, c5 = best_h.copy(), best_c5
    for scale in [0.005, 0.002, 0.001, 0.0004, 0.0001, 4e-5, 1e-5]:
        if time.time() - t0 > budget_s - 80: break
        for _it in range(25000):
            if time.time() - t0 > budget_s - 80: break
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                if c5 < best_c5: best_c5, best_h = c5, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d

    # Phase 4: Multi-start diversified
    if time.time() - t0 < budget_s - 120:
        for r in range(12):
            if time.time() - t0 > budget_s - 70: break
            s = rng.randint(5)
            if s == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n), n)
            elif s == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n), n)
            elif s == 2:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, n // 6))
                st = rng.randint(0, n - bs)
                hp[st:st + bs] += 0.02 * rng.randn(bs)
                hp = _repair(hp, n)
            elif s == 3:
                f = rng.randint(2, n // 4 + 1)
                hp = best_h + 0.004 * np.sin(2 * np.pi * f * np.arange(n) / n
                                               + rng.rand() * 6.28)
                hp = _repair(hp, n)
            else:
                hp = _repair(0.5 + 0.25 * rng.randn(5), n)
                hp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, 5), hp)
                hp = _repair(hp + 0.03 * rng.randn(n), n)
            hn = _slsqp(hp, 10000, 1500)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h = c5, hn.copy()

    # Phase 5: Short active-set polish at best
    if time.time() - t0 < budget_s - 30:
        h, c5 = best_h.copy(), best_c5
        for _rd in range(1500):
            if time.time() - t0 > budget_s - 15: break
            M = 1 if _rd % 3 != 0 else 3
            g = _grad_topM(h, M)
            improved = False
            for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 1e-4]:
                hn = _proj(h - step * g, n * 0.5)
                c5n = _eval(hn)
                if c5n < c5 - 1e-14:
                    h, c5 = hn, c5n; improved = True
                    if c5 < best_c5: best_c5, best_h = c5, h.copy()
                    break
            if not improved:
                h = _repair(h + 0.001 * rng.randn(n), n)
                c5 = _eval(h)

    # Phase 6: Final SLSQP + ultra-fine pairwise
    if time.time() - t0 < budget_s - 8:
        hn = _slsqp(best_h, 50000, 2000)
        c5 = _eval(hn)
        if c5 < best_c5: best_c5, best_h = c5, hn.copy()

    if time.time() - t0 < budget_s - 2:
        h, c5 = best_h.copy(), best_c5
        for _it in range(30000):
            if time.time() - t0 > budget_s - 1: break
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = 2e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5 - 1e-16:
                    c5 = c5n
                    if c5 < best_c5: best_c5, best_h = c5, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
