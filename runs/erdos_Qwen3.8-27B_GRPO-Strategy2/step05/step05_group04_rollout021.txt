    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]

            w = 1.0 - h_cur[sh:]  # length n-sh

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]

        sh = k - (n-1)
        if sh >= 0:
            w = (1.0 - h_cur[sh:])  # length n-sh
            constraints.append(cp.sum(cp.multiply(w, h_var[:n-sh])) * dx <= t_var)
        else:
            sp = -sh
            w = (1.0 - h_cur[:n-sp])  # length n-sp
            constraints.append(cp.sum(cp.multiply(w, h_var[sp:])) * dx <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver)
    if h_var.value is None:
        return h_cur.copy()
    return np.asarray(h_var.value)

def _grad_direction(h, n, top_m=None):
    """Compute weighted subgradient of max of correlation values."""
    c = _corr(h)
    m = c.max()
    # Use top-m lags
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[-top_m:]
        w = c[idx] - m + 1.0  # positive weights for top lags
        w = w / w.sum()
    else:
        w = np.ones(len(c)) / len(c)
        # Actually use soft-max
        alpha = 200.0
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s)
        w /= w.sum()
    
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        sh = k - (n-1)
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += wk * (1.0 - h[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2-sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1-sp])
            i2 = np.arange(n-sp)
            g[i2] -= wk * h[i2+sp]
    return g

    lo, hi = -2.0, 2.0

            if _el() > budget_s - 200:
                break
            h_lp = _lp_solve(h2, nc, 60, solver)
            h_lp = _repair(h_lp, nc)
            h2 = 0.5 * h2 + 0.5 * h_lp
            h2 = _repair(h2, nc)
            c5 = _eval(h2)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 5: Gradient + LP interleaved polish
    if _el() < budget_s - 100:
        h = best_h.copy()
        c5_cur = _eval(h)
        for it in range(2000):
            if _el() > budget_s - 100:
                break
            if it % 5 == 0:
                h_lp = _lp_solve(h, nc, 30, solver)
                h_lp = _repair(h_lp, nc)
                h = 0.4 * h + 0.6 * h_lp
                h = _repair(h, nc)
            else:
                g = _grad(h, nc, top_m=20)
                step = 0.002
                h_new = _proj_sum(h - step * g, nc)
                c5_test = _eval(h_new)
                if c5_test < c5_cur:
                    h = h_new
                    c5_cur = c5_test
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Fine pairwise local search
    if _el() < budget_s - 10:
        h = best_h.copy()
        c5c = _eval(h)
        remaining = budget_s - 10 - _el()
        n_tries = int(remaining / 0.004)
        for _it in range(n_tries):
            if _el() > budget_s - 10:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 8e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
