                # So positions j where h[j]=1 and h[j-s2]=0
                bad = np.where((h[s2:] > 0.5) & (h[:m2] < 0.5))[0]
                if len(bad) == 0:
                    continue
                n_tries = min(6, len(bad))
                for _ in range(n_tries):
                    j = bad[rng.randint(len(bad))] + s2
                    # h[j]=1, h[j-s2]=0
                    # Option: h[j-s2]→1, find k with h[k]=1→0
                    ones = np.where(h > 0.5)[0]
                    if len(ones) == 0:
                        continue
                    if s2 < n:
                        ov = ones[ones + s2 < n]
                        if len(ov) > 0:
                            prev = ov[h[ov + s2] > 0.5]
                            k = prev[rng.randint(len(prev))] if len(prev) > 0 else ov[rng.randint(len(ov))]
                        else:
                            k = ones[rng.randint(len(ones))]
                    else:
                        k = ones[rng.randint(len(ones))]
                    
                    h[j - s2] = 1.0; h[k] = 0.0
                    nc = _c5(h, dx)
                    if nc < best_c or (T > 0 and rng.random() < np.exp(-(nc - best_c)/T)):
                        if nc < best_c:
                            best_c = nc
                        improved = True
                        stall = 0
                        break
                    h[j - s2] = 0.0; h[k] = 1.0
        
        if not improved:
            stall += 1
        if T > 0:
            T *= 0.9995
    
    return h, best_c

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _periodic_pattern(n, p, k):
    """Generate best periodic pattern with period p, k ones per period."""
    from itertools import combinations
    best_pat = None
    best_max = float('inf')
    for combo in combinations(range(p), k):
        pat = np.zeros(p)
        pat[list(combo)] = 1.0
        # Autocorrelation of pattern
        ac = np.correlate(pat, pat, mode='full')
        # For the full sequence, the correlation at lag s depends on s mod p
        # For lag s: approximately (n-s)/p * ac[(s+p-1)+p-1]... 
        # Actually for periodic extension, corr at lag s = (n-|s|)/p * ac_base[s mod p]
        # where ac_base is the circular autocorrelation
        circ_ac = np.zeros(p)
        for t in range(p):
            circ_ac[t] = np.sum(pat * np.roll(pat, -t))
        # The "overlap" at lag s ≈ (n-|s|)/n * n/2 - (n-|s|)/p * circ_ac[s%p]
        # = (n-|s|) * (n/(2n) - circ_ac[s%p]/p)
        # = (n-|s|) * (1/2 - circ_ac[s%p]/p)
        max_over = 0
        for s in range(1, n):
            over = (n - s) * (0.5 - circ_ac[s % p] / p)
            max_over = max(max_over, over)
        # Also check small s where the (n-s) factor makes it larger
        if max_over < best_max:
            best_max = max_over
            best_pat = pat.copy()
    # Tile
    if best_pat is not None:
        h = np.tile(best_pat, n // p)
        if n % p > 0:
            h = np.concatenate([h, best_pat[:n % p]])
        h = h[:n]
        return h, best_max
    return None, float('inf')
