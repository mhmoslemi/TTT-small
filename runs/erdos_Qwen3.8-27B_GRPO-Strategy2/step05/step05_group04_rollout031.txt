def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

def _lag_grad(h, k, n):

    dx = 2.0 / n
    g = np.zeros(n)

    """Subgradient method cycling through top-K worst lags with Adam-like update."""
    best_h = h.copy()

    v = np.zeros(n)  # Adam second moment

        if c_new < c_max + 0.001:  # accept if not too much worse

    if sh >= 0:

        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:

            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return c[k_idx].sum() / len(k_idx)

def _topk_grad(h, k_idx, n):
    """Gradient of sum of overlaps at specified lag indices."""
    g = np.zeros(n)
    for k in k_idx:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    sh = (n - 1) - k

    else:
        sp = -sh
        g[sp:] += 1.0 - h[:n - sp]
        g[:n - sp] -= h[sp:]
    return g * dx

            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    return x.sum() - n * 0.5

    return np.ones(n)

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

        else:

            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

    except Exception:
