from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj_pen(h, temp, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    z = (c - m) / temp
    smax = m + temp * np.log(np.mean(np.exp(z - z.max())))
    pen = mu * (h.sum() - n / 2.0) ** 2
    return smax + pen

def _smax_grad_pen(h, temp, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    z = (c - m) / temp
    w = np.exp(z - z.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            l = lag
            g[l:] += wk * (1.0 - h[:n - l])
            g[:n - l] -= wk * h[l:]
        else:
            l = -lag
            g[:n - l] += wk * (1.0 - h[l:])
            g[l:] -= wk * h[:n - l]
    pen_val = h.sum() - n / 2.0
    g += 2.0 * mu * pen_val
    return g

def _lbfgs(h, temp, mu, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_obj_pen, h, jac=_smax_grad_pen, args=(temp, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 80})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fista(h, temp, n_iter, step_init):
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    m = 1.0
    step = step_init
    for it in range(n_iter):
        y = h_cur + ((m - 1) / (m + 1)) * (h_cur - h_prev)
        gy = _smax_grad_pen(y, temp, 1e8)
        h_new = _repair(y - step * gy, n)
        m_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * m * m))
        h_prev, h_cur, m = h_cur, h_new, m_new
        step *= 0.999
    return h_cur

def _upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)

def _coord_descent(h, n_max_iter=30000):
    nc = len(h)
    c5c = _c5(h)
    for _ in range(n_max_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        k_max = c.argmax()
        lag = k_max - (nc - 1)
        l = abs(lag)
        if l == 0 or l >= nc:
            break
        if lag > 0:
            i1 = np.arange(nc - l)
        else:
            i1 = np.arange(l, nc)
        for j in i1:
            partner = j + lag if lag > 0 else j + lag
            if partner < 0 or partner >= nc:
                continue
            for delta in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]:
                h[j] += delta
                h[partner] += delta
                h_rep = _repair(h, nc)
                c5n = _c5(h_rep)
                if c5n < c5c - 1e-15:
                    h, c5c = h_rep, c5n
                    break
            else:
                h[j] -= delta
                h[partner] -= delta
    return h, c5c

def _pairwise_search(h, n_iter=50000, scale=1e-4):
    nc = len(h)
    rng = np.random.RandomState(123)
    c5c = _c5(h)
    for _ in range(n_iter):
        i, j = rng.randint(nc), rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        h[i], h[j] = h[i] + d, h[j] - d
        c5n = _c5(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return _repair(h, nc), c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _c5(h), n0

    def elapsed():
        return time.time() - t0

    def time_left():
        return deadline - time.time()

    # Phase 1: L-BFGS-B at n0 with increasing temperature
    for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time_left() < 120:
            break
        h_new = _lbfgs(best_h, temp, 1e5, 800, 1e-18)
        c5 = _c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: FISTA polish at n0
    if time_left() > 150:
        h_new = _fista(best_h, 0.001, 5000, 0.01)
        c5 = _c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Upsample to 2*n0 and optimize
    if time_left() > 300:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time_left() < 200:
                break
            h_hi = _lbfgs(h_hi, temp, 1e5, 600, 1e-18)
            c5 = _c5(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase 4: Multi-start with diverse perturbations
    nc = best_n
    if time_left() > 200:
        for _r in range(20):
            if time_left() < 120:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = best_h + 0.03 * rng.randn(nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = rng.randint(10, nc // 4)
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.04 * rng.randn(bs)
            elif strat == 2:
                f = rng.randint(2, nc // 5 + 1)
                amp = 0.01 * rng.uniform(0.5, 1.5) * (1 if rng.rand() < 0.5 else -1)
                t_arr = np.arange(nc)
                hp = best_h + amp * np.sin(2 * np.pi * f * t_arr / nc + rng.rand() * 6.28)
            elif strat == 3:
                hp = 0.5 + 0.4 * rng.randn(8)
                hp = np.interp(np.linspace(0, nc, nc), np.linspace(0, 8, nc), hp)
            else:
                hp = best_h * rng.uniform(0.9, 1.1)
            hp = _repair(hp, nc)
            h_new = _lbfgs(hp, 0.01, 1e5, 500, 1e-17)
            c5 = _c5(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 5: Focused coordinate descent on active lags
    if time_left() > 100:
        h = best_h.copy()
        nc = len(h)
        c5c = _c5(h)
        for _round in range(200):
            if time_left() < 60:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            dx = 2.0 / nc
            thresh = c.max() * 0.999
            for k in np.where(c >= thresh)[0]:
                lag = k - (nc - 1)
                l = abs(lag)
                if l == 0 or l >= nc:
                    continue
                j = rng.randint(l, nc - l) if l < nc - l else rng.randint(nc)
                partner = j + lag
                if partner < 0 or partner >= nc:
                    continue
                for step in [0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002]:
                    h[j] += step
                    h[partner] -= step
                    h_rep = _repair(h, nc)
                    c5n = _c5(h_rep)
                    if c5n < c5c - 1e-15:
                        h, c5c = h_rep, c5n
                        if c5c < best_c5:
                            best_c5, best_h = c5c, h.copy()
                        break
                else:
                    h[j] -= step
                    h[partner] += step
                    h = _repair(h, nc)
                    c5c = _c5(h)

    # Phase 6: Pairwise random search (fine)
    if time_left() > 50:
        h, c5c = _pairwise_search(best_h, n_iter=30000, scale=5e-5)
        if c5c < best_c5:
            best_c5, best_h = c5c, h.copy()

    # Phase 7: Final L-BFGS polish with very tight temperature
    if time_left() > 15:
        h_new = _lbfgs(best_h, 0.0003, 1e6, 1000, 1e-20)
        c5 = _c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Final ultra-fine pairwise
    if time_left() > 5:
        h, c5c = _pairwise_search(best_h, n_iter=20000, scale=1e-6)
        if c5c < best_c5:
            best_c5, best_h = c5c, h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
