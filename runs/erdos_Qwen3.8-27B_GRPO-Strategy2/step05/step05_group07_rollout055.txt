        hn = _slsqp_trunc(best_h, 150000, 25, 500, _t0, t_end - 110)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 3: Focus on worst lag with steepest descent
    if time.time() < t_end - 100:
        hn, fn = _focus_lag(best_h, 6000, 3, 0.01, _t0, t_end - 80)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 4: More accelerated gradient with small top_k
    if time.time() < t_end - 80:
        for top_k in [3, 5]:
            if time.time() > t_end - 60:
                break
            hn, fn = _accel_descend(best_h, 6000, top_k, 0.004, _t0, t_end - 50)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()

    # Phase 5: High-alpha SLSQP
    if time.time() < t_end - 60:
        hn = _slsqp_trunc(best_h, 500000, 15, 500, _t0, t_end - 40)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 6: Multi-start perturbations
    if time.time() < t_end - 50:
        for sp in [0.01, 0.005, 0.002]:
            if time.time() > t_end - 35:
                break
            hp = _repair(best_h + sp * rng.randn(n), n)
            hn, fn = _accel_descend(hp, 2000, 5, 0.008, _t0, t_end - 30)
            if fn < best_c5:
                best_c5 = fn
                best_h = hn.copy()

    # Phase 7: Pairwise fine-tuning (progressively smaller steps)
    for scale, nit in [(0.004, 6000), (0.002, 8000), (0.001, 10000), (0.0005, 8000), (0.0002, 6000)]:
        if time.time() > t_end - 12:
            break
        hn, fn = _pairwise(best_h, nit, scale, _t0, t_end - 8, seed + int(scale * 1e7))
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    # Phase 8: Final SLSQP polish
    if time.time() < t_end - 25:
        hn = _slsqp_trunc(best_h, 1000000, 10, 400, _t0, t_end - 5)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 9: Final tiny pairwise
    if time.time() < t_end - 5:
        hn, fn = _pairwise(best_h, 4000, 0.0001, _t0, t_end - 1, seed + 777)
        if fn < best_c5:
            best_c5 = fn
            best_h = hn.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
