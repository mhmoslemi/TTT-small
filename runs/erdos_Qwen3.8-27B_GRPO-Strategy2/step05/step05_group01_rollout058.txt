def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

                    break
    return h

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    mid = n - 1
    sh1 = np.arange(0, n - i)
    k1 = mid - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = mid - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = mid + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

    m = c.max()

    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:

            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
