def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

from scipy.optimize import minimize, linprog

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h = h * (tgt / s)
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, t):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t: lo = m
        else: hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth(h, al):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(al * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / al) * (2.0 / n)

def _sg(h, al):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(al * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    return _smooth(x, al) + 1e5 * (x.sum() - n * 0.5) ** 2

def _pgrad(x, al, n):
    return _sg(x, al) + 2e5 * (x.sum() - n * 0.5)

def _eqf(x):
    return x.sum() - len(x) * 0.5

def _eqj(x):
    return np.ones(len(x))

def _lbfgs(h, al, mi):
    n = len(h)
    b = [(0.0, 1.0)] * n
    r = minimize(_pobj, _repair(h, n), jac=_pgrad, args=(al, n),
                 method='L-BFGS-B', bounds=b, options={'maxiter': mi, 'gtol': 1e-14})
    return _repair(r.x, n)

def _slsqp(h, al, mi):
    n = len(h)
    b = [(0.0, 1.0)] * n
    r = minimize(_smooth, _repair(h, n), jac=_sg, args=(al,),
                 method='SLSQP', bounds=b,
                 constraints=[{'type': 'eq', 'fun': _eqf, 'jac': _eqj}],
                 options={'maxiter': mi, 'ftol': 1e-20})
    return _repair(r.x, n)
