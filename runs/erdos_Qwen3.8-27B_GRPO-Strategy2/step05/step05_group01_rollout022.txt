def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

    h = np.clip(h, 0.0, 1.0).copy()

            else:
                hi = mid

                    h[idx] += a
                    diff -= a
                else:

                    diff += a

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B with increasing alpha
    for alpha in [500, 2000, 8000, 30000, 100000, 400000]:
        if time.time() - t0 > 80:
            break
        hn = _lbfgs(best_h, alpha, 300)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn

    # Phase 2: Block hill-climbing for exploration
    if time.time() - t0 < budget_s - 350:
        hbd = best_h.copy()
        c5bd = best_c5
        for _ in range(4000):
            if time.time() - t0 > budget_s - 350:
                break
            bs = int(rng.choice([3, 4, 5, 6, 8, 10]))
            if bs >= n - 1:
                continue
            st = rng.randint(0, n - bs + 1)
            idx = np.arange(st, st + bs)
            bsum = hbd[idx].sum()
            newb = hbd[idx] + 0.015 * rng.randn(bs)
            newb = np.clip(newb, 0, 1)
            newb += (bsum - newb.sum()) / bs
            newb = np.clip(newb, 0, 1)
            ht = hbd.copy()
            ht[idx] = newb
            ht = _repair(ht, n)
            c5t = _eval(ht)
            if c5t < c5bd:
                c5bd, hbd = c5t, ht
        if c5bd < best_c5:
            best_c5, best_h = c5bd, hbd

    # Phase 3: Simulated annealing with pair moves
    if time.time() - t0 < budget_s - 180:
        hsa = best_h.copy()
        csa = np.correlate(hsa, 1.0 - hsa, mode='full')
        c5sa = float(csa.max() * 2.0 / n)
        T = 0.025
        for it in range(25000):
            if time.time() - t0 > budget_s - 180:
                break
            if it % 2500 == 0 and it > 0:
                T *= 0.91
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = min(T, 0.04) * rng.randn()
            ni, nj = hsa[i] + d, hsa[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            dc = _pairdc(hsa, i, j, d, n)
            nc5 = float((csa + dc).max() * 2.0 / n)
            delta = nc5 - c5sa
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                hsa[i], hsa[j] = ni, nj
                csa += dc
                c5sa = nc5
        hsa = _repair(hsa, n)
        c5 = _eval(hsa)
        if c5 < best_c5:
            best_c5, best_h = c5, hsa

    # Phase 4: Multi-scale fine pairing
    if time.time() - t0 < budget_s - 50:
        hfp = best_h.copy()
        cfp = np.correlate(hfp, 1.0 - hfp, mode='full')
        c5fp = float(cfp.max() * 2.0 / n)
        for scale in [0.03, 0.015, 0.007, 0.003, 0.001, 0.0004, 0.0001, 5e-5, 1e-5]:
            if time.time() - t0 > budget_s - 50:
                break
            nit = 25000 if scale > 0.002 else 40000
            hfp, cfp, c5fp = _pair_search(hfp, cfp, c5fp, rng, nit, scale, n)
        hfp = _repair(hfp, n)
        c5 = _eval(hfp)
        if c5 < best_c5:
            best_c5, best_h = c5, hfp

    # Phase 5: Multi-start with L-BFGS-B + pairing
    if time.time() - t0 < budget_s - 100:
        for r in range(10):
            if time.time() - t0 > budget_s - 100:
                break
            s = rng.randint(5)
            if s == 0:
                hs = _repair(best_h + 0.025 * rng.randn(n), n)
            elif s == 1:
                hs = _repair(1.0 - best_h + 0.015 * rng.randn(n), n)
            elif s == 2:
                hs = best_h.copy()
                for _ in range(rng.randint(2, 4)):
                    bb = rng.randint(3, max(4, n // 6))
                    st2 = rng.randint(0, n - bb)
                    hs[st2:st2 + bb] += 0.035 * rng.randn(bb)
                hs = _repair(hs, n)
            elif s == 3:
                f = rng.uniform(2.0, 5.0)
                hs = 0.5 + 0.18 * np.sin(np.arange(n) * np.pi * f / n)
                hs = _repair(hs, n)
            else:
                hs = 0.5 + 0.12 * np.cos(np.arange(n) * np.pi * rng.uniform(1, 4) / n)
                hs += 0.08 * np.cos(np.arange(n) * np.pi * rng.uniform(4, 8) / n)
                hs = _repair(hs, n)
            hm = _lbfgs(hs, 50000, 200)
            cm = np.correlate(hm, 1.0 - hm, mode='full')
            hm, cm, _ = _pair_search(hm, cm, float(cm.max() * 2.0 / n), rng, 20000, 0.005, n)
            hm = _repair(hm, n)
            c5 = _eval(hm)
            if c5 < best_c5:
                best_c5, best_h = c5, hm

    # Phase 6: Final L-BFGS-B polish + ultra-fine pairing
    if time.time() - t0 < budget_s - 8:
        hp = _lbfgs(best_h, 200000, 200)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h = c5, hp
        hp = _repair(best_h, n)
        cp = np.correlate(hp, 1.0 - hp, mode='full')
        hp, cp, _ = _pair_search(hp, cp, float(cp.max() * 2.0 / n), rng, 60000, 3e-5, n)
        hp, cp, _ = _pair_search(hp, cp, float(cp.max() * 2.0 / n), rng, 40000, 1e-5, n)
        hp = _repair(hp, n)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h = c5, hp

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
