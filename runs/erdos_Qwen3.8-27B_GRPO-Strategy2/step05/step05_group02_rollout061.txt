def _optimal_delta(c, a, h_i):
    """Find δ that minimizes max_k(c_k + δ*a_k) subject to 0 ≤ h_i+δ ≤ 1."""
    # Breakpoints where c_k + δ*a_k = c_m + δ*a_m
    # δ = (c_m - c_k) / (a_k - a_m)
    # Also boundaries: δ = -h_i and δ = 1 - h_i
    
    n_lags = len(c)
    candidates = [max(0.0, -h_i + 1e-15), min(1.0, 1.0 - h_i - 1e-15)]
    
    # Actually boundaries are: 0 ≤ h[i]+δ ≤ 1, so -h[i] ≤ δ ≤ 1-h[i]
    lo = -h_i
    hi = 1.0 - h_i
    
    if lo >= hi:
        return 0.0
    
    candidates = [lo, hi]
    
    # Find breakpoints
    for k in range(n_lags):
        for m in range(k+1, n_lags):
            denom = a[k] - a[m]
            if abs(denom) > 1e-12:
                d = (c[m] - c[k]) / denom
                if lo < d < hi:
                    candidates.append(d)
    
    # Evaluate at all candidates
    best_d = 0.0
    best_val = np.inf
    for d in candidates:
        val = np.max(c + d * a)
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d, best_val

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Scale and clip
    h = np.clip(h * target / s, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj_sum(h, n):
    """Project h to [0,1]^n with sum = n/2 using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _influence(h, i, n):
    """Compute a_k = (1-h[i-s_k])_valid - h[i+s_k]_valid for all k."""
    shifts = np.arange(n) - (n - 1)  # shifts[k] = k - (n-1) for k=0..2n-2
    a = np.zeros(2 * n - 1)
    for k, s in enumerate(shifts):
        # h[i] is in first factor when i = j + s, i.e., j = i - s
        ia = i - s
        if 0 <= ia < n:
            a[k] += (1.0 - h[ia])
        # h[i] is in second factor when i = j, so term is h[j+s]*(1-h[j])
        # change in (1-h[i]) = -δ, multiplied by h[i+s]
        ib = i + s
        if 0 <= ib < n:
            a[k] -= h[ib]

def _coord_step(h, i, c, n, dx):
    """Compute optimal delta for coordinate i given current correlation c."""
    a = _influence_all(h, i, n)
    # c[k] + delta * a[k] for all k
    # Find delta minimizing max_k(c[k] + delta*a[k]) subject to 0 ≤ h[i]+delta ≤ 1
    lo = -h[i]
    hi = 1.0 - h[i]
    if lo >= hi:
        return 0.0, np.max(c)
    
    # Find all breakpoints
    # c[k] + d*a[k] = c[m] + d*a[m] => d = (c[m]-c[k])/(a[k]-a[m])
    c_ext = c  # (2n-1,)
    a_ext = a  # (2n-1,)
    
    # Vectorized breakpoint computation
    diff_a = a_ext[:, None] - a_ext[None, :]
    diff_c = c_ext[None, :] - c_ext[:, None]
    valid = np.abs(diff_a) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        deltas = np.where(valid, diff_c / np.where(valid, diff_a, 1.0), 0.0)
    deltas = deltas[valid]
    deltas = deltas[np.isfinite(deltas) & (deltas > lo) & (deltas < hi)]
    
    candidates = np.concatenate([deltas, [lo, hi]])
    # Evaluate max at each candidate
    # vals[j] = max_k(c[k] + candidates[j]*a[k])
    vals = np.max(c_ext[None, :] + candidates[:, None] * a_ext[None, :], axis=1)
    best_idx = np.argmin(vals)
    best_d = candidates[best_idx]
    return best_d, vals[best_idx]

    ib = np.arange(n)[:, None] + shifts[None, :]

    vb = (ib >= 0) & (ib < n)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = np.clip(h * target / s, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    amt = min(diff, 1.0 - h[idx])
                    h[idx] += amt
                    diff -= amt
                else:
                    amt = min(-diff, h[idx])
                    h[idx] -= amt
                    diff += amt
                if abs(diff) < 1e-12:
                    break
    return h

def _all_influences(h, n):
    shifts = np.arange(2 * n - 1) - (n - 1)
    rows = np.arange(n)[:, None]
    ia = rows - shifts[None, :]
    ib = rows + shifts[None, :]
    va = (ia >= 0) & (ia < n)
    vb = (ib >= 0) & (ib < n)
    infl = np.zeros((n, 2 * n - 1))
    infl += np.where(va, 1.0 - h[np.clip(ia, 0, n - 1)], 0.0)
    infl -= np.where(vb, h[np.clip(ib, 0, n - 1)], 0.0)
    return infl

        return 0.0, np.max(c)

            idx3 = np.argmin(vals3)
            return ds3[idx3], vals3[idx3]
        return ds2[idx2], vals2[idx2]
    return ds[idx], vals[idx]

def _slsqp_lse(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - hv[i1 + sh])
                if sh > 0:
                    g[sh:] -= wk * hv[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - hv[:n - sp])
                g[:n - sp] -= wk * hv[sp:]
        return g
    def sf(x):
        return x.sum() - n * 0.5
    def sj(x):
        return np.ones(n)
