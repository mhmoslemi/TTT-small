```python
def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _coord_direction(h, i, n):
    """Compute d[k] = dc[k]/dh[i] for all correlation indices k."""
    d = np.zeros(2 * n - 1)
    # First factor: h[i] * (1-h[i+s]), valid when 0<=i+s<n
    # k = (n-1)+s, so s = k-(n-1)
    # s ranges: -i to n-1-i
    if i < n:
        s_start = -i
        s_end = n - 1 - i
        k_start = (n - 1) + s_start
        k_end = (n - 1) + s_end
        d[k_start:k_end+1] += (1.0 - h[i+s_start:i+s_end+1])
    # Second factor: h[i-s] * (1-h[i]), valid when 0<=i-s<n
    # s = i-j where j = i-s, need 0<=j<n => 0<=i-s<n => i-n<s<=i
    s_start2 = i - n + 1
    s_end2 = i
    k_start2 = (n - 1) + s_start2
    k_end2 = (n - 1) + s_end2
    d[k_start2:k_end2+1] -= h[i-s_start2:i-s_start2+(s_end2-s_start2+1)]
    return d

    """Coordinate descent with exact 1D optimization."""
    c = _corr(h)
    best_c5 = c.max()
    for sweep in range(n_sweeps):

            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:

                t_new = 0.5  # will be refined
            elif old > 1.0 - 1e-12:
                d = _coord_direction(np.clip(h,0,1), i, n)
                t_new = 0.5
            else:
                d = _coord_direction(h, i, n)
                # c(t) = c + (t - old) * d
                # C(t) = max(c + (t-old)*d), convex in t
                # Find min on [0,1]
                lo, hi = 0.0, 1.0
                base = c - old * d  # c(t) = base + t*d

                    if np.max(base + m1 * d) < np.max(base + m2 * d):
                        hi = m2
                    else:
                        lo = m1
                t_new = (lo + hi) / 2.0

                h[i] = t_new
                new_c5 = c.max()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                improved = True
        if not improved:
            break
    return h, c, best_c5

def _swap_optimize(h, c, n, rng, n_iters=100000):
    """Try swapping values between pairs of coordinates."""
    dx = 2.0 / n
    best_c5 = c.max() * dx
    for _ in range(n_iters):

        j = rng.randint(n)

        # Actually let's try: move amount t from i to j

        # Need 0<=h[i]+t<=1, 0<=h[j]-t<=1

        t_max_j = h[j]
        t_range = min(t_max_i, t_max_j)

            continue

        # Direction for h[j] decreasing by 1: -d_j

        d_j = _coord_direction(h, j, n)
        dc = d_i - d_j
        # Find best t in [0, t_range]... but also could be negative
        # Actually let's allow t in [-h[i], h[j]] (i.e., either direction)
        lo_t = -h[i]

        # 0 <= h[j]-t <= 1 => h[j]-1 <= t <= h[j]

        base = c  # c(0) = c (no change)

        # Convex in t
        lo, hi = lo_t, hi_t
        for _ in range(30):
            m1 = lo + (hi-lo)/3.0
            m2 = hi - (hi-lo)/3.0
            if np.max(base + m1*dc) < np.max(base + m2*dc):
                hi = m2
            else:
                lo = m1
        t_opt = (lo+hi)/2.0
        new_c = c + t_opt * dc
        new_c5 = new_c.max() * dx
        if new_c5 < best_c5 - 1e-12:
            h[i] += t_opt
            h[j] -= t_opt
            c = new_c
            best_c5 = new_c5
    return h, c, best_c5
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

                a = -a

            diff -= a

    """Find t in [lo,hi] minimizing max(base + t*d)."""

            break

        j = rng.randint(n)
        if i == j:
            continue

        hj = h[j]
        h[i] = 0.0

        h[i] = 1.0

        h[i] = hi
        h[j] = 0.0
        cj0 = _corr(h)
        h[j] = 1.0
```

```python
def _coord_dir(h, i, n):
    """Direction dc/dh[i]. Computed analytically in O(n)."""
    d = np.zeros(2 * n - 1)
    # First factor: h[i] * (1-h[i+s]), valid when 0 <= i+s < n
    # k = (n-1) + s, so k ranges from (n-1)+(-i) to (n-1)+(n-1-i)
    # i.e., k from n-1-i to 2(n-1)-i
    # d[k] += (1 - h[i+s]) = (1 - h[k-(n-1)+i]) = (1 - h[k - (n-1-i)])
    
    # Let u = k - (n-1-i), then when k = n-1-i, u=0; when k=2(n-1)-i, u=n-1
    # h[k-(n-1)+i] = h[k-(n-1-i)] = h[u]
    # So d[k] += (1-h[k-(n-1-i)]) for k in [n-1-i, 2(n-1)-i]
    
    k_start = n - 1 - i
    k_end = 2 * (n - 1) - i
    idx = np.arange(k_start, k_end + 1)
    d[idx] += 1.0 - h[idx - (n - 1 - i)]
    
    # Second factor: h[i-s] * (1-h[i]), valid when 0 <= i-s < n
    # s = k-(n-1), so i-s = i-k+(n-1) = (n-1+i)-k
    # Let v = (n-1+i)-k, then v ranges from (n-1+i)-(n-1+i)=0 to (n-1+i)-0=n-1
    # Wait: k ranges from 0 to 2(n-1), s = k-(n-1) ranges from -(n-1) to (n-1)
    # i-s = i - (k-(n-1)) = i + (n-1) - k = (n-1+i) - k
    # Valid: 0 <= (n-1+i)-k < n => (n-1+i)-n < k <= n-1+i => i-1 < k <= n-1+i
    # So k from i to n-1+i (integer: k >= i, k <= n-1+i, also k >= 0 and k <= 2n-2)
    
    k_start2 = i  # need k > i-1, so k >= i
    k_end2 = n - 1 + i  # need k <= n-1+i, also k <= 2n-2
    k_end2 = min(k_end2, 2 * n - 2)
    k_start2 = max(k_start2, 0)
    
    idx2 = np.arange(k_start2, k_end2 + 1)
    # h[i-s] = h[(n-1+i)-k]
    h_vals = h[(n - 1 + i) - idx2]
    d[idx2] -= h_vals
    
    return d
```

```python
def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = -min(-diff, h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

def _coord_dir(h, i, n):
    d = np.zeros(2 * n - 1)
    ks = n - 1 - i
    ke = 2 * (n - 1) - i
    idx = np.arange(ks, ke + 1)
    d[idx] += 1.0 - h[idx - ks]

    ke2 = min(n - 1 + i, 2 * n - 2)
    idx2 = np.arange(ks2, ke2 + 1)
    d[idx2] -= h[(n - 1 + i) - idx2]
    return d

            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

            h[i] = t_new
            c += (t_new - old) * d
    return h, c

def _block_swap(h, c, n, rng, n_iters, block_size=3, deadline=np.inf):
    best_c5 = c.max()

        if abs(diff_sum) > 1e-12:
            m = np.argmax(1.0 - new_vals) if diff_sum > 0 else np.argmax(new_vals)
            adj = min(abs(diff_sum), 1.0 - new_vals[m]) if diff_sum > 0 else min(abs(diff_sum), new_vals[m])
            new_vals[m] += adj if diff_sum > 0 else -adj
```
