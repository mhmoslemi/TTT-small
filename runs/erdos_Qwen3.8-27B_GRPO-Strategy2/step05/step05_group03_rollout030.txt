def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d/n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

    b[idx[:n_ones]] = 1.0
    return b

        h[j] = 1.0

    n = len(h)

    bds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = _corr(np.clip(x, 0, 1))
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    
    def eq(x):
        return x.sum() - tgt

                     constraints=[{'type': 'eq', 'fun': eq}],

        return hc, _c5(hc, dx)
    except Exception:
