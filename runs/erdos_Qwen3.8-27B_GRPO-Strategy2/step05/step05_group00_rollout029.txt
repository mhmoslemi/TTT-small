    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -200, 200)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            r = n - sh
            g[:r] += wk * (1.0 - h[sh:sh + r])
            if sh > 0:
                g[sh:] -= wk * h[:r]
            else:
                g[:r] -= wk * h[:r]
        else:
            p = -sh
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] -= wk * h[p:]
    return g

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _pg_fit(h, n, t0, max_s, budget_fn):
    v = np.zeros(n)
    beta = 0.85
    lr = 0.8 / n
    alpha = 2000.0
    hc = h.copy()
    c5 = _eval(hc)
    cnt = 0
    for it in range(20000):
        if budget_fn(t0, max_s):
            break
        cnt += 1
        g = _grad(hc, alpha)
        v = beta * v - lr * g
        h_try = _proj(hc + v, n)
        c5_try = _eval(h_try)
        if c5_try < c5:
            c5 = c5_try
            hc = h_try
            lr = min(lr * 1.02, 5.0 / n)
        else:
            v = 0.5 * v
            lr *= 0.7
            h_try = _proj(hc - 0.1 * g, n)
            c5_try2 = _eval(h_try)
            if c5_try2 < c5:
                hc = h_try
                c5 = c5_try2
                v = -0.1 * g
            else:
                beta *= 0.99
        if cnt % 500 == 0:
            alpha *= 1.3
    return hc, c5

def _binary_local(h, n, rng, t0, max_s, budget_fn):
    hb = (h > 0.5).astype(float)
    hb = _repair(hb, n)
    c5b = _eval(hb)
    ones = np.where(hb > 0.5)[0]
    zeros = np.where(hb < 0.5)[0]
    T = 0.01
    for it in range(50000):
        if budget_fn(t0, max_s):
            break
        T *= 0.9997
        if T < 0.001:
            T = 0.001
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        hb[a], hb[b] = 0.0, 1.0
        c5_try = _eval(hb)
        if c5_try < c5b or rng.rand() < np.exp(-(c5_try - c5b) / T):
            c5b = c5_try
            ones = np.where(hb > 0.5)[0]
            zeros = np.where(hb < 0.5)[0]
        else:
            hb[a], hb[b] = 1.0, 0.0
    return hb, c5b

def _binary_hill(h, n, rng, t0, max_s, budget_fn):
    hb = (h > 0.5).astype(float)
    hb = _repair(hb, n)
    c5b = _eval(hb)
    for it in range(30000):
        if budget_fn(t0, max_s):
            break
        improved = False
        for _ in range(50):
            a = rng.choice(np.where(hb > 0.5)[0])
            b = rng.choice(np.where(hb < 0.5)[0])
            hb[a], hb[b] = 0.0, 1.0
            c5_try = _eval(hb)
            if c5_try < c5b - 1e-14:
                c5b = c5_try
                improved = True
            else:
                hb[a], hb[b] = 1.0, 0.0
        if not improved:
            break
    return hb, c5b

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old := h, period=None) if False else np.interp(xn, xo, h)

    t0 = time.time()
    rng = np.random.RandomState(seed)

        return time.time() - base > limit

        h4 = _repair(_resample(best_h, n4), n4)

            best_n = n4
        if not budget_fn(t0, budget_s - 150):

    if c5_5 < best_c5:
