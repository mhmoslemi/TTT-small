            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    def _time_left(margin=10):
        return time.time() - t0 < budget_s - margin

    # Phase 1: PGD with alpha annealing on original n
    for alpha in [50, 200, 1000, 5000, 20000, 80000]:
        if not _time_left(120):
            break
        n_steps = 800 if alpha < 1000 else 400
        h, c5 = _pgd(best_h, alpha, n_steps, t0, budget_s - 120)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 2: Multi-restart with perturbations
    for _rep in range(4):
        if not _time_left(180):
            break
        h_p = _perturb(best_h, rng, 0.05)
        for alpha in [500, 5000, 30000]:
            if not _time_left(180):
                break
            h_p, c5 = _pgd(h_p, alpha, 300, t0, budget_s - 180)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p

    # Phase 3: Periodic reduction
    for T in [7, 10, 14, 20, 22, 28, 30, 35, 44, 55, 70, 77, 110]:
        if not _time_left(250):
            break
        h_p, c5 = _periodic_opt(best_h, T, t0, budget_s - 250)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_p, len(h_p)

    # Phase 4: Try different n values
    for dn in [-30, -20, -10, 10, 20, 30, 50]:
        if not _time_left(300):
            break
        n_try = best_n + dn
        if n_try < 60:
            continue
        h_r = _resample(best_h, n_try)
        h_r = _repair(h_r, n_try)
        for alpha in [1000, 10000, 50000]:
            if not _time_left(300):
                break
            h_r, c5 = _pgd(h_r, alpha, 300, t0, budget_s - 300)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r, n_try

    # Phase 5: Complement trick
    if _time_left(350):
        h_c = _repair(1.0 - best_h, best_n)
        for alpha in [1000, 10000, 50000]:
            if not _time_left(350):
                break
            h_c, c5 = _pgd(h_c, alpha, 400, t0, budget_s - 350)
            if c5 < best_c5:
                best_c5, best_h = c5, h_c

    # Phase 6: Aggressive PGD with very high alpha
    for alpha in [100000, 300000, 1000000]:
        if not _time_left(60):
            break
        h, c5 = _pgd(best_h, alpha, 500, t0, budget_s - 60)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 7: Coordinate descent polish
    if _time_left(20):
        h, c5 = _coord_descent(best_h, 20000, t0, budget_s - 5, rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Final repair and evaluate
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
