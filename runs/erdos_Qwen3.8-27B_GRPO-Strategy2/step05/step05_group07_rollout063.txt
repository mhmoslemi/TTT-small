- The optimal δ makes the max as small as possible

For the top active lags, we want δ*(g_i[k]-g_j[k]) < 0, i.e., δ and (g_i[k]-g_j[k]) have opposite signs.

If for all active k, (g_i[k]-g_j[k]) has the same sign, we can choose δ to reduce all active lags simultaneously.

```python
def _pair_optimal(h, i, j, c, topm):
    """Find optimal δ for h[i]+=δ, h[j]-=δ to reduce max of top-m lags"""
    n = len(h)
    top = np.argsort(c)[::-1][:topm]
    # For each active lag k, the change is δ * d_k where d_k = g_i[k] - g_j[k]
    # We want to minimize max_k (c[k] + δ*d_k) over active k
    # Also need 0 ≤ h[i]+δ ≤ 1 and 0 ≤ h[j]-δ ≤ 1
    
    # Compute d_k for active lags
    di = _grad_single_vec(h, i)  # gradient of all c_k wrt h[i]
    dj = _grad_single_vec(h, j)
    d = (di - dj)[top]  # change in c_k for active k per unit δ
    
    # Also need to consider non-active lags (they might become active)
    # For simplicity, just consider active lags
    
    # Bounds on δ
    delta_max = min(1.0 - h[i], h[j])
    delta_min = max(-h[i], -(1.0 - h[j]))
    
    # Minimize max_k (c[top][k] + δ*d[k]) for δ in [delta_min, delta_max]
    # This is a piecewise linear function minimization
    # The minimum is at a breakpoint or at the bounds
    
    c_top = c[top]
    # Breakpoints: where c_top[k] + δ*d[k] = c_top[m] + δ*d[m] for some k,m
    # i.e., δ = (c_top[m] - c_top[k]) / (d[k] - d[m])
    
    candidates = [delta_min, delta_max, 0.0]
    for a in range(len(top)):
        for b in range(a+1, len(top)):
            if abs(d[a] - d[b]) > 1e-15:
                δ = (c_top[b] - c_top[a]) / (d[a] - d[b])
                if delta_min < δ < delta_max:
                    candidates.append(δ)
    
    best_δ = 0.0
    best_val = np.max(c_top)
    for δ in candidates:
        val = np.max(c_top + δ * d)
        if val < best_val:
            best_val = val
            best_δ = δ
    return best_δ
```

            hi = mid

                h3, f3 = _subgrad(h3, 2000, 4, tgt3, 0.001)
                h3, f3 = _pair_descent(h3, 100, 2000, 5, 789)
                if f3 < best_f:
                    best_h, best_f = h3.copy(), f3

    # Phase 13: Final subgradient polish at best resolution
    nf = len(best_h)
    if _el() < _BUDGET - 5:
        hn, fn = _subgrad(best_h, 3000, 3, nf / 2.0, 0.0005)
        if fn < best_f:
            best_h, best_f = hn.copy(), fn

    # Phase 14: Final pair polish
    if _el() < _BUDGET - 3:
        hn, fn = _pair_descent(best_h, 50, 1000, 4, 321)
        if fn < best_f:
            best_h, best_f = hn.copy(), fn

    best_h = _proj(best_h, len(best_h) / 2.0)
    best_f = _eval(best_h)
    return (best_h, best_f, len(best_h))
