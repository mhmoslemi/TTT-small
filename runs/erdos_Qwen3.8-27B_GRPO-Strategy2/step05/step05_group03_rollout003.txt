    n = len(h)

    for w, ki in zip(weights, top_idx):

def _adam_opt(h, dx, t_end, K=15, alpha_init=100, alpha_end=50000, lr=0.03):

        w_full = np.exp(s_arr)
        w_sum = w_full.sum()
        if w_sum < 1e-300:
            break
        
        # Determine K adaptively
        K_eff = max(K, int(np.sum(w_full / w_sum > 1.0 / n)))
        K_eff = min(K_eff, 2 * n)
        
        top_idx = np.argsort(c)[::-1][:K_eff]
        wt = w_full[top_idx]
        wt /= wt.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(wt, top_idx):
            grad += wi * _lag_grad(h, ki)
        
        # Project out the mean (to preserve sum)
        grad -= grad.mean()
        
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad**2
        m_hat = m_v / (1 - b1**t)
        v_hat = v_v / (1 - b2**t)

            h = h_new

            cn2 = _c5(h_half, dx)
            if cn2 < best:
                best = cn2
                h = h_half
            # Otherwise keep h
    
    return h, best

def _slsqp(h, dx, alpha, maxiter=300):

    return np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14: break

            if d > 0:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a

def _lag_grad(h, ki):
    n = len(h)
    s = abs(int(ki - (n-1)))
    if s == 0: return np.zeros(n)
    m = n - s
    g = np.zeros(n)
    g[:m] += 1.0 - h[s:]
    g[s:] -= h[:m]
    return g
