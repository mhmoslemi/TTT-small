from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(k, h):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        if s > 0:
            g[:n-s] -= h[s:]
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:])
        g[n-sp:] -= h[:sp]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = _corrs(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corrs(h) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    w = np.exp(s - s.max())

    for k in range(2 * n - 1):

        g += w[k] * _grad_lag(k, h)

    target = n * 0.5
    h = h.copy()

    f = _eval(h)
    step = step0
    t_end = time.time() + budget_left
    best_local = h.copy()
    best_f_local = f
    stall = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corrs(h)
        # Top-k selection with threshold
        thresh = c.mean() + 0.3 * c.std()
        top_idx = np.where(c > thresh)[0]
        if len(top_idx) < topm:
            top_idx = np.argsort(c)[::-1][:topm]

            top_idx = np.argsort(c)[::-1][:topm * 3]

        f_new = _eval(h_new)

            h = h_new
            if f < best_f_local:
                best_local = h.copy()
                best_f_local = f
                stall = 0
            else:
                stall += 1
        else:
            step *= 0.5
            if step < 1e-6:
                stall += 1
        
        if stall > 50:
            # Annealing restart
            h = _repair(h + 0.01 * np.random.randn(n), n)
            v = np.zeros(n)
            step = step0
            stall = 0

        d = scale * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i], h[j] = h[i] + d, h[j] - d
            fn = _eval(h)
            if fn < f - 1e-14:
                f = fn
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, f
