def _grad_k(h, k, n):

    sh = (n - 1) - k

    if sh >= 0:

        g[i1] += (1.0 - h[i1 + sh])

            g[i2] -= h[i2 - sh]

        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

    g_prev = None

    stall = 0
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        c = _eval_c(x)
        idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(x, k, n)
        
        # BB step
        if g_prev is not None:
            s_vec = x - x_prev
            y_vec = g - g_prev
            sy = s_vec @ y_vec
            yy = y_vec @ y_vec
            if yy > 1e-10 and sy > 1e-10:
                step = (s_vec @ s_vec) / sy
            elif yy > 1e-10:
                step = np.abs(x @ g) / yy
            step = float(np.clip(step, 1e-4, 0.02))
        
        x_new = _proj_to_sum(x - step * g, n * 0.5)
        x_prev = x.copy()
        g_prev = g.copy()
        x = x_new

        if c5n < c5c - 1e-13:
            c5c = c5n
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

    h = np.clip(h, 0.0, 1.0).copy()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _pgd(h, max_iters, top_k, t_end):
    n = len(h)
    c5c = _eval(h)
    x = h.copy()
    x_prev = x.copy()
    g_prev = None
    step = 0.005
    stall = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        c = _eval_c(x)
        idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(x, k, n)
        if g_prev is not None and it > 5:
            s_vec = x - x_prev
            y_vec = g - g_prev
            sy = s_vec @ y_vec
            yy = y_vec @ y_vec
            if yy > 1e-10 and sy > 1e-10:
                step = (s_vec @ s_vec) / sy
            elif yy > 1e-10:
                step = max(abs(x @ g), 1e-6) / yy
            step = float(np.clip(step, 5e-5, 0.015))
        x_new = _proj_to_sum(x - step * g, n * 0.5)
        x_prev = x.copy()
        g_prev = g.copy()
        x = x_new
        c5n = _eval(x)
        if c5n < c5c - 1e-13:
            c5c = c5n
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break
    return x, c5c

def _pairwise(h, n_iter, scale, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _structured_pair(h, n_rounds, rng, t_end):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        a = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = (n - 1) - k
            ci = 0.0
            cj = 0.0
            if sh >= 0:
                if i <= n - 1 - sh:
                    ci += (1.0 - h[i + sh])
                if 0 <= i - sh and i - sh <= n - 1 - sh:
                    ci -= h[i - sh]
                if j <= n - 1 - sh:
                    cj += (1.0 - h[j + sh])
                if 0 <= j - sh and j - sh <= n - 1 - sh:
                    cj -= h[j - sh]
            else:
                sp = -sh
                if sp <= i < n:
                    ci += (1.0 - h[i - sp])
                if 0 <= i + sp < n - sp:
                    ci -= h[i + sp]
                if sp <= j < n:
                    cj += (1.0 - h[j - sp])
                if 0 <= j + sp < n - sp:
                    cj -= h[j + sp]
            a[k] = ci - cj
        best_d = 0.0
        best_v = np.max(c0)
        dgrid = np.linspace(dlo, dhi, 17)
        for d in dgrid:
            v = np.max(c0 + dx * d * a)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-12:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i] -= best_d
                h[j] += best_d
    return h, c5c

def _upsample(h, factor):
    return np.repeat(h, factor)

def _slsqp_smooth(h, alpha, maxiter, t_end):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    bnds = [(0.0, 1.0)] * n
    cfunc = lambda x: x.sum() - n * 0.5
    cjac = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': cfunc, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=obj_jac, args=(),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or _eval(res.x) < _eval(h):
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Fast PGD at n0 with adaptive top-k
    for tk in [3, 5, 8, 12]:
        if time.time() > t_end - 500:
            break
        h_new, c5 = _pgd(best_h.copy(), 15000, tk, t_end - 450)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Structured pairwise at n0
    if time.time() < t_end - 400:
        h_new, c5 = _structured_pair(best_h.copy(), 3000, rng, t_end - 380)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: SLSQP polish at n0
    if time.time() < t_end - 350:
        h_new = _slsqp_smooth(best_h.copy(), 200000, 1500, t_end - 340)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 4: Upsample to 2x and optimize
    if time.time() < t_end - 350:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for tk in [3, 5, 8]:
            if time.time() > t_end - 200:
                break
            h_hi, c5 = _pgd(h_hi, 15000, tk, t_end - 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if time.time() < t_end - 150:
            h_new, c5 = _structured_pair(best_h.copy(), 2000, rng, t_end - 140)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(best_h)
        if time.time() < t_end - 100:
            h_new = _slsqp_smooth(best_h.copy(), 500000, 1000, t_end - 95)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase 5: Multi-start exploration
    n_cur = best_n
    if time.time() < t_end - 150:
        for r in range(15):
            if time.time() > t_end - 120:
                break
            strat = r % 5
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(np.clip(1.0 - best_h + 0.01 * rng.randn(n_cur), 0, 1), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                t_arr = np.arange(n_cur)
                for _f in range(3):
                    f = rng.randint(1, n_cur // 3 + 1)
                    hp += 0.004 * rng.randn() * np.sin(2 * np.pi * f * t_arr / n_cur)
                hp = _repair(hp, n_cur)
            elif strat == 3:
                hp = _repair(np.clip(best_h + 0.04 * (rng.rand(n_cur) - 0.5), 0, 1), n_cur)
            else:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, n_cur // 8))
                st = rng.randint(0, n_cur - bs)
                hp[st:st + bs] += 0.02 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            h_new, c5 = _pgd(hp, 5000, 5, t_end - 120)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase 6: Final PGD polish with small top-k
    if time.time() < t_end - 40:
        h_new, c5 = _pgd(best_h.copy(), 30000, 3, t_end - 40)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 7: Pairwise refinement at multiple scales
    if time.time() < t_end - 10:
        for sc in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() > t_end - 5:
                break
            h_new, c5 = _pairwise(best_h.copy(), 3000, sc, rng, t_end - 5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 8: Final structured pair
    if time.time() < t_end - 3:
        h_new, c5 = _structured_pair(best_h.copy(), 2000, rng, t_end - 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
