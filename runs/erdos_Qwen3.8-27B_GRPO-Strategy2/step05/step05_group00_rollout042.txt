def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_c(h, n):
    """Compute all shift gradients, stored as (shift, magnitude) pairs for sparse use."""
    c = np.correlate(h, 1.0 - h, mode='full')
    return c

    """Compute gradient as sum of w[k] * grad_c[k] for given weights w."""
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:n]
    return g

def _projected_descent(h, n, alpha, lr_init, n_iter, momentum=0.9, top_k=5):
    """Projected subgradient descent with momentum on top-k shifts."""
    h = h.copy()
    v = np.zeros(n)
    c5 = _eval(h)
    dx = 2.0 / n
    lr = lr_init
    
    for it in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        # Softmax weights on top-k
        top_idx = np.argsort(c)[-top_k:]
        c_top = c[top_idx]
        s = alpha * (c_top - c_top.max())
        w_top = np.exp(s)
        w_top /= w_top.sum()
        
        w = np.zeros(2*n-1)
        w[top_idx] = w_top
        
        g = _sparse_grad(h, w, n)
        g *= dx
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = h + v
        
        # Project: clip and fix sum
        h_new = _repair(h_new, n)
        
        c5_new = _eval(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            # Decay learning rate slowly
            lr *= 0.999
        else:
            # Revert
            h = h - v  # undo
            v *= 0.5
            lr *= 0.95
        
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, n, top_k, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-top_k:]
    c_top = c[top_idx]
    s = alpha * (c_top - c_top.max())
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for j, idx in enumerate(top_idx):
        sh = (n - 1) - int(idx)
        wk = w[j]
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:n]
    return g

def _proj_descent(h, n, alpha, lr0, n_iter, momentum, top_k, time_limit=None, t0=None):
    h = h.copy().astype(float)
    v = np.zeros(n)
    best_c5 = _eval(h)
    best_h = h.copy()
    dx = 2.0 / n
    target = n * 0.5
    lr = lr0
    patience = 0
    for it in range(n_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c5 = c.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        g = _topk_grad(h, n, top_k, alpha) * dx
        sum_dev = h.sum() - target
        g += 0.5 * sum_dev / n
        v = momentum * v - lr * g
        h_trial = np.clip(h + v, 0.0, 1.0)
        c5_trial = float(np.max(np.correlate(h_trial, 1.0 - h_trial, mode='full')) * dx)
        if c5_trial < c5 + 1e-12:
            h = h_trial
            lr = min(lr * 1.001, 0.1)
        else:
            v *= 0.5
            lr *= 0.85
        if patience > 200:
            v *= 0.3
            patience = 0
    return best_h, best_c5

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = _topk_grad(x, n, len(w), 1.0)
        for k in range(len(w)):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * 2.0 / n
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:n])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:n]
        return g
    def _sum_eq(x):
        return x.sum() - n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
