import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _interp_to(h, n_new):
    return np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h)), h)

def _directed_local(h, dx, rng, t_end, K=8, max_iter=50000):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    it = 0
    scale = 0.02
    while time.time() < t_end and it < max_iter:
        c = _corr(h)
        top = np.argsort(c)[::-1][:K]
        # Build combined gradient
        G = np.zeros(n)
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            gs = np.zeros(n)
            gs[:m] += (1.0 - h[s:s+m])
            gs[s:] -= h[:m]
            G += gs
        # Greedy: find the best swap
        # Positions to decrease: high G
        # Positions to increase: low G
        hi_idx = np.argpartition(G, -20)[-20:]
        lo_idx = np.argsort(G)[:20]
        found = False
        for _ in range(30):
            i = lo_idx[rng.randint(20)]
            j = hi_idx[rng.randint(20)]
            if i == j:
                continue
            d = scale * (0.5 + rng.random())
            h[j] -= d; h[i] += d
            if 0 <= h[j] and h[i] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                    h = _repair(h, tgt)
                    found = True
                else:
                    h[j] += d; h[i] -= d
            else:
                h[j] += d; h[i] -= d
        if not found:
            scale *= 0.7
            if scale < 0.001:
                break
        it += 1
    return h, best

def _slsqp_smooth(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        s -= s.max()
        return m + float(np.sum(np.exp(s))) / alpha
    def grad(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        s -= s.max()
        e = np.exp(s)
        weights = e / e.sum()
        n_c = len(c)
        g = np.zeros(n)
        for ki in range(n_c):
            if weights[ki] < 1e-10:
                continue
            s_lag = abs(int(ki - (n - 1)))
            if s_lag == 0:
                m2 = n
                g[:m2] += weights[ki] * (1.0 - 2.0 * h[:m2])
            else:
                m2 = n - s_lag
                g[:m2] += weights[ki] * (1.0 - h[s_lag:s_lag+m2])
                g[s_lag:] -= weights[ki] * h[:m2]
        return g * (dx if False else 1.0)  # we optimize c, not c*dx
    bds = [(0.0, 1.0)] * n
    try:

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:

                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:

                h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

    best_h, best_c5 = _directed_local(best_h, 2.0 / nb, rng, remaining, K=8)

    if budget_s - (time.time() - t0) > 12:
        h, c = _do_slsqp(best_h, 50000, 150)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = _repair(best_h, len(best_h) / 2.0)
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
