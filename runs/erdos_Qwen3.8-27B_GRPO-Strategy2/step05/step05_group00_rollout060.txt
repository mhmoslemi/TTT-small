def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    return h

Strategy: use a "continuation" approach - start with low alpha, gradually increase, using the previous solution as initialization. This is what the previous program does, but let me make it more aggressive.

Let me also try: after the main optimization, do a "worst-shift" targeted descent. Identify the k values where c_k is close to the max, and do a line search to reduce specifically those.

Here's another idea: the problem is equivalent to maximizing min_k Aut(k) where Aut(k) = sum_i h[i]*h[i+k]*dx. This is a "maximin" problem. The antiderivative of such problems is usually at a point where multiple constraints are active (multiple k values achieving the same minimum). We can target this: find the solution where the bottom few autocorrelation values are as equal as possible.

Let me code a more targeted approach:

def _targeted_descent(h, n, top_k=5, max_rounds=100):
    """Focus on the worst shifts and try to balance them."""
    c5_current = _eval(h)
    for _ in range(max_rounds):
        ov = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
        # Find the worst shifts (highest overlap)
        worst_idx = np.argsort(ov)[-top_k:]
        # Compute gradient only from these shifts
        g = np.zeros(n)
        for idx in worst_idx:
            sh = idx - (n-1)
            if sh >= 0:
                g[sh:] += (1.0 - h[:n-sh])
                g[:n-sh] -= h[sh:]
            else:
                sp = -sh
                g[:n-sp] += (1.0 - h[sp:])
                g[sp:] -= h[:n-sp]
        g *= (2.0/n)
        # Project gradient to maintain sum constraint
        g -= g.mean()
        # Line search
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _repair(h - step * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5_current - 1e-14:
                h = h_new
                c5_current = c5_new
                break
    return h, c5_current

import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    val += pen * (x.sum() - n * 0.5) ** 2
    return val

def _grad(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g

def _lbfgs(h, alpha, pen, maxiter=3000):
    n = len(h)
    try:
        res = fmin_l_bfgs_b(_obj, h, fprime=_grad, args=(alpha, pen),
                            bounds=[(0.0, 1.0)] * n, maxiter=maxiter,
                            iprint=0, pgtol=1e-14, maxls=50)
        return _repair(res[0], n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _targeted_descent(h, n, top_k=3, max_rounds=200):
    c5c = _eval(h)
    for _ in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        worst = np.argsort(ov)[-top_k:]
        g = np.zeros(n)
        for idx in worst:
            sh = idx - (n - 1)
            if sh >= 0:
                g[sh:] += (1.0 - h[:n - sh])
                g[:n - sh] -= h[sh:]
            else:
                sp = -sh
                g[:n - sp] += (1.0 - h[sp:])
                g[sp:] -= h[:n - sp]
        g *= (2.0 / n) / top_k
        g -= g.mean()
        moved = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h, c5c = h_new, c5n
                moved = True
                break
        if not moved:
            break
    return h, c5c

def _binary_ls(p, T, max_iters=3000):
    best_min = -1
    ac0 = np.zeros(T, dtype=float)
    for r in range(1, T):
        ac0[r] = sum(p[j] * p[(j + r) % T] for j in range(T))
    best_min = min(ac0[1:])
    for _ in range(max_iters):
        improved = False
        for i in range(T):
            p[i] = 1 - p[i]
            if p.sum() == T // 2:
                mn = -1
                for r in range(1, T):
                    ac = sum(p[j] * p[(j + r) % T] for j in range(T))
                    mn = max(mn, -1) if mn < 0 else mn
                    if ac < mn:
                        mn = ac
                if mn > best_min:
                    best_min = mn
                    improved = True
                else:
                    p[i] = 1 - p[i]
            else:
                p[i] = 1 - p[i]
        if not improved:
            break
    return p, best_min

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Continuation L-BFGS-B
    for alpha in [30, 100, 500, 2000, 10000, 50000]:

    # Phase 2: Aggressive perturbation + re-optimize
    for trial in range(25):

        scale = 0.04 * (0.85 ** trial)
        h_p = _repair(best_h + scale * rng.randn(best_n), best_n)

    # Phase 3: Binary periodic patterns as seeds
    if time.time() - t0 < budget_s - 350:

            if time.time() - t0 > budget_s - 200:

            h_r = _lbfgs(h_r, alpha, 1000.0, maxiter=2500)

        if time.time() - t0 > budget_s - 25:

        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin.copy()
            best_n = len(h_fin)

    # Phase 7: Fine coordinate descent
    if time.time() - t0 < budget_s - 3:
        nc = best_n
        h_cd = best_h.copy()
        c5_cd = _eval(h_cd)
        for scale in [0.002, 0.0005, 0.0001]:
            if time.time() - t0 > budget_s - 3:
                break
            for _it in range(60000):
                if time.time() - t0 > budget_s - 3:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h_cd[i] + d, h_cd[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                if c5n < c5_cd - 1e-14:
                    c5_cd = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h_cd.copy()
                else:
                    h_cd[i], h_cd[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
