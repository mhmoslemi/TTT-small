import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    cinit = _corr(h)
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _binary_search(h, dx, rng, t_end):

    n = len(h)
    tgt = n // 2
    hb = (h > 0.5).astype(float)
    ones = np.where(hb == 1)[0]
    if len(ones) > tgt:
        for i in rng.choice(len(ones), len(ones) - tgt, replace=False):
            hb[ones[i]] = 0
    elif len(ones) < tgt:
        zeros = np.where(hb == 0)[0]
        for i in rng.choice(len(zeros), tgt - len(ones), replace=False):
            hb[zeros[i]] = 1
    best_c = _c5(hb, dx)
    best_h = hb.copy()
    T = 0.5
    while time.time() < t_end:
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        a, b = rng.choice(ones), rng.choice(zeros)
        hb[a], hb[b] = 0, 1
        c = _c5(hb, dx)
        if c < best_c or rng.random() < np.exp((best_c - c) / T * 100):
            if c < best_c:
                best_c = c
                best_h = hb.copy()
            T *= 0.999
        else:
            hb[a], hb[b] = 1, 0
        T = max(T, 0.01)
    return best_h, best_c

def _targeted(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:3]

            m = n - s

                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:

                    if cn < best:
                        best = cn
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
        for _ in range(6):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:

            d = rng.uniform(-0.01, 0.01)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def _coord_descent(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.003, 0.001, 0.0003, 1e-4, 3e-5, 1e-5]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

                c = _c5(hc, dx)
                if c < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

class _QP:
    def __init__(self, alpha, tgt):
        self.alpha = alpha
        self.tgt = tgt
    def obj(self, x):
        c = _corr(x)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -300, 300)
        return m + float(np.log(np.sum(np.exp(s)))) / self.alpha
    def eq(self, x):
        return x.sum() - self.tgt

def _slsqp(h, dx, alpha, maxiter=250):
    n = len(h)
    q = _QP(alpha, n / 2.0)
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(q.obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': q.eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, n / 2.0)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_sweep(h, dx, t_end, alphas, mi=250):
    best_h, best_c = h.copy(), _c5(h, dx)
    for a in alphas:
        if time.time() > t_end:
            break
        for _ in range(2):
            if time.time() > t_end:
                break
            hc, c = _slsqp(best_h, dx, a, mi)
            if c < best_c:
                best_h, best_c = hc, c
    return best_h, best_c

def _binary_ls(h, dx, rng, t_end, max_it=300000):

    hb = (h > 0.5).astype(float)
    ones = np.where(hb == 1)[0]

            hb[ones[i]] = 0
    elif len(ones) < tgt:
        z = np.where(hb == 0)[0]
        for i in rng.choice(len(z), tgt - len(ones), replace=False):
            hb[z[i]] = 1
    best_c = _c5(hb, dx)
    best_h = hb.copy()
    it, T = 0, 0.4
    while time.time() < t_end and it < max_it:
        it += 1
        ones = np.where(hb == 1)[0]
        z = np.where(hb == 0)[0]
        a, b = int(rng.choice(ones)), int(rng.choice(z))
        hb[a], hb[b] = 0.0, 1.0

        else:
            hb[a], hb[b] = 1.0, 0.0
    return best_h, best_c

    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
