    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    diff -= a

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _lse_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g * (2.0 / n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _to_binary(h, n):
    hb = np.where(h > 0.5, 1.0, 0.0)
    target = n // 2
    cnt = int(hb.sum())
    if cnt > target:
        one_idx = np.where(hb == 1)[0]
        perm = np.argsort(h[one_idx])[:cnt - target]
        hb[one_idx[perm]] = 0
    elif cnt < target:
        zero_idx = np.where(hb == 0)[0]
        perm = np.argsort(-h[zero_idx])[:target - cnt]
        hb[zero_idx[perm]] = 1
    return hb

def _sa(n, h_init, rng, t_end):
    h = h_init.copy().astype(np.int8)
    c5c = float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)
    best = h.copy()
    best_c5 = c5c
    T = 0.4
    cool = 0.99997
    no_improve = 0
    while time.time() < t_end:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i] = 1
        h[j] = 0
        c5n = float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)
        d = c5n - c5c
        if d < 0 or (T > 0.001 and rng.rand() < np.exp(-d / T)):
            c5c = c5n
            if c5c < best_c5:
                best_c5 = c5c
                best = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 0
            h[j] = 1
            no_improve += 1
        if no_improve > 3000:
            T = max(T * 2.5, 0.01)
            no_improve = 0
        T = max(T * cool, 0.0005)
    return best, best_c5

def _det_refine(h, t_end, max_rounds):
    n = len(h)
    h = h.copy().astype(np.int8)
    c5c = float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)
    best = h.copy()
    best_c5 = c5c
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in zeros:
            if time.time() > t_end:
                break
            for j in ones:
                h[i] = 1
                h[j] = 0
                c5n = float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    improved = True
                    if c5c < best_c5:
                        best_c5 = c5c
                        best = h.copy()
                    break
                h[i] = 0
                h[j] = 1
        if not improved:
            break
    return best, best_c5

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP from initial (25s)
    for alpha in [5000, 50000, 500000]:
        if time.time() - t0 > 25:
            break
        h_new = _slsqp(h, alpha, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    # Phase 2: Binary SA at n0 (350s)
    h_bin = _to_binary(best_h, best_n)
    h_sa, c5_sa = _sa(best_n, h_bin, rng, t0 + 380)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.astype(float)
        best_n = len(h_sa)

    # Phase 3: Deterministic refinement (60s)
    h_det, c5_det = _det_refine(best_h, t0 + 440, 30)
    if c5_det < best_c5:
        best_c5 = c5_det
        best_h = h_det.astype(float)
        best_n = len(h_det)

    # Phase 4: SLSQP from best binary (80s)
    h4 = best_h.astype(float).copy()
    for alpha in [20000, 200000, 2000000]:
        if time.time() - t0 > 520:
            break
        h4 = _slsqp(h4, alpha, 3000, 1e-19)
        c5 = _eval(h4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h4.copy()
            best_n = len(h4)

    # Phase 5: Try n=320 (150s)
    n5 = 320
    h5b = _to_binary(_resample(best_h, n5), n5)
    h5_sa, c5_5 = _sa(n5, h5b, rng, t0 + 670)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5_sa.astype(float)
        best_n = n5

    # Phase 6: Det refine at best_n (80s)
    h6, c5_6 = _det_refine(best_h, t0 + 750, 30)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.astype(float)
        best_n = len(h6)

    # Phase 7: More SA at best_n (150s)
    h7b = _to_binary(best_h, best_n)
    h7_sa, c5_7 = _sa(best_n, h7b, rng, t0 + 900)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7_sa.astype(float)
        best_n = len(h7_sa)

    # Phase 8: Det refine again (40s)
    h8, c5_8 = _det_refine(best_h, t0 + 940, 20)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.astype(float)
        best_n = len(h8)

    # Phase 9: Final SLSQP (remaining)
    if time.time() - t0 < budget_s - 10:
        h9 = best_h.astype(float).copy()
        h9 = _slsqp(h9, 5000000, 3000, 1e-20)
        c5 = _eval(h9)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h9.copy()
            best_n = len(h9)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
