    # Phase 1: Nesterov at base resolution with varying alpha

        if _el() > 120:
            break
        hn, fn = _nesterov(best_h, alpha, 4, 4000, 0.08)
        if fn < best_c5:
            best_h, best_c5, best_n = hn, fn, len(hn)

    # Phase 2: Round-robin on top lags
    if _el() < _budget - 120:
        hn, fn = _round_robin(best_h, 50, 6, 3)
        if fn < best_c5:
            best_h, best_c5, best_n = hn, fn, len(hn)

    # Phase 3: SLSQP polish
    if _el() < _budget - 100:
        hn = _slsqp_quick(best_h, 50000, 3, 800)
        cn = _eval(hn)
        if cn < best_c5:
            best_h, best_c5, best_n = hn, cn, len(hn)

    # Phase 4: Upsample to 2x
    if _el() < _budget - 250:
        n2 = best_n * 2
        h2 = _upsample(best_h, 2)
        for alpha in [20000, 80000]:
            if _el() > _budget - 180:
                break
            hn, fn = _nesterov(h2, alpha, 5, 5000, 0.04)
            if fn < best_c5:
                best_h, best_c5, best_n = hn, fn, len(hn)
        if _el() < _budget - 150:
            hn, fn = _round_robin(best_h, 40, 5, 3)
            if fn < best_c5:
                best_h, best_c5, best_n = hn, fn, len(hn)
        if _el() < _budget - 120:
            hn = _slsqp_quick(best_h, 100000, 4, 600)
            cn = _eval(hn)
            if cn < best_c5:
                best_h, best_c5, best_n = hn, cn, len(hn)

    # Phase 5: Complement (1-h) trick + re-optimize
    if _el() < _budget - 100:
        h_comp = 1.0 - best_h
        h_comp = _repair(h_comp, best_n)
        fc = _eval(h_comp)
        if fc < best_c5:
            best_h, best_c5, best_n = h_comp, fc, best_n
        hn, fn = _nesterov(best_h, 50000, 4, 3000, 0.05)
        if fn < best_c5:
            best_h, best_c5, best_n = hn, fn, len(hn)

    # Phase 6: Multi-start with small perturbations
    nc = best_n
    for _r in range(15):
        if _el() > _budget - 80:
            break
        for amp in [0.004, 0.008, 0.015]:
            if _el() > _budget - 75:
                break
            hp = best_h + amp * rng.randn(nc)
            hp = _repair(hp, nc)
            hn, fn = _nesterov(hp, 50000, 3, 2000, 0.04)
            if fn < best_c5:
                best_h, best_c5, best_n = hn, fn, len(hn)

    # Phase 7: Fine round-robin + pairwise
    if _el() < _budget - 50:
        hn, fn = _round_robin(best_h, 60, 4, 2)
        if fn < best_c5:
            best_h, best_c5, best_n = hn, fn, len(hn)
    for scale, nit in [(0.002, 6000), (0.0005, 8000), (0.0001, 6000)]:
        if _el() > _budget - 12:
            break
        hn, fn = _pairwise(best_h, nit, scale, seed + 7 * int(scale * 1e6))
        if fn < best_c5:
            best_h, best_c5, best_n = hn, fn, len(hn)

    # Phase 8: Final SLSQP polish
    if _el() < _budget - 10:
        hn = _slsqp_quick(best_h, 200000, 3, 400)
        cn = _eval(hn)
        if cn < best_c5:
            best_h, best_c5, best_n = hn, cn, len(hn)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
