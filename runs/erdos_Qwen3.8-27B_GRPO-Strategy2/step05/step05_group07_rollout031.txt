from scipy.interpolate import interp1d

_t0 = 0.0

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def grad_f(h, w_active, lags_active):
    n = len(h)
    g = np.zeros(n)
    for (k, wk) in zip(lags_active, w_active):
        s = k - (n - 1)
        if s >= 0:
            end = n - s
            g[:end] += wk * (1.0 - h[s:s + end])

        else:
            sp = -s
            end = n - sp
            g[sp:] += wk * (1.0 - h[:end])
            g[:end] -= wk * h[sp:sp + end]
            g[sp:] += wk
    return g

            if best_f is None or f_val < best_f:

                best_h = h.copy()
    return best_h
