def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

        return m + float(np.log(np.sum(np.exp(s)))) / alpha

        # Simpler: just use numerical gradient or let SLSQP do it
        return None  # Let SLSQP compute numerically
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],

    n = len(h)
    tgt = n / 2.0
    c = _corr(h)
    nk = 2 * n - 1
    
    # Build linearization matrix
    # G[k,j] = d c[k]/d h[j] at h
    # c[k] = sum_i h[i+s](1-h[i]) where s = k-(n-1)
    # d/dh[j] = (1-h[j-s])*[0<=j-s<n] - h[j+s]*[0<=j+s<n]
    A = np.zeros((nk, n + 1))
    b = np.zeros(nk)
    for j in range(n):
        for s in range(-(n-1), n):
            k = s + n - 1
            g = 0.0
            if 0 <= j - s < n:
                g += 1.0 - h[j - s]
            if 0 <= j + s < n:
                g -= h[j + s]
            A[k, j] = dx * g
    A[:, n] = -1.0
    b = -c * dx + A[:, :n] @ h

    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp(h, dx, alpha, maxiter=300):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lp_step(h, dx, delta=0.1, lam=1.0):
    n = len(h)
    tgt = n / 2.0
    c = _corr(h)
    nk = 2 * n - 1
    s_vals = np.arange(-(n - 1), n)
    j_vals = np.arange(n)
    js = j_vals[:, None] - s_vals[None, :]
    jp = j_vals[:, None] + s_vals[None, :]
    m1 = (js >= 0) & (js < n)
    m2 = (jp >= 0) & (jp < n)
    t1 = np.where(m1, 1.0 - h[np.clip(js, 0, n - 1)], 0.0)
    t2 = np.where(m2, h[np.clip(jp, 0, n - 1)], 0.0)
    G = (t1 - t2).T
    dG = dx * G
    A = np.hstack([dG, -np.ones((nk, 1))])
    b = -c * dx + dG.dot(h)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([tgt])
    bnds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bnds, method='highs')
    if not res.success:
        return h, _c5(h, dx)
    h_lp = res.x[:n]
    diff = h_lp - h
    ni = np.max(np.abs(diff))
    if ni > 1e-12:
        if ni > delta:
            h_new = h + delta * diff / ni
        else:
            h_new = h + lam * diff
    else:
        h_new = h.copy()
    h_new = _repair(np.clip(h_new, 0, 1), tgt)
    return h_new, _c5(h_new, dx)

def _top_lag_nudge(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:6]
        imp = False
        for ki in top:
            s = int(ki - (n - 1))
            if s == 0:
                continue
            m = n - abs(s)
            for _ in range(12):
                a = rng.randint(m)
                i1 = a + abs(s) if s > 0 else a
                i2 = a if s > 0 else a + abs(s)
                d = rng.uniform(0.002, 0.015)
                h[i1] -= d; h[i2] += d
                if 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1:
                    h = _repair(h, tgt)
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn; imp = True
                    else:
                        h[i1] += d; h[i2] -= d
                        h = _repair(h, tgt)
                else:
                    h[i1] += d; h[i2] -= d
                    h = _repair(h, tgt)
        for _ in range(6):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.008, 0.008)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                h = _repair(h, tgt)
                cn = _c5(h, dx)
                if cn < best:
                    best = cn; imp = True
                else:
                    h[i] -= d; h[j] += d
                    h = _repair(h, tgt)
            else:
                h[i] -= d; h[j] += d
                h = _repair(h, tgt)
    return h, best

def _coord_descent(h, dx, t_end, n_passes=30):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    s_vals = np.arange(-(n - 1), n)
    j_vals = np.arange(n)
    js_all = j_vals[:, None] - s_vals[None, :]
    jp_all = j_vals[:, None] + s_vals[None, :]
    for p in range(n_passes):
        if time.time() > t_end:
            break
        c = _corr(h)
        for j in range(n):
            js = js_all[j]
            jp = jp_all[j]
            m1 = (js >= 0) & (js < n)
            m2 = (jp >= 0) & (jp < n)
            g = np.where(m1, 1.0 - h[np.clip(js, 0, n - 1)], 0.0) - np.where(m2, h[np.clip(jp, 0, n - 1)], 0.0)
            base = c - h[j] * g
            f0 = np.max(base)
            f1 = np.max(base + g)
            t_opt = 0.0
            f_opt = f0
            if f1 < f_opt:
                t_opt, f_opt = 1.0, f1
            order = np.argsort(base)[::-1]
            for _ in range(3):
                for idx2 in range(1, min(4, len(order))):
                    s1, s2 = order[0], order[idx2]
                    dg = g[s1] - g[s2]
                    if abs(dg) > 1e-12:
                        tc = (base[s2] - base[s1]) / dg
                        if 0 <= tc <= 1:
                            fc = np.max(base + tc * g)
                            if fc < f_opt:
                                f_opt, t_opt = fc, tc
            h[j] = np.clip(t_opt, 0.0, 1.0)
            h = _repair(h, tgt)
            cn = _c5(h, dx)
            if cn < best:
                best = cn
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
