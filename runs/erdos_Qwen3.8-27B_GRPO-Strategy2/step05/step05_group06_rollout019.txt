def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lsq_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lsq_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:topk]

    for k in idx:

            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:

                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def _nesterov(h, alpha, steps, lr0, mu, t_end):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    lr = lr0
    for t in range(steps):
        if time.time() > t_end:
            break
        g = _lsq_grad(h, alpha)
        v = mu * v - lr * g
        h = _proj_to_sum(h + v, target)
        lr = lr0 * (1.0 - t / max(steps, 1)) * 0.5 + lr0 * 0.5
    return h

def _subsgrad(h, steps, topk, lr0, t_end):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h)
    lr = lr0
    for t in range(steps):

            lr = min(lr * 1.02, lr0 * 2)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
    return h, c5c

            break
        # Random permutation of pairs
        perm = rng.permutation(n)
        used = set()
        for idx in perm:
            if idx in used:
                continue
            j = None
            for j2 in perm:
                if j2 != idx and j2 not in used:
                    j = j2
                    break
            if j is None:
                break
            used.add(idx)
            used.add(j)
            s_fixed = h[idx] + h[j]
            # Optimize h[idx] in [max(0, s_fixed-1), min(1, s_fixed)]
            lo = max(0.0, s_fixed - 1.0)
            hi = min(1.0, s_fixed)
            if hi - lo < 1e-10:
                continue
            for _ in range(25):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                h[idx], h[j] = m1, s_fixed - m1
                c1 = _eval(h)
                h[idx], h[j] = m2, s_fixed - m2
                c2 = _eval(h)
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            h[idx] = (lo + hi) / 2
            h[j] = s_fixed - h[idx]
        h = _repair(h, n)
        c5c = _eval(h)
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lsq_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lsq_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov(h, alpha, steps, lr0, mu, t_end):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    lr = lr0
    for t in range(steps):
        if time.time() > t_end:
            break
        g = _lsq_grad(h, alpha)
        v = mu * v - lr * g
        h = _proj_to_sum(h + v, target)
        lr = lr0 * (1 - 0.5 * t / max(steps, 1))
        if lr < 1e-8:
            lr = 1e-8
    return h

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    dx = 2.0 / n
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def _subsgrad(h, steps, topk, lr0, t_end):
    n = len(h)
    target = n * 0.5
    c5c = _eval(h)
    lr = lr0
    for t in range(steps):
        if time.time() > t_end:
            break
        g = _topk_grad(h, topk)
        h_new = _proj_to_sum(h - lr * g, target)
        c5n = _eval(h_new)
        if c5n < c5c:
            h, c5c = h_new, c5n
            lr = min(lr * 1.02, lr0 * 3)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
    return h, c5c
