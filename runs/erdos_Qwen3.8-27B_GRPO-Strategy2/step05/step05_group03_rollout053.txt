    h = np.clip(h, 0.0, 1.0).copy()

        n2, dx2 = 2 * n, 1.0 / n
        h2 = _repair(_interp(best_h, n2), n2 / 2.0)
        t_end6 = min(t0 + budget_s - 90, time.time() + 130)
        h2, c2 = _subgrad(h2, dx2, 0.004, min(t_end6 - 50, time.time() + 40), lr_init=0.007)
        if time.time() < t_end6 - 25:
            h2, c2 = _lse_solve(h2, dx2, 0.0008, 200)
        if time.time() < t_end6 - 10:
            h2, c2 = _local_refine(h2, dx2, rng, t_end6, 800)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2, c2, n2

    # P7: Final subgradient + SLSQP at best resolution
    dxb = 2.0 / best_n
    if rem() > 50:
        h, c = _subgrad(best_h, dxb, 0.002, min(t0 + budget_s - 40, time.time() + 40), lr_init=0.004)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 30:
        h, c = _lse_solve(best_h, dxb, 0.0002, 250)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 20:
        h, c = _pnorm_solve(best_h, dxb, 200, 200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P8: Final local + anneal
    if rem() > 15:
        best_h, best_c5 = _local_refine(best_h, dxb, rng, min(t0 + budget_s - 8, time.time() + 12), 500)
    if rem() > 8:
        h, c = _anneal(best_h, dxb, rng, t0 + budget_s - 3, 0.001, 0.0001)
        if c < best_c5:
            best_h = h
            best_c5 = c

    best_h = _repair(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
