**Key algorithmic improvement: Active-set QP**

At the optimum, suppose lags k_1, ..., k_m are "active" (achieve the max). Then we want:
min_t t
s.t. c_{k_i}(h) ≤ t for all i
     0 ≤ h ≤ 1
     sum(h) = n/2

This is a non-convex QCQP (because c_k is quadratic in h). But if we linearize c_k around current h:
c_k(h+δ) ≈ c_k(h) + ∇c_k(h)ᵀ δ

Then we can solve a QP:
min_t t
s.t. c_k(h_i) + ∇c_k(h_i)ᵀ δ ≤ t
     0 ≤ h_i + δ_i ≤ 1
     sum(δ) = 0

This is a linear program (LP)! We can solve it with scipy or cvxpy.

Then take a step δ, re-evaluate, and iterate. This is essentially a cutting-plane / sequential linearization approach.

def _lp_step(h, top_k=20, max_step=0.1):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top lags
    idx = np.argsort(c)[::-1][:top_k]
    
    # Build LP: minimize t
    # Variables: delta[0..n-1], t (total n+1 variables)
    # Constraints:
    #   For each active lag k: sum_j grad_k[j] * delta[j] + c[k] <= t
    #     => -sum_j grad_k[j] * delta[j] - t <= -c[k]  (A_ub * x <= b_ub form)
    #   0 <= h[j]+delta[j] <= 1 => -h[j] <= delta[j] <= 1-h[j]
    #   sum(delta) = 0
    #
    # In linprog form: minimize c @ x
    #   c_obj = [0]*n + [1]  (minimize t)
    #   A_ub * x <= b_ub
    #   A_eq * x = b_eq
    #   bounds
    
    # A_ub rows for correlation constraints:
    #   For lag k: sum_j grad_k[j]*delta[j] + c[k] - t <= 0
    #   => grad_k[0]*d0 + ... + grad_k[n-1]*d_{n-1} + 1*t <= c[k]
    #   Wait: we want c[k] + grad_k @ delta <= t
    #   => grad_k @ delta - t <= c[k] - c[k] = 0... no
    #   c[k] + grad_k @ delta <= t
    #   grad_k @ delta - t <= -c[k]
    #   So A_ub row: [grad_k[0], ..., grad_k[n-1], -1], b = -c[k]
    
    A_rows = []
    b_rows = []
    for lk in idx:
        gk = _lag_grad(h, lk)
        row = np.append(gk, -1.0)
        A_rows.append(row)
        b_rows.append(-c[lk])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[j] <= delta[j] <= 1-h[j], t unbounded below (but will be >= max(c))
    bounds = []
    for j in range(n):
        bounds.append((-h[j], 1.0 - h[j]))
    bounds.append((None, None))  # t is free
    
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        delta = result.x[:n]
        # Limit step size
        step_norm = np.linalg.norm(delta)
        if step_norm > max_step:
            delta *= max_step / step_norm
        h_new = np.clip(h + delta, 0, 1)
        # Re-project to maintain sum constraint
        h_new = _repair(h_new, n)
        return h_new

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k < n:
        j = np.arange(k + 1)
        g[j] = 1.0 - 2.0 * h[k - j]
    else:
        j = np.arange(k - n + 1, n)
        g[j] = 1.0 - 2.0 * h[k - j]
    return g

def _active_lags(h, tol=0.001):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    return np.where(c > cmax - tol)[0], cmax

def _lp_direction(h, lags):

def _smooth_obj(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _lag_grad(h, k)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _gp_step(h, lags, max_step):
    """Gradient projection step on active lags"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')

    # Scale

        h_try = _repair(h - scale * g, n)
        c_try = _eval(h_try)
        if c_try < best_c:

    return best_h

import numpy as np
from scipy.optimize import minimize, linprog

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k < n:
        j = np.arange(k + 1)
        g[j] = 1.0 - 2.0 * h[k - j]
    else:
        j = np.arange(k - n + 1, n)
        g[j] = 1.0 - 2.0 * h[k - j]
    return g

def _active_lags(h, frac=0.003, cap=30):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    mask = c > cmax - frac * cmax - 1e-5
    idx = np.where(mask)[0]
    if len(idx) > cap:
        idx = np.argsort(c)[::-1][:cap]
    return idx

def _lp_dir(h, lags):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    A_rows, b_rows = [], []

        gk = _lag_grad(h, k)

    try:

                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        g += w[k] * _lag_grad(h, k)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

    lo, hi = -1.5, 1.5

            lo = mid
        else:
