    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp_direct(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    bds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_tanh(h, dx, alpha, maxiter=250):
    n = len(h)
    tgt = n / 2.0
    hc = np.clip(h, 1e-8, 1.0 - 1e-8)
    x0 = np.arctanh(2.0 * hc - 1.0)
    def obj_t(x):
        ht = (1.0 + np.tanh(x)) / 2.0
        return _smooth_max(_corr(ht), alpha)
    def eq_t(x):
        ht = (1.0 + np.tanh(x)) / 2.0
        return ht.sum() - tgt
    try:
        r = minimize(obj_t, x0, method='SLSQP',
                     constraints=[{'type': 'eq', 'fun': eq_t}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hh = _repair((1.0 + np.tanh(r.x)) / 2.0, tgt)
        return hh, _c5(hh, dx)
    except Exception:
        return h, _c5(h, dx)

def _sa(h, dx, rng, t_end):
    n = len(h)
    h = h.copy()
    cur_c = _c5(h, dx)
    best_c = cur_c
    best_h = h.copy()
    total_time = max(1.0, t_end - time.time())
    while time.time() < t_end:
        frac = max(0.0, (t_end - time.time()) / total_time)
        temp = 0.008 * frac ** 2
        if temp < 1e-12:
            break
        r = rng.random()
        if r < 0.5:
            bs = int(rng.choice([2, 3, 5, 8, 15]))
            i = rng.randint(0, n - bs)
            j = rng.randint(0, n - bs)
            if abs(i - j) < bs:
                continue
            h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
        elif r < 0.8:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            dv = rng.uniform(0.02, 0.12)
            if h[i] < dv or h[j] > 1.0 - dv:
                continue
            h[i] -= dv; h[j] += dv
        new_c = _c5(h, dx)
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / temp):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            if r < 0.5:
                h[i:i+bs], h[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
            elif r < 0.8:
                h[i], h[j] = h[j], h[i]
            else:
                h[i] += dv; h[j] -= dv
    return best_h, best_c

def _targeted(h, dx, rng, t_end):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        improved = False
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            bad = h[s:s+m] * (1.0 - h[:m]) + h[:m] * (1.0 - h[s:s+m])
            for wi in np.argsort(bad)[::-1][:5]:
                for dv in [0.03, 0.06, 0.1]:
                    h2 = h.copy()
                    if h[wi] > h[s+wi]:
                        h2[wi] -= dv; h2[s+wi] += dv
                    else:
                        h2[s+wi] -= dv; h2[wi] += dv
                    if h2.min() < 0 or h2.max() > 1:
                        continue
                    h2 = _repair(h2, tgt)
                    cn = _c5(h2, dx)
                    if cn < best:
                        best = cn; h = h2; improved = True
                        break
        if not improved:
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                h2 = h.copy()
                dv = rng.uniform(-0.04, 0.04)
                h2[i] += dv; h2[j] -= dv
                if 0.0 <= h2[i] <= 1.0 and 0.0 <= h2[j] <= 1.0:
                    h2 = _repair(h2, tgt)
                    cn = _c5(h2, dx)
                    if cn < best:
                        best = cn; h = h2
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    for sc in [0.004, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            dv = sc * (2.0 * rng.random() - 1.0)
            h[i] += dv; h[j] -= dv
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= dv; h[j] += dv
            else:
                h[i] -= dv; h[j] += dv
    return h, best

def _perturb(h, rng, mode):
    n = len(h)
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.04 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        bs = int(rng.choice([5, 10, 15]))
        i = rng.randint(0, n - bs)
        hp[i:i+bs] = 1.0 - hp[i:i+bs]
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.03 * np.sin(2.0*np.pi*rng.uniform(2,8)*x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, n//3, replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values)/2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)

    # Phase 1: SLSQP direct sweep [0-50s]
    for alpha in [500, 2000, 8000, 30000]:
        if time.time() > t0 + 12:
            break
        h, c = _slsqp_direct(best_h, dx, alpha, 200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: SLSQP tanh parameterization [50s-90s]
    for alpha in [3000, 15000, 60000]:
        if time.time() > t0 + 90:
            break
        h, c = _slsqp_tanh(best_h, dx, alpha, 200)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Simulated annealing exploration [90s-420s]
    h, c = _sa(best_h, dx, rng, t0 + 420)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 4: SLSQP polish [420s-520s]
    for alpha in [10000, 50000, 200000]:
        if time.time() > t0 + 520:
            break
        h, c = _slsqp_direct(best_h, dx, alpha, 180)
        if c < best_c5:
            best_h, best_c5 = h, c
        h, c = _slsqp_tanh(best_h, dx, alpha, 180)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 5: Targeted worst-lag search [520s-720s]
    h, c = _targeted(best_h, dx, rng, t0 + 720)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 6: Multi-start with perturbation [720s-890s]
    for r in range(12):
        if time.time() > t0 + 890:
            break
        hp = _perturb(best_h, rng, r % 4)
        hp, c = _sa(hp, dx, rng, min(t0 + 890, time.time() + 10))
        if c < best_c5:
            best_h, best_c5 = hp, c
        hp, c = _slsqp_tanh(hp, dx, 80000, 120)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 7: Fine local search [890s-1030s]
    best_h, best_c5 = _fine(best_h, dx, rng, t0 + 1030)

    # Phase 8: Final SLSQP high-alpha [1030s-1080s]
    if time.time() < t0 + budget_s - 25:
        h, c = _slsqp_tanh(best_h, dx, 400000, 150)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 9: Micro fine-tune
    best_h, best_c5 = _fine(best_h, dx, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, n / 2.0)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
