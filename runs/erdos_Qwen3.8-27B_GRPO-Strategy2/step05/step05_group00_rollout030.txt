import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _fft_corr_full(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    R = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([R[N - n + 1:], R[:n]])

def _eval(h):
    n = len(h)
    c = _fft_corr_full(h)
    return float(np.max(c) * 2.0 / n)

def _softmax_obj(h, alpha):
    n = len(h)
    c = _fft_corr_full(h)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_corr(h, alpha):
    """Gradient of soft-max objective."""
    n = len(h)
    c = _fft_corr_full(h)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_grad(h, alpha, lr, momentum, max_iter, time_limit):
    """Nesterov accelerated projected gradient descent."""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    c5_prev = _eval(h)
    f_prev = _softmax_obj(h, alpha)
    
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # Nesterov lookahead
        h_look = h - momentum * v
        h_look = np.clip(h_look, 0, 1)
        
        g = _grad_corr(h_look, alpha)
        
        # Adapt learning rate
        # Barzilai-Borwein or fixed with decay
        v = momentum * v - lr * g
        h_new = h + v
        h_new = _repair(h_new, n)
        
        c5_new = _eval(h_new)
        if c5_new < c5_prev - 1e-15:
            h = h_new
            c5_prev = c5_new
            f_prev = _softmax_obj(h, alpha)
        else:
            # Halve learning rate
            lr *= 0.5
            v = np.zeros(n)
            # Retry with smaller step
            h_new = h - lr * _grad_corr(h, alpha)
            h_new = _repair(h_new, n)
            c5_new = _eval(h_new)
            if c5_new < c5_prev - 1e-15:
                h = h_new
                c5_prev = c5_new
            else:
                momentum *= 0.9
    
    return h, c5_prev
