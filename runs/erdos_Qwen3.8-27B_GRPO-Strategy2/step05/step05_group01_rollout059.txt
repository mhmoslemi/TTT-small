    # Phase 5: Pair search with decreasing scale (120s)
    if _el() < budget_s - 300:
        h_p = best_h.copy()
        c_p = _corr(h_p)
        for sc in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if _el() > budget_s - 250:
                break
            h_p, c_p = _pair_search(h_p, c_p, n0, 50000, rng, sc)
        h_p = _repair(h_p, n0)
        c5 = _eval_c5(h_p)
        if c5 < best_c5:
            best_c5, best_h = c5, h_p

    # Phase 6: Binary relaxation attempt (30s)
    if _el() < budget_s - 250:
        hb = _try_binary(best_h, n0, best_c5)
        if hb is not None:
            hb, _ = _coord_descent(hb, 20)
            hb = _repair(hb, n0)
            c5 = _eval_c5(hb)
            if c5 < best_c5:
                best_c5, best_h = c5, hb

    # Phase 7: Coarse grid exploration (60s)
    if _el() < budget_s - 180:
        for n_coarse in [40, 56, 80, 140]:
            if _el() > budget_s - 140:
                break
            # Interpolate best_h to coarse grid
            x_full = np.linspace(0, 2, n0)
            x_co = np.linspace(0, 2, n_coarse)
            h_co = np.interp(x_co, x_full, best_h)
            h_co = _repair(h_co, n_coarse)
            h_co, _ = _coord_descent(h_co, 30)
            h_co = _repair(h_co, n_coarse)
            c5_co = _eval_c5(h_co)
            if c5_co < best_c5:
                # Upsample back
                h_up = np.interp(x_full, x_co, h_co)
                h_up = _repair(h_up, n0)
                h_up, _ = _coord_descent(h_up, 30)
                h_up = _repair(h_up, n0)
                c5_up = _eval_c5(h_up)
                if c5_up < best_c5:
                    best_c5, best_h = c5_up, h_up

    # Phase 8: Multi-start perturbation (80s)
    if _el() < budget_s - 100:
        for _r in range(8):
            if _el() > budget_s - 80:
                break
            hp = best_h + 0.03 * rng.randn(n0)
            hp = _repair(hp, n0)
            hp = _proj_gradient(hp, 50000, 100, 0.004)
            hp = _repair(hp, n0)
            hp, _ = _coord_descent(hp, 20)
            hp = _repair(hp, n0)
            c5 = _eval_c5(hp)
            if c5 < best_c5:
                best_c5, best_h = c5, hp

    # Phase 9: Final intensive coordinate + pair search (remaining)
    if _el() < budget_s - 20:
        h_f, c_f = _coord_descent(best_h.copy(), 50)
        h_f = _repair(h_f, n0)
        c_f = _corr(h_f)
        for sc in [0.001, 0.0003, 0.0001]:
            if _el() > budget_s - 10:
                break
            h_f, c_f = _pair_search(h_f, c_f, n0, 60000, rng, sc)
        h_f = _repair(h_f, n0)
        c5 = _eval_c5(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f

    # Final coordinate descent polish
    if _el() < budget_s - 5:
        h_f, _ = _coord_descent(best_h.copy(), 30)
        h_f = _repair(h_f, n0)
        c5 = _eval_c5(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f

    best_h = _repair(best_h, best_n)
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, best_n)
