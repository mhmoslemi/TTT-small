                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return h

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx / topk

def _nm_soft(h, thr_lo, thr_hi, maxiter):
    n = len(h)
    soft = (h > thr_lo) & (h < thr_hi)
    soft_idx = np.where(soft)[0]
    m = len(soft_idx)
    if m < 2 or m > 60:
        return h
    fixed_sum = h[~soft].sum()
    soft_target = n * 0.5 - fixed_sum
    if soft_target < 0 or soft_target > m:
        return h
    def _obj(sv):
        ht = h.copy()
        ht[soft_idx] = np.clip(sv, 0, 1)
        ht = _repair(ht, n)
        return _eval(ht)
    x0 = h[soft_idx].copy()
    try:
        res = minimize(_obj, x0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-15})
        ht = h.copy()
        ht[soft_idx] = np.clip(res.x, 0, 1)
        return _repair(ht, n)
    except Exception:
        return h

def _active_grad_desc(h, topk, nsteps, base_lr):
    n = len(h)
    c5c = _eval(h)
    lr = base_lr
    for _ in range(nsteps):
        g = _topk_grad(h, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        for attempt in range(20):
            h_try = _proj_to_sum(h - lr * g, n * 0.5)
            c5n = _eval(h_try)
            if c5n < c5c:
                h, c5c = h_try, c5n
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                lr = base_lr
        lr = min(lr * 1.005, base_lr * 2)
    return h, c5c

def _fine_pairwise(h, rng, scale, iters):
    n = len(h)
    c5c = _eval(h)
    for _ in range(iters):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _block_moves(h, rng, iters, bs_options):
    n = len(h)
    c5c = _eval(h)
    for _ in range(iters):
        bs = bs_options[rng.randint(len(bs_options))]
        st = rng.randint(0, max(1, n - bs))
        h_new = h.copy()
        h_new[st:st + bs] += 0.02 * rng.randn(bs)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            h, c5c = h_new, c5n
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: SLSQP with escalating alpha
    for alpha in [800, 5000, 30000, 150000]:
        if _el() > 250:
            break
        h_new = _slsqp(best_h, alpha, 3500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Active-set gradient descent (equalize top correlations)
    if _el() < budget_s - 400:
        h2 = best_h.copy()
        for topk in [2, 4, 8]:
            if _el() > budget_s - 350:
                break
            h2, _ = _active_grad_desc(h2, topk, 400, 0.008)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: Nelder-Mead on soft variables (key novel step)
    if _el() < budget_s - 250:
        for thr in [(0.01, 0.99), (0.03, 0.97), (0.08, 0.92)]:
            if _el() > budget_s - 200:
                break
            h_nm = _nm_soft(best_h, thr[0], thr[1], 8000)
            c5 = _eval(h_nm)
            if c5 < best_c5:
                best_c5, best_h = c5, h_nm.copy()

    # Phase 4: SLSQP final polish at very high alpha
    if _el() < budget_s - 150:
        h_new = _slsqp(best_h, 400000, 2500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()
        h_new = _slsqp(best_h, 800000, 1500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: Another NM pass after SLSQP reshaped the soft set
    if _el() < budget_s - 100:
        h_nm = _nm_soft(best_h, 0.02, 0.98, 10000)
        c5 = _eval(h_nm)
        if c5 < best_c5:
            best_c5, best_h = c5, h_nm.copy()

    # Phase 6: Block + fine pairwise search
    if _el() < budget_s - 60:
        h_blk, c5b = _block_moves(best_h, rng, 3000, [3, 5, 7, 10, 14])
        if c5b < best_c5:
            best_c5, best_h = c5b, h_blk.copy()
        for scale in [0.003, 0.001, 3e-4, 1e-4]:
            if _el() > budget_s - 50:
                break
            h_f, c5f = _fine_pairwise(best_h, rng, scale, 8000)
            if c5f < best_c5:
                best_c5, best_h = c5f, h_f.copy()

    # Phase 7: Final micro-polish
    if _el() < budget_s - 10:
        h_new = _slsqp(best_h, 1000000, 1000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 8: Ultra-fine last steps
    if _el() < budget_s - 3:
        h_f, c5f = _fine_pairwise(best_h, rng, 5e-5, 15000)
        if c5f < best_c5:
            best_c5, best_h = c5f, h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
