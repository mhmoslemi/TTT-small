                    if c5_n < c5_cur:
                        c5_cur = c5_n
                    else:
                        h_cur[i], h_cur[j] = ni - d, nj + d
    return best_h, best_c5

def _sa_search(h, t_end, n_iters):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    rng = np.random.RandomState(4242)
    T = 0.0015
    cool = 0.9997
    for it in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = max(T, 1e-6) * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5_new = _eval(h_cur)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        elif rng.rand() < np.exp(-(c5_new - c5_cur) / max(T, 1e-12)):
            c5_cur = c5_new
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
        T *= cool
    return best_h, best_c5

def _upsample(h, factor=2):
    n = len(h)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n * factor)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n0 = len(initial_h_values)
    h = _proj_sum(np.array(initial_h_values, dtype=float).copy(), n0 * 0.5)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP at n=280 with increasing alpha
    for alpha in [500, 5000, 50000]:
        if time.time() > t0 + 140:
            break
        h_new = _slsqp_lse(h, alpha, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Gradient-directed coord search at n=280
    if time.time() < t0 + 240:
        h2, c5_2 = _grad_coord_search(best_h, t0 + 240, 50000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Upsample to 560, add Nyquist texture, SLSQP
    if time.time() < t0 + 560:
        h_hi = _upsample(best_h, 2)
        nh = len(h_hi)
        # Add small alternating perturbation for anti-aliasing
        alt = np.tile(np.array([1.0, -1.0]), nh // 2)
        h_hi = _proj_sum(h_hi + 0.005 * alt, nh * 0.5)
        for alpha in [2000, 20000, 100000]:
            if time.time() > t0 + 560:
                break
            h_new = _slsqp_lse(h_hi, alpha, 2500)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = nh
                h_hi = h_new.copy()

    # Phase 4: SA at best resolution
    if time.time() < t0 + 880:
        h4, c5_4 = _sa_search(best_h, t0 + 880, 200000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Final SLSQP polish at high alpha
    if time.time() < t0 + budget_s - 50:
        h5 = _slsqp_lse(best_h, 500000, 4000)
        c5_5 = _eval(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    # Phase 6: Ultra-fine coordinate search
    if time.time() < t0 + budget_s - 3:
        n = len(best_h)
        h6 = best_h.copy()
        c5_6 = _eval(h6)
        rng6 = np.random.RandomState(seed + 13)
        for it in range(400000):
            if time.time() > t0 + budget_s - 3:
                break
            i = rng6.randint(n)
            j = rng6.randint(n)
            if i == j:
                continue
            d = 3e-5 * rng6.randn()
            ni, nj = h6[i] + d, h6[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h6[i], h6[j] = ni, nj
                c5_n = _eval(h6)
                if c5_n < c5_6 - 1e-18:
                    c5_6 = c5_n
                    if c5_n < best_c5:
                        best_c5 = c5_n
                        best_h = h6.copy()
                else:
                    h6[i], h6[j] = ni - d, nj + d

    best_h = _proj_sum(best_h, len(best_h) * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
