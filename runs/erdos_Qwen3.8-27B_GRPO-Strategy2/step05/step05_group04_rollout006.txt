import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, K, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-c)[:K]
    top_c = c[idx]
    mx = top_c.max()
    w = np.exp(alpha * (top_c - mx))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(K):
        ki = idx[ii]
        sh = (n - 1) - ki
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _pgd(h, K, alpha, steps, step0):
    n = len(h)
    h_c = h.copy()
    f_c = _eval(h_c)
    step = step0
    v = np.zeros(n)
    mom = 0.4
    for s in range(steps):
        g = _grad_topk(h_c, K, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        v = mom * v + (1 - mom) * g / gn
        h_new = _proj(h_c - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f_c:
            h_c = h_new; f_c = f_new
            step = min(step * 1.04, step0 * 4.0)
        else:
            step *= 0.55
            h_c = _proj(h_c - step * v, n * 0.5)
            f_c = _eval(h_c)
    return h_c, f_c

def _smooth_max_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 600)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sah(h, n, iters, rng, temp0, temp1):
    c_best = _eval(h)
    best = h.copy()
    c_cur = c_best
    for i in range(iters):
        t = temp0 * (temp1 / temp0) ** (i / max(1, iters - 1))
        bsz = rng.randint(3, max(4, n // 8))
        st = rng.randint(0, n - bsz)
        h2 = h.copy()
        h2[st:st + bsz] += rng.randn(bsz) * (0.015 + 0.04 * t)
        h2 = _repair(h2, n)
        c2 = _eval(h2)
        if c2 < c_cur or rng.rand() < np.exp(-(c2 - c_cur) / max(t, 1e-8)):
            h = h2; c_cur = c2
            if c2 < c_best:
                c_best = c2; best = h.copy()
    return best, c_best

def _lbfgs_smooth(h, alpha, maxit):
    n = len(h)
    try:
        res = minimize(_smooth_max_obj, h.copy(), args=(alpha,),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxit, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)
