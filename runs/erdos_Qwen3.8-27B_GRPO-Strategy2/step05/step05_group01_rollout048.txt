def _targeted_pair(h, c, k_star, rng, n_iter, n):
    """Try to reduce correlation at specific lag k_star."""
    c5c = c.max() * 2.0 / n
    # k_star is in [0, 2n-2], the lag with maximum correlation
    # For k_star > n-1: the shift is k_star - (n-1)
    # For k_star < n-1: the shift is (n-1) - k_star
    # In both cases, the "effective shift" is |k_star - (n-1)|
    
    shift = abs(k_star - (n - 1))
    if shift == 0:
        return h, c, c5c  # zero lag can't be reduced by pair moves
    
    accepted = 0
    for _ in range(n_iter):
        # Pick i, try to move h[i] toward h[i+shift] or h[i-shift]
        i = rng.randint(0, n - shift)
        j = i + shift
        # We want to reduce h[i](1-h[j]) + h[j](1-h[i])
        # d/dh_i [h_i(1-h_j) + h_j(1-h_i)] = (1-h_j) - h_j = 1 - 2h_j
        # So if h_j > 0.5, decrease h_i; if h_j < 0.5, increase h_i
        # But we need to maintain sum, so also change another position
                
        # Actually, for the pair (i, j), moving δ from j to i:
        # Δc at this lag ≈ δ[(1-h[j]) - h[i]] (simplified)
        # We want Δc < 0, so (1-h[j]) < h[i], i.e., h[i] + h[j] > 1
        
        if h[i] + h[j] > 1.0:
            # Moving from j to i reduces the product
            d = 0.01 * rng.uniform(0.5, 1.0)
            if h[j] - d < 0 or h[i] + d > 1:
                d = min(h[j], 1 - h[i]) * 0.5
                if d <= 0:
                    continue
            # Move δ from j to i, but we need a third position to maintain sum
            # Actually, pair move: h[i]+=d, h[j]-=d maintains sum

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a

                    h[idx] -= a
                    diff += a

            i2 = np.arange(n - sp)

    return g

    return x.sum() - len(x) * 0.5

    n = len(h)

        delta = new_c5 - c5
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            h[i] = ni
            h[j] = nj
            c = new_c
            c5 = new_c5
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
                best_c = c.copy()
    return best_h, best_c, best_c5

def _greedy_pair(h, c, rng, n_iter, scale, n, deadline_fn):
    dx = 2.0 / n
    c5 = c.max() * dx
    km = c.argmax()
    for _ in range(n_iter):
        if deadline_fn() > 0:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
        if dc[km] > 1e-15:
            continue
        new_c = c + dc
        new_c5 = new_c.max() * dx
        if new_c5 < c5 - 1e-14:
            h[i] = ni
            h[j] = nj
            c = new_c
            c5 = new_c5
            km = c.argmax()
    return h, c, c5

def _targeted_moves(h, c, rng, n_passes, n, deadline_fn):
    dx = 2.0 / n
    for _ in range(n_passes):
        if deadline_fn() > 0:
            break
        top3 = np.argsort(c)[-3:]
        for k_s in top3:
            shift = abs(k_s - (n - 1))
            if shift == 0 or shift >= n:
                continue
            for _t in range(200):
                i = rng.randint(0, n - shift)
                j = i + shift
                s_ij = h[i] + h[j]
                if s_ij > 1.0:
                    d = min(0.008, (s_ij - 1.0) * 0.4, h[j], 1.0 - h[i])
                else:
                    d = -min(0.008, (1.0 - s_ij) * 0.4, h[i], 1.0 - h[j])
                if abs(d) < 1e-12:
                    continue
                ni = h[i] + d
                nj = h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                dc = _pos_change(h, i, d, n) + _pos_change(h, j, -d, n)
                new_c = c + dc
                if new_c.max() * dx < c.max() * dx - 1e-13:
                    h[i] = ni
                    h[j] = nj
                    c = new_c
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _remain():
        return budget_s - (time.time() - t0)

    def _spent():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP polish
    for alpha in [5000, 50000, 200000, 500000]:
        if _remain() < 80:
            break
        h_new = _slsqp(best_h, alpha, 1500, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Simulated annealing (escape local optima)
    if _remain() > 400:
        h_sa = best_h.copy()
        c_sa = _corr(h_sa)
        n_sa = len(h_sa)
        for T0, T1, nit in [(0.015, 0.001, 60000), (0.008, 0.0002, 80000), (0.003, 5e-5, 100000)]:
            if _remain() < 300:
                break
            h_sa, c_sa, c5_sa = _sa_round(h_sa, c_sa, rng, nit, T0, T1, n_sa)
            if c5_sa < best_c5:
                best_c5, best_h, best_n = c5_sa, h_sa.copy(), n_sa

    # Phase 3: Targeted moves on max lags
    if _remain() > 250:
        h_t = best_h.copy()
        c_t = _corr(h_t)
        n_t = len(h_t)
        n_passes = 500
        dl3 = budget_s - 250 + t0
        h_t, c_t = _targeted_moves(h_t, c_t, rng, n_passes, n_t, lambda: time.time())
        # Wrap with deadline
        c5 = _eval(h_t)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_t.copy(), n_t

    # Phase 4: Greedy pairwise with decreasing scale
    if _remain() > 120:
        h_g = best_h.copy()
        c_g = _corr(h_g)
        n_g = len(h_g)
        dl_end = budget_s - 100 + t0
        for scale in [0.01, 0.005, 0.002, 0.001, 0.0003, 0.0001]:
            h_g, c_g, c5_g = _greedy_pair(h_g, c_g, rng, 60000, scale, n_g,
                                           lambda: time.time() - tl_end)
            c5 = _eval(h_g)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_g.copy(), n_g

    # Fix: restructure phase 4 deadline
    if _remain() > 100:
        h_g = best_h.copy()
        c_g = _corr(h_g)
        n_g = len(h_g)
        for scale in [0.008, 0.004, 0.002, 0.0008, 0.0003, 0.0001, 5e-5]:
            if _remain() < 80:
                break
            iters = 50000
            for _ in range(iters):
                if _remain() < 80:
                    break
                i = rng.randint(n_g)
                j = rng.randint(n_g)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni = h_g[i] + d
                nj = h_g[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                dc = _pos_change(h_g, i, d, n_g) + _pos_change(h_g, j, -d, n_g)
                km = c_g.argmax()
                if dc[km] > 1e-15 and c_g[km] + dc[km] > c_g[km]:
                    pass
                new_c5 = (c_g + dc).max() * (2.0 / n_g)
                if new_c5 < c_g.max() * (2.0 / n_g) - 1e-14:
                    h_g[i] = ni
                    h_g[j] = nj
                    c_g = c_g + dc
            c5 = _eval(h_g)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_g.copy(), n_g

    # Phase 5: Final SLSQP
    if _remain() > 40:
        h_pol = _slsqp(best_h, 800000, 2000, 1e-20)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_pol.copy(), best_n

    # Phase 6: Ultra-fine greedy
    if _remain() > 10:
        h_u = best_h.copy()
        c_u = _corr(h_u)
        n_u = len(h_u)
        dx_u = 2.0 / n_u
        for _ in range(80000):
            if _remain() < 5:
                break
            i = rng.randint(n_u)
            j = rng.randint(n_u)
            if i == j:
                continue
            d = 2e-4 * rng.randn()
            ni = h_u[i] + d
            nj = h_u[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            dc = _pos_change(h_u, i, d, n_u) + _pos_change(h_u, j, -d, n_u)
            new_c5 = (c_u + dc).max() * dx_u
            if new_c5 < c_u.max() * dx_u - 1e-15:
                h_u[i] = ni
                h_u[j] = nj
                c_u += dc
        h_u = _repair(h_u, n_u)
        c5 = _eval(h_u)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_u, n_u

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
