        return _smooth_max_grad(_corr(x), x, alpha)
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, jac=grad, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _perturb(h, rng, mode):
    n = len(h)
    hp = h.copy()
    x = np.linspace(0, 2, n)
    if mode == 0:
        hp += 0.03 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(5, n // 4), replace=False)
        hp[idx] += 0.06 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        hp += 0.02 * np.sin(2 * np.pi * rng.uniform(1, 8) * x + rng.uniform(0, 6.28))
    elif mode == 3:
        idx = rng.choice(n, max(3, n // 10), replace=False)
        hp[idx] = 1.0 - hp[idx]
    elif mode == 4:
        idx = rng.choice(n, max(5, n // 3), replace=False)
        hp[idx] = np.roll(hp[idx], 1)
    else:
        k = n // 2
        hp[:k] += 0.015
        hp[k:] -= 0.015
    return _repair(np.clip(hp, 0, 1), n / 2.0)

def _targeted(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < t_end:
        c = _corr(h)
        top = np.argsort(c)[::-1][:4]
        for ki in top:
            lag = abs(int(ki - (n - 1)))
            if lag == 0:
                continue
            m = n - lag
            for _ in range(5):
                a = rng.randint(m)
                i1, i2 = a + lag, a
                d = rng.uniform(0.001, 0.008)
                h[i1] -= d; h[i2] += d
                ok = 0 <= h[i1] <= 1 and 0 <= h[i2] <= 1
                if ok:
                    cn = _c5(h, dx)
                    if cn < best:
                        best = cn
                    else:
                        h[i1] += d; h[i2] -= d
                else:
                    h[i1] += d; h[i2] -= d
    return h, best

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.002, 0.001, 5e-04, 2e-04, 1e-04, 5e-05, 2e-05]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP sweep on initial (fast with analytic grad)
    for a in [300, 1000, 5000, 20000]:
        if rem() < 40:
            break
        hc, c = _slsqp(best_h, dx, a, 500)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 2: Multi-start (diverse perturbations + SLSQP)
    t2 = t0 + min(350, budget_s * 0.32)
    for r in range(24):
        if time.time() > t2 or rem() < 300:
            break
        hp = _perturb(best_h, rng, r % 6)
        hp, c = _slsqp(hp, dx, 3000, 350)
        if c < best_c5:
            hp, c = _slsqp(hp, dx, 20000, 400)
            if c < best_c5:
                best_h, best_c5 = hp, c

    # Phase 3: High-alpha SLSQP refinement
    t3 = t0 + min(550, budget_s * 0.5)
    for a in [5000, 20000, 60000, 150000]:
        if time.time() > t3 or rem() < 250:
            break
        hc, c = _slsqp(best_h, dx, a, 600)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Phase 4: Targeted worst-lag reduction
    t4 = t0 + min(750, budget_s * 0.68)
    best_h, best_c5 = _targeted(best_h, dx, rng, t4)

    # Phase 5: SLSQP after targeted
    if rem() > 200:
        for a in [30000, 120000]:
            if rem() < 150:
                break
            hc, c = _slsqp(best_h, dx, a, 500)
            if c < best_c5:
                best_h, best_c5 = hc, c

    # Phase 6: Second multi-start round
    t6 = t0 + min(920, budget_s * 0.83)
    for r in range(12):
        if time.time() > t6 or rem() < 160:
            break
        hp = _perturb(best_h, rng, rng.randint(6))
        hp, c = _slsqp(hp, dx, 15000, 350)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 7: Fine local search
    t7 = t0 + budget_s - 40
    best_h, best_c5 = _fine(best_h, dx, rng, t7)

    # Phase 8: Final SLSQP polish
    if rem() > 30:
        hc, c = _slsqp(best_h, dx, 80000, 400)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # Final micro-search
    if rem() > 10:
        best_h, best_c5 = _fine(best_h, dx, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, n / 2.0)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
