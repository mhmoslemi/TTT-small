import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    # Clip and fix sum
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smooth_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _obj_lbfgs(h, alpha, lam):
    n = len(h)
    c = _corr(h)
    return _smooth_max(c, alpha) + lam * (h.sum() - n / 2.0) ** 2

def _grad_lbfgs(h, alpha, lam):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s_shifted = s - s.max()
    w = np.exp(s_shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

                h_try = h.copy()
                h_try[start:end] = h_try_blk
                h_try = _repair(h_try, n)
                c5n = _eval(h_try)
                if c5n < c5 - 1e-15:
                    h, c5 = h_try, c5n
                    h_blk = h[start:end].copy()
                    break
                step *= 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase A: L-BFGS-B with penalty at base resolution
    for alpha, lam in [(500, 5.0), (3000, 20.0), (15000, 80.0), (60000, 300.0)]:
        if _el() > 120:
            break
        h_new = _lbfgs_opt(best_h, alpha, lam, 800)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase B: SLSQP polish with high alpha
    for alpha in [30000, 100000, 300000]:
        if _el() > 200:
            break
        h_new = _slsqp_opt(best_h, alpha, 2000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase C: Multi-start with diverse perturbations
    if _el() < budget_s - 500:
        for _r in range(20):
            if _el() > budget_s - 400:
                break
            strat = rng.randint(8)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n0), n0)
            elif strat == 2:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, n0 // 6))
                st = rng.randint(0, n0 - bs)
                hp[st:st + bs] += 0.06 * rng.randn(bs)
                hp = _repair(hp, n0)
            elif strat == 3:
                sh = rng.randint(1, max(2, n0 // 10))
                hp = _repair(np.roll(best_h, sh), n0)
            elif strat == 4:
                hp = best_h.copy()
                t_arr = np.arange(n0)
                for _f in range(5):
                    f = rng.randint(1, n0 // 3 + 1)
                    hp += 0.01 * rng.randn() * np.sin(2 * np.pi * f * t_arr / n0)
                hp = _repair(hp, n0)
            elif strat == 5:
                hp = best_h.copy()
                mask = rng.rand(n0) < 0.2
                hp[mask] = np.clip(hp[mask] + 0.1 * rng.randn(mask.sum()), 0, 1)
                hp = _repair(hp, n0)
            elif strat == 6:
                hp = 0.5 + 0.25 * np.sin(2 * np.pi * rng.randint(1, 5) * np.arange(n0) / n0)
                hp = _repair(hp, n0)
            else:
                hp = 0.5 + 0.2 * rng.randn(n0)
                hp = _repair(hp, n0)
            alpha_pick = [5000, 30000, 100000][rng.randint(3)]
            lam_pick = [10.0, 50.0, 200.0][rng.randint(3)]
            h_new = _lbfgs_opt(hp, alpha_pick, lam_pick, 400)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase D: Top-k subgradient descent
    if _el() < budget_s - 250:
        for topk in [3, 5, 8]:
            if _el() > budget_s - 200:
                break
            h_new = _topk_descent(best_h, 300, topk, 0.005)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new

    # Phase E: Fine pairwise search with adaptive scale
    if _el() < budget_s - 150:
        for scale in [0.004, 0.001, 0.0003, 0.00008]:
            if _el() > budget_s - 100:
                break
            h_new = _fine_search(best_h.copy(), rng, 6000, scale)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new

    # Phase F: Upsample to 2x and optimize
    if _el() < budget_s - 200:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha, lam in [(5000, 20.0), (30000, 100.0), (100000, 400.0)]:
            if _el() > budget_s - 120:
                break
            h_hi = _lbfgs_opt(h_hi, alpha, lam, 500)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if _el() < budget_s - 80:
            h_hi = _slsqp_opt(best_h, 150000, 1500)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), len(h_hi)
        if _el() < budget_s - 40:
            for scale in [0.002, 0.0005, 0.0001]:
                if _el() > budget_s - 35:
                    break
                h_new = _fine_search(best_h.copy(), rng, 4000, scale)
                c5 = _eval(h_new)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_new

    # Phase G: Final SLSQP polish
    if _el() < budget_s - 20:
        h_new = _slsqp_opt(best_h, 500000, 2000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    # Phase H: Ultra-fine search
    if _el() < budget_s - 3:
        h_new = _fine_search(best_h.copy(), rng, 15000, 3e-5)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
