Let me also think about the block move. If we move a block of m consecutive 1s from position a to position b (replacing m consecutive 0s), the change in c_s is:
Δc_s = ∑_{r=0}^{m-1} [contribution from h[a+r]: 1→0] + [contribution from h[b+r]: 0→1]

Each position contributes O(1) to each s, so total is O(m) per s, or O(m * (2n)) total. For m=1, it's the same as a pair swap. For m=2,3, it's slightly more expensive but still fast.

Actually, for the swap to be valid (preserving the sum), we need to swap one 1 with one 0. A block move of m 1s to a position of m 0s is equivalent to m simultaneous swaps. The total Δc is the sum of individual swaps.

from scipy.optimize import minimize
import time

                    h[idx] += a
                    diff -= a
                else:

                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

def _corr(h):

def _binarize(h, n):
    # Convert to binary while preserving sum = n/2
    h_sorted = np.zeros(n)
    order = np.argsort(h)[::-1]
    h_sorted[order[:n//2]] = 1.0
    return h_sorted  # Wait, this loses the original ordering
    
def _to_binary(h, n):
    # Threshold at 0.5, then fix count
    b = (h >= 0.5).astype(float)
    cnt = b.sum()
    target = n // 2
    if cnt > target:
        # Too many 1s, flip some 1s to 0s (the ones with smallest h value)
        ones = np.where(b == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for idx in ones_sorted[:cnt - target]:
            b[idx] = 0
    elif cnt < target:
        zeros = np.where(b == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        for idx in zeros_sorted[:target - cnt]:
            b[idx] = 1
    return b

def _swap_delta(h, i, j, c, s_vals):
    """Compute delta in correlation when h[i]:1->0 and h[j]:0->1.
    s_vals[k] = k - (n-1), the actual shift for lag k.
    Returns delta_c array of length 2n-1."""
    n = len(h)
    nc = 2 * n - 1
    delta = np.zeros(nc)
    
    # h[i] goes 1->0
    # For each lag s, the contribution changes:
    # -1[i in A_s] + h[i+s][i in A_s] + h[i-s][i-s in A_s]
    # [i in A_s] = [0<=i<n and 0<=i+s<n] = [0<=i+s<n] (since i in [0,n))
    # [i-s in A_s] = [0<=i-s<n and 0<=i<n] = [i-n < s <= i-1] (roughly)
    
    for k in range(nc):
        s = k - (n - 1)
        contrib = 0.0
        # i in A_s: 0 <= i+s < n
        if 0 <= i + s < n:
            contrib += -1.0 + h[i + s]
        # i-s in A_s: 0 <= i-s < n (and i in [0,n) which is true)
        if 0 <= i - s < n:
            contrib += h[i - s]
        
        # j in A_s: 0 <= j+s < n
        if 0 <= j + s < n:
            contrib += 1.0 - h[j + s]
        # j-s in A_s
        if 0 <= j - s < n:
            contrib += -h[j - s]
        
        delta[k] = contrib
    
    return delta
```

For a given i, the set of s where i∈A_s is s ∈ [-i, n-i-1].
The set of s where i-s∈A_s is s ∈ [i-(n-1), i] (i.e., i-n < i-s ≤ i-1, s > i-n and s ≤ i).

Wait, i-s ∈ A_s means 0 ≤ i-s < n AND 0 ≤ i < n. The second is automatic. So 0 ≤ i-s < n means s ≤ i and s > i-n.

So the range of s for position i is:
- As the "first" element: s ∈ [-i, n-i-1] (length n, clipped to [-(n-1), n-1])
- As the "second" element: s ∈ [i-n+1, i-1] (length n-1, clipped to [-(n-1), n-1])

The union is s ∈ [-(n-1), n-1] (all lags), but the specific contributions vary.

def _swap_delta_vec(h, i, j):
    n = len(h)
    s = np.arange(-(n-1), n)  # all shifts
    nc = 2*n - 1
    
    # i as first element: valid when 0 <= i+s < n
    mask_i1 = (i + s >= 0) & (i + s < n)
    # i as second element: valid when 0 <= i-s < n
    mask_i2 = (i - s >= 0) & (i - s < n)
    
    # j as first element
    mask_j1 = (j + s >= 0) & (j + s < n)
    # j as second element
    mask_j2 = (j - s >= 0) & (j - s < n)
    
    delta = np.zeros(nc)
    delta[mask_i1] += -1.0 + h[i + s[mask_i1]]
    delta[mask_i2] += h[i - s[mask_i2]]
    delta[mask_j1] += 1.0 - h[j + s[mask_j1]]
    delta[mask_j2] += -h[j - s[mask_j2]]
    
    return delta
```

This is O(n) vectorized, much faster than a Python loop.

Actually, for binary h (values 0 or 1), the formula simplifies:
- If h[i]=1, h[i]=0 after swap
- delta from i (1→0): -[i∈A_s] + h[i+s]*[i∈A_s] + h[i-s]*[i-s∈A_s]
  For binary: -[i∈A_s] + h[i+s]*[i∈A_s] + h[i-s]*[i-s∈A_s]
  = h[i+s]*[i∈A_s] - [i∈A_s] + h[i-s]*[i-s∈A_s]
  = (h[i+s]-1)*[i∈A_s] + h[i-s]*[i-s∈A_s]
  = -[i∈A_s and i+s is 0] + [i-s ∈ A_s and h[i-s]=1]
  For binary h:
  = -[i∈A_s and h[i+s]=0] + [i-s∈A_s and h[i-s]=1]

Similarly for j (0→1):
= [j∈A_s] - h[j+s]*[j∈A_s] - h[j-s]*[j-s∈A_s]
= [j∈A_s and h[j+s]=0] - [j-s∈A_s and h[j-s]=1]

So the total delta is:
= [j∈A_s and h[j+s]=0] - [j-s∈A_s and h[j-s]=1] - [i∈A_s and h[i+s]=0] + [i-s∈A_s and h[i-s]=1]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _to_binary(h, n):
    b = (h >= 0.5).astype(np.float64)
    cnt = int(b.sum())
    tgt = n // 2
    if cnt == tgt: return b
    if cnt > tgt:
        ones = np.where(b == 1)[0]
        for idx in ones[np.argsort(h[ones])][:cnt - tgt]: b[idx] = 0
    else:
        zeros = np.where(b == 0)[0]
        for idx in zeros[np.argsort(-h[zeros])][:tgt - cnt]: b[idx] = 1
    return b

def _swap_delta(h, i, j, s_all):
    n = len(h)
    m1 = (i + s_all >= 0) & (i + s_all < n)
    m2 = (i - s_all >= 0) & (i - s_all < n)
    m3 = (j + s_all >= 0) & (j + s_all < n)
    m4 = (j - s_all >= 0) & (j - s_all < n)
    d = np.zeros(len(s_all))
    d[m1] += h[i + s_all[m1]] - 1.0
    d[m2] += h[i - s_all[m2]]
    d[m3] += 1.0 - h[j + s_all[m3]]
    d[m4] += -h[j - s_all[m4]]
    return d

def _anneal_binary(b, n, T_init, T_min, n_swaps, rng, deadline):
    s_all = np.arange(-(n-1), n)
    c = np.correlate(b, 1.0 - b, mode='full')
    c5 = float(c.max() * 2.0 / n)
    best_c = b.copy()
    best_c5 = c5
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for it in range(n_swaps):
        if time.time() > deadline: break
        T = T_init * (T_min / T_init) ** (it / max(n_swaps, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j: continue
        d = _swap_delta(b, i, j, s_all)
        new_c = c + d
        new_c5 = float(new_c.max() * 2.0 / n)
        accept = new_c5 < c5 or (rng.rand() < np.exp(-(new_c5 - c5) / max(T, 1e-15)))
        if accept:
            b[i] = 0; b[j] = 1
            c = new_c
            c5 = new_c5
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_c = b.copy()
    return best_c, best_c5

def _exhaustive_swap(b, n, rounds=5):
    s_all = np.arange(-(n-1), n)
    c = np.correlate(b, 1.0 - b, mode='full')
    c5 = float(c.max() * 2.0 / n)
    for _ in range(rounds):
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        for i in ones:
            for j in zeros:
                d = _swap_delta(b, i, j, s_all)
                new_c5 = float((c + d).max() * 2.0 / n)
                if new_c5 < c5 - 1e-15:
                    b[i] = 0; b[j] = 1
                    c += d
                    c5 = new_c5
                    improved = True
                    zeros = np.where(b == 0)[0]
                    # Note: ones set changes too, but we'll recompute next round
        if not improved: break
    return b, c5

def _slsqp_spm(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15: continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)

        if c5_ex < best_c5:
            best_c5 = c5_ex
            best_h = b2.copy()
    
    # Phase 3: Multi-start with different n
    for n_try in [n0 - 10, n0, n0 + 10, n0 + 20, n0 - 20]:
        if time.time() > t0 + budget_s * 0.7: break
        if n_try < 50 or n_try % 2 != 0: continue
        # Resample or create new
        if n_try == n0:
            b_try = best_h.copy()
        else:
            xo = np.linspace(0, 2.0, n0, endpoint=False)
            xn = np.linspace(0, 2.0, n_try, endpoint=False)
            b_try = _to_binary(np.interp(xn, xo, best_h), n_try)
        dl = t0 + budget_s * 0.7
        b_t, c5_t = _anneal_binary(b_try, n_try, 0.005, 1e-7, 5_000_000, rng, dl)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = b_t.copy()
    
    # Phase 4: SLSQP continuous polish
    if time.time() < t0 + budget_s * 0.9:
        h_cont = best_h.copy()
        # Add small noise to escape binary
        h_cont += rng.randn(len(h_cont)) * 0.01
        h_cont = _repair(h_cont, len(h_cont))
        for alpha in [10000, 100000, 500000]:
            if time.time() > t0 + budget_s * 0.9: break
            h_new = _slsqp_spm(h_cont, alpha, 2000, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_cont = h_new
    
    # Phase 5: Final exhaustive swap on best

        b_fin, c5_fin = _exhaustive_swap(best_h, len(best_h), rounds=5)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = b_fin.copy()
    
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))

def _spmv_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _spmv_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_spmv_obj, h, jac=_spmv_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h = np.clip(h * (tgt / s), 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _tob(h, n):
    b = (h >= 0.5).astype(float)
    c, t = int(b.sum()), n // 2
    if c == t: return b
    if c > t:
        ix = np.where(b == 1)[0]
        for k in ix[np.argsort(h[ix])][:c - t]: b[k] = 0
    else:
        ix = np.where(b == 0)[0]
        for k in ix[np.argsort(-h[ix])][:t - c]: b[k] = 1
    return b
