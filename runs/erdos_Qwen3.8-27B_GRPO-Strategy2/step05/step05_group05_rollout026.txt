    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lag_grad(h, k_idx):
    """Gradient of the k-th correlation entry w.r.t. h."""
    n = len(h)
    sh = (n - 1) - k_idx
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _smooth_obj_topk(h, alpha, topk):
    c = _corr_scaled(h)
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad_topk(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _corr_scaled(h)
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for wk, ki in zip(w, idx):
        g += wk * dx * _lag_grad(h, ki)
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp_topk(h, alpha, topk, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_smooth_obj_topk, h, jac=_smooth_grad_topk,
                       args=(alpha, topk), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent_topk(h, max_iters, n_top, k_exp):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_iters):
        c = _corr_scaled(h)
        idx = np.argsort(c)[::-1][:n_top]
        cvals = c[idx]
        m = cvals.max()
        s = np.clip(k_exp * (cvals - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for wk, ki in zip(w, idx):
            g += wk * dx * _lag_grad(h, ki)
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _subgrad_single(h, n_iter):
    """Subgradient on the single worst lag with adaptive step."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    lr = 0.01
    for it in range(n_iter):
        c = _corr_scaled(h)
        k_star = int(np.argmax(c))
        g = dx * _lag_grad(h, k_star)
        lr_adapt = lr / (1.0 + 0.0005 * it)
        h_new = _proj_to_sum(h - lr_adapt * g, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            h = h_new
            c5c = c5n
            lr = min(lr * 1.05, 0.08)
        else:
            lr *= 0.85
    return h, c5c

def _binary_push(h, n_iter, rng):
    """Push values toward 0 or 1 to exploit binary structure."""
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        above = np.where(h > 0.52)[0]
        below = np.where(h < 0.48)[0]
        if len(above) == 0 or len(below) == 0:
            break
        i = rng.choice(above)
        j = rng.choice(below)
        d = min(rng.uniform(0.02, 0.08), 1.0 - h[i], h[j])
        if d < 1e-7:
            continue
        h[i] += d
        h[j] -= d
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i] -= d
            h[j] += d
    return h, c5c

def _pairwise_refine(h, n_iter, scale, rng):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase A: SLSQP with decreasing topk (sharper focus)
    for alpha, topk in [(500, 20), (2000, 10), (10000, 5), (50000, 3), (200000, 2)]:
        if _el() > 180:
            break
        h_new = _slsqp_topk(best_h, alpha, topk, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase B: Active set with exponential top-k weighting
    if _el() < budget_s - 450:
        for k_exp in [20, 100, 500, 2000, 10000]:
            h_new, c5 = _active_descent_topk(best_h.copy(), 3000, 6, k_exp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        h_new, c5 = _active_descent_topk(best_h.copy(), 5000, 3, 50000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase C: Subgradient single-lag descent
    if _el() < budget_s - 380:
        h_new, c5 = _subgrad_single(best_h.copy(), 4000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase D: Binary push
    if _el() < budget_s - 320:
        h_new, c5 = _binary_push(best_h.copy(), 10000, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase E: 2x upsample and optimize
    if _el() < budget_s - 400:
        n1 = n0 * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha, topk in [(2000, 10), (10000, 5), (100000, 2)]:
            if _el() > budget_s - 320:
                break
            h_hi = _slsqp_topk(h_hi, alpha, topk, 1500, 1e-20)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _el() < budget_s - 200:
            h_new, c5 = _active_descent_topk(best_h.copy(), 3000, 4, 10000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(best_h)

    # Phase F: Multi-start
    n_cur = best_n
    if _el() < budget_s - 250:
        for _r in range(12):
            if _el() > budget_s - 200:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.008 * rng.randn(n_cur), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                t = np.arange(n_cur)
                for _f in range(2):
                    f = rng.randint(1, n_cur // 4 + 1)
                    hp += 0.008 * np.sin(2 * np.pi * f * t / n_cur + rng.uniform(0, 6.28))
                hp = _repair(hp, n_cur)
            else:
                hp = best_h.copy()
                bs = max(3, n_cur // 15)
                st = rng.randint(0, n_cur - bs)
                hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            h_new = _slsqp_topk(hp, 50000, 3, 1000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase G: Fine polish - active set + subgrad + pairwise
    if _el() < budget_s - 120:
        h_new, c5 = _active_descent_topk(best_h.copy(), 5000, 3, 50000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    if _el() < budget_s - 80:
        h_new, c5 = _subgrad_single(best_h.copy(), 3000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
    if _el() < budget_s - 50:
        for scale in [0.005, 0.002, 0.001, 0.0003, 0.0001]:
            if _el() > budget_s - 20:
                break
            h_new, c5 = _pairwise_refine(best_h.copy(), 5000, scale, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase H: Final ultra-sharp SLSQP (topk=1, very high alpha)
    if _el() < budget_s - 15:
        h_new = _slsqp_topk(best_h, 5000000, 1, 5000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase I: Final active set with top-2
    if _el() < budget_s - 5:
        h_new, c5 = _active_descent_topk(best_h.copy(), 5000, 2, 200000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
