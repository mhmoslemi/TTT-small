def _subgrad(h, num_worst=5):
    """Return (c5, subgradient of max as weighted combo of worst shifts)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(c.max() * dx)
    # Find top num_worst shifts
    top_idx = np.argsort(c)[::-1][:num_worst]
    top_c = c[top_idx]
    # Weight: softmax over top values
    w = np.exp((top_c - top_c[0]) * 10)
    w /= w.sum()
    g = np.zeros(n)
    for w_k, k in zip(w, top_idx):
        sh = k - (n - 1)
        if sh >= 0:
            m = n - sh
            g[:m] += w_k * (1.0 - h[sh:sh+m])
            if sh > 0:
                g[sh:] -= w_k * h[:m]
        else:
            sp = -sh
            m = n - sp
            g[sp:] += w_k * (1.0 - h[:m])
            g[:m] -= w_k * h[sp:]
    g *= dx
    return c5_val, g

def _shift_grad(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:sh + m])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        m = n - sp
        g[sp:] += (1.0 - h[:m])
        g[:m] -= h[sp:]
    return g

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    top = np.argsort(c)[::-1][:n_worst]
    tc = c[top]

    w /= w.sum()
    g = np.zeros(n)
    for wk, k in zip(w, top):
        sh = k - (n - 1)
        g += wk * _shift_grad(h, sh, n)
    return c5_val, g * dx

        sh = k - (n - 1)
        g += w[k] * _shift_grad(h, sh, n)
