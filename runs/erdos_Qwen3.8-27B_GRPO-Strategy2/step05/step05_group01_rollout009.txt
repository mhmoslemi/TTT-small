import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            g[i2] -= w[k] * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pos_change(h, i, di, n):
    dc = np.zeros(2 * n - 1)
    if di == 0.0:
        return dc
    sh1 = np.arange(0, n - i)
    k1 = (n - 1) - sh1
    dc[k1] += di * (1.0 - h[i:i + n - i])
    sh2 = np.arange(0, i + 1)
    k2 = (n - 1) - sh2
    dc[k2] -= di * h[i::-1][:i + 1]
    sp = np.arange(1, n)
    k3 = (n - 1) + sp
    m1 = sp <= i
    if m1.any():
        dc[k3[m1]] += di * (1.0 - h[i - sp[m1]])
    m2 = sp < (n - i)
    if m2.any():
        dc[k3[m2]] -= di * h[i + sp[m2]]
    return dc

def _incr_pair(h, c, rng, n_iter, scale, n):

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:
            hi = mid

def _proj_grad(h, alpha, nsteps, step0):

    step = step0
    for _ in range(nsteps):

            step *= 0.5
        if best_h is None:
            break
        h = best_h
        f = best_f

            h_ls, c_ls, _ = _incr_pair(h_ls, c_ls, rng, n_it, scale, n_ls)
        h_ls = _repair(h_ls, n_ls)
        c5 = _eval(h_ls)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_ls.copy(), n_ls

    # Phase 5: Greedy local search (semi-greedy partner selection)
    if _el() < budget_s - 60:
        h_gr = best_h.copy()
        c_gr = _corr(h_gr)
        n_gr = len(h_gr)
        for scale in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            if _el() > budget_s - 60:
                break
            n_it = 15000 if scale > 0.001 else 25000
            h_gr, c_gr, _ = _incr_greedy(h_gr, c_gr, rng, n_it, scale, n_gr)
        h_gr = _repair(h_gr, n_gr)
        c5 = _eval(h_gr)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_gr.copy(), n_gr

    # Phase 6: Multi-start exploration with diverse initializations
    if _el() < budget_s - 120:
        nc = len(best_h)
        for r in range(12):
            if _el() > budget_s - 120:
                break
            strat = r % 4
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                a = rng.randint(0, nc // 3)
                blen = rng.randint(nc // 10, nc // 3)
                b = min(a + blen, nc)
                c = rng.randint(b, max(b + 1, nc))
                d = min(c + (b - a), nc)
                if d <= nc:
                    hp[a:b], hp[c:d] = hp[c:d].copy(), hp[a:b].copy()
                hp = _repair(hp, nc)
            elif strat == 2:
                hp = np.clip(np.round(best_h * 6.0) / 6.0 + 0.02 * rng.randn(nc), 0, 1)
                hp = _repair(hp, nc)
            else:
                hp = best_h.copy()
                perm = rng.permutation(nc)
                k = rng.randint(nc // 8, nc // 4)
                hp[:k] = best_h[perm[:k]]
                hp = _repair(hp, nc)

            h_ms = _slsqp(hp, 300000, 800, 1e-19)
            c_ms = _corr(h_ms)
            h_ms, c_ms, _ = _incr_pair(h_ms, c_ms, rng, 40000, 0.004, nc)
            h_ms, c_ms, _ = _incr_pair(h_ms, c_ms, rng, 30000, 0.001, nc)
            h_ms = _repair(h_ms, nc)
            c5 = _eval(h_ms)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_ms.copy(), nc

    # Phase 7: SLSQP polish at very high alpha
    if _el() < budget_s - 40:
        h_pol = _slsqp(best_h, 1500000, 2000, 1e-20)
        c5 = _eval(h_pol)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_pol.copy(), len(h_pol)

    # Phase 8: Ultra-fine incremental search
    if _el() < budget_s - 10:
        h_fine = best_h.copy()
        c_fine = _corr(h_fine)
        n_fine = len(h_fine)
        for scale in [5e-5, 2e-5, 8e-6, 3e-6, 1e-6]:
            if _el() > budget_s - 10:
                break
            n_it = 80000 if scale > 1e-5 else 120000
            h_fine, c_fine, _ = _incr_pair(h_fine, c_fine, rng, n_it, scale, n_fine)
        h_fine = _repair(h_fine, n_fine)
        c5 = _eval(h_fine)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fine.copy(), n_fine

    # Final sweep
    if _el() < budget_s - 3:
        h_fin = best_h.copy()
        c_fin = _corr(h_fin)
        n_fin = len(h_fin)
        h_fin, c_fin, _ = _incr_pair(h_fin, c_fin, rng, 100000, 5e-6, n_fin)
        h_fin = _repair(h_fin, n_fin)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin.copy(), n_fin

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
