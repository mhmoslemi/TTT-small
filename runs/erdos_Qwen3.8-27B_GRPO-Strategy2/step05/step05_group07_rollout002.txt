def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if abs(diff) < 1e-12:
                    break
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
    return h

            if f_new < f - 1e-13:
                h = h_new
                f = f_new
                found = True
                break
            step *= 0.5
        if not found:
            step = min(step * 1.1, step0)
        else:
            step = min(step * 1.03, step0)
    return h, f

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _cfun(x):
        return x.sum() - n * 0.5

    def _cjac(x):
        return np.ones(len(x))

    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, alpha),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    x_old = np.linspace(0, 2, n)
    x_new = np.linspace(0, 2, n * factor)
    f = interp1d(x_old, h, kind='cubic', fill_value=(h[0], h[-1]), bounds_error=False)
    return _repair(np.clip(f(x_new), 0, 1), n * factor)

def _coord_descent(h, n_pass, n_iter):
    n = len(h)
    target = n * 0.5
    h = h.copy().astype(float)
    f = _eval(h)
    for p in range(n_pass):
        if time.time() - _t0 > _budget - 15:
            break
        for i in range(n):
            if n_iter and time.time() - _t0 > _budget - 12:
                break
            v0 = h[i]
            for dv in np.linspace(-0.03, 0.03, 13):
                if dv == 0:
                    continue
                h[i] = np.clip(v0 + dv, 0, 1)
                fn = _eval(h)
                if fn < f - 1e-14:
                    f = fn
                    break
            h[i] = v0
        h = _repair(h, n)
        f = _eval(h)
    return h, f

def _slsqp_run_wrapped(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        return _smooth_obj(x, alpha)
    def _jac(x):
        return _smooth_grad(x, alpha)
    def _cfun(x):
        return x.sum() - n * 0.5
    def _cjac(x):
        return np.ones(len(x))
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _budget
    _t0 = time.time()
    _budget = budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _el():
        return time.time() - _t0

    # Phase 1: SLSQP at n0 with increasing alpha
    for alpha in [3000, 20000, 100000]:
        if _el() > 60:
            break
        hn = _slsqp_run_wrapped(best_h, alpha, 2000, 1e-17)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 2: Momentum PG at n0
    if _el() < _budget - 200:
        for alpha, nit in [(5000, 800), (30000, 600)]:
            if _el() > _budget - 180:
                break
            hn, fc = _momentum_pg(best_h, alpha, nit, 0.9, 0.008)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 3: Active-set PG at n0
    if _el() < _budget - 180:
        for eps in [0.01, 0.005, 0.002]:
            if _el() > _budget - 160:
                break
            hn, fc = _active_pg(best_h, eps, 1500, 0.005)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 4: Upsample to 2x and optimize
    n1 = n0 * 2
    h2 = _upsample(best_h, 2)
    c5_2 = _eval(h2)
    if c5_2 < best_c5 + 0.002:
        best_h = h2.copy()
        best_n = n1
        best_c5 = c5_2

    if _el() < _budget - 200:
        for alpha in [10000, 60000]:
            if _el() > _budget - 180:
                break
            hn = _slsqp_run_wrapped(best_h, alpha, 1500, 1e-17)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn.copy()
                best_n = len(hn)

    # Phase 5: Momentum PG at higher resolution
    if _el() < _budget - 150:
        for alpha, nit in [(5000, 600), (40000, 500)]:
            if _el() > _budget - 130:
                break
            hn, fc = _momentum_pg(best_h, alpha, nit, 0.85, 0.006)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 6: Active PG at higher resolution
    if _el() < _budget - 120:
        for eps, nit in [(0.01, 2000), (0.003, 1500)]:
            if _el() > _budget - 100:
                break
            hn, fc = _active_pg(best_h, eps, nit, 0.004)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 7: High-alpha SLSQP polish
    if _el() < _budget - 80:
        hn = _slsqp_run_wrapped(best_h, 200000, 2000, 1e-18)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 8: Multi-start perturbations
    nc = len(best_h)
    if _el() < _budget - 60:
        n_starts = 8
        for _r in range(n_starts):
            if _el() > _budget - 45:
                break
            scale = 0.006 + 0.003 * _r
            hp = best_h + scale * rng.randn(nc)
            hp = _repair(hp, nc)
            hn, fc = _momentum_pg(hp, 30000, 400, 0.85, 0.005)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()
            hn, fc = _active_pg(hn, 0.005, 1000, 0.003)
            if fc < best_c5:
                best_c5 = fc
                best_h = hn.copy()

    # Phase 9: Final active PG + SLSQP
    if _el() < _budget - 30:
        hn, fc = _active_pg(best_h, 0.001, 2000, 0.002)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    if _el() < _budget - 15:
        hn = _slsqp_run_wrapped(best_h, 500000, 1500, 1e-19)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()

    # Phase 10: Final tiny active PG polish
    if _el() < _budget - 5:
        hn, fc = _active_pg(best_h, 0.0005, 3000, 0.001)
        if fc < best_c5:
            best_c5 = fc
            best_h = hn.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
