                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1) - k; wk = w[k]*dx
        if sh >= 0:
            i1 = np.arange(n-sh); g[i1] += wk*(1.0-x[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n); g[i2] -= wk*x[i2-sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n); g[i1] += wk*(1.0-x[i1-sp])
            i2 = np.arange(n-sp); g[i2] -= wk*x[i2+sp]
    return g

def _sum_eq(x): return x.sum() - len(x)*0.5
def _sum_jac(x): return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol=1e-20):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun':_sum_eq, 'jac':_sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons, options={'maxiter':maxiter, 'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _to_binary(h, target):
    h_bin = (h >= 0.5).astype(float)
    ones = np.where(h_bin==1)[0]; zeros = np.where(h_bin==0)[0]
    if len(ones) > target:
        h_bin[ones[np.argsort(h[ones])[:len(ones)-target]]] = 0
    elif len(ones) < target:
        h_bin[zeros[np.argsort(h[zeros])[::-1]][:target-len(ones)]] = 1
    return h_bin

def _build_fv(h, n, pos):
    m_len = 2*n - 1
    v = np.zeros(m_len)
    lo, hi = pos, pos + n
    v[lo:hi] = 2.0*h[lo-pos:hi-pos] - 1.0
    v[lo:hi] = 2.0*h - 1.0
    v[lo:hi] = (2.0*h[np.arange(lo,hi)-pos] - 1.0)
    if 2*pos >= lo and 2*pos < hi:
        v[2*pos] = 0.0
    return v

def _binary_il_search(h, n, t0, time_limit, rng):
    target = n//2
    m_len = 2*n - 1
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    cur_max = c.max()
    best_c5 = cur_max * 2.0/n
    best_h = h.copy()
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    n_ones = len(ones); n_zeros = len(zeros)
    
    fhm1 = 2.0*h - 1.0
    
    def make_fv(pos):
        v = np.zeros(m_len)
        lo, hi = pos, pos+n
        v[lo:hi] = fhm1[lo-pos:hi-pos]
        if 2*pos >= lo and 2*pos < hi:
            v[2*pos] = 0.0
        return v
    
    FV = np.zeros((2*n, m_len))
    for ii in range(n_ones):
        FV[ii] = make_fv(ones[ii])
    for jj in range(n_zeros):
        FV[n_ones+jj] = -(make_fv(zeros[jj]))
    
    FV_ones = FV[:n_ones]
    FV_zeros = FV[n_ones:]
    
    stall = 0
    max_stall = 5000
    
    for it in range(50000000):
        if time.time() - t0 > time_limit:
            break
        if stall > max_stall:
            # Reheat: random kick
            for _ in range(rng.randint(5, 30)):
                ii = rng.randint(n_ones); jj = rng.randint(n_zeros)
                ip, jp = ones[ii], zeros[jj]
                h[ip]=0; h[jp]=1
                ones[ii]=jp; zeros[jj]=ip
                fhm1[ip]=-1; fhm1[jp]=1
            c = np.correlate(h, 1-h, mode='full')
            cur_max = c.max()
            for ii in range(n_ones): FV_ones[ii] = make_fv(ones[ii])
            for jj in range(n_zeros): FV_zeros[jj] = -(make_fv(zeros[jj]))
            stall = 0
            max_stall = int(max_stall*1.3)
            continue
        
        improved = False
        for ii in range(n_ones):
            c_i = c + FV_ones[ii]
            maxes = (c_i[None,:] + FV_zeros).max(axis=1)
            jj = maxes.argmin()
            if maxes[jj] < cur_max - 1e-12:
                ip = ones[ii]; jp = zeros[jj]
                c = c_i + FV_zeros[jj]
                cur_max = maxes[jj]
                h[ip]=0; h[jp]=1
                ones[ii]=jp; zeros[jj]=ip
                fhm1[ip]=-1; fhm1[jp]=1
                for p in range(2*n):
                    pos = ones[p] if p < n_ones else zeros[p-n_ones]
                    if pos == ip or pos == jp:
                        continue
                    for other in [ip, jp]:
                        m_idx = pos + other
                        if m_idx >= 0 and m_idx < m_len and m_idx != 2*pos and m_idx >= pos and m_idx < pos+n:
                            FV[p][m_idx] = fhm1[other]
                            if p >= n_ones:
                                FV[p][m_idx] = -fhm1[other]
                FV_ones[ii] = make_fv(ones[ii])
                FV_zeros[jj] = -(make_fv(zeros[jj]))
                improved = True
                if cur_max * 2.0/n < best_c5:
                    best_c5 = cur_max * 2.0/n
                    best_h = h.copy()
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    
    # Phase 1: SLSQP polish (60s)
    for alpha in [5000, 50000, 200000, 1000000]:
        if time.time() - t0 > 60: break
        h_new = _slsqp(best_h, alpha, 3000)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
    
    # Phase 2: Binary ILS (500s)
    if time.time() - t0 < budget_s - 320:
        h_bin = _to_binary(best_h, n//2)
        h_best, c5_best = _binary_il_search(h_bin, n, t0, budget_s - 320, rng)
        if c5_best < best_c5:
            best_c5 = c5_best; best_h = h_best.copy()
    
    # Phase 3: SLSQP refinement (250s)
    if time.time() - t0 < budget_s - 70:
        for alpha in [10000, 100000, 500000, 2000000]:
            if time.time() - t0 > budget_s - 70: break
            h_new = _slsqp(best_h, alpha, 5000, 1e-21)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()
    
    # Phase 4: Second binary ILS from refined (150s)
    if time.time() - t0 < budget_s - 30:
        h_bin = _to_binary(best_h, n//2)
        h_best, c5_best = _binary_il_search(h_bin, n, t0, budget_s - 30, rng)
        # Also try complement
        h_comp = _to_binary(1.0-best_h, n//2)
        for hc in [h_best, h_comp]:
            c5t = _eval(hc)
            if c5t < c5_best:
                c5_best = c5t; h_best = hc.copy()
        if c5_best < best_c5:
            best_c5 = c5_best; best_h = h_best.copy()
    
    # Phase 5: Final SLSQP
    if time.time() - t0 < budget_s - 3:
        h_new = _slsqp(best_h, 5000000, 5000, 1e-21)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
    
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
