def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _mom_subgrad(h, dx, t_end, n_mom=10, top_k=8):

    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()
    v = np.zeros(n)  # momentum
    lr = 0.01
    beta = 0.9
    for it in range(2000000):
        if time.time() > t_end:
            break
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        # Softmax weights
        w = np.exp(80.0 * (c[top] - c[top].max()))
        w /= w.sum()
        # Compute gradient
        grad = np.zeros(n)

            s = abs(int(ki - (n - 1)))
            if s == 0:

            m = n - s
            g = np.zeros(n)
            g[s:n] += (1.0 - h[:m])
            g[:m] -= h[s:n]
            grad += wi * g
        # Also include lag 0 gradient for the sum constraint

def _slsqp_sm(h, dx, alpha, maxiter=500):

    tgt = n / 2.0

        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return x.sum() - tgt
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(np.clip(r.x, 0, 1), tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

    t0sa, t1sa = 0.8, 0.005
    for it in range(ntot):
        if time.time() > t_end:
            break
        t = t0sa * (t1sa / t0sa) ** (it / ntot)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.02, 0.02)
        h[i] += d; h[j] -= d
        if not (0 <= h[i] <= 1 and 0 <= h[j] <= 1):
            h[i] -= d; h[j] += d
            continue
        h = _repair(h, tgt)
        cn = _c5(h, dx)
        if cn < cur:
            cur = cn
            if cur < best:
                best = cur
        else:
            if rng.random() < np.exp(-(cn - cur) / max(t * dx, 1e-8)):
                cur = cn
            else:
                h[i] -= d; h[j] += d
                h = _repair(h, tgt)
    return h, best

def _structured_fine(h, dx, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    for _ in range(200000):
        if time.time() > t_end:
            break
        c = _corr(h)
        top = np.argsort(c)[::-1][:3]
        for ki in top:
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m = n - s
            pairs = h[:m] * (1.0 - h[s:n])
            for a in np.argsort(pairs)[::-1][:4]:
                for d in [2e-4, 1e-4, 5e-5]:
                    h[a] -= d; h[a + s] += d
                    if 0 <= h[a] <= 1 and 0 <= h[a + s] <= 1:
                        cn = _c5(h, dx)
                        if cn < best:
                            best = cn
                        else:
                            h[a] += d; h[a + s] -= d
                    else:
                        h[a] += d; h[a + s] -= d
    return h, best

def _perturb(h, rng, mode, n):
    tgt = n / 2.0
    hp = h.copy()
    if mode == 0:
        hp += 0.04 * (2.0 * rng.random(n) - 1.0)
    elif mode == 1:
        idx = rng.choice(n, max(5, n // 4), replace=False)
        hp[idx] += 0.07 * (2.0 * rng.random(len(idx)) - 1.0)
    elif mode == 2:
        x = np.linspace(0, 2, n)
        hp += 0.03 * np.sin(np.pi * int(rng.randint(2, 7)) * x + rng.uniform(0, 6.28))
    else:
        idx = rng.choice(n, max(3, n // 6), replace=False)
        hp[idx] = 1.0 - hp[idx]
    return _repair(np.clip(hp, 0, 1), tgt)

def _fine(h, dx, rng, t_end):
    n = len(h)
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.0015, 7e-4, 3e-4, 1.5e-4, 7e-5, 3e-5, 1e-5]
    for sc in scales:
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            h[i] += d; h[j] -= d
            if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                cn = _c5(h, dx)
                if cn < best:
                    best = cn
                else:
                    h[i] -= d; h[j] += d
            else:
                h[i] -= d; h[j] += d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(),
                len(initial_h_values) / 2.0)
    n = len(h)
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), _c5(h, dx)

    # P1: SLSQP ramp-up
    for alpha in [100, 300, 1000, 3000, 10000]:
        if time.time() > t0 + 90:
            break
        hc, c = _slsqp(best_h, dx, alpha, 500)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # P2: Subgradient with momentum
    if time.time() < t0 + 160:
        h, c = _subgrad(best_h, dx, t0 + 160, top_k=10)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P3: Coordinate descent
    if time.time() < t0 + 260:
        h, c = _coord_descent(best_h, dx, rng, t0 + 260, n_top=7)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P4: SLSQP high-alpha
    for alpha in [10000, 40000, 100000]:
        if time.time() > t0 + 340:
            break
        hc, c = _slsqp(best_h, dx, alpha, 400)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # P5: Simulated annealing
    if time.time() < t0 + 430:
        h, c = _sim_anneal(best_h, dx, rng, t0 + 430)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P6: Multi-start
    t_end6 = t0 + 570
    for r in range(12):
        if time.time() > t_end6:
            break
        hp = _perturb(best_h, rng, r % 4, n)
        hp, c = _slsqp(hp, dx, 3000, 250)
        hp, c = _coord_descent(hp, dx, rng, min(t_end6 - 1, time.time() + 10), n_top=5)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # P7: SLSQP polish
    for alpha in [30000, 100000, 300000]:
        if time.time() > t0 + 650:
            break
        hc, c = _slsqp(best_h, dx, alpha, 400)
        if c < best_c5:
            best_h, best_c5 = hc, c

    # P8: Extended coord descent
    if time.time() < t0 + 750:
        h, c = _coord_descent(best_h, dx, rng, t0 + 750, n_top=8)
        if c < best_c5:
            best_h, best_c5 = h, c

    # P9: Structured fine + random fine
    if time.time() < t0 + 900:
        best_h, best_c5 = _structured_fine(best_h, dx, t0 + 900)
    best_h, best_c5 = _fine(best_h, dx, rng, t0 + 1050)

    # P10: Final SLSQP + micro
    if budget_s - (time.time() - t0) > 20:
        hc, c = _slsqp(best_h, dx, 500000, 300)
        if c < best_c5:
            best_h, best_c5 = hc, c
    if budget_s - (time.time() - t0) > 5:
        best_h, best_c5 = _fine(best_h, dx, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, n / 2.0)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
