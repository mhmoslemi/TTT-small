def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _adam_subgrad(h, dx, t_end, n_lags=10):
    n = len(h)
    tgt = n / 2.0
    best = _c5(h, dx)
    h = h.copy()

    va = np.zeros(n)

    lr = 0.015
    t = 0
    while time.time() < t_end:
        t += 1
        c = _corr(h)
        top = np.argsort(c)[::-1][:n_lags]
        cmax = c[top].max()
        w = np.exp(80.0 * (c[top] - cmax))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            s = abs(int(ki - (n - 1)))
            if s == 0:
                continue
            m_idx = n - s
            g = np.zeros(n)
            g[s:] += 1.0 - h[:m_idx]
            g[:m_idx] -= h[s:]
            grad += wi * g
        grad -= grad.mean()
        ma = b1 * ma + (1 - b1) * grad
        va = b2 * va + (1 - b2) * grad**2
        mhat = ma / (1 - b1**t)

        m = c.max()

        return m + float(np.log(np.sum(np.exp(s)))) / alpha

        return x.sum() - tgt
    try:
        r = minimize(_obj, h, method='SLSQP', bounds=bds,
                     constraints=[{'type': 'eq', 'fun': _eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, tgt)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

    return (best_h, best_c5, n)
