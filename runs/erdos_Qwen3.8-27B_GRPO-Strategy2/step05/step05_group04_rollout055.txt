import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        sp = -shift
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _pobj(h, p):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    cp = np.clip(c, 0, None)
    sp = float(np.sum(cp ** p))
    return sp ** (1.0 / p)

def _pgrad(h, p):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    cp = np.clip(c, 0, None)
    sp = float(np.sum(cp ** p))
    if sp < 1e-30:
        return np.zeros(n)
    coeff = sp ** (1.0 / p - 1.0)
    w = cp ** (p - 1.0) * coeff
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            g[shift:] += w[k] * dx * (1.0 - h[:n - shift])
            g[:n - shift] -= w[k] * dx * h[shift:]
        else:
            sp_ = -shift
            g[sp_:] += w[k] * dx * (1.0 - h[:n - sp_])
            g[:n - sp_] -= w[k] * dx * h[sp_:]
    return g

def _lsqeq_fun(x, n):
    return x.sum() - n * 0.5

def _lsqeq_jac(x, n):
    return np.ones(n)

def _slsqp(h, p, maxiter, dt):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _lsqeq_fun(x, n), 'jac': lambda x: _lsqeq_jac(x, n)}]
    try:
        res = minimize(_pobj, h, jac=_pgrad, args=(p,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': dt})
        return res.x
    except Exception:
        return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _worst_lag_descent(h, max_rounds):
    n = len(h)
    target = n * 0.5
    c5 = _eval(h)
    for _ in range(max_rounds):
        c = _corr(h) * (2.0 / n)
        k = int(c.argmax())
        g = _grad_lag(h, k)
        improved = False
        for step in (0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002):
            h_new = _proj_sum(h - step * g, target)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            # Try multi-lag: top 3 lags
            top_k = np.argsort(c)[-3:]
            g_multi = np.zeros(n)
            for kk in top_k:
                g_multi += _grad_lag(h, int(kk))
            for step in (0.02, 0.01, 0.005, 0.001):
                h_new = _proj_sum(h - step * g_multi, target)
                c5n = _eval(h_new)
                if c5n < c5 - 1e-15:
                    h = h_new
                    c5 = c5n
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def _binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) > 0.5).astype(float)
    return _proj_sum(h, n * 0.5)

def _harmonic_init(n, freq, seed):
    rng = np.random.RandomState(seed)
    t = np.arange(n)
    phase = rng.rand() * 2.0 * np.pi
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t / n + phase)
    return _proj_sum(h, n * 0.5)

def _multi_harm_init(n, seed):
    rng = np.random.RandomState(seed)
    t = np.arange(n)
    h = np.zeros(n)
    for f in range(1, 6):
        h += 0.15 * np.sin(2.0 * np.pi * f * t / n + rng.rand() * 6.28)
    h += 0.5
    return _proj_sum(h, n * 0.5)
