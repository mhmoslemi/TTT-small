import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_c_k(h, k):
    """Gradient of c_k with respect to h, where k is the lag index in full correlation."""
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _proj_sum(h, target):
    """Project h to [0,1]^n with sum=target using bisection."""
    lo, hi = -2.0, 2.0

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:

        if 0 <= j+sh < n:
            cj += (1.0 - h[j+sh])
        if 0 <= j-sh < n:
            cj -= h[j-sh]
        delta[k] = ci - cj
    return delta

def _lse_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2*n-1):

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _slsqp(h, alpha, maxiter, ftol, bnds):
    n = len(h)
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def _delta_vec(h, i, j):
    n = len(h)
    sh = np.arange(n - 1, -n, -1).astype(np.int64)
    delta = np.zeros(2 * n - 1)
    ip = i + sh
    m = (ip >= 0) & (ip < n)
    delta[m] += 1.0 - h[ip[m]]
    im = i - sh
    m = (im >= 0) & (im < n)
    delta[m] -= h[im[m]]
    jp = j + sh
    m = (jp >= 0) & (jp < n)
    delta[m] -= 1.0 - h[jp[m]]
    jm = j - sh
    m = (jm >= 0) & (jm < n)
    delta[m] += h[jm[m]]
    return delta

def _lse_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fast_pairwise(h, n, budget_end, rng, max_iters=200000):
    c = _corr(h)
    c5 = c.max() * 2.0 / n
    h = h.copy()
    scales = [0.004, 0.0015, 0.0005, 0.00015, 0.00005]
    total_iters = 0
    for scale in scales:
        its = max_iters // len(scales)
        for _it in range(its):
            if time.time() > budget_end:
                return h, c5
            total_iters += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.rand() - 1.0)
            if abs(d) < 1e-15:
                continue
            if h[i] + d < 0 or h[i] + d > 1:
                continue
            if h[j] - d < 0 or h[j] - d > 1:
                continue
            delta = _delta_vec(h, i, j)
            new_max = np.max(c + d * delta)
            if new_max < c5 * n / 2.0 - 1e-12:
                h[i] += d
                h[j] -= d
                c += d * delta
                c5 = new_max * 2.0 / n
    return h, c5

def _subgrad_pass(h, n, budget_end, rng, n_rounds=200):
    h = h.copy()
    c5 = _eval(h)
    dx = 2.0 / n
    for _r in range(n_rounds):
        if time.time() > budget_end:
            break
        c = _corr(h)
        top5 = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        for k in top5:
            g += _lag_grad(h, n, k)
        g *= dx / len(top5)
        g -= g.mean()
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = _proj_sum(h - step * g, n / 2)
            c5_try = _eval(h_try)
            if c5_try < c5 - 1e-15:
                h = h_try
                c5 = c5_try
                improved = True
                break
        if not improved:
            h = _repair(h + 0.002 * rng.randn(n), n)
            c5 = _eval(h)
    return h, c5

def _lag_grad(h, n, k):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)

    best_c5 = _eval(best_h)

    for alpha in [500, 3000, 20000, 80000]:

        if c5 < best_c5:

            best_h = h_new.copy()

    # Phase 2: Fast pairwise search (main workhorse)

    # Phase 3: Subgradient descent
    if time.time() - t0 < budget_s - 250:
        best_h, c5 = _subgrad_pass(best_h, best_n, budget_s - 220, rng, n_rounds=150)
        if c5 < best_c5:

    # Phase 4: Another SLSQP polish

    # Phase 5: More pairwise with smaller steps
    if time.time() - t0 < budget_s - 150:

    # Phase 6: Upsample to 2x and optimize
    n1 = best_n * 2
    if time.time() - t0 < budget_s - 250:
