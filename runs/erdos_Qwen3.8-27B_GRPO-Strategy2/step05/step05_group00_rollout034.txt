def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _logsumexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _logsumexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter=3000, ftol=1e-18):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_sa(h, n_trials_per_round, n_rounds, rng, T0=0.05, T1=0.001):
    """Simulated annealing on binary swaps."""
    n = len(h)
    h = (h > 0.5).astype(float)
    # Fix to exactly n/2 ones
    ones_count = h.sum()
    if ones_count > n/2:
        idx = np.where(h > 0.5)[0]
        for i in idx[:int(ones_count - n/2)]:
            h[i] = 0
    elif ones_count < n/2:
        idx = np.where(h < 0.5)[0]
        for i in idx[:int(n/2 - ones_count)]:
            h[i] = 1
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    for rnd in range(n_rounds):
        T = T0 * (T1 / T0) ** (rnd / max(n_rounds - 1, 1))
        n_accept = 0
        for _ in range(n_trials_per_round):
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] != 0 or h[j] != 1 or i == j:
                continue
            h[i], h[j] = 1.0, 0.0
            c5_new = _eval(h)
            delta = c5_new - c5
            if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-9)):
                c5 = c5_new
                n_accept += 1
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = 0.0, 1.0
        if n_accept == 0:
            break
    return best_h, best_c5

def _central_descent(h, n_rounds=200, rng=None):
    """Coordinate descent focusing on worst shifts."""
    n = len(h)
    c5 = _eval(h)
    dx = 2.0 / n
    for _ in range(n_rounds):
        c = np.correlate(h, 1.0 - h, mode='full')
        # Focus on the top shifts
        k_max = np.argmax(c)
        sh = (n - 1) - k_max
        if sh >= 0:
            i1 = np.arange(n - sh)
            g = (1.0 - h[i1 + sh]).copy()
            g2 = -np.zeros(n)
            g2[i1 += 0]  # just for shape
            full_g = np.zeros(n)
            full_g[:n-sh] += (1.0 - h[sh:]) 
            full_g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            full_g = np.zeros(n)
            full_g[sp:] += (1.0 - h[:n-sp])
            full_g[:n-sp] -= h[sp:]
        # Take a small step
        for step in [0.01, 0.005, 0.002, 0.001]:
            h_new = _repair(h - step * full_g * dx, n)
            c5_new = _eval(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                break
    return h, c5

def _tile_h(pattern, reps):
    return np.tile(pattern, reps)

def _periodic_search(n, T, alpha=50000):
    """Optimize a T-length pattern tiled to length n."""
    if n % T != 0:
        return None
    reps = n // T
    # Try binary patterns
    from itertools import combinations
    hw = T // 2
    best_c5 = 1.0
    best_pat = None
    positions = list(range(T))
    for combo in combinations(positions, hw):
        pat = np.zeros(T)
        pat[list(combo)] = 1.0
        h = _tile_h(pat, reps)
        if h.sum() != n / 2:
            continue
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_pat = pat
    if best_pat is None:
        return None
    h = _tile_h(best_pat, reps)
    # Polish with SLSQP
    h = _slsqp(h, alpha, 2000, 1e-18)
    return h, _eval(h)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):

        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Binary SA (explore discrete landscape)
    if time.time() - t0 < budget_s - 450:
        h_bin, c5_bin = _binary_sa(best_h, 8000, 40, rng, T0=0.04, T1=0.0002)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin.copy()
        h_rel = _slsqp(h_bin, 100000, 3000, 1e-18)
        c5 = _eval(h_rel)
        if c5 < best_c5:
            best_c5, best_h = c5, h_rel.copy()

    # Phase 3: Steepest binary then SA
    if time.time() - t0 < budget_s - 350:
        h_st, c5_st = _steepest_binary(best_h, 20)
        if c5_st < best_c5:
            best_c5, best_h = c5_st, h_st
        h_sa2, c5_sa2 = _binary_sa(h_st, 5000, 30, rng, T0=0.02, T1=0.0002)
        if c5_sa2 < best_c5:
            best_c5, best_h = c5_sa2, h_sa2.copy()
        h_rel2 = _slsqp(h_sa2, 200000, 2000, 1e-18)
        c5 = _eval(h_rel2)
        if c5 < best_c5:
            best_c5, best_h = c5, h_rel2.copy()

    # Phase 4: Warp descent on best continuous
    if time.time() - t0 < budget_s - 300:
        h_w, c5_w = _warp_descent(best_h, 80, rng)
        if c5_w < best_c5:
            best_c5, best_h = c5_w, h_w.copy()

    # Phase 5: Try different n values
    for n_try in [250, 300, 336, 350, 400, 500]:
        if time.time() - t0 > budget_s - 200 or n_try < 40:
            break
        if n_try == len(best_h):
            continue
        h_r = _repair(_resample(best_h, n_try), n_try)
        h_r = _slsqp(h_r, 50000, 1500, 1e-18)
        c5 = _eval(h_r)
        if c5 < best_c5:
            best_c5, best_h = c5, h_r.copy()
        if time.time() - t0 < budget_s - 150:
            h_b, c5_b = _binary_sa(h_r, 3000, 20, rng, T0=0.02, T1=0.0003)
            if c5_b < best_c5:
                best_c5, best_h = c5_b, h_b.copy()
                h_r = _slsqp(h_b, 100000, 2000, 1e-18)
                c5 = _eval(h_r)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_r.copy()

    # Phase 6: Periodic pattern enumeration
    if time.time() - t0 < budget_s - 150:
        n_cur = len(best_h)
        for T in [10, 14, 20, 28, 40]:
            if time.time() - t0 > budget_s - 120:
                break
            if n_cur % T != 0:
                continue
            res = _periodic_enum(n_cur, T)
            if res is not None:
                h_p, c5_p = res
                if c5_p < best_c5:
                    best_c5, best_h = c5_p, h_p.copy()

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s - 30:
        for alpha in [500000, 2000000]:
            if time.time() - t0 > budget_s - 10:
                break
            h_fin = _slsqp(best_h, alpha, 3000, 1e-19)
            c5 = _eval(h_fin)
            if c5 < best_c5:
                best_c5, best_h = c5, h_fin.copy()

    # Phase 8: Fine coordinate descent
    if time.time() - t0 < budget_s - 5:
        h8 = best_h.copy()
        nc = len(h8)
        c5c = _eval(h8)
        for scale in [0.002, 0.0005, 0.0001]:
            if time.time() - t0 > budget_s - 3:
                break
            for _it in range(4000):
                if time.time() - t0 > budget_s - 3:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h8.copy()
                else:
                    h8[i], h8[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
