import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -100, 100)
    s0 = s - s.max()
    return m + np.log(np.exp(s0).sum()) / alpha

def _smax_grad(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -100, 100)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _lag_grad(h, shift, n):
    dx = 2.0 / n
    g = np.zeros(n)
    k = n - 1 - shift
    if shift >= 0:
        i1 = np.arange(n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        if shift > 0:
            i2 = np.arange(shift, n)
            g[i2] -= h[i2 - shift]
    else:
        sp = -shift
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _lbfgs_opt(h, n, alpha, maxiter):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return _smax(c, alpha)
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        w = _smax_grad(c, alpha)
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wx = w[k] * dx
            if sh >= 0:
                g[:n - sh] += wx * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wx * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += wx * (1.0 - x[:n - sp])
                g[:n - sp] -= wx * x[sp:]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_opt(h, n, alpha, steps, lr0):
    best_h = h.copy()
    best_f = _eval_c5(best_h)
    h_cur = h.copy()
    h_prev = h.copy()
    mu = 0.85
    lr = lr0
    for _s in range(steps):
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        w = _smax_grad(c, alpha)
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wx = w[k] * dx
            if sh >= 0:
                g[:n - sh] += wx * (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= wx * h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += wx * (1.0 - h_cur[:n - sp])
                g[:n - sp] -= wx * h_cur[sp:]
        h_y = _proj_sum(h_cur + mu * (h_cur - h_prev), n * 0.5)
        h_new = _proj_sum(h_y - lr * g, n * 0.5)
        f_new = _eval_c5(h_new)
        if f_new < best_f:
            best_h = h_new.copy()
            best_f = f_new
        h_prev = h_cur
        h_cur = h_new
        lr *= 0.9995
    return best_h

def _top_lag_descent(h, n, max_rounds):
    c5_cur = _eval_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for _r in range(max_rounds):
        c = np.correlate(h, 1.0 - h, mode='full')
        kf = c.argmax()
        sh = (n - 1) - kf
        g = _lag_grad(h, sh, n)
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj_sum(h - step * g, n * 0.5)
            c5n = _eval_c5(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            # Try second-worst lag
            c2 = c.copy()
            c2[kf] = -np.inf
            k2 = c2.argmax()
            sh2 = (n - 1) - k2
            g2 = _lag_grad(h, sh2, n)
            for step in [0.02, 0.01, 0.005, 0.001]:
                h_new = _proj_sum(h - step * g2, n * 0.5)
                c5n = _eval_c5(h_new)
                if c5n < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5n
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5

def _binary_sa(b, n, n_iter, rng):
    c5_cur = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * 2.0 / n)
    best_b = b.copy()
    best_c = c5_cur
    t0, t1 = 0.5, 0.005
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        ones = np.flatnonzero(b == 1)
        zeros = np.flatnonzero(b == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        b[i1], b[i0] = 0, 1
        c5_new = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * 2.0 / n)
        if c5_new < c5_cur or rng.rand() < np.exp(-(c5_new - c5_cur) / max(t, 1e-12)):
            c5_cur = c5_new
            if c5_new < best_c:
                best_c = c5_new
                best_b = b.copy()
        else:
            b[i1], b[i0] = 1, 0
    return best_b, best_c

def _fine_local(h, n, rng, n_tries):
    c5_cur = _eval_c5(h)
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval_c5(h)
        if c5n < c5_cur - 1e-18:
            c5_cur = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5_cur
