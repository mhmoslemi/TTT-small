def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _periodic_binary(n, period, pattern, rng=None):

    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,

    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

            hb, c5b = _binary_sa(hg, best_n, hl, rng, n_iter=15000)
            if c5b < best_c5:
                best_c5, best_h, best_n = c5b, hb.copy(), best_n

    # Phase 4: Continuous polish
    d4 = t0 + budget_s - 60
    if time.time() < d4:
        hp = best_h.copy()
        for alpha in [20000, 200000, 1000000]:
            if time.time() > d4:
                break
            hp = _slsqp(hp, alpha, 2000)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5, best_h = c5, hp.copy()

    # Phase 5: Final targeted SA at best_n
    rem = budget_s - (time.time() - t0) - 5
    if rem > 15:
        hb, c5b = _binary_sa(best_h, best_n, time.time() + rem - 3, rng,
                             n_iter=int(rem * 600))
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()

    # Final
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
