import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smobj(h, alpha, mu):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s - s.max()
    e = np.exp(sm)
    Q = e.sum()
    dx = 2.0 / n
    val = (m + np.log(Q) / alpha) * dx + mu * (h.sum() - n * 0.5) ** 2
    return val

def _smgrad(h, alpha, mu):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s - s.max()
    e = np.exp(sm)
    w = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):

            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * mu * (h.sum() - n * 0.5)
    return g

def _obj_grp(x, alpha, mu):
    return _smobj(x, alpha, mu)

def _jac_grp(x, alpha, mu):
    return _smgrad(x, alpha, mu)

def _lbfgs_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    mu = 50.0
    try:
        res = minimize(_obj_grp, h, jac=_jac_grp, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-12})
