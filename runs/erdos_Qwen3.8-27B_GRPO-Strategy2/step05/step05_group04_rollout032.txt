        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k_idx):
    n = len(h); s = k_idx - (n - 1); dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += -(1.0 - h[s:]) * dx
        g[s:] += (1.0 - h[:n-s]) * dx
    else:
        sp = -s
        g[:n-sp] += (1.0 - h[sp:]) * dx
        g[sp:] += -h[:n-sp] * dx
    return g

def _slsqp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max(); s2 = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s2))) / alpha

def _slsqp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max(); s2 = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s2 - s2.max()); w /= w.sum()
    dx = 2.0 / n; g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            g[sh:] += wk * (1.0 - x[:n-sh])
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - x[sp:])
            g[sp:] += -wk * x[:n-sp]
    return g

def _con_fun(x, n): return x.sum() - n * 0.5
def _con_jac(x, n): return np.ones(n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h); bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _directed(h, n_rounds, rng, topk):
    n = len(h); target = n * 0.5; c5c = _eval(h)
    for _r in range(n_rounds):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-topk:]
        m = c[idx].max()
        w = np.exp((c[idx] - m) * 30); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            g += w[ii] * _grad_lag(h, idx[ii])
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h - step * g, target)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-13:
                h = h_new; c5c = c5n; improved = True; break
        if not improved:
            h = _repair(h + 0.003 * rng.randn(n), n)
            c5c = _eval(h)
    return h, c5c

def _bimodal_push(h, strength, target):
    n = len(h); hn = h.copy()
    hn[hn < 0.5] *= (1.0 - strength)
    hn[hn >= 0.5] = 1.0 - (1.0 - hn[hn >= 0.5]) * (1.0 - strength)
    return _proj(hn, target)

def _sa(h, n, t0, t1, n_iter, rng):
    target = n * 0.5; c_cur = _eval(h)
    best_h = h.copy(); best_c = c_cur
    for i in range(n_iter):
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        mv = rng.randint(6); hn = h.copy()
        if mv == 0:
            blk = rng.randint(3, max(4, n // 8)); st = rng.randint(0, n - blk)
            hn[st:st+blk] += rng.randn(blk) * (0.01 + 0.06 * t)
        elif mv == 1:
            f = rng.randint(1, n // 4 + 1); ta = np.arange(n)
            hn += 0.02 * (0.5 + t) * np.sin(2 * np.pi * f * ta / n + rng.rand() * 6.28)
        elif mv == 2:
            hn = _bimodal_push(hn, 0.1 + 0.3 * t, target)
        elif mv == 3:
            i1, i2 = rng.choice(n, 2, replace=False)
            hn[i1], hn[i2] = hn[i2], hn[i1]
        elif mv == 4:
            bs = rng.randint(max(3, n // 12), max(4, n // 4)); st = rng.randint(0, n - bs)
            hn[st:st+bs] = 1.0 - hn[st:st+bs]
        else:
            idx = rng.choice(n, size=rng.randint(5, 20), replace=False)
            hn[idx] += rng.randn(len(idx)) * 0.015
        hn = _repair(hn, n); cn = _eval(hn)
        if cn < c_cur or rng.rand() < np.exp(-(cn - c_cur) / max(t, 1e-8)):
            h = hn; c_cur = cn
            if c_cur < best_c: best_c = c_cur; best_h = h.copy()
    return best_h, best_c

def _upsample(h, f): return np.repeat(h, f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(h); best_n = n0

    # Phase A: Directed descent (fast, many iterations)
    if _el() < budget_s * 0.15:
        h_d, c5_d = _directed(h, 400, rng, 3)
        if c5_d < best_c5:
            best_c5 = c5_d; best_h = h_d.copy()

    # Phase B: Multi-start with structural variants
    if _el() < budget_s * 0.35:
        for trial in range(3):
            if _el() > budget_s * 0.35: break
            if trial == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n0), n0)
            elif trial == 1:
                hp = _bimodal_push(best_h.copy(), 0.25, n0 * 0.5)
            else:
                hp = best_h.copy()
                ta = np.arange(n0)
                for f in [1, 2, 3, 5]:
                    hp += 0.01 * rng.randn() * np.sin(2 * np.pi * f * ta / n0 + rng.rand() * 6.28)
                hp = _repair(hp, n0)
            h_d, c5_d = _directed(hp, 150, rng, 3)
            hc = _slsqp(h_d, 100000, 1500, 1e-18)
            c5d = _eval(hc)
            if c5d < best_c5:
                best_c5 = c5d; best_h = hc.copy()

    # Phase C: SLSQP polish at base resolution
    if _el() < budget_s * 0.5:
        for al in [500000, 2000000]:
            if _el() > budget_s * 0.5: break
            hc = _slsqp(best_h, al, 3000, 1e-19); c5 = _eval(hc)
            if c5 < best_c5: best_c5 = c5; best_h = hc.copy()

    # Phase D: Directed descent after SLSQP
    if _el() < budget_s - 300:
        h_d, c5_d = _directed(best_h, 200, rng, 3)
        if c5_d < best_c5: best_c5 = c5_d; best_h = h_d.copy()

    # Phase E: Upsample to 2x
    if _el() < budget_s - 220:
        n1 = n0 * 2
        hh = _upsample(best_h, 2); hh = _repair(hh, n1)
        hh, c5h = _directed(hh, 120, rng, 3)
        if c5h < best_c5: best_c5 = c5h; best_h = hh; best_n = n1
        hc = _slsqp(best_h, 300000, 2000, 1e-19); c5 = _eval(hc)
        if c5 < best_c5: best_c5 = c5; best_h = hc; best_n = n1

    # Phase F: SA with diverse moves
    nc = best_n
    if _el() < budget_s - 80:
        t_rem = budget_s - 80 - _el()
        n_it = min(int(t_rem / 0.012), 30000)
        best_h, best_c5 = _sa(best_h, nc, 0.002, 1e-5, n_it, rng)

    # Phase G: Final directed + SLSQP
    if _el() < budget_s - 40:
        h_d, c5_d = _directed(best_h, 100, rng, 3)
        if c5_d < best_c5: best_c5 = c5_d; best_h = h_d.copy()
    if _el() < budget_s - 15:
        hc = _slsqp(best_h, 3000000, 4000, 1e-20); c5 = _eval(hc)
        if c5 < best_c5: best_c5 = c5; best_h = hc; best_n = len(hc)

    # Phase H: Ultra-fine pairwise
    if _el() < budget_s - 3:
        nc = len(best_h); hf = best_h.copy(); c5c = _eval(hf)
        n_tr = min(int((budget_s - 3 - _el()) / 0.002), 50000)
        for _ in range(n_tr):
            if _el() > budget_s - 3: break
            i = rng.randint(nc); j = rng.randint(nc)
            if i == j: continue
            d = 4e-5 * rng.randn()
            if 0 <= hf[i] + d <= 1 and 0 <= hf[j] - d <= 1:
                hf[i] += d; hf[j] -= d
                c5n = _eval(hf)
                if c5n < c5c: c5c = c5n
                else: hf[i] -= d; hf[j] += d
        if c5c < best_c5: best_c5 = c5c; best_h = hf

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
