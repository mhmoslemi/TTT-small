So the objective max_k corr[k] is the max of (2n-1) linear functions of h[i], which is a convex piecewise-linear function. The minimum over h[i] ∈ [0,1] can be found by checking the breakpoints (where two adjacent linear functions cross) and the endpoints.

And coordinate descent for a max-of-linear-functions converges (it's a convex optimization in each coordinate). It might not find the global optimum (the overall problem is non-convex), but it's a powerful tool.

def coordinate_descent(h, n, max_sweeps):
    # For each coordinate i, find the optimal h[i] given all others fixed
    # corr[k] is linear in h[i]: corr[k] = a_k * h[i] + b_k
    # where a_k and b_k depend on which other h[j] values participate
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Compute a_k and b_k for all k
            # corr[k] involves h[i] if and only if i is in the valid range for shift (n-1-k)
            # For shift s = (n-1)-k:
            #   If s >= 0: h[i] appears in corr[k] if i < n-s (as h[i]) or i >= s (as h[i-s] in (1-h[i-s+s]))
            #   
            # Let me compute: corr[k] = sum_{valid j} h[j]*(1-h[j+s])
            # h[i] appears in the term j=i (if valid) and in the term j=i-s (if s>0 and i-s is valid, as 1-h[i])
            
            # Actually, let me just compute corr[k] with h[i] removed, then add the h[i] terms.
            
            # Simpler: compute the full correlation, then for each k that involves h[i],
            # compute the coefficient of h[i] and the constant term.
            
            coeff = np.zeros(2*n-1)
            base = np.zeros(2*n-1)
            
            for k in range(2*n-1):
                s = (n-1) - k
                if s >= 0:
                    # corr[k] = sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])
                    # h[i] appears as h[i] if i <= n-1-s, and as h[j+s]=h[i] if j=i-s, i.e., j=i-s >= 0 and i <= n-1, j <= n-1-s
                    c = 0
                    for j in range(n-s):
                        if j == i:
                            # term is h[i]*(1-h[i+s]) → coeff: (1-h[i+s]), base: 0
                            if i+s < n:
                                coeff[k] += (1-h[i+s])
                            # else: h[i+s] doesn't exist, this term doesn't contribute
                        elif j+s == i:
                            # term is h[j]*(1-h[i]) → coeff: -h[j], base: h[j]
                            base[k] += h[j]
                            coeff[k] += -h[j]
                        else:
                            c += h[j]*(1-h[j+s])
                    base[k] += c
                else:
                    p = -s
                    # corr[k] = sum_{j=p}^{n-1} h[j]*(1-h[j-p])
                    c = 0
                    for j in range(p, n):
                        if j == i:
                            if j-p >= 0:
                                coeff[k] += (1-h[j-p])
                        elif j-p == i:
                            base[k] += h[j]
                            coeff[k] += -h[j]
                        else:
                            c += h[j]*(1-h[j-p])
                    base[k] += c
            
            # Now min h[i] ∈ [0,1] of max_k (base[k] + coeff[k]*h[i])
            # This is a 1D convex piecewise-linear minimization
            # The minimum is at one of the breakpoints or at 0 or 1
            # Breakpoints: where base[k] + coeff[k]*x = base[m] + coeff[m]*x
            # i.e., x = (base[m]-base[k])/(coeff[k]-coeff[m])
            
            # Find the minimum by ternary search or by evaluating at all breakpoints
            # Since it's convex PL, we can do binary search on the derivative
            # Or just evaluate at a few candidate points
            
            # For efficiency, use ternary search on [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                f1 = np.max(base + coeff*m1)
                f2 = np.max(base + coeff*m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            new_hi = (lo+hi)/2
            if abs(new_hi - h[i]) > 1e-10:
                h[i] = new_hi
                improved = True
        if not improved:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam_opt(h, alpha_schedule, nsteps, lr, beta1, beta2, eps):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_f = _smooth_obj(h, alpha_schedule[0])
    for t in range(nsteps):
        alpha = alpha_schedule[min(t, len(alpha_schedule)-1)]
        g = _smooth_grad(h, alpha)
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*g*g
        m_hat = m / (1-beta1**(t+1))
        v_hat = v / (1-beta2**(t+1))
        h_new = _proj_to_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), n*0.5)
        f = _smooth_obj(h_new, alpha)
        if f < best_f
