def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    m = c.max()

def _grad_single(h, j, n):
    """Gradient of corr vector w.r.t. h[j]. Returns array of length 2n-1."""
    m = 2 * n - 1
    g = np.zeros(m)
    one_minus_h = 1.0 - h
    for k in range(m):
        sh = (n - 1) - k
        if sh >= 0:
            if j < n - sh:
                g[k] += one_minus_h[j + sh]
            if sh > 0 and 0 <= j - sh < n - sh:
                g[k] -= h[j - sh]
        else:
            sp = -sh
            if sp <= j < n:
                g[k] += one_minus_h[j - sp]
            if 0 <= j + sp < n:
                g[k] -= h[j + sp]
    return g

def _min_line_max(c, d, lo, hi):
    """Minimize max_k(c[k] + d[k]*t) for t in [lo, hi]. c, d are arrays."""
    # Golden section search on convex function
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = np.max(c + d * c1)
    f2 = np.max(c + d * c2)
    for _ in range(60):
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = np.max(c + d * c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = np.max(c + d * c2)
    return 0.5 * (a + b)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        if _el() > 120:
            break
        h_new = _slsqp_opt(best_h, alpha, 2500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n0

    # Phase 2: Coordinate descent (main workhorse)
    if _el() < budget_s - 500:
        h_cd, _ = _coord_descent(best_h.copy(), 300, lambda: _el() > budget_s - 500)
        c5 = _eval(h_cd)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_cd.copy(), n0

    # Phase 3: Paired coordinate descent (preserves sum exactly)
    if _el() < budget_s - 400:
        h_p, _ = _paired_cd(best_h.copy(), 8000, lambda: _el() > budget_s - 400, rng)
        c5 = _eval(h_p)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_p.copy(), n0

    # Phase 4: More coordinate descent
    if _el() < budget_s - 300:
        h_cd2, _ = _coord_descent(best_h.copy(), 400, lambda: _el() > budget_s - 300)
        c5 = _eval(h_cd2)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_cd2.copy(), n0

    # Phase 5: SLSQP polish after coordinate descent
    if _el() < budget_s - 250:
        hs = _slsqp_opt(best_h, 500000, 2000, 1e-19)
        c5 = _eval(hs)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hs.copy(), n0
        # Another round of coordinate descent
        if _el() < budget_s - 250:
            h_cd3, _ = _coord_descent(best_h.copy(), 300, lambda: _el() > budget_s - 250)
            c5 = _eval(h_cd3)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_cd3.copy(), n0

    # Phase 6: Fine local search with decreasing steps
    if _el() < budget_s - 80:
        h_ls = best_h.copy()
        nc = len(h_ls)
        c5c = _eval(h_ls)
        for scale in [0.003, 0.0008, 0.0002, 5e-5, 1e-5]:
            if _el() > budget_s - 80:
                break
            for _it in range(10000):
                if _el() > budget_s - 80:
                    break
                i, j = rng.randint(nc), rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h_ls[i] + d, h_ls[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h_ls[i], h_ls[j] = ni, nj
                c5n = _eval(h_ls)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h, best_n = c5n, h_ls.copy(), nc
                else:
                    h_ls[i], h_ls[j] = h_ls[i] - d, h_ls[j] + d

    # Phase 7: Final paired CD + SLSQP
    if _el() < budget_s - 40:
        h_f, _ = _paired_cd(best_h.copy(), 5000, lambda: _el() > budget_s - 40, rng)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_f.copy(), n0
    if _el() < budget_s - 8:
        h_fin = _slsqp_opt(best_h, 500000, 1500, 1e-19)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_fin.copy(), len(h_fin)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
