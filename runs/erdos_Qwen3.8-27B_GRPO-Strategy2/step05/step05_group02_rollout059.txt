import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _lse_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _topk_obj(h, k_now, tau):
    c = _corr(h)
    idx = np.argsort(c)[::-1][:k_now]
    exc = np.maximum(c[idx] - tau, 0.0)
    return np.sum(exc * exc)

def _topk_grad(h, k_now, tau):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:k_now]
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in idx:
        if c[ki] <= tau:
            continue
        exc = 2.0 * (c[ki] - tau) * dx
        sh = (n - 1) - ki
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += exc * (1.0 - h[sh:sh + cnt])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += exc * (1.0 - h[:n - sp])
            g[:n - sp] -= exc * h[sp:]
    return g

    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_lse(h, alpha, maxiter, ftol):

    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

            h_pol = _slsqp_lse(best_h, 80000, 2000, 1e-19)
            c5p = _eval(h_pol)
            if c5p < best_c5:
                best_c5 = c5p; best_h = h_pol.copy()

    # Phase 4: Upsample to 2x
    if time.time() - t0 < budget_s - 200:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [5000, 50000, 200000]:
            if time.time() - t0 > budget_s - 120:
                break
            h_hi = _slsqp_lse(h_hi, alpha, 3000, 1e-20)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_hi.copy(); best_n = n1

    # Phase 5: Multi-start diverse
    nc = best_n
    if time.time() - t0 < budget_s - 80:
        for r in range(18):
            if time.time() - t0 > budget_s - 40:
                break
            strat = r % 6
            if strat == 0:
                hp = _repair(best_h + 0.025 * rng.randn(nc), nc)
            elif strat == 1:
                hp = best_h.copy()
                bs = rng.randint(8, max(9, nc // 6))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, nc)
            elif strat == 2:
                hp = best_h.copy()
                f = rng.randint(2, nc // 4 + 1)
                t_arr = np.arange(nc, dtype=float)
                hp += 0.008 * np.sin(2 * np.pi * f * t_arr / nc + rng.rand() * 2 * np.pi)
                hp = _repair(hp, nc)
            elif strat == 3:
                hp = best_h.copy()
                bs = rng.randint(5, max(6, nc // 12))
                st = rng.randint(0, nc - bs)
                hp[st:st + bs] = 1.0 - hp[st:st + bs]
                hp = _repair(hp, nc)
            elif strat == 4:
                hp = 0.5 + 0.35 * np.sin(2 * np.pi * np.arange(nc, dtype=float) / nc * rng.randint(3, 15))
                hp += 0.06 * rng.randn(nc)
                hp = _repair(hp, nc)
            else:
                half = nc // 2
                hp = best_h.copy()
                d = 0.02 * rng.randn(half)
                hp[:half] += d
                if nc % 2 == 0:
                    hp[half:] -= d
                else:
                    hp[half:] -= np.concatenate([d[1:], d[:1]])
                hp = _repair(hp, nc)
            h_new = _slsqp_lse(hp, 100000, 2500, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 6: Gradient targeted descent
    if time.time() - t0 < budget_s - 30:
        h = best_h.copy()
        nc = len(h); c5c = _eval(h)
        for _round in range(500):
            if time.time() - t0 > budget_s - 30:
                break
            c = _corr(h)
            k_w = c.argmax()
            sh = (nc - 1) - k_w
            g = np.zeros(nc)
            if sh >= 0:
                g[:nc - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:nc - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:nc - sp])
                g[:nc - sp] -= h[sp:]
            g *= 2.0 / nc
            improved = False
            for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_new = _proj_sum(h - step * g, nc * 0.5)
                c5n = _eval(h_new)
                if c5n < c5c - 1e-14:
                    h = h_new; c5c = c5n; improved = True
                    if c5c < best_c5:
                        best_c5 = c5c; best_h = h.copy()
                    break
            if not improved:
                h = _repair(h + 0.002 * rng.randn(nc), nc)
                c5c = _eval(h)

    # Phase 7: Fine pairwise
    if time.time() - t0 < budget_s - 10:
        h = best_h.copy()
        nc = len(h); c5c = _eval(h)
        for scale in [0.001, 0.0003, 0.0001]:
            if time.time() - t0 > budget_s - 10:
                break
            for _it in range(12000):
                if time.time() - t0 > budget_s - 10:
                    break
                i = rng.randint(nc); j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-14:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d

    # Phase 8: Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        h = _slsqp_lse(best_h, 500000, 5000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
