def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    # Fix remaining discrepancy
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(np.max(_corr_full(h)))

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pm_obj(h, p):
    c = _corr_full(h)
    return float(np.mean(c ** p) ** (1.0 / p))

def _pm_grad(h, p):
    n = len(h)
    c = _corr_full(h)
    w = c ** (p - 1)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-18:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            end = n - sh
            g[:end] += wk * (1.0 - h[sh:sh+end])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    scale = (1.0 / p) * (np.mean(c ** p)) ** (1.0 / p - 1.0)
    return scale * g

def _lbfgs_pow(h, p, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pm_obj, h, jac=_pm_grad, args=(p,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fourier_flatten(h, n_iter):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        H = np.fft.rfft(h)
        # Target: flat power for non-DC
        dc = H[0].real
        power = np.abs(H) ** 2
        if len(power) > 1:
            non_dc = power[1:]
            target_p = non_dc.mean()
            new_mag = np.sqrt(target_p) * np.ones(len(H))
            new_mag[0] = 0  # will set from DC
            new_H = np.zeros(len(H), dtype=complex)
            new_H[0] = dc
            for m in range(1, len(H)):
                if power[m] > 1e-20:
                    new_H[m] = H[m] / np.sqrt(power[m]) * new_mag[m]
                else:
                    new_H[m] = new_mag[m]
            h_new = np.fft.irfft(new_H, n=n)
        else:
            h_new = h.copy()
        h_new = _proj_sum(h_new, n / 2.0)
        c5n = _eval(h_new)
        if c5n < c5c:
            h = h_new
            c5c = c5n
    return h, c5c

def _active_grad(h, n_top, n_iter, lr0, decay):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for it in range(n_iter):
        c = _corr_full(h)
        idx = np.argsort(c)[::-1][:n_top]
        g = np.zeros(n)
        for k in idx:
            sh = k - (n - 1)
            if sh >= 0:
                end = n - sh
                g[:end] += (1.0 - h[sh:sh+end])
                if sh > 0:
                    g[sh:] -= h[:n-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n-sp])
                g[:n-sp] -= h[sp:]
        g *= dx / n_top
        lr = lr0 * (decay ** it)
        improved = False
        for _ls in range(25):
            h_new = _proj_sum(h - lr * g, n / 2.0)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h = h_new
                c5c = c5n
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return h, c5c

def _pair_search(h, n_iter, seed):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    rng = np.random.RandomState(seed)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        c0 = np.correlate(h, 1.0 - h, mode='full')
        delta = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            sh = k - (n - 1)
            di = 0.0
            dj = 0.0
            if sh >= 0:
                if i + sh < n:
                    di += (1.0 - h[i + sh])
                if sh > 0 and i - sh >= 0:
                    di -= h[i - sh]
                if j + sh < n:
                    dj += (1.0 - h[j + sh])
                if sh > 0 and j - sh >= 0:
                    dj -= h[j - sh]
            else:
                sp = -sh
                if i >= sp:
                    di += (1.0 - h[i - sp])
                if i + sp < n - sp:
                    di -= h[i + sp]
                if j >= sp:
                    dj += (1.0 - h[j - sp])
                if j + sp < n - sp:
                    dj -= h[j + sp]
            delta[k] = di - dj
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-12:
            continue
        dgrid = np.linspace(dlo, dhi, 21)
        best_d = 0.0
        best_v = np.max(c0)
        for d in dgrid:

            if v < best_v:

                best_d = d
        if abs(best_d) > 1e-10:

            h[j] -= best_d
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i] -= best_d
                h[j] += best_d
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(np.max(_corr(h)))

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pm_obj(h, p):
    c = _corr(h)
    return float(np.mean(c ** p) ** (1.0 / p))

def _pm_grad(h, p):
    n = len(h)
    c = _corr(h)
    w = c ** (p - 1)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-20:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            e = n - sh
            g[:e] += wk * (1.0 - h[sh:sh+e])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return (1.0 / p) * (np.mean(c ** p)) ** (1.0 / p - 1.0) * g

def _lbfgs(h, p, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pm_obj, h, jac=_pm_grad, args=(p,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fourier_flatten(h, n_iter):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        H = np.fft.rfft(h)
        if len(H) > 2:
            dc = H[0].real
            pw = np.abs(H[1:])**2
            tp = pw.mean()
            new_H = np.zeros(len(H), dtype=complex)
            new_H[0] = dc
            nf = np.sqrt(tp)
            for m in range(1, len(H)):
                if pw[m-1] > 1e-15:
                    new_H[m] = H[m] / np.sqrt(pw[m-1]) * nf
                else:
                    new_H[m] = nf * (0.5 + 0.5j)
            h_new = np.fft.irfft(new_H, n=n)
        else:
            h_new = h.copy()
        h_new = _proj_sum(h_new, n / 2.0)
        c5n = _eval(h_new)
        if c5n < c5c:
            h = h_new
            c5c = c5n
    return h, c5c

def _active_grad(h, n_top, n_iter, lr0, decay):

    dx = 2.0 / n
    c5c = _eval(h)

    num_k = 2 * n - 1

    c = np.correlate(h, 1.0 - h, mode='full')

def _pair_search(h, n_iter, seed):

    dx = 2.0 / n
    c5c = _eval(h)
    rng = np.random.RandomState(seed)
    D, c_raw = _build_D(h)
    success = 0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = D[i] - D[j]
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-12:
            continue
        c_base = c_raw * dx
        dgrid = np.linspace(dlo, dhi, 25)
        best_d = 0.0
        best_v = np.max(c_base)
        for d in dgrid:
            v = np.max(c_base + d * dx * delta)
            if v < best_v:
                best_v = v
                best_d = d
        if abs(best_d) > 1e-10:

            h[j] -= best_d
            h = np.clip(h, 0, 1)
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                success += 1
                if success % 50 == 0:
                    D, c_raw = _build_D(h)
            else:

    return h, c5c

def _sinusoid_init(n, f1, a1, f2, a2, rng):

    if time.time() - t0 < budget_s - 380:
        h_new, c5 = _fourier_flatten(h.copy(), 150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        for p in [64, 256, 1024]:
            if time.time() - t0 > budget_s - 330:
                break
            h_new = _lbfgs(h, p, 400)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h = h_new

    # Phase 3: Active gradient
    if time.time() - t0 < budget_s - 320:
        h_new, c5 = _active_grad(h.copy(), 8, 3000, 0.08, 0.9995)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new

    # Phase 4: Pairwise line search
    if time.time() - t0 < budget_s - 250:
        h_new, c5 = _pair_search(h.copy(), 5000, seed + 1)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new

    # Phase 5: Second Fourier + L-BFGS cycle
    if time.time() - t0 < budget_s - 230:
        h_new, c5 = _fourier_flatten(h.copy(), 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        h_new = _lbfgs(h, 1024, 400)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new

    # Phase 6: Multi-start with diverse inits
    if time.time() - t0 < budget_s - 180:
        for _r in range(15):
            if time.time() - t0 > budget_s - 130:
                break
            strat = rng.randint(5)
            if strat == 0:
                hp = _repair(h + 0.015 * rng.randn(n0), n0)
            elif strat == 1:
                hp = _repair(np.clip(1.25 * (h - 0.5) + 0.5, 0, 1), n0)
            elif strat == 2:
                hp = _sinusoid_init(n0, rng.randint(2, n0//4+1), 0.25*rng.uniform(0.3,1),
                                    rng.randint(n0//4, n0//2+1), 0.15*rng.uniform(0.3,1), rng)
            elif strat == 3:
                hp = h.copy()
                bs = rng.randint(8, 30)
                st = rng.randint(0, n0 - bs)
                hp[st:st+bs] += 0.05 * rng.randn(bs)
                hp = _repair(hp, n0)
            else:
                hp = _repair(h + 0.03 * rng.randn(n0), n0)
                hp = np.clip(1.3 * (hp - 0.5) + 0.5, 0, 1)
                hp = _repair(hp, n0)
            h_new = _lbfgs(hp, 256, 300)
            c5 = _eval(h_new)
            if time.time() - t0 < budget_s - 130:
                h_new, c5 = _active_grad(h_new, 5, 1000, 0.04, 0.999)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h = h_new

    # Phase 7: Final pairwise + active
    if time.time() - t0 < budget_s - 80:
        h_new, c5 = _pair_search(best_h.copy(), 8000, seed + 200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    if time.time() - t0 < budget_s - 40:
        h_new, c5 = _active_grad(best_h.copy(), 3, 2500, 0.02, 0.9997)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    if time.time() - t0 < budget_s - 15:
        h_new = _lbfgs(best_h, 4096, 400)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
