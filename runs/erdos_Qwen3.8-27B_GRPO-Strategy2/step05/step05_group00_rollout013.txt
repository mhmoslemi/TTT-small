from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    diff -= a

                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lsobj(x, alpha, lam):
    c = _cvec(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha + lam * (x.sum() - len(x) * 0.5) ** 2

def _lsgrad(x, alpha, lam):
    n = len(x)
    c = _cvec(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _lsbfgs(h, alpha, lam, maxiter=3000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lsobj, h, jac=_lsgrad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_descent(h, max_rounds, step_init=0.005):
    n = len(h)
    c5c = _eval(h)
    step = step_init
    for _ in range(max_rounds):
        c = _cvec(h)
        # Find top few lags
        top_idx = np.argsort(c)[-3:]
        w = np.zeros(2*n-1)
        w[top_idx] = 1.0
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
                if sh > 0:
                    g[sh:] -= wk * h[:cnt]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n-sp])
                g[:n-sp] -= wk * h[sp:]
        # Line search
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1]:
            h_new = _repair(h - step * mult * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:

                c5c = c5n
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
    return h, c5c

def _coord_descent(h, max_iter, rng):
    n = len(h)
    c5c = _eval(h)
    for scale in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        for _ in range(max_iter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:

            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lsobj(x, alpha, lam):
    c = _cvec(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha + lam * (x.sum() - len(x) * 0.5) ** 2

def _lsgrad(x, alpha, lam):
    n = len(x)
    c = _cvec(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _lsbfgs(h, alpha, lam, maxiter=3000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lsobj, h, jac=_lsgrad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, max_rounds, step_init=0.004, top_k=2):
    n = len(h)
    c5c = _eval(h)
    step = step_init
    stall = 0
    for r in range(max_rounds):
        c = _cvec(h)
        top_idx = np.argsort(c)[-top_k:]
        w = np.zeros(2*n-1)
        w[top_idx] = 1.0
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
                if sh > 0:
                    g[sh:] -= wk * h[:cnt]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n-sp])
                g[:n-sp] -= wk * h[sp:]
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _repair(h - step * mult * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new; c5c = c5n; improved = True; break
        if improved:
            stall = 0
        else:
            stall += 1
            step *= 0.7
            if step < 1e-9 or stall > 30:
                break
    return h, c5c

def _coord_cd(h, max_iter, rng):
    n = len(h)
    c5c = _eval(h)
    for scale in [0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
        n_it = max_iter if scale > 0.003 else max_iter * 2
        for _ in range(n_it):
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_local(h, n_swaps, rng):

    hb = (h > 0.5).astype(float)
    hb = _repair(hb, n)
    c5c = _eval(hb)

    zeros = np.where(hb == 0.0)[0]
    for _ in range(n_swaps):

            zeros = np.where(hb == 0.0)[0]
        else:
            hb[o] = 1.0; hb[z] = 0.0
    return hb, c5c
