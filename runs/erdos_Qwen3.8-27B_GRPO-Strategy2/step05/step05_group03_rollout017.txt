    c = np.correlate(h, 1-h, mode='full')
    nm1 = n - 1
    # For each lag k (0 to 2n-2), compute b_k
    # b_k = (1 - h[i-k+nm1]) if 0 <= i-k+nm1 < n, else 0
    #      - h[i+k-nm1] if 0 <= i+k-nm1 < n, else 0

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _pair_coeff(h, i, n):
    """Compute coefficient b_k of h[i] in the full correlation."""
    nm1 = n - 1
    k_vals = np.arange(2 * n - 1)
    j1 = i - k_vals + nm1
    j2 = i + k_vals - nm1
    b = np.zeros(2 * n - 1)
    m1 = (j1 >= 0) & (j1 < n)
    m2 = (j2 >= 0) & (j2 < n)
    if m1.any():
        b[m1] += (1.0 - h[j1[m1]])
    if m2.any():
        b[m2] -= h[j2[m2]]
    return b

def _pair_optimal_d(c, db, d_lo, d_hi):
    """Find d that minimizes max_k(c_k + d*db_k) for d in [d_lo, d_hi]."""
    if d_lo >= d_hi:
        return 0.0
    lo, hi = d_lo, d_hi
    # Check boundaries
    f_lo = np.max(c + lo * db)
    f_hi = np.max(c + hi * db)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c + mid * db
        mx = vals.max()
        S = np.where(vals >= mx - 1e-10)[0]
        if len(S) == 0:
            break
        if db[S].min() > 1e-14:
            hi = mid
        elif db[S].max() < -1e-14:
            lo = mid
        else:
            break
    t = (lo + hi) * 0.5
    f_mid = np.max(c + t * db)
    if f_lo < f_mid:
        t = lo
    if f_hi < f_mid:
        t = hi
    return t

def _coord_desc(h, dx, rng, t_end, n_pairs=80, n_iters=5000):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best = _c5(h, dx)
    c = _corr(h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        # Recompute correlation every few iterations
        if it % 5 == 0:
            c = _corr(h)
        # Pick random pairs
        idx = rng.choice(n, n_pairs * 2, replace=False)
        pairs = [(idx[2*k], idx[2*k+1]] for k in range(n_pairs)]
        improved = False
        for i, j in pairs:
            bi = _pair_coeff(h, i, n)
            bj = _pair_coeff(h, j, n)
            db = bi - bj
            d_lo = max(-h[i], 0.0)  # h[i]+d >= 0
            d_hi = min(1.0 - h[i], h[j])  # h[i]+d <= 1, h[j]-d >= 0
            if d_lo >= d_hi:
                continue
            d = _pair_optimal_d(c, db, d_lo, d_hi)
            if abs(d) > 1e-12:
                h[i] += d
                h[j] -= d
                improved = True
        if improved:
            h = _repair(h, tgt)
            c_new = _corr(h)
            c_new = c_new * dx
            cn = float(c_new.max())
            if cn < best:
                best = cn
                c = c_new / dx  # store without dx for reuse
            # Actually let me just recompute c each time for simplicity
            c = _corr(h)
    return h, best
