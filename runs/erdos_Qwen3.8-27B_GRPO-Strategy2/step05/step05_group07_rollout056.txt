    return h_k, f

def _fista_topm(h, topm, lr_init, n_iter, slack):
    n = len(h)
    h_k = h.copy()
    h_prev = h.copy()
    t = 1.0
    lr = lr_init
    f = _eval(h_k)
    for k in range(n_iter):
        if time.time() > _deadline - slack:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = h_k + (t - 1.0) / t_next * (h_k - h_prev)
        z = np.clip(z, 0.0, 1.0)
        grad = _topm_grad(z, topm)
        step = lr
        accepted = False
        for _ in range(20):
            h_new = _proj_sum(h_k - step * grad, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f - 1e-10:
                accepted = True
                break
            step *= 0.5
        if not accepted:
            break
        h_prev = h_k
        h_k = h_new
        f = f_new
        t = t_next
        lr = min(lr * 1.02, 20.0)
    return h_k, f

def _pairwise(h, n_iter, scale, seed_val, slack):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_iter):
        if time.time() > _deadline - slack:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-15:
                f = fn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, f

def _slsqp_polish(h, alpha, maxiter, slack):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def _obj(x):
        c = _corrs(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _objg(x):
        c = _corrs(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        obj = m + np.log(np.sum(np.exp(s - s.max()))) / alpha
        g = _smooth_grad(x, alpha)
        return obj, g
    try:
        res = minimize(_objg, h, jac=True, method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(len(x))}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() > _deadline - 5:
            return _repair(h, n), _eval(h)
        return _repair(res.x, n), _eval(_repair(res.x, n))
    except Exception:
        return _repair(h, n), _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0, _deadline
    _t0 = time.time()
    _deadline = _t0 + budget_s
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    rng = np.random.RandomState(seed)

    # Phase 1: FISTA smooth max, increasing alpha
    for alpha, lr, niter in [(100, 2.0, 8000), (500, 1.0, 6000),
                             (2000, 0.5, 5000), (10000, 0.3, 4000), (50000, 0.15, 3000)]:
        if time.time() > _deadline - 150:
            break
        hn, f = _fista_smooth(best_h, alpha, lr, niter, 30)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()

    # Phase 2: Top-m active set descent
    for topm, niter in [(1, 5000), (3, 4000), (5, 4000), (8, 3000), (15, 2000)]:
        if time.time() > _deadline - 120:
            break
        hn, f = _fista_topm(best_h, topm, 0.8, niter, 20)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()

    # Phase 3: SLSQP polish
    if time.time() < _deadline - 100:
        hn, f = _slsqp_polish(best_h, 50000, 800, 30)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()
    if time.time() < _deadline - 60:
        hn, f = _slsqp_polish(best_h, 200000, 600, 20)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()

    # Phase 4: Multi-start with FISTA
    if time.time() < _deadline - 60:
        for _r in range(30):
            if time.time() > _deadline - 30:
                break
            amp = 0.005 + 0.02 * (_r / 30.0)
            hp = best_h + amp * rng.randn(n)
            hp = _repair(hp, n)
            hn, f = _fista_smooth(hp, 20000, 0.3, 2000, 5)
            if f < best_c5:
                best_c5 = f
                best_h = hn.copy()

    # Phase 5: Pairwise fine-tuning
    for scale, nit in [(0.005, 4000), (0.002, 6000), (0.001, 8000),
                       (0.0005, 10000), (0.0001, 12000)]:
        if time.time() > _deadline - 5:
            break
        hn, f = _pairwise(best_h, nit, scale, seed + int(scale * 1e7), 3)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()

    # Phase 6: Final top-m polish
    if time.time() < _deadline - 6:
        hn, f = _fista_topm(best_h, 3, 0.3, 3000, 3)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()
        hn, f = _fista_topm(best_h, 1, 0.2, 2000, 2)
        if f < best_c5:
            best_c5 = f
            best_h = hn.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
