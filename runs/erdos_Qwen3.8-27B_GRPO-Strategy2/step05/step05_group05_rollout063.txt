import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _shift_gradient(h, shift, n):
    """Gradient of c_shift w.r.t. h, where c_shift = sum h[i]*(1-h[i+shift]) for shift>=0"""
    g = np.zeros(n)
    if shift >= 0:
        if shift == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-shift] += (1.0 - h[shift:])
            g[shift:] -= h[:n-shift]
    else:
        sp = -shift
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _top_k_descent(h, n_top, max_iters, n_point_evals):
    """Descent focusing on top n_top shifts with exact 2D moves."""
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[::-1][:n_top]
        # For top shifts, compute gradient
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            g += _shift_gradient(h, sh, n)
        g *= dx / n_top  # average gradient of top shifts
        # Try steepest descent
        improved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            # Try random 2D moves
            for _ in range(n_point_evals):
                i, j = np.random.randint(n, 2)
                if i == j:
                    continue
                d = 0.01 * np.random.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h[i], h[j] = ni - d, nj + d
    return h, c5c

def _compute_delta_vec(h, i, j):
    n = len(h)
    sh = np.arange(1, n)  # shape (n-1,)
    # Positive shift contribution for index x: (1-h[x+sh]) where x+sh<n, minus h[x-sh] where x>=sh
    def _grad_pos(x):
        g = np.zeros(n-1)
        valid1 = x + sh < n  # x+sh < n
        g[valid1] = 1.0 - h[x + sh[valid1]]
        valid2 = x >= sh
        g[valid2] -= h[x - sh[valid2]]
        return g
    
    def _grad_neg(x):
        g = np.zeros(n-1)
        valid1 = x >= sh  # x-sh >= 0
        g[valid1] = 1.0 - h[x - sh[valid1]]
        valid2 = x + sh < n
        g[valid2] -= h[x + sh[valid2]]
        return g
    
    d_pos = _grad_pos(i) - _grad_pos(j)
    d_neg = _grad_neg(i) - _grad_neg(j)
    
    delta = np.zeros(2*n-1)
    delta[n-1-sh] = d_pos  # positive shifts
    delta[n-1+sh] = d_neg  # negative shifts
    delta[n-1] = 2.0*(h[j] - h[i])  # zero shift
    return delta

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _c_fun(x):
    return x.sum() - len(x) * 0.5

def _c_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, max_iters, n_top):
    n = len(h); dx = 2.0 / n; c5c = _eval(h)
    for _ in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[::-1][:n_top]
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= dx / n_top
        improved = False
        for step in [0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new; c5c = c5n; improved = True; break
        if not improved:
            break
    return h, c5c

def _delta_vec(h, i, j):
    n = len(h)
    sh = np.arange(1, n)
    def _gp(x):
        g = np.zeros(n - 1)
        v1 = x + sh < n; g[v1] = 1.0 - h[x + sh[v1]]
        v2 = x >= sh; g[v2] -= h[x - sh[v2]]
        return g
    def _gn(x):
        g = np.zeros(n - 1)
        v1 = x >= sh; g[v1] = 1.0 - h[x - sh[v1]]
        v2 = x + sh < n; g[v2] -= h[x + sh[v2]]
        return g
    dp = _gp(i) - _gp(j)
    dn = _gn(i) - _gn(j)
    delta = np.zeros(2 * n - 1)
    delta[n - 1 - sh] = dp
    delta[n - 1 + sh] = dn
    delta[n - 1] = 2.0 * (h[j] - h[i])
    return delta

def _exact_2d(h, n_iters, rng, t0, budget_s):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c5r = c.max()
    stall = 0
    for it in range(n_iters):
        if (it + 1) % 5000 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
            c5r = c.max()
        if time.time() - t0 > budget_s - 30:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-14:
            continue
        delta = _delta_vec(h, i, j)
        a, b = dlo, dhi
        for _ in range(35):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            if np.max(c + m1 * delta) < np.max(c + m2 * delta):
                b = m2
            else:
                a = m1
        d = (a + b) * 0.5
        if abs(d) < 1e-13:
            continue
        nm = np.max(c + d * delta)
        if nm < c5r - 1e-14:
            h[i] += d; h[j] -= d
            c += d * delta; c5r = nm; stall = 0
        else:
            stall += 1
    return h, c5r * 2.0 / n

def _binary_push(h, strength):
    n = len(h); hp = h.copy()
    lo_mask = hp < 0.1
    hp[lo_mask] *= (1.0 - strength)
    hi_mask = hp > 0.9
    hp[hi_mask] = 1.0 - (1.0 - hp[hi_mask]) * (1.0 - strength)
    return _repair(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float).copy()
    h = _repair(h, n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP alpha sweep
    for alpha in [500, 2000, 8000, 40000, 200000, 800000]:
        if time.time() - t0 > 200:
            break
        h_new = _slsqp(best_h, alpha, 1500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 2: Active set descent
    if time.time() - t0 < budget_s - 400:
        h_new, c5 = _active_descent(best_h.copy(), 3000, 5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 3: Exact 2D coordinate descent (main refinement)
    if time.time() - t0 < budget_s - 60:
        h_new, c5 = _exact_2d(best_h.copy(), 60000, rng, t0, budget_s)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 4: Binary push variants
    if time.time() - t0 < budget_s - 120:
        for strength in [0.3, 0.6, 0.8]:
            if time.time() - t0 > budget_s - 100:
                break
            hp = _binary_push(best_h, strength)
            hp = _slsqp(hp, 300000, 800, 1e-18)
            c5 = _eval(hp)
            if c5 < best_c5:
                best_c5 = c5; best_h = hp
            h_new, c5 = _exact_2d(best_h.copy(), 15000, rng, t0, budget_s)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 5: Multi-start
    if time.time() - t0 < budget_s - 200:
        n_cur = len(best_h)
        for _r in range(8):
            if time.time() - t0 > budget_s - 160:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.02 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n_cur), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(1, 3)):
                    bs = rng.randint(5, max(6, n_cur // 8))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st + bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            else:
                hp = _repair(np.clip(best_h + 0.04 * (2.0 * rng.rand(n_cur) - 1.0), 0, 1), n_cur)
            h_new = _slsqp(hp, 500000, 500, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()
            h_new, c5 = _exact_2d(best_h.copy(), 10000, rng, t0, budget_s)
            if c5 < best_c5:
                best_c5 = c5; best_h = h_new.copy()

    # Phase 6: Final long 2D polish
    if time.time() - t0 < budget_s - 15:
        h_new, c5 = _exact_2d(best_h.copy(), 80000, rng, t0, budget_s)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    # Phase 7: Final SLSQP
    if time.time() - t0 < budget_s - 5:
        h_new = _slsqp(best_h, 2000000, 1500, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
