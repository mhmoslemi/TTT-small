            if len(xs) != n_m:
                xs = project_grid(xs, best_n, n_m)
        elif i == 4:
            xs = perturb_white(h_m, n_m, 0.12, rng)
        else:
            xs = perturb_smooth(h_m, n_m, 0.07, rng)
        x_b, v_b = refine_anneal(xs, n_m, dx_m, alloc, rng)
        if v_b < best_val:
            best_val = v_b
            best_h = x_b
            best_n = n_m
    
    # Phase 3: Fine grid (25% time)
    rem3 = budget_s - (time.time() - t_start) - 20
    if rem3 > 40:
        n_f = 2 * best_n + 1
        dx_f = 2.0 / n_f
        h_f = project_grid(best_h, best_n, n_f)
        t3 = min(rem3 * 0.45, 150)
        x_f, v_f = refine_anneal(h_f, n_f, dx_f, t3, rng)
        if v_f < best_val:
            best_val = v_f
            best_h = x_f
            best_n = n_f
    
    # Phase 4: Final polish with tight tau (remaining)
    rem4 = budget_s - (time.time() - t_start) - 5
    if rem4 > 12:
        dx_b = 2.0 / best_n
        xp = perturb_smooth(best_h, best_n, 0.02, rng)
        for t_au in [0.003, 0.001, 0.0005]:
            xp = slsqp_step(xp, best_n, dx_b, t_au, 250, 1e-14)
        vp = compute_c5(xp, dx_b)
        if vp < best_val:
            best_val = vp
            best_h = xp
        # Direct polish without perturbation
        xp2 = slsqp_step(best_h, best_n, dx_b, 0.001, 300, 1e-14)
        xp2 = slsqp_step(xp2, best_n, dx_b, 0.0005, 200, 1e-14)
        vp2 = compute_c5(xp2, dx_b)
        if vp2 < best_val:
            best_val = vp2
            best_h = xp2
    
    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    
    return (best_h, final_val, best_n)
