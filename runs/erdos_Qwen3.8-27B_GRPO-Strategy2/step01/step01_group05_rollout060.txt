Since each f_s is concave, f_s(h) ≤ t is a non-convex constraint. But for small k, we can use a sequential QP approach or just use SLSQP on the smooth max of these few functions.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return float(np.max(_corr(h, dx)))

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sm_obj(x, dx, n, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    sh = np.clip((cv - mx) / eps, -60, 0)
    return mx + eps * np.log(np.sum(np.exp(sh)))

def _sm_grad(x, dx, n, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    sh = np.clip((cv - mx) / eps, -60, 0)
    w = np.exp(sh)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n) & (jm + s >= 0) & (jm + s < n)
        if vm.any():
            g[jm[vm]] += wt * (1.0 - x[jm[vm] + s])
        jp = j + s
        vp = (jp >= 0) & (jp < n) & (jp - s >= 0) & (jp - s < n)
        if vp.any():
            g[jp[vp]] -= wt * x[jp[vp] - s]
    return g

def _slsqp(h0, dx, n, target, eps, maxiter=400):
    def fun(x):
        return _sm_obj(x, dx, n, eps)
    def jac(x):
        return _sm_grad(x, dx, n, eps)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, target)

def _subgrad(h, dx, n, target, lr, tol=1e-10):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - tol)[0]
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n) & (jm + s >= 0) & (jm + s < n)
        if vm.any():
            g[jm[vm]] += dx * (1.0 - h[jm[vm] + s])
        jp = j + s
        vp = (jp >= 0) & (jp < n) & (jp - s >= 0) & (jp - s < n)
        if vp.any():
            g[jp[vp]] -= dx * h[jp[vp] - s]
    g /= max(len(active), 1)
    return _proj(h - lr * g, target)

def _active_set_solve(h, dx, n, target, k_active=5, eps=0.0005, maxiter=200):
    cv = _corr(h, dx)
    idxs = np.argsort(cv)[-k_active:]
    def obj(x):
        c = _corr(x, dx)
        mx = c[idxs].max()
        sh = np.clip((c[idxs] - mx) / eps, -60, 0)
        return mx + eps * np.log(np.sum(np.exp(sh)))
    def jac(x):
        c = _corr(x, dx)
        mx = c[idxs].max()
        sh = np.clip((c[idxs] - mx) / eps, -60, 0)
        w = np.exp(sh)
        w /= w.sum()
        g = np.zeros(n)
        j = np.arange(n)
        for ii in range(len(idxs)):
            idx = idxs[ii]
            if w[ii] < 1e-15:
                continue
            wt = w[ii] * dx
            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n) & (jm + s >= 0) & (jm + s < n)
            if vm.any():
                g[jm[vm]] += wt * (1.0 - x[jm[vm] + s])
            jp = j + s
            vp = (jp >= 0) & (jp < n) & (jp - s >= 0) & (jp - s < n)
            if vp.any():
                g[jp[vp]] -= wt * x[jp[vp] - s]
        return g
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, target)

def _optimize_cycle(h, dx, n, target, rng, t_end):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    cur = h.copy()
    eps_schedule = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003, 0.0001, 0.00005]
    for eps in eps_schedule:
        if time.time() > t_end - 18:
            break
        try:
            h_new = _slsqp(cur, dx, n, target, eps, maxiter=350)
        except Exception:
            break
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        cur = h_new
    lr = 0.08
    for it in range(2000):
        if time.time() > t_end - 8:
            break
        h_new = _subgrad(cur, dx, n, target, lr)
        c_new = _c5(h_new, dx)
        c_cur = _c5(cur, dx)
        if c_new <= c_cur + 1e-14:
            cur = h_new
            lr = min(lr * 1.015, 1.5)
        else:
            lr = max(lr * 0.96, 0.003)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if it % 150 == 75:
            cur = _proj(best_h + rng.normal(0, 0.025, n), target)
    if time.time() < t_end - 6:
        for k_act in [3, 5, 8]:
            if time.time() > t_end - 3:
                break
            try:
                h2 = _active_set_solve(best_h, dx, n, target, k_act=k_act, eps=0.0003, maxiter=200)
                c2 = _c5(h2, dx)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()
            except Exception:
                pass
    if time.time() < t_end - 4:
        for eps in [0.0002, 0.0001, 0.00005]:
            if time.time() > t_end - 2:
                break
            try:
                h2 = _slsqp(best_h, dx, n, target, eps, maxiter=250)
                c2 = _c5(h2, dx)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()
            except Exception:
                break
    return best_h, best_c5

def _try_residue(n, dx, target, h_init, rng, t_end):
    best_h, best_c5 = _optimize_cycle(h_init, dx, n, target, rng, t_end)
    for restart in range(3):
        if time.time() > t_end:
            break
        amp = rng.uniform(0.04, 0.12)
        h_p = _proj(best_h + rng.normal(0, amp, n), target)
        c_p = _c5(h_p, dx)
        if c_p < best_c5:
            best_h = h_p.copy()
            best_c5 = c_p
        sub = min(time.time() + 15.0, t_end)
        if sub - time.time() > 5:
            h2, c2 = _optimize_cycle(h_p, dx, n, target, rng, sub)
            if c2 < best_c5:
                best_h = h2
                best_c5 = c2
    return best_h, best_c5, n
