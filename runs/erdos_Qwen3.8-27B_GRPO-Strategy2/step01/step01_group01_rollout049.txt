c_full[2n-2-k] = Σ_m h[m+k'']*(1-h[m]) where k'' = (2n-2-k)-(n-1) = n-1-k

So c_full[2n-2-k] uses lag n-1-k while c_full[k] uses lag k-(n-1).
c_full[k] with lag l = k-(n-1): Σ_m h[m+l]*(1-h[m])
c_full[2n-2-k] with lag l' = n-1-k = -(k-(n-1)) = -l: Σ_m h[m-l]*(1-h[m])

Are these the same? Let's substitute m' = m-l in the second:
Σ_{m'} h[m']*(1-h[m'+l]) = Σ_m' h[m'] - Σ_m' h[m']*h[m'+l]
First: Σ_m h[m+l]*(1-h[m]) = Σ_m h[m+l] - Σ_m h[m+l]*h[m]

These are not the same in general. So no simple symmetry to exploit.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    safe = (c - cmax) * temp
    safe = np.clip(safe, -400, 400)
    return (cmax + np.log(np.sum(np.exp(safe))) / temp) * dx


def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    n = n0
    h = repair(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP with adaptive decreasing temperature
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]

    temps = [500, 350, 250, 180, 130, 100, 80, 60, 50, 40, 35, 30, 25, 20, 18, 15]
    for temp in temps:
        if remaining() < budget_s * 0.55:
            break
        phase_budget = min(remaining() * 0.06, 50.0)
        maxiter = max(200, int(phase_budget / 0.008))
        res = minimize(smooth_obj, h.copy(), args=(temp,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 100})
        h_new = repair(res.x, n)
        v = compute_c5(h_new)
        if v < best_c5 - 1e-15:
            best_c5 = v
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Aggressive coordinate descent with many candidates
    if remaining() > budget_s * 0.2:
        n = len(h)
        target = n * 0.5
        dx = 2.0 / n
        time_per_sweep = 0.35
        max_sweeps = int(min(remaining() * 0.55, 250.0) / time_per_sweep)
        max_sweeps = max(max_sweeps, 50)

        for sweep in range(max_sweeps):
            if remaining() < budget_s * 0.25:
                break
            improved = False
            order = np.random.permutation(n) if (sweep % 3 == 0) else range(n)
            for i in order:
                old = h[i]
                if old < 1e-10:
                    cands = [0.0, 0.03, 0.07, 0.15, 0.3, 0.5]

                    cands.append(max(0.0, min(1.0, 1.0 - old * 0.5)))

                    if val < best_val - 1e-14:

                        best_cand = cand

                h[i] = best_cand

                    diff = target - h.sum()
                    if abs(diff) > 1e-15:
                        j = (i + 1) % n
                        h[j] = np.clip(h[j] + diff, 0.0, 1.0)

                    improved = True
                else:
                    h[i] = old
                    diff = target - h.sum()
                    if abs(diff) > 1e-15:

                        h[j] = np.clip(h[j] + diff, 0.0, 1.0)

                h = h_new.copy()

    # Phase 4: Small random perturbations + SLSQP
    if remaining() > budget_s * 0.04:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                 'jac': lambda x: np.ones(n)}]
        for attempt in range(8):
            if remaining() < budget_s * 0.015:
                break
            scale = 0.02 + 0.01 * attempt
            h_try = best_h + np.random.randn(n) * scale
            h_try = repair(h_try, n)
            phase_budget = min(remaining() * 0.005, 10.0)
            maxiter = max(100, int(phase_budget / 0.01))
            res = minimize(smooth_obj, h_try, args=(50,),
                         method='SLSQP', bounds=bnds, constraints=cons,

            if v < best_c5 - 1e-15:

    # Phase 5: Try half resolution
    if remaining() > budget_s * 0.03 and len(best_h) > 100:
        n_half = len(best_h) // 2
        if n_half >= 50:
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n_half)
            h_half = np.interp(xn, xo, best_h)
            h_half = repair(h_half, n_half)
            v_half = compute_c5(h_half)
            if v_half < best_c5:
                best_c5 = v_half
                best_h = h_half.copy()
            if remaining() > budget_s * 0.015:
                bnds_h = [(0.0, 1.0)] * n_half
                cons_h = [{'type': 'eq', 'fun': lambda x: x.sum() - n_half * 0.5,
                           'jac': lambda x: np.ones(n_half)}]
                maxiter = max(100, int(min(remaining() * 0.01, 15.0) / 0.008))

                             method='SLSQP', bounds=bnds_h, constraints=cons_h,

                h_new = repair(res.x, n_half)
                v = compute_c5(h_new)
                if v < best_c5:

                    best_h = h_new.copy()

    # Final evaluation
    best_h = repair(best_h, len(best_h))
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, len(best_h))
