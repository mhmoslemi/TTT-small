import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    return float(np.max(np.correlate(h, 1-h, mode='full'))) * (2.0/len(h))

def _proj(h, target):
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        # bisection for uniform shift
        lo, hi = -max(h)-0.01, 1-max(h)+0.01
        for _ in range(60):
            m = (lo+hi)/2
            if np.clip(h+m, 0, 1).sum() > target: hi = m
            else: lo = m
        h = np.clip(h + (lo+hi)/2, 0, 1)
    return h

def _obj_grad(h, alpha):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    m = np.max(corr)
    obj = (m + np.log(np.sum(np.exp(alpha*(corr-m))))/alpha) * dx
    w = np.exp(alpha*(corr-m)); w /= w.sum()
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12: continue
        s = k - (n-1)
        if s >= 0:
            g[:n-s] += wk*(1.0-h[s:])
            g[s:]  -= wk*h[:n-s]
        else:
            g[-s:] += wk*(1.0-h[:n+s])
            g[:n+s] -= wk*h[-s:]
    return obj, g*dx

def _slsqp(h0, n, alpha, iters):
    target = n/2.0
    bounds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-target,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _obj_grad(x,alpha)[0], h0, method='SLSQP',
                       jac=lambda x: _obj_grad(x,alpha)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter':iters,'ftol':1e-12})
        return _proj(res.x, target)
    except: return h0

def _coord_desc(h, target, sweeps=50):
    """Fast coordinate descent using analytical structure"""
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    cmax = corr.max()
    
    h_ext = np.zeros(3*n-1)
    h_ext[n-1:n-1+n] = h
    ek = np.arange(2*n-1)  # index k
    
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in [(i+1)%n, (i-1+n)%n, (i+n//2)%n, (i+n//3)%n]:
                if j == i: continue
                d_lo = max(-h[i], h[j]-1.0)
                d_hi = min(1.0-h[i], h[j])
                if d_hi - d_lo < 1e-12: continue
                
                # Update h_ext for i and j
                h_ext[n-1+i] = h[i]
                h_ext[n-1+j] = h[j]
                
                hp_i = h_ext[i:i+2*n-1]
                hm_i = h_ext[i:i+2*n-1][::-1]
                hp_j = h_ext[j:j+2*n-1]
                hm_j = h_ext[j:j+2*n-1][::-1]
                
                c = hp_j + hm_j - hp_i - hm_i
                
                # Ternary search for optimal delta
                lo, hi = d_lo, d_hi
                for _ in range(50):
                    m1 = lo + (hi-lo)/3
                    m2 = hi - (hi-lo)/3
                    if np.max(corr + m1*c) < np.max(corr + m2*c):
                        hi = m2
                    else:
                        lo = m1
                delta = (lo+hi)/2
                fn = np.max(corr + delta*c)
                if fn < cmax - 1e-11:
                    h[i] += delta; h[j] -= delta
                    corr = corr + delta*c
                    cmax = fn
                    improved = True
        
        # Rebuild h_ext
        h_ext[n-1:n-1+n] = h
        if not improved: break
    
    return h, cmax*dx

import time

_g_target = 0.0
_g_alpha = 100.0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        lo, hi = -float(h.max()) - 0.01, 1.0 - float(h.max()) + 0.01
        for _ in range(50):
            m = 0.5*(lo+hi)
            if np.clip(h+m, 0.0, 1.0).sum() > target:
                hi = m
            else:
                lo = m
        h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    return h

def _obj_grad(h, alpha):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    mx = float(np.max(corr))
    obj = (mx + np.log(np.sum(np.exp(alpha*(corr-mx))))/alpha) * dx
    w = np.exp(alpha*(corr-mx))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n-1)
        if s >= 0:
            g[:n-s] += wk*(1.0-h[s:])
            g[s:] -= wk*h[:n-s]
        else:
            g[-s:] += wk*(1.0-h[:n+s])
            g[:n+s] -= wk*h[-s:]
    return obj, g*dx

def _obj_fn(x):
    return _obj_grad(x, _g_alpha)[0]

def _obj_jac(x):
    return _obj_grad(x, _g_alpha)[1]

def _cons_fun(x):
    return x.sum() - _g_target

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h0, n, alpha, iters):
    global _g_target, _g_alpha
    _g_target = n/2.0
    _g_alpha = alpha
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':_cons_fun,'jac':_cons_jac}]
    try:
        res = minimize(_obj_fn, h0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter':iters,'ftol':1e-12})
        return _proj(res.x, n/2.0)
    except Exception:
        return h0

def _coord_desc(h, target, sweeps):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    cmax = float(corr.max())
    h_ext = np.zeros(3*n-1)
    h_ext[n-1:n-1+n] = h
    
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in [(i+1)%n, (i-1+n)%n, (i+n//2)%n, (i+n//3)%n]:
                if j == i:
                    continue
                d_lo = max(-h[i], h[j]-1.0)
                d_hi = min(1.0-h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                hp_i = h_ext[i:i+2*n-1]
                hm_i = hp_i[::-1]
                hp_j = h_ext[j:j+2*n-1]
                hm_j = hp_j[::-1]
                c = hp_j + hm_j - hp_i - hm_i
                lo, hi = d_lo, d_hi
                for _ in range(40):
                    m1 = lo + (hi-lo)/3.0
                    m2 = hi - (hi-lo)/3.0
                    if np.max(corr + m1*c) <= np.max(corr + m2*c):
                        hi = m2
                    else:
                        lo = m1
                delta = 0.5*(lo+hi)
                fn = np.max(corr + delta*c)
                if fn < cmax - 1e-11:
                    h[i] += delta
                    h[j] -= delta
                    h_ext[n-1+i] = h[i]
                    h_ext[n-1+j] = h[j]
                    corr = corr + delta*c
                    cmax = float(fn)
                    improved = True
        if not improved:
            break
    return h, cmax*dx
