def project_feasible(h, n):
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h_new = np.clip(h - lam, 0.0, 1.0)
    # Final adjustment for floating point
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

    x = np.clip(x, 0.0, 1.0)

    diff = target - x_proj.sum()

        mask = (x_proj > 1e-6) & (x_proj < 1.0 - 1e-6)
        if mask.sum() > 0:
            x_proj[mask] += diff / mask.sum()
            x_proj = np.clip(x_proj, 0.0, 1.0)
    return x_proj

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        m = (x > 1e-8) & (x < 1.0 - 1e-8)
        cnt = m.sum()
        if cnt > 0:
            x[m] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def get_gradient(h, p, n):
    grad = np.zeros(n)
    oh = 1.0 - h
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += pm * oh[:n - k]
            grad[:n - k] -= pm * h[k:]
        else:
            kk = -k
            grad[:n - kk] += pm * oh[kk:]
            grad[kk:] -= pm * h[:n - kk]
    return grad

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def pgd_optimize(h0, n, time_limit, n_starts=3, seed=0):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = compute_c5(h0, dx)
    
    # Starting points
    starts = [h0.copy()]
    for i in range(n_starts - 1):
        sc = 0.05 + 0.03 * i
        starts.append(project_feasible(h0 + sc * rng.standard_normal(n), n))
    
    taus = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    per_budget = (time_limit - 1.0) / (n_starts * len(taus))
    
    for si, s in enumerate(starts):
        h = s.copy()
        h_prev = h.copy()
        momentum = 0.85
        for tau in taus:
            elapsed = time.time() - t0
            if elapsed > time_limit - 1.0:
                return best_h, best_val
            n_steps = int(per_budget / (time_limit / (n_starts * len(taus))))
            n_steps = max(n_steps, 50)
            lr = 0.5 / np.sqrt(n)
            
            for step in range(n_steps):
                if time.time() - t0 > time_limit - 0.5:
                    return best_h, best_val
                # Decay learning rate
                lr_step = lr * (0.999 ** step)
                
                corr = get_corr(h)
                c = corr - corr.max()
                p = np.exp(c / tau)
                p_sum = p.sum()
                p /= p_sum
                f_sm = corr.max() + tau * np.log(p_sum)
                
                grad = get_gradient(h, p, n)
                h_new = project_feasible(h + momentum * (h - h_prev) - lr_step * grad, n)
                h_prev = h
                h = h_new
                
                val = compute_c5(h, dx)
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
    
    return best_h, best_val

def upsample(h, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, h, kind='linear', fill_value='extrapolate')(xd)
    return project_feasible(xp, n_dst)

def slsqq_refine(h, n, dx, maxiter=300, tau=0.01):
    global _N, _dx_g, _tau_g
    _N = n
    _dx_g = dx
    _tau_g = tau
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        sh = np.clip((corr - m) / _tau_g, -50.0, 0.0)
        return (m + _tau_g * np.log(np.sum(np.exp(sh)))) * _dx_g
    
    def cfun(x):
        return np.sum(x) - _N / 2.0
    
    def cjac(x):
        return np.ones(_N)

                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

_tau_g = 1.0

    best_h = project_feasible(best_h, best_n)
    final_val = compute_c5(best_h, dx_fin)
    
    return (best_h, final_val, best_n)
