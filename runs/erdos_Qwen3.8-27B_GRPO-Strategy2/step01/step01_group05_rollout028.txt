def _slsqp_polish(h, dx, n, target, eps, maxiter=200):

    N = n
    EP = eps

        mx = cv.max()
        if EP < 1e-15:

        mx = cv.max()
        w = np.exp((cv - mx) / EP)
        w /= w.sum()

            jm = j_arr - s

            jp = j_arr + s
            vp = (jp >= 0) & (jp < N)
            if vp.any():
                g[vp] -= wt * x[jp[vp]]
        return g
    
    def efun(x):
        return x.sum() - target
    
    def ejac(x):
        return np.ones(N)
    
    try:
        res = minimize(sobj, h, jac=sgrad,
                       bounds=[(0.0,1.0)]*N,
                       constraints=[{'type':'eq','fun':efun,'jac':ejac}],
                       method='SLSQP', options={'maxiter':maxiter,'ftol':1e-14})
        return _proj(res.x, target)
    except:
