def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:

            ones.remove(i)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def swap_eval(h, i, j):
    """Evaluate C5 after swapping h[i] and h[j]."""
    # h[i]=1, h[j]=0 -> h[i]=0, h[j]=1
    h[i] = 0.0
    h[j] = 1.0
    c = compute_c5(h)
    h[i] = 1.0
    h[j] = 0.0
    return c

def local_search(h, n, time_limit):
    """Try all single swaps until no improvement."""
    best_c5 = compute_c5(h)
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    t0 = time.time()
    
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        for i in list(ones):
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    ones.discard(i)
                    zeros.discard(j)
                    ones.add(j)
                    zeros.add(i)
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
    return h, best_c5

def sa_search(h, n, t_start, duration, seed):
    """Simulated annealing with swap moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    
    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h < 0.5)[0].tolist()
    
    T = 0.005
    T_min = 1e-7
    # Determine cooling rate based on duration
    n_steps = 500000  # target number of steps
    cooling = (T_min / T) ** (1.0 / n_steps)
    
    end_t = t_start + duration
    steps = 0
    while time.time() < end_t and T > T_min:
        T *= cooling
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones_list[ones_list.index(i)] = i  # update
            # Actually need to swap in the lists
            ones_list[ones_list.index(i)] = j
            zeros_list[zeros_list.index(j)] = i
        else:
            h[i] = 1.0
            h[j] = 0.0
        steps += 1
    
    return best_h, best_c5

One more important consideration: the problem says n_points samples over [0,2], and the correlation is computed on these n points. The shift k in the correlation goes from -(n-1) to (n-1), representing shifts from -(n-1)*dx to (n-1)*dx.

For a 0/1 solution, the "count" at shift k is the number of positions where h has 1 and h shifted by k has 0 (within the valid overlap region). This is NOT the same as the cyclic count |A Δ (A+k)| that I was computing in Z_n.

The valid overlap region matters: for shift k (0 ≤ k ≤ n-1), the count is over i = 0 to n-1-k. So there are n-k terms, not n terms.

This means large shifts have fewer terms and thus smaller counts automatically. The worst shifts are small k (like k=1, 2, ..., 10) where most terms contribute.

But this is a per-coordinate argument. The joint optimum might not be 0/1 if the coordinates are correlated through the constraint.

Counter-example: suppose max of {2x, -2x} over x ∈ [0,1] is minimized at x = 0.5 (value 1), not at x=0 or x=1 (value 2). Wait no: max(2*0, 0) = 0 at x=0, max(2*1, -2) = 2 at x=1. So min is at x=0, value 0. OK so max of linears IS minimized at the boundary of [0,1].

More carefully: max of a family of linear functions of x is a convex piecewise-linear function of x. The minimum of a convex function on [0,1] is at the boundary (0 or 1) UNLESS the function is flat (constant) on an interval, in which case any point in that interval works (including the boundary).

So YES, for each coordinate individually, the optimal is 0 or 1. And since we can do this for all coordinates simultaneously (the constraint sum=h[n/2] is satisfied by any 0/1 vector with n/2 ones), the global optimum IS 0/1.
