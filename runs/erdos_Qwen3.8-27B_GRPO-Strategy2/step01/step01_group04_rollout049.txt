import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gN = 0
_gdx = 1.0

def _corr(x):
    return np.correlate(x, 1.0 - x, mode="full")

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = 0.5 * n
    s = np.sum(x)
    if s < 1e-12:
        return np.full(n, 0.5)
    x *= tgt / s
    x = np.clip(x, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        mask = (x < 1.0) if diff > 0 else (x > 0.0)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
    return np.clip(x, 0.0, 1.0)

def _oproj(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = 0.5 * n
    if abs(np.sum(x) - tgt) < 1e-10:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        t = np.sum(np.clip(x - mid, 0.0, 1.0))
        if t > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _smax(x, tau, topK):
    c = _corr(x)
    m = c.max()
    if topK is not None and topK < len(c):
        idx = np.argpartition(-c, topK)[:topK]
        t = c[idx]
        return (m + tau * np.log(np.sum(np.exp((t - m) / tau)))) * _gdx
    sh = (c - m) / tau
    return (m + tau * np.log(np.sum(np.exp(sh)))) * _gdx

def _cf(x):
    return np.sum(x) - _gN * 0.5

def _cj(x):
    return np.ones(_gN)

def _slsqp(x0, n, dx, tau, topK, mi, ft):
    global _gN, _gdx
    _gN = n
    _gdx = dx
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]

    def f(x):
        return _smax(x, tau, topK)

    try:
        r = minimize(f, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': mi, 'ftol': ft})
        if r.x is not None:
            if _c5(r.x, dx) < _c5(x0, dx) + 1e-12:
                return _repair(r.x, n)
    except Exception:
        pass
    return x0.copy()

def _corr_grad(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g

def _subgrad(h, n, dx, niter, a0, topK):
    h = h.copy()
    best = h.copy()
    bv = _c5(h, dx)
    for it in range(niter):
        c = _corr(h)
        ks = np.argsort(-c)[:topK]
        tc = c[ks]
        w = np.exp(tc / 0.005 - tc.max())
        w /= w.sum()
        g = np.zeros(n)
        for wj, kj in zip(w, ks):
            g += wj * _corr_grad(h, int(kj) - (n - 1), n)
        a = a0 / (1.0 + 0.03 * it)
        h = _oproj(h - a * g, n)
        v = _c5(h, dx)
        if v < bv:
            bv = v
            best = h.copy()
    return best, bv

def _proj(x, ns, nd):
    if ns == nd:
        return _repair(x.copy(), nd)
    xs = np.linspace(0.0, 2.0, ns)
    xd = np.linspace(0.0, 2.0, nd)
    xp = interp1d(xs, x, kind='linear', fill_value='extrapolate')(xd)
    return _repair(xp, nd)

def _pert(x, sc, rng):
    return _repair(x + sc * rng.standard_normal(len(x)), len(x))
