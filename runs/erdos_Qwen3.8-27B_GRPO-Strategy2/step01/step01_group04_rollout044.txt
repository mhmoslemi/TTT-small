    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if abs(s) > 1e-12:

    x = np.clip(x, 0.0, 1.0)

        # Distribute to elements farthest from bounds
        room_up = 1.0 - x
        room_down = x
        if diff > 0:
            # Need to increase
            order = np.argsort(-room_up)
            remaining = diff
            for i in order:
                add = min(remaining, room_up[i])
                x[i] += add
                remaining -= add
                if remaining < 1e-12:
                    break
        else:
            order = np.argsort(-room_down)
            remaining = -diff
            for i in order:
                sub = min(remaining, room_down[i])
                x[i] -= sub
                remaining -= sub
                if remaining < 1e-12:
                    break

    """Approximate max using p-norm."""
    corr_scaled = corr * dx
    m = np.max(corr_scaled)
    if m <= 0:
        return 0.0
    return m * np.sum((corr_scaled / m) ** p) ** (1.0 / p)

def obj_smooth(x, n, dx, p, tau):
    """Smooth objective combining p-norm and log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if tau > 0:
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return m + tau * np.log(np.sum(np.exp(sh)))
    else:
        # p-norm
        if p >= 1:
            return m * np.sum((np.maximum(corr, 0) / m) ** p) ** (1.0 / p)
        return m
