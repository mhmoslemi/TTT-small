def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()

    s = np.sum(h)

    diff = target - np.sum(h)

        mask = (h > 1e-8) & (h < 1.0 - 1e-8)

        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:

            h = np.clip(h + diff/n, 0.0, 1.0)
    return h

    """Compute subgradient direction for the max correlation."""
    dx = 2.0 / n
    corr = compute_corr(h, n)
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol)[0]
    
    sg = np.zeros(n)
    center = n - 1
    for m in active:
        d = m - center  # effective shift
        # dg/ dh[j] = (1-h[j+d]) if j+d in [0,n) else 0

        j = np.arange(n)

        jd_neg = j - d
        sg += np.where((jd >= 0) & (jd < n), 1.0 - h[jd], 0.0)
        sg -= np.where((jd_neg >= 0) & (jd_neg < n), h[jd_neg], 0.0)

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 70

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) > 0:
            h[interior[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lse(h, n, alpha):
    corr = compute_corr(h, n)
    scaled = alpha * corr
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p /= np.sum(p)
    center = n - 1
    j = np.arange(n)
    g = np.zeros(n)
    for idx in range(len(corr)):
        if p[idx] < 1e-7:
            continue
        d = idx - center
        jd = j + d
        jm = j - d
        g += p[idx] * (np.where((jd >= 0) & (jd < n), 1.0 - h[jd], 0.0)
                     - np.where((jm >= 0) & (jm < n), h[jm], 0.0))
    return g

def subgrad_top(h, n, k_top=3, tol_frac=0.05):
    corr = compute_corr(h, n)
    max_val = np.max(corr)
    tol = tol_frac * max_val + 1e-12
    active = np.where(corr >= max_val - tol)[0]
    active = active[:k_top]
    center = n - 1
    j = np.arange(n)
    sg = np.zeros(n)
    for m in active:
        d = m - center
        jd = j + d
        jm = j - d
        sg += np.where((jd >= 0) & (jd < n), 1.0 - h[jd], 0.0)
        sg -= np.where((jm >= 0) & (jm < n), h[jm], 0.0)
    sg /= max(len(active), 1)
    return sg

def _slsqp_obj(x):
    n = _global_n
    return compute_c5(x, n)

def _slsqp_eq(x):
    n = _global_n
    return np.sum(x) - n / 2.0

def make_init(n, rng):
    best_h = np.full(n, 0.5)
    best_c = compute_c5(best_h, n)
    for _ in range(30):
        f1 = rng.uniform(1, 10)
        f2 = rng.uniform(10, 30)
        a1 = rng.uniform(0.1, 0.45)
        a2 = rng.uniform(0.05, 0.25)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        trial = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2)
        trial = repair(trial, n)
        c = compute_c5(trial, n)
        if c < best_c:
            best_c = c
            best_h = trial.copy()
    return best_h

def optimize_n(h, n, t0, t_end, rng):
    global _global_n
    _global_n = n
    c5 = compute_c5(h, n)
    best_local = h.copy()
    best_c5_local = c5
    stall = 0
    it = 0
    momentum = np.zeros(n)
    mu = 0.3

    while time.time() < t_end:
        it += 1
        improved = False

        # A: Nesterov-accelerated subgradient on LSE
        alpha_lse = 8.0 + 4.0 * (it % 8)
        g = grad_lse(h, n, alpha_lse)
        lookup = 0.5 / (1.0 + 0.05 * it)
        h Lookup = repair(h - lookup * momentum, n)
        g2 = grad_lse(h_lookup, n, alpha_lse)
        momentum = mu * momentum + (1 - mu) * g2
        h_lookup2 = repair(h_lookup - lookup * momentum, n)
        c_lookup = compute_c5(h_lookup2, n)
        if c_lookup < c5 - 1e-14:
            h, c5 = h_lookup2, c_lookup
            improved = True
            if c5 < best_c5_local:
                best_c5_local, best_local = c5, h.copy()
        else:
            # Backtracking
            for factor in [0.5, 0.25, 0.1]:
                h_try = repair(h - lookup * factor * momentum, n)
                c_try = compute_c5(h_try, n)
                if c_try < c5 - 1e-14:
                    h, c5 = h_try, c_try
                    improved = True
                    if c5 < best_c5_local:
                        best_c5_local, best_local = c5, h.copy()
                    break

        # B: Exact top-k subgradient step
        if not improved and it % 2 == 0:
            sg = subgrad_top(h, n, k_top=3, tol_frac=0.03)
            for step in [0.15, 0.08, 0.04, 0.02]:
                h_try = repair(h - step * sg, n)
                c_try = compute_c5(h_try, n)
                if c_try < c5 - 1e-14:
                    h, c5 = h_try, c_try
                    improved = True
                    if c5 < best_c5_local:
                        best_c5_local, best_local = c5, h.copy()
                    break

        # C: Targeted pair moves on active lags
        if not improved and it % 4 == 0:
            corr = compute_corr(h, n)
            max_c = np.max(corr)
            center = n - 1
            active = np.where(corr >= max_c - 0.1 * max_c)[0][:4]
            for m in active:
                d = m - center
                j = np.arange(n)
                jd = j + d
                jm = j - d
                valid = (jd >= 0) & (jd < n) & (jm >= 0) & (jm < n)
                # Direction: reduce h[jd] where high, increase h[jm] where low
                for eps in [0.04, 0.02, 0.01]:
                    h_try = h.copy()
                    idx_high = np.where(valid & (h[jd] > 0.5))[0]
                    idx_low = np.where(valid & (h[jm] < 0.5))[0]
                    if len(idx_high) > 0 and len(idx_low) > 0:
                        pi = int(rng.randint(len(idx_high)))
                        pj = int(rng.randint(len(idx_low)))
                        h_try[jd[idx_high[pi]]] -= eps
                        h_try[jm[idx_low[pj]]] += eps
                    h_try = repair(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < c5 - 1e-14:
                        h, c5 = h_try, c_try
                        improved = True
                        if c5 < best_c5_local:
                            best_c5_local, best_local = c5, h.copy()
                        break

        # D: Annealing perturbation
        if not improved and stall > 4:
            sigma = 0.02 + 0.004 * min(stall, 30)
            h_pert = h + rng.normal(0, sigma, n)
            h_pert = repair(h_pert, n)
            c_pert = compute_c5(h_pert, n)
            temp = 0.002 * (1.0 + 0.05 * stall)
            if c_pert < c5 - 1e-14 or rng.random() < np.exp(-(c_pert - c5) / temp):
                h, c5 = h_pert, c_pert
                improved = True
                if c5 < best_c5_local:
                    best_c5_local, best_local = c5, h.copy()

        # E: Block flip (exploration)
        if not improved and it % 12 == 0:
            blen = int(rng.randint(2, min(7, n // 5)))
            bstart = int(rng.randint(0, n - blen))
            h_flip = h.copy()
            h_flip[bstart:bstart + blen] = 1.0 - h_flip[bstart:bstart + blen]
            h_flip = repair(h_flip, n)
            c_flip = compute_c5(h_flip, n)
            if c_flip < c5 + 0.001:
                h, c5 = h_flip, c_flip
                if c5 < best_c5_local:
                    best_c5_local, best_local = c5, h.copy()
                improved = True

        # F: SLSQP polish when stalling
        if not improved and stall > 15 and it % 15 == 0 and time.time() < t_end - 10:
            try:
                bnd = [(0.0, 1.0)] * n
                r = minimize(_slsqp_obj, h, method='SLSQP', bounds=bnd,
                             constraints=[{'type': 'eq', 'fun': _slsqp_eq}],
                             options={'maxiter': 300, 'ftol': 1e-14})
                h_s = repair(r.x, n)
                c_s = compute_c5(h_s, n)
                if c_s < c5 - 1e-14:
                    h, c5 = h_s, c_s
                    stall = 0
                    if c5 < best_c5_local:
                        best_c5_local, best_local = c5, h.copy()
            except Exception:
                pass

        stall = 0 if improved else stall + 1

    # Final polish
    if time.time() < t_end - 2:
        for _ in range(300):
            if time.time() > t_end - 1:
                break
            g = grad_lse(h, n, 25.0)
            for a in [0.05, 0.02, 0.01, 0.005]:
                h_try = repair(h - a * g, n)
                c_try = compute_c5(h_try, n)
                if c_try < c5 - 1e-15:
                    h, c5 = h_try, c_try
                    if c5 < best_c5_local:
                        best_c5_local, best_local = c5, h.copy()
                    break
    return best_local, best_c5_local

def run(seed=42, budget_s=1100, **kwargs):
    global _global_n
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    best_h = None
    best_c5_global = 1.0
    best_n_global = 0

    # Determine n values to try
    n_primary = len(init) if init is not None else 70
    n_list = sorted(set([n_primary, 50, 60, 70, 80, 90, 100]))

    # Time allocation: primary gets 40%, others split the rest
    total_time = budget_s - 15
    primary_share = 0.40
    other_share = (1.0 - primary_share) / max(len(n_list) - 1, 1)

    for n in n_list:
        if time.time() > t0 + total_time - 5:
            break
        if n == n_primary:
            t_end = t0 + total_time * primary_share + 30
        else:
            t_end = t0 + total_time * other_share + 5
        t_end = min(t_end, t0 + budget_s - 12)
        if time.time() > t_end:
            continue

        # Initialize
        if init is not None and len(init) == n:
            h = repair(np.array(init, dtype=float).copy(), n)
        else:
            h = make_init(n, rng)

        # Coarse-to-fine: if n > 60, also try upsampled from n//2
        if n > 60:
            n_coarse = n // 2
            h_c = make_init(n_coarse, rng)
            hc_res = optimize_n(h_c, n_coarse, t0, min(time.time() + 5, t_end - 20), rng)
            # Upsample
            x_c = np.linspace(0, 1, n_coarse, endpoint=False)
            x_f = np.linspace(0, 1, n, endpoint=False)
            h_up = np.interp(x_f, x_c, hc_res[0])
            h_up = repair(h_up, n)
            if compute_c5(h_up, n) < compute_c5(h, n):
                h = h_up

        # Main optimization
        h_res, c5_res = optimize_n(h, n, t0, t_end, rng)
        if c5_res < best_c5_global:
            best_c5_global = c5_res
            best_h = h_res.copy()
            best_n_global = n

    # Final SLSQP on best
    global _global_n
    if best_h is not None and time.time() < t0 + budget_s - 5:
        _global_n = best_n_global
        try:
            bnd = [(0.0, 1.0)] * best_n_global
            r = minimize(_slsqp_obj, best_h, method='SLSQP', bounds=bnd,
                         constraints=[{'type': 'eq', 'fun': _slsqp_eq}],
                         options={'maxiter': 500, 'ftol': 1e-15})
            hf = repair(r.x, best_n_global)
            cf = compute_c5(hf, best_n_global)
            if cf < best_c5_global:
                best_h, best_c5_global = hf, cf
        except Exception:
            pass

    if best_h is None:
        best_h = np.full(n_primary, 0.5)
        best_n_global = n_primary
        best_c5_global = compute_c5(best_h, best_n_global)

    best_h = repair(best_h, best_n_global)
    best_c5_global = compute_c5(best_h, best_n_global)
    return (best_h, best_c5_global, best_n_global)
