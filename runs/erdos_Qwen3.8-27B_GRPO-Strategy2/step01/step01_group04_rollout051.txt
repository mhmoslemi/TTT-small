        mask = x < 1.0 if diff > 0 else x > 0.0
        if np.any(mask):
            room = np.sum(1.0 - x[mask]) if diff > 0 else np.sum(x[mask])
            if room > 1e-15:
                x[mask] += diff * (1.0 - x[mask]) / room if diff > 0 else diff * x[mask] / room
            x = np.clip(x, 0.0, 1.0)
    return x

def project_box_sum(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(50):
        s = np.sum(x)
        if abs(s - tgt) < 1e-11:
            break
        if s > tgt:
            mask = x > 1e-14
            if not np.any(mask):
                break
            excess = s - tgt
            x[mask] -= excess * x[mask] / np.sum(x[mask])
        else:
            mask = x < 1.0 - 1e-14
            if not np.any(mask):
                break
            deficit = tgt - s
            room = np.sum(1.0 - x[mask])
            if room < 1e-14:
                break
            x[mask] += deficit * (1.0 - x[mask]) / room
        x = np.clip(x, 0.0, 1.0)
    return x

def soft_obj(x, tau, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

def soft_grad(x, tau, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    ew = np.exp(sh)
    sw = np.sum(ew)
    w = ew / sw
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wv = w[m_idx]
        if wv < 1e-14:
            continue
        l = m_idx - (n - 1)
        j1_lo = max(0, l)
        j1_hi = min(n - 1, l + n - 1)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += wv * (1.0 - x[j1_lo - l:j1_hi - l + 1])
        j2_lo = max(0, -l)
        j2_hi = min(n - 1, n - 1 - l)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] -= wv * x[j2_lo + l:j2_hi + l + 1]
    return grad * dx

def project_gradient(x, n, dx, tau, max_iter=5000, t_limit=None, t_start=None):
    x = project_box_sum(x.copy(), n)
    best_x = x.copy()
    best_val = compute_c5(x, dx)
    obj0 = soft_obj(x, tau, dx)
    step = 1.0
    prev_x = x.copy()
    prev_g = soft_grad(x, tau, dx)
    prev_obj = obj0
    stall = 0
    for it in range(max_iter):
        if t_start is not None and time.time() - t_start > t_limit:
            break
        g = soft_grad(x, tau, dx)
        s = np.dot(x - prev_x, g - prev_g)
        d = np.dot(x - prev_x, x - prev_x)
        if s > 1e-12 and d > 1e-12:
            step = np.clip(d / s, 0.1, 5.0)
        prev_x = x.copy()
        prev_g = g
        x_new = project_box_sum(x - step * g, n)
        obj_new = soft_obj(x_new, tau, dx)
        if obj_new < prev_obj or it == 0:
            x = x_new
            prev_obj = obj_new
            val = compute_c5(x, dx)
            if val < best_val:
                best_val = val
                best_x = x.copy()
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break
        else:
            step *= 0.7
        prev_obj = obj_new
    return best_x, best_val

def slsqp_opt(x0, n, dx, tau, max_iter=300, ftol=1e-15):
    def objf(xx):
        corr = np.correlate(xx, 1.0 - xx, mode="full")
        m = np.max(corr)
        sh = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    bd = [(0.0, 1.0)] * n
    def cfun(xx):
        return np.sum(xx) - n / 2.0
    def cjac(xx):
        return np.ones(n)
    from scipy.optimize import minimize
    try:
        r = minimize(objf, x0, method='SLSQP', bounds=bd,
                     constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                     options={'maxiter': max_iter, 'ftol': ftol})
        if r.success or r.fun < objf(x0):
            return repair(r.x, n)
    except Exception:
        pass
    return x0.copy()

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    xp = interp1d(xs, x_src, kind='linear', fill_value=(x_src[0], x_src[-1]),
                  bounds_error=False)(xd)
    return repair(xp, n_dst)

def coord_polish(x, n, dx, max_sweeps=5, step_sz=0.02):
    x = x.copy()
    best_val = compute_c5(x, dx)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            for delta in [step_sz, -step_sz]:
                if x[j] + delta < 0 or x[j] + delta > 1:
                    continue
                x_old = x[j]
                x[j] += delta
                x = project_box_sum(x, n)
                val = compute_c5(x, dx)
                if val < best_val - 1e-10:
                    best_val = val
                    improved = True
                else:
                    x[j] = x_old
                    x = project_box_sum(x, n)
        if not improved:
            break
    return x, best_val

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float).ravel()
    else:
        h0_raw = np.full(191, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)
    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    if time.time() - t0 > budget_s * 0.05:
        return (best_h, best_val, best_n)

    # Phase 1: PG optimization on base grid with tau annealing
    taus = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    x_curr = h0.copy()
    p1_end = t0 + budget_s * 0.40
    for idx, tau in enumerate(taus):
        if time.time() > p1_end:
            break
        remaining_frac = max(0.0, 1.0 - (time.time() - t0) / (p1_end - t0))
        ni = int(2000 + 3000 * remaining_frac)
        x_curr, val = project_gradient(x_curr, n0, dx0, tau, max_iter=ni,
                                        t_start=t0, t_limit=p1_end - t0)
        if val < best_val:
            best_val = val
            best_h = x_curr.copy()
            best_n = n0

    # Phase 2: SLSQP refinement at best tau
    if time.time() - t0 < t0 + budget_s * 0.55:
        p2_end = t0 + budget_s * 0.55
        x_s = slsqp_opt(best_h, best_n, 2.0 / best_n, 0.002, 400)
        val_s = compute_c5(x_s, 2.0 / best_n)
        if val_s < best_val:
            best_val = val_s
            best_h = x_s.copy()
        # Try another SLSQP from slight perturbation
        if time.time() < p2_end - 2:
            x_p = perturb(best_h, best_n, 0.08, rng)
            x_s2 = slsqp_opt(x_p, best_n, 2.0 / best_n, 0.005, 200)
            val_s2 = compute_c5(x_s2, 2.0 / best_n)
            if val_s2 < best_val:
                best_val = val_s2
                best_h = x_s2.copy()

    # Phase 3: Project to finer grid (2n+1)
    if time.time() - t0 < t0 + budget_s * 0.70:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        x_fin = project_grid(best_h, best_n, n1)
        p3_end = t0 + budget_s * 0.70
        for tau in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            if time.time() > p3_end:
                break
            ni = 1500
            x_fin, val = project_gradient(x_fin, n1, dx1, tau, max_iter=ni,
                                           t_start=t0, t_limit=p3_end - t0)
            if val < best_val:
                best_val = val
                best_h = x_fin.copy()
                best_n = n1

    # Phase 4: Try coarser grid for diversity
    if time.time() - t0 < t0 + budget_s * 0.78:
        n_coarse = best_n // 2 + 1
        if n_coarse >= 50:
            dx_c = 2.0 / n_coarse
            x_co = project_grid(best_h, best_n, n_coarse)
            for tau in [0.1, 0.03, 0.01]:
                if time.time() > t0 + budget_s * 0.78:
                    break
                x_co, val = project_gradient(x_co, n_coarse, dx_c, tau, max_iter=1500,
                                              t_start=t0, t_limit=t0 + budget_s * 0.78 - t0)
                if val < best_val:
                    best_val = val
                    best_h = x_co.copy()
                    best_n = n_coarse

    # Phase 5: Multiple PG restarts with perturbations
    if time.time() - t0 < t0 + budget_s * 0.88:
        dx_best = 2.0 / best_n
        p5_end = t0 + budget_s * 0.88
        scales = [0.05, 0.1, 0.15, 0.03]
        for sc in scales:
            if time.time() > p5_end:
                break
            x_r = perturb(best_h, best_n, sc, rng)
            x_r, val = project_gradient(x_r, best_n, dx_best, 0.01, max_iter=2000,
                                         t_start=t0, t_limit=p5_end - t0)
            if val < best_val:
                best_val = val
                best_h = x_r.copy()

    # Phase 6: Coordinate descent polish
    remaining = budget_s - (time.time() - t0) - 3
    if remaining > 10:
        dx_best = 2.0 / best_n
        best_h, best_val = coord_polish(best_h, best_n, dx_best,
                                         max_sweeps=min(5, int(remaining / 3)))

    # Final repair and evaluation
    best_h = repair(best_h, best_n)
    dx_f = 2.0 / best_n
    final_val = compute_c5(best_h, dx_f)
    if final_val < best_val:
        best_val = final_val
    return (best_h, best_val, best_n)
