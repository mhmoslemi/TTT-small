import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0

def _cons_eq(x):
    return x.sum() - _TARGET_SUM

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    global _TARGET_SUM
    _TARGET_SUM = float(n) / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = _TARGET_SUM
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
            h = np.clip(h, 0.0, 1.0)
            d2 = target - h.sum()
            if d2 > 1e-10:
                idxs = np.where(h < 1.0 - 1e-10)[0]
                for idx in idxs:
                    adj = min(d2, 1.0 - h[idx])
                    h[idx] += adj
                    d2 -= adj
                    if abs(d2) < 1e-12:
                        break
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
            h = np.clip(h, 0.0, 1.0)
            d2 = target - h.sum()
            if d2 < -1e-10:
                idxs = np.where(h > 1e-10)[0]
                for idx in idxs:
                    adj = max(d2, -h[idx])
                    h[idx] += adj
                    d2 -= adj
                    if abs(d2) < 1e-12:
                        break
    return h

def slsqp_opt(h, n, dx, maxiter=500, ftol=1e-13):
    global _TARGET_SUM
    _TARGET_SUM = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq}]
    try:
        res = minimize(eval_c5, h, args=(dx,), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        c = eval_c5(ho, dx)
        if np.isfinite(c) and c < 2.0:
            return ho, c
    except:
        pass
    return h, eval_c5(h, dx)

def de_opt(h_init, n, dx, t_deadline, pop_size=45, F=0.55, CR=0.75):
    rng = np.random
    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    # Diverse initial population
    for i in range(1, pop_size):
        if i % 5 == 1:
            # Structured periodic patterns
            period = 3 + (i % 5)
            base = rng.uniform(0, 1, period)
            base = (base - base.mean()) + 0.5
            pat = np.tile(base, n // period + 1)[:n]
            pop[i] = project_feasible(pat, n)
        elif i % 5 == 2:
            # Binary-like
            h0 = (rng.random(n) > 0.5).astype(float)
            pop[i] = project_feasible(h0, n)
        elif i % 5 == 3:
            # Smooth periodic
            t = np.linspace(0, n, n)
            freq = rng.uniform(0.5, 3.0)
            pop[i] = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * freq * t / n), n)
        elif i % 5 == 4:
            # Perturbed best
            pop[i] = project_feasible(h_init + 0.2 * rng.randn(n), n)
        else:
            pop[i] = project_feasible(rng.uniform(0, 1, n), n)
    objs = np.array([eval_c5(p, dx) for p in pop])
    best_c5 = float(np.min(objs))
    best_h = pop[np.argmin(objs)].copy()
    no_improve = 0
    F_cur = F
    while time.time() < t_deadline:
        improved = False
        for i in range(pop_size):
            if time.time() > t_deadline:
                break
            for _tries in range(10):
                a, b, c = rng.randint(0, pop_size, 3)
                if len(set([a, b, c, i])) == 4:
                    break
            v = pop[a] + F_cur * (pop[b] - pop[c])
            j0 = rng.randint(n)
            mask = rng.random(n) < CR
            mask[j0] = True
            u = np.where(mask, v, pop[i])
            u = project_feasible(u, n)
            fu = eval_c5(u, dx)
            if fu <= objs[i]:
                pop[i] = u
                objs[i] = fu
                if fu < best_c5 - 1e-14:
                    best_c5 = fu
                    best_h = u.copy()
                    improved = True
        if not improved:
            no_improve += 1
            F_cur = max(0.3, F_cur * 0.98)
            if no_improve % 80 == 0:
                worst = np.argsort(objs)[-max(5, pop_size // 5):]
                for idx in worst:
                    pop[idx] = project_feasible(pop[idx] + 0.4 * rng.randn(n), n)
                    objs[idx] = eval_c5(pop[idx], dx)
                F_cur = F
            if no_improve > 600:
                break
        else:
            no_improve = 0
            F_cur = min(0.8, F_cur * 1.01)
    return best_h, best_c5

def basin_hop(h_init, n, dx, t_deadline, n_outer=50, step=0.15):
    best_h = h_init.copy()
    best_c5 = eval_c5(h_init, dx)
    rng = np.random
    for _ in range(n_outer):
        if time.time() > t_deadline:

        h0 = project_feasible(best_h + step * rng.randn(n), n)
        h, c = slsqp_opt(h0, n, dx, maxiter=200, ftol=1e-12)
        if c < best_c5:
            best_h, best_c5 = h, c
            step = max(0.05, step * 0.95)
        else:
            step = min(0.3, step * 1.1)
    return best_h, best_c5
