import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def all_corrs(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def proj(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(m) > 0:
            h[m] += diff / np.sum(m)
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_one(h, n, tau):
    """Subgradient of f_tau at h. tau is the lag index offset from 0 in 'full' mode."""
    # In 'full' mode, index k corresponds to lag = k - (n-1)
    # So tau here is the actual lag (positive or negative)
    dx = 2.0 / n
    g = np.zeros(n)
    if tau > 0:
        # f = sum_{i=0}^{n-1-tau} h[i]*(1-h[i+tau])
        g[:n-tau] += (1.0 - h[tau:n]) * dx
        g[tau:] -= h[:n-tau] * dx
    elif tau < 0:
        t = -tau
        # f = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])
        g[t:] += (1.0 - h[:n-t]) * dx
        g[:n-t] -= h[t:n] * dx
    else:
        g[:] = (1.0 - 2.0 * h) * dx
    return g

def subgrad_topk(h, n, corr_vals, k_top):
    """Average subgradient over top-k correlation lags."""
    idx_sorted = np.argsort(corr_vals)[::-1]
    top_idx = idx_sorted[:k_top]
    g = np.zeros(n)
    for idx in top_idx:
        tau = int(idx) - (n - 1)
        g += subgrad_one(h, n, tau)
    g /= k_top
    return g, top_idx

def sg_descent(h, n, t_end, rng, k_init=5, step_init=0.3):
    """Projected subgradient descent with momentum."""
    h = proj(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step_init
    mom = 0.7
    vel = np.zeros(n)
    no_improve = 0
    k = k_init
    
    it = 0
    while time.time() < t_end:
        corr_vals = all_corrs(h, n)
        c5 = corr_vals.max()
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive k_top
        if no_improve > 100:
            k = min(10, k + 1)
        elif no_improve < 20:
            k = max(3, k - 1)
        
        g, top_idx = subgrad_topk(h, n, corr_vals, k)
        
        # Nesterov-style momentum
        vel = mom * vel - step * g
        h_new = proj(h + vel, n)
        
        # Armijo-like check
        c5_new = compute_c5(h_new, n)
        if c5_new > c5 + 0.01:
            # Bad step, reduce
            step *= 0.5
            h_new = proj(h - step * g, n)
            c5_new = compute_c5(h_new, n)
        
        h = h_new
        
        # Periodic perturbation
        if no_improve > 80:
            h = proj(h + rng.normal(0, 0.03, n), n)
            no_improve = 0
            step = max(step, 0.1)
        
        # Anneal
        if no_improve > 200:
            step *= 0.9
            mom = min(0.9, mom + 0.01)
        
        it += 1
    
    return best_h, best_c5, it

def smooth_obj(h, n, beta):
    """Log-sum-exp surrogate for C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    init = globals().get('initial_h_values', None)
    if init is not None:
        n0 = len(init)
        h0 = np.array(init, dtype=float).copy()
    else:

    h0 = proj(h0, n0)

    # Phase 1: Subgradient descent on n0

    h1, c5_1, _ = sg_descent(h0, n0, t1, rng)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    
    # Phase 2: SLSQP polish on smooth surrogate
    if time.time() < t0 + 0.7 * budget_s:
        try:
            bounds = [(0.0, 1.0)] * n0
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n0/2.0}]
            for beta in [20, 40, 80]:
                r = minimize(lambda x: smooth_obj(x, n0, beta),
                            best_h, method='SLSQP', bounds=bounds,
                            constraints=cons,
                            options={'maxiter': 300, 'ftol': 1e-12})
                hc = proj(r.x, n0)
                cc = compute_c5(hc, n0)
                if cc < best_c5:
                    best_h, best_c5 = hc, cc
        except:
            pass
    
    # Phase 3: Try n=2*n0 (interpolated)
    n2 = 2 * n0
    if time.time() < t0 + 0.85 * budget_s:
        x_coarse = np.linspace(0, 2, n0, endpoint=False)
        x_fine = np.linspace(0, 2, n2, endpoint=False)
        h2_init = np.interp(x_fine, x_coarse, best_h, period=2.0)
        h2_init = proj(h2_init, n2)
        t3 = min(time.time() + 0.1 * budget_s, t0 + budget_s - 30)
        h2, c5_2, _ = sg_descent(h2_init, n2, t3, rng, k_init=5, step_init=0.2)
        if c5_2 < best_c5:
            best_h, best_c5, _ = h2, c5_2, n2
            n0 = n2  # update for subsequent phases
    
    # Phase 4: Final SLSQP on exact max (piecewise linear)
    if time.time() < t0 + budget_s - 10:
        try:
            bounds = [(0.0, 1.0)] * len(best_h)
            nn = len(best_h)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - nn/2.0}]
            r = minimize(lambda x: compute_c5(x, nn),
                        best_h, method='SLSQP', bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 500, 'ftol': 1e-14})
            hc = proj(r.x, nn)
            cc = compute_c5(hc, nn)
            if cc < best_c5:
                best_h, best_c5 = hc, cc
        except:
            pass
    
    # Phase 5: Final SG polish
    if time.time() < t0 + budget_s - 5:
        h5, c5_5, _ = sg_descent(best_h, len(best_h), t0 + budget_s - 3, rng,
                                  k_init=3, step_init=0.05)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
    
    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))

def coord_sweep(h, n, t_end, sweep_pts=50):
    h = proj(h.copy(), n)
    best = compute_c5(h, n)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        rng2 = np.random.RandomState(int(time.time() * 1000) % 100000)
        order = rng2.permutation(n)
        for j in order:
            if time.time() > t_end:
                break
            best_val = h[j]
            for t in np.linspace(0.0, 1.0, sweep_pts):
                hp = h.copy()
                hp[j] = t
                diff = n / 2.0 - np.sum(hp)
                if abs(diff) > 1e-12:
                    other = (np.arange(n) != j)
                    mask = (hp[other] > 1e-8) & (hp[other] < 1.0 - 1e-8)
                    if np.sum(mask) > 0:
                        idx_other = np.where(other)[0]
                        idx_mask = idx_other[mask]
                        hp[idx_mask] += diff / len(idx_mask)
                        np.clip(hp, 0.0, 1.0, out=hp)
                c = compute_c5(hp, n)
                if c < best - 1e-14:
                    best = c
                    h = hp
                    best_val = t
                    improved = True
            if h[j] != best_val:
                pass
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    global _gn, _gb, best_h_local, best_c5_local
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        n0 = len(init)
        h0 = np.array(init, dtype=float).copy()
    else:
        n0 = 70
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * x * 3.0 / n0)

    h0 = proj(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_c5 = compute_c5(best_h, n0)

    # Phase 1: Subgradient descent
    t1 = t0 + 0.30 * budget_s
    h1, c5_1 = sg_descent(h0, n0, t1, rng, k_init=5, step_init=0.25)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: SLSQP with log-sum-exp (multiple betas)
    if time.time() < t0 + 0.60 * budget_s:
        nn = len(best_h)
        _gn = nn
        bounds = [(0.0, 1.0)] * nn
        cons = [{"type": "eq", "fun": _sum_wrap}]
        for beta in [20, 40, 80, 150]:
            if time.time() > t0 + 0.60 * budget_s:
                break
            _gb = beta
            try:
                r = minimize(_smooth_wrap, best_h, method="SLSQP",
                             bounds=bounds, constraints=cons,
                             options={"maxiter": 500, "ftol": 1e-14})
                hc = proj(r.x, nn)
                cc = compute_c5(hc, nn)
                if cc < best_c5:
                    best_h, best_c5 = hc, cc
            except Exception:
                pass

    # Phase 3: Coordinate sweep
    if time.time() < t0 + 0.78 * budget_s:
        nn = len(best_h)
        h3, c5_3 = coord_sweep(best_h, nn, t0 + 0.78 * budget_s, sweep_pts=40)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: More SG from current best
    if time.time() < t0 + 0.88 * budget_s:
        nn = len(best_h)
        h4, c5_4 = sg_descent(best_h, nn, t0 + 0.88 * budget_s, rng,
                              k_init=3, step_init=0.08)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Final SLSQP on exact max
    if time.time() < t0 + budget_s - 8:
        nn = len(best_h)
        _gn = nn
        try:
            bounds = [(0.0, 1.0)] * nn
            cons = [{"type": "eq", "fun": _sum_wrap}]
            r = minimize(_exact_wrap, best_h, method="SLSQP",
                         bounds=bounds, constraints=cons,
                         options={"maxiter": 600, "ftol": 1e-15})
            hc = proj(r.x, nn)
            cc = compute_c5(hc, nn)
            if cc < best_c5:
                best_h, best_c5 = hc, cc
        except Exception:
            pass

    # Phase 6: Final tiny SG polish
    if time.time() < t0 + budget_s - 3:
        nn = len(best_h)
        h6, c5_6 = sg_descent(best_h, nn, t0 + budget_s - 2, rng,
                              k_init=3, step_init=0.02)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return (best_h, best_c5, len(best_h))
