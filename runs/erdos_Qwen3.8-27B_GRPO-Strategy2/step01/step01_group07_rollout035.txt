def _grad_smooth(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -200.0/alpha, 0.0)
    w = np.exp(alpha * sh)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        s = k - (n-1)  # shift
        for j in range(n):
            # Term 1: (1-h[j+s]) * [0<=j+s<n]
            ip = j + s
            if 0 <= ip < n:
                grad[j] += w[k] * (1.0 - h[ip])
            # Term 2: -h[j] * [0<=j-s<n]
            ip2 = j - s
            if 0 <= ip2 < n:
                grad[j] -= w[k] * h[j]
    
    return grad * dx

from scipy.optimize import minimize
import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _smooth_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -200.0/alpha, 0.0)
    return m + np.log(np.sum(np.exp(alpha * sh))) / alpha

def _smooth_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -200.0/alpha, 0.0)
    w = np.exp(alpha * sh)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= ws
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        # For each j, the contribution is:
        # (1-h[j+s]) if 0<=j+s<n, minus h[j] if 0<=j-s<n
        lo1 = max(0, -s)      # j >= -s => j+s >= 0
        hi1 = min(n, n - s)   # j+s < n => j < n-s
        lo2 = max(0, s)       # j >= s => j-s >= 0  (wait, 0<=j-s<n => s<=j<n+s)
        hi2 = min(n, n + s)   # j-s < n => j < n+s
        
        for j in range(n):
            ip = j + s
            if 0 <= ip < n:
                grad[j] += wk * (1.0 - h[ip])
            ip2 = j - s
            if 0 <= ip2 < n:
                grad[j] -= wk * h[j]
    
    return grad * dx

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-14:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-13:
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        c = max(1, int(np.sum(m)))
        h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def _pgd(h, target, tlim=30.0):
    """Projected gradient descent with adaptive learning rate"""
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    best_c = _c5(h)
    best_h = h.copy()
    
    alphas = [50, 100, 200, 500, 1000]
    for al in alphas:
        if time.time() - t0 > tlim:
            break
        lr = 0.01
        for it in range(2000):
            if time.time() - t0 > tlim:
                break
            g = _smooth_grad(h, al, dx)
            h_new = h - lr * g
            h_new = _proj(h_new, target)
            c_new = _c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                lr = min(lr * 1.1, 0.1)
            else:
                lr *= 0.7
                if lr < 1e-6:
                    break
            h = h_new
    
    return best_h, best_c

def _local(h, target, tlim=15.0):
    n = len(h)
    bc = _c5(h)
    t0 = time.time()
    steps = [0.05, 0.02, 0.01, 0.005, 0.002]
    for st in steps:
        if time.time() - t0 > tlim:
            break
        imp = True
        while imp:
            if time.time() - t0 > tlim:
                break
            imp = False
            for i in range(n):
                if imp or time.time() - t0 > tlim:
                    break
                for ds in [st, -st]:
                    h2 = h.copy()
                    h2[i] += ds
                    if h2[i] < 0 or h2[i] > 1:
                        continue
                    # Compensate: try several j
                    for j in [(i+1)%n, (i-1+n)%n, (i+n//2)%n, (i+3)%n]:
                        h2[j] -= ds

                            bc = c
                            imp = True
                            break
                        h2[j] += ds
                    if imp:
                        break
    return h, bc

            if budget_s - (time.time() - t0) < 20:
                break
            
            # PGD
            h1, c1 = _pgd(h0, target, min(40, budget_s-(time.time()-t0)-10))
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
                best_n = n
            
            # SLSQP polish
            if budget_s - (time.time() - t0) > 15:
                for al in [200, 500, 1000]:
                    if budget_s - (time.time() - t0) < 5:
                        break
                    _G_ALPHA = al
                    try:
                        res = minimize(_slsqp_obj, h1, method='SLSQP',
                                      bounds=bounds, constraints=cons,
                                      options={'maxiter': 60, 'ftol': 1e-12})
                        h1 = _proj(res.x, target)
                    except:
                        pass
            
            c2 = _c5(h1)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h1.copy()
                best_n = n
            
            # Local search
            if budget_s - (time.time() - t0) > 10:
                h1, c3 = _local(h1, target, min(15, budget_s-(time.time()-t0)-5))
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h1.copy()
                    best_n = n
    
    # Basin hopping
    if best_h is not None and budget_s - (time.time() - t0) > 20:
        n = best_n
        target = n / 2.0
        for bh in range(20):
            if budget_s - (time.time() - t0) < 3:
                break
            noise = rng.normal(0, 0.03 + 0.01*bh, n)
            hp = _proj(best_h + noise, target)
            hp, c = _local(hp, target, min(8, budget_s-(time.time()-t0)-1))
            if c < best_c5:
                best_c5 = c
                best_h = hp.copy()
    
    h_ret = best_h.copy()
    target = best_n / 2.0
    d = target - np.sum(h_ret)
    if abs(d) > 1e-13:
        m = (h_ret > 1e-10) & (h_ret < 1.0 - 1e-10)
        c = max(1, int(np.sum(m)))
        h_ret[m] += d / c
    h_ret = np.clip(h_ret, 0.0, 1.0)
    c5f = _c5(h_ret)
    return h_ret, c5f, best_n
