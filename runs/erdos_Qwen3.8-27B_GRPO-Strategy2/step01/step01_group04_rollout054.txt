from scipy.interpolate import interp1d

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s < 1e-12:
        x = np.full(n, 0.5)
        return x
    # Iterative proportional fitting to satisfy both constraints
    for _ in range(100):
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
        s_new = np.sum(x)
        if abs(s_new - tgt) < 1e-12:
            break
        s = s_new
    return x

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    f = interp1d(xs, x_src, kind='linear', fill_value='extrapolate', bounds_error=False)
    return repair(f(xd), n_dst)

def obj_grad_lf(h, dx, tau, lam):
    """L-BFGS-B objective with penalty for sum constraint."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / tau, -50.0, 0.0)
    ew = np.exp(shifted)
    sw = ew.sum()
    w = ew / sw
    obj = (m + tau * np.log(sw)) * dx
    
    # Penalty for sum constraint
    s = np.sum(h) - n / 2.0
    obj += lam * s * s
    
    # Gradient of smooth max part
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        L = j - (n - 1)
        # First part: (1-h[k-L]) for k in [max(0,L), min(n-1, n-1+L)]
        klo = max(0, L)
        khi = min(n - 1, n - 1 + L)
        grad[klo:khi+1] += wj * (1.0 - h[klo-L:khi+1-L])
        # Second part: -h[k+L] for k+L in [max(0,L), min(n-1, n-1+L)]
        if L >= 0:
            klo2, khi2 = 0, n - 1 - L
        else:
            klo2, khi2 = -L, n - 1
        grad[klo2:khi2+1] -= wj * h[klo2+L:khi2+1+L]
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * lam * s * np.ones(n)
    
    return obj, grad

def slsqp_optimize(x0, n, dx, tau, maxiter=300):
    """SLSQP optimization with smooth max objective."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = np.clip((corr - m) / tau, -50.0, 0.0)
        return (m + tau * np.log(np.sum(np.exp(shifted)))) * dx
    
    def con(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
        return repair(res.x, n)
    except:
        return x0.copy()

def lfgbss_optimize(x0, n, dx, tau, maxiter=500, lam=100.0):
    """L-BFGS-B optimization with penalty for sum constraint."""
    def obj(x):
        return obj_grad_lf(x, dx, tau, lam)[0]

        return obj_grad_lf(x, dx, tau, lam)[1]
