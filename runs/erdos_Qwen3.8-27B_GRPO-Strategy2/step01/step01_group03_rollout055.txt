    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr * dx
    max_s = scaled.max()
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / beta * dx

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr * dx
    max_s = scaled.max()
    p = np.exp(scaled - max_s)
    p /= p.sum()
    
    # p has length 2n-1, p[m] corresponds to shift a_m = m-(n-1)
    a = np.arange(-(n-1), n)  # length 2n-1
    
    grad = np.zeros(n)
    for j in range(n):
        # (1-h[j-a_m]) for valid j-a_m

        valid1 = (idx1 >= 0) & (idx1 < n)
        # h[j+a_m] for valid j+a_m  
        idx2 = j + a
        valid2 = (idx2 >= 0) & (idx2 < n)
        
        grad[j] = dx * np.sum(p[valid1] * (1 - h[idx1[valid1]])) - dx * np.sum(p[valid2] * h[idx2[valid2]])
    
    return grad

import numpy as np
from scipy.optimize import minimize, lbfgsb
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr * dx
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    val = (max_s + np.log(exp_s.sum())) / beta * dx
    
    # Gradient
    a = np.arange(-(n-1), n, dtype=float)
    grad = np.zeros(n)
    # Vectorized: for each j, grad[j] = dx * (sum_m p[m]*(1-h[j-a[m]]) - sum_m p[m]*h[j+a[m]])
    # where out-of-bounds are 0
    for j in range(n):
        i1 = j - a.astype(int)
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + a.astype(int)
        v2 = (i2 >= 0) & (i2 < n)
        grad[j] = dx * (np.sum(p[v1] * (1.0 - h[i1[v1]])) - np.sum(p[v2] * h[i2[v2]]))
    
    return val, grad

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    step = diff / len(idx)
    h_copy = h.copy()
    h_copy[idx] += step
    h_copy = np.clip(h_copy, 0.0, 1.0)
    # Fix residual
    residual = target_sum - h_copy.sum()
    if abs(residual) > 1e-10:
        h_copy[idx[0]] = np.clip(h_copy[idx[0]] + residual, 0, 1)
    return h_copy

def proj_grad_descent(h0, n, target_sum, beta_schedule, max_iter=5000, lr_init=0.01, momentum=0.9):
    """Nesterov accelerated projected gradient descent"""
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iter):
        beta = beta_schedule(it)
        val, grad = soft_max_val_grad(h, beta)
        
        # Nesterov step
        h_new = h + momentum * v
        _, grad_new = soft_max_val_grad(project(h_new, target_sum), beta)
        
        v = momentum * v - lr * grad_new
        h = project(h + v, target_sum)
        
        # Adaptive lr
        val_true = compute_c5(h)
        if val_true < best_val:
            best_val = val_true
            best_h = h.copy()
        
        # Decay lr
        lr *= 0.9995
    
    return best_h, best_val

The pair move is interesting: change h[i] by +δ and h[j] by -δ (maintaining the sum constraint). This explores the feasible set more effectively than single-coordinate moves.

def local_search(h, n, target_sum, max_time=60):
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 0.01
    
    while time.time() - t0 < max_time:
        improved = False
        # Single coordinate moves
        for i in range(n):
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                h_new = h.copy()
                h_new[i] += delta
                h_new = project(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        
        # Pair moves
        for _ in range(n):  # random pairs
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            for delta in [0.02, -0.02, 0.05, -0.05]:
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                h_new = project(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        
        if not improved:
            # Decay learning rate
            lr *= 0.5
            if lr < 0.001:
                break
    
    return h, best_c5
