def smooth_obj_and_grad(h, n, beta):
    """Compute smoothed max objective and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = vals.max()
    # shifted for stability
    shifted = (vals - m) * beta
    exps = np.exp(shifted)
    Z = exps.sum()
    vals_s = exps / Z  # softmax weights
    obj = m + np.log(Z) / beta
    
    # Gradient: dx * sum_p w_p * d(corr_p)/d(h)
    # d(corr_p)/dh[j] where corr_p = sum_i h[i]*(1-h[i+shift_p])
    # = (1-h[j+shift_p])*I(valid) - h[j-shift_p]*I(valid)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = vals_s[p]
        if w < 1e-14:
            continue
        s = p - (n - 1)
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:n])
            grad[s:n] -= w * h[:n-s]
        else:
            sa = -s
            grad[sa:n] += w * (1.0 - h[0:n-sa])
            grad[0:n-sa] -= w * h[sa:n]
    grad *= dx
    return obj, grad

def proj_grad_solve(h, n, beta_schedule, steps_per_beta=500, lr=None):
    """Projected gradient descent with momentum on smoothed objective."""
    ts = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    
    for beta in beta_schedule:
        if lr is None:
            step = 2.0 / (beta * n)
        else:
            step = lr
        mu = 0.9  # momentum
        for it in range(steps_per_beta):
            obj, grad = smooth_obj_and_grad(h, n, beta)
            v = mu * v - step * grad
            h_new = h + v
            h_new = project_feasible(h_new, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        # Reset velocity between betas
        v = np.zeros(n)
    
    return h, best_h, best_c5

def local_search_swaps(h, n, max_sweeps=200, time_limit=60):

                    improved = True
                # else: leave as is (val) since we're trying to find any improvement
            # Try swap: set h[i]=1, find j with h[j]=0, swap
            # Actually, let's try pairwise swaps separately

                if time.time() - t0 > time_limit:

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-9)[0]
    else:
        idx = np.where(h > 1e-9)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    rem = target_sum - h.sum()
    if abs(rem) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + rem, 0, 1)
    return h

def grad_smooth_max(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = vals.max()
    shifted = (vals - m) * beta
    exps = np.exp(shifted)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-13:
            continue
        s = p - (n - 1)
        if s >= 0:
            grad[:n - s] += wp * (1.0 - h[s:])
            grad[s:] -= wp * h[:n - s]
        else:
            sa = -s
            grad[sa:] += wp * (1.0 - h[:n - sa])
            grad[:n - sa] -= wp * h[sa:]
    return grad * dx

def slsqp_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = corr * dx
    m = vals.max()
    return m + np.log(np.sum(np.exp((vals - m) * beta))) / beta

def slsqp_obj_grad(x, n, beta):
    return grad_smooth_max(x, n, beta)

    return x.sum() - ts

    return np.ones(len(x))

                   jac=partial(slsqp_obj_grad, n=n, beta=beta),
                   method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=con, options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)
