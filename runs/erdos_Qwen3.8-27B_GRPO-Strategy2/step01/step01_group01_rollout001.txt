import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def lse_obj(h, n, dx, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    return float((cmax + np.log(np.sum(np.exp((c - cmax) * temp))) / temp) * dx)

                improved = True

def slsqp_optimize(h0, n, temp, maxiter, ftol):
    dx = 2.0 / n
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(lse_obj, h0.copy(), args=(n, dx, temp),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
