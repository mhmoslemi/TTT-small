    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    maxC = C.max()
    return float((np.log(np.sum(np.exp(beta * (C - maxC)))) + beta * maxC) / beta)

def smooth_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    maxC = C.max()
    exps = np.exp(beta * (C - maxC))
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + lag:hi2 + lag]
    grad *= dx
    return grad

    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    ceq = lambda x: [x.sum() - target]

    cons = [{'type': 'eq', 'fun': ceq, 'jac': jac_eq}]
    try:
        res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, dx, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)
