```python
    #   h[i] as (1-h[n]): n=i, term was h[i+τ]*(1-1)=0 → h[i+τ]*(1-0)=h[i+τ]. Δ: +h[i+τ], valid if 0≤i+τ<n
    # Change from h[j] 0→1:
    #   h[j] as h[n+τ]: n+τ=j → n=j-τ, valid if 0≤j-τ<n. Term: 0*(1-h[j-τ])=0 → 1*(1-h[j-τ]). Δ: +(1-h[j-τ])
    #   h[j] as (1-h[n]): n=j, term was h[j+τ]*(1-0)=h[j+τ] → h[j+τ]*(1-1)=0. Δ: -h[j+τ], valid if 0≤j+τ<n
    
    for tau in range(-(n-1), n):
        d = 0.0
        # h[i] 1→0, as h[n+τ]
        ni = i - tau
        if 0 <= ni < n:
            d -= (1.0 - h[ni])
        # h[i] 1→0, as (1-h[n]) at n=i
        nj = i + tau
        if 0 <= nj < n:
            d += h[nj]
        # h[j] 0→1, as h[n+τ]
        nj2 = j - tau
        if 0 <= nj2 < n:
            d += (1.0 - h[nj2])
        # h[j] 0→1, as (1-h[n]) at n=j
        nj3 = j + tau
        if 0 <= nj3 < n:
            d -= h[nj3]
        delta[tau + (n-1)] = d
    return delta
```
