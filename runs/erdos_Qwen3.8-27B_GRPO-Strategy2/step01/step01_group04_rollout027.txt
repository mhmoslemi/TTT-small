    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    corr = np.correlate(x, 1.0 - x, mode="full")
    K = len(corr)
    m = corr.max()
    sh = np.clip((corr - m) / tau, -50.0, 0.0)
    exps = np.exp(sh)
    total = exps.sum()
    return m + tau * np.log(total), exps / total

    val, _ = lse_corr_value(x, tau)
    return val * dx

    if n_src == n_dst:
        return h_src.copy()
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, h_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)


def multi_tau(x0, n, taus, t0, time_limit, maxiter=300):
    x = x0.copy()
    best_val = compute_c5(x)
    best_x = x.copy()
    for tau in taus:
        elapsed = time.time() - t0
        if elapsed > time_limit - 5:
            break
        mi = max(50, min(maxiter, int((time_limit - (time.time() - t0)) * 1.5)))
        x = slsqp_opt(x, n, tau, mi)
        val = compute_c5(x)
        if val < best_val:
            best_val = val
            best_x = x.copy()
    return best_x, best_val


def make_alternate(n, rng):
    i = np.arange(n)
    base = 0.5 + 0.25 * np.sin(np.pi * i / n * 2)
    base = base + 0.1 * np.sin(4 * np.pi * i / n)
    base = base + 0.05 * rng.standard_normal(n)
    return repair(base, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    h0 = repair(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0)

    # Phase 1: Coarse n=64 for fast structural optimization
    n_c = 64
    h_c = project_grid(h0, n0, n_c)
    t_lim = min(50, budget_s * 0.06)
    h_c, _ = multi_tau(h_c, n_c, [0.4, 0.2, 0.1, 0.05, 0.02], t_start, t_lim, 200)

    # Phase 2: Upsample to n0, optimize
    h_m = project_grid(h_c, n_c, n0)
    t_lim = min(140, budget_s * 0.18)
    h_m, val = multi_tau(h_m, n0, [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002], t_start, t_lim + 50, 300)
    if val < best_val:
        best_val = val
        best_h = h_m
        best_n = n0

    # Phase 2b: From original h0 with different taus
    rem = budget_s - (time.time() - t_start)
    if rem > 80:
        h_m2, val2 = multi_tau(h0.copy(), n0, [0.1, 0.05, 0.02, 0.01, 0.005], t_start, time.time() - t_start + min(80, rem - 10), 250)
        if val2 < best_val:
            best_val = val2
            best_h = h_m2
            best_n = n0

    # Phase 2c: Alternate structured start
    rem = budget_s - (time.time() - t_start)
    if rem > 60:
        h_alt = make_alternate(n0, rng)
        h_alt, val3 = multi_tau(h_alt, n0, [0.2, 0.1, 0.05, 0.02, 0.01], t_start, time.time() - t_start + min(60, rem - 10), 200)
        if val3 < best_val:
            best_val = val3
            best_h = h_alt
            best_n = n0

    # Phase 3: Fine grid n=383
    rem = budget_s - (time.time() - t_start)
    n_f = 2 * n0 + 1  # 383
    if rem > 100:
        h_f = project_grid(best_h, best_n, n_f)
        t_alloc = min(280, rem - 40)
        h_f, val = multi_tau(h_f, n_f, [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001], t_start, time.time() - t_start + t_alloc, 250)
        if val < best_val:
            best_val = val
            best_h = h_f
            best_n = n_f

    # Phase 4: Very fine grid n=512 if time allows
    rem = budget_s - (time.time() - t_start)
    if rem > 120 and best_n < 512:
        n_vf = min(512, 2 * best_n + 1)
        if n_vf > best_n:
            h_vf = project_grid(best_h, best_n, n_vf)
            t_alloc = min(300, rem - 20)
            h_vf, val = multi_tau(h_vf, n_vf, [0.05, 0.02, 0.01, 0.005, 0.002, 0.001], t_start, time.time() - t_start + t_alloc, 200)
            if val < best_val:
                best_val = val
                best_h = h_vf
                best_n = n_vf

    # Phase 5: Final polish at best grid with very small tau
    rem = budget_s - (time.time() - t_start) - 5
    if rem > 20:
        t_alloc = min(rem, 120)
        h_pol, val = multi_tau(best_h.copy(), best_n, [0.005, 0.002, 0.001, 0.0005], t_start, time.time() - t_start + t_alloc, 400)
        if val < best_val:
            best_val = val
            best_h = h_pol

    # Final repair and evaluation
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h)
    return (best_h, final_val, best_n)
