        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _subgrad_step(h, target, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    mask = corr > m - 0.005
    active = np.where(mask)[0]
    g = np.zeros(n)
    for ma in active:
        d = ma - (n - 1)
        idx1 = np.arange(n)
        valid1 = (idx1 + d >= 0) & (idx1 + d < n)
        valid2 = (idx1 - d >= 0) & (idx1 - d < n)
        g[valid1] += (1.0 - h[np.clip(idx1[valid1] + d, 0, n - 1)])
        g[valid2] -= h[np.clip(idx1[valid2] - d, 0, n - 1)]
    g /= max(1, len(active))
    h -= lr * g
    return _project(h, target)

def _coord_pair(h, target, max_sweeps=30):
    n = len(h)
    bc = _c5(h)
    for _ in range(max_sweeps):
        moved = False
        rng_i = np.random.permutation(n)
        for i in rng_i:
            if h[i] < 0.05:
                continue
            for delta in (0.02, 0.05, 0.1, -0.02, -0.05, -0.1):
                h2 = h.copy()
                h2[i] += delta
                if h2[i] < 0 or h2[i] > 1:
                    continue
                j = (i + 1 + int(np.floor(np.random.random() * (n - 1)))) % n
                if j == i:
                    j = (i + 2) % n
                h2[j] -= delta
                if h2[j] < 0 or h2[j] > 1:
                    continue
                h2 = _project(h2, target)
                c = _c5(h2)
                if c < bc - 1e-11:
                    h = h2
                    bc = c
                    moved = True
                    break
        if not moved:
            break
    return h, bc

def _sim_anneal(h, target, t0, t1, n_iter, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    bc = _c5(h)
    h_best = h.copy()
    for it in range(n_iter):
        t = t0 * (t1 / t0) ** (it / max(1, n_iter))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.08, 0.08)
        h2 = h.copy()
        h2[i] += d
        h2[j] -= d
        if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
            continue
        h2 = _project(h2, target)
        c2 = _c5(h2)
        if c2 < bc or rng.random() < np.exp(-(c2 - bc) / max(t, 1e-8)):
            h = h2
            bc = c2
            if c2 < _c5(h_best) - 1e-12:
                h_best = h2.copy()
    return h_best, _c5(h_best)

def _slsqp_opt(h0, target, dx, alpha, maxiter=80, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, alpha, dx)
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = alpha * (corr - m)
        sh = np.clip(sh, -500.0, 0.5)
        w = np.exp(sh)
        w /= np.sum(w)
        sm = m + np.log(np.sum(w)) / alpha
        # gradient of smooth max w.r.t. corr
        dc = w / alpha * dx  # actually d(smooth)/d(corr[m]) = w[m] * dx... 
        # Let me just use numerical or skip gradient
        return None
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _project(res.x, target)
    except Exception:
        return _project(h0, target)

def _gen_start(n, target, rs, style):
    if style == 0:
        h = np.full(n, 0.5)
        for _ in range(rs.randint(2, 6)):
            f = rs.randint(2, max(3, n // 3))
            p = rs.uniform(0, 2 * np.pi)
            a = rs.uniform(0.15, 0.4)
            h += a * np.sin(2 * np.pi * f * np.arange(n) / n + p)
    elif style == 1:
        h = np.full(n, 0.5)
        for _ in range(rs.randint(3, 8)):
            f = rs.randint(1, max(2, n // 2))
            p = rs.uniform(0, 2 * np.pi)
            a = rs.uniform(0.1, 0.35)
            h += a * np.cos(2 * np.pi * f * np.arange(n) / n + p)
    elif style == 2:
        h = (rs.rand(n) > 0.5).astype(float)
    else:
        h = np.full(n, 0.5)
        h[::2] = 0.3
        h[1::2] = 0.7
    return _project(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    best_c5 = _c5(parent)
    best_h = parent.copy()
    best_n = len(parent)

    n_list = sorted(set([48, 56, 64, 70, 80, 90, 96, 112, 120]))

    for n in n_list:
        if elapsed() > budget_s - 80:
            break
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n

        n_restarts = 4 if n <= 80 else 3
        for r in range(n_restarts):
            if elapsed() > budget_s - 40:
                break
            rs = np.random.RandomState(seed * 1000 + n * 7 + r * 13)

            if r == 0 and n == best_n and 'initial_h_values' in globals():
                hc = parent.copy()
            else:
                style = r % 4
                hc = _gen_start(n, target, rs, style)

            # Phase 1: Smooth optimization with increasing alpha
            for alpha in (50, 200, 1000, 4000):
                if elapsed() > budget_s - 20:
                    break
                n_iter = 80 if n <= 70 else 50
                try:
                    hc = _slsqp_opt(hc, target, dx, alpha, maxiter=n_iter)
                except Exception:
                    pass
                cc = _c5(hc)
                if cc < best_c5 - 1e-10:
                    best_c5 = cc
                    best_h = hc.copy()
                    best_n = n

            # Phase 2: Subgradient polishing
            if elapsed() > budget_s - 30:
                break
            for phase in range(3):
                if elapsed() > budget_s - 20:
                    break
                lr = 0.005 / (phase + 1)
                for _ in range(300):
                    if elapsed() > budget_s - 15:
                        break
                    hc = _subgrad_step(hc, target, lr)
                cc = _c5(hc)
                if cc < best_c5 - 1e-10:
                    best_c5 = cc
                    best_h = hc.copy()
                    best_n = n

            # Phase 3: Pairwise coordinate search
            if elapsed() > budget_s - 25:
                break
            hc, cc = _coord_pair(hc, target, max_sweeps=25)
            if cc < best_c5 - 1e-10:
                best_c5 = cc
                best_h = hc.copy()
                best_n = n

            # Phase 4: Simulated annealing
            if elapsed() > budget_s - 20:
                break
            hc, cc = _sim_anneal(hc, target, 0.003, 0.00005, 2000,
                                 seed + n + r * 100)
            if cc < best_c5 - 1e-10:
                best_c5 = cc
                best_h = hc.copy()
                best_n = n

    # Final intensive polishing on the best solution
    if elapsed() < budget_s - 30:
        target = best_n / 2.0
        for _ in range(5):
            if elapsed() > budget_s - 10:
                break
            best_h, bc = _coord_pair(best_h, target, max_sweeps=40)
            if bc < best_c5 - 1e-10:
                best_c5 = bc
        # Subgradient micro-steps
        if elapsed() < budget_s - 15:
            for lr in (0.003, 0.001, 0.0005, 0.0002):
                if elapsed() > budget_s - 10:
                    break
                for _ in range(200):
                    best_h = _subgrad_step(best_h, target, lr)
                bc = _c5(best_h)
                if bc < best_c5 - 1e-10:
                    best_c5 = bc

    # Ensure exact feasibility
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-13:
        m = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(m))
        if cnt > 0:
            best_h[m] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
        diff2 = target - np.sum(best_h)
        if abs(diff2) > 1e-14:
            best_h = best_h - diff2 / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    c5_final = _c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    print(f"Final: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, best_n
