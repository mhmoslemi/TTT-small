import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = int(mask.sum())
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    residual = ts - h.sum()
    if abs(residual) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if residual > 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + residual, 0, 1))
    return h

def smooth_obj(x, n, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) * beta
    return m + np.log(np.sum(np.exp(shifted))) / beta

def slsqp_opt(h, n, ts, beta, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        return smooth_obj(x, n, beta, dx)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x if res.x is not None and np.all(np.isfinite(res.x)) else h, ts)

def sa_opt(h, n, ts, rng, n_iters, T_hi, T_lo, time_lim):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    t0 = time.time()
    no_improve = 0
    for it in range(n_iters):
        if time.time() - t0 > time_lim:
            break
        if no_improve > 120000 and it > 50000:
            break
        frac = it / max(n_iters, 1)
        T = T_hi * (T_lo / T_hi) ** frac
        T = max(T, 1e-8)
        step = 0.25 * (T / T_hi) ** 0.4 + 0.005
        r = rng.random()
        if r < 0.40:
            j, k = rng.choice(n, 2, replace=False)
            d = (rng.random() - 0.5) * 2.0 * step
            nh = cur_h.copy()
            nh[j] += d
            nh[k] -= d
        elif r < 0.65:
            j = int(rng.randint(n))
            nh = cur_h.copy()
            nh[j] = np.clip(cur_h[j] + (rng.random() - 0.5) * 2.0 * step, 0, 1)
        elif r < 0.85:
            m = int(rng.randint(2, 5))
            nh = cur_h.copy()
            idx = rng.choice(n, m, replace=False)
            nh[idx] += rng.normal(0, step * 0.4, m)
        else:
            nh = cur_h + rng.normal(0, step * 0.2, n)
        nh = project_feasible(nh, ts)
        nc = compute_c5(nh)
        if nc <= cur_c5 or rng.random() < np.exp(-(nc - cur_c5) / T):
            cur_h = nh
            cur_c5 = nc
            if nc < best_c5 - 1e-15:
                best_c5 = nc
                best_h = nh.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def binary_local(h, n, ts, rng, time_lim):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(300):
        if time.time() - t0 > time_lim:
            break
        improved = False
        perm = rng.permutation(n)
        for j in perm:
            if time.time() - t0 > time_lim:
                break
            for tv in (0.0, 1.0):
                if abs(h[j] - tv) < 1e-8:
                    continue
                saved = h.copy()
                h[j] = tv
                h = project_feasible(h, ts)
                ct = compute_c5(h)
                if ct < best_c5 - 1e-14:
                    best_c5 = ct
                    best_h = h.copy()
                    improved = True
                else:
                    h[:] = saved
        if not improved:
            break
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    xo = np.linspace(0, 2.0, n_src)
    xn = np.linspace(0, 2.0, n_dst)
    return np.interp(xn, xo, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n_init = 140
        h_init = np.zeros(n_init)
        h_init[rng.choice(n_init, n_init // 2, replace=False)] = 1.0

    n_init = len(h_init)
    ts_init = n_init / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP on parent grid
    if remaining() > 60:
        cur = best_h.copy()
        for beta in [5.0, 20.0, 80.0, 300.0, 800.0]:
            if remaining() < 15:
                break
            cur = slsqp_opt(cur, n_init, ts_init, beta, maxiter=350)
            c5v = compute_c5(cur)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = cur.copy()

    # Also try symmetric complement
    h_comp = 1.0 - best_h
    h_comp = project_feasible(h_comp, ts_init)
    c5_comp = compute_c5(h_comp)
    if c5_comp < best_c5:
        best_c5 = c5_comp
        best_h = h_comp

    # Phase 2: SA (primary improvement engine)
    if remaining() > 120:
        h_sa, c5_sa = sa_opt(best_h.copy(), n_init, ts_init, rng,
                             n_iters=800000, T_hi=0.06, T_lo=5e-6,
                             time_lim=min(250, remaining() - 40))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 3: Binary local search
    if remaining() > 30:
        h_bl, c5_bl = binary_local(best_h.copy(), n_init, ts_init, rng,
                                   time_lim=min(80, remaining() - 10))
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl

    # Phase 4: SLSQP polish
    if remaining() > 25:
        h_pol = slsqp_opt(best_h.copy(), n_init, ts_init, 1000, maxiter=600)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 5: Try alternate grid sizes
    n_candidates = [100, 120, 160, 200]
    for n_try in n_candidates:
        if remaining() < 70:
            break
        ts_try = n_try / 2.0
        h_try = interpolate_h(best_h, n_init, n_try)
        h_try = project_feasible(h_try, ts_try)
        if remaining() > 25:
            h_try = slsqp_opt(h_try, n_try, ts_try, 100, maxiter=200)
        if remaining() > 25:
            h_try, c5_try = sa_opt(h_try, n_try, ts_try, rng,
                                   n_iters=300000, T_hi=0.04, T_lo=5e-6,
                                   time_lim=min(35, remaining() - 15))
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 6: Second round of SA with perturbations on best
    if remaining() > 60:
        for restart in range(3):
            if remaining() < 25:
                break
            n_b = len(best_h)
            ts_b = n_b / 2.0
            h_pert = best_h + rng.normal(0, 0.08, n_b)
            h_pert = project_feasible(h_pert, ts_b)
            h_r, c5_r = sa_opt(h_pert, n_b, ts_b, rng,
                               n_iters=400000, T_hi=0.04, T_lo=3e-6,
                               time_lim=min(25, remaining() - 10))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
            h_sym = 1.0 - h_r
            h_sym = project_feasible(h_sym, ts_b)
            c5_sym = compute_c5(h_sym)
            if c5_sym < best_c5:
                best_c5 = c5_sym
                best_h = h_sym

    # Final projection and verification
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    # One last SLSQP if time permits
    if remaining() > 15:
        h_final = slsqp_opt(best_h, len(best_h), len(best_h) / 2.0, 2000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
