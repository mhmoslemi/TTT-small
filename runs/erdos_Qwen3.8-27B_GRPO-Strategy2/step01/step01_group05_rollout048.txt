    return _corr(h, dx).max()

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    h = h.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    ts = np.linspace(0.0, 1.0, grid_pts)
    for sweep in range(n_sweeps):
        if time.time() > t_end - 3:
            break
        order = rng.permutation(n)
        for i in order:
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full") * dx

            best_h = h.copy()
    return best_h, best_c5

def _slsqp_polish(h0, dx, n, target, eps, t_end, maxiter=200):
    # Similar to previous but streamlined
    N = n
    DX = dx
    EPS = float(eps)
    
    def obj(x):
        cv = np.correlate(x, 1.0-x, mode="full") * DX
        mx = cv.max()
        return mx + EPS * np.log(np.sum(np.exp((cv - mx) / EPS)))
    
    def grad(x):
        cv = np.correlate(x, 1.0-x, mode="full") * DX
        mx = cv.max()
        w = np.exp((cv - mx) / EPS)
        w /= w.sum()
        g = np.zeros(n)
        j = np.arange(n)
        for idx in range(2*n-1):
            if w[idx] < 1e-14:
                continue
            wt = w[idx] * DX
            s = idx - (n-1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g[vm] += wt * (1.0 - x[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= wt * x[jp[vp]]
        return g
    
    res = minimize(obj, h0, jac=grad,
                   bounds=[(0.0, 1.0)]*n,
                   constraints=[{'type':'eq','fun':lambda x: x.sum()-target}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
    return _proj(res.x, target)

        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    # Spread penalty gradient
    if mu > 0:
        mean_cv = cv.mean()
        g2 = np.zeros(n)
        for idx in range(2 * n - 1):
            d = cv[idx] - mean_cv
            if abs(d) < 1e-14:
                continue
            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g2[vm] += 2.0 * d * dx * (1.0 - x[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g2[vp] -= 2.0 * d * dx * x[jp[vp]]
        g += g2 / len(cv)
    return g

def _slsqp_polish(h0, dx, n, target, eps, mu, t_end, maxiter=200):
    def obj(x):
        return _slsqp_obj(x, dx, n, eps, mu)
    def grad(x):
        return _slsqp_grad(x, dx, n, eps, mu)
    def cons(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    res = minimize(obj, h0, jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cons, 'jac': cons_jac}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
    return _proj(res.x, target)

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
