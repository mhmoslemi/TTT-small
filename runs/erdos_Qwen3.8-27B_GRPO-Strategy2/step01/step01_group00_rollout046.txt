        if lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        elif lag < 0:
            l = -lag
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
        else:
            grad[:] += (1.0 - 2.0 * h)

    c = _corr(x)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    return np.sum(x) - _OPT['target']

    try:
        cons = [{'type': 'eq', 'fun': _eq_con}]
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hc = _repair(res.x, n)
        return hc
    except Exception:
        return h

    t0 = time.time()
    h = h.copy()
    best = _c5(h, dx)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break
        eta = 0.3 / np.sqrt(it + 20)
        g, _ = _subgrad(h, n, frac=0.04 + 0.02 * np.sin(it * 0.01))
        h_new = _repair(h - eta * g, n)
        c5 = _c5(h_new, dx)
        if c5 < best:
            best = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best

def _coord_descent(h, n, dx, n_steps, budget, rng):
    t0 = time.time()
    h = h.copy()
    best = _c5(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05])
    for _ in range(n_steps):
        if time.time() - t0 > budget:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in deltas:
            ni, nj = hi + d, hj - d
            if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                continue
            h[i] = ni
            h[j] = nj
            c5 = _c5(h, dx)
            if c5 < best - 1e-10:
                best = c5
            else:
                h[i] = hi
                h[j] = hj
                break
    return h, best
