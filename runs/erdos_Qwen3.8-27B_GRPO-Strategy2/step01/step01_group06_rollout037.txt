def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    h = h / h.sum() * target if h.sum() > 1e-14 else np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)

                                       min(30, budget_s - (time.time() - t0) - 50), rng)
        if c_bt < best_c5:
            best_h, best_c5 = h_bt, c_bt

    # Phase 4: Final polish with high beta
    if time.time() - t0 < budget_s - 10:
        n_f = len(best_h)
        rem_f = budget_s - (time.time() - t0)
        h_f, c_f = projected_smooth_opt(best_h, n_f, min(rem_f * 0.7, 80), [200, 400, 800])
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    # Phase 5: Final binary hill climb
    if time.time() - t0 < budget_s - 3:
        n_f2 = len(best_h)
        h_b2, c_b2 = binary_hill_climb(best_h, n_f2, 2.0 / n_f2,
                                       min(budget_s - (time.time() - t0) - 2, 40), rng)
        if c_b2 < best_c5:
            best_h, best_c5 = h_b2, c_b2

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = eval_c5(best_h, 2.0 / n_final)
    return (best_h, best_c5, n_final)
