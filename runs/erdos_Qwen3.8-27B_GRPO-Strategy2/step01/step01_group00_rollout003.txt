Let me think about what "top-lag focused optimization" looks like:

Given the current h, compute all correlations. Let the top K lags be k_1, ..., k_K.
Define: g(h) = sum_{k=1}^{K} corr[k_k]^2 + 0.1 * smooth_max(all corr, alpha=500)

Minimize g(h) with SLSQP. This is smooth (C^2) and the gradient points in the direction of reducing the worst correlations.

The gradient of corr[k] with respect to h[i]:
∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k] (for valid indices, k≠0)
∂corr[0]/∂h[i] = 1-2h[i]

So the gradient of g is:
∂g/∂h[i] = 2*sum_k corr[k_k]*(1-h[i+k_k]) - 2*sum_k corr[k_k]*h[i-k_k] + ... (similar for k=0)

This is easy to compute! And it gives a smooth objective for SLSQP.

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h


def _logsumexp_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _top_lags_obj(x, K, alpha_smooth):
    c = _corr(x)
    top = np.partition(c, -K)[-K:]
    obj = np.sum(top ** 2)
    m = np.max(c)
    s = np.clip(alpha_smooth * (c - m), -400, 400)
    obj += 0.05 * (m + np.log(np.sum(np.exp(s))) / alpha_smooth)
    return obj


def _exact_c5_search(h, n, dx, pairs, delta_range, n_trials):
    """Try small shifts between pairs; return best h and c5."""
    best_h = h.copy()
    best_c5 = _c5_val(h, dx)
    for _ in range(n_trials):
        i, j = pairs[rng.randint(len(pairs))]
        if i == j:
            continue
        hi, hj = h[i], h[j]
        d_lo = max(0.0, -hi)
        d_hi = min(1.0 - hi, hj)
        if d_hi <= d_lo:
            continue
        deltas = np.linspace(d_lo + 1e-6, d_hi - 1e-6, 8)
        for d in deltas:
            if abs(d) < 1e-9:
                continue
            h[i], h[j] = hi + d, hj - d
            c5 = _c5_val(h, dx)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[i], h[j] = hi, hj
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global rng
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons_sum = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5_val(best_h, dx)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: SLSQP with log-sum-exp, alpha ramp
    alphas_p1 = [30, 60, 120, 250, 500, 1000]
    for a in alphas_p1:
        if remaining() < 90:
            break
        try:
            res = minimize(_logsumexp_obj, best_h.copy(), args=(a,),
                          method='SLSQP', bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 300, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass

    # Phase 2: Top-lags focused SLSQP
    K_schedule = [3, 5, 8, 12, 15]
    for K in K_schedule:
        if remaining() < 80:
            break
        for alpha_sm in [100, 300]:
            if remaining() < 60:
                break
            try:
                res = minimize(_top_lags_obj, best_h.copy(), args=(K, alpha_sm),
                              method='SLSQP', bounds=bounds, constraints=cons_sum,
                              options={'maxiter': 250, 'ftol': 1e-15})
                hc = _repair(res.x, n)
                c5 = _c5_val(hc, dx)
                if c5 < best_c5:
                    best_h, best_c5 = hc, c5
            except Exception:
                pass

    # Phase 3: Multi-restart with diverse perturbations
    perturb_sizes = [0.05, 0.1, 0.2, 0.3, 0.15, 0.08, 0.25, 0.4, 0.12, 0.35]
    restart_count = 0
    while remaining() > 120 and restart_count < 30:
        sigma = perturb_sizes[restart_count % len(perturb_sizes)]
        h_p = best_h + sigma * rng.randn(n)
        h_p = _repair(h_p, n)
        a = 200 + 100 * (restart_count % 5)
        try:
            res = minimize(_logsumexp_obj, h_p, args=(a,),
                          method='SLSQP', bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 200, 'ftol': 1e-14})
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
        restart_count += 1

    # Phase 4: Coordinate-pair exact search
    if remaining() > 60:
        idx = np.arange(n)
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        pairs = np.array(pairs)
        n_rounds = min(200, int(remaining() // 3))
        for _rd in range(n_rounds):
            if remaining() < 40:
                break
            best_h, best_c5 = _exact_c5_search(best_h, n, dx, pairs,
                                               delta_range=None, n_trials=60)

    # Phase 5: Final high-alpha polish
    if remaining() > 20:
        for a in [2000, 5000]:
            if remaining() < 15:
                break
            try:
                res = minimize(_logsumexp_obj, best_h.copy(), args=(a,),
                              method='SLSQP', bounds=bounds, constraints=cons_sum,
                              options={'maxiter': 400, 'ftol': 1e-16})
                hc = _repair(res.x, n)
                c5 = _c5_val(hc, dx)
                if c5 < best_c5:
                    best_h, best_c5 = hc, c5
            except Exception:
                pass

    # Phase 6: Ultra-fine pair search around the best
    if remaining() > 10:
        hi_idx = np.arange(n)
        hi_pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                hi_pairs.append((i, j))
        hi_pairs = np.array(hi_pairs)
        for _ in range(50):
            if remaining() < 5:
                break
            i, j = hi_pairs[rng.randint(len(hi_pairs))]
            ohi, ohj = best_h[i], best_h[j]
            d_lo = max(0.0, -ohi)
            d_hi = min(1.0 - ohi, ohj)
            if d_hi - d_lo < 1e-8:
                continue
            deltas = np.linspace(d_lo + 0.5e-7, d_hi - 0.5e-7, 20)
            for d in deltas:
                if abs(d) < 0.5e-8:
                    continue
                best_h[i], best_h[j] = ohi + d, ohj - d
                c5 = _c5_val(best_h, dx)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                else:
                    best_h[i], best_h[j] = ohi, ohj

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
