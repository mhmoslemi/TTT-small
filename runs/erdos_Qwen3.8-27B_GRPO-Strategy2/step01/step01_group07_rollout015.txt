        mi = 80 if n <= 70 else (60 if n <= 90 else 40)

        for s in range(n_starts):
            rem = budget_s - (time.time() - t_start)
            if rem < 12:
                break

            if s == 0 and n == pn:
                h = parent.copy()
            elif s == 1 and n == pn:
                h = _make_bimodal(n, rng)
            elif s == 2 and n == pn:
                h = _make_sine(n, rng, 3, 6.0)
            else:
                steep = rng.uniform(2.0, 6.0)
                nc = rng.randint(2, 6)
                h = _make_sine(n, rng, nc, steep)

            h = _slsqp_run(h, n, mi, alphas)
            c = _c5(h)
            if c < best_c5:
                best_c5 = c; best_h = h.copy(); best_n = n
                print(f"n={n} s={s} slsqp: C5={c:.6f}")

            rem = budget_s - (time.time() - t_start)
            if rem > 8:
                tl = min(12.0, rem - 4)
                h, c = _pair_search(h, n, tl)
                if c < best_c5:
                    best_c5 = c; best_h = h.copy(); best_n = n
                    print(f"n={n} s={s} pair: C5={c:.6f}")

            rem = budget_s - (time.time() - t_start)
            if rem > 6:
                tl = min(8.0, rem - 3)
                h, c = _single_search(h, n, tl)
                if c < best_c5:
                    best_c5 = c; best_h = h.copy(); best_n = n
                    print(f"n={n} s={s} single: C5={c:.6f}")

    # Final re-optimization of best solution
    rem = budget_s - (time.time() - t_start)
    if rem > 30:
        h = best_h.copy()
        h = _slsqp_run(h, best_n, 60, (100, 500, 2000, 5000, 10000))
        c = _c5(h)
        if c < best_c5:
            best_c5 = c; best_h = h.copy()
            print(f"refine: C5={c:.6f}")
        if rem - (time.time() - t_start) > 8:
            tl = min(10.0, rem - (time.time() - t_start) - 3)
            h, c = _pair_search(h, best_n, tl)
            if c < best_c5:
                best_c5 = c; best_h = h.copy()
                print(f"refine pair: C5={c:.6f}")

    # Snap attempt
    rem = budget_s - (time.time() - t_start)
    if rem > 5:
        hs = best_h.copy()
        hs[hs > 0.93] = 1.0
        hs[hs < 0.07] = 0.0
        hs = _norm_h(hs, best_n)
        cs = _c5(hs)
        if cs < best_c5:
            best_c5 = cs; best_h = hs.copy()
            print(f"snap: C5={cs:.6f}")

    best_h = _fix_sum(best_h, best_n)
    c5f = _c5(best_h)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return best_h, c5f, best_n
