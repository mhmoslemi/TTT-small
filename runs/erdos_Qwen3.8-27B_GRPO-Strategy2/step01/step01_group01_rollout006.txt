def project_to_constraint(x, target_sum):
    """Project x onto {z : sum(z) = target_sum, 0 <= z <= 1}"""
    # Binary search on the Lagrange multiplier
    lo, hi = -1.0, 1.0  # or wider
    # Actually, z[i] = clamp(x[i] - theta, 0, 1)
    # We need sum(z) = target_sum
    # sum(z) = sum(clamp(x[i] - theta, 0, 1)) is non-increasing in theta
    for _ in range(50):
        mid = (lo + hi) / 2
        z = np.clip(x - mid, 0, 1)
        s = z.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix exact sum
    diff = target_sum - z.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = z < 1.0
        else:
            mask = z > 0.0
        if mask.any():
            z[mask] += diff / mask.sum()
    return np.clip(z, 0, 1)

    c = np.correlate(h, 1-h, mode='full')
    # Softmax weights
    c_max = c.max()
    weights = np.exp(t * (c - c_max))
    weights /= weights.sum()

    grad = np.zeros(n)

            best_h = h.copy()

        if val_now < best:
            lr = min(lr0, lr * 1.5)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def project_sum(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:

    z = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

            mask = z < 1.0
        else:
            mask = z > 0.0

def weighted_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    nw = len(weights)

        w = weights[ki]
        if w < 1e-13:
            continue
        d = ki - (n - 1)
        if d >= 0:
            if n - d > 0:
                grad[:n - d] += w * dx * (1.0 - h[d:])
            if d < n:
                grad[d:] += w * dx * (-h[:n - d])
        else:
            ad = -d
            if n - ad > 0:
                grad[ad:] += w * dx * (1.0 - h[:n - ad])

    return grad

    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    sm = (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx
    sv = h.sum() - n * 0.5
    return sm + penalty * sv * sv


def lbfgs_gradient(h, t, penalty):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    w = np.exp(t * (c - cmax))
    w /= w.sum()
    grad = weighted_gradient(h, w, dx)
    sv = h.sum() - n * 0.5
    grad += 2.0 * penalty * sv
    return grad
