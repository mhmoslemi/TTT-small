import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gt = 1.0
_cn = 0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gt, -50.0, 0.0)
    return (m + _gt * np.log(np.sum(np.exp(sh)))) * _gdx

def _cfun(x):
    return np.sum(x) - _cn / 2.0

def _cjac(x):
    return np.ones(_cn)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    diff = tgt - s2
    if abs(diff) > 1e-10:
        mask = (x < 0.999999) if diff > 0 else (x > 1e-6)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_opt(x0, n, dx, taus, time_limit, maxiter=2000):
    global _gdx, _gt, _cn
    _cn = n
    _gdx = dx
    x = repair(x0, n).copy()

    best_val = compute_c5(x, dx)
    t0 = time.time()

        if time.time() - t0 > time_limit - 2.0:
            break
        _gt = tau
        try:
            r = minimize(_obj, x, method='SLSQP', bounds=bd,
                         constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
            x = repair(r.x, n)
            val = compute_c5(x, dx)
            if val < best_val:
                best_val = val
                best = x.copy()
        except Exception:
            pass
    return best, best_val

def subgrad_refine(h0, n, dx, n_iter, lr0):

    best = x.copy()

    lr = lr0

        corr = np.correlate(x, 1.0 - x, mode="full")

        grad = np.zeros(n)
        for k in topk:

            for i in range(n):
                j1 = i + sh
                if 0 <= j1 < n:
                    grad[i] += (1.0 - x[j1])
                j2 = i - sh
                if 0 <= j2 < n:
                    grad[i] -= x[j2]
        gn = np.linalg.norm(grad)
        if gn > 1e-14:
            x = x - lr * grad / gn
            x = repair(x, n)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
            lr = min(lr * 1.02, lr0)
        else:
            lr *= 0.9
            x = best.copy()
    return best, best_val

def proj_refine(h0, n, dx, budget):

def block_flip(h, n, rng, nblocks=3, blen=5):
    x = h.copy()
    for _ in range(nblocks):
        start = rng.integers(0, max(1, n - blen))
        end = min(start + blen, n)
        x[start:end] = 1.0 - x[start:end]
    return repair(x, n)

def sine_perturb(h, n, rng):
    x = h.copy()
    period = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    alpha = rng.uniform(0.05, 0.15)
    xs = np.linspace(0, 2, n, endpoint=False)
    x += alpha * np.sin(2 * np.pi * xs / period + phase)
    return repair(x, n)
