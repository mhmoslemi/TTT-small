_G_alpha = 1.0
_G_n = 0
_G_eq_target = 0.0

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            free = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
            if len(free) > 0:
                h[free] = np.clip(h[free] + d2 / len(free), 0, 1)
            else:
                idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
                h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def corr_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        m1 = np.arange(k, n)
        G[n - 1 + k, m1] += (1.0 - h[m1 - k])
        m2 = np.arange(0, n - k)
        G[n - 1 + k, m2] -= h[m2 + k]
        m1b = np.arange(0, n - k)
        G[n - 1 - k, m1b] += (1.0 - h[m1b + k])
        m2b = np.arange(k, n)
        G[n - 1 - k, m2b] -= h[m2b - k]
    return G

    """Projected gradient with smooth-max and backtracking."""
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    lr = 0.01
    stall = 0
    
    for it in range(maxiter):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        m_val = np.max(corr)
        s = alpha * (corr - m_val)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        G = corr_grad_matrix(h_curr, n)
        grad = dx * (G.T @ w)
        
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = eval_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
            lr = min(lr * 1.05, 0.1)
            stall = 0
        else:
            stall += 1
            lr *= 0.8
            if stall > 100:
                lr = 0.01
                stall = 0
            if lr < 1e-9:
                break
    
    return best_h, best_c5

def binary_swap_search(h, n, dx, max_iter=5000):

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            free = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
            if len(free) > 0:
                h[free] = np.clip(h[free] + d2 / len(free), 0, 1)
            else:
                idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
                h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def corr_grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        G[n - 1 + k, k:] += (1.0 - h[:n - k])
        G[n - 1 + k, :n - k] -= h[k:]
        G[n - 1 - k, :n - k] += (1.0 - h[k:])
        G[n - 1 - k, k:] -= h[:n - k]
    return G

def _s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    a = _G['alpha']
    return (m + np.log(np.sum(np.exp(a * (corr - m))))) / a * _G['dx']

def _s_grad(x):
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    a = _G['alpha']
    s = a * (corr - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    Gm = corr_grad_matrix(x, n)
    return _G['dx'] * (Gm.T @ w)

def _s_eq(x):
    return x.sum() - _G['target']

def _s_eqj(x):
    return np.ones(len(x))

def slsqp_smooth(h, n, dx, alpha, maxiter=400):
    _G['dx'] = dx
    _G['alpha'] = alpha
    _G['n'] = n
    _G['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_eq, 'jac': _s_eqj}]
    try:
        res = minimize(_s_obj, h, jac=_s_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def proj_gradient(h, n, dx, alpha=100, maxiter=30000):
    h_c = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    lr = 0.008
    stall = 0
    for it in range(maxiter):
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        mv = np.max(corr)
        s = alpha * (corr - mv)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        Gm = corr_grad_matrix(h_c, n)
        grad = dx * (Gm.T @ w)
        h_new = h_c - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h_c = h_new
            lr = min(lr * 1.03, 0.05)
            stall = 0
        else:
            stall += 1
            lr *= 0.85
            if stall > 80:
                lr = 0.01
                stall = 0
        if lr < 1e-10:
            break
    return best_h, best_c5

def binary_swap(h, n, dx, max_iter=3000, n_try=400):

                best_swap = (i, j)
        if best_swap is not None:

            hb[best_swap[1]] = 1.0

                hb[j1] = 1.0; hb[j2] = 1.0
                c = eval_c5(hb, dx)
                hb[i1] = 1.0; hb[i2] = 1.0
                hb[j1] = 0.0; hb[j2] = 0.0
