    # Phase 3: Local search
    if time.time() < budget_s - 120:
        bh, bc = _local(bh, n, dx, budget_s - 120)

    # Phase 4: SLSQP polish
    for a in [800, 2000]:
        if time.time() - t0 > budget_s - 50:
            break
        try:
            r = minimize(_smm_obj, bh, args=(a,), method='SLSQP',
                        bounds=bds, constraints=cns,
                        options={'maxiter': 500, 'ftol': 1e-15})
            hc = _fix(r.x, n)
            c = _c5(hc, dx)
            if c < bc:
                bc, bh = c, hc
        except Exception:
            pass

    # Phase 5: Fine local search
    if time.time() < budget_s - 20:
        bh, bc = _local(bh, n, dx, budget_s - 20)

    # Phase 6: Restarts
    while time.time() < budget_s - 25:
        hp = _fix(bh + 0.1 * rng.randn(n), n)
