        if nf > 0:

        else:
            h += diff / n

def smooth_obj(h, n, beta):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = beta * corr

    return m + np.log(np.sum(np.exp(scaled - m))) / beta
