    return np.correlate(h, 1.0 - h, mode="full") * dx

    return _corr_vec(h, dx).max()

            hi = mid

    mx = c.max()
    return mx + eps * np.log(np.sum(np.exp((c - mx) / eps)))

def _soft_max_weights(c, eps):
    mx = c.max()
    w = np.exp((c - mx) / eps)
    w /= w.sum()
    return w

    j = np.arange(n)

            g[vm] += wt * (1.0 - h[j[vm] - shift])
            g[vp] -= wt * h[j[vp] + shift]

        w = _soft_max_weights(c, eps)
        return _grad_c5(x, n, dx, w)

        if it % 500 == 250:

    return best_h, best_c

def _optimize(h0, n, dx, target, rng, t_end):

    h = h0.copy()

            best_h = h2.copy()

import numpy as np
from scipy.optimize import minimize
import time, math

def _corr_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5_val(h, dx):
    return _corr_vec(h, dx).max()

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_from_w(x, w, dx):
    n = len(x)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        shift = idx - (n - 1)
        if shift >= 0:
            vm = (j >= shift) & (j < n)
            vp = (j >= 0) & (j < n - shift)
            g[vm] += wt * (1.0 - x[j[vm] - shift])
            g[vp] -= wt * x[j[vp] + shift]
        else:
            t = -shift
            vm = (j >= 0) & (j < n - t)
            vp = (j >= t) & (j < n)
            g[vm] += wt * (1.0 - x[j[vm] + t])
            g[vp] -= wt * x[j[vp] - t]
    return g

def _lbfgs_phase(h0, n, dx, target, eps, pen, maxiter):
    def obj(x):
        c = _corr_vec(x, dx)
        mx = c.max()
        sm = mx + eps * np.log(np.sum(np.exp((c - mx) / eps)))
        s = x.sum() - target
        return sm + pen * s * s
    def jac(x):
        c = _corr_vec(x, dx)
        mx = c.max()
        w = np.exp((c - mx) / eps)
        w /= w.sum()
        g = _grad_from_w(x, w, dx)
        s = x.sum() - target
        g += 2.0 * pen * s
        return g
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return _proj(res.x, target)
    except Exception:
        return h0

def _slsqp_phase(h0, n, dx, target, eps, maxiter):
    def obj(x):
        c = _corr_vec(x, dx)
        return c.max() + eps * np.log(np.sum(np.exp((c - c.max()) / eps)))
    def jac(x):
        c = _corr_vec(x, dx)
        mx = c.max()
        w = np.exp((c - mx) / eps)
        w /= w.sum()
        return _grad_from_w(x, w, dx)
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _proj(res.x, target)
    except Exception:
        return h0

def _grad_descent(h0, n, dx, target, rng, t_end):
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_val(h, dx)
    lr = 0.05
    for it in range(30000):
        if time.time() > t_end:
            break
        eps = max(0.0002, 0.03 * (0.97 ** (it // 50)))
        c = _corr_vec(h, dx)
        mx = c.max()
        w = np.exp((c - mx) / eps)
        w /= w.sum()
        g = _grad_from_w(h, w, dx)
        h_new = _proj(h - lr * g, target)
        c_new = _c5_val(h_new, dx)
        c_cur = _c5_val(h, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= c_cur + 1e-10:
            h = h_new
            lr = min(lr * 1.005, 0.3)
        else:
            lr = max(lr * 0.97, 0.002)
        if it % 400 == 200:
            h = _proj(best_h + rng.normal(0, 0.02, n), target)
    return best_h, best_c

def _optimize_round(h0, n, dx, target, rng, t_end):
    best_h = h0.copy()
    best_c = _c5_val(h0, dx)
    h = h0.copy()
    # L-BFGS-B phases with decreasing eps
    for eps, pen in [(0.1, 5.0), (0.05, 10.0), (0.02, 20.0), (0.01, 50.0), (0.005, 100.0)]:

        if c3 < best_c:
            best_c = c3
            best_h = h3.copy()
    return best_h, best_c

    t0 = time.time()

    dx_f = 2.0 / len(best_h)
    final_c = _c5_val(best_h, dx_f)
    return best_h, final_c, len(best_h)
