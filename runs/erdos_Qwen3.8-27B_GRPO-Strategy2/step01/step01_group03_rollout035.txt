def correlated_local_search(h, timestep, n_iter=500):
    """Fast local search using incremental correlation updates."""
    n = len(h)
    ts = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    dx = 2.0 / n
    current_max = corr.max() * dx
    
    for it in range(n_iter):
        improved = False
        for j in range(n):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp((corr - m) * beta))) / beta

        return soft_max_obj(x, beta)

                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})

        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

import numpy as np
from scipy.optimize import minimize
import time

_BETA = 10.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) > 0:
            h[free[0]] = np.clip(h[free[0]] + diff2, 0, 1)
    return h

def _smo_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = corr.max()
    b = _BETA
    return m + np.log(np.sum(np.exp((corr - m) * b))) / b + 50.0 * (x.sum() - n / 2.0) ** 2

def _slsqp_solve(h, beta, maxiter=500):
    global _BETA
    _BETA = beta
    n = len(h)
    res = minimize(_smo_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n / 2.0)
    return project_feasible(h, n / 2.0)

def _dc_matrix(h, n):
    """Precompute Δc[j] for all positions j (flip from 0 to 1)."""
    lag = np.arange(-(n - 1), n)
    dc = np.zeros((n, 2 * n - 1))
    for j in range(n):
        jl = j - lag
        jr = j + lag
        ml = (jl >= 0) & (jl < n)
        mr = (jr >= 0) & (jr < n)
        dc[j, ml] += (1.0 - h[jl[ml]])
        dc[j, mr] -= h[jr[mr]]
    return dc

def binary_2flip(h, ts, max_sweeps=300, time_limit=60):
    n = len(h)
    t0 = time.time()
    h_b = (h > 0.5).astype(float)
    h_b = project_feasible(h_b, ts)
    corr = np.correlate(h_b, 1.0 - h_b, mode="full")
    dx = 2.0 / n
    best_max = corr.max() * dx
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        dc = _dc_matrix(h_b, n)
        zeros = np.where(h_b < 0.5)[0]
        ones = np.where(h_b > 0.5)[0]
        improved = False
        
        # Sort by potential: positions with larger max decrease are tried first
        scores = np.zeros(n)
        for j in range(n):
            if h_b[j] < 0.5:
                scores[j] = (corr + dc[j]).max()  # want this to be small
            else:
                scores[j] = (corr - dc[j]).max()
        
        order = np.argsort(scores)
        
        for j1 in order:
            if h_b[j1] >= 0.5:
                continue  # not a zero position
            for j2 in order:
                if h_b[j2] <= 0.5:
                    continue  # not a one position
                new_max = (corr + dc[j1] - dc[j2]).max() * dx
                if new_max < best_max - 1e-15:
                    h_b[j1] = 1.0
                    h_b[j2] = 0.0
                    corr = corr + dc[j1] - dc[j2]
                    best_max = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h_b, best_max

def interp_resize(h, new_n):
    xo = np.linspace(0, 2, len(h))
    xn = np.linspace(0, 2, new_n)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, new_n / 2.0)

    t_start = time.time()
    
    def time_left():

    # Get initial values

    best_h = h_init.copy()

    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    
    # Phase 2: SLSQP refinement on n=140
    n = len(h_init)
    h_cur = h_init.copy()
    for beta in [5.0, 15.0, 50.0, 150.0, 400.0]:
        if time_left() < 40:
            break
        h_cur = _slsqp_solve(h_cur, beta, maxiter=400)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    
    # Phase 3: 2-flip on SLSQP result
    if time_left() > 50:
        h_b2, c5_b2 = binary_2flip(h_cur, n / 2.0, max_sweeps=300, time_limit=80)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            h_cur = h_b2.copy()
    
    # Phase 4: Try other n values
    for n_try in [100, 120, 160, 200]:
        if time_left() < 60:
            break
        h_try = interp_resize(best_h, n_try)
        
        # SLSQP
        for beta in [10.0, 50.0, 200.0]:
            if time_left() < 30:
                break
            h_try = _slsqp_solve(h_try, beta, maxiter=300)
            c5 = compute_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
        
        # 2-flip
        if time_left() > 40:
            h_bt, c5_bt = binary_2flip(h_try, n_try / 2.0, max_sweeps=200, time_limit=50)
            if c5_bt < best_c5:
                best_c5 = c5_bt
                best_h = h_bt.copy()
    
    # Phase 5: Final SLSQP polish on best
    if time_left() > 30:
        n_best = len(best_h)
        h_pol = _slsqp_solve(best_h.copy(), 300.0, maxiter=500)
        h_pol = _slsqp_solve(h_pol, 800.0, maxiter=300)
        c5_p = compute_c5(h_pol)
        if c5_p < best_c5:

            if c5_bf < best_c5:

                best_h = h_bf.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

from scipy.optimize import minimize
import time

_BETA = 10.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) > 0:
            h[free[0]] = np.clip(h[free[0]] + diff2, 0, 1)
    return h

def _smo_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = corr.max()
    b = _BETA
    return m + np.log(np.sum(np.exp((corr - m) * b))) / b + 50.0 * (x.sum() - n * 0.5) ** 2

def _slsqp(h, beta, maxiter):
    global _BETA
    _BETA = beta
    n = len(h)
    res = minimize(_smo_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n * 0.5)
    return project_feasible(h, n * 0.5)

def _dc_matrix(h, n):
    lag = np.arange(-(n-1), n)
    dc = np.zeros((n, 2*n-1))
    for j in range(n):
        jl = j - lag
        jr = j + lag
        ml = (jl >= 0) & (jl < n)
        mr = (jr >= 0) & (jr < n)
        dc[j, ml] += (1.0 - h[jl[ml]])
        dc[j, mr] -= h[jr[mr]]
    return dc

def two_flip_search(h, ts, max_sweeps, time_limit):
    n = len(h)
    t0 = time.time()
    h_b = (h > 0.5).astype(float)
    h_b = project_feasible(h_b, ts)
    corr = np.correlate(h_b, 1.0 - h_b, mode="full")
    dx = 2.0 / n
    best_max = corr.max() * dx
    n1 = n - 1  # index of lag=0 in full correlation
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:

        dc = _dc_matrix(h_b, n)
        base = corr[None, :] + dc  # shape (n, 2n-1)
        zeros = np.where(h_b < 0.5)[0]
        ones = np.where(h_b > 0.5)[0]
        improved = False

            b_j1[n1] -= 2  # quadratic correction for j1
            best_delta = -1e18
            best_j2 = -1
            for j2 in ones:

                tmp[n1] -= 2  # quadratic correction for j2
                ia = n1 + j2 - j1
                ib = n1 + j1 - j2
                if 0 <= ia < 2*n-1:

                h_b[j2] = 0.0
                corr = np.correlate(h_b, 1.0 - h_b, mode="full")
                best_max = corr.max() * dx
                improved = True
                break
        
        if not improved:
            break
    
    return h_b, best_max

valid_a = (ia_arr >= 0) & (ia_arr < 2*n-1)
valid_b = (ib_arr >= 0) & (ib_arr < 2*n-1)
diffs[np.arange(len(ones))[valid_a], ia_arr[valid_a]] += 1
diffs[np.arange(len(ones))[valid_b], ib_arr[valid_b]] += 1

import numpy as np
from scipy.optimize import minimize
import time

_BETA = 10.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) > 0:
            h[free[0]] = np.clip(h[free[0]] + diff2, 0, 1)
    return h

def _smo_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = corr.max()
    b = _BETA
    return m + np.log(np.sum(np.exp((corr - m) * b))) / b + 50.0 * (x.sum() - n * 0.5) ** 2

def _slsqp(h, beta, maxiter):
    global _BETA
    _BETA = beta
    n = len(h)
    res = minimize(_smo_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n * 0.5)
    return project_feasible(h, n * 0.5)

def _dc_matrix(h, n):
    lag = np.arange(-(n - 1), n)
    dc = np.zeros((n, 2 * n - 1))
    for j in range(n):
        jl = j - lag
        jr = j + lag
        ml = (jl >= 0) & (jl < n)
        mr = (jr >= 0) & (jr < n)
        dc[j, ml] += (1.0 - h[jl[ml]])
        dc[j, mr] -= h[jr[mr]]
    return dc

def two_flip_search(h, ts, max_sweeps, time_limit):
    n = len(h)
    t0 = time.time()
    h_b = (h > 0.5).astype(float)
    h_b = project_feasible(h_b, ts)
    corr = np.correlate(h_b, 1.0 - h_b, mode="full")
    dx = 2.0 / n
    best_max = corr.max() * dx
    n1 = n - 1
    mfull = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        dc = _dc_matrix(h_b, n)
        zeros = np.where(h_b < 0.5)[0]
        ones = np.where(h_b > 0.5)[0]
        improved = False

        for j1 in zeros:
            b_j1 = corr + dc[j1]
            b_j1 = b_j1.copy()
            b_j1[n1] -= 2
            diffs = b_j1[None, :] - dc[ones, :]
            diffs[:, n1] -= 2
            ia_arr = n1 + ones - j1
            ib_arr = n1 + j1 - ones
            idx_all = np.arange(len(ones))
            va = (ia_arr >= 0) & (ia_arr < mfull)
            vb = (ib_arr >= 0) & (ib_arr < mfull)
            diffs[idx_all[va], ia_arr[va]] += 1
            diffs[idx_all[vb], ib_arr[vb]] += 1
            maxes = diffs.max(axis=1) * dx
            bi = np.argmin(maxes)
            if maxes[bi] < best_max - 1e-15:
                best_j2 = ones[bi]
                h_b[j1] = 1.0
                h_b[best_j2] = 0.0
                corr = np.correlate(h_b, 1.0 - h_b, mode="full")
                best_max = corr.max() * dx
                improved = True
                break
        if not improved:
            break
    return h_b, best_max

def interp_resize(h, new_n):
    xo = np.linspace(0, 2, len(h))
    xn = np.linspace(0, 2, new_n)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, new_n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 140
        h_init = np.zeros(n0)
        h_init[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n_init = len(h_init)

    # Phase 1: Quick 2-flip on initial
    h_b, c5_b = two_flip_search(h_init, n_init * 0.5, 200, 80)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Phase 2: SLSQP on n_init
    h_cur = best_h.copy()
    for beta in [5.0, 20.0, 60.0, 200.0]:
        if budget_s - (time.time() - t_start) < 40:
            break
        h_cur = _slsqp(h_cur, beta, 400)
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()

    # Phase 3: 2-flip after SLSQP
    if budget_s - (time.time() - t_start) > 60:
        h_b2, c5_b2 = two_flip_search(h_cur, n_init * 0.5, 300, 100)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            h_cur = h_b2.copy()

    # Phase 4: SLSQP polish
    if budget_s - (time.time() - t_start) > 40:
        h_pol = _slsqp(h_cur, 300.0, 500)
        h_pol = _slsqp(h_pol, 600.0, 300)
        c5_p = compute_c5(h_pol)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_pol.copy()
            h_cur = h_pol.copy()

    # Phase 5: Try other n values
    for n_try in [100, 120, 160, 200]:
        if budget_s - (time.time() - t_start) < 80:
            break
        h_try = interp_resize(best_h, n_try)
        for beta in [10.0, 50.0, 200.0]:
            if budget_s - (time.time() - t_start) < 30:
                break
            h_try = _slsqp(h_try, beta, 250)
            c5 = compute_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
        if budget_s - (time.time() - t_start) > 50:
            h_bt, c5_bt = two_flip_search(h_try, n_try * 0.5, 200, 60)
            if c5_bt < best_c5:
                best_c5 = c5_bt
                best_h = h_bt.copy()

    # Phase 6: Random restarts on best n
    n_best = len(best_h)
    for _ in range(3):
        if budget_s - (time.time() - t_start) < 50:
            break
        h_r = best_h.copy()
        n_flip = max(3, n_best // 20)
        flip_idx = rng.choice(n_best, n_flip, replace=False)
        h_r[flip_idx] = 1.0 - h_r[flip_idx]
        h_r = project_feasible(h_r, n_best * 0.5)
        h_r = _slsqp(h_r, 50.0, 300)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        h_rb, c5_rb = two_flip_search(h_r, n_best * 0.5, 150, 40)
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb.copy()

    # Phase 7: Final polish
    if budget_s - (time.time() - t_start) > 25:
        h_fin = _slsqp(best_h.copy(), 400.0, 400)
        c5_f = compute_c5(h_fin)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fin.copy()
        h_fb, c5_fb = two_flip_search(best_h, len(best_h) * 0.5, 150, 20)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

    best_h = project_feasible(best_h, len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
