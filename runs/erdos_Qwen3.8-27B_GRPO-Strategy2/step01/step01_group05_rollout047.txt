import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _smooth_obj(h, dx, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _smooth_grad(h, dx, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():

        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp_solve(h0, dx, n, target, eps, maxiter=500):
    def obj(x):
        return _smooth_obj(x, dx, eps)
    def grad(x):
        return _smooth_grad(x, dx, eps)
    def cnst(x):
        return x.sum() - target
    def cnst_grad(x):
        return np.ones(len(x))
    res = minimize(
        obj, h0, jac=grad,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': cnst, 'jac': cnst_grad}],
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': 1e-18}
    )
    return _proj(res.x, target)

def _subgrad_polish(h, dx, target, t_end, rng, max_steps=8000):
    n = len(h)
    best_h = h.copy()
    best_c = _c5(h, dx)
    lr = 0.15
    for it in range(max_steps):
        if time.time() > t_end:

        cv = _corr(h, dx)
        mx = cv.max()

        w = np.exp((cv - mx) / 0.05)
        w /= w.sum()
        g = np.zeros(n)

            if w[idx] < 1e-12:
                continue
            wt = w[idx] * dx
            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():

            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= wt * h[jp[vp]]
        h_new = _proj(h - lr * g, target)
        c_new = _c5(h_new, dx)
        if c_new <= _c5(h, dx) + 1e-12:
            h = h_new
            lr = min(lr * 1.005, 1.2)
        else:
            lr = max(lr * 0.97, 0.003)
        if c_new < best_c:

            best_h = h_new.copy()
        if it % 200 == 100:

    return best_h, best_c
