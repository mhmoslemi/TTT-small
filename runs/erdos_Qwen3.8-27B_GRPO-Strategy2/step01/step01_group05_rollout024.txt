    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx]
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * dx * h[jp[vp]]
    return g

def _nesterov(h, dx, n, target, rng, t_end, v0=None):

    eps = 0.05
    lr = 0.05
    momentum = 0.9
    y = h.copy()
    v = v0 if v0 is not None else np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    
    while time.time() < t_end:
        # Nesterov lookahead
        y = _proj(h + momentum * v, target)
        g = _softmax_grad(y, dx, n, eps)
        # Adaptive step
        new_v = momentum * v - lr * g
        h_new = _proj(h + new_v, target)
        c_new = _c5(h_new, dx)
        c_old = _c5(h, dx)
        
        if c_new <= c_old + 1e-12:
            v = new_v
            h = h_new
            lr = min(lr * 1.02, 0.5)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            lr = max(lr * 0.8, 0.001)
            v *= 0.5
        
        # Periodic perturbation
        if rng.random() < 0.05:
            h = _proj(best_h + rng.normal(0, 0.1, n), target)
            v = np.zeros(n)
    
    return best_h, best_c5

def _optimize(h_init, dx, n, target, rng, t_end):
    """Full optimization pipeline for one n and one start"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    
    phase_end = min(time.time() + (t_end - time.time()) * 0.4, t_end)

    for eps in [0.1, 0.03, 0.01, 0.003, 0.001]:
        if time.time() > phase_end:
            break
        try:
            res = minimize(
                lambda x: _softmax_obj(x, dx, n, eps),

                bounds=[(0.0, 1.0)] * n,

                             'jac': lambda x: np.ones(n)}],
                method='SLSQP',
                options={'maxiter': 200, 'ftol': 1e-14}
            )
            h_new = _proj(res.x, target)
        except:
            break
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new

    phase2_end = min(time.time() + (t_end - time.time()) * 0.4, t_end)
    h, c5_nv = _nesterov(h, dx, n, target, rng, phase2_end)
    if c5_nv < best_c5:
        best_c5 = c5_nv
        best_h = h.copy()
    
    # Phase 3: Jitter / local search
    for it in range(200):
        if time.time() > t_end:
            break
        # Perturb 1-3 random coordinates
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, n_pert, replace=False)
        h_try = best_h.copy()
        h_try[idx] += rng.normal(0, 0.1, n_pert)
        h_try = _proj(h_try, target)
        c_try = _c5(h_try, dx)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

        for eps in [0.001, 0.0005]:
            if time.time() > t_end - 2:
                break
            try:
                res = minimize(
                    lambda x: _softmax_obj(x, dx, n, eps),
                    best_h,
                    jac=lambda x: _softmax_grad(x, dx, n, eps),
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                 'jac': lambda x: np.ones(n)}],
                    method='SLSQP',
                    options={'maxiter': 300, 'ftol': 1e-15}
                )
                h_new = _proj(res.x, target)
                c = _c5(h_new, dx)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_new.copy()
            except:
                pass
    
    return best_h, best_c5

    c = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = c.max()
    return mx + _EPS * np.log(np.sum(np.exp((c - mx) / _EPS)))


def _grad(x):
    c = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = c.max()
    w = np.exp((c - mx) / _EPS)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue

        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * _DX * x[jp[vp]]
    return g


def _eq_fun(x):
    return x.sum() - _TGT


def _eq_jac(x):
    return np.ones(len(x))


def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    _DX = dx
    _N = n
    _EPS = float(eps)
    _TGT = float(target)

        res = minimize(_obj, h0, jac=_grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = _proj(res.x, target)
    except Exception:
        h_out = h0.copy()
    return h_out
