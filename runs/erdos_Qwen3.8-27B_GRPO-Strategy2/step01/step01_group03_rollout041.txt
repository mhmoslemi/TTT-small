import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) * beta))) / beta

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def slsqp(h, n, ts, beta, maxiter=400):
    res = minimize(smooth_obj, h, args=(beta,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def grad_max_lag(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    offset = k_star - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        g_val = 0.0
        h_val = 0.0
        if 0 <= m - offset < n:
            g_val = (1.0 - h[m - offset])
        if 0 <= m + offset < n:
            h_val = h[m + offset]
        grad[m] = g_val - h_val
    grad -= np.mean(grad)
    return grad, k_star

def adversarial_descent(h, ts, t_start, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.008
    for it in range(20000):
        if time.time() > deadline:
            break
        grad, _ = grad_max_lag(h, len(h))
        h_new = h - lr * grad
        h_new = project_feasible(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
        lr = max(0.0003, 0.008 * np.exp(-it * 0.0005))
    return best_h, best_c5

    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = 0.004

    for it in range(5000000):
