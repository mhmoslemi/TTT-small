                idx2 = idx1 + s

                g[idx2] -= dx * h_curr[idx1]

        lr = max(gap / (np.linalg.norm(g)**2 + 1e-12), 0.001)

        c_new = _c5(h_new, dx)
        if c_new < c_best:
            c_best = c_new
            h_best = h_new.copy()
        h_curr = h_new
    return h_best, c_best

import numpy as np
from scipy.optimize import minimize
import time

_DX = 2.0 / 70
_N = 70
_EPS = 0.01
_TGT = 35.0
_MU = 200.0


def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _c5(h, dx):
    return _corr(h, dx).max()


def _proj(h, target):
    lo, hi = -2.5, 3.5
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _obj_lse(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))


def _grad_lse(x):
    n = _N
    cv = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = cv.max()
    w = np.exp((cv - mx) / _EPS)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        wt = w[k] * _DX
        s = k - (n - 1)
        if s == 0:
            g += wt * (1.0 - 2.0 * x)
        else:
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            idx1 = np.arange(i_lo, i_hi)
            idx2 = idx1 + s
            g[idx1] += wt * (1.0 - x[idx2])
            g[idx2] -= wt * x[idx1]
    return g


def _cons_eq(x):
    return x.sum() - _TGT


def _cons_eq_jac(x):
    return np.ones(_N)


def _obj_pen(x):
    c = x.sum() - _TGT
    return _obj_lse(x) + _MU * c * c


def _grad_pen(x):
    c = x.sum() - _TGT
    return _grad_lse(x) + 2.0 * _MU * c


def _slsqp_opt(h0, dx, n, target, eps, maxiter=600):
    global _DX, _N, _EPS, _TGT
    _DX, _N, _EPS, _TGT = dx, n, eps, target
    res = minimize(
        _obj_lse, h0, jac=_grad_lse,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}],
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 60}
    )
    return _proj(res.x, target)


def _lbfgs_opt(h0, dx, n, target, eps, mu, maxiter=5000):
    global _DX, _N, _EPS, _TGT, _MU
    _DX, _N, _EPS, _TGT, _MU = dx, n, eps, target, mu
    res = minimize(
        _obj_pen, h0, jac=_grad_pen,
        bounds=[(0.0, 1.0)] * n,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                 'maxcor': 20, 'maxls': 60}
    )
    return _proj(res.x, target)


def _subgrad_step(h, dx, target, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-10)[0]
    n = len(h)
    g = np.zeros(n)
    for idx in active:
        s = idx - (n - 1)
        if s == 0:
            g += dx * (1.0 - 2.0 * h)
        else:
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            idx1 = np.arange(i_lo, i_hi)
            idx2 = idx1 + s
            g[idx1] += dx * (1.0 - h[idx2])
            g[idx2] -= dx * h[idx1]
    g /= max(len(active), 1)
    return _proj(h - lr * g, target)


def _subgrad_polish(h, dx, target, t_end, maxit=50000):
