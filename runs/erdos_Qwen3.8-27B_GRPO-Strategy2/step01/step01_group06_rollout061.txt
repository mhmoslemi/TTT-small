import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lse_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    vmax = np.max(vals)
    return vmax + np.log(np.sum(np.exp(beta * (vals - vmax)))) / beta

def lse_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    vmax = np.max(vals)
    b_shifted = beta * (vals - vmax)
    exps = np.exp(b_shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    grad *= dx
    return grad

def slsqp_smooth(h, n, dx, beta, maxiter=2000):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return lse_obj(x, dx, beta)
    
    def obj_grad(x):
        return lse_grad(x, dx, beta)
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def subgrad_descent(h, n, dx, maxiter=5000, lr=0.002):
    h = h.copy()
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    target = float(n) / 2.0
    lr_cur = lr
    stall = 0
    for it in range(maxiter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        active = np.where(np.abs(corr - max_val) < 1e-10)[0]
        g = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            if lag >= 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                L = -lag
                g[:n - L] += (1.0 - h[L:])
                g[L:] -= h[:n - L]
        g *= dx / len(active)
        h_new = h - lr_cur * g
        h_new = project_feasible(h_new, n)
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            lr_cur *= 0.95
            if stall > 200:
                lr_cur = lr
                stall = 0
    return best_h, best_c5
