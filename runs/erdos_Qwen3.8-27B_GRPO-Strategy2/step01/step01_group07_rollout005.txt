import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 0.0
_g_alpha = 100.0
_g_target = 0.0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    c = np.clip(corr - m, -5.0, 0.0)
    return (m + np.log(np.sum(np.exp(_g_alpha * c))) / _g_alpha) * _g_dx

def _eq_fun(x):
    return np.sum(x) - _g_target

def _eq_jac(x):
    return np.ones_like(x)

def _norm(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _make_init(n, seed, typ):
    rng = np.random.RandomState(seed)
    x = np.arange(n) / n
    if typ == 0:
        h = 0.5 + rng.uniform(0.15, 0.4) * np.sin(2 * np.pi * rng.randint(1, 5) * x + rng.uniform(0, 6.28))
    elif typ == 1:
        h = np.zeros(n)
        k = rng.randint(2, 6)
        bounds = np.sort(rng.uniform(0, 1, k))
        for i in range(k + 1):
            lo = bounds[i - 1] if i > 0 else 0
            hi = bounds[i] if i < k else 1.001
            m = (x >= lo) & (x < hi)
            h[m] = rng.uniform(0.1, 0.9)
    elif typ == 2:
        h = 0.5
        for _ in range(rng.randint(2, 6)):
            h += rng.uniform(0.05, 0.2) * np.sin(2 * np.pi * rng.randint(1, n // 3) * x + rng.uniform(0, 6.28))
    elif typ == 3:
        h = 0.5 + rng.uniform(0.1, 0.35) * np.cos(2 * np.pi * rng.randint(1, 4) * x + rng.uniform(0, 6.28))
    else:
        h = 0.5 + rng.uniform(0.1, 0.3) * np.sin(6 * np.pi * x + rng.uniform(0, 6.28))
    return _norm(h, n / 2.0)

def _local_search(h, target, max_sweeps=800, step0=0.015, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    best_c = _c5(h)
    step = step0
    for sweep in range(max_sweeps):
        improved = False
        for _ in range(n):
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            for d in (step, -step):
                h2 = h.copy()
                h2[i] += d
                h2[j] -= d
                if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                    continue
                c = _c5(h2)
                if c < best_c - 1e-13:
                    h = h2
                    best_c = c
                    improved = True
                    break
        if not improved:
            step *= 0.65
            if step < 0.0004:
                break
    return h, best_c

def _active_grad_step(h, target, max_steps=300, lr0=0.005):
    n = len(h)
    dx = 2.0 / n
    best_c = _c5(h)
    eps = 1e-5
    for it in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = corr.max()
        active = np.where(corr > mx - 0.5)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        # Numerical gradient of sum of active lags
        grad = np.zeros(n)
        for i in range(n):
            hp = h.copy()
            hp[i] = min(1.0, hp[i] + eps)
            hm = h.copy()
            hm[i] = max(0.0, hm[i] - eps)
            cp = np.correlate(hp, 1.0 - hp, mode='full')[active]
            cm = np.correlate(hm, 1.0 - hm, mode='full')[active]
            grad[i] = (cp.sum() - cm.sum()) / (hp[i] - hm[i] + 1e-15)
        grad *= dx
        lr = lr0 / (1.0 + 0.1 * it)
        h_new = h - lr * grad
        # Project to {0<=h<=1, sum=target}
        for _ in range(60):
            h_new = np.clip(h_new, 0.0, 1.0)
            diff = target - np.sum(h_new)
            if abs(diff) < 1e-13:
                break
            free = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
            if not np.any(free):
                break
            h_new[free] += diff / np.sum(free)
        h_new = np.clip(h_new, 0.0, 1.0)
        c_new = _c5(h_new)
        if c_new < best_c:
            h = h_new
            best_c = c_new
        else:
            lr0 *= 0.7
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global _g_dx, _g_alpha, _g_target
    t0 = time.time()

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:

    best_h = parent.copy()
    best_n = pn
    print(f"Parent: n={pn}, C5={best_c5:.6f}")

    n_list = [50, 60, 70, 80, 90, 100, 120]

    for n in n_list:
        rem = budget_s - (time.time() - t0)
        if rem < 30:
            break
        target = n / 2.0
        _g_target = target
        _g_dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        inits = []
        if n == pn:
            inits.append(parent.copy())
        else:
            xp = np.linspace(0, 1, pn)
            xn = np.linspace(0, 1, n)
            inits.append(_norm(np.interp(xn, xp, parent), target))
        for ti in range(4):
            inits.append(_make_init(n, seed * 131 + n * 17 + ti, ti))

        n_restarts = max(2, min(len(inits), int(rem / 35)))

        for r in range(n_restarts):
            rem = budget_s - (time.time() - t0)
            if rem < 10:
                break
            hc = inits[r].copy()

            alphas = (20, 50, 150, 400, 1000, 3000)
            for al in alphas:
                rem = budget_s - (time.time() - t0)
                if rem < 5:
                    break
                _g_alpha = al
                try:
                    mi = 80 if n <= 80 else 50
                    res = minimize(_smooth_obj, hc, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': mi, 'ftol': 1e-12, 'iprint': 0})
                    hc = _norm(res.x, target)
                except Exception:
                    pass

            rem = budget_s - (time.time() - t0)
            if rem > 8:
                rng_ls = np.random.RandomState(seed + r * 7 + n * 3)
                hc, _ = _local_search(hc, target, max_sweeps=min(400, int(rem * 25)),
                                      step0=0.015, rng=rng_ls)

            cc = _c5(hc)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()
                best_n = n

    # Final refinement on best solution
    rem = budget_s - (time.time() - t0)
    if rem > 12:
        target = best_n / 2.0
        _g_target = target
        _g_dx = 2.0 / best_n
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        try:
            res = minimize(_smooth_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 80, 'ftol': 1e-14, 'iprint': 0})
            best_h = _norm(res.x, target)
        except Exception:
            pass

        rem = budget_s - (time.time() - t0)
        if rem > 6 and best_n <= 100:
            best_h, _ = _active_grad_step(best_h, target, max_steps=min(200, int(rem * 8)), lr0=0.004)

        rem = budget_s - (time.time() - t0)
        if rem > 4:
            rng_f = np.random.RandomState(999)
            best_h, _ = _local_search(best_h, target, max_sweeps=min(600, int(rem * 30)),
                                      step0=0.004, rng=rng_f)

    # Fix sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-13:
        free = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)

    c5f = _c5(best_h)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return best_h, c5f, best_n
