import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_objective(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    log_Z = corr_max * alpha + np.log(np.sum(np.exp(alpha * (corr - corr_max))))
    return dx / alpha * log_Z

def smooth_gradient(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - corr_max))
    Z = exp_corr.sum()
    w = exp_corr / Z
    v = 1.0 - 2.0 * h
    grad = dx * np.correlate(w, v, mode='full')[:n]
    return grad

def slsqp_smooth(h, n, dx, alpha, maxiter=500, ftol=1e-12):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_objective(x, dx, alpha)
    
    def obj_grad(x):
        return smooth_gradient(x, dx, alpha)
    
    def ceq(x):
        return x.sum() - target
    
    def cjac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def proj_grad_smooth(h, n, dx, alpha, maxiter=50000, lr=0.01):
    target = float(n) / 2.0
    h = h.copy()
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    lr_cur = lr
    momentum = np.zeros(n)
    beta = 0.9
    for it in range(maxiter):
        g = smooth_gradient(h, dx, alpha)
        momentum = beta * momentum + (1 - beta) * g
        h_new = h - lr_cur * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                room = 1.0 - h_new
                s = room.sum()
                if s > 1e-14:
                    h_new += diff * room / s
            else:
                s = h_new.sum()
                if s > 1e-14:
                    h_new += diff * h_new / s
            h_new = np.clip(h_new, 0.0, 1.0)
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            lr_cur *= 0.95
            if lr_cur < 1e-10:
                lr_cur = 0.01
                momentum *= 0.5
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            mask = (1.0 - h) > 1e-10 if d2 > 0 else h > 1e-10
            if mask.any():
                h[mask] += d2 / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    return (dx / alpha) * (alpha * cm + np.log(np.sum(np.exp(alpha * (corr - cm)))))

def smooth_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ec = np.exp(alpha * (corr - cm))
    w = ec / ec.sum()
    v = 1.0 - 2.0 * h
    return dx * np.correlate(w, v, mode='full')[:n]

def lbfgsb_smooth(h0, n, dx, alpha, lam, maxiter=200):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, dx, alpha) + lam * (x.sum() - target) ** 2
    def grad(x):
        return smooth_grad(x, dx, alpha) + 2.0 * lam * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h0, eval_c5(h0, dx)

def slsqp_smooth(h0, n, dx, alpha, maxiter=400, ftol=1e-14):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, dx, alpha)
    def jac(x):
        return smooth_grad(x, dx, alpha)
    def ceq(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': ceq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h0, eval_c5(h0, dx)

def proj_grad_smooth(h0, n, dx, alpha, maxiter=20000, lr0=0.008):
    target = float(n) / 2.0
    h = h0.copy()
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    lr = lr0
    mom = np.zeros(n)
    beta = 0.9
    stall = 0
    for it in range(maxiter):
        g = smooth_grad(h, dx, alpha)
        mom = beta * mom + (1.0 - beta) * g
        h_new = h - lr * mom
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h_new
                s = room.sum()
                if s > 1e-12:
                    h_new += diff * room / s
            else:
                s = h_new.sum()
                if s > 1e-12:
                    h_new += diff * h_new / s
            h_new = np.clip(h_new, 0.0, 1.0)
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0

                    lr = lr0
                    mom = np.zeros(n)
                    stall = 0
    return best_h, best_c5
