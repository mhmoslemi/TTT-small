    return float(np.max(c) * dx)

def project(h, target):
    """Project onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def worst_lag_grad(h, n):
    """Gradient of the worst (max) correlation value w.r.t. h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    kmax = np.argmax(c)
    m = kmax - (n - 1)  # lag
    grad = np.zeros(n)
    if m >= 0:

            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
        else:
            # m=0, c_0 = sum h[i](1-h[i]), grad = 1-2h
            grad[:] = 1.0 - 2.0 * h
    else:
        mm = -m
        if mm > 0:
            grad[:n-mm] += (1.0 - h[mm:])
            grad[mm:] -= h[:n-mm]
        # m can't be 0 here since we're in else
    return grad

def compute_c5(h):

def project(h, target):

        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def lag_gradient(h, n, m):

    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    else:
        mm = -m
        grad[:n - mm] += (1.0 - h[mm:])
        grad[mm:] -= h[:n - mm]
    return grad

def topk_grad(h, n, k, t):
    c = np.correlate(h, 1.0 - h, mode='full')
