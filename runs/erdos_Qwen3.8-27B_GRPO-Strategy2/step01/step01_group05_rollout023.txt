def _corr(h, dx):

    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()

        jp = j + s

        if vp.any():

        dx = 2.0 / n
        target = n / 2.0
        h_f, c_f = _optimize_n(best_h_global, dx, n, target, rng, deadline - 2)
        if c_f < best_c_global:
            best_c_global = c_f
            best_h_global = h_f

    return best_h_global, best_c_global, best_n_global
