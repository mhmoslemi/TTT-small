import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _grad_c5(h, n):
    # Gradient of c5 (the max correlation term) w.r.t. h
    # We compute gradients of all correlation terms, return weighted sum for active ones
    corr = _corr(h)
    max_val = np.max(corr)
    # Active set: terms within 1% of max
    tol = 0.01 * max_val + 1e-10
    active = corr >= (max_val - tol)
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2*n-1):
        if not active[k]:
            continue
        lag = k - (n-1)
        for j in range(n):
            g = 0.0
            # Term: h[j] appears in the sum
            # c[k] = sum_i h[i+lag]*(1-h[i])
            # d/dh[j]: 
            #   From h[i+lag]=h[j]: i = j-lag, contributes (1-h[i]) = (1-h[j-lag]) if j-lag in [0,n)
            #   From (1-h[i]) where i=j: contributes -h[i+lag] = -h[j+lag] if j+lag in [0,n)
            if 0 <= j - lag < n:
                g += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                g -= h[j + lag]
            grad[j] += g * active[k]  # unweighted for simplicity
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Scale to fix sum, then clip, then distribute remainder
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    tol = 1e-12
    if abs(diff) > tol:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

            grad[j] += w[list(active).index(k)] * g

    h_new = h - alpha_k * grad
    h_new = _repair(h_new, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

One more optimization idea: In the coordinate descent, instead of trying fixed deltas, I can try to compute the optimal step for a given pair (i,j). For a fixed pair (i,j) and step d, the correlation terms that change are:
- Terms involving h[i]: c[k] for k where i or i+lag is in the sum
- Terms involving h[j]: c[k] for k where j or j+lag is in the sum

For each such term, c[k](d) = c[k](0) + a_k*d + b_k*d² where a_k and b_k depend on the specific indices.

The max of these quadratics is a convex function of d (since max of convex functions is convex). We can minimize it by ternary search or golden section.
