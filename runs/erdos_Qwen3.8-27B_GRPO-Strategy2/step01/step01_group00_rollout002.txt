import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _local_search(h, dx, n, rng, max_iter=5000, t_end=None):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    step_sizes = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    stagnation = 0
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        # Strategy 1: random coordinate pair
        i = rng.randint(n)
        js = rng.choice(n, size=min(n, 15), replace=False)
        for j in js:
            if i == j:
                continue
            for d in step_sizes:
                for sgn in [1, -1]:
                    di = sgn * d
                    ni = h[i] + di
                    nj = h[j] - di
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5 = _compute_c5(h, dx)
                    if c5 < best_c5 - 1e-11:
                        best_c5 = c5
                        improved = True
                        break
                    h[i] -= di
                    h[j] += di
            if improved:
                break
        # Strategy 2: target the binding lag
        if not improved:
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n - 1)
            if 0 <= k < n - 1:
                for ti in range(n - k):
                    diff_v = h[ti] - h[ti + k]
                    if abs(diff_v) < 0.03:
                        continue
                    d = min(0.015, abs(diff_v) * 0.5)
                    if diff_v > 0:
                        di = -d
                    else:
                        di = d
                    ni = h[ti] + di
                    nj = h[ti + k] - di
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[ti] = ni
                    h[ti + k] = nj
                    c5 = _compute_c5(h, dx)
                    if c5 < best_c5 - 1e-11:
                        best_c5 = c5
                        improved = True
                        break
                    h[ti] -= di
                    h[ti + k] += di
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 300:
                break
    return h, best_c5


def _slsqp_optimize(h, n, bounds, cons):
    best = h.copy()
    best_c = _compute_c5(best, 2.0 / n)
    for alpha in [50, 100, 250, 500, 1000]:
        try:
            res = minimize(
                lambda x, a=alpha: _smooth_obj(x, a), best,
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 250, 'ftol': 1e-13}
            )
            hc = _repair(res.x, n)
            c = _compute_c5(hc, 2.0 / n)
            if c < best_c:
                best_c = c
                best = hc
        except Exception:
            pass
    return best, best_c

        target = n / 2.0

        # Phase 1: SLSQP
        if time.time() - t_start < budget_s - 160:
