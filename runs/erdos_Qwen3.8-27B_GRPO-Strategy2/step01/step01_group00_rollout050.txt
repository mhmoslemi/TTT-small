        for _pass in range(400):
            if time.time() - t0 > budget_s - 30:
                break
            improved = False
            for _ in range(60):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = deltas[rng.randint(len(deltas))]
                hi_orig, hj_orig = h[i], h[j]
                ni, nj = hi_orig + d, hj_orig - d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                if nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i] = min(max(ni, 0.0), 1.0)
                h[j] = min(max(nj, 0.0), 1.0)
                c5 = _c5(h, dx)
                if c5 < bl - 1e-10:
                    bl = c5
                    improved = True
                else:
                    h[i] = hi_orig
                    h[j] = hj_orig
            if not improved and _pass > 50:
                break
        if bl < best_c5:
            best_c5 = bl
            best_h = h

    # Phase 5: Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        try:
            res = minimize(_smooth_obj, best_h, args=(5000,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
