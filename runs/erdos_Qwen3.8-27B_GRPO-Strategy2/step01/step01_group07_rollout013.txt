import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -50.0, 0.0)
    es = np.exp(s)
    w = es / np.sum(es)
    val = (m + np.log(np.sum(es)) / alpha) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        j = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        if j == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif j > 0:
            grad[j:] += wk * (1.0 - h[:n - j])
            grad[:n - j] -= wk * h[j:]
        else:
            jj = -j
            grad[:n - jj] += wk * (1.0 - h[jj:])
            grad[jj:] -= wk * h[:n - jj]
    return val, grad

def _proj(x, target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(40):
        d = target - np.sum(x)
        if abs(d) < 1e-14:
            break
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += d / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _fista(h0, target, n_steps, alpha):

    y = x.copy()
    t_val = 1.0
    best_c = _c5(x)
    best_x = x.copy()
    grad_y = None
    for st in range(n_steps):
        val, grad = _smooth_obj_grad(y, alpha)
        if grad_y is not None:
            L = np.dot(grad - grad_y, y - x)
            if L > 1e-16:
                lr = 0.5 * (np.dot(grad - grad_y, y - x)) / (L + 1e-16)
                lr = min(lr, 50.0 / alpha)
            else:
                lr = 20.0 / (alpha + 4.0)
        else:
            lr = 20.0 / (alpha + 4.0)
        y_new = y - lr * grad
        x_new = _proj(y_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + (t_val - 1.0) / t_new * (x_new - x)
        y = _proj(y, target)
        x = x_new
        t_val = t_new
        grad_y = grad
        if st % 20 == 0:
            c = _c5(x)
            if c < best_c:
                best_c = c
                best_x = x.copy()
    c = _c5(x)
    if c < best_c:
        best_c = c
        best_x = x.copy()
    return best_x, best_c

    n = len(h)
    bc = _c5(h)
    rng = np.random.RandomState(7)

def _anneal(h0, target, rounds=25, T0=0.06, T1=0.003):

def _slsqp_smooth(h_init, target, alpha):
    n = len(h_init)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -50.0, 0.0)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 150, 'ftol': 1e-14})
