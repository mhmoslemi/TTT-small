import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_box_eq(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = int(np.argmax(x)) if diff < 0 else int(np.argmin(x))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def smooth_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = np.clip((c - cmax) * t, -60.0, 60.0)
    exps = np.exp(shifted)
    logsum = np.log(exps.sum())
    val = (cmax + logsum / t) * dx
    w = exps / exps.sum()
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        s = p - (n - 1)
        wp = w[p]
        if wp < 1e-12:
            continue
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[s:] += wp * (1.0 - h[:ns])
                grad[:ns] += wp * (-h[s:])
        else:
            sp = -s
            ns = n - sp
            if ns > 0:
                grad[:ns] += wp * (1.0 - h[sp:])
                grad[sp:] += wp * (-h[:ns])
    grad *= dx
    return val, grad

def proj_gradient(h, t, lr_start, n_iter):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    lr = lr_start
    for i in range(n_iter):
        val, grad = smooth_obj_grad(h, t)
        alpha = lr
        h_try = h
        val_try = val
        for _ in range(25):
            h_try = project_to_box_eq(h - alpha * grad, target)
            val_try, _ = smooth_obj_grad(h_try, t)
            if val_try <= val * (1 - 1e-10) or val_try < val - 1e-12:
                break
            alpha *= 0.5
        h = h_try
        lr = max(alpha, lr_start * 0.3)
    return h, compute_c5(h)

def coord_desc(h, max_sweeps=40):
    n = len(h)
    target = n * 0.5
    best = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_cand = old
            best_val = best
            cands = [0.0, 1.0, 0.5, max(0.0, old - 0.1), max(0.0, old - 0.04),
                     min(1.0, old + 0.04), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                h[i] = cand
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    adjust = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
                    adjust[i] = True
                    cnt = adjust.sum()
                    if cnt > 0:
                        h[adjust] += diff / cnt
                        h = np.clip(h, 0.0, 1.0)
                v = compute_c5(h)
                if v < best_val - 1e-14:
                    best_val = v
                    best_cand = cand
            if best_cand != old:
                h[i] = best_cand
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    adjust = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
                    adjust[i] = True
                    cnt = adjust.sum()
                    if cnt > 0:
                        h[adjust] += diff / cnt
                        h = np.clip(h, 0.0, 1.0)
                best = best_val
                improved = True
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = project_to_box_eq(h0, n0 * 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient with temperature annealing at n0
    schedules = [
        (50.0, 5.0, 400), (100.0, 3.0, 400), (200.0, 2.0, 400),
        (400.0, 1.0, 400), (800.0, 0.5, 300), (1500.0, 0.3, 300),
    ]
    for t, lr_s, n_iter in schedules:
        if time_left() < 80:
            break
        h_try, v = proj_gradient(best_h, t, lr_s, n_iter)
        if v < best_c5:
            best_c5 = v
            best_h = h_try.copy()

    # Phase 2: Coordinate descent polish
    if time_left() > 50:
        h_try, v = coord_desc(best_h, max_sweeps=60)
        if v < best_c5:
            best_c5 = v
            best_h = h_try.copy()

    # Phase 3: Another projected gradient round at higher temp
    if time_left() > 80:
        h_try, v = proj_gradient(best_h, 2000.0, 0.2, 300)
        if v < best_c5:
            best_c5 = v
            best_h = h_try.copy()
        h_try, v = coord_desc(best_h, max_sweeps=40)
        if v < best_c5:
            best_c5 = v
            best_h = h_try.copy()

    # Phase 4: Try at half resolution
    if time_left() > 100:
        n_half = n0 // 2
        idx = np.linspace(0, n0, n_half + 1).astype(int)
        h_half = (best_h[idx[:-1]] + best_h[idx[1:]]) / 2.0
        h_half = project_to_box_eq(h_half, n_half * 0.5)
        for t, lr_s, n_iter in [(80.0, 4.0, 300), (200.0, 2.0, 300), (500.0, 1.0, 300)]:
            if time_left() < 50:
                break
            h_try, v = proj_gradient(h_half, t, lr_s, n_iter)
            if v < best_c5:
                best_c5 = v
                best_h = h_try.copy()
        if time_left() > 30:
            h_try, v = coord_desc(h_half, max_sweeps=40)
            if v < best_c5:
                best_c5 = v
                best_h = h_try.copy()

    # Phase 5: Random restarts
    if time_left() > 60:
        n_use = n0
        for attempt in range(5):
            if time_left() < 30:
                break
            scale = [0.1, 0.2, 0.3, 0.15, 0.25][attempt]
            noise = np.random.randn(n_use) * scale
            h_try = project_to_box_eq(best_h[:n_use] + noise, n_use * 0.5)
            h_try, v = proj_gradient(h_try, 300.0, 2.0, 250)
            h_try, v = coord_desc(h_try, max_sweeps=25)
            if v < best_c5:
                best_c5 = v
                best_h = h_try.copy()

    # Phase 6: SLSQP final polish
    if time_left() > 40 and len(best_h) <= 400:
        n = len(best_h)
        t_slqp = 500.0
        def slqp_obj(x):
            return smooth_obj_grad(x, t_slqp)[0]
        def slqp_grad(x):
            return smooth_obj_grad(x, t_slqp)[1]
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5,
                 'jac': lambda x, n=n: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(slqp_obj, best_h.copy(), jac=slqp_grad, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            h_try = project_to_box_eq(res.x, n * 0.5)
            v = compute_c5(h_try)
            if v < best_c5:
                best_c5 = v
                best_h = h_try.copy()
        except Exception:
            pass

    # Phase 7: Final coordinate descent
    if time_left() > 15:
        h_try, v = coord_desc(best_h, max_sweeps=30)
        if v < best_c5:
            best_c5 = v
            best_h = h_try.copy()

    n_final = len(best_h)
    best_h = project_to_box_eq(best_h, n_final * 0.5)
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n_final)
