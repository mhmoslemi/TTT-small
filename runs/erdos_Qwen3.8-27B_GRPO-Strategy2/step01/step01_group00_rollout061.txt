```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += diff * room / tot
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += diff * h / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_polish(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    
    def eqfun(x):
        return x.sum() - target
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eqfun}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return _repair(res.x, n)

                # To reduce h[m]*(1-h[ms]): increase h[ms] or decrease h[m]
                hi = h[ms]
                target_j = m  # decrease h[m] or some other
                for d_pos in [0.002, 0.005, 0.01, 0.02]:
```

```python
def _lag_constraint_direction(h, n, s):
    """Compute the direction that makes h[j+s]-h[j-s] more constant."""
    # For all valid j, we want h[j+s] - h[j-s] = c
    # Estimate c as the average
    diffs = []
    for j in range(n):
        js = j + s
        js2 = j - s
        if 0 <= js < n and 0 <= js2 < n:
            diffs.append(h[js] - h[js2])
    if not diffs:
        return None
    c = np.mean(diffs)
    # Direction: reduce variance of (h[j+s] - h[j-s])
    # Gradient direction for each h[k]:
    # h[k] appears as h[j+s] when j = k-s: contributes +1 to diff
    # h[k] appears as h[j-s] when j = k+s: contributes -1 to diff
    # So grad[k] = (h[k-s] - c) * (1/n_chain) - (h[k+s] - c) * ... 
    # Actually, to minimize sum_j (h[j+s]-h[j-s]-c)^2:
    # d/dh[k] = 2 * sum over j where k=j+s: (h[j+s]-h[j-s]-c)
    #          - 2 * sum over j where k=j-s: (h[j+s]-h[j-s]-c)
    # = 2*(diff[k-s] if valid) - 2*(diff[k+s] if valid)
    direction = np.zeros(n)
    for j in range(n):
        js = j + s
        js2 = j - s
        if 0 <= js < n and 0 <= js2 < n:
            d = h[js] - h[js2] - c
            direction[js] += d
            direction[js2] -= d
    return direction
```
