```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0-h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj_sum(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    t = 0.5*(lo+hi)
    return np.clip(h-t, 0, 1)

def _sobj(h, dx, n, eps, pen):
    cv = _corr(h, dx)
    mx = cv.max()
    sm = mx + eps*np.log(np.sum(np.exp((cv-mx)/eps)))
    v = h.sum()-n*0.5
    return sm + pen*v*v

def _sgrad(h, dx, n, eps, pen):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv-mx)/eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2*n-1):
        if w[idx] < 1e-13:
            continue
        wt = w[idx]*dx
        s = idx-(n-1)
        jm = j-s
        vm = (jm>=0)&(jm<n)
        if vm.any():
            g[vm] += wt*(1.0-h[jm[vm]])
        jp = j+s
        vp = (jp>=0)&(jp<n)
        if vp.any():
            g[vp] -= wt*h[jp[vp]]
    g += 2.0*pen*(h.sum()-n*0.5)
    return g

def _lfb(h0, dx, n, eps, pen, maxiter=500):
    res = minimize(_sobj, h0, jac=_sgrad, args=(dx,n,eps,pen),
                   bounds=[(0.0,1.0)]*n, method='L-BFGS-B',
                   options={'maxiter':maxiter,'ftol':1e-14,'gtol':1e-7})
    return res.x

def _subgrad(h, dx, n, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = cv >= mx - 1e-9
    g = np.zeros(n)
    j = np.arange(n)
    na = active.sum()
    for idx in np.where(active)[0]:
        s = idx-(n-1)
        jm = j-s
        vm = (jm>=0)&(jm<n)
        if vm.any():
            g[vm] += dx*(1.0-h[jm[vm]])
        jp = j+s
        vp = (jp>=0)&(jp<n)
        if vp.any():
            g[vp] -= dx*h[jp[vp]]
    g /= max(na,1)
    return h - lr*g

def _optimize(h0, dx, n, t_end):
    target = n*0.5
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = h0.copy()
    eps_list = [0.3,0.1,0.05,0.02,0.01,0.005,0.002,0.001,0.0005,0.0002]
    for eps in eps_list:
        if time.time() > t_end - 20:
            break
        pen = 5.0
        try:
            h_new = _lfb(h, dx, n, eps, pen, maxiter=400)
            h_new = _proj_sum(h_new, target)
        except:
            break
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    # Subgradient polish
    lr = 0.1
    for it in range(3000):
        if time.time() > t_end - 10:
            break
        h2 = _subgrad(h, dx, n, lr)
        h2 = _proj_sum(h2, target)
        c2 = _c5(h2, dx)
        cc = _c5(h, dx)
        if c2 <= cc:
            h = h2
            lr = min(lr*1.02, 1.0)
        else:
            lr = max(lr*0.95, 0.01)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s*0.97
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0/n
    target = n*0.5
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = np.interp(np.linspace(0,1,n), np.linspace(0,1,len(h0)), h0)
    else:
        h0 = np.full(n, 0.5)
    
    h0 = _proj_sum(h0, target)
    
    # Phase 1: Main optimization
    t1 = t0 + (deadline-t0)*0.45
    best_h, best_c5 = _optimize(h0, dx, n, t1)
    
    # Phase 2: Restarts
    t2 = t0 + (deadline-t0)*0.85
    while time.time() < t2:
        amp = rng.uniform(0.05, 0.25)
        h_pert = _proj_sum(best_h + rng.normal(0, amp, n), target)
        sub_end = min(time.time()+30, t2)
        if sub_end - time.time() < 5:
            break
        h_new, c_new = _optimize(h_pert, dx, n, sub_end)
        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new
    
    # Phase 3: Final polish
    if time.time() < deadline - 10:
        h_f, c_f = _optimize(best_h, dx, n, deadline)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f
    
    return best_h, best_c5, n
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj_sum(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _sobj(h, dx, n, eps, pen):
    cv = _corr(h, dx)
    mx = cv.max()
    sm = mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))
    v = h.sum() - n * 0.5
    return sm + pen * v * v

def _sgrad(h, dx, n, eps, pen):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-13:
            continue
        s = idx - (n - 1)
        wt = w[idx] * dx
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            g[lo1:hi1] += wt * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            g[lo2:hi2] -= wt * h[lo2 - s:hi2 - s]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lfb(h0, dx, n, eps, pen, maxiter=600):
    res = minimize(_sobj, h0, jac=_sgrad, args=(dx, n, eps, pen),
                   bounds=[(0.0, 1.0)] * n, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                            'maxls': 50, 'factr': 1e12})
    return res.x

def _subgrad_step(h, dx, n, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = cv >= mx - 1e-9
    na = max(active.sum(), 1)
    g = np.zeros(n)
    for idx in np.where(active)[0]:
        s = idx - (n - 1)
        wt = 1.0 / na
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            g[lo1:hi1] += wt * dx * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            g[lo2:hi2] -= wt * dx * h[lo2 - s:hi2 - s]
    return h - lr * g

def _optimize(h0, dx, n, t_end, max_restarts=1):
    target = n * 0.5
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = h0.copy()
    eps_list = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002]
    for eps in eps_list:
        if time.time() > t_end - 30:
            break
        pen = 3.0
        try:
            h_new = _lfb(h, dx, n, eps, pen, maxiter=500)
            h_new = _proj_sum(np.clip(h_new, 0, 1), target)
        except Exception:
            break
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    lr = 0.15
    for it in range(4000):
        if time.time() > t_end - 10:
            break
        h2 = _subgrad_step(h, dx, n, lr)
        h2 = _proj_sum(h2, target)
        c2 = _c5(h2, dx)
        cc = _c5(h, dx)
        if c2 <= cc:
            h = h2
            lr = min(lr * 1.03, 1.2)
        else:
            lr = max(lr * 0.94, 0.02)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
    return best_h, best_c5

def _quick_optimize(h0, dx, n, t_end):
    target = n * 0.5
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = h0.copy()
    for eps in [0.05, 0.02, 0.01, 0.003, 0.001]:
        if time.time() > t_end - 8:
            break
        try:
            h_new = _lfb(h, dx, n, eps, 3.0, maxiter=300)
            h_new = _proj_sum(np.clip(h_new, 0, 1), target)
        except Exception:
            break
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target = n * 0.5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
    else:
        h0 = np.full(n, 0.5)
    h0 = _proj_sum(h0, target)

    t1 = t0 + (deadline - t0) * 0.40
    best_h, best_c5 = _optimize(h0, dx, n, t1)

    t2 = t0 + (deadline - t0) * 0.80
    attempts = 0
    while time.time() < t2 and attempts < 60:
        attempts += 1
        amp = rng.uniform(0.04, 0.20)
        h_pert = _proj_sum(best_h + rng.normal(0, amp, n), target)
        c_pert = _c5(h_pert, dx)
        if c_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c_pert
        sub_t = min(time.time() + min(20.0, t2 - time.time()), t2)
        if sub_t - time.time() > 4:
            h_new, c_new = _quick_optimize(h_pert, dx, n, sub_t)
            if c_new < best_c5:
                best_h = h_new
                best_c5 = c_new

    # Try n=100 if time allows
    if time.time() < deadline - 120:
        n2 = 100
        dx2 = 2.0 / n2
        target2 = n2 * 0.5
        h0_2 = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n), best_h)
        h0_2 = _proj_sum(h0_2, target2)
        t3 = min(deadline - 60, time.time() + (deadline - time.time()) * 0.5)
        best_h2, best_c5_2 = _optimize(h0_2, dx2, n2, t3)
        if best_c5_2 < best_c5:
            return best_h2, best_c5_2, n2

    # Final polish on best solution found
    if time.time() < deadline - 15:
        h_f, c_f = _optimize(best_h, dx, n, deadline)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    return best_h, best_c5, n
```
