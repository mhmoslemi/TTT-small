import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def _obj_slsqp(x, alpha, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -100.0 / alpha, 100.0 / alpha)
    return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx

def _eq_constraint(x, target):
    return np.sum(x) - target

def _eq_jacobian(x, target):
    return np.ones_like(x)

def _slsqp_opt(h0, n, alpha, target, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jacobian, 'args': (target,)}]
    try:
        res = minimize(_obj_slsqp, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(alpha, target),
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0

def _renorm(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)
