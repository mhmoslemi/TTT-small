import time
import cvxpy as cp
from scipy.optimize import minimize

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _alt_lp(h0, dx, n, max_iter=5000, t_end=None, damp=1.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [
            cp.sum(h_var) * dx == 1.0,
            h_var >= 0,
            h_var <= 1,
        ]
        for s in range(-(n-1), n):
            i_min = max(0, -s)
            i_max = min(n-1, n-1-s)
            coeffs = np.zeros(n)
            coeffs[i_min:i_max+1] = g[i_min+s:i_max+s+1] * dx
            cons.append(cp.sum(h_var * coeffs) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver='CLARABEL')
        except Exception:
            try:
                prob.solve(solver='ECOS')
            except Exception:
                break
        if h_var.value is None:
            break
        h_raw = np.clip(h_var.value, 0.0, 1.0)
        if damp < 1.0:
            h_raw = damp * h_raw + (1.0 - damp) * h
        h_new = _proj(h_raw, n / 2.0)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h)) < 1e-9:
            h = h_new
            break
        h = h_new
    return best_h, best_c5

def _sobj_cv(h, cv_list, dx):
    # Smooth max of cv_list
    mx = max(cv_list)
    e = 0.01
    return mx + e * np.log(sum(np.exp((c - mx) / e) for c in cv_list))

def _slsqp_polish(h0, dx, n, maxiter=200):
    target = n / 2.0
    def obj(h):
        cv = _corr(h, dx)
        mx = cv.max()
        e = 0.005
        return mx + e * np.log(np.sum(np.exp((cv - mx) / e)))
    def grad(h):
        cv = _corr(h, dx)
        mx = cv.max()
        e = 0.005
        w = np.exp((cv - mx) / e)
        w /= w.sum()
        n_pts = len(h)
        g = np.zeros(n_pts)
        j = np.arange(n_pts)
        for idx in range(2 * n_pts - 1):
            if w[idx] < 1e-14:
                continue
            wt = w[idx] * dx
            s = idx - (n_pts - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n_pts)
            if vm.any():
                g[vm] += wt * (1.0 - h[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n_pts)
            if vp.any():
                g[vp] -= wt * h[jp[vp]]
        return g
    def eq_fun(h):
        return h.sum() - target
    def eq_jac(h):
        return np.ones(len(h))
    
    res = minimize(obj, h0, jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, target)

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _shift_ranges(n):
    ranges = []
    for s in range(-(n - 1), n):
        i_min = max(0, -s)
        i_max = min(n - 1, n - 1 - s)
        ranges.append((s, i_min, i_max))
    return ranges

def _lp_solve(fixed_g, dx, n, ranges):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) * dx == 1.0, h_var >= 0, h_var <= 1]
    for s, i_min, i_max in ranges:
        if i_min > i_max:
            continue
        coeffs = np.zeros(n)
        coeffs[i_min:i_max + 1] = fixed_g[i_min + s:i_max + s + 1] * dx
        cons.append(cp.sum(h_var * coeffs) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in ['ECOS', 'CBC', 'CVXOPT', 'PDLP']:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                return np.clip(h_var.value, 0.0, 1.0), t_var.value
        except Exception:
            continue
    return None, np.inf

def _alt_lp(h0, dx, n, max_iter=5000, t_end=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    ranges = _shift_ranges(n)
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        g = 1.0 - h
        h_new, _ = _lp_solve(g, dx, n, ranges)
        if h_new is None:
            break
        h_new = _proj(h_new, n / 2.0)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h)) < 1e-9:
            h = h_new
            break
        h = h_new
    return best_h, best_c5

def _slsqp_polish(h0, dx, n, maxiter=200, eps=0.005):
    target = n / 2.0

    def obj(h):
        cv = _corr(h, dx)
        mx = cv.max()
        return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

    def grad(h):
        cv = _corr(h, dx)
        mx = cv.max()
        w = np.exp((cv - mx) / eps)
        w /= w.sum()
        nn = len(h)
        g = np.zeros(nn)
        j = np.arange(nn)
        for idx in range(2 * nn - 1):
            if w[idx] < 1e-14:
                continue
            wt = w[idx] * dx
            s = idx - (nn - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < nn)
            if vm.any():
                g[vm] += wt * (1.0 - h[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < nn)
            if vp.any():
                g[vp] -= wt * h[jp[vp]]
        return g

    def eq_fun(h):
        return h.sum() - target

    def eq_jac(h):
        return np.ones(len(h))

    res = minimize(obj, h0, jac=grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, target)

def _local_search(h, dx, n, max_rounds=50, t_end=None):
    best = h.copy()
    best_c5 = _c5(h, dx)
    target = n / 2.0
    for rnd in range(max_rounds):

                h_try = best.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                # compensate on another index
                diff = target - h_try.sum()
                if abs(diff) < 1e-10:
                    pass
                elif diff > 0:
                    j = (i + 1) % n
                    h_try[j] = np.clip(h_try[j] + diff, 0.0, 1.0)
                elif diff < 0:
                    j = (i + 1) % n
                    h_try[j] = np.clip(h_try[j] + diff, 0.0, 1.0)
                if h_try.sum() * (2.0 / n) != 1.0:
                    h_try = _proj(h_try, target)
                c5_try = _c5(h_try, dx)
                if c5_try < best_c5 - 1e-10:
                    best = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5
