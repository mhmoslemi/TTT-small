def _grad_active(h, top_k=3):

    n = len(h)
    corr = _corr(h)
    # Find top-k shifts
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)

            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:

            grad[j] += g
    grad /= top_k

def _project(h, target_sum):
    """Project to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) > 1e-10:
        # Adjust: add/subtract from elements not at bounds
        diff = target_sum - s
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        # Final fix
        diff2 = target_sum - np.sum(h)
        if abs(diff2) > 1e-12:
            h += diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

    diff = target - np.sum(h)
    if abs(diff) > 1e-10:

        if cnt > 0:
            h[free] += diff / cnt

        h = np.clip(h, 0.0, 1.0)

                        continue

                        best_c = c
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            h += diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)

def _shift_grad(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    else:
        t = -s
        if t < n:
            g[t:] += (1.0 - h[:n - t])
            g[:n - t] -= h[t:]
    return g

def _lse_gradient(h, alpha, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mmax = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_k]

        grad += (w[i] / wsum) * _shift_grad(h, s)
