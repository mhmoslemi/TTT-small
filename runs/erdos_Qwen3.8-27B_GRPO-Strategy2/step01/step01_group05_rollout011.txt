import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def _calc_c(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    lo, hi = -5.0, 6.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj(x, dx, eps, target):
    h = np.clip(x, 0.0, 1.0)
    c = _calc_c(h, dx)
    m = c.max()
    val = m + eps * np.log(np.sum(np.exp((c - m) / eps)))
    val += 100.0 * (h.sum() - target) ** 2
    return val

def _subgrad_iter(h, dx, target, lr, n_steps, rng, t_end):
    best_h = h.copy()
    best_c5 = _calc_c(best_h, dx).max()
    v = np.zeros_like(h)
    n = len(h)
    j = np.arange(n)
    for it in range(n_steps):
        if time.time() > t_end - 5:
            break
        c = _calc_c(h, dx)
        mx = c.max()
        active = np.where(c >= mx - 1e-9)[0]
        g = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g[vm] += dx * (1.0 - h[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= dx * h[jp[vp]]
        g /= max(len(active), 1)
        v = 0.9 * v - lr * g
        h_new = _proj_sum(h + v, target)
        c_new = _calc_c(h_new, dx).max()
        c_cur = _calc_c(h, dx).max()
        if c_new <= c_cur:
            h = h_new
            lr = min(lr * 1.003, 3.0)
        else:
            v *= 0.5
            lr = max(lr * 0.96, 0.005)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if it % 300 == 150:
            h = _proj_sum(best_h + rng.normal(0, 0.025, n), target)
            v *= 0.1
    return best_h, best_c5

def _optimize_block(h0, dx, target, rng, t_end):
    n = len(h0)
    best_h = _proj_sum(h0, target)
    best_c5 = _calc_c(best_h, dx).max()
    h = best_h.copy()
    epsilons = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for eps in epsilons:
        if time.time() > t_end - 30:
            break
        xopt = fmin_l_bfgs_b(
            _smooth_obj, h, args=(dx, eps, target),
            fgtol=1e-15, pgtol=1e-15, maxiter=400
        )
        h_new = _proj_sum(xopt, target)
        c = _calc_c(h_new, dx).max()
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    h_final, c_final = _subgrad_iter(h, dx, target, 0.4, 15000, rng, t_end)
    if c_final < best_c5:
        best_h, best_c5 = h_final, c_final

            c2 = _calc_c(h2, dx).max()

def _structured_perturb(h, rng, n):
    k = rng.randint(3, max(4, n // 5))
    idx = rng.choice(n, size=k, replace=False)
    delta = rng.uniform(-0.15, 0.15, k)
    h_new = h.copy()
    h_new[idx] += delta
    return _proj_sum(h_new, n / 2.0)

    rng = np.random.RandomState(seed)

    best_c5 = _calc_c(best_h, dx).max()
    return best_h, best_c5, n
