                best_c5, best_h = c_ls, h_ls.copy()

    # Phase 3: Finer resolution
    for nn in [170, 196, 220]:
        if rem() < 55:
            break
        tsn = nn / 2.0
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, nn)
        h = np.interp(xn, xo, best_h)
        h = project_feasible(h, tsn)
        for beta, mk in [(10,200),(80,150),(250,120)]:
            if rem() < 15:
                break
            h = slsqp_solve(h, nn, tsn, beta, mk, 10)
            c = compute_c5(h)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        if rem() > 25:
            h_ls, c_ls = local_moves((best_h.copy() if len(best_h)==nn else h), nn, tsn, rng, min(35, rem()-15))
            if c_ls < best_c5:
                best_c5, best_h = c_ls, h_ls.copy()

    # Final SA
    if rem() > 50:
        nn = len(best_h)
        tsn = nn / 2.0
        h_sa, c_sa = sa_search(best_h.copy(), nn, tsn, rng, min(rem()-10, 150))
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa.copy()

    # Final local moves + polish
    if rem() > 20:
        nn = len(best_h)
        tsn = nn / 2.0
        h_ls, c_ls = local_moves(best_h.copy(), nn, tsn, rng, min(rem()-10, 60))
        if c_ls < best_c5:
            best_c5, best_h = c_ls, h_ls.copy()

    if rem() > 12:
        nn = len(best_h)
        tsn = nn / 2.0
        hp = slsqp_solve(best_h.copy(), nn, tsn, 500, 200, 20)
        cp = compute_c5(hp)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
