import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-11:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(-h)[:int(np.ceil(abs(diff)))]
            add = min(diff / len(idx), 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argsort(h)[:int(np.ceil(abs(diff)))]
            sub = min(abs(diff) / len(idx), h[idx])
            h[idx] -= sub
    return h

def build_lp(h_star, n):
    dx = 2.0 / n
    m = 2 * n - 1
    A = np.zeros((m, n))
    b = np.zeros(m)
    for s in range(m):
        k = s - (n - 1)
        if k >= 0:
            for i in range(n - k):
                A[s, i] += (1.0 - h_star[i + k])
                A[s, i + k] += -h_star[i]
                b[s] += h_star[i] * h_star[i + k]
        else:
            m2 = -k
            for i in range(m2, n):
                A[s, i] += (1.0 - h_star[i - m2])
                A[s, i - m2] += -h_star[i]
                b[s] += h_star[i] * h_star[i - m2]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([A, np.full((m, 1), -1.0 / dx)])
    b_ub = -b
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds_lp

def solve_lp_step(h_star, n):
    c_obj, A_ub, b_ub, A_eq, b_eq, bounds_lp = build_lp(h_star, n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if not res.success:
        return None
    return np.asarray(res.x[:n])

def logsumexp_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

                continue
            h_b, _ = successive_linearize(h_a, n, max_iters=60)
            h_c, _ = slsqp_polish(h_b, n, maxiter=200)
            h_d, _ = successive_linearize(h_c, n, max_iters=60)
            try_solution(h_d, n)

    # Phase 3: Random restarts on best n
    if time.time() - t0 < budget_s - 40:
        n_best = best_n
        for _ in range(15):
            if time.time() - t0 > budget_s - 30:
                break
            if best_h is not None:
                h_r = best_h + rng.normal(0, 0.12, n_best)
            else:
                h_r = make_sawtooth(n_best, rng.uniform(2, 7), rng.uniform(0, 6), 0.4)
            h_r = repair(h_r, n_best)
            h_a, _ = successive_linearize(h_r, n_best, max_iters=50)
            h_b, _ = slsqp_polish(h_a, n_best, maxiter=150)
            h_c, _ = successive_linearize(h_b, n_best, max_iters=50)
            try_solution(h_c, n_best)

    # Phase 4: Final intensive polish
    if best_h is not None and time.time() - t0 < budget_s - 15:
        h_f, _ = slsqp_polish(best_h, best_n, maxiter=600)
        try_solution(h_f, best_n)
        h_g, _ = successive_linearize(h_f, best_n, max_iters=100)
        try_solution(h_g, best_n)
        h_j, _ = smooth_polish(h_g, best_n, maxiter=100)
        try_solution(h_j, best_n)
        h_k, _ = slsqp_polish(h_j, best_n, maxiter=500)
        try_solution(h_k, best_n)

    if best_h is None:
        best_n = 70
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
