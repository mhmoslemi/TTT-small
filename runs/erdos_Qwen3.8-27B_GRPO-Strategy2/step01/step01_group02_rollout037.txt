def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s == 0:
        return np.full(n, 0.5)
    if s > target:
        # Reduce: reduce values that are > 0 proportionally
        mask = h > 1e-10
        excess = s - target
        # Reduce proportionally but not below 0
        reduce_amount = np.minimum(h[mask], excess / mask.sum() * 1.1)
        h[mask] -= reduce_amount
        h = np.clip(h, 0, 1)
    else:
        # Increase: increase values that are < 1
        mask = h < 1.0 - 1e-10
        deficit = target - s
        if mask.sum() == 0:
            return np.full(n, 0.5)
        increase_amount = np.minimum(1.0 - h[mask], deficit / mask.sum() * 1.1)
        h[mask] += increase_amount
        h = np.clip(h, 0, 1)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def logsumexp_c5(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return ((m + np.log(np.sum(np.exp(beta * corr - beta * m)))) / beta) * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_corr = np.exp(beta * corr - beta * m)
    weights = exp_corr / np.sum(exp_corr)
    # gradient of (1/beta) * log(sum exp(beta*corr)) = sum weights * corr
    # d/dh[i] of corr[k] for each k
    # This is expensive to do exactly, so use finite differences or analytic
    # Actually, let's compute it analytically
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:

        shift = k - (n-1)
        if shift >= 0:
            for j in range(n - shift):
                grad[j+shift] += weights[k] * (1 - h[j])

                grad[j] += weights[k] * (1 - h[j-s])
                grad[j-s] -= weights[k] * h[j]

    m = np.max(corr)
    exp_c = np.exp(beta * (corr - m))

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return np.full(n, 0.5)
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            m2 = h < 1.0 - 1e-10
        else:
            m2 = h > 1e-10
        if m2.sum() > 0:
            h[m2] += diff2 / m2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_c = np.exp(beta * (corr - m))
    s = exp_c.sum()
    w = exp_c / s
    val = (m + np.log(s) / beta) * dx
    grad = np.zeros(n)

            sa = -shift
            grad[:n - sa] += wk * (1.0 - h[sa:])
