from scipy.optimize import minimize, differential_evolution, minimize as sp_minimize

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
    d2 = target - h.sum()
    if abs(d2) > 1e-14:
        # distribute remainder
        if d2 > 0:
            idx = np.argsort(1.0 - h)[-int(np.ceil(abs(d2))):-int(np.ceil(abs(d2)))-1][::-1]
        else:
            idx = np.argsort(h)[:int(np.ceil(abs(d2)))]
        h[idx[0]] = float(np.clip(h[idx[0]] + d2, 0.0, 1.0))
    return h

def slsqp_opt(h, n, dx, maxiter=500, ftol=1e-12):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return eval_c5(x, dx)
    
    def ceq(x):
        return x.sum() - target
    
    cons = [{'type': 'eq', 'fun': ceq, 'jac': lambda x: np.ones(n)}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

        if c < best_c5:
            best_h, best_c5 = h, c
    if time.time() < t0 + budget_s - 2:
        h, c = slsqp_opt(best_h, n, dx, maxiter=300, ftol=1e-15)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
