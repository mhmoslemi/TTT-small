                       constraints=[{'type': 'eq', 'fun': _eqcon, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h


def _coord_descend(h, n, dx, rng, t_end, seed_rng):
    c5 = _c5(h, dx)
    deltas = np.array([0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05])
    big = np.array([0.1, 0.15, -0.1, -0.15])
    stuck = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        improved = False
        dl = deltas if stuck < 30 else big
        for d in dl:
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                continue
            h[i] = np.clip(ni, 0, 1)
            h[j] = np.clip(nj, 0, 1)
            c5c = _c5(h, dx)
            if c5c < c5 - 1e-12:
                c5 = c5c
                improved = True
                stuck = 0
            else:
                h[i] = _repair(h, n)[i] if False else ni if ni >= 0 and ni <= 1 else h[i]
                h[j] = nj if nj >= 0 and nj <= 1 else h[j]
                # Undo
                h[i] = h[i] - (h[i] - (h[i] + d - d))  # just reset
                h[i] = min(max(ni, 0), 1) if c5c >= c5 - 1e-12 else h[i]
                h[j] = min(max(nj, 0), 1) if c5c >= c5 - 1e-12 else h[j]
                if c5c >= c5 - 1e-12:
                    h[i] = ni if ni >= 0 and ni <= 1 else h[i]
                    h[j] = nj if nj >= 0 and nj <= 1 else h[j]
        if not improved:
            stuck += 1
            if stuck > 40:
                h = _repair(h + 0.08 * rng.randn(n), n)
                c5 = _c5(h, dx)
                stuck = 0
    return h, c5


def _coord_descend_v2(h, n, dx, rng, t_end):
    c5 = _c5(h, dx)
    small = [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]
    med = [0.03, 0.05, 0.08, -0.03, -0.05, -0.08]
    stuck = 0
    while time.time() < t_end:
        i = rng.randint(n)
        js = [(i + 1) % n, (i - 1) % n, (i + n // 3) % n, (i + n // 2) % n, rng.randint(n)]
        improved = False
        for j in js:
            if j == i:
                continue
            dl = small if stuck < 50 else med
            for d in dl:
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i] = ni
                h[j] = nj
                c5c = _c5(h, dx)
                if c5c < c5 - 1e-11:
                    c5 = c5c
                    improved = True
                    stuck = 0
                else:
                    h[i] = ni - d
                    h[j] = nj + d
        if not improved:
            stuck += 1
            if stuck > 60:
                h = _repair(h + 0.1 * rng.randn(n), n)
                c5 = _c5(h, dx)
                stuck = 0
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 70

    # Phase 1: Explore configs (35% budget)
    t_p1 = t0 + budget_s * 0.35

    n70 = len(initial_h_values)
    configs = [(n70, _repair(np.array(initial_h_values, dtype=float).copy(), n70))]
    for n in [50, 60, 80, 100]:
        configs.append((n, _golden_start(n, rng)))
        configs.append((n, _cyclic_start(n, rng)))
    h_rand = rng.rand(70)
    configs.append((70, _repair(h_rand, 70)))

    for n, h0 in configs:
        if time.time() > t_p1:
            break
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        h = h0.copy()
        c5 = _c5(h, dx)
        for alpha in [30, 80, 200, 500]:
            if time.time() > t_p1:
                break
            h = _slsqp_optimize(h, n, dx, target, bounds, alpha, 150, t_p1)
            cc = _c5(h, dx)
            if cc < c5:
                c5 = cc
        h = _repair(h, n)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 2: Intensive local optimization of best (50% budget)
    t_p2 = t0 + budget_s * 0.85
    if best_h is not None and time.time() < t_p2:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        h = best_h.copy()

        # SLSQP refinement with perturbations
        for alpha in [100, 300, 800, 2000]:
            if time.time() > t_p2:
                break
            for _r in range(2):
                if time.time() > t_p2:
                    break
                hp = _repair(h + 0.04 * rng.randn(n), n)
                hp = _slsqp_optimize(hp, n, dx, target, bounds, alpha, 250, t_p2)
                cc = _c5(hp, dx)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = hp.copy()
                    h = hp

        # Extended coordinate descent
        if time.time() < t_p2:
            h = best_h.copy()
            h, c5 = _coord_descend_v2(h, n, dx, rng, t_p2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    # Phase 3: Final polish (15% budget)
    t_p3 = t0 + budget_s - 5
    if best_h is not None and time.time() < t_p3:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for alpha in [3000, 8000]:
            if time.time() > t_p3:
                break
            hp = _slsqp_optimize(best_h, n, dx, target, bounds, alpha, 800, t_p3)
            cc = _c5(hp, dx)
            if cc < best_c5:
                best_c5 = cc
                best_h = hp

        # Tiny coord descent
        if time.time() < t_p3:
            best_h, cc = _coord_descend_v2(best_h, n, dx, rng, t_p3)
            if cc < best_c5:
                best_c5 = cc

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
