import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def _smooth(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return (m + float(np.log(np.sum(np.exp(s)))) / alpha) * (2.0 / len(x))


def _slsqp_step(h, n, alpha, maxit, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    res = minimize(_smooth, h, args=(alpha,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxit, 'ftol': ftol})
    hc = _repair(res.x, n)
    return hc, _c5(hc, dx)

        p = int(np.argmax(corr))
