```python
        lag = L - (n-1)
        for j in range(n):
            # dcorr/dh[j] = (1-h[j-lag]) [if 0<=j-lag<n] - h[j+lag] [if 0<=j+lag<n]
            val = 0.0
            if 0 <= j - lag < n:
                val += 1.0 - h[j - lag]
            if 0 <= j + lag < n:
                val -= h[j + lag]
            grad[j] += w[L] * val
    grad *= dx
    
    obj = (m + np.log(sum_exp) / beta) * dx
    return obj, grad
```

This is O(n * (2n-1)) = O(n²) per gradient evaluation. For n=140, that's ~40000 ops. Fast!

But the inner loop in Python will be slow. Let me vectorize:

For a given lag L (offset = L - (n-1)):
dcorr/dh[j] for all j:
- Term 1: (1-h[j-offset]) for j where 0 <= j-offset < n, i.e., offset <= j <= offset+n-1, intersected with [0, n-1]
- Term 2: -h[j+offset] for j where 0 <= j+offset < n, i.e., -offset <= j <= n-1-offset, intersected with [0, n-1]

I can compute this vectorized for each lag:

```python
def smoothed_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    
    grad = np.zeros(n)
    for L in range(2*n-1):
        wL = w[L]
        if wL < 1e-14:
            continue
        offset = L - (n-1)  # this is the lag
        # Term 1: (1-h[j-offset]) for j in [max(0,offset), min(n-1, offset+n-1)]
        j1_start = max(0, offset)
        j1_end = min(n, n + offset)  # exclusive
        if j1_start < j1_end:
            grad[j1_start:j1_end] += wL * (1.0 - h[j1_start-offset:j1_end-offset])
        # Term 2: -h[j+offset] for j in [max(0,-offset), min(n-1, n-1-offset)]
        j2_start = max(0, -offset)
        j2_end = min(n, n - offset)
        if j2_start < j2_end:
            grad[j2_start:j2_end] -= wL * h[j2_start+offset:j2_end+offset]
    
    grad *= dx
    obj = (m + np.log(sum_exp) / beta) * dx
    return obj, grad
```

This is O(n) per lag, O(n²) total, all vectorized within each lag. For n=140, 279 lags * O(140) = ~40000 numpy operations total. Very fast!

```python
def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:

    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def make_binary(h, target_ones, rng):
    h = h.copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    ones = np.where(h == 1)[0]
```

```python
def smoothed_obj_and_grad(x, n, beta):
    """Returns (objective, gradient) for smoothed C5."""
    dx = 2.0 / n
    h = x  # assume already projected to [0,1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    
    grad = np.zeros(n)
    nm1 = n - 1
    for L in range(2 * n - 1):
        wL = w[L]
        if wL < 1e-12:
            continue
        offset = L - nm1
        # Term 1: (1-h[j-offset]) for valid j
        js = max(0, offset)
        je = min(n, n + offset)
        if js < je:
            grad[js:je] += wL * (1.0 - h[js-offset:je-offset])
        # Term 2: -h[j+offset] for valid j
        js2 = max(0, -offset)
        je2 = min(n, n - offset)
        if js2 < je2:
            grad[js2:je2] -= wL * h[js2+offset:je2+offset]
    
    grad *= dx
    obj = (m + np.log(sum_exp) / beta) * dx
    return obj, grad

def fractional_refine(h_init, n, beta, maxiter=500, tol=1e-12):
    """SLSQP refinement with analytical gradient."""
    target_sum = n / 2.0
    h0 = project_feasible(h_init.copy(), target_sum)
    
    def obj(x):
        return smoothed_obj_and_grad(x, n, beta)[0]
    
    def grad(x):
        return smoothed_obj_and_grad(x, n, beta)[1]
    
    def constraint_fun(x):
        return x.sum() - target_sum
    
    def constraint_grad(x):
        return np.ones(n)
    
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_grad}],
                   options={'maxiter': maxiter, 'ftol': tol})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, target_sum)
    return project_feasible(h0, target_sum)
```

Wait, I realize there might be an issue with the gradient. Let me double-check.

The smoothed objective is:
F(h) = (m + (1/β) ln(Σ_L exp(β(c_L - m)))) * dx

where c_L = correlate_L (the L-th element of np.correlate(h, 1-h, mode="full"))

∂F/∂h[j] = dx * (β * c_m - β * c_m) / ... 

Hmm wait, m = max_L c_L depends on h! This makes the gradient tricky.

Actually, let me use a simpler smooth max that doesn't have the max subtraction:
F(h) = (1/β) ln(Σ_L exp(β c_L)) * dx

This is a well-known smooth approximation of max(c_L) * dx, and it's always differentiable.

∂F/∂h[j] = (1/β) * (1/Σ exp) * Σ_L β exp(β c_L) * ∂c_L/∂h[j] * dx
= dx * Σ_L w_L * ∂c_L/∂h[j]

where w_L = exp(β c_L) / Σ exp(β c_L)

This is cleaner! No need to track the max separately.

```python
            continue
        xo = np.linspace(0, 2, n_parent)
        xn = np.linspace(0, 2, n_try)
        h_interp = np.interp(xn, xo, h_parent)
        h_b = make_binary(h_interp, n_try // 2, rng)
        t0t = time.time()
        h_try, c5_try = binary_sa(h_b, n_try, rng, 50, t0t)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Fractional refinement with increasing beta
    n_best = len(best_h)
    for beta in [10, 30, 80, 200, 500, 1200]:
        if time.time() - t_start > 0.92 * budget_s:
            break
        h_ref = fractional_refine(best_h, n_best, beta, maxiter=300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 5: One more SA round on best (now possibly fractional -> binary)
    if time.time() - t_start < 0.97 * budget_s:
        h_bin = make_binary(best_h, n_best // 2, rng)
        t0f = time.time()
        h_f, c5_f = binary_sa(h_bin, n_best, rng, min(20, budget_s - 5 - (time.time() - t_start)), t0f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            # Quick fractional polish
            h_ref = fractional_refine(best_h, n_best, 500, maxiter=200)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
