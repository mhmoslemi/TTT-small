_g_dx = 2.0 / 70.0
_g_target = 35.0

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip((corr - m) * _g_alpha, -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(sh))) / _g_alpha) * _g_dx

def _pen_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip((corr - m) * _g_alpha, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(sh))) / _g_alpha
    pen = 10000.0 * (np.sum(x) - _g_target) ** 2
    return sm * _g_dx + pen

def _eq_f(x):
    return np.sum(x) - _g_target

def _eq_j(x):
    return np.ones_like(x)

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

                    if c2 < best_c - 1e-13:

    #   For ell != 0, these are different indices.
    
    # Actually for the product h[i]*(1-h[i+ell]):
    # d/dh[j] = delta(i,j)*(1-h[i+ell]) + h[i]*(-delta(i+ell,j))
    #         = delta(i,j)*(1-h[j+ell]) - h[j-ell]*delta(i, j-ell)
    
    # Summing over i:
    # = (1-h[j+ell]) - h[j-ell] (with appropriate boundary conditions)
    
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        # h[j] as first factor: valid if 0 <= j+ell < n
        if 0 <= j + ell and j + ell < n:
            g += (1.0 - h[j + ell])
        # h[j] as second factor (in 1-h): valid if 0 <= j-ell < n
        if 0 <= j - ell and j - ell < n:
            g -= h[j - ell]
        grad[j] = g
    
    # Projected subgradient step
    h_new = h - lr * grad
    # Project to [0,1] and fix sum
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h_new)
    if abs(diff) > 1e-12:
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
```

Wait, there's an issue. When ell = 0, the gradient is:
d/dh[j] of h[j]*(1-h[j]) = 1 - 2*h[j]

My formula gives: (1-h[j]) - h[j] = 1 - 2*h[j]. ✓

For ell ≠ 0:
d/dh[j] of sum_i h[i]*(1-h[i+ell])
= (1-h[j+ell]) - h[j-ell] (with boundary handling)

This looks correct.

But the subgradient method might be too slow or oscillatory. Let me use it as one component of a hybrid approach.

Actually, let me step back and think about what's most likely to give a 0.0002 improvement (from 0.38085 to 0.38065 or similar). 

The most reliable approach is probably:
1. A very thorough local search with many small steps
2. Combined with occasional gradient-based improvements
3. Starting from the known good solution

Let me also consider: maybe the improvement isn't in the shape of h but in the resolution. What if the n=70 discretization is not optimal, and there's a better solution at n=50 or n=100?

Let me write a solution that:
1. Thoroughly optimizes at n=70 (the parent's resolution)
2. Also tries n=50 and n=100 via interpolation
3. Uses a combination of SLSQP, L-BFGS-B, subgradient, and local search
4. Is very time-aware

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 100.0
_g_dx = 2.0 / 70.0
_g_target = 35.0

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip((corr - m) * _g_alpha, -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(sh))) / _g_alpha) * _g_dx

def _pen_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip((corr - m) * _g_alpha, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(sh))) / _g_alpha
    pen = 10000.0 * (np.sum(x) - _g_target) ** 2
    return sm * _g_dx + pen

def _eq_f(x):
    return np.sum(x) - _g_target

def _eq_j(x):
    return np.ones_like(x)

def _cnorm(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _interp(parent, n):
    xp = np.linspace(0.0, 2.0, len(parent), endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return _cnorm(np.interp(x, xp, parent, period=2.0), n / 2.0)

def _subgrad(h, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    ell = k_max - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j + ell < n:
            g += (1.0 - h[j + ell])
        if 0 <= j - ell < n:
            g -= h[j - ell]
        grad[j] = g
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h_new)
    h_new += diff / n
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _local(h, seed_ls, max_time):
    t_s = time.time()
    n = len(h)
    bc = _c5(h)
    bh = h.copy()
    rng = np.random.RandomState(seed_ls)
    for ss in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        for _ in range(300):
            if time.time() - t_s > max_time:
                return bh, bc
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            d = ss * rng.uniform(0.5, 1.5)
            for sg in (1, -1):
                h2 = bh.copy()
                h2[i] += sg * d
                h2[j] -= sg * d
                if 0.0 <= h2[i] <= 1.0 and 0.0 <= h2[j] <= 1.0:
                    c2 = _c5(h2)
                    if c2 < bc - 1e-13:
                        bh = h2
                        bc = c2
                        break
    return bh, bc

def _sweep(h, max_time):
    """Systematic sweep: for each i, try moving to nearby j's"""
    t_s = time.time()
    n = len(h)
    bc = _c5(h)
    bh = h.copy()
    for round in range(10):
        if time.time() - t_s > max_time:
            break
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                for d in (0.01, -0.01, 0.03, -0.03):
                    if time.time() - t_s > max_time:
                        return bh, bc
                    h2 = bh.copy()
                    h2[i] += d
                    h2[j] -= d
                    if 0.0 <= h2[i] <= 1.0 and 0.0 <= h2[j] <= 1.0:
                        c2 = _c5(h2)
                        if c2 < bc - 1e-13:
                            bh = h2
                            bc = c2
                            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    global _g_alpha, _g_dx, _g_target
    t0 = time.time()
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    best_h = parent.copy()
    best_n = pn

    _g_dx = 2.0 / pn
    _g_target = pn / 2.0

    cons = [{'type': 'eq', 'fun': _eq_f, 'jac': _eq_j}]

        if budget_s - (time.time() - t0) < 10:
            break
        _g_alpha = al
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 150, 'ftol': 1e-13})
            h_new = _cnorm(res.x, pn / 2.0)
            c_new = _c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = pn
                print(f"  SLSQ al={al}: C5={c_new:.6f}")
            h_cur = h_new
        except Exception:
            pass
    
    # L-BFGS-B
    if budget_s - (time.time() - t0) > 15:
        for al in [1000, 10000]:
            if budget_s - (time.time() - t0) < 10:
                break
            _g_alpha = al
            try:

                             bounds=bounds,
                             options={'maxiter': 500, 'ftol': 1e-15})
                h_new = _cnorm(res.x, pn / 2.0)
                c_new = _c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
                    best_n = pn
