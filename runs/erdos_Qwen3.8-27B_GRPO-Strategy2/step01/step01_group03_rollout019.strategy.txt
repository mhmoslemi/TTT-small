def compute_c5(h):

    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode="full")
    return float(np.max(corr)*dx)

def project_feasible(h, ts):
    """Project solution to feasible set while preserving sum"""
    h = np.clip(h, 0.0, 1.0)
    diff = ts - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff/len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff/len(idx)
    
    if abs(ts - np.sum(h)) > 1e-10 and len(idx) > 0:
        # Further adjust one point to meet sum constraint
        if diff > 0:
            h[idx[0]] = np.clip(h[idx[0]] + diff, 0, 1)
        else:
            h[idx[0]] = np.clip(h[idx[0]] + diff, 0, 1)
    return project_feasible(h, ts)
