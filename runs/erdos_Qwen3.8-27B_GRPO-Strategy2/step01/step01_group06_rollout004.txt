import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
            h = np.clip(h, 0.0, 1.0)
            d2 = target - h.sum()
            if abs(d2) > 1e-14:
                idx = int(np.argmax(1.0 - h))
                h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
            h = np.clip(h, 0.0, 1.0)
            d2 = target - h.sum()
            if abs(d2) > 1e-14:
                idx = int(np.argmin(h))
                h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lse_gradient(h, dx, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ev = np.exp(T * (corr - cm))
    p = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        pk = p[k]
        if pk < 1e-15:
            continue
        mlo = max(0, -lag)
        mhi = min(n, n - lag)
        if mhi > mlo:
            grad[mlo:mhi] += pk * (1.0 - h[mlo + lag:mhi + lag])
        mlo2 = max(0, lag)
        mhi2 = min(n, n + lag)
        if mhi2 > mlo2:
            grad[mlo2:mhi2] -= pk * h[mlo2 - lag:mhi2 - lag]
    return dx * grad

def lse_optimize(h, n, dx, T_list, maxiter_per_T, lr_base):
    h = h.copy()
    target = float(n) / 2.0
    for T in T_list:
        # Nesterov accelerated projected gradient
        v = np.zeros(n)
        h_nest = h.copy()
        for it in range(maxiter_per_T):
            g = lse_gradient(h_nest, dx, n, T)
            # Nesterov momentum
            z = h - 0.5 * v  # or use standard momentum
            # Actually let's use standard projected gradient with momentum
            v = 0.9 * v + lr_base * g
            h_new = h - v
            h_new = project_feasible(h_new, n)
            h = h_new
    return h

def slsqp_opt(h, n, dx, maxiter=200, ftol=1e-14):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return eval_c5(x, dx)
    
    def ceq(x):
        return x.sum() - target
    
    cons = [{'type': 'eq', 'fun': ceq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c = eval_c5(h_bin, dx)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    n = 70
    dx = 2.0 / n
    best_h = project_feasible(np.array(initial_h_values, dtype=float), n)
    best_c5 = eval_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: LSE + momentum with T annealing
    T_list = [3, 5, 10, 20, 50, 100, 200, 400]
    h1 = pg_momentum(best_h, n, dx, T_list, 800, lr_base=0.4)
    c1 = eval_c5(h1, dx)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: SLSQP polish
    if elapsed() < budget_s - 60:
        h2, c2 = slsqp_opt(best_h, n, dx, maxiter=300)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Binary local search
    if elapsed() < budget_s - 30:
        hb, cb = binary_ls(best_h, n, dx, max_passes=80)
        if cb < best_c5:
            best_c5, best_h = cb, hb

    # Phase 4: SLSQP after binary
    if elapsed() < budget_s - 30:
        h4, c4 = slsqp_opt(best_h, n, dx, maxiter=200)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Random restarts
    attempt = 0
    while elapsed() < budget_s - 5:
        attempt += 1
        if attempt <= 8:
            noise_scale = 0.03 + 0.05 * (attempt % 4)
            h0 = best_h + noise_scale * np.random.randn(n)
        elif attempt <= 14:
            h0 = best_h + 0.1 * np.random.randn(n)
        else:
            h0 = np.where(np.random.rand(n) > 0.5, 1.0, 0.0)
        h0 = project_feasible(h0, n)
        h0 = pg_momentum(h0, n, dx, [10, 50, 200], 500, lr_base=0.3)
        h0, c0 = slsqp_opt(h0, n, dx, maxiter=150)
        hb0, cb0 = binary_ls(h0, n, dx, max_passes=40)
        if cb0 < best_c5:
            best_c5, best_h = cb0, hb0
        if c0 < best_c5:
            best_c5, best_h = c0, h0
        # Small SLSQP polish if improvement found
        if cb0 < best_c5 + 1e-8 or c0 < best_c5 + 1e-8:
            hp, cp = slsqp_opt(best_h, n, dx, maxiter=200)
            if cp < best_c5:
                best_c5, best_h = cp, hp

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
