def _compute_dcorr_vec(h, n):
    L = 2*n - 1
    dcorr = np.zeros((n, L))
    for i in range(n):
        # For shift k = m-(n-1), m = 0..L-1
        # dcorr[i, m] = (1-h[i-k]) * [0<=i-k<n] - h[i+k] * [0<=i+k<n]

        # Actually: for each m, k = m-(n-1)
        # i-k = i - m + n - 1
        # i+k = i + m - n + 1
        idx1 = i - np.arange(L) + (n-1)  # i - k = i - (m-(n-1)) = i + n - 1 - m
        idx2 = i + np.arange(L) - (n-1)  # i + k = i + m - n + 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        
        term1_arr = np.zeros(L)
        term1_arr[valid1] = 1.0 - h[idx1[valid1]]
        
        term2_arr = np.zeros(L)
        term2_arr[valid2] = h[idx2[valid2]]
        
        dcorr[i, :] = term1_arr - term2_arr
    return dcorr

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _compute_dcorr(h, n):
    L = 2 * n - 1
    mc = n - 1
    dcorr = np.zeros((n, L))
    for i in range(n):
        idx1 = i + mc - np.arange(L)  # i - k for each m
        idx2 = i + np.arange(L) - mc  # i + k for each m
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(L)
        t1[v1] = 1.0 - h[idx1[v1]]
        t2 = np.zeros(L)
        t2[v2] = h[idx2[v2]]
        dcorr[i] = t1 - t2
    return dcorr

    ]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _compute_dcorr(h, n):
    L = 2 * n - 1
    mc = n - 1
    dcorr = np.zeros((n, L))
    for i in range(n):
        idx1 = i + mc - np.arange(L)
        idx2 = i + np.arange(L) - mc
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(L)
        t1[v1] = 1.0 - h[idx1[v1]]
        t2 = np.zeros(L)
        t2[v2] = h[idx2[v2]]
        dcorr[i] = t1 - t2
    return dcorr

def _obj(x, p, dx, target):
    c = _corr(x) * dx
    val = np.sum(c ** p) ** (1.0 / p)
    s = x.sum() - target
    return val + 50.0 * s * s

def _fast_cd(h, dx, n, time_limit_s):
    t_start = time.time()
    best = _c5(h, dx)
    corr = _corr(h).astype(float)
    dcorr = _compute_dcorr(h, n)
    L = 2 * n - 1
    delta_groups = [
        [0.1, 0.05, 0.02, 0.01, 0.005, -0.1, -0.05, -0.02, -0.01, -0.005],
        [0.03, 0.015, 0.007, 0.003, 0.001, -0.03, -0.015, -0.007, -0.003, -0.001],
        [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001],
        [0.004, 0.002, 0.001, -0.004, -0.002, -0.001],
    ]
    gi = 0
    no_improve = 0
    while time.time() - t_start < time_limit_s:
        deltas = delta_groups[min(gi, len(delta_groups) - 1)]
        improved = False
        for i in range(n - 1):
            if time.time() - t_start > time_limit_s:
                break
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in deltas:
                    ni = hi + d
                    nj = hj - d
                    if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                        continue
                    nc = corr + d * (dcorr[i] - dcorr[j])
                    if np.max(nc) * dx < best - 1e-13:
                        h[i] = ni
                        h[j] = nj
                        corr = _corr(h)
                        best = np.max(corr) * dx
                        dcorr = _compute_dcorr(h, n)
                        improved = True
                        hi = ni
                        hj = nj
        if not improved:
            no_improve += 1
            if no_improve >= 2:
                gi += 1
                no_improve = 0
                if gi >= len(delta_groups):
                    break
        else:
            no_improve = 0
    h = _repair(h, n)
    best = _c5(h, dx)
    return h, best

    n = len(initial_h_values)

        0.5 + 0.35 * np.cos(2 * np.pi * x / n),

        np.linspace(0.1, 0.9, n),

    for alt in alts:

            best_h = alt_h

    p_sched = [4, 8, 16, 32, 64]
    bounds = [(0.0, 1.0)] * n

                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    elapsed = time.time() - t0
    cd_budget = min(budget_s - 80 - elapsed, 700)
    if cd_budget > 20:

            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

            pass

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
