def compute_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    tc = tau * c * dx
    tc -= tc.max()  # numerical stability
    weights = np.exp(tc)
    weights /= weights.sum()
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += weights[k] * (1.0 - 2.0 * h)
        elif s > 0:
            g = np.zeros(n)
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
            grad += weights[k] * g
        else:
            t = -s
            g = np.zeros(n)
            g[t:] += (1.0 - h[:n-t])
            g[:n-t] -= h[t:]
            grad += weights[k] * g
    
    return dx * grad
