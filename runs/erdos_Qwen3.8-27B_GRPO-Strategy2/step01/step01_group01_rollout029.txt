from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def project_feasible(v, n):
    target = n * 0.5
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, n, t):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    return cmax + np.log(np.sum(np.exp(c - cmax))) / t


def smooth_grad(h, n, t):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    p = np.exp(c - cmax)
    p /= p.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        m = k - (n - 1)
        pk = p[k] * dx
        if m >= 0:
            grad[:n - m] += pk * (1.0 - h[m:])
            grad[m:] -= pk * h[:n - m]
        else:
            m2 = -m
            grad[m2:] += pk * (1.0 - h[:n - m2])
            grad[:n - m2] -= pk * h[m2:]
    return grad


def projected_gradient(h, n, t, max_iter=2000, gap=0.05):
    h = h.copy()
    best_val = smooth_obj(h, n, t)
    alpha = 5.0 * n
    for it in range(max_iter):
        grad = smooth_grad(h, n, t)
        # Backtracking
        for _ in range(30):
            h_new = project_feasible(h - alpha * grad, n)
            val_new = smooth_obj(h_new, n, t)
            if val_new <= best_val - 0.01 * alpha * np.dot(grad, h_new - h):
                break
            alpha *= 0.5
        else:
            break
        if val_new < best_val - 1e-12:
            h = h_new
            best_val = val_new
        else:
            break
    return h, smooth_obj(h, n, t)


def coordinate_descent(h, n, max_sweeps=100):
    h = h.copy()
    target = n * 0.5
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            candidates = [0.0, 1.0, old * 0.5, 2.0 - old * 0.5]
            best_val = best + 1e-12
            best_cand = old
            for cand in [0.0, 1.0, max(0, old - 0.1), min(1, old + 0.1)]:
                h[i] = cand
                diff = target - h.sum()
                j = (i + 1) % n
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_cand = cand
            if best_cand != old:
                h[i] = best_cand
                diff = target - h.sum()
                h[(i + 1) % n] = np.clip(h[(i + 1) % n] + diff, 0.0, 1.0)
                best = best_val
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, best
