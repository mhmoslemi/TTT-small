                add = min(diff2, 1.0 - h[k])
                h[k] += add
                diff2 -= add
            else:
                rem = min(-diff2, h[k])
                h[k] -= rem
                diff2 += rem
            if abs(diff2) < 1e-14:
                break
    return h


def subgrad_c5(h, n, top_frac=0.3):
    corr = compute_corr(h, n)
    mx = corr.max()
    dx = 2.0 / n
    thresh = mx - top_frac / dx
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        return np.zeros(n)
    n1 = n - 1
    grad = np.zeros(n)
    for m in active:
        offset = m - n1
        g = np.zeros(n)
        if offset >= 0:
            if n - offset > 0:
                g[:n - offset] += (1.0 - h[offset:])
                g[offset:] -= h[:n - offset]
        else:
            if n + offset > 0:
                g[-offset:] += (1.0 - h[:n + offset])
                g[:n + offset] -= h[-offset:]
        grad += g
    return grad / len(active)


def ls_obj_wrapper(x, n, beta):
    corr = compute_corr(x, n)
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n)


def eq_constr(x, n):
    return np.sum(x) - n / 2.0


def make_obj(x, n):
    return compute_c5(x, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        h_best = np.array(init, dtype=float).copy()
        n = len(h_best)
    else:
        n = 70
        h_best = np.full(n, 0.5)

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constr, 'args': (n,)}]

    it = 0
    stall = 0

    while time.time() - t0 < budget_s - 30:
        it += 1
        frac = (time.time() - t0) / budget_s

        if stall < 4:
            sigma = 0.02
        elif stall < 10:
            sigma = 0.05
        elif stall < 25:
            sigma = 0.10
        else:
            sigma = 0.15 + 0.01 * min(stall - 25, 20)

        h_pert = repair(h_best + rng.normal(0, sigma, n), n)
        c_pert = compute_c5(h_pert, n)
        improved = False

        # Phase 1: Subgradient descent with line search
        if time.time() - t0 < budget_s - 60:
            grad = subgrad_c5(h_pert, n, 0.35)
            gn = np.linalg.norm(grad)
            if gn > 1e-10:
                for s in [0.03, 0.05, 0.08, 0.12, 0.18]:
                    h_try = repair(h_pert - s * grad / gn, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < c_pert:
                        h_pert, c_pert = h_try, c_try
                if c_pert < best_c5 - 1e-14:
                    h_best, best_c5 = h_pert, c_pert
                    improved = True
                elif c_pert < compute_c5(h_best, n):
                    h_best, best_c5 = h_pert, c_pert
                    improved = True

        # Phase 2: Targeted coordinate moves on top correlations
        if time.time() - t0 < budget_s - 40 and it % 2 == 1:
            corr = compute_corr(h_best, n)
            n1 = n - 1
            top_idx = np.argsort(corr)[-6:]
            epsilons = [0.04, 0.07, 0.10, 0.14, 0.20]
            for m_idx in top_idx:
                if improved:
                    corr = compute_corr(h_best, n)
                offset = m_idx - n1
                for trial in range(15):
                    i = int(rng.randint(n))
                    if offset == 0:
                        j = (i + 1 + int(rng.randint(n - 1))) % n
                    else:
                        j = i + offset
                        if j < 0 or j >= n:
                            j = int(rng.randint(n))
                    if j == i:
                        continue
                    for eps in epsilons:
                        hp = h_best.copy()
                        if hp[i] + eps <= 1.0 and hp[j] - eps >= 0.0:
                            hp[i] += eps
                            hp[j] -= eps
                            hp = repair(hp, n)
                            c = compute_c5(hp, n)
                            if c < best_c5 - 1e-14:
                                h_best, best_c5, improved = hp, c, True
                                break
                        if hp[i] - eps >= 0.0 and hp[j] + eps <= 1.0:
                            hp2 = h_best.copy()
                            hp2[i] -= eps
                            hp2[j] += eps
                            hp2 = repair(hp2, n)
                            c2 = compute_c5(hp2, n)
                            if c2 < best_c5 - 1e-14:
                                h_best, best_c5, improved = hp2, c2, True
                                break
                    if improved:
                        break
                if improved:
                    break

        # Phase 3: SLSQP with exact max objective
        if time.time() - t0 < budget_s - 50 and (improved or it % 4 == 0):
            try:
                h_start = h_pert if not improved else h_best
                r = minimize(make_obj, repair(h_start, n), args=(n,),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 200, 'ftol': 1e-13})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc, c, True
            except Exception:
                pass

        # Phase 4: Log-sum-exp smoothing (higher beta for sharper max)
        if time.time() - t0 < budget_s - 50 and it % 3 == 0:
            beta = 25.0 + 15.0 * (it % 5)
            try:
                r2 = minimize(ls_obj_wrapper, h_best, args=(n, beta),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 150, 'ftol': 1e-12})
                hc2 = repair(r2.x, n)
                c2 = compute_c5(hc2, n)
                if c2 < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc2, c2, True
            except Exception:
                pass

        # Phase 5: Symmetry exploitation - try complemented-reflected version
        if it % 15 == 0 and time.time() - t0 < budget_s - 80:
            h_sym = 1.0 - h_best[::-1].copy()
            h_sym = repair(h_sym, n)
            try:
                r5 = minimize(make_obj, h_sym, args=(n,),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 300, 'ftol': 1e-13})
                hc5 = repair(r5.x, n)
                c5 = compute_c5(hc5, n)
                if c5 < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc5, c5, True
            except Exception:
                pass

        # Phase 6: Diverse restart with structured patterns
        if it % 25 == 0 and time.time() - t0 < budget_s - 100:
            pat_type = int(rng.randint(4))
            if pat_type == 0:
                freq = rng.choice([2, 3, 4, 5, 6])
                h_r = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n)
            elif pat_type == 1:
                h_r = 0.5 + 0.25 * np.cos(4 * np.pi * np.arange(n) / n) + 0.15 * np.sin(6 * np.pi * np.arange(n) / n)
            elif pat_type == 2:
                h_r = np.linspace(0.65, 0.35, n)
            else:
                h_r = 0.5 + 0.3 * np.sin(np.pi * np.arange(n) / n * 3)
            h_r = repair(h_r, n)
            try:
                r6 = minimize(make_obj, h_r, args=(n,),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-13})
                hc6 = repair(r6.x, n)
                c6 = compute_c5(hc6, n)
                if c6 < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc6, c6, True
            except Exception:
                pass

        # Phase 7: Aggressive local search - small random flips
        if time.time() - t0 < budget_s - 25 and it % 5 == 0:
            for _ in range(30):
                idx1, idx2 = int(rng.randint(n)), int(rng.randint(n))
                if idx1 == idx2:
                    continue
                for amt in [0.05, 0.1, 0.15]:
                    hp = h_best.copy()
                    hp[idx1] += amt
                    hp[idx2] -= amt
                    if 0 <= hp[idx1] <= 1 and 0 <= hp[idx2] <= 1:
                        hp = repair(hp, n)
                        c = compute_c5(hp, n)
                        if c < best_c5 - 1e-14:
                            h_best, best_c5, improved = hp, c, True
                            break
                    if improved:
                        break
                if improved:
                    break

        stall = 0 if improved else stall + 1

    # Final polish
    if time.time() - t0 < budget_s - 5:
        try:
            rf = minimize(make_obj, h_best, args=(n,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-15})
            hf = repair(rf.x, n)
            cf = compute_c5(hf, n)
            if cf < best_c5:
                h_best, best_c5 = hf, cf
        except Exception:
            pass

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
