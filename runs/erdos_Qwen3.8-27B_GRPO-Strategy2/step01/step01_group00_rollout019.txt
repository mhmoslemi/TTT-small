def _coeff_for_coord(h, i, n):
    """Compute coefficient array for coordinate i."""
    center = n - 1
    coeff = np.zeros(2 * n - 1)
    for lag in range(-(n-1), n):
        c = 0.0
        j1 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        j2 = i + lag
        if 0 <= j2 < n:
            c -= h[j2]

    return coeff


def _opt1d(corr, coeff, old, lo=0.0, hi=1.0):

    c = b - gr * (b - a)
    d = a + gr * (b - a)

    """Vectorized coefficient computation for coordinate i."""
    lags = np.arange(-(n - 1), n)
    j1 = i - lags
    j2 = i + lags
    coeff = np.zeros(2 * n - 1)

    coeff[m2] -= h[j2[m2]]

def _opt1d(corr, coeff, old):
    """Minimize max(corr + (v - old) * coeff) over v in [0, 1]."""
    # Convex 1D: use golden section

    c = b - (b - a) * (1.0 - gr)  # Hmm let me just use standard golden section
    # Actually standard: c = a + (1-gr)*(b-a), d = a + gr*(b-a)
    phi = (1.0 - 5.0 ** 0.5) / 2.0  # 0.381966
    c = a + phi * (b - a)
    d = a + (1.0 - phi) * (b - a)
    fc = np.max(corr + (c - old) * coeff)
    fd = np.max(corr + (d - old) * coeff)
    for _ in range(60):

            b = d
            d = c
            fd = fc
            c = a + phi * (b - a)

            a = c
            c = d
            fc = fd
            d = a + (1.0 - phi) * (b - a)
            fd = np.max(corr + (d - old) * coeff)
    return (a + b) * 0.5

    corr = _corr_arr(h)

    else:
        order = list(rand_order)

        v = _opt1d(corr, coeff, old)
        if abs(v - old) > 1e-14:
            corr += (v - old) * coeff
            h[i] = v
    return corr


def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
