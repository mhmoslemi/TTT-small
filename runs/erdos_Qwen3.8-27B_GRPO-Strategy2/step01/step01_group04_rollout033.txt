def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = h.copy()
    target = n / 2.0
    # Iteratively adjust
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        # Distribute diff to elements not at boundary
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    # Final scale
    s = np.sum(h)
    if s > 1e-12 and abs(s - target) > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

            old = h[i]
            best_t = old
            best_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
            for t in np.concatenate([[0.0, 1.0], np.linspace(0, 1, 11)]):
                h[i] = t
                c = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c < best_c:
                    best_c = c
                    best_t = t
            h[i] = best_t
    return project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = project_box_sum(h0, n0 / 2.0)

    best_h = h0.copy()
    best_val = compute_c5(h0, dx0)
    best_n = n0

    taus_full = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004]

    def elapsed():
        return time.time() - t_start

    # Phase 1: Main optimization at n0
    p1 = min(budget_s * 0.35, 200)
    x, v = optimize_lsqp(h0, n0, dx0, taus_full, p1)
    x = coord_polish(x, n0, dx0, 2)
    v = compute_c5(x, dx0)
    if v < best_val:
        best_val, best_h = v, x.copy()

    # Phase 2: Perturbed restarts
    for ri in range(5):
        if elapsed() > budget_s * 0.55:
            break
        al = budget_s * 0.55 - elapsed()
        if al < 20:
            break
        sc = 0.06 + 0.04 * ri
        x_r = project_box_sum(best_h + sc * rng.standard_normal(n0), n0 / 2.0)
        x, v = optimize_lsqp(x_r, n0, dx0, taus_full[:5], al * 0.7)
        x = coord_polish(x, n0, dx0, 1)
        v = compute_c5(x, dx0)
        if v < best_val:
            best_val, best_h = v, x.copy()

    # Phase 3: Finer grid
    if elapsed() < budget_s * 0.75:
        n1 = min(2 * n0, 400)
        dx1 = 2.0 / n1
        xs = np.linspace(0, 2, n0)
        xd = np.linspace(0, 2, n1)
        h_interp = np.interp(xd, xs, best_h)
        h_interp = project_box_sum(h_interp, n1 / 2.0)
        p3 = min(budget_s * 0.18, budget_s - elapsed() - 60)
        if p3 > 30:
            x, v = optimize_lsqp(h_interp, n1, dx1, taus_full[:6], p3)
            x = coord_polish(x, n1, dx1, 1)
            v = compute_c5(x, dx1)
            if v < best_val:
                best_val, best_h, best_n = v, x.copy(), n1

    # Phase 4: Another structure - sinusoidal perturbation
    if elapsed() < budget_s * 0.85:
        rem = budget_s - elapsed() - 30
        if rem > 40:
            f = rng.integers(3, 12)
            x_sin = best_h + 0.15 * np.sin(2 * np.pi * f * np.linspace(0, 2, n0) / 2.0)
            x_sin = project_box_sum(x_sin, n0 / 2.0)
            x, v = optimize_lsqp(x_sin, n0, dx0, taus_full[:4], rem * 0.5)
            x = coord_polish(x, n0, dx0, 1)
            v = compute_c5(x, dx0)
            if v < best_val:
                best_val, best_h, best_n = v, x.copy(), n0

    # Phase 5: Final polish
    rem = budget_s - elapsed() - 10
    if rem > 20:
        dx_b = 2.0 / best_n
        x, v = optimize_lsqp(best_h, best_n, dx_b, [0.003, 0.001, 0.0005], rem * 0.6)
        x = coord_polish(x, best_n, dx_b, 3)
        v = compute_c5(x, dx_b)
        if v < best_val:
            best_val, best_h = v, x.copy()

    # Final projection and evaluation
    dx_f = 2.0 / best_n
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
