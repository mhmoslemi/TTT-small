import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, dx, alpha, mu, n):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    penalty = mu * (h.sum() - n/2.0)**2
    return val + penalty

def lse_slsqp(h0, n, dx, alpha, mu, maxiter=500):
    def obj(x):
        return lse_obj(x, dx, alpha, mu, n)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    h_out = project_feasible(res.x, n)
    return h_out, eval_c5(h_out, dx)

def hard_slsqp(h0, n, dx, maxiter=500):
    target = n / 2.0
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return [x.sum() - target]
    def ceq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq, 'jac': ceq_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, n)
        return h_out, eval_c5(h_out, dx)
    except:
        return h0, eval_c5(h0, dx)

The gradient of LSE(α * f) with respect to h[i]:
∂/∂h[i] [max(c) + log(sum_k exp(α*(c_k - max(c))))/α]
= (1/α) * sum_k [α * (c_k - max(c)) - log(sum_k exp(α*(c_k - max(c))))] / sum_k exp(α*(c_k - max(c))) * ∂c_k/∂h[i]
= sum_k exp(α*(c_k - max(c))) / sum_k exp(α*(c_k - max(c))) * ∂c_k/∂h[i]
= sum_k w_k * ∂c_k/∂h[i]

where w_k = exp(α*(c_k - max(c))) / sum_j exp(α*(c_j - max(c)))

And ∂c_k/∂h[i] where c_k = dx * sum_j h[j] * (1-h[j+L_k]):
- If i = j: ∂/∂h[i] = dx * (1 - h[i+L_k]) [if 0 ≤ i+L_k < n]
- If i = j+L_k: ∂/∂h[i] = -dx * h[i-L_k] [if 0 ≤ i-L_k < n]

So ∂c_k/∂h[i] = dx * [(1-h[i+L_k]) * 1_{0≤i+L_k<n} - h[i-L_k] * 1_{0≤i-L_k<n}]

This is O(1) per (k,i) pair, so the full gradient is O(n * 2n) = O(n²).

def lse_grad(x, n, dx, alpha):

    m = np.max(corr)
    if alpha * np.max(corr - m) > 500:

        active = np.where(np.abs(corr - m) < 1e-12)[0]
        grad = np.zeros(n)
        for ki in active:
            L = ki - (n - 1)
            for i in range(n):
                if 0 <= i + L < n:
                    grad[i] += dx * (1.0 - x[i + L])
                if 0 <= i - L < n:
                    grad[i] -= dx * x[i - L]
        grad /= len(active)
        return grad
    
    w = np.exp(alpha * (corr - m))
