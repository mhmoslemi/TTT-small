            while abs(delta) > 1e-7 and count < n:
                i = index_add[count]
                space = 1.0 - h[i]
                if space > 0:
                    add = min(delta, space)
                    h[i] += add
                    delta -= add
                    count += 1
                else:
                    # Decrease at some positions
                    space = h[i]
                    sub_val = min(-delta, space)
                    h[i] -= sub_val
                    delta += sub_val
                    count += 1
        else:  # delta < 0
            addable = True
            index_add = indices
            count = 0
            while abs(delta) > 1e-7 and count < n:
                i = index_add[count]
                space = h[i]
                sub_val = min(-delta, space)
                h[i] -= sub_val
                delta += sub_val
                count += 1

    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    n_parent = 70
    dx_parent = 2.0 / n_parent
    target_sum_parent = n_parent / 2.0

    # Use provided initial_h_values with n_parent samples
    if 'initial_h_values' not in globals():
        initial_h_values = np.random.rand(n_parent) * 0.5
        sum_h = np.sum(initial_h_values)
        if abs(sum_h - target_sum_parent) > 1e-7:
            alpha = 10.0
            res = minimize(lambda x, alpha=alpha: smooth_obj(x, alpha), 
                           x0=initial_h_values, 
                           method='SLSQP', 
                           bounds=[(0,1)]*n_parent,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_parent}],
                           options={'maxiter': 50})
            if 'res' in locals() and res.success:
                initial_h_values = repair(res.x, n_parent)
            else:
                initial_h_values = np.clip(initial_h_values, 0,1.0)
                # Adjust integral by adding/removing mass
                def adjust_integral(v):
                    s = np.sum(v)
                    if abs(s - target_sum_parent) < 1e-7:
                        return v
                    ratio = target_sum_parent / s
                    if abs(ratio - 1.0) < 1e-7:
                        return v
                    v1 = v * ratio
                    v1 =np.clip(v1, 0,1.0)
                    s1 = np.sum(v1)
                    delta = target_sum_parent - s1
                    if abs(delta) < 1e-7:
                        return v1
                    # Identify and adjust with small steps
                    indices = np.argsort(v1)
                    add_ends = delta > 0
                    for idx in indices:
                        if add_ends and delta>0 and 1-v1[idx] > 0:
                            add = min(delta, 1-v1[idx])
                            v1[idx] += add
                            delta -= add
                            if abs(delta) < 1e-7:
                                break
                        elif delta < 0 and v1[idx]>0:
                            sub = min(-delta, v1[idx])
                            v1[idx] -= sub
                            delta += sub
                            if abs(delta) < 1e-7:
                                break
                    return v1
                initial_h_values = adjust_integral(initial_h_values)
    
    initial_h_values = np.array(initial_h_values)
    n_start = len(initial_h_values)
    
    # Decide on resolution based on available time
    if time.time() - t_start > budget_s * 0.7:
        n_points = 100
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        # Transfer parent construction: downsample to n_points (2/3 -> 1)
        indices = np.linspace(0, n_start-1, n_points, endpoint=True)
        indices = indices.astype(int)
        initial_h_values = initial_h_values[indices]
    else:
        n_points = n_start
        dx = 2.0 / n_points
        target_sum = target_sum_parent  # Because initial was 70 points

    # Repair the initial condition
    h0 = repair(initial_h_values.copy(), n_points)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    global best_known_c5
    if 'best_known_c5' not in globals():
        best_known_c5 = best_c5 * 1.0
    
    # Successive SLSQP with increasing alpha
    iter_count = 0
    found_improvement = False

    for alpha in [20, 40, 80, 160, 320, 640]:
        if time.time() - t_start > budget_s:
            break
        if iter_count >= 5 or time.time() - t_start > budget_s - 150:
            break
            
        print(f"Phase1, alpha={alpha} (Iteration {iter_count})")
        
        bounds = [(0.0, 1.0)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        res = minimize(lambda x, a=alpha: smooth_obj(x, a), 
                        best_h, 
                        method='SLSQP', 
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
        
        if res.success:
            hc = repair(res.x, n_points)
            current_c5 = compute_c5(hc, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = hc.copy()
                found_improvement = True
                if best_c5 < best_known_c5:
                    best_known_c5 = current_c5
                    print(f"New best: C5 = {best_c5:.6f}")
                
                # Early termination if we beat the previous record
                if best_c5 < 0.3808:
                    print(f"Exceeded target C5 < 0.3808 ({best_c5:.6f}), halting search.")
                    return best_h, best_c5, n_points
                    
                # Reset iteration count after a big jump
                iter_count = 0
                # Also, use the found solution as a new starting point
                best_h = hc.copy()
        else:
            # Penalize failure
            h_pert = best_h.copy() + 0.1 * rng.randn(n_points)
            h_pert = repair(h_pert, n_points)
            current_c5 = compute_c5(h_pert, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_pert.copy()
                found_improvement = True
        
        iter_count += 1
        
        # Early termination if budget is short
        if time.time() - t_start > budget_s - 150:
            break

    # Perturbation-restart with exponential mixing
    sigma_schedule = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]
    handled_improvement = False
    pert_count = 0
    while time.time() - t_start < budget_s and not handled_improvement and pert_count < 10:
        pert_count += 1
        
        if rng.random() < 0.5:  # sometimes do repair first
            new_h = best_h.copy() + sigma_schedule[min(pert_count-1, len(sigma_schedule)-1)] * rng.randn(n_points)
            new_h = repair(new_h, n_points)
            if compute_c5(new_h, dx) < best_c5:
                best_c5 = compute_c5(new_h, dx)
                best_h = new_h.copy()
                found_improvement = True
        
        # Objective-based restart
        min_corr_diff = None
        def obj_restart(x, restart_alpha=300):
            c = smooth_obj(x, restart_alpha)
            if min_corr_diff is None:
                min_corr_diff = -np.inf
            return c, min_corr_diff
        
        # Check improvement over objective landscape
        if compute_c5(best_h, dx) > 0.4 and time.time() - t_start < budget_s - 120:
            # Restart only if stuck and improvement is possible
            print(f"Restoring improvement at {compute_c5(best_h, dx):.6f}")
            restart_point = best_h.copy()
            # Try a more random restart
            restart_point[:int(0.3 * n_points)] = 0.2 * rng.random(int(0.3 * n_points))
            restart_point[int(0.3 * n_points):] = 0.8 * rng.random(int(0.7 * n_points))
            restart_point = repair(restart_point, n_points)
            
            res = minimize(lambda x, a=300: smooth_obj(x, a), 
                            restart_point, 
                            method='SLSQP', 
                            bounds=bounds,
                            constraints=cons,
                            options={'maxiter': 50, 'ftol': 1e-9, 'disp': False})
                            
            if res.success:
                new_h = repair(res.x, n_points)
                if compute_c5(new_h, dx) < best_c5:
                    best_c5 = compute_c5(new_h, dx)
                    best_h = new_h.copy()
                    handled_improvement = True
                    print(f"Restart improved C5={best_c5:.6f}")
            else:
                # Last resort: perturbation
                new_h = best_h.copy()
                new_h[:] = repair(new_h, n_points)
                
                n_mod = int(0.02 * n_points)  # small number to perturb
                indices = rng.choice(n_points, n_mod, replace=False)
                new_h[indices] = np.abs(rng.randn(n_mod))
                # Force repair
                if compute_c5(new_h, dx) < best_c5:
                    best_c5 = compute_c5(new_h, dx)
                    best_h = new_h.copy()
                    handled_improvement = True

    # Intensive local search around best found
    if time.time() - t_start < budget_s - 60 and best_c5 > 0.37 and n_points < 150:
        print(f"Intensive local search for C5 = {best_c5:.6f}")
        deltas = [0.005, 0.01, 0.02, 0.05, 0.1, -0.005, -0.01, -0.02, -0.05, -0.1]
        best_local = best_c5
        
        # Ensure bounds against local maxima
        try:
            # Simulated annealing on a grid around best_h?
            # We'll try small coordinate swaps
            for cycle in range(100):
                if time.time() - t_start > budget_s - 60:
                    break
                i agencies = rng.randint(n_points)

                    if best_c5 < best_known_c5:

                        print(f"Local search: new best C5={best_known_c5:.6f}")
                        
                else:  # Rejection with higher temperature
                    # Record history and try again
                    pass
                    
        except Exception as e:
            print("Local search failed", e)
            
    # If we are above 50% time remaining, do not finalize
    if (time.time() - t_start) < budget_s * 0.5:
        final_phase = time.time() - t_start
        budget_remaining = budget_s - final_phase
        
        # Minimum relaxation time
        if budget_remaining > 30:

                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

                                   options={'maxiter': 40, 'ftol': 1e-12})
                
                if result.success:
                    best_h = repair(result.x, n_points)
                    best_c5 = compute_c5(best_h, dx)
                    
            except:
                # Standard repair
                best_h = repair(best_h, n_points)
    
    # Final cleaning
    best_h = repair(best_h, n_points)
    best_c5 = compute_c5(best_h, dx)
    
    # Record known best
    if best_c5 < best_known_c5:
        best_known_c5 = best_c5
    
    # For record keeping
    return best_h, best_c5, n_points
