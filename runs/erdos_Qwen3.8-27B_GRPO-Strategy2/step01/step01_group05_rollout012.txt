from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _sobj_cv(x, dx, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad_cv(x, dx, eps, n):
    cv = _corr(x, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _proj_binary(h, target):
    """Project to binary-like: keep top 'target' values as 1, rest as 0"""
    sorted_idx = np.argsort(h)[::-1]
    out = np.zeros(len(h))
    out[sorted_idx[:int(target)]] = 1.0
    return out

def _proj(h, target):
    """Project to [0,1] with sum=target"""
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _gen_binary_pattern(n, target, rng):
    """Generate a random binary pattern with exactly 'target' ones"""
    idx = rng.choice(n, size=int(target), replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _binary_local_search(h, dx, n, target, max_iters, t_end):
    """Local search on binary patterns: swap 1->0 and 0->1"""
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Try a random swap
        oi = ones[rng.randint(len(ones))] if len(ones) > 0 else None
        zi = zeros[rng.randint(len(zeros))] if len(zeros) > 0 else None
        if oi is None or zi is None:
            break
        h[oi] = 0.0
        h[zi] = 1.0
        new_ones = np.where(h == 1.0)[0]
        new_zeros = np.where(h == 0.0)[0]
        
        c_new = _c5(h, dx)
        if c_new < best_c5:
            best_h = h.copy()
            best_c5 = c_new
            ones = new_ones
            zeros = new_zeros
        else:

            h[oi] = 1.0
            h[zi] = 0.0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_N = 70
_DX = 2.0 / _N
_TGT = _N / 2.0

def _c5(h):
    return np.correlate(h, 1.0 - h, mode="full").max() * _DX

def _cvdx(h):
    return np.correlate(h, 1.0 - h, mode="full") * _DX

def _smax(h, eps):
    cv = _cvdx(h)
    m = cv.max()
    return m + eps * np.log(np.sum(np.exp((cv - m) / eps)))

def _sgrad(h, eps):
    cv = _cvdx(h)
    m = cv.max()
    w = np.exp((cv - m) / eps)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wt = w[k] * _DX
        s = k - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]
    return g

def _proj(h, tgt):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp(h0, eps, mxit, t_end):
    n = _N
    def f(x):
        return _smax(x, eps)
    def jf(x):
        return _sgrad(x, eps)
    def cf(x):
        return x.sum() - _TGT
    def cjf(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=jf, bounds=[(0.0, 1.0)] * n,
                       constraints={'type': 'eq', 'fun': cf, 'jac': cjf},
                       method='SLSQP', options={'maxiter': mxit, 'ftol': 1e-15})
        if time.time() > t_end:
            return _proj(h0, _TGT)
        return _proj(res.x, _TGT)
    except Exception:
        return _proj(h0, _TGT)

def _active_opt(h, tgt, t_end, lr0=0.05, tol=0.02, maxit=50000):
    n = _N
    best = h.copy()
    bc = _c5(h)
    h = h.copy()
    cur_c = bc
    lr = lr0
    j = np.arange(n)
    for it in range(maxit):
        if time.time() > t_end:
            break
        cv = _cvdx(h)
        mx = cv.max()
        act = np.where(cv >= mx - tol)[0]
        if len(act) > 12:
            act = act[np.argsort(cv[act])[::-1][:12]]
        g = np.zeros(n)
        for k in act:
            s = k - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g[vm] += _DX * (1.0 - h[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= _DX * h[jp[vp]]
        g /= max(len(act), 1)
        hn = _proj(h - lr * g, tgt)
        cn = _c5(hn)
        if cn <= cur_c:
            h = hn
            cur_c = cn
            lr = min(lr * 1.004, 0.5)
        else:
            lr = max(lr * 0.93, 0.0005)
        if cn < bc:
            bc = cn
            best = hn.copy()
    return best, bc

def _coord_refine(h, tgt, t_end):

                h = h2
