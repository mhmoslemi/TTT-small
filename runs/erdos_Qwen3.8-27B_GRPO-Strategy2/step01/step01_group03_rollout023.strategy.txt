    time_limit = budget
    current_time = 0
    
    # SA steps:
    best_sol = h_current.copy()
    best_score = compute_c5(best_sol)
    
    for cooling_cycle in range(int(5000)):

        # Temperature schedule
        temp *= cooling_rate

        new_score = compute_c5(new_h)
        cost_diff = new_score - current_score
        accept_prob = min(1.0, exp(-cost_diff/temp))
        
        if random() < accept_prob:
            h_current = new_h
            current_score = new_score
            if new_score < best_score:
                best_sol = new_h.copy()
                best_score = new_score
    return best_sol
