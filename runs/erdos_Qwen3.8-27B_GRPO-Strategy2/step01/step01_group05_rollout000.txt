        g = np.zeros(n)
        j = np.arange(n)
        for idx in active:
            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g[vm] += dx * (1.0 - h[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= dx * h[jp[vp]]
        g /= max(len(active), 1)
        step = _proj(h - lr * g, target)
        c_new = _c5(step, dx)
        c_old = _c5(h, dx)
        if c_new <= c_old + 1e-15:
            h = step
            lr = min(lr * 1.01, 0.8)
        else:
            lr = max(lr * 0.93, 0.002)
    return h

def _coord_descend(h, dx, n, target, t_end, rounds=5):
    h = h.copy()
    rng = np.random.RandomState(12345)
    for r in range(rounds):
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h
            old = h[i]
            x_try = max(0.0, min(1.0, old))
            cv_full = _corr_cv(h, dx)
            mx_old = cv_full.max()
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                h[i] = max(0.0, min(1.0, old + delta))
                h[i] = _proj(h, target)[i]
                c = _c5(h, dx)
                if c < mx_old - 1e-12:
                    mx_old = c
                    x_try = h[i]
            h[i] = x_try
            h = _proj(h, target)
    return h

def _optimize(h, dx, n, target, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    epsilons = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for eps in epsilons:
        if time.time() > t_end - 20:
            break
        h = _lbfgs(h, dx, n, target, eps, t_end, maxiter=200)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < t_end - 10:
        h = _subgrad(best_h, dx, n, target, t_end, iters=1500)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < t_end - 3:
        h = _coord_descend(best_h, dx, n, target, time.time() + (t_end - time.time()) * 0.5)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = _proj(h0, target)

    best_h = h0.copy()
    best_c = _c5(h0, dx)

    # Phase 1: Main optimization (40% of time)
    t1_end = t0 + (budget_s - 5) * 0.35
    best_h, best_c = _optimize(h0, dx, n, target, t1_end)

    # Phase 2: Diverse restarts (45% of time)
    t2_end = t0 + (budget_s - 5) * 0.82
    attempts = 0
    while time.time() < t2_end:
        attempts += 1
        pat = attempts % 5
        amp = rng.uniform(0.04, 0.15)
        if pat == 0:
            h_pert = best_h + rng.normal(0, amp, n)
        elif pat == 1:
            phase = rng.uniform(0, 2 * np.pi)
            freq = rng.uniform(1, 4) * np.pi / n
            h_pert = best_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif pat == 2:
            h_pert = best_h[::-1] + rng.normal(0, amp * 0.4, n)
            # adjust for integral
        elif pat == 3:
            h_pert = np.roll(best_h, rng.randint(1, max(2, n // 10)))
        else:
            h_pert = best_h.copy()
            for _ in range(rng.randint(5, 15)):
                idx = rng.randint(n)
                h_pert[idx] += rng.normal(0, amp)
        h_pert = _proj(h_pert, target)
        c_pert = _c5(h_pert, dx)
        if c_pert < best_c:
            best_h = h_pert.copy()
            best_c = c_pert
        sub_end = min(time.time() + min(45.0, t2_end - time.time()), t2_end)
        if sub_end - time.time() > 5:
            h2, c2 = _optimize(h_pert, dx, n, target, sub_end)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 3: Final fine-tuning (remaining time)
    if time.time() < t_end - 5:
        h3, c3 = _optimize(best_h, dx, n, target, t_end)
        if c3 < best_c:
            best_h, best_c = h3, c3

    return best_h, best_c, n
