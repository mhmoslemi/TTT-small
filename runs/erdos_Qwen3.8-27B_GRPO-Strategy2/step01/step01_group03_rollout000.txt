def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0.0, 1.0)
    return h

def smooth_obj(x, n, beta):
    corr = compute_corr(x)
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def weighted_grad(h, n, beta):
    corr = compute_corr(h)
    shifted = beta * (corr - corr.max())
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

            best_h = h.copy()

    if remaining() > 80:
        tl = min(remaining() * 0.2, 180)
        h, c5 = continuous_local_search(h, n, ts, 60, tl, rng)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Pair local search
    if remaining() > 60:
        tl = min(remaining() * 0.15, 120)
        h, c5 = pair_local_search(h, n, ts, 80, tl, rng)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Random restart with perturbation + re-optimize
    if remaining() > 120:
        for restart in range(4):
            if remaining() < 40:
                break
            h_r = best_h + rng.normal(0, 0.05, n)
            h_r = project_feasible(h_r, ts)
            h_r, c5_r = grad_descent(h_r, n, ts, 200.0, 0.005, 3000, min(remaining() * 0.1, 50))
            h_r = slsqp_polish(h_r, n, ts, 300.0, 300, min(remaining() * 0.05, 30))
            c5_r = compute_c5(h_r)
            if c5_r < best_c5 - 1e-14:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final local search + polish
    if remaining() > 30:
        h, c5 = continuous_local_search(best_h, n, ts, 40, min(remaining() * 0.4, 80), rng)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if remaining() > 20:
            best_h = slsqp_polish(best_h, n, ts, 600.0, 400, min(remaining() * 0.3, 60))
            c5_final = compute_c5(best_h)
            if c5_final < best_c5 - 1e-14:
                best_c5 = c5_final

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
