```python
    """Gradient of C[k] w.r.t. h, where k is the shift (can be negative)"""
    n = len(h)
    g = np.zeros(n)

        # So j in [k, n-1]
        g[k:] = 1.0 - h[:-k]  # h[j-k] for j in [k, n-1] is h[0:n-k]
```

```python
def project_box_sum(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    x = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = x.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        # Find free variables
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if not free.any():
            break
        # Distribute evenly among free variables
        x[free] += diff / free.sum()
        x = np.clip(x, 0.0, 1.0)
    # Final check
    x = np.clip(x, 0.0, 1.0)
    return x
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max(corr, tau):
    m = np.max(corr)
    sh = (corr - m) / tau
    sh = np.clip(sh, -50, 0)
    e = np.exp(sh)
    logsum = np.log(np.sum(e))
    return m + tau * logsum, e / np.sum(e)

def grad_soft_max(h, w, tau):
    """Gradient of soft-max objective w.r.t. h. w is softmax weights over corr values."""
    n = len(h)
    g = np.zeros(n)
    # For each shift m (0 to 2n-2), k = m - (n-1)
    # grad_m[j] = (1-h[j-k])*I[0<=j-k<n] - h[j+k]*I[0<=j+k<n]
    # g = sum_m w[m] * grad_m
    # Use convolution structure:
    # Term1[j] = sum_m w[m] * (1-h[j-(m-(n-1))])*I[0<=j-(m-(n-1))<n]
    #          = sum_{i: 0<=i<n} (1-h[i]) * w[j-i+(n-1)] (with w indexed 0..2n-2)
    # Term2[j] = sum_m w[m] * h[j+(m-(n-1))]*I[0<=j+(m-(n-1))<n]
    #          = sum_{i: 0<=i<n} h[i] * w[j+i-(n-1)] (with w indexed 0..2n-2)
    
    # Term1: (1-h) correlated with w
    # (1-h)[i] * w[j-i+n-1]: this is like sum_i (1-h)[i] * w[(n-1+j)-i]
    # = cross-correlation of (1-h) and w at lag (j+n-1)
    # In numpy: np.correlate(w, 1-h, 'full')[...]
    
    # Let me just use np.correlate directly
    # np.correlate(w, f, 'full')[k] = sum_i w[i] * f[i+k]
    # We want: sum_i f[i] * w[j+n-1-i] = sum_i w[j+n-1-i] * f[i]
    # Let l = j+n-1, then sum_i w[l-i] * f[i]
    # This is the cross-correlation (w ⊛ f)[l] = sum_i w[l-i] * f[i]
    # numpy's correlate: np.correlate(w, f, 'full')[k] = sum_i w[i+k] * f[i] (checking docs)
    
    # Actually numpy docs: For mode='full', the output is of length N+M-1 where N=len(a), M=len(v)
    # out[k] = sum_n a[n] * v[n+k-N+1]... no.
    
    # Let me just verify with a simple example and use the right formula.
    # For now, let me use the direct loop approach since n is small.
    
    for m in range(2*n-1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        # (1-h[j-k]) for j where 0 <= j-k < n
        js = max(0, k)
        je = min(n - 1, n - 1 + k)
        if js <= je:
            g[js:je+1] += wm * (1.0 - h[js-k:je+1-k])
        # -h[j+k] for j where 0 <= j+k < n
        js2 = max(0, -k)
        je2 = min(n - 1, n - 1 - k)
        if js2 <= je2:
            g[js2:je2+1] -= wm * h[js2+k:je2+1+k]
    return g

def project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    x = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        x[free] += diff / nf
        x = np.clip(x, 0.0, 1.0)
    return x

def pg_optimize(h0, n, dx, time_limit, seed, verbose=False):
    """Projected gradient descent with soft-max"""
    rng = np.random.default_rng(seed)
    t0 = time.time()
    h = h0.copy()
    target = n / 2.0
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    
    n_corr = 2 * n - 1
    step_size = 0.05  # initial
    
    phases = [
        (0.3, 500), (0.15, 500), (0.08, 500), (0.04, 500), (0.02, 500),
        (0.01, 1000), (0.005, 1000), (0.002, 1000), (0.001, 1000), (0.0005, 1000)
    ]
    
    for tau, n_iters in phases:
        if time.time() - t0 > time_limit - 1:
            break
        for it in range(n_iters):
            if time.time() - t0 > time_limit - 1:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            sh = np.clip((corr - m) / tau, -50, 0)
            e = np.exp(sh)
            w = e / e.sum()
            g = grad_soft_max(h, w, tau)
            
            # Step
            h_new = h - step_size * g * tau  # scale by tau to get gradient of the objective
            h_new = project_box_sum(h_new, target)
            
            val_new = np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()
            
            h = h_new
            # Adapt step size
            if it % 100 == 99:
                step_size *= 0.9
    
    return best_h, best_val
```

```python
def coord_descent(h, n, dx, time_limit, seed, n_sweeps=10):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    target = n / 2.0
    h = project_box_sum(h, target).copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    corr0 = np.correlate(h, 1.0 - h, mode="full")
    
    order = rng.permutation(n)
    
    for sweep in range(n_sweeps):
        for j in order:
            if time.time() - t0 > time_limit - 1:
                return best_h, best_val
            
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode="full")
            
            # Coefficients for h[j]:
            # c_k = h[j]*a_k + b_k
            a = np.zeros(2*n-1)
            for m in range(2*n-1):
                k = m - (n-1)
                aj = 0.0
                # h[j] appears as h[i] in term h[i](1-h[i+k]) when i=j: (1-h[j+k])
                if 0 <= j+k < n:
                    aj += (1.0 - h[j+k])
                # h[j] appears as h[i+k] in term h[i](1-h[i+k]) when i+k=j, i=j-k: (1-h[j-k])... wait no
                # Actually in c_k = sum_i h[i]*(1-h[i+k]), h[j] appears when:
                # i = j: contributes h[j]*(1-h[j+k]) → coefficient of h[j] is (1-h[j+k])
                # i+k = j, i = j-k: contributes h[j-k]*(1-h[j]) → coefficient of h[j] is -h[j-k]
                if 0 <= j-k < n:
                    aj -= h[j-k]

            b = corr - h[j] * a
            
            # Minimize max_m (a[m]*t + b[m]) over t in [0,1]
            # Evaluate on grid

                best_h = h.copy()
    
    return best_h, best_val
```

```python
        # Periodic full recompute to avoid drift
        corr = np.correlate(h, 1.0 - h, mode="full")
    
    return best_h, best_val
```

Hmm, the sum constraint handling is tricky. When I change h[j] by delta, the sum changes by delta. I need to compensate elsewhere, which changes the correlation further.

Simple approach: after updating h[j], just rescale all other coordinates slightly to maintain the sum. But that breaks the "incremental" update of corr.

Alternative: don't enforce the sum exactly during coordinate descent; just clip to [0,1] and rescale at the end of each sweep.

Actually, let me take a step back. The cleanest approach might be:

1. Use the soft-max gradient method (like before) but with better step size control and more iterations
2. Add coordinate descent as a polishing step
3. Use multiple starting points

Or even better: use the structure that the problem is a "minimax" problem and use a specific algorithm for that.

**Sion's minimax / iterative method:**
min_h max_k f(h,k) ↔ max_{p∈Δ} min_h Σ_k p_k f(h,k)

The inner problem (LP if we linearize) can be solved, then update p via multiplicative weights or gradient ascent.

But f(h,k) is bilinear, so the inner problem is a QP. Not as clean.

Let me just go with a practical approach:

1. Start from initial_h_values
2. Do projected gradient descent with soft-max (faster than SLSQP)
3. Do coordinate descent for polishing
4. Try a few different initializations
5. Maybe try a bigger grid for final answer

from scipy.interpolate import interp1d
import time

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _compute_c5(h, dx):
    return float(np.max(_compute_corr(h)) * dx)

def _project(h, target):
    x = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        x[free] += diff / nf
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _pg_step(h, tau, step, target):
    n = len(h)
    corr = _compute_corr(h)
    m = np.max(corr)
    sh = np.clip((corr - m) / tau, -50, 0)
    e = np.exp(sh)
    w = e / e.sum()
    
    # Gradient: sum_k w[k] * grad_k
    n_c = 2 * n - 1
    g = np.zeros(n)
    for mi in range(n_c):
        wi = w[mi]
        if wi < 1e-13:
            continue
        k = mi - (n - 1)
        js = max(0, k)
        je = min(n - 1, n - 1 + k)
        if js <= je:
            g[js:je+1] += wi * (1.0 - h[js-k:je+1-k])
        js2 = max(0, -k)
        je2 = min(n - 1, n - 1 - k)
        if js2 <= je2:
            g[js2:je2+1] -= wi * h[js2+k:je2+1+k]
    
    h_new = h - step * g
    return _project(h_new, target)

def _coord_sweep(h, n, dx, time_limit, sweep_order):
    t0 = time.time()
    target = n / 2.0
    n_c = 2 * n - 1
    corr = _compute_corr(h)
    for j in sweep_order:
        if time.time() - t0 > time_limit - 0.5:
            break
        hj = h[j]
        # Compute a[m] = ∂corr[m]/∂h[j]
        a = np.zeros(n_c)
        for mi in range(n_c):
            k = mi - (n - 1)
            ak = 0.0
            if 0 <= j + k < n:
                ak += 1.0 - h[j + k]
            if 0 <= j - k < n:
                ak -= h[j - k]
            a[mi] = ak
        b = corr - hj * a
        # Minimize max(a*t + b) over t in [0,1]
        ts = np.linspace(0, 1, 80)
        vals = np.max(a * ts[None, :] + b[:, None], axis=0)
        t_opt = ts[np.argmin(vals)]
        h[j] = t_opt
        corr += (t_opt - hj) * a
        h = np.clip(h, 0, 1)
    # Fix sum
    return h, corr, target
