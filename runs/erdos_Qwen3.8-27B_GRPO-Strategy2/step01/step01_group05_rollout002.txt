                c_try = _c5(h, dx)
                if c_try <= best_c5 or rng.rand() < np.exp(-(c_try - best_c5) / max(T, 1e-12)):
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h.copy()
                    improved = True
                else:
                    h[i] = 1
                    h[j] = 0
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                T *= cool_rate
                if T < 1e-8:
                    break
                stall = 0
    return best_h, best_c5

def _slsqp_smooth(h0, dx, n, target, eps, maxiter=400):
    cv_max0 = np.max(np.correlate(h0, 1.0 - h0, mode="full"))

    def sobj(x):
        cv_x = np.correlate(x, 1.0 - x, mode="full") * dx
        m = cv_x.max()
        return m + eps * np.log(np.sum(np.exp((cv_x - m) / eps)))

    def sgrad(x):
        cv_x = np.correlate(x, 1.0 - x, mode="full") * dx
        m = cv_x.max()
        w = np.exp((cv_x - m) / eps)
        w /= w.sum()
        g = np.zeros(n)
        j = np.arange(n)
        for idx in range(2 * n - 1):
            if w[idx] < 1e-14:
                continue
            wt = w[idx] * dx
            s = idx - (n - 1)
            jm = j - s
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                g[vm] += wt * (1.0 - x[jm[vm]])
            jp = j + s
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                g[vp] -= wt * x[jp[vp]]
        return g

    def efun(x):
        return x.sum() - target

    def ejac(x):
        return np.ones(len(x))

    res = minimize(sobj, h0, jac=sgrad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': efun, 'jac': ejac}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-16})
    return _proj(res.x, target)

def _adv_gradient(h, dx, n, top_k=3):
    cv = _corr(h, dx)
    g = np.zeros(n)
    j = np.arange(n)
    top = np.argsort(cv)[::-1][:top_k]
    for idx in top:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= top_k
    return g

def _adv_descend(h, dx, n, target, rng, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    lr = 0.3
    while time.time() < t_end:
        g = _adv_gradient(h, dx, n, top_k=3)
        h_new = _proj(h - lr * g, target)
        c_new = _c5(h_new, dx)
        c_cur = _c5(h, dx)
        if c_new <= c_cur + 1e-12:
            h = h_new
            lr = min(lr * 1.01, 3.0)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        else:
            lr = max(lr * 0.97, 0.02)
        if rng.rand() < 0.02:
            h = _proj(best_h + rng.normal(0, 0.04, n), target)
    return best_h, best_c5

def _optimize_n(n, h0, dx, target, rng, t_end):
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)

    # Phase 1: Binary local search
    t_p1 = min(time.time() + (t_end - time.time()) * 0.45, t_end - 30)
    h_bin = _proj_binary(h0, target)
    h_bin, c_bin = _binary_sa(h_bin, dx, n, rng, t_p1)
    if c_bin < best_c5:
        best_c5, best_h = c_bin, h_bin.copy()

    # Phase 2: SLSQP from best binary and from original
    t_p2 = min(time.time() + (t_end - time.time()) * 0.30, t_end - 15)
    starts = [h_bin.copy(), h0.copy()]
    if best_h is not h_bin and best_h is not h0:
        starts.append(best_h.copy())
    for eps in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > t_p2:
            break
        for st in starts[:2]:
            if time.time() > t_p2:
                break
            try:
                h_s = _slsqp_smooth(st, dx, n, target, eps, maxiter=350)
                c_s = _c5(h_s, dx)
                if c_s < best_c5:
                    best_c5, best_h = c_s, h_s.copy()
            except Exception:
                pass
        starts = [best_h.copy()]

    # Phase 3: Adversarial descent
    t_p3 = min(time.time() + (t_end - time.time()) * 0.50, t_end - 5)
    h_adv, c_adv = _adv_descend(best_h, dx, n, target, rng, t_p3)
    if c_adv < best_c5:
        best_c5, best_h = c_adv, h_adv

    # Phase 4: Final SLSQP polish
    if time.time() < t_end - 5:
        for eps in [0.001, 0.0005, 0.0002]:
            if time.time() > t_end:
                break
            try:
                h_f = _slsqp_smooth(best_h, dx, n, target, eps, maxiter=250)
                c_f = _c5(h_f, dx)
                if c_f < best_c5:
                    best_c5, best_h = c_f, h_f.copy()
            except Exception:
                pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial values
    h_init_global = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init_global = np.array(initial_h_values, dtype=float).copy()

    # Primary: n=70 with initial
    n_list = []
    if h_init_global is not None:
        n_list.append(len(h_init_global))
    for nd in [70, 72, 68, 74, 76, 80, 84]:
        if nd % 2 == 0 and nd not in n_list and 40 <= nd <= 120:
            n_list.append(nd)

    # Allocate time: primary n gets 60%, rest split
    primary_n = n_list[0]
    t_primary = t0 + (deadline - t0) * 0.55

    if h_init_global is not None and len(h_init_global) == primary_n:
        h0 = h_init_global.copy()
    else:
        h0 = rng.uniform(0.2, 0.8, primary_n)
    h0 = _proj(h0, primary_n / 2.0)
    dx_p = 2.0 / primary_n

    hp, cp = _optimize_n(primary_n, h0, dx_p, primary_n / 2.0, rng, t_primary)
    best_h, best_c5, best_n = hp, cp, primary_n

    # Secondary: try other n values with shorter time each
    remaining_ks = n_list[1:]
    if remaining_ks:
        time_per = max((deadline - primary_n * 0) - time.time(), 5) / len(remaining_ks)
        time_per = min(time_per, 120)
        for ni in remaining_ks:
            if time.time() > deadline - 20:
                break
            dx_i = 2.0 / ni
            tgt = ni / 2.0
            # Various starts
            rng2 = np.random.RandomState(seed + ni)
            h0i = rng2.uniform(0.2, 0.8, ni)
            h0i = _proj(h0i, tgt)
            # Also try interpolated best
            h0b = np.interp(np.linspace(0, 1, ni), np.linspace(0, 1, len(best_h)), best_h)
            h0b = _proj(h0b, tgt)
            t_sub = min(time.time() + time_per * 0.6, deadline - 20)
            hi1, ci1 = _optimize_n(ni, h0i, dx_i, tgt, rng2, t_sub)
            t_sub2 = min(time.time() + time_per * 0.6, deadline - 10)
            hi2, ci2 = _optimize_n(ni, h0b, dx_i, tgt, rng2, t_sub2)
            if ci1 < best_c5:
                best_h, best_c5, best_n = hi1, ci1, ni
            if ci2 < best_c5:
                best_h, best_c5, best_n = hi2, ci2, ni

    # Final polish on best
    if time.time() < deadline - 15:
        dx_b = 2.0 / best_n
        tgt_b = best_n / 2.0
        rng3 = np.random.RandomState(seed + 999)
        hf, cf = _optimize_n(best_n, best_h, dx_b, tgt_b, rng3, deadline)
        if cf < best_c5:
            best_h, best_c5, best_n = hf, cf, best_n

    return best_h, best_c5, best_n
