import numpy as np
from scipy.optimize import minimize
import time

_slsqp_n = 0
_slsqp_ts = 0.0
_slsqp_beta = 1.0

def slsqp_obj(x):
    n = _slsqp_n
    ts = _slsqp_ts
    beta = _slsqp_beta
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    pen = (x.sum() - ts) ** 2 * 1e6
    return (m + np.log(np.sum(np.exp((corr - m) * beta))) / beta) * (2.0 / n) + pen

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

def round_to_01(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    h01 = np.zeros(n)
    h01[order[:target]] = 1.0
    return h01

def make_random_01(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def sa_search(cur, n, deadline, rng, T0=0.015, cool_steps=4000, kick_size=3):
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T = T0
    it = 0
    no_improve = 0
    while time.time() < deadline:
        if no_improve > 400000 and it > 10000:
            # Kick: flip kick_size random pairs
            for _ in range(kick_size):
                ones = np.where(cur == 1.0)[0]
                zeros = np.where(cur == 0.0)[0]
                if len(ones) < 2 or len(zeros) < 2:
                    break
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                cur[i1] = 0; cur[j1] = 1
                cur[i2] = 0; cur[j2] = 1
            new_c5 = compute_c5(cur)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = cur.copy()
            cur_c5 = new_c5
            T = T0
            no_improve = 0
        else:
            ones = np.where(cur == 1.0)[0]
            zeros = np.where(cur == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            cur[i] = 0.0
            cur[j] = 1.0
            new_c5 = compute_c5(cur)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.rand() < np.exp(-min(delta / max(T, 1e-15), 200.0)):
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    best = cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                cur[i] = 1.0
                cur[j] = 0.0
                no_improve += 1
        it += 1
        if it % cool_steps == 0:
            T *= 0.996
        if T < 1e-12:
            T = T0 * 0.3
    return best, best_c5

def local_search_01(h01, n, deadline):
    cur = h01.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    while time.time() < deadline:
        improved = False
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        for i in ones:
            if time.time() > deadline:
                return best, best_c5
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = compute_c5(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                if c < cur_c5 - 1e-14:
                    cur[i] = 0.0
                    cur[j] = 1.0
                    cur_c5 = c
                    improved = True
                    ones = np.where(cur == 1.0)[0]
                    zeros = np.where(cur == 0.0)[0]
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best = cur.copy()
    return best, best_c5

def slsqp_refine(h, n, beta, maxiter=400):
    global _slsqp_n, _slsqp_ts, _slsqp_beta
    _slsqp_n = n
    _slsqp_ts = n / 2.0
    _slsqp_beta = beta
    res = minimize(slsqp_obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n / 2.0)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 2: Second SA with lower T (10%)

        xo = np.linspace(0, 2, n_init)

        # Also try random init

        dl5 = t0 + 0.82 * budget_s
