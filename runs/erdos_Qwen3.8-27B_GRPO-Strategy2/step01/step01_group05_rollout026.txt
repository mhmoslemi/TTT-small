        mx = cv.max()

            jp = j + s
            vp = (jp >= 0) & (jp < nn)

        return x.sum() - target

        return np.ones(len(x))

                   method='SLSQP',
                   options={'maxiter': maxiter, 'ftol': 1e-16})

import time

_G_N = 70
_G_EPS = 0.01
_G_TGT = 35.0

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

    return _corr_full(h).max() * dx

    for i in range(n):
        c = i
        # count bits mod 2
        c = bin(c).count('1') % 2
        h[i] = c
    return h

def _sa_binary(h, n, t_end, rng, n_iters=300000):

    h = h.astype(float).copy()
    ones = np.where(h == 1)[0].tolist()

    best_h = h.copy()

    cur_e = best_e
    T = 0.6
    for it in range(n_iters):
        if time.time() > t_end:
            break
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        h[a] = 0.0
        h[b] = 1.0

            ones[ia] = b
            zeros[ib] = a

                best_e = new_e
                best_h = h.copy()
        else:

            h[b] = 0.0

def _local_search(h, n, max_rounds=10):
    dx = 2.0 / n
    h = h.astype(float).copy()

    threshold = 0.5
    h = (h > threshold).astype(float)
    # Ensure exactly n/2 ones

    need = n // 2
    if n_ones < need:
        zeros = np.where(h == 0)[0]
        rng_flip = np.random.RandomState(0)
        idx = rng_flip.choice(len(zeros), need - int(n_ones), replace=False)
        h[zeros[idx]] = 1
    elif n_ones > need:
        ones = np.where(h == 1)[0]
        rng_flip = np.random.RandomState(0)
        idx = rng_flip.choice(len(ones), int(n_ones) - need, replace=False)
        h[ones[idx]] = 0

    for _ in range(max_rounds):
        improved = False
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]

def _set_globals(dx, n, eps, tgt):
    global _G_DX, _G_N, _G_EPS, _G_TGT
    _G_DX = dx
    _G_N = n
    _G_EPS = eps
    _G_TGT = tgt

def _sobj(x):
    cv = _corr_full(x) * _G_DX
    mx = cv.max()
    return mx + _G_EPS * np.log(np.sum(np.exp((cv - mx) / _G_EPS)))

def _sgrad(x):
    cv = _corr_full(x) * _G_DX
    mx = cv.max()
    w = np.exp((cv - mx) / _G_EPS)
    w /= w.sum()
    n = _G_N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:

        wt = w[idx] * _G_DX

def _efun(x):

def _ejac(x):

                   method='SLSQP',

        t_end_i = t0 + t_per_init * (i + 1)

        if c_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c_opt
    
    # Restart phase: perturb best and re-optimize
    t_restart_end = deadline - 60
    while time.time() < t_restart_end:
        amp = rng.uniform(0.05, 0.2)
        h_pert = _proj(best_h + rng.normal(0, amp, n), target)
        t_sub = min(time.time() + 30, t_restart_end)
        h_opt, c_opt = _optimize_from(h_pert, dx, n, target, rng, t_sub)
        if c_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c_opt

        for eps in [0.001, 0.0005, 0.0002]:
            if time.time() > deadline - 2:

            h_ref = _slsqp_refine(best_h, dx, n, target, eps, maxiter=200)
            c_ref = _c5(h_ref, dx)

import numpy as np
from scipy.optimize import minimize
import time

_G_DX = 2.0 / 70
_G_N = 70
_G_EPS = 0.01
_G_TGT = 35.0

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _c5(h, dx):
    return _corr_full(h).max() * dx

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def _fix_binary(h, n):
    h = (h > 0.5).astype(float)
    need = n // 2
    nc = int(h.sum())
    if nc < need:
        z = np.where(h == 0)[0]
        idx = np.random.RandomState(42).choice(len(z), need - nc, replace=False)
        h[z[idx]] = 1.0
    elif nc > need:
        o = np.where(h == 1)[0]
        idx = np.random.RandomState(42).choice(len(o), nc - need, replace=False)
        h[o[idx]] = 0.0
    return h

def _sa_binary(h, n, t_end, rng, n_iters=500000):
    dx = 2.0 / n
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    best_h = h.copy()
    best_e = _c5(h, dx)
    cur_e = best_e
    T = 0.8
    for it in range(n_iters):
        if time.time() > t_end:
            break
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        h[a] = 0.0
        h[b] = 1.0
        new_e = _c5(h, dx)
        dE = new_e - cur_e
        if dE < 0 or (T > 1e-8 and rng.random() < np.exp(-dE / T)):
            cur_e = new_e
            ones[ia] = b
            zeros[ib] = a
            if new_e < best_e:
                best_e = new_e
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        if it % 500 == 0:
            T = max(T * 0.99, 0.002)
    return best_h, best_e

def _local_search(h, n, max_rounds=15):
    dx = 2.0 / n
    h = _fix_binary(h, n).copy()
    best_e = _c5(h, dx)
    for _ in range(max_rounds):
        improved = False
        ol = np.where(h == 1)[0]
        zl = np.where(h == 0)[0]
        best_new_e = best_e
        best_swap = None
        for a in ol:
            for b in zl:
                h[a], h[b] = h[b], h[a]
                e = _c5(h, dx)
                h[a], h[b] = h[b], h[a]
                if e < best_new_e - 1e-14:
                    best_new_e = e
                    best_swap = (a, b)
        if best_swap:
            a, b = best_swap
            h[a], h[b] = h[b], h[a]
            best_e = best_new_e
            improved = True
        if not improved:
            break
    return h, best_e

def _set_globals(dx, n, eps, tgt):
    global _G_DX, _G_N, _G_EPS, _G_TGT
    _G_DX = dx
    _G_N = n
    _G_EPS = eps
    _G_TGT = tgt

def _sobj(x):
    cv = _corr_full(x) * _G_DX
    mx = cv.max()
    return mx + _G_EPS * np.log(np.sum(np.exp((cv - mx) / _G_EPS)))

def _sgrad(x):
    cv = _corr_full(x) * _G_DX
    mx = cv.max()
    w = np.exp((cv - mx) / _G_EPS)
    w /= w.sum()
    nn = _G_N
    g = np.zeros(nn)
    j = np.arange(nn)
    for idx in range(2 * nn - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _G_DX
        s = idx - (nn - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < nn)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < nn)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _efun(x):
    return x.sum() - _G_TGT

def _ejac(x):
    return np.ones(_G_N)

def _slsqp_refine(h0, dx, n, target, eps, maxiter):
    _set_globals(dx, n, eps, target)
    res = minimize(_sobj, h0, jac=_sgrad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                   method='SLSQP',
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, target)

def _optimize_from(h_init, dx, n, target, rng, t_end):

    best_c5 = _c5(h_init, dx)
    h_bin = _fix_binary(h_init, n)
    t_sa = min(time.time() + 45, t_end - 35)
    h_sa, e_sa = _sa_binary(h_bin, n, t_sa, rng)
    if e_sa < best_c5:

        best_h = h_sa.copy()

    if e_ls < best_c5:

        best_h = h_ls.copy()
    if time.time() < t_end - 8:
        for eps in [0.05, 0.01, 0.003, 0.001, 0.0004]:
            if time.time() > t_end - 3:
                break
            h_r = _slsqp_refine(best_h, dx, n, target, eps, 400)
            c_r = _c5(h_r, dx)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()
    return best_h, best_c5
