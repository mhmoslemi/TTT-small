import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(x, n):
    """Project onto {x: 0<=x<=1, sum(x)=n/2}"""
    tgt = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > tgt:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def autocorr(h, n):
    """R(s) for s=0..n-1"""
    R = np.empty(n)
    for s in range(n):
        R[s] = np.dot(h[:n-s], h[s:])
    return R

def softmin_f_grad(h, n, tau):
    """Return f and grad where f = -softmin(R) (to minimize)"""
    R = autocorr(h, n)
    x = -R / tau
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    f = tau * (x_max + np.log(exp_x.sum()))
    
    grad = -2.0 * w[0] * h
    for s in range(1, n):
        ws = w[s]
        if ws < 1e-14:
            break
        grad[:n-s] -= ws * h[s:]
        grad[s:] -= ws * h[:n-s]
    
    return f, grad

def nesterov_cd(h0, n, dx, tau, max_iter, time_limit, t_start, step_init=0.005, mom=0.9):
    h = h0.copy()
    v = np.zeros(n)
    step = step_init
    best_local = compute_c5(h, dx)
    best_local_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        # Nesterov lookahead
        h_nest = h - step * v
        h_nest = project_constraint(h_nest, n)
        _, g_nest = softmin_f_grad(h_nest, n, tau)
        
        # Update momentum
        v = mom * v + g_nest
        
        # Step
        h_new = h - step * v
        h_new = project_constraint(h_new, n)
        
        # Evaluate
        f_new, _ = softmin_f_grad(h_new, n, tau)
        f_old, _ = softmin_f_grad(h, n, tau)
        
        if f_new > f_old * (1 + 1e-10) + 1e-12:
            # Line search: halve step
            step /= 2
            h_new = h - step * v
            h_new = project_constraint(h_new, n)
            if step < 1e-10:
                step = 1e-10
                h_new = h
        
        h = h_new
        
        # Track best C5 every 50 iterations
        if it % 50 == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_local:
                best_local = c5
                best_local_h = h.copy()
    
    c5 = compute_c5(h, dx)
    if c5 < best_local:
        best_local = c5
        best_local_h = h.copy()
    
    return best_local_h, best_local

def slsqp_polish(h0, n, dx, tau, max_iter=200, time_limit_s=30):
    """Final SLSQP polish"""
    # Same as before but tailored
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        m = corr.max()
        sh = np.clip((corr - m) / tau, -50, 0)
        return (m + tau * np.log(np.sum(np.exp(sh)))) * dx
    
    def cfun(x):
        return np.sum(x) - n/2.0
    
    bd = [(0,1)]*n
    cs = [{'type':'eq', 'fun':cfun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bd, constraints=cs,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.success or res.fun < obj(h0):
            h = project_constraint(res.x, n)
            return h
    except:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(191, 0.5)
    n = len(h0)
    dx = 2.0 / n
    h0 = project_constraint(h0, n)
    
    best_h = h0.copy()
    best_val = compute_c5(h0, dx)
    
    # Phase 1: Main gradient descent with tau annealing
    taus = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.025, 0.02, 0.015, 0.012, 0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001]
    
    h = h0.copy()
    total_iters = 0
    for idx, tau in enumerate(taus):
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.75:
            break
        remaining = budget_s * 0.75 - elapsed
        # Allocate iterations proportional to remaining time
        iters = max(200, int(remaining * 200))  # ~200 iters/sec target
        h, val = nesterov_cd(h, n, dx, tau, iters, t_start + budget_s * 0.75, t_start)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    # Phase 2: A few restarts with perturbation
    for restart in range(5):
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.88:
            break
        scale = 0.05 + 0.03 * restart
        h_pert = project_constraint(best_h + scale * rng.standard_normal(n), n)
        h_r, val_r = nesterov_cd(h_pert, n, dx, 0.05, 2000, t_start + budget_s * 0.88, t_start)
        if val_r < best_val:
            best_val = val_r
            best_h = h_r.copy()
    
    # Phase 3: Final SLSQP polish
    elapsed = time.time() - t_start
    if elapsed < budget_s - 10:
        h_pol = slsqp_polish(best_h, n, dx, 0.005, 300, min(30, budget_s - elapsed - 5))
        val_pol = compute_c5(h_pol, dx)
        if val_pol < best_val:
            best_val = val_pol
            best_h = h_pol
    
    # Final evaluation
    best_h = project_constraint(best_h, n)
    final_val = compute_c5(best_h, dx)
    if final_val < best_val:
        best_val = final_val
