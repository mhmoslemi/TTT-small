    """Compute smooth C5 and its gradient using log-sum-exp."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # log-sum-exp for numerical stability
    max_c = np.max(corr)

def eval_c5(h, dx):

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_c5_grad(h, dx, s):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    sh = s * (corr - mc)
    es = np.exp(sh)
    ve = es.sum()
    smooth_val = mc + np.log(ve) / s
    p = es / ve
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if p[j] < 1e-14:
            continue
        lag = j - (n - 1)
        w = p[j] * dx
        if lag >= 0:
            grad[lag:n] += w * (1.0 - h[0:n - lag])
            grad[0:n - lag] -= w * h[lag:n]
        else:
            grad[0:n + lag] += w * (1.0 - h[-lag:n])
            grad[-lag:n] -= w * h[0:n + lag]
    return smooth_val, grad

def lsbfgsb_opt(h, n, dx, s, mu, maxiter):
    target = float(n) / 2.0
    ones_n = np.ones(n)

    def obj(x):
        val, g = smooth_c5_grad(x, dx, s)
        d = x.sum() - target
        return val + mu * d * d

    def grad(x):
        val, g = smooth_c5_grad(x, dx, s)
        d = x.sum() - target
        return g + 2.0 * mu * d * ones_n

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxfun': maxiter * 3})
        ho = project_feasible(res.x, n)
    except Exception:
        ho = h
    return ho, eval_c5(ho, dx)

def nelder_mead_polish(h, n, dx, maxiter=5000):
    target = float(n) / 2.0

    def obj(x):
        x = project_feasible(x, n)
        return eval_c5(x, dx)

    try:
        res = minimize(obj, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        ho = project_feasible(res.x, n)
    except Exception:
        ho = h
    return ho, eval_c5(ho, dx)
