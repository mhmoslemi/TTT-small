import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def smooth_max_c5(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_smooth(h0, n, dx, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    target = float(n) / 2.0

    def obj(x):
        return smooth_max_c5(np.clip(x, 0.0, 1.0), dx, alpha)

    def constr_eq(x):
        return float(x.sum() - target)

    cons = [{'type': 'eq', 'fun': constr_eq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, n)
        return h_out, eval_c5(h_out, dx)
    except Exception:
        return h0, eval_c5(h0, dx)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, n)

        return h0, eval_c5(h0, dx)
