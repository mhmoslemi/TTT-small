import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lse_val(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(tau * (corr - mc)))) / tau

def lse_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    w = np.exp(tau * (corr - mc))
    sw = w.sum()
    p = dx * w / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n - s]
        else:
            t = -s
            grad[t:] += pk * (1.0 - h[:n - t])
            grad[:n - t] -= pk * h[t:]
    return grad

def lbfgs_opt(h0, n, dx, tau, maxiter, t_deadline):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return lse_val(x, dx, tau) + 500.0 * (x.sum() - target) ** 2
    def jac(x):
        return lse_grad(x, dx, tau) + 1000.0 * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h0, eval_c5(h0, dx)

def binary_ls(h0, n, dx, max_sweeps=50, t_deadline=None):
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    c5 = eval_c5(h, dx)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for sweep in range(max_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        best_c5 = c5
        best_swap = None
        for ia in range(len(ones)):
            a = ones[ia]
            for ib in range(len(zeros)):
                b = zeros[ib]
                if a == b:
                    continue
                h[a] = 0.0
                h[b] = 1.0
                c5_new = eval_c5(h, dx)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_swap = (ia, ib)
                h[a] = 1.0
                h[b] = 0.0
        if best_swap is not None:
            ia, ib = best_swap
            a, b = ones[ia], zeros[ib]
            h[a] = 0.0
            h[b] = 1.0
            ones = [x for x in ones if x != a]
            zeros = [x for x in zeros if x != b]
            ones.append(b)
            zeros.append(a)
            c5 = best_c5
        else:
            break
    return h, c5

def make_structured(n, variant, rng):
    h = np.zeros(n)
    if variant == 0:
        for i in range(n):
            h[i] = 1.0 if np.sin(2.0 * np.pi * 7.0 * i / n) > 0 else 0.0
    elif variant == 1:
        for i in range(n):
            h[i] = 1.0 if np.sin(2.0 * np.pi * 5.0 * i / n + 0.3) > 0 else 0.0
    elif variant == 2:
        a = rng.randint(3, max(4, n // 3))
        for i in range(n):
            h[i] = 1.0 if (a * i) % n < n // 2 else 0.0
    elif variant == 3:
        h = (rng.rand(n) > 0.5).astype(float)
    elif variant == 4:
        for i in range(n):
            h[i] = 1.0 if (i * i) % (n // 2 + 1) < n // 4 else 0.0
    else:
        h = rng.rand(n)
    return project_feasible(h, n)

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_deadline = t0 + budget_s - 5.0

    best_h = None
    best_c5 = 1.0
    best_n = 70

    def update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Initialize from parent if available
    init_parent = None
    try:

            iv = np.array(initial_h_values, dtype=float)
            if len(iv) > 0:

    except Exception:
        pass

    n_list = [50, 60, 70, 80, 90]
    if init_parent is not None:
        n_list = [len(init_parent)] + [n for n in n_list if n != len(init_parent)]

    for n in n_list:
        if time.time() > t_deadline - 10:
            break
        dx = 2.0 / n
        starts = []
        if init_parent is not None and n == len(init_parent):
            starts.append(project_feasible(init_parent.copy(), n))
        for v in range(5):
            starts.append(make_structured(n, v, rng))
        starts.append(project_feasible(rng.rand(n), n))

        for h0 in starts:
            if time.time() > t_deadline - 10:
                break
            # Multi-tau LSE optimization
            h_cur = h0.copy()
            c5_cur = eval_c5(h_cur, dx)
            for tau in [50.0, 200.0, 800.0]:
                if time.time() > t_deadline - 10:

                if c5_new < c5_cur:

            update(h_cur, c5_cur, n)

            hb, c5b = binary_ls(h_cur, n, dx, max_sweeps=40, t_deadline=t_deadline)

            h_noisy = project_feasible(h_noisy, n)
            hb2, c5b2 = binary_ls(h_noisy, n, dx, max_sweeps=30, t_deadline=t_deadline)
            update(hb2, c5b2, n)

    # Final refinement on best solution found

        n_b = best_n
        dx_b = 2.0 / n_b
        h_ref = best_h.copy()
        c5_ref = best_c5
        # Gradient polish with very high tau

            if time.time() > t_deadline - 10:

        best_h = h_ref

    dx_final = 2.0 / best_n
    best_c5 = eval_c5(best_h, dx_final)
    return (best_h, best_c5, best_n)
