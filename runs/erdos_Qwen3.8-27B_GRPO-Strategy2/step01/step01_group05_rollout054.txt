def _active_qp(h, dx, target, n_active=5, n_inner_iters=200):
    """Active set QP: minimize weighted sum of top overlaps"""
    n = len(h)
    h = h.copy()
    for outer in range(n_inner_iters):
        cv = np.correlate(h, 1.0 - h, mode="full") * dx
        # Find top n_active shifts
        order = np.argsort(cv)[::-1]
        active = order[:n_active]
        
        # Build Q and c for the active set
        # overlap_k = dx * (∑ h[i+s] - ∑ h[i+s]*h[i])
        # = dx * c_k^T h - dx * h^T Q_k h (where Q_k is 0/1 matrix)
        
        Q = np.zeros((n, n))
        c = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            # ∑_{i: valid} h[i+s] → linear term
            # ∑_{i: valid} h[i+s]*h[i] → quadratic term
            for i in range(n):

                    # quadratic: h[i+s]*h[i]

import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _sobj(h, dx, eps):
    cv = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad(h, dx, eps):
    n = len(h)
    cv = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        ks = j - s
        v1 = (ks >= 0) & (ks < n)
        if v1.any():
            g[v1] += wt * (1.0 - h[ks[v1]])
        kp = j + s
        v2 = (kp >= 0) & (kp < n)
        if v2.any():
            g[v2] -= wt * h[kp[v2]]
    return g

def _smooth_opt(h0, dx, target, eps, maxiter=500):
    n = len(h0)
    best_h = h0.copy()
    best_c = _c5(h0, dx)
    
    def obj(h):
        return _sobj(h, dx, eps)
    
    def grad(h):
        return _sgrad(h, dx, eps)
    
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        h_new = _proj(res.x, target)
        c = _c5(h_new, dx)
        if c < best_c:
            best_c = c
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c

def _pgrad(h0, dx, target, eps, niter=5000):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.85
    lr = 0.5
    best_h = h.copy()
    best_c = _c5(h, dx)
    
    for it in range(niter):
        g = _sgrad(h, dx, eps)
        v = mu * v - lr * g
        h_new = _proj(h + v, target)
        c = _c5(h_new, dx)
        if c <= best_c + 1e-12:
            h = h_new
            lr = min(lr * 1.01, 3.0)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            lr *= 0.8
            v *= 0.5
    return best_h, best_c

def _coord_descent(h, dx, target, n_rounds=500):
    n = len(h)
    h = h.copy()
    best_c = _c5(h, dx)
    
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            # Try h[i] + ε and h[i] - ε (adjusting h[j] to maintain sum)
            # Find j to adjust (the one with most "room")

                    best_c = c
                    improved = True
                    break

            break
