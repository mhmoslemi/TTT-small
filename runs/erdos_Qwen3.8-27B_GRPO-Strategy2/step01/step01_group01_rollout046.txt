        target = n * 0.5
        h = best_h.copy()
        c5 = best_c5
        step = max(1, n // 300)
        for sweep in range(30):
            if tl() < 5:
                break
            improved = False
            for i in range(0, n, step):
                old = h[i]
                best_cand = old
                best_v = c5
                for cand in (0.0, 1.0):
                    if abs(cand - old) < 1e-10:
                        continue
                    h[i] = cand
                    diff = target - h.sum()
                    if abs(diff) > 1e-10:
                        fi = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
                        h[fi] = np.clip(h[fi] + diff, 0.0, 1.0)
                    v = compute_c5(h)
                    if v < best_v:
                        best_v = v
                        best_cand = cand
                        best_fix = fi
                    h[i] = old
                if best_cand != old:
                    h[i] = best_cand
                    diff = target - h.sum()
                    if abs(diff) > 1e-10:
                        fi = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
                        h[fi] = np.clip(h[fi] + diff, 0.0, 1.0)
                    c5 = best_v
                    improved = True
            if not improved:
                break
        if c5 < best_c5:
            best_h = repair(h, n)
            best_c5 = compute_c5(best_h)

    # Final check
    n_final = len(best_h)
    best_h = repair(best_h, n_final)
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n_final)
