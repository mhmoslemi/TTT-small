from scipy.interpolate import interp1d
import time

def project_feasible(x, n):
    tgt = n / 2.0
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        xp = np.minimum(np.maximum(x - mid, 0.0), 1.0)
        if xp.sum() > tgt:
            lo = mid
        else:
            hi = mid
    xf = np.minimum(np.maximum(x - (lo+hi)*0.5, 0.0), 1.0)
    diff = tgt - xf.sum()
    if abs(diff) > 1e-12:
        mask = (xf > 1e-9) & (xf < 1.0 - 1e-9)
        if mask.sum() > 0:
            xf[mask] += diff / mask.sum()
            xf = np.minimum(np.maximum(xf, 0.0), 1.0)

    w /= sw

    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def project_feasible(x, n):
    tgt = n / 2.0
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        xp = np.minimum(np.maximum(x - mid, 0.0), 1.0)
        if xp.sum() > tgt:
            lo = mid
        else:
            hi = mid
    xf = np.minimum(np.maximum(x - (lo + hi) * 0.5, 0.0), 1.0)
    diff = tgt - xf.sum()
    if abs(diff) > 1e-12:
        mask = (xf > 1e-9) & (xf < 1.0 - 1e-9)
        if mask.sum() > 0:
            xf[mask] += diff / mask.sum()
            xf = np.minimum(np.maximum(xf, 0.0), 1.0)
    return xf

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, n, tau):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    m = corr.max()
    sh = np.clip((corr - m) / tau, -40.0, 0.0)
    w = np.exp(sh)
    sw = w.sum()
    w /= sw
    obj = m + tau * np.log(sw)
    grad = np.zeros(n)
    thresh = 1e-12
    for s in range(-(n - 1), n):
        ws = w[s + (n - 1)]
        if ws < thresh:
            continue
        ps, pe = max(0, -s), min(n, n - s)
        if ps < pe:
            grad[ps:pe] += ws * (1.0 - h[ps + s:pe + s])
        ps2, pe2 = max(0, s), min(n, n + s)
        if ps2 < pe2:
            grad[ps2:pe2] -= ws * h[ps2 - s:pe2 - s]
    return obj, grad

    best_x = x.copy()

    stall = 0
    for tau in taus:
        for it in range(per_tau):
            if time.time() - t0 > time_limit:
                return best_x, best_val
            obj, grad = smooth_obj_and_grad(x, n, tau)
            converged = False
            for bt in range(25):
                x_tri = project_feasible(x - lr * grad, n)
                obj_tri, _ = smooth_obj_and_grad(x_tri, n, tau)
                if obj_tri < obj:
                    x = x_tri
                    converged = True
                    break
                lr *= 0.5
            if not converged:
                lr *= 0.5
                x = project_feasible(x - lr * grad, n)
            stall += 0 if converged else 1
            if stall > 500:
                lr *= 0.5
                stall = 0
            if it % 300 == 0:
                val = compute_c5(x, dx)
                if val < best_val:
                    best_val = val
                    best_x = x.copy()
        lr = 1.0
        stall = 0
    val = compute_c5(x, dx)
    if val < best_val:
        best_val = val

    return best_x, best_val

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src + 1)
    xd = np.linspace(0.0, 2.0, n_dst + 1)
    xp = interp1d(xs, np.concatenate([[x_src[0]], x_src, [x_src[-1]]]),
                  kind='linear', fill_value='extrapolate', assume_sorted=True)(xd)
    return project_feasible(xp, n_dst)
