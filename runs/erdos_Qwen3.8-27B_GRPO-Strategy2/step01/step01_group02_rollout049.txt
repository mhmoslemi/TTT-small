```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    # Iterative bisection for sum constraint
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Add diff to entries not at boundary
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h += diff / n
            h = np.clip(h, 0, 1)
        else:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = scaled.max()
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx
    return val

def smooth_grad(h, n, beta):
    """Compute gradient of the smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Weights for each lag
    scaled = beta * corr
    m = scaled.max()
    w = np.exp(scaled - m) / np.sum(np.exp(scaled - m))
    
    # Gradient of correlation at lag k w.r.t. h
    # corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])
    # For the full correlate, index j corresponds to lag k = j - (n-1)
    # corr[j] = sum_i h[i] * (1-h[i+k]) where k = j-(n-1)
    
    # grad_k[i] = (1-h[i+k]) - h[i-k] (with boundary handling)
    n_corr = 2*n - 1
    grad = np.zeros(n)
    
    for j in range(n_corr):
        k = j - (n-1)
        if w[j] < 1e-15:
            continue
        # corr[j] = sum_i h[i] * (1-h[i+k]) for valid i
        # d/dh[t] corr[j] = (1-h[t+k]) if t+k in [0,n-1] and t in [0,n-1]
        #                  - h[t-k] if t-k in [0,n-1] and t in [0,n-1]
        for t in range(n):
            g = 0.0
            if 0 <= t + k < n:
                g += (1.0 - h[t + k])
            if 0 <= t - k < n:
                g -= h[t - k]
            grad[t] += w[j] * g
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T = [35.0]
_ACTIVE = slice(None, None)

def con_fun(x):
    return x.sum() - _T[0]

def con_jac(x):
    return np.ones_like(x)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    s = beta * c
    m = s.max()
    return (m - np.logaddexp.reduce(s - m) + np.log(len(s))) / beta * dx

def exact_obj(x, n):
    return compute_c5(x, n)

def active_max_obj(x, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(c[_ACTIVE])) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(m.sum())
        if cnt > 0:
            h[m] += diff / cnt
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def find_active_lags(corr, threshold_frac=0.02):
    mx = corr.max()
    return np.where(corr >= mx * (1.0 - threshold_frac))[0]

def coord_descent(h, n, rng, iters=300):
    best = compute_c5(h, n)
    improved_any = False
    for _ in range(iters):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        for eps in (0.005, 0.01, 0.02, 0.05):
            for sgn in (1, -1):
                hi = h[i] + sgn * eps
                hj = h[j] - sgn * eps
                if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                    continue
                hp = h.copy()
                hp[i] = hi
                hp[j] = hj
                c = compute_c5(hp, n)
                if c < best - 1e-15:
                    h = hp
                    best = c
                    improved_any = True
                    break
    return h, best, improved_any
```
