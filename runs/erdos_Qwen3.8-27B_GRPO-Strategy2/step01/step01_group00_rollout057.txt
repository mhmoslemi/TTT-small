def delta_corr(h, i, j, n):
    """Compute the vector of delta_correlations for all lags when h[i]+=1, h[j]-=1"""
    # For each lag k, the change is:
    # δ * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
    # where out-of-range terms are 0
    
    lags = np.arange(-(n-1), n)  # all lags
    delta = np.zeros(len(lags))
    
    for k in lags:
        dc = 0.0
        if 0 <= i + k < n:
            dc += (1.0 - h[i + k])
        if 0 <= i - k < n:
            dc -= h[i - k]
        if 0 <= j + k < n:
            dc -= (1.0 - h[j + k])
        if 0 <= j - k < n:
            dc += h[j - k]
        delta[k + (n-1)] = dc
    
    return delta
```

def delta_corr_vec(h, i, j, n):
    """Vectorized: compute delta correlation for all lags when h[i]+=1, h[j]-=1"""
    # lags ranges from -(n-1) to (n-1)
    lags = np.arange(-(n-1), n)
    
    # (1-h[i+k]) for valid k: i+k in [0,n-1] → k in [-i, n-1-i]
    term1 = np.zeros(len(lags))
    mask1 = (i + lags >= 0) & (i + lags < n)
    term1[mask1] = 1.0 - h[i + lags[mask1]]
    
    # h[i-k] for valid k: i-k in [0,n-1] → k in [i-(n-1), i]
    term2 = np.zeros(len(lags))
    mask2 = (i - lags >= 0) & (i - lags < n)
    term2[mask2] = h[i - lags[mask2]]
    
    # (1-h[j+k]) for valid k
    term3 = np.zeros(len(lags))
    mask3 = (j + lags >= 0) & (j + lags < n)
    term3[mask3] = 1.0 - h[j + lags[mask3]]
    
    # h[j-k] for valid k
    term4 = np.zeros(len(lags))
    mask4 = (j - lags >= 0) & (j - lags < n)
    term4[mask4] = h[j - lags[mask4]]
    
    return term1 - term2 - term3 + term4
```

def repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:
        if diff > 0:

        else:
            order = np.argsort(h)

            if diff > 0:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    """Compute dc vector: change in corr when h[i]+=1, h[j]-=1"""
    lags = np.arange(-(n - 1), n)
    dc = np.zeros(len(lags))

    # +h[j-k] where 0 <= j-k < n
    idx = j - lags
    mask = (idx >= 0) & (idx < n)
    dc[mask] += h[idx[mask]]
    
    return dc

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_of(corr, dx):
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, t / n)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if d > 0:
                add = min(d, 1.0 - h[idx])
                h[idx] += add
                d -= add
            else:
                sub = min(-d, h[idx])
                h[idx] -= sub
                d += sub
            if abs(d) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr_full(x)
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _eq_fun(x, target):
    return np.sum(x) - target

def _delta_corr(h, i, j, n):
    lags = np.arange(-(n - 1), n)
    dc = np.zeros(len(lags))
    idx = i + lags
    m = (idx >= 0) & (idx < n)
    dc[m] += 1.0 - h[idx[m]]
    idx = i - lags
    m = (idx >= 0) & (idx < n)
    dc[m] -= h[idx[m]]
    idx = j + lags
    m = (idx >= 0) & (idx < n)
    dc[m] -= 1.0 - h[idx[m]]
    idx = j - lags
    m = (idx >= 0) & (idx < n)
    dc[m] += h[idx[m]]
    return dc

def _pair_step(h, corr, i, j, n):
    hi, hj = h[i], h[j]
    dc = _delta_corr(h, i, j, n)
    dmin = max(-hi, hj - 1.0) + 1e-10
    dmax = min(1.0 - hi, hj) - 1e-10
    if dmax <= dmin:
        return 0.0, dc
    nc = len(corr)
    tk = min(10, nc)
    top = np.argpartition(-corr, tk)[:tk]
    cands = [(dmin + dmax) * 0.001 + 0.999 * (dmin + dmax) * 0.5,
             (dmin + dmax) * 0.5,
             (dmin + dmax) * 0.999 + 0.001 * dmin + 0.001 * dmax]
    cands = [dmin + 1e-5, dmax - 1e-5]
    for a in range(tk):
        for b in range(a + 1, tk):
            ka, kb = int(top[a]), int(top[b])
            dd = dc[ka] - dc[kb]
            if abs(dd) > 1e-10:
                dv = (corr[kb] - corr[ka]) / dd
                if dmin < dv < dmax:
                    cands.append(dv)
    cands.extend(np.linspace(dmin, dmax, 8).tolist())
    cands = np.array(cands)
    cands = cands[(cands >= dmin) & (cands <= dmax)]
    if len(cands) == 0:
        return 0.0, dc
    new_c = corr[None, :] + cands[:, None] * dc[None, :]
    mx = np.max(new_c, axis=1)
    bi = int(np.argmin(mx))
    if mx[bi] < np.max(corr) - 1e-13:
        return cands[bi], dc
    return 0.0, dc
