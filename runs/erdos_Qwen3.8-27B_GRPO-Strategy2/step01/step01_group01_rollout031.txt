**Multi-cut Frank-Wolfe**: Maintain a set of active cuts (lags). At each iteration:
1. Add the current worst lag to the active set

3. The linear step for the average: the gradient is the sum of gradients over active lags
4. Solve the LP: min (sum of grads) ^T h' s.t. 0≤h'≤1, sum=en2

Actually, the average of linear functions is still linear, so the LP is still easy.

But wait, c_k(h) = sum_i h_i (1-h_{i+k}) is QUADRATIC, not linear. So the linearization at the current point is:
c_k(h + d) ≈ c_k(h) + ∇c_k(h)^T d

And the LP becomes: min_{h'} sum_{k∈active} c_k(h) + ∇c_k(h)^T (h' - h)
= const + (sum_{k∈active} ∇c_k(h))^T h'
= const + G^T h' where G = sum of gradients

So we minimize G^T h' subject to 0≤h'≤1, sum(h')=n/2. The solution is: set h'=1 where G is smallest, h'=0 where G is largest, and adjust one element for the sum constraint.

```python
def grad_smooth(h, t):
    c = corr_vec(h)
    cmax = c.max()
    weights = np.exp((c - cmax) * t)
    weights /= weights.sum()
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    # Vectorize: for each lag m, the contribution to grad is:
    # (1-h)[i + lag] - h[i - lag]
    # This is a shifted (1-h) minus a shifted h
    # Sum over all lags weighted by weights
    # = convolve(1-h, weights) at position i - convolve(h, weights) at position i (approximately)
    
    # Let's use np.convolve or correlate
    # weights is indexed 0 to 2n-2, corresponding to lags -(n-1) to (n-1)
    # We want: sum_m weights[m] * h[i + (m - center)]
    # This is (h * W_reversed)[i] where W is properly aligned
    
    # Actually, let's just do it with a loop but vectorized inner:
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-16:
            continue
        w = weights[m]
        lag = m - center
        if lag >= 0:
            # (1-h)[i+lag] for i = 0 to n-1-lag
            grad[:n-lag] += w * (1.0 - h[lag:])
            # -h[i-lag] for i = lag to n-1
            grad[lag:] -= w * h[:n-lag]
        else:
            ll = -lag
            # (1-h)[i+lag] = (1-h)[i-ll] for i = ll to n-1
            grad[ll:] += w * (1.0 - h[:n-ll])
            # -h[i-lag] = -h[i+ll] for i = 0 to n-1-ll
            grad[:n-ll] -= w * h[ll:]
    return grad * dx
```

    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        lag = m - center
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += wm * (1.0 - h[lag:])
                grad[lag:] -= wm * h[:n - lag]
        else:
            ll = -lag
            if n - ll > 0:
                grad[ll:] += wm * (1.0 - h[:n - ll])
                grad[:n - ll] -= wm * h[ll:]
    return grad * dx

def slsqp_opt(h, n, t, maxiter):
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: smooth_obj(x, t), h.copy(),
                   jac=lambda x: smooth_grad(x, t),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return repair(res.x, n)

def coord_descent(h, n, max_sweeps, n_cands):
    h = h.copy()
    best = compute_c5(h)
    target = n * 0.5
    cands = np.linspace(0.0, 1.0, n_cands)
    improved_sweeps = 0
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = h[i]
            best_val = best
            best_cand = old
            h[i] = old
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                h[i] = cand
                diff = target - h.sum()
                if abs(diff) > 1e-15:
                    if diff > 0:
                        mask = h < 1.0
                    else:
                        mask = h > 0.0
                    cnt = mask.sum()
                    if cnt > 0:
                        h[mask] += diff / cnt
                        h = np.clip(h, 0.0, 1.0)
                val = compute_c5(h)
                if val < best_val - 1e-14:
                    best_val = val
                    best_cand = cand
            h[i] = best_cand
            if best_cand != old:
                diff = target - h.sum()
                if abs(diff) > 1e-15:
                    if diff > 0:
                        mask = h < 1.0
                    else:
                        mask = h > 0.0
                    cnt = mask.sum()
                    if cnt > 0:
                        h[mask] += diff / cnt
                        h = np.clip(h, 0.0, 1.0)
                best = best_val
                improved = True
        if improved:
            improved_sweeps += 1
        else:
            break
    h = repair(h, n)
    return h, compute_c5(h)

def frank_wolfe(h, n, max_iter, k_active):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    target = n * 0.5
    center = n - 1
    for it in range(max_iter):
        a = 2.0 / (it + 2.0)
        c = corr_vec(h)
        top_k = np.argpartition(c, -k_active)[-k_active:]
        grad = np.zeros(n)
        for m in top_k:
            lag = m - center
            if lag >= 0:
                if n - lag > 0:
                    grad[:n - lag] += (1.0 - h[lag:])
                    grad[lag:] -= h[:n - lag]
            else:
                ll = -lag
                if n - ll > 0:
                    grad[ll:] += (1.0 - h[:n - ll])
                    grad[:n - ll] -= h[ll:]
        sorted_idx = np.argsort(grad)
        h_cw = np.zeros(n)
        h_cw[sorted_idx[:int(target)]] = 1.0
        h = (1.0 - a) * h + a * h_cw
        h = np.clip(h, 0.0, 1.0)
        h = repair(h, n)
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
            best_h = h.copy()
    return best_h, best

def binary_local(h, n, max_pairs, seed):
    if n % 2 != 0:
        return h, compute_c5(h)
    b = (h > 0.5).astype(float)
    n_ones = int(b.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(b == 1)[0]
        order = ones[np.argsort(h[ones])]
        b[order[:n_ones - target]] = 0.0
    elif n_ones < target:
        zeros = np.where(b == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        b[order[:target - n_ones]] = 1.0
    best = compute_c5(b)
    rng = np.random.RandomState(seed)
    for _ in range(max_pairs):
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        b[i] = 0.0
        b[j] = 1.0
        val = compute_c5(b)
        if val < best - 1e-14:
            best = val
        else:
            b[i] = 1.0
            b[j] = 0.0
    return b, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def time_ok(frac):
        return elapsed() < budget_s * frac

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP with temperature annealing
    for t_val in [800, 400, 200, 100, 50]:
        if not time_ok(0.12):
            break
        h_new = slsqp_opt(best_h, n, t_val, maxiter=150)
        v = compute_c5(h_new)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()

    # Phase 2: Fine coordinate descent
    if time_ok(0.30):
        h_new, v = coord_descent(best_h, n, max_sweeps=80, n_cands=20)
        if v < best_c5:
            best_c5, best_h = v, h_new

    # Phase 3: Frank-Wolfe
    if time_ok(0.40):
        h_new, v = frank_wolfe(best_h, n, max_iter=800, k_active=15)
        if v < best_c5:
            best_c5, best_h = v, h_new
        h_new, v = frank_wolfe(best_h, n, max_iter=500, k_active=5)
        if v < best_c5:
            best_c5, best_h = v, h_new

    # Phase 4: Binary local search
    if time_ok(0.50):
        h_new, v = binary_local(best_h, n, max_pairs=20000, seed=123)
        if v < best_c5:
            best_c5, best_h = v, h_new

    # Phase 5: SLSQP polish on best
    if time_ok(0.60):
        h_new = slsqp_opt(best_h, n, 100, maxiter=200)
        v = compute_c5(h_new)
        if v < best_c5:
            best_c5, best_h = v, h_new
        h_new, v = coord_descent(best_h, n, max_sweeps=50, n_cands=15)
        if v < best_c5:
            best_c5, best_h = v, h_new

    # Phase 6: Structured restarts
    if time_ok(0.78):
        for attempt in range(5):
            if not time_ok(0.78):
                break
            rng = np.random.RandomState(seed + 1000 + attempt)
            if attempt % 2 == 0:
                noise = rng.randn(n) * 0.05
                h_try = best_h + noise
            else:
                shift = rng.randint(1, n // 4)
                h_try = np.roll(best_h, shift)
                h_try = (h_try + best_h) / 2.0
            h_try = repair(h_try, n)
            h_new = slsqp_opt(h_try, n, 150, maxiter=100)
            v = compute_c5(h_new)
            if v < best_c5:
                best_c5, best_h = v, h_new
            h_new2, v2 = coord_descent(h_new, n, max_sweeps=30, n_cands=10)
            if v2 < best_c5:
                best_c5, best_h = v2, h_new2

    # Phase 7: Higher resolution n=752
    if time_ok(0.90) and n < 752:
        n2 = n * 2
        if n2 <= 800:
            xo = np.linspace(0.0, 2.0, n)
            xn = np.linspace(0.0, 2.0, n2)
            h_init = np.interp(xn, xo, best_h)
            h_init = repair(h_init, n2)
            h_new = slsqp_opt(h_init, n2, 200, maxiter=80)
            v = compute_c5(h_new)
            if v < best_c5:
                best_c5, best_h = v, h_new
            if time_ok(0.93):
                h_new, v = coord_descent(best_h, len(best_h), max_sweeps=20, n_cands=10)
                if v < best_c5:
                    best_c5, best_h = v, h_new

    # Phase 8: Final fine coordinate descent polish
    if time_ok(0.98):
        nf = len(best_h)
        h_new, v = coord_descent(best_h, nf, max_sweeps=40, n_cands=25)
        if v < best_c5:
            best_c5, best_h = v, h_new

    # Final repair and evaluation
    nf = len(best_h)
    best_h = repair(best_h, nf)
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, nf)
