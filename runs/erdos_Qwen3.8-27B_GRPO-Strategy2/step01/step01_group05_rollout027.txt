from scipy.optimize import minimize
import time

_N = 70
_EPS = 0.01
_TGT = 35.0

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _soft_max(vals, eps):
    mx = vals.max()
    if eps < 1e-15:
        return mx
    return mx + eps * np.log(np.sum(np.exp((vals - mx) / eps)))

def _soft_max_grad(vals, eps):
    mx = vals.max()
    w = np.exp((vals - mx) / eps)
    w /= w.sum()
    return w

def _sobj(x, dx, eps):
    cv = _corr(x, dx)
    return _soft_max(cv, eps)

def _sgrad(x, dx, eps):
    cv = _corr(x, dx)
    w = _soft_max_grad(cv, eps)
    n = len(x)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _proj(h, target):

        if np.clip(h - mid, 0.0, 1.0).sum() > target:

        else:
            hi = mid

def _efun(x, tgt):
    return x.sum() - tgt

def _ejac(x):
    return np.ones(len(x))

_G_N = 0
_G_EPS = 0.01
_G_TGT = 0.0

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _sobj(x):
    cv = _corr(x, _G_DX)
    mx = cv.max()
    return mx + _G_EPS * np.log(np.sum(np.exp((cv - mx) / _G_EPS)))

def _sgrad(x):
    cv = _corr(x, _G_DX)
    mx = cv.max()
    w = np.exp((cv - mx) / _G_EPS)
    w /= w.sum()
    n = _G_N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _G_DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():

        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return _proj(res.x, target)

        return _proj(h0, target)

    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - tol)[0]

    g = np.zeros(n)
    j = np.arange(n)

        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    if len(active) > 0:
        g /= len(active)
    return _proj(h - lr * g, target)

def _coord_descent(h, dx, target, rng, n_iter=2000, t_end=None):
    """Pair-swap coordinate descent."""
    n = len(h)
    best_c5 = _c5(h, dx)
    step = 0.05
    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)

            continue
        for delta in [step, -step, step*0.5, -step*0.5]:
            h_trial = h.copy()
            h_trial[i] += delta
            h_trial[j] -= delta
            if h_trial[i] < -0.01 or h_trial[i] > 1.01 or h_trial[j] < -0.01 or h_trial[j] > 1.01:

            c = _c5(h_trial, dx)
            if c < best_c5:

                best_c5 = c

    return h, best_c5

    return np.correlate(h, 1.0 - h, mode="full") * dx
