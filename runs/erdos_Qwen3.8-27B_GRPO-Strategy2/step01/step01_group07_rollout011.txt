            h = np.clip(h, 0.0, 1.0)

    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break

        for i, j in neighbor_pairs:

                if a[i] >= d and a[j] + d <= 1.0:

                    m3 = (j + s >= 0) & (j + s < n)

                b[i] -= di
                b[j] -= dj
                corr += dc
                cur_max = best_new_max
                improved = True
        if not improved:
            break
    return a, cur_max * (2.0 / n)

import numpy as np
import time

def _c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / len(h))

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-14:
        h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _compute_dc(a, b, corr, shifts, i, j, di, dj):
    """Compute delta_corr for a[i]+=di, a[j]+=dj (so b[i]-=di, b[j]-=dj)."""
    n = len(a)
    s = shifts
    dc = np.zeros(2 * n - 1)
    m1 = (i + s >= 0) & (i + s < n)
    m2 = (i - s >= 0) & (i - s < n)
    m3 = (j + s >= 0) & (j + s < n)
    m4 = (j - s >= 0) & (j - s < n)
    dc[m1] += di * b[i + s[m1]]
    dc[m2] += -di * a[i - s[m2]]
    dc[m3] += dj * b[j + s[m3]]
    dc[m4] += -dj * a[j - s[m4]]
    return dc

def _fast_cd(a, target, max_sweeps=300, t_end=None, max_neighbor=6, steps=None):
    n = len(a)
    a = a.copy()
    b = 1.0 - a
    corr = np.correlate(a, b, mode='full')
    cur_max = float(np.max(corr))
    shifts = np.arange(2 * n - 1) - (n - 1)
    if steps is None:
        steps = [0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
    pairs = [(i, j) for i in range(n) for j in range(i + 1, min(i + max_neighbor + 1, n))]
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        for i, j in pairs:
            done = False
            for d in steps:
                if done:
                    break
                for signs in [(-1, 1), (1, -1)]:
                    di, dj = signs[0] * d, signs[1] * d
                    if a[i] + di < 0 or a[i] + di > 1 or a[j] + dj < 0 or a[j] + dj > 1:
                        continue
                    dc = _compute_dc(a, b, corr, shifts, i, j, di, dj)
                    nm = float(np.max(corr + dc))
                    if nm < cur_max - 1e-13:
                        a[i] += di
                        a[j] += dj
                        b[i] -= di
                        b[j] -= dj
                        corr += dc
                        cur_max = nm
                        improved = True
                        done = True
                        break
        if not improved:
            break
    return a, cur_max * (2.0 / n)

def _random_perturb(a, target, rng, scale=0.05):

    a2 = a.copy()
    idx = rng.randint(0, n, size=rng.randint(3, 10))
