```python
    """Project y onto {x : 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = -1.0, 1.0  # y_i ± lambda should cover [0,1]
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune for exact sum
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # Minor adjustment
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            x[mask] += diff / mask.sum()
            x[mask] = np.clip(x[mask], 0, 1)
    return x
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if abs(s - tgt) > 1e-12 and s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-14:
        mask = (x > 1e-15) & (x < 1.0 - 1e-15)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y + lam, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
    else:
        a = -lag
        if n - a > 0:
            g[:n-a] += (1.0 - h[a:])
            g[a:] -= h[:n-a]
    return g

def projected_gradient(h, n, dx, max_iter, tol=1e-10, top_k=3, alpha_init=0.01):
    """Polish using projected gradient on the max correlation."""
    h = h.copy()
    alpha = alpha_init
    best = h.copy()
    best_val = compute_c5(h, dx)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        c5 = m * dx
        
        # Find top-k active constraints
        thresh = m - 0.05 * m  # within 5% of max
        active = np.where(corr > thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > top_k:
            idx = np.argsort(corr)[::-1][:top_k]
            active = idx
        
        # Compute subgradient
        grad = np.zeros(n)
        for m_idx in active:
            lag = m_idx - (n - 1)
            grad += grad_lag(h, lag, n)
        grad *= dx / len(active)
        
        # Gradient descent step
        h_new = project_feasible(h - alpha * grad, n)
        
        # Line search / armijo
        val_new = compute_c5(h_new, dx)
        if val_new < c5:
            h = h_new
            alpha = min(alpha * 1.1, 0.5)
            if val_new < best_val:
                best_val = val_new
                best = h.copy()
        else:
            alpha *= 0.7
            if alpha < 1e-8:
                break
    
    return best, best_val

# SLSQP with soft-max
_gdx = 1.0
_gtau = 1.0
_cN = 0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx

def _obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    exp_sh = np.exp(sh)
    w = exp_sh / np.sum(exp_sh)
    # Gradient: dx/tau * sum_m w[m] * grad_lag(x, lag_m, n)
    n = _cN
    grad = np.zeros(n)
    for m_idx in range(2*n - 1):
        if w[m_idx] > 1e-10:
            lag = m_idx - (n - 1)
            grad += w[m_idx] * grad_lag(x, lag, n)
    return _gdx / _gtau * grad

def _cfun(x):
    return np.sum(x) - _cN / 2.0

def _cjac(x):
    return np.ones(_cN)

def slsqp_opt(x0, n, dx, tau, maxiter=300):
    global _gdx, _gtau, _cN
    _cN = n
    _gdx = dx
    _gtau = tau
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_obj, x0, jac=_obj_grad, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def slsqp_anneal(x0, n, dx, time_limit, taus=None):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    if taus is None:
        taus = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for tau in taus:
        if time.time() - t0 > time_limit - 2.0:
            break
        iters = 400 if n <= 200 else 200
        x = slsqp_opt(x, n, dx, tau, iters)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
    return best, best_val

def project_grid(x_src, n_src, n_dst):
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gtau = 1.0
_cN = 0


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if abs(s - tgt) > 1e-12 and s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-14:
        mask = (x > 1e-15) & (x < 1.0 - 1e-15)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y + lam, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        a = n - lag
        if a > 0:
            g[lag:] += (1.0 - h[:a])
            g[:a] -= h[lag:]
    else:
        a = -lag
        if n - a > 0:
            g[:n - a] += (1.0 - h[a:])
            g[a:] -= h[:n - a]
    return g


def projected_gradient(h, n, dx, max_iter, top_k=5, alpha_init=0.005):
    h = h.copy().astype(float)
    alpha = alpha_init
    best = h.copy()
    best_val = compute_c5(h, dx)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        c5 = m * dx
        thresh = m - 0.02 * m - 1e-6
        active = np.where(corr > thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > top_k:
            active = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for m_idx in active:
            lag = m_idx - (n - 1)
            grad += grad_lag(h, lag, n)
        grad *= dx / len(active)
        h_new = project_feasible(h - alpha * grad, n)
        val_new = compute_c5(h_new, dx)
        if val_new < c5 - 1e-14:
            h = h_new
            alpha = min(alpha * 1.05, 0.5)
            if val_new < best_val:
                best_val = val_new
                best = h.copy()
        else:
            alpha *= 0.6
            if alpha < 1e-9:
                break
    return best, best_val


def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx


def _obj_grad(x):
    n = _cN
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    exp_sh = np.exp(sh)
    w = exp_sh / np.sum(exp_sh)
    active = np.where(w > 1e-6)[0]
    grad = np.zeros(n)
    for m_idx in active:
        lag = m_idx - (n - 1)
        grad += w[m_idx] * grad_lag(x, lag, n)
    return _gdx / _gtau * grad


def _cfun(x):
    return np.sum(x) - _cN / 2.0


def _cjac(x):
    return np.ones(_cN)


def slsqp_opt(x0, n, dx, tau, maxiter=400):
    global _gdx, _gtau, _cN
    _cN = n
    _gdx = dx
    _gtau = tau
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_obj, x0, jac=_obj_grad, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()


def slsqp_anneal(x0, n, dx, time_limit, taus=None):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    if taus is None:
        taus = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    iters = 400 if n <= 200 else 250
    for tau in taus:
        if time.time() - t0 > time_limit - 3.0:
            break
        x = slsqp_opt(x, n, dx, tau, iters)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
    return best, best_val


def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    interp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate', bounds_error=False)
    return repair(interp(xd), n_dst)


def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)


def structured_perturb(x, n, rng):
    """Block swap perturbation: pick a random block and flip it."""
    x2 = x.copy()
    blk = max(3, n // 15)
    start = rng.integers(0, n - blk)
    x2[start:start + blk] = 1.0 - x[start:start + blk]
    return repair(x2, n)

    h0 = repair(h0_raw, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Phase 1: SLSQP at base grid with multiple restarts
    p1_end = t_start + budget_s * 0.42
    starts = [h0.copy(), perturb(h0, n0, 0.06, rng), perturb(h0, n0, 0.12, rng),
              structured_perturb(h0, n0, rng)]
    for idx, xs in enumerate(starts):
        if elapsed() > p1_end:
            break
        alloc = p1_end - elapsed() - 5
        if alloc < 15:
            break
        x_best, val = slsqp_anneal(xs, n0, dx0, alloc)
        if val < best_val:
            best_val = val
            best_h = x_best
            best_n = n0

    # Phase 2: PG polish at base grid
    if remaining() > 40:
        p2_alloc = min(remaining() * 0.12, 50)
        x_pg, val_pg = projected_gradient(best_h, best_n, 2.0 / best_n,
                                          int(p2_alloc * 200), top_k=5)
        if val_pg < best_val:
            best_val = val_pg
            best_h = x_pg

    # Phase 3: Refine to 2x+1 grid
    if remaining() > 100:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        x_p = project_grid(best_h, best_n, n1)
        p3_alloc = min(remaining() * 0.28, 250)
        taus_fine = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        x_b1, v1 = slsqp_anneal(x_p, n1, dx1, p3_alloc, taus=taus_fine)
        if v1 < best_val:
            best_val = v1
            best_h = x_b1
            best_n = n1
        # PG on finer grid
        if remaining() > 30:
            x_pg1, v_pg1 = projected_gradient(best_h, best_n, 2.0 / best_n,
                                              int(min(remaining() * 0.1, 60) * 150), top_k=7)
            if v_pg1 < best_val:
                best_val = v_pg1
                best_h = x_pg1
            best_n = best_n  # keep track

    # Phase 4: Another restart at fine grid + final polish
    if remaining() > 60:
        n_f = best_n
        dx_f = 2.0 / n_f
        x_r = perturb(best_h, n_f, 0.03, rng)
        p4_alloc = min(remaining() * 0.2, 120)
        x_b4, v4 = slsqp_anneal(x_r, n_f, dx_f, p4_alloc,
                                 taus=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
        if v4 < best_val:
            best_val = v4
            best_h = x_b4
            best_n = n_f

    # Phase 5: Final PG squeeze
    if remaining() > 20:
        dx_fin = 2.0 / best_n
        x_final, v_final = projected_gradient(best_h, best_n, dx_fin,
                                              int(min(remaining() * 0.6, 100) * 200), top_k=5,
                                              alpha_init=0.002)
        if v_final < best_val:
            best_val = v_final
            best_h = x_final

    # Final repair and evaluation
    dx_f2 = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f2)
    return (best_h, final_val, best_n)
```
