                h2, c2 = slsqp_polish(h2, n, dx, maxiter=300)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
