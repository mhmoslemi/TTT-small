        if d >= 0:
            idx1 = np.arange(max(0, n - d))
            grad[idx1] += (1.0 - h[idx1 + d])
            if d > 0:
                idx2 = np.arange(d, n)
                grad[idx2] -= h[idx2 - d]
        else:
            dp = -d
            idx1 = np.arange(dp, n)
            grad[idx1] += (1.0 - h[idx1 - dp])
            idx2 = np.arange(0, n - dp)
            grad[idx2] -= h[idx2 + dp]
    return h - alpha * grad

def projected_subgradient(h, ts, n_steps=400, alpha0=0.02, top_k=5):
    h = h.copy()
    for step in range(n_steps):
        alpha = alpha0 / (1.0 + step * 0.005)
        h = subgradient_step(h, alpha, top_k)
        h = project_feasible(h, ts)
    return h

def slsqp_solve(h, n, ts, beta, maxiter=600):
    def obj(x):
        return smooth_obj(x, n, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def binary_local_search(h, n, ts, rng, max_sweeps=300, time_limit=150):
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            for t_val in (0.0, 1.0):
                if abs(t_val - h[j]) < 1e-9:
                    continue
                old = h[j]
                h[j] = t_val
                h = project_feasible(h, ts)
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    improved = True
                else:
                    h[j] = old
                    h = project_feasible(h, ts)
        # Swap moves
        if time.time() - t0 > time_limit * 0.6:
            break
        for _ in range(n):
            i, j = rng.choice(n, 2, replace=False)
            if abs(h[i] - h[j]) < 1e-9:
                continue
            h[i], h[j] = h[j], h[i]
            h = project_feasible(h, ts)
            c5_t = compute_c5(h)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
                h = project_feasible(h, ts)
        if not improved:
            break
    return h, best_c5

def sim_anneal_binary(h, n, ts, rng, time_limit=120, T0=0.005, T_final=1e-7):
    # Work on binary state
    state = (h > 0.5).astype(float)
    n_ones = int(round(state.sum()))
    state = np.where(h > 0.5, 1.0, 0.0)
    if state.sum() != n_ones:
        idx = np.argsort(h)[::-1]
        state = np.zeros(n)
        state[idx[:n_ones]] = 1.0
    best_state = state.copy()
    best_c5 = compute_c5(project_feasible(state, ts))
    cur_c5 = best_c5
    t0 = time.time()
    n_iters = 200000
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        T = T0 * (T_final / T0) ** (it / max(1, n_iters))
        idx1 = np.where(state == 1.0)[0]
        idx0 = np.where(state == 0.0)[0]
        if len(idx1) == 0 or len(idx0) == 0:
            break
        i = idx1[rng.randint(len(idx1))]
        j = idx0[rng.randint(len(idx0))]
        state[i], state[j] = 0.0, 1.0
        new_c5 = compute_c5(project_feasible(state, ts))
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_state = state.copy()
        else:
            state[i], state[j] = 1.0, 0.0
    return best_state, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([n_init, 280]))

    for n in n_list:
        if remaining() < 120:
            break
        ts = n / 2.0

        if n == n_init:
            h = h_init.copy()
        else:
            xo = np.linspace(0, 2, n_init)
            xn = np.linspace(0, 2, n)
            h = np.interp(xn, xo, best_h)
            h = project_feasible(h, ts)

        # SLSQP with annealed beta
        for beta in [4.0, 12.0, 40.0, 120.0, 400.0]:
            if remaining() < 50:
                break
            h = slsqp_solve(h, n, ts, beta, maxiter=400)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Subgradient refinement
        if remaining() > 80:
            h_sg = projected_subgradient(h.copy(), ts, n_steps=500, alpha0=0.025, top_k=3)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

        # Simulated annealing on binary
        if remaining() > 60:
            h_bin, c5_bin = sim_anneal_binary(h, n, ts, rng, time_limit=min(100, remaining() * 0.25))
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = project_feasible(h_bin, ts)

        # Binary local search
        if remaining() > 40:
            h_ls = best_h.copy() if len(best_h) == n else h
            tl = min(150, max(25, remaining() * 0.3))
            h_ls, c5_ls = binary_local_search(h_ls, n, ts, rng, max_sweeps=400, time_limit=tl)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

        # Final SLSQP polish
        if remaining() > 30 and len(best_h) == n:
            h_pol = slsqp_solve(best_h.copy(), n, ts, 600.0, maxiter=500)
            c5_p = compute_c5(h_pol)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol.copy()

    # Random restarts
    if remaining() > 100:
        n_best = len(best_h)
        ts = n_best / 2.0
        for _ in range(5):
            if remaining() < 40:
                break
            noise_scale = 0.1 + 0.1 * rng.random()
            h_r = best_h + rng.normal(0, noise_scale, n_best)
            h_r = project_feasible(h_r, ts)
            h_r = slsqp_solve(h_r, n_best, ts, 20.0, maxiter=300)
            h_r = slsqp_solve(h_r, n_best, ts, 150.0, maxiter=300)
            if remaining() > 25:
                h_r, c5_r = binary_local_search(h_r, n_best, ts, rng, max_sweeps=150, time_limit=50)
            else:
                c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
