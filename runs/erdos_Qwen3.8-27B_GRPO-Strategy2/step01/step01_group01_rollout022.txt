            best_v = best
            best_c = old
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                dj = cand - old
                if h[j] + dj < -1e-12 or h[j] + dj > 1.0 + 1e-12:
                    continue
                h[i] = cand
                h[j] = h[j] + dj
                v = compute_c5(h)
                h[i] = old
                h[j] = h[j] - dj
                if v < best_v - 1e-13:
                    best_v = v
                    best_c = cand
            if best_c != old:
                h[i] = best_c
                h[j] = h[j] + (best_c - old)
                best = best_v
                imp = True
        if not imp:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def tleft():
        return budget_s - elapsed()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    best_h = repair(h0, n0)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytic gradient at n0, temperature annealing
    temps_p1 = [30, 100, 400, 1500, 6000]
    for t in temps_p1:
        if tleft() < 80:
            break
        bnds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': partial(_csum, n=n0),
                 'jac': partial(_cjac, n=n0)}]
        tr = min(100, tleft() * 0.12)
        miter = max(60, int(tr / 0.08))
        try:
            res = minimize(partial(_efun, n=n0, t=t), best_h.copy(),
                           jac=partial(_ejac, n=n0, t=t),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': miter, 'ftol': 1e-15})
            h = repair(res.x, n0)
            v = compute_c5(h)
            if v < best_c5:
                best_c5, best_h = v, h.copy()
        except Exception:
            pass

    # Phase 2: CD polish at n0
    if tleft() > 150:
        cl = min(80, tleft() * 0.1)
        h, v = cd_polish(best_h, n0, cl)
        h = repair(h, n0)
        v = compute_c5(h)
        if v < best_c5:
            best_c5, best_h = v, h.copy()

    # Phase 3: Multi-resolution SLSQP
    for mult in [2, 3, 4, 6]:
        n = n0 * mult
        if n > 2800 or tleft() < 120:
            break
        xo = np.linspace(0.0, 2.0, n0)
        xn = np.linspace(0.0, 2.0, n)
        h_init = np.interp(xn, xo, best_h)
        h_init = repair(h_init, n)
        v_init = compute_c5(h_init)
        if v_init < best_c5:
            best_c5, best_h = v_init, h_init.copy()

        temps_p3 = [100, 500, 2000, 8000]
        for t in temps_p3:
            if tleft() < 80:
                break
            bnds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': partial(_csum, n=n),
                     'jac': partial(_cjac, n=n)}]
            tr = min(120, tleft() * 0.13)
            miter = max(40, int(tr / 0.3))
            try:
                res = minimize(partial(_efun, n=n, t=t), h_init,
                               jac=partial(_ejac, n=n, t=t),
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': miter, 'ftol': 1e-15})
                h = repair(res.x, n)
                v = compute_c5(h)
                if v < best_c5:
                    best_c5, best_h = v, h.copy()
                    h_init = h.copy()
            except Exception:
                pass

        # CD polish at this resolution
        if tleft() > 80 and n <= 1500:
            cl = min(50, tleft() * 0.08)
            h, v = cd_polish(best_h, n, cl)
            h = repair(h, n)
            v = compute_c5(h)
            if v < best_c5:
                best_c5, best_h = v, h.copy()

    # Phase 4: Random restarts at best resolution
    if tleft() > 200:
        n = len(best_h)
        for attempt in range(6):
            if tleft() < 100:
                break
            noise_scale = 0.15 + 0.1 * attempt
            noise = np.random.randn(n) * noise_scale
            h_try = np.clip(best_h + noise, 0, 1)
            h_try = repair(h_try, n)
            for t in [300, 2000]:
                if tleft() < 60:
                    break
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': partial(_csum, n=n),
                         'jac': partial(_cjac, n=n)}]
                tr = min(70, tleft() * 0.1)
                miter = max(30, int(tr / 0.3))
                try:
                    res = minimize(partial(_efun, n=n, t=t), h_try,
                                   jac=partial(_ejac, n=n, t=t),
                                   method='SLSQP', bounds=bnds, constraints=cons,
                                   options={'maxiter': miter, 'ftol': 1e-15})
                    h = repair(res.x, n)
                    v = compute_c5(h)
                    if v < best_c5:
                        best_c5, best_h = v, h.copy()
                        h_try = h.copy()
                except Exception:
                    pass

    # Phase 5: Final CD polish
    if tleft() > 30:
        n = len(best_h)
        h, v = cd_polish(best_h, n, min(60, tleft() * 0.3))
        h = repair(h, n)
        v = compute_c5(h)
        if v < best_c5:
            best_c5, best_h = v, h.copy()

    best_h = repair(best_h, len(best_h))
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, len(best_h))
