import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _sm_obj(h, dx, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    vals = np.exp((cv - mx) / eps)
    return mx + eps * np.log(vals.sum())

def _sm_grad(h, dx, eps):
    n = len(h)
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]
    return g

def _slsqp(h0, dx, n, target, eps, maxiter=250):
    def obj(x):
        return _sm_obj(x, dx, eps)
    def jac(x):
        return _sm_grad(x, dx, eps)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    res = minimize(obj, h0, jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                   method='SLSQP',
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return _proj(res.x, target)
