def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
            else:
                break
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
            else:
                break

    d = target - h.sum()
    if abs(d) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if d > 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + d, 0.0, 1.0))

    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = beta * corr * dx
    m = np.max(scaled)
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    target = float(n) / 2.0
    val += penalty * (x.sum() - target) ** 2
    return val

def sm_grad(x, n, dx, beta, penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = beta * corr * dx
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= w.sum()
    omh = 1.0 - x
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-16:

        lag = i - (n - 1)

            grad[lag:] -= wi * x[:n - lag]

    return [x.sum() - float(n) / 2.0]

def cjac_fn(x, n):
    return np.ones((1, len(x)))

def do_lbfgsb(h, n, dx, time_budget):
    ts = time.time()
    h_c = h.copy()
    schedule = [(3, 500, 40), (10, 500, 40), (30, 1000, 40), (100, 1000, 50),
                (300, 2000, 50), (1000, 2000, 50), (3000, 5000, 60), (10000, 5000, 60)]
    for beta, pen, mit in schedule:
        if time.time() - ts > time_budget:
            break
        bnds = [(0.0, 1.0)] * n
        try:
            r = minimize(sm_obj, h_c, args=(n, dx, beta, pen),
                        jac=sm_obj_grad, method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': mit, 'ftol': 1e-11, 'maxfun': mit * 8})
            h_c = project_feasible(r.x, n)
        except Exception:
            pass
    return h_c, eval_c5(h_c, dx)

def do_slsqp(h, n, dx, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ceq_fn, 'jac': cjac_fn, 'args': (n,)}]
    try:
        r = minimize(slsqp_obj, h, args=(dx,), method='SLSQP', bounds=bnds,
                    constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        ho = project_feasible(r.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def coord_bin(h, n, dx, passes=25):
    h = h.copy()
    c5 = eval_c5(h, dx)
    tgt = float(n) / 2.0
    for _ in range(passes):
        imp = False
        for i in range(n):
            for nv in (0.0, 1.0, 0.25, 0.75):
                if abs(h[i] - nv) < 1e-6:
                    continue
                ht = h.copy()
                ht[i] = nv
                d = tgt - ht.sum()
                if abs(d) > 1e-12:
                    j_best = -1
                    best_r = 0.0
                    for j in range(n):
                        if j == i:
                            continue
                        if d > 0:
                            r = 1.0 - ht[j]
                        else:
                            r = ht[j]
                        if j_best < 0 or (d > 0 and r > best_r) or (d < 0 and r < best_r):
                            best_r = r
                            j_best = j
                    if j_best < 0:
                        continue
                    ht[j_best] = float(np.clip(ht[j_best] + d, 0.0, 1.0))
                if abs(ht.sum() - tgt) > 1e-10:
                    continue
                ct = eval_c5(ht, dx)
                if ct < c5 - 1e-14:
                    h, c5 = ht, ct
                    imp = True
                    break
        if not imp:
            break
    return h, c5

def subgrad_step(h, n, dx, lr):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.where(np.abs(corr - mx) < 1e-8)[0]
    g = np.zeros(n)
    omh = 1.0 - h
    for i in active:
        lag = i - (n - 1)
        if lag >= 0:
            g[lag:] += omh[:n - lag]
            g[:n - lag] -= h[lag:]
        else:
            a = -lag
            g[:n - a] += omh[a:]
            g[a:] -= h[:n - a]
    g /= len(active)
    g *= dx
    h_new = h - lr * g
    h_new = project_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 70

    h0 = project_feasible(np.array(initial_h_values, dtype=float).copy(), 70)
    c5_0 = eval_c5(h0, 2.0 / 70)
    best_c5, best_h, best_n = c5_0, h0.copy(), 70

    n_values = [70, 60, 80, 100]
    for n in n_values:
        if budget_s - (time.time() - t0) < 70:
            break
        dx = 2.0 / n
        if n == 70:
            h = h0.copy()
        else:
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, 70), h0)
            h = project_feasible(h, n)
        c5 = eval_c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

        h, c5 = do_lbfgsb(h, n, dx, time_budget=40.0)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h.copy(), n

        if budget_s - (time.time() - t0) > 30:
            h, c5 = do_slsqp(h, n, dx, maxiter=200)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), n

    n = best_n
    dx = 2.0 / n

    if budget_s - (time.time() - t0) > 25:
        h, c5 = coord_bin(best_h, n, dx, passes=20)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    if budget_s - (time.time() - t0) > 40:
        h, c5 = do_lbfgsb(best_h, n, dx, time_budget=35.0)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    if budget_s - (time.time() - t0) > 20:
        h, c5 = do_slsqp(best_h, n, dx, maxiter=400)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Subgradient polish with decreasing lr
    if budget_s - (time.time() - t0) > 15:
        h = best_h.copy()
        lr = 0.02
        for _ in range(3000):
            if budget_s - (time.time() - t0) < 5:
                break
            h = subgrad_step(h, n, dx, lr)
            c5 = eval_c5(h, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
                lr = max(lr * 0.99, 0.002)
            else:
                lr = max(lr * 0.95, 0.002)

    # Random restarts
    ri = 0
    while budget_s - (time.time() - t0) > 10:
        ri += 1
        sc = 0.12 / (1.0 + 0.12 * ri)
        h0r = best_h + sc * np.random.randn(n)
        h0r = project_feasible(h0r, n)
        h, c5 = do_slsqp(h0r, n, dx, maxiter=80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
            if budget_s - (time.time() - t0) > 15:
                h, c5 = coord_bin(h, n, dx, passes=5)
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = eval_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
