from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_arg(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    return float(corr[idx] * dx), idx, corr

def project_to_n(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    from scipy.interpolate import interp1d
    xp = interp1d(xs, h_src, kind='linear', fill_value='extrapolate', bounds_error=False)(xd)
    x = np.clip(xp, 0.0, 1.0)
    s = np.sum(x)
    tgt = n_dst / 2.0
    if s > 0:
        x *= tgt / s
    x = np.clip(x, 0.0, 1.0)
    # Fix sum exactly
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-10:
        x *= tgt / s2
        x = np.clip(x, 0.0, 1.0)
    return x

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s > 1e-12:
        x *= tgt / s
    x = np.clip(x, 0.0, 1.0)
    # Final exact fix: adjust largest element
    s2 = np.sum(x)
    diff = tgt - s2
    if abs(diff) > 1e-12:
        # Add diff to the element closest to 0.5
        mid = np.argmin(np.abs(x - 0.5))
        x[mid] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def binarize(x, n):

    n = int(n)
    tgt = n / 2.0
    if n % 2 == 0:
        # Exactly n/2 ones
        idx = np.argsort(-x)
        b = np.zeros(n)
        b[idx[:n//2]] = 1.0
    else:

        idx = np.argsort(-x)
        b = np.zeros(n)
        b[idx[:(n-1)//2]] = 1.0
        b[idx[(n-1)//2]] = 0.5
    return b

    best = h.copy()
    best_val = compute_c5(h, dx)
    
    t0 = time.time()
    temp = 0.005  # initial temperature (in units of C5)
    cool = 0.9997
    
    n_ones = int(np.sum(h >= 0.5))  # count of "1"s
    n_zeros = n - n_ones
    
    iters_per_check = 1000
    last_check = t0
    stall = 0

        if n_ones > 0 and n_zeros > 0:
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]

            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
            
            new_val = compute_c5(h, dx)

                    best = h.copy()
                    stall = 0
                # Accept

                    j = zeros[rng.integers(len(zeros))]

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    return x

def project_to_n(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, h_src, kind='linear', bounds_error=False, fill_value="extrapolate")(xd)
    return repair(np.clip(xp, 0, 1), n_dst)

def binarize(x, n):
    n = int(n)
    b = np.zeros(n)
    idx = np.argsort(-x)
    if n % 2 == 0:
        b[idx[:n//2]] = 1.0
    else:
        b[idx[:n//2]] = 1.0
        b[idx[n//2]] = 0.5
    return b

def binary_sa(h_init, n, dx, time_budget, rng):
    h = h_init.copy()
    best = h.copy()
    best_val = compute_c5(h, dx)
    curr_val = best_val
    
    t0 = time.time()
    temp = 0.003
    min_temp = 1e-6
    cool_rate = 0.99995
    
    ones_mask = h >= 0.5
    
    while time.time() - t0 < time_budget:
        if ones_mask.sum() == 0 or ones_mask.sum() == n:
            break
        
        # Pick random swap
        one_idxs = np.where(ones_mask)[0]
        zero_idxs = np.where(~ones_mask)[0]
        i0 = one_idxs[rng.integers(len(one_idxs))]
        j0 = zero_idxs[rng.integers(len(zero_idxs))]
        
        # Tentative swap
        h[i0], h[j0] = 0.0, 1.0
        ones_mask[i0] = False
        ones_mask[j0] = True
        
        new_val = compute_c5(h, dx)
        
        delta = new_val - curr_val
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, min_temp)):
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = h.copy()
        else:
            # Revert
            h[i0], h[j0] = 1.0, 0.0
            ones_mask[i0] = True
            ones_mask[j0] = False
        
        temp = max(temp * cool_rate, min_temp)
        
        # Periodic restart if stuck
        if int((time.time() - t0) * 10) % 10 == 0:  # every 0.1s
            # Check if we should restart
            pass
    
    return best, best_val

    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
