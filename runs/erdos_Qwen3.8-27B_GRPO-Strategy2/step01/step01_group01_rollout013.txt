    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue

        a1 = max(0, -p)
        b1 = min(n, n - p)
        if b1 > a1:

        b2 = min(n, n + p)
        if b2 > a2:

    grad *= dx
    
    # Gradient step
    h_new = h - step_size * grad
    # Project
    h_new = project_box_hyperplane(h_new, n)

        if wk < 1e-14:
            continue
        p = k - n + 1
        a1, b1 = max(0, -p), min(n, n - p)
        if b1 > a1:
            grad[a1:b1] += wk * (1.0 - h[a1+p:b1+p])
        a2, b2 = max(0, p), min(n, n + p)
        if b2 > a2:
            grad[a2:b2] -= wk * h[a2-p:b2-p]
    return val, grad

def projected_gradient(h, n, t_frames, steps_per_frame, lr0):
    h = h.copy()
    dx = 2.0 / n
    best = compute_c5(h)
    lr = lr0
    for frame in range(t_frames):
        t = 30.0 * (5.0 ** (frame / max(t_frames - 1, 1)))
        for step in range(steps_per_frame):
            val, grad = smooth_max_weighted_grad(h, t, n)
            obj = val * dx
            if obj < best:
                best = obj
            h_new = h - lr * grad
            h_new = project_box_hyperplane(h_new, n)
            val_new = smooth_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), t)
            if val_new * dx > obj * 1.3 + 1e-10:
                lr *= 0.5
                h_new = h - lr * grad
                h_new = project_box_hyperplane(h_new, n)
            h = h_new
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
    return h, best

def coord_descent(h, n, max_sweeps=80):
    h = h.copy()
    target = n / 2.0
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        c_base = np.correlate(h, 1.0 - h, mode='full')
        n_lags = len(c_base)
        improved = False
        for j in range(n):
            old_j = h[j]
            B = np.zeros(n_lags)
            for k in range(n_lags):
                p = k - n + 1
                bv = 0.0
                if 0 <= j + p < n:
                    bv += 1.0 - h[j + p]
                if 0 <= j - p < n:
                    bv -= h[j - p]
                B[k] = bv
            A = c_base - old_j * B
            zero_mask = np.abs(B) < 1e-12
            if zero_mask.any():
                zv = np.max(A[zero_mask])
            else:
                zv = -1e9
            lo_v = max(min(np.max(A), np.max(A + B)), zv)
            hi_v = max(np.max(A), np.max(A + B))
            if hi_v - lo_v < 1e-14:
                continue
            for _ in range(35):
                mid_v = (lo_v + hi_v) / 2.0
                if mid_v < zv:
                    lo_v = mid_v
                    continue
                pos = B > 1e-12
                neg = B < -1e-12
                x_max = 1.0
                x_min = 0.0
                if pos.any():
                    x_max = min(x_max, np.min((mid_v - A[pos]) / B[pos]))
                if neg.any():
                    x_min = max(x_min, np.max((mid_v - A[neg]) / B[neg]))
                if x_min <= x_max + 1e-14:
                    hi_v = mid_v
                else:
                    lo_v = mid_v
            v_opt = hi_v
            pos = B > 1e-12
            neg = B < -1e-12
            x_max = 1.0 if not pos.any() else np.min((v_opt - A[pos]) / B[pos])
            x_min = 0.0 if not neg.any() else np.max((v_opt - A[neg]) / B[neg])
            x_opt = np.clip(old_j, max(x_min, 0.0), min(x_max, 1.0))
            if abs(x_opt - old_j) < 1e-12:
                continue
            h[j] = x_opt
            diff = target - h.sum()
            if abs(diff) > 1e-13:
                if diff > 0:
                    mask = h < 1.0
                else:
                    mask = h > 0.0
                mask[j] = False
                if mask.any():
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0.0, 1.0)
            val = compute_c5(h)
            if val <= best + 1e-12:
                best = val
                improved = True
                c_base = np.correlate(h, 1.0 - h, mode='full')
            else:
                h[j] = old_j
        if not improved:
            break
    return h, best

def upscale(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 1.0, n_old)
    xn = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(xn, xo, h_old)
    return repair(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def elapsed():
        return time.time() - t0

    def time_ok(frac):
        return (budget_s * frac - elapsed()) > 8.0

    # Phase 1: Projected gradient at original n
    if time_ok(0.15):
        h1, c1 = projected_gradient(best_h, n0, t_frames=5, steps_per_frame=200, lr0=0.02)
        h1 = repair(h1, n0)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

    # Phase 2: Coordinate descent at original n
    if time_ok(0.25):
        h2, c2 = coord_descent(best_h, n0, max_sweeps=60)
        h2 = repair(h2, n0)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Multi-resolution
    for mult in [2, 3, 4, 6]:
        n = n0 * mult
        if n > 2600 or not time_ok(0.80):
            break
        h_init = upscale(best_h, n)
        if time_ok(0.75):
            hpg, cpg = projected_gradient(h_init, n, t_frames=4, steps_per_frame=100, lr0=0.015)
            hpg = repair(hpg, n)
            cpg = compute_c5(hpg)
            if cpg < best_c5:
                best_c5, best_h = cpg, hpg.copy()
        if time_ok(0.78) and n <= 1500:
            hdc, chdc = coord_descent(best_h, n, max_sweeps=30)
            hdc = repair(hdc, n)
            chdc = compute_c5(hdc)
            if chdc < best_c5:
                best_c5, best_h = chdc, hdc.copy()

    # Phase 4: SLSQP refinement on best
    if time_ok(0.88):
        n = len(best_h)
        t = 500.0
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5,
                 'jac': lambda x, n=n: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n

        def obj_slsqp(x):
            return smooth_max_val(np.correlate(x, 1.0 - x, mode='full'), t)

        def grad_slsqp(x):
            _, g = smooth_max_weighted_grad(x, t, n)
            return g

        maxiter = max(50, int((budget_s * 0.90 - elapsed()) / 0.5))
        res = minimize(obj_slsqp, best_h.copy(), jac=grad_slsqp, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hs = repair(res.x, n)
        cs = compute_c5(hs)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()

    # Phase 5: Random restarts
    if time_ok(0.92):
        n = len(best_h)
        for attempt in range(4):
            if not time_ok(0.92):
                break
            noise = np.random.randn(n) * 0.08
            h_try = repair(best_h + noise, n)
            ht, ct = projected_gradient(h_try, n, t_frames=3, steps_per_frame=80, lr0=0.02)
            ht = repair(ht, n)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht.copy()

    # Phase 6: Final coord descent polish
    if time_ok(0.97):
        n = len(best_h)
        hfin, cfin = coord_descent(best_h, n, max_sweeps=40)
        hfin = repair(hfin, n)
        cfin = compute_c5(hfin)
        if cfin < best_c5:
            best_c5, best_h = cfin, hfin.copy()

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
