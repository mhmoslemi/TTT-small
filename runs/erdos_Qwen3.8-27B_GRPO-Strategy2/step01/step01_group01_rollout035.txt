Wait, I'm concerned about the CD being too slow. Let me optimize it. The issue is that for each coordinate j, I'm recomputing the full correlate (O(n²) in the naive case, but np.correlate is O(n²) anyway for full mode... actually np.correlate for arrays of length n gives output of length 2n-1, and the computation is O(n * (2n-1)) ≈ O(n²)).

For n=376: one correlate is about 376² = 141K multiply-adds. Very fast in C (numpy). The overhead is in Python loop iteration.

Per CD sweep: 376 coordinates * 10 candidates * (recompute correlate) = 3760 correlates. Each correlate is ~141K ops ≈ maybe 0.1ms. Total per sweep: ~376ms. 100 sweeps: ~37s. That's manageable.

Actually wait, I don't need to recompute the full correlate for each candidate. I can compute the sensitivity once per coordinate j, then evaluate all candidates using the sensitivity. Let me optimize:

For coordinate j, the change in c when h[j] goes from old to v is:
c_new[m] = c[m] + (v - old) * s_j[m]

where s_j[m] = dx * [(1-h[j-k_m]) * 1_{0<=j-k_m<n} - h[j+k_m] * 1_{0<=j+k_m<n}]

and k_m = m - (n-1).

I can precompute s_j once (O(n) cost), then for each candidate v:
max(c + (v-old)*s_j) 

This is O(n) per candidate. Total per coordinate: O(n) for s_j + O(n * num_cands) for evaluations.
Per sweep: O(n²) total. Much better than O(n³) (full correlate per candidate).

Let me implement this optimization:

def cd_polish_fast(h, n, max_sweeps, time_limit):
    t_start = time.time()
    h = h.copy().astype(float)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    best = np.max(c)
    L = 2 * n - 1  # length of c
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            old_v = h[j]
            # Compute sensitivity s_j of length L
            s_j = np.zeros(L)
            # For shift k = m-(n-1): 
            #   s_j[m] = dx * [(1-h[j-k])*1_{0<=j-k<n} - h[j+k]*1_{0<=j+k<n}]
            # j-k in [0,n-1] → k in [j-(n-1), j] → m in [j, j+n-1]
            # j+k in [0,n-1] → k in [-j, n-1-j] → m in [n-1-j, n-1+j]
            
            m_lo1 = j  # first m where j-k is valid (k=j → m=2j-(n-1)...)
            # Actually let me just iterate over valid shifts
            # k ranges from max(-(n-1), -j) to min(n-1, n-1-j) for j+k valid
            # and from max(-(n-1), j-(n-1)) to min(n-1, j) for j-k valid
            
            # Simpler: for m in range(L):
            #   k = m - (n-1)
            #   if 0<=j-k<n: s_j[m] += dx*(1-h[j-k])
            #   if 0<=j+k<n: s_j[m] -= dx*h[j+k]
            
            # Vectorized:
            # j-k in [0,n-1]: k in [j-(n-1), j], i.e., k in [-(n-1-j), j]
            # m = k + (n-1): m in [n-1-(n-1-j), n-1+j] = [j, n-1+j]
            m1_start = max(0, j)
            m1_end = min(L-1, n-1+j)
            k_vals1 = np.arange(m1_start, m1_end+1) - (n-1)  # k values
            h_idx1 = j - k_vals1  # = j - (m-(n-1)) = j + n - 1 - m
            # h_idx1 should be in [0, n-1]
            s_j[m1_start:m1_end+1] += dx * (1.0 - h[h_idx1])
            
            # j+k in [0,n-1]: k in [-j, n-1-j]
            # m = k + (n-1): m in [n-1-j, n-1+j]
            m2_start = max(0, n-1-j)
            m2_end = min(L-1, n-1+j)
            k_vals2 = np.arange(m2_start, m2_end+1) - (n-1)
            h_idx2 = j + k_vals2
            s_j[m2_start:m2_end+1] -= dx * h[h_idx2]
            
            # Now find best v in [0,1] minimizing max(c + (v-old_v)*s_j)
            # Try candidates
            deltas = [0.0, 0.05, 0.1, 0.2, 0.5]
            best_val = best
            best_v = old_v
            
            for d in deltas:
                for sign in [-1, 1]:
                    v = old_v + sign * d
                    if v < 0.0 or v > 1.0:
                        continue
                    if d == 0:
                        v = 0.5
                    c_try = c + (v - old_v) * s_j
                    val = np.max(c_try)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_v = v
            
            # Also try v=0 and v=1 explicitly
            for v in [0.0, 1.0]:
                if abs(v - old_v) > 1e-10:
                    c_try = c + (v - old_v) * s_j
                    val = np.max(c_try)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_v = v
            
            if best_v != old_v:
                h[j] = best_v
                c += (best_v - old_v) * s_j
                best = np.max(c)
                improved = True
        
        if not improved:
            break
    
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_shifts(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def project_poly(h, n):
    target = n * 0.5
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def fw_optimize(h, n, max_iter, time_limit):
    t_start = time.time()
    h = h.copy().astype(float)
    c = compute_all_shifts(h)
    best = np.max(c)
    dx = 2.0 / n
    n_half = n // 2
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        k_max = int(np.argmax(c))
        k_shift = k_max - (n - 1)
        s = np.zeros(n)
        j1lo, j1hi = max(0, k_shift), min(n - 1, n - 1 + k_shift)
        if j1lo <= j1hi:
            s[j1lo:j1hi + 1] += (1.0 - h[j1lo - k_shift:j1hi - k_shift + 1]) * dx
        j2lo, j2hi = max(0, -k_shift), min(n - 1, n - 1 - k_shift)
        if j2lo <= j2hi:
            s[j2lo:j2hi + 1] -= h[j2lo + k_shift:j2hi + k_shift + 1] * dx
        part = np.argpartition(s, n_half)[:n_half]
        v = np.zeros(n)
        v[part] = 1.0
        gamma = 2.0 / (it + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        c_new = compute_all_shifts(h_new)
        val = np.max(c_new)
        if val < best:
            best = val
            h = h_new
            c = c_new
    return h, best

def cd_polish(h, n, max_sweeps, time_limit):
    t_start = time.time()
    h = h.copy().astype(float)
    dx = 2.0 / n
    c = compute_all_shifts(h)
    best = np.max(c)
    L = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            old_v = h[j]
            s_j = np.zeros(L)
            m1s, m1e = max(0, j), min(L - 1, n - 1 + j)
            if m1s <= m1e:
                kv = np.arange(m1s, m1e + 1) - (n - 1)
                s_j[m1s:m1e + 1] += dx * (1.0 - h[j - kv])
            m2s, m2e = max(0, n - 1 - j), min(L - 1, n - 1 + j)
            if m2s <= m2e:
                kv = np.arange(m2s, m2e + 1) - (n - 1)
                s_j[m2s:m2e + 1] -= dx * h[j + kv]
            best_val = best
            best_v = old_v
            cands = [0.0, 1.0]
            for d in [0.02, 0.05, 0.1, 0.2]:
                for sg in [-1, 1]:
                    vv = old_v + sg * d
                    if 0.0 <= vv <= 1.0 and abs(vv - old_v) > 1e-10:
                        cands.append(vv)
            for v in cands:
                if abs(v - old_v) < 1e-10:
                    continue
                val = np.max(c + (v - old_v) * s_j)
                if val < best_val - 1e-14:
                    best_val = val
                    best_v = v
            if abs(best_v - old_v) > 1e-12:
                h[j] = best_v
                c += (best_v - old_v) * s_j
                best = np.max(c)
                improved = True
        if improved:
            h = repair(h, n)
            c = compute_all_shifts(h)
            best = np.max(c)
        else:
            break
    return h, best

def proj_subgrad(h, n, max_iter, time_limit):
    t_start = time.time()
    h = h.copy().astype(float)
    dx = 2.0 / n
    c = compute_all_shifts(h)
    best = np.max(c)
    eta = 0.02
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        k_max = int(np.argmax(c))
        k_shift = k_max - (n - 1)
        s = np.zeros(n)
        j1lo, j1hi = max(0, k_shift), min(n - 1, n - 1 + k_shift)
        if j1lo <= j1hi:
            s[j1lo:j1hi + 1] += (1.0 - h[j1lo - k_shift:j1hi - k_shift + 1]) * dx
        j2lo, j2hi = max(0, -k_shift), min(n - 1, n - 1 - k_shift)
        if j2lo <= j2hi:
            s[j2lo:j2hi + 1] -= h[j2lo + k_shift:j2hi + k_shift + 1] * dx
        h_new = project_poly(h - eta * s, n)
        c_new = compute_all_shifts(h_new)
        val = np.max(c_new)
        if val < best:
            best = val
            h = h_new
            c = c_new
            eta = min(eta * 1.005, 0.1)
        else:
            eta *= 0.95
    return h, best
