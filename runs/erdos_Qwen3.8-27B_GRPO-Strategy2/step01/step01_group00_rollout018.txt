import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < 1e-15:
            h[:] = 0.5
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

def _exact_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _cyclic_descent(h, n, dx, max_passes=2000, tol=1e-11):
    h = h.astype(float).copy()
    prev_c5 = _exact_c5(h, dx)
    for _pass in range(max_passes):
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            slope = c1 - c0
            base = c0
            a, b = 0.0, 1.0
            for _ in range(80):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                v1 = np.max(base + slope * m1)
                v2 = np.max(base + slope * m2)
                if v1 < v2:
                    b = m2
                else:
                    a = m1
            h[i] = (a + b) / 2.0
        h = _repair(h, n)
        cur_c5 = _exact_c5(h, dx)
        if cur_c5 >= prev_c5 - tol:
            if cur_c5 < prev_c5:
                prev_c5 = cur_c5
            else:
                break
        else:
            prev_c5 = cur_c5
    return h

def _pnorm_obj(x, p, dx):
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return float(np.sum(np.abs(c) ** p) ** (1.0 / p))

def _pair_search(h, n, dx, rng, n_steps=200000):
    h = h.copy()
    best = _exact_c5(h, dx)
    for _ in range(n_steps):
        i = rng.randint(n)
        delta = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2, -0.01, -0.02, -0.05, -0.1, -0.2])
        new_val = h[i] + delta
        if new_val < -1e-9 or new_val > 1.0 + 1e-9:
            continue
        old_i = h[i]
        h[i] = np.clip(new_val, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-9:
            j = rng.randint(n)
            old_j = h[j]
            h[j] = np.clip(h[j] - diff, 0.0, 1.0)
            h[i] = np.clip(h[i] + (n / 2.0 - h.sum()), 0.0, 1.0)
            c5 = _exact_c5(h, dx)
            if c5 < best - 1e-12:
                best = c5
            else:
                h[i] = old_i
                h[j] = old_j
        else:
            c5 = _exact_c5(h, dx)
            if c5 < best - 1e-12:
                best = c5
            else:
                h[i] = old_i
    return h, best

def _push_binary(h, n, dx, rng, n_trials=50000):
    h = h.copy()
    best = _exact_c5(h, dx)
    for _ in range(n_trials):
        i = rng.randint(n)

        if c5 < best - 1e-12:

def _slsqp_pnorm(h, n, dx, p, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    def eq_con(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(
            lambda x: _pnorm_obj(x, p, dx),
            h, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc
    except Exception:
        return h
