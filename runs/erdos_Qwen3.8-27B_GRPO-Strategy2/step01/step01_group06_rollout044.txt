import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h = h + diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h = h + diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def slsqp_opt(h, n, dx, maxiter=1000, ftol=1e-14):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return [x.sum() - target]
    def cjac(x):
        return np.ones((1, n))
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def coord_descent(h, n, dx, max_passes=50):
    """Exact piecewise-linear coordinate descent."""
    target = float(n) / 2.0
    h = h.copy()
    best_c5 = eval_c5(h, dx)
    
    for p in range(max_passes):
        improved = False
        for j in range(n):
            # Compute alpha_m and beta_m for all lags
            # corr[m] corresponds to lag m in np.correlate(h, 1-h, 'full')
            # which has length 2n-1, indexed 0 to 2n-2
            # lag = idx - (n-1)
            ncorr = 2 * n - 1
            alpha = np.zeros(ncorr)
            beta_val = np.zeros(ncorr)
            
            # For each lag, compute coefficient of h[j]
            for m in range(ncorr):
                lag = m - (n - 1)
                # h[j] appears as h[i] when i=j, i+m valid: j + lag in [0, n-1]
                # h[j] appears in (1-h[i+m]) when i+m=j, i valid: j-lag in [0, n-1]
                a = 0.0
                if 0 <= j + lag < n:
                    a += 1.0 - h[j + lag]
                if 0 <= j - lag < n:
                    a -= h[j - lag]
                alpha[m] = a
                beta_val[m] = 0.0  # we'll compute from current corr
            
            # Current correlation
            corr = np.correlate(h, 1.0 - h, mode='full')
            # beta = corr - alpha * h[j]
            beta = corr - alpha * h[j]
            
            # Find optimal h[j]: minimize max_m (alpha[m]*t + beta[m]) for t in [0,1]
            # Collect candidate t values: breakpoints where alpha[m]*t+beta[m] = alpha[m']*t+beta[m']
            candidates = set()
            candidates.add(0.0)
            candidates.add(1.0)
            
            for m1 in range(ncorr):
                for m2 in range(m1+1, ncorr):
                    da = alpha[m1] - alpha[m2]
                    if abs(da) > 1e-12:
                        t = (beta[m2] - beta[m1]) / da
                        if 0.0 <= t <= 1.0:
                            candidates.add(t)
            
            # Evaluate at each candidate
            best_t = h[j]
            best_obj = float('inf')
            for t in candidates:
                vals = alpha * t + beta
                obj = np.max(vals)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            
            if abs(best_t - h[j]) > 1e-10:
                h[j] = best_t
                improved = True
        
        # Project to feasible set
        h = project_feasible(h, n)
        c5 = eval_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    
    h = project_feasible(h, n)
    return h, eval_c5(h, dx)
