from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def obj_fn(h, n, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    val = (cmax + np.log(np.sum(np.exp(t * (c - cmax)))) / t) * dx
    return val

def grad_fn(h, n, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = t * (c - cmax)
    w = np.exp(shifted)
    w /= w.sum()
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k in range(n):
        v1 = (k + lags >= 0) & (k + lags < n)
        v2 = (k - lags >= 0) & (k - lags < n)
        grad[k] = 1.0 - w[v1] @ h[k + lags[v1]] - w[v2] @ h[k - lags[v2]]
    grad *= dx / t
    return grad

def sa_01(h, n, max_iter, t0, t1, rng):

                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

def l_bfgs_refine(h, n, dx, t, max_iter, time_limit):

    bnds = [(0.0, 1.0)] * n
    
    def fun(x):
        x = np.clip(x, 0.0, 1.0)
        return obj_fn(x, n, dx, t)
    
    def gfun(x):
        x = np.clip(x, 0.0, 1.0)

                         pgtol=1e-10, maxiter=max_iter,
                         factr=1e8, iprint=0,

    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h
