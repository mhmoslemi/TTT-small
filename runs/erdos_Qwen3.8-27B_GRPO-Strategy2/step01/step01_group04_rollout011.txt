def repair(x, n):
    """Project onto {0<=x<=1, sum(x)=n/2}"""
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s < 1e-14:
        x[:] = 0.5
        return x
    x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)

    for _ in range(100):
        s = np.sum(x)
        diff = tgt - s
        if abs(diff) < 1e-12:
            break

def local_search_binary(h, n, max_iters=5000, rng=None):
    """Local search on binary h with exactly n/2 ones"""
    if rng is None:
        rng = np.random.default_rng()
    h = h.copy()
    n1 = np.sum(h > 0.5)
    n0 = n - n1
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    best_corr = compute_c5_binary(h)
    # We work with the unscaled correlation; the dx factor doesn't change the argmax
    
    for it in range(max_iters):
        # Try a random swap: move one 1 to a 0 position
        i = rng.integers(n1)
        j = rng.integers(n0)
        oi = ones[i]
        oz = zeros[j]
        
        h[oi] = 0
        h[oz] = 1
        new_corr = compute_c5_binary(h)
        
        if new_corr > best_corr + 1e-12:
            # Revert
            h[oi] = 1
            h[oz] = 0
        else:
            best_corr = new_corr
            # Update lists
            ones[i] = oz
            zeros[j] = oi
            # Could also do a few more tries from this improvement
    
    return h

        if nf == 0:

    _gn = n
    x = x0.copy()

    for i in range(ntaus):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        f = i / (ntaus - 1)
        tau = taustart * (tauend / taustart) ** f
        _gtau = tau

        p = rng.uniform(0, 2 * np.pi)
        noise += a * np.sin(np.pi * f * xx + p)
    return repair(x + noise, n)
