def project(h, target):
    """Project onto {0≤h≤1, sum(h)=target}"""
    # Binary search for shift
    lo, hi = -h.max(), 1 - h.min()
    for _ in range(60):
        mid = (lo + hi) / 2
        h_shifted = np.clip(h - mid, 0, 1)
        s = h_shifted.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Final adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-12) & (h_proj < 1 - 1e-12)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def obj_smooth(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def projected_grad_descent(h0, n, beta, max_iter=500, step0=0.5, tol=1e-14):
    h = h0.copy()
    best_h = h.copy()
    best_f = compute_c5(h, n)
    f_smooth = obj_smooth(h, n, beta)
    step = step0
    for it in range(max_iter):
        g = grad_smooth(h, n, beta)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break
        for _ in range(30):
            h_try = project_feasible(h - step * g, n)
            f_try = obj_smooth(h_try, n, beta)
            if f_try < f_smooth - 1e-16:
                break
            step *= 0.5
        else:
            step *= 0.5
            if step < 1e-10:
                break
        f_smooth = f_try
        h = h_try
        c = compute_c5(h, n)
        if c < best_f - 1e-13:
            best_f = c
            best_h = h.copy()
        step = min(step * 1.1, 1.0)
    return best_h, best_f

def nelder_mead_step(h0, n, max_iter=200):
    def f(x):
        return compute_c5(x, n)
    try:
        r = minimize(f, h0, method='Nelder-Mead',
                     options={'maxiter': max_iter, 'xatol': 1e-13, 'fatol': 1e-13,
                              'adaptive': True})
        hc = project_feasible(r.x, n)
        return hc, compute_c5(hc, n)
    except Exception:
        return h0, f(h0)

def slsqp_polish(h0, n, max_iter=500, ftol=1e-15):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    obj = lambda x: compute_c5(x, n)
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': max_iter, 'ftol': ftol})
        hc = project_feasible(r.x, n)
        return hc, compute_c5(hc, n)
    except Exception:
        return h0, compute_c5(h0, n)

def coordinate_descent(h0, n, rounds=200):
    h = h0.copy()
    best = h.copy()
    best_c = compute_c5(h, n)
    for rd in range(rounds):
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            for delta in (0.01, -0.01, 0.03, -0.03, 0.005, -0.005):
                hp = h.copy()
                hp[i] += delta
                if hp[i] < 0 or hp[i] > 1:
                    continue
                hp = project_feasible(hp, n)
                c = compute_c5(hp, n)
                if c < best_c - 1e-14:
                    h = hp
                    best_c = c
                    best = hp.copy()
                    improved = True
                    break
        if not improved:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)

    h_best = project_feasible(h_best, n)
    best_c5 = compute_c5(h_best, n)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    # Phase 1: Projected gradient with softmax annealing (40% budget)
    p1_end = budget_s * 0.40
    betas = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5]
    for bi, beta in enumerate(betas):
        if elapsed() > p1_end - 5:
            break
        iters = 300 if bi < 4 else 500
        # Small perturbation for diversity
        h_start = project_feasible(h_best + rng.normal(0, 0.01, n), n)
        h_cand, c_cand = projected_grad_descent(h_start, n, beta, max_iter=iters)
        if c_cand < best_c5 - 1e-13:
            h_best, best_c5 = h_cand, c_cand
        # Also try from a larger perturbation
        if elapsed() < p1_end - 20 and bi % 2 == 0:
            h_start2 = project_feasible(h_best + rng.normal(0, 0.05, n), n)
            h_c2, c2 = projected_grad_descent(h_start2, n, beta, max_iter=200)
            if c2 < best_c5 - 1e-13:
                h_best, best_c5 = h_c2, c2

    # Phase 2: Nelder-Mead escapes + SLSQP (25% budget)
    p2_end = budget_s * 0.65
    nm_scales = [0.02, 0.04, 0.08, 0.15, 0.03, 0.06, 0.12, 0.2]
    ni = 0
    while elapsed() < p2_end - 10:
        scale = nm_scales[ni % len(nm_scales)]
        h_pert = project_feasible(h_best + rng.normal(0, scale, n), n)
        h_cand, c_cand = nelder_mead_step(h_pert, n, max_iter=150)
        if c_cand < best_c5 - 1e-13:
            h_best, best_c5 = h_cand, c_cand
        if elapsed() < p2_end - 10:
            h_cand2, c_cand2 = slsqp_polish(h_best, n, max_iter=300)
            if c_cand2 < best_c5 - 1e-13:
                h_best, best_c5 = h_cand2, c_cand2
        ni += 1

    # Phase 3: Coordinate descent + fine SLSQP (20% budget)
    p3_end = budget_s * 0.85
    while elapsed() < p3_end - 10:
        h_cand, c_cand = coordinate_descent(h_best, n, rounds=100)
        if c_cand < best_c5 - 1e-14:
            h_best, best_c5 = h_cand, c_cand
        h_cand2, c_cand2 = slsqp_polish(h_best, n, max_iter=400, ftol=1e-15)
        if c_cand2 < best_c5 - 1e-14:
            h_best, best_c5 = h_cand2, c_cand2

    # Phase 4: Final intensive polish
    for _ in range(5):
        if time_left() < 15:
            break
        h_cand, c_cand = slsqp_polish(h_best, n, max_iter=600, ftol=1e-16)
        if c_cand < best_c5 - 1e-15:
            h_best, best_c5 = h_cand, c_cand
        h_cd, c_cd = coordinate_descent(h_best, n, rounds=50)
        if c_cd < best_c5 - 1e-15:
            h_best, best_c5 = h_cd, c_cd

    h_best = project_feasible(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
