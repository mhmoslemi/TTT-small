    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        hp = np.clip(h + mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    hp = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = hp < 1.0
        else:
            mask = hp > 0.0
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smooth_max_and_grad(h, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = t * (c - cmax)
    max_s = shifted.max()
    exps = np.exp(shifted - max_s)
    s = exps.sum()
    f = cmax + (max_s + np.log(s)) / t
    w = exps / s
    lags = np.arange(len(c)) - (n - 1)
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for idx in active:
        lag = int(lags[idx])
        wt = w[idx]
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[0:m] += wt * (1.0 - h[lag:n])
                grad[lag:n] -= wt * h[0:m]
        else:
            m = n + lag  # = n - |lag|
            if m > 0:
                grad[-lag:n] += wt * (1.0 - h[0:m])
                grad[0:m] -= wt * h[-lag:n]
    return f, grad

    h = project_feasible(h.copy(), n)
    best_f = compute_c5(h)
    alpha = 0.01
    for it in range(max_iter):
        f, g = smooth_max_and_grad(h, t)
        g_norm = np.linalg.norm(g)
        if g_norm < tol:
            break
        # Try a few step sizes
        best_h = h.copy()
        best_f = f
        for a in [alpha * 0.25, alpha * 0.5, alpha, alpha * 2.0]:
            h_try = project_feasible(h - a * g, n)
            f_try = compute_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_h = h_try
        if np.allclose(best_h, h):
            break
        h = best_h
        # Decay alpha
        alpha *= 0.995
    return h, compute_c5(h)

    target = n * 0.5
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_local = best
            best_val = old
            for cand in [0.0, 1.0, 0.25, 0.5, 0.75, old * 0.5, 0.5 + (old - 0.5) * 0.5, 0.5 + (old - 0.5) * 1.5]:
                if abs(cand - old) < 1e-10 or cand < -0.01 or cand > 1.01:

                diff = target - h.sum()

                    mask = np.ones(n, dtype=bool)
                    mask[i] = False

                    else:
                        sub = mask & (h > 0.0)

                    if cnt > 0:

                if val < best_local - 1e-12:
                    best_local = val
                    best_val = np.clip(cand, 0, 1)
            if best_val != old:

                diff = target - h.sum()
                if abs(diff) > 1e-12:

                    mask[i] = False

                    else:
                        sub = mask & (h > 0.0)
                    cnt = sub.sum()
                    if cnt > 0:
                        h[sub] += diff / cnt
                        h = np.clip(h, 0, 1)
                best = best_local
                improved = True
            else:

        if not improved:

    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            hi = mid
        else:
            lo = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            m = hp < 1.0
        else:
            m = hp > 0.0
        cnt = m.sum()
        if cnt > 0:
            hp[m] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smooth_max_grad(h, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    sh = t * (c - cmax)
    ms = sh.max()
    exps = np.exp(sh - ms)
    s = exps.sum()
    f = cmax + (ms + np.log(s)) / t
    w = exps / s
    lags = np.arange(len(c)) - (n - 1)
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for idx in active:
        lag = int(lags[idx])
        wt = w[idx]
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += wt * (1.0 - h[lag:n])
                grad[lag:] -= wt * h[:m]
        else:
            m = n + lag
            if m > 0:
                grad[-lag:] += wt * (1.0 - h[:m])
                grad[:m] -= wt * h[-lag:]
    return f, grad

def subgradient_descent(h, t, max_iter=5000, tol=1e-10):
    n = len(h)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    alpha = 0.05
    for it in range(max_iter):
        f, g = smooth_max_grad(h, t)
        gn = np.linalg.norm(g)
        if gn < tol:
            break
        g = g / gn
        best_step = 0.0
        best_h = h.copy()
        best_f = f
        for a in [alpha * 0.2, alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0]:
            h_try = project_feasible(h - a * g, n)
            f_try = compute_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_h = h_try
                best_step = a
        if best_step == 0.0:
            alpha *= 0.8
            if alpha < 1e-6:
                break
        else:
            alpha = min(alpha * 1.05, 0.5)
        h = best_h
        if best_f < best_c5:
            best_c5 = best_f
    return h, compute_c5(h)

def coord_descent(h, n, max_sweeps=80):
    h = h.copy()
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = h[i]
            best_local = best
            best_new = old
            cands = np.array([0.0, 1.0, 0.25, 0.5, 0.75,
                              old - 0.08, old + 0.08,
                              old - 0.15, old + 0.15])
            cands = cands[(cands >= 0) & (cands <= 1)]
            for cand in cands:
                if abs(cand - old) < 1e-9:
                    continue
                h[i] = cand
                diff = n * 0.5 - h.sum()
                if abs(diff) > 1e-13:
                    j = (i + 1) % n
                    if diff > 0:
                        add = min(diff, 1.0 - h[j])
                    else:
                        add = max(diff, -h[j])
                    h[j] += add
                    h[j] = np.clip(h[j], 0.0, 1.0)
                val = compute_c5(h)
                if val < best_local - 1e-13:
                    best_local = val
                    best_new = h.copy()
            h[i] = old
            diff = n * 0.5 - h.sum()
            if abs(diff) > 1e-13:
                j = (i + 1) % n
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
            if best_new != old:
                h = best_new
                best = best_local
                improved = True
        if not improved:
            break
    return project_feasible(h, n), best
