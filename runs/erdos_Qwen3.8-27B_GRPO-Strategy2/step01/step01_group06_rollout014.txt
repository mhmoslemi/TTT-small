    try:
        h0 = project_feasible(np.array(initial_h_values, dtype=float), n)
    except (NameError, ValueError, Exception):
        h0 = project_feasible(0.5 * np.ones(n), n)

    best_h = h0.copy()
    best_c5 = eval_c5(h0, dx)

    # Phase 1: SLSQP with soft-max (T=30) - ~250s
    t1 = t0 + 0.25 * budget_s
    if time.time() < t1:
        h, c = slsqp_soft(best_h, n, dx, T=30, maxiter=2000)
        if c < best_c5:
            best_c5, best_h = c, h

    # Phase 2: Gradient descent with annealing T - ~200s
    t2 = t0 + 0.45 * budget_s
    if time.time() < t2:
        h, c = softgrad_descend(best_h, n, dx, t2, T_init=5.0, T_end=200.0)
        if c < best_c5:
            best_c5, best_h = c, h

    # Phase 3: SLSQP with higher T soft-max - ~150s
    t3 = t0 + 0.6 * budget_s
    if time.time() < t3:
        h, c = slsqp_soft(best_h, n, dx, T=80, maxiter=3000)
        if c < best_c5:
            best_c5, best_h = c, h

    # Phase 4: Hard SLSQP polish - ~100s
    t4 = t0 + 0.7 * budget_s
    if time.time() < t4:
        h, c = slsqp_hard(best_h, n, dx, maxiter=3000)
        if c < best_c5:
            best_c5, best_h = c, h

    # Phase 5: Structured restarts - ~250s
    t5 = t0 + budget_s - 30
    starts = gen_starts(n, seed)
    si = 0
    while si < len(starts) and time.time() < t5 - 10:
        s = project_feasible(starts[si], n)
        si += 1
        # Quick softgrad descent
        t_sub = min(time.time() + 12, t5 - 15)
        h_s, c_s = softgrad_descend(s, n, dx, t_sub, T_init=10.0, T_end=80.0)
        # SLSQP polish
        if time.time() < t5 - 15:
            h_s, c_s = slsqp_soft(h_s, n, dx, T=50, maxiter=500)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s

    # Phase 6: Final hard SLSQP
    if time.time() < budget_s - 8:
        h, c = slsqp_hard(best_h, n, dx, maxiter=5000)
        if c < best_c5:
            best_c5, best_h = c, h

    # Phase 7: Tiny perturbation + polish
    if time.time() < budget_s - 4:
        hp = project_feasible(best_h + 0.005 * np.random.randn(n), n)
        h, c = slsqp_hard(hp, n, dx, maxiter=2000)
        if c < best_c5:
            best_c5, best_h = c, h

    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
