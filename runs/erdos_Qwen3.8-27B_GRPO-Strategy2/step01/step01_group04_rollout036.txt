        x = lbgfb(h1, n1, tau, 200.0, 200, dl3)
        h1 = x
        v = compute_c5(h1, dx1)
        if v < bv:
            bv = v
            bh = h1.copy()
            bn = n1

    # Phase 4: Final coordinate polish at best grid
    rem = budget_s - (time.time() - t0) - 5
    if rem > 10:
        db = 2.0 / bn
        bh, bv = coord_polish(bh, bn, db, rem, rng)

    # Final repair and evaluation
    df = 2.0 / bn
    bh = repair(bh, bn)
    fv = compute_c5(bh, df)
    return (bh, fv, bn)
