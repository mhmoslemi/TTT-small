def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        # Distribute remaining to the most flexible coordinates
        if diff2 > 0:
            order = idx[np.argsort(1.0 - h[idx])]
        else:
            order = idx[np.argsort(h[idx])]
        for j in order:
            step = np.clip(diff2, h[j] if diff2 < 0 else 1.0 - h[j])
            h[j] += step
            diff2 -= step
            if abs(diff2) < 1e-12:
                break
    return h

def smooth_obj(h, n, beta, ref_val=0.0):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    # Smooth max: m + (1/beta)*log(sum(exp(beta*(x-m))))
    shifted = (corr - m) * beta

    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val * (2.0 / n)

        pass

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-11:
        if diff2 > 0:
            order = idx[np.argsort(1.0 - h[idx])]
        else:
            order = idx[np.argsort(h[idx])]
        for j in order:
            step = min(abs(diff2), (1.0 - h[j]) if diff2 > 0 else h[j])
            h[j] += np.sign(diff2) * step
            diff2 -= np.sign(diff2) * step
            if abs(diff2) < 1e-12:
                break
    return h

def smooth_max_obj(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip((corr - m) * beta, -60, 60)
    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val * (2.0 / n)

def power_mean_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = np.mean(corr ** alpha) ** (1.0 / alpha)
    return val * (2.0 / n)

def slsqp_solve(h_in, ts, beta, maxiter=400, time_limit_s=20):
    n = len(h_in)
    t0 = time.time()
    def obj(x):
        if time.time() - t0 > time_limit_s:
            return 1e6
        return smooth_max_obj(x, beta)
    def con_fun(x):
        return x.sum() - ts
    try:
        res = minimize(obj, h_in, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h_in, ts)

def nelder_mead_polish(h, ts, t_start, deadline_s, maxiter=3000):
    n = len(h)
    t0 = time.time()
    def obj(x):
        if time.time() > deadline_s:
            return 1e6
        return compute_c5(x)
    def con(x):
        return x.sum() - ts
    best = h.copy()
    best_c5 = compute_c5(h)
    try:
        res = minimize(obj, h.copy(), method='Nelder-Mead',
                       xatol=1e-10, fatol=1e-12, maxiter=maxiter,
                       options={'adaptive': True})
        if res.x is not None and np.all(np.isfinite(res.x)):
            res_x = project_feasible(res.x, ts)
            c5 = compute_c5(res_x)
            if c5 < best_c5:
                best, best_c5 = res_x, c5
    except Exception:
        pass
    return best, best_c5

def simulated_anneal(h, ts, t_start, deadline_s, rng, t0=0.04, t1=0.0005):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    stagnation = 0
    while time.time() < deadline_s:
        elapsed = max(0.001, time.time() - t_start)
        total = max(0.001, deadline_s - t_start)
        frac = min(1.0, elapsed / total)
        temp = t0 * (t1 / t0) ** frac
        step_size = 0.08 * (1.0 - 0.7 * frac)
        idx = rng.randint(n)
        new_h = cur_h.copy()
        new_h[idx] += rng.normal(0, step_size)
        new_h = project_feasible(new_h, ts)
        new_c5 = compute_c5(new_h)
        d = new_c5 - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(temp, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5, best_h = cur_c5, cur_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
        if stagnation > 8000 and frac > 0.3:
            cur_h = best_h.copy()
            cur_c5 = best_c5
            stagnation = 0
        if time.time() < deadline_s - 15 and (stagnation % 6000 == 0):
            cur_h = slsqp_solve(cur_h, ts, 150.0, maxiter=150, time_limit_s=6)
            cur_c5 = compute_c5(cur_h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, cur_h.copy()
    return best_h, best_c5

def two_opt_search(h, ts, t_start, deadline_s, rng):
    n = len(h)
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    while time.time() < deadline_s:
        improved = False
        pairs = [(rng.randint(n), rng.randint(n)) for _ in range(500)]
        for i, j in pairs:
            if i == j:
                continue
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
            new_h = project_feasible(new_h, ts)
            c5 = compute_c5(new_h)
            if c5 < best_c5 - 1e-14:
                cur_h, best_c5 = new_h, c5
                improved = True
                break
        if not improved:
            break
    return cur_h, best_c5

def coord_descent(h, ts, t_start, deadline_s, rng):
    n = len(h)
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    steps = [0.04, 0.015, 0.006, 0.002, 0.0008]
    while time.time() < deadline_s:
        improved = False
        for j in range(n):
            if time.time() >= deadline_s:
                break
            for s in steps:
                for d in (1, -1):
                    nh = cur_h.copy()
                    nh[j] = np.clip(nh[j] + d * s, 0, 1)
                    nh = project_feasible(nh, ts)
                    c5 = compute_c5(nh)
                    if c5 < best_c5 - 1e-14:
                        cur_h, best_c5 = nh, c5
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            idx = rng.randint(n)
            nh = cur_h.copy()
            nh[idx] = np.clip(nh[idx] + rng.normal(0, 0.02), 0, 1)
            nh = project_feasible(nh, ts)
            if compute_c5(nh) < best_c5:
                cur_h, best_c5 = nh, compute_c5(nh)
            else:
                break
    return cur_h, best_c5
