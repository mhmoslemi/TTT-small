        jm = j - sh
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            contrib[vm] -= h[jm[vm]]
        g += wt * contrib
    return g

def _project(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _lbfgsb_step(h0, dx, n, target, eps, mu, maxiter):
    def obj(x):
        return _smax_cv(x, dx, eps) + mu * (np.sum(x) - target) ** 2
    def grd(x):
        g = _smax_grad(x, dx, eps)
        g += 2.0 * mu * (np.sum(x) - target)
        return g
    res = minimize(obj, h0, jac=grd, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 60})
    return np.clip(res.x, 0.0, 1.0)

def _optimize_seq(h0, dx, n, target, t_end):
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = h0.copy()
    eps_list = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    mu_list = [50.0, 200.0, 800.0]
    for eps in eps_list:
        if time.time() > t_end - 8:
            break
        for mu in mu_list:
            if time.time() > t_end - 3:
                break
            h_new = _lbfgsb_step(h, dx, n, target, eps, mu, 300)
            h_new = _project(h_new, target)
            c = _c5(h_new, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def _binary_ls(h0, dx, n, rng, t_end):
    target = n // 2
    h = np.where(h0 > 0.5, 1.0, 0.0).astype(float)
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx)] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx)] = 1
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    it = 0
    while time.time() < t_end:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        jj = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[jj] = 1.0
        c5_new = _c5(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h[i] = 1.0
            h[jj] = 0.0
        it += 1
    return best_h, best_c5

def _subgrad(h, dx, n, target, t_end, lr=0.08):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    j = np.arange(n)
    for it in range(200000):
        if time.time() > t_end:
            break
        cv = _corr(h, dx)
        mx = cv.max()
        active = np.where(cv >= mx - 1e-9)[0]
        g = np.zeros(n)
        for idx in active:
            sh = idx - (n - 1)
            contrib = np.ones(n)
            jp = j + sh
            vp = (jp >= 0) & (jp < n)
            if vp.any():
                contrib[vp] -= h[jp[vp]]
            jm = j - sh
            vm = (jm >= 0) & (jm < n)
            if vm.any():
                contrib[vm] -= h[jm[vm]]
            g += dx * contrib
        g /= len(active)
        h_new = _project(h - lr * g, target)
        c_new = _c5(h_new, dx)
        c_cur = _c5(h, dx)
        if c_new <= c_cur + 1e-12:
            h = h_new
            lr = min(lr * 1.005, 1.5)
        else:
            lr = max(lr * 0.97, 0.005)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
    return best_h, best_c5

def _make_structured(n, rng, kind):
    h = np.zeros(n)
    if kind == 0:
        h[:n // 2] = 1.0
    elif kind == 1:
        pat = np.array([1, 1, 0, 0, 1, 0])
        h[:] = np.tile(pat, n // len(pat))[:n]
        h = np.where(h > 0.5, 1.0, 0.0)
    elif kind == 2:
        tm = np.array([bin(i).count("1") & 1 for i in range(n)])
        h[:] = tm
    elif kind == 3:
        h[:] = rng.choice([0.0, 1.0], n, p=[0.5, 0.5])
    elif kind == 4:
        h[:] = (np.arange(n) % 7).astype(float)
        h[h < 3] = 1.0
        h[h >= 3] = 0.0
    else:
        h[:] = rng.uniform(0, 1, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target = float(n) / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
    else:
        h0 = rng.uniform(0.0, 1.0, n)
    h0 = _project(h0, target)
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)

    # Phase 1: Main L-BFGS-B optimization from parent (~45% budget)
    t1 = t0 + (deadline - t0) * 0.45
    h1, c1 = _optimize_seq(h0, dx, n, target, t1)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Binary local search from best + structured starts (~20%)
    t2 = t0 + (deadline - t0) * 0.65
    t2_sub = (t2 - t1) / 4.0
    hbs, cbs = _binary_ls(best_h, dx, n, rng, min(t1 + t2_sub, t2))
    if cbs < best_c5:
        best_h, best_c5 = hbs, cbs
    for kind in [1, 2, 4]:
        hr = _make_structured(n, rng, kind)
        hr = _project(hr, target)
        t_sub = min(time.time() + t2_sub, t2)
        hbs2, cbs2 = _binary_ls(hr, dx, n, rng, t_sub)
        if cbs2 < best_c5:
            best_h, best_c5 = hbs2, cbs2

    # Phase 3: Subgradient polish (~10%)
    t3 = t0 + (deadline - t0) * 0.75
    hs, cs = _subgrad(best_h, dx, n, target, t3, lr=0.06)
    if cs < best_c5:
        best_h, best_c5 = hs, cs

    # Phase 4: L-BFGS-B restarts with perturbations (~15%)
    t4 = t0 + (deadline - t0) * 0.92
    while time.time() < t4:
        amp = rng.uniform(0.04, 0.12)
        h_pert = _project(best_h + rng.normal(0, amp, n), target)
        t_sub = min(time.time() + 40.0, t4)
        h_new, c_new = _optimize_seq(h_pert, dx, n, target, t_sub)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new

    # Phase 5: Final fine polish
    if time.time() < deadline - 10:
        t5 = deadline
        hf, cf = _optimize_seq(best_h, dx, n, target, t5)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
        if time.time() < deadline - 3:
            hs2, cs2 = _subgrad(best_h, dx, n, target, deadline, lr=0.03)
            if cs2 < best_c5:
                best_h, best_c5 = hs2, cs2

    return best_h, best_c5, n
