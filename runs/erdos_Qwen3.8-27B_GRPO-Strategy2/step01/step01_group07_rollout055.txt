from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _norm_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _interp_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def _sle_obj(h, alpha, dx, target, pw):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -500.0/alpha, 500.0/alpha)
    sm = (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx
    pen = pw * (np.sum(h) - target)**2
    return sm + pen

def _coord_desc(h, target, rounds=40):
    n = len(h)
    bc = _c5(h)
    for _ in range(rounds):
        best_c = bc
        best_move = None
        for i in range(n):
            for d in (0.01, 0.03, 0.05, 0.1, -0.01, -0.03, -0.05, -0.1):
                vi = h[i] + d
                if vi < 0 or vi > 1:
                    continue
                for j in ((i+1)%n, (i-1+n)%n, (i+n//2)%n):
                    if j == i:
                        continue
                    vj = h[j] - d
                    if vj < 0 or vj > 1:
                        continue
                    h2 = h.copy()
                    h2[i] = vi
                    h2[j] = vj
                    c = _c5(h2)
                    if c < best_c - 1e-12:
                        best_c = c
                        best_move = (i, j, d)
        if best_move:
            h[best_move[0]] += best_move[2]
            h[best_move[1]] -= best_move[2]
            bc = best_c
        else:
            break
    return h, bc

def _simul_anneal(h, target, t_start=0.01, n_iter=5000, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)
    bc = _c5(h)
    h_best = h.copy()
    for it in range(n_iter):
        T = t_start * (1.0 - it / n_iter) + 1e-5
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            continue
        d = rng.choice((0.01, -0.01, 0.02, -0.02, 0.05, -0.05))
        vi = h[i] + d
        vj = h[j] - d
        if vi < 0 or vi > 1 or vj < 0 or vj > 1:
            continue
        h2 = h.copy()
        h2[i] = vi
        h2[j] = vj
        c2 = _c5(h2)
        if c2 < bc or rng.rand() < np.exp(-(c2 - bc) / T):
            h = h2
            bc = c2
            if bc < _c5(h_best):
                h_best = h.copy()
    return h_best, _c5(h_best)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 100.0
_g_dx = 2.0 / 70
_g_target = 35.0
_g_pw = 5000.0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _norm_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _interp_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def _obj_func(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -500.0 / _g_alpha, 500.0 / _g_alpha)
    sm = (m + np.log(np.sum(np.exp(_g_alpha * sh))) / _g_alpha) * _g_dx
    pen = _g_pw * (np.sum(x) - _g_target) ** 2
    return sm + pen

def _coord_desc(h, target, rounds=100, steps=None):
    n = len(h)
    if steps is None:
        steps = (0.01, 0.02, 0.04, 0.07, 0.1, -0.01, -0.02, -0.04, -0.07, -0.1)
    bc = _c5(h)
    for _ in range(rounds):
        best_c = bc
        best_move = None
        for i in range(n):
            for d in steps:
                vi = h[i] + d
                if vi < 0 or vi > 1:
                    continue
                for j in ((i + 1) % n, (i - 1 + n) % n, (i + n // 2) % n, (i + n // 3) % n):
                    if j == i:
                        continue
                    vj = h[j] - d
                    if vj < 0 or vj > 1:
                        continue
                    h2 = h.copy()
                    h2[i] = vi
                    h2[j] = vj
                    c = _c5(h2)
                    if c < best_c - 5e-12:
                        best_c = c
                        best_move = (i, j, d)
        if best_move is not None:
            h[best_move[0]] = h[best_move[0]] + best_move[2]
            h[best_move[1]] = h[best_move[1]] - best_move[2]
            bc = best_c
        else:
            break
    return h, bc

def _simul_anneal(h, target, t_start=0.004, n_iter=30000, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)
    bc = _c5(h)
    h_best = h.copy()
    bc_best = bc
    deltas = (0.01, -0.01, 0.02, -0.02, 0.04, -0.04, 0.07, -0.07)
    for it in range(n_iter):
        T = t_start * (1.0 - it / n_iter) + 1e-6
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            continue
        d = deltas[rng.randint(len(deltas))]
        vi = h[i] + d
        vj = h[j] - d
        if vi < 0 or vi > 1 or vj < 0 or vj > 1:
            continue
        h2 = h.copy()
        h2[i] = vi
        h2[j] = vj
        c2 = _c5(h2)
        if c2 < bc or rng.rand() < np.exp(-(c2 - bc) / max(T, 1e-7)):
            h = h2
            bc = c2
            if bc < bc_best:
                bc_best = bc
                h_best = h.copy()
    return h_best, bc_best

def _multi_restart(n, parent_h, target, seed, time_limit):
    t0 = time.time()
    dx = 2.0 / n
    _g_dx = dx
    _g_target = target
    _g_pw = 5000.0
    bounds = [(0.0, 1.0)] * n
    best_c = _c5(_norm_h(parent_h, target))
    best_h = _norm_h(parent_h, target).copy()
    
    n_restarts = max(2, min(5, int(time_limit / 40.0)))
    for r in range(n_restarts):
        if time.time() - t0 > time_limit:
            break
        if r == 0:
            hc = best_h.copy()
        else:
            rng = np.random.RandomState(seed + r * 1000)
            hc = np.full(n, 0.5)
            for k in range(rng.randint(1, 4)):
                freq = rng.randint(1, max(2, n // 5))
                phase = rng.uniform(0, 2.0 * np.pi)
                amp = rng.uniform(0.05, 0.3)
                hc += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
            hc = _norm_h(hc, target)
        
        for al in [30, 100, 500, 2000, 6000]:
            if time.time() - t0 > time_limit:
                break
            _g_alpha = float(al)
            mi = 150 if n <= 90 else 80
            try:
                res = minimize(_obj_func, hc, method='SLSQP',
                               bounds=bounds,
                               options={'maxiter': mi, 'ftol': 1e-13, 'maxfev': 2000})
                hc = _norm_h(res.x, target)
            except Exception:
                pass
            cn = _c5(hc)
            if cn < best_c:
                best_c = cn
                best_h = hc.copy()
        
        if time.time() - t0 > time_limit * 0.7:
            break
        hc, cg = _coord_desc(hc, target, rounds=150)
        cg = _c5(hc)
        if cg < best_c:
            best_c = cg
            best_h = hc.copy()
        
        if time.time() - t0 > time_limit * 0.85:
            break
        rng_sa = np.random.RandomState(seed + r * 3000 + 7)
        hc, cs = _simul_anneal(hc, target, t_start=0.003, n_iter=25000, rng=rng_sa)
        cs = _c5(hc)
        if cs < best_c:
            best_c = cs
            best_h = hc.copy()
        
        if time.time() - t0 > time_limit * 0.95:
            break
        hc, _ = _coord_desc(hc, target, rounds=80,
                            steps=(0.005, 0.01, 0.02, 0.03, -0.005, -0.01, -0.02, -0.03))
        cn = _c5(hc)
        if cn < best_c:
            best_c = cn
            best_h = hc.copy()
    
    return best_h, best_c

def _final_polish(h, target, time_s=30):
    t0 = time.time()
    best_c = _c5(h)
    best_h = h.copy()
    fine_steps = (0.003, 0.007, 0.01, -0.003, -0.007, -0.01)
    for _ in range(300):
        if time.time() - t0 > time_s:
            break
        h, c = _coord_desc(best_h, target, rounds=50, steps=fine_steps)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    if 'initial_h_values' in globals():

    n_list = [70, 80, 90, 100]

    for n in n_list:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        alloc = min(remaining * 0.55, 300.0)
        
        if n == pn:
            init_h = best_h.copy()
        else:
            init_h = _interp_to_n(parent, n)
            init_h = _norm_h(init_h, n / 2.0)
        
        h_n, c_n = _multi_restart(n, init_h, n / 2.0, seed + n, alloc)
        
        if c_n < best_c5:
            best_c5 = c_n
            best_h = h_n.copy()
            best_n = n
            print(f"n={n} multi: C5={c_n:.6f}")
        else:
            print(f"n={n} multi: C5={c_n:.6f} (no improvement)")
    
    # Final polish on best
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = _final_polish(best_h, best_n / 2.0, min(remaining * 0.7, 120))

    # Enforce exact sum constraint
    h_ret = best_h.copy()
    tgt = best_n / 2.0
    diff = tgt - np.sum(h_ret)

        mask = (h_ret > 1e-10) & (h_ret < 1.0 - 1e-10)

        if cnt > 0:
            h_ret[mask] += diff / cnt
        h_ret = np.clip(h_ret, 0.0, 1.0)
        diff2 = tgt - np.sum(h_ret)
        if abs(diff2) > 1e-14:
            h_ret = h_ret - diff2 / best_n
            h_ret = np.clip(h_ret, 0.0, 1.0)
    
    c5_final = _c5(h_ret)
