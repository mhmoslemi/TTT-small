    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5_val(h, dx):
    return _make_corr(h, dx).max()

def _softmax_obj(x, dx, n, eps, target):
    cv = _make_corr(x, dx)
    mx = cv.max()
    # Soft max for smoothness
    shifted = (cv - mx) / eps
    max_shifted = shifted.max()
    val = mx + eps * (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))))
    return val

    cv = _make_corr(x, dx)
    mx = cv.max()
    shifted = (cv - mx) / eps
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w /= w.sum()
    # Gradient of the soft-max of cv with respect to x
    # cv[j] = sum_i x[i + j - (n-1)] * (1 - x[i]) for valid i
    # dcv[j]/dx[k] = (1-x[k]) if k = i + j - (n-1) for some valid i
    #              = -x[k] if k = i for some valid i where i + j - (n-1) is valid

    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)  # shift in x-units (in sample units)
        # cv[idx] = sum_i x[i+s] * (1-x[i]) for i where 0<=i<n and 0<=i+s<n

        #   contributes (1-x[k]) for k where 0<=k-s<n => s<=k<s+n
        # Second term: k = i => i+s = k+s, need 0<=i<n and 0<=k+s<n => -s<=k<n-s
        vm1 = (j >= s) & (j < s + n) & (j >= 0) & (j < n)
        if vm1.any():

        vm2 = (j >= -s) & (j < n - s) & (j >= 0) & (j < n)

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, tgt)

def _active_grad_step(h, dx, n, tgt, k, lr):
    cv = _corr_of(h, dx)
    top_idx = np.argsort(cv)[-k:]
    g = np.zeros(n)
    j = np.arange(n)
    cnt = 0
    for idx in top_idx:
        if cv[idx] < cv.max() - 1e-8:
            continue
        sh = idx - (n - 1)
        v1 = (j >= sh) & (j < sh + n)
        if v1.any():
            g[v1] += dx * (1.0 - h[j[v1]])
        v2 = (j >= -sh) & (j < n - sh)
        if v2.any():
            g[v2] -= dx * h[j[v2]]
        cnt += 1
    if cnt == 0:
        return h
    g /= cnt
    return _proj(h - lr * g, tgt)

def _bit_flip_search(h, dx, n, tgt, rng, max_sweeps=500):
    hb = (h > 0.5).astype(float)
    diff = tgt - hb.sum()
    if diff > 0:
        zeros = np.where(hb == 0)[0]
        for _ in range(int(diff)):
            best_i = -1; best_v = 1e9
            for i in zeros[:50]:
                hb[i] = 1; v = _c5_of(hb, dx); hb[i] = 0
                if v < best_v: best_v = v; best_i = i
            if best_i >= 0:
                hb[best_i] = 1; zeros = np.where(hb == 0)[0]
            else:
                break
    elif diff < 0:
        ones = np.where(hb == 1)[0]
        for _ in range(int(-diff)):
            best_i = -1; best_v = 1e9
            for i in ones[:50]:
                hb[i] = 0; v = _c5_of(hb, dx); hb_i = hb[i]
                hb[i] = 1
                if v < best_v: best_v = v; best_i = i
            if best_i >= 0:
                hb[best_i] = 0; ones = np.where(hb == 1)[0]
            else:
                break
    best_c = _c5_of(hb, dx)
    best_hb = hb.copy()
    t_end = time.time() + 60
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        # Try random subset of moves for speed
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            a = int(rng.choice(ones))
            b = int(rng.choice(zeros))
            hb[a] = 0; hb[b] = 1
            c_new = _c5_of(hb, dx)
            if c_new < best_c - 1e-12:
                best_c = c_new; best_hb = hb.copy(); improved = True
            else:
                hb[a] = 1; hb[b] = 0
        if not improved:
            break
    return best_hb, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = _proj(h0, tgt)

    best_h = h0.copy()
    best_c5 = _c5_of(h0, dx)

    # Phase 1: SLSQP with decreasing eps on parent
    eps_seq = [0.02, 0.005, 0.001, 0.0003, 0.0001, 0.00003, 0.00001]
    t_p1 = t0 + (deadline - t0) * 0.45
    for eps in eps_seq:
        if time.time() > t_p1 - 8:
            break
        try:
            h_new = _slsqp(best_h, dx, n, tgt, eps, maxiter=500)
        except Exception:
            break
        c_new = _c5_of(h_new, dx)
        if c_new < best_c5:
            best_c5 = c_new; best_h = h_new

    # Phase 2: Perturbation restarts
    t_p2 = t0 + (deadline - t0) * 0.72
    attempt = 0
    while time.time() < t_p2 - 8:
        attempt += 1
        amp = 0.03 + 0.12 * (attempt % 5) / 5.0
        h_p = _proj(best_h + rng.normal(0, amp, n), tgt)
        if _c5_of(h_p, dx) < best_c5:
            best_c5 = _c5_of(h_p, dx); best_h = h_p
        t_sub = min(time.time() + 18, t_p2 - 4)
        for eps in [0.0005, 0.0001]:
            if time.time() > t_sub:
                break
            try:
                h_t = _slsqp(h_p, dx, n, tgt, eps, maxiter=250)
            except Exception:
                break
            c_t = _c5_of(h_t, dx)
            if c_t < best_c5:
                best_c5 = c_t; best_h = h_t

    # Phase 3: Active-set gradient polish
    if time.time() < deadline - 25:
        lr = 0.05
        for it in range(3000):
            if time.time() > deadline - 20:
                break
            k = 3 if it < 1000 else 5
            h_a = _active_grad_step(best_h, dx, n, tgt, k, lr)
            c_a = _c5_of(h_a, dx)
            if c_a < best_c5:
                best_c5 = c_a; best_h = h_a; lr = min(lr * 1.02, 0.3)
            else:
                lr = max(lr * 0.95, 0.005)

    # Phase 4: Binary local search
    if time.time() < deadline - 15:
        h_bin, c_bin = _bit_flip_search(best_h, dx, n, tgt, rng, max_sweeps=400)
        if c_bin < best_c5:
            best_c5 = c_bin; best_h = h_bin

    # Phase 5: Final SLSQP polish from both
    if time.time() < deadline - 4:
        for start in [best_h, h_bin]:
            if time.time() > deadline - 2:
                break
            for eps in [0.00003, 0.00001]:
                if time.time() > deadline - 1:
                    break
                try:
                    h_f = _slsqp(start, dx, n, tgt, eps, maxiter=200)
                except Exception:
                    break
                c_f = _c5_of(h_f, dx)
                if c_f < best_c5:
                    best_c5 = c_f; best_h = h_f

    return best_h, best_c5, n
