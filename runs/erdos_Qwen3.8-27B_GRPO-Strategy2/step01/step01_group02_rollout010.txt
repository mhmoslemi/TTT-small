def project_h(h, n):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.array(h, dtype=float)
    # Bisection for lambda
    lo, hi = -1.0, 1.0

        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def _obj_exact(x, n):
    return _compute_c5(x, n)

def _obj_lse(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def _cons_eq(x, n):
    return np.sum(x) - n / 2.0

def _subgrad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.where(corr > mx - 1e-9)[0]
    g = np.zeros(n)
    for k in active:
        off = k - n + 1
        lo = max(0, -off)
        hi = min(n, n - off)
        if lo < hi:
            g[lo:hi] += 1.0 - h[lo + off:hi + off]
        lo2 = max(0, off)
        hi2 = min(n, n + off)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 - off:hi2 - off]
    g /= len(active)
    return g

def _slsqp(h0, n, obj_kind, beta, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    if obj_kind == 0:
        fn = _obj_exact
        args = (n,)
    else:
        fn = _obj_lse
        args = (n, beta)
    cons = [{'type': 'eq', 'fun': _cons_eq, 'args': (n,)}]
    try:
        res = minimize(fn, h0, args=args, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_p = _project(res.x, n)
        return h_p, _compute_c5(h_p, n)
    except Exception:
        return h0, _compute_c5(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)

    h_best = _project(h_best, n)
    best_c5 = _compute_c5(h_best, n)

    # Phase A: Continuation method (LSE with increasing beta)
    for beta in [3, 8, 15, 30, 60]:
        if time.time() - t0 > budget_s - 80:
            break
        h_try, c_try = _slsqp(h_best, n, 1, beta, 400, 1e-12)
        if c_try < best_c5:
            h_best, best_c5 = h_try, c_try

    # Phase B: Exact objective SLSQP
    if time.time() - t0 < budget_s - 60:
        h_best, best_c5 = _slsqp(h_best, n, 0, 0, 500, 1e-14)

    # Phase C: Iterated Local Search with subgradient + SLSQP
    h_curr = h_best.copy()
    vel = np.zeros(n)
    stall = 0
    it = 0
    best_time = t0

    while time.time() - t0 < budget_s - 40:
        it += 1
        improved = False

        # Subgradient steps (cheap, handles non-smoothness)
        for _ in range(5):
            g = _subgrad(h_curr, n)
            alpha = 0.02 / (1.0 + 0.02 * stall)
            if stall > 15:
                alpha *= 2.0
            vel = 0.75 * vel + 0.25 * g
            h_new = _project(h_curr - alpha * vel, n)
            c_new = _compute_c5(h_new, n)
            # SA acceptance
            T = max(0.0005, 0.02 * 0.998 ** it)
            if c_new < best_c5 - 1e-12:
                h_best, best_c5 = h_new, c_new
                h_curr = h_best.copy()
                vel = np.zeros(n)
                stall = 0
                best_time = time.time()
                improved = True
            elif rng.random() < np.exp(-(c_new - best_c5) / T):
                h_curr = h_new
            else:
                h_curr = h_new
                vel *= 0.5

        # Periodic SLSQP polish
        if it % 8 == 0 and time.time() - t0 < budget_s - 60:
            hc, cc = _slsqp(h_best, n, 0, 0, 300, 1e-14)
            if cc < best_c5 - 1e-12:
                h_best, best_c5 = hc, cc
                h_curr = h_best.copy()
                vel = np.zeros(n)
                stall = 0
                best_time = time.time()
                improved = True

        # Periodic LSE refinement with high beta
        if it % 12 == 0 and time.time() - t0 < budget_s - 50:
            beta_h = 40 + 20 * (it % 6)
            hc2, cc2 = _slsqp(h_best, n, 1, beta_h, 250, 1e-12)
            if cc2 < best_c5 - 1e-12:
                h_best, best_c5 = hc2, cc2
                h_curr = h_best.copy()
                vel = np.zeros(n)
                stall = 0
                best_time = time.time()
                improved = True

        # ILS perturbation every 15 iters
        if it % 15 == 0 and time.time() - t0 < budget_s - 70:
            if stall < 5:
                k_p = 2
                sig = 0.12
            elif stall < 15:
                k_p = 4
                sig = 0.25
            else:
                k_p = min(8, n // 8)
                sig = 0.4
            h_pert = h_best.copy()
            idx = rng.choice(n, size=k_p, replace=False)
            h_pert[idx] = rng.uniform(0.0, 1.0, k_p)
            h_pert = _project(h_pert, n)
            hc3, cc3 = _slsqp(h_pert, n, 0, 0, 250, 1e-13)
            if cc3 < best_c5 - 1e-12:
                h_best, best_c5 = hc3, cc3
                h_curr = h_best.copy()
                vel = np.zeros(n)
                stall = 0
                best_time = time.time()
                improved = True

        # Binarization + re-optimize every 25 iters
        if it % 25 == 0 and time.time() - t0 < budget_s - 60:
            h_bin = (h_best > 0.5).astype(float)
            h_bin = _project(h_bin, n)
            hc4, cc4 = _slsqp(h_bin, n, 0, 0, 350, 1e-14)
            if cc4 < best_c5 - 1e-12:
                h_best, best_c5 = hc4, cc4
                h_curr = h_best.copy()
                vel = np.zeros(n)
                stall = 0
                best_time = time.time()
                improved = True

        # Large structural move every 40 iters
        if it % 40 == 0 and time.time() - t0 < budget_s - 80:
            h_flip = 1.0 - h_best
            h_flip = _project(h_flip, n)
            hc5, cc5 = _slsqp(h_flip, n, 0, 0, 300, 1e-13)
            if cc5 < best_c5 - 1e-12:
                h_best, best_c5 = hc5, cc5
                h_curr = h_best.copy()
                stall = 0
                best_time = time.time()
                improved = True

        if not improved:
            stall += 1

        # Reheat if stuck long
        if stall > 40:
            h_curr = _project(h_best + rng.normal(0, 0.15, n), n)
            vel = np.zeros(n)
            stall = 0

    # Final polish: multiple SLSQP passes
    if time.time() - t0 < budget_s - 15:
        for _ in range(3):
            if time.time() - t0 > budget_s - 10:
                break
            hc, cc = _slsqp(h_best, n, 0, 0, 500, 1e-15)
            if cc < best_c5:
                h_best, best_c5 = hc, cc

    # Try one more from binarized version
    if time.time() - t0 < budget_s - 10:
        h_b2 = (h_best > 0.5).astype(float)
        h_b2 = _project(h_b2, n)
        hc_f, cc_f = _slsqp(h_b2, n, 0, 0, 400, 1e-15)
        if cc_f < best_c5:
            h_best, best_c5 = hc_f, cc_f

    h_best = _project(h_best, n)
    best_c5 = _compute_c5(h_best, n)
    return (h_best, best_c5, n)
