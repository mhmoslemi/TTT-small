        hc = _slsqp(best_h, dx, n, target, alpha, maxiter=300)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    
    # Phase 2: Subgradient polish
    if elapsed() < budget_s - 500:
        hc, c5 = _subgrad_polish(best_h, dx, n, target, steps=3000, lr0=0.01)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    
    # Phase 3: Exact active-constraint SLSQP (iterative)
    for _ in range(5):
        if elapsed() > budget_s - 400:
            break
        hc = _exact_active_slsqp(best_h, dx, n, target, maxiter=300)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    
    # Phase 4: Pair coordinate descent (exact)
    if elapsed() < budget_s - 200:
        hc, c5 = _pair_cd(best_h, dx, n, target, rng, max_passes=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    
    # Phase 5: More SLSQP + subgradient cycles
    for _ in range(3):
        if elapsed() > budget_s - 150:
            break
        hc = _slsqp(best_h, dx, n, target, 2000, maxiter=200)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
        hc, c5 = _subgrad_polish(best_h, dx, n, target, steps=2000, lr0=0.005)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    
    # Phase 6: Restart with perturbation
    if elapsed() < budget_s - 200:
        for sigma in [0.1, 0.2, 0.05, 0.15]:
            if elapsed() > budget_s - 150:
                break
            h_pert = _repair(best_h + sigma * rng.randn(n), n)
            hc = _slsqp(h_pert, dx, n, target, 500, maxiter=300)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
            hc, c5 = _subgrad_polish(best_h, dx, n, target, steps=2000, lr0=0.008)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
            hc, c5 = _pair_cd(best_h, dx, n, target, rng, max_passes=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
    
    # Phase 7: Final fine coordinate descent
    if elapsed() < budget_s - 50:
        hc, c5 = _pair_cd(best_h, dx, n, target, rng, max_passes=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    
    # Phase 8: Final SLSQP squeeze
    if elapsed() < budget_s - 20:
        for alpha in [3000, 5000, 8000]:
            if elapsed() > budget_s - 15:
                break
            try:
                hc = _slsqp(best_h, dx, n, target, alpha, maxiter=300)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass
    
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
