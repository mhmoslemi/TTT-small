            g[:n-mm] += w[k] * (-h[mm:])    # second term
    g *= dx
    # Step
    h_new = h - step * g
    h_new = project(h_new, n)
    return h_new

from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    for _ in range(50):
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def smooth_max_obj(h, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    # log-sum-exp
    return float((cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx)

def lp_step(h, n, dx, tol_frac=0.02):
    """One LP-based step: linearize F at h, solve LP, return new h."""
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    t_star = c.max()
    # Active lags: within tol_frac * range of max
    active = np.where(c > t_star - tol_frac * (t_star - c.min()))[0]
    if len(active) < 3:
        active = np.where(c > t_star - 0.005)[0]
    
    m_vals = np.arange(-(n-1), n)
    A_rows = []
    for k in active:
        m = m_vals[k]
        g = np.zeros(n)
        if m >= 0:
            if m < n:
                g[:n-m] += (1.0 - h[m:]) * dx
            if m > 0:
                g[m:] += (-h[:-m]) * dx
        else:
            mm = -m
            if mm < n:
                g[mm:] += (1.0 - h[:-mm]) * dx
            if mm > 0:
                g[:n-mm] += (-h[mm:]) * dx
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return h, t_star
    return res.x[:n].copy(), res.x[-1]

def coord_descent_fine(h, n, max_sweeps=20):
    """Coordinate descent: for each i, try many values of h_i."""
    h = h.copy()
    target = n * 0.5
    best = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            # Try a grid of values for h_i
            candidates = np.linspace(0, 1, 11)
            candidates = np.unique(np.concatenate([candidates, [old]]))
            best_local = best
            best_cand = old
            for cand in candidates:
                if abs(cand - old) < 1e-10:
                    continue
                h[i] = cand
                diff = target - h.sum()
                # Distribute correction to all other elements
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                h[mask] += diff / (n-1)
                h = np.clip(h, 0, 1)
                # Re-adjust
                diff2 = target - h.sum()
                if abs(diff2) > 1e-10:
                    mask2 = (h < 1) if diff2 > 0 else (h > 0)
                    mask2[i] = False
                    if mask2.any():
                        h[mask2] += diff2 / mask2.sum()
                        h = np.clip(h, 0, 1)
                val = compute_c5(h)
                if val < best_local:
                    best_local = val
                    best_cand = cand
            if best_cand != old:
                h[i] = best_cand
                diff = target - h.sum()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                h[mask] += diff / (n-1)
                h = np.clip(h, 0, 1)
                best = best_local
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    h = repair(h, n)
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    if abs(target - h.sum()) > 1e-12:
        idx = int(np.argmin(h)) if h.sum() < target else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + (target - h.sum()), 0.0, 1.0)
    return h


def smooth_max_obj(h, dx, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    return float((cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * dx)


def lp_refine(h, n, dx, n_active=None):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    t_star = c.max()
    if n_active is None:
        n_active = 30
    idx = np.argsort(c)[::-1][:n_active]
    m_vals = np.arange(-(n - 1), n)
    G = np.zeros((n_active, n))
    b = np.zeros(n_active)
    for i, k in enumerate(idx):
        m = m_vals[k]
        g = np.zeros(n)
        if m >= 0:
            if m < n:
                g[:n - m] += (1.0 - h[m:]) * dx
            if m > 0:
                g[m:] += (-h[:-m]) * dx
        else:
            mm = -m
            if mm < n:
                g[mm:] += (1.0 - h[:-mm]) * dx
            if mm > 0:
                g[:n - mm] += (-h[mm:]) * dx
        G[i] = g
        b[i] = -c[k] + g.dot(h)
    A_ub = np.hstack([G, -np.eye(n_active)])
    b_ub = b
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return h, t_star
    return res.x[:n].copy(), res.x[-1]


def coord_search(h, n, max_sweeps=15, grid=9):
    h = h.copy()
    target = n * 0.5
    best = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            cands = np.linspace(0, 1, grid)
            cands = np.concatenate([cands, [old, 0.5]])
            best_local = best
            best_cand = old
            for cand in cands:
                if abs(cand - old) < 1e-9:
                    continue
                h[i] = cand
                diff = target - h.sum()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                h[mask] += diff / (n - 1)
                h = np.clip(h, 0, 1)
                d2 = target - h.sum()
                if abs(d2) > 1e-12:
                    m2 = (h < 1) if d2 > 0 else (h > 0)
                    m2[i] = False
                    if m2.any():
                        h[m2] += d2 / m2.sum()
                        h = np.clip(h, 0, 1)
                v = compute_c5(h)
                if v < best_local:
                    best_local = v
                    best_cand = cand
            if best_cand != old:
                h[i] = best_cand
                diff = target - h.sum()
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                h[mask] += diff / (n - 1)
                h = np.clip(h, 0, 1)
                best = best_local
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return repair(h, n), compute_c5(repair(h, n))

                    best_h = h_new

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
