                for delta in [0.001, -0.001, 0.003, -0.003, 0.008, -0.008, 0.015, -0.015]:
                    hp = best_h.copy()
                    hp[idx] += delta
                    if hp[idx] < 0 or hp[idx] > 1:
                        continue
                    diff = n / 2.0 - np.sum(hp)
                    if abs(diff) > 1e-10:
                        others = [j for j in range(n) if j != idx]
                        oa = np.array(others)
                        dist = np.abs(hp[oa] - 0.5)
                        j = oa[np.argmin(dist)]
                        hp[j] -= diff
                        if hp[j] < 0 or hp[j] > 1:
                            continue
                    hp = repair(hp, n)
                    c_new = compute_c5(hp, n)
                    if c_new < best_c5 - 1e-15:
                        best_h, best_c5 = hp, c_new
                        improved = True
                        break
                if not improved:
                    break

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
