```python
def project_feasible(h, ts):
    """Project onto [0,1]^n ∩ {sum = ts}"""
    h = h.copy()
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0, 1)

def lse_grad(h, beta):
    """Gradient of (1/beta)*log(sum exp(beta*corr_k))*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    bc_max = bc.max()
    p = np.exp(bc - bc_max)
    p /= p.sum()
    
    grad = np.zeros(n)
    for l in range(-n+1, n):
        w = p[l + n - 1]
        if w < 1e-10:
            continue
        # d corr[l]/d h[j] = (1-h[j+l])*I(0<=j+l<n) - h[j-l]*I(0<=j-l<n)
        # First term: j in [max(0,-l), min(n-1,n-1-l)]
        a1 = max(0, -l)
        b1 = min(n-1, n-1-l)
        if a1 <= b1:
            grad[a1:b1+1] += w * (1.0 - h[a1+l:b1+1+l])
        # Second term: j in [max(0,l), min(n-1,n-1+l)]  
        a2 = max(0, l)
        b2 = min(n-1, n-1+l)
        if a2 <= b2:
            grad[a2:b2+1] -= w * h[a2-l:b2+1-l]
    
    return dx * grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = h.copy()
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    bc_max = bc.max()
    p = np.exp(bc - bc_max)
    p /= p.sum()
    grad = np.zeros(n)
    for l in range(-n + 1, n):
        w = p[l + n - 1]
        if w < 1e-10:
            continue
        a1 = max(0, -l)
        b1 = min(n - 1, n - 1 - l)
        if a1 <= b1:
            grad[a1:b1+1] += w * (1.0 - h[a1+l:b1+1+l])
        a2 = max(0, l)
        b2 = min(n - 1, n - 1 + l)
        if a2 <= b2:
            grad[a2:b2+1] -= w * h[a2-l:b2+1-l]
    return dx * grad

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    return (bc.max() + np.log(np.sum(np.exp(bc - bc.max()))) / beta) * dx

def nesterov_smooth(h, ts, beta, max_iter, t_limit):
    """Nesterov accelerated gradient with projection"""
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    t_k = 1.0
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_cur + (t_k - 1.0) / t_next * (h_cur - h_prev)
        y = project_feasible(y, ts)
        
        grad = lse_grad(y, beta)
        f_y = smooth_obj(y, beta)
        
        # Backtracking
        alpha = 2.0 / (beta * 2.0)  # step size
        h_new = h_cur
        for _ in range(30):
            h_cand = y - alpha * grad
            h_cand = project_feasible(h_cand, ts)
            f_cand = smooth_obj(h_cand, beta)
            if f_cand <= f_y - 1e-4 * alpha * np.dot(grad, grad):
                h_new = h_cand
                break
            alpha *= 0.5
        
        h_prev = h_cur
        h_cur = h_new
        t_k = t_next
    return h_cur

def slsqp_polish(h, ts, beta, maxiter):
    n = len(h)
    def obj(x):
        return smooth_obj(x, beta)
    def obj_grad(x):
        return lse_grad(x, beta)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except:
        pass
    return project_feasible(h, ts)

def binary_local_search(h, ts, max_sweeps, time_limit):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            for t_val in (0.0, 1.0):
                old = h[j]
                if abs(t_val - old) < 1e-8:
                    continue
                h[j] = t_val
                h = project_feasible(h, ts)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    h[j] = old
                    h = project_feasible(h, ts)
        if not improved:
            break
    return h, best_c5
```

```python
    h = h.copy()
    for _ in range(20):
```
```python
        np.clip(h, 0.0, 1.0, out=h)
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break

        else:
            idx = np.where(h > 1e-10)[0]

            break

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m))) / beta) * dx

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    p = np.exp(bc - bc.max())
    p /= p.sum()
    grad = np.zeros(n)

        w = p[l + n - 1]
        if w < 1e-10:

        if a1 <= b1:

        a2, b2 = max(0, l), min(n - 1, n - 1 + l)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_eq_ts = [0.0]
_sb = [1.0]

def _eq_con(x):
    return float(np.sum(x) - _eq_ts[0])

def _sb_jac(x):
    return lse_grad(x, _sb[0])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = h.copy()
    for _ in range(20):
        np.clip(h, 0.0, 1.0, out=h)
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = bc.max()
    return (m + np.log(np.sum(np.exp(bc - m))) / beta) * dx

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    p = np.exp(bc - bc.max())
    p /= p.sum()
    grad = np.zeros(n)
    for l in range(1 - n, n):
        w = p[l + n - 1]
        if w < 1e-10:
            continue
        a1, b1 = max(0, -l), min(n - 1, n - 1 - l)
        if a1 <= b1:
            grad[a1:b1 + 1] += w * (1.0 - h[a1 + l:b1 + 1 + l])
        a2, b2 = max(0, l), min(n - 1, n - 1 + l)
        if a2 <= b2:
            grad[a2:b2 + 1] -= w * h[a2 - l:b2 + 1 - l]
    return dx * grad

def accel_grad(h, ts, beta, max_iter, t_limit):
    hc = h.copy()
    hp = h.copy()
    tk = 1.0
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = project_feasible(hc + (tk - 1.0) / tn * (hc - hp), ts)
        g = lse_grad(y, beta)
        fy = lse_smooth(y, beta)
        gg = np.dot(g, g)
        if gg < 1e-18:
            break
        alpha = 1.0 / max(beta, 1.0)
        hn = y.copy()
        for _ in range(30):
            cand = project_feasible(y - alpha * g, ts)
            if lse_smooth(cand, beta) <= fy - 1e-4 * alpha * gg:
                hn = cand
                break
            alpha *= 0.5
        hp, hc, tk = hc, hn, tn
    return hc

def slsqp_polish(h, ts, beta, maxiter):
    n = len(h)
    _eq_ts[0] = ts
    _sb[0] = beta
    try:
        res = minimize(lse_smooth, h, jac=_sb_jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _eq_con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def bin_local_search(h, ts, max_sweeps, t_limit):
    n = len(h)
    bc5 = compute_c5(h)
    t0 = time.time()
    for _ in range(max_sweeps):
        if time.time() - t0 > t_limit:
            break
        imp = False
        for j in range(n):
            for tv in (0.0, 1.0):
                ov = h[j]
                if abs(tv - ov) < 1e-8:
                    continue
                h[j] = tv
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < bc5 - 1e-14:
                    bc5, imp = c, True
                else:
                    h[j] = ov
                    h = project_feasible(h, ts)
        if not imp:
            break
    return h, bc5

def cont_local_search(h, ts, max_sweeps, t_limit, step):
    n = len(h)
    bc5 = compute_c5(h)
    t0 = time.time()
    for _ in range(max_sweeps):
        if time.time() - t0 > t_limit:
            break
        imp = False
        for j in range(n):

                h[j] = nv
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < bc5 - 1e-14:
                    bc5, imp = c, True
                else:
                    h[j] = ov
                    h = project_feasible(h, ts)
        if not imp:
            break
    return h, bc5

def swap_search(h, ts, n_trials, t_limit, rng):
    n = len(h)
    bc5 = compute_c5(h)
    t0 = time.time()
    for _ in range(n_trials):
        if time.time() - t0 > t_limit:
            break
        imp = False
        for _ in range(n):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < bc5 - 1e-14:
                bc5, imp = c, True
            else:
                h[i], h[j] = h[j], h[i]
        if not imp:
            break
    return h, bc5

def make_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def optimize_at_n(h, ts, n, t_start, budget, rng):
    best_h, best_c5 = h.copy(), compute_c5(h)
    for beta in [15, 50, 150, 500, 1500]:
        if budget - (time.time() - t_start) < 10:

        il = min(budget - time.time() + t_start - 5, 35)
        if il < 2:
            break
        hh = accel_grad(h, ts, float(beta), 500, il)
        hh = project_feasible(hh, ts)
        cc = compute_c5(hh)
        if cc < best_c5:
            best_c5, best_h = cc, hh.copy()
        h = hh
    if budget - (time.time() - t_start) > 18:
        hp = slsqp_polish(best_h.copy(), ts, 500.0, 60)
        cp = compute_c5(hp)
        if cp < best_c5:

    if budget - (time.time() - t_start) > 15:
        hl, cl = bin_local_search(best_h.copy(), ts, 60,
                                  min(budget - time.time() + t_start - 3, 60))
        if cl < best_c5:
            best_c5, best_h = cl, hl.copy()
    return best_h, best_c5
```
