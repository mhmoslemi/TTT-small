import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    shifted = np.clip((corr - m) * beta, -500, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Swap local search (systematic, more thorough than binary flip)
    if remaining() > 25:
        h_sw = np.round(best_h).clip(0, 1)
        h_sw, c5_sw = swap_local_search(h_sw, n, ts, time_limit=min(25, remaining() - 15))
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    # Phase 5: Fractional local search for fine-tuning
    if remaining() > 20:
        h_fr, c5_fr = frac_local_search(best_h, n, ts, time_limit=min(20, remaining() - 10))
        if c5_fr < best_c5:
            best_c5 = c5_fr
            best_h = h_fr.copy()

    # Phase 6: Final binary polish
    if remaining() > 15:
        h_final = np.round(best_h).clip(0, 1)
        h_final = project_feasible(h_final, ts)
        h_final, c5_final = binary_local_search(h_final, n, ts, time_limit=12)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: Try smaller n (less discretization overfit)
    if remaining() > 50:
        n2 = min(100, (n_init // 2) * 2)
        ts2 = n2 / 2.0
        xo = np.linspace(0, 2, n_init)
        xn2 = np.linspace(0, 2, n2)
        h2 = np.interp(xn2, xo, best_h)
        h2 = np.round(h2).clip(0, 1)
        h2 = project_feasible(h2, ts2)
        h2 = slsqp_opt(h2, n2, ts2, 30.0, maxiter=250)
        h2b = np.round(h2).clip(0, 1)
        h2b = project_feasible(h2b, ts2)
        h2b, c5_2 = binary_local_search(h2b, n2, ts2, time_limit=min(25, remaining() - 15))
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2b.copy()

    # Final
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
