def project_box_sum(v, target_sum):
    """Project v onto {0<=x<=1, sum(x)=target_sum}"""
    if target_sum <= 0: return np.zeros_like(v)
    if target_sum >= len(v): return np.ones_like(v)
    
    # Bisection on lambda
    # p(lambda) = sum clip(v[i] - lambda, 0, 1)
    # Find lambda such that p(lambda) = target_sum
    lo, hi = v.max() - target_sum + 1, v.min() - 1  # wide bounds
    # Actually: p is decreasing in lambda

    lo = v.min() - 1.0

        p = np.sum(np.clip(v - mid, 0, 1))
        if p > target_sum:

        else:
            hi = mid

    u = 1.0 - h

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    return np.clip(v - 0.5*(lo+hi), 0.0, 1.0)

def compute_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = corr / tau
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    n1 = n - 1
    t = np.arange(n)[:, None]
    m = np.arange(2*n-1)[None, :]
    idx1 = np.clip(t - m + n1, 0, n-1)
    idx2 = np.clip(t + m - n1, 0, n-1)
    mask1 = (t - m + n1 >= 0) & (t - m + n1 < n)
    mask2 = (t + m - n1 >= 0) & (t + m - n1 < n)
    w_b = w[None, :]
    term1 = ((1.0 - h)[idx1] * w_b * mask1).sum(axis=1)
    term2 = (h[idx2] * w_b * mask2).sum(axis=1)
    return term1 - term2

            g = compute_grad(h, tau)
            h = project_box_sum(h - lr * g, n / 2.0)
            v = compute_c5(h, dx)
            if v < best_val:

                best_h = h.copy()
            # Adaptive lr
            if i % 100 == 99:

                pass
    return best_h, best_val

    xd = np.linspace(0.0, 2.0, n_dst)
    f = interp1d(xs, h_src, kind='linear', fill_value='extrapolate')
    h_dst = f(xd)
