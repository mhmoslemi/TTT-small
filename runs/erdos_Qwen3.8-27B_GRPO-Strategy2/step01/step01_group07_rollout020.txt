        if time.time() > t_end - 35:
            break
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        # Interpolation init
        x_p = np.linspace(0.0, 2.0, len(parent))
        x_n = np.linspace(0.0, 2.0, n)
        h_interp = project(np.interp(x_n, x_p, parent), target)

        # Random sinusoid init for diversity
        rng = np.random.RandomState(seed + n * 7 + 1)
        h_rand = np.full(n, 0.5)
        for _ in range(4):
            f = rng.randint(1, max(2, n // 4))
            p = rng.uniform(0.0, 2.0 * np.pi)
            a = rng.uniform(0.08, 0.35)
            h_rand += a * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
        h_rand = project(h_rand, target)

        starts = [h_interp]
        if c5_val(h_rand) < c5_val(h_interp) + 0.03:
            starts.append(h_rand)

        for h0 in starts:
            if time.time() > t_end - 12:
                break
            # SLSQP phase
            hc = slsqp_round(h0, n, target, bounds, cons, t_end)
            c = c5_val(hc)
            if c < best_c5:
                best_c5 = c
                best_h = hc.copy()
                best_n = n

            # Subgradient polish
            if time.time() < t_end - 18:
                nsg = 250 if n <= 100 else 150
                hc, cs = subgrad_opt(hc, target, nsg, 0.04, t_end - 15)
                if cs < best_c5:
                    best_c5 = cs
                    best_h = hc.copy()
                    best_n = n

            # Fine coordinate descent
            if time.time() < t_end - 5:
                hc, cc = fine_cd(hc, target, t_end - 2)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = hc.copy()
                    best_n = n

    # Final constraint fix
    target = best_n / 2.0
    best_h = project(best_h, target)
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-13:
        mask = (best_h > 0.01) & (best_h < 0.99)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / float(cnt)
        best_h = np.clip(best_h, 0.0, 1.0)
        diff2 = target - np.sum(best_h)
        if abs(diff2) > 1e-14:
            best_h = best_h - diff2 / float(best_n)
            best_h = np.clip(best_h, 0.0, 1.0)

    c5f = c5_val(best_h)
    if c5f < best_c5:
        best_c5 = c5f

    return best_h, best_c5, best_n
