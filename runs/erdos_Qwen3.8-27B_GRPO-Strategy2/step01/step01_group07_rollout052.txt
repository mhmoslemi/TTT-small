        if gnorm < 1e-10:
            break
        grad = grad / gnorm * 0.001
        best_h = h
        best_c = _c5(h)
        for thunk in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = _project(h - thunk * grad, target)
            c_try = _c5(h_try)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_h = h_try
        h = best_h
    return h, _c5(h)

def _local_search(h, target, max_rounds=200):
    n = len(h)
    best_c = _c5(h)
    deltas = (0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.015, -0.015)
    for rd in range(max_rounds):
        grad, _ = _smooth_max_grad(h, 500)
        order = np.argsort(-np.abs(grad))
        improved = False
        for ii in range(min(30, n)):
            i = order[ii]
            for d in deltas:
                h2 = h.copy()
                h2[i] += d
                if h2[i] < 0 or h2[i] > 1:
                    continue
                h2 = _project(h2, target)
                c2 = _c5(h2)
                if c2 < best_c - 1e-11:
                    h = h2
                    best_c = c2
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try pair moves
            pair_improved = False
            for ii in range(min(15, n)):
                i = order[ii]
                for j in [(i + 1) % n, (i + n // 3) % n, (i + n // 2) % n]:
                    if j == i:
                        continue
                    for d in (0.01, -0.01):
                        h2 = h.copy()
                        h2[i] += d
                        h2[j] -= d
                        if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                            continue
                        h2 = _project(h2, target)
                        c2 = _c5(h2)
                        if c2 < best_c - 1e-11:
                            h = h2
                            best_c = c2
                            pair_improved = True
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    return h, best_c

def _upscale(parent, n_new):
    n_old = len(parent)
    x_old = np.arange(n_old, dtype=float)
    x_new = (np.arange(n_new, dtype=float) * n_old / n_new)
    h_new = np.interp(x_new, x_old, parent)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = len(parent)

    n_targets = [70, 80, 90, 100, 110]
    for n in n_targets:
        if budget_s - (time.time() - t0) < 40:
            break
        target = n / 2.0
        h_base = parent.copy() if n == len(parent) else _upscale(parent, n)
        h_base = _project(h_base, target)
        n_restarts = 4 if n <= 90 else 3

        for r in range(n_restarts):
            if budget_s - (time.time() - t0) < 20:
                break
            rng = np.random.RandomState(seed * 31 + n * 7 + r)
            if r == 0 and n == len(parent):
                h0 = parent.copy()
            elif r == 0:
                h0 = h_base.copy()
            else:
                h0 = h_base + rng.uniform(-0.03, 0.03, n)
                h0 = _project(h0, target)

            # Multiscale gradient descent
            h_cur = h0
            for (a_s, a_e, n_it) in [(10, 100, 800), (50, 500, 1500), (200, 3000, 2000)]:
                if budget_s - (time.time() - t0) < 10:
                    break
                h_cur, c_cur = _grad_descent(h_cur, target, n_iter=n_it,
                                             alpha_start=a_s, alpha_end=a_e)
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print(f"  GD n={n} r={r} a=[{a_s},{a_e}]: C5={c_cur:.6f}")

            # Local search polish
            if budget_s - (time.time() - t0) > 12:
                h_ls, c_ls = _local_search(h_cur, target, max_rounds=150)
                if c_ls < best_c5:
                    best_c5 = c_ls
                    best_h = h_ls.copy()
                    best_n = n
                    print(f"  LS n={n} r={r}: C5={c_ls:.6f}")
                h_base = h_ls

    # Final constraint enforcement
    h_ret = best_h.copy()
    tgt = best_n / 2.0
    diff = tgt - h_ret.sum()
    if abs(diff) > 1e-13:
        mask = (h_ret > 1e-10) & (h_ret < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            h_ret[mask] += diff / cnt
        h_ret = np.clip(h_ret, 0.0, 1.0)
        d2 = tgt - h_ret.sum()
        if abs(d2) > 1e-14:
            h_ret -= d2 / best_n
            h_ret = np.clip(h_ret, 0.0, 1.0)

    c5f = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return h_ret, c5f, best_n
