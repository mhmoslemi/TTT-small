def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_objective(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    s = np.clip((corr - m) * beta, -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / beta) * (2.0 / n)
