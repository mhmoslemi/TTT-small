    """One pass of coordinate descent. Returns modified h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for i in range(n):
        # Compute a_k for each lag k
        # Position i participates in lags k where i is either in the "first" or "second" factor
        # First factor: h[i+k-n+1] = h[i] means k = i + n - 1 - m for m = 0..i-1... 
        # Let me just compute it directly
        
        # For k in range(2n-1):
        #   the valid range of m (second index) is: max(0, k-(n-1)) <= m <= min(n-1, k)
        #   first index = m + k - (n-1)
        #   h[i] appears as first index when m + k - (n-1) = i, i.e., m = i - k + n - 1
        #   h[i] appears as second index when m = i
        
        # Coefficient of t (new value of h[i]) in c[k]:
        # From first-index terms: sum of (1-h[m]) for valid m where m = i-k+n-1
        # From second-index terms: -h[m+k-n+1] for m = i (if valid)
        
        a = np.zeros(2*n-1)
        for k in range(2*n-1):
            # m range: max(0, k-(n-1)) to min(n-1, k)
            m_lo = max(0, k - (n-1))
            m_hi = min(n-1, k)
            # First-index contribution: m = i - k + n - 1
            m1 = i - k + n - 1
            if m_lo <= m1 <= m_hi:
                a[k] += (1.0 - h[m1])
            # Second-index contribution: m = i
            if m_lo <= i <= m_hi:
                first_idx = i + k - (n-1)
                if 0 <= first_idx <= n-1:
                    a[k] -= h[first_idx]
        
        # c[k](t) = corr[k] + (t - h[i]) * a[k]
        # f(t) = max_k (corr[k] + (t-h[i])*a[k]) * dx
        # Try t=0 and t=1
        t_try = [0.0, 1.0]
        # Also try the current value (should be a local optimum already)
        
        best_t = h[i]
        best_f = np.max(corr + (best_t - h[i]) * a)  # which is just max(corr)
        
        for t in [0.0, 1.0]:
            val = np.max(corr + (t - h[i]) * a)
            if val < best_f:
                best_f = val
                best_t = t
        
        h[i] = best_t
    
    return h

import time

def c5_val(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def coord_descent(h, target, n_passes=50):
    """Coordinate descent: for each i, try h[i]=0 and h[i]=1, keep the better."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    second_partners = np.arange(n)

            a[i:i+n] += (1.0 - h[first_partners])

                any_change = True
