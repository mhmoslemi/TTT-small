import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_max(corr, beta):
    m = corr.max()
    shifted = (corr - m) * beta
    return m + np.log(np.sum(np.exp(shifted))) / beta

def sa_optimize(h, n, ts, T_init, T_min, n_iter, seed, time_limit=60):
    rng = np.random.RandomState(seed)
    h = h.copy()
    h = project_feasible(h, ts)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        T = T_init * (T_min / T_init) ** (it / max(n_iter - 1, 1))
        T = max(T, T_min)
        r = rng.rand()
        h_new = h.copy()
        if r < 0.20:
            i = rng.randint(n)
            h_new[i] = 1.0 - h[i]
        elif r < 0.40:
            i = rng.randint(n)
            h_new[i] += rng.normal(0, 0.3)
        elif r < 0.55:
            blen = rng.randint(3, 10)
            start = rng.randint(0, max(1, n - blen))
            h_new[start:start + blen] += rng.normal(0, 0.12, blen)
        elif r < 0.70:
            i, j = rng.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h[j], h[i]
        elif r < 0.85:
            blen = rng.randint(4, 15)
            start = rng.randint(0, max(1, n - blen))
            h_new[start:start + blen] = 1.0 - h[start:start + blen]
        else:
            h_new = np.clip(h + rng.normal(0, 0.06, n), 0, 1)
        h_new = project_feasible(h_new, ts)
        new_c5 = compute_c5(h_new)
        if new_c5 <= cur_c5 or rng.rand() < np.exp(-(new_c5 - cur_c5) / T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5

def slsqp_solve(h, n, ts, beta, maxiter=500):
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        return smooth_max(corr, beta)
    try:
        res = minimize(objective, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def focused_local_search(h, n, ts, time_limit=30):
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(300):
        if time.time() - t0 > time_limit:
            break
        free = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(free) == 0:
            break
        improved = False
        for j in free:
            if time.time() - t0 > time_limit:
                break
            for t_val in (0.0, 1.0):
                if abs(t_val - h[j]) < 1e-8:
                    continue
                h[j] = t_val
                h = project_feasible(h, ts)
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    improved = True
        if not improved:
            break
    return h, best_c5
