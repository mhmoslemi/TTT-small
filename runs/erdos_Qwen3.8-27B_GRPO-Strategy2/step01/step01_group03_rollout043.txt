import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    s = np.clip((corr - m) * beta, -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / beta) * (2.0 / n)

def slsqp(h, n, beta, iters=300):
    ts = n / 2.0
    def f(x):
        return smooth_obj(x, n, beta)
    def c(x):
        return x.sum() - ts
    r = minimize(f, h, method='SLSQP', bounds=[(0,1)]*n,
                 constraints=[{'type':'eq','fun':c}],
                 options={'maxiter':iters,'ftol':1e-12})
    if r.x is not None and np.isfinite(r.x).all():
        x = np.clip(r.x, 0, 1)
        d = ts - x.sum()
        if abs(d) > 1e-12:
            mask = (x < 1-1e-10) if d > 0 else (x > 1e-10)
            idx = np.where(mask)[0]
            if len(idx):
                x[idx] += d/len(idx)
                x = np.clip(x, 0, 1)
        return x
    return np.clip(h, 0, 1)

def to_binary(h, n):
    t = n // 2
    h_r = np.round(h)
    no = int(h_r.sum())
    if no > t:
        idx = np.where(h_r==1)[0]
        h_r[idx[np.argsort(h[idx])[:no-t]]] = 0
    elif no < t:
        idx = np.where(h_r==0)[0]
        h_r[idx[np.argsort(-h[idx])[:t-no]]] = 1
    return h_r

def sa_solve(h0, n, time_limit, rng, T0=0.25, alpha=0.99985, Tmin=0.002):
    h = h0.astype(float).copy()
    bh = h.copy()
    bc = compute_c5(h)
    cc = bc
    T = T0

        if time.time() - t0 > time_limit: break
        if T < Tmin: break
        u = rng.random()
        if u < 0.4:
            z = np.where(h==0)[0]; o = np.where(h==1)[0]
            i, j = int(z[rng.randint(len(z))]), int(o[rng.randint(len(o))])
            h[i], h[j] = 1., 0.
            nc = compute_c5(h)
            d = nc - cc
            if d < 0 or rng.random() < np.exp(-d/T):
                cc = nc
                if cc < bc: bc = cc; bh = h.copy()
            else:
                h[i], h[j] = 0., 1.
        elif u < 0.7:
            i = rng.randint(n-1)
            if h[i] == h[i+1]:
                T *= alpha; continue
            h[i], h[i+1] = h[i+1], h[i]
            nc = compute_c5(h)
            d = nc - cc
            if d < 0 or rng.random() < np.exp(-d/T):
                cc = nc
                if cc < bc: bc = cc; bh = h.copy()
            else:
                h[i], h[i+1] = h[i+1], h[i]
        elif u < 0.85:
            i = rng.randint(n-2)
            if h[i] == h[i+2]:
                T *= alpha; continue
            h[i], h[i+2] = h[i+2], h[i]
            nc = compute_c5(h)
            d = nc - cc
            if d < 0 or rng.random() < np.exp(-d/T):

            else:

        else:
            # 2-swap: pick two 1s and two 0s, cross-swap
            o = np.where(h==1)[0]; z = np.where(h==0)[0]
            if len(o) < 2 or len(z) < 2:
