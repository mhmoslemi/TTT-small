import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h, n, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
            else:
                d = max(diff, -h[idx])
            h[idx] += d
            diff -= d
            if abs(diff) < 1e-10:
                break
    return h

def _exhaustive_pair_descent(h, best_c5, dx, deltas, max_passes, t0, budget_s, margin=100):
    n = len(h)
    for _pass in range(max_passes):
        if time.time() - t0 > budget_s - margin:
            break
        improved = False
        for j in range(n):
            hj_save = h[j]
            for i in range(n):
                if i == j:
                    continue
                hi_save = h[i]
                for d in deltas:
                    # Transfer d from j to i
                    nj_, ni_ = hj_save - d, hi_save + d
                    if nj_ < -1e-12 or ni_ > 1.0 + 1e-12:
                        continue
                    h[j] = max(0, nj_)
                    h[i] = min(1, ni_)
                    c5 = _compute_c5(h, dx)
                    if c5 < best_c5 - 1e-12:
                        best_c5 = c5
                        hj_save = h[j]
                        hi_save = h[i]
                        improved = True
                    else:
                        h[j] = hj_save
                        h[i] = hi_save
                        break

                hi_save = h[i]
                hj_save = h[j]
                for d in deltas:
                    ni_, nj_ = hi_save - d, hj_save + d
                    if ni_ < -1e-12 or nj_ > 1.0 + 1e-12:
                        continue
                    h[i] = max(0, ni_)
                    h[j] = min(1, nj_)
                    c5 = _compute_c5(h, dx)

                        best_c5 = c5
                        hj_save = h[j]
                        hi_save = h[i]
                        improved = True
                    else:
                        h[i] = hi_save
                        h[j] = hj_save

import numpy as np
import time


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _repair(h, n, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    if s > 1e-10:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
            else:
                d = max(diff, -h[idx])
            h[idx] += d
            diff -= d
            if abs(diff) < 1e-10:
                break
    return h


def _pair_descent(h, best_c5, dx, deltas, max_passes, t0, budget_s, margin=80):

        if time.time() - t0 > budget_s - margin:

def _sa(h, best_c5, dx, t0, budget_s, n_steps=3000000, margin=60):

    T_init = 0.002
    T_min = 1e-8
