    """Gradient of correlation at lag m w.r.t. h"""

    shift = m - (n - 1)

    i_start = max(0, n - 1 - m)
    i_end = min(n - 1, n - 2 + m)

        grad[i] += (1.0 - h[j])
        # d/dh[j] = -h[i]
        grad[j] -= h[i]
    return grad

        if time.time() - t0 > budget_s - 0.45 * budget_s:
            break
        alloc = min(remaining * 0.08, 120)
        if time.time() - t0 + alloc > budget_s - 0.4 * budget_s:
            break
        bh, bc = _binary_local_search(cand, n, rng, alloc, time.time(), best_c5)
        if bc < best_c5:
            best_c5 = bc
            h_best = bh

    # Phase 2: SLSQP + logsumexp refinement (35% of time)
    t_slip_end = t0 + 0.80 * budget_s
    it = 0
    stall = 0
    while time.time() < t_slip_end:
        it += 1
        improved = False
        sigma = 0.02 + 0.03 * min(4, it // 6)
        h_pert = repair(h_best + rng.normal(0, sigma, n), n)

        try:
            r = minimize(_max_obj, h_pert, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 350, 'ftol': 1e-13})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5, improved = hc, c, True
        except Exception:
            pass

        if not improved and time.time() < t_slip_end:
            beta = 20.0 + 10.0 * (it % 6)
            try:
                r2 = minimize(make_lse(beta), h_pert, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-12})
                hc2 = repair(r2.x, n)
                c2 = compute_c5(hc2, n)
                if c2 < best_c5 - 1e-14:
                    h_best, best_c5, improved = hc2, c2, True
            except Exception:
                pass

        if not improved and it % 4 == 0:
            # Subgradient step on top-3 lags
            corrs = np.correlate(h_best, 1.0 - h_best, mode='full')
            top3 = np.argsort(corrs)[-3:]
            g = np.zeros(n)
            for m in top3:
                g += grad_lag(h_best, int(m), n)
            g /= 3.0
            gn = np.linalg.norm(g)
            if gn > 1e-12:
                for alpha_step in [0.03, 0.01, 0.005]:
                    h_sg = repair(h_best - alpha_step * g / gn, n)
                    c_sg = compute_c5(h_sg, n)
                    if c_sg < best_c5 - 1e-14:
                        h_best, best_c5, improved = h_sg, c_sg, True
                        break

        stall = 0 if improved else stall + 1
        if stall > 12:
            h_best = repair(h_best + rng.normal(0, 0.1, n), n)
            stall = 0

    # Phase 3: Iterate binary-continuous (12% of time)
    t_iter_end = t0 + 0.92 * budget_s
    for round_i in range(4):
        if time.time() > t_iter_end:
            break
        # Round to binary
        h_r = (h_best > 0.5).astype(float)
        h_r = repair(h_r, n)
        # Binary local search
        alloc = min(30, (budget_s - (time.time() - t0)) * 0.1)
        bh, bc = _binary_local_search(h_r, n, rng, alloc, time.time(), best_c5)
        if bc < best_c5:
            best_c5 = bc
            h_best = bh
        # SLSQP polish
        if time.time() < t_iter_end:
            try:
                r = minimize(_max_obj, h_best, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 400, 'ftol': 1e-13})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5:
                    h_best, best_c5 = hc, c
            except Exception:
                pass

    # Phase 4: Final polish
    if time.time() < t0 + budget_s - 5:
        try:
            r = minimize(_max_obj, h_best, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 600, 'ftol': 1e-15})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5:
                h_best, best_c5 = hc, c
        except Exception:
            pass

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)

def make_lse(beta):
    return _LSE(beta)

class _LSE:
    def __init__(self, beta):
        self.beta = beta
    def __call__(self, x):
        return logsumexp_obj(x, _N, self.beta)
