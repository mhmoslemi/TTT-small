    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:

        h = np.clip(h, 0.0, 1.0)

        if abs(d2) > 1e-10:

    return h

def lse_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    e = np.exp(alpha * (corr - m))

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lse_obj(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(corr.max())
    return m + float(np.log(np.sum(np.exp(alpha * (corr - m))))) / alpha

def lse_grad(x, alpha, dx, n):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(corr.max())
    e = np.exp(alpha * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            t = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                t += 1.0 - x[j1]
            j2 = i - lag
            if 0 <= j2 < n:
                t -= x[j2]
            s += w[k] * t
        grad[i] = s
    return grad

def ceq_fun(x, target):
    return x.sum() - target

def ceq_jac(x, target):
    return np.ones(len(x))

def slsqp_smooth(h0, n, dx, alpha, maxiter=300):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return lse_obj(x, alpha, dx)
    def g(x):
        return lse_grad(x, alpha, dx, n)
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}
    try:
        res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                       constraints=[con], options={'maxiter': maxiter, 'ftol': 1e-12})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)

    _t = float(n) / 2.0
    _dx_g = dx
    b = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _cfun, 'jac': _cjac}
    try:
        r = minimize(_obj_true, h0, method='SLSQP', bounds=b,
                     constraints=[con], options={'maxiter': maxit, 'ftol': 1e-11})
        h = project(r.x, n)
        return h, eval_c5(h, dx)
    except Exception:
        return h0.copy(), eval_c5(h0, dx)

def lag_obj(x, n, dx, lag):
    s = 0.0
    for i in range(n):
        j = i + lag
        if 0 <= j < n:
            s += x[i] * (1.0 - x[j])
    return s * dx

def lag_grad(x, n, lag):
    g = np.zeros(n)
    for i in range(n):
        j = i + lag
        if 0 <= j < n:
            g[i] += 1.0 - x[j]
        j2 = i - lag
        if 0 <= j2 < n:
            g[i] -= x[j2]
    return g

def _obj_lag(x):
    return lag_obj(x, _n_g, _dx_g, _lag_g)

def _grad_lag(x):
    return lag_grad(x, _n_g, _lag_g)

_lag_g = 0

def opts_lag(h0, n, dx, lag, maxit=200):
    global _t, _dx_g, _n_g, _lag_g
    _t = float(n) / 2.0
    _dx_g = dx
    _n_g = n
    _lag_g = int(lag)
    b = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _cfun, 'jac': _cjac}
    try:
        r = minimize(_obj_lag, h0, jac=_grad_lag, method='SLSQP', bounds=b,
                     constraints=[con], options={'maxiter': maxit, 'ftol': 1e-12})
        h = project(r.x, n)
        return h, eval_c5(h, dx)
    except Exception:
        return h0.copy(), eval_c5(h0, dx)

def anneal(h0, n, dx, t_limit):
    best_h = h0.copy()
    best_c5 = eval_c5(h0, dx)
    alphas = [3.0, 8.0, 15.0, 30.0, 60.0, 120.0, 250.0, 500.0]
    for a in alphas:
        if time.time() > t_limit:
            break
        h, c = opts_smooth(best_h, n, dx, a, maxit=350)
        if c < best_c5:
            best_h, best_c5 = h, c
    if time.time() < t_limit:
        h, c = opts_true(best_h, n, dx, maxit=400)
        if c < best_c5:
            best_h, best_c5 = h, c
    return best_h, best_c5

def lag_polish(h, n, dx, t_limit, rounds=20):
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    for _ in range(rounds):
        if time.time() > t_limit:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        h, c = opts_lag(best_h, n, dx, lag, maxit=150)
        if c < best_c5:
            best_h, best_c5 = h, c
        else:
            h2, c2 = opts_smooth(best_h, n, dx, 100.0, maxit=100)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
            else:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 10.0

    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    best_h = project(np.array(initial_h_values, dtype=float), n0)
    best_c5 = eval_c5(best_h, dx0)
    best_n = n0

    # Phase 1: Annealed smooth optimization on initial (120s)
    t1 = min(t0 + 120.0, t_end - 200.0)
    best_h, best_c5 = anneal(best_h, n0, dx0, t1)
    best_n = n0

    # Phase 2: Lag-targeted polish (60s)
    t2 = min(time.time() + 60.0, t_end - 150.0)
    best_h, best_c5 = lag_polish(best_h, best_n, dx0, t2, rounds=25)

    # Phase 3: Try other n values
    n_candidates = [60, 80, 64, 72, 84, 90, 100]
    for nc in n_candidates:
        if time.time() > t_end - 60.0:
            break
        if nc == best_n:
            continue
        dxc = 2.0 / nc
        # Interpolate best to new n
        xs_old = np.linspace(0.0, 2.0, len(best_h))
        xs_new = np.linspace(0.0, 2.0, nc)
        h_init = np.interp(xs_new, xs_old, best_h)
        h_init = project(h_init, nc)
        h_c, c_c = anneal(h_init, nc, dxc, min(time.time() + 45.0, t_end - 30.0))
        if c_c < best_c5:
            best_c5 = c_c
            best_h = h_c
            best_n = nc
            best_h, best_c5 = lag_polish(best_h, best_n, dxc, min(time.time() + 30.0, t_end - 20.0), rounds=10)

    # Phase 4: Random restarts with smooth + true optimization
    dx_b = 2.0 / best_n
    n_b = best_n
    attempt = 0
    while time.time() < t_end - 5.0:
        attempt += 1
        if attempt % 5 == 1:
            h0 = best_h + 0.08 * np.random.randn(n_b)
        elif attempt % 5 == 2:
            h0 = best_h + 0.15 * np.random.randn(n_b)
        elif attempt % 5 == 3:
            h0 = best_h + 0.25 * np.random.randn(n_b)
        elif attempt % 5 == 4:
            h0 = np.clip(best_h[np.random.permutation(n_b)], 0.0, 1.0)
        else:
            h0 = 0.5 + 0.3 * np.random.randn(n_b)
        h0 = project(h0, n_b)
        h, c = opts_smooth(h0, n_b, dx_b, 50.0, maxit=250)
        c = eval_c5(h, dx_b)
        if c < best_c5:
            h, c = opts_true(h, n_b, dx_b, maxit=200)
            if c < best_c5:
                best_h, best_c5 = h, c

    # Phase 5: Final polish
    if time.time() < t_end - 3.0:
        best_h, best_c5 = lag_polish(best_h, best_n, 2.0 / best_n, t_end - 1.0, rounds=15)
    if time.time() < t_end - 1.0:
        h, c = opts_true(best_h, best_n, 2.0 / best_n, maxit=300)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = project(best_h, best_n)
    best_c5 = eval_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
