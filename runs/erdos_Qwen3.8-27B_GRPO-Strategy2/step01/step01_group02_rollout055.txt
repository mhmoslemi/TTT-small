def subgradient_step(h, n, step_size=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    # The lag corresponding to max_idx in full correlation
    # full correlation: index k corresponds to lag k - (n-1)
    lag = max_idx - (n - 1)
    
    # Gradient of corr[max_idx] w.r.t. h
    grad = np.zeros(n)
    if lag >= 0:
        # corr[k] = sum_i h[i] * (1-h[i-lag]) for i=lag to n-1

        for j in range(lag, n):
            grad[j] += (1.0 - h[j - lag])
        for j in range(0, n - lag):
            grad[j] += -h[j + lag]
    else:
        # corr[k] where lag = k - (n-1) < 0, so actual shift is -lag
        # corr[n-1-lag] = sum_i h[i-lag] * (1-h[i]) for i=0 to lag+n-1
