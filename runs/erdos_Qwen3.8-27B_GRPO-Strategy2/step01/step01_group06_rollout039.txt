If lag >= 0: S_lag = {0, 1, ..., n-1-lag}
If lag < 0: S_lag = {-lag, -lag+1, ..., n-1}

d corr[k] / d h[m]:
- h[m] appears as h[i] when i=m (if m in S_lag): contributes (1-h[m+lag])
- h[m] appears as h[i+lag] when i+lag=m, i=m-lag (if m-lag in S_lag): contributes -h[m-lag]

Wait, the second term: we need m-lag >= 0 and m-lag < n and also (m-lag)+lag = m must be < n (which it is) and >= 0 (which it is). And we need m-lag in S_lag, which means 0 <= m-lag < n and 0 <= (m-lag)+lag = m < n. So we just need 0 <= m-lag < n.

So:
d corr[k] / d h[m] = (1-h[m+lag]) * [0 <= m < n and 0 <= m+lag < n] - h[m-lag] * [0 <= m-lag < n]

For lag >= 0:
- First term: valid when m+lag < n, i.e., m < n-lag. So for m = 0, ..., n-lag-1: (1-h[m+lag])
- Second term: valid when m-lag >= 0, i.e., m >= lag. So for m = lag, ..., n-1: -h[m-lag]

For lag < 0 (let l = -lag > 0):
- First term: valid when m+lag = m-l >= 0, i.e., m >= l. So for m = l, ..., n-1: (1-h[m-l])
- Second term: valid when m-lag = m+l < n, i.e., m < n-l. So for m = 0, ..., n-l-1: -h[m+l]

So the vectorized computation for a given lag:
If lag >= 0:
  grad_lag[:n-lag] += (1-h[lag:])      # first term for m=0,...,n-lag-1
  grad_lag[lag:] -= h[:n-lag]           # second term for m=lag,...,n-1

If lag < 0 (l = -lag):

  grad_lag[:n-l] -= h[l:]               # second term for m=0,...,n-l-1

import numpy as np
from scipy.optimize import minimize
import time

_gT = 0.0
_gN = 0
_gD = 0.0
_gB = 1.0

def eval_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = float(n) / 2.0
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-14:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += d * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = tgt - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def socg(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    s = beta * c * dx
    mx = s.max()
    e = np.exp(s - mx)
    se = e.sum()
    w = e / se
    obj = (mx + np.log(se)) / beta
    g = np.zeros(n)
    om = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - n + 1
        if lag >= 0:
            g[:n-lag] += wk * om[lag:]
            g[lag:] -= wk * h[:n-lag]
        else:
            l = -lag
            g[l:] += wk * om[:n-l]
            g[:n-l] -= wk * h[l:]
    return obj, g * dx

def _f(x):
    return socg(x, _gD, _gB)[0]

def _g(x):
    return socg(x, _gD, _gB)[1]

def _eq(x):
    return x.sum() - _gT

def _eqj(x):
    return np.ones(_gN)

def slsqp(h, n, dx, beta, mx=5000):
    global _gT, _gN, _gD, _gB
    _gT, _gN, _gD, _gB = float(n)/2.0, n, dx, beta
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_f, h.copy(), jac=_g, method='SLSQP', bounds=bnds,
                     constraints=[{'type':'eq','fun':_eq,'jac':_eqj}],
                     options={'maxiter':mx,'ftol':1e-17})
        h2 = project_feasible(r.x, n)
        return h2, eval_c5(h2, dx)
    except:
        h2 = project_feasible(h, n)
        return h2, eval_c5(h2, dx)

    xd = (np.arange(n_dst) + 0.5) / n_dst

import numpy as np
from scipy.optimize import minimize
import time

_gT = 0.0
_gN = 0
_gD = 0.0
_gB = 1.0

def eval_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = float(n) / 2.0
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-14:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += d * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = tgt - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def socg(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    s = beta * c * dx
    mx = s.max()
    e = np.exp(s - mx)
    se = e.sum()
    w = e / se
    obj = (mx + np.log(se)) / beta
    g = np.zeros(n)
    om = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - n + 1
        if lag >= 0:
            g[:n - lag] += wk * om[lag:]
            g[lag:] -= wk * h[:n - lag]
        else:
            l = -lag
            g[l:] += wk * om[:n - l]
            g[:n - l] -= wk * h[l:]
    return obj, g * dx

def _f(x):
    return socg(x, _gD, _gB)[0]

def _g(x):
    return socg(x, _gD, _gB)[1]

def _eq(x):
    return x.sum() - _gT

def _eqj(x):
    return np.ones(_gN)

def slsqp(h, n, dx, beta, mx=5000):
    global _gT, _gN, _gD, _gB
    _gT, _gN, _gD, _gB = float(n) / 2.0, n, dx, beta
    bnds = [(0.0, 1.0)] * n
    mn = max(1000, 8000 - 40 * n)
    mx = min(mx, mn)
    try:
        r = minimize(_f, h.copy(), jac=_g, method='SLSQP', bounds=bnds,
                     constraints=[{'type': 'eq', 'fun': _eq, 'jac': _eqj}],
                     options={'maxiter': mx, 'ftol': 1e-17})
        h2 = project_feasible(r.x, n)
        return h2, eval_c5(h2, dx)
    except Exception:
        h2 = project_feasible(h, n)
        return h2, eval_c5(h2, dx)

def momentum_opt(h, n, dx, beta, maxiter=50000, lr0=0.3, mom=0.9):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    lr = lr0
    stagnant = 0
    for it in range(maxiter):
        _, g = socg(h, dx, beta)
        v = mom * v - lr * g
        h_new = np.clip(h + v, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
            if stagnant % 50 == 0:
                pass
        else:
            h = h_new
            stagnant += 1
            if stagnant > 200:
                lr *= 0.7
                v *= 0.5
                stagnant = 0
            elif stagnant > 100:
                lr *= 0.9
        if lr < 1e-7:
            break
    return best_h, best_c5

def interp_h(h_s, n_s, n_d):
    xs = (np.arange(n_s) + 0.5) / n_s
    xd = (np.arange(n_d) + 0.5) / n_d
    return np.interp(xd, xs, h_s)
