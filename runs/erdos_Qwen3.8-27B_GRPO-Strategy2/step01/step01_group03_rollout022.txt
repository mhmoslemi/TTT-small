import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgradient_c5(h):
    """Return (c5_value, gradient_array) for the maximizing shift."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    p = np.argmax(corr)
    s = p - (n - 1)  # shift
    c5 = corr[p] * dx
    
    # Compute gradient of f_s with respect to h
    g = np.zeros(n)
    if s >= 0:
        # f_s(h) = sum_{k=0}^{n-1-s} h[k]*(1-h[k+s])
        valid1 = np.arange(n - s)  # k where h[k] appears as first factor
        g[valid1] += (1.0 - h[valid1 + s])
        valid2 = np.arange(s, n)  # j where h[j] appears as h[k+s], k=j-s
        g[valid2] -= h[valid2 - s]
    else:
        m = -s
        # f_{-m}(h) = sum_{k=m}^{n-1} h[k]*(1-h[k-m])
        valid1 = np.arange(m, n)  # k where h[k] appears as first factor
        g[valid1] += (1.0 - h[valid1 - m])
        valid2 = np.arange(n - m)  # j where h[j] appears as h[k-m], k=j+m
        g[valid2] -= h[valid2 + m]
    
    return c5, g * dx

def project_feasible(h, ts):
    """Project h to [0,1]^n with sum = ts."""
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    # Bisection for better projection
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0.0, 1.0)
        s = h_trial.sum()
        if s < ts:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        idx2 = np.where((h > 0) & (h < 1))[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + diff2, 0, 1)
    return h

def subgradient_descent(h_init, n, budget_s, seed=42):
    """Subgradient descent with adaptive step size."""
    ts = n / 2.0
    h = h_init.copy()
    h = project_feasible(h, ts)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    rng = np.random.RandomState(seed)
    t0 = time.time()
    alpha = 0.1  # initial step size
    alpha_decay = 0.9999
    no_improve = 0
    max_no_improve = 500
    
    while time.time() - t0 < budget_s:
        c5, g = subgradient_c5(h)
        
        # Random perturbation to escape flat regions
        if no_improve > 200:
            h += rng.normal(0, 0.05, n)
            h = project_feasible(h, ts)
        
        # Subgradient step
        h_new = h - alpha * g
        h_new = project_feasible(h_new, ts)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        alpha *= alpha_decay
        if alpha < 0.001:
            alpha = 0.01  # restart with random perturbation
            h += rng.normal(0, 0.1, n)
            h = project_feasible(h, ts)
    
    return best_h, best_c5

def slsqp_polish(h, n, beta, maxiter=500):
    """SLSQP with log-sum-exp smoothing."""
    ts = n / 2.0
    dx = 2.0 / n
    m0 = np.max(np.correlate(h, 1-h, mode="full"))
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        shifted = (corr - m0) * beta
        max_shifted = np.max(shifted)
        logsumexp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
        val = (m0 + logsumexp / beta) * dx
        return val
    
    res = minimize(obj_and_grad, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def local_refine(h, n, budget_s, seed=42):
    """Local refinement with small perturbations and binary flips."""
    ts = n / 2.0
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    
    while time.time() - t0 < budget_s:
        improved = False
        # Try random small perturbations
        for _ in range(50):
            if time.time() - t0 > budget_s:
                break
            scale = 0.05
            idx = np.random.randint(0, n)
            delta = rng.uniform(-scale, scale)
            h_trial = best_h.copy()
            h_trial[idx] = np.clip(h_trial[idx] + delta, 0, 1)
            h_trial = project_feasible(h_trial, ts)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_trial
                improved = True
        
        # Try binary flips
        for _ in range(20):
            if time.time() - t0 > budget_s:
                break
            idx = rng.randint(n)
            target = 1.0 if h[idx] < 0.5 else 0.0
            if abs(target - h[idx]) < 0.01:
                continue
            h_trial = best_h.copy()
            h_trial[idx] = target
            h_trial = project_feasible(h_trial, ts)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_trial
                improved = True
        
        if not improved:
            # Try larger perturbation
            h_trial = best_h + rng.normal(0, 0.1, n)
            h_trial = project_feasible(h_trial, ts)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_trial
            else:
                break
    
    return best_h, best_c5

def interpolate_h(h_from, n_from, n_to):
    """Interpolate h from n_from to n_to samples."""
    xo = np.linspace(0, 2, n_from)
    xn = np.linspace(0, 2, n_to)
    h_new = np.interp(xn, xo, h_from)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def remaining():

    rng = np.random.RandomState(seed)

            g[idx1] += weights[p] * (1.0 - h[idx1 - m])

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_val = np.clip(h + mid, 0.0, 1.0).sum()
        if s_val < ts:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        idx = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + diff2, 0.0, 1.0)
    return h


def obj_lse_val(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    ms = sh.max()
    return (m + (ms + np.log(np.sum(np.exp(sh - ms)))) / beta) * dx


def obj_lse_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    sh = (corr - m) * beta
    ms = sh.max()
    lse = m + (ms + np.log(np.sum(np.exp(sh - ms)))) / beta
    val = lse * dx
    w = np.exp(sh - ms)
    w /= w.sum()
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-12:
            continue
        s = p - (n - 1)
        if s >= 0:
            idx1 = np.arange(n - s)
            grad[idx1] += w[p] * (1.0 - x[idx1 + s])
            if s == 0:
                grad[idx1] -= w[p] * x[idx1]
            else:
                idx2 = np.arange(s, n)
                grad[idx2] -= w[p] * x[idx2 - s]
        else:
            mm = -s
            idx1 = np.arange(mm, n)
            grad[idx1] += w[p] * (1.0 - x[idx1 - mm])
            idx2 = np.arange(n - mm)
            grad[idx2] -= w[p] * x[idx2 + mm]
    grad *= dx
    return val, grad


def sum_con(x, n):
    return x.sum() - 0.5 * n


def slsqp_grad(h, n, beta, maxiter=500):
    ts = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    res = minimize(obj_lse_val, h, args=(n, beta), jac=obj_lse_grad,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)


def binary_sa(h_init, n, budget_s, seed=42):
    rng_t = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    ones = [int(i) for i in np.where(h > 0.5)[0]]
    zeros = [int(i) for i in np.where(h <= 0.5)[0]]
    t0 = time.time()
    temp = 0.04
    while time.time() - t0 < budget_s:
        if temp < 1e-7:
            break
        ip = rng_t.randint(len(ones))
        jp = rng_t.randint(len(zeros))
        i = ones[ip]
        j = zeros[jp]
        h[i] = 0.0
        h[j] = 1.0
        ones[ip] = j
        zeros[jp] = i
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        elif rng_t.random() < np.exp(-(c5_new - best_c5) / temp):
            pass
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones[ip] = i
            zeros[jp] = j
        temp *= 0.99997
    return best_h, best_c5


def continuous_local(h_init, n, budget_s, seed=42):
    rng_c = np.random.RandomState(seed)
    ts = 0.5 * n
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    deltas = [0.01, 0.02, 0.05, 0.1, 0.2, -0.01, -0.02, -0.05, -0.1, -0.2]
    while time.time() - t0 < budget_s:
        improved = False
        idx_order = rng_c.permutation(n)
        for jj in idx_order:
            if time.time() - t0 > budget_s:
                break
            for d in deltas:
                h_t = best_h.copy()
                h_t[jj] = np.clip(h_t[jj] + d, 0.0, 1.0)
                h_t = project_feasible(h_t, ts)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_t
                    improved = True
                    break
        if not improved:
            h_t = best_h + rng_c.normal(0, 0.08, n)
            h_t = project_feasible(h_t, ts)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-13:
                best_c5 = c5_t
                best_h = h_t
            else:
                break
    return best_h, best_c5


def interp_h(h_from, n_to):
    xo = np.linspace(0.0, 2.0, len(h_from))
    xn = np.linspace(0.0, 2.0, n_to)
    return np.interp(xn, xo, h_from)

                best_h = h_c.copy()
        if remaining() > 30:
            h_c = slsqp_grad(h_c, nc, 200, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
        # Local search
        if remaining() > 25:
            h_lr2, c5_lr2 = continuous_local(h_c, nc, min(25, remaining() - 10), seed + nc)
            if c5_lr2 < best_c5:
                best_c5 = c5_lr2
                best_h = h_lr2

    # Phase 5: Final intensive polish
    nb = len(best_h)
    if remaining() > 40:
        h_f = slsqp_grad(best_h.copy(), nb, 200, 500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if remaining() > 30:
        h_f = slsqp_grad(best_h.copy(), nb, 500, 400)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if remaining() > 20:
        h_lr3, c5_lr3 = continuous_local(best_h.copy(), nb, min(20, remaining() - 5), seed + 777)
        if c5_lr3 < best_c5:
            best_c5 = c5_lr3
            best_h = h_lr3

    best_h = project_feasible(best_h, 0.5 * len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
