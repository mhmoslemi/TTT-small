                    hn, cn = _subgrad_desc(hc, target, n_it, 0.004, k, eb)
                    if cn < best_c5:
                        best_c5 = cn
                        best_h = hn.copy()
                        best_n = n
                        print(f"n={n} a={att} SG k={k} eb={eb}: C5={cn:.6f}")
                    hc = hn

            # Phase C: Greedy local
            if time.time() - t0 < budget_s - 8:
                hc, cg = _greedy(hc, target, 6)
                if cg < best_c5:
                    best_c5 = cg
                    best_h = hc.copy()
                    best_n = n
                    print(f"n={n} a={att} GRD: C5={cg:.6f}")
                    hc = hc

            # Phase D: Block swap
            if time.time() - t0 < budget_s - 6:
                hc, cb = _block_swap(hc, target, 6, 40)
                if cb < best_c5:
                    best_c5 = cb
                    best_h = hc.copy()
                    best_n = n
                    print(f"n={n} a={att} BSW: C5={cb:.6f}")
                    hc = hc

    # Final polish: SLSQP on best
    if time.time() - t0 < budget_s - 8:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: _eq_fun(x, target),
                 'jac': lambda x: _eq_jac(x, target)}]
        for al in [200, 1000, 5000]:
            if time.time() - t0 > budget_s - 4:
                break
            try:
                res = minimize(_slsqp_smooth, best_h, args=(al, target, dx),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-14})
                hn = _proj(res.x, target)
                cn = _c5(hn)
                if cn < best_c5:
                    best_c5 = cn
                    best_h = hn.copy()
                    print(f"FINAL SLS a={al}: C5={cn:.6f}")
            except Exception:
                pass

    # Final greedy
    if time.time() - t0 < budget_s - 2:
        n = best_n
        target = n / 2.0
        best_h, best_c5 = _greedy(best_h, target, 10)
        print(f"FINAL GRD: C5={best_c5:.6f}")

    # Ensure exact constraint
    target = best_n / 2.0
    best_h = _proj(best_h, target)
    s = np.sum(best_h)
    diff = target - s
    if abs(diff) > 1e-14:
        free = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if free.any():
            best_h[free] += diff / np.sum(free)
        best_h = np.clip(best_h, 0.0, 1.0)

    c5_final = _c5(best_h)
    print(f"RESULT: n={best_n}, C5={c5_final:.7f}")
    return best_h, c5_final, best_n
