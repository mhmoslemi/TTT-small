import numpy as np
from scipy.optimize import minimize
import time

def c5_val(h, dx):
    cv = np.correlate(h, 1.0 - h, mode="full") * dx
    return cv.max()

def corr_full(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project(h, target):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def softmax_obj(h, dx, eps):
    cv = corr_full(h, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def softmax_grad(h, dx, eps):
    cv = corr_full(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]
    return g

def subgrad_step(h, dx, target, lr):
    cv = corr_full(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-12)[0]
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    g /= max(len(active), 1)
    return project(h - lr * g, target)

def slsqp_run(h0, dx, n, target, eps, maxiter=500):
    def obj(x):
        return softmax_obj(x, dx, eps)
    def jac(x):
        return softmax_grad(x, dx, eps)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    res = minimize(
        obj, h0, jac=jac,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': 1e-18}
    )
    return project(res.x, target)

_N = 0

_TGT = 0.0

def _obj(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = cv.max()
    return mx + _EPS * np.log(np.sum(np.exp((cv - mx) / _EPS)))

def _grad(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _DX
    mx = cv.max()
    w = np.exp((cv - mx) / _EPS)
    w /= w.sum()
    n = _N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * _DX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _cf(x):
    return x.sum() - _TGT

def _cj(x):
    return np.ones(len(x))

    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def proj(h, tgt):
    lo, hi = -4.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def slsqp(h0, dx, n, tgt, eps, mit):
    global _DX, _N, _EPS, _TGT

                     bounds=[(0.0, 1.0)] * n,

                     method='SLSQP', options={'maxiter': mit, 'ftol': 1e-18})
