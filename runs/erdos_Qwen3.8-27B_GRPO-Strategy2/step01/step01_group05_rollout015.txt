import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _smo_vals(cv, eps):
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _smo_grad(h, dx, n, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]
    return g

def _lbfgs(h0, dx, n, target, eps, maxiter=400):
    lam = 50.0
    def obj(x):
        return _smo_vals(_corr(x, dx), eps) + 0.5 * lam * (x.sum() - target) ** 2
    def grad(x):
        return _smo_grad(x, dx, n, eps) + lam * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40})
        return _proj(res.x, target)
    except Exception:
        return _proj(h0, target)

def _pgd(h0, dx, n, target, eps, n_iter=4000, t_end=None):
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    lr = 0.3
    prev_h = h.copy()
    prev_g = _smo_grad(h, dx, n, eps)
    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        g = _smo_grad(h, dx, n, eps)
        if it > 2:
            sv = h - prev_h
            yv = g - prev_g
            sy = sv @ yv
            yy = yv @ yv
            if yy > 1e-10 and sy > 1e-10:
                lr_b = 0.5 * (sv @ sv) / sy
                lr = float(np.clip(lr_b, 0.01, 1.5))
        h_new = _proj(h - lr * g, target)
        prev_h, prev_g = h.copy(), g
        h = h_new
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _top_shift_grad(h, dx, n, top_k):
    cv = _corr(h, dx)
    idxs = np.argsort(cv)[-top_k:]
    j = np.arange(n)
    g = np.zeros(n)
    for idx in idxs:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)

        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= h[jp[vp]]
    g /= len(idxs)
    return g
