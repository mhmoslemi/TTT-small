    mo = n - 1

    mks = np.arange(2*n-1) - mo  # lag values
    # grad_j[k] = (1-h[j+mk]) - h[j-mk] (with validity)
    gj = np.zeros(2*n-1)
    gl = np.zeros(2*n-1)

        if 0 <= j-mk < n: gj[k] -= h[j-mk]
        if 0 <= l+mk < n: gl[k] += 1.0 - h[l+mk]
        if 0 <= l-mk < n: gl[k] -= h[l-mk]

    # C5(d) = max_k (corr[k] + d * sl[k]) * dx
    # Vectorized: (2n-1) × n_eval matrix
    C = corr[:, None] + dvals[None, :] * sl[:, None]
    c5_vals = C.max(axis=0) * dx
    
    best_idx = c5_vals.argmin()
    best_d = dvals[best_idx]
    
    # Refine
    if 0 < best_idx < n_eval - 1:
        lo, hi = dvals[best_idx-1], dvals[best_idx+1]
    else:
        lo, hi = dlo, dhi
    dvals2 = np.linspace(lo, hi, 40)
    C2 = corr[:, None] + dvals2[None, :] * sl[:, None]

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break

        if free.any():

    m = corr.max()

    w = np.exp(np.clip(alpha * (corr - corr.max()), -50.0, 50.0))

        if 0 <= l - m < n: gl[k] -= h[l - m]

    h = np.zeros(n)
