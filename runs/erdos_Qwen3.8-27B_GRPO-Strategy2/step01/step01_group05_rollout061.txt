import numpy as np
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return float(_corr(h, dx).max())

def _proj_sum(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _smooth_grad(h, dx, n, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    sw = w.sum()
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]
    return g

def _subgrad(h, dx, n):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-10)[0]
    na = max(len(active), 1)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += dx * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= dx * h[jp[vp]]
    return g / na

def _phase_smooth(h, dx, n, target, t_end):
    for eps in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > t_end:
            break
        g = _smooth_grad(h, dx, n, eps)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            continue
        g = g / gn
        for step in [0.05, 0.02, 0.01, 0.005, 0.002]:
            h_try = _proj_sum(h - step * g * gn, target)
            c_try = _c5(h_try, dx)
            if c_try < _c5(h, dx):
                h = h_try
                break
    return h

def _phase_subgrad(h, dx, n, target, rng, n_iter, t_end, lr_init=0.004):
    h = h.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    lr = lr_init
    stag = 0
    prev_c = best_c
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _subgrad(h, dx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _proj_sum(h - lr * g / gn * gn, target)
        c_new = _c5(h_new, dx)
        if c_new <= prev_c:
            h = h_new
            lr = min(lr * 1.004, 0.3)
            stag = 0
        else:
            lr = max(lr * 0.96, 0.0003)
            stag += 1
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        prev_c = _c5(h, dx)
        if stag > 300:
            amp = min(0.02 * stag / 300, 0.06)
            h = _proj_sum(best_h + rng.normal(0, amp, n), target)
            stag = 0
            lr = lr_init
    return best_h, best_c

def _binary_search(h0, dx, target, rng, t_end, max_it=40000):
    n = len(h0)
    n_ones = int(round(target))
    hb = np.round(h0).astype(float)
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    while len(ones) > n_ones and len(zeros) > 0:
        i = rng.choice(ones)
        hb[i] = 0
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
    while len(ones) < n_ones and len(zeros) > 0:
        i = rng.choice(zeros)
        hb[i] = 1
        zeros = np.where(hb == 0)[0]
        ones = np.where(hb == 1)[0]
    best_h = hb.copy()
    best_c = _c5(hb, dx)
    for it in range(max_it):
        if time.time() > t_end:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        hb[i], hb[j] = 0, 1
        c = _c5(hb, dx)
        if c < best_c:
            best_c = c
            best_h = hb.copy()
        else:
            hb[i], hb[j] = 1, 0
            if it % 60 == 30:
                o2 = np.where(hb == 1)[0]
                z2 = np.where(hb == 0)[0]
                if len(o2) > 0 and len(z2) > 0:
                    hb[rng.choice(o2)], hb[rng.choice(z2)] = 0, 1
    return best_h, best_c

def _polish_fine(h, dx, n, target, rng, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    for it in range(30000):
        if time.time() > t_end:
            break
        g = _subgrad(h, dx, n)
        lr = 0.0015 * (1.0 + 0.5 * np.cos(it * 0.005))
        h_new = _proj_sum(h - lr * g, target)
        c_new = _c5(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            if it % 200 == 100:
                i = rng.randint(n)
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + rng.choice([-0.02, 0.02]), 0, 1)
                h_try = _proj_sum(h_try, target)
                c_try = _c5(h_try, dx)
                if c_try < _c5(h, dx):
                    h = h_try
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target = float(n) / 2.0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
    else:
        h0 = np.full(n, 0.5)
        h0[:35] = 1.0
        h0[35:] = 0.0
    h0 = _proj_sum(h0, target)
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    t1 = min(t0 + 90, t0 + (deadline - t0) * 0.15)
    h_s = _phase_smooth(h0, dx, n, target, t1)
    cs = _c5(h_s, dx)
    if cs < best_c5:
        best_c5 = cs
        best_h = h_s.copy()
    t2 = t0 + (deadline - t0) * 0.42
    h, c = _phase_subgrad(h_s, dx, n, target, rng, 20000, t2, lr_init=0.004)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    t3 = t0 + (deadline - t0) * 0.65
    hb, cb = _binary_search(best_h, dx, target, rng, t3, max_it=40000)
    if cb < best_c5:
        best_c5 = cb
        best_h = hb.copy()
        h0 = np.clip(best_h + rng.normal(0, 0.06, n), 0, 1)
        h0 = _proj_sum(h0, target)
    else:
        h0 = best_h.copy()
    t4 = t0 + (deadline - t0) * 0.82
    h, c = _phase_subgrad(h0, dx, n, target, rng, 15000, t4, lr_init=0.003)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    if time.time() < deadline - 5:
        h, c = _polish_fine(best_h, dx, n, target, rng, deadline)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    best_h = _proj_sum(best_h, target)
    best_c5 = _c5(best_h, dx)
    return best_h, best_c5, n
