        x = np.clip(x, 0.0, 1.0)

def block_perturb(x, rng, nblocks=5, block_frac=0.1):
    n = len(x)
    x2 = x.copy()
    bs = max(3, int(n * block_frac))
    for _ in range(nblocks):

        x2[s:s+bs] += rng.standard_normal() * 0.1
    return repair(x2, n)

    """Circular shift by ps positions."""
    return repair(np.roll(x, ps), len(x))

import numpy as np
from scipy.optimize import minimize
import time

_S = {}

def _f(x):
    t = _S['tau']; d = _S['dx']; n = _S['n']
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    sh = np.clip((c - m) / t, -50.0, 0.0)
    e = np.exp(sh)
    se = e.sum()
    obj = (m + t * np.log(se)) * d
    if not _S.get('ug', False):
        return obj
    w = e / se
    A = np.zeros(n); B = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12: continue
        s = n - 1 - k
        j0 = max(0, -s); j1 = min(n - 1, n - 1 - s)
        if j0 <= j1:
            A[j0:j1 + 1] += wk * x[j0 + s:j1 + s]
        j0b = max(0, s); j1b = min(n - 1, s + n - 1)
        if j0b <= j1b:
            B[j0b:j1b + 1] += wk * x[j0b - s:j1b - s]
    return obj, d * (1.0 - A - B)

def _c(x):
    return np.sum(x) - _S['n'] / 2.0

def _cj(x):
    return np.ones(_S['n'])

def c5v(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def rep(x, n):
    x = np.clip(x.astype(float), 0.0, 1.0)
    tg = n / 2.0
    for _ in range(15):
        s = x.sum()
        if abs(s - tg) < 1e-10: break
        if s > 1e-12: x = x * (tg / s)
        x = np.clip(x, 0.0, 1.0)
    return x

def p2(h, ns, nd):
    ls = np.linspace(0, 2, ns, endpoint=False)
    ld = np.linspace(0, 2, nd, endpoint=False)
    return rep(np.interp(ld, ls, h, period=2.0), nd)

def slq(x0, n, dx, tau, mi=400):
    _S.update(n=n, tau=tau, dx=dx, ug=True)
    try:
        r = minimize(_f, x0, method='SLSQP', jac='exact',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _c, 'jac': _cj}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        return rep(r.x, n)
    except Exception:
        return x0.copy()

def ref(x0, n, dx, tl, taus=None):
    t0 = time.time()
    if taus is None:
        taus = [0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    x = rep(x0, n).copy()
    bv = c5v(x, dx)
    best = x.copy()
    for t in taus:
        if time.time() - t0 > tl - 1.0: break
        x = slq(x, n, dx, t)
        v = c5v(x, dx)
        if v < bv: bv, best = v, x.copy()
    return best, bv

def bpert(x, rng, nb=6, bf=0.08):
    n = len(x); x2 = x.copy()
    bs = max(3, int(n * bf))
    for _ in range(nb):
        s = int(rng.integers(0, n - bs))
        x2[s:s + bs] += 0.12 * rng.standard_normal(bs)
    return rep(x2, n)

        if v < bv: bv, best, bn = v, x, n0

        bw = n0 // 4

        xt[n0 // 2:n0 // 2 + bw] = 1.0
        xt = rep(xt, n0)
        x, v = ref(xt, n0, dx0, min(35, rem() * 0.07))
        if v < bv: bv, best, bn = v, x, n0
    
    # B: Coarse-to-fine

    m = c.max()
    sh = np.clip((c - m) / t, -50.0, 0.0)

        wk = w[k]
