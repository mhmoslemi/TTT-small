    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

    corr = _corr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n, target_sum):
    """Repair h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            count = mask.sum()
            if count == 0:
                break
            step = min(diff, count * 0.1)
            h[mask] = np.minimum(h[mask] + step / count, 1.0)
        else:
            mask = h > 0.0
            count = mask.sum()
            if count == 0:
                break
            step = min(-diff, count * 0.1)
            h[mask] = np.maximum(h[mask] - step / count, 0.0)
    return h

def _slsqp_optimize(h, n, target_sum, alpha, maxiter, t_start, budget_s, margin=60):
    if time.time() - t_start > budget_s - margin:
        return h, _c5(h, 2.0/n)
    dx = 2.0 / n
    
    def obj(x):
        corr = _corr(x)
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def grad(x):
        # Numerical gradient (expensive but n=70 is fine)
        g = np.zeros(n)
        eps = 1e-7
        f0 = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - f0) / eps
            x[i] -= eps
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                      'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n, target_sum)
        return hc, _c5(hc, dx)

import numpy as np
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_obj(x, alpha):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.clip(alpha * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] = np.minimum(h[m] + d * 0.4 / c, 1.0)
        else:
            m = h > 0.0
            c = int(m.sum())
            if c == 0:
                break
            h[m] = np.maximum(h[m] - (-d) * 0.4 / c, 0.0)
    return h


def _slsqp(h, n, alpha, maxiter, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n, target)
    except Exception:
        return h.copy()
