def compute_c5(h):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:

                h[idx] += diff2
    return h

def smooth_obj(h, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    e = np.exp(t * (c - cmax))
    return (cmax + np.log(np.sum(e)) / t)

    e = np.exp(t * (c - cmax))
    p = e / np.sum(e)

            if p_start2 < p_end2:
                grad[p_start2:p_end2] -= p[m] * x[p_start2 - offset:p_end2 - offset]
        return grad

    res = minimize(obj, h, jac=grads, method='SLSQP',

                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def subgradient_step(h, n, step_size, iters=100):
    """Simple projected subgradient method for the non-smooth max."""
    dx = 2.0 / n
    h = h.copy()
    target = n * 0.5
    for it in range(iters):
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(c)
        # Subgradient: average of gradients of active constraints
        active = np.where(c > cmax - 1e-6)[0]

            p_start2 = max(0, offset)

            if p_start2 < p_end2:
                grad[p_start2:p_end2] -= p[m] * h[p_start2 - offset:p_end2 - offset]

        h += diff / n

                h = np.clip(h, 0.0, 1.0)
