        except Exception:
            break
    # ProjGrad phase
    lr = 0.15
    v = np.zeros(n)
    mu = 0.8
    for it in range(8000):
        if time.time() > t_end - 8:
            break
        h_new = _projgrad(h, dx, n, target, 0.008, lr)
        c_new = _c5(h_new, dx)
        c_cur = _c5(h, dx)
        if c_new <= c_cur:
            v = mu * v + (h_new - h)
            h = h_new
            lr = min(lr * 1.003, 1.2)
        else:
            v = mu * v
            h = _proj(h - 0.5 * lr * v, target)
            lr = max(lr * 0.96, 0.02)
        if c_new < best_c:
            best_c = c_new
            best = h.copy()
        if it % 150 == 75:
            h = _proj(best + rng.normal(0, 0.035, n), target)
            v = np.zeros(n)
    # Final SLSQP polish
    if time.time() < t_end - 4:
        for eps in [0.0002, 0.00008, 0.00003]:
            if time.time() > t_end - 2:
                break
            try:
                h2 = _slsqp(best, dx, n, target, eps, maxiter=300)
                c2 = _c5(h2, dx)
                if c2 < best_c:
                    best_c = c2
                    best = h2.copy()
            except Exception:
                pass
    return best, best_c

def _try_n(n, h_init, dx, target, rng, t_end):
    h0 = _proj(h_init, target)
    h, c = _optimize_phase(h0, dx, n, target, rng, t_end)
    return h, c, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)

    # Primary: n=70
    n70 = 70
    dx70 = 2.0 / n70
    tgt70 = n70 / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if len(h0) != n70:
            h0 = np.interp(np.linspace(0, 1, n70), np.linspace(0, 1, len(h0)), h0)
    else:
        h0 = rng.uniform(0.2, 0.8, n70)
    h0 = _proj(h0, tgt70)

    best_h = h0.copy()
    best_c5 = _c5(h0, dx70)

    # Phase 1: Main optimization on n=70 (50% budget)
    t1 = t0 + (deadline - t0) * 0.48
    best_h, best_c5 = _optimize_phase(h0, dx70, n70, tgt70, rng, t1)

    # Quick check n=64 with subset of time
    if time.time() < deadline - 120:
        n64 = 64
        dx64 = 2.0 / n64
        tgt64 = n64 / 2.0
        h64_init = np.interp(np.linspace(0, 1, n64), np.linspace(0, 1, n70), best_h)
        h64_init = _proj(h64_init, tgt64)
        t64_end = min(time.time() + 40, deadline - 100)
        h64, c64, _ = _optimize_phase(h64_init, dx64, n64, tgt64, rng, t64_end)
        if c64 < best_c5 - 0.0003:
            best_c5 = c64
            # interpolate back to n70 for final output
            best_h = np.interp(np.linspace(0, 1, n70), np.linspace(0, 1, n64), h64)
            best_h = _proj(best_h, tgt70)

    # Phase 2: Restarts (35% of remaining)
    t2 = t0 + (deadline - t0) * 0.88
    restart_count = 0
    while time.time() < t2:
        restart_count += 1
        strategy = rng.random()
        if strategy < 0.4:
            amp = rng.uniform(0.02, 0.12)
            h_pert = _proj(best_h + rng.normal(0, amp, n70), tgt70)
        elif strategy < 0.7:
            # Block flip
            blk = rng.randint(5, 20)
            start = rng.randint(0, n70 - blk)
            h_pert = best_h.copy()
            h_pert[start:start + blk] = 1.0 - h_pert[start:start + blk]
            h_pert = _proj(h_pert, tgt70)
        else:
            # Region swap: move a block
            blk = rng.randint(4, 12)
            src = rng.randint(0, n70 - blk)
            dst = rng.randint(0, n70 - blk)
            h_pert = best_h.copy()
            tmp = h_pert[src:src + blk].copy()
            h_pert[src:src + blk] = h_pert[dst:dst + blk]
            h_pert[dst:dst + blk] = tmp
            h_pert = _proj(h_pert, tgt70)

        c_pert = _c5(h_pert, dx70)
        if c_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c_pert

        sub_end = min(time.time() + min(22.0, t2 - time.time()), t2)
        if sub_end - time.time() > 5:
            two, c5o = _optimize_phase(h_pert, dx70, n70, tgt70, rng, sub_end)
            if c5o < best_c5:
                best_h = two
                best_c5 = c5o

    # Phase 3: Final polish
    if time.time() < deadline - 12:
        hp, cp = _optimize_phase(best_h, dx70, n70, tgt70, rng, deadline)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    return best_h, best_c5, n70
