    return np.correlate(h, 1.0 - h, mode="full") * dx

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):

        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _obj_smooth(x, dx, eps):
    cv = _corr(x, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _grad_smooth(x, dx, eps, n):
    cv = _corr(x, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    k = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx]
        s = idx - (n - 1)
        kp = k + s
        mp = (kp >= 0) & (kp < n)
        if mp.any():
            g[mp] += wt * (1.0 - x[kp[mp]])
        km = k - s
        mm = (km >= 0) & (km < n)
        if mm.any():
            g[mm] -= wt * x[km[mm]]
    g *= dx
    return g

def _lbfgs_step(h0, dx, n, target, eps, maxiter=200):
    def fun(x):
        return _obj_smooth(x, dx, eps)
    def jac(x):
        return _grad_smooth(x, dx, eps, n)
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'disp': False})
    return _proj(res.x, target)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sobj(x, dx, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad(x, dx, eps, n):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    k = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        kp = k + s
        mp = (kp >= 0) & (kp < n)
        if mp.any():
            g[mp] += wt * (1.0 - x[kp[mp]])
        km = k - s
        mm = (km >= 0) & (km < n)
        if mm.any():
            g[mm] -= wt * x[km[mm]]
    return g

def _lbfgs(h0, dx, n, target, eps, maxiter=500):
    res = minimize(
        lambda x: _sobj(x, dx, eps), h0,
        jac=lambda x: _sgrad(x, dx, eps, n),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'disp': False}
    )
    return _proj(res.x, target)

            s = idx - (n - 1)
            kp = k + s
            mp = (kp >= 0) & (kp < n)

            km = k - s
            mm = (km >= 0) & (km < n)
            if mm.any():
                g[mm] -= h[km[mm]]
        g *= dx / max(len(active), 1)
        h_new = _proj(h - lr * g, target)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sobj(x, dx, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad(x, dx, eps, n):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    k = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        kp = k + s
        mp = (kp >= 0) & (kp < n)
        if mp.any():
            g[mp] += wt * (1.0 - x[kp[mp]])
        km = k - s
        mm = (km >= 0) & (km < n)
        if mm.any():
            g[mm] -= wt * x[km[mm]]
    return g

def _lbfgs_run(h0, dx, n, target, eps, maxiter=500):
    res = minimize(
        _sobj, h0, args=(dx, eps),
        jac=_sgrad, jac_args=(dx, eps, n),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'disp': False}
    )
    return _proj(res.x, target)

def _subgrad(h, dx, n, target, max_it=8000, seed=0):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h = h.copy()
    lr = 0.4
    stag = 0
    rng = np.random.RandomState(seed)
    k = np.arange(n)
    for it in range(max_it):
        cv = _corr(h, dx)
        mx = cv.max()
        active = np.where(cv > mx - 1e-9)[0]
        g = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            kp = k + s
            mp = (kp >= 0) & (kp < n)
            if mp.any():
                g[mp] += (1.0 - h[kp[mp]])
            km = k - s
            mm = (km >= 0) & (km < n)
            if mm.any():
                g[mm] -= h[km[mm]]
        g *= dx / max(len(active), 1)
        h_new = _proj(h - lr * g, target)
        c_new = _c5(h_new, dx)
        if c_new <= mx + 1e-12:
            h = h_new
            lr = min(lr * 1.015, 2.5)
            stag = 0
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            lr = max(lr * 0.88, 0.008)
            stag += 1
            if stag > 400:
                break
            if stag % 100 == 50:
                h = _proj(h + rng.normal(0, 0.04, n), target)
                stag = 0
    return best_h, best_c5

def _binary_sa(h0, dx, n, target, n_iter=30000, seed=0):
    """Simulated annealing on near-binary solutions"""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    # Threshold to get near-binary
    thr = 0.5
    h = np.where(h > thr, 1.0, 0.0)
    h = _proj(h, target)
    # Adjust to satisfy constraint
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    T = 0.005
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_new = h.copy()
        if h_new[i] < 0.5:
            h_new[i] = 1.0
        else:
            h_new[i] = 0.0
        if h_new[j] < 0.5:
            h_new[j] = 1.0
        else:
            h_new[j] = 0.0
        # Check sum constraint
        if abs(h_new.sum() - target) > 1e-9:
            continue
        c_new = _c5(h_new, dx)
        c_cur = _c5(h, dx)
        if c_new < c_cur or rng.rand() < np.exp((c_cur - c_new) / T):
            h = h_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        T *= 0.9999
    return best_h, best_c5

def _periodic_init(n, p, target, rng):
    if n % p != 0:
        return _proj(rng.uniform(0.2, 0.8, n), target)
    nper = n // p
    tgt_p = target / nper
    hp = rng.uniform(0.1, 0.9, p)
    hp = _proj(hp, tgt_p)
    h = np.tile(hp, nper)
    return _proj(h, target)

                    best_c5 = c3
                    best_h = h3.copy()
            except Exception:
                break
    return best_h, best_c5
