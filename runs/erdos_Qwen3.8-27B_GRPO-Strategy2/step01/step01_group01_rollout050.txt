def solve_lp_for_h(g, n):
    """Solve min t s.t. c_k(h) <= t, 0<=h<=1, sum(h)=n/2. g is fixed."""
    from scipy.optimize import linprog
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # c_vec for linprog: minimize c^T x
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each output index m (0 to 2n-2):
    #   sum_i h[i] * g[i + (m - n + 1)] - t <= 0
    num_constr = 2 * n - 1
    A_ub = np.zeros((num_constr, n + 1))
    b_ub = np.zeros(num_constr)
    
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        for i in range(max(0, -lag), min(n, n - lag)):
            A_ub[m, i] = g[i + lag]
        A_ub[m, n] = -1.0  # -t
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    
    # Bounds: 0 <= h_i <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_opt = res.x[:n]
        t_opt = res.x[n]
    else:
        # Fallback
        h_opt = np.full(n, 0.5)
        t_opt = float('inf')
    
    return h_opt, t_opt

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def smooth_obj(h, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    c_shifted = np.clip((c - cmax) * t, -500, 500)
    return (cmax + np.log(np.sum(np.exp(c_shifted))) / t) * (2.0 / len(h))

def solve_lp_h(g, n):
    """LP: min t s.t. corr_k(h, g) <= t, 0<=h<=1, sum(h)=n/2"""
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    nc = 2 * n - 1
    A_ub = np.zeros((nc, n + 1))
    for m in range(nc):
        lag = m - (n - 1)
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo < i_hi:
            A_ub[m, i_lo:i_hi] = g[i_lo + lag:i_hi + lag]
        A_ub[m, n] = -1.0
    b_ub = np.zeros(nc)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

def alt_minimize(h_init, n, max_iters=200, time_limit=120):
    h = repair(h_init.copy(), n)
    g = 1.0 - h
    best_h = h.copy()
    best_v = compute_c5(h)
    t_start = time.time()
    stuck = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        h_opt, t_val = solve_lp_h(g, n)
        if h_opt is None:
            break
        h_new = repair(h_opt, n)
        v = compute_c5(h_new)
        if v < best_v:
            best_v = v
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        g = 1.0 - h_new
        # Check convergence
        if np.max(np.abs(h_new - h)) < 1e-10:
            if stuck > 5:
                break
        h = h_new
    
    return best_h, best_v

def subgrad_refine(h, n, n_steps=300, lr_init=0.003):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        k = int(np.argmax(c))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx + lag] += (1.0 - h[idx])
            grad[idx] -= h[idx + lag]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx] += (1.0 - h[idx + la])
            grad[idx + la] -= h[idx]
        lr = lr_init / (1.0 + step * 0.1)
        h_new = repair(h - lr * grad, n)
        v = compute_c5(h_new)
        if v < best:
            best = v
            best_h = h_new.copy()
        h = h_new
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: SLSQP with adaptive temperature
    for t_val in [80.0, 300.0, 1000.0]:
        if time.time() - t0 > budget_s * 0.25:
            break
        cons = [{'type': 'eq', 'fun': lambda x, n=n0: x.sum() - n0 * 0.5}]
        bnds = [(0.0, 1.0)] * n0
        maxiter = max(50, int(min(200, (budget_s * 0.25 - (time.time() - t0)) / 3.0)))
        res = minimize(smooth_obj, best_h.copy(), args=(t_val,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h = repair(res.x, n0)
        v = compute_c5(h)
        if v < best_c5:

    # Phase 2: Alternating LP minimization

        h_alt, v_alt = alt_minimize(best_h, n0, max_iters=300, time_limit=budget_s * 0.3)
        if v_alt < best_c5:
            best_c5, best_h = v_alt, h_alt.copy()
    
    # Phase 3: Subgradient refinement
    if time.time() - t0 < budget_s * 0.75:
        h_sg, v_sg = subgrad_refine(best_h, n0, n_steps=500, lr_init=0.002)
        if v_sg < best_c5:

    for n_hi in [n0 * 2, n0 * 3]:
        if n_hi > 2000 or time.time() - t0 > budget_s * 0.85:

        xo = np.linspace(0, 2, len(best_h))

        h_hi = np.interp(xn, xo, best_h)
        h_hi = repair(h_hi, n_hi)
        # LP at higher res

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def smooth_obj(h, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    shifted = np.clip((c - cmax) * t, -500.0, 500.0)
    return (cmax + np.log(np.sum(np.exp(shifted))) / t) * (2.0 / len(h))


def sum_constraint(x, n):
    return x.sum() - n * 0.5


def sum_jacobian(x):
    return np.ones(len(x))


def solve_lp_h(g, n):
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    nc = 2 * n - 1
    A_ub = np.zeros((nc, n + 1))
    for m in range(nc):
        lag = m - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if lo < hi:
            A_ub[m, lo:hi] = g[lo + lag:hi + lag]
        A_ub[m, n] = -1.0
    b_ub = np.zeros(nc)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), res.x[n]
    except Exception:
        pass
    return None, None


def alt_minimize(h_init, n, max_iters, time_limit):
    h = repair(h_init.copy(), n)
    g = 1.0 - h
    best_h = h.copy()
    best_v = compute_c5(h)
    t_start = time.time()
    prev_h = h.copy()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        h_opt, t_val = solve_lp_h(g, n)
        if h_opt is None:
            break
        h_new = repair(h_opt, n)
        v = compute_c5(h_new)
        if v < best_v - 1e-14:
            best_v = v
            best_h = h_new.copy()
        g = 1.0 - h_new
        if np.max(np.abs(h_new - prev_h)) < 1e-10:
            if it > 20:
                break
        prev_h = h_new
        h = h_new
    return best_h, best_v


def subgrad_refine(h, n, n_steps, lr_init):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        k = int(np.argmax(c))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx + lag] += (1.0 - h[idx])
            grad[idx] -= h[idx + lag]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx] += (1.0 - h[idx + la])
            grad[idx + la] -= h[idx]
        lr = lr_init / (1.0 + 0.05 * step)
        h_new = repair(h - lr * grad, n)
        v = compute_c5(h_new)
        if v < best:
            best = v
            best_h = h_new.copy()
        h = h_new
    return best_h, best


def nm_objective(x):
    n = len(x)
    return compute_c5(repair(x, n))
