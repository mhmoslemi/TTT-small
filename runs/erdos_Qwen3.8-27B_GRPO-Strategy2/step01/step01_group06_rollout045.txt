import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def slsqp_opt(h, n, dx, maxiter=2000, ftol=1e-18):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

    h_new = np.interp(x_new, x_old, h_old, period=2.0)

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def slsqp_opt(h, n, dx, maxiter=3000, ftol=1e-18):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return eval_c5(x, dx)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

    best_h = h.copy()
    h = h.copy()

    corr = np.correlate(h, 1.0 - h, mode='full')
    k_range = np.arange(2 * n - 1) - (n - 1)
    two_n_1 = 2 * n - 1

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            factor = np.zeros(two_n_1)
            valid1 = (i - k_range >= 0) & (i - k_range < n)
            factor[valid1] += (1.0 - h[i - k_range[valid1]])
            valid2 = (i + k_range >= 0) & (i + k_range < n)
            factor[valid2] -= h[i + k_range[valid2]]
            new_corrs = corr[None, :] + (grid[:, None] - old_val) * factor[None, :]
            maxes = np.max(new_corrs, axis=1)
            best_idx = int(np.argmin(maxes))
            best_t = grid[best_idx]
