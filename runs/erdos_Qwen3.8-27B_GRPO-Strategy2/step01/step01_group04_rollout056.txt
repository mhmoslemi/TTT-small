        x = np.clip(x, 0.0, 1.0)
    return x

def slsqp(x0, n, dx, tau, maxiter, use_full=True):
    global _g_n, _g_tau, _g_dx
    _g_n = n
    _g_tau = tau
    _g_dx = dx
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    obj = _obj_full if use_full else _obj_active
    try:
        r = minimize(obj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value=0.0)(xd)
    return repair(xp.astype(float), n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def update_active(x, topk=12):
    global _g_active
    corr = np.correlate(x, 1.0 - x, mode="full")
    _g_active = np.argsort(corr)[-topk:]

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values, _g_active
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = repair(h0.copy(), n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Multi-restart smooth-max optimization at n0
    taus_p1 = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    n_restarts = 4
    for r in range(n_restarts):
        if remaining() < budget_s * 0.45:
            break
        if r == 0:
            x_cur = h0.copy()
        else:
            x_cur = perturb(h0, n0, 0.03 + 0.02 * r, rng)
        for ti, tau in enumerate(taus_p1):
            if remaining() < 15:
                break
            mi = 500 + 300 * (len(taus_p1) - ti)
            x_cur = slsqp(x_cur, n0, dx0, tau, mi, True)
            val = compute_c5(x_cur, dx0)
            if val < best_val:
                best_val = val
                best_h = x_cur.copy()
                best_n = n0

    # Phase 2: Active-set iterative refinement at best grid
    if remaining() > 100:
        dx_b = 2.0 / best_n
        x_cur = best_h.copy()
        for rnd in range(8):
            if remaining() < 20:
                break
            update_active(x_cur, topk=14)
            x_cur = slsqp(x_cur, best_n, dx_b, 0.0003, 2000, False)
            val = compute_c5(x_cur, dx_b)
            if val < best_val:
                best_val = val
                best_h = x_cur.copy()

    # Phase 3: Even tighter active-set with smaller tau
    if remaining() > 60:
        dx_b = 2.0 / best_n
        x_cur = best_h.copy()
        for tau_act in [0.0001, 0.00005, 0.00002]:
            if remaining() < 15:
                break
            update_active(x_cur, topk=10)
            x_cur = slsqp(x_cur, best_n, dx_b, tau_act, 3000, False)
            val = compute_c5(x_cur, dx_b)
            if val < best_val:
                best_val = val
                best_h = x_cur.copy()
                best_n = best_n

    # Phase 4: Try a slightly different grid size
    if remaining() > 120:
        n_alt = n0 + 49  # e.g., 240 if n0=191
        dx_alt = 2.0 / n_alt
        x_p = project_grid(best_h, best_n, n_alt)
        x_cur = x_p
        for tau in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if remaining() < 25:
                break
            x_cur = slsqp(x_cur, n_alt, dx_alt, tau, 600, True)
            val = compute_c5(x_cur, dx_alt)
            if val < best_val:
                best_val = val
                best_h = x_cur.copy()
                best_n = n_alt
        # Active refinement on alt grid
        if remaining() > 40 and best_n == n_alt:
            for _ in range(3):
                if remaining() < 15:
                    break
                update_active(x_cur, topk=10)
                x_cur = slsqp(x_cur, n_alt, dx_alt, 0.0001, 2000, False)
                val = compute_c5(x_cur, dx_alt)
                if val < best_val:
                    best_val = val
                    best_h = x_cur.copy()
                    best_n = n_alt

    # Phase 5: Final micro-perturbation + polish at best grid
    if remaining() > 30:
        dx_f = 2.0 / best_n
        x_cur = perturb(best_h, best_n, 0.005, rng)
        update_active(x_cur, topk=12)
        x_cur = slsqp(x_cur, best_n, dx_f, 0.00005, 3000, False)
        val = compute_c5(x_cur, dx_f)
        if val < best_val:
            best_val = val
            best_h = x_cur.copy()
            best_n = best_n

    # Final evaluation
    dx_final = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_final)

    return (best_h, final_val, best_n)
