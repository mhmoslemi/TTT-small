def projected_adam(h, n, dx, tau_schedule, max_iter, lr, rng):
    # ADAM state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best = h.copy()
    best_val = compute_c5(h, dx)
    
    for tau in tau_schedule:
        for it in range(max_iter):
            t += 1
            h = repair(h, n)
            corr = np.correlate(h, 1-h, mode='full')
            # Soft-max gradient
            exp_c = np.exp(corr / tau)
            exp_c = exp_c - np.max(exp_c)  # numerical stability
            p = exp_c / np.sum(exp_c)
            # Gradient
            grad = compute_gradient(h, p, n)
            # ADAM update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0, 1)
            h = repair(h, n)
            val = compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best = h.copy()
    return best, best_val

def compute_grad(h, p, n):
    # p has length 2n-1, indexed by k = l + (n-1)
    # h has length n
    grad = np.zeros(n)
    for k in range(2*n-1):
        l = k - (n-1)
        pk = p[k]
        if pk < 1e-15:
            continue
        if l >= 0:
            # (1-h[j-l]) for j in [l, n-1]
            j_range = range(l, n)
            grad[j_range] += pk * (1 - h[j_range - l])
            # -h[j+l] for j in [0, n-1-l]
            j_range2 = range(0, n-l)
            grad[j_range2] -= pk * h[j_range2 + l]

            m = -l

            j_range = range(0, n-m)

            j_range2 = range(m, n)
            grad[j_range2] -= pk * h[j_range2 - m]

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    tgt = n / 2.0
    s = np.sum(x)
    if s < 1e-12:
        x[:] = 0.5
    else:

    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:

        if diff > 0:
            mask = x < 0.999
        else:
            mask = x > 0.001
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def softmax_grad(h, n, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    c = corr / tau
    c -= np.max(c)
    ec = np.exp(c)
    p = ec / np.sum(ec)
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        pk = p[k]
        if pk < 1e-12:
            continue
        if l >= 0:
            # +p_k * (1-h[j-l]) for j in [l, n-1]
            if l < n:
                js = np.arange(l, n)
                grad[js] += pk * (1.0 - h[js - l])
            # -p_k * h[j+l] for j in [0, n-1-l]
            if l < n:
                js2 = np.arange(0, n - l)
                grad[js2] -= pk * h[js2 + l]
        else:
            m = -l
            # +p_k * (1-h[j+m]) for j in [0, n-1-m]
            js = np.arange(0, n - m)
            grad[js] += pk * (1.0 - h[js + m])
            # -p_k * h[j-m] for j in [m, n-1]
            js2 = np.arange(m, n)
            grad[js2] -= pk * h[js2 - m]
    return grad * tau, float(np.max(corr) * (2.0 / n))

def adam_optimize(h0, n, dx, tau_schedule, iters_per_tau, lr0, time_limit):

            t += 1

            h = repair(h, n)
            val = compute_c5(h, dx)
            if val < best_val:

Where autocorr_g[k] = sum_i g[i]*g[i+k] (with appropriate boundary handling for the discrete correlation).

This is a much cleaner formulation! We want to find g (with sum=0, |g| ≤ 0.5, i.e., |h-0.5| ≤ 0.5) that maximizes the minimum autocorrelation.

This is related to the "flat autocorrelation" problem in signal processing. The optimal g would have flat (constant) autocorrelation, which is impossible for finite sequences, but we can get close.

This reformulation might give better intuition but doesn't change the optimization much. The key insight is that we want the autocorrelation to be as flat as possible (high at all lags).

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    s = np.sum(x)
    if s < 1e-14:
        x[:] = 0.5
    else:
        x *= tgt / s
        x = np.clip(x, 0.0, 1.0)
    for _ in range(30):
        diff = tgt - np.sum(x)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = x < 0.999
        else:
            mask = x > 0.001
        cnt = np.sum(mask)
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def softgrad(h, n, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr / tau
    c -= np.max(c)
    ec = np.exp(c)
    p = ec / np.sum(ec)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        pk = p[k]
        if pk < 1e-14:
            continue
        if l >= 0:
            if l < n:
                js = np.arange(l, n)
                grad[js] += pk * (1.0 - h[js - l])
                js2 = np.arange(0, n - l)
                grad[js2] -= pk * h[js2 + l]
        else:
            m = -l
            js = np.arange(0, n - m)
            grad[js] += pk * (1.0 - h[js + m])
            js2 = np.arange(m, n)
            grad[js2] -= pk * h[js2 - m]
    return grad

def adam_opt(h0, n, dx, taus, iters, lr0, tlim):
    t0 = time.time()
    h = repair(h0, n).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best = h.copy()
    bv = compute_c5(h, dx)
    for tau in taus:
        for it in range(iters):
            if time.time() - t0 > tlim:
                return best, bv
            t += 1
            h = repair(h, n)
            g = softgrad(h, n, tau)
            m = b1 * m + (1.0 - b1) * g
            v = b2 * v + (1.0 - b2) * g * g
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            lr = lr0 / (1.0 + 0.002 * it)
            h = h - lr * mh / (np.sqrt(vh) + eps)
            h = np.clip(h, 0.0, 1.0)
            h = repair(h, n)
            val = compute_c5(h, dx)
            if val < bv:
                bv = val
                best = h.copy()
    return best, bv

def projekto(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    f = interp1d(xs, x_src, kind='linear', bounds_error=False,
                 fill_value='extrapolate')
    return repair(f(xd), n_dst)

_gdx = 1.0
_gtau = 1.0
_gn = 0

def _sobj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -100.0, 0.0)
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx

def _scfun(x):
    return np.sum(x) - _gn / 2.0

def _sjac(x):
    return np.ones(_gn)

def slsqp_polish(x0, n, dx, tau, maxit):
    global _gdx, _gtau, _gn
    _gn = n
    _gdx = dx
    _gtau = tau
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _scfun, 'jac': _sjac}]
    try:
        r = minimize(_sobj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxit, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(191, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)

    best_h = h0.copy()
    best_val = compute_c5(h0, dx0)
    best_n = n0

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed() - 8

    # Phase 1: ADAM on base grid with restarts
    taus1 = [2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    n_restarts = 6
    p1_total = budget_s * 0.28
    p1_per = p1_total / n_restarts

    starts = [h0.copy()]
    for i in range(n_restarts - 1):
        sc = 0.05 + 0.025 * i
        noise = sc * rng.standard_normal(n0)
        fr = rng.integers(1, 5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        xx = np.arange(n0, dtype=float)
        noise += 0.03 * np.sin(2.0 * np.pi * fr * xx / n0 + ph)
        starts.append(h0 + noise)

    for idx, xs in enumerate(starts):
        if remaining() < 40:
            break
        if elapsed() > budget_s * 0.42:
            break
        alloc = min(p1_per, budget_s * 0.42 - elapsed())
        if alloc < 12:
            break
        xb, val = adam_opt(xs, n0, dx0, taus1, 350, 0.03, alloc)
        if val < best_val:
            best_val = val
            best_h = xb.copy()
            best_n = n0

    # Phase 2: SLSQP polish at base grid
    if remaining() > 30 and elapsed() < budget_s * 0.60:
        p2 = min(budget_s * 0.15, budget_s * 0.60 - elapsed())
        if p2 > 15:
            xp = slsqp_polish(best_h, best_n, dx0, 0.01, 1200)
            val = compute_c5(xp, dx0)
            if val < best_val:
                best_val = val
                best_h = xp
            xp2 = slsqp_polish(best_h, best_n, dx0, 0.004, 800)
            val = compute_c5(xp2, dx0)
            if val < best_val:
                best_val = val
                best_h = xp2

    # Phase 3: Project to finer grid and optimize
    if remaining() > 60 and elapsed() < budget_s * 0.82:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        xp = projekto(best_h, best_n, n1)
        p3_total = min(budget_s * 0.22, budget_s * 0.82 - elapsed())
        if p3_total > 25:
            taus3 = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002]
            xb, val = adam_opt(xp, n1, dx1, taus3, 250, 0.02, p3_total * 0.55)
            if val < best_val:
                best_val = val
                best_h = xb.copy()
                best_n = n1
            rem3 = remaining()
            if rem3 > 20:
                xp2 = slsqp_polish(best_h, best_n, 2.0 / best_n, 0.005, 600)
                val = compute_c5(xp2, 2.0 / best_n)
                if val < best_val:
                    best_val = val
                    best_h = xp2

    # Phase 4: Final polish
    rem = remaining()
    if rem > 20:
        dx_b = 2.0 / best_n
        tau_f = 0.002
        xp = slsqp_polish(best_h, best_n, dx_b, tau_f, max(300, int(rem * 8)))
        val = compute_c5(xp, dx_b)
        if val < best_val:
            best_val = val
            best_h = xp
        if rem > 40:
            xp2 = slsqp_polish(best_h, best_n, dx_b, 0.001, 500)
            val = compute_c5(xp2, dx_b)
            if val < best_val:
                best_val = val
                best_h = xp2

    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
