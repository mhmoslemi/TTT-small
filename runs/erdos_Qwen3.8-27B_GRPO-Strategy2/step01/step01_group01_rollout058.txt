def compute_grad_softmax(h, n, t):
    c = np.correlate(h, 1.0-h, mode='full')  # length 2n-1
    # softmax weights
    w = np.exp(t * (c - c.max()))
    w /= w.sum()
    # w has length 2n-1, w[k] corresponds to shift m_k = k-(n-1)
    # grad[j] = sum_k w[k] * [(1-h[j+m_k])*I(0<=j+m_k<n) - h[j-m_k]*I(0<=j-m_k<n)]
    grad = np.zeros(n)

        for j in range(n):
            if 0 <= j+m < n:
                grad[j] += w[k] * (1.0 - h[j+m])
            if 0 <= j-m < n:
                grad[j] -= w[k] * h[j-m]
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def project_onto_set(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.count()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_max_grad(h, n, t):
    """Gradient of (smoothed max of correlation) * dx w.r.t. h"""
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    w = np.exp(t * (c - cmax))
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            rem = n - m
            grad[:rem] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:rem]
        else:
            am = -m
            rem = n - am
            grad[am:] += wk * (1.0 - h[:rem])
            grad[:rem] -= wk * h[am:]
    return grad * dx

    t_start = time.time()

def compute_c5(h):

    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def project_set(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10

            break

def smooth_grad(h, n, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    w = np.exp(t * (c - cm))
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            rem = n - m
            if rem > 0:
                grad[:rem] += wk * (1.0 - h[m:])
                grad[m:] -= wk * h[:rem]
        else:
            am = -m
            rem = n - am
            grad[am:] += wk * (1.0 - h[:rem])
            grad[:rem] -= wk * h[am:]
    return grad * dx

            c = np.correlate(x, 1.0 - x, mode='full')
            cm = c.max()
            return (cm + np.log(np.sum(np.exp(250.0 * (c - cm)))) / 250.0) * dx

        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                 'jac': lambda x: np.ones(n)}]
        bnds = [(0.0, 1.0)] * n
        tl = budget_s * 0.86 - elapsed()
        maxiter = max(80, int(tl / 1.5))
        try:
            res = minimize(obj_slqp, best_h.copy(), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            h = repair(res.x, n)
            v = compute_c5(h)
            if v < best_c5:
                best_c5, best_h = v, h.copy()
        except Exception:
            pass

    # Phase 6: Binary search from current best
    if time_ok(0.93):
        h_bin = (best_h > 0.45).astype(float)
        h_bin = repair(h_bin, n0)
        h_b, v_b = binary_search(h_bin, n0, time_limit=budget_s * 0.05, seed=seed + 1)
        h_b = repair(h_b, n0)
        v_b = compute_c5(h_b)
        if v_b < best_c5:
            best_c5, best_h = v_b, h_b.copy()

    # Phase 7: Final coordinate descent
    if time_ok(0.98):
        h_cur, v = coord_desc(best_h, n0, max_sweeps=60, time_limit=budget_s * 0.02)
        h_cur = repair(h_cur, n0)
        v = compute_c5(h_cur)
        if v < best_c5:
            best_c5, best_h = v, h_cur.copy()

    best_h = repair(best_h, n0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n0)
