- Use L-BFGS-B with penalty for the bulk of the optimization (fast, handles many iterations)
- Use SLSQP for final polish (handles constraint exactly)
- Use subgradient for the "kink" regions

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smoobj(h, dx, n, eps, mu, target):
    cv = _corr(h, dx)
    mx = cv.max()
    base = mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))
    pen = mu * (h.sum() - target) ** 2
    return base + pen

def _smograd(h, dx, n, eps, mu, target):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        v1 = (j + s >= 0) & (j + s < n)
        v2 = (j - s >= 0) & (j - s < n)
        if v1.any():
            g[v1] += wt * (1.0 - h[j[v1] + s])
        if v2.any():
            g[v2] -= wt * h[j[v2] - s]
    g += 2.0 * mu * (h.sum() - target)
    return g

def _slqp_obj(h, dx, n, eps, target):
    cv = _corr(h, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _slqp_grad(h, dx, n, eps, target):
    return _smograd(h, dx, n, eps, 0.0, target)

def _cons_eq(h, target):
    return h.sum() - target

def _cons_eq_jac(h, target):
    return np.ones(len(h))

def _do_lbfgsb(h0, dx, n, target, eps, mu, maxiter):
    args = (dx, n, eps, mu, target)
    res = minimize(_smoobj, h0, args=args, jac=_smograd, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
    return _proj(res.x, target)

def _do_slqp(h0, dx, n, target, eps, maxiter):
    args = (dx, n, eps, target)
    res = minimize(_slqp_obj, h0, args=args, jac=_slqp_grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return _proj(res.x, target)

def _subgrad_step(h, dx, n, target, lr):
    cv = _corr(h, dx)
    mx = cv.max()
    active = np.where(cv >= mx - 1e-8)[0]
    g = np.zeros(n)
    j = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        v1 = (j + s >= 0) & (j + s < n)
        v2 = (j - s >= 0) & (j - s < n)
        if v1.any():
            g[v1] += dx * (1.0 - h[j[v1] + s])
        if v2.any():
            g[v2] -= dx * h[j[v2] - s]
    g /= max(len(active), 1)
    return _proj(h - lr * g, target)

def _full_optimize(h0, dx, n, target, rng, t_end):
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
    h = best_h.copy()
    # Phase 1: L-BFGS-B with penalty, decreasing eps
    for eps, mu in [(0.1, 10.0), (0.05, 50.0), (0.02, 100.0), (0.01, 500.0),
                    (0.005, 1000.0), (0.002, 5000.0), (0.001, 10000.0)]:
        if time.time() > t_end - 25:
            break
        try:
            h_new = _do_lbfgsb(h, dx, n, target, eps, mu, maxiter=200)
        except Exception:
            h_new = h
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    # Phase 2: SLSQP for exact constraint
    for eps in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > t_end - 15:
            break
        try:
            h_new = _do_slqp(h, dx, n, target, eps, maxiter=400)
        except Exception:
            h_new = h
        c = _c5(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    # Phase 3: Subgradient with momentum
    lr = 0.1
    mom = np.zeros(n)
    beta = 0.5
    for it in range(6000):
        if time.time() > t_end - 8:
            break
        h_new = _subgrad_step(h, dx, n, target, lr)
        c_new = _c5(h_new, dx)
        c_old = _c5(h, dx)
        if c_new <= c_old:
            h = h_new
            lr = min(lr * 1.008, 2.0)
            beta = min(beta * 1.01, 0.9)
        else:
            lr = max(lr * 0.93, 0.005)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if it % 400 == 200:
            h = _proj(best_h + rng.normal(0, 0.025, n), target)
            mom = np.zeros(n)
    # Phase 4: Final SLSQP polish
    for eps in [0.0001, 0.00005]:
        if time.time() > t_end - 2:
            break
        try:
            h2 = _do_slqp(best_h, dx, n, target, eps, maxiter=300)
            c2 = _c5(h2, dx)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target = n / 2.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) == n:
            h0 = h0.copy()
        else:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
        h0 = _proj(h0, target)
    else:
        h0 = _proj(rng.uniform(0.3, 0.7, n), target)

    # Stage A: Primary optimization from parent (50% time)
    tA = t0 + (deadline - t0) * 0.50
    best_h, best_c5 = _full_optimize(h0, dx, n, target, rng, tA)

    # Stage B: Restarts with diverse perturbations (30% time)
    tB = t0 + (deadline - t0) * 0.80
    n_restarts = 0
    while time.time() < tB:
        n_restarts += 1
        r = rng.random()
        if r < 0.4:
            amp = rng.uniform(0.02, 0.20)
            h_pert = _proj(best_h + rng.normal(0, amp, n), target)
        elif r < 0.7:
            freq = rng.randint(1, 6)
            amp = rng.uniform(0.03, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            h_pert = _proj(best_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase), target)
        else:
            # Flip a random block
            i0, i1 = sorted(rng.choice(n, 2, replace=False))
            h_pert = best_h.copy()
            block = h_pert[i0:i1+1]
            h_pert[i0:i1+1] = 1.0 - block
            h_pert = _proj(h_pert, target)
        c_pert = _c5(h_pert, dx)
        if c_pert < best_c5:
            best_c5 = c_pert
            best_h = h_pert.copy()
        remaining = tB - time.time()
        if remaining > 8:
            sub_end = time.time() + min(18.0, remaining)
            h_opt, c_opt = _full_optimize(h_pert, dx, n, target, rng, sub_end)
            if c_opt < best_c5:
                best_h = h_opt
                best_c5 = c_opt

    # Stage C: Try n=100 refinement (10% time)
    if time.time() < deadline - 60:
        n2 = 100
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h0_2 = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n), best_h)
        h0_2 = _proj(h0_2, target2)

        # Downsample n=100 to n=70 for comparison
        h2_70 = np.interp(np.linspace(0, 1, 70), np.linspace(0, 1, n2), h2)
        h2_70 = _proj(h2_70, target)
        c2_70 = _c5(h2_70, dx)
        if c2_70 < best_c5:

            best_c5 = c2_70

    # Stage D: Final polish (remaining time)
    if time.time() < deadline - 12:
        h_f, c_f = _full_optimize(best_h, dx, n, target, rng, deadline)
        if c_f < best_c5:
            best_h = h_f
            best_c5 = c_f

    # Final verification
    best_c5 = _c5(best_h, dx)
    assert abs(best_h.sum() - target) < 1e-6, f"sum={best_h.sum()}"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj(best_h, target)

    return best_h, best_c5, n
