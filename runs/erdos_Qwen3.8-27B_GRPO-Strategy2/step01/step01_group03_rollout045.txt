import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def corr_grad(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[:n-s] += (1.0 - h[s:n])
            grad[s:] -= h[:n-s]
    else:
        sp = -s
        if n - sp > 0:
            grad[sp:] += (1.0 - h[:n-sp])
            grad[:n-sp] -= h[sp:]
    return grad

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-14)[0]
    else:
        idx = np.where(h > 1e-14)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        order = idx[np.argsort(h if diff2 < 0 else (1.0 - h))]
        for i in order:
            adj = np.sign(diff2) * min(abs(diff2), abs(h[i] if diff2 < 0 else (1.0 - h[i])))
            h[i] += adj
            diff2 -= adj
            if abs(diff2) < 1e-14:
                break
    return h

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = corr.max()
    shifted = (corr - m) * beta
    return m + np.log(np.sum(np.exp(shifted))) / beta

def slsqp_solve(h, n, ts, beta, maxiter=400):
    res = minimize(smooth_obj, h.copy(), args=(n, beta), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def subgradient_hc(h, n, ts, t_start, time_limit, alpha0=0.04, topk=3, momentum=0.85):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    alpha = alpha0
    for it in range(500000):
        if time.time() - t_start > time_limit:
            break
        corr = compute_corr_vec(h)
        top_idx = np.argsort(corr)[-topk:]
        weights = np.exp((corr[top_idx] - corr[top_idx].max()) * 5.0)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            s = k - (n - 1)
            grad += w * corr_grad(h, s, n)
        vel = momentum * vel - alpha * grad
        h_new = project_feasible(h + vel, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        alpha *= 0.9998
    return best_h, best_c5

def sa_binary(h, n, ts, rng, t_start, time_limit, T0=0.004, T1=0.00005):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    # Ensure exactly n/2 ones
    ones_count = int(h.sum())
    if ones_count > n // 2:
        excess = np.sort(np.where(h == 1.0)[0])[:ones_count - n//2]
        h[excess] = 0.0
    elif ones_count < n // 2:
        deficit = np.sort(np.where(h == 0.0)[0])[:n//2 - ones_count]
        h[deficit] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    total_steps = max(1, int((time_limit - (time.time() - t_start)) * 8000))
    log_ratio = np.log(T1 / T0)
    cool_per_step = np.exp(log_ratio / max(total_steps, 1))
    T = T0
    for it in range(total_steps + 100000):
        if time.time() - t_start > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < -1e-15:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            if delta < T and rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
            else:
                h[i], h[j] = 1.0, 0.0
        T *= cool_per_step
    return best_h, best_c5
