def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    residual = ts - h.sum()
    if abs(residual) > 1e-12 and len(idx) > 0:
        h[idx[0]] += residual
        h[idx[0]] = np.clip(h[idx[0]], 0, 1)
    return h

def smooth_max_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    if beta < 1e6:
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    else:
        return m * (2.0 / n)

def slsqp_step(h, n, ts, beta, maxiter=500):
    def obj(x):
        return smooth_max_obj(x, n, beta)
    def cons_fun(x):
        return x.sum() - ts
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cons_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

                    improved = True
                else:

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_corr(h):
    """Returns (c5, corr) for potential reuse"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = ts - h.sum()
    if abs(residual) > 1e-12 and len(idx) > 0:
        # Distribute residual to non-bound indices
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + residual, 0, 1)
    return h

def smooth_max_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    if beta < 1e6:
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    else:
        return m * (2.0 / n)

        return smooth_max_obj(x, n, beta)

        return x.sum() - ts

        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,

                       options={'maxiter': maxiter, 'ftol': 1e-14})

        pass
    return project_feasible(h, ts)

def simulated_annealing(h, n, ts, t_start_sa=0.005, t_end_sa=0.0001, n_iter=500000, time_limit=60):

    rng = np.random.RandomState(int(time.time() * 1000) % 100000)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = ts - h.sum()
    if abs(residual) > 1e-12 and len(idx) > 0:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + residual, 0, 1)
    return h

def smooth_max_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    if beta < 1e6:
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    return m * (2.0 / n)

def slsqp_step(h, n, ts, beta, maxiter=500):
    def obj(x):
        return smooth_max_obj(x, n, beta)
    def cons_fun(x):
        return x.sum() - ts
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, ts)
    except Exception:
        pass
    return project_feasible(h, ts)

def greedy_swap_search(h, n, ts, max_sweeps=300, time_limit=60):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, ts)
    best_c5 = compute_c5(h_bin)
    t0 = time.time()

        if time.time() - t0 > time_limit:
            break
        zeros = np.where(h_bin < 0.5)[0]
        ones = np.where(h_bin > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in zeros:
            h_bin[i] = 1.0
            for j in ones:
                h_bin[j] = 0.0
                c5_new = compute_c5(h_bin)
                if c5_new < best_c5 + best_delta - 1e-15:
                    best_delta = c5_new - best_c5
                    best_i, best_j = i, j
                h_bin[j] = 1.0
            h_bin[i] = 0.0
        if best_i >= 0:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
            best_c5 = compute_c5(h_bin)
        else:
            break
    return h_bin, best_c5

def simulated_annealing_binary(h, n, ts, t_lo=0.003, t_hi=0.00003, n_iter=500000, time_limit=60, seed=0):
    rng = np.random.RandomState(seed)

    h_bin = project_feasible(h_bin, ts)
    cur_c5 = compute_c5(h_bin)
    best_c5 = cur_c5
    best_h = h_bin.copy()
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        alpha = it / max(n_iter, 1)
        T = t_lo * (t_hi / t_lo) ** alpha
        zeros = np.where(h_bin < 0.5)[0]
        ones = np.where(h_bin > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:

        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i] = 1.0
        h_bin[j] = 0.0

                best_h = h_bin.copy()
        else:
            h_bin[i] = 0.0
            h_bin[j] = 1.0
    return best_h, best_c5
