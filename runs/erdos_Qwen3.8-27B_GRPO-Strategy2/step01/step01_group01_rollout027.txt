```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            step = diff / mask.sum()
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def topk_smooth(h, K=20, t=50.0):
    c = correlations(h)
    # Use top-K soft max
    # Sort and take top K
    vals = np.sort(c)[::-1][:K]
    cmax = vals[0]
    return (cmax + np.log(np.sum(np.exp((vals - cmax) * t))) / t) * (2.0 / len(h))

def subgradient_topk(h, K=20):
    n = len(h)
    c = correlations(h)
    # Find top K lags
    top_idx = np.argsort(c)[::-1][:K]
    # Subgradient: for each active lag m, grad[h[j]] += (1-h[j-d]) - h[j+d] where d = m-(n-1)
    grad = np.zeros(n)
    for m in top_idx:
        d = m - (n - 1)
        # c[m] = sum_i h[i] * (1-h[i+d]) for valid i
        # dc[m]/dh[j] = (1-h[j+d]) - h[j-d] (with boundary = 0)
        for j in range(n):
            g = 0.0
            if 0 <= j + d < n:
                g += (1.0 - h[j + d])
            if 0 <= j - d < n:
                g -= h[j - d]
            grad[j] += g
    return grad / K  # average over top K

def projected_subgradient(h, n, max_iters=5000, K=20, momentum=0.9):
    best_h = h.copy()
    best_f = compute_c5(h)
    v = np.zeros(n)
    # Polyak: need upper bound
    U = best_f
    L = 0.0  # lower bound
    c = best_f + 1e-6  # slight offset
    
    for it in range(max_iters):
        grad = subgradient_topk(h, K)
        f = compute_c5(h)
        if f < best_f:
            best_f = f
            best_h = h.copy()
        
        # Polyak step
        diff = f - L  # simplified
        norm_sq = np.dot(grad, grad)
        if norm_sq < 1e-12:
            break
        step = diff / norm_sq
        step = min(step, 0.5)  # cap
        
        # Nesterov
        v = momentum * v - step * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        
        # Check improvement (accept/reject)
        f_new = compute_c5(h_new)
        if f_new < f + 1e-10:
            h = h_new
        else:
            # Reject, reduce step
            step *= 0.5
            h_new = np.clip(h + step * grad * (-1), 0.0, 1.0)
            h_new = repair(h_new, n)
            h = h_new
    
    return best_h, best_f
```

```python
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for it in range(iterations):
        # Compute gradient of smooth max over top K
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[::-1][:K]
        c_top = c[top_idx]
        # softmax weights
        c_max = c_top[0]
        weights = np.exp((c_top - c_max) * t)
        weights /= weights.sum()
        
        # Gradient: weighted sum of subgradients
        grad = np.zeros(n)
        for k_idx, m in enumerate(top_idx):
            d = m - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j + d < n:
                    g += (1.0 - h[j + d])
                if 0 <= j - d < n:
                    g -= h[j - d]
                grad[j] += weights[k_idx] * g
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_f
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def grad_smooth_max(h, n, K, t):
    """Gradient of smoothed max over top-K lags."""
    c = get_corr(h)
    top_idx = np.argpartition(c, -K)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    c_top = c[top_idx]
    c_max = c_top[0]
    w = np.exp((c_top - c_max) * t)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(K):
        d = n - 1 - top_idx[k]
        g = np.zeros(n)
        if d >= 0:
            g[:n-d] += (1.0 - h[d:n])
            g[d:n] -= h[0:n-d]
        else:
            nd = -d
            g[nd:n] += (1.0 - h[0:n-nd])
            g[0:n-nd] -= h[nd:n]
        grad += w[k] * g
    return grad * (2.0 / n)  # scale by dx

def adam_optimize(h, n, iters, lr, K, t, seed=42):
    best_h = h.copy()
    best_f = compute_c5(h)
    m_val = np.zeros(n)
    v_val = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(iters):
        g = grad_smooth_max(h, n, K, t)
        m_val = b1 * m_val + (1 - b1) * g
        v_val = b2 * v_val + (1 - b2) * g**2
        m_hat = m_val / (1 - b1**(it+1))
        v_hat = v_val / (1 - b2**(it+1))
        h_new = np.clip(h - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0, 1.0)
        h_new = repair(h_new, n)
        f = compute_c5(h_new)
        if f < best_f:
            best_f = f
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_f

def obj_slsqp(h, n, K, t):
    c = get_corr(h)
    top_idx = np.argpartition(c, -K)[-K:]
    c_top = c[top_idx]
    c_max = c_top.max()
    logsumexp = c_max + np.log(np.sum(np.exp((c_top - c_max) * t)))
    return (logsumexp / K) * (2.0 / n)

def cd_polish(h, n, max_sweeps=200, time_limit=120):
    t0 = time.time()
    best = compute_c5(h)
    target = n * 0.5
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0, 0.5, old + 0.1, old - 0.1]
            cands = [max(0.0, min(1.0, c)) for c in cands]
            cands.append(old)
            best_val = best
            best_c = old
            for cand in cands:
                if abs(cand - best_c) < 1e-10:
                    continue
                h[i] = cand
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    j = (i + 1) % n
                    h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                val = compute_c5(h)
                if val < best_val - 1e-13:
                    best_val = val
                    best_c = cand
            if best_c != old:
                h[i] = best_c
                diff = target - h.sum()
                if abs(diff) > 1e-12:
                    j = (i + 1) % n
                    h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                best = best_val
                improved = True
            else:
                h[i] = old
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    def remaining(frac=1.0):
        return budget_s * frac - elapsed()
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Adam at original resolution with various K, t
    if remaining(0.3) > 10:
        for K, t, lr in [(10, 50, 0.3), (20, 100, 0.2), (30, 200, 0.1), (15, 80, 0.25)]:
            if remaining(0.3) < 5:
                break
            iters = max(500, int(remaining(0.3) / 0.05))
            h_try, f_try = adam_optimize(best_h, n0, iters, lr, K, t)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try
    
    # Phase 2: SLSQP polish
    if remaining(0.55) > 30:
        for K, t in [(10, 200), (20, 100)]:
            if remaining(0.55) < 15:
                break
            def obj(x):
                return obj_slsqp(x, n0, K, t)
            def obj_grad(x):
                return grad_smooth_max(x, n0, K, t)
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n0*0.5,
                     'jac': lambda x: np.ones(n0)}]
            bnds = [(0.0, 1.0)] * n0
            maxiter = max(100, int(remaining(0.55) / 0.3))
            res = minimize(obj, best_h, jac=obj_grad, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_try = repair(res.x, n0)
            f_try = compute_c5(h_try)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try
    
    # Phase 3: CD polish
    if remaining(0.75) > 30:
        h_try, f_try = cd_polish(best_h, n0, max_sweeps=200, time_limit=remaining(0.75)-5)
        if f_try < best_c5:
            best_c5, best_h = f_try, h_try
    
    # Phase 4: Resolution increase
    for mult in [2, 3]:
        n = n0 * mult
        if n > 1000 or remaining(0.92) < 60:
            break
        xo = np.linspace(0, 2, n0)
        xn = np.linspace(0, 2, n)
        h_init = np.interp(xn, xo, best_h)
        h_init = repair(h_init, n)
        
        # Adam at new resolution
        iters = max(1000, int(remaining(0.92) / 0.05))
        h_try, f_try = adam_optimize(h_init, n, iters, 0.2, 15, 100)
        if f_try < best_c5:
            best_c5, best_h = f_try, h_try
            n0 = n
        
        # CD at new resolution
        if remaining(0.95) > 20:
            h_try, f_try = cd_polish(best_h, n0, max_sweeps=50, time_limit=remaining(0.95)-5)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try
    
    # Phase 5: Final perturbation + Adam restart
    if remaining(0.98) > 30:
        n_final = len(best_h)
        for _ in range(3):
            if remaining(0.98) < 10:
                break
            noise = np.random.randn(n_final) * 0.05
            h_try = repair(best_h + noise, n_final)
            iters = max(300, int(remaining(0.98) / 0.05))
            h_try, f_try = adam_optimize(h_try, n_final, iters, 0.15, 15, 150)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try
    
    # Final
    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
```

```python
            # old_j - δ ∈ [0,1] → δ ∈ [old_j-1, old_j]
            delta_min = max(-old_i, old_j - 1)
            delta_max = min(1 - old_i, old_j)

            # f(δ) = max_m (c_base[m] + δ * delta_b[m]) is convex in δ
            # So we can use ternary search or bisection on subgradient
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            d2 = target - h.sum()
            if abs(d2) > 1e-14:
                idx = int(np.argmin(h)) if d2 > 0 else int(np.argmax(h))
                h[idx] += d2
    return h

def compute_b(h, n, pos):
    D = n - 1 - np.arange(2 * n - 1)
    b = np.zeros(2 * n - 1)
    idx1 = pos + D
    idx2 = pos - D
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    b[v1] += (1.0 - h[idx1[v1]])
    b[v2] -= h[idx2[v2]]
    return b

def grad_smooth_topk(h, n, K, t):
    c = get_corr(h)
    top_idx = np.argpartition(c, -K)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    c_top = c[top_idx]
    c_max = c_top[0]
    w = np.exp((c_top - c_max) * t)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(K):
        grad += w[k] * compute_b(h, n, 0) * 0 + compute_b_weighted(h, n, top_idx[k])
    return grad * (2.0 / n)

def compute_b_weighted(h, n, m):
    D = n - 1 - m
    b = np.zeros(n)
    if D >= 0:
        b[:n - D] += (1.0 - h[D:n])
        b[D:n] -= h[0:n - D]
    else:
        nd = -D
        b[nd:n] += (1.0 - h[0:n - nd])
        b[0:n - nd] -= h[nd:n]
    return b

def adam_opt(h, n, iters, lr, K, t):
    best_h = h.copy()
    best_f = compute_c5(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    c_cache = get_corr(h)
    for it in range(iters):
        top_idx = np.argpartition(c_cache, -K)[-K:]
        top_idx = top_idx[np.argsort(c_cache[top_idx])[::-1]]
        c_top = c_cache[top_idx]
        c_max = c_top[0]
        w = np.exp((c_top - c_max) * t)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(K):
            grad += w[k] * compute_b_weighted(h, n, top_idx[k])
        grad *= (2.0 / n)
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad * grad
        m_hat = m_v / (1 - b1 ** (it + 1))
        v_hat = v_v / (1 - b2 ** (it + 1))
        h_new = np.clip(h - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0, 1.0)
        h_new = repair(h_new, n)
        f = compute_c5(h_new)
        if f < best_f:
            best_f = f
            best_h = h_new.copy()
        h = h_new
    return best_h, best_f

def cd_vectorized(h, n, max_sweeps=200, n_cands=31, time_limit=120):
    t0 = time.time()
    best = compute_c5(h)
    c_base = get_corr(h)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            j = (i + 1) % n
            old_i = h[i]
            old_j = h[j]
            b_i = compute_b(h, n, i)
            b_j = compute_b(h, n, j)
            db = b_i - b_j
            d_lo = max(-old_i, old_j - 1.0)
            d_hi = min(1.0 - old_i, old_j)
            if d_hi - d_lo < 1e-12:
                continue
            best_val = c_base.max()
            best_d = 0.0
            for d in np.linspace(d_lo, d_hi, n_cands):
                if abs(d) < 1e-13:
                    continue
                c_try = c_base + d * db
                val = c_try.max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            if best_d != 0.0:
                h[i] = old_i + best_d
                h[j] = old_j - best_d
                c_base += best_d * db
                if c_base.max() * (2.0 / n) < best:
                    best = float(c_base.max() * (2.0 / n))
                    improved = True
        if not improved:
            break
    return h, best

def obj_slsqp_topk(x, n, K, t):
    c = get_corr(x)
    top = np.partition(c, -K)[-K:]
    cm = top.max()
    lse = cm + np.log(np.sum(np.exp((top - cm) * t)))
    return (lse / K) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem(frac=1.0):
        return budget_s * frac - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Adam with multiple hyperparameter settings
    if rem(0.25) > 20:
        configs = [(12, 60, 0.4), (20, 120, 0.25), (8, 40, 0.5), (25, 200, 0.2), (15, 100, 0.35)]
        for K, t, lr in configs:
            if rem(0.25) < 15:
                break
            iters = max(800, int(rem(0.25) / 0.008))
            h_try, f_try = adam_opt(best_h, n0, iters, lr, K, t)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try.copy()

    # Phase 2: CD polish at n0
    if rem(0.45) > 20:
        h_try, f_try = cd_vectorized(best_h, n0, max_sweeps=300, n_cands=31,
                                     time_limit=max(10, rem(0.45) - 5))
        h_try = repair(h_try, n0)
        f_try = compute_c5(h_try)
        if f_try < best_c5:
            best_c5, best_h = f_try, h_try

    # Phase 3: SLSQP with top-K objective
    if rem(0.65) > 40:
        for K, t in [(10, 300), (20, 150)]:
            if rem(0.65) < 20:
                break
            cons = [{'type': 'eq', 'fun': lambda x, nn=n0: x.sum() - nn * 0.5,
                     'jac': lambda x, nn=n0: np.ones(nn)}]
            bnds = [(0.0, 1.0)] * n0
            maxiter = max(80, int(rem(0.65) / 0.4))
            try:
                res = minimize(lambda x: obj_slsqp_topk(x, n0, K, t), best_h.copy(),
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-14})
                h_try = repair(res.x, n0)
                f_try = compute_c5(h_try)
                if f_try < best_c5:
                    best_c5, best_h = f_try, h_try
            except Exception:
                pass

    # Phase 4: CD again after SLSQP
    if rem(0.75) > 10:
        h_try, f_try = cd_vectorized(best_h, n0, max_sweeps=200, n_cands=31,
                                     time_limit=max(5, rem(0.75) - 3))
        h_try = repair(h_try, n0)
        f_try = compute_c5(h_try)
        if f_try < best_c5:
            best_c5, best_h = f_try, h_try

    # Phase 5: Resolution increase
    for mult in [2, 3]:
        n_new = n0 * mult
        if n_new > 1000 or rem(0.93) < 60:
            break
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_new)
        h_init = np.interp(xn, xo, best_h)
        h_init = repair(h_init, n_new)
        # Adam at new resolution
        iters = max(500, int(rem(0.93) / 0.01))
        h_try, f_try = adam_opt(h_init, n_new, iters, 0.2, 15, 100)
        if f_try < best_c5:
            best_c5, best_h = f_try, h_try
            n0 = n_new
        # CD at new resolution
        if rem(0.96) > 15:
            h_try, f_try = cd_vectorized(best_h, n0, max_sweeps=100, n_cands=21,
                                         time_limit=max(5, rem(0.96) - 3))
            h_try = repair(h_try, n0)
            f_try = compute_c5(h_try)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try

    # Phase 6: Random restarts + Adam
    if rem(0.99) > 30:
        n_final = len(best_h)
        for attempt in range(4):
            if rem(0.99) < 10:
                break
            noise_scale = 0.04 + 0.02 * attempt
            h_try = repair(best_h + np.random.randn(n_final) * noise_scale, n_final)
            iters = max(400, int(rem(0.99) / 0.01))
            h_try, f_try = adam_opt(h_try, n_final, iters, 0.15, 12, 150)
            h_try = repair(h_try, n_final)
            f_try = compute_c5(h_try)
            if f_try < best_c5:
                best_c5, best_h = f_try, h_try
            # Quick CD
            if rem(0.99) > 8:
                h_try2, f_try2 = cd_vectorized(h_try, n_final, max_sweeps=50, n_cands=21,
                                               time_limit=max(3, rem(0.99) - 5))
                h_try2 = repair(h_try2, n_final)
                f_try2 = compute_c5(h_try2)
                if f_try2 < best_c5:
                    best_c5, best_h = f_try2, h_try2

    # Final
    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
