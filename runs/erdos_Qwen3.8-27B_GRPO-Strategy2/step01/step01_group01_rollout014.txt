            best_c5, best_h = v, h_new.copy()

    # Phase 2: Continuous local search
    if elapsed() < budget_s * 0.40:
        h_new, v = local_search(best_h, n0, 3000, seed)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()

    # Phase 3: SLSQP polish with high temp
    if elapsed() < budget_s * 0.50:
        h_new = slsqp_step(best_h, n0, 5000, 200)
        v = compute_c5(h_new)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()
        h_new, v = local_search(best_h, n0, 2000, seed + 7)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()

    # Phase 4: Binary local search
    if elapsed() < budget_s * 0.65:
        h_bin, v = binary_search(best_h, n0, 8000, seed + 1)
        if v < best_c5:
            best_c5, best_h = v, h_bin.copy()

    # Phase 5: Interpolate between continuous and binary
    if elapsed() < budget_s * 0.72:
        h_bin2, v_b = binary_search(best_h, n0, 4000, seed + 50)
        best_bin = min(v_b, best_c5)
        for alpha in [0.2, 0.4, 0.6, 0.8]:
            h_mix = (1.0 - alpha) * best_h + alpha * h_bin2
            h_mix = repair(h_mix, n0)
            v = compute_c5(h_mix)
            if v < best_c5:
                best_c5, best_h = v, h_mix.copy()

    # Phase 6: Simulated annealing from current best
    if elapsed() < budget_s * 0.82:
        h_new, v = anneal_search(best_h, n0, 3000, seed + 200)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()
        h_new, v = local_search(best_h, n0, 2000, seed + 300)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()

    # Phase 7: Try coarser resolution
    if elapsed() < budget_s * 0.90:
        for nc in [200, 150]:
            if elapsed() > budget_s * 0.88:
                break
            xo = np.linspace(0, 2, n0)
            xn = np.linspace(0, 2, nc)
            h_c = np.interp(xn, xo, best_h)
            h_c = repair(h_c, nc)
            h_c = slsqp_step(h_c, nc, 500, 150)
            v = compute_c5(h_c)
            h_c, v2 = local_search(h_c, nc, 1500, seed + nc)
            if v2 < v:
                h_c = h_c
                v = v2
            if v < best_c5:
                best_c5, best_h = v, h_c.copy()

    # Phase 8: Final polish
    if elapsed() < budget_s * 0.97:
        h_new = slsqp_step(best_h, len(best_h), 10000, 150)
        v = compute_c5(h_new)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()
        h_new, v = local_search(best_h, len(best_h), 1500, seed + 500)
        if v < best_c5:
            best_c5, best_h = v, h_new.copy()

    n_final = len(best_h)
    best_h = repair(best_h, n_final)
    c5_final = compute_c5(best_h)
    if c5_final > best_c5:
        best_c5 = c5_final
    return (best_h, best_c5, n_final)
