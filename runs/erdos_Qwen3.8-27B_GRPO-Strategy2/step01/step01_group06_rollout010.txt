            s = h.sum()

    cmax = np.max(corr)

    weights /= weights.sum()

            jm = m + lag

                val -= h[jm]
            grad[m] += w * val
    return dx * grad

from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_c5(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    s = np.sum(np.exp(T * (corr - cmax)))
    return cmax + np.log(s) / T

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def grad_softmax(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    weights = np.exp(T * (corr - cmax))
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        for m in range(n):
            val = 0.0
            im = m - lag
            if 0 <= im < n:
                val += (1.0 - h[im])
            jm = m + lag
            if 0 <= jm < n:
                val -= h[jm]
            grad[m] += w * val
    return dx * grad

    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return soft_max_c5(x, dx, T)
    def obj_grad(x):
        return grad_softmax(x, dx, T)
    def ceq(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

        order = np.argsort(-h[zero_idx])

            h_bin[zero_idx[order[i]]] = 1.0

    curr_c5 = best_c5

        T_curr = t_start * (t_end / t_start) ** (it / n_iter)
        # Pick random swap
        ones = np.where(curr_h == 1)[0]
        zeros = np.where(curr_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = eval_c5(new_h, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T_curr, 1e-10)):
            curr_h = new_h
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr_h.copy()
