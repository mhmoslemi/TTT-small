    eps_sched = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for eps in eps_sched:
        if time.time() > t_end - 8:
            break
        h_new = _slsqp(h, eps, maxiter=800)
        c = _c5(h_new, _dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    _eps = 0.001
    lr = 0.05
    for it in range(4000):
        if time.time() > t_end - 4:
            break
        h_new = _proj_grad_step(h, lr)
        c_new = _c5(h_new, _dx)
        c_cur = _c5(h, _dx)
        if c_new <= c_cur + 1e-14:
            h = h_new
            lr = min(lr * 1.02, 2.0)
        else:
            lr = max(lr * 0.95, 0.002)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if it % 200 == 100:
            h = _proj(best_h + rng.normal(0, 0.02, _n), _target)
    return best_h, best_c5

def _polish(h, t_end):
    global _eps
    best_h = h.copy()
    best_c5 = _c5(h, _dx)
    for eps in [0.0001, 0.00005, 0.00002, 0.00001]:
        if time.time() > t_end - 2:
            break
        hn = _slsqp(best_h, eps, maxiter=1000)
        cn = _c5(hn, _dx)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn
    return best_h, best_c5

def _try_n(h_source, n, t_end, rng):
    global _dx, _n, _target
    _n = n
    _dx = 2.0 / n
    _target = n / 2.0
    if len(h_source) != n:
        h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_source)), h_source)
    else:
        h0 = h_source.copy()
    h0 = _proj(h0, _target)
    best_h, best_c5 = _optimize_track(h0, t_end, rng)
    if time.time() < t_end - 3:
        best_h, best_c5 = _polish(best_h, t_end)
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)
    global _dx, _n, _target
    n70 = 70
    dx70 = 2.0 / n70
    tgt70 = n70 / 2.0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h70 = np.array(initial_h_values, dtype=float).copy()
        if len(h70) != n70:
            h70 = np.interp(np.linspace(0, 1, n70), np.linspace(0, 1, len(h70)), h70)
    else:
        h70 = rng.uniform(0.2, 0.8, n70)
    h70 = _proj(h70, tgt70)
    global _dx, _n, _target
    _dx, _n, _target = dx70, n70, tgt70
    best_overall = h70.copy()
    best_c5_overall = _c5(h70, dx70)
    best_n_overall = n70
    phase1_end = t0 + (deadline - t0) * 0.40
    h1, c1 = _optimize_track(h70, phase1_end, rng)
    if c1 < best_c5_overall:
        best_c5_overall, best_overall, best_n_overall = c1, h1, n70
    n_trials = [64, 80]
    phase2_end = t0 + (deadline - t0) * 0.70
    for ntr in n_trials:
        if time.time() > phase2_end - 15:
            break
        t_sub = min(phase2_end, time.time() + 60)
        hr, cr, nr = _try_n(best_overall, ntr, t_sub, rng)
        if cr < best_c5_overall:
            best_c5_overall, best_overall, best_n_overall = cr, hr, nr
    # Final phase: more restarts and polish on best config
    global _dx, _n, _target
    _dx, _n, _target = 2.0 / best_n_overall, best_n_overall, best_n_overall / 2.0
    best_h = best_overall.copy()
    best_c5 = best_c5_overall
    restart_end = deadline - 30
    n_restarts = 0
    while time.time() < restart_end:
        n_restarts += 1
        amp = max(0.01, 0.12 * (1.0 - n_restarts * 0.08))
        h_p = _proj(best_h + rng.normal(0, amp, _n), _target)
        c_p = _c5(h_p, _dx)
        if c_p < best_c5:
            best_c5, best_h = c_p, h_p.copy()
        sub_end = min(time.time() + min(18.0, restart_end - time.time()), restart_end)
        if sub_end - time.time() > 4:
            h2, c2 = _optimize_track(h_p, sub_end, rng)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
    # Final polish
    if time.time() < deadline - 5:
        hp, cp = _polish(best_h, deadline)
        if cp < best_c5:
            best_c5, best_h = cp, hp
    # Verify
    final_c5 = _c5(best_h, _dx)
    if abs(final_c5 - best_c5) > 1e-10:
        best_c5 = final_c5
    # Ensure exact feasibility
    s = best_h.sum()
    if abs(s - _target) > 1e-8:
        best_h = _proj(best_h, _target)
    return best_h, best_c5, best_n_overall
