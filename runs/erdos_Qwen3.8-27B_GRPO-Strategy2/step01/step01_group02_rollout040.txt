    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def eq_constraint(x):
    return np.sum(x) - _target_sum

def obj_max(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr)) * dx

def obj_lse(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def obj_pmean(x, n, p):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    corr_pos = np.maximum(corr, 1e-12)
    return float(np.mean(corr_pos ** p) ** (1.0 / p)) * dx

def obj_topk(x, n, k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    topk = np.sort(corr)[-k:]
    return float(np.sum(topk)) / k * dx

def do_slsqp(h, n, obj_fn, args, maxiter, ftol):
    global _target_sum
    _target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint}]
    try:
        r = minimize(obj_fn, h, args=args, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        hc = repair(r.x, n)
        return hc
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    global _target_sum
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)

    _target_sum = n / 2.0
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    bounds = [(0.0, 1.0)] * n

    t1 = budget_s * 0.12
    t2 = budget_s * 0.50
    t3 = budget_s * 0.78

    # Phase 1: Quick multi-surrogate optimization from parent
    if elapsed() < t1:
        surrogates = [
            (obj_max, (n,), 400, 1e-14),
            (obj_lse, (n, 25), 300, 1e-13),
            (obj_lse, (n, 60), 300, 1e-13),
            (obj_pmean, (n, 15), 300, 1e-13),
            (obj_topk, (n, 3), 300, 1e-13),
            (obj_topk, (n, 5), 300, 1e-13),
        ]
        for fn, a, mi, ft in surrogates:
            if elapsed() > t1:
                break
            hc = do_slsqp(h_best, n, fn, a, mi, ft)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-15:
                h_best, best_c5 = hc, c

    # Phase 2: Main iterative loop with diverse methods
    stall = 0
    it = 0
    while elapsed() < t2:
        it += 1
        improved = False

        if stall < 5:
            sigma = 0.015
        elif stall < 12:
            sigma = 0.05
        elif stall < 25:
            sigma = 0.10
        elif stall < 50:
            sigma = 0.18
        else:
            sigma = 0.25 + 0.01 * min(stall - 50, 10)

        h_pert = repair(h_best + rng.normal(0, sigma, n), n)

        # A: SLSQP exact max
        if time_left() > 25:
            hc = do_slsqp(h_pert, n, obj_max, (n,), 500, 1e-14)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-15:
                h_best, best_c5, improved = hc, c, True

        # B: Log-sum-exp with cycling beta
        if time_left() > 25 and it % 2 == 0:
            beta = 15.0 + 15.0 * (it % 6)
            hc = do_slsqp(h_pert, n, obj_lse, (n, beta), 300, 1e-13)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-15:
                h_best, best_c5, improved = hc, c, True

        # C: Top-k focus
        if time_left() > 25 and it % 3 == 0:
            k_val = 3 if it % 6 == 0 else 4
            hc = do_slsqp(h_pert, n, obj_topk, (n, k_val), 300, 1e-13)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-15:
                h_best, best_c5, improved = hc, c, True

        # D: Power mean
        if time_left() > 25 and it % 4 == 0:
            p_val = 8.0 + 6.0 * (it % 5)
            hc = do_slsqp(h_pert, n, obj_pmean, (n, p_val), 250, 1e-13)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-15:
                h_best, best_c5, improved = hc, c, True

        # E: Directed coordinate moves on near-active lags
        if time_left() > 15 and it % 2 == 0:
            dx = 2.0 / n
            corr = np.correlate(h_best, 1.0 - h_best, mode='full')
            threshold = np.max(corr) * 0.93
            active = np.where(corr >= threshold)[0]
            for _ in range(40):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                for eps in (0.008, 0.02, 0.05, 0.10):
                    hp = h_best.copy()
                    hp[i] += eps
                    hp[j] -= eps
                    if 0.0 <= hp[i] <= 1.0 and 0.0 <= hp[j] <= 1.0:
                        hp = repair(hp, n)
                        c = compute_c5(hp, n)
                        if c < best_c5 - 1e-15:
                            h_best, best_c5, improved = hp, c, True
                            break

        # F: Structural moves when stalled
        if time_left() > 20 and stall > 8:
            move_type = stall % 4
            if move_type == 0:
                start = int(rng.randint(max(1, n // 3)))
                length = int(rng.randint(2, min(6, n - start)))
                hp = h_best.copy()
                hp[start:start + length] = 1.0 - hp[start:start + length]
            elif move_type == 1:
                s1 = int(rng.randint(max(1, n // 3)))
                s2 = int(rng.randint(max(1, n // 3)))
                length = int(rng.randint(2, min(4, n // 4)))
                hp = h_best.copy()
                if s1 + length <= n and s2 + length <= n:
                    b1 = hp[s1:s1 + length].copy()
                    b2 = hp[s2:s2 + length].copy()
                    hp[s1:s1 + length] = b2
                    hp[s2:s2 + length] = b1
            elif move_type == 2:
                hp = np.roll(h_best, int(rng.randint(-3, 4)))
            else:
                hp = 1.0 - h_best

            hp = repair(hp, n)
            c = compute_c5(hp, n)
            if c < best_c5 - 1e-15:
                h_best, best_c5, improved = hp, c, True

        stall = 0 if improved else stall + 1

    # Phase 3: Try alternate n values
    if elapsed() < t3:
        for n_try in [60, 80, 100, 55, 90]:
            if elapsed() > t3:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_try = np.interp(x_new, x_old, h_best)
            h_try = repair(h_try, n_try)
            c_try = compute_c5(h_try, n_try)
            hc = do_slsqp(h_try, n_try, obj_max, (n_try,), 300, 1e-13)
            c_try = compute_c5(hc, n_try)
            if c_try < best_c5 - 1e-15:
                h_best, best_c5, n = hc, c_try, n_try
                bounds = [(0.0, 1.0)] * n
            hc2 = do_slsqp(h_try, n_try, obj_lse, (n_try, 40), 250, 1e-13)
            c_try2 = compute_c5(hc2, n_try)
            if c_try2 < best_c5 - 1e-15:
                h_best, best_c5, n = hc2, c_try2, n_try

    # Phase 4: Final intensive polish
    if time_left() > 8:
        for _ in range(3):
            if time_left() < 8:
                break
            hc = do_slsqp(h_best, n, obj_max, (n,), 600, 1e-15)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-16:
                h_best, best_c5 = hc, c
        for _ in range(8):
            if time_left() < 6:
                break
            hp = repair(h_best + rng.normal(0, 0.003, n), n)
            hc = do_slsqp(hp, n, obj_max, (n,), 400, 1e-14)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-16:
                h_best, best_c5 = hc, c
        if time_left() > 5:
            hc = do_slsqp(h_best, n, obj_lse, (n, 100), 500, 1e-15)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-16:
                h_best, best_c5 = hc, c

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
