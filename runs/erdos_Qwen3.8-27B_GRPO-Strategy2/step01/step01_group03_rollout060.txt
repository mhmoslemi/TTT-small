def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute difference to non-bound entries
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        idx = np.arange(len(h))
    per = diff / len(idx)
    h[idx] += per
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0)[0]
        if len(idx2) > 0:
            h[idx2[0]] = np.clip(h[idx2[0]] + diff2, 0, 1)
    return h

def smooth_max_top(corr, n, beta, topk):
    """Smooth max focusing on top-k correlation values."""
    dx = 2.0 / n
    c = corr * dx
    # Sort and take top-k
    c_sorted = np.sort(c)[::-1]
    k = min(topk, len(c_sorted))
    top_part = c_sorted[:k]
    # Smooth max of top-k
    m = top_part[0]
    shifted = (top_part - m) * beta
    smax = m + np.log(np.sum(np.exp(shifted))) / beta
    # Small penalty for the rest
    rest = c_sorted[k:]
    if len(rest) > 0:
        rest_penalty = np.max(rest) * 0.1
    else:
        rest_penalty = 0
    return smax + rest_penalty

def slsqp_solve(h, n, ts, beta, topk, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_top(corr, n, beta, topk)
    
    res = minimize(obj, h, method='SLSQP',

                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],

                h[j] = v

                    best_local_c5 = c5_t
                    best_local_val = v
            h[j] = best_local_val
            h = project_feasible(h, ts)
            if best_local_c5 < best_c5 - 1e-13:

                j, k = rng.choice(n, 2, replace=False)
                ov, ow = h[j], h[k]
                h[j], h[k] = (0.0, 1.0) if h[j] > 0.5 else (1.0, 0.0)
                h = project_feasible(h, ts)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                else:
                    h[j], h[k] = ov, ow
                    h = project_feasible(h, ts)
        
        if not improved:
            if sweep > 30:
                h1 = h + rng.normal(0, 0.06, n)
                h1 = project_feasible(h1, ts)
                c51 = compute_c5(h1)
                if c51 < best_c5:
                    best_c5 = c51
                    h = h1
                    improved = True
                elif sweep > 200:
                    break
            elif sweep > 20:
                break
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def remaining():
        return budget_s - (time.time() - t_start)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0
    
    best_h = project_feasible(h_init, len(h_init) / 2.0)
    best_c5 = compute_c5(best_h)
    
    n = len(best_h)
    ts = n / 2.0
    
    # Phase 1: SLSQP with annealed beta at initial resolution
    for beta, topk in [(25, 4), (60, 5), (150, 6), (400, 8)]:
        if remaining() < 35:
            break
        h = slsqp_opt(best_h.copy(), n, ts, beta, topk, maxiter=350)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Intensive local search
    if remaining() > 50:
        h_ls, c5_ls = local_search(best_h.copy(), n, ts, time_limit=min(90, remaining() - 15), seed=7)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 3: SLSQP polish
    for beta in [(250, 6), (500, 5)]:
        if remaining() < 25:
            break
        h = slsqp_opt(best_h.copy(), n, ts, beta, beta[1] if isinstance(beta, tuple) else 6, maxiter=300)
        if not isinstance(beta, tuple):
            h = slsqp_opt(best_h.copy(), n, ts, beta, 6, maxiter=300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 4: Multi-start with perturbation
    for r in range(6):
        if remaining() < 40:
            break
        sigma = 0.10 + 0.03 * r
        h_r = best_h + rng.normal(0, sigma, n)
        h_r = project_feasible(h_r, ts)
        h_r = slsqp_opt(h_r, n, ts, 80, 6, maxiter=200)
        h_r, c5_r = local_search(h_r, n, ts, time_limit=max(8, remaining() - 25), seed=100 + r * 13)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Binary projection + re-optimize
    if remaining() > 50:
        hbin = (best_h > 0.5).astype(float)
        hbin = project_feasible(hbin, ts)
        hbin = slsqp_opt(hbin, n, ts, 100, 6, maxiter=250)
        hbin, c5b = local_search(hbin, n, ts, time_limit=max(8, remaining() - 10), seed=500)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hbin.copy()
    
    # Phase 6: Try other resolutions
    for n_try in [100, 120, 160, 200, 280]:
        if remaining() < 50:
            break
        if n_try == n:
            continue
        ts_t = n_try / 2.0
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n_try)
        h_t = np.interp(xn, xo, best_h)
        h_t = project_feasible(h_t, ts_t)
        h_t = slsqp_opt(h_t, n_try, ts_t, 80, 6, maxiter=200)
        h_t, c5_t = local_search(h_t, n_try, ts_t, time_limit=max(8, remaining() - 18), seed=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
    
    # Final polish
    if remaining() > 15:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        best_h = slsqp_opt(best_h, n_f, ts_f, 400, 5, maxiter=250)
        best_h, best_c5 = local_search(best_h, n_f, ts_f, time_limit=min(12, remaining() - 5), seed=999)
    
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
