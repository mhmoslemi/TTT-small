    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * (2.0 / n)

def _project(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _subgradient_step(h, target, lr, top_k=50):
    """One subgradient step to reduce max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k lags
    idx = np.argsort(corr)[-top_k:]
    # For each top lag, compute subgradient
    # corr[k] = sum_i h[i] * (1-h[i+k]) (approximately)
    # d(corr[k])/d(h[j]) = (1-h[j+k]) - h[j-k+...] (depends on position)
    # This is complex. Let's use a simpler approach.
    
    # Alternative: for the lag with max correlation, we want to reduce h[i]*(1-h[i+k])
    # This means either decreasing h[i] or increasing h[i+k] (or vice versa)
    
    best_lag = np.argmax(corr)
    offset = best_lag - (n - 1)  # shift amount
    # corr at this lag: sum_{i} h[i+offset] * (1-h[i]) for valid i
    # Subgradient wrt h[j]: 
    #   if j appears as i+offset: contribution is (1-h[j-offset])
    #   if j appears as i: contribution is -h[j+offset]
    
    grad = np.zeros(n)
    if offset >= 0:
        for i in range(n - offset):
            grad[i + offset] += (1 - h[i])  # h[i+offset] factor
            grad[i] += (-h[i + offset])  # (1-h[i]) factor → deriv wrt h[i] is -h[i+offset]
    else:
        o = -offset
        for i in range(o, n):
            grad[i] += (1 - h[i - o])
            grad[i - o] += (-h[i])
    
    # Normalize gradient
    gn = np.linalg.norm(grad)
    if gn > 1e-12:
        grad /= gn
    
    # Step and project
    h_new = h - lr * grad
    h_new = _project(h_new, target)

import numpy as np
import time
from scipy.optimize import minimize

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _project_sum(h, target):
    """Adjust h so that sum(h) = target, staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Distribute difference among elements not at boundary
    can_up = h < 1.0 - 1e-12
    can_dn = h > 1e-12
    if diff > 0:
        mask = can_up
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    else:
        mask = can_dn
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _adaptive_project(h, target):
    """More careful projection maintaining sum."""
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        # Don't overshoot boundaries
        if diff > 0:
            max_step = 1.0 - h[mask]
        else:
            max_step = h[mask]
        actual = min(abs(step), max_step.min() if cnt > 0 else abs(step))
        h[mask] += np.sign(diff) * actual
    h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_corr(h, alpha):
    """Return (smoothed_max, weights) for log-sum-exp smoothing."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -200.0/alpha, 0.0)
    w = np.exp(alpha * sh)
    sw = np.sum(w)
    w /= sw
    s = m + np.log(sw) / alpha
    return s, w, corr

def _grad_smoothed(h, alpha):
    """Gradient of smoothed max correlation."""
    n = len(h)
    s, w, corr = _smooth_corr(h, alpha)
    A = 1.0 - 2.0 * h  # length n
    # grad[j] = sum_{s=0}^{n-1} A[s] * w[j+s]
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(A, w[j:j+n])
    return grad * (2.0 / n), w  # factor dx = 2/n

def _adam_step(h, target, alpha, lr, state, step_idx):
    """One Adam step with projection."""
    n = len(h)
    grad, _ = _grad_smoothed(h, alpha)
    
    m = state['m']
    v = state['v']
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    m = b1 * m + (1-b1) * grad
    v = b2 * v + (1-b2) * grad**2
    m_hat = m / (1 - b1**(step_idx+1))
    v_hat = v / (1 - b2**(step_idx+1))
    
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = _project_sum(h_new, target)
    
    state['m'] = m
    state['v'] = v
    return h_new

def _greedy_pairwise(h, target, max_rounds=50):
    """Greedy pairwise optimization."""
    n = len(h)
    best_c = _c5(h)
    rng = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n * 3):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:

    return h, best_c

def _slsqp_optimize(h, n, target, maxiter=80):
    """SLSQP optimization with soft-max objective."""
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        m = np.max(corr)
        sh = np.clip(corr - m, -0.5, 0.0)
        alpha = 500.0
        return (m + np.log(np.sum(np.exp(alpha*sh)))/alpha) * (2.0/n)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    result = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    return _project_sum(result.x, target)

import numpy as np
import time
from scipy.optimize import minimize

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -200.0 / alpha, 0.0)
    w = np.exp(alpha * sh)
    sw = np.sum(w)
    w /= sw
    A = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(A, w[j:j + n])
    return grad * (2.0 / n), w

def _adam_step(h, target, alpha, lr, state, idx):
    grad, _ = _smooth_grad(h, alpha)
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['m'] = b1 * state['m'] + (1 - b1) * grad
    state['v'] = b2 * state['v'] + (1 - b2) * grad ** 2
    mh = state['m'] / (1 - b1 ** (idx + 1))
    vh = state['v'] / (1 - b2 ** (idx + 1))
    h_new = h - lr * mh / (np.sqrt(vh) + eps)
    return _project_sum(h_new, target)

def _greedy_refine(h, target, rounds=40, rnd_seed=0):
    n = len(h)
    best_c = _c5(h)
    rng = np.random.RandomState(rnd_seed)
    deltas = [0.02, -0.02, 0.05, -0.05, 0.08, -0.08, 0.12, -0.12]
    for _ in range(rounds):
        moved = False
        order = rng.permutation(n)
        for i in order:
            if moved:
                continue
            for d in deltas:
                h2 = h.copy()
                h2[i] += d
                if h2[i] < 0 or h2[i] > 1:
                    continue
                for off in [1, 2, 3, -1, -2, -3]:
                    j = (i + off) % n
                    h2[j] -= d
                    if h2[j] < 0 or h2[j] > 1:
                        continue
                    c2 = _c5(h2)
                    if c2 < best_c - 1e-12:
                        h = h2
                        best_c = c2
                        moved = True
                        break
                if moved:
                    break
        if not moved:
            break
    return h, best_c

def _sine_init(n, rng, ns=0):
    x = np.arange(n)
    h = np.full(n, 0.5)
    if ns == 0:
        ns = rng.randint(2, 6)
    for _ in range(ns):
        f = rng.randint(1, max(2, n // 3))
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.05, 0.3)
        h += a * np.sin(2 * np.pi * f * x / n + p)
    return _project_sum(h, n / 2.0)

def _binary_init(n, rng, typ=0):
    if typ == 0:
        t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
        h = 0.3 + 0.4 * t
    elif typ == 1:
        h = rng.randint(0, 2, size=n).astype(float) * 0.35 + 0.3
    elif typ == 2:
        h = (np.arange(n) ** 2) % n
        h = (h < n // 2).astype(float) * 0.4 + 0.3
    else:
        h = rng.uniform(0.25, 0.75, size=n)
    return _project_sum(h, n / 2.0)

def _slsqp(h, n, target, maxiter=80, alpha=800.0):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(corr - m, -1.0, 0.0)
        return (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-11})
    return _project_sum(res.x, target)
