def repair(h, n):

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:

        h = np.clip(h, 0, 1)

    m = np.max(scaled)
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    return val * dx

            h = repair(np.array(init, dtype=float), trial_n)
        elif best_h is not None:
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, trial_n)
            h = np.interp(xn, xo, best_h)
            h = repair(h, trial_n)
        else:
            ta = np.linspace(0, 2, trial_n)
            h = 0.5 + 0.2 * np.sin(2 * np.pi * ta * 0.7) + 0.08 * np.cos(4 * np.pi * ta * 1.3)
            h = repair(h, trial_n)

        c5_cur = compute_c5(h, trial_n)
        if c5_cur < best_c5:
            best_h, best_c5, best_n = h.copy(), c5_cur, trial_n

        it = 0
        while time.time() < t_limit:
            it += 1
            improved = False

            # A: Nelder-Mead global exploration
            if it % 4 == 1 and time.time() < t_limit - 15:
                obj_fn = _nm_penalized if it < 8 else _nm_tight
                try:
                    r = minimize(obj_fn, h, method='Nelder-Mead',
                               options={'maxiter': 1500, 'xatol': 1e-9,
                                        'fatol': 1e-12, 'adaptive': True})
                    hc = repair(r.x, trial_n)
                    c = compute_c5(hc, trial_n)
                    if c < c5_cur - 1e-14:
                        h, c5_cur = hc, c
                        improved = True
                    if c5_cur < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5_cur, trial_n
                except Exception:
                    pass

            # B: SLSQP log-sum-exp (annealed beta)
            if it % 4 == 2 and time.time() < t_limit - 15:
                betas = [5, 15, 40, 100, 300, 600]
                _b_g = betas[min(it % len(betas), len(betas) - 1)]
                sig = 0.015 + 0.005 * min(it // 10, 5)
                hp = repair(h + rng.normal(0, sig, trial_n), trial_n)
                try:
                    r2 = minimize(_obj_lse, hp, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-14})
                    hc2 = repair(r2.x, trial_n)
                    c2 = compute_c5(hc2, trial_n)
                    if c2 < c5_cur - 1e-14:
                        h, c5_cur = hc2, c2
                        improved = True
                    if c5_cur < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5_cur, trial_n
                except Exception:
                    pass

            # C: SLSQP exact max
            if it % 4 == 3 and time.time() < t_limit - 15:
                try:
                    r3 = minimize(_obj_exact, h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 350, 'ftol': 1e-14})
                    hc3 = repair(r3.x, trial_n)
                    c3 = compute_c5(hc3, trial_n)
                    if c3 < c5_cur - 1e-14:
                        h, c5_cur = hc3, c3
                        improved = True
                    if c5_cur < best_c5:
                        best_h, best_c5, best_n = h.copy(), c5_cur, trial_n
                except Exception:
                    pass

            # D: Targeted coordinate moves
            if it % 2 == 0 and time.time() < t_limit - 8:
                for _ in range(25):
                    i = int(rng.randint(trial_n))
                    j = int(rng.randint(trial_n))
                    if i == j:
                        continue
                    for eps in (0.012, 0.03, 0.06, 0.1):
                        hp = h.copy()
                        hp[i] += eps
                        hp[j] -= eps
                        if hp[i] > 1.0 or hp[j] < 0.0:
                            continue
                        c_try = compute_c5(hp, trial_n)
                        if c_try < c5_cur - 1e-14:
                            h = repair(hp, trial_n)
                            c5_cur = compute_c5(h, trial_n)
                            improved = True
                            if c5_cur < best_c5:
                                best_h, best_c5, best_n = h.copy(), c5_cur, trial_n
                            break

            # E: Periodic structured perturbation
            if it % 5 == 4 and time.time() < t_limit - 5:
                freq = int(rng.choice([2, 3, 4, 5, 7, 9, 11]))
                amp = rng.uniform(0.01, 0.06)
                phase = rng.uniform(0, 2 * np.pi)
                idx = np.arange(trial_n)
                hp = h + amp * np.sin(2.0 * np.pi * freq * idx / trial_n + phase)
                hp = repair(hp, trial_n)
                c_try = compute_c5(hp, trial_n)
                if c_try < c5_cur - 1e-14:
                    h, c5_cur = hp, c_try
                    improved = True

            # F: Stagnation escape
            if not improved and it > 12:
                sig = 0.08 + 0.015 * min(it - 12, 10)
                hp = repair(h + rng.normal(0, sig, trial_n), trial_n)
                c_try = compute_c5(hp, trial_n)
                if c_try < c5_cur:
                    h, c5_cur = hp, c_try
                    improved = True

    # Final polish phase
    if time.time() - t0 < budget_s - 8:
        _n_g = best_n
        _t_g = best_n / 2.0
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': _eq_con}]
        h = best_h.copy()

        for rd in range(6):
            if time.time() - t0 > budget_s - 4:
                break
            _b_g = 50 + rd * 150
            try:
                r = minimize(_obj_lse, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15})
                hc = repair(r.x, best_n)
                c = compute_c5(hc, best_n)
                if c < best_c5:
                    h, best_c5 = hc, c
            except Exception:
                pass

        if time.time() - t0 < budget_s - 4:
            try:
                r = minimize(_obj_exact, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-15})
                hc = repair(r.x, best_n)
                c = compute_c5(hc, best_n)
                if c < best_c5:
                    h, best_c5 = hc, c
            except Exception:
                pass

        if time.time() - t0 < budget_s - 3:
            try:
                r = minimize(_nm_tight, h, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-12,
                                    'fatol': 1e-14, 'adaptive': True})
                hc = repair(r.x, best_n)
                c = compute_c5(hc, best_n)
                if c < best_c5:
                    h, best_c5 = hc, c
            except Exception:
                pass

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
