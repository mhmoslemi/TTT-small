The key insight: at the local minimum, the max C5 is achieved at (probably) 2-3 lags. To improve, we need to find a direction that reduces ALL of these simultaneously.

**Better approach: Active set quadratic programming**

At the optimum, we have:
- max_k c_k(h) = T (target)
- Active set A = {k : c_k(h) ≈ T}
- We want to find h' such that c_k(h') < T for all k in A (and ≤ T for all k)

This is equivalent to: find the maximum reduction δ such that there exists h' with:
- c_k(h') ≤ T - δ for all k
- 0 ≤ h' ≤ 1
- sum(h') = n/2

For small δ, we can linearize: c_k(h') ≈ c_k(h) + grad_k(h)·(h'-h) = T - δ requires:
- grad_k(h)·(h'-h) ≤ -δ for all k in A

This is a linear feasibility problem! We can solve it:
- Maximize δ subject to:
  - grad_k(h)·d ≤ -δ for all k in A (where d = h' - h)
  - -h ≤ d ≤ 1-h
  - sum(d) = 0
  - δ ≥ 0

This can be solved as a linear program! And it gives us a direction to move.

Then we can do a line search along this direction to find the actual improvement.

```python
def active_set_direction(h, n, top_k=5):
    """Find a feasible descent direction using active set LP"""
    from scipy.optimize import linprog
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, 'full')
    T = np.max(corr) * dx
    
    # Get top_k lags
    top_indices = np.argsort(corr)[-top_k:]
    
    # Compute gradients for these lags
    grads = []
    for k in top_indices:
        lag = k - n + 1
        g = np.zeros(n)
        a, b = max(0, lag), min(n, lag + n)
        if b > a:
            g[a:b] = 1.0 - h[a-lag:b-lag]
        a2, b2 = max(0, -lag), min(n, n - lag)
        if b2 > a2:
            g[a2:b2] -= h[a2+lag:b2+lag]
        grads.append(g * dx)
    
    grads = np.array(grads)  # shape (top_k, n)
    
    # LP: maximize delta
    # Variables: d[0..n-1], delta (total n+1 variables)
    # Constraints:
    #   For each k: grads[k] @ d + delta <= 0  (i.e., grads[k]@d <= -delta)
    #   -h[i] <= d[i] <= 1-h[i]
    #   sum(d) = 0
    #   delta >= 0
    
    # In linprog form: minimize -delta
    # c = [0, ..., 0, -1] (last variable is delta)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # grads[k] @ d + delta <= 0 for each k
    A_ub = np.hstack([grads, np.ones((top_k, 1))])
    b_ub = np.zeros(top_k)
    
    # -h <= d <= 1-h → -d <= h and d <= 1-h
    # In linprog (minimize): d <= 1-h → row of A_ub
    I = np.eye(n)
    A_ub2 = np.vstack([I, -I])
    b_ub2 = np.concatenate([1-h, h])
    
    A_ub_full = np.vstack([A_ub, A_ub2])
    b_ub_full = np.concatenate([b_ub, b_ub2])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d[i] in [-h[i], 1-h[i]], delta in [0, inf)
    res = linprog(c, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq, b_eq=b_eq,
                 bounds=list(zip(-h, 1-h)) + [(0, None)],
                 method='highs', options={'maxiter': 10000})
    
    if res.status == 0:
        d = res.x[:n]
        return d, -res.fun  # direction, delta value
    return None, 0.0
```

```python
        else:
            hi = mid

                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-13: break
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, beta, pw):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, 'full')
    s = beta * corr
    m = s.max()
    e = np.exp(s - m)
    total = e.sum()
    obj = (m + np.log(total)) / beta * dx
    w = e / total
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-15: continue
        lag = k - n + 1
        a, b = max(0, lag), min(n, lag + n)
        if b > a: g[a:b] += w[k] * (1.0 - h[a-lag:b-lag])
        a2, b2 = max(0, -lag), min(n, n - lag)
        if b2 > a2: g[a2:b2] -= w[k] * h[a2+lag:b2+lag]
    g *= beta * dx
    diff = h.sum() - n / 2.0
    obj += 0.5 * pw * diff * diff
    g += pw * diff
    return obj, g

def active_dir(h, n, n_lags=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    T = corr.max()
    thresh = T * 0.9
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.argsort(corr)[-n_lags:]
    active = np.sort(active)[-n_lags:]
    grads = np.zeros((len(active), n))
    for i, k in enumerate(active):
        lag = k - n + 1
        a, b = max(0, lag), min(n, lag + n)
        if b > a: grads[i, a:b] = 1.0 - h[a-lag:b-lag]
        a2, b2 = max(0, -lag), min(n, n - lag)
        if b2 > a2: grads[i, a2:b2] -= h[a2+lag:b2+lag]
    # Linprog: max delta s.t. grads[k]@d + delta <= 0, -h<=d<=1-h, sum(d)=0, delta>=0
    c = np.zeros(n + 1); c[-1] = -1.0
    A_ub = np.hstack([grads * dx, np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    I = np.eye(n)
    A_full = np.vstack([A_ub, I, -I])
    b_full = np.concatenate([b_ub, 1.0 - h, h])
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    bounds = list(zip(-h, 1.0 - h)) + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=[0.0],
                     bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.status == 0:
            return res.x[:n], -res.fun
    except:
        pass
    return None, 0.0

    return h, best_c
```
