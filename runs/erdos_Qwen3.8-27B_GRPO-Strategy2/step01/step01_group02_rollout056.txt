def compute_corr(h, n):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    # Iterative repair: scale, clip, adjust
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Distribute to non-boundary elements
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.sum(h > 0.1) + np.sum(h < 0.9)) // 2
        if 0 <= idx < n:
            h[idx] += diff
            h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def project_point(h, n, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.clip(h, 0.0, 1.0)
    # Water-filling for the sum constraint
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_gradient(h, n, top_m=3):
    """Compute approximate gradient of C5 using top M active lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top M lags
    m = min(top_m, len(corr))
    top_idx = np.argsort(corr)[-m:]
    # For each active lag, compute gradient of corr[k] w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i+k_offset]) -- but we need to be careful
    # Use numerical gradient for reliability
    grad = np.zeros(n)
    for k in top_idx:
        # Numerical gradient of corr[k]
        eps = 1e-7
        for j in range(n):
            h[j] += eps
            c_plus = np.correlate(h, 1.0 - h, mode='full')[k]
            h[j] -= 2*eps
            c_minus = np.correlate(h, 1.0 - h, mode='full')[k]
            h[j] += eps
            grad[j] += (c_plus - c_minus) / (2*eps)
    return grad / m

    # soft_max = (1/beta)*log(sum exp(scaled)) = (1/beta)*(m + log(sum exp(scaled-m)))
    
    c5_soft = (m + np.log(np.sum(exps))) / beta
    # Wait, (1/beta)*log(sum exp(scaled)) where scaled = beta*corr*dx
    # = (1/beta)*log(sum exp(beta*corr*dx))
    # This has units of corr*dx = integral. Yes, correct.
    
    # Gradient:
    # d/dh[j] of soft_max = (1/beta) * d/dh[j] log(sum_m exp(beta*corr_m*dx))
    # = (1/beta) * sum_m exp(beta*corr_m*dx) / sum_l exp(beta*corr_l*dx) * beta*dx * d(corr_m)/dh[j]
    # = dx * sum_m p[m] * d(corr_m)/dh[j]
    # where p[m] = exp(scaled[m])/sum_l exp(scaled[l])
    
    # d(corr_m)/dh[j] = (1-h[j-L_m])*I(0<=j-L_m<n) - h[j+L_m]*I(0<=j+L_m<n)
    # where L_m = m-(n-1)
    
    grad = np.zeros(n)
    L = np.arange(-(n-1), n)  # L values
    # For each j, compute the gradient
    for j in range(n):
        # First term: sum over L where 0<=j-L<n
        idx1 = (j - L >= 0) & (j - L < n) & (L >= -(n-1)) & (L < n)
        # L = m-(n-1), so m = L+(n-1), and p[m] = p[L+(n-1)]
        p_vals = p[L + (n-1)]
        # First sum
        i1 = j - L[idx1]  # indices into h
        grad[j] += np.sum(p_vals[idx1] * (1 - h[i1]))
        # Second sum: sum over L where 0<=j+L<n
        idx2 = (j + L >= 0) & (j + L < n) & (L >= -(n-1)) & (L < n)
        i2 = j + L[idx2]
        grad[j] -= np.sum(p_vals[idx2] * h[i2])
    
    grad *= dx
    return c5_soft, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_c5_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr * dx
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    p = exps / np.sum(exps)
    c5_soft = (mx + np.log(np.sum(exps))) / beta
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for m in range(2 * n - 1):
            L = m - (n - 1)
            i1 = j - L
            i2 = j + L
            if 0 <= i1 < n:
                s1 += p[m] * (1.0 - h[i1])
            if 0 <= i2 < n:
                s2 += p[m] * h[i2]
        grad[j] = dx * (s1 - s2)
    return c5_soft, grad

def grad_exact_max(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    p = np.zeros(2 * n - 1)
    p[idx] = 1.0 / top_k
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for m in idx:
            L = m - (n - 1)
            i1 = j - L
            i2 = j + L
            if 0 <= i1 < n:
                s1 += (1.0 - h[i1])
            if 0 <= i2 < n:
                s2 += h[i2]
        grad[j] = dx * (s1 - s2) / top_k
    return grad

    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h_new)
        if abs(diff) < 1e-13:

        mask = (h_new > 1e-9) & (h_new < 1.0 - 1e-9)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h_new[mask] += diff / cnt
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def slsqp_polish(h, n, time_limit=30):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    t_start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for maxiter in [100, 200, 300]:
        if time.time() - t_start > time_limit:
            break
        try:
            r = minimize(lambda x: compute_c5(x, n), best_h, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c:
                best_h, best_c = hc, c
        except Exception:
            pass
    return best_h, best_c
