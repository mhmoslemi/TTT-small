import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gtau = 1.0
_cN = 0

def _obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    pen = 500.0 * (np.sum(x) - _cN * 0.5) ** 2
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx + pen

def _cfun(x):
    return np.sum(x) - _cN * 0.5

def _cjac(x):
    return np.ones(_cN)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n * 0.5
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10 and n > 0:
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def coord_sweep(h, n, n_t=81):
    h = h.copy()
    ts = np.linspace(0.0, 1.0, n_t)
    k0 = n - 1
    order = np.arange(n)
    for j in order:
        h[j] = 0.0
        cb = np.correlate(h, 1.0 - h, mode="full")
        h[j] = 1.0
        co = np.correlate(h, 1.0 - h, mode="full")
        d = co - cb
        ct = cb[None, :] + ts[:, None] * d[None, :]
        ct[:, k0] = cb[k0] + ts * (1.0 - ts)
        maxes = np.max(ct, axis=1)
        h[j] = ts[np.argmin(maxes)]
    return repair(h, n)

def coord_sweep_shuffled(h, n, n_t=81, rng=None):
    h = h.copy()
    ts = np.linspace(0.0, 1.0, n_t)
    k0 = n - 1
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    for j in order:
        h[j] = 0.0
        cb = np.correlate(h, 1.0 - h, mode="full")
        h[j] = 1.0
        co = np.correlate(h, 1.0 - h, mode="full")
        d = co - cb
        ct = cb[None, :] + ts[:, None] * d[None, :]
        ct[:, k0] = cb[k0] + ts * (1.0 - ts)
        maxes = np.max(ct, axis=1)
        h[j] = ts[np.argmin(maxes)]
    return repair(h, n)

def binarize_push(h, n, gamma=1.3):
    h2 = 0.5 + (h - 0.5) * gamma
    return repair(np.clip(h2, 0.0, 1.0), n)

def slsqp_optimize(x0, n, dx, tau, maxiter=200):
    global _gdx, _gtau, _cN
    _cN = n
    _gdx = dx
    _gtau = tau
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_obj_lse, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def perturb_h(h, scale, rng):
    return repair(h + scale * rng.standard_normal(len(h)), len(h))

def project_grid(h_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = np.interp(xd, xs, h_src)
    return repair(xp, n_dst)
