def subgradient_step(h, n, alpha, beta):
    """Take a subgradient step using log-sum-exp smoothing"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max via log-sum-exp
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    weights = exps / np.sum(exps)
    # Gradient is weighted sum of individual gradients
    n_corr = len(corr)  # 2n-1
    lag_offset = n - 1  # corr index k corresponds to lag k - lag_offset
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - lag_offset  # actual lag
        for j in range(n):
            gj = 0.0
            # h[j] appears as h[i+k] where i = j-k: contribution (1-h[j-k])
            if 0 <= j - k < n:

            if 0 <= j + k < n:
                gj -= h[j + k]
            grad[j] += weights[k_idx] * gj
    h_new = h - alpha * grad
    return h_new

        if it % 20 == 0 and stalled > 10:
            h = project_feasible(h + 0.05 * np.random.randn(n), n)
            stalled = 0
    if time.time() < t_end - 20:
        h2 = slsqp_polish(best_h, n, maxiter=500)
        c2 = compute_c5(h2, n)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    if time.time() < t_end - 15:
        for b in [80, 150, 300]:
            if time.time() > t_end - 15:
                break
            h3 = slsqp_lse(best_h, n, beta=b, maxiter=150)
            c3 = compute_c5(h3, n)
            if c3 < best_c5:
                best_h, best_c5 = h3, c3
    if time.time() < t_end - 8:
        h4 = slsqp_polish(best_h, n, maxiter=600, ftol=1e-15)
        c4 = compute_c5(h4, n)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4
    return best_h, best_c5

def pairwise_refine(h, n, t_end, n_iter=200):
    target = n / 2.0
    best_c5 = compute_c5(h, n)
    rng = np.random.RandomState(12345)
    for _ in range(n_iter):
        if time.time() > t_end - 10:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        for eps in [0.01, 0.03, 0.06, 0.1]:
            hp = h.copy()
            hp[i] += eps
            hp[j] -= eps
            if hp[i] > 1.0 or hp[j] < 0.0:
                continue
            hp = project_feasible(hp, n)
            c = compute_c5(hp, n)
            if c < best_c5 - 1e-14:
                h, best_c5 = hp, c
                break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    best_overall_c5 = 1.0
    best_overall_h = None
    best_overall_n = 70
    
    n_values = [70, 55, 65, 80]
    init = globals().get('initial_h_values', None)
    
    for n in n_values:
        if time.time() > t_end - 80:
            break
        starts = []
        if init is not None and len(init) == n:
            starts.append(np.array(init, dtype=float))
        starts.append(np.full(n, 0.5))
        x = np.linspace(0, 2, n, endpoint=False)
        starts.append(project_feasible(0.5 + 0.35 * np.sin(2 * np.pi * x * 1.0), n))
        starts.append(project_feasible(0.5 + 0.3 * np.cos(2 * np.pi * x * 2.0), n))
        starts.append(project_feasible(rng.rand(n), n))
        
        n_starts = min(len(starts), 4)
        time_per_start = (t_end - 80 - time.time()) / max(1, n_starts)
        for idx in range(n_starts):
            if time.time() > t_end - 50:
                break
            t_sub = min(t_end - 40, t0 + time.time() - t0 + time_per_start)
            h_c, c_c = optimize_from_start(starts[idx], n, t_sub)
            h_c, c_c = pairwise_refine(h_c, n, t_sub, n_iter=80)
            if c_c < best_overall_c5:
                best_overall_c5 = c_c
                best_overall_h = h_c
                best_overall_n = n
    
    if best_overall_h is not None and time.time() < t_end - 10:
        hp = slsqp_polish(best_overall_h, best_overall_n, maxiter=600, ftol=1e-15)
        cp = compute_c5(hp, best_overall_n)
        if cp < best_overall_c5:
            best_overall_h, best_overall_c5 = hp, cp
        hp2 = pairwise_refine(best_overall_h, best_overall_n, t_end, n_iter=300)
        cp2 = compute_c5(hp2[0], best_overall_n)
        if cp2 < best_overall_c5:
            best_overall_h, best_overall_c5 = hp2[0], cp2
        hp3 = slsqp_polish(best_overall_h, best_overall_n, maxiter=500, ftol=1e-15)
        cp3 = compute_c5(hp3, best_overall_n)
        if cp3 < best_overall_c5:
            best_overall_h, best_overall_c5 = hp3, cp3
    
    if best_overall_h is None:
        best_overall_h = np.full(70, 0.5)
        best_overall_c5 = 0.5
        best_overall_n = 70
    
    return (best_overall_h, best_overall_c5, best_overall_n)
