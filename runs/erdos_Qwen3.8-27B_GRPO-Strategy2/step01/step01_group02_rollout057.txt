Actually let me just compute it directly. Let me define:

- output[m] for m = 0, ..., 2n-2
- output[m] = sum_i a[i] * b[i+m-(n-1)] for valid i

- From (1-h[i+d]) factor: i+d=j, i=j-d, valid if 0<=j-d<n → contributes -h[j-d]

1. Compute all correlations
2. Find active set (top few lags)
3. Compute average gradient over active set

    lo, hi = -5.0, 5.0

    """Compute subgradient of max correlation using top-k active lags."""

                g += (1.0 - h[j + d])
            if 0 <= j - d < n:
                g -= h[j - d]
            grad[j] += g / k_active

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(x, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def active_gradient(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    thresh = cmax - 0.005 * max(cmax, 1e-10)
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = [int(np.argmax(corr))]
    if len(active) > top_k:
        top_idx = np.argsort(corr[active])[::-1][:top_k]
        active = active[top_idx]
    k_act = len(active)
    grad = np.zeros(n)
    for m in active:
        d = m - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + d < n:
                g += (1.0 - h[j + d])
            if 0 <= j - d < n:
                g -= h[j - d]
            grad[j] += g / k_act
    return grad

    return compute_c5(x, _gn)

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(corr - m)))) / beta * dx

    return np.sum(x) - _gn / 2.0
