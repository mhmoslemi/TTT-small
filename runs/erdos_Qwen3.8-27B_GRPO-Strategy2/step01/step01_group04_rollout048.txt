```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x *= tgt / s
    x = np.clip(x, 0.0, 1.0)
    return x

def soft_obj(x, dx, tau, topk_frac=1.0):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    if topk_frac < 1.0:
        k = max(int(len(corr) * topk_frac), 5)
        idx = np.argpartition(corr, -k)[-k:]
        corr = corr[idx]
    m = np.max(corr)
    if tau < 1e-14:
        return m
    sh = np.clip((corr - m) / tau, -55.0, 0.0)
    return m + tau * np.log(np.sum(np.exp(sh)))

def opt(x0, n, dx, tau, mi=500, kfrac=1.0):
    def f(x):
        return soft_obj(x, dx, tau, kfrac)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': lambda v: np.sum(v) - n/2.0,
           'jac': lambda v: np.ones(n)}]
    try:
        r = minimize(f, x0, method='SLSQP', bounds=bd, constraints=cs,
                     options={'maxiter': mi, 'ftol': 1e-16})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def project(x, ns, nd):
    xs = np.linspace(0, 2, ns)
    xd = np.linspace(0, 2, nd)
    xp = interp1d(xs, x, kind='linear', fill_value='extrapolate',
                  bounds_error=False)(xd)
    return repair(xp, nd)

def perturb_diverse(x, n, rng):
    out = []
    out.append(repair(x + 0.06 * rng.standard_normal(n), n))
    out.append(repair(x + 0.13 * rng.standard_normal(n), n))
    xb = x.copy()
    bl = max(n // 8, 2)
    for _ in range(3):
        s = int(rng.integers(0, n - bl))
        xb[s:s+bl] += 0.18 * rng.standard_normal(bl)
    out.append(repair(xb, n))
    f = rng.uniform(1.5, 5.0)
    p = rng.uniform(0, 6.28)
    t = np.linspace(0, 6.28 * f, n)
    out.append(repair(x + 0.09 * np.sin(t + p), n))
    xa = x.copy()
    h = n // 2
    xa[h:] += 0.08 * rng.standard_normal(n - h)
    xa[:h] -= 0.08 * rng.standard_normal(h)
    out.append(repair(xa, n))
    return out

def binary_push(x, n, threshold=0.5):
    x2 = x.copy()
    x2[x2 > threshold] = 1.0
    x2[x2 < threshold] = 0.0
    s = np.sum(x2)
    diff = n / 2.0 - s
    if diff != 0:
        if diff > 0:
            idx = np.argsort(x)[::-1]
            for i in range(int(diff)):
                x2[idx[i]] = min(1.0, x2[idx[i]] + 1.0)
        else:
            idx = np.argsort(x)
            for i in range(int(-diff)):
                x2[idx[i]] = max(0.0, x2[idx[i]] - 1.0)
    return repair(x2, n)

def refine_chain(x, n, dx, time_budget, t_last):
    t0 = time.time()
    best = x.copy()
    bv = compute_c5(best, dx)
    elapsed = lambda: time.time() - t0
    taus = [0.25, 0.1, 0.04, 0.015, 0.006, 0.0025, 0.001, 0.0004]
    kfracs = [1.0, 1.0, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2]
    for i, (tau, kf) in enumerate(zip(taus, kfracs)):
        if elapsed() > time_budget - 3:
            break
        mi = 500 if i < 3 else 350
        x_new = opt(x, n, dx, tau, mi, kf)
        v = compute_c5(x_new, dx)
        if v < bv:
            bv = v
            best = x_new.copy()
            x = x_new
        else:
            x_new2 = opt(best, n, dx, tau, 200, kf)
            v2 = compute_c5(x_new2, dx)
            if v2 < bv:
                bv = v2
                best = x_new2.copy()
                x = x_new2
    return best, bv

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = repair(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    remaining = lambda: budget_s - (time.time() - t_start)

    # Phase 1: Core optimization on initial grid
    p1_bud = budget_s * 0.30
    x = h0.copy()
    for tau in [0.3, 0.12, 0.05, 0.02, 0.008, 0.003]:
        if remaining() < 10:
            break
        mi = 550
        x = opt(x, n0, dx0, tau, mi, 1.0)
        v = compute_c5(x, dx0)
        if v < best_val:
            best_val = v
            best_h = x.copy()

    # Phase 2: Multi-restart with diverse perturbations
    p2_budget = min(budget_s * 0.35, remaining() - 60)
    if p2_budget > 20:
        n_r = 6
        for r in range(n_r):
            if remaining() < 20 or (time.time() - t_start) > t_start + budget_s * 0.65:
                break
            pts = perturb_diverse(best_h, n0, rng)
            for pt in pts:
                if remaining() < 12:
                    break
                pt_budget = min(p2_budget / (n_r * len(pts)), remaining() - 8)
                if pt_budget < 5:
                    continue
                x_r, v_r = refine_chain(pt, n0, dx0, pt_budget, t_start)
                if v_r < best_val:
                    best_val = v_r
                    best_h = x_r.copy()

    # Phase 3: Binary push heuristic
    if remaining() > 40:
        x_b = binary_push(best_h, n0, 0.5)
        dx_b = 2.0 / n0
        x_b2, v_b = refine_chain(x_b, n0, dx_b, min(30, remaining() - 15), t_start)
        if v_b < best_val:
            best_val = v_b
            best_h = x_b2.copy()

    # Phase 4: Coarser grid exploration
    n_c = max(n0 // 2, 50)
    if remaining() > 40 and n_c >= 50:
        dx_c = 2.0 / n_c
        x_c = project(best_h, n0, n_c)
        x_c2, v_c = refine_chain(x_c, n_c, dx_c, min(35, remaining() - 15), t_start)
        if v_c < best_val:
            best_val = v_c
            best_h = x_c2.copy()
            best_n = n_c

    # Phase 5: Finer grid if time permits
    n_f = 2 * best_n + 1
    if remaining() > 60 and n_f <= 800:
        dx_f = 2.0 / n_f
        x_f = project(best_h, best_n, n_f)
        x_f2, v_f = refine_chain(x_f, n_f, dx_f, min(50, remaining() - 20), t_start)
        if v_f < best_val:
            best_val = v_f
            best_h = x_f2.copy()
            best_n = n_f

    # Phase 6: Final fine polish
    if remaining() > 15:
        dx_fin = 2.0 / best_n
        for tau in [0.0015, 0.0006, 0.00025, 0.0001]:
            if remaining() < 6:
                break
            x_p = opt(best_h, best_n, dx_fin, tau, 300, 0.3)
            v = compute_c5(x_p, dx_fin)
            if v < best_val:
                best_val = v
                best_h = x_p.copy()

    dx_fn = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_fn)
    return (best_h, final_val, best_n)
```
