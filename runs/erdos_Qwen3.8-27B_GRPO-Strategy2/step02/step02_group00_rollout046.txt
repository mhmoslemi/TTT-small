import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b


def _binary_ls(b, dx, max_passes=30):
    n = len(b)
    best_c = _c5(b, dx)
    ones = set(np.where(b == 1.0)[0].tolist())
    zeros = set(np.where(b == 0.0)[0].tolist())
    for _ in range(max_passes):
        improved = False
        for i in list(ones):
            for j in list(zeros):
                b[i] = 0.0
                b[j] = 1.0
                c = _c5(b, dx)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                    improved = True
                else:
                    b[i] = 1.0
                    b[j] = 0.0
        if not improved:
            break
    return b.copy(), best_c


def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)


def _obj_topk(x, k, n):
    corr = _corr(x)
    if k >= len(corr):
        return float(corr.sum())
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum()) * (2.0 / n)


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _fine_local(h, n, dx, best_c5, deltas, max_j=6):
    h = h.copy()
    improved = False
    for i in range(n):
        vi = h[i]
        for d in deltas:
            ni = vi + d
            if ni < 0.0 or ni > 1.0:
                continue
            need = -d
            if need > 0:
                order = np.argsort(h)[:max_j]
            else:
                order = np.argsort(-h)[:max_j]
            for j in order:
                if j == i:
                    continue
                nj = h[j] + need
                if nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                c_try = _c5(h, dx)
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    improved = True
                    break
                else:
                    h[i] = vi
                    h[j] = h[j] - need
    return h, best_c5, improved
