```python
    # Find top-k correlations
    idx = np.argpartition(corr, -top_k)[-top_k:]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.max(corr))

def _binarize(h, n, rng):
    """Round to binary with exactly n/2 ones"""
    # Sort by value, pick top n/2
    h = np.clip(h, 0, 1)
    thresh = np.partition(h, n//2)[n//2]
    binary = (h >= thresh).astype(float)
    # Fix count
    ones = int(binary.sum())
    target = n // 2
    if ones > target:
        # Too many ones, flip some
        one_idx = np.where(binary == 1)[0]
        # Flip those with lowest h values
        order = np.argsort(h[one_idx])
        for k in range(ones - target):
            binary[one_idx[order[k]]] = 0
    elif ones < target:
        zero_idx = np.where(binary == 0)[0]
        order = np.argsort(-h[zero_idx])
        for k in range(target - ones):
            binary[zero_idx[order[k]]] = 1
    return binary

def _local_search_binary(h, n, max_iter=1000):
    """Local search on binary space with swap moves"""
    h = h.copy()
    target = n // 2
    mc = _max_corr(h)
    
    for it in range(max_iter):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_mc = mc
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_mc = _max_corr(h)
                h[i] = 1
                h[j] = 0
                if new_mc < best_mc:
                    best_mc = new_mc
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            mc = best_mc
        else:
            break
    
    return h, mc
```

```python
def _swap_delta(h, i, j, n):
    """Compute change in correlation array when h[i]: 1→0, h[j]: 0→1"""
    # i is currently 1, j is currently 0
    delta = np.zeros(2*n-1)
    
    # Effect of h[i] going from 1 to 0:
    # corr[k] includes h[i]*(1-h[i+k-(n-1)]) for valid index
    # This term changes from (1-h[i+k-(n-1)]) to 0
    # So delta[k] -= (1-h[j_idx]) where j_idx = i+k-(n-1)
    for k in range(2*n-1):
        j_idx = i + k - (n-1)
        if 0 <= j_idx < n:
            delta[k] -= (1.0 - h[j_idx])
    
    # Effect on corr[k] from the j-side: h[j_idx]*(1-h[i]) where j_idx+k-(n-1)=i
    # i.e., j_idx = i - k + (n-1)
    # This term changes from h[j_idx]*(1-1)=0 to h[j_idx]*(1-0)=h[j_idx]
    for k in range(2*n-1):
        j_idx = i - k + (n-1)
        if 0 <= j_idx < n:
            delta[k] += h[j_idx]

    # corr[k] includes h[j]*(1-h[j+k-(n-1)]) for valid index
    # This term changes from 0 to (1-h[j_idx]) where j_idx = j+k-(n-1)
    for k in range(2*n-1):

        if 0 <= j_idx < n:
```

```python
def _swap_delta_vec(h, i, j, n):

    delta = np.zeros(2*n-1)

    # For shift s: pair (i-s, i) was (h[i-s], 0)→(h[i-s], 1), change: +h[i-s]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _mc(h):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = float(h.sum())
    if s < 1e-15: return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - float(h.sum())
    if abs(d) > 1e-10:
        if d > 0: order = np.argsort(1.0-h)
        else: order = np.argsort(h)
        for idx in order:
            if d > 0:
                a = min(d, 1.0-h[idx]); h[idx]+=a; d-=a
            else:
                a = min(-d, h[idx]); h[idx]-=a; d+=a
            if abs(d) < 1e-10: break
    return h

def _bin(h, n):
    tgt = n // 2
    h = np.clip(h, 0, 1)
    ix = np.argsort(h)
    b = np.zeros(n)
    b[ix[tgt:]] = 1.0
    return b

def _greedy_bin(h, n, max_pass=500):
    h = h.copy()
    mc = _mc(h)
    for p in range(max_pass):
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        best = mc; bi, bj = -1, -1
        for i in ones:
            for j in zeros:
                h[i]=0; h[j]=1
                v = _mc(h)
                h[i]=1; h[j]=0
                if v < best: best=v; bi,bj=i,j
        if bi < 0: break
        h[bi]=0; h[bj]=1; mc=best
    return h, mc

def _sa_bin(h, n, rng, t_end, T0=2.0, Tend=0.01):
    h = h.copy()
    mc = _mc(h); bm = mc; bh = h.copy()
    it = 0; mx = 3000000
    while time.time() < t_end and it < mx:
        it += 1
        f = it/mx
        T = T0*(Tend/T0)**f
        on = np.where(h==1)[0]; zo = np.where(h==0)[0]
        i = on[rng.randint(len(on))]; j = zo[rng.randint(len(zo))]
        h[i]=0; h[j]=1
        nm = _mc(h)
        d = nm - mc
        if d < 0 or rng.rand() < np.exp(-d/max(T,1e-10)):
            mc = nm
            if nm < bm: bm=nm; bh=h.copy()
        else:
            h[i]=1; h[j]=0
    return bh, bm

def _slsqp_obj(x, n):
    corr = np.correlate(x, 1.0-x, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(300.0*(corr-m), -50, 50)))
    return (m + s/300.0) * (2.0/n)

def _slsqp_eq(x, tgt):
    return float(x.sum() - tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Use n=70 from initial_h_values

    best_h = h0.copy()
    best_c5 = _c5(best_h, n)
    
    # Phase 1: Binary local search on n=70
    t1 = t0 + min(budget_s*0.25, 300)
    hb = _bin(h0, n)
    hb, mcb = _greedy_bin(hb, n, max_pass=300)
    c5b = mcb * 2.0/n
    if c5b < best_c5:
        best_c5 = c5b; best_h = hb.copy()
    
    # SA on best binary
    hb2, mcb2 = _sa_bin(hb, n, rng, t1)
    c5b2 = mcb2 * 2.0/n
    if c5b2 < best_c5:
        best_c5 = c5b2; best_h = hb2.copy()
    
    # Phase 2: Try other n values
    if time.time() < t0 + budget_s*0.6:
        for nn in [60, 80, 100]:
            if time.time() > t0 + budget_s*0.6: break
            tgt = nn // 2
            # Multiple random starts
            for attempt in range(5):
                if time.time() > t0 + budget_s*0.6: break
                hr = np.zeros(nn)
                ones_pos = rng.choice(nn, tgt, replace=False)
                hr[ones_pos] = 1.0
                hr, mcr = _greedy_bin(hr, nn, max_pass=200)
                c5r = mcr * 2.0/nn
                if c5r < best_c5:
                    best_c5 = c5r; best_h = hr.copy(); n = nn
                    # SA polish
                    t_sa = t0 + budget_s*0.6
                    hr2, mcr2 = _sa_bin(hr, nn, rng, t_sa)
                    c5r2 = mcr2 * 2.0/nn
                    if c5r2 < best_c5:
                        best_c5 = c5r2; best_h = hr2.copy(); n = nn
    
    # Phase 3: Continuous SLSQP polish
    if time.time() < t0 + budget_s - 15:
        try:
            cons = [{'type':'eq','fun':_slsqp_eq,'args':(n/2.0,)}]
            bnds = [(0.0,1.0)]*n
            res = minimize(_slsqp_obj, best_h, args=(n,), method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter':300,'ftol':1e-15})
            hc = _repair(res.x, n)
            c5c = _c5(hc, n)
            if c5c < best_c5:
                best_c5 = c5c; best_h = hc
        except: pass
    
    # Phase 4: Final binary + continuous
    if time.time() < t0 + budget_s - 5:
        hb_f = _bin(best_h, n)
        hb_f, mcf = _greedy_bin(hb_f, n, max_pass=200)
        c5f = mcf * 2.0/n
        if c5f < best_c5:
            best_c5 = c5f; best_h = hb_f.copy()
    
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
```

```python
def _subgrad_step(h, n, step_size, top_k=3):
    corr = np.correlate(h, 1-h, mode='full')
    # Find top-k
    idx = np.argpartition(corr, -top_k)[-top_k:]
    # Average subgradient over active correlations
    grad = np.zeros(n)
    for k in idx:
        s = k - (n-1)
        for j in range(n):
            g = 0.0
            if 0 <= j+s < n: g += (1.0 - h[j+s])
            if 0 <= j-s < n: g += (-h[j-s])
            grad[j] += g / len(idx)
    # Take step
    h_new = h - step_size * grad
    return _repair(h_new, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _mc(h):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')))

def _corr(h):
    return np.correlate(h, 1.0-h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0, 1)
    tgt = n/2.0
    s = h.sum()
    if s < 1e-15: return np.full(n, 0.5)
    h *= tgt/s
    h = np.clip(h, 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        order = np.argsort(1-h) if d>0 else np.argsort(h)
        for j in order:
            if d>0: a=min(d,1-h[j]); h[j]+=a; d-=a
            else: a=min(-d,h[j]); h[j]-=a; d+=a
            if abs(d)<1e-10: break
    return h

def _subgrad(h, n, top_k=5):
    c = _corr(h)
    m = c.max()
    idx = np.where(c >= m - 1e-6)[0]
    if len(idx) > top_k:
        idx = np.argpartition(c, -top_k)[-top_k:]
    
    grad = np.zeros(n)
    for k in idx:
        s = k - (n-1)
        for j in range(n):
            g = 0.0
            if 0 <= j+s < n: g += 1.0 - h[j+s]
            if 0 <= j-s < n: g += -h[j-s]
            grad[j] += g
    grad /= len(idx)
    return grad

def _lse_obj(x, n, alpha):
    c = _corr(x)
    m = c.max()
    s = np.sum(np.exp(np.clip(alpha*(c-m), -60, 60)))
    return (m + np.log(s)/alpha) * (2.0/n)
```
