    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    if abs(s - target) < 1e-12:
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

    if time.time() < t_start + budget_s - 15:
        for perturb in [0.0, 0.01, -0.01]:
            if time.time() > t_start + budget_s - 10:
                break
            try:
                x0 = best_h.copy()
                if perturb != 0.0:
                    x0 = np.clip(x0 + perturb * rng.randn(n) * 0.02, 0.0, 1.0)
                    x0 = _repair_sum(x0, n)
                res = minimize(_slsqp_obj, x0, args=(n,), method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 300, 'ftol': 1e-16})
                hc = _repair_sum(res.x, n)
                c5 = _c5_eval(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Final micro-refinement
    if time.time() < t_start + budget_s - 2:
        best_h, best_c5 = _cont_refine(best_h, dx, t_start + budget_s - 1)

    best_h = _repair_sum(best_h, n)
    best_c5 = _c5_eval(best_h, dx)
    return (best_h, best_c5, n)
