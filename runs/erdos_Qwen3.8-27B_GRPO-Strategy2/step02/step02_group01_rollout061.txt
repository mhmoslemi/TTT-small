def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:

                    diff += sub

                h[i], h[j] = hi, hj
            T = max(T * 0.9997, 1e-10)

    # Phase 7: Ultra-fine refinement
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5_of(h, dx)
        for it in range(20000):
            if elapsed() > budget_s - 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.002, 0.002)
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5_of(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i], h[j] = hi, hj

    best_h = _repair(best_h, n)
    best_c5 = _c5_of(best_h, dx)
    return (best_h, best_c5, n)
