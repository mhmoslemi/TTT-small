        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) == 0:
                break
            add = diff / len(free)
            h[free] = np.minimum(h[free] + add, 1.0)
        else:
            free = np.where(h > 0.0)[0]
            if len(free) == 0:
                break
            sub = -diff / len(free)
            h[free] = np.maximum(h[free] - sub, 0.0)
    return h


def _lse_fh(x, alpha, mu, target):
    n = len(x)
    corr = _corr_full(x)
    ac = alpha * corr
    ac_shifted = ac - ac.max()
    f_lse = (ac.max() + np.log(np.sum(np.exp(ac_shifted)))) / alpha
    w = np.exp(ac_shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-12:
            continue
        shift = m - (n - 1)
        if shift >= 0:
            hi = n - shift
            if hi > 0:
                idx = np.arange(0, hi)
                grad[idx] += w[m] * alpha * (1.0 - x[idx + shift])
            if shift < n and shift > 0:
                idx2 = np.arange(shift, n)
                grad[idx2] -= w[m] * alpha * x[idx2 - shift]
            elif shift == 0:
                grad -= w[m] * alpha * x
                grad += w[m] * alpha * np.ones(n)
        else:
            s = -shift
            idx = np.arange(s, n)
            grad[idx] += w[m] * alpha * (1.0 - x[idx - s])
            hi2 = n - s
            if hi2 > 0:
                idx2 = np.arange(0, hi2)
                grad[idx2] -= w[m] * alpha * x[idx2 + s]
    viol = x.sum() - target
    f_total = f_lse + mu * viol * viol
    grad_total = grad + 2.0 * mu * viol
    return f_total, grad_total


def _run_lbfgs(h0, n, dx, alphas, mu, t_end):
    best_h = h0.copy()
    best_c5 = _c5_eval(h0, dx)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for al in alphas:
        if time.time() > t_end:
            break
        try:
            def _fo(x):
                return _lse_fh(x, al, mu, target)
            res = minimize(_fo, best_h, jac=True, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12})
            hc = _repair(res.x, n)
            c5 = _c5_eval(hc, dx)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    return best_h, best_c5


def _sa_block(h0, n, dx, t_end, rng):
    h = h0.copy()
    best_c5 = _c5_eval(h, dx)
    best_h = h.copy()
    T0, Tend = 0.003, 0.00002
    max_it = 2000000
    for it in range(max_it):
        if time.time() > t_end:
            break
        if it % 2000 == 0 and (it // 2000) % 10 == 0:
            pass
        frac = min(1.0, it / max_it)
        T = T0 * (Tend / T0) ** frac
        # Block move: pick a block of size 2-5 and perturb
        bsize = rng.randint(2, 6)
        idxs = rng.randint(0, n, size=bsize)
        step = 0.04 * (1.0 - 0.7 * frac)
        old_vals = h[idxs].copy()
        deltas = (rng.rand(bsize) - 0.5) * 2.0 * step
        # Ensure sum constraint: adjust uniformly
        net = deltas.sum()
        rest = np.setdiff1d(np.arange(n), idxs)
        if len(rest) > 0:
            comp = -net / len(rest)
            h[rest] += comp
        h[idxs] = old_vals + deltas
        # Clamp and repair
        h = np.clip(h, 0.0, 1.0)
        h = _repair(h, n)
        new_c5 = _c5_eval(h, dx)
        if new_c5 <= best_c5 + T:
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[idxs] = old_vals
            if len(rest) > 0:
                h[rest] -= comp
            h = np.clip(h, 0.0, 1.0)
            h = _repair(h, n)
    return best_h, best_c5


def _coord_descent(h0, n, dx, t_end, step_sizes):
    h = h0.copy()
    best_c5 = _c5_eval(h, dx)
    target = n / 2.0
    n_passes = 0
    while time.time() < t_end and n_passes < 20:
        n_passes += 1
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in step_sizes:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                # Find best j to compensate
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                found = False
                for j in order[:15]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5_eval(h, dx)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        improved = True
                        found = True
                        break
                    else:
                        h[i] = vi
                        h[j] = nj - need
                if found:
                    vi = h[i]  # update for next delta attempt
                    break
        if not improved:
            break
    return h, best_c5


def _interp_to_n(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return _repair(h_dst, n_dst)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n_parent = len(initial_h_values)
    dx_parent = 2.0 / n_parent
    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), n_parent)
    best_h = h_parent.copy()
    best_c5 = _c5_eval(best_h, dx_parent)
    best_n = n_parent

    # Phase 1: L-BFGS-B on parent n (40% budget)
    p1_end = t_start + budget_s * 0.35
    target_n = n_parent
    bounds = [(0.0, 1.0)] * n_parent
    for al in [200, 500, 1000, 2000, 4000, 8000]:
        if time.time() > p1_end:
            break
        best_h, c5 = _run_lbfgs(best_h, n_parent, dx_parent, [al], 5e4, p1_end)
        if c5 < best_c5:
            best_c5, best_h = c5, best_h

    # Phase 2: SA with block moves (25% budget)
    p2_end = t_start + budget_s * 0.60
    if time.time() < p2_end:
        h_sa, c5_sa = _sa_block(best_h, n_parent, dx_parent, p2_end, rng)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa

    # Phase 3: Coordinate descent (25% budget)
    p3_end = t_start + budget_s * 0.85
    if time.time() < p3_end:
        steps = np.array([0.005, 0.01, 0.02, 0.04, -0.005, -0.01, -0.02, -0.04])
        best_h, c5_cd = _coord_descent(best_h, n_parent, dx_parent, p3_end, steps)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, best_h

    # Phase 4: Try different n values (15% budget)
    p4_end = t_start + budget_s - 5
    if time.time() < p4_end:
        for n_try in [65, 72, 75, 60]:
            if time.time() > p4_end:
                break
            dx_try = 2.0 / n_try
            h_try = _interp_to_n(best_h, n_parent, n_try)
            c5_try = _c5_eval(h_try, dx_try)
            # Quick L-BFGS polish
            al_final = 5000
            for al2 in [1000, al_final]:
                if time.time() > p4_end:
                    break
                try:
                    def _fo2(x):
                        return _lse_fh(x, al2, 5e4, n_try / 2.0)
                    res = minimize(_fo2, h_try, jac=True, method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * n_try,
                                   options={'maxiter': 200, 'ftol': 1e-16})
                    h_try = _repair(res.x, n_try)
                    c5_try = _c5_eval(h_try, dx_try)
                except Exception:
                    pass
            # Quick coord descent
            if time.time() < p4_end - 5:
                steps_fine = np.array([0.003, 0.008, 0.015, -0.003, -0.008, -0.015])
                h_try, c5_try = _coord_descent(h_try, n_try, dx_try, p4_end, steps_fine)
            if c5_try < best_c5:
                best_c5, best_h, best_n = c5_try, h_try, n_try

    # Final polish
    if time.time() < t_start + budget_s - 2:
        dx_fin = 2.0 / best_n
        steps_final = np.array([0.001, 0.002, 0.005, -0.001, -0.002, -0.005])
        best_h, c5_fin = _coord_descent(best_h, best_n, dx_fin, t_start + budget_s - 1, steps_final)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, best_h

    best_h = _repair(best_h, best_n)
    best_c5 = _c5_eval(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
