        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.array(h_var.value).flatten()
        h_new = _repair(h_new, n)
        c5_new = _compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            # Damping
            alpha = 0.5
            h_damped = (1 - alpha) * h + alpha * h_new
            h_damped = _repair(h_damped, n)
            c5_damped = _compute_c5(h_damped, dx)
            if c5_damped < best_c5:
                h = h_damped
                best_c5 = c5_damped
            else:
                break  # Converged
        
        # Check convergence
        if abs(c5_new - best_c5) < 1e-15:
            break
    
    return h, best_c5

The linearized correlation at lag L:
corr_lin[L] = sum over valid (i1, i2) pairs of [(1-h0[i2]) * h[i1] - h0[i1] * h[i2]] + const[L]

where const[L] = sum h0[i1]*h0[i2] over valid pairs.

So corr_lin[L] = c[L] + sum_j A[L,j] * h[j] where A and c are precomputed.

Let me vectorize this. For each L:
- If L >= 0: pairs are (i+L, i) for i=0..n-1-L
  - Contribution to h[i+L]: (1-h0[i])
  - Contribution to h[i]: -h0[i+L]
  - Constant: h0[i+L]*h0[i]
- If L < 0 (let L' = -L > 0): pairs are (i, i-L) for i=L'..n-1, i.e., (i, i+L) for i=L'..n-1
  - Wait, for L < 0: corr = sum_{i=-L}^{n-1} h[i]*(1-h[i+L])
  - Pairs: (i, i+L) for i from -L to n-1
  - Contribution to h[i]: (1-h0[i+L])
  - Contribution to h[i+L]: -h0[i]
  - Constant: h0[i]*h0[i+L]

So the matrix A has rows indexed by L (m = L+n-1, from 0 to 2n-2) and columns by j (0 to n-1).

A[m, j] = sum of contributions of h[j] to corr at lag L=m-(n-1).

For a given j, which lags does it appear in?
- As h[i+L] with i+L=j → i=j-L: valid if 0<=j-L<=n-1-L, i.e., L<=j and j-L<=n-1-L → L<=j and L<=n-1-j. So L <= min(j, n-1-j).
  Also need i=j-L >= 0 → L <= j. And i <= n-1-L → j-L <= n-1-L → j <= n-1. Always true.
  Contribution: (1-h0[j-L]) = (1-h0[i])
  
- As h[i] with i=j: valid if 0<=j<=n-1-L, i.e., L <= n-1-j.
  Contribution: -h0[j+L]

- For L < 0, similar but with roles swapped.

    """Build A and c such that corr_lin[m] = c[m] + A[m,:] @ h for all m."""
    A = np.zeros((2*n-1, n))
    c = np.zeros(2*n-1)

        m = L + (n-1)
        if L >= 0:

                c[m] += h0[i1] * h0[i2]

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        else:

            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:

                else:
                    continue
                break  # Found improvement for this (i,j), move on
        if not improved:
            break
    return h, best_c5

def _build_lin(h0, n):

def _coord_descent(h, n, dx, deadline, max_sweeps=200, fine=False):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    if fine:
        d_vals = [0.002, -0.002, 0.005, -0.005, 0.01, -0.01,

def _slsqp_refine(h, n, dx, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    h = h.copy()
    for alpha in [200, 500, 1500, 4000]:
        if time.time() >= deadline:

        def obj(x, a=alpha):
            corr = _corr_full(x)
            m = np.max(corr)
            s = a * (corr - m)
            s = np.clip(s, -400, 400)
            return m + np.log(np.sum(np.exp(s))) / a
        try:

                           constraints=[eq_con],

            hc = _repair(res.x, n)
