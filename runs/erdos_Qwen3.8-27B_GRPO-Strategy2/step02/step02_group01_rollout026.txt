def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(-h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smooth_max_and_grad(h, alpha):
    c = _corr_full(h)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)

        if wk < 1e-10:

        L = k - (n - 1)

        jmax2 = min(n - 1, n - 1 + L)
        if jmin2 <= jmax2:
            idx2 = np.arange(jmin2, jmax2 + 1)
            grad[idx2] -= wk * h[idx2 - L]
    return f, grad

from scipy.optimize import minimize

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(-h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smooth_max_and_grad(h, alpha):
    c = _corr_full(h)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    f = m + np.log(e.sum()) / alpha
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        L = k - (n - 1)
        jmin = max(0, -L); jmax = min(n - 1, n - 1 - L)
        if jmin <= jmax:
            idx = np.arange(jmin, jmax + 1)
            grad[idx] += wk * (1.0 - h[idx + L])
        jmin2 = max(0, L); jmax2 = min(n - 1, n - 1 + L)
        if jmin2 <= jmax2:
            idx2 = np.arange(jmin2, jmax2 + 1)
            grad[idx2] -= wk * h[idx2 - L]
    return f, grad
