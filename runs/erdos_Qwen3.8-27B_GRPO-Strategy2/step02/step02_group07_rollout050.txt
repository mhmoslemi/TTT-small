import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:

    return h

def _obj_at_t(h, j, k, t, dx):

def _ternary_search(h, j, k, dx, n_iter=60):

        h = _repair(h, n)
        pairs = adj_pairs + [(i, (i + 3) % n) for i in range(n) if (i + 3) % n != i]
        rng.shuffle(pairs)
        _cd_round(h, n, dx, pairs, 48)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 6: Multi-restart with CD
    deadline = t0 + budget_s * 0.97
    for sig in [0.10, 0.18, 0.06, 0.25, 0.12, 0.30]:
        if elapsed() > deadline:
            break
        h_p = _repair(best_h + sig * rng.randn(n), n)
        for _ in range(40):
            if elapsed() > deadline:
                break
            _cd_round(h_p, n, dx, adj_pairs, 25)
        c = _c5(h_p, dx)
        if c < best_c5:
            best_c5, best_h = c, h_p.copy()
            h2, c2 = _try_slsqp(best_h.copy(), n, dx, target, bounds, cons, 300, 300, t0, deadline)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 7: Final fine polish
    if elapsed() < t0 + budget_s - 8:
        h = best_h.copy()
        for _ in range(3):
            _cd_round(h, n, dx, adj_pairs, 55)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
