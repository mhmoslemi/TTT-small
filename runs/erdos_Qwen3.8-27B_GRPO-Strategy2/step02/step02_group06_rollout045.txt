import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p + k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p - k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid, max_passes=40):
    improved = True
    passes = 0
    while improved and passes < max_passes:
        passes += 1
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-12:
                    continue
                gd = _g_vec(h, i, n) - _g_vec(h, j, n)
                ds1 = np.linspace(d_min, d_max, n_coarse)
                nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
                c5s1 = np.max(nc1, axis=1) * dx
                bi1 = np.argmin(c5s1)
                cur_max = float(np.max(corr)) * dx
                if c5s1[bi1] < cur_max - 1e-14:
                    d_best = ds1[bi1]
                    if n_fine > 2 and fine_wid > 0:
                        lo = max(d_min, d_best - fine_wid)
                        hi2 = min(d_max, d_best + fine_wid)
                        ds2 = np.linspace(lo, hi2, n_fine)
                        nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                        c5s2 = np.max(nc2, axis=1) * dx
                        bi2 = np.argmin(c5s2)
                        d_best = ds2[bi2]
                    h[i] = hi + d_best
                    h[j] = hj - d_best
                    corr = corr + d_best * gd
                    improved = True
    return h, corr

def _de_population(pop, n, dx, rng, F, CR, n_gens, deadline):
    P = len(pop)
    pop_c5 = np.array([_c5(pop[i], dx) for i in range(P)])
    for gen in range(n_gens):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        sp = pop[order].copy()
        sc = pop_c5[order].copy()
        for i in range(P):
            a, b, c = rng.choice(P, 3, replace=False)
            v = sp[a] + F * (sp[b] - sp[c])
            j0 = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[j0] = True
            trial = sp[i].copy()
            trial[mask] = v[mask]
            trial = _repair(trial, n)
            tc5 = _c5(trial, dx)
            if tc5 < sc[i]:
                sp[i] = trial
                sc[i] = tc5
        pop = sp
        pop_c5 = sc
    return pop, pop_c5

def _slsqp_refine(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
