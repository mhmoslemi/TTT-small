    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _cd_sweep(h, n):
    ncorr = 2 * n - 1
    d_vals = np.arange(ncorr) - (n - 1)
    corr = _corr_full(h)
    improved = 0.0
    for i in range(n):
        old_hi = h[i]
        ipd = i + d_vals
        imd = i - d_vals
        g = np.zeros(ncorr)
        v1 = (ipd >= 0) & (ipd < n)
        v2 = (imd >= 0) & (imd < n)
        g[v1] = 1.0 - h[ipd[v1]]
        g[v2] -= h[imd[v2]]
        L = -old_hi
        U = 1.0 - old_hi
        if U - L < 1e-14:
            continue
        lo, hi_b = L, U
        for _ in range(55):
            mid = (lo + hi_b) / 2.0
            vals = corr + mid * g
            fval = np.max(vals)
            active = vals >= fval - max(1e-12, abs(fval) * 1e-12)
            if not np.any(active):
                active = np.full(ncorr, True)
            gmin = np.min(g[active])
            gmax = np.max(g[active])
            if gmin > 1e-13:
                hi_b = mid
            elif gmax < -1e-13:
                lo = mid
            else:
                lo = hi_b = mid
                break
        delta = (lo + hi_b) / 2.0
        new_hi = np.clip(old_hi + delta, 0.0, 1.0)
        dd = new_hi - old_hi
        if abs(dd) > 1e-13:
            h[i] = new_hi
            improved += abs(dd)
            corr += dd * g
    return h, improved

def _coord_descent(h, n, max_sweeps=200):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = _c5(h, dx)
    for sweep in range(max_sweeps):
        h, improved = _cd_sweep(h, n)
        h = _repair(h, n)
        curr_c5 = _c5(h, dx)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
        if improved < 1e-11:
            break
    return h, best_c5

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_obj(x, alpha, dx):
    return _smooth_max(_corr_full(x), alpha) * dx

def _con_fun(x, target):
    return np.sum(x) - target

def _slsqp_polish(h, n, dx, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return _repair(h, n)

def _structured_init(n, seed_offset):
    rng2 = np.random.RandomState(1000 + seed_offset)
    t = np.linspace(0, 2.0, n, endpoint=False)
    base = 0.5 + 0.3 * np.sin(2 * np.pi * t * (2 + seed_offset % 3))
    base += 0.1 * rng2.randn(n)
    return _repair(base, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    starts = [h_init]
    for k in range(2):
        starts.append(_binary_init(n, np.random.RandomState(seed + 7 * (k + 1))))
        starts.append(_structured_init(n, k + 1))
    best_h = h_init.copy()
    best_c5 = _c5(best_h, dx)

    def elig(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: CD from each start
    for idx in range(len(starts)):
        if not elig(300):
            break
        hc = starts[idx].copy()
        hc, c5 = _coord_descent(hc, n, max_sweeps=250)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 2: SLSQP + CD cycles with increasing alpha
    for a in [60, 150, 400, 1200]:
        if not elig(180):
            break
        hc = _slsqp_polish(best_h, n, dx, a, 350)
        hc, c5 = _coord_descent(hc, n, max_sweeps=120)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()

    # Phase 3: Random perturbation + CD (plus occasional SLSQP)
    it = 0
    sigmas = [0.1, 0.2, 0.3, 0.15, 0.25, 0.4, 0.08, 0.35, 0.2, 0.5, 0.12, 0.18]
    while elig(60):
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc, c5 = _coord_descent(hp, n, max_sweeps=130)
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
        elif c5 < best_c5 + 0.004 and elig(30):
            hc2 = _slsqp_polish(hc, n, dx, 600, 200)
            hc2, c52 = _coord_descent(hc2, n, max_sweeps=70)
            if c52 < best_c5:
                best_c5, best_h = c52, hc2.copy()
        it += 1

    # Phase 4: Final SLSQP + CD polish
    if elig(45):
        for a in [800, 2500]:
            if not elig(20):
                break
            hc = _slsqp_polish(best_h, n, dx, a, 500)
            hc, c5 = _coord_descent(hc, n, max_sweeps=120)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()

    # Phase 5: Final exhaustive CD
    if time.time() < t0 + budget_s - 3:
        best_h, best_c5 = _coord_descent(best_h, n, max_sweeps=400)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
