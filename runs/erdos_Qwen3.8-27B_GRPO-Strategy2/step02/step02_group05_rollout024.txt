                ni, nj = best[i] + d, best[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    oi, oj = best[i], best[j]
                    best[i], best[j] = ni, nj
                    nc = _c5(best, dx)
                    if nc < best_c5 - 1e-12:
                        best_c5 = nc
                        improved = True
                    else:
                        best[i], best[j] = oi, oj
        if not improved:
            stall += 1
            if stall > 40:
                break
        else:
            stall = 0
    return best, best_c5

def _block_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    for it in range(8000):
        if time.time() > deadline:
            break
        blen = int(rng.randint(2, 5))
        start = int(rng.randint(max(1, n - blen)))
        end = start + blen
        if end > n:
            end = n
        start_b = max(0, start - blen)
        if end - start >= 2:
            adj_start = int(rng.randint(start_b, start))
            adj_cnt = end - start
        else:
            adj_start = 0
            adj_cnt = 1
        delta = float((rng.rand() - 0.5) * 0.15)
        block_slice = slice(start, end)
        adj_slice = slice(adj_start, adj_start + adj_cnt)
        trial = best.copy()
        trial[block_slice] += delta
        total = trial[adj_slice].sum() * (end - start) / max(1, adj_cnt)
        adj_scale = (end - start) / max(1, adj_cnt)
        trial[adj_slice] -= delta * (end - start) / max(1, adj_cnt)
        if np.any(trial < -1e-9) or np.any(trial > 1.0 + 1e-9):
            continue
        nc = _c5(trial, dx)
        if nc < best_c5 - 1e-12:
            best_c5 = nc
            best = trial
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    def elapsed():
        return time.time() - t0

    def time_left(margin):
        return elapsed() < budget_s - margin

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)

    # Phase 1: SLSQP with alpha schedule
    alphas = [30, 60, 120, 250, 500, 1000, 2000]
    hc = h_init.copy()
    for a in alphas:
        if not time_left(500):
            break
        hc = _slsqp(hc, n, dx, a, 400, 1e-14)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5 = nc
            best_h = hc.copy()

    # Phase 2: Directed local search (focus on worst lags)
    if time_left(400):
        dl = t0 + budget_s - 400
        best_h, best_c5 = _directed_search(best_h, n, dx, rng, dl)

    # Phase 3: SLSQP restarts with perturbations
    sigmas = [0.15, 0.3, 0.5, 0.2, 0.4, 0.6, 0.1, 0.25, 0.35, 0.45, 0.7, 0.12]
    it = 0
    while it < len(sigmas) * 3 and time_left(250):
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 200, 250, 1e-13)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5 = nc
            best_h = hc.copy()
            if time_left(200):
                dl = t0 + budget_s - 200
                best_h, best_c5 = _directed_search(best_h, n, dx, rng, dl)
        it += 1

    # Phase 4: Simulated annealing
    if time_left(150):
        dl = t0 + budget_s - 150
        best_h, best_c5 = _sa_local_search(best_h, n, dx, np.random.RandomState(seed + 99), dl)

    # Phase 5: Block search
    if time_left(100):
        dl = t0 + budget_s - 100
        best_h, best_c5 = _block_search(best_h, n, dx, np.random.RandomState(seed + 200), dl)

    # Phase 6: Final SLSQP polish
    if time_left(30):
        for a in [1500, 3000, 5000]:
            if not time_left(15):
                break
            hc = _slsqp(best_h, n, dx, a, 500, 1e-15)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5 = nc
                best_h = hc.copy()

    # Phase 7: Final directed polish
    if time_left(10):
        best_h, best_c5 = _directed_search(best_h, n, dx, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
