def _binary_round(h, n):
    """Round h to binary maintaining sum=n/2."""
    h = h.copy()
    ones = np.sum(h >= 0.5)
    target = n // 2
    if ones > target:
        # Need to set some 0.5+ values to 0
        idx = np.where(h >= 0.5)[0]
        order = idx[np.argsort(h[idx])]  # smallest first
        for i in order[:ones - target]:
            h[i] = 0.0
    elif ones < target:
        idx = np.where(h < 0.5)[0]
        order = idx[np.argsort(-h[idx])]  # largest first
        for i in order[:target - ones]:
            h[i] = 1.0
    else:
        h[h >= 0.5] = 1.0
        h[h < 0.5] = 0.0
    # Force exact binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Verify sum
    while h.sum() != target:
        if h.sum() > target:
            i = np.where(h == 1.0)[0]
            h[i[0]] = 0.0
        else:
            i = np.where(h == 0.0)[0]
            h[i[0]] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _make_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def _binary_swap_n(n, seed, target_max, max_rounds, budget_end, rng):
    """Binary swap search trying to get max(corr) <= target_max."""
    h = _make_binary(n, seed)
    corr = _corr(h)
    max_c = np.max(corr)
    best_h = h.copy()
    best_max = max_c
    n_corr = 2 * n - 1
    P = np.arange(n_corr) - n + 1  # shifts: -(n-1) to (n-1)
    
    for rnd in range(max_rounds):
        if time.time() > budget_end:
            break
        if max_c <= target_max:
            break

        # delta[i, k] = effect of flipping h[i] (1->0 or 0->1)
        # For 1->0: delta[i,k] = -(1-h[i-P[k]]) + h[i+P[k]] (with 0 for OOB)
        # For 0->1: it's the negative

        for i in range(n):

            mask1 = (idx1 >= 0) & (idx1 < n)

def _slsqp_optimize(h, n, alpha, maxiter, budget_end):
    global _g_alpha, _g_target
    _g_alpha = alpha
    _g_target = n / 2.0
    
    def _objective(x):
        corr = _corr(x)
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def _constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint}]
    try:
        if time.time() > budget_end:
            return h
        res = minimize(_objective, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad(h, n, steps, step_size, budget_end):
    """Subgradient descent on active lags."""
    h = h.copy()
    n_corr = 2 * n - 1
    P = np.arange(n_corr) - n + 1
    
    for step in range(steps):
        if time.time() > budget_end:
            break
        corr = _corr(h)
        m = np.max(corr)
        # Active lags (within 0.5 of max)
        active = np.where(corr >= m - 0.5)[0]
        
        # Compute gradient for active lags
        grad = np.zeros(n)
        for k in active[:10]:  # Limit to top 10 active lags
            p = P[k]
            for j in range(n):
                g = 0.0
                idx1 = j - p
                if 0 <= idx1 < n:
                    g += (1.0 - h[idx1])
                idx2 = j + p
                if 0 <= idx2 < n:
                    g -= h[idx2]
                grad[j] += g
        
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        h = h - step_size * grad / gn
        h = _repair(h, n)
    
    return h
