    return np.correlate(h, 1.0 - h, mode='full')

    return np.max(_corr(h)) * dx

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to achieve exact sum
    for _ in range(50):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-14:
            break
        # Find values not at bounds
        if diff > 0:
            # Need to decrease values not at 0
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # Need to increase values not at 1
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

        else:

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-14:
        mask = (h > 1e-14) if diff > 0 else (h < 1.0 - 1e-14)
        h[mask] -= diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def _grad_lag(h, n, out_idx):
    L = out_idx - (n - 1)
    l = abs(L)
    g = np.zeros(n)
    if l == 0:
        return 1.0 - 2.0 * h
    g[l:] += (1.0 - h[:n - l])
    g[:n - l] -= h[l:]
    return g


def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _init_h(n, rng, seed=0):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * 1.5 * x + rng.rand() * 6.28)
    h += 0.1 * rng.randn(n)
    return _project(h, n)


def _optimize_one(n, h0, seed, t_budget, t0):
    rng = np.random.RandomState(seed + n)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    best_h = _project(h0.copy(), n)
    best_c5 = _c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP
    for alpha in [50, 200, 1000, 5000]:
        if elapsed() > t0 + t_budget * 0.35:
            break
        for rst in range(2):
            if elapsed() > t0 + t_budget * 0.35:
                break
            x0 = best_h + (0.003 if rst else 0.0) * rng.randn(n)
            x0 = _project(x0, n)
            try:
                res = minimize(lambda x, a=alpha: _smooth_obj(x, a), x0,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-18})
                hc = _project(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 2: Analytical projected gradient (top-2 lags)
    if elapsed() < t0 + t_budget * 0.65:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        step = 0.01
        for it in range(3000):
            if elapsed() > t0 + t_budget * 0.65:
                break
            c = _corr(h)
            top2 = np.argsort(c)[::-1][:2]
            w = np.exp(20 * (c[top2] - c[top2][0]))
            w /= w.sum()
            g = w[0] * _grad_lag(h, n, top2[0])
            if len(top2) > 1:
                g += w[1] * _grad_lag(h, n, top2[1])
            step = max(step, 0.0005)
            improved = False

    # Phase 3: Pair local search (coarse to fine)

                            else:

                    h[i] = hi
                if not any_imp:

            if c5_cur < best_c5:

            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur or (T > 1e-12 and rng.rand() < np.exp(-(c - c5_cur) / T)):
                c5_cur = c
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                h[i], h[j] = hi, hj
            T = max(T * 0.999, 1e-14)

    return best_h, best_c5, n
