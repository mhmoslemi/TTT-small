def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, alpha):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.clip(alpha * (corr - m), -500, 500)
    exp_s = np.exp(s)
    Z = np.sum(exp_s)
    weights = exp_s / Z
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        w = weights[ki]
        if w < 1e-7:
            continue
        L = ki - (n - 1)
        # dcorr/dh[i] = (1-h[i-L])_{0<=i-L<n} - h[i+L]_{0<=i+L<n}
        if L >= 0:
            # i-L in [0,n-1] → i in [L, n-1]
            # i+L in [0,n-1] → i in [0, n-1-L]
            grad[L:n] += w * (1.0 - x[0:n-L])
            grad[0:n-L] -= w * x[L:n]
        else:
            L2 = -L
            # i-L = i+L2 in [0,n-1] → i in [0, n-1-L2]
            # i+L = i-L2 in [0,n-1] → i in [L2, n-1]
            grad[0:n-L2] += w * (1.0 - x[L2:n])
            grad[L2:n] -= w * x[0:n-L2]
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, dx, bounds, eq, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    try:
        res = minimize(
            _smooth_obj, h, args=(alpha,),
            jac=lambda x, a=alpha: _smooth_grad(x, a),
            method='SLSQP', bounds=bounds, constraints=[eq],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _pair_cd(h, n, dx, rng, deadline, max_sweeps=5000):

    h = h.copy()
    n_pairs_per_sweep = min(500, n * (n - 1) // 2)
    
    for sweep in range(max_sweeps):

            n_tried += 1

def _smooth_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()  # softmax weights
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-8:
            continue
        L = ki - (n - 1)
        if L >= 0:
            g[L:n] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            L2 = -L
            g[:n-L2] += wk * (1.0 - x[L2:])
            g[L2:] -= wk * x[:n-L2]
    return g

def _repair(h, n):

    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _lse(s):
    m = s.max()
    return m + np.log(np.sum(np.exp(s - m)))

def _sobj(x, alpha):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500, 500)
    return m + _lse(s) / alpha

def _sgrad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-8:
            continue
        L = ki - (n - 1)
        if L >= 0:
            g[L:n] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            L2 = -L
            g[:n - L2] += wk * (1.0 - x[L2:])
            g[L2:] -= wk * x[:n - L2]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _slsqp(h, n, bounds, eq, alpha, iters, dl):
    if time.time() >= dl:
        return h
    try:
        r = minimize(_sobj, h, args=(alpha,), jac=lambda x, a=alpha: _sgrad(x, a),
                     method='SLSQP', bounds=bounds, constraints=[eq],
                     options={'maxiter': iters, 'ftol': 1e-14})
        return _repair(r.x, n)
    except Exception:
        return h

def _gp_cd(h, n, dx, rng, dl):
    h = h.copy()
    best = _c5(h, dx)
    alpha = 500.0
    stall_count = 0
    step_scale = 1.0
    it = 0
    while time.time() < dl:
        it += 1
        alpha = max(50.0, 500.0 * (0.999 ** it))
        g = _sgrad(h, alpha)
        order = np.argsort(g)
        improved = False
        half = max(1, n // 2)
        for _ in range(300):
            if time.time() >= dl:
                break
            i = int(order[rng.randint(half)])
            j = int(order[n - 1 - rng.randint(half)])
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-10:
                continue
            d0 = 0.03 * step_scale
            if g[i] >= g[j]:
                d0 = -d0
            d0 = float(np.clip(d0, lo, hi))
            cands = np.array([d0, 0.5 * d0, 2.0 * d0, 0.5 * (lo + hi), 0.1 * (hi - lo) * (1 if d0 >= 0 else -1)])
            cands = cands[(cands >= lo) & (cands <= hi) & (np.abs(cands) > 1e-11)]
            bd, bv = 0.0, best
            for d in cands:
                h[i] += d; h[j] -= d
                v = _c5(h, dx)
                h[i] -= d; h[j] += d
                if v < bv - 1e-13:
                    bv, bd = v, d

                    best = bv
