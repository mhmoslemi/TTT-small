def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slope_max(c, alpha):
    """Smooth max with gradient"""
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    sse = np.sum(e)
    val = m + np.log(sse) / alpha
    grad = e / sse  # softmax weights
    return val, grad

def _obj_smax(x, alpha):
    c = _corr(x)
    return _smooth_max(c, alpha)

    return np.sum(x) - target

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    c = _corr(x)
    return _smooth_max(c, alpha)

def _obj_topk(x, peaks, alpha):
    c = _corr(x)
    return _smooth_max(c[peaks], alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _pair_sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    best_curr = float(np.max(corr) * dx)
    for i in range(n):
        for j in range(i + 1, n):

            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-13:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s = np.max(nc, axis=1) * dx
            bi = int(np.argmin(c5s))
            if c5s[bi] < best_curr - 1e-14:
                d_best = ds1[bi]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = int(np.argmin(c5s2))
                    d_best = ds2[bi2]
                    c5s[bi] = c5s2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                best_curr = c5s[bi]
                improved = True
    return h, corr, improved

    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

    return _slsqp_opt(h, n, target, bounds, obj, maxiter)
