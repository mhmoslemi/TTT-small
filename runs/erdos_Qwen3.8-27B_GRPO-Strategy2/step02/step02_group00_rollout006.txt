```python
            idx2 = np.arange(0, n - lag)  # j where j+lag is valid
            grad[idx2] += w * (-h[idx2 + lag])
        else:
            l = -lag
            # corr[k] = sum_{i=l}^{n-1} h[i-l]*(1-h[i])
            # d/dh[j]: (1-h[j+l]) if 0<=j<n-l, minus h[j-l] if l<=j<n
            idx1 = np.arange(0, n - l)  # j where j+l is valid (< n)
            grad[idx1] += w * (1.0 - h[idx1 + l])
            idx2 = np.arange(l, n)  # j where j-l is valid (>= 0)
            grad[idx2] += w * (-h[idx2 - l])
    grad /= alpha
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _grad_smoothed(h, alpha, n):
    corr = _corr(h)
    m = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m), -700, 700)
    exp_vals = np.exp(shifted)
    weights = exp_vals / float(np.sum(exp_vals))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - n + 1
        if lag >= 0:
            idx1 = np.arange(lag, n)
            if len(idx1) > 0:
                grad[idx1] += w * (1.0 - h[idx1 - lag])
            idx2 = np.arange(0, n - lag)
            if len(idx2) > 0:
                grad[idx2] += w * (-h[idx2 + lag])
        else:
            l = -lag
            idx1 = np.arange(0, n - l)
            if len(idx1) > 0:
                grad[idx1] += w * (1.0 - h[idx1 + l])
            idx2 = np.arange(l, n)
            if len(idx2) > 0:
                grad[idx2] += w * (-h[idx2 - l])
    grad /= alpha
    return grad


def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(-h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return h


def _obj_lse(x, alpha):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.sum(np.exp(np.clip(alpha * (corr - m), -700, 700)))
    return (m + np.log(s) / alpha) * (2.0 / len(x))


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _coord_search(h, n, dx, deadline, steps_per_round=6):
    local_best = _c5(h, dx)
    deltas = np.array([0.001, 0.003, 0.006, 0.012, 0.025, 0.05])
    if steps_per_round >= 5:
        deltas = np.concatenate([deltas, -deltas])
    else:
        deltas = np.array([0.001, 0.004, -0.001, -0.004])
    improved = True
    rounds = 0
    max_rounds = 30
    while improved and rounds < max_rounds and time.time() < deadline:
        improved = False
        rounds += 1
        for i in range(n):
            if time.time() > deadline:
                break
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                for j in order[:12]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-14:
                        local_best = c5_try
                        improved = True
                        break
                    else:
                        h[i] = vi
                        h[j] = nj - need
                if improved:
                    break
    return h, local_best

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    # Phase 1: Projected gradient with momentum, increasing alpha
    p1_end = t_start + min(budget_s * 0.45, 450)
    alphas = [40, 80, 150, 300, 600, 1200, 2500, 5000]

    for ai, alpha in enumerate(alphas):

        phase_budget = (p1_end - t_start) / (n_alpha - ai)
        alpha_deadline = min(p1_end, time.time() + phase_budget)

        h = best_h.copy()
        v = np.zeros(n)

                else:
                    order = np.argsort(-h)

                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:

                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-14:
                        local_best = c5_try
                        break
                    else:
                        h[i] = vi
                        h[j] = nj - need
        if local_best < best_c5:

    # Phase 5: Final SLSQP polish
    if time.time() < t_start + budget_s - 3:

        except Exception:
            pass

    best_h = _project(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
