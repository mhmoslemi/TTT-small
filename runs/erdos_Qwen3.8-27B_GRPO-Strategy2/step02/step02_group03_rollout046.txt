def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _soft_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):

import numpy as np
from scipy.optimize import minimize
import time

_GA = 500.0
_GT = 35.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _obj_smooth(x):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(_GA * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _GA

def _obj_flat(x):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(_GA * (c - m), -500, 500)
    base = m + np.log(np.sum(np.exp(s))) / _GA
    sc = np.sort(c)[::-1][:5]
    return base + 0.04 * np.var(sc)

def _con(x):
    return np.sum(x) - _GT

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def _slsqp(h, n, alpha, maxiter=300, flat=False):
    global _GA, _GT
    _GA, _GT = alpha, n / 2.0
    bounds = [(0.0, 1.0)] * n

    obj = _obj_flat if flat else _obj_smooth
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_sweep(h, n, dx, n_passes, deadline, rng):
    h = h.copy()
    best = _c5(h, dx)
    for p in range(n_passes):
        if time.time() > deadline - 15:
            break
        improved = False
        step = max(0.003, 0.025 * (0.93 ** p))
        for i in range(n):
            if time.time() > deadline - 15:
                break
            hi = h[i]
            for d in [step, -step, 0.5*step, -0.5*step]:
                ni = hi + d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                jcs = [(i+1)%n, (i+n-1)%n, (i+n//2)%n, (i+7)%n, (i+n-7)%n, rng.randint(n)]
                for j in jcs:
                    if j == i:
                        continue
                    nj = h[j] - d
                    if nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-15:
                        best = c
                        improved = True
                    else:
                        h[i], h[j] = hi, h[j] + d
        if not improved and p > 35:
            break
    return h, best
