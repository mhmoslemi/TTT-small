def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-8})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _subgrad_loop(h, n, t_start, t_limit, n_iters=800):
    dx = 2.0 / n
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(n_iters):
        if time.time() - t_start > t_limit:
            break
        m = c.max()
        top_idx = np.argsort(c)[-3:]
        weights = np.maximum(c[top_idx] - m + 0.005, 0.0) + 0.15
        grad = np.zeros(n)
        for wi, kk in enumerate(top_idx):
            kk = int(kk)
            for j in range(n):
                g = 0.0
                idx2 = j + kk - n + 1
                if 0 <= idx2 < n:
                    g += (1.0 - h[idx2])
                idx1 = j - kk + n - 1
                if 0 <= idx1 < n:
                    g -= h[idx1]
                grad[j] += weights[wi] * g
        lr = 0.0015 * (4.0 / (it + 5.0))
        h_new = h - lr * grad
        h_new = _repair(h_new, n)
        c = np.correlate(h_new, 1.0 - h_new, mode='full')
        m_new = c.max()
        if m_new * dx < best_c5 - 1e-14:
            best_c5 = m_new * dx
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _cd_pass(h, n, n_grid):
    dx = 2.0 / n
    improved = False
    hi_c5 = _c5(h, dx)
    for i in range(n):
        for j in range(i + 1, n):
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            orig_i = h[i]
            orig_j = h[j]
            best_t = orig_i
            best_c = hi_c5
            for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                h[i] = t
                h[j] = s - t
                cv = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if cv < best_c:
                    best_c = cv
                    best_t = t
            if best_c < hi_c5 - 1e-13:
                h[i] = best_t
                h[j] = s - best_t
                hi_c5 = best_c
                improved = True
            else:
                h[i] = orig_i
                h[j] = orig_j
    return h, hi_c5, improved

def _cd_loop(h, n, t_start, t_limit, n_grid, max_passes=2000):
    dx = 2.0 / n
    for p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        h, c5, imp = _cd_pass(h, n, n_grid)
        if not imp:
            break
    return h, _c5(h, dx)

def _optimize_block(h0, n, t_start, t_end):
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    t_end = min(t_end, t_start + 400)
    for a in [40, 80, 160, 320, 600, 1000]:
        if time.time() - t_start > t_end - 60:
            break
        h = _slsqp(h, n, a, 500, 1e-14)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
    if time.time() - t_start < t_end - 80:
        h, c = _subgrad_loop(best_h, n, t_start, t_end - 100, 400)
        if c < best_c5:
            best_c5, best_h = c, h
    if time.time() - t_start < t_end - 50:
        h, c = _cd_loop(best_h, n, t_start, t_end - 60, 19, 300)
        if c < best_c5:
            best_c5, best_h = c, h
    if time.time() - t_start < t_end - 10:
        h = _slsqp(best_h, n, 1500, 600, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h
    if time.time() - t_start < t_end - 5:
        h, c = _cd_loop(best_h, n, t_start, t_end, 27, 500)
        if c < best_c5:
            best_c5, best_h = c, h
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 70

    # Main: n=70 from initial
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    h, c = _optimize_block(h, n, t0, t0 + 500)
    if c < best_c5:
        best_c5, best_h, best_n = c, h.copy(), n

    # Multi-restart n=70
    for sig in [0.1, 0.15, 0.06, 0.2]:
        if time.time() - t0 > budget_s - 250:
            break
        h_p = _repair(best_h + sig * rng.randn(n), n)
        h2, c2 = _optimize_block(h_p, n, t0, time.time() - t0 + 80)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), n

    # Try n=50
    if time.time() - t0 < budget_s - 120:
        n2 = 50
        h2 = _repair(0.5 + 0.3 * rng.randn(n2), n2)
        h2, c2 = _optimize_block(h2, n2, t0, time.time() - t0 + 100)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), n2

    # Try n=100
    if time.time() - t0 < budget_s - 100:
        n3 = 100
        h3 = _repair(0.5 + 0.25 * rng.randn(n3), n3)
        h3, c3 = _optimize_block(h3, n3, t0, time.time() - t0 + 80)
        if c3 < best_c5:
            best_c5, best_h, best_n = c3, h3.copy(), n3

    # Final polish on best
    if time.time() - t0 < budget_s - 10:
        h_f, c_f = _cd_loop(best_h, best_n, t0, budget_s - 5, 31, 800)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        h_f2 = _slsqp(best_h, best_n, 2500, 400, 1e-15)
        c_f2 = _c5(h_f2, 2.0 / best_n)
        if c_f2 < best_c5:
            best_c5, best_h = c_f2, h_f2

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
