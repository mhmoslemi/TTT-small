import numpy as np
from scipy.optimize import minimize
import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_arr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lap(h_half, n):
    half = n // 2
    h = np.empty(n)
    h[:half] = h_half
    h[half:] = 1.0 - h_half
    return h

def _ap_c5(h_half, n):
    h = _lap(h_half, n)
    return _compute_c5(h, 2.0 / n)

    return _ap_c5(x, n)

def _lse_val(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _full_obj(x, alpha):
    return _lse_val(x, alpha)

def _eq_con(x, target):
    return np.sum(x) - target

        return hc, _compute_c5(hc, dx)
    except Exception:
        return h0, _compute_c5(h0, dx)
