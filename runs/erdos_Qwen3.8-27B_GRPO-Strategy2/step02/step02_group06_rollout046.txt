def project_onto_simplex_mod(h, target):
    """Project h onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(h)
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Exact fix
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1 - h_proj)
        else:
            order = np.argsort(h_proj)
        for idx in order:
            if diff > 0:
                add = min(diff, 1 - h_proj[idx])
                h_proj[idx] += add
                diff -= add
            else:
                sub = min(-diff, h_proj[idx])
                h_proj[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h_proj

    """Compute subgradient for the max correlation lag"""
    corr = _corr(h)
    m_star = int(np.argmax(corr))
    # m_star corresponds to shift k = m_star - (n-1)
    k = m_star - (n - 1)

        if 0 <= j1 < n:
            g[i] += (1.0 - h[j1])
        j2 = i - k
        if 0 <= j2 < n:
            g[i] -= h[j2]
    return g, m_star

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0, 1)
    diff = target - h_p.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h_p)
        else:
            order = np.argsort(h_p)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h_p[idx])
                h_p[idx] += add
                diff -= add
            else:
                sub = min(-diff, h_p[idx])
                h_p[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h_p

def _multilag_subgrad(h, n, thresh_frac):
    corr = _corr(h)
    cmax = float(corr.max())
    active = corr > thresh_frac * cmax
    n_act = max(1, int(active.sum()))
    g = np.zeros(n)
    for idx in np.where(active)[0]:
        k = idx - (n - 1)
        j1 = np.arange(n) + k
        m1 = (j1 >= 0) & (j1 < n)
        g[m1] += (1.0 - h[j1[m1]]) / n_act
        j2 = np.arange(n) - k
        m2 = (j2 >= 0) & (j2 < n)
        g[m2] -= h[j2[m2]] / n_act
    return g, cmax

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min > d_max - 1e-12:
                continue
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds1 = np.linspace(d_min, d_max, n_coarse)
            nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
            c5s1 = np.max(nc1, axis=1) * dx
            bi1 = int(np.argmin(c5s1))
            cur_max = float(np.max(corr)) * dx
            if c5s1[bi1] < cur_max - 1e-14:
                d_best = ds1[bi1]
                if n_fine > 2 and fine_wid > 0:
                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    c5s2 = np.max(nc2, axis=1) * dx
                    bi2 = int(np.argmin(c5s2))
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _sgd_phase(h, n, dx, target, n_iter, beta, lr0):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _c5(h, dx)
    lr = lr0
    for it in range(n_iter):
        g, _ = _multilag_subgrad(h, n, 0.92)
        v = beta * v + (1.0 - beta) * g
        h_new = _project(h - lr * v, target)
        h_new = _repair(h_new, n)
        c = _c5(h_new, dx)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        lr *= 0.997
    return best_h, best_c

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _binary_push(h, n, dx):
    h = h.copy()
    corr = _corr(h)
    improved = any(True for _ in range(1))
    while improved:
        improved = False
        for i in range(n):
            for target_val in [0.0, 1.0]:
                d = target_val - h[i]
                if abs(d) < 1e-8:
                    continue
                if target_val == 0.0 and h[i] < 0.35:
                    pass
                elif target_val == 1.0 and h[i] > 0.65:
                    pass
                else:
                    continue
                h2 = h.copy()
                h2[i] = target_val
                h2 = _repair(h2, n)
                c2 = _c5(h2, dx)
                if c2 < float(np.max(corr)) * dx - 1e-12:
                    h = h2
                    corr = _corr(h)
                    improved = True
                    break
    return h
