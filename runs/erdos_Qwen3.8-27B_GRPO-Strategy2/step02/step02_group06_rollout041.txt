import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse=25, n_bisect=35):
    """One full round of coordinate-pair sweeps with bisection."""
    improved = False
    order = np.random.permutation(n)
    n_pairs = 0
    n_improved = 0
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            n_pairs += 1
            if time.time() > time_limit:
                return h, corr, improved, n_pairs, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            range_d = d_max - d_min
            if range_d < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_pairs, n_improved

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h
