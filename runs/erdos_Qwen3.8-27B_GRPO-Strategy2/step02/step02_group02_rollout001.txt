    hi = float(y.max()) + 1.0

            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    # Phase 3: Focused CD on high-overlap lags
    if _rem(t0, B) > 40:
        h, c = _cd_focused(bh, dx, n, 200, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 4: FISTA polish on current best
    if _rem(t0, B) > 50:
        h, c = _fista(bh, 5000, tgt, 3000)
        if c < bc:
            bc = c
            bh = h
        h, c = _cd_rr(bh, dx, n, 200, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 5: Structured restarts
    inits = []
    # Smooth bump
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            xs = np.linspace(0, 2, n)
            xb = np.exp(-0.5 * ((xs - 2 * center) / width) ** 2)
            inits.append(xb)
    # Two bumps
    for c1 in [0.2, 0.4, 0.6]:
        for c2 in [0.6, 0.7, 0.85]:
            if abs(c1 - c2) < 0.15:
                continue
            for w in [0.2, 0.3]:
                xs = np.linspace(0, 2, n)
                xb = 0.5 * (np.exp(-0.5 * ((xs - 2 * c1) / w) ** 2) + np.exp(-0.5 * ((xs - 2 * c2) / w) ** 2))
                inits.append(xb)
    # Triangular wave
    for freq in [2, 3, 4, 5]:
        xs = np.linspace(0, 2, n)
        xt = 1.0 - np.abs(2 * (freq * xs / 2 % 1.0) - 1.0)
        inits.append(xt)

    for ini in inits:
        if _rem(t0, B) < 45:
            break
        hp = _proj(np.array(ini, dtype=float), tgt)
        h, c = _fista(hp, 300, tgt, 3000)
        if c < bc:
            bc = c
            bh = h
            if _rem(t0, B) > 30:
                h2, c2 = _cd_rr(bh, dx, n, 150, rng)
                if c2 < bc:
                    bc = c2
                    bh = h2

    # Phase 6: Random perturbations (small)
    for s in [0.05, 0.08, 0.12, 0.06, 0.1, 0.15, 0.07, 0.1]:
        if _rem(t0, B) < 40:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        h, c = _fista(hp, 800, tgt, 2500)
        if c < bc:
            bc = c
            bh = h
            h2, c2 = _cd_rr(bh, dx, n, 100, rng)
            if c2 < bc:
                bc = c2
                bh = h2

    # Phase 7: SLSQP with smooth surrogate
    bnds = [(0.0, 1.0)] * n
    for a_p in [3000, 8000]:
        if _rem(t0, B) < 30:
            break
        try:
            r = minimize(_sobj, bh, args=(a_p,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': lambda x, t=tgt: float(x.sum()) - t}],
                         options={'maxiter': 400, 'ftol': 1e-14})
            hf = _proj(np.array(r.x), tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    # Phase 8: Final intensive CD
    if _rem(t0, B) > 30:
        h, c = _cd_rr(bh, dx, n, 500, rng)
        if c < bc:
            bc = c
            bh = h
    if _rem(t0, B) > 20:
        h, c = _cd_focused(bh, dx, n, 300, rng)
        if c < bc:
            bc = c
            bh = h

    # Phase 9: Final micro CD
    if _rem(t0, B) > 10:
        h = bh.copy()
        for _ in range(200):
            imp = False
            for i in range(n):
                if h[i] < 0.01 or h[i] > 0.99:
                    continue
                j = (i + 1 + rng.randint(n - 1)) % n
                if h[j] < 0.01 or h[j] > 0.99:
                    continue
                hi0, hj0 = h[i], h[j]
                for sgn in [1, -1]:
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        t = sgn * frac * min(hi0, hj0, 1 - hi0, 1 - hj0)
                        if abs(t) < 1e-8:
                            continue
                        h[i], h[j] = hi0 + t, hj0 - t
                        c = _c5(h, dx)
                        if c < bc - 1e-13:
                            bc = c
                            imp = True
                            break
                        h[i], h[j] = hi0, hj0
                    if imp:
                        break
            if not imp:
                break
        bh = _proj(h, tgt)
        bc = _c5(bh, dx)

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
