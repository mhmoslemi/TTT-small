                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_obj(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    target = n / 2.0
    try:
        res = minimize(_smooth_obj, h, args=(alpha, dx), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _coord_descent(h, n, dx, n_passes, deadline, grid_n=20):
    best = h.copy()
    best_c5 = _c5(best, dx)
    coarse_grid = np.linspace(0, 1, grid_n)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = best[i]
            best_nc = best_c5
            best_v = old
            for c in coarse_grid:
                if abs(c - old) < 1e-5:
                    continue
                best[i] = c
                best = _repair(best, n)
                nc = _c5(best, dx)
                if nc < best_nc - 1e-13:
                    best_nc = nc
                    best_v = c
            if best_v != old:
                best[i] = best_v
                best = _repair(best, n)
                best_c5 = best_nc
                improved = True
            else:
                best[i] = old
                best = _repair(best, n)
    if improved:
        for i in range(n):
            if time.time() > deadline:
                break
            center = best[i]
            lo = max(0.0, center - 0.04)
            hi = min(1.0, center + 0.04)
            fine_grid = np.linspace(lo, hi, 25)
            best_nc = best_c5
            best_v = center
            for c in fine_grid:
                if abs(c - center) < 1e-7:
                    continue
                best[i] = c
                best = _repair(best, n)
                nc = _c5(best, dx)
                if nc < best_nc - 1e-13:
                    best_nc = nc
                    best_v = c
            best[i] = best_v
            best = _repair(best, n)
            if best_v != center:
                best_c5 = best_nc
    return best, best_c5

def _local_search(h, n, dx, rng, deadline, min_step=0.001, max_step=0.04):
    best = h.copy()
    best_c5 = _c5(best, dx)
    n_steps = 12
    steps = np.linspace(min_step, max_step, n_steps)
    steps = np.concatenate([steps, -steps])
    stall = 0
    for outer in range(8000):
        if time.time() > deadline:
            break
        improved = False
        ntrials = 60
        for _ in range(ntrials):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = oi, oj
                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break
    return best, best_c5

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return h

def _sin_init(n, freq, phase, rng):
    t = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * t + phase) + 0.05 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)

    def elapsed():
        return time.time() - t0

    def remaining(margin):
        return budget_s - margin - elapsed() > 0

    # Phase 1: SLSQP from multiple starts with alpha sweep
    starts = [h_init]
    starts.append(_repair(_binary_init(n, np.random.RandomState(seed + 3)), n))
    starts.append(_sin_init(n, 2, 0.5, rng))
    starts.append(_sin_init(n, 3, 1.7, rng))
    starts.append(_sin_init(n, 1, 2.3, rng))

    for si in range(len(starts)):
        if not remaining(300):
            break
        hc = starts[si].copy()
        c5 = _c5(hc, dx)
        for a in [30, 60, 120, 250, 500, 1000]:
            if not remaining(300):
                break
            hc2 = _slsqp(hc, n, dx, a, 300, 1e-14)
            nc = _c5(hc2, dx)
            if nc < c5:
                c5 = nc
                hc = hc2
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Coordinate descent (key new component)
    if remaining(200):
        dl = t0 + budget_s - 200
        best_h, best_c5 = _coord_descent(best_h, n, dx, n_passes=15, deadline=dl, grid_n=20)

    # Phase 3: Fine local search
    if remaining(150):
        dl = t0 + budget_s - 150
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl, min_step=0.001, max_step=0.03)

    # Phase 4: Iterative shake + coord descent + local search
    it = 0
    sigmas = [0.08, 0.12, 0.05, 0.15, 0.1, 0.2, 0.06, 0.18, 0.04, 0.1]
    while remaining(120):
        sigma = sigmas[it % len(sigmas)]
        hp = best_h + sigma * rng.randn(n)
        hp = _repair(hp, n)
        # Quick SLSQP
        hc = _slsqp(hp, n, dx, 200, 150, 1e-13)
        c5 = _c5(hc, dx)
        # Coord descent refinement
        if remaining(130):
            dl_cd = t0 + budget_s - 130
            hc, c5_cd = _coord_descent(hc, n, dx, n_passes=5, deadline=dl_cd, grid_n=15)
            if c5_cd < c5:
                c5 = c5_cd
        # Fine local search if promising
        if c5 < best_c5 + 0.005 and remaining(135):
            dl_ls = t0 + budget_s - 135
            hc, c5 = _local_search(hc, n, dx, rng, dl_ls, min_step=0.001, max_step=0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()
        it += 1

    # Phase 5: Aggressive coordinate descent polish
    if remaining(60):
        dl = t0 + budget_s - 60
        best_h, best_c5 = _coord_descent(best_h, n, dx, n_passes=30, deadline=dl, grid_n=25)

    # Phase 6: Final ultra-fine local search
    if remaining(20):
        dl = t0 + budget_s - 5
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl, min_step=0.0005, max_step=0.01)

    # Phase 7: One last SLSQP with very high alpha
    if remaining(8):
        hc = _slsqp(best_h, n, dx, 2000, 500, 1e-15)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_h = hc
            best_c5 = nc
        if remaining(4):
            _dl = t0 + budget_s - 2
            best_h, best_c5 = _coord_descent(best_h, n, dx, n_passes=5, deadline=_dl, grid_n=15)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
