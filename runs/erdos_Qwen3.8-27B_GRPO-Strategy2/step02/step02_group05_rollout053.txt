import numpy as np
from scipy.optimize import minimize
import time

def _c5val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idxs = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in idxs:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _smmax(x, alpha):
    m = x.max()
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_func(x, alpha, dx):
    return _smmax(np.correlate(x, 1.0 - x, mode='full'), alpha) * dx

def _con_func(x, target):
    return x.sum() - target

def _slsqp(h, n, dx, alpha, iters, ftol):
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    try:
        r = minimize(_obj_func, h, args=(alpha, dx), method='SLSQP',
                     bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': _con_func, 'args': (tgt,)}],
                     options={'maxiter': iters, 'ftol': ftol})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _pairwise_pass(h, n, dx, ngrid, deadline, refine=True):
    best = h.copy()
    bc5 = _c5val(best, dx)
    for _ in range(300):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                hi, hj = best[i], best[j]
                lo = max(-hj, hi - 1.0) + 1e-9
                hi_v = min(1.0 - hj, hi) - 1e-9
                if hi_v <= lo:
                    continue
                best_c = bc5
                best_d = 0.0
                step = (hi_v - lo) / ngrid
                for d in np.linspace(lo, hi_v, ngrid):
                    best[i], best[j] = hi - d, hj + d
                    v = _c5val(best, dx)
                    if v < best_c - 1e-14:
                        best_c = v
                        best_d = d
                if best_d != 0.0:
                    if refine:
                        lo_r = max(best_d - step, lo)
                        hi_r = min(best_d + step, hi_v)
                        for _ in range(12):
                            m1 = lo_r + (hi_r - lo_r) / 3.0
                            m2 = hi_r - (hi_r - lo_r) / 3.0
                            best[i], best[j] = hi - m1, hj + m1
                            v1 = _c5val(best, dx)
                            best[i], best[j] = hi - m2, hj + m2
                            v2 = _c5val(best, dx)
                            if v1 < v2:
                                hi_r = m2
                            else:
                                lo_r = m1
                        best_d = (lo_r + hi_r) / 2.0
                        best[i], best[j] = hi - best_d, hj + best_d
                        best_c = _c5val(best, dx)
                    else:
                        best[i], best[j] = hi - best_d, hj + best_d
                    bc5 = best_c
                    improved = True
                else:
                    best[i], best[j] = hi, hj
        if not improved:
            break
    return best, bc5

def _perturb_run(h_best, n, dx, rng, deadline, seed_off):

    hp = _repair(h_best + sigma * rng.randn(n), n)
