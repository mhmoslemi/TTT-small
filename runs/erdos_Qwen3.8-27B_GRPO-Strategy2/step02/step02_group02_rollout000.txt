from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            a = min(abs(diff), (1.0 - x[k]) if diff > 0 else x[k])
            x[k] += a if diff > 0 else -a
            diff -= a if diff > 0 else a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    return m + float(np.log(e.sum())) / a

def _sgrad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = w[n-1] * (1.0 - 2.0 * h)
    for L in range(1, n):

        if wn > 1e-14:

def _fista(h0, alpha, target, iters=5000, t_start=None, t_budget=None, min_time=2.0):

    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 10:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5/step * float(np.dot(xn-z, xn-z))
            if fn <= quad + 1e-9:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t-1.0)/tn * (xn - x), target)
        x = xn
        t = tn
        if (it + 1) % 200 == 0:

            if t_start is not None and (time.time() - t_start) > t_budget - min_time:

                    h[i] = hi0; h[j] = hj0
                else:
                    h[i] = hi0; h[j] = hj0
            if not imp:
                break
    return h, bc

def _sim_anneal(h, dx, n, target, t0, budget_s, min_time=10.0, max_it=3000):
    h = h.copy()
    bc = _c5(h, dx)
    bh = h.copy()
    T = 0.01
    rng = np.random.RandomState(int(time.time()) % 10000)
    for it in range(max_it):
        if time.time() - t0 > budget_s - min_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (0.01 + 0.05 * T) * rng.randn()
        ni, nj = h[i] + delta, h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        old_i, old_j = h[i], h[j]
        h[i] = ni; h[j] = nj
        c_new = _c5(h, dx)
        if c_new < bc or rng.rand() < np.exp(-(c_new - bc) / max(T, 1e-6)):
            bc = min(bc, c_new)
            if c_new < bc:
                bh = h.copy()
        else:
            h[i] = old_i; h[j] = old_j
        T *= 0.999
    return bh, bc

def _fourier_restart(h, n, freq, amp, phase, tgt, rng):
    x = np.arange(n)
    basis = amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    hp = h + basis
    return _proj(hp, tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)
    h0 = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: FISTA with vectorized gradient, increasing alpha
    alphas = [50, 100, 200, 400, 800, 1600, 3200, 5000]
    iters_list = [6000, 5000, 5000, 4000, 3000, 2500, 2000, 1500]
    for ai, a in enumerate(alphas):
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        est = iters_list[ai] * 0.003
        if est > rem - 20:
            continue
        h, c = _fista(bh, a, tgt, iters=iters_list[ai], t_start=t0, t_budget=budget_s, min_time=40)
        if c < bc:
            bc = c; bh = h

    # Phase 2: Coordinate descent (multi-scale)
    if budget_s - (time.time() - t0) > 30:
        h, c = _cd(bh, dx, n, 200, rng, t0, budget_s)
        if c < bc:
            bc = c; bh = h

    # Phase 3: Simulated annealing
    if budget_s - (time.time() - t0) > 60:
        h, c = _sim_anneal(bh, dx, n, tgt, t0, budget_s, min_time=30, max_it=5000)
        if c < bc:
            bc = c; bh = h
        # FISTA polish after SA
        if budget_s - (time.time() - t0) > 40:
            h, c = _fista(bh, 2000, tgt, iters=2000, t_start=t0, t_budget=budget_s, min_time=20)
            if c < bc:
                bc = c; bh = h

    # Phase 4: Fourier-mode restarts
    if budget_s - (time.time() - t0) > 80:
        for freq in [1, 2, 3, 4, 5, 7]:
            for amp in [0.05, 0.1, 0.15, 0.2]:
                if budget_s - (time.time() - t0) < 30:
                    break
                phase = rng.rand() * 2.0 * np.pi
                hp = _fourier_restart(bh, n, freq, amp, phase, tgt, rng)
                h, c = _fista(hp, 500, tgt, iters=2000, t_start=t0, t_budget=budget_s, min_time=20)
                if c < bc:
                    bc = c; bh = h
                    if budget_s - (time.time() - t0) > 20:
                        h2, c2 = _cd(bh, dx, n, 80, rng, t0, budget_s, min_time=10)
                        if c2 < bc:
                            bc = c2; bh = h2

    # Phase 5: Gaussian restarts (fine-tuning)
    if budget_s - (time.time() - t0) > 60:
        for s in [0.03, 0.05, 0.08, 0.04, 0.06, 0.02, 0.1, 0.07]:
            if budget_s - (time.time() - t0) < 30:
                break
            hp = _proj(bh + s * rng.randn(n), tgt)
            h, c = _fista(hp, 1000, tgt, iters=2000, t_start=t0, t_budget=budget_s, min_time=20)
            if c < bc:
                bc = c; bh = h
                h2, c2 = _cd(bh, dx, n, 60, rng, t0, budget_s, min_time=10)
                if c2 < bc:
                    bc = c2; bh = h2

    # Phase 6: SLSQP polish
    bnds = [(0.0, 1.0)] * n
    if budget_s - (time.time() - t0) > 25:
        try:
            r = minimize(_sobj, bh, args=(3000,), method='SLSQP', bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - tgt}],
                         options={'maxiter': 400, 'ftol': 1e-14})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf; bh = hf
        except Exception:
            pass

    # Phase 7: Final intensive CD
    if budget_s - (time.time() - t0) > 15:
        h, c = _cd(bh, dx, n, 300, rng, t0, budget_s, min_time=3.0)
        if c < bc:
            bc = c; bh = h

    # Phase 8: Final SLSQP
    if budget_s - (time.time() - t0) > 10:
        try:
            r = minimize(_sobj, bh, args=(8000,), method='SLSQP', bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - tgt}],
                         options={'maxiter': 500, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf; bh = hf
        except Exception:
            pass

    # Final CD
    if budget_s - (time.time() - t0) > 5:
        h, c = _cd(bh, dx, n, 200, rng, t0, budget_s, min_time=1.0)
        if c < bc:
            bc = c; bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
