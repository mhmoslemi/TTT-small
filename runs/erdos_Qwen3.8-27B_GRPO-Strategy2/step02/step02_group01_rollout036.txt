df/dx = dM/dx + (1/α) * Σ_j [exp(z_j) / Σ_k exp(z_k)] * dz_j/dx

dz_j/dx = α(dc_j/dx - dM/dx)

So: df/dx = dM/dx + (1/α) * Σ_j w_j * α(dc_j/dx - dM/dx)

         = dM/dx + Σ_j w_j * dc_j/dx - dM/dx
         = Σ_j w_j * dc_j/dx

where w_j = exp(z_j) / Σ_k exp(z_k)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

    m = c.max()

    return m + np.log(np.sum(np.exp(s))) / a

        dsets = [
            np.array([0.02, 0.05, 0.1, 0.15, -0.02, -0.05, -0.1, -0.15]),
            np.array([0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05]),
            np.array([0.001, 0.003, 0.005, 0.01, -0.001, -0.003, -0.005, -0.01]),
            np.array([0.0002, 0.0005, 0.001, 0.003, -0.0002, -0.0005, -0.001, -0.003]),
        ]
        for ds in dsets:
            if _el() > budget_s - 80:
                break
            for _pass in range(300):
                if _el() > budget_s - 80:
                    break
                imp = False
                for i in range(n):
                    if _el() > budget_s - 80:
                        break
                    hi = h[i]
                    for j in range(i + 1, n):
                        if _el() > budget_s - 80:
                            break
                        hj = h[j]
                        for d in ds:
                            ni, nj = hi + d, hj - d
                            if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                                continue
                            h[i], h[j] = ni, nj
                            c5n = _eval(h)
                            if c5n < c5c - 1e-16:
                                c5c = c5n
                                hi, hj = ni, nj
                                imp = True
                                if c5c < best_c5:
                                    best_c5 = c5c
                                    best_h = h.copy()
                            else:
                                h[i], h[j] = hi, hj
                        h[j] = hj
                    h[i] = hi
                if not imp:
                    break

    # Phase 4: Random fine-tuning with adaptive step size
    if _el() < budget_s - 30:
        h = best_h.copy()
        c5c = _eval(h)
        scale = 0.003
        no_improve = 0
        for _it in range(50000):
            if _el() > budget_s - 30:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                no_improve = 0
                scale = min(scale * 1.2, 0.01)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                no_improve += 1
                if no_improve > 500:
                    scale *= 0.7
                    no_improve = 0
                    if scale < 1e-6:
                        break

    # Phase 5: Final SLSQP polish with very high alpha
    if _el() < budget_s - 15:
        h = _opt(best_h, n, 1000000, 3000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Ultra-fine coordinate moves
    if _el() < budget_s - 5:
        h = best_h.copy()
        c5c = _eval(h)
        for _it in range(30000):
            if _el() > budget_s - 5:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-4 * rng.choice([1, 2, 5])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
