```python
def _smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -400, 400)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = shifts[k]
        # Term 1: (1-h[i-s]) for valid i
        i_lo = max(0, s)
        i_hi = min(n, n+s)  # exclusive
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo - s:i_hi - s])
        # Term 2: -h[i+s] for valid i
        j_lo = max(0, -s)
        j_hi = min(n, n-s)  # exclusive
        if j_lo < j_hi:
            grad[j_lo:j_hi] -= w * h[j_lo + s:j_hi + s]
    
    fval = m + np.log(exps.sum()) / alpha
    return fval, grad
```

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(np.max(corr))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(alpha * (corr - m), -400, 400)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = shifts[k]
        i_lo = max(0, s)
        i_hi = min(n, n+s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo - s:i_hi - s])
        j_lo = max(0, -s)
        j_hi = min(n, n-s)
        if j_lo < j_hi:
            grad[j_lo:j_hi] -= w * h[j_lo + s:j_hi + s]
    
    fval = m + np.log(exps.sum()) / alpha
    return fval, grad

def _slsqp_opt(h, n, alpha, maxiter, ftol):
    def obj(x):
        return _smooth_obj_grad(x, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()
