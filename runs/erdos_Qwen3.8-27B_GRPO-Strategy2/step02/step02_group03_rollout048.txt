from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, n):
    return float(np.max(_corr_full(h)) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(500):
        h *= (target / max(s, 1e-15))
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        if abs(diff) < 1e-12:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _grad_top_lags(h, n, top_idx):
    grad = np.zeros(n)
    offset = n - 1
    for idx in top_idx:
        k = idx - offset
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] += (-h[k:])
        else:
            k2 = -k
            grad[:n - k2] += (1.0 - h[k2:])
            grad[k2:] += (-h[:n - k2])
    return grad

def _gradient_descent(h, n, t_end, rng):

    best_c5 = _compute_c5(h, n)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:

        h_try = _repair(h_try, n)
        c5_try = _compute_c5(h_try, n)

            h = h_try
            best_c5 = c5_try
            step *= 1.05
            if step > 0.02:

        else:

            if step < 1e-5:

    best_c5 = _compute_c5(h, n)

def _slsqp_refine(h, n, t_end, rng):
    h = h.copy()
    best_c5 = _compute_c5(h, n)
    for alpha in [200, 500, 1500]:
        if time.time() > t_end:
            break
        def _obj(x):
            corr = _corr_full(x)
            m = np.max(corr)
            s = alpha * (corr - m)
            s = np.clip(s, -500, 500)
            return m + np.log(np.sum(np.exp(s))) / alpha
        def _con(x):
            return np.sum(x) - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con}]
        try:
            res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-18})
            h_new = _repair(res.x, n)
            c5_new = _compute_c5(h_new, n)
            if c5_new < best_c5:

                best_c5 = c5_new
