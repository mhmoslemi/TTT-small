            lo = mid
        else:
            hi = mid

            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _project_sum(h, target):
    """Project h onto {x: sum(x)=target} staying in [0,1] approximately."""
    # Iterative
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + adj)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            adj = diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] + adj)
    return h

def _grad_c5(h, n, top_k=5):
    """Compute subgradient of c5 objective."""
    cvals = _corr_vals(h)
    f = cvals.max()
    idx = np.argsort(cvals)[::-1][:top_k]
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k_idx in idx:
        s = shifts[k_idx]
        if s == 0:
            grad += (1 - 2*h)
        elif s > 0:
            grad[s:] += (1 - h[:n-s])
            grad[:n-s] -= h[s:]
        else:
            s2 = -s
            grad[:n-s2] += (1 - h[s2:])
            grad[s2:] -= h[:n-s2]
    grad /= len(idx)
    return grad, f

def _pg_descent(h, n, dx, t_limit, t_start, n_iter=20000, top_k=5):
    """Projected gradient descent with momentum."""
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    v = np.zeros(n)  # momentum
    beta = 0.9
    eta = 0.1
    
    for it in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        
        grad, f = _grad_c5(h, n, top_k)
        if f < best_c5 - 1e-12:
            best_c5 = f
            best_h = h.copy()
        
        # Momentum
        v = beta * v - eta * grad / max(1.0, np.max(np.abs(grad)))
        h_new = h + v
        h_new = np.clip(h_new, 0, 1)
        h = _project_sum(h_new, target)
        
        # Decay step size
        if it % 1000 == 0 and it > 0:
            eta *= 0.8
    
    return best_h, best_c5

def _cd_pairs(h, n, dx, t_limit, t_start, n_grid=15):
    """Coordinate descent on pairs."""
    best_c5 = _c5(h, dx)
    n_passes = 0
    for _ in range(5000):
        if time.time() - t_start > t_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                hs = h[i] + h[j]
                lo = max(0.0, hs - 1.0)
                hi = min(1.0, hs)
                if hi - lo < 1e-10:
                    continue
                orig_i, orig_j = h[i], h[j]
                best_t, best_c = orig_i, best_c5
                for t in np.linspace(lo + 1e-8, hi - 1e-8, n_grid):
                    h[i], h[j] = t, hs - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c, best_t = c, t
                if best_c < best_c5 - 1e-12:
                    h[i], h[j] = best_t, hs - best_t
                    best_c5 = best_c
                    improved = True
                else:
                    h[i], h[j] = orig_i, orig_j
            if time.time() - t_start > t_limit:
                break
        n_passes += 1
        if not improved:
            break
    return h, best_c5, n_passes

def _slsqp_smooth(h, n, dx, alpha, maxiter, t_limit, t_start):
    """SLSQP with log-sum-exp smoothing."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def smooth_obj(x):
        c = _corr_vals(x) * dx
        m = c.max()
        s = np.clip(20 - alpha * (m - c), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    try:
        res = minimize(smooth_obj, h, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_vals(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _project_sum(h, target):
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = mask.sum()
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + adj)
        else:
            mask = h > 1e-9
            cnt = mask.sum()
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.maximum(0.0, h[mask] + adj)
    return h

def _grad_sub(h, n, top_k):
    cvals = _corr_vals(h)
    f = cvals.max()
    idx = np.argsort(cvals)[::-1][:top_k]
    shifts = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for kk in idx:
        s = shifts[kk]
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        else:
            s2 = -s
            grad[:n - s2] += (1.0 - h[s2:])
            grad[s2:] -= h[:n - s2]
    grad /= float(len(idx))
    return grad, f

def _pg_descent(h, n, dx, t_limit, t_start, n_iter, top_k):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    v = np.zeros(n)
    beta = 0.85
    eta = 0.15
    stall = 0
    for it in range(n_iter):
        if time.time() - t_start > t_limit:
            break
        grad, f = _grad_sub(h, n, top_k)
        if f < best_c5 - 1e-10:
            best_c5 = f
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                eta *= 0.9
                stall = 0
        gnorm = np.max(np.abs(grad))
        if gnorm > 1e-12:
            v = beta * v - eta * grad / gnorm
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h = _project_sum(h_new, target)
        if it % 2000 == 1999:
            eta *= 0.85
    return best_h, best_c5

def _cd_sweep(h, n, dx, t_limit, t_start, n_grid, max_passes, random_pairs):
    target = n / 2.0
    best_c5 = _c5(h, dx)
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        if random_pairs:
            pairs = [(np.random.randint(n), np.random.randint(n)) for _ in range(3000)]
            pairs = [(min(a,b), max(a,b)) for a,b in pairs if a != b]
        else:
            pairs = all_pairs
        for i, j in pairs:
            hs = h[i] + h[j]
            lo = max(0.0, hs - 1.0)
            hi = min(1.0, hs)
            if hi - lo < 1e-10:
                continue
            oi, oj = h[i], h[j]
            best_t, best_c = oi, best_c5
            grid = np.linspace(lo + 1e-9, hi - 1e-9, n_grid)
            for t in grid:
                h[i], h[j] = t, hs - t
                c = _c5(h, dx)
                if c < best_c:
                    best_c, best_t = c, t
            if best_c < best_c5 - 1e-13:
                h[i], h[j] = best_t, hs - best_t
                best_c5 = best_c
                improved = True
            else:
                h[i], h[j] = oi, oj
        if not improved and not random_pairs:
            break
    return h, best_c5

def _slsqp_smooth(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = _corr_vals(x) * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def cons_fun(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: Projected gradient (long)
    if elapsed() < budget_s - 300:
        h, c = _pg_descent(best_h, n, dx, budget_s - 320, t0, 50000, 5)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Randomized CD
    if elapsed() < budget_s - 200:
        h, c = _cd_sweep(best_h, n, dx, budget_s - 220, t0, 15, 30, True)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 3: SLSQP polish
    if elapsed() < budget_s - 150:
        h = _slsqp_smooth(best_h, n, dx, 1500, 600)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 4: Full CD sweep (coarse)
    if elapsed() < budget_s - 100:
        h, c = _cd_sweep(best_h, n, dx, budget_s - 120, t0, 13, 30, False)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 5: PG with top-3 (sharper)
    if elapsed() < budget_s - 80:
        h, c = _pg_descent(best_h, n, dx, budget_s - 95, t0, 30000, 3)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 6: Fine CD (randomized)
    if elapsed() < budget_s - 50:
        h, c = _cd_sweep(best_h, n, dx, budget_s - 65, t0, 21, 40, True)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 7: SLSQP final
    if elapsed() < budget_s - 25:
        h = _slsqp_smooth(best_h, n, dx, 4000, 1000)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 8: Multi-restart
    sigmas = [0.06, 0.1, 0.15, 0.04]
    for si in range(len(sigmas)):
        if elapsed() > budget_s - 20:
            break
        hp = best_h + sigmas[si] * rng.randn(n)
        hp = _repair(hp, n)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
