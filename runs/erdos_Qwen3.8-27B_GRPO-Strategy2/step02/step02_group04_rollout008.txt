import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    mse = np.logsumexp(s) if hasattr(np, 'logsumexp') else np.log(np.sum(np.exp(s)))
    return m + mse / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    if time.time() >= deadline:
        return h, _c5(h, dx)
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smooth_obj(x, alpha)

                       constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)
