def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _c5_lag(h, dx):
    c = _corr(h)
    idx = int(np.argmax(c))
    return float(c[idx] * dx), idx

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_obj(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _slsqp_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'args': (n / 2.0,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 250, 200, 1e-13)
        c5 = _c5(hc, dx)
        if rem(100) > 5:
            dl = t0 + budget_s - 100
            hc, c5 = _grad_descent(hc, n, dx, dl, 150)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        elif c5 < best_c5 + 0.004 and rem(130) > 0:
            dl = t0 + budget_s - 130
            hl, cl = _local_search(hc, n, dx, rng, dl, 2000)
            if cl < best_c5:
                best_c5, best_h = cl, hl
        it += 1

    # Phase 3: Final SLSQP polish + gradient + fine LS
    if rem(20) > 5:
        for a in [1500, 3000, 6000]:
            if rem(10) < 0:
                break
            hc = _slsqp(best_h, n, dx, a, 500, 1e-15)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
    if rem(15) > 5:
        dl = t0 + budget_s - 12
        best_h, best_c5 = _grad_descent(best_h, n, dx, dl, 200)
    if rem(6) > 3:
        dl = t0 + budget_s - 2
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl, 4000)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
