import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = diff
        for idx in order:
            if remaining > 0:
                amt = min(remaining, 1.0 - h[idx])
                h[idx] += amt
                remaining -= amt
            else:
                amt = min(-remaining, h[idx])
                h[idx] -= amt
                remaining += amt
            if abs(remaining) < 1e-15:
                break
    return h

def _nm_obj(x, n):
    # Raw max correlation + boundary penalty
    c = _corr(x)
    val = np.max(c)
    # Penalty for constraint violation
    s = x.sum() - n / 2.0
    val += 0.1 * s * s
    # Penalty for out of bounds
    val += 10.0 * np.sum(np.maximum(0, -x)**2) + 10.0 * np.sum(np.maximum(0, x-1)**2)
    return val

        return _repair(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                amt = min(diff, 1.0 - h[idx])
                h[idx] += amt
                diff -= amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt
                diff += amt
            if abs(diff) < 1e-14:
                break
    return h

def _targeted_cd(h, n, dx, budget_end, rng, max_rounds=20000):
    h = h.copy()
    best = _c5(h, dx)
    rnd = 0
    while time.time() < budget_end and rnd < max_rounds:
        rnd += 1
        c = _corr(h)
        m = np.max(c)
        top_lags = np.where(c >= m - 0.3)[0]
        improved = False
        for lag_idx in top_lags[:6]:
            if time.time() > budget_end:
                break
            offset = lag_idx - (n - 1)
            if offset >= 0:
                valid_range = range(n - offset)
            else:
                valid_range = range(-offset, n)
            pairs = []
            for i in valid_range:
                j = i + offset
                if 0 <= j < n:
                    prod = h[i] * (1.0 - h[j])
                    if prod > 0.2:
                        pairs.append((i, j, prod))
            pairs.sort(key=lambda x: -x[2])
            for i, j, _ in pairs[:4]:
                if h[i] > h[j]:
                    d = 0.005
                else:
                    d = -0.005
                old_i, old_j = h[i], h[j]
                h[i] = np.clip(h[i] + d, 0, 1)
                h[j] = np.clip(h[j] - d, 0, 1)
                c5_new = _c5(h, dx)
                if c5_new < best - 1e-15:

                    improved = True
                else:

                idx = rng.randint(n)
                h[idx] += rng.uniform(-0.02, 0.02)
            h = _repair(h, n)
            best = _c5(h, dx)
    return h, best

def _paired_cd(h, n, dx, budget_end, rng, max_rounds=30000):
    h = h.copy()
    best = _c5(h, dx)
    steps = [0.001, 0.002, 0.005, 0.01, 0.02]
    rnd = 0
    while time.time() < budget_end and rnd < max_rounds:
        rnd += 1
        sc = steps[min(rnd // 5000, 4)]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in [sc, 2 * sc]:
            for s in [1, -1]:
                ni, nj = hi + s * d, hj - s * d
                if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c5_new = _c5(h, dx)
                if c5_new < best - 1e-15:
                    best = c5_new
                    break
                h[i], h[j] = hi, hj
            else:
                continue
            break
    return h, best

def _sa_refine(h, n, dx, budget_end, rng, n_iter=50000):
    h = h.copy()
    best = _c5(h, dx)
    T = 0.003
    for _ in range(n_iter):
        if time.time() > budget_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.015, 0.015)
        old_i, old_j = h[i], h[j]
        h[i] = np.clip(h[i] + d, 0, 1)
        h[j] = np.clip(h[j] - d, 0, 1)
        c5_new = _c5(h, dx)
        if c5_new < best:
            best = c5_new
        elif rng.rand() < np.exp(-(c5_new - best) / max(T, 1e-12)):
            pass
        else:
            h[i], h[j] = old_i, old_j
        T = max(T * 0.99995, 0.0005)
    return _repair(h, n), best

def _smooth_obj(x, n, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += 0.5 * (np.sum(x) - n / 2.0) ** 2
    return val

def _slsqp_smooth(h, n, alpha, maxiter):
    g_alpha = alpha
    g_target = n / 2.0
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(g_alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / g_alpha
    def con(x):
        return np.sum(x) - g_target
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h
