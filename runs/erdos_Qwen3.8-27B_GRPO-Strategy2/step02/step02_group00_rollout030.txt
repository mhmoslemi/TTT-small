import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(c, -k)[-k:]
    return float(c[idx].sum()) * dx

    m = float(c.max())
    s = np.exp(np.clip(alpha * (c - m), -50, 50))
    return (m + float(s.sum()) / alpha) * dx

    c = _eval(tm, dx)
    if c < best_c5:
        best_c5, best_h = c, tm.copy()
    tm2 = _repair(1.0 - _thue_morse(n), n)
    c = _eval(tm2, dx)
    if c < best_c5:
        best_c5, best_h = c, tm2.copy()
    for _ in range(4):
        rb = _repair((rng.rand(n) > 0.5).astype(float), n)
        c = _eval(rb, dx)
        if c < best_c5:
            best_c5, best_h = c, rb.copy()

    # Phase 1: SLSQP
    p1 = t0 + budget_s * 0.09
    for k in (3, 5, 8, 15):
        if time.time() > p1:
            break
        try:
            r = minimize(_topk_obj, best_h.copy(), args=(k, n, dx),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 250, 'ftol': 1e-14})
            hc = _repair(r.x, n)
            c5 = _eval(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    for a in (1500, 4000, 10000):
        if time.time() > p1:
            break
        try:
            r = minimize(_lse_obj, best_h.copy(), args=(a, n, dx),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 350, 'ftol': 1e-14})
            hc = _repair(r.x, n)
            c5 = _eval(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Intensive SA with restarts
    sa_end = t0 + budget_s * 0.62
    h_sa = best_h.copy()
    sa_best = _eval(h_sa, dx)
    sa_best_h = h_sa.copy()
    it = 0
    T0, Tend = 0.007, 0.000002
    max_it = 18000000
    last_imp = 0

    while it < max_it and time.time() < sa_end:
        frac = min(1.0, it / max_it)
        T = T0 * (Tend / T0) ** frac
        r = rng.rand()
        if r < 0.38:
            dm = 0.06 * (1.0 - 0.7 * frac)
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                it += 1
                continue
            d = (rng.rand() - 0.5) * 2.0 * dm
            ni, nj = h_sa[i] + d, h_sa[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                it += 1
                continue
            h_sa[i], h_sa[j] = ni, nj
            c5 = _eval(h_sa, dx)
            if c5 <= sa_best + T:
                if c5 < sa_best - 1e-15:
                    sa_best = c5
                    sa_best_h = h_sa.copy()
                    last_imp = it
            else:
                h_sa[i], h_sa[j] = ni - d, nj + d
        elif r < 0.68:
            i, j = rng.randint(n), rng.randint(n)
            if i == j or abs(h_sa[i] - h_sa[j]) < 1e-8:
                it += 1
                continue
            h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            c5 = _eval(h_sa, dx)
            if c5 <= sa_best + T:
                if c5 < sa_best - 1e-15:
                    sa_best = c5
                    sa_best_h = h_sa.copy()
                    last_imp = it
            else:
                h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
        else:
            bl = rng.randint(2, 5)
            i = rng.randint(n - bl + 1)
            j = rng.randint(n - bl + 1)
            if abs(i - j) < bl:
                it += 1
                continue
            amax = min(h_sa[i:i + bl].min(), 1.0 - h_sa[j:j + bl].max())
            if amax < 1e-5:
                it += 1
                continue
            a = rng.rand() * amax
            h_sa[i:i + bl] -= a
            h_sa[j:j + bl] += a
            c5 = _eval(h_sa, dx)
            if c5 <= sa_best + T:
                if c5 < sa_best - 1e-15:
                    sa_best = c5
                    sa_best_h = h_sa.copy()
                    last_imp = it
            else:
                h_sa[i:i + bl] += a
                h_sa[j:j + bl] -= a
        it += 1
        if it - last_imp > 1500000 and it > 500000:
            h_sa = sa_best_h.copy()
            amp = 0.02 * max(0.1, 1.0 - frac)
            h_sa += rng.randn(n) * amp
            h_sa = _repair(h_sa, n)
            last_imp = it

    if sa_best < best_c5:
        best_c5, best_h = sa_best, sa_best_h

    # Phase 3: Coarse-to-fine local search
    p3 = t0 + budget_s * 0.86
    h = best_h.copy()
    h, lb = _local_search(h, dx, n, p3, (0.01, 0.004, 0.002, 0.001, 0.0004, 0.0002), 12)
    if lb < best_c5:
        best_c5, best_h = lb, h.copy()

    # Phase 4: Second SA round (shorter, colder)
    if time.time() < t0 + budget_s * 0.93:
        h_sa2 = best_h.copy()
        sb2 = _eval(h_sa2, dx)
        sb2_h = h_sa2.copy()
        it2 = 0
        max_it2 = 4000000
        for _ in range(max_it2):
            if time.time() >= t0 + budget_s * 0.94:
                break
            it2 += 1
            T = 0.001 * (0.00001 / 0.001) ** (it2 / max_it2)
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = (rng.rand() - 0.5) * 0.015
            ni, nj = h_sa2[i] + d, h_sa2[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h_sa2[i], h_sa2[j] = ni, nj
            c5 = _eval(h_sa2, dx)
            if c5 <= sb2 + T:
                if c5 < sb2:
                    sb2 = c5
                    sb2_h = h_sa2.copy()
            else:
                h_sa2[i], h_sa2[j] = ni - d, nj + d
        if sb2 < best_c5:
            best_c5, best_h = sb2, sb2_h

    # Phase 5: Final SLSQP + micro local search
    if time.time() < t0 + budget_s - 6:
        try:
            r = minimize(_lse_obj, best_h.copy(), args=(12000, n, dx),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-16})
            hc = _repair(r.x, n)
            c5 = _eval(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    if time.time() < t0 + budget_s - 2:
        h = best_h.copy()
        h, lb = _local_search(h, dx, n, t0 + budget_s - 1, (0.0001, 0.00005, 0.00002), 8)
        if lb < best_c5:
            best_c5, best_h = lb, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h, dx)
    return (best_h, best_c5, n)
