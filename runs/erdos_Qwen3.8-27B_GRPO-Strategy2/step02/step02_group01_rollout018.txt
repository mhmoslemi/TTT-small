def _smooth_obj_and_grad(x, alpha):
    n = len(x)

    m = np.max(c)
    # Numerical stability
    shifted = c - m
    weights = np.exp(alpha * shifted)
    wsum = np.sum(weights)

    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1)

import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx, c


def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _smooth_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    g = np.zeros(n)
    ar = np.arange(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        delta = k - (n - 1)
        idx1 = ar - delta
        mask1 = (idx1 >= 0) & (idx1 < n)
        if mask1.any():
            g[mask1] += p[k] * (1.0 - x[idx1[mask1]])
        idx2 = ar + delta
        mask2 = (idx2 >= 0) & (idx2 < n)
        if mask2.any():
            g[mask2] -= p[k] * x[idx2[mask2]]
    return g


def _obj_wrap(x, alpha):
    return _smooth_obj(x, alpha)


def _grad_wrap(x, alpha):
    return _smooth_grad(x, alpha)


def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _grad_worst_lag(h, n, k_star):
    delta = k_star - (n - 1)
    g = np.zeros(n)
    ar = np.arange(n)
    idx1 = ar - delta
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = ar + delta
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    return g
