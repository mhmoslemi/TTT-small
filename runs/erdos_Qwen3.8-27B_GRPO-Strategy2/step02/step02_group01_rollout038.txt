import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_lag(h, k, n):
    d = k - (n - 1)
    ad = abs(d)
    g = np.zeros(n)
    if ad == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[ad:] += 1.0 - h[:-ad]
        g[:n - ad] -= h[ad:]
    else:
        g[:n - ad] += 1.0 - h[ad:]
        g[ad:] -= h[:-ad]
    return g

def _project_feasible(h, c):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    # Tiny correction for exact sum
    diff = c - result.sum()
    if abs(diff) > 1e-15:
        adjustable = np.where((result > 1e-15) & (result < 1 - 1e-15))[0]
        if len(adjustable) > 0:
            result[adjustable] += diff / len(adjustable)
            result = np.clip(result, 0.0, 1.0)
    return result

    a = alpha * c * dx
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_lag(h, k, n):
    d = k - (n - 1)
    ad = abs(d)
    g = np.zeros(n)
    if ad == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[ad:] += 1.0 - h[:n - ad]
        g[:n - ad] -= h[ad:]
    else:
        g[:n - ad] += 1.0 - h[ad:]
        g[ad:] -= h[:n - ad]
    return g

def _project_feasible(h, c):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _lse_weights(c_arr, alpha, dx):
    a = alpha * c_arr * dx
    am = a.max()
    w = np.exp(a - am)
    w /= w.sum()
    return w

def _cont_opt(h_init, n, dx, t0, t_limit):
    ct = n / 2.0
    h = _project_feasible(h_init.copy(), ct)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    nl = 2 * n - 1
    alphas = [30, 80, 250, 800, 2500, 8000, 25000, 80000]
    for alpha in alphas:
        if time.time() - t0 > t_limit:
            break
        step = 1.0
        for _it in range(400):
            if time.time() - t0 > t_limit:
                break
            c_arr = _corr(h)
            w = _lse_weights(c_arr, alpha, dx)
            g = np.zeros(n)
            for k in range(nl):
                if w[k] > 1e-14:
                    g += w[k] * _grad_lag(h, k, n)
            g *= dx
            cur_c5 = float(c_arr.max() * dx)
            for _bt in range(60):
                h_new = _project_feasible(h - step * g, ct)
                if _c5(h_new, dx) <= cur_c5 + 1e-16:
                    break
                step *= 0.5
            h = h_new
            step = min(step * 1.15, 80.0)
            c5n = _c5(h, dx)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
    return best_h, best_c5

def _make_binary(h, n, rng):
    ct = n // 2
    h = (h >= 0.5).astype(float)
    while h.sum() > ct:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < ct:
        h[rng.choice(np.where(h == 0)[0])] = 1
    return h

def _bin_descent(h, n, dx, t0, t_limit, rng):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for _outer in range(300):
        if time.time() - t0 > t_limit:
            break
        if _outer > 0:
            nk = min(3 + _outer % 5, 12)
            for _ in range(nk):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) < 1 or len(zeros) < 1:
                    break
                h[ones[rng.randint(len(ones))]] = 0
                h[zeros[rng.randint(len(zeros))]] = 1
        cur_c5 = _c5(h, dx)
        improved = True
        while improved and time.time() - t0 < t_limit:
            improved = False
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            bi, bj = -1, -1
            bd = -1e-12
            for ii in range(len(ones)):
                for jj in range(len(zeros)):
                    i, j = ones[ii], zeros[jj]
                    h[i], h[j] = 0.0, 1.0
                    c5v = _c5(h, dx)
                    h[i], h[j] = 1.0, 0.0
                    if c5v - cur_c5 < bd:
                        bd = c5v - cur_c5
                        bi, bj = i, j
            if bi >= 0:
                h[bi], h[bj] = 0.0, 1.0
                cur_c5 += bd
                improved = True
        c5f = _c5(h, dx)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = h.copy()
    return best_h, best_c5

def _refine(h_init, n, dx, t0, t_limit, rng):
    ct = n / 2.0
    h = _project_feasible(h_init.copy(), ct)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    scales = np.array([0.0005, 0.001, 0.003, 0.01, 0.03])
    while time.time() - t0 < t_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scales[rng.randint(5)] * (1.0 if rng.rand() < 0.5 else -1.0)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5v = _c5(h, dx)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    return best_h, best_c5

    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    ct = n / 2.0
    rng = np.random.RandomState(seed)

    best_h = h0.copy()
    best_c5 = _c5(h0, dx)
