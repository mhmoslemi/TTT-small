import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _grad_sobj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - n1
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - x[lag:])
            g[lag:] -= wk * x[:n - lag]
        else:
            L = -lag
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
    return g

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _fista(h0, alpha, target, iters, t_budget):
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        if time.time() > t_budget:
            break
        gz = _grad_sobj(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = xn + (t - 1.0) / tn * (xn - x)
        x = xn
        t = tn
    return x

def _opt_1d(a, c):
    n = len(a)
    def _feasible(t):
        lo_x = 0.0
        hi_x = 1.0
        for k in range(n):
            if a[k] > 1e-14:
                hi_x = min(hi_x, (t - c[k]) / a[k])
            elif a[k] < -1e-14:
                lo_x = max(lo_x, (t - c[k]) / a[k])
            else:
                if c[k] > t + 1e-12:
                    return False
        return lo_x <= hi_x + 1e-12
    t_hi = float(np.max(np.maximum(a, a + c)) + 0.1)
    t_lo = 0.0
    for _ in range(50):
        t_mid = 0.5 * (t_lo + t_hi)
        if _feasible(t_mid):
            t_hi = t_mid
        else:
            t_lo = t_mid
    t = t_hi
    lo_x = 0.0
    hi_x = 1.0
    for k in range(n):
        if a[k] > 1e-14:
            hi_x = min(hi_x, (t - c[k]) / a[k])
        elif a[k] < -1e-14:
            lo_x = max(lo_x, (t - c[k]) / a[k])
    x = 0.5 * (lo_x + hi_x)
    return float(np.clip(x, 0.0, 1.0))

def _cyc_cd(h, target, n_cycles, t_budget):
    n = len(h)
    h = h.copy()
    n1 = n - 1
    bc = _c5(h, 2.0 / n)
    for cyc in range(n_cycles):
        if time.time() > t_budget:
            break
        for i in range(n):
            if time.time() > t_budget:
                break
            a_k = np.zeros(2 * n - 1)
            c_k = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - n1
                A = 0.0
                B = 0.0
                if 0 <= i + lag < n:
                    A = 1.0 - h[i + lag]
                if 0 <= i - lag < n:
                    B = h[i - lag]
                a_k[k] = A - B
                c_k[k] = a_k[k] * (-h[i]) + B  # overlap_k - (A-B)*h[i] = c_k
                # overlap_k = (A-B)*x + c_k, c_k = overlap_k - (A-B)*h[i]
                # But we need actual overlap_k
            # Compute actual overlaps once
            ov = np.correlate(h, 1.0 - h, mode='full')
            for k in range(2 * n - 1):
                c_k[k] = ov[k] - (a_k[k]) * h[i]
            x_new = _opt_1d(a_k, c_k)
            h[i] = x_new
            c = _c5(h, 2.0 / n)
            if c < bc - 1e-14:
                bc = c
    return h, bc

def _rand_cd(h, dx, target, n_passes, rng, t_budget):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    deltas = [0.005, 0.01, 0.02, 0.04, 0.07, 0.1, -0.005, -0.01, -0.02, -0.04, -0.07, -0.1]
    for p in range(n_passes):
        if time.time() > t_budget:
            break
        imp = 0
        for _ in range(n * 6):
            if time.time() > t_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            best_d = 0.0
            best_c = bc
            for d in deltas:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best_c - 1e-13:
                    best_c = c
                    best_d = d
                h[i], h[j] = hi0, hj0
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                bc = best_c
                imp += 1
        if imp == 0:
            break
    return h, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target
