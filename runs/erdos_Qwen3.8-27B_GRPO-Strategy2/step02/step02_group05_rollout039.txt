def _corr(h):

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:

                h[idx] -= sub
                diff += sub

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CP = True
except ImportError:
    HAS_CP = False

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_obj(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _sm_jac(x, alpha, dx):
    n = len(x)
    m = 2 * n - 1
    corr = _corr(x)
    c_max = np.max(corr)
    s = np.clip(alpha * (corr - c_max), -500, 500)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    obj = (c_max + np.log(np.sum(exp_s)) / alpha) * dx
    lags = np.arange(m) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        mask1 = (i - lags >= 0) & (i - lags < n)
        if np.any(mask1):
            grad[i] += np.dot(w[mask1], 1.0 - x[i - lags[mask1]])
        mask2 = (i + lags >= 0) & (i + lags < n)
        if np.any(mask2):
            grad[i] += np.dot(w[mask2], -x[i + lags[mask2]])
    grad *= dx
    return obj, grad

def _con_fun(x, target):
    return np.sum(x) - target

def _con_jac(x, target):
    return np.ones_like(x)

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target,)}]
    try:
        res = minimize(_sm_obj, h, args=(alpha, dx), jac=_sm_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _lp_step(h, n):
    if not HAS_CP:
        return h.copy()
    b = 1.0 - h
    m = 2 * n - 1
    lags = np.arange(m) - (n - 1)
    A = np.zeros((m, n))
    for i in range(n):
        valid = (i - lags >= 0) & (i - lags < n)
        A[valid, i] = b[i - lags[valid]]
    hv = cp.Variable(n)
    t = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n / 2.0, A @ hv <= t]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if hv.value is not None:
            return _repair(np.array(hv.value).flatten(), n)
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
            if hv.value is not None:
                return _repair(np.array(hv.value).flatten(), n)
        except Exception:
            pass
    return h.copy()

def _lp_phase(h, n, dx, max_iter, time_limit):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h_cur = h.copy()
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        h_new = _lp_step(h_cur, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h_cur)) < 1e-7:
            break
        h_cur = 0.75 * h_new + 0.25 * h_cur
        h_cur = _repair(h_cur, n)
    return best_h, best_c5

def _subgrad(h, n, dx, rng, time_limit, num_steps=2000):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h_cur = h.copy()
    alpha = 50.0
    step = 0.05
    for it in range(num_steps):
        if time.time() > time_limit:
            break
        _, g = _sm_jac(h_cur, alpha, dx)
        h_trial = h_cur - step * g
        h_trial = _repair(h_trial, n)
        c5_tri = _c5(h_trial, dx)
        if c5_tri < best_c5:
            best_c5 = c5_tri
            best_h = h_trial.copy()
            h_cur = h_trial
        else:
            step *= 0.7
        if it % 100 == 0:
            alpha = min(alpha * 1.5, 500.0)
            step = min(step * 1.3, 0.15)
    return best_h, best_c5

def _local_search(h, n, dx, rng, time_limit):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = np.array([0.002, 0.005, 0.01, 0.015, 0.025, 0.04, 0.06, 0.09, 0.13,
                      -0.002, -0.005, -0.01, -0.015, -0.025, -0.04, -0.06, -0.09, -0.13])
    stall = 0
    while time.time() < time_limit:
        improved = False
        ntrials = 50 + 40 * min(stall // 5, 5)
        for _ in range(ntrials):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                old_i, old_j = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = old_i, old_j
                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    return best, best_c5

def _block_search(h, n, dx, rng, time_limit):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    while time.time() < time_limit:
        improved = False
        ntrials = 30 + 20 * min(stall // 5, 5)
        for _ in range(ntrials):
            sz = int(rng.randint(2, 4))
            idxs = rng.choice(n, sz, replace=False)
            d = rng.choice([0.02, 0.04, 0.08, 0.12]) * rng.choice([1, -1])
            mags = np.abs(rng.dirichlet(np.ones(sz))) * 2 * abs(d)
            signs = rng.choice([1, -1], sz)
            cand = best.copy()
            for ii in range(sz):
                cand[idxs[ii]] += signs[ii] * mags[ii]
            if cand.min() < -1e-9 or cand.max() > 1.0 + 1e-9:
                continue
            if abs(cand.sum() - n / 2) > 1e-6:
                continue
            nc = _c5(cand, dx)
            if nc < best_c5 - 1e-13:
                best = cand
                best_c5 = nc
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break
    return best, best_c5
