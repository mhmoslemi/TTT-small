    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr_full(h)
    return float(np.max(c) * dx)

    """Compute gradients of all c_s w.r.t. h. Shape: (2n-1, n)"""
    G = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        s = idx - (n-1)

                G[idx, i] += (1.0 - h[i + s])

            for i in range(ss, n):
                G[idx, i] += (1.0 - h[i - ss])

    b_eq = np.array([target])

                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    G = _compute_G(h0, n)
    rhs = corr - G @ h0
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = rhs
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, t_ub)]

    c_obj[-1] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return _repair(res.x[:n], n)
    except Exception:
        pass
    return h0

    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

                    h[i] = hi + d_best
                    h[j] = hj - d_best
                    corr = _corr_full(h)
                    G = _compute_G(h, n)
                    any_improve = True

            improved_any = True
        else:

    return np.correlate(h, 1.0 - h, mode='full')

def _compute_G(h, n):
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        if s >= 0:
            for i in range(n - s):
                G[idx, i] += (1.0 - h[i + s])
                G[idx, i + s] += (-h[i])
        else:
            ss = -s
            for i in range(ss, n):
                G[idx, i] += (1.0 - h[i - ss])
                G[idx, i - ss] += (-h[i])
    return G

def _repair(h, n):

                diff -= a
