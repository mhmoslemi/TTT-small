import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for ki in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[ki])
                x[ki] += a
                diff -= a
            else:
                a = min(-diff, x[ki])
                x[ki] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    one = 1.0 - x
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - lag] += wk * one[lag:]
            g[lag:] -= wk * x[:n - lag]
        else:
            L = -lag
            g[L:] += wk * one[:n - L]
            g[:n - L] -= wk * x[L:]
    return g

def _fista(h0, alpha, target, iters, t0, budget):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (2.0 * alpha + 4.0 * n + 4.0)
    for it in range(iters):
        if it > 0 and (it % 50 == 0 or it < 5):
            if time.time() - t0 > budget - 25:
                break
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        if it % 20 == 0 and it > 0:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn > quad + 1e-7:
                step *= 0.6
                y = z - step * gz
                xn = _proj(y, target)
        else:
            if step < 1.0 / (2.0 * alpha + 4.0 * n):
                step = 1.0 / (2.0 * alpha + 4.0 * n)
        cn = _c5(xn, dx)
        if cn < bc:
            bc = cn
            bx = xn.copy()
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = xn + (t - 1.0) / tn * (xn - x)
        x = xn
        t = tn
    return bx, bc
