    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute diff to elements with most room
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    remaining = diff
    for idx in order:
        if remaining > 0:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
        else:
            sub = min(-remaining, h[idx])
            h[idx] -= sub
            remaining += sub
        if abs(remaining) < 1e-14:
            break
    return h

            break
        h_cur = best_h.copy()
        v = np.zeros(n)
        lr = 2.0 / alpha
        mu = 0.85
        for it in range(iters_per_alpha):
            if time.time() - t_start > t_end:
                break
            g = _smooth_grad(h_cur, alpha)
            v = mu * v + g
            y = h_cur - (1.0 + mu) * lr * v
            h_new = _project(y, n)
            c5_new = _c5(h_new, dx)
            c5_cur = _c5(h_cur, dx)
            if c5_new < c5_cur:
                h_cur = h_new
                lr = min(lr * 1.03, 10.0 / alpha)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
            else:
                lr *= 0.5
            if lr < 1e-7 / alpha:
                break
    return best_h, best_c5

def _binary_phase(h, n, dx, rng, t_start, t_end):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h_cur = h.copy()
    c5_cur = best_c5
    iters = 0
    stuck = 0
    while time.time() - t_start < t_end and stuck < 2000:
        iters += 1
        ones = np.where(h_cur > 0.3)[0]
        zeros = np.where(h_cur < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_cur[i], h_cur[j] = 0.0, 1.0
        c5_new = _c5(h_cur, dx)
        if c5_new <= c5_cur:
            c5_cur = c5_new
            stuck = 0
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = 1.0, 0.0
            stuck += 1
    return best_h, best_c5

def _fine_local(h, n, dx, rng, t_start, t_end):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h_cur = h.copy()
    c5_cur = best_c5
    deltas = np.array([0.0005, 0.001, 0.002, 0.005, 0.01, -0.0005, -0.001, -0.002, -0.005, -0.01])
    while time.time() - t_start < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        ni = h_cur[i] + d
        nj = h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5_new = _c5(h_cur, dx)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return best_h, best_c5

def _sa_phase(h, n, dx, rng, t_start, t_end):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    h_cur = h.copy()
    c5_cur = best_c5
    T = 0.002
    while time.time() - t_start < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        scale = T * 5.0 + 0.002
        d = rng.uniform(-scale, scale)
        ni = h_cur[i] + d
        nj = h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            T *= 0.999
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5_new = _c5(h_cur, dx)
        if c5_new < c5_cur or (T > 1e-12 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
        T = max(T * 0.999, 1e-13)
    return best_h, best_c5

def _worst_lag_grad(h, alpha):
    c = _corr(h)
    n = len(h)
    m = np.max(c)
    sh = alpha * (c - m)
    sh = np.clip(sh, -500, 500)
    p = np.exp(sh)
    p /= p.sum()
    top_k = int(max(3, np.sum(p > 0.05)))
    top_idx = np.argsort(p)[-top_k:]
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in top_idx:
        mk = lags[k]
        w = p[k]
        jlo = max(0, mk)
        jhi = min(n, n + mk)
        if jlo < jhi:
            idx1 = np.arange(jlo, jhi) - mk
            grad[jlo:jhi] += w * (1.0 - h[idx1])
        jlo2 = max(0, -mk)
        jhi2 = min(n, n - mk)
        if jlo2 < jhi2:
            idx2 = np.arange(jlo2, jhi2) + mk
            grad[jlo2:jhi2] -= w * h[idx2]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = _repair(h0, n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Nesterov projected gradient
    alphas = [20, 50, 150, 400, 1000, 3000, 8000]
    best_h, best_c5 = _nesterov_phase(best_h, n, dx, t0, 250.0, alphas, 600)

    # Phase 2: Binary local search
    if time.time() - t0 < budget_s - 300:
        best_h, best_c5 = _binary_phase(best_h, n, dx, rng, t0, budget_s - 280.0)

    # Phase 3: Nesterov again from potentially binary solution
    if time.time() - t0 < budget_s - 250:
        alphas2 = [100, 300, 1000, 3000]
        best_h, best_c5 = _nesterov_phase(best_h, n, dx, t0, budget_s - 240.0, alphas2, 400)

    # Phase 4: Simulated annealing
    if time.time() - t0 < budget_s - 180:
        best_h, best_c5 = _sa_phase(best_h, n, dx, rng, t0, budget_s - 170.0)

    # Phase 5: Fine local search
    if time.time() - t0 < budget_s - 80:
        best_h, best_c5 = _fine_local(best_h, n, dx, rng, t0, budget_s - 75.0)

    # Phase 6: Multi-restart Nesterov
    if time.time() - t0 < budget_s - 40:
        for r in range(6):
            if time.time() - t0 > budget_s - 40:
                break
            h_pert = best_h + 0.015 * rng.randn(n)
            h_pert = _project(h_pert, n)
            h_pert = _repair(h_pert, n)
            hh, cc = _nesterov_phase(h_pert, n, dx, t0, budget_s - 40.0, [500, 2000], 200)
            if cc < best_c5:
                best_c5, best_h = cc, hh

    # Phase 7: Final ultra-fine polish
    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = _fine_local(best_h, n, dx, rng, t0, budget_s - 3.0)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
