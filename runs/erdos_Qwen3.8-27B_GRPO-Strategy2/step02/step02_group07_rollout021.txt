    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 + offset:hi2 + offset]
    return g

def _lp_step(h, n, dx, tol=1e-3, max_lags=6):
    c = _corr(h) * dx
    c5 = c.max()
    active = np.where(c >= c5 - tol)[0]
    if len(active) > max_lags:
        active = np.argsort(c)[::-1][:max_lags]
    if len(active) == 0:
        return h.copy(), c5
    grads = np.zeros((len(active), n))
    for row, k_idx in enumerate(active):
        offset = int(k_idx) - (n - 1)
        grads[row, :] = _grad_lag(h, offset) * dx
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for i in range(len(active)):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = c[active[i]] - grads[i] @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.status == 0:
            h_new = _repair(res.x[:n], n)
            return h_new, _c5(h_new, dx)
    except Exception:
        pass
    return h.copy(), c5

def _lp_loop(h, n, dx, t_start, t_limit, max_iters=500, base_tol=1e-3):
    best_c5 = _c5(h, dx)
    st = 0
    for it in range(max_iters):
        if time.time() - t_start > t_limit:
            break
        tol = base_tol * (1 + 0.5 * it / max(1, max_iters))
        ml = 4 + (it % 4)
        h_new, c_new = _lp_step(h, n, dx, tol=tol, max_lags=ml)
        if c_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c_new
            st = 0
        else:
            st += 1
            if st > 30:
                break
    return h, _c5(h, dx)

def _slsqp_smooth(h, n, dx, alpha, maxiter=600):
    def obj(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400.0, 400.0)
        return m + np.log(np.sum(np.exp(s))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_refine(h, n, dx, n_grid, max_passes, t_start, t_limit):
    hi_c5 = _c5(h, dx)
    for _p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        any_impr = False
        for i in range(n):
            for j in range(i + 1, n):
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi = min(1.0, hsum)
                if hi - lo < 1e-10:
                    continue
                oi, oj = h[i], h[j]
                bt, bc = oi, hi_c5
                for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                    h[i] = t
                    h[j] = hsum - t
                    cv = _c5(h, dx)
                    if cv < bc:
                        bc = cv
                        bt = t
                if bc < hi_c5 - 1e-13:
                    h[i], h[j] = bt, hsum - bt
                    hi_c5 = bc
                    any_impr = True
                else:
                    h[i], h[j] = oi, oj
        if not any_impr:
            break
    return h, _c5(h, dx)

def _structured_perturb(h, n, rng):
    k = rng.randint(1, 8)
    h2 = h.copy()
    mask = rng.random(n) < 0.3
    h2[mask] += 0.1 * rng.randn(mask.sum())
    for _ in range(k):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            swap = 0.05 * rng.randn()
            h2[i] += swap
            h2[j] -= swap
    return _repair(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    def elapsed():
        return time.time() - t0
    def remaining(limit_margin=5):
        return budget_s - limit_margin - elapsed()

    # Phase 1: LP cutting-plane
    if remaining(250) > 0:
        h, c = _lp_loop(h, n, dx, t0, budget_s - 250, max_iters=400, base_tol=1e-3)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: SLSQP with increasing alpha
    for alpha in [80, 200, 500, 1000, 2000]:
        if remaining(200) < 10:
            break
        h_s = _slsqp_smooth(h, n, dx, alpha, 500)
        c = _c5(h_s, dx)
        if c < best_c5:
            best_c5, best_h = c, h_s.copy()
            h = h_s

    # Phase 3: LP again
    if remaining(180) > 0:
        h, c = _lp_loop(h, n, dx, t0, budget_s - 180, max_iters=300, base_tol=5e-4)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 4: Moderate CD
    if remaining(120) > 10:
        h, c = _cd_refine(h, n, dx, 15, 150, t0, budget_s - 130)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 5: Multi-restart
    sigmas = [0.06, 0.1, 0.04, 0.15, 0.08, 0.05, 0.2, 0.12]
    for si in range(len(sigmas)):
        if remaining(60) < 30:
            break
        h_p = _repair(best_h + sigmas[si] * rng.randn(n), n)
        for _rp in range(15):
            if remaining(60) < 20:
                break
            h_p, c_p = _lp_step(h_p, n, dx, tol=8e-4, max_lags=5)
        c_p = _c5(h_p, dx)
        if c_p < best_c5:
            best_c5, best_h = c_p, h_p.copy()
        h_p2 = _structured_perturb(best_h, n, rng)
        for _rp in range(10):
            if remaining(60) < 20:
                break
            h_p2, c_p2 = _lp_step(h_p2, n, dx, tol=8e-4, max_lags=5)
        c_p2 = _c5(h_p2, dx)
        if c_p2 < best_c5:
            best_c5, best_h = c_p2, h_p2.copy()

    # Phase 6: SLSQP polish
    if remaining(80) > 20:
        h_s = _slsqp_smooth(best_h, n, dx, 3000, 600)
        c = _c5(h_s, dx)
        if c < best_c5:
            best_c5, best_h = c, h_s.copy()

    # Phase 7: Fine CD
    if remaining(40) > 15:
        h, c = _cd_refine(best_h, n, dx, 21, 400, t0, budget_s - 45)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 8: Final LP polish
    if remaining(10) > 3:
        for _ in range(300):
            if remaining(5) < 1:
                break
            h_new, c_new = _lp_step(best_h, n, dx, tol=3e-4, max_lags=4)
            if c_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c_new
            else:
                break

    # Phase 9: Ultra-fine CD on last few pairs
    if remaining(15) > 5:
        c = _corr(best_h)
        c5_raw = c.max()
        top = np.argsort(c)[-8:]
        h, c = _cd_refine(best_h, n, dx, 25, 100, t0, budget_s - 10)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
