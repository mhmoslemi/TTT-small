def _subgradient(h, n):
    """Compute subgradient of max_k R[k] where R = correlate(h, 1-h, 'full')."""
    R = np.correlate(h, 1.0 - h, mode='full')
    Rmax = np.max(R)
    active = R >= Rmax - 1e-12
    nact = np.sum(active)
    if nact == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        for k in range(2*n-1):
            if not active[k]:
                continue
            # dR[k]/dh[m] = (1-h[m-k+(n-1)]) - h[m+k-(n-1)]
            idx1 = m - k + (n-1)
            idx2 = m + k - (n-1)
            term1 = 1.0 - (h[idx1] if 0 <= idx1 < n else 0.0)
            term2 = h[idx2] if 0 <= idx2 < n else 0.0
            g += (term1 - term2) / nact
        grad[m] = g
    return grad

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx]); h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx]); h[idx] -= sub; diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _coord_sweep(h, n, lam=0.0):
    best = h.copy()
    for j in range(n):
        h0 = best.copy(); h0[j] = 0.0
        c0 = np.correlate(h0, 1.0 - h0, mode='full')
        h1 = best.copy(); h1[j] = 1.0
        c1 = np.correlate(h1, 1.0 - h1, mode='full')
        slope = c1 - c0
        ts = np.linspace(0.0, 1.0, 256)
        vals = np.max(c0[None, :] + slope[None, :] * ts[:, None], axis=1) + lam * ts
        idx = int(np.argmin(vals))
        t_best = ts[idx]
        if 0 < idx < 255:
            lo, hi = ts[idx - 1], ts[idx + 1]
            ts2 = np.linspace(lo, hi, 256)
            vals2 = np.max(c0[None, :] + slope[None, :] * ts2[:, None], axis=1) + lam * ts2
            t_best = ts2[int(np.argmin(vals2))]
        best[j] = t_best
    return best

def _lp_refine(h, n, dx):
    """Linearize max correlation and solve LP."""
    R = np.correlate(h, 1.0 - h, mode='full')
    # Linearize: R[k] ≈ R_cur[k] + sum_j dR[k]/dh[j] * (h[j] - h_cur[j])
    # = sum_j h[j] * (1 - h_cur[j-k+(n-1)])  [linearized in h]
    # c[k] = sum_j h[j] * w_k[j] where w_k[j] = 1 - h_cur[j-k+(n-1)] if in range
    m = 2 * n - 1
    A_ineq = np.zeros((m, n + 1))
    b_ineq = np.zeros(m)
    for k in range(m):
        for j in range(n):
            idx = j - k + (n - 1)
            w = 1.0 - h[idx] if 0 <= idx < n else 1.0
            A_ineq[k, j] = w
            if w < 0:
                A_ineq[k, j] = 0.0  # should be in [0,1] since h in [0,1]
        A_ineq[k, n] = -1.0  # -T
    b_ineq[:] = 0.0
    # sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize T
    bounds = [(0, 1)] * n + [(0, 2.0)]
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        new_h = np.clip(res.x[:n], 0, 1)
        return _repair(new_h, n)
    return h

def _subgrad_h(h, n):
    """Analytical subgradient of max_k R[k]."""
    R = np.correlate(h, 1.0 - h, mode='full')
    Rmax = np.max(R)
    active = R >= Rmax - 1e-10
    nact = max(1, int(active.sum()))
    grad = np.zeros(n)
    for m_idx in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            i1 = m_idx - k + (n - 1)
            i2 = m_idx + k - (n - 1)
            t1 = 1.0 - (h[i1] if 0 <= i1 < n else 0.0)
            t2 = h[i2] if 0 <= i2 < n else 0.0
            g += (t1 - t2)
        grad[m_idx] = g / nact
    return grad

def _subgrad_descent(h, n, dx, deadline, maxiter=5000):
    best = h.copy()
    best_c5 = _c5(best, dx)
    lr = 0.02
    for it in range(maxiter):
        if time.time() > deadline:
            break
        g = _subgrad_h(best, n)
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        # Projected step
        cand = best - lr * g
        # Also try to maintain sum
        s = cand.sum()
        target = n / 2.0
        if abs(s - target) > 1e-10:
            cand += (target - s) / n
        cand = np.clip(cand, 0, 1)
        new_c5 = _c5(cand, dx)
        if new_c5 < best_c5 - 1e-14:
            best = cand
            best_c5 = new_c5
            if it % 100 == 0:
                lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.95
            if lr < 1e-5:
                break
    return best, best_c5

def _coord_descend(h, n, dx, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    lam = 0.0
    for sweep in range(5000):
        if time.time() > deadline:
            break
        best = _coord_sweep(best, n, lam)
        s = best.sum()
        target = n / 2.0
        err = s - target
        if err > 0.01:

        c5 = _c5(best, dx)

def _slsqq(h, n, dx, alpha, maxiter=200):
    def _obj(x):
        m = np.max(np.correlate(x, 1.0 - x, mode='full'))
        s = np.clip(alpha * (np.correlate(x, 1.0 - x, mode='full') - m), -400, 400)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    def _con(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(_obj, h, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _local_kbest(h, n, dx, rng, deadline, max_iter=3000):
    """Local search: pick coordinate with largest gradient, adjust it."""
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = np.array([0.01, 0.02, 0.05, 0.1, 0.2, -0.01, -0.02, -0.05, -0.1, -0.2,
                      0.005, -0.005, 0.03, -0.03, 0.07, -0.07])
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _subgrad_h(best, n)
        # Pick coordinate most aligned with descent
        scores = np.abs(g)
        j = int(np.argmax(scores))
        improved = False
        for d in steps:
            ni = best[j] + d
            if ni < -1e-9 or ni > 1.0 + 1e-9:
                continue
            old = best[j]
            best[j] = ni
            # More than j: also adjust another to keep sum
            k = int(rng.randint(n))
            if k == j:
                k = (j + 1) % n
            nj = best[k] - (ni - old)
            if nj < -1e-9 or nj > 1.0 + 1e-9:
                best[j] = old
                continue
            ok_val = best[k]
            best[k] = nj
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-13:
                best_c5 = nc
                improved = True
            else:
                best[j] = old
                best[k] = ok_val
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best, best_c5

    return (best_h, best_c5, n)
