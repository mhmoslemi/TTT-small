def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)


def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def subgrad_c5(h, n):
    corr = compute_corr(h)
    m = int(np.argmax(corr))
    lag = m - (n - 1)
    a = abs(lag)
    g = np.zeros(n)

        i1 = j - a
        if 0 <= i1 <= n - 1 - a:
            g[j] += (1.0 - h[i1])
        if 0 <= j <= n - 1 - a:
            g[j] -= h[j + a]
