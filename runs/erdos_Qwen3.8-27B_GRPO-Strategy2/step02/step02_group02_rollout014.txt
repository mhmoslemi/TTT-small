def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a; diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _tobj(x, a, K):
    c = np.correlate(x, 1.0 - x, mode='full')
    top = np.sort(c)[::-1][:K]
    mx = top[0]
    s = np.clip(a * (top - mx), -500.0, 500.0)
    return mx + float(np.log(np.exp(s).sum())) / a

def _tgrad(x, a, K):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    ti = np.argsort(c)[::-1][:K]
    ct = c[ti]
    mx = ct[0]
    s = np.clip(a * (ct - mx), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for ki in range(K):
        wk = w[ki]
        if wk < 1e-14:
            continue
        lag = ti[ki] - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

                break

def _slsqp_obj(x, a, K):
    return _tobj(x, a, K)

def _slsqp_grad(x, a, K):
    return _tgrad(x, a, K)

def _slsqp_con(x, tgt):
    return float(x.sum()) - tgt

def _slsqp_conjac(x, n):
    return np.ones(n)

def _lbfgs_obj(x, a, K, pen, tgt, n):
    return _tobj(np.clip(x, 0, 1), a, K) + pen * (float(x.sum()) - tgt) ** 2

def _lbfgs_grad(x, a, K, pen, tgt, n):
    xc = np.clip(x, 0, 1)
    g = _tgrad(xc, a, K)
    g += 2.0 * pen * (float(xc.sum()) - tgt) * np.ones(n)
    return np.where((xc > 1e-8) & (xc < 1.0 - 1e-8), g, 0.0)
