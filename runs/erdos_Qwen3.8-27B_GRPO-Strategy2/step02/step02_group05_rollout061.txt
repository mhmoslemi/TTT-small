When h[p] changes, only correlations m in [p, p+n-1] are affected. That's at most n correlations. For each, the new value is:
- If m ≠ 2p: C[m] + (x - old_x)*(1 - 2*h[m-p])
- If m = 2p: C[m] + x*(1-x) - old_x*(1-old_x)

And correlations outside this range are unchanged.

So the new max is max(max(C[non-affected]), max(C_new[affected])).

Precompute C_non_affected_max = max of C outside [p, p+n-1] (and outside {2p} if we handle it separately).

For the affected range, for each x:
- Linear terms: C[m] + (x-old_x)*(1-2*h[m-p]) for m in range, m≠2p
- Quadratic term: C[2p] + x*(1-x) - old_x*(1-old_x) (if 2p in range)

def _coord_descent_fast(h, n, dx, n_sweeps=500, grid_size=200, tol=1e-14):

    x_grid = np.linspace(0.0, 1.0, grid_size)

        C = np.correlate(h, 1.0 - h, mode='full')  # (2n-1,)
        c5 = np.max(C) * dx
        max_change = 0.0
        for p in range(n):
            old_x = h[p]
            m_lo = p
            m_hi = min(2*n-2, p+n-1)
            
            # Indices in affected range
            m_idx = np.arange(m_lo, m_hi+1)
            j_idx = m_idx - p
            
            # Check if 2p is in affected range
            two_p = 2*p
            has_quad = (two_p >= m_lo) and (two_p <= m_hi)
            
            # Build linear coefficients
            lin_coeff = 1.0 - 2.0 * h[j_idx]  # (x - old_x) * lin_coeff
            lin_const = C[m_idx] - old_x * lin_coeff  # C[m] - old_x*(1-2*h[m-p])
            # new_C[m] = lin_const[m] + x * lin_coeff[m]
            
            if has_quad:
                # Separately handle m=2p
                quad_k = np.where(m_idx == two_p)[0][0]
                quad_const = C[two_p] - old_x*(1-old_x)
                # new_C[2p] = quad_const + x*(1-x) = quad_const + x - x²
            
            # Max of non-affected C
            affected_mask = np.zeros(2*n-1, dtype=bool)
            affected_mask[m_idx] = True
            non_affected_max = np.max(C[~affected_mask]) if np.any(~affected_mask) else -np.inf
            
            # Evaluate on grid
            # Linear part: shape (grid_size, len(m_idx))
            # x_grid[:,None] * lin_coeff[None,:] + lin_const[None,:]
            lin_vals = x_grid[:,None] * lin_coeff[None,:] + lin_const[None,:]
            lin_max = np.max(lin_vals, axis=1)
            
            if has_quad:
                quad_vals = quad_const + x_grid - x_grid**2
                obj = np.maximum(np.maximum(lin_max, quad_vals), non_affected_max)
            else:
                obj = np.maximum(lin_max, non_affected_max)
            obj *= dx
            
            best_i = np.argmin(obj)
            best_x = x_grid[best_i]
            
            if best_x != old_x:
                h[p] = best_x
                max_change = max(max_change, abs(best_x - old_x))
        
        if max_change < tol:
            break
    
    return h

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:

                diff -= add
            else:
