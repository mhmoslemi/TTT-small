import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _smooth_obj(h, alpha):
    c = _corr_full(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h = h * (tgt / max(s, 1e-15))
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-10:
            break
        diff = tgt - s
        if abs(diff) < 1e-12:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _slsqp(h, n, alpha, maxiter):
    tgt = n / 2.0
    def obj(x):
        return _smooth_obj(x, alpha)
    def con(x):
        return np.sum(x) - tgt
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

    best = _c5_val(h, dx)
    corr = _corr_full(h)
    top_idx = np.argsort(corr)[-topk:]

                   -0.003, -0.007, -0.012, -0.02, -0.035, -0.06]

    while time.time() < budget_end:
        step_scale *= 0.9985
        if step_scale < 0.05:
            step_scale = 0.05

        improved = False
        n_attempts = 10

        for _ in range(n_attempts):
            if rng.rand() < 0.5:
                k_idx = top_idx[rng.randint(topk)]
                shift = abs(k_idx - n_half)
                if shift == 0 or shift >= n:
                    i = rng.randint(n)
                    j = (i + 1 + rng.randint(n - 1)) % n
                else:
                    i = rng.randint(max(1, n - shift))
                    j = i + shift
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue

            hi, hj = h[i], h[j]
            for d in deltas:
                ni, nj = hi + d, hj - d
