    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:

    return h

    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    # Top-k weights
    idx = np.argsort(x)[-k:]
    wm = np.zeros(len(x))
    wm[idx] = 1.0 / k
    # Blend
    return np.sum(w * x)  # soft max weighted average

    return _smooth_max(_corr(h), alpha) * dx

def _obj_topk(h, k, alpha, dx):

    # Focus on top-k correlations
    top_k_idx = np.argsort(c)[-k:]
    # Weighted sum of top-k with large weights
    w = np.zeros(len(c))
    w[top_k_idx] = alpha
    # Add small weight to all
    w += 1.0
    return (np.sum(w * c) / np.sum(w)) * dx

                       bounds=bounds, constraints=cons,

        return h.copy()

def _grad_topk(h, top_idx):
    """Compute gradient of sum of corr[k] for k in top_idx."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in top_idx:
        # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # d/dh[m] = (1-h[m+k-(n-1)]) - h[m-k+(n-1)]
        # First term: i=m, need j=m+k-(n-1) valid
        # Second term: i+k-(n-1)=m, i.e., i=m-k+(n-1), need i valid
        # corr[k] = sum over valid i of h[i] * (1-h[i+k-n+1])
        # Let offset = k - (n-1)
        # i ranges so that both i and i+offset are in [0, n-1]
        offset = k - (n-1)
        for m in range(n):
            j1 = m + offset  # index in (1-h) factor
            j2 = m - offset  # index in h factor (where h[m] is the second factor)
            if 0 <= j1 < n:
                grad[m] += one_minus_h[j1]
            if 0 <= j2 < n:
                grad[m] -= h[j2]
    return grad

def _grad_topk(h, top_idx):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in top_idx:
        offset = k - (n - 1)
        # First term: for each m, if m+offset in [0,n), add one_minus_h[m+offset]
        # Second term: for each m, if m-offset in [0,n), subtract h[m-offset]
        # Vectorized:
        if offset >= 0:
            # m + offset < n => m < n - offset
            valid1 = np.arange(n - offset)
            grad[valid1] += one_minus_h[valid1 + offset]
            # m - offset >= 0 => m >= offset
            valid2 = np.arange(offset, n)
            grad[valid2] -= h[valid2 - offset]
        else:

            valid1 = np.arange(-offset, n)
            grad[valid1] += one_minus_h[valid1 + offset]
            # m - offset < n => m < n + offset
            valid2 = np.arange(0, n + offset)
            grad[valid2] -= h[valid2 - offset]

import time

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_smooth(h, alpha, dx):
    return _smooth_max(_corr(h), alpha) * dx

def _grad_corr_k(h, k):
    """Analytic gradient of corr[k] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    offset = k - (n - 1)
    if offset >= 0:
        v1 = np.arange(n - offset)
        grad[v1] += one_minus_h[v1 + offset]
        v2 = np.arange(offset, n)
        grad[v2] -= h[v2 - offset]
    else:
        v1 = np.arange(-offset, n)
        grad[v1] += one_minus_h[v1 + offset]
        v2 = np.arange(0, n + offset)
        grad[v2] -= h[v2 - offset]
    return grad

def _grad_topk(h, top_idx):
    """Analytic gradient of sum of top-k correlations."""
    n = len(h)
    grad = np.zeros(n)
    for k in top_idx:
        grad += _grad_corr_k(h, k)
    return grad

    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}]
    try:

        return _repair(res.x, n)

    best = h.copy()
    best_c5 = _c5(best, dx)

    s = h.sum()

        else:

                h[idx] += add
                diff -= add

    lo, hi = v.min() - 1, v.max() + 1

        s = np.clip(v - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _project(h, grad, lr, n):
    target = n / 2.0
    v = h - lr * grad
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_smooth(h, alpha, dx):
    return _smooth_max(_corr(h), alpha) * dx

def _con_sum(x, target):
    return np.sum(x) - target

def _grad_corr_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    omd = 1.0 - h
    offset = k - (n - 1)
    if offset >= 0:
        v1 = np.arange(n - offset)
        grad[v1] += omd[v1 + offset]
        v2 = np.arange(offset, n)
        grad[v2] -= h[v2 - offset]
    else:
        v1 = np.arange(-offset, n)
        grad[v1] += omd[v1 + offset]
        v2 = np.arange(0, n + offset)
        grad[v2] -= h[v2 - offset]
    return grad

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_sum, 'args': (target,)}]
    try:
        res = minimize(_obj_smooth, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _proj_grad(h, n, dx, max_iter, deadline, top_k):
    h = h.copy()
    lr = 0.04
    best_c5 = _c5(h, dx)
    patience = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += _grad_corr_k(h, k)
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
        h_new = _project(h, grad, lr, n)
        c5n = _c5(h_new, dx)
        if c5n < best_c5 - 1e-10:
            best_c5 = c5n
            h = h_new
            lr = min(lr * 1.05, 0.4)
            patience = 0
        else:
            lr *= 0.85
            patience += 1
            if patience > 60:
                break
    return h, best_c5

def _local_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = np.array([0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1])
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        improved = False
        for _ in range(60):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    improved = True
                    break
                best[i], best[j] = ni - d, nj + d
            if improved:
                break
        stall = 0 if improved else stall + 1
    return best, best_c5

def _block_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    while time.time() < deadline and stall < 25:
        improved = False
        for _ in range(50):
            w = int(rng.randint(2, min(8, n // 3)))
            a = int(rng.randint(0, n - w))
            b = int(rng.randint(0, n - w))
            if abs(a - b) < w:
                continue
            d = rng.uniform(0.03, 0.15) * (1.0 if rng.rand() < 0.5 else -1.0)
            ba = best[a:a+w].copy()
            bb = best[b:b+w].copy()
            na = np.clip(ba + d, 0, 1)
            nb = np.clip(bb - d, 0, 1)
            if abs((na.sum() - ba.sum()) + (nb.sum() - bb.sum())) > 0.06:
                continue
            best[a:a+w] = na
            best[b:b+w] = nb
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-10:
                best_c5 = nc
                improved = True
                break
            best[a:a+w] = ba
            best[b:b+w] = bb
        stall = 0 if improved else stall + 1
    return best, best_c5

def _interp(h_src, n_dst):
    n_src = len(h_src)
    xs = np.linspace(0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0, 2.0, n_dst, endpoint=False)
    return _repair(np.interp(xd, xs, np.append(h_src, h_src[0])), n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # Phase 1: SLSQP with increasing alpha
    for a in [30, 60, 150, 300, 600, 1200]:

            break
        hc = _slsqp(best_h, n, dx, a, 250, 1e-13)

        if c5 < best_c5:

    # Phase 2: Analytic projected gradient (top-5 shifts)
    if time.time() < deadline - 140:

    # Phase 3: Pairwise local search
    if time.time() < deadline - 100:

    # Phase 4: Block search
    if time.time() < deadline - 70:
        best_h, best_c5 = _block_search(best_h, n, dx, rng, deadline - 70)

    # Phase 5: SLSQP + local restarts from perturbed best
    sigmas = [0.2, 0.35, 0.5, 0.15, 0.3, 0.45, 0.6, 0.25, 0.4, 0.1]
    it = 0
    while time.time() < deadline - 55:
        sigma = sigmas[it % 10]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 250, 150, 1e-12)

        if c5 < best_c5:

                best_c5, best_h = cl, hl

            if time.time() > deadline - 80:

            dxt = 2.0 / nt

            ct = _c5(ht, dxt)

        if c5 < best_c5:

        h9, c9 = _proj_grad(best_h, n, dx, 100, deadline - 3, 9)
        if c9 < best_c5:

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
