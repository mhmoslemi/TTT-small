```python
def _min_convex_pl(c, gd, d_min, d_max, iters=60):
    """Minimize max_k (c[k] + d*gd[k]) for d in [d_min, d_max] using ternary search."""
    a, b = d_min, d_max
    for _ in range(iters):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        f1 = np.max(c + m1 * gd)
        f2 = np.max(c + m2 * gd)
        if f1 < f2:
            b = m2
        else:
            a = m1
    d_opt = (a + b) / 2
    return d_opt, np.max(c + d_opt * gd)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _obj_topk(x, k):
    c = _corr(x)
    idx = np.argsort(c)[::-1][:k]
    return float(np.mean(c[idx]))

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub; diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _ternary_min(c, gd, d_min, d_max):
    a, b = d_min, d_max
    if b - a < 1e-15:
        return d_min
    for _ in range(55):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c + m1 * gd)
        f2 = np.max(c + m2 * gd)
        if f1 < f2:
            b = m2
        else:
            a = m1
    d_opt = (a + b) / 2.0
    return d_opt

def _exact_sweep(h, n, dx, corr, max_passes=5):
    best_c5 = float(np.max(corr)) * dx
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-14:
                    continue
                gd = _g_vec(h, i, n) - _g_vec(h, j, n)
                d_best = _ternary_min(corr, gd, d_min, d_max)
                new_max = float(np.max(corr + d_best * gd))
                if new_max < best_c5 / dx - 1e-14:
                    h[i] = hi + d_best
                    h[j] = hj - d_best
                    corr = corr + d_best * gd
                    best_c5 = new_max * dx
                    improved = True
        if not improved:
            break
    return h, corr, best_c5

def _slsqp_smax(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _slsqp_topk(h, n, dx, target, bounds, k, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_topk, h, args=(k,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h
```
