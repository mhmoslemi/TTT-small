        try:
            res = minimize(
                lambda x, a=alpha, p=pen: _smooth_obj_pen(x, a, target, p),
                best_h, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7}
            )
            hc = _repair(res.x, n)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _slsqp_opt(h, n, dx, alpha, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    try:
        res = minimize(
            lambda x, a=alpha: _smooth_obj(x, a), h,
            method='SLSQP', bounds=bounds, constraints=[eq_con],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

def _struct_perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        block = rng.randint(4, 12)
        i = rng.randint(0, n - block)
        seg = hp[i:i+block]
        hp[i:i+block] = np.roll(seg, rng.randint(1, block))
    elif kind == 1:
        f = rng.randint(1, 5)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.15)
        hp = hp + amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        f1, f2 = rng.randint(1, 4), rng.randint(1, 4)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        amp = rng.uniform(0.04, 0.12)
        t = np.arange(n) / n
        hp = hp + amp * np.sin(2*np.pi*f1*t + p1) * np.sin(2*np.pi*f2*t + p2)
    elif kind == 3:
        j = rng.randint(n // 4, 3 * n // 4)
        hp = np.concatenate([h[j:], h[:j]])
    else:
        hp = h + rng.uniform(-0.06, 0.06, n)
    return _repair(hp, n)

def _fine_local(h, n, dx, rng, deadline, n_tries=80000):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
    per_scale = max(500, n_tries // len(scales))
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            h[i], h[j] = ni, nj
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and scale < 0.003:
            break
    return h, best_c5

def _block_local(h, n, dx, rng, deadline, n_iters=8000):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    for bs in [2, 3, 2]:
        for _ in range(n_iters // 3):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if h[i+di] + eps < -1e-10 or h[i+di] + eps > 1.0 + 1e-10:
                    ok = False
                if h[j+di] - eps < -1e-10 or h[j+di] - eps > 1.0 + 1e-10:
                    ok = False
            if not ok:
                continue
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
            else:
                h[i:i+bs] -= eps
                h[j:j+bs] += eps
    return h, best_c5

def _anneal(h, n, dx, rng, deadline):
    best_c5 = _compute_c5(h, dx)
    h_cur = h.copy()
    c5_cur = best_c5
    h_best = h_cur.copy()
    T = 0.003
    T_min = 0.00005
    cool = 0.9997
    while time.time() < deadline and T > T_min:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.008 * (2.0 * rng.random() - 1.0)
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _compute_c5(h_cur, dx)
        if c5n < c5_cur or rng.random() < np.exp(-(c5n - c5_cur) / T):
            c5_cur = c5n
            if c5n < best_c5:
                best_c5 = c5n
                h_best = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
        T *= cool
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Phase 1: Fast L-BFGS-B warm-up (50s)
    tEnd = t_start + 50
    for alpha in [50, 200, 600, 1500]:
        if time.time() >= tEnd or remaining() < 550:
            break
        h_t, c5_t = _lbfgs_opt(best_h, n, dx, alpha, maxiter=300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 2: SLSQP precision refinement (180s)
    tEnd = t_start + 230
    for alpha in [300, 800, 2000, 5000, 10000, 20000]:
        if time.time() >= tEnd or remaining() < 400:
            break
        h_t, c5_t = _slsqp_opt(best_h, n, dx, alpha, maxiter=600)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 3: Structured multi-restart (200s)
    tEnd = t_start + 430
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() >= tEnd or remaining() < 250:
            break
        start = _struct_perturb(best_h, n, rng, r % 5)
        h_t, c5_t = _lbfgs_opt(start, n, dx, 500, maxiter=200)
        h_t, c5_t = _slsqp_opt(h_t, n, dx, 3000, maxiter=300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 4: Block local search (120s)
    if remaining() > 200:
        tEnd = time.time() + min(120, remaining() - 150)
        h_t, c5_t = _block_local(best_h, n, dx, rng, tEnd, n_iters=6000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 5: Simulated annealing (100s)
    if remaining() > 150:
        tEnd = time.time() + min(100, remaining() - 100)
        h_t, c5_t = _anneal(best_h, n, dx, rng, tEnd)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 6: SLSQP final polish (100s)
    if remaining() > 100:
        tEnd = time.time() + min(80, remaining() - 60)
        for alpha in [5000, 15000, 40000, 100000]:
            if time.time() >= tEnd:
                break
            h_t, c5_t = _slsqp_opt(best_h, n, dx, alpha, maxiter=400)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 7: Intensive fine local search (80s)
    if remaining() > 60:
        tEnd = time.time() + min(80, remaining() - 30)
        h_t, c5_t = _fine_local(best_h, n, dx, rng, tEnd, n_tries=50000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 8: Ultra-fine final polish
    if remaining() > 30:
        h_t, c5_t = _slsqp_opt(best_h, n, dx, 50000, maxiter=500)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
    if remaining() > 10:
        tEnd = time.time() + max(2, min(10, remaining() - 2))
        h_t, c5_t = _fine_local(best_h, n, dx, rng, tEnd, n_tries=15000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
