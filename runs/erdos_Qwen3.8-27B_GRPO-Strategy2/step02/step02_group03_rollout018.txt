    for rnd in range(max_rounds):
        if time.time() > budget_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[::-1][:4]
        improved = False
        for tk in top_k:
            sh = int(tk - (n - 1))
            lo = max(0, -sh)
            hi = min(n - 1, n - 1 - sh)
            step = max(1, (hi - lo + 1) // 12)
            for i in range(lo, hi + 1, step):
                j = i + sh
                if j < 0 or j >= n or i == j:
                    continue
                for d in deltas:
                    ni = h[i] + d
                    nj = h[j] - d
                    if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                        continue
                    h2 = h.copy()
                    h2[i] = ni
                    h2[j] = nj
                    h2 = _proj(h2, n)
                    c52 = _c5(h2, n)
                    if c52 < best - 1e-12:
                        h = h2
                        best = c52
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and rnd > 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                d = rng.choice(deltas)
                if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                    h2 = h.copy()
                    h2[i] += d
                    h2[j] -= d
                    h2 = _proj(h2, n)
                    c52 = _c5(h2, n)
                    if c52 < best + 0.0003:
                        h = h2
                        if c52 < best:
                            best = c52
    return h, best

def _make_perturb(h, n, rng):
    h2 = h.copy()
    kind = rng.randint(4)
    if kind == 0:
        freq = rng.randint(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.15)
        h2 += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif kind == 1:
        a = rng.randint(0, n - 5)
        b = rng.randint(a + 3, n)
        h2[a:b] = 1.0 - h2[a:b]
    elif kind == 2:
        perm = rng.permutation(n)
        mask = rng.rand(n) < 0.3
        h2 = np.where(mask, h2[perm], h2)
    else:
        h2 += rng.uniform(-0.08, 0.08, n)
    return _repair(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_alpha, _g_target, _g_pen
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        init_n = len(initial_h_values)
        h_curr = _repair(np.array(initial_h_values, dtype=float).copy(), init_n)
    except NameError:
        init_n = 70
        h_curr = _repair(0.5 + 0.2 * rng.randn(70), init_n)

    best_h = h_curr.copy()
    best_c5 = _c5(best_h, init_n)
    best_n = init_n

    # Phase 1: L-BFGS-B with increasing alpha
    n = best_n
    for alpha in [100, 300, 1000, 3000, 10000]:
        if time.time() > deadline - 120:
            break
        _g_alpha = float(alpha)
        _g_target = n / 2.0
        _g_pen = 2000.0
        res = minimize(_softmax_fn, h_curr, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 2000, 'ftol': 1e-18})
        h_curr = _repair(res.x, n)
        c5_new = _c5(h_curr, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_curr.copy()

    # Phase 2: Subgradient descent
    if time.time() < deadline - 100:
        h_sg, c5_sg = _subgrad_descent(best_h, best_n, 8000, deadline - 100, 300)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = best_n

    # Phase 3: More L-BFGS-B
    for alpha in [5000, 15000, 50000]:
        if time.time() > deadline - 60:
            break
        _g_alpha = float(alpha)
        _g_target = best_n / 2.0
        _g_pen = 5000.0
        res = minimize(_softmax_fn, best_h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * best_n,
                       options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12})
        h_new = _repair(res.x, best_n)
        c5_new = _c5(h_new, best_n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Local search
    if time.time() < deadline - 40:
        h_ls, c5_ls = _local_search(best_h, best_n, deadline - 40, rng, 600)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 5: Perturbation + reoptimize loop
    for trial in range(6):
        if time.time() > deadline - 20:
            break
        h_pert = _make_perturb(best_h, best_n, rng)
        for alpha in [1000, 10000]:
            if time.time() > deadline - 20:
                break
            _g_alpha = float(alpha)
            _g_target = best_n / 2.0
            _g_pen = 3000.0
            res = minimize(_softmax_fn, h_pert, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 1500, 'ftol': 1e-17})
            h_new = _repair(res.x, best_n)
            c5_new = _c5(h_new, best_n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Try other resolutions quickly
    for n_try in [50, 60, 80, 90]:
        if n_try == best_n or time.time() > deadline - 10:
            continue
        h_t = _repair(0.5 + 0.25 * rng.randn(n_try), n_try)
        for alpha in [500, 5000]:
            if time.time() > deadline - 10:
                break
            _g_alpha = float(alpha)
            _g_target = n_try / 2.0
            _g_pen = 2000.0
            res = minimize(_softmax_fn, h_t, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n_try,
                           options={'maxiter': 1000, 'ftol': 1e-16})
            h_t = _repair(res.x, n_try)
            c5_t = _c5(h_t, n_try)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try

    # Phase 7: Final polish
    if time.time() < deadline:
        h_sg2, c5_sg2 = _subgrad_descent(best_h, best_n, 5000, deadline - 2, 500)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
        h_ls2, c5_ls2 = _local_search(best_h, best_n, max(deadline, t0 + 1), rng, 200)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
