                diff -= add

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, i, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = i - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = i + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

    return _smooth_max(_corr(x), alpha)

    return np.sum(x) - target

        if hasattr(res, 'x') and res.x is not None:

        pass

    cur_max = float(np.max(corr))
    improved = False
    changes = 0
    for i in range(n):
        for j in range(i + 1, n):
            gd = gvecs[i] - gvecs[j]
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-10:
                continue
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = ds[max(bi - 1, 0)]
            hi2 = ds[min(bi + 1, n_coarse - 1)]
            ds2 = np.linspace(lo, hi2, n_fine)
            vals2 = np.max(corr[None, :] + ds2[:, None] * gd[None, :], axis=1)
            bi2 = int(np.argmin(vals2))
            d_best = float(ds2[bi2])
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * gd
            nc = float(np.max(corr))
            if nc < cur_max:
                cur_max = nc
            improved = True
            changes += 1
            if changes % n_recalc == 0:
                gvecs = np.array([_g_vec(h, p, n) for p in range(n)])
    return h, corr, improved
