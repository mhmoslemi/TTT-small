import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _greedy_cd(h, dx, n, max_passes, rng, grid_pts=None):
    """Greedy coordinate descent: for each i, find best h[i] on a grid."""
    if grid_pts is None:
        grid_pts = np.concatenate([np.linspace(0, 1, 21), np.linspace(0.3, 0.7, 21)[1:-1]])
        grid_pts = np.unique(np.sort(grid_pts))
    best_c = _c5(h, dx)
    imp_total = 0
    for p in range(max_passes):
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            best_val = old
            best_f = _c5(h, dx)

                improved = True
                imp_total += 1
            else:

        if not improved:

                best_c = best_c_try
                imp += 1
            else:
                h[i], h[j] = hi0, hj0
        if imp == 0:
            break
    return h, best_c

def _sa_search(h, dx, n, max_iter, rng, T0=0.01, T1=0.001):
    """Simulated annealing on individual points."""
    best_c = _c5(h, dx)
    best_h = h.copy()
    T = T0
    decay = (T1 / T0) ** (1.0 / max_iter)
    for it in range(max_iter):
        i = rng.randint(n)
        old = h[i]
        # Propose new value
        if rng.random() < 0.5:
            nv = np.clip(old + rng.randn() * 0.05, 0, 1)
        else:
            nv = rng.random()
        h[i] = nv
        c_new = _c5(h, dx)
        c_old = _c5(best_h, dx)  # Use best so far as reference
        if c_new < best_c or rng.random() < np.exp(-(c_new - best_c) / max(T, 1e-10)):
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            # Recompute to fix sum
            h = _proj(h, n / 2.0)
        else:
            h[i] = old
        T *= decay
    # Ensure constraint
    h = _proj(best_h, n / 2.0)
    return best_h, _c5(best_h, dx)

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters):
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, 2.0/n)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        # Backtracking
        for _ in range(5):
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, 2.0/n)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            a = min(abs(diff), (1.0 - x[k]) if diff > 0 else x[k])
            x[k] += a if diff > 0 else -a
            diff += -a if diff > 0 else a
            if abs(diff) < 1e-13:
