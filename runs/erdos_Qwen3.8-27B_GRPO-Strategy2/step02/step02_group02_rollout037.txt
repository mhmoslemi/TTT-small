    hi = float(y.max()) + 1.0

        mid = 0.5 * (lo + hi)

        if s > target:

                diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters=3000):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _swap_coef(h, i, j, n):
    """Compute coefficient vector for swap (i,j) in correlation"""
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        L = abs(k - (n - 1))
        ci = 0.0
        cj = 0.0
        if i >= L:
            ci += 1.0 - h[i - L]
        if i + L < n:
            ci -= h[i + L]
        if j >= L:
            cj += 1.0 - h[j - L]
        if j + L < n:
            cj -= h[j + L]
        coef[k] = ci - cj
    return coef

def _cd_exact(h, dx, n, target, passes, rng):
    """Coordinate descent with exact 1D optimization"""
    bc = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        c = np.correlate(h, 1.0 - h, mode='full')
        imp = 0
        for _ in range(n * 20):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            ci, cj = h[i], h[j]
            # Feasible range for d: 0<=ci+d<=1, 0<=cj-d<=1
            d_lo = max(-ci, cj - 1.0)
            d_hi = min(1.0 - ci, cj)
            if d_lo >= d_hi:
                continue
            # Get coefficient vector (recompute c if needed)
            coef = _swap_coef(h, i, j, n)
            # Ternary search on [d_lo, d_hi] for min of max(c + d*coef)
            a, b = d_lo, d_hi
            for _ in range(20):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = float(np.max(c + m1 * coef))
                f2 = float(np.max(c + m2 * coef))
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            d_opt = 0.5 * (a + b)
            # Also check d=0
            f0 = float(c.max())
            folds = float(np.max(c + d_opt * coef))
            if folds < f0 - 1e-12:
                h[i] = ci + d_opt
                h[j] = cj - d_opt
                c = np.correlate(h, 1.0 - h, mode='full')
                bc = float(c.max()) * dx
                imp += 1
        if imp == 0:
            break
    return h, bc
