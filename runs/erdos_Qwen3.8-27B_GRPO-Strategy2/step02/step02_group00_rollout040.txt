import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _c5_topk(h, dx, k):
    c = _corr(h)
    if k >= len(c):
        return float(c.max()) * dx
    idx = np.argpartition(c, -k)[-k:]
    return float(c[idx].mean()) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            step = diff if diff > 0 else diff
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            amt = min(abs(step), room) * (1 if diff > 0 else -1)
            h[idx] += amt
            diff -= amt
            if abs(diff) < 1e-14:
                break
    return h


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n) + 1e6 * abs(np.sum(x) - n / 2.0)


def _sa_phase(h, n, dx, rng, t_budget, use_topk=True):
    h = h.copy()
    target = n / 2.0
    best_val = _c5(h, dx)
    best_h = h.copy()
    T0 = 0.003
    Tend = 0.00002
    t_end = time.time() + t_budget
    it = 0
    dmax_base = 0.05
    no_improve = 0
    patience = 50000
    while time.time() < t_end:
        frac = min(1.0, it / max(1, int(t_budget * 120)))
        T = T0 * (Tend / T0) ** frac
        dmax = dmax_base * (1.0 - 0.7 * frac)
        n_move = 2 if rng.rand() < 0.8 else 3
        indices = rng.randint(n, size=n_move)
        if len(set(indices)) < n_move:
            it += 1
            continue
        deltas = (rng.rand(n_move) - 0.5) * 2.0 * dmax
        # Compensate to maintain sum
        net = deltas.sum()
        deltas[0] -= net
        new_vals = h[indices] + deltas
        if np.any(new_vals < -1e-9) or np.any(new_vals > 1.0 + 1e-9):
            it += 1
            continue
        h[indices] = np.clip(new_vals, 0.0, 1.0)
        # Recompute sum constraint
        s = h.sum()
        if abs(s - target) > 1e-10:
            adj = (target - s) / n
            h = np.clip(h + adj, 0.0, 1.0)
        if use_topk and frac < 0.6:
            new_val = _c5_topk(h, dx, 3)
        else:
            new_val = _c5(h, dx)
        if new_val <= best_val + T:
            if new_val < best_val - 1e-14:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[indices] = h[indices] - deltas
            s = h.sum()
            if abs(s - target) > 1e-10:
                adj = (target - s) / n
                h = np.clip(h + adj, 0.0, 1.0)
            no_improve += 1
        if no_improve > patience and frac > 0.3:
            # Perturb to escape
            for _ in range(5):
                ii = rng.randint(n)
                jj = rng.randint(n)
                if ii != jj:
                    d = (rng.rand() - 0.5) * 0.04
                    h[ii] += d
                    h[jj] -= d
                    h = np.clip(h, 0.0, 1.0)
            s = h.sum()
            h = _repair(h, n)
            no_improve = 0
        it += 1
    return _c5(best_h, dx), best_h


def _coord_descent(h, n, dx, t_budget, fine=False):
    h = h.copy()
    target = n / 2.0
    best = _c5(h, dx)
    t_end = time.time() + t_budget
    if fine:
        deltas = np.array([0.0003, 0.0006, 0.001, 0.002, -0.0003, -0.0006, -0.001, -0.002])
        n_candidates = 6
    else:
        deltas = np.array([0.002, 0.004, 0.008, 0.015, 0.03, -0.002, -0.004, -0.008, -0.015, -0.03])
        n_candidates = 12
    no_round = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                found = False
                for j in order[:n_candidates]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < best - 1e-14:
                        best = c5_try
                        improved = True
                        found = True
                        break
                    else:
                        h[i] = vi
                        h[j] = nj - need
                if found:
                    break
        if improved:
            no_round = 0
        else:
            no_round += 1
            if no_round >= 4:
                break
    return best, h
