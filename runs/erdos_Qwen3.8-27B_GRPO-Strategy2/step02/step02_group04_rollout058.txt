import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _soft_max(corr, alpha):
    m = np.max(corr)
    s = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(s)) / alpha

def _slsqp(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    def obj(x):
        c = _corr(x)
        return _soft_max(c, alpha)
    def obj_grad(x):
        c = _corr(x)
        w = np.exp(alpha * (c - np.max(c)))
        w /= w.sum()

                        g[i] += wi * (1.0 - h[i-L])
                    if i + L < n:  # h[i] is in (1-h[i]) term, i.e., i is the second factor
                        g[i] -= wi * h[i+L]

                    if i + l < n:

    max_val = corr.max()
    k_star = np.argmax(corr)
    L = k_star - (n-1)  # lag
    
    # Compute A[i] = d(corr[k_star])/dh[i]
    A = np.zeros(n)
    if L >= 0:
        for i in range(n):
            if i >= L:
                A[i] += (1.0 - h[i-L])
            if i + L < n:
                A[i] -= h[i+L]
    else:
        l = -L
        for i in range(n):
            if i >= l:
                A[i] += (1.0 - h[i-l])
            if i + l < n:
                A[i] -= h[i+l]
    
    # Find best swap (i,j): minimize A[i] - A[j] (i.e., maximize A[j] - A[i])
    # We want to increase h at position with small A and decrease at position with large A
    # Change in corr[k*] = d * (A[i] - A[j]) where h[i]+=d, h[j]-=d
    # To decrease corr[k*]: A[i] < A[j]
    
    best_score = 0  # want most negative (A[i]-A[j])
    best_i, best_j = -1, -1
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            score = A[i] - A[j]  # we want this to be negative
            if score < best_score:
                best_score = score
                best_i, best_j = i, j
    
    if best_i < 0:
        return h, max_val, False
    
    # Optimal step size
    d_max = min(1.0 - h[best_i], h[best_j])
    d = d_max  # take full step
    
    h[best_i] += d
    h[best_j] -= d
    
    new_corr = _corr(h)
    new_max = new_corr.max()
    
    # Accept with simulated annealing
    if new_max <= max_val + temperature:
        return h, new_max, True
    else:
        h[best_i] -= d
        h[best_j] += d
        return h, max_val, False

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 1.0 / 2.0)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
                h[idx] += d
                diff -= d
            else:
                d = min(-diff, h[idx])
                h[idx] -= d
                diff += d
            if abs(diff) < 1e-14:
                break
    return h

def _soft_max_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def _soft_max_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    p = np.exp(alpha * (c - m))
    p /= p.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            lo = k
            hi = n - 1
            for i in range(lo, hi + 1):
                g[i] += w * (1.0 - x[i - k])
            for i in range(0, n - k):
                g[i] -= w * x[i + k]
        else:
            l = -k
            for i in range(0, n - l):
                g[i] += w * (1.0 - x[i + l])
            for i in range(l, n):
                g[i] -= w * x[i - l]
    return g

    if time.time() >= deadline:
        return h, _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj(x):
        return _soft_max_obj(x, alpha)
    def grad(x):
        return _soft_max_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:

                for i in range(n - k):
                    g[i] -= wi * h[i + k]
            else:

                for i in range(n - l):
                    g[i] += wi * (1.0 - h[i + l])
                for i in range(l, n):
                    g[i] -= wi * h[i - l]

        # This reduces the weighted corr

        # We want g[i] - g[j] < 0, i.e., g[i] < g[j]
        # Best: minimize g[i] - g[j] = maximize g[j] - g[i]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(-(1.0 - h)) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
                h[idx] += d
                diff -= d
            else:
                d = min(-diff, h[idx])
                h[idx] -= d
                diff += d
            if abs(diff) < 1e-14:
                break
    return h

def _soft_max(x, alpha):
    c = _corr(x)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def _soft_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    p = np.exp(alpha * (c - m))
    p /= p.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            for i in range(k, n):
                g[i] += w * (1.0 - x[i - k])
            for i in range(n - k):
                g[i] -= w * x[i + k]
        else:
            l = -k
            for i in range(n - l):
                g[i] += w * (1.0 - x[i + l])
            for i in range(l, n):
                g[i] -= w * x[i - l]
    return g

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj(x):
        return _soft_max(x, alpha)
    def grad(x):
        return _soft_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _greedy_search(h, n, dx, rng, deadline):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h = h.copy()
    temperature = 0.008
    stall_count = 0
    while time.time() < deadline:
        corr = _corr(h)
        max_val = np.max(corr)
        thresh = max_val - 0.03 * max_val
        active = np.where(corr > thresh)[0]
        w = np.zeros(len(corr))
        w[active] = 1.0
        w /= w.sum()
        g = np.zeros(n)
        for idx in active:
            wi = w[idx]
            k = int(idx) - (n - 1)
            if k >= 0:
                for i in range(k, n):
                    g[i] += wi * (1.0 - h[i - k])
                for i in range(n - k):
                    g[i] -= wi * h[i + k]
            else:
                l = -k
                for i in range(n - l):
                    g[i] += wi * (1.0 - h[i + l])
                for i in range(l, n):
                    g[i] -= wi * x[i - l] if False else h[i - l]
        min_idx = np.argsort(g)[:15]
        max_idx = np.argsort(g)[-15:]
        best_score = 0.0
        best_i, best_j = -1, -1
        for ii in min_idx:
            for jj in max_idx:
                if ii == jj:
                    continue
                sc = g[ii] - g[jj]
                if sc < best_score:
                    best_score = sc
                    best_i, best_j = ii, jj
        if best_i < 0 or best_score > -1e-12:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                j = (j + 1) % n
            d_max = min(1.0 - h[i], h[j])
            d = 0.1 * d_max if d_max > 1e-10 else 0.0
            if d < 1e-12:
                continue
            h[i] += d
            h[j] -= d
        else:
            d_max = min(1.0 - h[best_i], h[best_j])
            if d_max <= 1e-12:
                i = rng.randint(n)
                j = rng.randint(n)

            else:

        new_c5 = _c5(h, dx)
        if new_c5 < best_c5 - 1e-13:
            best_c5 = new_c5

            stall_count = 0

        else:
            h = best_h if new_c5 > best_c5 + 0.01 * temperature else h
            if new_c5 > best_c5:

            stall_count += 1

            h = _repair(h, n)
            temperature = 0.01
            stall_count = 0

def _subgradient(h, n, dx, deadline, alphas):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h = h.copy()
    n_iter = 0
    for alpha in alphas:
        alpha_end = alpha * 3
        a = alpha
        while time.time() < deadline and a < alpha_end:
            g = _soft_grad(h, a)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            lr = 0.02 / (a * gn)
            h_new = h - lr * g
            h_new = _repair(h_new, n)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                lr *= 0.5
                h_new = _repair(h - lr * g, n)
                c5_new = _c5(h_new, dx)
                if c5_new >= best_c5:
                    h_new = h
            h = h_new
            a += alpha * 0.05
            n_iter += 1
            if n_iter % 5000 == 0:
                h = _repair(h, n)
    return best_h, best_c5
