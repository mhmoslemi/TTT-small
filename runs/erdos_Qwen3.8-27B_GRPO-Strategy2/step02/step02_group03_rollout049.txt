def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s < 1e-12:
            h = np.full(n, 0.5)
            break
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

    """Gradient of c[k] with respect to h."""
    grad = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    for i in range(i_min, i_max + 1):
        j2 = k - i
        grad[i] += (1.0 - h[j2])
        grad[j2] -= h[i]
    return grad

def _smooth_grad(h, alpha, dx):
    """Gradient of smooth max approximation."""
    corr = _corr(h)
    ac = alpha * corr * dx
    ac_max = np.max(ac)
    w = np.exp(ac - ac_max)
    w /= w.sum()
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        grad += w[k] * _grad_c(h, k, n)
    return grad * dx

def _smooth_obj(h, alpha, dx):
    corr = _corr(h)
    ac = alpha * corr * dx
    ac_max = np.max(ac)
    return ac_max / alpha + np.log(np.sum(np.exp(ac - ac_max))) / alpha

def _lbfgs_phase(h, n, alpha, maxiter, dx):
    def obj(x):
        return _smooth_obj(x, alpha, dx)
    def grad(x):
        return _smooth_grad(x, alpha, dx)
    bounds = [(0.0, 1.0)] * n
    # Use penalty for sum constraint
    target = n / 2.0
    def pen_obj(x):
        return obj(x) + 1000.0 * (np.sum(x) - target) ** 2
    def pen_grad(x):
        return grad(x) + 2000.0 * (np.sum(x) - target)
    try:
        res = minimize(pen_obj, h, jac=pen_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return _repair(res.x, n)
    except:
        return h

def _subgrad_step(h, n, dx, lr):
    """One subgradient step on the true max."""
    corr = _corr(h)
    k_star = np.argmax(corr)
    g = _grad_c(h, k_star, n) * dx
    # Also include near-max for stability
    top_k = np.argsort(corr)[::-1][:5]
    w = np.exp(50 * (corr[top_k] - np.max(corr)))
    w /= w.sum()
    g_total = np.zeros(n)
    for idx in top_k:
        g_total += w[idx] * _grad_c(h, idx, n)
    g_total *= dx
    h_new = h - lr * g_total
    h_new = _repair(h_new, n)
    return h_new

def _coord_hill_climb(h, n, dx, max_passes, time_limit, rng):
    """Systematic coordinate descent with pair moves."""
    h = h.copy()
    best_c = _c5(h, dx)
    step_sizes = [0.01, 0.02, 0.05, 0.1]
    for p in range(max_passes):
        if time.time() > time_limit:
            break
        improved = False
        # Try all pairs of nearby indices
        for i in range(n):
            for j in [i+1, i-1, (i+n//2)%n] if j < n and j >= 0 and j != i:
                for d in step_sizes:
                    for sign in [1, -1]:
                        hi, hj = h[i], h[j]
                        ni, nj = hi + sign*d, hj - sign*d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi, hj
        if not improved:
            step_sizes = [s*0.5 for s in step_sizes]
            if step_sizes[0] < 1e-6:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s < 1e-12:
            return np.full(n, 0.5)
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _grad_single_k(h, k, n):
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    for i in range(lo, hi + 1):
        j2 = k - i
        g[i] += (1.0 - h[j2])
        g[j2] -= h[i]
    return g

def _smooth_obj(x, alpha, dx):
    c = _corr(x)
    z = alpha * c * dx
    zmax = np.max(z)
    return zmax / alpha + np.log(np.sum(np.exp(z - zmax))) / alpha

def _smooth_grad(x, alpha, dx, n):
    c = _corr(x)
    z = alpha * c * dx
    z -= np.max(z)
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-14:
            g += w[k] * _grad_single_k(x, k, n)
    return g * dx

                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)

    def objg(x): return _smooth_grad(x, alpha, dx, n) + 10000.0 * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n

        return h

def _coord_search(h, n, dx, max_passes, time_limit, rng):
