def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_arr(h))) * dx

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    corr = _corr_arr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    lags = np.arange(-(n - 1), n)
    grad = np.ones(n)
    for j in range(n):
        idx1 = j - lags
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + lags
        mask2 = (idx2 >= 0) & (idx2 < n)
        t1 = float(np.sum(w[mask1] * h[idx1[mask1]])) if mask1.any() else 0.0
        t2 = float(np.sum(w[mask2] * h[idx2[mask2]])) if mask2.any() else 0.0
        grad[j] = 1.0 - t1 - t2
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _nesterov_descent(h, n, dx, deadline, alpha=300):
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    x = h.copy()
    v = np.zeros(n)
    lr = 0.015
    mu = 0.8
    while time.time() < deadline:
        x_nest = x - mu * v
        g = _smooth_grad(x_nest, alpha)
        g -= g.mean()
        v = mu * v - lr * g
        x_new = _repair(x + v, n)
        obj_new = _smooth_obj(x_new, alpha)
        obj_old = _smooth_obj(x, alpha)
        if obj_new > obj_old + 1e-12:
            lr *= 0.85
            v *= 0.4
        elif lr < 0.04:
            lr *= 1.03
        x = x_new
        c5 = _compute_c5(x, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
    return best_h, best_c5
