∂corr_k/∂h[j] = (1 - h[j+lag]) * [j+lag valid] + (-1) * h[j-lag] * [j-lag valid]

where "valid" means the index is in [0, n-1] and the pair is within the correlation range.

This can be computed in O(n) per k. For 139 k's, that's O(139*n) = O(9730). Very fast.

Let me implement this properly:

```python
def _grad_all_corr(h):
    """Compute gradient of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n) where G[k, j] = d corr[k] / d h[j]
    """
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        for i in range(i_start, i_end):
            j = i + lag
            G[k, i] += (1.0 - h[j])  # d h[i]*(1-h[j]) / d h[i]
            G[k, j] += (-1.0) * h[i]  # d h[i]*(1-h[j]) / d h[j]
    return G
```

from scipy.optimize import minimize, minimize_scalar

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_vals(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_max_obj(x, alpha, mu, n):
    c = _corr_vals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    f = m + np.log(np.sum(np.exp(s))) / alpha
    target = n / 2.0
    f += mu * (x.sum() - target) ** 2
    return f

def _smooth_max_grad(x, alpha, mu, n):
    c = _corr_vals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    # Gradient of smooth max: sum_k w_k * dc_k/dx
    grad = np.zeros_like(x)
    for k in range(len(c)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        for i in range(i_start, i_end):
            j = i + lag
            grad[i] += w[k] * (1.0 - x[j])
            grad[j] += w[k] * (-1.0) * x[i]
    # Sum penalty gradient
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _lbfgsb(h, n, dx, alpha, mu, maxiter=3000, ftol=1e-16):
    def obj(x):
        return _smooth_max_obj(x, alpha, mu, n)
    def jac(x):
        return _smooth_max_grad(x, alpha, mu, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_pass(h, n, dx, n_grid, max_passes_inner=1):

    improved = False
    n_imp = 0
    for i in range(n):
        for j in range(i + 1, n):
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:

            orig_i = h[i]
            orig_j = h[j]
            best_t = orig_i
            best_c = hi_c5
            for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                h[i] = t
                h[j] = hsum - t
                c = _c5(h, dx)
                if c < best_c:

                    best_t = t

                h[j] = hsum - best_t
                hi_c5 = best_c
                improved = True
                n_imp += 1
            else:
                h[i] = orig_i
                h[j] = orig_j
    return h, hi_c5, improved, n_imp

def _cd_loop(h, n, dx, t_limit, t_start, n_grid, max_passes=1000, tol=1e-14):

So corr_k as a function of t is at most quadratic (with a t² term only if i and j are directly paired in that correlation). The max of quadratics is convex but piecewise smooth.

For a max of quadratics, golden section search should work well. Let me use it:

```python
def _golden_search(lo, hi, f, n_iter=25):
    """Golden section minimization"""
    ratio = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    a, b = lo, hi
    c_ = b - ratio * (b - a)
    d_ = a + ratio * (b - a)
    fc = f(c_)
    fd = f(d_)
    for _ in range(n_iter):
        if fc < fd:
            b, d_, fd = d_, c_, fc
            c_ = b - ratio * (b - a)
            fc = f(c_)
        else:
            a, c_, fc = c_, d_, fd
            d_ = a + ratio * (b - a)
            fd = f(d_)
    return (a + b) / 2
```

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lbfgsb(h, n, alpha, mu, maxiter=2000):
    def obj(x):
        c = _corr(x); m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        return m + np.log(np.sum(np.exp(s)))/alpha + mu*(x.sum()-n/2.0)**2
    def jac(x):
        c = _corr(x); m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        e = np.exp(s); w = e/e.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-12: continue
            lag = k-(n-1); i0 = max(0,-lag); i1 = min(n,n-lag)
            idx = np.arange(i0,i1); jdx = idx+lag
            g[idx] += w[k]*(1.0-x[jdx]); g[jdx] -= w[k]*x[idx]
        g += 2.0*mu*(x.sum()-n/2.0)
        return g
    try:

                     bounds=[(0.0,1.0)]*n, options={'maxiter':maxiter,'ftol':1e-16})
