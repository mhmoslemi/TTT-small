def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(h_init, dx)
    
    def remaining(margin=0):
        return t0 + budget_s - margin - time.time()
    
    # Phase 1: SLSQP from multiple starting points
    starts = [h_init]
    # Add a couple of structured inits
    for si in range(2):
        t = np.linspace(0, 2.0, n, endpoint=False)
        base = 0.5 + 0.25 * np.sin(2 * np.pi * (1 + si) * t + si * 0.7)
        base += 0.05 * rng.randn(n)
        starts.append(_repair(base, n))
    
    for hc in starts:
        if remaining(300) < 0:
            break
        c5 = _c5(hc, dx)
        for a in [60, 150, 400]:
            if remaining(300) < 0:
                break
            hc = _slsqp_step(hc, n, dx, a, 300, 1e-14)
            nc = _c5(hc, dx)
            if nc < c5:
                c5 = nc
        if c5 < best_c5:
            best_c5, best_h = c5, hc.copy()
    
    # Phase 2: Systematic pairwise descent
    if remaining(200) > 30:
        best_h, best_c5 = _pairwise_descent(best_h, n, dx, t0 + budget_s - 200)

    if remaining(120) > 10:

            if c5 < best_c5:
                best_c5, best_h = c5, hc
