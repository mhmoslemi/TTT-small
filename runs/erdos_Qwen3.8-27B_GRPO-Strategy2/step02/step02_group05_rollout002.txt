import numpy as np
from scipy.optimize import minimize
import time

def _cv(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_cv(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * tgt
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                s2 = min(-d, h[idx])
                h[idx] -= s2
                d += s2
            if abs(d) < 1e-10:
                break
    return h

def _obj_full(x, alpha, dx):
    cv = _cv(x)
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _obj_top(x, alpha, dx, top_k):
    cv = _cv(x)
    idx = np.argsort(cv)[-top_k:]
    cvt = cv[idx]
    m = cvt.max()
    s = np.clip(alpha * (cvt - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _ceq(x, tgt):
    return np.sum(x) - tgt

def _ceq_jac(x, tgt):
    return np.ones_like(x)

def _slsqp(h, n, dx, alpha, maxiter, ftol, top_k=0):
    bounds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceq_jac, 'args': (tgt,)}]
    if top_k > 0:
        try:
            res = minimize(_obj_top, h, args=(alpha, dx, top_k), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.fun < _obj_top(h, alpha, dx, top_k):
                return _repair(res.x, n)
        except Exception:
            pass
    try:
        res = minimize(_obj_full, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < _obj_full(h, alpha, dx):
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()
