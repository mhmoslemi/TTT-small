import numpy as np
from scipy.optimize import minimize
import time

_g_target = 35.0
_g_alpha = 100.0

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _smooth_obj(x):
    corr = _corr_full(x)
    m = np.max(corr)
    s = _g_alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _g_alpha

def _con_fun(x):
    return np.sum(x) - _g_target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, max(0.0, 1.0 - h[idx]))
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, max(0.0, h[idx]))
                h[idx] -= sub
                diff += sub
    return h

def _slsqp(h, n, alpha, maxiter):
    global _g_alpha, _g_target
    _g_alpha = alpha
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

                t = (a + b) / 2.0

                h[j] -= t

                    imp = True

def _toplag_fix(h, n, dx, n_iter, budget_end):

    best = _c5(h, dx)
    for _ in range(n_iter):

        top = np.argsort(corr)[::-1][:5]
