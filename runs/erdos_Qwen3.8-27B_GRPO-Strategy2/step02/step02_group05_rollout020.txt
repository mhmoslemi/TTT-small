def _obj(x, alpha, dx):
    return _sm(_corr(x), alpha) * dx

def _con(x, target):
    return np.sum(x) - target

def _slsqp(h0, n, dx, alpha, maxiter=400, ftol=1e-14):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'args': (target,)}]
    try:
        res = minimize(_obj, h0, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < 1e10:
            return _repair(res.x, n)
    except Exception:
        pass
    return h0.copy()

def _focused_ls(h, n, dx, rng, deadline, fine=True):
    best = h.copy()
    best_c5 = _c5(best, dx)
    if fine:
        steps = np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05,
                          -0.001, -0.002, -0.005, -0.01, -0.02, -0.03, -0.05])
    else:
        steps = np.array([0.01, 0.02, 0.04, 0.06, 0.1,
                          -0.01, -0.02, -0.04, -0.06, -0.1])
    for outer in range(5000):
        if time.time() > deadline:
            break
        c = _corr(best)
        k_max = int(np.argmax(c))
        j_max = k_max - (n - 1)
        # Identify entries contributing to max lag
        if j_max >= 0:
            hot = set(range(0, n - j_max)) | set(range(j_max, n))
        else:
            hot = set(range(-j_max, n)) | set(range(0, n + j_max))
        hot_list = list(hot)
        improved = False
        ntrials = 120
        for _ in range(ntrials):
            if rng.random() < 0.7 and len(hot_list) > 2:
                i = hot_list[int(rng.randint(len(hot_list)))]
            else:
                i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = steps[int(rng.randint(len(steps)))]
            ni, nj = best[i] + d, best[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-13:
                best_c5 = nc
                improved = True
            else:
                best[i], best[j] = ni - d, nj + d
    return best, best_c5

def _pair_optimize(h, n, dx, deadline, top_k=3):
    """Optimize pairs that contribute to the top lags."""
    best = h.copy()
    best_c5 = _c5(best, dx)
    grid = np.linspace(-0.15, 0.15, 31)
    for outer in range(2000):
        if time.time() > deadline:
            break
        c = _corr(best)
        top_idx = np.argsort(c)[-top_k:]
        hot_pairs = set()
        for k in top_idx:
            j = k - (n - 1)
            if j >= 0:
                for i in range(n - j):
                    hot_pairs.add((i, i + j))
            else:
                for i in range(-j, n):
                    hot_pairs.add((i, i + j))
        hp_list = list(hot_pairs)
        if not hp_list:
            break
        improved = False
        rng_pass = np.random.RandomState(outer * 7 + 13)
        samples = rng_pass.choice(len(hp_list), size=min(60, len(hp_list)), replace=False)
        for si in samples:
            i, j = hp_list[si]
            for d in grid:
                if abs(d) < 1e-10:
                    continue
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
        if not improved:
            break
    return best, best_c5

def _periodic_init(n, period, rng, t0):
    h = np.zeros(n)
    base = np.zeros(period)
    base[:(period+1)//2] = 1.0
    for i in range(n):
        h[i] = base[i % period] + 0.1 * rng.randn()
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h_init.copy(), _c5(h_init, dx)

    def elig(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: SLSQP adaptive alpha on initial
    hc = h_init.copy()
    for a in [30, 60, 120, 250, 500, 1000, 2000]:
        if not elig(300):
            break
        hc = _slsqp(hc, n, dx, a, 500, 1e-15)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc.copy()

    # Phase 2: Focused local search
    if elig(200):
        best_h, best_c5 = _focused_ls(best_h, n, dx, rng, t0 + budget_s - 200, fine=False)

    # Phase 3: Pair optimization on top lags
    if elig(150):
        best_h, best_c5 = _pair_optimize(best_h, n, dx, t0 + budget_s - 150, top_k=3)

    # Phase 4: SLSQP restarts with noise
    sigmas = [0.3, 0.5, 0.8, 0.2, 0.4, 0.6, 0.15, 0.35, 0.7, 0.25]
    it = 0
    while elig(100):
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        for a in [200, 500]:
            if not elig(100):
                break
            hc = _slsqp(hp, n, dx, a, 300, 1e-14)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc.copy()
        it += 1

    # Phase 5: Periodic diverse starts
    for p in [7, 10, 14]:
        if not elig(120):
            break
        hp = _periodic_init(n, p, rng, t0)
        for a in [100, 300, 800]:
            if not elig(120):
                break
            hc = _slsqp(hp, n, dx, a, 300, 1e-14)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc.copy()
                best_h, best_c5 = _focused_ls(best_h, n, dx, rng, t0 + budget_s - 120, fine=False)

    # Phase 6: Fine local search
    if elig(60):
        best_h, best_c5 = _focused_ls(best_h, n, dx, rng, t0 + budget_s - 55, fine=True)

    # Phase 7: Pair optimize fine
    if elig(40):
        best_h, best_c5 = _pair_optimize(best_h, n, dx, t0 + budget_s - 35, top_k=5)

    # Phase 8: Final SLSQP polish
    if elig(20):
        for a in [1000, 3000, 5000]:
            if not elig(15):
                break
            hc = _slsqp(best_h, n, dx, a, 600, 1e-15)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc.copy()

    # Phase 9: Ultra-fine local search
    if elig(8):
        deadline = t0 + budget_s - 3
        best_h, best_c5 = _focused_ls(best_h, n, dx, rng, deadline, fine=True)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
