import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                h[idx] += min(diff, 1.0 - h[idx])
                diff -= h[idx] - h[idx] + min(diff, 1.0 - h[idx])
            else:
                h[idx] += max(diff, -h[idx])
                diff -= h[idx] + max(diff, -h[idx])
    return np.clip(h, 0.0, 1.0)

def _compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_opt(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_opt(h, n, alpha, pen, maxiter):
    target = n / 2.0
    def obj(x):
        return _smooth_obj(x, alpha) + pen * (np.sum(x) - target) ** 2
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _compute_effects(h, n):
    eff = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for k in range(-(n - 1), n):
            ki = k + n - 1
            v = 0.0
            if 0 <= i - k < n:
                v += 1.0 - h[i - k]
            if 0 <= i + k < n:
                v -= h[i + k]
            eff[i, ki] = v
    return eff
