    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

                h[idx] += add
                diff -= add
            else:

                diff += sub

    target = n / 2.0

        return _smooth(_corr(x), alpha)

                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-18})

            best_c5, best_h = cl, hl.copy()

    # Phase 2: Full optimization at primary n (300s)
    end2 = t_start + 420
    rem = budget_s - (time.time() - t_start)
    if rem > 100:
        hd, cd, _ = _opt_at_n(best_h, best_n, t_start, min(end2, t_start + rem * 0.45), rng)
        if cd < best_c5:
            best_c5, best_h, best_n = cd, hd.copy(), len(hd)

    # Phase 3: Try other n values (250s)
    end3 = t_start + 670
    for n_try in [50, 60, 80, 90, 100]:
        if time.time() >= end3 or budget_s - (time.time() - t_start) < 150:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_try)
        hi = np.interp(x_new, x_old, best_h)
        sub_dl = min(end3 - 5, time.time() + 50)
        ht, ct, _ = _opt_at_n(hi, n_try, t_start, sub_dl, rng)
        if ct < best_c5:
            best_c5, best_h, best_n = ct, ht.copy(), n_try

    # Phase 4: Intensive refinement at best (200s)
    end4 = t_start + 870
    rem = budget_s - (time.time() - t_start)
    if rem > 80:
        dx_b = 2.0 / best_n
        h = best_h.copy()
        for alpha in [500, 2000, 8000]:
            if time.time() >= end4 or rem - (time.time() - t_start) < 20:
                break
            h, c = _slsqp(h, best_n, dx_b, alpha, 800, min(end4, time.time() + 40))
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        if time.time() < end4 - 10:
            h, c = _lag_push(best_h, best_n, dx_b, rng, min(end4 - 5, time.time() + 60), 80000)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        if time.time() < end4 - 5:
            best_h, best_c5 = _local_search(best_h, best_n, dx_b, rng, min(end4 - 2, time.time() + 40), True)

    # Phase 5: Final polish (remaining ~230s)
    rem = budget_s - (time.time() - t_start)
    if rem > 30:
        dx_b = 2.0 / best_n
        for alpha in [5000, 15000, 30000]:
            if rem - (time.time() - t_start) < 15:
                break
            dl = min(t_start + budget_s - 20, time.time() + 25)
            h, c = _slsqp(best_h, best_n, dx_b, alpha, 500, dl)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        rem = budget_s - (time.time() - t_start)
        if rem > 20:
            h, c = _anneal(best_h, best_n, dx_b, rng, min(t_start + budget_s - 10, time.time() + 20), 0.3, 0.01)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        rem = budget_s - (time.time() - t_start)
        if rem > 10:
            best_h, best_c5 = _local_search(best_h, best_n, dx_b, rng,
                                             min(t_start + budget_s - 3, time.time() + 12), False)
        rem = budget_s - (time.time() - t_start)
        if rem > 5:
            best_h, best_c5 = _lag_push(best_h, best_n, dx_b, rng,
                                         min(t_start + budget_s - 1, time.time() + 4), 30000)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
