import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, n):
    target = n / 2.0
    # Bisection for lambda: sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _grad_active(h, n, top_idx, n_active):
    """Compute sum of gradients for top active shifts."""
    grad = np.zeros(n)
    for m in top_idx[:n_active]:
        s = m - (n - 1)  # actual shift
        if s == 0:
            # c_0 = sum h[i]*(1-h[i]), d/dh[j] = 1-2*h[j]
            grad += (1 - 2 * h)
        elif s > 0:
            # d(c_s)/d(h[j]) = (1-h[j-s]) for s<=j<n, minus h[j+s] for 0<=j<=n-1-s
            for j in range(s, n):
                grad[j] += (1 - h[j - s])
            for j in range(0, n - s):
                grad[j] -= h[j + s]
        else:
            t = -s
            # d(c_{-t})/d(h[j]) = (1-h[j+t]) for 0<=j<=n-1-t, minus h[j-t] for t<=j<=n-1
            for j in range(0, n - t):
                grad[j] += (1 - h[j + t])
            for j in range(t, n):
                grad[j] -= h[j - t]
    return grad

def _proj_grad(h, n, dx, n_iters, t_end, rng):
    """Projected gradient descent targeting active shifts."""
    h = h.astype(float).copy()
    best = _c5val(h, dx)
    lr = 0.05
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = _corr(h)
        c = float(np.max(corr)) * dx
        if c < best:
            best = c
            lr = min(lr * 1.1, 0.5)
        else:
            lr = max(lr * 0.8, 0.001)
        
        # Top active shifts
        top_idx = np.argsort(corr)[::-1][:4]
        grad = _grad_active(h, n, top_idx, 4)
        
        # Projected gradient step
        h_new = _project(h - lr * grad, n)
        c_new = _c5val(h_new, dx)
        
        if c_new > best * 1.05 + 0.01:
            # Rejection: step too big, reduce lr more
            lr = max(lr * 0.5, 0.001)
            h_new = _project(h - lr * grad, n)
            c_new = _c5val(h_new, dx)
        
        h = h_new

def _S_obj(x):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(_g_a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _g_a

def _S_con(x):
    return np.sum(x) - _g_n / 2.0

def _slsqp(h, n, alpha, maxiter):
    global _g_a, _g_n
    _g_a, _g_n = alpha, n
    try:

                     constraints=[{'type':'eq', 'fun': _S_con}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_shift(h, n, s):
    """Gradient of correlation at shift s w.r.t. h."""
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        idx1 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)
        g[idx2] -= h[idx2 + s]
    else:
        t = -s
        idx1 = np.arange(0, n - t)
        g[idx1] += (1.0 - h[idx1 + t])
        idx2 = np.arange(t, n)
        g[idx2] -= h[idx2 - t]
    return g

def _proj_grad(h, n, t_end, n_iters, rng):
    h = h.astype(float).copy()

    lr = 0.03

            break

        c = float(np.max(corr)) * (2.0 / n)

            best = c
            lr = min(lr * 1.08, 0.3)
        else:
            lr = max(lr * 0.85, 0.003)
        top = np.argsort(corr)[::-1][:4]
        grad = np.zeros(n)
        for m in top:
            grad += _grad_shift(h, n, m - (n - 1))
        grad /= len(top)
        for trial in range(5):
            h_new = _project(h - lr * grad, n)
            c_new = _c5val(h_new, 2.0 / n)
            if c_new <= best:

def _S_obj(x):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(_g_a * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / _g_a

def _S_con(x):
    return np.sum(x) - _g_n / 2.0

def _slsqp(h, n, alpha, maxiter):

    _g_a, _g_n = alpha, n
    try:
        r = minimize(_S_obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _S_con}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(r.x, n)
    except Exception:
        return h

def _lbfgs_obj(x):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(_g_a * (c - m), -500.0, 500.0)
    base = m + np.log(np.sum(np.exp(s))) / _g_a
    pen = _g_lam * (np.sum(x) - _g_n / 2.0) ** 2
    return base + pen
