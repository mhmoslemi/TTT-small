```python
def corr_gradient(h, n):
    """Return array of gradients: grad[k, m] = d corr[k] / d h[m] for k=0..2n-2"""
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)  # shift
        if s > 0:
            for m in range(n):
                g = 0.0
                if m < n - s:
                    g += (1.0 - h[m+s])
                if m >= s:
                    g -= h[m-s]
                grad[k, m] = g
        elif s < 0:
            sp = -s
            for m in range(n):
                g = 0.0
                if m >= sp:
                    g += (1.0 - h[m-sp])
                if m + sp < n:
                    g -= h[m+sp]
                grad[k, m] = g
        else:  # s == 0
            grad[k, :] = 1.0 - 2.0 * h
    return grad
```

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _proj(x, target):

        if np.sum(np.clip(x + mid, 0.0, 1.0)) > target:
            hi = mid
        else:

def _grad_mat(h, n):
    g = np.zeros((2 * n - 1, n))
    g[n - 1] = 1.0 - 2.0 * h
    for s in range(1, n):
        g[n - 1 + s, :n - s] += (1.0 - h[s:])
        g[n - 1 + s, s:] -= h[:n - s]
        g[n - 1 - s, s:] += (1.0 - h[:n - s])
        g[n - 1 - s, :n - s] -= h[s:]
    return g

def _slsqp_obj(x):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(_G_A * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _G_A

def _slsqp_con(x):
    return np.sum(x) - _G_T

                     constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _proj(r.x, _G_T)

        return h
