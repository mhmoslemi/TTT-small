            if c5b < best_c5:
                best_h, best_c5 = h2.copy(), c5b

    # Phase 4: Alternative starts + optimize
    for mode in range(4):
        if not rem(60):
            break
        hr = _random_start(n, rng, mode)
        hr, cr = _sweep(hr, n, dx, 2, t0, budget_s)
        if rem(50):
            hr = _slsqp_polish(hr, n, dx, 200, 150)
            cr = _c5(hr, dx)
        if rem(40):
            hr, cr = _sweep(hr, n, dx, 2, t0, budget_s)
        if cr < best_c5:
            best_h, best_c5 = hr.copy(), cr

    # Phase 5: Final anneal + sweep + polish
    if rem(50):
        h, c5 = _anneal_search(best_h, n, dx, rng, t0, budget_s, 20000)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
    if rem(30):
        h, c5 = _sweep(best_h, n, dx, 5, t0, budget_s)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
    if rem(15):
        for alpha in [3000, 8000]:
            if not rem(10):
                break
            hp = _slsqp_polish(best_h, n, dx, alpha, 400)
            c5 = _c5(hp, dx)
            if c5 < best_c5:
                best_h, best_c5 = hp.copy(), c5
    if rem(8):
        h, c5 = _sweep(best_h, n, dx, 3, t0, budget_s)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
