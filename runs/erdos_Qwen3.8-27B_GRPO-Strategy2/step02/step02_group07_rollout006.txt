    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 40)
    return m + np.log(np.sum(np.exp(s))) / alpha

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

import numpy as np
import time
from scipy.optimize import minimize

def _corr_max(h):
    return float(np.correlate(h, 1.0 - h, mode='full').max())

def _c5_val(h, dx):
    return _corr_max(h) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

class _EqCons:
    def __init__(self, target):
        self._t = target
    def __call__(self, x):
        return x.sum() - self._t

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 40)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _EqCons(target)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _golden_cd(h, i, j, n, n_iter):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return _corr_max(h)
    hi_orig = h[i]
    hj_orig = h[j]
    orig_max = _corr_max(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_v = lo + 1e-14
    b_v = hi - 1e-14
    if b_v - a_v < 1e-12:
        return orig_max
    c_v = b_v - gr * (b_v - a_v)
    d_v = a_v + gr * (b_v - a_v)
    h[i] = c_v; h[j] = s - c_v
    fc = _corr_max(h)
    h[i] = d_v; h[j] = s - d_v
    fd = _corr_max(h)
    for _ in range(n_iter):
        if fc < fd:
            b_v = d_v; d_v = c_v; fd = fc
            c_v = b_v - gr * (b_v - a_v)
            h[i] = c_v; h[j] = s - c_v
            fc = _corr_max(h)
        else:
            a_v = c_v; c_v = d_v; fc = fd
            d_v = a_v + gr * (b_v - a_v)
            h[i] = d_v; h[j] = s - d_v
            fd = _corr_max(h)
    best_f = min(fc, fd)
    t_best = c_v if fc < fd else d_v
    if best_f < orig_max:
        h[i] = t_best
        h[j] = s - t_best
        return best_f
    else:
        h[i] = hi_orig
        h[j] = hj_orig
        return orig_max

def _cd_pass(h, n, t_lim, t0, shifts, n_iter):
    improved = False
    best_c = _corr_max(h)
    if shifts is not None:
        pairs = set()
        for k in shifts:
            k_abs = abs(k)
            if k_abs == 0:
                continue
            for ii in range(n - k_abs):
                pairs.add((ii, ii + k_abs))
        pair_list = list(pairs)
    else:
        pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
    rng_local = np.random.RandomState(42)
    order = rng_local.permutation(len(pair_list))
    for ii in order:
        if time.time() - t0 > t_lim:
            break
        i, j = pair_list[ii]
        new_c = _golden_cd(h, i, j, n, n_iter)
        if new_c < best_c - 1e-15:
            best_c = new_c
            improved = True
    return h, best_c, improved

def _active_shifts(h, n, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_m]
    return [int(x) - (n - 1) for x in idx]

def _bipolar_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return _repair(h, n)
