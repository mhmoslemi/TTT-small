import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth(x, alpha):
    # Gradient of smooth max for the correlation
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    w = np.exp(np.clip(alpha * (c - m), -500, 500))
    w /= w.sum()
    # dc/dx is complex; use numerical gradient or skip
    return None  # Let L-BFGS-B use numerical gradient

def _lbfgs(h, n, alpha, maxiter, opts):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, **opts})
        return _repair(res.x, n)
    except:
        try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except:
        return h.copy()

from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _obj_alpha(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -500, 500)))) / alpha

def _sqp(h, n, alpha, maxit):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(_obj_alpha, h, args=(alpha,), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _cd_pass(h, n, dx, grid, idx_list):

    nf = len(idx_list)

            bt, bc = h[i], c5
            ts = np.linspace(lo + 1e-9, hi - 1e-9, grid)

                h[i], h[j] = bt, s - bt
                c5 = bc
                imp = True
            else:
                h[i], h[j] = bt, s - bt
    return c5, imp
