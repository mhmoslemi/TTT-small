    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    
    def _slsqp(h0, alpha, mi, ft):
        def f(x):
            return _smooth_obj_val(np.correlate(x, 1.0-x, mode='full'), alpha)
        def jf(x):
            g, _ = _smooth_grad(x, n, alpha)
            return g
        r = minimize(f, h0, jac=jf, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    
    # Phase 1: SLSQP annealing
    for alpha in [30, 75, 200, 500, 1200, 3000, 8000, 20000]:
        if _el() > budget_s - 200:
            break
        try:
            hc = _slsqp(best_h.copy(), alpha, 500, 1e-14)
            c5 = _c5_of(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except:
            pass
    
    # Phase 2: Projected gradient
    if _el() < budget_s - 180:
        h = best_h.copy()
        c5c = _c5_of(h, dx)
        for alpha in [100, 500, 2000, 8000]:
            if _el() > budget_s - 180:
                break
            lr = 0.05
            for it in range(5000):
                if _el() > budget_s - 180:
                    break
                g, _ = _smooth_grad(h, n, alpha)
                gn = np.linalg.norm(g)
                if gn < 1e-10:
                    break
                found = False
                rl = lr
                for _ in range(50):
                    hn = _repair(h - rl * g / max(gn, 1e-10), n)
                    c5n = _c5_of(hn, dx)
                    if c5n < c5c - 1e-15:
                        h = hn; c5c = c5n; found = True
                        if c5c < best_c5:
                            best_c5, best_h = c5c, h.copy()
                        break
                    rl *= 0.55
                if found:
                    lr = min(lr * 1.08, 0.1)
                else:
                    lr = max(rl * 0.5, 1e-6)
    
    # Phase 3: Diversified SLSQP restarts
    if _el() < budget_s - 120:
        for r in range(40):
            if _el() > budget_s - 120:
                break
            s = r % 5
            if s == 0:
                hp = best_h + 0.01 * rng.randn(n)
            elif s == 1:
                hp = best_h.copy()
                hp[rng.choice(n, n//6, replace=False)] = 1.0 - hp[rng.choice(n, n//6, replace=False)]
                # Hmm this might select different indices. Let me fix.
                idx = rng.choice(n, n//6, replace=False)
                hp = best_h.copy()
                hp[idx] = 1.0 - best_h[idx]
            elif s == 2:
                hp = best_h.copy()
                b0 = rng.randint(0, n-20)
                hp[b0:b0+20] = np.clip(best_h[b0:b0+20] + 0.03*rng.randn(20), 0, 1)
            elif s == 3:
                hp = best_h.copy()
                f = rng.randint(1, 6)
                hp += 0.008 * np.sin(2*np.pi*f*np.arange(n)/n + rng.rand()*6.28)
            else:
                hp = best_h.copy()
                a = rng.randint(0, n-25)
                b = (a + 25) % n
                hp[a:a+25], hp[b:b+25] = hp[b:b+25].copy(), hp[a:a+25].copy()
            hp = _repair(hp, n)
            try:
                hc = _slsqp(hp, 5000, 400, 1e-13)
                c5 = _c5_of(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except:
                pass
