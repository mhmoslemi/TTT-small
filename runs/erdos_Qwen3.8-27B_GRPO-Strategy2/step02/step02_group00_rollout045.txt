from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _obj_lse(x, alpha, n, target, penalty):
    corr = _corr(x)
    m = float(np.max(corr))
    shifted = np.clip(alpha * (corr - m), -50, 50)
    val = (m + np.sum(np.exp(shifted)) / alpha) * (2.0 / n)
    # Penalty for integral constraint
    s = float(np.sum(x))
    val += penalty * (s - target) ** 2
    return val

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    # Scale to meet target
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    target = n // 2
    # Sort by value, set top target to 1, rest to 0
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b
