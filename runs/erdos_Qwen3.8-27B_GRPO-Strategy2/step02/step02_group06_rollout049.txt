                diff -= add
            else:

                h[idx] -= sub
                diff += sub

    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = i - k

    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

    return _smooth_max(_corr(x), alpha)

    return np.sum(x) - target

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, i, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = i - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = i + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

    return _smooth_max(_corr(x), alpha)

        gd = _g_vec(h, i, n) - _g_vec(h, j, n)
        ds1 = np.linspace(d_min, d_max, n_coarse)
        nc1 = corr[None, :] + ds1[:, None] * gd[None, :]
        c5s1 = np.max(nc1, axis=1) * dx
        bi1 = np.argmin(c5s1)
        cur_max = float(np.max(corr)) * dx
        if c5s1[bi1] < cur_max - 1e-14:
            d_best = ds1[bi1]
            if n_fine > 2 and fine_wid > 0:

                hi2 = min(d_max, d_best + fine_wid)
                ds2 = np.linspace(lo, hi2, n_fine)

            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
