    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP with increasing alpha for better max approximation
    for alpha in [50, 150, 400, 1000, 3000, 8000]:
        if time.time() - t0 > budget_s - 450:
            break
        hc = _slsqp(best_h, n, dx, target, bounds, alpha, 1200)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 2: Logit parameterization (avoids boundary issues)
    if time.time() - t0 < budget_s - 400:
        for alpha in [400, 1500]:
            if time.time() - t0 > budget_s - 400:
                break
            hc = _logit_slsqp(best_h, n, dx, target, alpha, 1000)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 3: Coordinate pair sweeps (coarse to fine)
    if time.time() - t0 < budget_s - 350:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(50):
            if time.time() - t0 > budget_s - 350:
                break
            h, corr, imp = _sweep(h, n, dx, corr, 20, 41, 0.025)
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 4: Differential evolution (global exploration)
    if time.time() - t0 < budget_s - 250:
        hc = _de_search(n, dx, target, seed, 35, 35)
        if hc is not None:
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
                # Quick polish after DE
                hc2 = _slsqp(best_h, n, dx, target, bounds, 2000, 800)
                c52 = _c5(hc2, dx)
                if c52 < best_c5:
                    best_c5 = c52
                    best_h = hc2

    # Phase 5: Diversified restarts with local optimization
    for sr in range(12):
        if time.time() - t0 > budget_s - 130:
            break
        scale = 0.04 + 0.025 * sr
        h_pert = best_h + scale * rng.randn(n)
        h_pert = _repair(h_pert, n)
        hc = _slsqp(h_pert, n, dx, target, bounds, 1500, 600)
        corr_r = _corr(hc)
        hc, corr_r, _ = _sweep(hc, n, dx, corr_r, 18, 31, 0.02)
        c5 = float(np.max(corr_r) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 6: Fine coordinate sweeps
    if time.time() - t0 < budget_s - 80:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(50):
            if time.time() - t0 > budget_s - 80:
                break
            h, corr, imp = _sweep(h, n, dx, corr, 25, 61, 0.01)
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 7: Ultra-fine final polish
    if time.time() - t0 < budget_s - 20:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep(h, n, dx, corr, 30, 81, 0.005)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 8: Final SLSQP with very high alpha
    if time.time() - t0 < budget_s - 5:
        try:
            hc = _slsqp(best_h, n, dx, target, bounds, 10000, 2000)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
