import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h)) * (2.0 / n))

def _softmax_val(v, alpha):
    m = np.max(v)
    s = np.clip(alpha * (v - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softmax_weights(v, alpha):
    m = np.max(v)
    s = np.clip(alpha * (v - m), -500, 500)
    e = np.exp(s)
    return e / e.sum()

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
    return h

def _grad_softcor(h, n, alpha):
    c = _corr(h)
    w = _softmax_weights(c, alpha)
    g = np.zeros(n)
    j = np.arange(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-11:
            continue
        m = k - (n - 1)
        v1 = (j + m >= 0) & (j + m < n)
        if np.any(v1):
            g[v1] += wk * (1.0 - h[j[v1] + m])
        v2 = (j - m >= 0) & (j - m < n)
        if np.any(v2):
            g[v2] -= wk * h[j[v2] - m]
    return g

def _slsqp_opt(h0, n, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    def obj(x):
        return _softmax_val(_corr(x), alpha)
    def obj_grad(x):
        return _grad_softcor(x, n, alpha)
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h0

def _focused_slq(h0, n, topk, alpha, maxiter=200):
    """Add penalty on top-k correlations exceeding a threshold."""
    c = _corr(h0)
    thr = np.sort(c)[::-1][min(topk, len(c)-1)] * 0.999
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    mu = 50.0
    def obj(x):
        c2 = _corr(x)
        base = _softmax_val(c2, alpha)
        pen = mu * np.sum(np.maximum(c2 - thr, 0.0)**2)
        return base + pen
    def obj_grad(x):
        c2 = _corr(x)
        g = _grad_softcor(x, n, alpha)
        over = np.maximum(c2 - thr, 0.0)
        wg = np.zeros(len(c2))
        wg[over > 0] = 2.0 * mu * over[over > 0]
        j = np.arange(n)
        for k in range(len(c2)):
            wk = wg[k]
            if wk < 1e-12:
                continue
            m = k - (n - 1)
            v1 = (j + m >= 0) & (j + m < n)
            if np.any(v1):
                g[v1] += wk * (1.0 - x[j[v1] + m])
            v2 = (j - m >= 0) & (j - m < n)
            if np.any(v2):
                g[v2] -= wk * x[j[v2] - m]
        return g
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h0

def _block_coord(h, n, n_passes, budget_end, rng):
    best_c = _c5(h, n)
    for p in range(n_passes):
        if time.time() > budget_end:

        for _ in range(10):

            if abs(i - j) < blen:

            for d in [0.005, 0.01, 0.02, 0.04, -0.005, -0.01, -0.02, -0.04]:

                    continue
                c2 = _c5(h2, n)
                if c2 < best_c:
                    h = h2
                    best_c = c2
                    break
    return h, best_c
