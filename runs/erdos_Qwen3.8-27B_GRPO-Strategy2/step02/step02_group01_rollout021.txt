import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] = min(1.0, h[idx] + add)
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] = max(0.0, h[idx] - sub)
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n - 1, n - 1 + shift)
        if lo1 <= hi1:
            g[lo1:hi1+1] += wk * (1.0 - x[lo1-shift:hi1-shift+1])
        lo2 = max(0, -shift)
        hi2 = min(n - 1, n - 1 - shift)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= wk * x[lo2+shift:hi2+shift+1]
    return g

def _slsqp_step(h, dx, n, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)})
    fun = lambda x: _smooth_obj(x, alpha)
    jac = lambda x: _smooth_grad(x, alpha)
    res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _pair_descent(h, dx, n, iterations, rng, step_list):
    c5_cur = _c5(h, dx)
    for _ in range(iterations):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in step_list:
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur:
                c5_cur = c
                hi, hj = ni, nj
            else:
                h[i], h[j] = hi, hj
    return h, c5_cur

def _nelder_mead_active(h, n, dx, maxiter=200):
    """Nelder-Mead on interior variables only."""
    active = np.where((h > 0.01) & (h < 0.99))[0]
    if len(active) < 2:
        return h, _c5(h, dx)
    h_act = h[active].copy()
    target_act = h_act.sum()
    
    def obj(z):
        h_test = h.copy()
        # Adjust active vars to maintain sum
        h_test[active] = z
        diff = target_act - z.sum()
        if abs(diff) > 1e-12:

            h_test[active] += diff / len(active)
            h_test = np.clip(h_test, 0, 1)
        return _c5(h_test, dx)

                   options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
    h_new = h.copy()
    h_new[active] = res.x
    h_new = _repair(h_new, n)
    return h_new, _c5(h_new, dx)

From numpy docs: For 'full' mode, the output is a linear cross-correlation of length M+N-1, where M and N are the lengths of the two inputs (here both n).

The formula is: `c_k = sum_n a[n] * conj(b[n+k])` for k = -(N-1), ..., M-1

In numpy's implementation: `np.correlate(a, b, 'full')[k] = sum_i a[i] * b[i + k - (len(b)-1)]`

The standard definition: c(k) = sum_n a(n+k) * conj(b(n)), for k = -(n_b-1) to n_a-1

So c(0) = a0*b0 + a1*b1 + a2*b2
c(1) = a1*b0 + a2*b1
c(-1) = a0*b1 + a1*b2
c(2) = a2*b0
c(-2) = a0*b2

In numpy, the output is ordered: [c(-2), c(-1), c(0), c(1), c(2)]
So output[0] = c(-2) = a0*b2
    output[1] = c(-1) = a0*b1 + a1*b2
    output[2] = c(0) = a0*b0 + a1*b1 + a2*b2
    output[3] = c(1) = a1*b0 + a2*b1
    output[4] = c(2) = a2*b0

So output[k] = c(k-(n-1)) where c(shift) = sum_i a[i+shift]*b[i] for valid i.

For our case: a = h, b = 1-h
output[k] = sum_i h[i+shift] * (1-h[i]) where shift = k-(n-1), and i is valid if 0≤i<n and 0≤i+shift<n.

- h[j] appears as h[i+shift] when i+shift = j, i.e., i = j-shift. Valid if 0≤j-shift<n and 0≤j<n (latter always true).
  Contribution: (1-h[j-shift]) if 0≤j-shift<n
  
- h[j] appears in (1-h[i]) when i=j. Valid if 0≤j<n (always) and 0≤j+shift<n.
  Contribution: -h[j+shift] if 0≤j+shift<n

Total: d/dh[j] output[k] = (1-h[j-shift])*[0≤j-shift<n] - h[j+shift]*[0≤j+shift<n]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] = min(1.0, h[idx] + add)
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] = max(0.0, h[idx] - sub)
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-15:
            continue
        shift = k - (n - 1)
        lo1 = max(0, shift)
        hi1 = min(n - 1, n - 1 + shift)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += wk * (1.0 - x[lo1 - shift:hi1 - shift + 1])
        lo2 = max(0, -shift)
        hi2 = min(n - 1, n - 1 - shift)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= wk * x[lo2 + shift:hi2 + shift + 1]
    return g

def _slsqp(h, n, alpha, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - target,
            'jac': lambda x: np.ones(n)}
    fun = lambda x: _smooth_obj(x, alpha)
    jac = lambda x: _smooth_grad(x, alpha)
    res = minimize(fun, h, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _pair_sd(h, dx, n, iters, rng, steps):
    c5_cur = _c5(h, dx)
    for _ in range(iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in steps:
            ni, nj = hi + d, hj - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur:
                c5_cur = c
                hi, hj = ni, nj
            else:
                h[i], h[j] = hi, hj
    return h, c5_cur

def _nm_active(h, n, dx, maxiter=300):
    active = np.where((h > 0.02) & (h < 0.98))[0]
    if len(active) < 2:
        return h, _c5(h, dx)
    h_act = h[active].copy()
    target_act = h_act.sum()

    def obj(z):
        ht = h.copy()
        ht[active] = np.clip(z, 0.0, 1.0)
        diff = target_act - ht[active].sum()
        if abs(diff) > 1e-12:
            ht[active] += diff / len(active)
            ht = np.clip(ht, 0.0, 1.0)
        return _c5(ht, dx)

    res = minimize(obj, h_act, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-13})
    h_new = h.copy()
    h_new[active] = res.x
    h_new = _repair(h_new, n)
    return h_new, _c5(h_new, dx)

def _worst_lag_push(h, n, dx, iters, step):
    """Targeted descent: reduce the current worst lag specifically."""
    c5_cur = _c5(h, dx)
    for _ in range(iters):
        c = _corr(h)
        m_idx = int(np.argmax(c))
        shift = m_idx - (n - 1)
        g = np.zeros(n)
        lo1 = max(0, shift)
        hi1 = min(n - 1, n - 1 + shift)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += (1.0 - h[lo1 - shift:hi1 - shift + 1])
        lo2 = max(0, -shift)
        hi2 = min(n - 1, n - 1 - shift)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= h[lo2 + shift:hi2 + shift + 1]

        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur
