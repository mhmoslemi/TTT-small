                for mu in (80, 400):
                    hn = _opt_lbfgs(h, n, alpha, mu, 350)
                    _update(st, hn, n)
                    if _c5(hn, dx) < _c5(h, dx):
                        h = hn

            # Nelder-Mead on exact (non-smooth) objective
            if _tleft(st, 70) <= 0:
                break
            hn = _opt_nm(h, n, 300, 600)
            _update(st, hn, n)
            if _c5(hn, dx) < _c5(h, dx):
                h = hn

            # Powell's method
            if _tleft(st, 60) <= 0:
                break
            hn = _opt_powell(h, n, 500, 300)
            _update(st, hn, n)
            if _c5(hn, dx) < _c5(h, dx):
                h = hn

            # Another L-BFGS-B with very high alpha
            if _tleft(st, 50) <= 0:
                break
            hn = _opt_lbfgs(h, n, 8000, 800, 300)
            _update(st, hn, n)
            if _c5(hn, dx) < _c5(h, dx):
                h = hn

            # Nelder-Mead again after global methods
            if _tleft(st, 45) <= 0:
                break
            hn = _opt_nm(h, n, 600, 500)
            _update(st, hn, n)
            if _c5(hn, dx) < _c5(h, dx):
                h = hn

            # Coordinate descent (pair moves)
            if _tleft(st, 35) <= 0:
                break
            h, _ = _coord(h, n, dx, t0 + _tleft(st, 15), rng, 100)
            _update(st, h, n)

            # Block coordinate descent
            if _tleft(st, 30) <= 0:
                break
            h, _ = _block_coord(h, n, dx, t0 + _tleft(st, 12), rng, 60)
            _update(st, h, n)

    # Memetic: perturb best and re-optimize
    if st['bh'] is not None and _tleft(st, 60) > 0:
        n_b = st['bn']
        dx_b = 2.0 / n_b
        for _m in range(3):
            if _tleft(st, 30) <= 0:
                break
            hp = st['bh'] + np.random.RandomState(seed + 9000 + _m).randn(n_b) * 0.05
            hp = _repair(hp, n_b)
            hn = _opt_lbfgs(hp, n_b, 2000, 500, 300)
            _update(st, hn, n_b)
            if _c5(hn, dx_b) < _c5(st['bh'], dx_b):
                harness = hn
            else:
                harness = st['bh']
            hn = _opt_nm(harness, n_b, 400, 500)
            _update(st, hn, n_b)
            if _c5(hn, dx_b) < _c5(st['bh'], dx_b):
                st['bh'] = hn.copy()

    # Final polish
    if st['bh'] is not None and _tleft(st, 20) > 0:
        n_b = st['bn']
        dx_b = 2.0 / n_b
        hp, _ = _coord(st['bh'], n_b, dx_b, t0 + _tleft(st, 8), rng, 150)
        _update(st, hp, n_b)
        if _tleft(st, 12) > 0:
            hn = _opt_nm(st['bh'], st['bn'], 800, 400)
            _update(st, hn, st['bn'])

    if st['bh'] is None:
        n = 70
        bh = _repair(np.full(n, 0.5), n)
        return (bh, _c5(bh, 2.0 / n), n)

    st['bh'] = _repair(st['bh'], st['bn'])
    st['bc5'] = _c5(st['bh'], 2.0 / st['bn'])
    return (st['bh'], st['bc5'], st['bn'])
