            j = (i + rng.randint(1, n)) % n
            hi, hj = h[i], h[j]
            for d in np.concatenate([steps, -steps]):
                ni, nj = hi + d, hj - d
                if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best - 1e-12:
                    best = c
                    improved = True
                    break
                h[i], h[j] = hi, hj
        if not improved and it % 40 == 39:
            scale *= 0.5
            if scale < 0.01:
                break
    return h, best

def _wave_perturb(h, n, dx, rng, n_trials):
    h = h.copy()
    best = _c5(h, dx)
    for _ in range(n_trials):
        freq = rng.randint(1, 6)
        amp = rng.uniform(0.02, 0.12)
        ph = rng.uniform(0, 2 * np.pi)
        pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        pert -= pert.mean()
        h_new = _repair(h + pert, n)
        c = _c5(h_new, dx)
        if c < best:
            best = c
            h = h_new
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    parent = None
    parent_n = 0
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent)
        if parent_n > 0:
            parent = _repair(parent, parent_n)
    except NameError:
        pass

    ref_n = parent_n if parent_n > 0 else 70
    n_list = [ref_n]
    for dn in [-4, -2, 2, 4, 6, 8]:
        nn = ref_n + dn
        if 40 <= nn <= 120 and nn not in n_list:
            n_list.append(nn)

    total_opt_time = budget_s - 80
    alloc = {}
    alloc[ref_n] = 0.55 * total_opt_time
    others = [n for n in n_list if n != ref_n]
    if others:
        per_other = 0.45 * total_opt_time / len(others)
        for nn in others:
            alloc[nn] = min(per_other, 80.0)

    for n in n_list:
        t_end = t0 + alloc.get(n, 30) + (0 if n != ref_n else 0)
        if n == ref_n:
            t_end = t0 + budget_s - 100
        else:
            t_end = min(t_end, t0 + budget_s - 60)
        dx = 2.0 / n

        starts = []
        if parent is not None and parent_n == n:
            starts.append(parent.copy())
        elif parent is not None and n != parent_n:
            xp = np.linspace(0, 2, parent_n)
            xn = np.linspace(0, 2, n)
            starts.append(_repair(np.interp(xn, xp, parent), n))
        for k in range(3):
            starts.append(_make_start(n, seed + k * 997 + n * 13, k))
        starts.append(_make_start(n, seed + 555 + n, 3))

        for h0 in starts:
            if time.time() > t_end - 30:
                break
            h = h0.copy()
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

            h_deadline = t_end - 15
            for alpha in [100, 400, 1200, 3000, 6000]:
                if time.time() > h_deadline:
                    break
                h_new = _slsqp(h, n, alpha, 500)
                c_new = _c5(h_new, dx)
                if c_new < best_c5:
                    best_c5, best_h, best_n = c_new, h_new.copy(), n
                if c_new < _c5(h, dx):
                    h = h_new

            h, _ = _coord_descent(h, n, dx, 250, h_deadline, rng, 1.0)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h, best_n = c, h.copy(), n

            if time.time() < h_deadline - 10:
                h, _ = _wave_perturb(h, n, dx, rng, 25)
                c = _c5(h, dx)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h.copy(), n
                if time.time() < h_deadline:
                    h_new = _slsqp(h, n, 3000, 300)
                    c = _c5(h_new, dx)
                    if c < best_c5:
                        best_c5, best_h, best_n = c, h_new.copy(), n
                    h = h_new if c < _c5(h, dx) else h

    # Final polish
    if best_h is not None:
        dx_b = 2.0 / best_n
        h_pol = best_h.copy()
        h_pol, _ = _coord_descent(h_pol, best_n, dx_b, 400, t0 + budget_s - 15, rng, 0.4)
        c = _c5(h_pol, dx_b)
        if c < best_c5:
            best_c5, best_h = c, h_pol.copy()
        if time.time() < t0 + budget_s - 5:
            h_fin = _slsqp(best_h, best_n, 8000, 200)
            c = _c5(h_fin, 2.0 / best_n)
            if c < best_c5:
                best_c5, best_h = c, h_fin.copy()

    if best_h is None:
        best_h = _repair(np.full(70, 0.5), 70)
        best_c5 = _c5(best_h, 2.0 / 70)
        best_n = 70

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
