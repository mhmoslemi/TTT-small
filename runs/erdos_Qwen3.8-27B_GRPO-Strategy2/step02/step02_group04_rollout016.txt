def _project(z, n, target):
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            s = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += s if diff > 0 else -s
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def _grad_smooth(h, dx, alpha):
    n = len(h)
    corr = _corr(h)
    a = alpha * corr
    a -= a.max()
    w = np.exp(a)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[:n - m] += wk * dx * (1.0 - h[m:])
            grad[m:] -= wk * dx * h[:n - m]
        else:
            mm = -m
            grad[mm:] += wk * dx * (1.0 - h[:n - mm])
            grad[:n - mm] -= wk * dx * h[mm:]
    return grad

def _smooth_f(x, dx, alpha):
    corr = _corr(x)
    c = corr - corr.max()
    return (corr.max() + np.log(np.exp(alpha * c).sum()) / alpha) * dx

def _pgd_refine(h, n, dx, target, deadline, rng):
    best = _c5(h, dx)
    h = h.copy()
    alpha = 100.0
    step = 0.04
    stall = 0
    while time.time() < deadline:
        grad = _grad_smooth(h, dx, alpha)
        improved = False
        for _ in range(30):
            z = h - step * grad
            hn = _project(z, n, target)
            cn = _c5(hn, dx)
            if cn < best - 1e-14:
                best = cn
                h = hn
                improved = True
                step = min(step * 1.1, 0.12)
                break
            step *= 0.55
            if step < 1e-9:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            alpha = max(alpha * 0.75, 15.0)
            step = 0.04
            for _ in range(200):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = 0.008 * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                        break
                    h[i], h[j] = ni - d, nj + d
                if time.time() >= deadline:
                    break
            if stall > 40:
                break
    return h, best

def _slsqp_refine(h, n, dx, target, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    try:
        res = minimize(lambda x: _smooth_f(x, dx, alpha), h,
                       method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project(res.x, n, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _targeted_local(h, n, dx, target, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        corr = _corr(h)
        top_k = np.argsort(corr)[-5:]
        improved = False
        for k_star in top_k:
            if time.time() >= deadline:
                break
            m = k_star - (n - 1)
            if m >= 0:
                pairs = [(i, i + m) for i in range(n - m)]
            else:
                mm = -m
                pairs = [(i, i - mm) for i in range(mm, n)]
            contribs = []
            for (a, b) in pairs:
                c_val = h[a] * (1.0 - h[b])
                if c_val > 0.005:
                    contribs.append((c_val, a, b))
            contribs.sort(reverse=True)
            for (cv, a, b) in contribs[:30]:
                if time.time() >= deadline:
                    break
                for d in [0.004, 0.002, 0.001, -0.002, -0.004]:
                    na, nb = h[a] - d, h[b] + d
                    if na < 0 or na > 1 or nb < 0 or nb > 1:
                        continue
                    h[a], h[b] = na, nb
                    c = _c5(h, dx)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                        break
                    h[a], h[b] = na + d, nb - d
            if improved:
                break
        if not improved:
            for _ in range(1000):
                if time.time() >= deadline:
                    return h, best
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = 0.003 * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                        break
                    h[i], h[j] = ni - d, nj + d
            if not improved:
                break
    return h, best

def _fine_random(h, n, dx, target, rng, deadline, scale):
    best = _c5(h, dx)
    h = h.copy()
    no_improve = 0
    while time.time() < deadline:
        for _ in range(8000):
            if time.time() >= deadline:
                return h, best
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
                no_improve = 0
            else:
                h[i], h[j] = ni - d, nj + d
                no_improve += 1
        if no_improve > 20000:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _project(np.array(initial_h_values, dtype=float).copy(), n, target)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    t1 = min(t0 + 180, t0 + budget_s - 500)
    best_h, best_c5 = _pgd_refine(best_h, n, dx, target, t1, rng)

    t2 = min(t0 + 330, t0 + budget_s - 380)
    for a in [60, 200, 500]:
        if time.time() >= t2:
            break
        best_h, best_c5 = _slsqp_refine(best_h, n, dx, target, a, 300, t2)

    t3 = min(t0 + 530, t0 + budget_s - 250)
    best_h, best_c5 = _targeted_local(best_h, n, dx, target, rng, t3)

    t4 = min(t0 + 680, t0 + budget_s - 160)
    best_h, best_c5 = _pgd_refine(best_h, n, dx, target, t4, rng)

    t5 = min(t0 + 820, t0 + budget_s - 100)
    for a in [300, 800, 2000]:
        if time.time() >= t5:
            break
        best_h, best_c5 = _slsqp_refine(best_h, n, dx, target, a, 400, t5)

    t6 = min(t0 + 960, t0 + budget_s - 40)
    for sc in [0.004, 0.002, 0.001, 0.0005]:
        if time.time() >= t6:
            break
        best_h, best_c5 = _fine_random(best_h, n, dx, target, rng, t6, sc)

    t_end = t0 + budget_s - 10
    if time.time() < t_end - 20:
        best_h, best_c5 = _slsqp_refine(best_h, n, dx, target, 4000, 600, t_end - 20)
    if time.time() < t_end - 3:
        best_h, best_c5 = _targeted_local(best_h, n, dx, target, rng, t_end - 3)

    best_h = _project(best_h, n, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
