def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(h, target):
    h = h.astype(float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid

    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(h, alpha):
    corr = _corr(h)
    m = np.max(corr)
    s = np.exp(np.clip(alpha * (corr - m), -500, 500))
    w = s / s.sum()
    n = len(h)

                t += one_minus_h[p + d]
            if 0 <= p - d < n:
                t -= h[p - d]
            g += w[di] * t

    return alpha * grad
