    n = len(h)
    hx = 1.0 - h
    # Coefficients for each lag
    c = np.correlate(h, hx, mode='full')  # length 2n-1
    coeffs = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        coeff = 0.0
        # h[i] contribution: appears as h[i]*(1-h[i+lag]) when i+lag in [0,n)
        if 0 <= i+lag < n:
            coeff += 1.0 - h[i+lag]
        # h[i] appears in (1-h[i]) when i = j+lag, i.e., the term h[i-lag]*(1-h[i])
        if 0 <= i-lag < n:
            coeff -= h[i-lag]
        # h[j] contribution (h[j] changes by -d)
        if 0 <= j+lag < n:
            coeff -= (1.0 - h[j+lag])  # because h[j]→h[j]-d, so -(h[j]-d)*(1-h[j+lag]) change is -d*(1-h[j+lag])
        if 0 <= j-lag < n:
            coeff += h[j-lag]  # because h[j-lag]*(1-(h[j]-d)) = h[j-lag]*(1-h[j]+d), change is +d*h[j-lag]
        coeffs[k] = coeff
    
    # Feasible range for d
    d_lo = max(-h[i], h[j]-1.0)
    d_hi = min(1.0-h[i], h[j])
    if d_lo >= d_hi:
        return 0.0
    
    # Find min of max(c_k + d*coeff_k) for d in [d_lo, d_hi]
    # Evaluate at breakpoints
    candidates = [d_lo, d_hi]
    for a in range(2*n-1):
        for b in range(a+1, 2*n-1):
            dc = coeffs[a] - coeffs[b]
            if abs(dc) > 1e-15:
                dp = (c[b] - c[a]) / dc
                if d_lo < dp < d_hi:
                    candidates.append(dp)
    
    best_d = 0.0
    best_val = max(c + 0.0*coeffs)  # just c
    for d in candidates:
        val = np.max(c + d * coeffs)
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d

    hi = float(y.max()) + 1.0

                diff -= a
            else:

                x[k] -= a

def _exact_pair(h, i, j):
    n = len(h)
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    
    coeffs = np.zeros(m)
    for k in range(m):
        lag = k - (n - 1)
        co = 0.0
        if 0 <= i + lag < n:
            co += 1.0 - h[i + lag]
        if 0 <= i - lag < n:
            co -= h[i - lag]
        if 0 <= j + lag < n:
            co -= 1.0 - h[j + lag]
        if 0 <= j - lag < n:
            co += h[j - lag]
        coeffs[k] = co
    
    d_lo = max(-h[i], h[j] - 1.0) + 1e-14
    d_hi = min(1.0 - h[i], h[j]) - 1e-14
    if d_lo >= d_hi:
        return 0.0
    
    ca = c[:, None]
    cb = c[None, :]
    dc = coeffs[:, None] - coeffs[None, :]
    with np.errstate(divide='ignore', invalid='ignore'):
        dp = (cb - ca) / dc
    valid = (np.abs(dc) > 1e-15) & (d_lo < dp) & (dp < d_hi) & np.isfinite(dp)
    cand = dp[valid]
    extra = [d_lo, d_hi]
    if d_lo <= 0.0 <= d_hi:
        extra.append(0.0)
    cands = np.concatenate([cand, np.array(extra)])
    vals = np.max(c[None, :] + cands[:, None] * coeffs[None, :], axis=1)
    return float(cands[np.argmin(vals)])

    bc = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = _exact_pair(h, i, j)

        h[j] -= d
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        c = _c5(h, dx)
        if c < bc - 1e-14:
            bc = c
        else:
            h[i] -= d
            h[j] += d
    return h, bc
