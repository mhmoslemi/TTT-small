import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _obj_lse(h, alpha):
    corr = _corr(h)
    m = corr.max()
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_tops(h, topk):
    corr = _corr(h)
    idx = np.argsort(corr)[::-1][:topk]
    return np.sum(corr[idx])

def _slsqp_lse(h0, n, alpha, maxiter):
    target = n / 2.0
    def obj(x):
        return _obj_lse(x, alpha)
    def con(x):
        return np.sum(x) - target
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h0

    target = n / 2.0
    def obj(x):
        return _obj_tops(x, topk)

                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h0

        res = minimize(obj_pen, h, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12,
                                'fatol': 1e-14, 'adaptive': True})
        h_new = _repair(res.x, n)
        c_new = _c5(h_new, dx)
        if c_new < best:
            best = c_new
            h = h_new
    except:
        pass
    return h, best

def _fine_coord(h, n, dx, budget_end, rng, n_rounds=3000):
    h = h.copy()
    best = _c5(h, dx)
    deltas = [1e-4, 3e-4, 1e-3, 3e-3, 1e-2,
              -1e-4, -3e-4, -1e-3, -3e-3, -1e-2]
    for _ in range(n_rounds):

            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in deltas:
            ni, nj = hi + d, hj - d
            if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best - 1e-16:
                best = c
            else:
                h[i], h[j] = hi, hj
    return h, best
