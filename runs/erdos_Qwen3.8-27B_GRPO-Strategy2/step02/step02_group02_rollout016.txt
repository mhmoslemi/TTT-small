                    break
                if fc < fd:
                    b = d1; d1 = c1; fd = fc
                    c1 = b - gr * (b - a)
                    fc = _eval_pair(h, i, j, c1, dx)
                else:
                    a = c1; c1 = d1; fc = fd
                    d1 = a + gr * (b - a)
                    fd = _eval_pair(h, i, j, d1, dx)
            dopt = 0.5 * (a + b)
            cn = _eval_pair(h, i, j, dopt, dx)
            if cn < bc - 1e-13:
                h[i] += dopt
                h[j] -= dopt
                bc = cn
                imp += 1
        if imp == 0:
            break
    return h, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')
    tb = budget_s - 10

    # Phase 1: Optimize initial_h_values deeply (n=70)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    tgt0 = n0 / 2.0
    h0 = _proj(np.array(initial_h_values, dtype=float), tgt0)
    c0 = _c5(h0, dx0)
    if c0 < best_c:
        best_c = c0
        best_h = h0.copy()

    for a in [50, 200, 800, 3000, 8000]:
        if time.time() - t0 > tb - 300:
            break
        h_f, c_f = _fista(h0, a, tgt0, 2000, t0, tb - 200)
        if c_f < c0:
            c0 = c_f
            h0 = h_f
        h_g, c_g = _gscd(h0, dx0, tgt0, 100, 1e-9, t0, tb - 100)
        if c_g < c0:
            c0 = c_g
            h0 = h_g
        h_a, c_a = _adj_cd(h0, dx0, 50, t0, tb - 50)
        if c_a < c0:
            c0 = c_a
            h0 = h_a
        h0 = _proj(h0, tgt0)
        c0 = _c5(h0, dx0)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()

    # Deep golden-section CD on n=70
    h_deep, c_deep = _gscd(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 500, 1e-10, t0, tb - 200)
    if c_deep < best_c:
        best_c = c_deep
        best_h = h_deep.copy()

    # Phase 2: Try n=100 from scratch
    if time.time() - t0 < tb - 250:
        n1 = 100
        dx1 = 2.0 / n1
        tgt1 = n1 / 2.0
        xgrid = np.linspace(0.0, 2.0, n1, endpoint=False)
        starts = []
        starts.append(np.full(n1, 0.5))
        for f in [1, 2, 3, 4]:
            starts.append(_proj(0.5 + 0.4 * np.sin(2.0 * np.pi * f * xgrid / 2.0), tgt1))
        for w in [0.25, 0.35, 0.45, 0.55]:
            s = np.zeros(n1)
            mask = (xgrid < w) | (xgrid > 2.0 - w)
            s[mask] = 1.0 / (2.0 * w)
            starts.append(_proj(s, tgt1))
        starts.append(_proj(rng.rand(n1), tgt1))
        starts.append(_proj(0.5 + 0.3 * rng.rand(n1), tgt1))

        for s in starts:
            if time.time() - t0 > tb - 150:
                break
            h_s = _proj(s, tgt1)
            for a in [200, 1000]:
                if time.time() - t0 > tb - 150:
                    break
                h_s, c_s = _fista(h_s, a, tgt1, 1500, t0, tb - 100)
            h_c, c_c = _gscd(h_s, dx1, tgt1, 80, 1e-9, t0, tb - 50)
            h_a, c_a = _adj_cd(h_c, dx1, 30, t0, tb - 30)
            if c_a < best_c:
                best_c = c_a
                best_h = h_a.copy()

    # Phase 3: Try n=140
    if time.time() - t0 < tb - 150:
        n2 = 140
        dx2 = 2.0 / n2
        tgt2 = n2 / 2.0
        xgrid2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        starts2 = [np.full(n2, 0.5)]
        for f in [1, 2, 3]:
            starts2.append(_proj(0.5 + 0.3 * np.sin(2.0 * np.pi * f * xgrid2 / 2.0), tgt2))
        starts2.append(_proj(rng.rand(n2), tgt2))
        for s in starts2:
            if time.time() - t0 > tb - 80:
                break
            h_s = _proj(s, tgt2)
            h_s, _ = _fista(h_s, 500, tgt2, 1000, t0, tb - 50)
            h_c, c_c = _gscd(h_s, dx2, tgt2, 40, 1e-8, t0, tb - 30)
            if c_c < best_c:
                best_c = c_c
                best_h = h_c.copy()

    # Phase 4: Final intensive polish on best
    nb = len(best_h)
    dx_b = 2.0 / nb
    tgt_b = nb / 2.0
    if time.time() - t0 < tb - 30:
        hp, cp = _gscd(best_h, dx_b, tgt_b, 300, 1e-10, t0, tb - 20)
        if cp < best_c:
            best_c = cp
            best_h = hp
    if time.time() - t0 < tb - 15:
        ha, ca = _adj_cd(best_h, dx_b, 100, t0, tb - 5)
        if ca < best_c:
            best_c = ca
            best_h = ha

    # SLSQP polish
    if time.time() - t0 < tb - 8 and nb >= 50:
        bnds = [(0.0, 1.0)] * nb
        try:
            r = minimize(_slsqp_obj, best_h, args=(5000,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt_b,)}],
                         options={'maxiter': 400, 'ftol': 1e-15})
            hf = _proj(r.x, tgt_b)
            cf = _c5(hf, dx_b)
            if cf < best_c:
                best_c = cf
                best_h = hf
        except Exception:
            pass

    # Final CD after SLSQP
    if time.time() - t0 < tb - 3:
        hf, cf = _gscd(best_h, dx_b, tgt_b, 100, 1e-10, t0, budget_s - 2)
        if cf < best_c:
            best_c = cf
            best_h = hf

    best_h = _proj(best_h, len(best_h) / 2.0)
    best_c = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c, len(best_h))
