import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smobj(h, alpha, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    es = np.exp(s)
    return m + np.log(es.sum()) / alpha

def _smobj_grad(h, alpha, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    es = np.exp(s)
    w = es / es.sum()
    obj = m + np.log(es.sum()) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, lag + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[np.arange(lo1, hi1) - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[np.arange(lo2, hi2) + lag]
    return obj, grad

def _slsqp(h0, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    
    def obj(x):
        return _smobj(x, alpha, n)
    
    def obj_grad(x):
        return _smobj_grad(x, alpha, n)
    
    try:
        res = minimize(obj_grad, h0, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0.copy()

    """One pass of ternary-search coordinate descent over all pairs"""
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            orig_i, orig_j = h[i], h[j]
            best_t, best_c = orig_i, _c5(h, dx)
            a, b = lo, hi
            for _ in range(n_iter):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[i] = m1; h[j] = s - m1
                c1 = _c5(h, dx)
                h[i] = m2; h[j] = s - m2
                c2 = _c5(h, dx)
                if c1 < c2:
                    b = m2
                else:
                    a = m1
            h[i] = (a + b) / 2.0
            h[j] = s - h[i]
            c_mid = _c5(h, dx)
            ht = (a + b) / 2.0
            h[i] = ht; h[j] = s - ht
            c_mid = _c5(h, dx)
            if c_mid < best_c:
                best_c, best_t = c_mid, ht
                improved = True
            else:
                h[i] = orig_i
                h[j] = orig_j
            h[i] = best_t
            h[j] = s - best_t
    return improved

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smobj(h, alpha, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    es = np.exp(s)
    return m + np.log(es.sum()) / alpha

def _smobj_grad(h, alpha, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    es = np.exp(s)
    w = es / es.sum()
    obj = m + np.log(es.sum()) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        lo1, hi1 = max(0, lag), min(n, lag + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[np.arange(lo1, hi1) - lag])
        lo2, hi2 = max(0, -lag), min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[np.arange(lo2, hi2) + lag]
    return obj, grad

def _slsqp(h0, n, alpha, maxiter=600, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(_smobj_grad, h0, args=(alpha, n), jac=True,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0.copy()

def _cd_pass(h, n, dx, n_grid, refine=True):
    hi_c5 = _c5(h, dx)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            orig_i, orig_j = h[i], h[j]
            best_t, best_c = orig_i, hi_c5
            span = hi - lo
            for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                h[i] = t
                h[j] = hsum - t
                c = _c5(h, dx)
                if c < best_c:
                    best_c = c
                    best_t = t
            if refine and best_t != orig_i:
                lo2 = max(lo, best_t - span / n_grid)
                hi2 = min(hi, best_t + span / n_grid)
                for t in np.linspace(lo2, hi2, 15):
                    h[i] = t
                    h[j] = hsum - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c = c
                        best_t = t
            if best_c < hi_c5 - 1e-13:
                h[i] = best_t
                h[j] = hsum - best_t
                hi_c5 = best_c
                improved = True
            else:
                h[i] = orig_i
                h[j] = orig_j
    return h, hi_c5, improved

def _cd_loop(h, n, dx, t_limit, t_start, n_grid, max_passes=1000, refine=True):
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        h, c5, imp = _cd_pass(h, n, dx, n_grid, refine)
        if not imp:
            break
    return h, _c5(h, dx)
