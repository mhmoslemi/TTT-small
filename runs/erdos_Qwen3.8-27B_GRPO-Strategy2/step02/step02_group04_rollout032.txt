                for _ in range(30):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    h[i] += m1; h[j] -= m1
                    f1 = _c5(h, dx)
                    h[i] -= m1; h[j] += m1
                    h[i] += m2; h[j] -= m2
                    f2 = _c5(h, dx)
                    h[i] -= m2; h[j] += m2
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                t_opt = (a + b) / 2.0
                h[i] += t_opt
                h[j] -= t_opt
                best_c5 = _c5(h, dx)
                improved = True
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 35.0

def _eq_fun(x):
    return np.sum(x) - _TARGET

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(corr, alpha):
    m = float(np.max(corr))
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, dx, bounds, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    con = {'type': 'eq', 'fun': _eq_fun}
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[con], options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _pair_optimize(h, i, j, dx, best_c5):
    """Optimize transfer t: h[i]+=t, h[j]-=t. Returns (t_opt, improved)."""
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-13:
        return 0.0, False
    # Coarse grid: 30 points
    grid = np.linspace(t_lo, t_hi, 30)
    best_t, best_f = 0.0, best_c5
    for t in grid:
        if abs(t) < 1e-14:
            continue
        h[i] += t; h[j] -= t
        f = _c5(h, dx)
        h[i] -= t; h[j] += t
        if f < best_f:
            best_f = f; best_t = t
    if best_f >= best_c5 - 1e-15:
        return 0.0, False
    # Fine grid around best: 20 points in +/-1 grid spacing
    span = (t_hi - t_lo) / 30.0
    a = max(t_lo, best_t - span)
    b = min(t_hi, best_t + span)
    grid2 = np.linspace(a, b, 20)
    for t in grid2:
        if abs(t) < 1e-14:
            continue
        h[i] += t; h[j] -= t
        f = _c5(h, dx)
        h[i] -= t; h[j] += t
        if f < best_f:
            best_f = f; best_t = t
    # Apply best
    if abs(best_t) > 1e-15:
        h[i] += best_t; h[j] -= best_t
    return best_t, True

def _coord_descent(h, n, dx, rng, deadline, n_rounds=500, pairs_per_round=120):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for _r in range(n_rounds):
        if time.time() >= deadline:
            break
        npairs = pairs_per_round
        for _ in range(npairs):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            _pair_optimize(h, i, j, dx, best_c5)
            c5_now = _c5(h, dx)
            if c5_now < best_c5:
                best_c5 = c5_now
    return h, best_c5

def _systematic_cd(h, n, dx, rng, deadline, n_cycles=50):
    """Cycle through all pairs in random order."""
    best_c5 = _c5(h, dx)
    h = h.copy()
    idx = np.arange(n)
    for _cyc in range(n_cycles):
        if time.time() >= deadline:
            break
        order_i = rng.permutation(n)
        order_j = rng.permutation(n)
        for _ in range(n):
            if time.time() >= deadline:
                break
            i = order_i[_]; j = order_j[_]
            if i == j:
                continue
            _pair_optimize(h, i, j, dx, best_c5)
        best_c5 = _c5(h, dx)
    return h, best_c5

def _sim_anneal(h, n, dx, rng, deadline, t0=0.005, t_mult=0.85, n_iter=3000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    curr_c5 = best_c5
    T = t0
    for _ in range(n_iter):
        if time.time() >= deadline:
            break
        if _ % 50 == 0:
            T *= t_mult
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        t = 0.03 * (2.0 * rng.random() - 1.0)
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        t = max(t_lo, min(t_hi, t))
        if abs(t) < 1e-12:
            continue
        h[i] += t; h[j] -= t
        new_c5 = _c5(h, dx)
        if new_c5 < curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-15)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            h[i] -= t; h[j] += t
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t_start)

    n = len(initial_h_values)
    _TARGET = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-start (60s)
    t_end = t_start + 60
    for alpha in [50, 150]:
        if time.time() > t_end or rem() < 400:
            break
        best_h, best_c5 = _slsqp(best_h, n, dx, bounds, alpha, 400, t_end)

    # Phase 2: Coordinate descent + systematic cycling (250s)
    t_end = t_start + 310
    best_h, best_c5 = _coord_descent(best_h, n, dx, rng, t_end, 300, 150)
    if rem() > 300:
        t_end2 = time.time() + min(100, rem() - 250)
        best_h, best_c5 = _systematic_cd(best_h, n, dx, rng, t_end2, 40)

    # Phase 3: SLSQP multi-restart with perturbation (250s)
    t_end = t_start + 560
    for alpha in [300, 800, 2000, 5000]:
        if time.time() > t_end or rem() < 250:
            break
        for frac in [0.0, 0.015, 0.04]:
            if time.time() > t_end or rem() < 250:
                break
            start = best_h.copy()
            if frac > 0:
                start = _repair(best_h + frac * rng.randn(n), n)
            hc, c5c = _slsqp(start, n, dx, bounds, alpha, 800, t_end)
            if c5c < best_c5:
                best_h = hc; best_c5 = c5c

    # Phase 4: Simulated annealing for escape (100s)
    if rem() > 200:
        t_end = time.time() + min(100, rem() - 120)
        ha, ca = _sim_anneal(best_h, n, dx, rng, t_end, t0=0.008, n_iter=2000)
        if ca < best_c5:
            best_h = ha; best_c5 = ca
        # Quick refine after SA
        best_h, best_c5 = _coord_descent(best_h, n, dx, rng, time.time() + 10, 30, 200)

    # Phase 5: Intensive coordinate descent (180s)
    if rem() > 100:
        t_end = time.time() + min(180, rem() - 50)
        best_h, best_c5 = _coord_descent(best_h, n, dx, rng, t_end, 400, 200)
        best_h, best_c5 = _systematic_cd(best_h, n, dx, rng, time.time() + min(30, rem()-40), 30)

    # Phase 6: Final SLSQP high-alpha polish (60s)
    if rem() > 40:
        t_end = time.time() + min(60, rem() - 20)
        for alpha in [10000, 30000]:
            if time.time() > t_end:
                break
            best_h, best_c5 = _slsqp(best_h, n, dx, bounds, alpha, 1500, t_end)

    # Phase 7: Ultra-fine local search
    if rem() > 20:
        t_end = time.time() + min(20, rem() - 5)
        best_h, best_c5 = _coord_descent(best_h, n, dx, rng, t_end, 100, 300)

    # Final repair
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
