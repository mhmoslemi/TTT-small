```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 0) & (h < 1)
        if diff > 0:
            order = np.where(mask, 1.0 - h, -1.0)
        else:
            order = np.where(mask, h, -1.0)
        idx = np.argsort(order)[::-1]
        idx = idx[:n]
        per = diff / len(idx)
        hx = h.copy()
        for j in idx:
            if hx[j] > 0 and hx[j] < 1:
                hx[j] += per
        hx = np.clip(hx, 0, 1)
        h = hx
    # Exact fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _run_lbfgs(h, n, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smooth_obj(x, alpha)
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h
```

```python
        val = 0.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_max(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_max(x, alpha)
    def eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, n, alpha, pen, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_max(x, alpha) + pen * (np.sum(x) - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        return _repair(res.x, n)
    except Exception:
        return h

def _pairwise_cd(h, n, budget_end, rng, max_sweeps=200):
    dx = 2.0 / n
    m = 2 * n - 1
    shifts = np.arange(m) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_val = float(np.max(corr))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > budget_end:
            break
        improved = 0
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
        recompute = 0
        for (i, j) in pairs:
            if time.time() > budget_end:
                break
            a, b = h[i], h[j]
            d_lo = max(-a + 1e-9, -(1.0 - b) + 1e-9)
            d_hi = min(1.0 - a - 1e-9, b - 1e-9)
            if d_lo >= d_hi:
                continue
            dc = np.zeros(m)
            ip = i + shifts
            im_s = i - shifts
            vp = (ip >= 0) & (ip < n)
            vm = (im_s >= 0) & (im_s < n)
            dc[vp] += 1.0 - h[ip[vp]]
            dc[vm] += -h[im_s[vm]]
            jp = j + shifts
            jm_s = j - shifts
            vjp = (jp >= 0) & (jp < n)
            vjm = (jm_s >= 0) & (jm_s < n)
            dc[vjp] += -(1.0 - h[jp[vjp]])
            dc[vjm] += h[jm_s[vjm]]
            lo, hi = d_lo, d_hi
            for _ in range(35):
                mid = (lo + hi) * 0.5
                vals = corr + dc * mid
                mx = np.max(vals)
                act = vals >= mx - 1e-12
                if len(act) == 0:
                    break
                ms = np.min(dc[act])
                Ms = np.max(dc[act])
                if ms >= -1e-13:
                    hi = mid
                elif Ms <= 1e-13:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            d_opt = (lo + hi) * 0.5
            if abs(d_opt) < 1e-13:
                continue
            h[i] = np.clip(a + d_opt, 0.0, 1.0)
            h[j] = np.clip(b - d_opt, 0.0, 1.0)
            corr += dc * d_opt
            recompute += 1
            if recompute % 80 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full')
            new_mx = np.max(corr)
            if new_mx < best_val - 1e-12:
                best_val = float(new_mx)
                improved += 1
        if improved == 0:
            no_improve += 1
            if no_improve >= 3:
                break
            kick = 0.02 / (no_improve + 1)
            idx = rng.choice(n, size=max(2, n // 10), replace=False)
            for k in idx:
                h[k] = np.clip(h[k] + kick * (rng.rand() - 0.5), 0, 1)
            h = _repair(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            best_val = float(np.max(corr))
        else:
            no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(corr) * dx)

def _make_start(n, seed, kind, ref_h=None):
    rng = np.random.RandomState(seed)
    if ref_h is not None and len(ref_h) == n:
        h = np.array(ref_h, dtype=float).copy()
        h += 0.05 * rng.randn(n)
        return _repair(h, n)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0)
        h += 0.15 * rng.randn(n)
    elif kind == 2:
        h = np.zeros(n)
        a, b = n // 3, 2 * n // 3
        h[:a] = 1.0
        h[b:] = 1.0
        h += 0.1 * rng.randn(n)
    elif kind == 3:
        freq = rng.randint(2, 7)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h += 0.08 * rng.randn(n)
    elif kind == 4:
        h = np.zeros(n)
        block = max(1, n // 12)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.5 * h + 0.3 * rng.randn(n)
    else:
        h = 0.5 + 0.25 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    def _try(h, n):
        nonlocal best_h, best_c5, best_n
        dx = 2.0 / n
        c5 = _compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    init_arr = None
    init_n = 0
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        init_n = len(init_arr)
    except NameError:
        pass

    n_list = []
    if init_n > 0:
        n_list.append(init_n)
    for nn in [60, 70, 80, 90, 100, 120]:
        if nn not in n_list:
            n_list.append(nn)

    time_per_n = max(60.0, (budget_s - 120.0) / len(n_list))

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break
        dx = 2.0 / n
        n_starts = 3
        starts = []
        if init_arr is not None and init_n == n:
            starts.append(_repair(init_arr.copy(), n))
        for kind in range(5):
            starts.append(_make_start(n, seed + kind * 7919 + n * 13, kind))
        while len(starts) < n_starts + 2:
            starts.append(_make_start(n, seed + rng.randint(1000000), 5))

        for si, h0 in enumerate(starts):
            t_start = time.time()
            if t_start - t0 > budget_s - 60:
                break
            h = h0.copy()
            _try(h, n)
            elapsed_in_n = time.time() - t0
            remain = time_per_n - (elapsed_in_n - (time.time() - t0))
            slot_end = t0 + min(time_per_n * (si // n_starts + 1), budget_s - 80)

            for alpha in [300, 1000, 3000]:
                if time.time() > slot_end - 30:
                    break
                h_new = _slsqp(h, n, alpha, 200)
                _try(h_new, n)
                if _compute_c5(h_new, dx) < _compute_c5(h, dx):
                    h = h_new
            for alpha in [5000, 10000]:
                if time.time() > slot_end - 20:
                    break
                h_new = _slsqp(h, n, alpha, 150)
                _try(h_new, n)
                if _compute_c5(h_new, dx) < _compute_c5(h, dx):
                    h = h_new
            h_cd, c5_cd = _pairwise_cd(h, n, slot_end - 10, rng, max_sweeps=300)
            _try(h_cd, n)

    if best_h is not None:
        dx_b = 2.0 / best_n
        h_pol, _ = _pairwise_cd(best_h, best_n, t0 + budget_s - 20, rng, max_sweeps=500)
        _try(h_pol, best_n)
        if time.time() - t0 < budget_s - 15:
            h2 = _slsqp(best_h, best_n, 20000, 100)
            _try(h2, best_n)
            h3, _ = _pairwise_cd(h2, best_n, t0 + budget_s - 5, rng, max_sweeps=300)
            _try(h3, best_n)

    if best_h is None:
        n = 70
        best_h = _repair(np.full(n, 0.5), n)
        best_c5 = _compute_c5(best_h, 2.0 / n)
        best_n = n

    best_h = _repair(best_h, best_n)
    best_c5 = _compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
```
