import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lse_obj(h, alpha, dx):
    corr = _corr(h)
    m = np.max(corr)
    return (m + np.sum(np.exp(alpha*(corr - m))) / alpha) * dx

def _lse_grad(h, alpha, dx):
    corr = _corr(h)
    m = np.max(corr)
    w = np.exp(alpha*(corr - m)) / np.sum(np.exp(alpha*(corr - m)))
    # grad of corr[k] w.r.t. h[p]:
    # corr[k] = sum_i h[i+m_k]*(1-h[i]) where m_k = k-(n-1)
    n = len(h)

```python
def _project(h, target):
    # Project h onto {x: sum(x)=target, 0<=x<=1}
    # Find lambda: sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```
