from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    c = _corr(h)
    return float(np.max(c)) * dx

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

    for i in range(n):

            bi1 = np.argmin(c5s1)

**Alternative: Use the structure of the problem**

The Erdős minimum overlap problem has a known structure. The optimal function is believed to be close to a "ternary" function (taking values in {0, 1/2, 1}) or a specific step function. 

For the 1D case, the optimal function might have a periodic or quasi-periodic structure. Let me think about what structure minimizes the maximum overlap.

The overlap at lag k is: ∫ h(x)(1-h(x+k)) dx

For a step function with n points, this is: dx * sum_i h[i] * (1-h[i+k])

To minimize the maximum, we want all lags to have similar (and small) overlap.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _topk_smooth(c, topk, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    idx = np.argsort(c)
    thresh = c[idx[-topk]]
    pen = np.clip(thresh - c, 0.0, None) * alpha * 0.5
    s = s - pen
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _smax_obj(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _tk_obj(x, topk, alpha):
    return _topk_smooth(_corr(x), topk, alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    i1 = p - k; m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = p + k; m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _sweep(h, n, dx, corr, nc, nf, fw):
    imp = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi = h[i]; hj = h[j]
            dlo = max(-hi, hj - 1.0)
            dhi = min(1.0 - hi, hj)
            if dhi - dlo < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dlo, dhi, nc)
            nc1 = corr[None, :] + ds[:, None] * gd[None, :]
            v1 = np.max(nc1, axis=1) * dx
            b1 = int(np.argmin(v1))
            cm = float(np.max(corr)) * dx
            if v1[b1] < cm - 1e-14:
                db = ds[b1]
                if nf > 2 and fw > 0:
                    lo = max(dlo, db - fw); hi2 = min(dhi, db + fw)
                    ds2 = np.linspace(lo, hi2, nf)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]
                    v2 = np.max(nc2, axis=1) * dx
                    b2 = int(np.argmin(v2)); db = ds2[b2]
                h[i] += db; h[j] -= db
                corr = corr + db * gd
                imp = True
    return h, corr, imp

def _slsqp(h, n, tgt, bounds, objfn, args, maxit):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (tgt,)}]
    try:
        r = minimize(objfn, h, args=args, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

def _block_swap(h, n, rng, ns):
    h2 = h.copy()
    for _ in range(ns):
        i = rng.randint(0, n); j = rng.randint(0, n)
        if i != j:
            h2[i], h2[j] = h2[j], h2[i]
    return h2

def _interp_h(h, n_old, n_new):
    dxo = 2.0 / n_old; dxn = 2.0 / n_new
    xo = (np.arange(n_old) + 0.5) * dxo
    xn = (np.arange(n_new) + 0.5) * dxn
    return _repair(np.interp(xn, xo, h), n_new)
