= sum_{j=s}^{n-1} h0[j] - sum_{j=s}^{n-1} h0[j-s]*h0[j] - sum_{j=0}^{n-1-s} h0[j+s]*h0[j]
= (sum_{j=s}^{n-1} h0[j]) - 2*sum_{j=0}^{n-1-s} h0[j]*h0[j+s]

And c_s(h0) = sum_{j=0}^{n-1-s} h0[j+s]*(1-h0[j]) = sum_{j=0}^{n-1-s} h0[j+s] - sum_{j=0}^{n-1-s} h0[j+s]*h0[j]

So c_s(h0) - g_s^T h0 = [sum_{j=0}^{n-1-s} h0[j+s] - sum h0[j+s]h0[j]] - [sum_{j=s}^{n-1} h0[j] - 2*sum h0[j]h0[j+s]]
= [sum_{j=0}^{n-1-s} h0[j+s] - sum_{j=s}^{n-1} h0[j]] + sum h0[j]h0[j+s]

Note: sum_{j=0}^{n-1-s} h0[j+s] = sum_{m=s}^{n-1} h0[m] = sum_{j=s}^{n-1} h0[j]

So the first bracket is 0!

c_s(h0) - g_s^T h0 = sum_{j=0}^{n-1-s} h0[j]*h0[j+s] ≥ 0 ✓

Great! So c_s(h0) ≥ g_s^T h0 always (for s≥0), meaning the LP is always feasible at h0 (with t = c_s(h0) being a valid upper bound). The LP will find t* ≤ max_s c_s(h0), which is what we want.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_lag(s, h, n):
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        sp = -s
        if n - sp > 0:
            g[:n - sp] += 1.0 - h[sp:]
            g[sp:] -= h[:n - sp]
    return g

def _lp_solve(h, n, active_idx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(active_idx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, idx in enumerate(active_idx):
        s = idx - (n - 1)
        g = _grad_lag(s, h, n)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g @ h - corr[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n].copy(), res.x[n]
    return None, None

def _benders(h, n, dx, t_start, t_limit, M, max_iter=500, seed=42):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    rng = np.random.RandomState(seed)
    for it in range(max_iter):
        if time.time() - t_start > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(corr.max() * dx)
        if c5_now + 1e-12 < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if it % 7 == 0:
            shuffle = rng.permutation(2 * n - 1)[:max(3, M // 5)]
            active_idx = np.unique(np.concatenate([np.argsort(corr)[-M:], shuffle]))[:M]
        else:
            active_idx = np.argsort(corr)[-M:]
        h_lp, _ = _lp_solve(h, n, active_idx)
        if h_lp is None:
            break
        improved = False
        alphas = [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]
        for alpha in alphas:
            h_try = (1.0 - alpha) * h + alpha * h_lp
            h_try = _repair(h_try, n)
            c5_try = _c5(h_try, dx)
            if c5_try + 1e-12 < best_c5:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_pair(h, n, dx, t_start, t_limit, n_grid, max_passes=500):
    hi_c5 = _c5(h, dx)
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi = min(1.0, hsum)
                if hi - lo < 1e-10:
                    continue
                orig_i, orig_j = h[i], h[j]
                best_t, best_c = orig_i, hi_c5
                for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                    h[i], h[j] = t, hsum - t
                    c = _c5(h, dx)
                    if c < best_c:
                        best_c, best_t = c, t
                if best_t != orig_i:
                    lo2 = max(lo, best_t - (hi - lo) / n_grid)
                    hi2 = min(hi, best_t + (hi - lo) / n_grid)
                    for t in np.linspace(lo2, hi2, 9):
                        h[i], h[j] = t, hsum - t
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c, best_t = c, t
                if best_c < hi_c5 - 1e-12:
                    h[i], h[j] = best_t, hsum - best_t
                    hi_c5 = best_c
                    improved = True
                else:
                    h[i], h[j] = orig_i, orig_j
        if not improved:
            break
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

        # Follow up with SLSQP on improved solution
