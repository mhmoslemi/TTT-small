    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:

            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _smooth_obj(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(float(np.exp(s).sum())) / a

def _smooth_grad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]

            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - h[i + lag])
                g[i + lag] -= wk * h[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - h[i - L])
                g[i - L] -= wk * h[i]

    idx = np.argsort(g)
    s = np.zeros(n)
    ip = int(target)
    fr = target - ip
    s[idx[:ip]] = 1.0
    if fr > 1e-10:
        s[idx[ip]] = fr
    return s

def _fw_opt(h0, alpha, target, t0, budget, margin=30):

    dx = 2.0 / n
    h = h0.copy()
    best_h, best_c = h.copy(), _c5(h, dx)
    t = 1
    while time.time() - t0 < budget - margin:
        gamma = 2.0 / (t + 2.0)
        g = _smooth_grad(h, alpha)
        s = _fw_linear(g, n, target)
        h_new = (1.0 - gamma) * h + gamma * s
        c = _c5(h_new, dx)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        t += 1
        if t > 50000:
            break
    return best_h, best_c

def _fista(h0, alpha, target, t0, budget, margin=30):
    n = len(h0)
    dx = 2.0 / n
    x, z = h0.copy(), h0.copy()
    best_h, best_c = h0.copy(), _c5(h0, dx)
    t = 1.0
    step = 1.0 / (4.0 * alpha * n + 4.0)
    while time.time() - t0 < budget - margin:
        gz = _smooth_grad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _smooth_obj(xn, alpha)
            quad = _smooth_obj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)

def _cd_fast(h, dx, t0, budget, margin=20, n_passes=200, rng=None):
    n = len(h)
    h = h.copy()
    bc = _c5(h, dx)
    dvals = [0.001, -0.001, 0.003, -0.003, 0.007, -0.007, 0.015, -0.015,
             0.03, -0.03, 0.06, -0.06, 0.1, -0.1, 0.15, -0.15]
    for p in range(n_passes):
        if time.time() - t0 > budget - margin:
            break
        improved = False
        for _ in range(n * 6):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                h[i] = min(max(ni, 0.0), 1.0)
                h[j] = min(max(nj, 0.0), 1.0)
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    improved = True
                    break
                h[i], h[j] = hi0, hj0
            else:
                h[i], h[j] = hi0, hj0
        if not improved:
            break
    return h, bc
