                    h[i], h[j] = ni - d, nj + d
            stall = 0 if improved else stall + 1
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    _CONSTR['target'] = n / 2.0
    h0 = _repair(np.array(initial_h_values, dtype=np.float64).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm start
    t_end = t0 + 130
    for alpha in [50, 150, 400, 800]:
        if time.time() > t_end or (budget_s - (time.time() - t0)) < 600:
            break
        best_h, best_c5 = _slsqp(best_h, n, dx, alpha, 700, t_end)

    # Phase 2: Directed exchange
    if (budget_s - (time.time() - t0)) > 500:
        t_end = t0 + 450
        best_h, best_c5 = _directed(best_h, n, dx, rng, t_end)

    # Phase 3: SLSQP refinement
    if (budget_s - (time.time() - t0)) > 300:
        t_end = t0 + 700
        for alpha in [300, 800, 2000, 5000]:
            if time.time() > t_end or (budget_s - (time.time() - t0)) < 200:
                break
            best_h, best_c5 = _slsqp(best_h, n, dx, alpha, 1000, t_end)

    # Phase 4: Second directed pass
    if (budget_s - (time.time() - t0)) > 250:
        t_end = t0 + 920
        best_h, best_c5 = _directed(best_h, n, dx, rng, t_end)

    # Phase 5: SLSQP + local search polish
    if (budget_s - (time.time() - t0)) > 60:
        t_end = time.time() + min(35, (budget_s - (time.time() - t0)) - 25)
        best_h, best_c5 = _slsqp(best_h, n, dx, 8000, 1200, t_end)
    if (budget_s - (time.time() - t0)) > 30:
        t_end = time.time() + min(20, (budget_s - (time.time() - t0)) - 8)
        best_h, best_c5 = _random_local(best_h, n, dx, rng, t_end)

    # Final polish
    if (budget_s - (time.time() - t0)) > 15:
        t_end = time.time() + min(10, (budget_s - (time.time() - t0)) - 3)
        best_h, best_c5 = _slsqp(best_h, n, dx, 12000, 500, t_end)
        if (budget_s - (time.time() - t0)) > 5:
            t_end2 = time.time() + min(3, (budget_s - (time.time() - t0)) - 1)
            best_h, best_c5 = _directed(best_h, n, dx, rng, t_end2)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
