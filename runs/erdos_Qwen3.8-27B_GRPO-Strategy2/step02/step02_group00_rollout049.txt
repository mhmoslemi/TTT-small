def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _project(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - p) if diff > 0 else np.argsort(p)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - p[idx])
                p[idx] += a
                diff -= a
            else:
                a = min(-diff, p[idx])
                p[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return p
