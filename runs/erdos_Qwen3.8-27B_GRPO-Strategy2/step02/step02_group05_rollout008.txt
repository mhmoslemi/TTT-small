def _con_eq(x, target):
    return np.sum(x) - target

def _slsqp_run(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_eq, 'args': (target,)}]
    try:
        res = minimize(_obj_fn, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _active_grad(h, n):
    c = _corr(h)
    top_k = np.argsort(c)[-3:]
    n1 = n - 1
    grad = np.zeros(n)
    for k in top_k:
        L = k - n1
        for j in range(n):
            if 0 <= j + L < n:
                grad[j] += (1.0 - h[j + L])
            if 0 <= j - L < n:
                grad[j] -= h[j - L]
    return grad

def _local_search(h, n, dx, rng, deadline, use_blocks=True):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps_p = [0.003, 0.006, 0.01, 0.02, 0.04, 0.08, 0.12]
    steps_neg = [-s for s in steps_p]
    stall = 0
    for outer in range(4000):
        if time.time() > deadline:
            break
        improved = False
        grad = _active_grad(best, n)
        ntrials = 100 + 40 * min(stall // 12, 5)
        for _ in range(ntrials):
            i = int(rng.randint(n))
            if use_blocks and rng.rand() < 0.25 and i + 2 < n:
                j = int(rng.randint(n))
                if j in (i, i+1, i+2):
                    continue
                for d in [0.004, 0.008, 0.015, 0.03, -0.004, -0.008, -0.015, -0.03]:
                    vals = [best[i]+d, best[i+1]+d, best[i+2]+d, best[j]-3*d]
                    if min(vals) < -1e-9 or max(vals) > 1+1e-9:
                        continue
                    old = [best[i], best[i+1], best[i+2], best[j]]
                    best[i], best[i+1], best[i+2], best[j] = vals
                    nc = _c5(best, dx)
                    if nc < best_c5 - 1e-13:
                        best_c5 = nc
                        improved = True
                    else:
                        best[i], best[i+1], best[i+2], best[j] = old
                    if improved:
                        break
            else:
                j = int(rng.randint(n))
                if i == j:
                    continue
                if grad[i] > grad[j]:
                    d_list = steps_neg + steps_p[:3]
                else:
                    d_list = steps_p + steps_neg[:3]
                for d in d_list:
                    ni, nj = best[i]+d, best[j]-d
                    if ni < -1e-9 or ni > 1+1e-9 or nj < -1e-9 or nj > 1+1e-9:
                        continue
                    oi, oj = best[i], best[j]
                    best[i], best[j] = ni, nj
                    nc = _c5(best, dx)
                    if nc < best_c5 - 1e-13:
                        best_c5 = nc
                        improved = True
                    else:
                        best[i], best[j] = oi, oj
                    if improved:
                        break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 35:
                break
    return best, best_c5

def _sim_anneal(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    cur = h.copy()
    cur_c5 = _c5(cur, dx)
    t_lo, t_hi = 0.002, 0.06
    total = 8000
    cnt = 0
    while time.time() < deadline and cnt < total:
        cnt += 1
        t = t_hi * (t_lo / t_hi) ** (cnt / total)
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 0.12
        ni, nj = cur[i]+d, cur[j]-d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        oi, oj = cur[i], cur[j]
        cur[i], cur[j] = ni, nj
        nc = _c5(cur, dx)
        delta = nc - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-10)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[i], cur[j] = oi, oj
    return best, best_c5

def _diverse_init(n, rng_seed):
    rs = np.random.RandomState(5000 + rng_seed)
    t = np.linspace(0, 2.0, n, endpoint=False)
    f1, f2 = 1 + rng_seed % 4, 2 + (rng_seed * 3) % 5
    base = 0.5 + 0.22*np.sin(2*np.pi*f1*t/2.0) + 0.14*np.cos(2*np.pi*f2*t/2.0)
    base += 0.06 * rs.randn(n)
    return _repair(base, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)

    def _ok(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: Progressive SLSQP
    for a in [25, 50, 100, 200, 400, 800, 1600, 3200]:
        if not _ok(350):
            break
        mi = 600 if a <= 200 else 400
        hc = _slsqp_run(best_h, n, dx, a, mi, 1e-14)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc

    # Phase 2: Aggressive local search
    if _ok(280):
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0+budget_s-280)

    # Phase 3: Perturbed SLSQP restarts
    sigmas = [0.08, 0.15, 0.25, 0.35, 0.5, 0.12, 0.4, 0.2, 0.6, 0.1]
    it = 0
    while _ok(150):
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp_run(hp, n, dx, 200, 300, 1e-13)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        elif c5 < best_c5 + 0.004 and _ok(160):
            hl, cl = _local_search(hc, n, dx, rng, t0+budget_s-160)
            if cl < best_c5:
                best_c5, best_h = cl, hl
        if _ok(150):
            hc2 = _slsqp_run(hc if c5 < best_c5 else best_h, n, dx, 1000, 400, 1e-14)
            nc2 = _c5(hc2, dx)
            if nc2 < best_c5:
                best_c5, best_h = nc2, hc2
        it += 1

    # Phase 4: Try diverse starts with SLSQP
    for ds in range(4):
        if not _ok(120):
            break
        hs = _diverse_init(n, seed + ds * 7)
        hc = _slsqp_run(hs, n, dx, 100, 400, 1e-13)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    # Phase 5: Simulated annealing
    if _ok(70):
        ha, ca = _sim_anneal(best_h, n, dx, rng, t0+budget_s-70)
        if ca < best_c5:
            best_c5, best_h = ca, ha

    # Phase 6: Final SLSQP polish
    if _ok(25):
        for a in [2000, 5000, 10000]:
            if not _ok(25):
                break
            hc = _slsqp_run(best_h, n, dx, a, 600, 1e-15)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc

    # Phase 7: Final fine local search
    if _ok(8):
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0+budget_s-3)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
