def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s

                diff -= a

                h[idx] -= a
                diff += a

def _slsqp(h, n, alpha, maxiter, target, bounds):

    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

def _subgrad_step(h, corr, dx, eta, n):

    top_k = int(np.argmax(corr))
    offset = top_k - (n - 1)

        if 0 <= j2 < n:
            grad[i] -= h[j2]
    h_new = h - eta * grad

def _slsqp(h, n, alpha, maxiter, target, bounds):
    try:
        cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad(h, corr, eta, n):
    top = int(np.argmax(corr))
    off = top - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        a = i + off
        if 0 <= a < n:
            grad[i] += 1.0 - h[a]
        b = i - off
        if 0 <= b < n:
            grad[i] -= h[b]
    return _repair(h - eta * grad, n)
