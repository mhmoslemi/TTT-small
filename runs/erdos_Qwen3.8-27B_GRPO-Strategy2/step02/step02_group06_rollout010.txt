import numpy as np

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

                    span = (d_max - d_min) / max(grid_c - 1, 1)
                    lo = max(d_min, d_best - span)
                    hi2 = min(d_max, d_best + span)
                    if hi2 - lo > 1e-14:
                        ds2 = np.linspace(lo, hi2, grid_f)
                        vals2 = np.max(corr + ds2[:, None] * gd[None, :], axis=1) * dx
                        bi2 = np.argmin(vals2)
                        d_best = ds2[bi2]
                        cur_max = float(vals2[bi2])
                    else:
                        cur_max = float(vals[bi])
                    h[i] = hi + d_best
                    h[j] = hj - d_best
                    corr = corr + d_best * gd
                    improved += 1
        if improved == 0:
            break
    return h, corr, cur_max

def _sweep_triples(h, n, dx, grid, n_samp, max_passes, deadline, seed):
    corr = _corr(h)
    cur_max = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = 0
        idx = rng.choice(n, size=3 * n_samp, replace=False)
        for s in range(n_samp):
            i, j, k = int(idx[3*s]), int(idx[3*s+1]), int(idx[3*s+2])
            hi, hj, hk = h[i], h[j], h[k]
            d_lo = max(-hi, -hj, (hk - 1.0) / 2.0)
            d_hi = min(1.0 - hi, 1.0 - hj, hk / 2.0)
            if d_hi - d_lo < 1e-14:
                continue
            gd = _g_vec(h, i, n) + _g_vec(h, j, n) - 2.0 * _g_vec(h, k, n)
            ds = np.linspace(d_lo, d_hi, grid)
            vals = np.max(corr + ds[:, None] * gd[None, :], axis=1) * dx
            bi = np.argmin(vals)
            if vals[bi] < cur_max - 1e-14:
                h[i] = hi + ds[bi]
                h[j] = hj + ds[bi]
                h[k] = hk - 2.0 * ds[bi]
                corr = corr + ds[bi] * gd
                cur_max = float(vals[bi])
                improved += 1
        if improved == 0:
            break
    return h, corr, cur_max

def _slsqq(h, n, target, bounds, alpha, maxiter):
    def obj(x):
        c = _corr(x)
        m = float(np.max(c))
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / alpha
    def eq(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _make_start(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    positions = np.floor(np.arange(n) * phi).astype(int) % n
    order = np.argsort(positions)
    h[order[:n//2]] = 1.0
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial from parent
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx0)
    bounds0 = [(0.0, 1.0)] * n0

    deadline_late = t0 + budget_s - 20
    deadline_slsqp = t0 + budget_s - 80

    # Phase 1: Aggressive pair sweeps
    h, corr, cur_c5 = _sweep_pairs(h, n0, dx0, 25, 51, 500, deadline_slsqp)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()

    # Phase 2: Triple moves for escaping pair-local optima
    if time.time() < deadline_slsqp - 60:
        h = best_h.copy()
        h, corr, cur_c5 = _sweep_triples(h, n0, dx0, 31, 800, 100, deadline_slsqp - 60, 7)
        h, corr, cur_c5 = _sweep_pairs(h, n0, dx0, 25, 51, 300, deadline_slsqp - 30)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    # Phase 3: More pair sweeps from best
    if time.time() < deadline_slsqp - 30:
        h = best_h.copy()
        h, corr, cur_c5 = _sweep_pairs(h, n0, dx0, 15, 41, 600, deadline_slsqp - 30)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    # Phase 4: Triple moves again
    if time.time() < deadline_slsqp - 20:
        h = best_h.copy()
        h, corr, cur_c5 = _sweep_triples(h, n0, dx0, 31, 1200, 80, deadline_slsqp - 20, 42)
        h, corr, cur_c5 = _sweep_pairs(h, n0, dx0, 25, 51, 300, deadline_slsqp)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    # Phase 5: SLSQP polish at multiple alphas
    if time.time() < deadline_slsqp:
        for alpha in [500, 1500, 5000]:
            if time.time() > deadline_slsqp - 5:
                break
            hc = _slsqq(best_h, n0, target0, bounds0, alpha, 2000)
            c5 = _c5(hc, dx0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 6: Random restarts with pair optimization
    for r in range(8):
        if time.time() > deadline_late:
            break
        h_p = best_h + 0.12 * rng.randn(n0)
        h_p = _repair(h_p, n0)
        h_r, corr_r, c5_r = _sweep_pairs(h_p, n0, dx0, 20, 41, 300, deadline_late)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Structure-based restarts (different n)
    for n_try in [50, 60, 80]:
        if time.time() > deadline_late:
            break
        dx_t = 2.0 / n_try
        h_s = _make_start(n_try, rng)
        h_r, corr_r, c5_r = _sweep_pairs(h_s, n_try, dx_t, 20, 41, 300, deadline_late)
        h_r, corr_r, c5_r2 = _sweep_triples(h_r, n_try, dx_t, 25, 500, 40, deadline_late, 99)
        h_r, corr_r, c5_r3 = _sweep_pairs(h_r, n_try, dx_t, 20, 41, 200, deadline_late)
        if c5_r3 < best_c5:
            best_c5 = c5_r3
            best_h = h_r.copy()
            n0 = n_try

    # Phase 8: Final intensive pair sweep
    h = best_h.copy()
    n_final = len(h)
    dx_final = 2.0 / n_final
    h, corr, cur_c5 = _sweep_pairs(h, n_final, dx_final, 31, 61, 800, deadline_late)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()

    # Phase 9: Final SLSQP
    if time.time() < t0 + budget_s - 5:
        n_fin = len(best_h)
        target_fin = n_fin / 2.0
        bounds_fin = [(0.0, 1.0)] * n_fin
        hc = _slsqq(best_h, n_fin, target_fin, bounds_fin, 8000, 3000)
        c5 = _c5(hc, 2.0 / n_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    best_h = _repair(best_h, len(best_h))
    n_out = len(best_h)
    best_c5 = _c5(best_h, 2.0 / n_out)
    return (best_h, best_c5, n_out)
