from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400.0, 400.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _grad_max(h, n):
    c = _corr(h)
    mv = float(c.max())
    # Find active lags (within 1e-4 of max)
    active = np.where(c >= mv - 1e-4)[0]
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        for t in range(n):
            j1 = t - m
            if 0 <= j1 < n:
                grad[t] += (1.0 - h[j1])
            j2 = t + m
            if 0 <= j2 < n:
                grad[t] -= h[j2]
    grad /= len(active)
    return mv, grad

def _slsqp_opt(h, n, alpha, maxiter, ftol, bounds, cons):
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < _smooth_obj(h, alpha):
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

    """Gradient descent with backtracking line search on the true max."""
    best_c = _c5(h, dx)
    lr = lr_init
    for it in range(max_iter):
        if time.time() - t_start > t_limit:
            break
        mv, g = _grad_max(h, n)
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        g = g / gn
        # Backtracking line search

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _lag_grad(h, n, k):
    """Gradient of correlation at lag k (0-indexed in full correlation)."""
    m = k - (n - 1)
    g = np.zeros(n)
    for t in range(n):
        j1 = t - m
        if 0 <= j1 < n:
            g[t] += 1.0 - h[j1]
        j2 = t + m
        if 0 <= j2 < n:
            g[t] -= h[j2]
    return g

def _lag_val(h, k):
    """Value of correlation at lag k."""
    n = len(h)
    m = k - (n - 1)
    val = 0.0
    for j in range(n):
        j2 = j + m
        if 0 <= j2 < n:
            val += h[j2] * (1.0 - h[j])
    return val

def _cutting_plane(h, n, dx, max_iter=50, tol=1e-12):
    """Cutting-plane LP minimax."""
    cuts = []  # list of (gradient_vector, rhs_offset)
    h_cur = h.copy()
    
    for it in range(max_iter):
        c = _corr(h_cur)
        k_max = int(np.argmax(c))
        c_max = float(c[k_max])
        
        # Add constraint for k_max
        g = _lag_grad(h_cur, n, k_max)
        f_val = float(c[k_max])
        # Linearized: g^T h ≤ t - f_val + g^T h_cur
        rhs = g @ h_cur - f_val
        cuts.append((g, rhs, k_max))
        
        # Solve LP: min t s.t. g_i^T h - t ≤ rhs_i, 0≤h≤1, sum(h)=n/2
        # Variables: [h_0,...,h_{n-1}, t]
        # Objective: [0,...,0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub = []
        b_ub = []
        for (g_i, rhs_i, _) in cuts:
            row = np.zeros(n + 1)
            row[:n] = g_i
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(rhs_i)
        
        if A_ub:
            A_ub = np.array(A_ub)
            b_ub = np.array(b_ub)
        else:
            A_ub = np.empty((0, n+1))
            b_ub = np.empty(0)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        
        if not res.success:
            break
        
        h_new = _repair(res.x[:n], n)
        
        # Check true objective
        c_new = _corr(h_new)
        if float(c_new.max()) <= c_max * (1 - tol) + tol:
            # LP solution is good
            return h_new, float(c_new.max() * dx)
        
        h_cur = h_new
    
    return h_cur, _c5(h_cur, dx)

import numpy as np
from scipy.optimize import minimize, linprog
import time

_con_target = [0.0]

def _sum_con(x):
    return x.sum() - _con_target[0]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _lag_grad(h, n, k):
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        mm = -m
        g[:n - mm] += (1.0 - h[mm:])
        g[mm:] -= h[:n - mm]
    return g

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _slsqp(h, n, alpha, maxiter, ftol, bounds, cons):
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < _smooth_obj(h, alpha) + 1e-12:
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

    best_c = _c5(h, dx)

            best_c = c_val * dx
            best_h = h_cur.copy()
        g = _lag_grad(h_cur, n, k_max)
        rhs = float(g @ h_cur - c_val)
        cuts.append((g, rhs, k_max))
        if len(cuts) > 40:
            cuts = cuts[-40:]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((len(cuts), n + 1))
        b_ub = np.zeros(len(cuts))
        for i, (gg, rr, _) in enumerate(cuts):
            A_ub[i, :n] = gg
            A_ub[i, n] = -1.0
            b_ub[i] = rr
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
