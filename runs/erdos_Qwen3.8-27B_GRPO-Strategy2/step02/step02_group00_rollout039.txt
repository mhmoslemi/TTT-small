from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h


def _binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb


def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)


def _eq_con(x, target):
    return float(np.sum(x) - target)

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h))) * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h
