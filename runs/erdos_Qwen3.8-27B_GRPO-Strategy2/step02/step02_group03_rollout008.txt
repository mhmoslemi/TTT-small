```python
def _try_swap(h, corr, i, j, d, n):
    """Try swapping d from h[i] to h[j]. Returns True if it reduces max corr."""
    cmax = corr.max()
    s_all = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # Precompute for index i: which k values are affected
    # Effect on corr[k] from h[i]-=d:
    #   -d*(1-h[i+s_k]) if 0<=i+s_k<n
    #   +d*h[i-s_k] if 0<=i-s_k<n
    # Effect on corr[k] from h[j]+=d:
    #   +d*(1-h[j+s_k]) if 0<=j+s_k<n
    #   -d*h[j-s_k] if 0<=j-s_k<n
    
    delta = np.zeros(2*n-1)
    
    for k, s in enumerate(s_all):
        d_k = 0.0
        if 0 <= i+s < n:
            d_k += -d * (1 - h[i+s])
        if 0 <= i-s < n:
            d_k += d * h[i-s]
        if 0 <= j+s < n:
            d_k += d * (1 - h[j+s])
        if 0 <= j-s < n:
            d_k += -d * h[j-s]
        delta[k] = d_k
    
    new_corr_max = np.max(corr + delta)
    if new_corr_max < cmax - 1e-14:
        h[i] -= d
        h[j] += d
        corr += delta
        return True, new_corr_max
    return False, cmax
```

```python
def _swap_delta(h, i, j, d, n):
    """Compute delta to corr when h[i]-=d, h[j]+=d."""
    L = 2*n-1
    s_all = np.arange(L) - (n-1)  # shape (L,)
    delta = np.zeros(L)
    
    # For index i (decreased by d):
    # Term 1: -d*(1-h[i+s]) for valid i+s
    valid1 = (i + s_all >= 0) & (i + s_all < n)
    delta[valid1] += -d * (1 - h[i + s_all[valid1]])
    # Term 2: +d*h[i-s] for valid i-s
    valid2 = (i - s_all >= 0) & (i - s_all < n)
    delta[valid2] += d * h[i - s_all[valid2]]
    
    # For index j (increased by d):
    # Term 3: +d*(1-h[j+s]) for valid j+s
    valid3 = (j + s_all >= 0) & (j + s_all < n)
    delta[valid3] += d * (1 - h[j + s_all[valid3]])
    # Term 4: -d*h[j-s] for valid j-s
    valid4 = (j - s_all >= 0) & (j - s_all < n)
    delta[valid4] += -d * h[j - s_all[valid4]]
    
    return delta
```
