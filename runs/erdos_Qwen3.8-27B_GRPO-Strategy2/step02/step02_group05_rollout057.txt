import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * tgt
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * tgt
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_obj(x, alpha, dx):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
