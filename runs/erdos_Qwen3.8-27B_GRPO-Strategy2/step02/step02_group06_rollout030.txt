                nc = corr + G[i] - G[j]
                nm = float(np.max(nc))
                if nm < cm - 1e-14:
                    h[i] = 0; h[j] = 1
                    corr = nc; cm = nm
                    improved = True
        if not improved:
            if ones and zeros:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0; h[j] = 1
                corr = corr + G[i] - G[j]
                cm = float(np.max(corr))
            else:
                break
    return h, cm * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)

    # Phase 1: Binary local search from diverse starts
    t1 = t0 + 40
    for trial in range(6):
        if time.time() > t1:
            break
        if trial == 0:
            hb0 = best_h.copy()
        elif trial == 1:
            hb0 = (rng.rand(n) < 0.5).astype(float)
        else:
            hb0 = best_h + 0.3 * rng.randn(n)
        hb, cb = _binary_ls(hb0, n, dx, rng, 150)
        if cb < best_c5:
            best_c5 = cb
            best_h = _repair(hb, n)
            best_c5 = _c5(best_h, dx)

    # Phase 2: SLSQP smooth max (increasing alpha)
    t2 = t0 + 280
    for alpha in [20, 60, 200, 600]:
        if time.time() > t2:
            break
        try:
            res = minimize(_obj_smax, best_h, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-14})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5; best_h = hc
        except Exception:
            pass

    # Phase 3: Coarse multi-pass sweep (randomized order)
    t3 = t0 + 480
    best_h, best_c5 = _sweep(best_h, n, dx, 60, 18, 0, 0, t3, rng)

    # Phase 4: Fine two-stage sweep
    t4 = t0 + 720
    best_h, best_c5 = _sweep(best_h, n, dx, 80, 22, 41, 0.03, t4, rng)

    # Phase 5: Interpolate + re-sweep for escape
    if time.time() < t0 + budget_s - 200:
        h_alt = best_h + 0.05 * rng.randn(n)
        h_alt = _repair(h_alt, n)
        try:
            res = minimize(_obj_smax, h_alt, args=(300,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-14})
            h_alt = _repair(res.x, n)
        except Exception:
            pass
        h_alt, c5a = _sweep(h_alt, n, dx, 50, 22, 31, 0.02,
                           t0 + budget_s - 200, rng)
        if c5a < best_c5:
            best_c5 = c5a; best_h = h_alt

    # Phase 6: Ultra-fine sweep
    t6 = t0 + budget_s - 100
    best_h, best_c5 = _sweep(best_h, n, dx, 80, 25, 51, 0.01, t6)

    # Phase 7: Nelder-Mead with penalty (derivative-free)
    if time.time() < t0 + budget_s - 50:
        try:
            res = minimize(_obj_pen, best_h, args=(target, 50.0),
                          method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-12,
                                   'fatol': 1e-14, 'adaptive': True})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5; best_h = hc
        except Exception:
            pass

    # Phase 8: Final SLSQP + sweep
    if time.time() < t0 + budget_s - 30:
        try:
            res = minimize(_obj_smax, best_h, args=(3000,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1200, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5; best_h = hc
        except Exception:
            pass

    t_final = t0 + budget_s - 5
    best_h, best_c5 = _sweep(best_h, n, dx, 40, 28, 61, 0.005, t_final)
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
