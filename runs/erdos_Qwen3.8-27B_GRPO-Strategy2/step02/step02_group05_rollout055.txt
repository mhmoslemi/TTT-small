import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        direction = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in direction:
            if diff > 0:
                amt = min(diff, 1.0 - h[idx])
                h[idx] += amt
                diff -= amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt
                diff += amt
            if abs(diff) < 1e-10:
                break
    return h


def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _obj_func(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx


def _con_func(x, target):
    return np.sum(x) - target


def _run_slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_func, 'args': (target,)}]
    try:
        res = minimize(_obj_func, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()


def _coord_descent(h, n, dx, max_sweeps, time_limit):
    h = h.copy()
    t_start = time.time()
    best_c5 = _c5(h, dx)
    n_corr = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        improved = False
        corr = _corr(h)
        for i in range(n):
            if time.time() > time_limit:

            coeff = np.zeros(n_corr)
            for k in range(n_corr):
                m1 = i - k + (n - 1)
                m2 = i + k - (n - 1)
                c = 0.0
                if 0 <= m1 < n:
                    c += 1.0 - h[m1]
                if 0 <= m2 < n:
                    c -= h[m2]
            if c != 0 or True:  # always process
                coeff[k] = c
            # Find best d in [-h[i], 1-h[i]]
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            # Check boundaries
            f_lo = np.max(corr + lo * coeff)
            f_hi = np.max(corr + hi * coeff)
            f_cur = np.max(corr)
            # If current is already best, skip
            if f_cur <= f_lo - 1e-14 and f_cur <= f_hi - 1e-14:
                # Check if interior could be better
                f_mid = np.max(corr + (lo + hi) / 2 * coeff)

                    corr = _corr(h)

    h = _repair(h, n)
    return h, _c5(h, dx)

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        direction = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in direction:
            if diff > 0:
                amt = min(diff, 1.0 - h[idx])
                h[idx] += amt; diff -= amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt; diff += amt
            if abs(diff) < 1e-10:
                break
    return h

def _smax(x, a):
    m = np.max(x)
    s = np.clip(a * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _objf(x, a, dx):
    return _smax(_corr(x), a) * dx

def _con(x, t):
    return np.sum(x) - t

                     bounds=bnds, constraints=cs,
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

                else:

    h = _repair(h, n)
