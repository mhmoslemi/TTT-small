from scipy.optimize import minimize
import time

def _c5_from_corr(corr, dx):
    return float(np.max(corr)) * dx

def _corr_binary(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _to_binary(h, n, target):
    """Convert to binary with exactly target ones"""
    h_bin = (h >= 0.5).astype(float)
    current_sum = int(h_bin.sum())
    if current_sum == target:
        return h_bin
    if current_sum > target:
        # Need to flip (current_sum - target) ones to zeros
        # Prefer flipping 1s that are "less needed" (lower h value? or random?)
        ones = np.where(h_bin == 1)[0]
        # Sort by h value ascending (values closest to 0.5 first)
        order = np.argsort(h[ones])
        to_flip = order[:current_sum - target]
        h_bin[ones[to_flip]] = 0
    else:
        # Need to flip (target - current_sum) zeros to ones
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        to_flip = order[:target - current_sum]
        h_bin[zeros[to_flip]] = 1
    return h_bin

def _evaluate_binary(h):
    """Return (max_corr, corr_array) for binary h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), corr

def _swap_improves(h, i, j, current_max, corr):
    """Check if swapping h[i] (1) and h[j] (0) improves"""
    h[i], h[j] = h[j], h[i]
    new_corr = np.correlate(h, 1.0 - h, mode='full')
    new_max = float(np.max(new_corr))
    improved = new_max < current_max
    h[i], h[j] = h[j], h[i]
