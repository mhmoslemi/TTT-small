def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _fh(params, n, M):
    i = np.arange(n, dtype=float)
    h = np.full(n, 0.5)
    for k in range(1, M + 1):
        h += params[2*k-2] * np.cos(2*np.pi*k*i/n)
        h += params[2*k-1] * np.sin(2*np.pi*k*i/n)
    return h

    v = np.sum(np.maximum(0.0, -h)**2 + np.maximum(0.0, h-1)**2)

def _sgdir(h, n, lags):
    g = np.zeros(n)
    for m in lags:
        lag = m - (n - 1)
        for j in range(n):
            if 0 <= j + lag < n:
                g[j] += 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                g[j] -= h[j - lag]
    return g / len(lags)

def _slsqp(h, n, bounds, target, alpha, mi):
    def _sm(c, a):
        m = float(np.max(c))
        s = np.clip(a*(c-m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s)))) / a
    def _o(x, a):
        return _sm(_corr(x), a)

        return np.sum(x) - t

                     bounds=bounds, constraints=[{'type':'eq','fun':_e,'args':(target,)}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-15: return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _fh(p, n, M):
    i = np.arange(n, dtype=float)
    h = np.full(n, 0.5)
    for k in range(1, M+1):
        h += p[2*k-2]*np.cos(2*np.pi*k*i/n) + p[2*k-1]*np.sin(2*np.pi*k*i/n)
    return h

def _fobj(p, n, M, dx, pen):

    v = np.sum(np.maximum(0, -h)**2 + np.maximum(0, h-1)**2)
    return _c5(_repair(h, n), dx) + pen*v

    i = np.arange(n); d = h - 0.5; p = np.zeros(2*M)
    for k in range(1, M+1):

        p[2*k-1] = 2.0/n*np.dot(d, np.sin(2*np.pi*k*i/n))
    return p

def _sgdir(h, n, lags):
    g = np.zeros(n)
    for m in lags:
        lag = m - (n-1)
        for j in range(n):
            if 0 <= j+lag < n: g[j] += 1.0 - h[j+lag]
            if 0 <= j-lag < n: g[j] -= h[j-lag]
    return g / len(lags)

def _slsqp(h, n, bnds, tgt, alpha, mi):

        m = float(np.max(c)); s = np.clip(a*(c-m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s))))/a
    def _o(x, a): return _sm(_corr(x), a)
    def _e(x, t): return np.sum(x) - t
    try:
        r = minimize(_o, h, args=(alpha,), method='SLSQP', bounds=bnds,
                     constraints=[{'type':'eq','fun':_e,'args':(tgt,)}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception: return h

def _sweep(h, n, dx, nc, nc2=0, wd=0.0):

    k = np.arange(-(n-1), n)
    for i in range(n):
        for j in range(i+1, n):
            hi, hj = h[i], h[j]
            lo = max(-hi, hj-1.0); hi2 = min(1.0-hi, hj)
            if hi2-lo < 1e-12: continue

            a1=i-k; m1=(a1>=0)&(a1<n); gd[m1]+=(1.0-h[a1[m1]])
            a2=i+k; m2=(a2>=0)&(a2<n); gd[m2]-=h[a2[m2]]
            b1=j-k; m1=(b1>=0)&(b1<n); gd[m1]-=(1.0-h[b1[m1]])
            b2=j+k; m2=(b2>=0)&(b2<n); gd[m2]+=h[b2[m2]]
            ds = np.linspace(lo, hi2, nc)
            v = np.max(c[None,:]+ds[:,None]*gd[None,:], axis=1)*dx
            bi = int(np.argmin(v))
