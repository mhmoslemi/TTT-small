    best = _c5(h, dx)
    h = h.copy()
    
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        corr = _corr(h)
        max_val = np.max(corr)
        
        # Find top few lags
        top_k = 5
        top_lags = np.argsort(corr)[-top_k:]
        
        # Combined gradient from top lags
        grad = np.zeros(n)
        for lag in top_lags:
            weight = (corr[lag] - max_val + 0.01)  # weight by proximity to max
            shift = int(lag) - n + 1
            for j in range(n):
                if 0 <= j - shift < n:
                    grad[j] += weight * (1.0 - h[j - shift])
                if 0 <= j + shift < n:
                    grad[j] += weight * (-h[j + shift])
        
        # Find best donor (high grad, will decrease) and receiver (low grad, will increase)
        # Decreasing h[i] where grad[i] > 0 reduces the max correlation
        # Increasing h[j] where grad[j] < 0 reduces the max correlation
        donors = np.where(grad > 0.01)[0]  # these will be decreased
        receivers = np.where(grad < -0.01)[0]  # these will be increased

            j = int(rng.randint(n))
