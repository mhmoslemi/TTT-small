def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _c5_vec(corr, dx):
    return float(np.max(corr)) * dx

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)

def _obj_topk(x, k, n):
    corr = _corr(x)
    if k >= len(corr):
        return float(corr.sum()) * (2.0 / n)
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum()) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)


def _obj_topk(x, k, n):
    corr = _corr(x)
    if k >= len(corr):
        return float(corr.sum()) * (2.0 / n)
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum()) * (2.0 / n)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _fine_descend(h, n, dx, t_end, max_passes=50):
    local_best = _c5(h, dx)
    h = h.copy()
    base_deltas = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for pass_num in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        scale = 1.0 / (1.0 + 0.4 * pass_num)
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for base_d in base_deltas:
                for sign in (1, -1):
                    d = sign * base_d * scale
                    ni = vi + d
                    if ni < 0.0 or ni > 1.0:
                        continue
                    need = -d
                    if need > 0:
                        candidates = np.argsort(h)[:22]
                    else:
                        candidates = np.argsort(-h)[:22]
                    found = False
                    for j in candidates:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0.0 or nj > 1.0:
                            continue
                        h[i] = ni
                        h[j] = nj
                        c5_try = _c5(h, dx)
                        if c5_try < local_best - 1e-14:
                            local_best = c5_try
                            improved = True
                            found = True
                            break
                        else:
                            h[i] = vi
                            h[j] = nj - need
                    if found:
                        break
        if not improved:
            break
    return h, local_best


def _targeted_sa(h, n, dx, t_end, rng, iters=3000000):
    h_sa = h.copy()
    sa_best = _c5(h_sa, dx)
    sa_best_h = h_sa.copy()
    T0, Tend = 0.003, 0.00002
    for it in range(iters):
        if time.time() > t_end:
            break
        frac = it / iters
        T = T0 * (Tend / T0) ** frac
        dmax = 0.07 * (1.0 - 0.5 * frac)
        if it % 200 == 0:
            corr = _corr(h_sa)
            peak_idx = int(np.argmax(corr))
            shift = peak_idx - (n - 1)
            lo = max(0, shift)
            hi = min(n, n + shift)
            if hi <= lo:
                lo, hi = 0, n
            i = int(rng.randint(lo, hi))
        else:
            i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        delta = (rng.rand() - 0.5) * 2.0 * dmax
        ni = h_sa[i] + delta
        if ni < 0.0 or ni > 1.0:
            continue
        nj = h_sa[j] - delta
        if nj < 0.0 or nj > 1.0:
            continue
        h_sa[i] = ni
        h_sa[j] = nj
        new_c5 = _c5(h_sa, dx)
        if new_c5 <= sa_best + T:
            if new_c5 < sa_best - 1e-15:
                sa_best = new_c5
                sa_best_h = h_sa.copy()
        else:
            h_sa[i] -= delta
            h_sa[j] += delta
    return sa_best_h, sa_best


def _ils(h, n, dx, t_end, rng, max_cycles=40):
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    for cyc in range(max_cycles):
        if time.time() > t_end:
            break
        h_pert = best_h.copy()
        n_pert = rng.randint(2, 6)
        for _ in range(n_pert):
            i = rng.randint(n)
            h_pert[i] += (rng.rand() - 0.5) * 0.12
        h_pert = _repair(h_pert, n)
        h_new, c5_new = _fine_descend(h_pert, n, dx,
                                       min(time.time() + 6, t_end),
                                       max_passes=12)
        if c5_new < best_c5 - 1e-15:
            best_c5, best_h = c5_new, h_new
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    n = len(initial_h_values)

    best_c5 = _c5(best_h, dx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    p1_end = t_start + min(budget_s * 0.07, 50)

                                         t_start + budget_s - 2,
                                         max_passes=40)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
