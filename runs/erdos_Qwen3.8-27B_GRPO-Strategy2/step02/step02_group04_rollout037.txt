def _subgradient_step(h, n, dx, epsilon=0.01, step_size=0.005):
    corr = _corr_arr(h)
    max_val = np.max(corr)
    # Active set: correlations within epsilon of max
    active = np.where(corr > max_val - epsilon)[0]
    
    if len(active) == 0:
        return h
    
    # Numerical subgradient
    grad = np.zeros(n)
    h_perturb = 1e-7
    for j in range(n):
        for k in active:
            # Perturb h[j] and compute change in corr[k]
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += h_perturb
            h_minus[j] -= h_perturb
            # Need to maintain sum for fair comparison
            # Actually for gradient computation, we don't need to maintain sum
            # The constraint is handled by projection
            corr_plus = _corr_arr(h_plus)[k]
            corr_minus = _corr_arr(h_minus)[k]
            grad[j] += (corr_plus - corr_minus) / (2 * h_perturb)
    
    grad /= len(active)
    
    # Step
    h_new = h - step_size * grad
    h_new = _repair(h_new, n)
    return h_new

def _corr_gradient(h, k, n):
    """Gradient of the k-th element of np.correlate(h, 1-h, 'full') w.r.t. h"""
    # k is the index in the 'full' output (0 to 2n-2)
    # The actual lag is k - (n-1)
    lag = k - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        # First part: (1-h[j-lag]) if 0<=j-lag<n
        if 0 <= j - lag < n:
            grad[j] += 1.0 - h[j - lag]
        # Second part: -h[j+lag] if 0<=j+lag<n
        if 0 <= j + lag < n:
            grad[j] -= h[j + lag]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return np.max(_corr_arr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _corr_gradient(h, k, n):
    lag = k - (n - 1)
    grad = np.zeros(n)
    jl = j - lag
    # Vectorized: for each j in 0..n-1
    # Part 1: (1-h[j-lag]) if 0<=j-lag<n
    idx1 = np.arange(n) - lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += 1.0 - h[idx1[valid1]]
    # Part 2: -h[j+lag] if 0<=j+lag<n
    idx2 = np.arange(n) + lag
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return grad

def _subgradient_descent(h, n, dx, deadline, init_step=0.01):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    step = init_step
    t0 = time.time()
    total = max(1.0, deadline - t0)
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        progress = (time.time() - t0) / total
        step = init_step * (1.0 - 0.7 * progress) + 0.0005
        corr = _corr_arr(h)
        mx = np.max(corr)
        # Active set: within 2% of max
        active = np.where(corr > mx - 0.02 * dx * n)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        # Compute averaged subgradient
        grad = np.zeros(n)
        for k in active:
            grad += _corr_gradient(h, k, n)
        grad /= len(active)
        # Step
        h_new = h - step * grad
        h_new = _repair(h_new, n)
        c5_new = _compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Accept with some probability even if worse (annealing)
        if c5_new <= _compute_c5(h, dx) or (step > 0.002 and np.random.random() < 0.1):
            h = h_new
        else:
            step *= 0.8
    return best_h, best_c5

def _simulated_annealing(h, n, dx, rng, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    t0 = time.time()
    total = max(1.0, deadline - t0)
    T0 = 0.04
    while time.time() < deadline:
        progress = (time.time() - t0) / total
        T = max(0.0001, T0 * (1.0 - progress) ** 2.5)
        n_mov = 2 + rng.randint(3)
        idxs = rng.choice(n, size=n_mov, replace=False)
        deltas = rng.normal(0, 0.02 + T, n_mov)
        new_h = h.copy()
        for idx, d in zip(idxs, deltas):
            new_h[idx] += d
        new_h = _repair(new_h, n)
        new_c5 = _compute_c5(new_h, dx)
        old_c5 = _compute_c5(h, dx)
        delta = new_c5 - old_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5

def _slsqp_refine(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    try:
        res = minimize(
            lambda x, a=alpha: _smooth_obj(x, a), h,
            method='SLSQP', bounds=bounds, constraints=[eq_con],
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
        if res.x is not None:
            hc = _repair(res.x, n)
            c5 = _compute_c5(hc, dx)
            if c5 < 1.0:
                return hc, c5
    except Exception:
        pass
    return h, _compute_c5(h, dx)

def _fine_local(h, n, dx, rng, deadline):
    best_local = _compute_c5(h, dx)
    h = h.copy()
    scales = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
    for scale in scales:
        stalled = 0
        for _pass in range(500):
            if time.time() >= deadline:
                return h, best_local
            improved = False
            n_tries = 800 if scale > 0.003 else 2000
            for _ in range(n_tries):
                if time.time() >= deadline:
                    return h, best_local
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5 = _compute_c5(h, dx)
                if c5 < best_local - 1e-12:
                    best_local = c5
                    improved = True
                else:
                    h[i], h[j] = ni - d, nj + d
            if improved:
                stalled = 0
            else:
                stalled += 1
                if stalled >= 5:
                    break
    return h, best_local

def _balance_active(h, n, dx, rng, deadline):
    """Try to equalize the top correlations"""
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        corr = _corr_arr(h)
        mx = np.max(corr)
        active = np.where(corr > mx - 0.5 * dx * n / 2)[0]
        if len(active) < 2:
            break
        # Find the two most different active correlations
        active_vals = corr[active]
        ai_max = active[np.argmax(active_vals)]
        ai_min = active[np.argmin(active_vals)]
        # Try to reduce the max and increase the min
        g_max = _corr_gradient(h, ai_max, n)
        g_min = _corr_gradient(h, ai_min, n)
        # Direction: reduce max, increase min => -g_max + g_min
        direction = -g_max + g_min
        for step in [0.003, 0.002, 0.001, 0.0005]:
            h_try = h + step * direction
            h_try = _repair(h_try, n)
            c5_try = _compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def tl():
        return budget_s - (time.time() - t_start)
