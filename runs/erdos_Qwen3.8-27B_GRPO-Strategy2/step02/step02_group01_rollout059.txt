def _rep(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h, dx):
    return np.max(_rep(h)) * dx

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _smax_grad(h, alpha):
    n = len(h)
    c = _rep(h)
    mc = np.max(c)
    s = np.clip(alpha * (c - mc), -500, 500)

    w = np.exp(s)

    g = np.zeros(n)
