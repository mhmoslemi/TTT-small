import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth(x, a):
    c = _corr(x)
    m = c.max()
    s = np.clip(a*(c-m), -400, 400)
    return m + np.log(np.sum(np.exp(s)))/a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if d > 0:
                add = min(d, 1.0 - h[idx])
                h[idx] += add; d -= add
            else:
                sub = min(-d, h[idx])
                h[idx] -= sub; d += sub
            if abs(d) < 1e-13:
                break
    return h

            B[:, mi] = 1.0 - 2.0 * h
        else:
            B[:, mi] = 1.0
            # subtract h[p+L] where valid
            if L > 0:
                B[:n-L, mi] -= h[L:]       # p+L < n => p < n-L
            else:
                B[-L:, mi] -= h[:n+L]      # p+L >= 0 => p >= -L
            # subtract h[p-L] where valid
            if L > 0:
                B[L:, mi] -= h[:n-L]       # p-L >= 0 => p >= L
            else:
                B[:n+L, mi] -= h[-L:]      # p-L < n => p < n+L
    return B

def _optimal_d(corr, s, hi, hj):
    """Find optimal d for h[i]+=d, h[j]-=d"""
    dmin = max(-hi, -(1.0 - hj))
    dmax = min(1.0 - hi, hj)
    if dmin >= dmax - 1e-14:
        return 0.0
    a, b = dmin + 1e-12, dmax - 1e-12
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_ = b - gr*(b-a)
    d_ = a + gr*(b-a)
    fc = np.max(corr + c_*s)
    fd = np.max(corr + d_*s)
    for _ in range(40):
        if fc < fd:
            b, d_, fd = d_, c_, fc
            c_ = b - gr*(b-a); fc = np.max(corr + c_*s)
        else:
            a, c_, fc = c_, d_, fd
            d_ = a + gr*(b-a); fd = np.max(corr + d_*s)
    return 0.5*(a+b)

def _local_search(h, n, dx, rng, deadline):
    """Optimal-step local search"""
    best = _c5(h, dx)
    h = h.copy()
    n_pairs = 500
    for _pass in range(300):
        if time.time() >= deadline:
            break
        corr = _corr(h)
        B = _bmatrix(h, n)
        improved = False
        for _ in range(n_pairs):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            s = B[i] - B[j]
            d = _optimal_d(corr, s, h[i], h[j])
            if abs(d) < 1e-10:
                continue
            h[i] += d; h[j] -= d
            new_c5 = _c5(h, dx)
            if new_c5 < best - 1e-12:
                best = new_c5
                improved = True
            else:
                h[i] -= d; h[j] += d
                corr = _corr(h)  # restore
        if not improved:
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth(x, a):
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h * (tgt / s)).copy()
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if d > 0:
                add = min(d, 1.0 - h[idx]); h[idx] += add; d -= add
            else:
                sub = min(-d, h[idx]); h[idx] -= sub; d += sub
            if abs(d) < 1e-13:
                break
    return h

def _bmat(h, n):
    B = np.ones((n, 2 * n - 1))
    for mi in range(2 * n - 1):
        L = mi - (n - 1)
        if L == 0:
            B[:, mi] = 1.0 - 2.0 * h
        else:
            B[:, mi] = 1.0
            if L > 0:
                B[:n - L, mi] -= h[L:]
                B[L:, mi] -= h[:n - L]
            else:
                B[-L:, mi] -= h[:n + L]
                B[:n + L, mi] -= h[-L:]
    return B

def _opt_d(corr, s, hi, hj):
    dmin = max(-hi, -(1.0 - hj))
    dmax = min(1.0 - hi, hj)
    if dmin >= dmax - 1e-14:
        return 0.0
    a, b = dmin + 1e-12, dmax - 1e-12
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = np.max(corr + c_ * s)
    fd = np.max(corr + d_ * s)
    for _ in range(45):
        if fc < fd:
            b, d_, fd = d_, c_, fc
            c_ = b - gr * (b - a); fc = np.max(corr + c_ * s)
        else:
            a, c_, fc = c_, d_, fd
            d_ = a + gr * (b - a); fd = np.max(corr + d_ * s)
    return 0.5 * (a + b)

def _local_search(h, n, dx, rng, deadline, n_steps=400):
    best = _c5(h, dx)
    h = h.copy()
    no_improve = 0
    for _pass in range(500):
        if time.time() >= deadline:
            break
        if no_improve >= 6:
            break
        corr = _corr(h)
        B = _bmat(h, n)
        improved_pass = False
        for _ in range(n_steps):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            s = B[i] - B[j]
            d = _opt_d(corr, s, h[i], h[j])
            if abs(d) < 1e-12:
                continue
            h[i] += d
            h[j] -= d
            nc = np.max(_corr(h)) * dx
            if nc < best - 1e-13:
                best = nc
                improved_pass = True
                corr = _corr(h)
                B = _bmat(h, n)
            else:
                h[i] -= d
                h[j] += d
        if improved_pass:
            no_improve = 0
        else:
            no_improve += 1
    return h, best

def _slsqp(h, n, dx, alpha, maxit, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    tgt = n / 2.0
    try:
        res = minimize(
            lambda x, a=alpha: _smooth(x, a), h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt}],
            options={'maxiter': maxit, 'ftol': 1e-16}
        )
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    n = len(initial_h_values)

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
