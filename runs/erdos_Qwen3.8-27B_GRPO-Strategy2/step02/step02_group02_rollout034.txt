                r = minimize(_sg_obj, bh, args=(a_pol,), method='SLSQP',
                             bounds=bnds,
                             constraints=[{'type': 'eq', 'fun': _sg_con, 'args': (tgt,)}],
                             options={'maxiter': 600, 'ftol': 1e-15})
                hf = _proj(r.x, tgt)
                cf = _c5(hf, dx)
                if cf < bc:
                    bc, bh = cf, hf
            except Exception:
                pass

    # Phase 8: Final intensive pairwise burnishing
    if time.time() - t0 < budget_s - 10:
        h, c = _pwl_random(bh, dx, 800, rng)
        if c < bc:
            bc, bh = c, h
    if time.time() - t0 < budget_s - 5:
        h, c = _pwl_cyclic(bh, dx, 400, rng)
        if c < bc:
            bc, bh = c, h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
