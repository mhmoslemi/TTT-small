def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    n1 = n - 1
    k = np.arange(-n1, n + 1)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _precompute_gs(h, n):
    return np.array([_g_vec(h, p, n) for p in range(n)])

def _sweep_pass(h, n, dx, n_coarse, n_fine):
    corr = _corr(h).astype(float)
    gs = _precompute_gs(h, n)
    improved = False
    n1 = n - 1
    cols = np.arange(n)
    cur_max = float(np.max(corr))
    for i in range(n):
        for j in range(i + 1, n):
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            g = gs[i] - gs[j]
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + np.outer(ds, g), axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = ds[max(0, bi - 3)]
            hi_r = ds[min(n_coarse - 1, bi + 3)]
            ds2 = np.linspace(lo, hi_r, n_fine)
            vals2 = np.max(corr[None, :] + np.outer(ds2, g), axis=1)
            bi2 = int(np.argmin(vals2))
            d_best = float(ds2[bi2])
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * g
            gs[:, cols + n1 - i] -= d_best
            gs[:, i + n1 - cols] -= d_best
            gs[:, cols + n1 - j] += d_best
            gs[:, j + n1 - cols] += d_best
            nm = float(np.max(corr))
            if nm < cur_max:
                cur_max = nm
            improved = True
    return h, corr, cur_max, improved

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _slsqp(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        x = np.clip(res.x, 0, 1)
        diff = target - x.sum()
        if abs(diff) > 1e-10:
            order = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - x[idx]); x[idx] += a; diff -= a
                else:
                    a = min(-diff, x[idx]); x[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
        return x
    except Exception:
        return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

        h_cur = best_h.copy()

            h_cur, corr, cmax, imp = _sweep_pass(h_cur, n, dx, nc, nf)
            c5 = cmax * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
            if not imp or c5 > best_c5 - 1e-15:
                break

    # Phase 2: SLSQP polish with increasing alpha
    for alpha in [300, 800, 2000]:
        if time.time() - t0 > budget_s - 120:
            break
        h_s = _slsqp(best_h, n, target, alpha, 400)
        c5s = _c5(h_s, dx)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = h_s

    # Phase 3: Diversification - perturb and re-sweep
    for trial in range(8):
        if time.time() - t0 > budget_s - 80:
            break
        h_p = best_h + 0.05 * rng.randn(n)
        h_p = _repair(h_p, n)
        h_p, corr_p, cm_p, _ = _sweep_pass(h_p, n, dx, 200, 60)
        h_p, corr_p, cm_p, _ = _sweep_pass(h_p, n, dx, 300, 80)
        c5p = cm_p * dx
        if c5p < best_c5:
            best_c5 = c5p
            best_h = h_p.copy()
        # Also try SLSQP on perturbed
        if time.time() - t0 > budget_s - 80:
            break
        h_s = _slsqp(h_p, n, target, 500, 300)
        h_s, corr_s, cm_s, _ = _sweep_pass(h_s, n, dx, 300, 80)
        c5s = cm_s * dx
        if c5s < best_c5:
            best_c5 = c5s
            best_h = h_s.copy()

    # Phase 4: Final ultra-fine sweeps
    h_cur = best_h.copy()
    for it in range(100):
        if time.time() - t0 > budget_s - 30:
            break
        h_cur, corr, cmax, imp = _sweep_pass(h_cur, n, dx, 500, 120)
        c5 = cmax * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
        if not imp or c5 > best_c5 - 1e-15:
            break

    # Phase 5: Final SLSQP
    if time.time() - t0 < budget_s - 5:
        h_s = _slsqp(best_h, n, target, 5000, 800)
        c5s = _c5(h_s, dx)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = h_s
        # One more sweep after SLSQP
        if time.time() - t0 < budget_s - 2:
            best_h, _, cmax, _ = _sweep_pass(best_h, n, dx, 500, 120)
            c5s = cmax * dx
            if c5s < best_c5:
                best_c5 = c5s

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
