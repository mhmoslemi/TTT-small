                    best_c, best_h = nc, h.copy()
            else:
                h[j] = h[i]; h[i] = tmp
    return best_h, best_c

def _descend(h, n, dx, t_end):
    best_c = _c5(h, dx)
    h = h.copy().astype(float)
    ds = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                   -0.001, -0.003, -0.008, -0.02, -0.05])
    nat = 0
    while time.time() < t_end and nat < 5:
        imp = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in ds:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                nd = -d
                order = np.argsort(h if nd < 0 else 1.0 - h)
                for j in order:
                    if j == i:
                        continue
                    nj = h[j] + nd
                    if nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best_c - 1e-15:
                        best_c = c
                        imp = True
                        break
                    h[i], h[j] = vi, nj - nd
            if imp:
                break
        nat = 0 if imp else nat + 1
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    hp = np.array(initial_h_values, dtype=float).copy()
    np_ = len(hp)
    bh = _repair(hp, np_)
    bc = _c5(bh, 2.0 / np_)
    bn = np_

    # Phase 1: SLSQP polish on parent
    p1e = t0 + min(budget_s * 0.10, 100)
    for alpha in [500, 1200, 3000, 6000]:
        if time.time() > p1e:
            break
        try:
            r = minimize(_lse_obj, bh, args=(alpha, bn), method='SLSQP',
                         bounds=[(0, 1)] * bn,
                         constraints=[{'type': 'eq', 'fun': _eqcon, 'args': (bn / 2,)}],
                         options={'maxiter': 250, 'ftol': 1e-16})
            hc = _repair(r.x, bn)
            c = _c5(hc, 2.0 / bn)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass
    for k in [10, 5, 3]:
        if time.time() > p1e:
            break
        try:
            r = minimize(_topk_obj, bh, args=(k, bn), method='SLSQP',
                         bounds=[(0, 1)] * bn,
                         constraints=[{'type': 'eq', 'fun': _eqcon, 'args': (bn / 2,)}],
                         options={'maxiter': 150, 'ftol': 1e-16})
            hc = _repair(r.x, bn)
            c = _c5(hc, 2.0 / bn)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass

    # Phase 2: Try other n via interpolation
    p2e = t0 + min(budget_s * 0.28, 300)
    for nt in [80, 90, 100, 110]:
        if time.time() > p2e:
            break
        if nt == bn:
            continue
        hx = _interp(bh, bn, nt)
        hx = _repair(hx, nt)
        cx = _c5(hx, 2.0 / nt)
        if cx < bc + 0.0015:
            try:
                r = minimize(_lse_obj, hx, args=(2000, nt), method='SLSQP',
                             bounds=[(0, 1)] * nt,
                             constraints=[{'type': 'eq', 'fun': _eqcon, 'args': (nt / 2,)}],
                             options={'maxiter': 200, 'ftol': 1e-16})
                hc = _repair(r.x, nt)
                c = _c5(hc, 2.0 / nt)
                if c < bc:
                    bh, bc, bn = hc, c, nt
            except Exception:
                pass

    # Phase 3: Subgradient descent (main workhorse)
    p3e = t0 + min(budget_s * 0.55, 580)
    if time.time() < p3e:
        bh, bc = _subgrad(bh, bn, 2.0 / bn, rng, p3e, n_top=5)

    # Phase 4: SA for escape from local optima
    p4e = t0 + min(budget_s * 0.78, 840)
    if time.time() < p4e:
        bh, bc = _sa(bh, bn, 2.0 / bn, rng, p4e)

    # Phase 5: Fine coordinate descent
    p5e = t0 + budget_s - 35
    if time.time() < p5e:
        bh, bc = _descend(bh, bn, 2.0 / bn, p5e)

    # Phase 6: Final SLSQP polish
    if time.time() < t0 + budget_s - 5:
        try:
            r = minimize(_lse_obj, bh, args=(8000, bn), method='SLSQP',
                         bounds=[(0, 1)] * bn,
                         constraints=[{'type': 'eq', 'fun': _eqcon, 'args': (bn / 2,)}],
                         options={'maxiter': 300, 'ftol': 1e-17})
            hc = _repair(r.x, bn)
            c = _c5(hc, 2.0 / bn)
            if c < bc:
                bh, bc = hc, c
        except Exception:
            pass

    bh = _repair(bh, bn)
    bc = _c5(bh, 2.0 / bn)
    return (bh, bc, bn)
