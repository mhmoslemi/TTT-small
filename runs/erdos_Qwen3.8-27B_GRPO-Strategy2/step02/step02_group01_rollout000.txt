import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _project(y, n):
    target = n / 2.0
    x = np.clip(y, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(x + lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.sum() > 0:
            x[free] += diff / float(free.sum())
            x = np.clip(x, 0.0, 1.0)
    return x


def _smooth_c5(x, alpha, top_k):
    c = _corr(x)
    idx = np.argsort(c)[::-1][:top_k]
    ct = c[idx]
    m = ct[0]
    s = (ct - m) * alpha
    s = np.clip(s, -300, 300)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)


def _grad_smooth(x, alpha, top_k):
    c = _corr(x)
    n = len(x)
    idx = np.argsort(c)[::-1][:top_k]
    ct = c[idx]
    m = ct[0]
    s = (ct - m) * alpha
    s = np.clip(s, -300, 300)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        d = int(ki) - (n - 1)
        # Term 1: (1-h[j-d]) for valid j
        js1 = max(0, d)
        je1 = min(n, n + d)
        if js1 < je1:
            g[js1:je1] += wi * (1.0 - x[js1 - d: je1 - d])
        # Term 2: -h[j+d] for valid j
        js2 = max(0, -d)
        je2 = min(n, n - d)
        if js2 < je2:
            g[js2:je2] -= wi * x[js2 + d: je2 + d]
    return g

    return _smooth_c5(x, alpha, top_k)


def _slsqp_jac(x, alpha, top_k):
    return _grad_smooth(x, alpha, top_k)


def _sum_con(x, n):
    return n / 2.0 - np.sum(x)


def _sum_con_jac(x, n):
    return -np.ones(n)
