                break
    return best, best_c5

def _targeted_ls(h, n, dx, rng, deadline, top_k=10, n_inner=400):
    best = h.copy()
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for scale in scales:
        stalled = 0
        for _pass in range(250):
            if time.time() >= deadline:
                break
            improved = False
            for _ in range(n_inner):
                if time.time() >= deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c5 = _compute_c5(h, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
            if improved:
                stalled = 0
            else:
                stalled += 1
                if stalled >= 4:
                    break
    return best, best_c5

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _compute_c5(h, dx)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: _obj_grad(x, alpha)[0], h,
            jac=lambda x: _obj_grad(x, alpha)[1],
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq',
                          'fun': lambda x: np.sum(x) - target,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-18}
        )
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _multipoint_start(h_base, n, rng, n_starts=4):
    starts = [h_base.copy()]
    for s in range(n_starts - 1):
        noise = rng.randn(n)
        noise -= noise.mean()
        cands = [0.02, 0.05, 0.08, 0.12][s % 4]
        starts.append(_repair(h_base + cands * noise, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Projected gradient with increasing alpha (280s)
    t1_end = t0 + 280
    for alpha in [15, 40, 100, 250, 600, 1500, 4000]:
        if time.time() > t1_end or rem() < 350:
            break
        dl = min(t1_end, time.time() + 40)
        best_h, best_c5 = _proj_grad(best_h, alpha, n, dx, 15000, 0.01, dl)

    # Phase 2: Multi-start SLSQP (280s)
    t2_end = t0 + 560
    starts = _multipoint_start(best_h, n, rng, 3)
    for alpha in [400, 1000, 3000]:
        if time.time() > t2_end or rem() < 250:
            break
        for st in starts:
            if time.time() > t2_end or rem() < 250:
                break
            dl = min(t2_end, time.time() + 50)
            h_try, c5_try = _slsqp(st, n, dx, alpha, 600, dl)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: Targeted local search (200s)
    if rem() > 170:
        dl = time.time() + min(200, rem() - 130)
        best_h, best_c5 = _targeted_ls(best_h, n, dx, rng, dl, top_k=10, n_inner=500)

    # Phase 4: Final gradient + SLSQP polish (120s)
    if rem() > 100:
        dl = time.time() + min(40, rem() - 60)
        best_h, best_c5 = _proj_grad(best_h, 8000, n, dx, 20000, 0.002, dl)
    if rem() > 50:
        dl = time.time() + min(30, rem() - 20)
        best_h, best_c5 = _slsqp(best_h, n, dx, 10000, 800, dl)
    if rem() > 20:
        dl = time.time() + min(15, rem() - 5)
        best_h, best_c5 = _targeted_ls(best_h, n, dx, rng, dl, top_k=5, n_inner=300)

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
