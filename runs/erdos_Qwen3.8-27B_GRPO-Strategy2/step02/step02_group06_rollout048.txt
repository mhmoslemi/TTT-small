    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14: break
    return h

def _gvec_all(h, n):
    """Precompute all gradient vectors."""
    M = 2 * n - 1
    k = np.arange(-(n-1), n)
    ga = np.zeros((n, M))
    for p in range(n):
        idx1 = p - k
        m1 = (idx1 >= 0) & (idx1 < n)
        ga[p, m1] += (1.0 - h[idx1[m1]])
        idx2 = p + k
        m2 = (idx2 >= 0) & (idx2 < n)
        ga[p, m2] -= h[idx2[m2]]
    return ga

    s = np.clip(alpha * (c - m), -400, 400)
    return m + float(np.logsumexp(s)) / alpha
