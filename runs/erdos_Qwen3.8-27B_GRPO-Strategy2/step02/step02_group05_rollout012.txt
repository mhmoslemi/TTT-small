def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _project(h, target):

        h_p = np.clip(h - mid, 0, 1)

        if s > target:
            lo = mid
        else:
            hi = mid

    return h_p

                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = oi, oj
            else:
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                d1 = float(rng.uniform(-0.15, 0.15))
                d2 = float(rng.uniform(-0.15, 0.15))
                di = d1 - d2
                ni, nj = best[i] + d1, best[j] + d2
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                if abs(best.sum() - n / 2) > 1e-6:
                    best[i], best[j] = oi, oj
                    continue
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = oi, oj
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return best, best_c5

def _structured_init(n, seed_off):
    rng2 = np.random.RandomState(42 + seed_off * 137)
    t = np.linspace(0, 2.0, n, endpoint=False)
    base = 0.5 + 0.25 * np.sin(2 * np.pi * t * (1 + seed_off % 4))
    base += 0.15 * np.cos(4 * np.pi * t + seed_off)
    base += 0.05 * rng2.randn(n)
    return _repair(base, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(best_h, dx)

    def alive(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: Subgradient descent (multi-shift)
    if alive(550):
        dl = t0 + budget_s - 550
        for m in [1, 3, 5]:
            if not alive(400):
                break
            h_sub, c5_sub = _subgrad(best_h, n, dx, 500, rng, dl, m=m)
            if c5_sub < best_c5:
                best_c5, best_h = c5_sub, h_sub

    # Phase 2: SLSQP polish
    if alive(400):
        for a in [100, 250, 500]:
            if not alive(350):
                break
            hc = _slsqp(best_h, n, dx, a, 400)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc

    # Phase 3: Local search
    if alive(250):
        dl = t0 + budget_s - 250
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl)

    # Phase 4: Nelder-Mead + re-localize
    if alive(200):
        h_nm = _nelder_mead(best_h, n, dx, 2000, t0 + budget_s - 150)
        nc = _c5(h_nm, dx)
        if nc < best_c5:
            best_c5, best_h = nc, h_nm
        if alive(150):
            dl = t0 + budget_s - 150
            best_h, best_c5 = _local_search(best_h, n, dx, rng, dl)

    # Phase 5: Restart with perturbations
    sigmas = [0.1, 0.2, 0.3, 0.15, 0.25, 0.4, 0.12, 0.35]
    it = 0
    while alive(100):
        sigma = sigmas[it % len(sigmas)]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 300, 250)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc
        elif nc < best_c5 + 0.003 and alive(110):
            dl = t0 + budget_s - 110
            hl, cl = _local_search(hc, n, dx, rng, dl)
            if cl < best_c5:
                best_c5, best_h = cl, hl
        it += 1

    # Phase 6: Try structured inits
    for si in [1, 2, 3]:
        if not alive(80):
            break
        hs = _structured_init(n, si)
        hc = _slsqp(hs, n, dx, 300, 300)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc
            if alive(70):
                dl = t0 + budget_s - 70
                best_h, best_c5 = _local_search(best_h, n, dx, rng, dl)

    # Phase 7: Final polish - subgradient + SLSQP + local search
    if alive(40):
        h_sub, c5_sub = _subgrad(best_h, n, dx, 200, rng, t0 + budget_s - 30, m=3)
        if c5_sub < best_c5:
            best_c5, best_h = c5_sub, h_sub
    if alive(20):
        hc = _slsqp(best_h, n, dx, 800, 300)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc
    if alive(8):
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t0 + budget_s - 2)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
