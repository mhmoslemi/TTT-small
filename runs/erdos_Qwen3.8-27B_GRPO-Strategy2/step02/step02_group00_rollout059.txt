import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)

def _obj_topk(x, k, n):
    corr = _corr(x)
    if k >= len(corr):
        return float(corr.sum()) * (2.0 / n)
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum()) * (2.0 / n)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_lse, h, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_descent(h, n, dx, t_dead, fine=False):

    local_best = _c5(h, dx)
    if fine:

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)


def _obj_lse_pen(x, alpha, n, target, pen):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    base = (m + np.sum(s) / alpha) * (2.0 / n)
    cv = float(np.sum(x) - target)
    return base + pen * cv * cv


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _slsqp(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_lse, h, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h


def _powell_pen(h, n, alpha, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    try:
        res = minimize(_obj_lse_pen, h, args=(alpha, n, target, pen),
                       method='Powell', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h


def _sa(h, n, dx, t_dead, rng):
    h_sa = h.copy()
    sa_best = _c5(h_sa, dx)
    sa_best_h = h_sa.copy()
    T0 = 0.010
    Tend = 0.000002
    max_it = 100000000
    for it in range(max_it):
        if time.time() > t_dead:
            break
        frac = (it + 1) / max_it
        T = T0 * (Tend / T0) ** frac
        dmax = 0.05 * (1.0 - 0.7 * frac)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (rng.rand() - 0.5) * 2.0 * dmax
        ni = h_sa[i] + delta
        if ni < 0.0 or ni > 1.0:
            continue
        nj = h_sa[j] - delta
        if nj < 0.0 or nj > 1.0:
            continue
        h_sa[i] = ni
        h_sa[j] = nj
        new_c5 = _c5(h_sa, dx)
        if new_c5 <= sa_best + T:
            if new_c5 < sa_best - 1e-16:
                sa_best = new_c5
                sa_best_h = h_sa.copy()
        else:
            h_sa[i] -= delta
            h_sa[j] += delta
    return sa_best_h, sa_best


def _coord_descent(h, n, dx, t_dead, fine):
    h = h.copy()
    local_best = _c5(h, dx)
    if fine:
        deltas = np.array([0.0002, 0.0005, 0.001, 0.002, -0.0002, -0.0005, -0.001, -0.002])
    else:
        deltas = np.array([0.003, 0.006, 0.01, 0.02, 0.04, 0.08, -0.003, -0.006, -0.01, -0.02, -0.04, -0.08])
    no_improve = 0
    while time.time() < t_dead:
        improved = False
        for i in range(n):
            if time.time() > t_dead:
                break
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                for j in order[:15]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-15:
                        local_best = c5_try
                        improved = True
                        break
                    else:
                        h[i] = vi
                        h[j] = nj - need
                if improved:
                    break
            if improved:
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 4:
                break
    return h, local_best
