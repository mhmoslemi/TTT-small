import numpy as np
from scipy.optimize import minimize
import time

_ALPH = [100.0]
_TARG = [35.0]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, dx):
    return float(np.max(_corr(h))) * dx

def _obj(x, a):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(a * (c - m), -600, 600)
    e = np.exp(s)
    return m + np.log(e.sum()) / a

def _grad(x, a):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(a * (c - m), -600, 600)
    e = np.exp(s)
    w = e / e.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            p = -lag
            g[:n - p] += wk * (1.0 - x[p:])
            g[p:] -= wk * x[:n - p]
    return g

def _proj(h, t):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            for i in np.argsort(1.0 - h):
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
                if abs(d) < 1e-12:
                    break
        else:
            for i in np.argsort(h):
                a = min(-d, h[i])
                h[i] -= a
                d += a
                if abs(d) < 1e-12:
                    break
    return h

def _obj_w(x):
    return _obj(x, _ALPH[0])

def _grad_w(x):
    return _grad(x, _ALPH[0])

def _con_w(x):
    return x.sum() - _TARG[0]

def _slsqp(h, n, dx, t, a, mi, dl):
    if time.time() >= dl:
        return h, _c5v(h, dx)
    _ALPH[0] = a
    _TARG[0] = t
    try:
        r = minimize(_obj_w, h, jac=_grad_w, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_w}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        hc = _proj(r.x, t)
        return hc, _c5v(hc, dx)
    except Exception:
        return h, _c5v(h, dx)

def _sa(h, n, dx, t, dl, rng, iters=60000):

    s = int(h1.sum())

        m1 = m1[np.argsort(h[m1])]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, dx):
    return float(np.max(_corr(h))) * dx

def _obj(x, a):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(a * (c - m), -600, 600)
    e = np.exp(s)
    return m + np.log(e.sum()) / a

def _grad(x, a):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(a * (c - m), -600, 600)
    e = np.exp(s)
    w = e / e.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wk * (1.0 - x[:n - lag])
            g[:n - lag] -= wk * x[lag:]
        else:
            p = -lag
            g[:n - p] += wk * (1.0 - x[p:])
            g[p:] -= wk * x[:n - p]
    return g

def _proj(h, t):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            for i in np.argsort(1.0 - h):
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
                if abs(d) < 1e-12:

        else:
            for i in np.argsort(h):
                a = min(-d, h[i])
                h[i] -= a
                d += a

                    break
    return h
