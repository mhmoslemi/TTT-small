def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        signs = np.sign(1.0 - h) if diff > 0 else -np.sign(h)
        order = np.argsort(np.abs(signs))
        # Actually let's just do the simple approach
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:

            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, -h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-14:

    return h

def _grad_corr(h, n, m):
    """Compute gradient of correlation at lag m w.r.t. h"""
    g = np.zeros(n)
    # corr(m) = sum_i h[i]*(1-h[i+m]) for valid i
    # d/dh[j]: (1-h[j+m]) if j valid as first index
    #          -h[j-m] if j valid as second index (i.e., j-m is valid first index)
    for j in range(n):
        # As first index h[i] where i=j: need 0<=j<n and 0<=j+m<n
        if 0 <= j < n and 0 <= j + m < n:
            g[j] += (1.0 - h[j + m])
        # As second index h[i+m] where i+m=j, i=j-m: need 0<=j-m<n and 0<=j<n
        if 0 <= j - m < n and 0 <= j < n:
            g[j] -= h[j - m]
    return g

    g = np.zeros(n)
    for m in lags:
        g += _grad_corr(h, n, m)
    g /= len(lags)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, -h[idx])
            if add == 0:
                continue
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    corr = _corr(x)
    m = corr.max()
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    best_h, best_c5 = h, _c5(h, dx)
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=[eq_con],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    except Exception:
        pass
    return best_h, best_c5

def _de_step(pop, F, rng, n, deadline):
    """One DE/rand/1/bin step for each particle"""
    pops = len(pop)
    for i in range(pops):
        if time.time() >= deadline:
            break
        candidates = [j for j in range(pops) if j != i]
        a, b = rng.choice(candidates, 2, replace=False)
        v = pop[i] + F * (pop[a] - pop[b])
        v = _repair(v, n)
        if _c5(v, 2.0/n) < _c5(pop[i], 2.0/n):
            pop[i] = v
    return pop

def _diff_evo(pop, n, dx, F, rng, deadline):
    """Differential Evolution with adaptive F"""
    pops = len(pop)
    best_c5 = min(_c5(p, dx) for p in pop)
    fits = [_c5(p, dx) for p in pop]
    it = 0
    while time.time() < deadline:
        it += 1
        F_adapt = max(0.3, F - 0.3 * it / 80.0)
        _de_step(pop, F_adapt, rng, n, deadline)
        new_fits = [_c5(p, dx) for p in pop]
        new_best = min(new_fits)
        if new_best < best_c5:
            best_c5 = new_best
    best_idx = int(np.argmin(fits))
    return pop[best_idx], best_c5

def _corr_targeted_local(h, n, dx, rng, deadline):
    """Local search targeting active correlations"""
    best_c5 = _c5(h, dx)
    h = h.copy()
    n_corr = 2 * n - 1
    for scale in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        stale = 0
        while stale < 2000:
            if time.time() >= deadline:
                return h, best_c5
            stale += 1
            corr = _corr(h)
            mx = corr.max()
            threshold = mx - 0.002
            active = np.where(corr >= threshold)[0]
            improved = False
            for _ in range(300):
                if time.time() >= deadline:
                    return h, best_c5
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                h[i], h[j] = ni, nj
                c5 = _c5(h, dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                    stale = 0
                else:
                    h[i], h[j] = ni - d, nj + d
            if not improved:

    return h, best_c5

def _fine_local(h, n, dx, rng, deadline):
    """Very fine random search"""
    best_c5 = _c5(h, dx)
    h = h.copy()
    for scale in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
