def _grad_corr(h, n, idx):
    """Gradient of correlation at index idx w.r.t. h."""
    lag = idx - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    else:
        for j in range(n):
            if 0 <= j + lag < n:
                g[j] += 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                g[j] -= h[j - lag]
    return g

def _smooth_obj(x, n, alpha):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    ve = e / e.sum()
    return m + np.log(e.sum()) / alpha, ve, c

def _smooth_grad(x, n, alpha):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    ve = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if ve[k] > 1e-8:
            g += ve[k] * _grad_corr(x, n, k)
    g *= alpha
    # Also add subgradient of m (the max)
    active = np.where(np.abs(c - m) < 1e-12)[0]
    for k in active:
        g += (1.0 - alpha * 0) / len(active) * _grad_corr(x, n, k)
    # Actually the gradient of m + log(sum exp)/alpha is:
    # grad(m) + alpha * sum_k ve[k] * grad(c_k) - alpha * sum_k ve[k] * grad(m)
    # = grad(m) * (1 - alpha*sum(ve)) + alpha * sum(ve*grad(c_k))
    # = grad(m) + alpha * sum(ve*grad(c_k))  [since sum(ve)=1]
    # Wait no. Let me redo.
    # f(x) = m + log(sum_k exp(alpha*(c_k - m))) / alpha
    # = m + log(sum_k exp(alpha*c_k - alpha*m)) / alpha
    # = m + (log(sum_k exp(alpha*c_k)) - alpha*m) / alpha
    # = m + log(sum_k exp(alpha*c_k))/alpha - m
    # = log(sum_k exp(alpha*c_k))/alpha
    # So f(x) = log(sum_k exp(alpha*c_k))/alpha  (the log-sum-exp of alpha*c divided by alpha)
    # grad f = (1/alpha) * sum_k [alpha*exp(alpha*c_k)/sum(exp(alpha*c_k))] * grad(c_k)
    # = sum_k ve[k] * grad(c_k)  where ve[k] = exp(alpha*c_k)/sum(exp(alpha*c_k))
    
    # Let me recompute properly
    s2 = np.clip(alpha * c, -400, 400)
    e2 = np.exp(s2)
    ve2 = e2 / e2.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if ve2[k] > 1e-10:
            g += ve2[k] * _grad_corr(x, n, k)
    return g
