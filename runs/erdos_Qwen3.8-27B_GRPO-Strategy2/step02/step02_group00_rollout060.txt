import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lse_obj(h, alpha, lam, n, target):
    corr = _corr(h)
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(alpha * (corr - m), -50, 50)))
    lse_val = (m + np.log(s) / alpha)
    pen = lam * (np.sum(h) - target) ** 2
    return lse_val * (2.0 / n) + pen

def _lse_grad(h, alpha, lam, n, target):
    corr = _corr(h)
    m = np.max(corr)
    exps = np.exp(np.clip(alpha * (corr - m), -50, 50))
    w = exps / np.sum(exps)
    # w has length 2n-1
    grad = np.zeros(n)
    # k = m_idx - (n-1), where m_idx is 0..2n-2
    for m_idx in range(2 * n - 1):
        wi = w[m_idx]
        if wi < 1e-15:
            continue
        k = m_idx - (n - 1)
        # dcorr[m_idx]/dh[j]:
        # (1 - h[j+k]) for j where 0<=j<n and 0<=j+k<n
        # - h[j-k] for j where 0<=j<n and 0<=j-k<n
        j_lo1 = max(0, -k)
        j_hi1 = min(n, n - k)
        if j_lo1 < j_hi1:
            grad[j_lo1:j_hi1] += wi * (1.0 - h[j_lo1 + k:j_hi1 + k])
        j_lo2 = max(0, k)
        j_hi2 = min(n, n + k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wi * h[j_lo2 - k:j_hi2 - k]
    # Scale by 2/n for dx
    grad *= (2.0 / n)
    # Penalty gradient
    grad += 2.0 * lam * (np.sum(h) - target) * np.ones(n)
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _lse_og(h, alpha, lam, n, target):
    corr = _corr(h)
    m = float(corr.max())
    ex = np.exp(np.clip(alpha * (corr - m), -50, 50))
    se = float(ex.sum())
    obj = (m + np.log(se) / alpha) * (2.0 / n) + lam * (h.sum() - target) ** 2
    w = ex / se
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wm = w[mi]
        if wm < 1e-14:
            continue
        k = mi - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += wm * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 + k:hi2 + k]
    grad *= (2.0 / n)
    grad += 2.0 * lam * (h.sum() - target)
    return obj, grad

def _lse_fun(x, alpha, lam, n, target):
    o, _ = _lse_og(x, alpha, lam, n, target)
    return o

def _lse_jac(x, alpha, lam, n, target):
    _, g = _lse_og(x, alpha, lam, n, target)
    return g

def _run_lbfgs(h, alpha, lam, n, target, maxit):
    res = minimize(_lse_fun, h, method='L-BFGS-B', jac=_lse_jac,
                   args=(alpha, lam, n, target),
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
    return _repair(res.x, n)

def _sa_opt(h, dx, rng, t_end, n):
    h = h.copy()
    best = _c5(h, dx)
    best_h = h.copy()
    T0, Tend = 0.005, 0.00001
    it = 0
    total_it = 1000000
    while time.time() < t_end:
        frac = min(it / float(total_it), 1.0)
        T = T0 * (Tend / T0) ** frac
        dmax = 0.05 * (1.0 - 0.7 * frac)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            it += 1
            continue
        d = (rng.rand() - 0.5) * 2.0 * dmax
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            it += 1
            continue
        h[i], h[j] = ni, nj
        nc = _c5(h, dx)
        if nc <= best + T:
            if nc < best - 1e-15:
                best = nc
                best_h = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
        it += 1
    return best_h, best

def _coord_desc(h, dx, n, t_end):
    h = h.copy()
    best = _c5(h, dx)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.04])
    all_d = np.concatenate([deltas, -deltas])
    no_imp = 0
    while time.time() < t_end and no_imp < 5:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for d in all_d:
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                need = -d
                if need > 0:
                    order = np.argsort(h)
                else:
                    order = np.argsort(-h)
                found = False
                for j in order[:15]:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-15:
                        best = c
                        improved = True
                        found = True
                        break
                    else:
                        h[i], h[j] = vi, nj - need
                if found:
                    break
        if not improved:
            no_imp += 1
    return h, best

def _interp_to_n(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    f = interp1d(old_x, h, kind='linear', fill_value='extrapolate', bounds_error=False)
    new_h = f(new_x)
    return _repair(new_h, new_n)
