from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
            else:
                a = max(diff, -x[k])
            x[k] += a
            diff -= a
            if abs(diff) < 1e-13:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _fista(h0, alpha, target, iters):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for it in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        fn = _sobj(xn, alpha)
        quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
        cnt = 0
        while fn > quad + 1e-10 and cnt < 6:
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            fn = _sobj(xn, alpha)
            quad = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _cd_random(h, dx, n, rng, dvals, n_pairs):
    bc = _c5(h, dx)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi0, hj0 = h[i], h[j]
        best_c, best_d = bc, 0.0
        for d in dvals:
            ni, nj = hi0 + d, hj0 - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best_c - 1e-14:
                best_c, best_d = c, d
            h[i], h[j] = hi0, hj0
        if best_d != 0.0:
            h[i] = min(max(hi0 + best_d, 0.0), 1.0)
            h[j] = min(max(hj0 - best_d, 0.0), 1.0)
            bc = best_c
    return h, bc

def _cd_cyclic(h, dx, n, dvals):
    bc = _c5(h, dx)
    for i in range(n):
        for j in range(i + 1, n):
            hi0, hj0 = h[i], h[j]
            best_c, best_d = bc, 0.0
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best_c - 1e-14:
                    best_c, best_d = c, d
                h[i], h[j] = hi0, hj0
            if best_d != 0.0:
                h[i] = min(max(hi0 + best_d, 0.0), 1.0)
                h[j] = min(max(hj0 - best_d, 0.0), 1.0)
                bc = best_c
    return h, bc

def _cd_swap(h, dx, n, rng, n_swaps):
    bc = _c5(h, dx)
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c = _c5(h, dx)
        if c < bc - 1e-14:
            bc = c
        else:
            h[i], h[j] = h[j], h[i]
    return h, bc

def _guided_cd(h, dx, n, rng, t_end, dvals):
    bc = _c5(h, dx)
    stall = 0
    while time.time() < t_end:
        if stall > 30000:
            break
        ov = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(ov))
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += (1.0 - h[i + lag])
                g[i + lag] -= h[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += (1.0 - h[i - L])
                g[i - L] -= h[i]
        top_idx = np.argsort(-np.abs(g))[:12]
        i = int(top_idx[rng.randint(len(top_idx))])
        j = int(rng.randint(n))
        if j == i:
            j = (j + 1) % n
        hi0, hj0 = h[i], h[j]
        best_c, best_d = bc, 0.0
        for d in dvals:
            ni, nj = hi0 + d, hj0 - d
            ni2, nj2 = hj0 + d, hi0 - d
            if ni >= 0.0 and ni <= 1.0 and nj >= 0.0 and nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best_c - 1e-14:
                    best_c, best_d = c, d
                h[i], h[j] = hi0, hj0
            if ni2 >= 0.0 and ni2 <= 1.0 and nj2 >= 0.0 and nj2 <= 1.0:
                h[i], h[j] = ni2, nj2
                c = _c5(h, dx)
                if c < best_c - 1e-14:
                    best_c, best_d = c, -d
                h[i], h[j] = hi0, hj0
        if best_d != 0.0:
            if best_d > 0:
                h[i] = min(max(hi0 + best_d, 0.0), 1.0)
                h[j] = min(max(hj0 - best_d, 0.0), 1.0)
            else:
                h[i] = min(max(hi0 + best_d, 0.0), 1.0)
                h[j] = min(max(hj0 - best_d, 0.0), 1.0)
            bc = best_c
            stall = 0
        else:
            stall += 1
    return h, bc

def _slsqp_obj(x, a):
    return _sobj(x, a)

def _slsqp_con(x, target):
    return float(x.sum()) - target

    t0 = time.time()

    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)

    h0 = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    dvals_med = [0.05, 0.03, 0.02, 0.01, 0.005, -0.05, -0.03, -0.02, -0.01, -0.005]
    dvals_fine = [0.003, 0.002, 0.001, 0.0005, 0.0002, -0.003, -0.002, -0.001, -0.0005, -0.0002]
    dvals_all = dvals_med + dvals_fine

    # Phase 1: Quick FISTA to establish good basin
    for a in [200, 1000, 3000]:
        if budget_s - (time.time() - t0) < 60:
            break
        h, c = _fista(bh, a, tgt, 2500)
        if c < bc:
            bc, bh = c, h

    # Phase 2: Guided CD (uses subgradient to target active overlap)
    t_cd1_end = t0 + budget_s * 0.35
    if time.time() < t_cd1_end:
        h, c = _guided_cd(bh, dx, n, rng, t_cd1_end, dvals_all)
        if c < bc:
            bc, bh = c, h

    # Phase 3: Cyclic CD passes (systematic coverage)
    t_cd2_end = t0 + budget_s * 0.55
    while time.time() < t_cd2_end:
        h, c = _cd_cyclic(bh, dx, n, dvals_fine)
        if c < bc:
            bc, bh = c, h
        else:
            break

    # Phase 4: Random CD + swaps
    t_cd3_end = t0 + budget_s * 0.72
    while time.time() < t_cd3_end:
        h, c = _cd_random(bh, dx, n, rng, dvals_fine, 5000)
        if c < bc:
            bc, bh = c, h
        h, c = _cd_swap(bh, dx, n, rng, 3000)
        if c < bc:
            bc, bh = c, h
        if time.time() > t_cd3_end:
            break

    # Phase 5: Multi-start from best
    if budget_s - (time.time() - t0) > 60:
        sigmas = [0.08, 0.12, 0.15, 0.05, 0.1]
        for s in sigmas:
            if budget_s - (time.time() - t0) < 50:
                break
            hp = _proj(bh + s * rng.randn(n), tgt)
            h, c = _fista(hp, 800, tgt, 1500)
            if c < bc:
                bc, bh = c, h
                h, c = _guided_cd(bh, dx, n, rng, time.time() + 8, dvals_fine)
                if c < bc:
                    bc, bh = c, h

    # Phase 6: SLSQP polish
    if budget_s - (time.time() - t0) > 25:
        bnds = [(0.0, 1.0)] * n
        for a_p in [5000, 15000]:
            if budget_s - (time.time() - t0) < 15:
                break
            try:
                r = minimize(_slsqp_obj, bh, args=(a_p,), method='SLSQP',
                            bounds=bnds,
                            constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                            options={'maxiter': 600, 'ftol': 1e-15})
                hf = _proj(r.x, tgt)
                cf = _c5(hf, dx)
                if cf < bc:
                    bc, bh = cf, hf
            except Exception:
                pass

    # Phase 7: Final fine CD
    t_final = t0 + budget_s - 3
    if time.time() < t_final:
        h, c = _guided_cd(bh, dx, n, rng, t_final, dvals_fine)
        if c < bc:
            bc, bh = c, h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
