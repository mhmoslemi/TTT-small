    h = h_full.copy()
    h[block_idx] = block_vals

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

                break
    return h

    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep(h, n, dx, corr, n_coarse, n_fine, fine_wid):
    improved = False
    for i in range(n):
        for j in range(i+1, n):  # only i < j to avoid duplicates

            gd = _g_vec(h, i, n) - _g_vec(h, j, n)

                    lo = max(d_min, d_best - fine_wid)
                    hi2 = min(d_max, d_best + fine_wid)
                    ds2 = np.linspace(lo, hi2, n_fine)
                    nc2 = corr[None, :] + ds2[:, None] * gd[None, :]

                    bi2 = np.argmin(c5s2)
                    d_best = ds2[bi2]
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                improved = True
    return h, corr, improved

def _block_cd(h, n, dx, target, alpha, block_size, n_blocks, time_limit_fn):
    # Block coordinate descent
    rng_block = np.random.RandomState(12345)
    for _ in range(n_blocks):
        if time_limit_fn():
            break
        # Create a random partition into blocks
        perm = rng_block.permutation(n)

                break

                continue
            # The sum constraint for the block

            block_target = target - other_sum
            x0 = h[block_idx].copy()
            
            # Objective: smooth max of correlation
            def obj(xv):
                h_temp = h.copy()
                h_temp[block_idx] = xv
                return _smooth_max(_corr(h_temp), alpha)
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - block_target}]
            bounds = [(0, 1)] * len(block_idx)
            try:

                             constraints=cons, options={'maxiter': 50, 'ftol': 1e-12})
                if res.fun < obj(x0):
                    h[block_idx] = _repair_1d(res.x, block_target)
            except:
                pass
    return h

    if s < 1e-12:

        return np.clip(x, 0, 1)

                               constraints=cons,
                               options={'maxiter': 80, 'ftol': 1e-12})
                new_val = float(res.fun)
                old_val = _smooth_max(_corr(h), alpha)
                if new_val < old_val - 1e-14:
                    h[block_idx] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return h

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _con_full, 'args': (target,)}]
    try:
        res = minimize(_obj_full, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warmup with increasing alpha
    for alpha in [40, 100, 250, 600]:
        if time.time() - t0 > budget_s - 550:
            break
        hc = _slsqp_step(best_h, n, dx, target, bounds, alpha, 800)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 2: Block coordinate descent (block_size=7)
    dl = t0 + budget_s - 420
    h = best_h.copy()
    h = _block_cd(h, n, dx, target, 300, 7, 20, dl)
    c5 = _c5(h, dx)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    # Phase 3: Coarse pair sweeps
    if time.time() - t0 < budget_s - 350:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(50):
            if time.time() - t0 > budget_s - 350:
                break
            h, corr, imp = _sweep(h, n, dx, corr, 15, 0, 0)
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 4: Fine pair sweeps
    if time.time() - t0 < budget_s - 250:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(80):
            if time.time() - t0 > budget_s - 250:
                break
            h, corr, imp = _sweep(h, n, dx, corr, 15, 31, 0.03)
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 5: Block CD with smaller blocks (block_size=4)
    dl5 = t0 + budget_s - 180
    h = best_h.copy()
    h = _block_cd(h, n, dx, target, 800, 4, 15, dl5)
    c5 = _c5(h, dx)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    # Phase 6: Ultra-fine sweeps
    if time.time() - t0 < budget_s - 100:
        h = best_h.copy()
        corr = _corr(h)
        for _ in range(100):
            if time.time() - t0 > budget_s - 100:
                break
            h, corr, imp = _sweep(h, n, dx, corr, 21, 51, 0.008)
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if not imp:
                break

    # Phase 7: Random restarts
    for sr in range(8):
        if time.time() - t0 > budget_s - 35:
            break
        h_pert = best_h + 0.04 * rng.randn(n)
        h_pert = _repair(h_pert, n)
        hc = _slsqp_step(h_pert, n, dx, target, bounds, 400, 500)
        corr_r = _corr(hc)
        hc, corr_r, _ = _sweep(hc, n, dx, corr_r, 15, 21, 0.02)
        c5 = float(np.max(corr_r) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 8: Final high-alpha SLSQP polish
    if time.time() - t0 < budget_s - 12:
        try:
            hc = _slsqp_step(best_h, n, dx, target, bounds, 3000, 2000)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 9: Final ultra-fine sweep
    if time.time() - t0 < budget_s - 4:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep(h, n, dx, corr, 25, 81, 0.003)
        corr = _corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
