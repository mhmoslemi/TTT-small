import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _obj_max(h, n):
    """Compute the max correlation value (without dx)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr), corr

def _coord_step(h, n):
    """One full coordinate descent sweep. Returns new h and improvement."""
    h_new = h.copy()
    for i in range(n):
        t = h_new[i]
        c = np.correlate(h_new, 1.0 - h_new, mode='full')
        
        # Compute A_k = dc_k/dt
        A = np.zeros(2*n-1)
        for m in range(2*n-1):
            j1 = i - m + n - 1
            if 0 <= j1 < n:
                A[m] += (1.0 - h_new[j1])
            j2 = m - n + 1 + i
            if 0 <= j2 < n:
                A[m] -= h_new[j2]
        
        B = c - A * t
        B0 = B[n-1]
        
        # Candidate t values
        candidates = [0.0, 1.0]
        for m in range(2*n-1):
            if m == n-1:
                continue
            b_coef = A[m] - 1.0
            c_coef = B[m] - B0
            disc = b_coef**2 - 4.0*c_coef
            if disc >= 0:
                sq = np.sqrt(disc)
                t1 = (-b_coef + sq) / 2.0
                t2 = (-b_coef - sq) / 2.0
                if 0.0 <= t1 <= 1.0:
                    candidates.append(t1)
                if 0.0 <= t2 <= 1.0:
                    candidates.append(t2)
        
        # Add linspace points
        candidates.extend(np.linspace(0, 1, 15).tolist())
        candidates = np.array(candidates)
        
        # Evaluate
        best_t = t
        best_val = 1e18
        for tc in candidates:
            tc = max(0.0, min(1.0, tc))
            vals = A * tc + B
            vals[n-1] = -tc*tc + tc + B0
            v = np.max(vals)
            if v < best_val:
                best_val = v
                best_t = tc
        
        h_new[i] = best_t
    
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _coord_sweep(h, n):
    h_new = h.copy()
    for i in range(n):
        t = h_new[i]
        c = np.correlate(h_new, 1.0 - h_new, mode='full')
        
        m_range = np.arange(2 * n - 1)
        j1 = i - m_range + n - 1
        j2 = m_range - n + 1 + i
        mask1 = (j1 >= 0) & (j1 < n)
        mask2 = (j2 >= 0) & (j2 < n)
        A = np.zeros(2 * n - 1)
        A[mask1] += (1.0 - h_new[j1[mask1]])
        A[mask2] -= h_new[j2[mask2]]
        
        B = c - A * t
        B0 = B[n - 1]
        
        # Find optimal t
        candidates = np.array([0.0, 1.0])
        
        # Parabola-line intersections
        for m in range(2 * n - 1):
            if m == n - 1:
                continue
            b = A[m] - 1.0
            cc = B[m] - B0
            disc = b * b - 4.0 * cc
            if disc >= 0:
                sq = np.sqrt(disc)
                for tc in [(-b + sq) / 2.0, (-b - sq) / 2.0]:
                    if 0.0 <= tc <= 1.0:
                        candidates = np.append(candidates, tc)
        
        # Add grid points
        candidates = np.append(candidates, np.linspace(0, 1, 20))
        
        best_t = t
        best_val = 1e18
        for tc in candidates:
            tc = max(0.0, min(1.0, float(tc)))
            vals = A * tc + B
            vals[n - 1] = -tc * tc + tc + B0
            v = np.max(vals)
            if v < best_val:
                best_val = v
                best_t = tc
        
        h_new[i] = best_t
    
    return h_new

def _smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h_init, n, dx, target_sum, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    def eq_fun(x):
        return np.sum(x) - target_sum
    try:
        res = minimize(
            lambda x: _smooth_obj(x, alpha, n), h_init,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h_init, _compute_c5(h_init, dx)

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _coord_sweep(h, n):

def _slsqp_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_eq(x, target):
    return np.sum(x) - target

def _slsqp(h_init, n, target, alpha, maxiter, deadline):
    dx = 2.0 / n
    if time.time() >= deadline:
        return h_init, _compute_c5(h_init, dx)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _slsqp_obj, h_init, args=(alpha, n),
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': _slsqp_eq, 'args': (target,)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h_init, _compute_c5(h_init, dx)

                best_c5 = c5

    # Phase 5: Local search fine-tuning
    if rem() > 40:
        t_end = time.time() + min(40, rem() - 10)
        best_h, best_c5 = _local_search(best_h, n, dx, rng, t_end)

    # Phase 6: Final coordinate descent polish
    if rem() > 10:
        t_end = time.time() + min(10, rem() - 3)
        h_final = best_h.copy()
        for _ in range(200):
            if time.time() >= t_end:
                break
            h_final = _coord_sweep(h_final, n)
        h_final = _repair(h_final, n)
        c5 = _compute_c5(h_final, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
