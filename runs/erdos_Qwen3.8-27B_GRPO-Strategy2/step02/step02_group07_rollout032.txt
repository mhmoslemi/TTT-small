    # By symmetry, only need k=0..n-1, which are indices n-1..2n-2
    c_pos = c[n-1:]  # c_0, c_1, ..., c_{n-1}

Alternative: since the function is convex, we can find the minimum by:
1. Evaluate at t=lo and t=hi
2. Find the t where the "derivative" (slope of active line) crosses 0
3. The derivative at any t is A_{k*(t)} where k*(t) = argmax_k c_k(t)

To find where the slope crosses 0:
- Start at t=lo, slope = A_{k*(lo)}
- If slope > 0, minimum is at lo
- If slope < 0, advance to next breakpoint, recompute slope
- Continue until slope ≥ 0 or t reaches hi

This is O(n) per pair in the worst case (70 breakpoints to check), much faster than golden section (40 iterations × 2 evals).

minimize t
subject to: c_k ≤ t for all k, 0≤h≤1, sum(h)=n/2

This is an SDP-like problem (or rather, a LP in t and bilinear in h). Not directly solvable as LP, but we can use:
- Benders decomposition: fix h, compute t = max c_k; then optimize h to reduce the worst c_k
- Or: use the dual of the max: t ≥ c_k for all k → introduce multipliers λ_k ≥ 0, Σλ = 1
  Lagrangian: L = Σ_k λ_k c_k - t + t = Σ_k λ_k c_k (at optimum t = max c_k)

  This is a Frank-Wolfe / mirror descent in the dual!

Inner problem: minimize Σ_k λ_k c_k(h) = Σ_k λ_k Σ_{i=0}^{n-1-k} h[i+k](1-h[i])
= Σ_i h[i](1 - λ_0 h[i] - Σ_{k≥1, i+k<n} λ_k h[i+k] - Σ_{k≥1, i-k≥0} λ_k h[i-k])

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _project(x, target):
    lo, hi = -10.0, 10.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _subgrad_opt(h, n, dx, n_iter, t_limit, lr0, mom, act_frac):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    v = np.zeros(n)
    lr = lr0
    stall = 0
    target = n / 2.0
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        c = _c5_pos(h, n)
        m = c.max()
        tol = act_frac * max(m, 1e-8)
        active = c >= m - tol
        n_act = np.sum(active)
        if n_act == 0:
            break
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            if k == 0:
                grad += 1.0 - 2.0 * h
            else:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
        grad *= dx / n_act
        v = mom * v - lr * grad
        h_new = _project(h + v, target)
        cn = _c5(h_new, dx)
        if cn < best_c5 - 1e-12:
            best_c5 = cn
            best_h = h_new.copy()
            lr = min(lr * 1.015, 0.06)
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                lr *= 0.82
                v *= 0.5
            h_new = h
        h = h_new
        if lr < 1e-9:
            break
    return best_h, best_c5

def _cd_pass(h, n, dx):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for i in range(n):
        for j in range(i + 1, n):
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            a, b = lo + 1e-10, hi - 1e-10
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            h[i], h[j] = c1, s - c1
            f1 = _c5(h, dx)
            h[i], h[j] = c2, s - c2
            f2 = _c5(h, dx)
            for _ in range(30):
                if f1 < f2:
                    b = c2; c2 = c1; f2 = f1
                    c1 = b - gr * (b - a)
                    h[i], h[j] = c1, s - c1
                    f1 = _c5(h, dx)
                else:
                    a = c1; c1 = c2; f1 = f2
                    c2 = a + gr * (b - a)
                    h[i], h[j] = c2, s - c2
                    f2 = _c5(h, dx)
            t = (a + b) * 0.5
            h[i], h[j] = t, s - t
    return h

def _cd_loop(h, n, dx, t_limit, max_passes=500):
    prev = _c5(h, dx)
    for _ in range(max_passes):
        if time.time() > t_limit:
            break
        h = _cd_pass(h, n, dx)
        cur = _c5(h, dx)
        if prev - cur < 1e-11:
            break
        prev = cur
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: Subgradient (global shaping)
    tl1 = t0 + budget_s * 0.18
    h, c = _subgrad_opt(h.copy(), n, dx, 20000, tl1, 0.02, 0.8, 0.10)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 2: Coordinate descent
    tl2 = t0 + budget_s * 0.45
    h, c = _cd_loop(best_h.copy(), n, dx, tl2, 300)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 3: Subgradient polish (tighter active set)
    tl3 = t0 + budget_s * 0.62
    h, c = _subgrad_opt(best_h.copy(), n, dx, 25000, tl3, 0.004, 0.9, 0.02)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 4: CD polish
    tl4 = t0 + budget_s * 0.80
    h, c = _cd_loop(best_h.copy(), n, dx, tl4, 400)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 5: Multi-restart with structured perturbations
    sigmas = [0.04, 0.08, 0.12, 0.06, 0.15, 0.03, 0.10, 0.05]
    for si in range(len(sigmas)):
        if time.time() > t0 + budget_s - 30:
            break
        hp = _repair(best_h + sigmas[si] * rng.randn(n), n)
        # Also try cyclic shifts
        if si % 3 == 1:
            hp = _repair(np.roll(best_h, rng.randint(1, n)), n)
        elif si % 3 == 2:
            hp = _repair(1.0 - best_h, n)
            hp = _repair(best_h + 0.05 * rng.randn(n), n)
        cur = time.time()
        tl_sg = min(cur + 35, t0 + budget_s - 20)
        h, c = _subgrad_opt(hp, n, dx, 8000, tl_sg, 0.01, 0.85, 0.05)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
        tl_cd = min(time.time() + 18, t0 + budget_s - 8)
        h, c = _cd_loop(h, n, dx, tl_cd, 150)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 6: Final CD
    if time.time() < t0 + budget_s - 8:
        h, c = _cd_loop(best_h.copy(), n, dx, t0 + budget_s - 3, 150)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
