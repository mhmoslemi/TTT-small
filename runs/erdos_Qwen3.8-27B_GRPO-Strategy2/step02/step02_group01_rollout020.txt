import time

def corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, dx):
    return np.max(corr_vals(h)) * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def smooth_max_val(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    val = smooth_max_val(c, alpha)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)

    # grad = sum_k w[k] * corr_grad_lag(h, k-(n-1))

import numpy as np
from scipy.optimize import minimize
import time


def corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_val(h, dx):
    return float(np.max(corr_vals(h)) * dx)


def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h


def corr_grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    lo_f = max(0, lag)
    hi_f = min(n, n + lag)
    g[lo_f:hi_f] = 1.0
    lo_a = max(0, lag)
    hi_a = min(n, n + lag)
    g[lo_a:hi_a] -= h[lo_a - lag:hi_a - lag]
    lo_b = max(0, -lag)
    hi_b = min(n, n - lag)
    g[lo_b:hi_b] -= h[lo_b + lag:hi_b + lag]
    return g


def smooth_val_grad(h, alpha):
    n = len(h)
    c = corr_vals(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    ev = np.exp(s)
    val = m + np.log(ev.sum()) / alpha
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * corr_grad_lag(h, lag)
    return val, grad


def _obj(x, alpha, mu, target):
    h = np.clip(x, 0.0, 1.0)
    v, g = smooth_val_grad(h, alpha)
    d = h.sum() - target
    return v + mu * d * d


def _jac(x, alpha, mu, target):
    h = np.clip(x, 0.0, 1.0)
    v, g = smooth_val_grad(h, alpha)
    d = h.sum() - target
    g += 2.0 * mu * d
    return g


def _lbfgs(h0, bounds, alpha, mu, target, maxiter):
    try:
        res = minimize(_obj, h0, args=(alpha, mu, target), jac=_jac,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return repair(res.x, len(h0))
    except Exception:
        return h0


def _pair_descent(h, dx, n, deltas, max_iters, t_start, deadline):
    c5_cur = c5_val(h, dx)
    hi_arr = np.array(deltas)
    count = 0
    while count < max_iters:
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = hi_arr[np.random.randint(len(hi_arr))]
        a, b = h[i], h[j]
        na, nb = a + d, b - d
        if 0.0 <= na <= 1.0 and 0.0 <= nb <= 1.0:
            h[i], h[j] = na, nb
            c = c5_val(h, dx)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = a, b
        count += 1
    return h, c5_cur

    # Phase 6: Final ultra-fine
    if _elapsed() < max_time - 3:
        h_w = best_h.copy()
        tiny = [0.0001, 0.0003, 0.0005, -0.0001, -0.0003, -0.0005]
        h_w, c5_w = _pair_descent(h_w, dx, n, tiny, 50000, t0, max_time - 3)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w

    best_h = repair(best_h, n)
    best_c5 = c5_val(best_h, dx)
    return (best_h, best_c5, n)
