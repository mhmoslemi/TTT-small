```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, alpha, maxiter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _pair_cd(h, n, dx, t_limit, t_start, max_passes=500):
    """Analytical pairwise coordinate descent."""
    best_c5 = _c5(h, dx)
    for p in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi = min(1.0, hsum)
                if hi - lo < 1e-10:
                    continue
                # Compute slopes for each shift
                c = _corr(h)
                si = h[i]
                sj = h[j]
                # slope_k = (1-2*h[i+s])*[valid] - (1-2*h[j+s])*[valid]
                n_sh = len(c)
                slopes = np.zeros(n_sh)
                for k in range(n_sh):
                    s = k - (n - 1)
                    sl = 0.0
                    ii = i + s
                    jj = j + s
                    if 0 <= ii < n:
                        sl += (1.0 - 2.0 * h[ii])
                    if 0 <= jj < n:
                        sl -= (1.0 - 2.0 * h[jj])
                    slopes[k] = sl
                # c_k(t) = c[k] + (t - si) * slopes[k]
                # Minimize max_k c_k(t) over t in [lo, hi]
                # This is convex in t. Use ternary search.
                a, b = lo, hi
                for _ in range(40):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = np.max(c + (m1 - si) * slopes)
                    f2 = np.max(c + (m2 - si) * slopes)
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                t_new = (a + b) / 2.0
                # Evaluate
                f_new = np.max(c + (t_new - si) * slopes)
                if f_new < best_c5 - 1e-10:
                    h[i] = t_new
                    h[j] = hsum - t_new
                    best_c5 = f_new
                    improved = True
        if not improved:
            break
    return h, _c5(h, dx)
```
