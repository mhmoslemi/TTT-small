import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _exact_cd_one(h, p, c_full):
    """Solve 1D minimax for coordinate p given current correlation values."""
    n = len(h)
    m = 2 * n - 1  # number of lags
    # c_full[k] is correlation at shift s = k - (n-1)
    # For coordinate p, compute B_s for each shift s
    # B_s = (1-h[p-s]) * [0<=p-s<n] + (-h[p+s]) * [0<=p+s<n]
    
    B = np.zeros(m)
    A = np.zeros(m)
    for k in range(m):
        s = k - (n - 1)
        b = 0.0
        if 0 <= p - s < n:
            b += 1.0 - h[p - s]
        if 0 <= p + s < n:
            b += -h[p + s]
        B[k] = b
        A[k] = c_full[k] - b * h[p]
    
    # Now solve min_{x in [0,1]} max_k (A[k] + B[k]*x)
    # Evaluate at x=0 and x=1
    f0 = float(np.max(A))
    f1 = float(np.max(A + B))
    best_x = 0.0 if f0 <= f1 else 1.0
    best_f = min(f0, f1)
    
    # Check all pairwise intersections
    # For speed, only check pairs where B differs
    for i in range(m):
        for j in range(i+1, m):
            db = B[i] - B[j]
            if abs(db) < 1e-15:
                continue
            xstar = (A[j] - A[i]) / db
            if xstar < 0.0 or xstar > 1.0:
                continue
            # Check if this is a valid vertex (both lines i,j are on the upper envelope at xstar)
            val_i = A[i] + B[i] * xstar
            val = float(np.max(A + B * xstar))
            if val < best_f - 1e-15:
                best_f = val
                best_x = xstar
    
    return best_x, best_f

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _mm1d(A, B):
    lo, hi = 0.0, 1.0
    for _ in range(55):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(A + B * m1))
        f2 = float(np.max(A + B * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    x = 0.5 * (lo + hi)
    return x, float(np.max(A + B * x))

def _exact_cd(h0, target, max_sweeps=500):
    n = len(h0)
    m = 2 * n - 1
    dx = 2.0 / n
    h = _proj(h0.copy(), target)
    s_arr = np.arange(m) - (n - 1)
    bc = _c5(h, dx)
    for sweep in range(max_sweeps):
        c_full = np.correlate(h, 1.0 - h, mode='full')
        max_chg = 0.0
        for p in range(n):
            B = np.zeros(m)
            v1 = (p - s_arr >= 0) & (p - s_arr < n)
            v2 = (p + s_arr >= 0) & (p + s_arr < n)
            B[v1] += 1.0 - h[p - s_arr[v1]]
            B[v2] += -h[p + s_arr[v2]]
            A = c_full - B * h[p]
            xv, _ = _mm1d(A, B)
            d = xv - h[p]
            if abs(d) > 1e-14:
                h[p] = xv
                c_full += d * B
                max_chg = max(max_chg, abs(d))
        cv = _c5(h, dx)
        if cv < bc:
            bc = cv
        if max_chg < 1e-12:
            break
    h = _proj(h, target)
    return h, _c5(h, dx)

def _fista_sm(h0, alpha, target, iters=1500):
    n = len(h0)
    dx = 2.0 / n
    x = _proj(h0.copy(), target)
    z = x.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (8.0 * alpha * n + 4.0)
    for it in range(iters):
        c = np.correlate(z, 1.0 - z, mode='full')
        mx = float(c.max())
        s = np.clip(alpha * (c - mx), -500.0, 500.0)
        e = np.exp(s)
        w = e / float(e.sum())
        g = np.zeros(n)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                for i in range(n - lag):
                    g[i] += wk * (1.0 - z[i + lag])
                    g[i + lag] -= wk * z[i]
            else:

                    g[i] += wk * (1.0 - z[i - L])
                    g[i - L] -= wk * z[i]
        y = z - step * g

        x = xn
        t = tn
        cnow = _c5(x, dx)
        if cnow < bc:
            bc = cnow
            bx = x.copy()
