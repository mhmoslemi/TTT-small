    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

    return m + np.log(np.sum(np.exp(s))) / alpha

def _penalty_obj(x, alpha, pen):
    corr = _corr(x)
    val = _smooth_max(corr, alpha)
    target = len(x) / 2.0
    val += pen * (x.sum() - target) ** 2
    return val

def _project(h, target):

            if total < 1e-15:

    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += (diff / tr) * room
        else:
            if s < 1e-15:
                break
            h += (diff / s) * h
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in range(len(h)):
                a = min(diff, 1.0 - h[i])
                h[i] += a + (0 if a < 1e-15 else 0)
                diff -= a
                if diff < 1e-14:
                    break
        else:
            for i in range(len(h)):
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
                if -diff < 1e-14:
                    break
    return h

def _structured_local(h, n, dx, budget, n_passes=40, coarse_first=True):

    else:
        delta_groups = [[0.005, 0.01, 0.02], [0.03, 0.05, 0.08], [0.1, 0.12, 0.15]]
    pass_budget = budget / max(n_passes, 1)
    for gi, dgroup in enumerate(delta_groups):
        for d in dgroup:
            if time.time() > budget:
                return h, best
            improved = False
            for i in range(n):

                        delta = sign * d

                        nj = h[j] - delta

                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = ni - delta, nj + delta
            if not improved:
                pass
    return h, best

def _rand_local(h, n, dx, budget, rng, n_iter=5000):
    target = n / 2.0
    best = _c5(h, dx)
    deltas = [0.01, 0.02, 0.04, 0.08, 0.15, 0.25]
    for _ in range(n_iter):
        if time.time() > budget:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        s = 1 if rng.rand() < 0.5 else -1
        delta = s * d
        ni, nj = h[i] + delta, h[j] - delta
        if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
            continue
        h[i], h[j] = ni, nj
        c = _c5(h, dx)
        if c < best:
            best = c
        else:
            h[i], h[j] = ni - delta, nj + delta
    return h, best

def _lbfgs_opt(h, n, alpha, pen, iters, budget):
    target = n / 2.0

    def obj(x):
        return _smooth_max(_corr(x), alpha) + pen * (x.sum() - target) ** 2

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12})
        out = _project(res.x, target)
        return out
    except Exception:
        return h

def _slsqp_opt(h, n, alpha, iters, budget):
    target = n / 2.0

    def obj(x):
        return _smooth_max(_corr(x), alpha)

    def con(x):
        return x.sum() - target

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': iters, 'ftol': 1e-16})
        return _project(res.x, target)
    except Exception:
        return h
