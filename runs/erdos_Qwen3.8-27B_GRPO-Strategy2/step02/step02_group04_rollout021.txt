def _fix(h, n):
    """Ensure 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    t = n/2.0
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0-h)
            for i in o:
                a = min(d, 1.0-h[i]); h[i]+=a; d-=a
                if abs(d)<1e-14: break

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    d = t - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            o = np.argsort(1.0 - h)
            for i in o:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                if abs(d) < 1e-14: break
        else:
            o = np.argsort(h)
            for i in o:
                a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-14: break
    return h

def _sgrad(h, n, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    w = np.exp(a * (c - mx)); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-11: continue
        s = k - (n - 1)
        if s >= 0:
            g[:n-s] += w[k] * (1.0 - h[s:])
            g[s:] -= w[k] * h[:n-s]
        else:
            t = -s
            g[t:] += w[k] * (1.0 - h[:n-t])
            g[:n-t] -= w[k] * h[t:]
    return g

def _gd(h, n, dx, a, step, iters, deadline, rng):
    h = h.copy()
    bc5 = _c5(h, dx)
    for _ in range(iters):
        if time.time() >= deadline: break
        g = _sgrad(h, n, a)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-13: break
        d = -g / gn
        best = bc5; best_h = h
        s = step
        for _ in range(30):
            nh = _fix(h + s * d, n)
            nc = _c5(nh, dx)
            if nc < best - 1e-14:
                best = nc; best_h = nh
            s *= 0.55
            if s < 1e-7: break
        if best < bc5 - 1e-14:
            h = best_h; bc5 = best
        else:
            if rng.random() < 0.4:
                nh = _fix(h + 0.025 * rng.randn(n), n)
                if _c5(nh, dx) < bc5:
                    h = nh; bc5 = _c5(h, dx)
            else:
                break
    return h, bc5

def _ls(h, n, dx, deadline, rng, n_iters=200000):
    h = h.copy()
    bc5 = _c5(h, dx)
    deltas = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    stalls = 0
    for _ in range(n_iters):
        if time.time() >= deadline: break
        g = _sgrad(h, n, 120)
        pw_i = np.exp(g - g.max()); pw_i /= pw_i.sum()
        pw_j = np.exp(-g + g.min()); pw_j /= pw_j.sum()
        i = rng.choice(n, p=pw_i)
        j = rng.choice(n, p=pw_j)
        if i == j: continue
        found = False
        for d in deltas:
            for sgn in (1, -1):
                dd = sgn * d
                ni, nj = h[i] + dd, h[j] - dd
                if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12: continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                h[i], h[j] = h[i] - dd, h[j] + dd
                if c < bc5 - 1e-14:
                    h[i], h[j] = ni, nj
                    bc5 = c; found = True; break
            if found: break
        if found:
            stalls = 0
        else:
            stalls += 1
            if stalls > 300:
                h = _fix(h + 0.012 * rng.randn(n), n)
                bc5 = _c5(h, dx)
                stalls = 0
                if stalls > 5000: break
    return h, bc5

def _coord_sweep(h, n, dx, deadline, rng):
    h = h.copy()
    bc5 = _c5(h, dx)
    deltas = [0.015, 0.008, 0.004, 0.002]
    no_improve = 0
    for _pass in range(5000):
        if time.time() >= deadline: break
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for i in order:
            for j in order:
                if i == j: continue
                for d in deltas:
                    for sgn in (1, -1):
                        dd = sgn * d
                        ni, nj = h[i] + dd, h[j] - dd
                        if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12: continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        h[i], h[j] = h[i] - dd, h[j] + dd
                        if c < bc5 - 1e-14:
                            h[i], h[j] = ni, nj
                            bc5 = c; improved = True; break
                    if improved: break
                if improved: break
            if time.time() >= deadline: break
        if not improved:
            no_improve += 1
            if no_improve > 15: break
        else:
            no_improve = 0
    return h, bc5

def _slsqp(h, n, dx, a, iters, deadline):
    if time.time() >= deadline: return h, _c5(h, dx)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def smooth_obj(x, alpha=a):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    def smooth_grad(x, alpha=a):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        w = np.exp(np.clip(alpha * (c - m), -500, 500)); w /= w.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-11: continue
            s = k - (n-1)
            if s >= 0:
                g[:n-s] += w[k]*(1-x[s:]); g[s:] -= w[k]*x[:n-s]
            else:
                t = -s
                g[t:] += w[k]*(1-x[:n-t]); g[:n-t] -= w[k]*x[t:]
        return g * dx
    try:
        r = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                     bounds=bounds,
                     constraints=[{'type':'eq','fun':lambda x: x.sum()-target}],
                     options={'maxiter':iters,'ftol':1e-16})
        hc = _fix(r.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)
