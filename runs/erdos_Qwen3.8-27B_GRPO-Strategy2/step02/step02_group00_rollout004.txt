    b = np.zeros(len(h))
    b[order[:int(target)]] = 1.0
    return b


def _binary_sa(h_init, n, dx, target, t_end, rng, T0=0.008, Tend=0.0002, restarts=3):
    best = _c5(h_init, dx)
    best_h = h_init.copy()
    h_sa = h_init.copy()
    total_iters = 400000
    iters_per = total_iters // max(restarts, 1)
    
    for r in range(restarts):
        h_cur = best_h.copy() if r > 0 else h_sa.copy()
        if r > 0 and r < restarts:
            # Perturb: flip random pairs
            for _ in range(5):
                oi = rng.randint(n)
                if h_cur[oi] > 0.5:
                    zi = rng.choice(np.where(h_cur < 0.5)[0])
                    h_cur[oi] = 0.0
                    h_cur[zi] = 1.0
            cur = _c5(h_cur, dx)
            if cur > best + 0.02:
                h_cur = best_h.copy()
        
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        sa_best = _c5(h_cur, dx)
        sa_best_h = h_cur.copy()
        
        for it in range(iters_per):
            if time.time() > t_end:
                return best_h, best
            if it % 5000 == 0:
                ones = np.where(h_cur > 0.5)[0]
                zeros = np.where(h_cur < 0.5)[0]
            frac = (r * iters_per + it) / total_iters
            T = T0 * (Tend / T0) ** frac
            
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            new_c5 = _c5(h_cur, dx)
            
            if new_c5 < sa_best or rng.rand() < np.exp(-(new_c5 - sa_best) / max(T, 1e-12)):
                sa_best = min(sa_best, new_c5)
                if new_c5 < sa_best_h is not None:
                    pass
                if new_c5 < _c5(sa_best_h, dx) - 1e-15:
                    sa_best_h = h_cur.copy()
                oi = np.where(ones == i)[0][0]
                zi = np.where(zeros == j)[0][0]
                ones[oi] = j
                zeros[zi] = i
            else:
                h_cur[i] = 1.0
                h_cur[j] = 0.0
        
        if sa_best < best:
            best = sa_best
            best_h = sa_best_h.copy()
    
    return best_h, best


def _local_search(h, n, dx, target, t_end, rng):
    best = _c5(h, dx)
    h = h.copy()
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05])
    no_imp = 0
    
    while time.time() < t_end and no_imp < 8:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, best
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                need = -(ni - vi)
                if abs(need) < 1e-15:
                    continue
                if need > 0:
                    cand = np.argsort(h)[:6]
                else:
                    cand = np.argsort(-h)[:6]
                found = False
                for j in cand:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < best - 1e-14:
                        best = c5_try
                        improved = True
                        found = True
                        break
                    else:
                        h[i] = vi
                        h[j] -= need
                if found:
                    break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    
    # Prepare continuous starting point
    h_cont = _repair_sum(h_init.copy(), target)
    best_c5 = _c5(h_cont, dx)
    best_h = h_cont.copy()
    
    # Prepare binary starting point
    h_bin = _to_binary(h_cont, target)
    best_bin_c5 = _c5(h_bin, dx)
    if best_bin_c5 < best_c5:
        best_c5 = best_bin_c5
        best_h = h_bin.copy()
    
    # Phase 1: Binary Simulated Annealing (fast, many iterations)
    p1_end = t_start + min(budget_s * 0.35, 350)
    h_sa, sa_result = _binary_sa(h_bin, n, dx, target, p1_end, rng,
                                  T0=0.006, Tend=0.0001, restarts=4)
    if sa_result < best_c5:
        best_c5 = sa_result
        best_h = h_sa.copy()
    
    # Phase 2: More targeted SA with focus on reducing peaks
    p2_end = t_start + min(budget_s * 0.55, 550)
    if time.time() < p2_end:
        h_sa2 = best_h.copy()
        if not np.allclose(np.round(h_sa2), h_sa2):
            h_sa2 = _to_binary(h_sa2, target)
        best2 = _c5(h_sa2, dx)
        best2_h = h_sa2.copy()
        
        ones = np.where(h_sa2 > 0.5)[0]
        zeros = np.where(h_sa2 < 0.5)[0]
        T = 0.003
        
        for it in range(600000):
            if time.time() > p2_end:
                break
            if it % 10000 == 0:
                ones = np.where(h_sa2 > 0.5)[0]
                zeros = np.where(h_sa2 < 0.5)[0]
                T = 0.003 * (0.0001 / 0.003) ** (it / 600000.0)
            
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h_sa2[i] = 0.0
            h_sa2[j] = 1.0
            new_c5 = _c5(h_sa2, dx)
            
            if new_c5 < best2 or rng.rand() < np.exp(-(new_c5 - best2) / max(T, 1e-12)):
                if new_c5 < best2 - 1e-15:
                    best2 = new_c5
                    best2_h = h_sa2.copy()
                oi = np.where(ones == i)[0]
                zi = np.where(zeros == j)[0]
                if len(oi) > 0 and len(zi) > 0:
                    ones[oi[0]] = j
                    zeros[zi[0]] = i
            else:
                h_sa2[i] = 1.0
                h_sa2[j] = 0.0
        
        if best2 < best_c5:
            best_c5 = best2
            best_h = best2_h.copy()
    
    # Phase 3: Continuous SLSQP refinement
    p3_end = t_start + min(budget_s * 0.82, 880)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    
    h_opt = best_h.copy()
    if np.allclose(np.round(h_opt), h_opt):
        # Soften: add small noise to allow continuous movement
        h_opt = h_opt * (1 - 0.1) + 0.05
        h_opt = _repair_sum(h_opt, target)
    
    for k in [2, 3, 5]:
        if time.time() > p3_end:
            break
        try:
            res = minimize(_obj_topk, h_opt, args=(k, dx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-16})
            hc = _repair_sum(np.clip(res.x, 0, 1), target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
                h_opt = hc.copy()
        except Exception:
            pass
    
    for alpha in [800, 3000, 8000]:
        if time.time() > p3_end:
            break
        try:
            res = minimize(_obj_lse, h_opt, args=(alpha, dx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-16})
            hc = _repair_sum(np.clip(res.x, 0, 1), target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
                h_opt = hc.copy()
        except Exception:
            pass
    
    # Phase 4: Fine local search
    p4_end = t_start + budget_s - 10
    if time.time() < p4_end:
        h_fine, fine_result = _local_search(best_h, n, dx, target, p4_end, rng)
        if fine_result < best_c5:
            best_c5, best_h = fine_result, h_fine
    
    # Final: one more SLSQP push
    if time.time() < t_start + budget_s - 3:
        try:
            res = minimize(_obj_lse, best_h, args=(10000, dx),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-17})
            hc = _repair_sum(np.clip(res.x, 0, 1), target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    
    best_h = _repair_sum(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
