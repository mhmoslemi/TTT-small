def _project_to_box_sum(h, n):
    """Project h onto {x : 0≤x≤1, sum x = n/2}."""
    target = n / 2.0
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then adjust sum
    s = h.sum()
    if s == target:
        return h
    # Need to add diff = target - s
    diff = target - s
    if diff > 0:
        # Add to entries with most room (h < 1)
        # Water filling: find threshold tau such that sum(min(1, h+tau)) = target
        # Sort: entries at 1 contribute 0, others contribute min(1-h_i, tau)
        order = np.argsort(h)  # ascending, so most room first
        # Binary search for tau
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2
            new_sum = np.sum(np.minimum(1.0, h + mid))
            if new_sum < target:
                lo = mid
            else:
                hi = mid
        tau = (lo + hi) / 2
        h = np.minimum(1.0, h + tau)
    else:
        # Subtract from entries with most value (h > 0)
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2
            new_sum = np.sum(np.maximum(0.0, h - mid))
            if new_sum > target:
                lo = mid
            else:
                hi = mid
        tau = (lo + hi) / 2
        h = np.maximum(0.0, h - tau)
    # Final adjustment for numerical precision
    r = target - h.sum()
    if abs(r) > 1e-15:
        if r > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += r
    return h

∂c[k]/∂h[j] = (1-h[j+s]) * I(lo≤j<hi) + (-h[j-s]) * I(lo≤j-s<hi)

        lo = max(0, -s); hi = min(n, n-s)
        if hi <= lo: continue
        j = np.arange(lo, hi)
        grad[j] += wk * (1.0 - h[j+s])
        j2lo, j2hi = max(lo+s, 0), min(hi+s, n)
        if j2hi > j2lo:
            j2 = np.arange(j2lo, j2hi)
            grad[j2] -= wk * h[j2-s]
    return grad

def _lbfgs_run(h, n, alpha, maxiter, t_lim, t0):
    target = n/2.0
    bnds = [(0.0, 1.0)]*n
    def f(x):
        hx = _project(x, target)
        c = _corr(hx)
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        return m + np.log(np.sum(np.exp(s)))/alpha
    def g(x):
        hx = _project(x, target)
        c = _corr(hx)
        m = c.max()
        s = np.clip(alpha*(c-m), -500, 500)
        e = np.exp(s)
        return _grad_w(hx, n, e/e.sum())
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-13, 'ftol': 1e-15})
        return _project(res.x, target)
    except:
        return h.copy()

def _pair_cd(h, n, dx, t_lim, t0, n_grid=11, max_passes=5000):
    hi_c5 = _c5(h, dx)
    for _ in range(max_passes):
        if time.time() - t0 > t_lim: break
        imp = False
        for i in range(n):
            for j in range(i+1, n):
                hs = h[i] + h[j]
                lo, hi = max(0.0, hs-1.0), min(1.0, hs)
                if hi - lo < 1e-10: continue
                oi, oj = h[i], h[j]
                best_t, best_c = oi, hi_c5
                for t in np.linspace(lo+1e-9, hi-1e-9, n_grid):
                    h[i], h[j] = t, hs-t
                    c = _c5(h, dx)
                    if c < best_c: best_c, best_t = c, t
                    h[i], h[j] = oi, oj
                if best_c < hi_c5 - 1e-13:
                    h[i], h[j] = best_t, hs-best_t
                    hi_c5 = best_c; imp = True
        if not imp: break
    return h, hi_c5

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_w(h, n, w):

def _binary_h(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)

            h[i] -= swap
            h[j] += swap
    return _repair(h, n)
