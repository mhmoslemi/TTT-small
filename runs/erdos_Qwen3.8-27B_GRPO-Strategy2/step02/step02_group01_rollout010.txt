        return np.full(n, 0.5)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

    return m + np.log(Z) / _ALPHA + pen
