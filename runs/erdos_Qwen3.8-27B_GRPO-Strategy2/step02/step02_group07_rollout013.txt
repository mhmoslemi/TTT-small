        lo, hi = 0.0, 1.0
        for _ in range(50):

        if free.sum() > 0:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _project(h, n):
    """Project onto {x: 0<=x<=1, sum=n/2} via bisection"""
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        free = (h_new > 1e-8) & (h_new < 1 - 1e-8)
        if free.sum() > 0:
            h_new[free] = np.clip(h_new[free] + diff / free.sum(), 0, 1)
    return h_new

def _soft_weighted_grad(h, n, dx, ws):
    """Weighted gradient: sum_k ws[k] * grad(c_k)/sum h"""
    # ws has length 2n-1, corresponding to lags -(n-1) to (n-1)
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = ws[p]
        if w < 1e-12:
            continue
        l = lags[p]
        # Term 1: (1-h[j+l]) for valid j where 0<=j+l<n
        # Term 2: -h[j-l] for valid j where 0<=j-l<n
        if l >= 0:
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += w * (-h[:n - l])
        else:
            l2 = -l
            grad[l2:] += w * (1.0 - h[:n - l2])
            grad[:n - l2] += w * (-h[l2:])
    return dx * grad

def _proj_grad(h, n, dx, t0, t_end):
    """Projected gradient with momentum"""
    h = h.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    momentum = np.zeros(n)
    betas = 0.9
    lr = 1.0
    alpha = 50
    
    for it in range(300000):
        if time.time() > t_end:
            break
        c = _corr(h) * dx
        # Increase alpha over time
        alpha = min(50 + 2 * it, 2000)
        cs = c - c.max()
        ws = np.exp(alpha * cs)
        ws /= ws.sum()
        g = _soft_weighted_grad(h, n, dx, ws)
        momentum = betas * momentum + g
        h_new = _project(h - lr * momentum, n)
        c_new = _c5(h_new, dx)
        if c_new < c:
            h = h_new
            lr = min(lr * 1.02, 10.0)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 0.01:
                lr = 0.5
                momentum *= 0.1
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _cd_fast(h, n, dx, t0, t_end, n_grid=13):
    """Fast coordinate descent - try all pairs, 1D search on transfer"""
    h = h.copy()
    best_c = _c5(h, dx)
    no_improve = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi = min(1.0, hsum)
                if hi - lo < 1e-10:
                    continue
                orig_i, orig_j = h[i], h[j]
                best_t, best_c_local = orig_i, best_c
                ts = np.linspace(lo + 1e-9, hi - 1e-9, n_grid)
                for t in ts:
                    h[i] = t; h[j] = hsum - t
                    c = _c5(h, dx)
                    if c < best_c_local:
                        best_c_local = c; best_t = t
                if best_c_local < best_c - 1e-12:
                    h[i] = best_t; h[j] = hsum - best_t
                    best_c = best_c_local
                    improved = True
                else:
                    h[i] = orig_i; h[j] = orig_j
        if not improved:
            break
    return h, _c5(h, dx)

def _slsqp_brief(h, n, dx):
    target = n / 2.0
    def obj(x):
        c = _corr(x) * dx
        m = c.max()
        s = np.clip(300 * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / 300
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                   options={'maxiter': 300, 'ftol': 1e-13})
    return _repair(res.x, n)

Δc[k] = δ * [(1-h[i+k])*I(valid_ik) - h[i-k]*I(valid) - (1-h[j+k])*I(valid) + h[j-k]*I(valid)]

But this still requires checking all 2n-1 k values. For n=70, that's 139 values. So Δc computation is O(n).

Total CD per pair: O(n) for delta computation + O(n) for max check = O(n).
Total CD per pass: O(n² * n) = O(n³) = 70³ = 343K. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _project(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        free = (h_new > 1e-9) & (h_new < 1 - 1e-9)
        if free.sum() > 0:
            h_new[free] = np.clip(h_new[free] + diff / free.sum(), 0, 1)
    return h_new

def _grad_for_lag(h, n, l):
    g = np.zeros(n)
    if l >= 0:
        g[:n - l] += 1.0 - h[l:]
        g[l:] -= h[:n - l]
    else:
        l2 = -l
        g[l2:] += 1.0 - h[:n - l2]
        g[:n - l2] -= h[l2:]
    return g

def _proj_grad(h, n, dx, t0, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c = _c5(h, dx)
    momentum = np.zeros(n)
    betas = 0.88
    lr = 3.0
    lags_arr = np.arange(2 * n - 1) - (n - 1)
    it = 0
    while time.time() < t_end:
        c_arr = _corr(h) * dx
        m = c_arr.max()
        alpha = min(15.0 + 0.004 * it, 3000.0)
        cs = c_arr - m
        ws = np.exp(alpha * cs)
        ws /= ws.sum()
        thresh = 0.005 * ws.max()
        active = np.where(ws > thresh)[0]
        g = np.zeros(n)
        for p in active:
            g += ws[p] * _grad_for_lag(h, n, lags_arr[p])
        g *= dx
        momentum = betas * momentum + g
        h_new = _project(h - lr * momentum, n)
        c_new = _c5(h_new, dx)
        if c_new < m:
            h = h_new
            lr = min(lr * 1.005, 30.0)
        else:
            lr *= 0.6
            if lr < 0.3:
                lr = 2.0; momentum *= 0.2
        it += 1
        cv = _c5(h, dx)
        if cv < best_c:
            best_c = cv; best_h = h.copy()
    return best_h, best_c

def _cd_refine(h, n, dx, t0, t_end, n_grid=11):
    h = h.copy()
    best_c = _c5(h, dx)
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hs = h[i] + h[j]
                lo = max(0.0, hs - 1.0); hi = min(1.0, hs)
                if hi - lo < 1e-10:
                    continue
                oi, oj = h[i], h[j]
                bt, bv = oi, best_c
                for t in np.linspace(lo + 1e-8, hi - 1e-8, n_grid):
                    h[i] = t; h[j] = hs - t
                    cv = _c5(h, dx)
                    if cv < bv:
                        bv = cv; bt = t
                if bv < best_c - 1e-13:
                    h[i] = bt; h[j] = hs - bt; best_c = bv; improved = True
                else:
                    h[i] = oi; h[j] = oj
        if not improved:
            break
    return h, _c5(h, dx)

def _slsqp_p(h, n, dx):
    target = n / 2.0
    def obj(x):
        c = _corr(x) * dx
        m = c.max()
        return m + np.sum(np.maximum(0, c - m - 0.001)) + 0.1 * np.sum((c - m) ** 2)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                   options={'maxiter': 400, 'ftol': 1e-14})
    return _repair(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: Projected gradient (main, ~40% budget)
    e = time.time() - t0
    h1, c1 = _proj_grad(h, n, dx, t0, t0 + budget_s * 0.38)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: CD refinement (~15% budget)
    e = time.time() - t0
    if e < budget_s * 0.55:
        h2, c2 = _cd_refine(best_h, n, dx, t0, t0 + budget_s * 0.55, n_grid=11)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Second PG pass from improved point
    e = time.time() - t0
    if e < budget_s * 0.7:
        h3, c3 = _proj_grad(best_h, n, dx, t0, t0 + budget_s * 0.72)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Structured restarts
    e = time.time() - t0
    remain = budget_s * 0.85 - e
    if remain > 40:
        sigmas = [0.08, 0.15, 0.25, 0.05, 0.12, 0.2]
        per = max(10, remain / len(sigmas))
        for si, s in enumerate(sigmas):
            e = time.time() - t0
            if e > budget_s * 0.85 - 10:
                break
            # Mix Gaussian + sinusoidal perturbation
            freq = 1.0 + si * 0.7
            phase = rng.uniform(0, 6.28)
            xs = np.linspace(0, 2, n)
            pert = s * rng.randn(n) + 0.3 * s * np.sin(2 * np.pi * freq * xs + phase)
            h_p = _repair(best_h + pert, n)
            hp, cp = _proj_grad(h_p, n, dx, t0, time.time() + per)
            if cp < best_c5:
                best_c5, best_h = cp, hp.copy()
                h_cd, c_cd = _cd_refine(best_h, n, dx, t0, time.time() + 15, n_grid=13)
                if c_cd < best_c5:
                    best_c5, best_h = c_cd, h_cd.copy()

    # Phase 5: Final intensive CD
    e = time.time() - t0
    if e < budget_s * 0.94:
        h5, c5 = _cd_refine(best_h, n, dx, t0, t0 + budget_s * 0.94, n_grid=15)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: SLSQP polish
    e = time.time() - t0
    if e < budget_s - 15:
        h6 = _slsqp_p(best_h, n, dx)
        c6 = _c5(h6, dx)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Phase 7: Tiny final CD
    e = time.time() - t0
    if e < budget_s - 5:
        h7, c7 = _cd_refine(best_h, n, dx, t0, t0 + budget_s - 3, n_grid=13)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
