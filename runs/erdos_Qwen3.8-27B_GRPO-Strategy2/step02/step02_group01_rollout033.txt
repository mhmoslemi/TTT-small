    """Compute G[m, j] = d(corr[m])/d(h[j]) for all m, j."""
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            G[m, k:] += (1.0 - h[:n-k])
            G[m, :n-k] -= h[k:]
        else:
            G[m, :n+k] += (1.0 - h[-k:])
            G[m, -k:] -= h[:n+k]
    return G

                    options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 5: Nelder-Mead polish
    if tleft() > 25:
        h = best_h.copy()
        try:
            obj_nm = _smooth_obj(h, 50000)
            res = minimize(
                lambda x: _smooth_obj(x, 50000) + 500.0 * (_sum_con(x, target)) ** 2,
                h, method='Nelder-Mead',
                options={'maxiter': 5000, 'xatol': 1e-9, 'fatol': 1e-15, 'adaptive': True})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 6: Ultra-fine pair search
    if tleft() > 10:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _ in range(100000):
            if tleft() < 10:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            for d in (0.0003, 0.001, 0.003, -0.0003, -0.001, -0.003):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c5_cur:
                        c5_cur = c
                        best_c5 = c5_cur
                        best_h = h.copy()
                        break
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
