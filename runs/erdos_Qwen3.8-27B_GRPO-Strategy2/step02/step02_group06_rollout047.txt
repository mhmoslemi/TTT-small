        if _c5(best_h, dx) >= best_c5 - 1e-14:
            break

    # Phase 2: SLSQP with smooth max (increasing alpha)
    for alpha in [80, 150, 300, 600, 1200, 2500]:
        if time.time() - t0 > budget_s - 250:
            break
        hc = _slsqp(best_h, n, alpha, None, target, bounds, 600)
        best_c5, best_h = _try_solution(hc, n, dx, best_c5, best_h)

    # Phase 3: p-norm SLSQP
    for p in [20, 40, 80, 160]:
        if time.time() - t0 > budget_s - 200:
            break
        hc = _slsqp(best_h, n, None, p, target, bounds, 500)
        best_c5, best_h = _try_solution(hc, n, dx, best_c5, best_h)

    # Phase 4: CD with random ordering
    for rnd in range(20):
        if time.time() - t0 > budget_s - 150:
            break
        hc = _coord_descent(best_h, n, n_sweeps=25, n_grid=401, rng=rng)
        best_c5, best_h = _try_solution(hc, n, dx, best_c5, best_h)
        if _c5(best_h, dx) >= best_c5 - 1e-14:
            break

    # Phase 5: SLSQP + CD interleaved
    for alpha in [200, 500, 1000]:
        if time.time() - t0 > budget_s - 100:
            break
        hc = _slsqp(best_h, n, alpha, None, target, bounds, 800)
        hc2 = _coord_descent(hc, n, n_sweeps=20, n_grid=401)
        best_c5, best_h = _try_solution(hc2, n, dx, best_c5, best_h)

    # Phase 6: Perturbation + CD (escape local minima)
    amps = [0.12, 0.08, 0.05, 0.03, 0.02, 0.015, 0.01]
    for amp in amps:
        for sr in range(3):
            if time.time() - t0 > budget_s - 50:
                break
            h_pert = best_h + amp * rng.randn(n)
            h_pert = _repair(h_pert, n)
            hc = _coord_descent(h_pert, n, n_sweeps=30, n_grid=401, rng=rng)
            hc = _slsqp(hc, n, 300, None, target, bounds, 300)
            hc2 = _coord_descent(hc, n, n_sweeps=20, n_grid=401)
            best_c5, best_h = _try_solution(hc2, n, dx, best_c5, best_h)

    # Phase 7: Shift-based perturbations
    for shift in range(1, min(5, n // 2)):
        if time.time() - t0 > budget_s - 30:
            break
        h_shift = np.roll(best_h, shift)
        h_shift = _repair(h_shift, n)
        hc = _coord_descent(h_shift, n, n_sweeps=25, n_grid=401)
        best_c5, best_h = _try_solution(hc, n, dx, best_c5, best_h)

    # Phase 8: Final ultra-fine CD + SLSQP
    if time.time() - t0 < budget_s - 20:
        hc = _coord_descent(best_h, n, n_sweeps=50, n_grid=801)
        best_c5, best_h = _try_solution(hc, n, dx, best_c5, best_h)

    if time.time() - t0 < budget_s - 10:
        hc = _slsqp(best_h, n, 5000, None, target, bounds, 2000)
        hc2 = _coord_descent(hc, n, n_sweeps=40, n_grid=601)
        best_c5, best_h = _try_solution(hc2, n, dx, best_c5, best_h)

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
