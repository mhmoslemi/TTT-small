import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            amt = min(abs(diff), max(0.0, (1.0 - x[k]) if diff > 0 else x[k]))
            if diff > 0:
                x[k] += amt
            else:
                x[k] -= amt
            diff -= np.copysign(amt, diff)
            if abs(diff) < 1e-14:
                break
    return x

def _sobj(x, a, K):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    idx = np.argsort(c)[-K:]
    s = np.clip(a * (c[idx] - m), -500.0, 0.0)
    e = np.exp(s)
    return m + np.log(e.sum()) / a

def _sgrad(x, a, K):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    idx = np.argsort(c)[-K:]
    s = np.clip(a * (c[idx] - m), -500.0, 0.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for ii in range(K):
        k = idx[ii]
        wk = w[ii]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _fista(x0, a, K, target, iters, t0, budget):
    n = len(x0)
    dx = 2.0 / n
    x, z, t = x0.copy(), x0.copy(), 1.0
    bx, bc = x.copy(), _c5(x, dx)
    step = 1.0 / (4.0 * a * K + 8.0)
    for it in range(iters):
        if time.time() - t0 > budget - 3:
            break
        gz = _sgrad(z, a, K)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 10:
            fn = _sobj(xn, a, K)
            qd = _sobj(z, a, K) + float(gz @ (xn - z)) + 0.5 / step * float((xn - z) @ (xn - z))
            if fn <= qd + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        if (it + 1) % 400 == 0:
            c = _c5(x, dx)
            if c < bc:
                bc = c
                bx = x.copy()
    c = _c5(x, dx)
    if c < bc:
        bc, bx = c, x.copy()
    return bx, bc

def _cd_line(h, dx, n, iters, rng, t0, budget):
    bc = _c5(h, dx)
    h = h.copy()
    for _ in range(iters):
        if time.time() - t0 > budget - 3:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-10:
            continue
        a, b = d_lo, d_hi
        for _ in range(22):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            h[i] += m1; h[j] -= m1
            c1 = _c5(h, dx)
            h[i] -= m1; h[j] += m1
            h[i] += m2; h[j] -= m2
            c2 = _c5(h, dx)
            h[i] -= m2; h[j] += m2
            if c1 < c2:
                b = m2
            else:
                a = m1
        d_opt = (a + b) / 2.0
        h[i] += d_opt; h[j] -= d_opt
        c = _c5(h, dx)
        if c > bc + 1e-12:
            h[i] -= d_opt; h[j] += d_opt
    h = _proj(h, n / 2.0)
    return h, _c5(h, dx)

def _lbfgsb_solve(h0, a, K, target, n, t0, budget):
    bounds = [(0.0, 1.0)] * n
    lam_lo, lam_hi = -50.0, 50.0
    x = h0.copy()
    for outer in range(40):
        if time.time() - t0 > budget - 5:
            break
        lam = 0.5 * (lam_lo + lam_hi)
        def obj(xx):
            return _sobj(xx, a, K) + lam * (float(xx.sum()) - target)
        def grad(xx):
            return _sgrad(xx, a, K) + lam * np.ones(n)
        try:
            res = minimize(obj, x, jac=grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 100, 'gtol': 1e-12, 'ftol': 1e-15})
            x = res.x
        except Exception:
            break
        s = float(x.sum()) - target
        if s > 0:
            lam_lo = lam
        else:
            lam_hi = lam
        if abs(s) < 1e-10:
            break
    return x
