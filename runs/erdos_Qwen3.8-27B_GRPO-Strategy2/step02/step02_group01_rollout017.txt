def _corr_full(h):

def _smooth_c5(h, dx, alpha):
    c = _corr_full(h)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _smooth_grad(h, dx, alpha):

    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    weights = np.exp(s) / np.sum(np.exp(s))

    if s < 1e-12:

        if abs(diff) > 1e-12:

                    diff -= add

import numpy as np
from scipy.optimize import minimize
import time


def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return np.max(_corr_full(h)) * dx


def _smooth_c5(h, dx, alpha):
    c = _corr_full(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx


def _smooth_grad(h, dx, alpha):
    c = _corr_full(h)
    n = len(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            a = -sh
            g[:n - a] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:n - a]
    return g * dx


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h


def _proj_grad_desc(h, n, dx, alpha, max_iter, max_lr, time_limit, t0):
    lr = max_lr
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad = _smooth_grad(h, dx, alpha)
        for _ls in range(40):
            h_new = _repair(h - lr * grad, n)
            if _c5(h_new, dx) < _c5(h, dx) - 1e-16:
                h = h_new
                break
            lr *= 0.5
        else:
            break
        lr = min(lr * 1.2, max_lr)
    return h


def _gen_start(n, rng, mode):
    if mode == 0:
        h = rng.rand(n)
    elif mode == 1:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 2:
        x = np.linspace(0, 2, n)
        h = np.maximum(0.5 - 0.4 * np.abs(x - 1.0), 0)
    elif mode == 3:
        x = np.linspace(0, 2, n)
        h = np.clip(0.5 + 0.35 * np.sin(np.pi * x), 0, 1)
    elif mode == 4:
        h = np.zeros(n)
        h[(np.arange(n) * 3) % n] = 1.0
    elif mode == 5:
        x = np.linspace(0, 2, n)
        h = np.clip(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x), 0, 1)
    elif mode == 6:
        h = np.full(n, 0.5)
        idx = rng.choice(n, size=n // 3, replace=False)
        h[idx] = 1.0
    else:
        h = rng.rand(n)
    return _repair(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    def elapsed():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Smooth gradient descent from initial (increasing alpha)
    for alpha in [100, 300, 1000, 3000, 10000, 30000]:
        if elapsed() > budget_s - 200:
            break
        h = best_h.copy()
        h = _proj_grad_desc(h, n, dx, alpha, 3000, 0.05, budget_s - 200, t0)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Multi-restart with smooth gradient
    modes = [0, 1, 2, 3, 4, 5, 6]
    for mode in modes:
        if elapsed() > budget_s - 250:
            break
        h_s = _gen_start(n, rng, mode)
        for alpha in [500, 3000, 15000]:
            if elapsed() > budget_s - 250:
                break
            h_s = _proj_grad_desc(h_s, n, dx, alpha, 1500, 0.04, budget_s - 250, t0)
        c5 = _c5(h_s, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s.copy()

    # Phase 3: SLSQP polish
    if elapsed() < budget_s - 100:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
        for alpha in [5000, 30000]:
            if elapsed() > budget_s - 100:
                break
            try:
                res = minimize(
                    lambda x, a=alpha: _smooth_c5(x, dx, a), best_h,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 1000, 'ftol': 1e-18})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 4: Pair coordinate descent (thorough)
    if elapsed() < budget_s - 40:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.04,
                           -0.002, -0.005, -0.01, -0.02, -0.04])
        for _pass in range(60):
            if elapsed() > budget_s - 40:
                break
            improved = False
            for i in range(n):
                if elapsed() > budget_s - 40:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    if elapsed() > budget_s - 40:
                        break
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not improved:
                break

    # Phase 5: Simulated annealing (pair moves)
    if elapsed() < budget_s - 15:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        T = 0.0004
        min_T = 1e-10
        steps = np.array([0.001, 0.003, 0.01, 0.03, 0.08])
        while elapsed() < budget_s - 15:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = steps[rng.randint(len(steps))] * (1 if rng.rand() < 0.5 else -1)
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if ni < -1e-14 or ni > 1 + 1e-14 or nj < -1e-14 or nj > 1 + 1e-14:
                continue
            h[i] = max(0.0, min(1.0, ni))
            h[j] = max(0.0, min(1.0, nj))
            c5_new = _c5(h, dx)
            delta = c5_new - c5_cur
            if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
                c5_cur = c5_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h.copy()
            else:
                h[i], h[j] = hi, hj
            T = max(T * 0.9998, min_T)

    # Phase 6: Ultra-fine 3-point search
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([0.0003, 0.001, 0.003, -0.0003, -0.001, -0.003])
        for _ in range(20000):
            if elapsed() > budget_s - 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = hi, hj

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
