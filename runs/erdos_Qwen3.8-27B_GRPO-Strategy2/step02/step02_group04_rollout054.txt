    # sum_k w[k]*dx * [(1-h[j+k-(n-1)]) * [0<=j+k-(n-1)<n] - h[j-k+(n-1)] * [0<=j-k+(n-1)<n]]

    # Let me use convolution-like operations.

        if wk < 1e-13:

        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wk * h[j_lo2-lag:j_hi2+1-lag]
    
    grad *= dx
    return grad

            best_c = c
            best = h.copy()
    return best, best_c

def _local_search(h, n, dx, rng, deadline):
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0002]
    for scale in scales:
        stalled = 0
        for _pass in range(400):
            if time.time() >= deadline:
                return best_h, best_c
            improved = False
            n_try = 1200
            for _ in range(n_try):
                if time.time() >= deadline:
                    return best_h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
            if improved:
                stalled = 0
            else:
                stalled += 1
                if stalled >= 4:
                    break
    return best_h, best_c

def _coord_descent(h, n, dx, deadline, max_rounds=200):
    best_h = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    for _ in range(max_rounds):
        if time.time() >= deadline:
            return best_h, best_c
        improved = False
        for i in range(n):
            if time.time() >= deadline:
                return best_h, best_c
            orig = h[i]
            for val in [orig * 0.5, orig * 1.5, 1.0 - orig, 0.0, 1.0]:
                if val < 0 or val > 1 or abs(val - orig) < 1e-9:
                    continue
                delta = val - orig
                h[i] = val
                h_compensated = _repair(h, n)
                c = _c5(h_compensated, dx)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h_compensated.copy()
                    h = h_compensated.copy()
                    improved = True
                    break
                h = h_compensated
                h[i] = orig
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c = _c5(best_h, dx)

    # Phase 1: L-BFGS-B with analytic grad (180s)
    dl = t0 + 180
    for alpha in [30, 80, 150, 300, 600]:
        if time.time() > dl or elapsed() > budget_s - 400:
            break
        best_h, c = _lbgfbs(best_h, n, dx, alpha, 2000, dl)
        if c < best_c:
            best_c = c

    # Phase 2: SLSQP refinement (120s)
    dl = t0 + 300
    for alpha in [200, 500, 1000]:
        if time.time() > dl or elapsed() > budget_s - 300:
            break
        best_h, c = _slsqp(best_h, n, dx, alpha, 3000, dl)
        if c < best_c:
            best_c = c

    # Phase 3: Nesterov projected gradient (120s)
    dl = t0 + 420
    for alpha, lr in [(300, 0.008), (600, 0.004), (1000, 0.002), (2000, 0.001)]:
        if time.time() > dl or elapsed() > budget_s - 250:
            break
        best_h, c = _nesterov(best_h, n, dx, alpha, lr, 8000, dl)
        if c < best_c:
            best_c = c
        best_h, c = _lbgfbs(best_h, n, dx, 500, 500, dl)
        if c < best_c:
            best_c = c

    # Phase 4: Multi-restart with local search (250s)
    dl = t0 + 670
    for restart in range(4):
        if time.time() > dl or elapsed() > budget_s - 200:
            break
        if restart == 0:
            start = best_h.copy()
        else:
            jitter = 0.015 * rng.randn(n)
            start = _repair(best_h + jitter, n)
            start, c = _lbgfbs(start, n, dx, 200, 500, dl)
        best_h, c = _local_search(start, n, dx, rng, dl)
        if c < best_c:
            best_c = c
        best_h, c = _slsqp(best_h, n, dx, 500, 1000, dl)
        if c < best_c:
            best_c = c

    # Phase 5: Coordinate descent (100s)
    if elapsed() < budget_s - 100:
        dl = min(t0 + budget_s - 50, t0 + budget_s - 100)
        best_h, c = _coord_descent(best_h, n, dx, dl)
        if c < best_c:
            best_c = c

    # Phase 6: Final intensive local search (80s)
    if elapsed() < budget_s - 40:
        dl = min(t0 + budget_s - 20, t0 + budget_s - 40)
        best_h, c = _local_search(best_h, n, dx, rng, dl)
        if c < best_c:
            best_c = c

    # Phase 7: Final SLSQP + tiny local search (30s)
    if elapsed() < budget_s - 15:
        dl = t0 + budget_s - 10
        best_h, c = _slsqp(best_h, n, dx, 2000, 800, dl)
        if c < best_c:
            best_c = c
    if elapsed() < budget_s - 5:
        dl = t0 + budget_s - 2
        best_h, c = _local_search(best_h, n, dx, rng, dl)
        if c < best_c:
            best_c = c

    best_h = _repair(best_h, n)
    best_c = _c5(best_h, dx)
    return (best_h, best_c, n)
