def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _smooth_max_val_and_grad(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    return val, p

def _corr_grad(h):
    """Gradient matrix: G[j, k] = d(corr[k])/d(h[j])"""
    n = len(h)
    # corr[k] = sum_i h[i+k_offset] * (1-h[i])
    # where k_offset = k - (n-1) = lag
    # d/dh[j] corr[k]:
    #   From h term: i+k_offset = j => i = j-k_offset, contribution (1-h[j-k_offset])
    #   From (1-h) term: i = j, contribution -h[j+k_offset]
    # Valid: 0<=j-k_offset<n for first, 0<=j+k_offset<n for second
    
    # Build gradient as a function of lag
    # For efficiency with n=70, just loop
    grad = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        lag = k - (n-1)
        # First factor: h[j] is h[i+k_offset] when i=j-lag
        # j-lag in [0,n-1] => j in [lag, n-1+lag] intersect [0,n-1]
        for j in range(max(0,lag), min(n, n+lag)):
            grad[j, k] += (1.0 - h[j-lag])
        # Second factor: h[j] is in (1-h[i]) when i=j
        # j+lag in [0,n-1] => j in [-lag, n-1-lag] intersect [0,n-1]
        for j in range(max(0,-lag), min(n, n-lag)):
            grad[j, k] -= h[j+lag]
    return grad

def _grad_smooth_c5(h, alpha):
    """Gradient of smooth max C5 / dx w.r.t. h"""
    c = _corr(h)
    _, p = _smooth_max_val_and_grad(c, alpha)
    G = _corr_grad(h)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h
