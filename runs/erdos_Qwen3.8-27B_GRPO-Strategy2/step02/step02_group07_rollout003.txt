from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _projected_subgrad(h, n, dx, steps=2000, lr_init=0.05, lr_end=0.001, momentum=0.9):
    """Subgradient descent with momentum and projection."""
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _c5(h, dx)
    for s in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        c = m * dx
        if c < best_c:
            best_c = c
            best_h = h.copy()
        k = np.argmax(corr)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-h if diff > 0 else h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_smooth(h, n, alpha, maxiter=500, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _slsqp_bottleneck(h, n, target_val, n_top=5, maxiter=400):
    """SLSQP: minimize ||x-h||^2 s.t. top lags <= target."""
    corr = _corr(h)
    top_idx = np.argsort(corr)[-n_top:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    for k in top_idx:
        ki = int(k)
        cons.append({'type': 'ineq', 'fun': lambda x, ki=ki: target_val - _corr(x)[ki]})
    def obj(x):
        return np.sum((x - h) ** 2)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _subgrad_descent(h, n, dx, steps, lr0=0.15, mom=0.9):
    h = h.copy()
    v = np.zeros(n)
    best_h, best_c = h.copy(), _c5(h, dx)
    for s in range(steps):
        c = _c5(h, dx)
        if c < best_c:
            best_c, best_h = c, h.copy()
        corr = _corr(h)
        k = int(np.argmax(corr))
        sh = k - (n - 1)
        g = np.zeros(n)
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            g[:n - sh] -= h[sh:]
        else:
            p = -sh
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
        lr = lr0 / (1.0 + 0.003 * s) ** 0.6
        v = mom * v + (1.0 - mom) * g
        h = _repair(h - lr * v, n)
    return best_h, best_c

def _cd_pass(h, n, dx, n_grid):
    cur_c = np.max(_corr(h))
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            oi, oj = h[i], h[j]
            best_t, best_m = oi, cur_c
            for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
                h[i], h[j] = t, hsum - t
                m = np.max(_corr(h))
                if m < best_m:
                    best_m, best_t = m, t
            if best_t != oi and best_m < cur_c - 1e-13:
                step = (hi - lo) / n_grid
                lo2 = max(lo, best_t - step)
                hi2 = min(hi, best_t + step)
                for t in np.linspace(lo2, hi2, 13):
                    h[i], h[j] = t, hsum - t
                    m = np.max(_corr(h))
                    if m < best_m:
                        best_m, best_t = m, t
                h[i], h[j] = best_t, hsum - best_t
                cur_c = best_m
                improved = True
            else:
                h[i], h[j] = oi, oj
    return h, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = _c5(h, dx)

    # Phase 1: SLSQP smooth (quick, diverse alphas)
    for a in [80, 300, 1000, 3000]:
        if rem() < 350:
            break
        hh = _slsqp_smooth(best_h, n, a, 500, 1e-15)
        c = _c5(hh, dx)
        if c < best_c:
            best_c, best_h = c, hh.copy()

    # Phase 2: Subgradient descent (different exploration)
    if rem() > 250:
        steps = min(3000, int(rem() * 0.25 / 0.0008))
        hh, c = _subgrad_descent(best_h, n, dx, steps, lr0=0.18)
        if c < best_c:
            best_c, best_h = c, hh.copy()
        # Quick CD after subgrad
        if rem() > 200:
            _cd_pass(best_h, n, dx, 15)
            c = _c5(best_h, dx)
            if c < best_c:
                best_c, best_h = c, best_h.copy()

    # Phase 3: Bottleneck SLSQP (iterative target tightening)
    if rem() > 200:
        tgt = best_c / dx * 0.999
        for _ in range(8):
            if rem() < 180:
                break
            hh = _slsqp_bottleneck(best_h, n, tgt, n_top=6, maxiter=300)
            c = _c5(hh, dx)
            if c < best_c:
                best_c, best_h = c, hh.copy()
                tgt = c / dx * 0.999
            else:
                tgt = min(tgt, c / dx * 1.0005)

    # Phase 4: CD (coarse to fine)
    for ng in [15, 21, 27, 31]:
        if rem() < 80:
            break
        t_end = time.time() + max(10, rem() * 0.15)
        while time.time() < t_end and rem() > 60:
            hh, imp = _cd_pass(best_h, n, dx, ng)
            if not imp:
                break
            c = _c5(hh, dx)
            if c < best_c:
                best_c, best_h = c, hh.copy()
            else:
                break

    # Phase 5: SLSQP polish
    if rem() > 80:
        hh = _slsqp_smooth(best_h, n, 5000, 800, 1e-16)
        c = _c5(hh, dx)
        if c < best_c:
            best_c, best_h = c, hh.copy()

    # Phase 6: Multi-restart (diverse perturbations)
    for si in range(10):
        if rem() < 30:
            break
        if si < 3:
            # Shift perturbation (exploits near-translation invariance)
            s_shift = (si + 1) % n
            hp = np.roll(best_h, s_shift)
        elif si < 6:
            # Random Gaussian
            sig = [0.05, 0.1, 0.15][si - 3]
            hp = best_h + sig * rng.randn(n)
        else:
            # Sinusoidal + random
            freq = 2 + (si % 5)
            phase = rng.uniform(0, 2 * np.pi)
            hp = best_h + 0.06 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            hp += 0.03 * rng.randn(n)
        hp = _repair(hp, n)
        hp = _slsqp_smooth(hp, n, 800, 400, 1e-14)
        c = _c5(hp, dx)
        if c < best_c:
            best_c, best_h = c, hp.copy()
            # Quick CD on new best
            for _ in range(40):
                if rem() < 20:
                    break
                hh, imp = _cd_pass(best_h, n, dx, 21)
                c2 = _c5(hh, dx)
                if c2 < best_c:
                    best_c, best_h = c2, hh.copy()
                else:
                    break

    # Phase 7: Final CD + SLSQP
    if rem() > 20:
        for _ in range(100):
            if rem() < 10:
                break
            hh, imp = _cd_pass(best_h, n, dx, 27)
            if not imp:
                break
            c = _c5(hh, dx)
            if c < best_c:
                best_c, best_h = c, hh.copy()
            else:
                break
    if rem() > 8:
        hh = _slsqp_smooth(best_h, n, 8000, 400, 1e-16)
        c = _c5(hh, dx)
        if c < best_c:
            best_c, best_h = c, hh.copy()

    best_h = _repair(best_h, n)
    best_c = _c5(best_h, dx)
    return (best_h, best_c, n)
