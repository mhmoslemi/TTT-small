def _project(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to adjust sum
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Use bisection on the Lagrange multiplier
    # We want: sum(clipped(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0.0, 1.0).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

    return float(np.max(_corr(h)) * dx)

        else:
            hi = mid

    m = 2 * n - 1
    dcd = np.zeros((m, n))
    for k in range(m):

        if off >= 0:
            dcd[k, :n - off] += 1.0 - h[off:n]
            if off > 0:
                dcd[k, off:] -= h[:n - off]
        else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _dcd_mat(h, n):
    m = 2 * n - 1
    dcd = np.zeros((m, n))
    for k in range(m):
        off = k - (n - 1)
        if off >= 0:
            dcd[k, :n - off] += 1.0 - h[off:n]
            if off > 0:
                dcd[k, off:] -= h[:n - off]
        else:
            L = -off
            dcd[k, L:] += 1.0 - h[:n - L]
            dcd[k, :n - L] -= h[L:]
    return dcd

def _obj_grad(h, alpha, n, mu=200.0):
    c = _corr(h)
    m = c.max()
    sm = np.clip(alpha * (c - m), -300, 300)
    ex = np.exp(sm - sm.max())
    w = ex / ex.sum()
    obj = m + (sm.max() + np.log(ex.sum())) / alpha
    pen = mu * (h.sum() - n / 2.0) ** 2
    obj += pen
    dcd = _dcd_mat(h, n)
    g = dcd.T.dot(w)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return obj, g

def _cd_fast(h, n, dx, t_end, n_gs=25, recompute_every=30):
    hi_c5 = _c5(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    pair_count = 0
    while time.time() < t_end:
        c = _corr(h)
        dcd = _dcd_mat(h, n)
        improved = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                pair_count += 1
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi_ = min(1.0, hsum)
                if hi_ - lo < 1e-8:
                    continue
                dv = dcd[:, i] - dcd[:, j]
                oi = h[i]
                a, b = lo, hi_
                x1 = b - gr * (b - a)
                x2 = a + gr * (b - a)
                f1 = np.max(c + (x1 - oi) * dv)
                f2 = np.max(c + (x2 - oi) * dv)
                for _ in range(n_gs - 2):
                    if f1 < f2:
                        b, x2, f2 = x2, x1, f1
                        x1 = b - gr * (b - a)
                        f1 = np.max(c + (x1 - oi) * dv)
                    else:
                        a, x1, f1 = x1, x2, f2
                        x2 = a + gr * (b - a)
                        f2 = np.max(c + (x2 - oi) * dv)
                t_mid = 0.5 * (a + b)
                cf = np.max(c + (t_mid - oi) * dv)
                if cf < hi_c5 - 1e-12:
                    h[i] = t_mid
                    h[j] = hsum - t_mid
                    c = c + (t_mid - oi) * dv
                    hi_c5 = float(cf)
                    improved = True
                    if pair_count % recompute_every == 0:
                        c = _corr(h)
        if not improved:
            break
    h = _project(h, n)
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h = _project(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B with increasing smoothness
    for alpha in [50, 100, 200, 500, 1000]:
        if elapsed() > budget_s * 0.30:
            break
        res = minimize(_obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, args=(alpha, n),
                       options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-13})
        h = _project(res.x, n)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Fast coordinate descent (main workhorse)
    t_cd_end = t0 + budget_s * 0.82

        if c < best_c5:

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
