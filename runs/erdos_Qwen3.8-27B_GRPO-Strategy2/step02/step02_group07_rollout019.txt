    """Subgradient method iterations."""
    dx = 2.0 / n
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = c.max()
        # Active set: within tol of max
        active = c > cmax - 0.01
        if not active.any():
            active = np.zeros_like(c)
            active[c.argmax()] = True
        w = active / active.sum()
        
        # Compute gradient at active points
        # dc[nt]/dh[i] = (1-h[i-k])*valid1 - h[i+k]*valid2 where k=nt-(n-1)
        k_vals = np.arange(2*n-1) - (n-1)
        grad = np.zeros(n)
        for i in range(n):
            # Term 1: (1-h[i-k]) for k where 0<=i-k<n
            # k range: [i-n+1, i]
            k1_lo = max(k_vals[0], i - n + 1)
            k1_hi = min(k_vals[-1], i)
            # t indices where k_vals[t] in [k1_lo, k1_hi]
            t1 = (k_vals >= k1_lo) & (k_vals <= k1_hi)
            if t1.any():
                idx = i - k_vals[t1]  # 0<=idx<n guaranteed
                grad[i] += np.sum(w[t1] * (1 - h[idx]))
            # Term 2: -h[i+k] for k where 0<=i+k<n
            k2_lo = max(k_vals[0], -i)
            k2_hi = min(k_vals[-1], n - 1 - i)
            t2 = (k_vals >= k2_lo) & (k_vals <= k2_hi)
            if t2.any():
                idx = i + k_vals[t2]  # 0<=idx<n guaranteed
                grad[i] += np.sum(w[t2] * (-h[idx]))
        
        # Adaptive step size
        step = 1.0 / (it + 10)
        # Or: step = 2*(cmax*dx - target) / (||grad||^2 + eps)
        gn = np.linalg.norm(grad)
        if gn > 1e-10:
            step = min(step, 1.0 / gn)
        
        h_new = _project(h - step * grad, n)
        c_new = _c5_val(h_new, dx)
        if c_new > cmax * dx:
            # Rejection: halve step
            step *= 0.5
            h_new = _project(h - step * grad, n)
            c_new = _c5_val(h_new, dx)
        h = h_new
    
    return h

            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _smooth_val(x, n, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 50)
    e = np.exp(s)
    w = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    kv = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        m1 = (kv >= i - n + 1) & (kv <= i)
        if m1.any():
            g[i] += np.sum(w[m1] * (1.0 - x[i - kv[m1]]))
        m2 = (kv >= -i) & (kv <= n - 1 - i)
        if m2.any():
            g[i] += np.sum(w[m2] * (-x[i + kv[m2]]))
    return val, g

def _f_wrap(x, n, alpha):
    return _smooth_val(x, n, alpha)

def _g_wrap(x, n, alpha):
    return _smooth_grad(x, n, alpha)[1]

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def _slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_f_wrap, h, args=(n, alpha), jac=_g_wrap,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return h.copy()

def _subgrad(h, n, n_iters, t_start, t_limit, gap=0.003):
    dx = 2.0 / n
    kv = np.arange(2 * n - 1) - (n - 1)
    best = _c5(h, dx)
    for it in range(n_iters):
        if time.time() - t_start > t_limit:
            break
        c = _corr(h)
        cmax = c.max()
        active = c > cmax - gap
        if not active.any():
            active = np.zeros_like(c); active[c.argmax()] = 1
        w = active.astype(float) / active.sum()
        g = np.zeros(n)
        for i in range(n):
            m1 = (kv >= i - n + 1) & (kv <= i)
            if m1.any():
                g[i] += np.sum(w[m1] * (1.0 - h[i - kv[m1]]))
            m2 = (kv >= -i) & (kv <= n - 1 - i)
            if m2.any():
                g[i] += np.sum(w[m2] * (-h[i + kv[m2]]))
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        step = min(2.0 / (it + 5.0), 1.0 / gn)
        for _ in range(4):
            h_new = _project_sum(h - step * g, n)
            c_new = _c5(h_new, dx)
            if c_new < best:
                best = c_new; h = h_new; break
            step *= 0.5
    return h, best

def _cd_focused(h, n, dx, t_start, t_limit, n_grid=15):
    current = _c5(h, dx)
    rng = np.random.RandomState(12345)
    for _ in range(8000):
        if time.time() - t_start > t_limit:
            break
        improved = False
        c = _corr(h)
        cmax = c.max()
        perm = rng.permutation(n)
        for i in perm:
            for j in perm:
                if j <= i:
                    continue
                hsum = h[i] + h[j]
                lo = max(0.0, hsum - 1.0)
                hi = min(1.0, hsum)
                if hi - lo < 1e-10:
                    continue
                oi, oj = h[i], h[j]
                best_t, best_c = oi, cmax
                for t in np.linspace(lo + 1e-8, hi - 1e-8, n_grid):
                    h[i] = t; h[j] = hsum - t
                    cm = np.max(_corr(h))
                    if cm < best_c:
                        best_c = cm; best_t = t
                if best_c < cmax - 1e-13:
                    h[i] = best_t; h[j] = hsum - best_t
                    cmax = best_c; improved = True
                else:
                    h[i] = oi; h[j] = oj
        if not improved:
            break
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP with gradient, multiple alpha
    for alpha in [80, 200, 600, 1500]:
        if time.time() - t0 > budget_s - 500:
            break
        h2 = _slsqp(best_h, n, alpha, 600, 1e-14)
        c2 = _c5(h2, dx)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 2: Subgradient
    if time.time() - t0 < budget_s - 350:
        h2, c2 = _subgrad(best_h, n, 3000, t0, budget_s - 400, gap=0.002)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: SLSQP with high alpha
    if time.time() - t0 < budget_s - 250:
        h2 = _slsqp(best_h, n, 4000, 1000, 1e-15)
        c2 = _c5(h2, dx)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 4: Coordinate descent (coarse)
    if time.time() - t0 < budget_s - 150:
        h2, c2 = _cd_focused(best_h, n, dx, t0, budget_s - 120, n_grid=13)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 5: Multi-restart with perturbation
    sigmas = [0.04, 0.08, 0.15, 0.06, 0.2, 0.1, 0.03]
    for si, sig in enumerate(sigmas):
        if time.time() - t0 > budget_s - 80:
            break
        hp = best_h + sig * rng.randn(n)
        hp = _repair(hp, n)
        h2 = _slsqp(hp, n, 400 + 200 * si, 400, 1e-13)
        c2 = _c5(h2, dx)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
            if time.time() - t0 < budget_s - 60:
                h2b, c2b = _cd_focused(h2, n, dx, t0, min(budget_s - 40, time.time() - t0 + 80), 11)
                if c2b < best_c5:
                    best_c5, best_h = c2b, h2b.copy()

    # Phase 6: Subgradient polish
    if time.time() - t0 < budget_s - 40:
        h2, c2 = _subgrad(best_h, n, 2000, t0, budget_s - 25, gap=0.001)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 7: Final SLSQP
    if time.time() - t0 < budget_s - 10:
        h2 = _slsqp(best_h, n, 10000, 400, 1e-15)
        c2 = _c5(h2, dx)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Final CD
    if time.time() - t0 < budget_s - 5:
        h2, c2 = _cd_focused(best_h, n, dx, t0, budget_s - 3, n_grid=17)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
