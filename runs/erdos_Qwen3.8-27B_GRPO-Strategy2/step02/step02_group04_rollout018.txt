import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _smooth_max_val(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _weights(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return np.exp(s) / np.exp(s).sum()

def _grad_smooth_max(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        mk = k - (n-1)
        wk = w[k]
        # Term 1: (1-h[j+mk]) for valid j+mk
        if mk >= 0:
            j_hi = n - mk
            if j_hi > 0:
                grad[:j_hi] += wk * (1.0 - h[mk:n])
        else:
            mk2 = -mk
            grad[mk2:] += wk * (1.0 - h[:n-mk2])
        # Term 2: -h[j-mk] for valid j-mk
        if mk >= 0:
            grad[mk:] -= wk * h[:n-mk]
        else:
            mk2 = -mk
            grad[:n-mk2] -= wk * h[mk2:n]
    return grad

def _project(h, target_sum):
    """Project h onto {sum=target_sum, 0<=h<=1} via water-filling."""
    h = np.clip(h, 0, 1)
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(h + mid, 0, 1)
        if y.sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo+hi)/2, 0, 1)

def _gd_step(h, alpha, lr, momentum, v, rng):
    """One gradient descent step with momentum."""
    n = len(h)
    corr = _corr(h)
    w = _weights(corr, alpha)
    grad = _grad_smooth_max(h, w)
    v = momentum * v - lr * grad
    h_new = _project(h + v, n/2.0)
    return h_new, v, _c5(h_new, 2.0/n)

    """Run GD for n_iter steps."""
    n = len(h)
    dx = 2.0/n
    v = np.zeros(n)
    best = _c5(h, dx)
    best_h = h.copy()
    no_improve = 0
    for i in range(n_iter):
        if time.time() > deadline:
            break
        h, v, c5 = _gd_step(h, alpha, lr, momentum, v, rng)
        if c5 < best - 1e-12:
            best = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break
    return best_h, best

def _lbfgs_refine(h, alpha, mu, target_sum, deadline, maxiter=500):
    """L-BFGS-B refinement with penalty."""
    n = len(h)
    dx = 2.0/n
    
    def obj_and_grad(x):
        corr = _corr(x)
        w = _weights(corr, alpha)
        gm = _grad_smooth_max(x, w)
        f = _smooth_max_val(corr, alpha)
        pen = mu * (np.sum(x) - target_sum)**2
        g = gm + 2*mu*(np.sum(x) - target_sum)
        return f + pen, g

                       bounds=[(0,1)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = _project(res.x, target_sum)
        c5 = _c5(h_new, dx)
        return h_new, c5
    except:
        return h, _c5(h, dx)

                    h[j] += d
                    h = h_p  # restore
            if improved:
                stalled = 0
            else:
                stalled += 1
        stalled_total = 0
    
    return h, best

def _coordinate_descent(h, n, target_sum, deadline, max_sweeps=50):

    dx = 2.0/n
    best = _c5(h, dx)

            break

            # The sum of other coordinates is fixed

            # h[j] can range [max(0, others_sum - (n-1)), min(1, others_sum)]
            # (since other coords are in [0,1], their sum is in [0, n-1])
            lo = max(0.0, others_sum - (n-1))
            hi = min(1.0, others_sum)

            # Use golden section or just try a few points

            best_j = old
            # Try quantile points
            from scipy.optimize import minimize_scalar
            def f1d(t):
                h[j] = t
                return _c5(h, dx)
            
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                options={'maxiter': 100, 'xatol': 1e-8})
            if res.fun < best_val - 1e-12:
                h[j] = res.x
                best_val = res.fun
                best_j = res.x
                improved_any = True
        
        h[j] = best_j  # restore in case
        if not improved_any:
            break
    
    h = _project(h, target_sum)
    best = _c5(h, dx)
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max_val(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _weights(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return np.exp(s) / np.exp(s).sum()

def _grad_sm(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        mk = k - (n-1)
        wk = w[k]
        if mk >= 0:
            jhi = n - mk
            if jhi > 0:
                grad[:jhi] += wk * (1.0 - h[mk:n])
            if jhi > 0:
                grad[mk:] -= wk * h[:jhi]
        else:
            mk2 = -mk
            grad[mk2:] += wk * (1.0 - h[:n-mk2])
            grad[:n-mk2] -= wk * h[mk2:n]
    return grad

def _proj(h, ts):
    h = np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        s = np.clip(h+mid, 0, 1).sum()
        if s < ts: lo = mid
        else: hi = mid
    return np.clip(h+(lo+hi)/2, 0, 1)

def _gd(h0, alpha, lr, mom, niter, deadline, ts):
    n = len(h0); dx = 2.0/n
    h = h0.copy(); v = np.zeros(n)
    best = _c5(h, dx); bh = h.copy()
    for _ in range(niter):
        if time.time() > deadline: break
        c = _corr(h); w = _weights(c, alpha)
        g = _grad_sm(h, w)
        v = mom*v - lr*g
        hn = _proj(h+v, ts)
        c5 = _c5(hn, dx)
        h = hn
        if c5 < best-1e-13: best=c5; bh=h.copy()
    return bh, best

def _lbfgs(h0, alpha, mu, ts, deadline, mx=300):
    n = len(h0); dx = 2.0/n
    def fg(x):
        c = _corr(x); w = _weights(c, alpha)
        f = _smooth_max_val(c, alpha)
        g = _grad_sm(x, w)
        p = mu*(np.sum(x)-ts)
        return f+mu*p*p, g+2*mu*p
    try:
        r = minimize(fg, h0, method='L-BFGS-B', bounds=[(0,1)]*n,
                     options={'maxiter':mx,'ftol':1e-18})
        hn = _proj(r.x, ts)
        return hn, _c5(hn, dx)
    except: return h0, _c5(h0, dx)

def _ls(h0, rng, deadline, ts):
    n = len(h0); dx = 2.0/n
    h = h0.copy(); best = _c5(h, dx)
    for sc in [0.04,0.02,0.01,0.005,0.002,0.001,0.0005,0.0002,0.0001]:
        st = 0
        while st < 300 and time.time() < deadline:
            imp = False
            for _ in range(600):
                if time.time() > deadline: return h, best
                if rng.random()<0.25:
                    i = rng.randint(n)
                    d = sc*(2*rng.random()-1)
                    h[i]+=d
                    hp = _proj(h, ts)
                    c5 = _c5(hp, dx)
                    h[i]-=d
                    if c5<best-1e-12: best=c5; h=hp.copy(); imp=True
                else:
                    i,j = rng.randint(n), rng.randint(n)
                    if i==j: continue
                    d = sc*(2*rng.random()-1)
                    h[i]+=d; h[j]-=d
                    hp = _proj(h, ts)
                    c5 = _c5(hp, dx)
                    h[i]-=d; h[j]+=d
                    if c5<best-1e-12: best=c5; h=hp.copy(); imp=True
            st = 0 if imp else st+1
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max_val(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _weights(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return np.exp(s) / np.exp(s).sum()

def _grad_sm(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        mk = k - (n - 1)
        wk = w[k]
        if mk >= 0:
            jhi = n - mk
            if jhi > 0:
                grad[:jhi] += wk * (1.0 - h[mk:n])
                grad[mk:] -= wk * h[:jhi]
        else:
            mk2 = -mk
            grad[mk2:] += wk * (1.0 - h[:n - mk2])
            grad[:n - mk2] -= wk * h[mk2:n]
    return grad

def _proj(h, ts):
    h = np.clip(h, 0, 1)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0, 1).sum()
        if s < ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def _exact_fix(h, ts):
    h = np.clip(h, 0, 1)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-14:
            break
    return h

def _gd_run(h0, alpha_sched, lr_base, mom, steps_per, deadline, ts):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    best = _c5(h, dx)
    bh = h.copy()
    for alpha in alpha_sched:
        lr = lr_base / alpha
        for _ in range(steps_per):
            if time.time() > deadline:
                break
            c = _corr(h)
            w = _weights(c, alpha)
            g = _grad_sm(h, w)
            v = mom * v - lr * g
            hn = _exact_fix(_proj(h + v, ts), ts)
            c5 = _c5(hn, dx)
            h = hn
            if c5 < best - 1e-13:
                best = c5
                bh = h.copy()
    return bh, best

def _lbfgs_run(h0, alpha, mu, ts, deadline, mx=400):
    n = len(h0)
    dx = 2.0 / n
    def fg(x):
        c = _corr(x)
        w = _weights(c, alpha)
        f = _smooth_max_val(c, alpha)
        g = _grad_sm(x, w)
        p = mu * (np.sum(x) - ts)
        return f + p * p, g + 2.0 * p
    try:
        r = minimize(fg, h0, method='L-BFGS-B', bounds=[(0, 1)] * n,
                     options={'maxiter': mx, 'ftol': 1e-20, 'gtol': 1e-14})
        hn = _exact_fix(_proj(r.x, ts), ts)
        return hn, _c5(hn, dx)
    except Exception:
        return h0, _c5(h0, dx)

def _local_search(h0, rng, deadline, ts):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best = _c5(h, dx)
    scales = [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    for sc in scales:
        stalled = 0
        while stalled < 250 and time.time() < deadline:
            imp = False
            for _ in range(800):
                if time.time() > deadline:
                    return h, best
                if rng.random() < 0.2:
                    i = rng.randint(n)
                    d = sc * (2 * rng.random() - 1)
                    h[i] += d
                    hp = _exact_fix(_proj(h, ts), ts)
                    c5 = _c5(hp, dx)
                    h[i] -= d
                    if c5 < best - 1e-12:
                        best = c5
                        h = hp.copy()
                        imp = True
                else:
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d = sc * (2 * rng.random() - 1)
                    h[i] += d
                    h[j] -= d
                    hp = _exact_fix(_proj(h, ts), ts)
                    c5 = _c5(hp, dx)
                    h[i] -= d
                    h[j] += d
                    if c5 < best - 1e-12:
                        best = c5
                        h = hp.copy()
                        imp = True
            stalled = 0 if imp else stalled + 1
    return h, best

def _anneal_refine(h0, rng, deadline, ts, t0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best = _c5(h, dx)
    bh = h.copy()
    temp = 0.02
    for i in range(20000):
        if time.time() > deadline:
            break
        temp *= 0.9997
        d = 0.005 * (2 * rng.random() - 1)
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        h[i1] += d
        h[i2] -= d
        hp = _exact_fix(_proj(h, ts), ts)
        c5 = _c5(hp, dx)
        if c5 < best - 1e-12:
            best = c5
            bh = hp.copy()
            h = hp
        elif rng.random() < np.exp(-(c5 - best + 1e-10) / max(temp, 1e-10)):
            h = hp
        else:
            h[i1] -= d
            h[i2] += d
    return bh, best
