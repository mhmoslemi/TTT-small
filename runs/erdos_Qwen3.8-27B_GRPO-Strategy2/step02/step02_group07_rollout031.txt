```python
def _pair_min(t_lo, t_hi, P, Q, n_lags, n_iter=40):
    # P, Q: arrays of length n_lags, f_k(t) = P[k] + Q[k]*t
    # Find t in [t_lo, t_hi] minimizing max_k f_k(t)
    # Convex function, use ternary search
    for _ in range(n_iter):
        t1 = t_lo + (t_hi - t_lo) / 3
        t2 = t_hi - (t_hi - t_lo) / 3
        f1 = np.max(P + Q * t1)
        f2 = np.max(P + Q * t2)
        if f1 < f2:
            t_hi = t2
        else:
            t_lo = t1
    t_opt = (t_lo + t_hi) / 2
    return t_opt, np.max(P + Q * t_opt)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _cd_pass(h, n, dx, order, n_tern=35, T=0.0, rng=None):
    cur = np.max(np.correlate(h, 1.0 - h, mode='full'))
    count = 0
    for i, j in order:
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-11:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = 0.0, s
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i], h[j] = 1.0, s - 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        Q = c1 - c0
        a, b = lo, hi
        for _ in range(n_tern):
            t1 = a + (b - a) / 3.0
            t2 = b - (b - a) / 3.0
            if np.max(c0 + Q * t1) < np.max(c0 + Q * t2):
                b = t2
            else:
                a = t1
        t_opt = (a + b) / 2.0
        h[i], h[j] = t_opt, s - t_opt
        new_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
        delta = new_max - cur
        if delta < -1e-13 or (T > 0 and rng is not None and rng.random() < np.exp(-delta / T)):
            cur = new_max
            count += 1
        else:
            h[i], h[j] = old_i, old_j
    return h, cur, count

def _slsqp_polish(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _make_candidate(n, seed_val, mode):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    if mode == 0:
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
    elif mode == 1:
        h = np.tile(np.array([0.8, 0.5, 0.5, 0.2, 0.5]), n // 5 + 1)[:n]
    elif mode == 2:
        h = 0.5 + 0.3 * np.cos(np.pi * x + rng.rand()) + 0.05 * rng.randn(n)
    else:
        h = rng.rand(n) * 0.6 + 0.2
    return _repair(h, n)
```
