                v1, v2, v3 = best[a] + d1, best[b] + d2, best[c] + d3
                if v1 < -1e-9 or v1 > 1 + 1e-9 or v2 < -1e-9 or v2 > 1 + 1e-9 or v3 < -1e-9 or v3 > 1 + 1e-9:
                    continue
                best[a], best[b], best[c] = v1, v2, v3
                nc = _c5_val(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[a] -= d1
                    best[b] -= d2
                    best[c] -= d3
            else:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                d = rng.uniform(-0.2, 0.2)
                if abs(d) < 1e-8:
                    continue
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5_val(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    return best, best_c5

def _perturb(h, n, sigma, rng):
    hp = h + sigma * rng.randn(n)
    return _repair(hp, n)

def _eligible(t0, budget_s, margin):
    return time.time() < t0 + budget_s - margin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5_val(best_h, dx)

    # Phase 1: SLSQP from initial with increasing alpha
    h_cur = h_init.copy()
    alphas_p1 = [40, 80, 150, 300, 600, 1200]
    for a in alphas_p1:
        if not _eligible(t0, budget_s, 500):
            break
        h_new = _slsqp(h_cur, n, dx, a, 500, 1e-14)
        c5_new = _c5_val(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new

    # Phase 2: Extended local search
    if _eligible(t0, budget_s, 350):
        deadline = t0 + budget_s - 350
        best_h, best_c5 = _local_search(best_h, n, dx, rng, deadline)

    # Phase 3: Multi-restart with SLSQP + local search
    sigmas = [0.25, 0.4, 0.15, 0.6, 0.3, 0.5, 0.2, 0.35, 0.7, 0.1, 0.45, 0.55]
    restart_idx = 0
    while _eligible(t0, budget_s, 120):
        sigma = sigmas[restart_idx % len(sigmas)]
        hp = _perturb(best_h, n, sigma, rng)
        if _eligible(t0, budget_s, 150):
            hc = _slsqp(hp, n, dx, 250, 300, 1e-13)
        else:
            hc = hp
        c5 = _c5_val(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()
            if _eligible(t0, budget_s, 100):
                dl = t0 + budget_s - 100
                hl, cl = _local_search(best_h, n, dx, rng, dl)
                if cl < best_c5:
                    best_c5 = cl
                    best_h = hl.copy()
        elif c5 < best_c5 + 0.003 and _eligible(t0, budget_s, 100):
            dl = t0 + budget_s - 100
            hl, cl = _local_search(hc, n, dx, rng, dl)
            if cl < best_c5:
                best_c5 = cl
                best_h = hl.copy()
        restart_idx += 1

    # Phase 4: High-alpha SLSQP polish
    if _eligible(t0, budget_s, 40):
        for a in [2000, 5000, 10000]:
            if not _eligible(t0, budget_s, 25):
                break
            hc = _slsqp(best_h, n, dx, a, 600, 1e-15)
            c5 = _c5_val(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()

    # Phase 5: Final fine local search
    if _eligible(t0, budget_s, 8):
        deadline = t0 + budget_s - 3
        best_h, best_c5 = _local_search(best_h, n, dx, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
