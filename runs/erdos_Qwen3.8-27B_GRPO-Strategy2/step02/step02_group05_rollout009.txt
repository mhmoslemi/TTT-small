import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        lo = 1.0 - h if diff > 0 else h
        order = np.argsort(lo)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b
                diff += b
            if abs(diff) < 1e-9:
                break
    return h

def _smooth_max_obj(x, alpha, dx):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(s.exp().sum()) / alpha) * dx

def _smooth_max_grad(x, alpha, dx):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = s.exp()
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        sh = k - (n - 1)
        # dc_k/dh_j = (1-h[j+sh]) if valid - h[j-sh] if valid
        # First term: j+sh in [0,n-1]
        # j ranges over [max(0,-sh), min(n-1, n-1-sh)]
        jlo = max(0, -sh)
        jhi = min(n - 1, n - 1 - sh)
        for j in range(jlo, jhi + 1):
            g[j] += wk * (1.0 - x[j + sh])
        # Second term: j-sh in [0,n-1]
        # j ranges over [max(0, sh), min(n-1, n-1+sh)]
        jlo2 = max(0, sh)
        jhi2 = min(n - 1, n - 1 + sh)
        for j in range(jlo2, jhi2 + 1):
            g[j] -= wk * x[j - sh]
    return g * dx

def _con_fun(x, target):
    return x.sum() - target

def _slsqp(h0, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (target,)}]
    try:
        res = minimize(_smooth_max_obj, h0, jac=_smooth_max_grad,
                       args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < _c5(h0, dx) or True:
            return _repair(res.x, n)
    except Exception:
        pass
    return h0.copy()

def _systematic_ls(h, n, dx, rng, deadline):
    """Systematic pair-exchange local search."""
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps_list = [0.02, 0.05, 0.1, 0.02, -0.02, -0.05, -0.1, -0.02]
    eps_adaptive = 0.05
    no_improve_sweeps = 0
    
    # Build pair list (i < j)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    sweep = 0
    while time.time() < deadline:
        improved = False
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            for d in steps_list:
                ni = best[i] + d
                nj = best[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                if nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                best[i] = ni
                best[j] = nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    improved = True
                else:
                    best[i] -= d
                    best[j] += d
            # Also try the other direction
            for d in steps_list:
                ni = best[i] - d
                nj = best[j] + d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                if nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                best[i] = ni
                best[j] = nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-12:
                    best_c5 = nc
                    improved = True
                else:
                    best[i] += d
                    best[j] -= d
        sweep += 1
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                break
            # Try smaller steps
            steps_list = [s * 0.5 for s in steps_list]
            if abs(steps_list[0]) < 1e-5:
                break
        else:
            no_improve_sweeps = 0
            # Occasionally expand steps
            if sweep % 5 == 0:
                steps_list = [s * 1.5 for s in steps_list]
                steps_list = [max(-0.5, min(0.5, s)) for s in steps_list]
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        lo = (1.0 - h) if diff > 0 else h
        order = np.argsort(lo)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b
                diff += b
            if abs(diff) < 1e-9:
                break
    return h

def _smm_obj(x, alpha, dx):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s)))) / alpha * dx

def _smm_grad(x, alpha, dx):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-7:
            continue
        ilo = max(0, p - n + 1)
        ihi = min(n - 1, p)
        # (1-h[p-j]) if j in I_p
        for j in range(ilo, ihi + 1):
            ps = p - j
            if 0 <= ps < n:
                g[j] += wp * (1.0 - x[ps])
            if ps != j:
                g[j] -= wp * x[ps]
            else:
                g[j] += 0  # already counted
        # -h[p-j] if (p-j) in I_p
        for j in range(ilo, ihi + 1):
            ps = p - j
            if 0 <= ps < n and ps != j:
                g[j] -= 0  # already handled above via the second term
    # Fix: redo gradient properly
    g = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-7:
            continue
        ilo = max(0, p - n + 1)
        ihi = min(n - 1, p)
        for j in range(ilo, ihi + 1):
            ps = p - j
            if 0 <= ps < n:
                g[j] += wp * (1.0 - x[ps])
            if 0 <= ps < n and ps != j:
                g[j] -= wp * x[ps]
            elif ps == j:
                g[j] -= wp * x[ps]
    return g * dx

def _con(x, target):
    return x.sum() - target

def _slsqp(h0, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con, 'args': (target,)}]
    try:
        res = minimize(_smm_obj, h0, jac=_smm_grad, args=(alpha, dx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = _repair(res.x, n)
        if _c5(hx, dx) < _c5(h0, dx):
            return hx
    except Exception:
        pass
    return h0.copy()

def _sweep_ls(h, n, dx, deadline, base_eps=0.04):
    best = h.copy()
    best_c = _c5(best, dx)
    eps = base_eps
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    stalls = 0
    while time.time() < deadline:
        improved = False
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            for d in (eps, -eps, 2*eps, -2*eps):
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c - 1e-13:
                    best_c = nc
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
        if improved:
            stalls = 0
            if eps > 0.005:
                eps = max(0.003, eps * 0.85)
        else:
            stalls += 1
            eps *= 0.6
            if stalls > 5 or eps < 0.001:
                break
    return best, best_c

def _random_ls(h, n, dx, rng, deadline, nsteps=200):
    best = h.copy()
    best_c = _c5(best, dx)
    steps = np.array([0.01, 0.03, 0.06, 0.1, -0.01, -0.03, -0.06, -0.1])
    for _ in range(nsteps):
        if time.time() > deadline:
            break
        improved = False
        for _t in range(80):
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            d = steps[int(rng.randint(len(steps)))]
            ni, nj = best[i] + d, best[j] - d
            if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12:
                continue
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < best_c - 1e-13:
                best_c = nc
                improved = True
            else:
                best[i], best[j] = ni - d, nj + d
        if not improved:
            break
    return best, best_c
