```python
def _c5(h, dx):

def _c5_grad(h, dx):
    """Gradient of C5 w.r.t. h (subgradient based on argmax lag)."""
    c = _corr(h)
    k_max = np.argmax(c)
    # Gradient of c[k_max] w.r.t. h
    n = len(h)
    shift = k_max - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        # Term from h[i] * (1-h[i+shift])
        j = i + shift
        if 0 <= j < n:
            grad[i] += (1.0 - h[j])
        # Term from h[j] * (1-h[i]) where j+shift = i, i.e., j = i - shift
        j2 = i - shift
        if 0 <= j2 < n:
            grad[i] -= h[j2]
    return grad * dx
```

```python
def _project(h, n):
    """Project h onto {0≤x[i]≤1, sum(x)=n/2}."""
    x = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Water filling: add/subtract uniformly from non-saturated points
    if diff > 0:
        # Add: increase non-1 points
        mask = x < 1.0 - 1e-12
        count = mask.sum()
        if count == 0:
            return x
        add = min(diff / count, 1.0 - x[mask])
        x[mask] += min(diff/count, 1.0-np.max(x[mask]))
        # If still not enough, iterate (shouldn't need to)
        diff2 = target - x.sum()
        if abs(diff2) > 1e-12:
            # More careful redistribution
            for _ in range(10):
                mask = x < 1.0 - 1e-12
                if not mask.any():
                    break
                add = (target - x.sum()) / mask.sum()
                if add <= 0:
                    break
                x[mask] = np.minimum(x[mask] + add, 1.0)
    else:
        # Subtract: decrease non-0 points
        mask = x > 1e-12
        while (target - x.sum()) > 1e-12 and mask.any():
            sub = (-diff) / mask.sum()
            x[mask] = np.maximum(x[mask] - sub, 0.0)
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
            mask = x > 1e-12
    return x
```

Let me also add a "gradient pair" move: look at the gradient, find the pair (i,j) where moving i up and j down (or vice versa) most reduces the objective.

Actually, for a pair move (i up by d, j down by d), the change in objective is approximately:
Δ ≈ -d * (grad[i] - grad[j]) (for the LP approximation)

So the best pair move is to move from the index with most positive grad to the one with most negative grad.

```python
def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t}]
    def _obj(x):
        c = _corr(x)
        m = c.max()
        s = np.exp(np.clip(alpha * (c - m), -500, 500))
        return (m + np.log(s.sum()) / alpha) * dx
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = _project(res.x, n)
        return h_new
    except Exception:
        return _project(h, n)
```
