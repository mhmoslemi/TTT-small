def _sgrad_fast(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    # For each lag L, the correlation term is:
    # L >= 0: c_L = sum_{i=0}^{n-1-L} x[i]*(1-x[i+L])
    # L < 0: c_L = sum_{i=|L|}^{n-1} x[i]*(1-x[i+L]) = sum_{i=|L|}^{n-1} x[i]*(1-x[i-|L|])
    # 
    # The full correlation from numpy:
    # result[k] where k=0..2n-2
    # lag = k - (n-1)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n-1)
        if lag >= 0:

            idx1 = np.arange(n - lag)
            idx2 = idx1 + lag
            g[idx1] += wk * (1.0 - x[idx2])
            g[idx2] -= wk * x[idx1]
        else:

            # c = sum_{i=L}^{n-1} x[i]*(1-x[i-L])

            idx1 = idx2 - L
            g[idx2] += wk * (1.0 - x[idx1])
            g[idx1] -= wk * x[idx2]
    return g

                diff -= a
            else:
