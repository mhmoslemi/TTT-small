def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_smax(h, n, alpha):
    c = _corr(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -400, 400)
    p = np.exp(s)
    p /= p.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wk = p[m_idx]
        if wk < 1e-8:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            lo1, hi1 = k, n - 1
            if hi1 >= lo1:
                idx1 = np.arange(lo1, hi1 + 1)
                grad[idx1] += wk * (1.0 - h[idx1 - k])
            lo2, hi2 = 0, n - 1 - k
            if hi2 >= lo2:
                idx2 = np.arange(lo2, hi2 + 1)
                grad[idx2] -= wk * h[idx2 + k]
        else:
            kp = -k
            lo1, hi1 = 0, n - 1 - kp
            if hi1 >= lo1:
                idx1 = np.arange(lo1, hi1 + 1)
                grad[idx1] += wk * (1.0 - h[idx1 + kp])
            lo2, hi2 = kp, n - 1
            if hi2 >= lo2:
                idx2 = np.arange(lo2, hi2 + 1)
                grad[idx2] -= wk * h[idx2 - kp]
    return grad

def _slsqp_grad(h, n, dx, alpha, maxiter):
    def f(x):
        c = _corr(x)
        mx = c.max()
        s = np.clip(alpha * (c - mx), -400, 400)
        return (mx + np.log(s.sum() + 1e-300) / alpha) * dx
    def g(x):
        return _grad_smax(x, n, alpha) * dx
    def con_f(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h, method='SLSQP', bounds=bounds, constraints=cons,
                     jac=g, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _lfgsb_opt(h, n, dx, alpha, maxiter, pen_w):
    def f(x):
        c = _corr(x)
        mx = c.max()
        s = np.clip(alpha * (c - mx), -400, 400)
        pen = (x.sum() - n / 2.0) ** 2 * pen_w
        return (mx + np.log(s.sum() + 1e-300) / alpha) * dx + pen
    def g(x):
        gr = _grad_smax(x, n, alpha) * dx
        gr = gr + 2.0 * (x.sum() - n / 2.0) * pen_w * np.ones(n)
        return gr
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _local_search(h, n, dx, rng, deadline, max_rounds=8000):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = [0.001, 0.003, 0.007, 0.014, 0.028, 0.056,
             -0.001, -0.003, -0.007, -0.014, -0.028, -0.056]
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ntrials = 50 + 30 * min(stall // 15, 4)
        for _ in range(ntrials):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12:
                    continue
                if nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-14:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = oi, oj
                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return best, best_c5

def _penalty_push(h, n, dx, target_c5, maxiter):
    thresh = target_c5 / dx
    alpha = 600
    def f(x):
        c = _corr(x)
        excess = np.maximum(0.0, c - thresh)
        pen_sum = np.sum(excess * excess) * 50.0
        con_pen = (x.sum() - n / 2.0) ** 2 * 1e5
        mx = c.max()
        s = np.clip(alpha * (c - mx), -400, 400)
        sm = (mx + np.log(s.sum() + 1e-300) / alpha) * dx
        return sm + pen_sum + con_pen
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _sa_search(h, n, dx, rng, deadline, max_rounds=20000):
    cur = h.copy()
    cur_c5 = _c5(cur, dx)
    best = cur.copy()
    best_c5 = cur_c5
    T = 0.002
    T_min = 1e-6
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        T *= 0.9997
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        while j == i:
            j = int(rng.randint(n))
        d = 0.01 * (rng.random() * 2.0 - 1.0)
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        oi, oj = cur[i], cur[j]
        cur[i], cur[j] = ni, nj
        nc = _c5(cur, dx)
        if nc < cur_c5 or rng.random() < np.exp(-(nc - cur_c5) / max(T, 1e-9)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[i], cur[j] = oi, oj
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def elig(margin):
        return time.time() < t0 + budget_s - margin

    # Phase 1: SLSQP with analytical gradient, sweep alpha
    for a in [25, 60, 150, 400, 1000, 2500]:
        if not elig(220):
            break
        hc = _slsqp_grad(best_h.copy(), n, dx, a, 500)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc

    # Phase 2: L-BFGS-B with penalty (different optimization landscape)
    for a in [150, 400, 1000, 2000]:
        if not elig(180):
            break
        hc = _lfgsb_opt(best_h.copy(), n, dx, a, 600, 1e5)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc

    # Phase 3: Local search (fine-grained)
    if elig(120):
        dl = t0 + budget_s - 120
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl, 6000)

    # Phase 4: Alternate SLSQP and L-BFGS-B
    for _ in range(3):
        if not elig(160):
            break
        for a in [400, 1000, 2500, 5000]:
            if not elig(60):
                break
            hc = _slsqp_grad(best_h.copy(), n, dx, a, 350)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc
        if elig(80):
            hc = _lfgsb_opt(best_h.copy(), n, dx, 800, 400, 1e5)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc

    # Phase 5: Simulated annealing for global exploration
    if elig(120):
        dl = t0 + budget_s - 120
        sa_h, sa_c5 = _sa_search(best_h, n, dx, rng, dl, 15000)
        if sa_c5 < best_c5:
            best_c5, best_h = sa_c5, sa_h
            # Refine after SA
            if elig(60):
                hc = _slsqp_grad(best_h.copy(), n, dx, 2000, 300)
                nc = _c5(hc, dx)
                if nc < best_c5:
                    best_c5, best_h = nc, hc

    # Phase 6: Penalty push toward target
    if elig(80):
        target = best_c5 - 0.0003
        hc = _penalty_push(best_h.copy(), n, dx, target, 500)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc
        if elig(40):
            hc = _slsqp_grad(best_h.copy(), n, dx, 3000, 300)
            nc = _c5(hc, dx)
            if nc < best_c5:
                best_c5, best_h = nc, hc

    # Phase 7: Final local search
    if elig(30):
        dl = t0 + budget_s - 5
        best_h, best_c5 = _local_search(best_h, n, dx, rng, dl, 5000)

    # Final SLSQP polish
    if elig(8):
        hc = _slsqp.grad_polish(best_h.copy(), n, dx, 5000, 150) if hasattr(_slsqp, 'grad_polish') else _slsqp_grad(best_h.copy(), n, dx, 5000, 150)
        nc = _c5(hc, dx)
        if nc < best_c5:
            best_c5, best_h = nc, hc

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
