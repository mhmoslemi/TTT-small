def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lse(corr, alpha):
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return m + float(np.sum(s)) / alpha

def _obj_lse(x, alpha, dx):
    corr = _corr(x)
    return _lse(corr, alpha) * dx

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _obj_lse(x, alpha, dx):
    c = _corr(x)
    m = float(np.max(c))
    s = np.exp(np.clip(alpha * (c - m), -50, 50))
    return (m + float(np.sum(s)) / alpha) * dx

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _local_pair_search(h, dx, t_end, rng, deltas=None, j_limit=15, fine=False):
    n = len(h)

    best_c5 = _c5(best_h, dx)
    if deltas is None:

        else:

                              -0.02, -0.05, -0.1, -0.2, -0.35])
    no_impr = 0
    while time.time() < t_end and no_impr < (3 if fine else 5):
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            vi = best_h[i]
            # Precompute j ordering once per i
            for d in deltas:
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                comp = -d
                if comp > 0:
                    oj = np.argsort(best_h)
                else:
                    oj = np.argsort(-best_h)
                for j in oj[:j_limit]:
                    if j == i:
                        continue
                    nj = best_h[j] + comp
                    if nj < 0 or nj > 1:
                        continue
                    best_h[i], best_h[j] = ni, nj
                    c = _c5(best_h, dx)

                        break

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _obj_lse(x, alpha, dx):
    c = _corr(x)
    m = float(np.max(c))
    s = np.exp(np.clip(alpha * (c - m), -50, 50))
    return (m + float(np.sum(s)) / alpha) * dx


def _eq_con(x, target):
    return float(np.sum(x) - target)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sa_pair(h, dx, t_end, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    T0, Tend = 0.004, 0.00001
    total = 4000000
    for it in range(total):

            break
        frac = it / total
        T = T0 * (Tend / T0) ** frac

    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)
