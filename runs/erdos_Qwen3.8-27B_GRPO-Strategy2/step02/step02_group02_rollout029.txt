def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _smax(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(np.exp(s).sum()) / a

def _sgrad(x, a):
