```python
        for idx in order:

def _run_lse_alpha(h0, alpha, n, dx, iters, lr, t_end):

    best_c5 = _c5(h, dx)
```

```python
def _gradient_lse(h, w, n):
    """Compute gradient of LSE(corr(h)) w.r.t. h.
    w: weights from LSE (length 2n-1), sum to 1, non-negative.
    Returns gradient of length n."""
    n2 = 2 * n - 1
    # g[p] = 1 - h[p] - sum_k w[k] * h[p + k - (n-1)]
    # The last sum is a cross-correlation
    # For each p: sum over k where 0 <= p + k - (n-1) < n
    g = np.ones(n) - h.copy()
    # Compute sum_k w[k] * h[p + k - (n-1)] for each p
    # This is np.correlate-like: for lag j = k-(n-1), sum_j w[j+(n-1)] * h[p+j]
    # = (w_centered @ h reversed) ... let me just use the direct formula
    # For efficiency with n=70, direct computation is fine
    for p in range(n):
        # k ranges from max(0-(n-1), -(p)) to min((2n-2)-(n-1), n-1-p)
        # i.e., j = k-(n-1) ranges from max(-n+1, -p) to min(n-1, n-1-p)
        j_min = max(-(n-1), -p)
        j_max = min(n-1, n-1-p)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            # h indices: p + j, must be in [0, n-1]
            h_idx = p + js
            valid = (h_idx >= 0) & (h_idx < n)
            if valid.any():
                g[p] -= float(np.dot(w[js + (n-1)][valid], h[h_idx[valid]]))
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(hp.sum())
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - hp[idx])
                hp[idx] += add
                diff -= add
            else:
                sub = min(-diff, hp[idx])
                hp[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return hp

def _lse_weights(corr, alpha):
    m = float(np.max(corr))
    e = np.exp(alpha * (corr - m))
    return e / float(e.sum())

def _grad_lse(h, w, n):
    g = 1.0 - h.copy()
    nc = n - 1
    for p in range(n):
        j_min = max(-nc, -p)
        j_max = min(nc, nc - p)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            h_idx = p + js
            valid = (h_idx >= 0) & (h_idx < n)
            if valid.any():
                g[p] -= float(np.dot(w[js + nc][valid], h[h_idx[valid]]))
    return g

def _pgd(h0, alpha, n, dx, t_end, iters_max, lr0):
    h = h0.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    target = n / 2.0
    stagnant = 0
    for it in range(iters_max):
        if time.time() > t_end:
            break
        corr = _corr(h)
        w = _lse_weights(corr, alpha)
        g = _grad_lse(h, w, n)
        # Adaptive step size
        lr = lr0 / (1.0 + 0.01 * stagnant)
        # Try a few step sizes
        best_local_h = h.copy()
        best_local_c5 = _c5(h, dx)
        for lr_try in [lr, lr * 0.5, lr * 2.0]:
            h_try = _project(h - lr_try * g, target)
            c5_try = _c5(h_try, dx)
            if c5_try < best_local_c5:
                best_local_c5 = c5_try
                best_local_h = h_try
        h = best_local_h
        c5_now = _c5(h, dx)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                # Restart with perturbation
                rng = np.random.RandomState(it)
                pert = rng.normal(0, 0.05, n)
                h = _project(h + pert, target)
                stagnant = 0
    return best_h, best_c5

def _local_search(h0, n, dx, t_end):
    h = h0.copy()
    best_c5 = _c5(h, dx)
    target = n / 2.0
    no_improve = 0
    max_no_improve = 50
    step_sizes = np.array([0.01, 0.005, 0.002, 0.001, 0.0005])
    while time.time() < t_end and no_improve < max_no_improve:
        improved = False
        corr = _corr(h)
        # Find active k's (top 10)
        top_k = np.argsort(corr)[-10:]
        for i in range(n):
            if time.time() > t_end:
                break
            vi = h[i]
            for ss in step_sizes:
                for sign in [1, -1]:
                    d = sign * ss
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    need = -d
                    # Best j to compensate
                    if need > 0:
                        j_candidates = np.argsort(h)[:20]
                    else:
                        j_candidates = np.argsort(-h)[:20]
                    for j in j_candidates:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0 or nj > 1:
                            continue
                        h[i] = ni
                        h[j] = nj
                        c5_try = _c5(h, dx)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            improved = True
                            break
                        h[i] = vi
                        h[j] = nj - need
                    if improved:
                        break
                if improved:
                    break
            if improved:
                no_improve = 0
        if not improved:
            no_improve += 1
    return h, best_c5

        return (m + float(np.log(e.sum())) / alpha) * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - float(hp.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - hp)
        else:
            order = np.argsort(hp)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - hp[idx])
                hp[idx] += add
                diff -= add
            else:
                sub = min(-diff, hp[idx])
                hp[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return hp


def _lse_weights(corr, alpha):
    m = float(np.max(corr))
    e = np.exp(alpha * (corr - m))
    s = float(e.sum())
    if s < 1e-300:
        idx = np.argmin(np.abs(corr - m))
        w = np.zeros(len(corr))
        w[idx] = 1.0
        return w
    return e / s


def _grad_lse(h, w, n):
    nc = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        j = k - nc
        i_min = max(0, j)
        i_max = min(n - 1, nc + j)
        if i_max >= i_min:
            g[i_min:i_max + 1] += wk * (1.0 - h[i_min - j:i_max - j + 1])
        p_min = max(0, -j)
        p_max = min(n - 1, nc - j)
        if p_max >= p_min:
            g[p_min:p_max + 1] -= wk * h[p_min + j:p_max + j + 1]
    return g


def _pgd(h0, alpha, n, dx, t_end, lr0, iters_max):
    h = h0.copy()
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    target = n / 2.0
    stagnant = 0
    for it in range(iters_max):
        if time.time() > t_end:
            break
        corr = _corr(h)
        w = _lse_weights(corr, alpha)
        g = _grad_lse(h, w, n)
        lr = lr0 * (0.5 ** (stagnant // 100))
        h_new = _project(h - lr * g, target)
        c5_try = _c5(h_new, dx)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
        else:
            if c5_try < _c5(h, dx) + 1e-14:
                h = h_new
            stagnant += 1
            if stagnant > 300:
                rng = np.random.RandomState(it + 7)
                h = _project(h + rng.normal(0, 0.04, n), target)
                stagnant = 0
    return best_h, best_c5

    best_c5 = _c5(h, dx)
    target = n / 2.0
    no_improve = 0

def _slsqp_obj(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    e = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + float(np.log(e.sum())) / alpha) * (2.0 / n)
```
