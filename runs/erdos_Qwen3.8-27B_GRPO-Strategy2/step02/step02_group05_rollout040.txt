        d = rng.uniform(-0.15, 0.15)
        ni, nj = cur[i] + d, cur[j] - d
        if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
            d *= (0.3 * (1 - abs(ni if ni < 0 or ni > 1 else nj if nj < 0 or nj > 1 else 0)))
            ni, nj = cur[i] + d, cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
        cur[i], cur[j] = ni, nj
        nc5 = _c5(cur, dx)
        delta = nc5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = nc5
            if nc5 < best_c5 - 1e-13:
                best = cur.copy()
                best_c5 = nc5
                st = 0
            else:
                st += 1
        else:
            cur[i], cur[j] = ni - d, nj + d
        if st > 2000:
            break
        T *= cool
    return best, best_c5

def _structured_init(n, rng, mode):
    t = np.linspace(0, 2.0, n, endpoint=False)
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * t * 2) + 0.1 * np.sin(2 * np.pi * t * 5)
    elif mode == 1:
        h = 0.5 + 0.4 * np.cos(2 * np.pi * t * 3)
    elif mode == 2:
        h = 0.5 + 0.3 * np.sign(np.sin(2 * np.pi * t * 2)) * np.exp(-3 * (t - 1.0)**2)
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * t * 1.5 + rng.randn())
    h += 0.05 * rng.randn(n)
    return _repair(h, n)

def _mirror_init(h, n):
    hm = np.zeros(n)
    hm[:n//2] = 1.0 - h[n-1:n//2-1:-1]
    hm[n//2:] = h[:n//2]
    return _repair(hm, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(h0, dx)

    def el(m):
        return time.time() < t0 + budget_s - m

    def try_update(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 1: SLSQP from multiple starts with increasing alpha
    starts = [h0, _mirror_init(h0, n)]
    for m in [0, 1, 2, 3]:
        starts.append(_structured_init(n, np.random.RandomState(seed + 100 + m), m))

    for si in range(len(starts)):
        if not el(350):
            break
        hc = starts[si].copy()
        c5 = _c5(hc, dx)
        for a in [30, 80, 200, 500, 1200]:
            if not el(350):
                break
            hc = _slsqp(hc, n, dx, a, 400, 1e-15)
            nc5 = _c5(hc, dx)
            if nc5 < c5:
                c5 = nc5
        try_update(hc, c5)

    # Phase 2: Gradient-based local search
    if el(300):
        dl = t0 + budget_s - 290
        best_h, best_c5 = _grad_local_search(best_h, n, dx, dl, 4000)

    # Phase 3: Simulated annealing
    if el(250):
        dl = t0 + budget_s - 240
        best_h, best_c5 = _sa_local(best_h, n, dx, rng, dl, 200000)

    # Phase 4: SLSQP + gradient + SA iterations
    alphas = [500, 1000, 2000, 4000, 8000]
    it = 0
    while el(120) and it < 20:
        sigma = [0.15, 0.3, 0.5, 0.1, 0.25, 0.4][it % 6]
        hp = _repair(best_h + sigma * rng.randn(n), n)
        a = alphas[it % len(alphas)]
        hc = _slsqp(hp, n, dx, a, 500, 1e-16)
        c5 = _c5(hc, dx)
        if c5 < best_c5 + 0.001 and el(130):
            dl = t0 + budget_s - 125
            hc, c5g = _grad_local_search(hc, n, dx, dl, 2000)
            if c5g < best_c5:
                best_h, best_c5 = hc.copy(), c5g
            elif c5g < best_c5 + 0.0005 and el(60):
                dl2 = t0 + budget_s - 55
                hc2, c5s = _sa_local(hc, n, dx, rng, dl2, 100000)
                if c5s < best_c5:
                    best_h, best_c5 = hc2.copy(), c5s
            elif c5 < best_c5:
                best_h, best_c5 = hc.copy(), c5
        elif c5 < best_c5:
            best_h, best_c5 = hc.copy(), c5
        it += 1

    # Phase 5: Final high-alpha SLSQP polish
    if el(30):
        for a in [3000, 6000, 12000, 20000]:
            if not el(20):
                break
            hc = _slsqp(best_h, n, dx, a, 800, 1e-17)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc.copy(), c5

    # Phase 6: Final fine gradient + SA
    if el(15):
        dl = t0 + budget_s - 10
        best_h, best_c5 = _grad_local_search(best_h, n, dx, dl, 2000)
    if el(8):
        dl = t0 + budget_s - 5
        best_h, best_c5 = _sa_local(best_h, n, dx, rng, dl, 50000)

    # Phase 7: Final SLSQP at extreme alpha
    if el(5):
        hc = _slsqp(best_h, n, dx, 50000, 1000, 1e-18)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc.copy(), c5

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
