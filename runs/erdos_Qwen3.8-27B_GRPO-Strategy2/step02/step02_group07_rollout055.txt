        vals = c + mid * dc
        mx = vals.max()
        active = np.where(vals >= mx - 1e-15)[0]
        slopes = dc[active]
        if slopes.min() >= 0:
            lo = mid
        elif slopes.max() <= 0:
            hi = mid
        else:
            break  # 0 in subgradient
