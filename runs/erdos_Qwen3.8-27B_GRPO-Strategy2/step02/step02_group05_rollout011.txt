import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_s(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _con_fun(x, target):
    return x.sum() - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    try:
        res = minimize(_obj_s, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': _con_fun, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _nm(h, n, dx, alpha, timeout_s):
    start = time.time()
    target = n / 2.0
    def obj(x):
        if time.time() - start > timeout_s:
            return 1e10
        hx = np.clip(x, 0.0, 1.0)
        pen = 80.0 * (hx.sum() - target) ** 2
        return _smooth_max(_corr(hx), alpha) * dx + pen
    try:
        res = minimize(obj, h, method='Nelder-Mead',
                       options={'maxiter': 4000, 'xatol': 1e-10, 'fatol': 1e-13, 'maxfev': 10000})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _local_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps_pos = [0.002, 0.004, 0.008, 0.015, 0.03, 0.06, 0.1, 0.15]
    steps_neg = [-0.002, -0.004, -0.008, -0.015, -0.03, -0.06, -0.1, -0.15]
    steps = steps_pos + steps_neg
    stall = 0
    best_stall_steps = steps
    for outer in range(5000):
        if time.time() > deadline:
            break
        improved = False
        ntrials = 120
        for _ in range(ntrials):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in best_stall_steps:
                ni = best[i] + d
                nj = best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = oi, oj
                if improved:
                    break
        if improved:
            stall = 0
            best_stall_steps = steps
        else:
            stall += 1
            if stall > 15:
                best_stall_steps = steps_pos + steps_neg
            if stall > 60:
                break
    return best, best_c5

def _block_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    stall = 0
    for outer in range(2000):
        if time.time() > deadline:
            break
        improved = False
        bs = rng.randint(2, 5)
        idxs = rng.choice(n, bs, replace=False)
        for _ in range(30):
            trial = best.copy()
            delta_vals = (rng.rand(bs) - 0.5) * 0.1
            for k, idx in enumerate(idxs):
                trial[idx] += delta_vals[k]
            trial = _repair(trial, n)
            nc = _c5(trial, dx)
            if nc < best_c5 - 1e-12:
                best = trial
                best_c5 = nc
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best, best_c5

def _binary_push(h, n, dx, rng, threshold=0.15):
    best = h.copy()
    best_c5 = _c5(best, dx)
    ambiguous = np.where((best > threshold) & (best < 1.0 - threshold))[0]
    if len(ambiguous) == 0:
        return best, best_c5
    for idx in rng.permutation(ambiguous):
        for target_val in [0.0, 1.0]:
            trial = best.copy()
            delta = target_val + 0.1 * rng.randn() - best[idx]
            delta = np.clip(delta, -0.5, 0.5)
            trial[idx] = np.clip(best[idx] + delta, 0, 1)
            if delta > 0:
                others = np.where(best[1:] > 0.05)[0] + 1
            else:
                others = np.where(best[1:] < 0.95)[0] + 1
            if len(others) == 0:
                continue
            j = others[int(rng.randint(len(others)))]
            trial[j] = np.clip(best[j] - delta / max(1, len(others) - 1), 0, 1)
            trial = _repair(trial, n)
            nc = _c5(trial, dx)
            if nc < best_c5 - 1e-12:
                best = trial
                best_c5 = nc
    return best, best_c5
