```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _soft_obj(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _soft_grad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - h[i + lag])
                g[i + lag] -= wk * h[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - h[i - L])
                g[i - L] -= wk * h[i]
    return g

def _fista(h0, alpha, target, iters=3000):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    # Lipschitz estimate
    L = 4.0 * alpha * (2 * n) + 4.0
    step = 1.0 / L
    for it in range(iters):
        gz = _soft_grad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        # Backtracking
        cnt = 0
        while cnt < 10:
            fn = _soft_obj(xn, alpha)
            quad = _soft_obj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        if it % 50 == 0:
            c = _c5(x, dx)
            if c < bc:
                bc = c
                bx = x.copy()
    c = _c5(x, dx)
    if c < bc:
        bc = c
        bx = x.copy()
    return bx, bc

def _cco(h0, dx, n, iters=200, target=None):
    """Cyclic coordinate optimization with exact 1D minimization."""
    if target is None:
        target = n / 2.0
    h = h0.copy()
    best_c = _c5(h, dx)
    for _ in range(iters):
        improved = False
        for i in range(n):
            old_val = h[i]
            
            def obj_1d(xi):
                h[i] = xi
                c = np.correlate(h, 1.0 - h, mode='full')
                return float(c.max())
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            h[i] = res.x
            if abs(h[i] - old_val) > 1e-10:
                improved = True
        h = _proj(h, target)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
        if not improved:
            break
    return h, best_c

def _pair_cd(h, dx, n, target, iters=50000, rng=None):
    """Pair coordinate descent with various step sizes."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c = _c5(h, dx)
    dvals = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05,
             -0.001, -0.002, -0.005, -0.01, -0.02, -0.05]
    for _ in range(iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi0, hj0 = h[i], h[j]
        for d in dvals:
            ni, nj = hi0 + d, hj0 - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i] = ni
            h[j] = nj
            c = _c5(h, dx)
            if c < best_c - 1e-14:
                best_c = c
            else:
                h[i] = hi0
                h[j] = hj0
                break
    return h, best_c

def _rem(t0, budget):
    return budget - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)
    
    h = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh = h.copy()
    bc = _c5(bh, dx)
    
    # Phase 1: FISTA with multiple alphas
    for a in [50, 100, 200, 500, 1000, 2000, 5000]:
        if _rem(t0, budget_s) < 40:
            break
        h, c = _fista(bh, a, tgt, iters=3000)

    #   g_i += w[j] * (1-h_{i-l}) for i >= l  [h_{i-l} term]
    #   g_i -= w[j] * h_{i-l} for i >= l  ... wait let me redo

    # = [0<=i-l<=n-1]*(1-h_{i-l}) - [0<=i+l<=n-1]*h_{i+l}
    #
    # For l >= 0:
    #   [0<=i-l<=n-1] means i >= l (and i-l < n, i.e., i < n+l, which is always true for i<n)
    #   So for i >= l: (1-h_{i-l})
    #   [0<=i+l<=n-1] means i <= n-1-l
    #   So for i <= n-1-l: -h_{i+l}

    #   [0<=i+l<=n-1] = [0<=i-L<=n-1] means i >= L
    #   So for i >= L: (1-h_{i-L})
    #   [0<=i+l<=n-1]... wait, l < 0, i+l = i-L
    #   [0<=i+l<=n-1] for the second term: i+l valid means 0<=i-L... no
    #

    # c_j = sum_n h_{n+l}(1-h_n) for valid n.
    # 
    # ∂c_j/∂h_i:
    # h_i appears as h_{n+l} when n+l=i, i.e., n=i-l. Valid if 0<=i-l<=n-1.
    #   Contribution: (1-h_{i-l})

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _soft_obj(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _soft_grad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        l = j - (n - 1)
        if l >= 0:
            idx1 = np.arange(l, n)
            g[idx1] += wj * (1.0 - h[idx1 - l])
            idx2 = np.arange(0, n - l)
            g[idx2] -= wj * h[idx2 + l]
        else:
            L = -l
            idx1 = np.arange(0, n - L)
            g[idx1] += wj * (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            g[idx2] -= wj * h[idx2 - L]
    return g

def _fista(h0, alpha, target, iters):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n * 2 + 4.0)
    for it in range(iters):
        gz = _soft_grad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 10:
            fn = _soft_obj(xn, alpha)
            quad = _soft_obj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        if it % 100 == 0 or it == iters - 1:
            c = _c5(x, dx)
            if c < bc:
                bc = c
                bx = x.copy()
    return bx, bc

def _cco_obj(h, i, xi):
    old = h[i]
    h[i] = xi
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(c.max())
    h[i] = old
    return val

def _cco(h0, dx, n, iters, target):
    h = h0.copy()
    best_c = _c5(h, dx)
    for _ in range(iters):
        prev_max_change = 0.0
        for i in range(n):
            res = minimize_scalar(_cco_obj, bounds=(0.0, 1.0), method='bounded',
                                  args=(h, i), options={'xatol': 1e-10})
            old = h[i]
            h[i] = res.x
            prev_max_change = max(prev_max_change, abs(res.x - old))
        h = _proj(h, target)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
        if prev_max_change < 1e-9:
            break
    return h, best_c

def _rem(t0, budget):
    return budget - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values
