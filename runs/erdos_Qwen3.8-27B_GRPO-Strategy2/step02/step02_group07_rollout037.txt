import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, n):
    target = n / 2.0
    lo = h.min() - target
    hi = h.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _so_g(x, alpha):
    c = _corr(x)
    n = len(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    f = m + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for jj in range(2 * n - 1):
        if w[jj] < 1e-10:
            continue
        lag = jj - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += w[jj] * (1.0 - x[lo1 - lag:hi1 - lag + 1])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= w[jj] * x[lo2 + lag:hi2 + lag + 1]
    return f, g

def _nesterov(h, n, dx, alpha, maxit, t0, tlim):
    x = h.copy()
    xp = h.copy()
    bx = h.copy()
    bf = _c5(h, dx)
    mu = 0.0
    lr = 2.0 / (alpha + 20.0)
    for it in range(maxit):
        if time.time() - t0 > tlim:
            break
        tk = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) * 0.5
        y = x + (mu / tk) * (x - xp)
        xp = x
        mu = tk
        yp = _project(y, n)
        f, g = _so_g(yp, alpha)
        gsq = np.dot(g, g)
        if gsq < 1e-28:
            break
        step = lr
        for _ in range(60):
            xn = _project(yp - step * g, n)
            fn, _ = _so_g(xn, alpha)
            if fn <= f - 1e-4 * step * gsq or step < 1e-20:
                break
            step *= 0.5
        x = xn
        c = _c5(x, dx)
        if c < bf:
            bf = c
            bx = x.copy()
        lr = max(step * 1.05, lr * 0.98)
    return bx, bf

def _cd(h, n, dx, t0, tlim, mxp):
    hc = _c5(h, dx)
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    idx = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
    for _ in range(mxp):
        if time.time() - t0 > tlim:
            break
        imp = False
        for i, j in idx:
            hs = h[i] + h[j]
            lo = max(0.0, hs - 1.0)
            hi = min(1.0, hs)
            if hi - lo < 1e-12:
                continue
            oi, oj = h[i], h[j]
            a, b = lo + 1e-14, hi - 1e-14
            x1 = b - gr * (b - a)
            x2 = a + gr * (b - a)
            h[i], h[j] = x1, hs - x1
            f1 = _c5(h, dx)
            h[i], h[j] = x2, hs - x2
            f2 = _c5(h, dx)
            for _ in range(28):
                if f1 <= f2:
                    b = x2
                    x2, f2 = x1, f1
                    x1 = b - gr * (b - a)
                    h[i], h[j] = x1, hs - x1
                    f1 = _c5(h, dx)

                    a = x1
                    x1, f1 = x2, f2
                    x2 = a + gr * (b - a)
                    h[i], h[j] = x2, hs - x2
                    f2 = _c5(h, dx)
            to = (a + b) * 0.5
            h[i], h[j] = to, hs - to
            co = _c5(h, dx)

                hc = co
                imp = True
            else:

def _slsqp(h, n, alpha, mx, ft):
    target = n / 2.0
    bd = [(0.0, 1.0)] * n
    cn = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        r = minimize(lambda x: _so_g(x, alpha)[0], h,
                     jac=lambda x: _so_g(x, alpha)[1],
                     method='SLSQP', bounds=bd, constraints=cn,
                     options={'maxiter': mx, 'ftol': ft})
        if r.x is not None and np.all(np.isfinite(r.x)):
            return _project(r.x, n)
    except Exception:
        pass
    return h.copy()

            bc5, bh = c, h.copy()

    # Phase 6: Multi-restart with structured perturbations
    sigs = [0.04, 0.08, 0.12, 0.2, 0.06, 0.15, 0.3, 0.03]
    for si in range(len(sigs)):
        if el() > budget_s - 80:
            break
        hp = bh + sigs[si] * rng.randn(n)
        hp = _project(hp, n)
        h = _slsqp(hp, n, 500, 300, 1e-12)
        h2, c2 = _cd(h, n, dx, t0, budget_s - 60, 400)
        c = _c5(h2, dx)
        if c < bc5:
            bc5, bh = c, h2.copy()
            if el() < budget_s - 50:
                h3, c3 = _nesterov(h2, n, dx, 3000, 3000, t0, budget_s - 40)
                if c3 < bc5:
                    bc5, bh = c3, h3.copy()

    # Phase 7: Final intensive CD
    if el() < budget_s - 30:
        h, c = _cd(bh.copy(), n, dx, t0, budget_s - 15, 5000)
        if c < bc5:
            bc5, bh = c, h.copy()

    # Phase 8: Final SLSQP
    if el() < budget_s - 5:
        h = _slsqp(bh, n, 15000, 400, 1e-15)
        c = _c5(h, dx)
        if c < bc5:
            bc5, bh = c, h.copy()

    bh = _project(bh, n)
    bc5 = _c5(bh, dx)
    return (bh, bc5, n)
