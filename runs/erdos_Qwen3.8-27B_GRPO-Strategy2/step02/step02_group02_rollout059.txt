            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - x[k]); x[k] += a; diff -= a
            else:
                a = min(-diff, x[k]); x[k] -= a; diff += a
    return x

def _grad_corr(h, p):
    n = len(h)
    gc = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        k = lag + (n - 1)
        if lag >= 0:
            if p <= n - 1 - lag:
                gc[k] += 1.0 - h[p + lag]
            if p >= lag:
                gc[k] -= h[p - lag]
        else:
            L = -lag
            if p >= L:
                gc[k] += 1.0 - h[p - L]
            if p <= n - 1 - L:
                gc[k] -= h[p + L]
    return gc

def _sobj(x, a, tgt):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a + 1e8 * (float(x.sum()) - tgt) ** 2

def _sgrad(x, a, tgt):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    g += 2e8 * (float(x.sum()) - tgt)
    return g

def _fista(h0, alpha, tgt, iters=3000):
    n = len(h0)
    dx = 2.0 / n
    x, z = h0.copy(), h0.copy()
    t, bx = 1.0, h0.copy()
    bc = _c5(h0, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for _ in range(iters):
        gz = _sgrad(z, alpha, tgt)
        y = z - step * gz
        xn = _proj(y, tgt)
        for _ in range(6):
            fn = _sobj(xn, alpha, tgt)
            qd = _sobj(z, alpha, tgt) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= qd + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, tgt)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), tgt)
        x, t = xn, tn
        c = _c5(x, dx)
        if c < bc:
            bc, bx = c, x.copy()
    return bx, bc

def _cd_exact(h, dx, n, sweeps, rng):
    bc = _c5(h, dx)
    h = h.copy()
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    n_pairs = len(pairs)
    for _ in range(sweeps):
        imp = 0
        rng.shuffle(pairs)
        for idx in range(n_pairs):
            i, j = pairs[idx]
            hi0, hj0 = h[i], h[j]
            lo_d = max(-hi0, hj0 - 1.0) + 1e-15
            hi_d = min(1.0 - hi0, hj0) - 1e-15
            if lo_d >= hi_d:
                continue
            c = np.correlate(h, 1.0 - h, mode='full')
            dg = _grad_corr(h, i) - _grad_corr(h, j)
            best_d, best_c = 0.0, float(c.max()) * dx
            ds = np.linspace(lo_d, hi_d, 100)
            for d in ds:
                if abs(d) < 1e-14:
                    continue
                cm = float((c + d * dg).max()) * dx
                if cm < best_c - 1e-14:
                    best_c, best_d = cm, d
            if best_d != 0.0:
                span = abs(hi_d - lo_d) / 99.0
                for _ in range(30):
                    found = False
                    for frac in [0.5, -0.5, 0.25, -0.25, 0.125, -0.125]:
                        d2 = best_d + frac * span * 2.0
                        if d2 < lo_d or d2 > hi_d:
                            continue
                        cm = float((c + d2 * dg).max()) * dx
                        if cm < best_c - 1e-14:
                            best_c, best_d, found = cm, d2, True
                    if not found:
                        break
            if best_d != 0.0 and best_c < bc:
                h[i] = hi0 + best_d
                h[j] = hj0 - best_d
                bc = best_c
                imp += 1
        if imp == 0:
            break
    return h, bc

def _cd_steps(h, dx, n, sweeps, steps, rng):
    bc = _c5(h, dx)
    h = h.copy()
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    n_pairs = len(pairs)
    for _ in range(sweeps):
        imp = 0
        rng.shuffle(pairs)
        for idx in range(n_pairs):
            i, j = pairs[idx]
            hi0, hj0 = h[i], h[j]
            found = False
            for d in steps:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc, found = c, True
                    break
            if not found:
                h[i], h[j] = hi0, hj0
            else:
                imp += 1
        if imp == 0:
            break
    return h, bc

def _lbfgs(h0, alpha, tgt, maxiter=500):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        return _sobj(x, alpha, tgt)

    def grad(x):
        return _sgrad(x, alpha, tgt)

    try:
        r = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        hf = _proj(r.x, tgt)
        return hf, _c5(hf, 2.0 / n)
    except Exception:
        return h0, _c5(h0, 2.0 / n)
