import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha, target):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slope(v, i, j, n):
    h = 1.0 - v
    s = np.zeros(2 * n - 1)
    s[i:i+n] += v[n-1::-1]
    s[n-1-i:2*n-1-i] -= h
    s[j:j+n] -= v[n-1::-1]
    s[n-1-j:2*n-1-j] += h
    return s

def _opt_delta(C, s, dlo, dhi):
    if dhi - dlo < 1e-12:
        return 0.0
    dets = np.linspace(dlo, dhi, 25)
    mx = (C[None, :] + s[None, :] * dets[:, None]).max(axis=1)
    bk = int(np.argmin(mx))
    lo = dets[max(0, bk - 1)]
    hi = dets[min(24, bk + 1)]
    dets2 = np.linspace(lo, hi, 30)
    mx2 = (C[None, :] + s[None, :] * dets2[:, None]).max(axis=1)
    return float(dets2[int(np.argmin(mx2))])

def _cd_pass(h, n, t_start, t_limit):
    dx = 2.0 / n
    v = 1.0 - h
    C = np.correlate(h, v, mode='full')
    cm = float(C.max() * dx)
    improved = False
    for i in range(n):
        if time.time() - t_start > t_limit:
            break
        for j in range(i + 1, n):
            dlo = max(-h[i], h[j] - 1.0)
            dhi = min(1.0 - h[i], h[j])
            if dhi - dlo < 1e-10:
                continue
            s = _slope(v, i, j, n)
            bd = _opt_delta(C, s, dlo, dhi)
            if abs(bd) > 1e-10:
                nm = float((C + s * bd).max() * dx)
                if nm < cm - 1e-13:
                    h[i] += bd
                    h[j] -= bd
                    v[i] -= bd
                    v[j] += bd
                    C += s * bd
                    cm = nm
                    improved = True
    return h, cm, improved

def _cd_loop(h, n, t_start, t_limit, max_passes=5000):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() - t_start > t_limit:
            break
        h, c5, imp = _cd_pass(h, n, t_start, t_limit)
        if not imp:
            break
    return h, c5

def _run_slsqp(h, n, bounds, target, alpha, maxiter, ftol):
    try:
        res = minimize(_smooth_obj, h, (alpha, target), method='SLSQP',
                       bounds=bounds,

                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
