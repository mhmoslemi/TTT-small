import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'args': (target,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

Let me also consider: instead of evaluating on a uniform grid, I can find the exact minimum of the max-of-lines problem in 1D. Since the function g(δ) = max_p (base_corr[p] + δ * a[p]) is convex piecewise linear with at most n_corr pieces (≤ 139), the minimum is at:
- An endpoint, or
- A breakpoint where the active line changes

The breakpoints are at δ values where two lines cross: (base_corr[p1] - base_corr[p2]) / (a[p2] - a[p1])

There are O(n_corr²) ≈ 19,000 breakpoints. Finding the minimum among them:
- Filter to those in [delta_lo, delta_hi]
- Evaluate g at each valid breakpoint + endpoints
- Take the minimum

But evaluating g at each point is O(n_corr) = O(n). Total: O(n_corr² * n_corr) = O(n³) per pair. For n=70: 343,000 per pair. For 4900 pairs: 1.7B. Too slow.

Alternative: Use the convexity. Binary search on the subgradient:
- g'(δ) at the minimum contains 0
- g'(δ) = a[p*] where p* is the active (max) line
- If multiple lines are tied at max, g'(δ) = [min(a[p]: p active), max(a[p]: p active)]

Binary search:
- Evaluate at midpoint, find active lines
- If all active slopes > 0: min is to the left → hi = mid
- If all active slopes < 0: min is to the right → lo = mid
- If mixed: the min is between the crossing of the most-negative-slope active line and the most-positive-slope active line... 

Actually for a convex piecewise linear function, once we find a point where the subgradient contains 0, that's the minimum. And the subgradient contains 0 iff there exist active lines with both positive and negative slopes (or a zero-slope line).

The binary search will converge to a breakpoint (or interval containing 0 in the subgradient). After ~20 iterations (precision 1/2^20 of the interval), we can evaluate g at that point.

Let me implement this:

```python
def _min_convex_pl(base, a, lo, hi):
    # Minimize max_p (base[p] + a[p]*t) for t in [lo, hi]
    # base, a are 1D arrays of same length
    # Returns (optimal_t, optimal_value)
    
    def g(t):
        return np.max(base + t * a)
    
    # Binary search for where subgradient contains 0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = base + mid * a
        mx = vals.max()
        active = vals >= mx - 1e-14
        min_s = a[active].min()
        max_s = a[active].max()
        if min_s >= -1e-14:
            hi = mid
        elif max_s <= 1e-14:
            lo = mid
        else:
            # Found the minimum (subgradient contains 0)
            return mid, g(mid)
    mid = (lo + hi) / 2
    return mid, g(mid)
```

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _target_fun(x, target):
    return x.sum() - target

def _effect(idx, h, n):
    """Compute the effect vector: how full_corr changes when h[idx] increases by 1."""
    n_corr = 2 * n - 1
    m = np.arange(-n + 1, n)
    # p = m + n - 1
    eff = np.zeros(n_corr)
    # First factor: h[idx] * (1-h[idx+m]) valid when 0 <= idx+m < n
    im = idx + m
    mask1 = (im >= 0) & (im < n)
    eff[mask1] += (1.0 - h[im[mask1]])
    # Second factor: h[idx-m] * (1-h[idx]) valid when 0 <= idx-m < n
    jm = idx - m
    mask2 = (jm >= 0) & (jm < n)
    eff[mask2] -= h[jm[mask2]]
    return eff

def _min_convex_pl(base, a, lo, hi):
    """Minimize max_p(base[p] + a[p]*t) for t in [lo, hi]. Returns (t, value)."""
    def g(t):
        return np.max(base + t * a)
    
    g_lo = g(lo)
    g_hi = g(hi)
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = base + mid * a
        mx = vals.max()
        active = vals >= mx - 1e-12
        min_s = a[active].min()
        max_s = a[active].max()
        if min_s >= -1e-12:
            hi = mid
        elif max_s <= 1e-12:
            lo = mid
        else:
            break
    best_t = (lo + hi) / 2.0
    best_v = g(best_t)
    if g_lo < best_v:
        best_t, best_v = lo, g_lo
    if g_hi < best_v:
        best_t, best_v = hi, g_hi
    return best_t, best_v

def _fast_cd_pass(h, n, dx):
    """One pass of fast coordinate descent. Returns (h, c5, improved)."""
    n_corr = 2 * n - 1
    base_corr = _corr(h).copy()
    hi_c5 = float(np.max(base_corr) * dx)
    improved = False
    
    order = np.random.permutation(n)  # randomize to avoid systematic bias
    
    for ii in range(n):
        i = order[ii]
        for j in range(n):
            if j == i:
                continue
            hsum = h[i] + h[j]
            lo = max(0.0, hsum - 1.0)
            hi = min(1.0, hsum)
            if hi - lo < 1e-10:
                continue
            
            # Effect vectors
            eff_i = _effect(i, h, n)
            eff_j = _effect(j, h, n)
            a = eff_i - eff_j
            
            delta_lo = lo - h[i]
            delta_hi = hi - h[i]
            
            best_d, best_v = _min_convex_pl(base_corr, a, delta_lo, delta_hi)
            
            if best_v < hi_c5 / dx - 1e-12:
                h[i] += best_d
                h[j] -= best_d
                base_corr += best_d * a
                hi_c5 = best_v * dx
                improved = True
    
    return h, hi_c5, improved

def _slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _target_fun, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_loop(h, n, dx, t_start, t_end, seed_rng):
    """Run CD passes until time limit."""
    while time.time() < t_end:
        h, c5, imp = _fast_cd_pass(h, n, dx)
        if not imp:
            break
    return h, _c5(h, dx)
