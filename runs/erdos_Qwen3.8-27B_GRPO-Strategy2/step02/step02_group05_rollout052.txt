            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = obj_ev(c1)
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = obj_ev(c2)
    return (t_lo + t_hi) / 2.0

def _coord_descent(h, n, dx, deadline, max_passes=15):
    best = h.copy()
    best_c5 = _c5(best, dx)
    target = n / 2.0
    prev_c5 = best_c5
    for p in range(max_passes):
        if time.time() > deadline:
            break
        for d in range(1, n):
            for i in range(n - d):
                j = i + d
                c = target - (best.sum() - best[i] - best[j])
                t_lo = max(0.0, c - 1.0)
                t_hi = min(1.0, c)
                if t_hi - t_lo < 1e-14:
                    continue
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                tt_lo, tt_hi = t_lo, t_hi
                c1 = tt_hi - gr * (tt_hi - tt_lo)
                c2 = tt_lo + gr * (tt_hi - tt_lo)
                old_i, old_j = best[i], best[j]
                best[i], best[j] = c1, c - c1
                f1 = float(np.max(_corr(best)))
                best[i], best[j] = c2, c - c2
                f2 = float(np.max(_corr(best)))
                best[i], best[j] = old_i, old_j
                for _ in range(28):
                    if f1 < f2:
                        tt_hi = c2; c2 = c1; f2 = f1
                        c1 = tt_hi - gr * (tt_hi - tt_lo)
                        best[i], best[j] = c1, c - c1
                        f1 = float(np.max(_corr(best)))
                        best[i], best[j] = old_i, old_j
                    else:
                        tt_lo = c1; c1 = c2; f1 = f2
                        c2 = tt_lo + gr * (tt_hi - tt_lo)
                        best[i], best[j] = c2, c - c2
                        f2 = float(np.max(_corr(best)))
                        best[i], best[j] = old_i, old_j
                t_opt = (tt_lo + tt_hi) / 2.0
                best[i] = t_opt
                best[j] = c - t_opt
        nc = _c5(best, dx)
        if nc < best_c5 - 1e-13:
            best_c5 = nc
            prev_c5 = nc
        else:
            if p > 2:
                break
    return best, best_c5

def _local_search_fine(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    target = n / 2.0
    steps = np.array([0.001, 0.003, 0.007, 0.015, 0.03,
                      -0.001, -0.003, -0.007, -0.015, -0.03])
    stall = 0
    round_num = 0
    while time.time() < deadline and stall < 40:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        c = target - (best.sum() - best[i] - best[j])
        for d in steps:
            ni, nj = best[i] + d, best[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < best_c5 - 1e-14:
                best_c5 = nc
                stall = 0
            else:
                best[i], best[j] = oi, oj
        round_num += 1
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(best_h, dx)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP from multiple starts
    starts = [h_init]
    starts.append(_repair(h_init + 0.08 * rng.randn(n), n))
    h_bin = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h_bin[pos] = 1.0
    starts.append(h_bin)
    for start in starts:
        if remaining() < 150:
            break
        hc = start.copy()
        for alpha in [25, 60, 120, 250, 500]:
            if remaining() < 150:
                break
            hc = _slsqp(hc, n, dx, alpha, 250, 1e-14)
            c = _c5(hc, dx)
            if c < best_c5:
                best_c5 = c
                best_h = hc.copy()

    # Phase 2: 2-coordinate descent
    if remaining() > 100:
        deadline = t0 + budget_s - 80
        best_h, best_c5 = _coord_descent(best_h, n, dx, deadline, max_passes=12)

    # Phase 3: SLSQP polish
    if remaining() > 60:
        for alpha in [400, 800, 1500]:
            if remaining() < 60:
                break
            hc = _slsqp(best_h, n, dx, alpha, 350, 1e-15)
            c = _c5(hc, dx)
            if c < best_c5:
                best_c5 = c
                best_h = hc.copy()

    # Phase 4: More coordinate descent
    if remaining() > 50:
        deadline = t0 + budget_s - 35
        best_h, best_c5 = _coord_descent(best_h, n, dx, deadline, max_passes=10)

    # Phase 5: Restart with perturbation + coord descent
    for sigma in [0.15, 0.3, 0.5, 0.2, 0.4, 0.6]:
        if remaining() < 40:
            break
        hp = _repair(best_h + sigma * rng.randn(n), n)
        hc = _slsqp(hp, n, dx, 200, 200, 1e-14)
        deadline = t0 + budget_s - 30
        hc, c = _coord_descent(hc, n, dx, deadline, max_passes=5)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()

    # Phase 6: Fine local search
    if remaining() > 20:
        deadline = t0 + budget_s - 8
        best_h, best_c5 = _local_search_fine(best_h, n, dx, rng, deadline)

    # Phase 7: Final SLSQP + coord descent
    if remaining() > 10:
        hc = _slsqp(best_h, n, dx, 3000, 500, 1e-16)
        c = _c5(hc, dx)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()
        if remaining() > 5:
            best_h, best_c5 = _coord_descent(best_h, n, dx, t0 + budget_s - 2, max_passes=5)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
