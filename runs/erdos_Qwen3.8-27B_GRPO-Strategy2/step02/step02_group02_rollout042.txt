def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -500.0, 500.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += wk * (1.0 - x[i + lag])
                g[i + lag] -= wk * x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk * (1.0 - x[i - L])
                g[i - L] -= wk * x[i]
    return g

def _pen_obj(x, a, lam, tgt):
    return _sobj(x, a) + lam * (float(np.sum(x)) - tgt) ** 2

def _pen_grad(x, a, lam, tgt):
    return _sgrad(x, a) + 2.0 * lam * (float(np.sum(x)) - tgt) * np.ones_like(x)

def _cd(h, dx, n, n_sweeps, rng, t_rem):
    bc = _c5(h, dx)
    h = h.copy()
    dvals = [0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03,
             0.05, 0.08, 0.12, 0.18,
             -0.001, -0.002, -0.003, -0.005, -0.008, -0.01, -0.015, -0.02,
             -0.03, -0.05, -0.08, -0.12, -0.18]
    for sw in range(n_sweeps):
        if time.time() - t_rem < 5:
            break
        imp = False
        for _ in range(n * 12):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            for d in dvals:
                ni, nj = hi0 + d, hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = hi0, hj0
        if not imp:
            break
    return h, bc

def _fista_fast(h0, a, target, iters, t_rem):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * a * n + 4.0)
    for it in range(iters):
        if time.time() - t_rem > 0:
            break
        gz = _sgrad(z, a)
        y = z - step * gz
        xn = _proj(y, target)
        cnt = 0
        while cnt < 6:
            fn = _sobj(xn, a)
            quad = _sobj(z, a) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if fn <= quad + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
            cnt += 1
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _lbfgs_solve(h_s, a, lam, tgt, bnds, maxiter=800):
    try:
        res = minimize(_pen_obj, h_s, args=(a, lam, tgt),
                       method='L-BFGS-B', bounds=bnds, jac=_pen_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-16, 'maxls': 60})
        return _proj(res.x, tgt)
    except Exception:
        return h_s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    rng = np.random.RandomState(seed)
    bnds = [(0.0, 1.0)] * n

    h0 = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh = h0.copy()
    bc = _c5(bh, dx)

    # Phase 1: L-BFGS-B warm-started with increasing alpha
    alphas_p1 = [80, 200, 500, 1200, 3000, 8000, 20000, 50000]
    for a in alphas_p1:
        if time.time() - t0 > budget_s - 80:
            break
        lam = 1e4 if a < 5000 else 1e5
        bf = _lbfgs_solve(bh, a, lam, tgt, bnds, maxiter=1000)
        cf = _c5(bf, dx)
        if cf < bc:
            bc = cf
            bh = bf

    # Phase 2: FISTA for smoother landscape at moderate alpha
    if time.time() - t0 < budget_s - 60:
        h, c = _fista_fast(bh, 3000, tgt, 3000, time.time() - t0)
        if c < bc:
            bc = c
            bh = h
        h, c = _fista_fast(bh, 15000, tgt, 2000, time.time() - t0)
        if c < bc:
            bc = c
            bh = h

    # Phase 3: Intensive coordinate descent
    if time.time() - t0 < budget_s - 30:
        bh, bc = _cd(bh, dx, n, 400, rng, time.time() - t0)

    # Phase 4: More L-BFGS-B at very high alpha
    if time.time() - t0 < budget_s - 50:
        for a in [50000, 100000, 200000]:
            if time.time() - t0 > budget_s - 30:
                break
            bf = _lbfgs_solve(bh, a, 1e6, tgt, bnds, maxiter=1200)
            cf = _c5(bf, dx)
            if cf < bc:
                bc = cf
                bh = bf

    # Phase 5: Random restarts with L-BFGS + CD
    sigmas = [0.04, 0.07, 0.1, 0.15, 0.2, 0.3, 0.5, 0.08, 0.12, 0.25, 0.4, 0.6]
    for s in sigmas:
        if time.time() - t0 > budget_s - 40:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        bf = _lbfgs_solve(hp, 5000, 1e5, tgt, bnds, maxiter=600)
        cf = _c5(bf, dx)
        if cf < bc:
            bc = cf
            bh = bf
            if time.time() - t0 < budget_s - 20:
                bh, bc = _cd(bh, dx, n, 150, rng, time.time() - t0)
        elif cf < bc + 0.001:
            bf2 = _lbfgs_solve(hp, 20000, 1e5, tgt, bnds, maxiter=500)
            cf2 = _c5(bf2, dx)
            if cf2 < bc:
                bc = cf2
                bh = bf2

    # Phase 6: Structured restarts (Fourier modes, step functions)
    if time.time() - t0 < budget_s - 60:
        for freq in range(1, 10):
            for amp in [0.1, 0.2, 0.35]:
                if time.time() - t0 > budget_s - 30:
                    break
                ph = rng.rand() * 2.0 * np.pi
                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + ph)
                hs = _proj(hs, tgt)
                bf = _lbfgs_solve(hs, 2000, 1e4, tgt, bnds, maxiter=400)
                cf = _c5(bf, dx)
                if cf < bc:
                    bc = cf
                    bh = bf
                    bh, bc = _cd(bh, dx, n, 80, rng, time.time() - t0)

    # Phase 7: Step-function restarts
    if time.time() - t0 < budget_s - 40:
        for nblocks in [2, 3, 4, 5, 7]:
            if time.time() - t0 > budget_s - 20:
                break
            edges = np.sort(rng.choice(np.arange(1, n), size=nblocks - 1, replace=False))
            vals = rng.uniform(0.2, 0.8, size=nblocks)
            hs = np.zeros(n)
            prev = 0
            for bi, e in enumerate(edges):
                hs[prev:e] = vals[bi]
                prev = e
            hs[prev:] = vals[-1]
            hs = _proj(hs, tgt)
            bf = _lbfgs_solve(hs, 3000, 1e4, tgt, bnds, maxiter=500)
            cf = _c5(bf, dx)
            if cf < bc:
                bc = cf
                bh = bf
                bh, bc = _cd(bh, dx, n, 100, rng, time.time() - t0)

    # Phase 8: Final L-BFGS at extreme alpha + final CD
    if time.time() - t0 < budget_s - 25:
        for a in [100000, 300000, 500000]:
            if time.time() - t0 > budget_s - 10:
                break
            bf = _lbfgs_solve(bh, a, 1e6, tgt, bnds, maxiter=1500)
            cf = _c5(bf, dx)
            if cf < bc:
                bc = cf
                bh = bf
    if time.time() - t0 < budget_s - 8:
        bh, bc = _cd(bh, dx, n, 1500, rng, time.time() - t0)

    # Final feasibility
    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
