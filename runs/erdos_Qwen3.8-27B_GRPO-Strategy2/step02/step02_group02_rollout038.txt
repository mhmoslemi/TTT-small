    for _ in range(80):

        if float(np.clip(y - mid, 0, 1).sum()) > target:
            lo = mid
        else:

    x = np.clip(y - 0.5 * (lo + hi), 0, 1)

    if abs(d) > 1e-13:
        idx = np.argsort(1 - x) if d > 0 else np.argsort(x)
        for k in idx:
            if d > 0:
                a = min(d, 1 - x[k]); x[k] += a; d -= a
            else:
                a = min(-d, x[k]); x[k] -= a; d += a
            if abs(d) < 1e-13:
