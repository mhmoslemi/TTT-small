    m = float(c.max())

    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n-1)
        if lag >= 0:
            for i in range(n-lag):
                g[i] += wk*(1.0-x[i+lag])
                g[i+lag] -= wk*x[i]
        else:
            L = -lag
            for i in range(L, n):
                g[i] += wk*(1.0-x[i-L])
                g[i-L] -= wk*x[i]
    return g

def _pair_search(h, i, j, dx):
    """Find optimal d for h[i]+=d, h[j]-=d using 1D search."""
    lo = max(-h[i], h[j]-1.0)
    hi = min(1.0-h[i], 1.0-h[j])
    if lo >= hi:
        return 0.0, _c5(h, dx)
    def f(d):
        h[i] += d; h[j] -= d
        val = _c5(h, dx)
        h[i] -= d; h[j] += d
        return val
    # Quick scan then refine
    best_d = 0.0; best_f = f(0.0)
    for d in np.linspace(lo, hi, 11):
        v = f(d)
        if v < best_f:
            best_f = v; best_d = d
    # Refine with golden section
    # (skip for speed, the 11-point scan is usually good enough)
    return best_d, best_f

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(y - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a

    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    s = np.clip(a * (c - m), -400.0, 400.0)
    e = np.exp(s)
    w = e / float(np.sum(e))
    g = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-13:

        if lag >= 0:

                g[i + lag] -= wk * x[i]
