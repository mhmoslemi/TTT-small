    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr(h)
    return float(np.max(c) * dx)

def _g_vec(h, p, n):
    k = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _exact_sweep(c, g, d_min, d_max):
    """Find d in [d_min, d_max] minimizing max_k (c[k] + d*g[k])"""
    if d_max - d_min < 1e-15:
        return d_min
    # Bisection on optimal value t
    # Lower bound for t
    min_at = np.where(g > 0, c + d_min*g, c + d_max*g)
    lo_t = np.max(min_at)
    # Upper bound for t
    hi_t = np.max(np.maximum(c + d_min*g, c + d_max*g))

        # pos: d <= (mid_t - c[k])/g[k] for k in pos

        hi_d = d_max

        if np.any(~neg & ~pos):

def _sweep_pass(h, n, dx, max_rounds=50):
    """Exact coordinate-pair sweeps"""
    corr = _corr(h)
    improved = False
    for round_ in range(max_rounds):
        any_change = False
        for i in range(n):
            for j in range(i+1, n):
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_min > d_max - 1e-12:
                    continue
                g = _g_vec(h, i, n) - _g_vec(h, j, n)
                d_best = _exact_sweep(corr, g, d_min, d_max)
                if abs(d_best) > 1e-12:

                    new_max = np.max(corr + d_best * g)
                    if new_max < cur_max - 1e-14:
                        h[i] += d_best
                        h[j] -= d_best
                        corr += d_best * g
                        any_change = True
        if not any_change:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _exact_sweep(c, g, d_min, d_max):
    if d_max - d_min < 1e-15:
        return d_min
    min_at = np.where(g > 0, c + d_min * g, c + d_max * g)
    lo_t = float(np.max(min_at))
    hi_t = float(np.max(np.maximum(c + d_min * g, c + d_max * g)))
    if hi_t - lo_t < 1e-14:
        return (d_min + d_max) * 0.5
    neg_m = g < -1e-15
    pos_m = g > 1e-15
    zero_m = ~neg_m & ~pos_m
    for _ in range(35):
        mid_t = (lo_t + hi_t) * 0.5
        if zero_m.any() and np.max(c[zero_m]) > mid_t:
            lo_t = mid_t
            continue
        lo_d = d_min
        if neg_m.any():
            lo_d = max(lo_d, float(np.max((mid_t - c[neg_m]) / g[neg_m])))
        hi_d = d_max
        if pos_m.any():
            hi_d = min(hi_d, float(np.min((mid_t - c[pos_m]) / g[pos_m])))
        if lo_d <= hi_d:
            hi_t = mid_t
        else:
            lo_t = mid_t
    t = hi_t
    lo_d = d_min
    if neg_m.any():
        lo_d = max(lo_d, float(np.max((t - c[neg_m]) / g[neg_m])))
    hi_d = d_max
    if pos_m.any():
        hi_d = min(hi_d, float(np.min((t - c[pos_m]) / g[pos_m])))
    return (lo_d + hi_d) * 0.5

def _sweep_pass(h, n, dx, max_rounds=40):
    corr = _corr(h)
    for r in range(max_rounds):
        any_change = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_min > d_max - 1e-12:
                    continue
                g = _g_vec(h, i, n) - _g_vec(h, j, n)
                if np.max(np.abs(g)) < 1e-14:
                    continue
                d_best = _exact_sweep(corr, g, d_min, d_max)
                if abs(d_best) > 1e-13:
                    new_max = float(np.max(corr + d_best * g))
                    cur_max = float(np.max(corr))
                    if new_max < cur_max - 1e-14:
                        h[i] += d_best
                        h[j] -= d_best
                        corr += d_best * g
                        any_change = True
        if not any_change:
            break
    return h, corr

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, bounds, target, alpha, maxiter):
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
