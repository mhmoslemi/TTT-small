    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h))) * dx

def _obj_lse(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + np.sum(s) / alpha) * (2.0 / n)

def _obj_topk(x, k, n):
    corr = _corr(x)
    if k >= len(corr):
        return float(corr.sum()) * (2.0 / n)
    idx = np.argpartition(corr, -k)[-k:]
    return float(corr[idx].sum()) * (2.0 / n)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

                              (_obj_topk, (8, n), 'top8'), (_obj_lse, (800, n), 'lse800'),
                              (_obj_lse, (2000, n), 'lse2000')]:
        if time.time() > p1_end:
            break
        try:
            res = minimize(obj_fn, best_h, args=args, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Simulated annealing with diverse moves (45% budget)
    p2_end = t_start + min(budget_s * 0.45, 500)
    if time.time() < p2_end:
        sa_h, sa_c5 = _sa_search(best_h, dx, p2_end, rng)
        if sa_c5 < best_c5:
            best_c5, best_h = sa_c5, sa_h

    # Phase 3: Thorough pairwise local search (30% budget)
    p3_end = t_start + min(budget_s * 0.75, 830)
    if time.time() < p3_end:
        pw_h, pw_c5 = _pairwise_search(best_h, dx, p3_end, rng)
        if pw_c5 < best_c5:
            best_c5, best_h = pw_c5, pw_h
        # Second round of pairwise with finer steps
        if time.time() < p3_end - 20:
            pw_h2, pw_c52 = _pairwise_search(best_h, dx, p3_end, rng)
            if pw_c52 < best_c5:
                best_c5, best_h = pw_c52, pw_h2

    # Phase 4: SLSQP restarts from perturbed solutions (10% budget)
    p4_end = t_start + min(budget_s * 0.88, 970)
    if time.time() < p4_end:
        for attempt in range(6):
            if time.time() > p4_end:
                break
            pert = best_h.copy()
            mask = rng.rand(n) < 0.3
            pert[mask] += (rng.rand(mask.sum()) - 0.5) * 0.03
            pert = _repair(pert, n)
            for alpha in [1000, 3000]:
                if time.time() > p4_end:
                    break
                try:
                    res = minimize(_obj_lse, pert, args=(alpha, n), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 200, 'ftol': 1e-16})
                    hc = _repair(res.x, n)
                    c5 = _c5(hc, dx)
                    if c5 < best_c5:
                        best_c5, best_h = c5, hc
                        pert = best_h.copy()
                except Exception:
                    pass

    # Phase 5: Ultra-fine final polish
    p5_end = t_start + budget_s - 5
    if time.time() < p5_end:
        h = best_h.copy()
        local_best = best_c5
        fine_deltas = [0.0003, 0.0005, 0.001, -0.0003, -0.0005, -0.001]
        for _pass in range(20):
            if time.time() > p5_end:
                break
            improved = False
            for i in range(n):
                if time.time() > p5_end:
                    break
                vi = h[i]
                for d in fine_deltas:
                    ni = vi + d
                    if ni < 0 or ni > 1:
                        continue
                    # find best j to compensate
                    remaining = -d
                    if remaining > 0:
                        order = np.argsort(h)
                    else:
                        order = np.argsort(-h)
                    found = False
                    for j in order[:15]:
                        if j == i:
                            continue
                        nj = h[j] + remaining
                        if nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < local_best - 1e-14:
                            local_best = c
                            improved = True
                            found = True
                            break
                        h[i], h[j] = vi, h[j] - remaining
                    if found:
                        break
                # if not improved at i, h[i] unchanged
            if not improved:
                break
        if local_best < best_c5:
            best_c5, best_h = local_best, h

    # Final SLSQP polish
    if time.time() < t_start + budget_s - 2:
        try:
            res = minimize(_obj_lse, best_h, args=(5000, n), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
