import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _penalty_obj(x, alpha, dx, lam, target):
    c = _corr(x)
    sm = _smooth_max(c, alpha) * dx
    pen = lam * (np.sum(x) - target) ** 2
    return sm + pen

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
