                best_c5, best_h = c5_new, h_new

    # Phase 4: Projected gradient on best
    tl = budget_s - (time.time() - t0) - 80
    if tl > 10:
        h, c5 = proj_grad(best_h, n, time_limit=tl, t0=t0, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 5: Fine polish
    tl = budget_s - (time.time() - t0) - 30
    if tl > 5:
        h, c5 = fine_polish(best_h, n, rng, time_limit=tl, t0=t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 6: Final SLP polish
    if time.time() - t0 < budget_s - 8:
        h, c5 = slp_optimize(best_h, n, max_iters=8,
                             time_limit=budget_s - (time.time() - t0) - 3, t0=t0)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 7: Ultra-fine coordinate descent
    if time.time() - t0 < budget_s - 3:
        h_cur = best_h.copy()
        tiny = [0.0002, 0.0005, 0.001, 0.002, -0.0002, -0.0005, -0.001, -0.002]
        for _ in range(5000):
            if time.time() - t0 > budget_s - 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            hi, hj = h_cur[i], h_cur[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_cur[i], h_cur[j] = ni, nj
                c = compute_c5(h_cur, dx)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = hi, hj

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
