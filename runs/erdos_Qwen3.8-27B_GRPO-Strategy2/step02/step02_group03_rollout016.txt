import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(500):
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if abs(diff) < 1e-13:
            break
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _smmax(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _slsqp(h, n, alpha, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return _smmax(_corr(x), alpha)
    def jac(x):
        eps = 1e-5
        g = np.zeros(n)
        for i in range(n):
            xp = x.copy(); xp[i] = min(1.0, x[i] + eps)
            xm = x.copy(); xm[i] = max(0.0, x[i] - eps)
            d = xp[i] - xm[i]
            g[i] = (obj(xp) - obj(xm)) / d if d > 1e-14 else 0.0
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: ones}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol}, jac=jac)
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pair_desc(h, n, dx, budget_end, rng, deltas, max_rounds):
    h = h.copy()
    best = _c5(h, dx)
    for r in range(max_rounds):
        if time.time() > budget_end:
            break
        improved = False
        perm = rng.permutation(n)
        for ii in range(n):
            if time.time() > budget_end:
                break
            i = int(perm[ii])
            hi = h[i]
            for jj in range(ii + 1, n):
                j = int(perm[jj])
                hj = h[j]
                for d in deltas:
                    for sgn in (1, -1):
                        ni = hi + sgn * d
                        nj = hj - sgn * d
                        if ni < -1e-14 or ni > 1.0 + 1e-14 or nj < -1e-14 or nj > 1.0 + 1e-14:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best - 1e-14:
                            best = c; improved = True; hi = h[i]; hj = h[j]
                        else:
                            h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best

def _ternary_pair(h, n, dx, budget_end, rng):
    h = h.copy()
    best = _c5(h, dx)
    perm = rng.permutation(n)
    for ii in range(n):
        if time.time() > budget_end:
            break
        i = int(perm[ii])
        for jj in range(ii + 1, n):
            if time.time() > budget_end:
                break
            j = int(perm[jj])
            hi, hj = h[i], h[j]
            lo = max(hi - 1.0, -hj)
            hi_ = min(hi, 1.0 - hj)
            if hi_ <= lo + 1e-12:
                continue
            a, b = lo, hi_
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(22):
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                h[i], h[j] = hi - c1, hj + c1
                f1 = _c5(h, dx)
                h[i], h[j] = hi - c2, hj + c2
                f2 = _c5(h, dx)
                if f1 < f2:
                    b = c2
                else:
                    a = c1
            d_opt = 0.5 * (a + b)
            h[i], h[j] = hi - d_opt, hj + d_opt
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
                hi, hj = h[i], h[j]
            else:
                h[i], h[j] = hi, hj
