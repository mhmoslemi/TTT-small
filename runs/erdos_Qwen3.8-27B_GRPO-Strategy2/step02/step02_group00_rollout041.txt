        for i in range(n):
            if time.time() > t_end:
                return h, best
            vi = h[i]
            for j in range(i + 1, n):
                vj = h[j]
                for d in steps:
                    ni = vi + d
                    nj = vj - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c = float(np.max(_corr(h))) * dx
                    if c < best - 1e-14:
                        best = c
                        improved = True
                    else:
                        h[i] = vi
                        h[j] = vj
        if not improved:
            break
    return h, best


def _nudge_dir(h, n):
    corr = _corr(h)
    thresh = corr.max() * 0.999
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    eps = 1e-5
    grad = np.zeros(n)
    for idx in active[:5]:
        hi = h.copy()
        hi[idx] += eps
        corr_i = _corr(hi)[idx]
        hi[idx] -= eps
        hi2 = h.copy()
        hi2[idx] -= eps
        if hi2[idx] < 0:
            lo_eps = h[idx] / 2.0 if h[idx] > 0 else eps
            hi2[idx] = h[idx] - lo_eps
            corr_i2 = _corr(hi2)[idx]
            g = (corr_i - corr_i2) / (2 * lo_eps) if lo_eps > 0 else 0.0
        else:
            corr_i2 = _corr(hi2)[idx]
            g = (corr_i - corr_i2) / (2 * eps)
        grad[idx] += g
    return -grad


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [80, 250, 800, 2500, 6000, 12000]:
        if time.time() > t0 + budget_s * 0.15:
            break
        hc, c5 = _slsqp_pass(best_h, n, alpha, 350, t_end)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5

    # Phase 2: Coarse pairwise search
    s1 = np.array([0.08, 0.04, 0.02, 0.01, -0.08, -0.04, -0.02, -0.01])
    p2_end = t0 + budget_s * 0.30
    if time.time() < p2_end:
        best_h, best_c5 = _pairwise(best_h, n, dx, s1, 40, p2_end)

    # Phase 3: SLSQP refinement
    for alpha in [5000, 15000, 30000]:
        if time.time() > t0 + budget_s * 0.40:
            break
        hc, c5 = _slsqp_pass(best_h, n, alpha, 500, t_end)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5

    # Phase 4: Fine pairwise
    s2 = np.array([0.008, 0.004, 0.002, 0.001, -0.008, -0.004, -0.002, -0.001])
    p4_end = t0 + budget_s * 0.55
    if time.time() < p4_end:
        best_h, best_c5 = _pairwise(best_h, n, dx, s2, 60, p4_end)

    # Phase 5: Nudge + SLSQP cycles for escaping local optima
    p5_end = t0 + budget_s * 0.82
    nudge_trials = 0
    while time.time() < p5_end and nudge_trials < 20:
        nudge_trials += 1
        grad = _nudge_dir(best_h, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad = grad / gn
        hp = best_h + 0.015 * grad
        hp = _repair(hp, n)
        # Add small random perturbation
        for _ in range(10):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = (rng.rand() - 0.5) * 0.01
            ni, nj = hp[i] + d, hp[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                hp[i], hp[j] = ni, nj
        hp = _repair(hp, n)
        hc, c5 = _slsqp_pass(hp, n, 8000, 300, t_end)
        if c5 < best_c5 - 1e-15:
            best_h, best_c5 = hc, c5
            # Refine the new best
            hc2, c52 = _slsqp_pass(best_h, n, 20000, 400, t_end)
            if c52 < best_c5:
                best_h, best_c5 = hc2, c52
            s3 = np.array([0.003, 0.001, 0.0005, -0.003, -0.001, -0.0005])
            best_h, best_c5 = _pairwise(best_h, n, dx, s3, 25, p5_end)

    # Phase 6: Final ultra-fine pairwise
    s4 = np.array([0.001, 0.0005, 0.0002, -0.001, -0.0005, -0.0002])
    p6_end = t0 + budget_s - 15
    if time.time() < p6_end:
        best_h, best_c5 = _pairwise(best_h, n, dx, s4, 40, p6_end)

    # Final SLSQP polish
    if time.time() < t_end - 5:
        hc, c5 = _slsqp_pass(best_h, n, 25000, 500, t_end)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
