    corr = np.correlate(h, 1-h, 'full')  # length 2n-1
    dx = 2.0/n
    # for the objective, it's max_k corr[k] * dx, so gradient is same as max_k corr[k]
    m = np.max(corr)
    shifted = alpha * (corr - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)  # shift
        if weights[k] < 1e-15:

        # valid i: max(0,-s) <= i <= min(n-1, n-1-s)

        return g * dx

                       bounds=bounds, constraints=cons,

            h_out = _project(h_out, target)
        return h_out

def _slsqp_opt(h, n, dx, alpha, maxiter, ftol, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        if time.time() > deadline:
            raise InterruptedError()
        return _smooth_max_val(_corr(x), alpha) * dx
    def grad_obj(x):
        _, g = _smooth_max_grad(x, n, alpha)
        return g * dx
    try:
        res = minimize(obj, h, jac=grad_obj, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, rng, freq):
    t = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * freq * t) + 0.1 * np.sin(4 * np.pi * freq * t + 0.5)
    h += rng.randn(n) * 0.05
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0

    h0 = _repair(np.array(initial_h_values, dtype=float), n0)
    best_h, best_c5 = h0.copy(), _c5(h0, dx0)

    def rem():
        return t0 + budget_s - time.time()

    # Phase 1: PGD at n0
    for alpha in [40, 80, 200, 500]:
        if rem() < 60:
            break
        h_new, c5_new = _pgd(best_h.copy(), n0, dx0, alpha, 300, t0 + budget_s - 60)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    # Phase 2: Greedy + random LS at n0
    if rem() > 50:
        h_g, c5_g = _greedy_search(best_h, n0, dx0, 200, t0 + budget_s - 50, 500)
        if c5_g < best_c5:
            best_c5, best_h = c5_g, h_g
    if rem() > 40:
        h_r, c5_r = _random_ls(best_h, n0, dx0, rng, t0 + budget_s - 40, 300)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r

    # Phase 3: Upsample to higher n
    for n_try in [140, 210]:
        if rem() < 250:
            break
        dx_try = 2.0 / n_try
        h_try = _upsample(best_h, n_try)

        # PGD at higher n
        for alpha in [50, 100, 250, 500]:
            if rem() < 200:
                break
            h_try, c5_try = _pgd(h_try, n_try, dx_try, alpha, 400, t0 + budget_s - 200)

        # Greedy search
        if rem() > 120:
            h_try, c5_try = _greedy_search(h_try, n_try, dx_try, 300, t0 + budget_s - 120, 600)
        # Random LS
        if rem() > 60:
            h_try, c5_try = _random_ls(h_try, n_try, dx_try, rng, t0 + budget_s - 60, 300)

        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 4: Multi-start at best n
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    it = 0
    for freq in [1.0, 2.0, 3.0, 1.5, 2.5]:
        if rem() < 100:
            break
        h_s = _structured_init(n_best, rng, freq)
        for alpha in [80, 200]:
            if rem() < 90:
                break
            h_s, _ = _pgd(h_s, n_best, dx_best, alpha, 200, t0 + budget_s - 90)
        if rem() > 50:
            h_s, c5_s = _greedy_search(h_s, n_best, dx_best, 200, t0 + budget_s - 50, 400)
        else:
            c5_s = _c5(h_s, dx_best)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s
        it += 1

    # Phase 5: SLSQP polish
    if rem() > 40:
        for alpha in [300, 800, 1500]:
            if rem() < 35:
                break
            h_p = _slsqp_opt(best_h.copy(), n_best, dx_best, alpha, 300, 1e-14, t0 + budget_s - 30)
            c5_p = _c5(h_p, dx_best)
            if c5_p < best_c5:
                best_c5, best_h = c5_p, h_p

    # Phase 6: Final greedy + local search
    if rem() > 15:
        best_h, best_c5 = _greedy_search(best_h, n_best, dx_best, 500, t0 + budget_s - 8, 500)
    if rem() > 6:
        best_h, best_c5 = _random_ls(best_h, n_best, dx_best, rng, t0 + budget_s - 3, 200)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
