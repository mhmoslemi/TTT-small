    return np.max(_corr(h)) * dx

    m = np.max(c)
    s = _G_AL * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _G_AL

def _eqf(x):
    return np.sum(x) - _G_EQ

def _eqj(x):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        ix = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ix:
            if d > 0:

def _grad_step(h, n, dx, top_k, step_size, iters):
    """Gradient step to reduce top-k correlations."""
    best = _c5(h, dx)
    for _ in range(iters):
        c = _corr(h)
        # Find top-k active shifts
        idx_top = np.argsort(c)[-top_k:][::-1]
        # Compute gradient: weighted combination of active correlation gradients
        w = np.zeros(len(c))
        for idx in idx_top:
            w[idx] = c[idx]  # weight by correlation value
        w /= w.sum()
        # Gradient of sum(w[k] * c[k]) w.r.t. h
        # c[k] = sum_i h[i] * (1-h[i+k_shift])
        # This is complex; use numerical gradient
        grad = np.zeros(n)
        eps = 1e-6
        for m in range(n):
            h[m] += eps
            cp = _corr(h)
            h[m] -= eps
            h[m] -= eps
            cm = _corr(h)
            h[m] += eps
            grad[m] = (cp - cm)[idx_top].sum() / (2 * eps)
        # Projected gradient step
        h -= step_size * grad
        h = _repair(h, n)
        nc = _c5(h, dx)
        if nc > best:
            # Revert
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

_G_EQ = 0
_G_AL = 100

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _smo(x):
    c = _corr(x)
    m = np.max(c)
    s = _G_AL * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / _G_AL

def _eqf(x):
    return np.sum(x) - _G_EQ

def _eqj(x):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        ix = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ix:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-10:
                break
    return h

def _slq(h, n, al, mi, dl):
    global _G_EQ, _G_AL
    _G_EQ = n / 2.0
    _G_AL = al
    bd = [(0.0, 1.0)] * n
    cn = {'type': 'eq', 'fun': _eqf, 'jac': _eqj}
    try:
        r = minimize(_smo, h, method='SLSQP', bounds=bd, constraints=[cn],
                     options={'maxiter': mi, 'ftol': 1e-14})
        hc = _repair(r.x, n)
        return hc, _c5(hc, 2.0 / n)
    except Exception:
        return h, _c5(h, 2.0 / n)

def _pws(h, n, dx, sc, mp, te):
    best = _c5(h, dx)
    for _ in range(mp):
        if time.time() > te:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                for sg in (1, -1):
                    d = sg * sc
                    ni = h[i] + d
                    nj = h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        d *= 0.5
                        ni = h[i] + d
                        nj = h[j] - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                    h[i] = ni; h[j] = nj
                    c = _c5(h, dx)
                    if c < best - 1e-14:
                        best = c; imp = True
                    else:
                        h[i] -= d; h[j] += d
        if not imp:
            break
    return h, best

def _sa(h, n, dx, rng, te):
    bh = h.copy(); bc = _c5(h, dx)
    cur = h.copy(); cc = bc
    T = 8e-4
    while time.time() < te:
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = 0.025 * (2.0 * rng.random() - 1.0)
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i] = ni; cur[j] = nj
        nc = _c5(cur, dx)
        if nc < cc or rng.random() < np.exp(-(nc - cc) / T):
            cc = nc
            if nc < bc:
                bc = nc; bh = cur.copy()
        else:
            cur[i] -= d; cur[j] += d
        T = max(T * 0.9996, 1e-8)
    return bh, bc

def _multi_start(n, dx, rng, t_end):
    """Try a few structured starts alongside the parent."""
    candidates = []
    # Structured: two bumps

    h1 = np.clip(bump1 + bump2, 0, 1)

    # Structured: sinusoidal

    for h in candidates:
        if time.time() > t_end:
            break
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
