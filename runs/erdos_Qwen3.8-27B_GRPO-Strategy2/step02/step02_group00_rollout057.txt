    return p

def _fast_sa(h0, corr0, n, dx, t_end, rng):
    h = h0.copy()
    corr = corr0.copy()
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    sv = np.arange(-(n - 1), n).astype(np.int64)
    T0, Tend = 0.006, 0.00001
    max_it = 80000000
    no_imp = 0
    stall = 300000
    for it in range(max_it):
        if time.time() > t_end:
            break
        if no_imp > stall:
            h = best_h.copy()
            for _ in range(8):
                a, b = rng.randint(n), rng.randint(n)
                if a != b:
                    d = rng.randn() * 0.08
                    h[a] = np.clip(h[a] + d, 0, 1)
                    h[b] = np.clip(h[b] - d, 0, 1)
            h = _repair(h, n)
            corr = _corr_full(h)
            no_imp = 0
            continue
        frac = min(it / max_it, 1.0)
        T = T0 * (Tend / T0) ** frac
        dmax = 0.06 * (1.0 - 0.4 * frac)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (rng.rand() - 0.5) * 2.0 * dmax
        ni = h[i] + delta
        if ni < 0.0 or ni > 1.0:
            continue
        nj = h[j] - delta
        if nj < 0.0 or nj > 1.0:
            continue
        pat = _unit_pat(h, i, j, n, sv)
        new_corr = corr + delta * pat
        nm = float(np.max(new_corr))
        if nm * dx <= best_c5 + T * dx:
            h[i] = ni; h[j] = nj
            corr = new_corr
            if nm * dx < best_c5 - 1e-14:
                best_c5 = nm * dx
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
    return best_h, best_c5

def _pair_local(h0, corr0, n, dx, t_end):
    h = h0.copy()
    corr = corr0.copy()
    best = float(np.max(corr)) * dx
    sv = np.arange(-(n - 1), n).astype(np.int64)
    n_deltas = 12
    for sweep in range(200):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(n):
                if i == j:
                    continue
                dlo = max(-h[j], -h[i])
                dhi = min(1.0 - h[i], h[j])
                if dlo >= dhi - 1e-12:
                    continue
                pat = _unit_pat(h, i, j, n, sv)
                ds = np.linspace(dlo + 1e-6, dhi - 1e-6, n_deltas)
                best_d = 0.0
                best_m = float(np.max(corr))
                for d in ds:
                    nm = float(np.max(corr + d * pat))
                    if nm < best_m - 1e-10:
                        best_m = nm
                        best_d = d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    corr += best_d * pat
                    if best_m * dx < best - 1e-14:
                        best = best_m * dx
                        improved = True
                    else:
                        h[i] -= best_d
                        h[j] += best_d
                        corr -= best_d * pat
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(_corr_full(h), dx)

    # Phase 1: SLSQP with LSE (12% time)
    p1 = t0 + budget_s * 0.12
    for alpha in [3000, 8000, 15000]:
        if time.time() > p1:
            break
        try:
            res = minimize(_obj_lse, best_h, args=(alpha, n),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(_corr_full(hc), dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Fast SA with incremental updates (53% time)
    p2 = t0 + budget_s * 0.65
    if time.time() < p2:
        sh, sc = _fast_sa(best_h, _corr_full(best_h), n, dx, p2, rng)
        if _c5(_corr_full(sh), dx) < best_c5:
            best_c5, best_h = _c5(_corr_full(sh), dx), sh

    # Phase 3: Pair-based local search (20% time)
    p3 = t0 + budget_s * 0.85
    if time.time() < p3:
        ph, pc = _pair_local(best_h, _corr_full(best_h), n, dx, p3)
        if pc < best_c5:
            best_c5, best_h = pc, _repair(ph, n)

    # Phase 4: Another SA round from improved solution (8% time)
    p4 = t0 + budget_s * 0.93
    if time.time() < p4:
        sh, sc = _fast_sa(best_h, _corr_full(best_h), n, dx, p4, rng)
        if _c5(_corr_full(sh), dx) < best_c5:
            best_c5, best_h = _c5(_corr_full(sh), dx), sh

    # Phase 5: Final SLSQP polish
    if time.time() < t0 + budget_s - 5:
        try:
            res = minimize(_obj_lse, best_h, args=(20000, n),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(_corr_full(hc), dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
        # Tiny perturbation + re-SLSQP
        if time.time() < t0 + budget_s - 3:
            h2 = best_h.copy()
            for _ in range(4):
                a, b = rng.randint(n), rng.randint(n)
                if a != b:
                    d = rng.randn() * 0.003
                    h2[a] = np.clip(h2[a] + d, 0, 1)
                    h2[b] = np.clip(h2[b] - d, 0, 1)
            h2 = _repair(h2, n)
            try:
                res = minimize(_obj_lse, h2, args=(20000, n),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-16})
                hc = _repair(res.x, n)
                c5 = _c5(_corr_full(hc), dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(_corr_full(best_h), dx)
    return (best_h, best_c5, n)
