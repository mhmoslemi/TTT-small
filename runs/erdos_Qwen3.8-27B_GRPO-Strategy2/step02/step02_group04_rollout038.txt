import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    """Compute correlation values (before multiplying by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} via bisection"""
    if h.sum() == target:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mu = 0.5*(lo+hi)
        x = np.clip(h - mu, 0.0, 1.0)
        if x.sum() > target:
            lo = mu
        else:
            hi = mu
    x = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    return x

def _grad_corr(h, k):
    """Gradient of corr_k w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0*h
    elif k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:n]
    else:
        kk = -k
        grad[:n-kk] += (1.0 - h[kk:])
        grad[kk:] += -h[:n-kk]
    return grad

def _subgrad_max(corr, K=5):
    """Subgradient direction for max of corr values (indices into the corr array)"""
    # corr has length 2n-1, lag k corresponds to index k + (n-1)
    # Return the top-K indices and their gradients to be computed
    n_half = len(corr)//2
    idx = np.argsort(corr)[::-1][:K]
    return idx

def _run_subgrad(h, dx, target, n, iters, base_lr, deadline, topK=5):
    """Subgradient descent with projection"""
    h = h.copy()
    best = _c5(h, dx)
    best_h = h.copy()
    lr = base_lr
    momentum = np.zeros(n)
    
    for it in range(iters):
        if time.time() >= deadline:
            break
        corr = _corr(h)
        c5_val = np.max(corr) * dx
        if c5_val < best - 1e-12:
            best = c5_val
            best_h = h.copy()
        
        # Find top-K lags
        top_idx = np.argsort(corr)[::-1][:topK]
        
        # Compute subgradient (average of top-K gradients)
        # Convert corr indices to lags: lag = idx - (n-1)
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n-1)
            grad += _grad_corr(h, k)
        grad /= len(top_idx)
        grad *= dx  # scale for the actual objective
        
        # Momentum step
        momentum = 0.7*momentum + 0.3*grad
        h_new = h - lr*momentum
        h_new = _project_box_sum(h_new, target)
        
        # Check if we improved
        new_c5 = _c5(h_new, dx)
        if new_c5 < c5_val:
            h = h_new
            lr = min(lr*1.05, 0.1)
        else:
            lr = max(lr*0.5, 1e-5)
    
    return best_h, best

def _nelder_mead_reduced(h0, dx, target, n, basis, deadline, max_iter=500):
    """Nelder-Mead in reduced space: h = project(h0 + sum_j z_j * basis_j)"""
    m = len(basis)
    # Initial simplex
    z0 = np.zeros(m)
    simplex = [z0.copy()]
    step = 0.01
    for j in range(m):
        z = z0.copy()
        z[j] = step
        simplex.append(z.copy())
    
    def obj(z):
        h = h0 + sum(z[j]*basis[j] for j in range(m))
        h = _project_box_sum(h, target)
        return _c5(h, dx)
    
    values = [obj(z) for z in simplex]
    order = np.argsort(values)
    simplex = [simplex[i] for i in order]
    values = [values[i] for i in order]
    
    best_z = simplex[0].copy()
    best_val = values[0]
    
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # Reflect
        centroid = np.mean(simplex[:-1], axis=0)
        z_ref = centroid + (centroid - simplex[-1])
        f_ref = obj(z_ref)
        if f_ref < values[0]:
            # Expand
            z_exp = centroid + 2*(centroid - simplex[-1])
            f_exp = obj(z_exp)
            if f_exp < f_ref:
                simplex[-1] = z_exp
                values[-1] = f_exp
            else:
                simplex[-1] = z_ref
                values[-1] = f_ref
        elif f_ref < values[-2]:
            simplex[-1] = z_ref
            values[-1] = f_ref
        else:
            # Contract
            z_con = centroid + 0.5*(simplex[-1] - centroid)
            f_con = obj(z_con)
            if f_con < values[-1]:
                simplex[-1] = z_con
                values[-1] = f_con
            else:
                # Shrink
                for i in range(1, m+1):
                    simplex[i] = simplex[0] + 0.5*(simplex[i] - simplex[0])
                    values[i] = obj(simplex[i])
        
        # Reorder
        order = np.argsort(values)
        simplex = [simplex[i] for i in order]
        values = [values[i] for i in order]
        
        if values[0] < best_val:
            best_val = values[0]
            best_z = simplex[0].copy()
    
    h_best = _project_box_sum(h0 + sum(best_z[j]*basis[j] for j in range(m)), target)
    return h_best, _c5(h_best, dx)

def _local_search_h(h, dx, target, n, deadline, rng):
    """Gradient-informed local search"""
    h = h.copy()
    best = _c5(h, dx)
    
    for phase_scale in [0.05, 0.03, 0.01, 0.005, 0.002, 0.001]:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for _ in range(300):
                if time.time() >= deadline:
                    break
                # Pick a random pair
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = phase_scale * (2*rng.random() - 1)
                ni, nj = h[i]+d, h[j]-d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5 = _c5(h, dx)
                if c5 < best - 1e-12:
                    best = c5
                    improved = True
                else:
                    h[i], h[j] = h[i]-d, h[j]+d
    return h, best

def run(seed=42, budget_s=1100, **kwargs):

from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mu = 0.5*(lo+hi)
        x = np.clip(h - mu, 0.0, 1.0)
        if x.sum() > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def _grad_corr(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0*h
    elif k > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        kk = -k
        g[:n-kk] += (1.0 - h[kk:])
        g[kk:] -= h[:n-kk]
    return g

def _lp_move(h, corr, top_idx, eps):

    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    rows, bvals = [], []
    n1 = n - 1
    for idx in top_idx:
        k = idx - n1
        gr = _grad_corr(h, k)
        r = np.zeros(nv)
        r[:n] = gr
        r[n] = -1.0
        rows.append(r)
        bvals.append(-corr[idx])
    A_ub = np.array(rows)
    b_ub = np.array(bvals)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo_b = np.maximum(-eps, -h)
    hi_b = np.minimum(eps, 1.0-h)
    bounds = list(zip(lo_b, hi_b)) + [(0.0, np.max(corr)*1.5+0.1)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n]
    return np.zeros(n)

def _lp_descend(h, dx, target, deadline, topK=5, eps0=0.05):

    best_h = h.copy()
    h = h.copy()

        if nc < cv - 1e-14:

        else:

            if nc2 < cv:

            else:
                eps *= 0.7
                if eps < 1e-7:

    return best_h, best_c5

def _subgrad(h, dx, target, deadline, lr0=0.01, topK=5):

    h = h.copy()

def _local(h, dx, target, rng, deadline):
    best = _c5(h, dx)
    n = len(h)
    for scale in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:

                i, j = rng.randint(n), rng.randint(n)

                d = scale*(2*rng.random()-1)

                if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                if c < best - 1e-13:

                    stall = 0
                else:

        c = _corr(x)*dx
        m = np.max(c)
        return m

    def obj_smooth(x):
        c = _corr(x)*dx
        m = np.max(c)
        s = alpha*(c-m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s)))/alpha

    def eq(x):
        return np.sum(x) - target

                       bounds=[(0,1)]*n,

                       options={'maxiter':maxiter,'ftol':1e-14})
        hc = _project(res.x, target)
        return hc, _c5(hc, dx)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mu = 0.5 * (lo + hi)
        x = np.clip(h - mu, 0.0, 1.0)
        if x.sum() > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_corr(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        kk = -k
        g[:n - kk] += (1.0 - h[kk:])
        g[kk:] -= h[:n - kk]
    return g

def _lp_move(h, corr, top_idx, eps):
    n = len(h)
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    n1 = n - 1
    rows, bvals = [], []
    for idx in top_idx:
        k = idx - n1
        gr = _grad_corr(h, k)
        r = np.zeros(nv)
        r[:n] = gr
        r[n] = -1.0
        rows.append(r)
        bvals.append(-corr[idx])
    A_ub = np.array(rows)
    b_ub = np.array(bvals)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo_b = np.maximum(-eps, -h)
    hi_b = np.minimum(eps, 1.0 - h)
    bounds = list(zip(lo_b, hi_b)) + [(0.0, np.max(corr) * 1.5 + 0.1)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n]
    return np.zeros(n)

def _lp_descend(h, dx, target, deadline, topK=5, eps0=0.05):
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    h = h.copy()
    n = len(h)
    n1 = n - 1
    eps = eps0
    stuck = 0
    for it in range(200000):
        if time.time() >= deadline:
            break
        corr = _corr(h) * dx
        cv = np.max(corr)
        if cv < best_c5 - 1e-13:
            best_c5 = cv
            best_h = h.copy()
            stuck = 0
        top = np.argsort(corr)[::-1][:topK]
        delta = _lp_move(h, corr, top, eps)
        step = np.max(np.abs(delta))
        if step < 1e-10:
            stuck += 1
            if stuck > 30:
                eps *= 0.5
                if eps < 1e-7:
                    break
                stuck = 0
            continue
        stuck = 0
        h_new = _project(h + delta, target)
        nc = _c5(h_new, dx)
        if nc < cv - 1e-14:
            h = h_new
        else:
            h_half = _project(h + 0.3 * delta, target)
            nc2 = _c5(h_half, dx)
            if nc2 < cv - 1e-14:
                h = h_half
            else:
                eps *= 0.7
                if eps < 1e-7:
                    break
    return best_h, best_c5

def _subgrad(h, dx, target, deadline, lr0=0.01, topK=5):
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    h = h.copy()
    n = len(h)
    n1 = n - 1
    lr = lr0
    mom = np.zeros(n)
    for it in range(200000):
        if time.time() >= deadline:
            break
        corr = _corr(h)
        cv = np.max(corr) * dx
        if cv < best_c5 - 1e-13:
            best_c5 = cv
            best_h = h.copy()
        top = np.argsort(corr)[::-1][:topK]
        g = np.zeros(n)
        for idx in top:
            g += _grad_corr(h, idx - n1)
        g /= float(topK)
        g *= dx
        mom = 0.75 * mom + 0.25 * g
        h_new = _project(h - lr * mom, target)
        nc = _c5(h_new, dx)
        if nc < cv - 1e-14:
            h = h_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.6
            if lr < 1e-9:
                break
    return best_h, best_c5

def _local(h, dx, target, rng, deadline):
    best = _c5(h, dx)
    n = len(h)
    for scale in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        stall = 0
        while stall < 5 and time.time() < deadline:
            stall += 1
            for _ in range(600):
                if time.time() >= deadline:
                    break
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best - 1e-13:
                    best = c
                    stall = 0
                else:
                    h[i], h[j] = ni - d, nj + d
    return h, best

def _smooth_obj(x, dx, alpha):
    c = _corr(x) * dx
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _eq_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, target, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(dx, alpha), method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _project(res.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _project(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: subgradient warmup (80s)

        h, c = _subgrad(h, dx, target, d1, lr0=0.02, topK=7)
        if c < best_c5:
            best_h, best_c5 = h, c
        h, c = _subgrad(h, dx, target, min(t0 + 130, d1 + 50), lr0=0.004, topK=3)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: LP descent main (250s)
    d2 = t0 + 380
    if budget_s > 450 and time.time() < d2:

        if c < best_c5:

    # Phase 3: LP descent finer (180s)
    d3 = t0 + 560

        # Second LP from perturbation

        d3b = t0 + 680

    # Phase 4: SLSQP polish (150s)
    d4 = t0 + 830
    if budget_s > 900 and time.time() < d4:
        for a in [80, 200, 500, 1200]:
            if time.time() > d4:
                break
