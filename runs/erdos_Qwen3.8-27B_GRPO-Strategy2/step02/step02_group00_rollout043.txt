    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h


def _obj_topk(x, k, dx):
    c = _corr(x)
    if k >= len(c):
        return float(c.sum()) * dx
    idx = np.argpartition(c, -k)[-k:]
    return float(c[idx].sum()) * dx


def _obj_lse(x, alpha, dx):
    c = _corr(x)
    m = float(np.max(c))
    s = np.exp(np.clip(alpha * (c - m), -50, 50))
    return (m + np.sum(s) / alpha) * dx


def _eq_con(x, target):
    return float(np.sum(x) - target)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    # Phase 1: SLSQP (brief)
    p1_end = t_start + min(budget_s * 0.06, 60)
    for k in [5, 3]:
        if time.time() > p1_end:
            break
        try:
            res = minimize(_obj_topk, best_h, args=(k, dx),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    try:
        res = minimize(_obj_lse, best_h, args=(2000, dx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    except Exception:
        pass

    # Phase 2: Multi-pass coordinate-pair descent (main workhorse)
    p2_end = t_start + budget_s * 0.72
    h = best_h.copy()
    local_best = best_c5
    n_passes = 25
    for p in range(n_passes):
        if time.time() > p2_end:
            break
        scale = 0.12 * (0.65 ** p)
        if scale < 0.00005:
            break
        deltas = np.array([scale, 0.5 * scale, -scale, -0.5 * scale,
                          0.25 * scale, -0.25 * scale, 2.0 * scale, -2.0 * scale])
        improved = False
        order_i = rng.permutation(n)
        for i in order_i:
            if time.time() > p2_end:
                break
            vi = h[i]
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    jc = np.argsort(h)[:12]
                else:
                    jc = np.argsort(-h)[:12]
                for j in jc:
                    if j == i:
                        continue
                    nj = h[j] + need
                    if nj < 0.0 or nj > 1.0:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5_try = _c5(h, dx)
                    if c5_try < local_best - 1e-15:
                        local_best = c5_try
                        improved = True
                        vi = h[i]
                    else:
                        h[i] = vi
                        h[j] = nj - need
        if not improved and p > 5:
            break

    if local_best < best_c5:
        best_c5, best_h = local_best, h.copy()

    # Phase 3: Simulated annealing on the refined solution
    p3_end = t_start + budget_s * 0.88
    if time.time() < p3_end:
        h_sa = best_h.copy()
        sa_best = best_c5
        sa_best_h = best_h.copy()
        n_iter = int(min(budget_s * 0.15, 400000))
        T0, Tend = 0.0025, 0.000005
        for it in range(n_iter):
            if time.time() > p3_end:
                break
            frac = (it + 1) / n_iter
            T = T0 * (Tend / T0) ** frac
            dmax = 0.015 * (1.0 - 0.7 * frac)
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = (rng.rand() - 0.5) * 2.0 * dmax
            ni = h_sa[i] + delta
            nj = h_sa[j] - delta
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h_sa[i] = ni
            h_sa[j] = nj
            new_c5 = _c5(h_sa, dx)
            if new_c5 <= sa_best + T:
                if new_c5 < sa_best - 1e-15:
                    sa_best = new_c5
                    sa_best_h = h_sa.copy()
            else:
                h_sa[i] -= delta
                h_sa[j] += delta
        if sa_best < best_c5:
            best_c5, best_h = sa_best, sa_best_h

    # Phase 4: Very fine coordinate descent
    p4_end = t_start + budget_s - 8
    if time.time() < p4_end:
        h = best_h.copy()
        local_best = best_c5
        fine_d = np.array([0.0001, 0.0003, 0.0005, 0.001,
                          -0.0001, -0.0003, -0.0005, -0.001])
        for _ in range(5):
            if time.time() > p4_end:
                break
            order_i = rng.permutation(n)
            for i in order_i:
                if time.time() > p4_end:
                    break
                vi = h[i]
                for d in fine_d:
                    ni = vi + d
                    if ni < 0.0 or ni > 1.0:
                        continue
                    need = -d
                    if need > 0:
                        jc = np.argsort(h)[:8]
                    else:
                        jc = np.argsort(-h)[:8]
                    for j in jc:
                        if j == i:
                            continue
                        nj = h[j] + need
                        if nj < 0.0 or nj > 1.0:
                            continue
                        h[i] = ni
                        h[j] = nj
                        c5_try = _c5(h, dx)
                        if c5_try < local_best - 1e-15:
                            local_best = c5_try
                            break
                        else:
                            h[i] = vi
                            h[j] = nj - need
        if local_best < best_c5:
            best_c5, best_h = local_best, h.copy()

    # Phase 5: Final SLSQP polish
    if time.time() < t_start + budget_s - 2:
        try:
            res = minimize(_obj_max_obj, best_h, args=(dx,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-17})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
        try:
            res = minimize(_obj_lse, best_h, args=(5000, dx),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-17})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)


def _obj_max_obj(x, dx):
    return float(np.max(_corr(x))) * dx
