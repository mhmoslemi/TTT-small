def _eq_con(x, target):
    return float(np.sum(x) - target)

def _grad_worst(h, m_w, N):
    j_arr = np.arange(N)
    g = np.zeros(N)
    p1 = j_arr - m_w + N - 1
    m1 = (p1 >= 0) & (p1 < N)
    g[m1] += (1.0 - h[p1[m1]])
    p2 = j_arr + m_w - N + 1
    m2 = (p2 >= 0) & (p2 < N)
    g[m2] -= h[p2[m2]]
    return g

def _targeted_sa(h, dx, N, rng, t_deadline):
    T0, Tend = 0.005, 0.00001
    best = _c5(h, dx)
    best_h = h.copy()
    h_w = h.copy()
    accepted = 0
    total = 0
    step_scale = 1.0

    while time.time() < t_deadline:
        total += 1
        frac = min(1.0, total / 800000.0)
        T = T0 * (Tend / T0) ** (frac ** 0.5)
        dmax = 0.06 * step_scale * (1.0 - 0.6 * frac)

        corr = np.correlate(h_w, 1.0 - h_w, mode='full')
        m_w = int(np.argmax(corr))
        g = _grad_worst(h_w, m_w, N)

        pos_idx = np.flatnonzero(g > 0.03)
        neg_idx = np.flatnonzero(g < -0.03)

        if len(pos_idx) > 0 and len(neg_idx) > 0:
            i = pos_idx[rng.randint(len(pos_idx))]
            j = neg_idx[rng.randint(len(neg_idx))]
        elif len(pos_idx) > 0:
            i = pos_idx[rng.randint(len(pos_idx))]
            j = rng.randint(N)
            if j == i:
                j = (i + 1) % N
        else:
            i = rng.randint(N)
            j = rng.randint(N)
            if i == j:
                continue

        delta = (rng.rand() - 0.5) * 2.0 * dmax
        ni = h_w[i] + delta
        if ni < 0.0 or ni > 1.0:
            continue
        nj = h_w[j] - delta
        if nj < 0.0 or nj > 1.0:
            continue

        h_w[i] = ni
        h_w[j] = nj
        new_c5 = _c5(h_w, dx)

        if new_c5 <= best + T:
            accepted += 1
            if new_c5 < best - 1e-14:
                best = new_c5
                best_h = h_w.copy()
        else:
            h_w[i] -= delta
            h_w[j] += delta

        if total % 1000 == 0:
            acc_rate = accepted / max(1, total % 10000)
            if acc_rate < 0.15:
                step_scale *= 0.85
            elif acc_rate > 0.6:
                step_scale *= 1.1
            step_scale = max(0.2, min(2.0, step_scale))

    return best_h, best

def _coord_descent(h, dx, N, t_deadline):
    local_best = _c5(h, dx)
    h = h.copy()
    deltas = np.array([0.001, 0.003, 0.005, 0.01, -0.001, -0.003, -0.005, -0.01])
    no_imp = 0
    while time.time() < t_deadline and no_imp < 4:
        no_imp += 1
        for i in range(N):
            if time.time() > t_deadline:
                break
            vi = h[i]
            found = False
            for d in deltas:
                ni = vi + d
                if ni < 0.0 or ni > 1.0:
                    continue
                need = -d
                if need > 0:
                    j = int(np.argmin(h))
                else:
                    j = int(np.argmax(h))
                if j == i:
                    continue
                nj = h[j] + need
                if nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                c5_try = _c5(h, dx)
                if c5_try < local_best - 1e-14:
                    local_best = c5_try
                    found = True
                    no_imp = 0
                    break
                else:
                    h[i] = vi
                    h[j] -= need
            if found:
                break
    return h, local_best

def _slsqp_pass(h, dx, N, t_deadline, alpha=2000):
    target = N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    rem = t_deadline - time.time()
    if rem < 10:
        return h, _c5(h, dx)
    try:
        res = minimize(_obj_lse, h, args=(alpha, N), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-14})
        if res.x is not None:
            hc = _repair(res.x, N)
            return hc, _c5(hc, dx)
    except Exception:
        pass
    return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = None

    best_h = None
    best_c5 = float('inf')
    hards_deadline = t_start + budget_s - 5

    N_list = []
    if parent is not None:
        N_list.append(len(parent))
    for N_try in [80, 90, 100, 120]:
        if N_try not in N_list:
            N_list.append(N_try)
    if len(N_list) > 4:
        N_list = N_list[:4]

    n_configs = len(N_list)
    per_config = (budget_s - 40) / n_configs

    for cfg_idx, N in enumerate(N_list):
        if time.time() > hards_deadline:
            break
        cx_deadline = t_start + per_config * (cfg_idx + 1) - 5
        dx = 2.0 / N
        target = N / 2.0

        starts = []
        if parent is not None:
            if N == len(parent):
                starts.append(_repair(parent.copy(), N))
            else:
                idx = np.linspace(0, len(parent) - 1, N)
                i0 = np.floor(idx).astype(int)
                i1 = np.minimum(i0 + 1, len(parent) - 1)
                fr = idx - i0
                interp = parent[i0] * (1.0 - fr) + parent[i1] * fr
                starts.append(_repair(interp, N))

        h_alt = np.zeros(N)
        h_alt[::2] = 1.0
        starts.append(_repair(h_alt, N))

        h_blk = np.zeros(N)
        blen = max(N // 6, 1)
        for b in range(0, N, 2 * blen):
            h_blk[b:b + blen] = 1.0
        starts.append(_repair(h_blk, N))

        h_rnd = rng.rand(N)
        starts.append(_repair(h_rnd, N))

        n_starts = len(starts)
        per_start = (cx_deadline - time.time()) / max(1, n_starts) - 3

        for h0 in starts:
            t_final = time.time() + per_start
            if t_final > hards_deadline:
                t_final = hards_deadline
            if t_final - time.time() < 5:
                break

            c5_cur = _c5(h0, dx)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()

            sa_end = time.time() + (t_final - time.time()) * 0.65
            if sa_end - time.time() > 15:
                h_sa, c5_sa = _targeted_sa(h0, dx, N, rng, sa_end)
                if c5_sa < c5_cur:
                    c5_cur = c5_sa
                    h0 = h_sa.copy()

            cd_end = time.time() + (t_final - time.time()) * 0.25
            if cd_end - time.time() > 5:
                h_cd, c5_cd = _coord_descent(h0, dx, N, cd_end)
                if c5_cd < c5_cur:
                    c5_cur = c5_cd
                    h0 = h_cd.copy()

            if time.time() < t_final - 3:
                h_s, c5_s = _slsqp_pass(h0, dx, N, t_final - 2)
                if c5_s < c5_cur:
                    c5_cur = c5_s
                    h0 = h_s.copy()

            h0 = _repair(h0, N)
            c5_cur = _c5(h0, dx)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()

        if best_h is not None:
            best_h = _repair(best_h, len(best_h))
            best_c5 = _c5(best_h, 2.0 / len(best_h))

    if best_h is None:
        N = 70
        h = np.zeros(N)
        h[::2] = 1.0
        best_h = _repair(h, N)
        best_c5 = _c5(best_h, 2.0 / N)

    return (best_h, best_c5, len(best_h))
