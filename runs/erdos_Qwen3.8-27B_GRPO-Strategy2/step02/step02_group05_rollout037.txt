def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

    ip = i - p
    iq = i + p

    valid2 = (iq >= 0) & (iq < n)
    a = np.zeros(2 * n - 1)
    a[valid1] = 1.0 - h[ip[valid1]]
    a[valid2] -= h[iq[valid2]]
    return a

        if len(vc) > 0:
            cands = np.concatenate([cands, np.clip(vc, 0.0, 1.0)])
    objs = np.max(b[None, :] + cands[:, None] * a[None, :], axis=1)

    cvals = _corr(h)
    best_c5 = float(np.max(cvals) * dx)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_v = h[i]
            a = _coeff_vec(h, i, n)
            b = cvals - old_v * a  # intercepts
            new_v = _opt_1d(b, a)
            if abs(new_v - old_v) > 1e-10:
                h[i] = new_v
                cvals += (new_v - old_v) * a
                if float(np.max(cvals)) < best_c5 - 1e-14:
                    best_c5 = float(np.max(cvals))
                    improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:

        if (sweep + 1) % 20 == 0:
            hr = _repair(h, n)
            cr = _c5(hr, dx)
            if cr <= best_c5 + 0.005:
                h = hr
                cvals = _corr(h)
                best_c5 = cr
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _coeff_vec(h, i, n):
    p = np.arange(-(n - 1), n)
    ip = i - p
    iq = i + p
    valid1 = (ip >= 0) & (ip < n)
    valid2 = (iq >= 0) & (iq < n)
    a = np.zeros(2 * n - 1)
    a[valid1] = 1.0 - h[ip[valid1]]
    a[valid2] -= h[iq[valid2]]
    return a

def _opt_1d(b, a):
    nl = len(a)
    cands = np.array([0.0, 1.0])
    ia, ja = np.triu_indices(nl, 1)
    denom = a[ia] - a[ja]
    valid = np.abs(denom) > 1e-14
    if np.any(valid):
        vc = (b[ja[valid]] - b[ia[valid]]) / denom[valid]
        vc = vc[(vc >= -1e-12) & (vc <= 1.0 + 1e-12)]
        if len(vc) > 0:
            cands = np.concatenate([cands, np.clip(vc, 0.0, 1.0)])
    objs = np.max(b[None, :] + cands[:, None] * a[None, :], axis=1)
    return cands[int(np.argmin(objs))]

def _coord_descent(h, n, dx, rng, max_sweeps, deadline):
    h = h.copy()
    cvals = _corr(h)
    best_c5 = float(np.max(cvals) * dx)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_v = h[i]
            a = _coeff_vec(h, i, n)
            b = cvals - old_v * a
            new_v = _opt_1d(b, a)
            if abs(new_v - old_v) > 1e-10:
                h[i] = new_v
                cvals += (new_v - old_v) * a
                if float(np.max(cvals)) < best_c5 - 1e-14:
                    best_c5 = float(np.max(cvals))
                    improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        if (sweep + 1) % 15 == 0:
            hr = _repair(h, n)
            cr = _c5(hr, dx)
            if cr <= best_c5 + 0.005:
                h = hr
                cvals = _corr(h)
                best_c5 = cr
    return h, best_c5

def _local_search(h, n, dx, rng, deadline):
    best = h.copy()
    best_c5 = _c5(best, dx)
    steps = np.array([0.01, 0.02, 0.04, 0.08, 0.12, -0.01, -0.02, -0.04, -0.08, -0.12])
    stall = 0
    while time.time() < deadline and stall < 20:
        improved = False
        for _ in range(400):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-13:
                    best_c5 = nc
                    improved = True
                else:
                    best[i], best[j] = ni - d, nj + d
                if improved:
                    break
        stall = 0 if improved else stall + 1
    return best, best_c5

def _slsqp_obj(x, alpha, dx):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _slsqp_con(x, target):
    return np.sum(x) - target

def _slsqp_refine(h, n, dx, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'args': (n / 2.0,)}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.choice(n, n // 2, replace=False)
    h[pos] = 1.0
    return _repair(h, n)

def _sin_init(n, freq, rng):
    t = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t) + 0.1 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: Coordinate descent from initial
    d1 = t0 + budget_s * 0.20
    h1, c1 = _coord_descent(h, n, dx, rng, 5000, d1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Local search
    d2 = t0 + budget_s * 0.35
    if time.time() < d2:
        hl, cl = _local_search(best_h, n, dx, rng, d2)
        if cl < best_c5:
            best_c5, best_h = cl, hl

    # Phase 3: Coordinate descent again
    d3 = t0 + budget_s * 0.50
    if time.time() < d3:
        h3, c3 = _coord_descent(best_h, n, dx, rng, 3000, d3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: SLSQP polish
    d4 = t0 + budget_s * 0.62
    if time.time() < d4:
        for a in [80, 200, 500, 1000]:
            if time.time() > d4:
                break
            hs = _slsqp_refine(best_h, n, dx, a, 350)
            cs = _c5(hs, dx)
            if cs < best_c5:
                best_c5, best_h = cs, hs

    # Phase 5: Multi-start with different initializations
    d5 = t0 + budget_s * 0.82
    inits = [_binary_init(n, np.random.RandomState(seed + 7)),
             _sin_init(n, 2, np.random.RandomState(seed + 13)),
             _sin_init(n, 3, np.random.RandomState(seed + 19)),
             _sin_init(n, 1, np.random.RandomState(seed + 23)),
             _repair(best_h + 0.15 * rng.randn(n), n),
             _repair(best_h + 0.3 * rng.randn(n), n)]
    for h_init in inits:
        if time.time() > d5:
            break
        hr, cr = _coord_descent(h_init, n, dx, rng, 2000, min(time.time() + 50, d5))
        if cr < best_c5:
            best_c5, best_h = cr, hr
        elif cr < best_c5 + 0.002 and time.time() < d5 - 10:
            hl2, cl2 = _local_search(hr, n, dx, rng, d5 - 5)
            if cl2 < best_c5:
                best_c5, best_h = cl2, hl2

    # Phase 6: Perturbation restarts from best
    d6 = t0 + budget_s * 0.92
    sigmas = [0.08, 0.15, 0.25, 0.12, 0.35, 0.06, 0.2, 0.4, 0.1, 0.3]
    it = 0
    while time.time() < d6:
        s = sigmas[it % len(sigmas)]
        hp = _repair(best_h + s * rng.randn(n), n)
        hr, cr = _coord_descent(hp, n, dx, rng, 1000, min(time.time() + 35, d6))
        if cr < best_c5:
            best_c5, best_h = cr, hr
        elif cr < best_c5 + 0.002 and time.time() < d6 - 12:
            hl2, cl2 = _local_search(hr, n, dx, rng, d6 - 8)
            if cl2 < best_c5:
                best_c5, best_h = cl2, hl2
        it += 1

    # Phase 7: Final local search
    d7 = t0 + budget_s - 25
    if time.time() < d7:
        hl, cl = _local_search(best_h, n, dx, rng, d7)
        if cl < best_c5:
            best_c5, best_h = cl, hl

    # Phase 8: Final coordinate descent polish
    if time.time() < t0 + budget_s - 5:
        hf, cf = _coord_descent(best_h, n, dx, rng, 1000, t0 + budget_s - 5)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
