import time

                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:

def _binarize(h, n, rng):
    """Round to nearest binary with exact count."""
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(h_b.sum())
    diff = target - s
    if diff > 0:

        idx = rng.choice(zeros, diff, replace=False)
        h_b[idx] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        idx = rng.choice(ones, -diff, replace=False)
        h_b[idx] = 0.0
    return h_b

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    s = np.exp(np.clip(alpha * (corr - m), -50, 50))
    return (m + s.sum() / alpha) * (2.0 / n)

def _obj_topk(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    return corr[idx].sum() * (2.0 / n)

    best_c5 = _c5_val(h, dx)
    best_h = h.copy()

    it = 0
    max_it = 500000
    while time.time() < t_end and it < max_it:

        frac = it / max_it
        T = 3.0 * (0.05 / 3.0) ** frac

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_c5

    cur_h = h.copy()
    cur_c5 = _c5_val(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    while time.time() < t_end:

        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]

                best_c5 = cur_c5
                best_h = cur_h.copy()

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _binarize(h, n, rng):
    h_b = (h + 0.1 * rng.randn(n) >= 0.5).astype(float)  # slight randomization
    target = n // 2
    s = int(h_b.sum())
    diff = target - s
    if diff > 0:
        zeros = np.where(h_b == 0)[0]
        idx = rng.choice(zeros, diff, replace=False)
        h_b[idx] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        idx = rng.choice(ones, -diff, replace=False)
        h_b[idx] = 0.0
    return h_b

def _obj_lse(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    return (m + np.sum(np.exp(np.clip(alpha * (corr - m), -50, 50))) / alpha) * (2.0 / n)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _sa_binary(h, dx, n, rng, t_end, iters_per_sec=50000):
    """SA on binary solutions with swap moves."""
    best_c5 = _c5_val(h, dx)
    best_h = h.copy()
    cur_h = h.copy()
    cur_c5 = best_c5
    T = 2.0
    T_min = 0.01
    count = 0
    while time.time() < t_end:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        ii = ones[i]
        jj = zeros[j]
        cur_h[ii] = 0
        cur_h[jj] = 1
        new_c5 = _c5_val(cur_h, dx)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[ii] = 1
            cur_h[jj] = 0
        count += 1
        if count % 5000 == 0:
            T = max(T_min, T * 0.999)
    return best_h, best_c5

def _binary_hill_climb(h, dx, n, t_end):
    """Steepest-descent binary hill climbing."""
    cur_h = h.copy()
    cur_c5 = _c5_val(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    while time.time() < t_end:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        best_delta = 0
        best_i = best_j = -1
        for ii in ones:
            cur_h[ii] = 0
            for jj in zeros:
                cur_h[jj] = 1
                d = _c5_val(cur_h, dx) - cur_c5
                if d < best_delta:
                    best_delta = d
                    best_i, best_j = ii, jj
                cur_h[jj] = 0
            cur_h[ii] = 1
        if best_i >= 0:
            cur_h[best_i] = 0
            cur_h[best_j] = 1
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            break
    return best_h, best_c5

1. Compute gradient of lse objective
2. Take a step
3. Project onto {0 ≤ h[i] ≤ 1, sum(h) = target}

The projection onto {0 ≤ h[i] ≤ 1, sum(h) = target} can be done by:
1. Clip to [0,1]
2. Adjust to make sum = target (similar to _repair but done more carefully)

Actually, the Wasserstein projection or the "dual" approach for projection onto the simplex with bounds:
- Start with y = x - step * grad
- Find λ such that sum(clamp(y - λ, 0, 1)) = target
- h = clamp(y - λ, 0, 1)

This is solvable by bisection on λ!

```python
def _project(y, target, n):
    """Project y onto {0<=h[i]<=1, sum(h)=target}."""
    lo, hi = -1.0, 2.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h
```

corr[k] = sum_j h[j] * (1-h[j-(k-(n-1))])

The valid range of j-(k-(n-1)) is [0, n-1], so j is in [k-(n-1), k-(n-1)+n-1] = [k-n+1, k].

d(corr[k])/d(h[i]) appears in two terms:

        if len(ones) >= -diff:

    m = corr.max()
    return (m + np.sum(np.exp(np.clip(alpha * (corr - m), -50, 50))) / alpha) * (2.0 / n)

    """Sum of squared correlations (smooth)."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(corr * corr)) * (2.0 / n)

    """LSE with penalty on sum constraint."""
    c5part = _obj_lse(x, alpha, n)
    pen = lam * (float(np.sum(x)) - target) ** 2
    return c5part + pen

Actually, one more idea: **what if I use the "top-k sum" as the objective for SLSQP?** For k=2 or k=3, this is differentiable (almost) and directly targets the top correlations. The issue is it's not smooth at the points where the ordering changes.

A better approach: use the "smooth max" which is:
F_α(h) = (1/α) * log(sum_k exp(α * corr[k]))

This is always smooth and converges to max as α→∞. For α=1000, it's very close to max.

The gradient of F_α is:
dF_α/dh[i] = sum_k p_k * d(corr[k])/d(h[i])

where p_k = exp(α*corr[k]) / sum_j exp(α*corr[j])

This is a "soft" version of the max gradient.

For L-BFGS-B, I need to provide the gradient (or let it compute numerically). Numerical gradient for n=70 means 70 function evaluations per gradient step. Each eval is O(n) for the correlate (actually O(n^2) but small constant). So 70 * (2n-1) ≈ 70*139 ≈ 10K ops per gradient. At Python speed, maybe 5-10ms per gradient. L-BFGS-B does ~n evaluations per iteration, so ~70 * 10ms = 700ms per iteration. In 200s, that's ~280 iterations. Should be enough for convergence.

Actually, scipy's L-BFGS-B uses numerical differentiation by default (finite differences), which costs n function evaluations per gradient. With maxfun=30000, that's 30000/70 ≈ 430 gradient evaluations. Each takes ~10ms, so ~4.3s total. That's fine.

import numpy as np
from scipy.optimize import minimize
import time

def _c5_val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _binarize(h, n, rng):
    h_b = (h + 0.05 * rng.randn(n) >= 0.5).astype(float)
    target = n // 2
    s = int(h_b.sum())
    diff = target - s
    if diff > 0:
        zeros = np.where(h_b == 0)[0]
        if len(zeros) >= diff:
            h_b[rng.choice(zeros, diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        if len(ones) >= -diff:
            h_b[rng.choice(ones, -diff, replace=False)] = 0.0
    return h_b

def _obj_lse(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    return (m + np.sum(np.exp(np.clip(alpha * (corr - m), -50, 50))) / alpha) * (2.0 / n)

def _obj_pen(x, alpha, n, target, lam):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    lse = (m + np.sum(np.exp(np.clip(alpha * (corr - m), -50, 50))) / alpha) * (2.0 / n)
    pen = lam * (float(np.sum(x)) - target) ** 2
    return lse + pen

def _obj_topk(x, k, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    return corr[idx].sum() * (2.0 / n)

def _eq_con(x, target):
    return float(np.sum(x) - target)

def _sa_binary(h, dx, n, rng, t_end):
    best_c5 = _c5_val(h, dx)
    best_h = h.copy()
    cur_h = h.copy()
    cur_c5 = best_c5
    T = 3.0
    while time.time() < t_end:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ii = ones[rng.randint(len(ones))]
        jj = zeros[rng.randint(len(zeros))]
        cur_h[ii] = 0
        cur_h[jj] = 1
        new_c5 = _c5_val(cur_h, dx)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[ii] = 1
            cur_h[jj] = 0
        T = max(0.005, T * 0.99997)
    return best_h, best_c5

def _hill_climb_bin(h, dx, n, t_end):
    cur_h = h.copy()
    cur_c5 = _c5_val(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    while time.time() < t_end:
        ones = list(np.where(cur_h == 1)[0])
        zeros = list(np.where(cur_h == 0)[0])
        best_d = 0
        bi = bj = -1
        for ii in ones:
            cur_h[ii] = 0
            for jj in zeros:
                cur_h[jj] = 1
                d = _c5_val(cur_h, dx) - cur_c5
                if d < best_d:
                    best_d = d
                    bi, bj = ii, jj
                cur_h[jj] = 0
            cur_h[ii] = 1
        if bi >= 0:
            cur_h[bi] = 0
            cur_h[bj] = 1
            cur_c5 += best_d
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            break
    return best_h, best_c5
