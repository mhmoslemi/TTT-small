    s = alpha * (corr - m)

    return m + np.log(np.sum(np.exp(s))) / alpha

    if s < 1e-12:
        return np.full(n, 0.5)

                diff -= add

                diff += sub

    m = np.max(corr)

    active = np.where(corr >= m - 0.001 * m)[0]

    G = np.zeros(n)

        g = np.zeros(n)
        for j in range(n):
            val = 0.0
            if k >= 0:
                if j + k < n:
                    val += (1.0 - h[j + k])
                if j - k >= 0:
                    val -= h[j - k]
            else:
                kk = -k
                if j - kk >= 0:
                    val += (1.0 - h[j - kk])
                if j + kk < n:
                    val -= h[j + kk]
            g[j] = val
        G += g

    G_projected = G - np.mean(G)

    h_new = h - step_size * G_projected
    h_new = np.clip(h_new, 0, 1)
    # Repair sum
    h_new = _repair(h_new, n)

    # corr_k = sum_{valid} h[p]*(1-h[q]) where q=p+k (or p=q+k)
    # Changing h[i] by d affects terms where i is p or q
    # Changing h[j] by d (with -d) affects terms where j is p or q

    # If h[i] is the first factor (p=i): d*corr/d(h[i]) = (1-h[i+k])
    # If h[i] is the second factor (q=i, p=i-k): d*corr/d(h[i]) = -h[i-k]

    # But with 139 lines, there could be many breakpoints

    return h, best_v * dx

def _pair_local_search(h, n, dx, rng, deadline, n_iter=20000):
    """Local search: cycle through random pairs with exact optimal step."""
    best = _c5(h, dx)
    h = h.copy()
    it = 0
    stall = 0
    while time.time() < deadline and it < n_iter:
        it += 1
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        h, c = _exact_pair(h, dx, i, j)
        if c < best - 1e-14:
            best = c
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
    return h, best

def _fine_search(h, n, dx, rng, deadline):
    """Finer pair search with restricted step range."""
    best = _c5(h, dx)
    h = h.copy()
    # Restrict pairs to nearby indices for smoother changes
    while time.time() < deadline:
        improved = False
        for _ in range(5000):
            if time.time() >= deadline:
                return h, best
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h2 = h.copy()
            h2, c = _exact_pair(h2, dx, i, j)
            if c < best - 1e-14:
                h = h2
                best = c
                improved = True
        if not improved:
            break
    return h, best

    else: h = 0.5 + 0.2*rng.randn(n)
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, alpha):
    corr = _corr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500.0, 50.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, dx, bounds, eq_con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    try:
        res = minimize(_smooth_obj, h, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=[eq_con],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _exact_pair(h, dx, i, j):
    N = len(h)
    corr0 = _corr(h)
    eps = 1e-9
    hp = h.copy()
    hp[i] += eps; hp[j] -= eps
    a = (_corr(hp) - corr0) / eps
    d_lo = max(-h[i], h[j] - 1.0) + 1e-12
    d_hi = min(1.0 - h[i], h[j]) - 1e-12
    if d_lo >= d_hi:
        return float(np.max(corr0) * dx)
    gr = 0.6180339887498949
    lo, hi = d_lo, d_hi
    best_d, best_v = 0.0, float(np.max(corr0))
    for _ in range(60):
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        f1 = float(np.max(corr0 + a * x1))
        f2 = float(np.max(corr0 + a * x2))
        if f1 < f2:
            hi = x2
        else:
            lo = x1
        mid = 0.5 * (lo + hi)
        fm = float(np.max(corr0 + a * mid))
        if fm < best_v:
            best_v = fm; best_d = mid
    h[i] += best_d; h[j] -= best_d
    return best_v * dx

def _pair_sweep(h, n, dx, rng, deadline, max_iter):
    best = _c5(h, dx)
    h = h.copy()
    it = 0
    stall = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            it -= 1; continue
        c = _exact_pair(h, dx, i, j)
        if c < best - 1e-14:
            best = c
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                break
    return h, best

def _pair_cycling(h, n, dx, deadline):
    best = _c5(h, dx)
    h = h.copy()
    pairs = []
    for i in range(n):
