def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h


def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _slsqp_opt(h, n, alpha, maxiter):
    target = n / 2.0

    def obj(x):
        return _smooth_max(_corr(x), alpha)

    def constr(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h


def _lbfgs_opt(h, n, alpha, pen, maxiter):
    target = n / 2.0

    def obj(x):
        return _smooth_max(_corr(x), alpha) + pen * (np.sum(x) - target) ** 2

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return _repair(res.x, n)
    except Exception:
        return h


def _fine_search(h, n, budget_end, rng, use_anneal=True):
    h = h.copy()
    dx = 2.0 / n
    best = _c5(h, n)
    current = best
    max_iter = 500000
    t_init = 0.008
    t_min = 0.00005
    steps = [0.0005, 0.001, 0.002, 0.005, 0.01]
    triple_steps = [0.003, 0.008, 0.015]
    no_improve = 0

    for it in range(max_iter):
        if time.time() > budget_end:
            break
        frac = min(1.0, it / (max_iter * 0.8))
        temp = t_init * (t_min / t_init) ** frac if use_anneal else 0.0

        r = rng.rand()
        if r < 0.75:
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = h[i], h[j]
            accepted = False
            for step in steps:
                if accepted:
                    break
                for sign in (1, -1):
                    ni, nj = hi + sign * step, hj - sign * step
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    new_c5 = _c5(h, n)
                    delta = new_c5 - current
                    if delta < -1e-15 or (use_anneal and rng.rand() < np.exp(-max(delta, 0) / max(temp, 1e-15))):
                        current = new_c5
                        hi, hj = ni, nj
                        accepted = True
                        if current < best:
                            best = current
                            no_improve = 0
                    else:
                        h[i], h[j] = hi, hj
            if not accepted:
                no_improve += 1
        elif r < 0.92:
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = idx[0], idx[1], idx[2]
            hi, hj, hk = h[i], h[j], h[k]
            accepted = False
            for step in triple_steps:
                if accepted:
                    break
                for sgn in (1, -1):
                    ni, nj, nk = hi + sgn * step, hj - sgn * step * 0.5, hk - sgn * step * 0.5
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1 or nk < 0 or nk > 1:
                        continue
                    h[i], h[j], h[k] = ni, nj, nk
                    new_c5 = _c5(h, n)
                    delta = new_c5 - current
                    if delta < -1e-15 or (use_anneal and rng.rand() < np.exp(-max(delta, 0) / max(temp, 1e-15))):
                        current = new_c5
                        accepted = True
                        if current < best:
                            best = current
                            no_improve = 0
                    else:
                        h[i], h[j], h[k] = hi, hj, hk
            if not accepted:
                no_improve += 1
        else:
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            new_c5 = _c5(h, n)
            if new_c5 < current or (use_anneal and rng.rand() < np.exp(-(new_c5 - current) / max(temp, 1e-15))):
                current = new_c5
                if current < best:
                    best = current
                    no_improve = 0
            else:
                h[i], h[j] = hi, hj
            no_improve += 1

        if no_improve > 20000 and it > 50000:
            break

    return h, best


def _block_perturb(h, n, rng, nblocks):
    h = h.copy()
    perm = rng.permutation(n)
    boundaries = np.sort(rng.choice(np.arange(1, n), size=min(nblocks, n - 1), replace=False))
    new_h = np.zeros(n)
    prev = 0
    for b in list(boundaries) + [n]:
        seg = h[perm[prev:b]]
        new_h[perm[prev:b]] = seg
        prev = b
    return _repair(new_h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except NameError:
        n = 70
        h = np.full(n, 0.5)

    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _c5(h, n)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [500, 2000, 8000, 30000]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = _slsqp_opt(h, n, alpha, 250)
        c5_new = _c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

    # Phase 2: L-BFGS penalty
    for pen in [100, 1000, 5000]:
        if time.time() - t0 > budget_s - 180:
            break
        h_new = _lbfgs_opt(h, n, 3000, pen, 400)
        c5_new = _c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

    # Phase 3: Main fine search with annealing
    p3_end = t0 + budget_s * 0.65
    h3, c5_3 = _fine_search(best_h, n, p3_end, rng, use_anneal=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3

    # Phase 4: SLSQP refinement on improved solution
    for alpha in [10000, 50000]:
        if time.time() - t0 > budget_s - 80:
            break
        h_new = _slsqp_opt(best_h, n, alpha, 400)
        c5_new = _c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Very fine search (no annealing, tiny steps)
    p5_end = t0 + budget_s - 30
    h5, c5_5 = _fine_search(best_h, n, p5_end, rng, use_anneal=False)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5

    # Phase 6: Block perturbation + re-optimize (diversity)
    for _ in range(3):
        if time.time() - t0 > budget_s - 15:
            break
        h_pert = _block_perturb(best_h, n, rng, 4)
        h_new = _slsqp_opt(h_pert, n, 5000, 200)
        h_new, c5_t = _fine_search(h_new, n, t0 + budget_s - 10, rng, use_anneal=True)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_new

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        h_fin = _slsqp_opt(best_h, n, 80000, 200)
        if _c5(h_fin, n) < best_c5:
            best_c5 = _c5(h_fin, n)
            best_h = h_fin

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
