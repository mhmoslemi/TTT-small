import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _con_fun(x, target):
    return np.sum(x) - target

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (target,)}]
    try:
        res = minimize(_obj, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _pair_cd(h, n, dx, deadline, max_sweeps=15, iters=35):
    best_c5 = _c5(h, dx)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                a, b = lo, hi
                for _ in range(iters):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    h[i], h[j] = m1, s - m1
                    f1 = float(np.max(_corr(h)))
                    h[i], h[j] = m2, s - m2
                    f2 = float(np.max(_corr(h)))
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                t_opt = (a + b) / 2.0
                h[i], h[j] = t_opt, s - t_opt
                new_c5 = _c5(h, dx)
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    improved = True
        if not improved and sweep >= 2:
            break
    return h, best_c5

def _rand_pair_cd(h, n, dx, rng, deadline, n_trials=50000):
    best_c5 = _c5(h, dx)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-12:
            continue
        a, b = lo, hi
        for _ in range(30):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            h[i], h[j] = m1, s - m1
            f1 = float(np.max(_corr(h)))
            h[i], h[j] = m2, s - m2
            f2 = float(np.max(_corr(h)))
            if f1 < f2:
                b = m2
            else:
                a = m1
        t_opt = (a + b) / 2.0
        new_c5_val = float(np.max(_corr(h))) * dx
        h[i], h[j] = t_opt, s - t_opt
        nc = _c5(h, dx)
        if nc < best_c5 - 1e-14:
            best_c5 = nc
    return h, best_c5

def _project(h, S):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgradient(h, n, dx, deadline):
    S = n / 2.0
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    lr = 0.005
    prev_c5 = best_c5
    for it in range(20000):
        if time.time() > deadline:
            break
        c = _corr(h)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        g = np.ones(n)
        if lag >= 0:
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
        else:
            L = -lag
            g[L:] -= h[:n - L]
            g[:n - L] -= h[L:]
        h = _project(h - lr * g, S)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if c5 >= prev_c5:
            lr *= 0.95
        prev_c5 = c5
        if it > 500:
            lr = max(lr * 0.9995, 0.0001)
    if _c5(best_h, dx) < best_c5:
        return best_h, _c5(best_h, dx)
    return h, best_c5

def _sa(h, n, dx, rng, deadline, T0=0.0005):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    cur_c5 = best_c5
    n_iter = 0
    while time.time() < deadline:
        n_iter += 1
        frac = min(n_iter / 8000.0, 1.0)
        T = T0 * (0.001) ** frac
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-12:
            continue
        t_new = rng.uniform(lo, hi)
        oti, otj = h[i], h[j]
        h[i], h[j] = t_new, s - t_new
        nc5 = _c5(h, dx)
        delta = nc5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[i], h[j] = oti, otj
    return best_h, best_c5

def _block_init(n, rng):
    h = np.zeros(n)
    pattern = rng.permutation(n)
    for k in range(n // 2):
        h[pattern[k]] = 1.0
    return _repair(h, n)
