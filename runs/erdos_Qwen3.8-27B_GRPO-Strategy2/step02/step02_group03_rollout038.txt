def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                add = min(diff, max(0.0, 1.0 - h[idx]))
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, max(0.0, h[idx]))
                h[idx] -= sub
                diff += sub
    return h

def _sobj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sojb(x, alpha, pen, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += pen * (np.sum(x) - target) ** 2
    return val

def _scn(x, alpha, target):
    return np.sum(x) - target

        return h

                    c5 = _compute_c5(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if s > 1e-12:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                add = min(diff, max(0.0, 1.0 - h[idx]))
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, max(0.0, h[idx]))
                h[idx] -= sub
                diff += sub
    return h

def _sobj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sojb(x, alpha, pen, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += pen * (np.sum(x) - target) ** 2
    return val

def _scn(x, alpha, target):
    return np.sum(x) - target

def _slsqp(h, n, alpha, maxit):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scn, 'args': (alpha, target)}]
    try:
        res = minimize(_sobj, h, args=(alpha, target), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, n, alpha, pen, maxit):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sojb, h, args=(alpha, pen, target), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _sys_refine(h, n, dx, deadline):
    h = h.copy()
    best = _compute_c5(h, dx)
    steps = [0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.0025, 0.0015, 0.001, 0.0006, 0.0004, 0.00025, 0.00015, 0.0001]
    for step in steps:
        if time.time() > deadline:
            break
        passes_at_step = 0
        while time.time() < deadline and passes_at_step < 3:
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    if time.time() > deadline:
                        return h, best
                    for d in (step, -step):
                        hi, hj = h[i], h[j]
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c5 = _compute_c5(h, dx)
                        if c5 < best - 1e-14:
                            best = c5
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
            passes_at_step += 1
            if not improved:
                break
    return h, best

def _random_refine(h, n, dx, deadline, rng):
    h = h.copy()
    best = _compute_c5(h, dx)
    temp = 0.008
    for _ in range(300000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1, 1]) * rng.uniform(0.001, 0.025)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5 = _compute_c5(h, dx)
        if c5 < best:
            best = c5
        elif rng.rand() > np.exp(-(c5 - best) / max(temp, 1e-15)):
            h[i], h[j] = hi, hj
        temp = max(temp * 0.9997, 0.0003)
    return h, best

def _gradient_refine(h, n, dx, deadline, rng):
    h = h.copy()
    best = _compute_c5(h, dx)
    alpha = 3000.0
    for it in range(300):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        exp_s = np.exp(np.clip(alpha * (corr - m), -500, 500))
        w = exp_s / exp_s.sum()
        eps = 1e-5
        grad = np.zeros(n)
        for i in range(n):
            h[i] += eps
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] -= 2.0 * eps
            c2 = np.correlate(h, 1.0 - h, mode='full')
            h[i] += eps
            grad[i] = np.dot(w, c1 - c2) / (2.0 * eps)
        order = np.argsort(grad)
        found = False
        for trial in range(8):
            idx = order[trial]
            d = 0.005 * rng.uniform(0.5, 1.5)
            j_order = np.argsort(np.abs(grad - grad[idx]))
            jdx = j_order[(trial + 1) % (n - 1)]
            if jdx == idx:
                jdx = j_order[(trial + 2) % (n - 1)]
            hi, hj = h[idx], h[jdx]
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                ni, nj = hi - d, hj + d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
            h[idx], h[jdx] = ni, nj
            c5 = _compute_c5(h, dx)
            if c5 < best - 1e-14:
                best = c5
                found = True
            else:
                h[idx], h[jdx] = hi, hj
    return h, best

def _make_start(n, seed, kind):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
    else:
        h = rng.uniform(0, 1, n)
    return _repair(h, n)
