def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

                h[idx] += add
                diff -= add
            else:

                diff += sub

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

            else:

                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:

def _proj(v, target):

    for k in range(n):
        for idx in range(2 * n - 1):
            ell = idx - (n - 1)
            v = 0.0
            kp, km = k + ell, k - ell
            if 0 <= kp < n:
                v += 1.0 - h[kp]
            if 0 <= km < n:
                v -= h[km]
            G[k, idx] = v
    return G

    for alpha in alphas:

        m = np.zeros(n)
        v = np.zeros(n)
        t = 0

            t += 1

            cmax = c.max()

            w /= w.sum()
