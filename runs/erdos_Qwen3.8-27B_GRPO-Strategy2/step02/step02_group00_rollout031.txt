        ds = np.array([0.001, 0.003, 0.008, 0.02, 0.05, -0.001, -0.003, -0.008, -0.02, -0.05])
    stuck = 0
    while time.time() < t_limit:
        imp = False
        for i in range(n):
            if time.time() > t_limit or imp:
                break
            vi = best[i]
            for d in ds:
                if imp:
                    break
                ni = vi + d
                if ni < 0 or ni > 1:
                    continue
                need = -d
                ord_ = np.argsort(best) if need > 0 else np.argsort(-best)
                for j in ord_[:jmax]:
                    if j == i:
                        continue
                    nj = best[j] + need
                    if nj < 0 or nj > 1:
                        continue
                    best[i], best[j] = ni, nj
                    tc = _c5(best, dx)
                    if tc < bc - 1e-14:
                        bc = tc
                        imp = True
                        break
                    best[i], best[j] = vi, nj - need
        if not imp:
            stuck += 1
            if stuck >= 5:
                break
        else:
            stuck = 0
    return best, bc

def _sa(h, n, dx, t_limit, t0, seed):
    rng = np.random.RandomState(seed)
    cur = h.copy()
    best = h.copy()
    bc = _c5(best, dx)
    T0v, T1v = 0.006, 0.00001
    N = 5000000
    for it in range(N):
        if time.time() > t_limit:
            break
        f = min(it / N, 1.0)
        T = T0v * (T1v / T0v) ** f
        dm = 0.07 * (1.0 - 0.6 * f)
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = (rng.rand() - 0.5) * 2.0 * dm
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        tc = _c5(cur, dx)
        if tc <= bc + T:
            if tc < bc:
                bc, best = tc, cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
    return best, bc

def _critical_lags(h, n, frac=0.995):
    c = _corr(h)
    mx = float(c.max())
    thresh = mx * frac
    return np.where(c >= thresh)[0]

def _perturb(h, rng, sig):
    return _repair(h + rng.randn(len(h)) * sig, len(h))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bc = _c5(h, dx)

    # Phase 1: SLSQP with multiple objectives (10%)
    t1 = t0 + budget_s * 0.10
    for a in [300, 800, 2000, 5000]:
        if time.time() > t1:
            break
        h, c = _slsqp(h, n, dx, _obj_lse, (a, n), 200)
        if c < bc:
            bc, h = c, h
    for m in [10, 5, 3, 2]:
        if time.time() > t1:
            break
        h, c = _slsqp(h, n, dx, _obj_topm, (m, n), 200)
        if c < bc:
            bc, h = c, h
    # Lag-focused SLSQP
    if time.time() < t1:
        lags = _critical_lags(h, n)
        h, c = _slsqp(h, n, dx, _obj_lagsum, (lags, n), 200)
        if c < bc:
            bc, h = c, h

    # Phase 2: SA (28%)
    t2 = t0 + budget_s * 0.38
    h, c = _sa(h, n, dx, t2, t0, seed)
    if c < bc:
        bc, h = c, h

    # Phase 3: Coarse local search (22%)
    t3 = t0 + budget_s * 0.60
    h, c = _local(h, n, dx, t3, t0, fine=False, jmax=15)
    if c < bc:
        bc, h = c, h

    # Phase 4: SLSQP + local iteration cycle (15%)
    t4 = t0 + budget_s * 0.75
    for cyc in range(6):
        if time.time() > t4:
            break
        lags = _critical_lags(h, n, 0.99)
        h, c = _slsqp(h, n, dx, _obj_lagsum, (lags, n), 150)
        if c < bc:
            bc = c
        h, c = _local(h, n, dx, min(30, t4 - time.time()), t0, fine=False, jmax=10)
        if c < bc:
            bc, h = c, h

    # Phase 5: Multi-restart (12%)
    t5 = t0 + budget_s * 0.87
    for tr in range(12):
        if time.time() > t5:
            break
        sig = 0.02 + 0.08 * (1.0 - tr / 12.0)
        ht = _perturb(h, rng, sig)
        ht, ct = _slsqp(ht, n, dx, _obj_lse, (2000, n), 120)
        ht, ct = _local(ht, n, dx, min(15, t5 - time.time()), t0, fine=True, jmax=8)
        if ct < bc:
            bc, h = ct, ht

    # Phase 6: Fine local + final SLSQP (10%)
    t6 = t0 + budget_s * 0.96
    h, c = _local(h, n, dx, t6, t0, fine=True, jmax=20)
    if c < bc:
        bc, h = c, h
    if time.time() < t0 + budget_s - 10:
        h, c = _slsqp(h, n, dx, _obj_lse, (8000, n), 300)
        if c < bc:
            bc, h = c, h
    if time.time() < t0 + budget_s - 5:
        h, c = _local(h, n, dx, t0 + budget_s - 3, t0, fine=True, jmax=15)
        if c < bc:
            bc, h = c, h

    h = _repair(h, n)
    bc = _c5(h, dx)
    return (h, bc, n)
