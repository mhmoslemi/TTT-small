        order = np.argsort(-x) if diff > 0 else np.argsort(x)
        for k in order:
            a = min(abs(diff), (1.0 - x[k]) if diff > 0 else x[k])
            x[k] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-14:
                break
    return x

def _sobj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -400.0, 400.0)
    return m + np.log(float(np.exp(s).sum())) / a

def _sgrad(z, a):
    n = len(z)
    c = np.correlate(z, 1.0 - z, mode='full')
    m = float(c.max())
    s = np.clip(a * (c - m), -400.0, 400.0)
    e = np.exp(s)
    w = e / float(e.sum())
    g = np.zeros(n)
    wk = w[n - 1]
    g += wk * (1.0 - 2.0 * z)
    for L in range(1, n):
        wp = w[n - 1 + L]
        wm = w[n - 1 - L]
        if wp < 1e-12 and wm < 1e-12:
            continue
        sl = n - L
        g[0:sl] += wp * (1.0 - z[L:n]) - wm * z[L:n]
        g[L:n] += wm * (1.0 - z[0:sl]) - wp * z[0:sl]
    return g

def _fista(h0, alpha, target, iters):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    bx = x.copy()
    bc = _c5(x, dx)
    step = 1.0 / (4.0 * alpha * n + 4.0)
    for _ in range(iters):
        gz = _sgrad(z, alpha)
        y = z - step * gz
        xn = _proj(y, target)
        for _ in range(5):
            q = _sobj(z, alpha) + float(np.dot(gz, xn - z)) + 0.5 / step * float(np.dot(xn - z, xn - z))
            if _sobj(xn, alpha) <= q + 1e-10:
                break
            step *= 0.5
            y = z - step * gz
            xn = _proj(y, target)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = _proj(xn + (t - 1.0) / tn * (xn - x), target)
        x = xn
        t = tn
        c = _c5(x, dx)
        if c < bc:
            bc = c
            bx = x.copy()
    return bx, bc

def _pair_cd(h, dx, n, iters, rng, max_lag=None):
    bc = _c5(h, dx)
    h = h.copy()
    for _ in range(iters):
        i = rng.randint(n)
        if max_lag is not None:
            off = rng.randint(1, max_lag + 1) * rng.choice([-1, 1])
            j = (i + off) % n
        else:
            j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        lo_d = max(-hi, hj - 1.0)
        hi_d = min(1.0 - hi, hj)
        if hi_d - lo_d < 1e-10:
            continue
        best_d = 0.0
        best_c = bc
        for d in np.linspace(lo_d, hi_d, 10):
            h[i] = hi + d
            h[j] = hj - d
            c = _c5(h, dx)
            if c < best_c:
                best_c = c
                best_d = d
        if best_c < bc - 1e-14:
            bc = best_c
        else:
            best_d = 0.0
        h[i] = hi + best_d
        h[j] = hj - best_d
    return h, bc

def _sim_anneal(h, dx, n, iters, rng):
    bc = _c5(h, dx)
    h = h.copy()
    for it in range(iters):
        T = 0.05 * 0.02 ** (it / max(iters, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        d = rng.uniform(-0.2, 0.2)
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c = _c5(h, dx)
        if c < bc:
            bc = c
        elif rng.rand() < np.exp(-(c - bc) / max(T, 1e-12)):
            pass
        else:
            h[i], h[j] = hi, hj
    return h, bc

def _cons_eq(x, tgt):
    return float(np.sum(x)) - tgt

def _rem(t0, budget_s):
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    tgt0 = n0 / 2.0
    h0 = _proj(np.array(initial_h_values, dtype=float), tgt0)
    bh = h0.copy()
    bc = _c5(bh, dx0)
    bn = n0

    for a in [80, 200, 600, 2000, 5000, 12000]:
        if _rem(t0, budget_s) < 40:
            break
        h, c = _fista(bh, a, tgt0, 2000)
        if c < bc:
            bc, bh = c, h

    if _rem(t0, budget_s) > 30:
        h, c = _pair_cd(bh, dx0, n0, 40000, rng)
        if c < bc:
            bc, bh = c, h

    if _rem(t0, budget_s) > 30:
        h, c = _sim_anneal(bh, dx0, n0, 50000, rng)
        if c < bc:
            bc, bh = c, h
        if _rem(t0, budget_s) > 20:
            h, c = _pair_cd(bh, dx0, n0, 25000, rng)
            if c < bc:
                bc, bh = c, h

    if _rem(t0, budget_s) > 20:
        h, c = _pair_cd(bh, dx0, n0, 30000, rng, max_lag=max(2, n0 // 8))
        if c < bc:
            bc, bh = c, h

    for n in [100, 140, 200, 300]:
        if _rem(t0, budget_s) < 90:
            break
        dx = 2.0 / n
        tgt = n / 2.0
        h_interp = _proj(np.interp(np.arange(n), np.arange(bn), bh), tgt)
        h_start = _proj(h_interp + 0.08 * rng.randn(n), tgt)
        h, c = _fista(h_start, 500, tgt, 2500)
        if c < bc:
            bc, bh, bn = c, h, n
            if _rem(t0, budget_s) > 50:
                h, c = _pair_cd(bh, dx, n, 20000, rng)
                if c < bc:
                    bc, bh, bn = c, h, n
            if _rem(t0, budget_s) > 30:
                h, c = _sim_anneal(bh, dx, n, 30000, rng)
                if c < bc:
                    bc, bh, bn = c, h, n
                if _rem(t0, budget_s) > 20:
                    h, c = _pair_cd(bh, dx, n, 15000, rng)
                    if c < bc:
                        bc, bh, bn = c, h, n

    for sigma in [0.06, 0.12, 0.2, 0.35, 0.1, 0.5, 0.15, 0.25]:
        if _rem(t0, budget_s) < 50:
            break
        dx = 2.0 / bn
        hp = _proj(bh + sigma * rng.randn(bn), bn / 2.0)
        h, c = _fista(hp, 400, bn / 2.0, 2000)
        if c < bc:
            bc, bh = c, h
            if _rem(t0, budget_s) > 20:
                h, c = _pair_cd(bh, dx, bn, 10000, rng)
                if c < bc:
                    bc, bh = c, h

    if _rem(t0, budget_s) > 25:
        bnds = [(0.0, 1.0)] * bn
        try:
            r = minimize(_sobj, bh, args=(8000,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _cons_eq, 'args': (bn / 2.0,)}],
                         options={'maxiter': 500, 'ftol': 1e-15})
            hf = _proj(r.x, bn / 2.0)
            cf = _c5(hf, 2.0 / bn)
            if cf < bc:
                bc, bh = cf, hf
        except Exception:
            pass

    if _rem(t0, budget_s) > 15:
        h, c = _pair_cd(bh, 2.0 / bn, bn, 25000, rng)
        if c < bc:
            bc, bh = c, h

    if _rem(t0, budget_s) > 20:
        h, c = _sim_anneal(bh, 2.0 / bn, bn, 30000, rng)
        if c < bc:
            bc, bh = c, h
        if _rem(t0, budget_s) > 10:
            h, c = _pair_cd(bh, 2.0 / bn, bn, 15000, rng)
            if c < bc:
                bc, bh = c, h

    if _rem(t0, budget_s) > 10:
        h, c = _pair_cd(bh, 2.0 / bn, bn, 10000, rng, max_lag=max(2, bn // 10))
        if c < bc:
            bc, bh = c, h

    bh = _proj(bh, bn / 2.0)
    bc = _c5(bh, 2.0 / bn)
    return (bh, bc, bn)
