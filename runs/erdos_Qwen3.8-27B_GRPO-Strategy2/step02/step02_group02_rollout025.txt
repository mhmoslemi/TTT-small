def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0-h, mode='full')))*dx

def _sobj(h, a):
    c = np.correlate(h, 1.0-h, mode='full')
    m = float(c.max())
    s = np.clip(a*(c-m), -500.0, 500.0)
    return m + np.log(float(np.exp(s).sum()))/a

def _sgrad(h, a):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = float(c.max())
    s = np.clip(a*(c-m), -500.0, 500.0)
    e = np.exp(s)
    w = e/float(e.sum())
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14: continue
        lag = k-(n-1)
        if lag >= 0:
            g[:n-lag] += wk*(1.0-h[lag:])
            g[lag:] -= wk*h[:n-lag]
        else:
            L = -lag
            g[L:] += wk*(1.0-h[:n-L])
            g[:n-L] -= wk*h[L:]
    return g

def _fista(h0, a, tgt, iters, t0, budget):
    n = len(h0); dx = 2.0/n
    x = h0.copy(); z = h0.copy(); t = 1.0
    bx = x.copy(); bc = _c5(x, dx)
    step = 1.0/(4.0*a*n+4.0)
    for it in range(iters):
        if time.time()-t0 > budget-10: break
        gz = _sgrad(z, a)
        for _bt in range(8):
            y = z - step*gz
            xn = _proj(y, tgt)
            fn = _sobj(xn, a)
            qd = _sobj(z, a) + float(np.dot(gz, xn-z)) + 0.5/step*float(np.dot(xn-z, xn-z))
            if fn <= qd + 1e-10: break
            step *= 0.5
        tn = 0.5*(1.0+np.sqrt(1.0+4.0*t*t))
        z = xn + (t-1.0)/tn*(xn-x)
        x = xn; t = tn
        if it % 20 == 0:
            c = _c5(x, dx)
            if c < bc: bc = c; bx = x.copy()
    c = _c5(x, dx)
    if c < bc: bc = c; bx = x.copy()
    return bx, bc

def _cd(h, dx, n, rng, time_limit):
    bc = _c5(h, dx)
    h = h.copy()
    # Adaptive step sizes
    base_steps = np.array([1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0, 30.0, 50.0]) * 0.005
    steps = np.concatenate([base_steps, -base_steps])
    
    t_end = time.time() + time_limit
    improved = True
    while improved and time.time() < t_end:
        improved = False
        npass = max(1, int(30 * (t_end - time.time()) / max(0.1, time_limit)))
        for _ in range(n * 4 * npass):
            if time.time() > t_end: break
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            hi0, hj0 = h[i], h[j]
            for d in steps:
                ni, nj = hi0+d, hj0-d
                if ni < -1e-15 or ni > 1+1e-15 or nj < -1e-15 or nj > 1+1e-15: continue
                h[i] = min(max(ni, 0), 1); h[j] = min(max(nj, 0), 1)
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc = c; improved = True; break
                h[i] = hi0; h[j] = hj0
            else:
                h[i] = hi0; h[j] = hj0
    return h, bc

def _slsqp_con(x, t):
