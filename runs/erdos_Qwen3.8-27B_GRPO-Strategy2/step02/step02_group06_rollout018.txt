```python
def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    nc = 2 * n - 1
    k = np.arange(-(n - 1), n)
    g = np.zeros(nc)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _smax_grad(h, alpha):

    m = corr.max()
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:

        w[corr == m] = 1.0 / np.sum(corr == m)
    else:
        w /= ws
    n = len(h)
    grad = np.zeros(n)
    for p in range(n):
        g = _g_vec(h, p, n)
        grad[p] = np.dot(w, g)
    return grad, m

    m = corr.max()
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    d_min = max(-hi, hj - 1.0)

    gd = _g_vec(h, i, n) - _g_vec(h, j, n)

        hi = min(d_max, d_best + fine_wid)
        ds2 = np.linspace(lo, hi, n_fine)
        nc2 = corr[None, :] + ds2[:, None] * gd[None, :]

        bi2 = np.argmin(c5s2)
        d_best = ds2[bi2]
    h[i] = hi + d_best
    h[j] = hj - d_best
    corr += d_best * gd
```

```python
def _lp_direction(h, n, target):
    corr = _corr(h)
    nc = len(corr)
    # Build LP: min t s.t. corr_k + (h-h0)^T g_k <= t, sum(h)=target, 0<=h<=1
    # Variables: h (n vars) + t (1 var) = n+1 vars
    # Constraints:
    #   For each k: sum_p g_k[p] * h[p] <= t - corr[k]
    #   sum(h) = target
    #   0 <= h[p] <= 1
    
    # g_k[p] = d corr[k] / d h[p] = _g_vec(h, p, n)[k]
    # So the constraint is: sum_p g_vec(p)[k] * h[p] - t <= -corr[k]
    
    from scipy.optimize import linprog
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_p gk[p] * h[p] - t <= -corr[k]
    A = np.zeros((nc, n + 1))
    b = np.zeros(nc)
    for p in range(n):
        g = _g_vec(h, p, n)
        A[:, p] = g
    A[:, n] = -1.0  # coefficient of t
    b = -corr
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds_lp = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x[:n]
    return h
```
