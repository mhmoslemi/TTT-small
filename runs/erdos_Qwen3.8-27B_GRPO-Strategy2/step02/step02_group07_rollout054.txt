import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            a = min(abs(diff), room)
            h[idx] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-14:
                break
    return h

def _coefs(h, n):
    shifts = np.arange(2 * n - 1) - (n - 1)
    P = np.arange(n)[:, None]
    S = shifts[None, :]
    PS = P + S
    PM = P - S
    v1 = (PS >= 0) & (PS < n)
    v2 = (PM >= 0) & (PM < n)
    t1 = np.where(v1, 1.0 - h[np.clip(PS, 0, n - 1)], 0.0)
    t2 = np.where(v2, h[np.clip(PM, 0, n - 1)], 0.0)
    return t1 - t2

def _minimax_delta(corr, A, lo, hi):
    if lo >= hi:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(corr + A * c)
    fd = np.max(corr + A * d)
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + A * c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(corr + A * d)
    return (a + b) / 2.0

    C = _coefs(h, n)
    hi_c5 = float(np.max(corr) * dx)

        for j in range(i + 1, n):
            lo = -h[j]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            A = C[i] - C[j]
            d = _minimax_delta(corr, A, lo, hi)
            if abs(d) < 1e-13:
                continue
            h[i] += d
            h[j] -= d
            corr = np.correlate(h, 1.0 - h, mode='full')
            nc = float(np.max(corr) * dx)
            if nc < hi_c5 - 1e-14:
                hi_c5 = nc
                C = _coefs(h, n)
                improved = True
            else:
                h[i] -= d
                h[j] += d
    return h, hi_c5, improved

def _smooth_obj(x, alpha, target):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha, target), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            a = min(abs(diff), room)
            h[idx] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-14:
                break
    return h

def _coefs(h, n):
    shifts = np.arange(2 * n - 1) - (n - 1)
    P = np.arange(n)[:, None]
    S = shifts[None, :]
    PS = P + S
    PM = P - S
    v1 = (PS >= 0) & (PS < n)
    v2 = (PM >= 0) & (PM < n)
    t1 = np.where(v1, 1.0 - h[np.clip(PS, 0, n - 1)], 0.0)
    t2 = np.where(v2, h[np.clip(PM, 0, n - 1)], 0.0)
    return t1 - t2

def _minimax_delta(corr, A, lo, hi):
    if hi - lo < 1e-15:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(corr + A * c))
    fd = float(np.max(corr + A * d))
    for _ in range(55):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(corr + A * c))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = float(np.max(corr + A * d))
    return (a + b) * 0.5

def _cd_pass(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = _coefs(h, n)
    hi_c5 = float(np.max(corr) * dx)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-14:
                continue
            A = C[i] - C[j]
            d = _minimax_delta(corr, A, lo, hi)
            if abs(d) < 1e-14:
                continue
            h[i] += d
            h[j] -= d
            corr = np.correlate(h, 1.0 - h, mode='full')
            nc = float(np.max(corr) * dx)
            if nc < hi_c5 - 1e-15:
                hi_c5 = nc
                C = _coefs(h, n)
                improved = True
            else:
                h[i] -= d
                h[j] += d
    return h, hi_c5, improved

def _smooth_obj(x, alpha, target):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha, target),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _cbound_cons(x, target):
    return x.sum() - target

def _lgbfsg_pass(h, n, target, t_limit, t0):
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [('eq', _cbound_cons, target)]
    best = h.copy()
    best_c = _c5(h, 2.0 / n)
    while time.time() - t0 < t_limit:
        try:
            res = minimize(_smooth_obj, h, args=(2000, target), method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16})
            h = _repair(res.x, n)
            c = _c5(h, 2.0 / n)
            if c < best_c:
                best_c, best = c, h.copy()
            else:
                break
        except Exception:
            break
    return best, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

        return time.time() - t0

        c = _c5(h, dx)

    # Phase 2: Intensive coordinate descent (exact 1D minimization)
    h = best_h.copy()
    stall = 0
    while elapsed() < budget_s - 300:
        h, c, imp = _cd_pass(h, n, dx)
        if c < best_c5:

            stall = 0
        else:
            stall += 1
        if not imp:

                break
            # Nudge to escape plateaus
            h += rng.randn(n) * 0.001
            h = _repair(h, n)
            stall = 0

    # Phase 3: SLSQP polish on best
    if elapsed() < budget_s - 200:
        h = _slsqp(best_h, n, target, 2000, 700)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    h = best_h.copy()
    stall = 0

            best_c5, best_h = c, h.copy()

        else:
            stall += 1

    # Phase 5: Perturbation restarts
    sigmas = [0.04, 0.08, 0.12, 0.18, 0.06, 0.25]
    for sig in sigmas:

            break
        hp = best_h + sig * rng.randn(n)

        h = _slsqp(hp, n, target, 300, 300)
        h2, c2, _ = _cd_pass(h, n, dx)
        c2 = _c5(h2, dx)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
            h3, c3, _ = _cd_pass(best_h.copy(), n, dx)
            c3 = _c5(h3, dx)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()

    # Phase 6: Final CD until time runs out
    if elapsed() < budget_s - 20:
        h = best_h.copy()
        while elapsed() < budget_s - 10:
            h, c, imp = _cd_pass(h, n, dx)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
            if not imp:

    # Phase 7: Very final SLSQP
    if elapsed() < budget_s - 3:
        h = _slsqp(best_h, n, target, 5000, 300)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
