def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)

                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

_slsqp_alpha = 5000.0
_slsqp_sum = 35.0

def _project_box_sum(h, n):
    S = n / 2.0
    lo, hi = h.min() - 1.0, h.max()
    for _ in range(50):
        tau = (lo + hi) * 0.5
        proj = np.clip(h - tau, 0.0, 1.0)
        if proj.sum() > S:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _c5_val(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def _smooth_grad(h, dx, alpha, top_k):
    c = np.correlate(h, 1.0 - h, 'full')
    n = len(h)
    nc = len(c)
    if top_k is not None and top_k < nc:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_sel = c[idx]
        s = alpha * (c_sel - c_sel.max())
        w = np.exp(s)
        w /= w.sum()
    else:
        idx = np.arange(nc)
        s = alpha * (c - c.max())
        s = np.clip(s, -500, 0)
        w = np.exp(s)
        w /= w.sum()
    val = float((c[idx] * w).sum() * dx)
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        lag = int(idx[ii]) - (n - 1)
        wk = w[ii]
        j = np.arange(n)
        vp = (j + lag >= 0) & (j + lag < n)
        vm = (j - lag >= 0) & (j - lag < n)
        g[vp] += wk * (1.0 - h[j[vp] + lag])
        g[vm] -= wk * h[j[vm] - lag]
    g *= dx
    return val, g, c, w, idx

def _slsqp_obj(x):
    c = np.correlate(x, 1.0 - x, 'full')
    m = c.max()
    s = _slsqp_alpha * (c - m)
    s = np.clip(s, -500, 0)
    return m + float(np.log(np.sum(np.exp(s)))) / _slsqp_alpha

def _cons_eq(x):
    return np.sum(x) - _slsqp_sum

def run(seed=42, budget_s=1100, **kwargs):
    global _slsqp_alpha, _slsqp_sum
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    _slsqp_sum = n / 2.0

    h = _project_box_sum(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5_val(best_h, dx)

    def elapsed():
        return time.time() - t0

    nc = 2 * n - 1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq}]

    # Phase 1: Momentum gradient descent with schedule
    schedule = [
        (100, None, 5000),
        (300, None, 5000),
        (800, 25, 4000),
        (2000, 15, 3000),
        (5000, 10, 3000),
        (10000, 7, 2500),
        (20000, 5, 2000),
        (50000, 4, 2000),
    ]
    for alpha, tk, max_it in schedule:
        if elapsed() > budget_s - 80:
            break
        h_cur = best_h.copy()
        vel = np.zeros(n)
        lr = 0.05
        momentum = 0.9
        for it in range(max_it):
            if elapsed() > budget_s - 80:
                break
            val, g, c, w, idx = _smooth_grad(h_cur, dx, alpha, tk)
            g -= np.mean(g)
            vel = momentum * vel - lr * g
            h_new = _project_box_sum(h_cur + vel, n)
            c5_new = _c5_val(h_new, dx)
            if c5_new > best_c5 + 0.002:
                lr *= 0.6
                h_cur = h_new
            else:
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                lr = min(lr * 1.02, 0.3)
                h_cur = h_new
                if it % 500 == 0:
                    h_cur = best_h.copy()
                    vel = np.zeros(n)

    # Phase 2: SLSQP polish
    if elapsed() < budget_s - 60:
        for a_s in [3000, 10000, 30000]:
            if elapsed() > budget_s - 60:
                break
            _slsqp_alpha = float(a_s)
            try:
                res = minimize(_slsqp_obj, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-15})
                hc = _project_box_sum(res.x, n)
                c5 = _c5_val(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 3: Focused coordinate descent on top lags
    if elapsed() < budget_s - 40:
        h_cur = best_h.copy()
        c_arr = np.correlate(h_cur, 1.0 - h_cur, 'full')
        top5 = np.argpartition(c_arr, -5)[-5:]
        for _pass in range(60):
            if elapsed() > budget_s - 40:
                break
            improved = False
            val, g, c, w, idx = _smooth_grad(h_cur, dx, 50000, 3)
            g -= np.mean(g)
            order = np.argsort(-np.abs(g))[:12]
            for ii in range(len(order)):
                for jj in range(ii + 1, len(order)):
                    i, j = int(order[ii]), int(order[jj])
                    for delta in [0.0005, 0.001, 0.003, 0.01,
                                  -0.0005, -0.001, -0.003, -0.01]:
                        ni, nj = h_cur[i] + delta, h_cur[j] - delta
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h_cur[i], h_cur[j] = ni, nj
                            cc = _c5_val(h_cur, dx)
                            if cc < best_c5 - 1e-15:
                                best_c5 = cc
                                best_h = h_cur.copy()
                                improved = True
                            else:
                                h_cur[i], h_cur[j] = best_h[i], best_h[j]
            if not improved:
                h_cur = best_h.copy()
                c_arr = np.correlate(h_cur, 1.0 - h_cur, 'full')
                top5 = np.argpartition(c_arr, -5)[-5:]

    # Phase 4: Random perturbation + gradient restart
    if elapsed() < budget_s - 20:
        for _r in range(10):
            if elapsed() > budget_s - 20:
                break
            h_p = best_h + 0.005 * rng.randn(n)
            h_p = _project_box_sum(h_p, n)
            val, g, c, _, _ = _smooth_grad(h_p, dx, 20000, 5)
            g -= np.mean(g)
            for _it in range(500):
                if elapsed() > budget_s - 20:
                    break
                h_p = _project_box_sum(h_p - 0.02 * g, n)
                val, g, c, _, _ = _smooth_grad(h_p, dx, 20000, 5)
                g -= np.mean(g)
                cc = _c5_val(h_p, dx)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = h_p.copy()

    # Phase 5: Ultra-fine random search
    if elapsed() < budget_s - 5:
        h_cur = best_h.copy()
        for _ in range(8000):
            if elapsed() > budget_s - 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.005, 0.005)
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h_cur[i], h_cur[j] = ni, nj
                cc = _c5_val(h_cur, dx)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = best_h[i], best_h[j]

    best_h = _project_box_sum(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
