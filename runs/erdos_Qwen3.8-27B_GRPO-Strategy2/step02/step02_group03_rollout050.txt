import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 1000.0
_g_n = 70
_g_target = 35.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _lse_obj(x):
    c = _corr(x)
    m = c.max()
    s = np.clip(_g_alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / _g_alpha

def _con_eq(x):
    return np.sum(x) - _g_target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h = np.clip(h * (target / s), 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
    return h

def _slsqp(h, n, alpha, maxiter):
    global _g_alpha, _g_target
    _g_alpha = alpha
    _g_target = n / 2.0
    try:
        res = minimize(_lse_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _cyclic_pair(h, n, dx, t_end):
    h = h.astype(float).copy()
    best = _c5(h, dx)
    scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    for scale in scales:
        if time.time() > t_end:
            break
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                for d in (scale, -scale):
                    ni, nj = h[i] + d, h[j] - d
                    if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < best - 1e-16:
                        best = c
                    else:
                        h[i], h[j] = ni - d, nj + d
    return h, best

def _rand_pair_cd(h, n, dx, passes, t_end, rng):
    h = h.astype(float).copy()
    best = _c5(h, dx)
    for p in range(passes):
        if time.time() > t_end:
            break
        scale = [0.02, 0.01, 0.005, 0.003, 0.001][min(p // 20, 4)]
        improved = False
        temp = 0.003 / (1.0 + p * 0.15)
        for _ in range(40):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            for d in (scale, -scale, 2 * scale, -2 * scale):
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < best - 1e-16:
                    best = c
                    improved = True
                    break
                elif rng.rand() < np.exp(-(c - best) / max(temp, 1e-15)):
                    improved = True
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
        if not improved and p > 40:
            break
    return h, best

def _flat_top_slsq(h, n, dx, top_frac, t_end):
    """Penalize spread among top correlations to flatten the max."""
    c = _corr(h)
    m = c.max()
    thresh = m - top_frac * (m - c.min())
    top_mask = c >= thresh
    top_count = np.sum(top_mask)
    if top_count < 2:
        return h
    def _flat_obj(x):
        cc = _corr(x)
        mc = cc.max()
        sc = np.clip(2000.0 * (cc - mc), -400, 400)
        base = mc + np.log(np.sum(np.exp(sc))) / 2000.0
        top_idx = np.where(cc >= cc.max() - top_frac * (cc.max() - cc.min()))[0]
        var_pen = 0.5 * 100.0 * np.var(cc[top_idx])
        return base + var_pen
    def _flat_con(x):
        return np.sum(x) - n / 2.0
    try:

                       constraints=[{'type': 'eq', 'fun': _flat_con}],

        h2 = _repair(res.x, n)
        if _c5(h2, dx) < _c5(h, dx):
            return h2
    except Exception:
        pass
    return h

        h += 0.15 * rng.randn(n)
    else:
        h = rng.uniform(0, 1, n)
    return _repair(h, n)
