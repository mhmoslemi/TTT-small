import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _lse_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_opt(h, n, alpha, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    try:
        res = minimize(_lse_obj, h.copy(), args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _cd_pass(h, n, dx, n_grid, order=None):
    """One pass of coordinate descent. If order is None, use natural order."""
    hi_c5 = _c5(h, dx)
    improved = False
    n2 = h.sum()  # should be n/2
    if order is None:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    else:
        pairs = order
    for (i, j) in pairs:
        hsum = h[i] + h[j]
        lo = max(0.0, hsum - 1.0)
        hi = min(1.0, hsum)
        if hi - lo < 1e-10:
            continue
        orig_i = h[i]
        orig_j = h[j]
        best_t = orig_i
        best_c = hi_c5
        span = hi - lo
        # Coarse grid
        for t in np.linspace(lo + 1e-9, hi - 1e-9, n_grid):
            h[i] = t
            h[j] = hsum - t
            c = _c5(h, dx)
            if c < best_c:

                best_t = t
        # Fine grid around best
        if best_t != orig_i:
            lo2 = max(lo, best_t - span / n_grid)
            hi2 = min(hi, best_t + span / n_grid)
            for t in np.linspace(lo2, hi2, 7):

            h[j] = hsum - best_t
            hi_c5 = best_c
            improved = True
        else:

            h[j] = orig_j
    return h, hi_c5, improved

def _cd_loop(h, n, dx, t_limit, t_start, n_grid, max_passes=2000, reverse_every=5):
    order = None
    for p in range(max_passes):
        if time.time() - t_start > t_limit:

        if p > 0 and p % reverse_every == 0 and order is None:
            # Reverse the order for diversity
            order = [(i, j) for i in reversed(range(n)) for j in reversed(range(i + 1, n))]
        elif p > 0 and p % reverse_every == 1 and order is not None:

        h, c5, imp = _cd_pass(h, n, dx, n_grid, order)

def _gradient_pairs(h, n, dx, top_k=30):
    c = _corr(h)
    active_k = np.argsort(c)[::-1][:3]
    pairs = []
    for k in active_k:
        ik = k - (n - 1)
        if ik == 0:
            continue
        g = np.zeros(n)
        for i in range(n):
            ip = i + ik
            im = i - ik
            contrib = 0.0
            if 0 <= i < n and 0 <= ip < n:
                contrib += (1.0 - h[ip])
            if 0 <= im < n:
                contrib -= h[im]
            g[i] = contrib
        neg = np.argsort(g)[:top_k // 2]
        pos = np.argsort(g)[::-1][:top_k // 2]
        for a in neg:
            for b in pos:
                if a != b:
                    pairs.append((int(a), int(b)))

        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 2: Coarse coordinate descent
    if tleft() > 250:
        h, c = _cd_loop(best_h, n, dx, t0 + budget_s - 300, t0, 11, 400, 4)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 3: Fourier structured moves
    if tleft() > 200:
        h, c, _ = _fourier_moves(best_h, n, best_c5, dx, 35, 12)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
            h2, c2 = _cd_loop(h, n, dx, t0 + budget_s - 250, t0, 13, 200, 4)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 4: Fine coordinate descent
    if tleft() > 180:
        h, c = _cd_loop(best_h, n, dx, t0 + budget_s - 210, t0, 17, 600, 3)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 5: Block moves + polish
    if tleft() > 130:
        h, c, _ = _block_move(best_h, n, best_c5, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
            h2, c2 = _cd_loop(h, n, dx, t0 + budget_s - 160, t0, 15, 250, 4)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 6: SLSQP medium polish
    if tleft() > 100:
        h = _slsqp_opt(best_h, n, 1500, 500, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 7: More Fourier + CD
    if tleft() > 120:
        h, c, _ = _fourier_moves(best_h, n, best_c5, dx, 25, 10)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
        h2, c2 = _cd_loop(best_h, n, dx, t0 + budget_s - 140, t0, 19, 400, 3)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 8: Multi-restart
    sigmas = (0.02, 0.04, 0.06, 0.09, 0.12, 0.18, 0.03, 0.15)
    for si in range(len(sigmas)):
        if tleft() < 50:
            break
        h_p = _repair(best_h + sigmas[si] * rng.randn(n), n)
        a = 150 + 120 * si
        h = _slsqp_opt(h_p, n, a, 250, 1e-13)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
            h2, c2 = _cd_loop(h, n, dx, t0 + time.time() - t0 + 50, t0, 17, 150, 4)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 9: Final fine CD
    if tleft() > 30:
        h, c = _cd_loop(best_h, n, dx, t0 + budget_s - 35, t0, 23, 500, 2)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 10: Final SLSQP polish
    if tleft() > 12:
        h = _slsqp_opt(best_h, n, 3000, 300, 1e-15)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 11: Last CD sweep
    if tleft() > 5:
        h, c = _cd_loop(best_h, n, dx, t0 + budget_s - 3, t0, 19, 200, 2)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
