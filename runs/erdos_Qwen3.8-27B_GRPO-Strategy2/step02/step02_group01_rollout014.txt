    return np.correlate(h, 1.0-h, mode='full')

    c = _corr_full(h, n)
    m_max = np.max(c)
    exp_vals = np.exp(alpha * (c - m_max))

    c = np.correlate(h, 1.0-h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _smooth_obj_and_grad(h, n, alpha, penalty=0):
    c = _corr(h)
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    logsum = np.log(np.sum(np.exp(shifted)))
    val = (m + logsum / alpha) * (2.0 / n)
    if penalty > 0:
        s = h.sum()
        val += penalty * (s - n / 2.0) ** 2
    
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    g = np.zeros(n)
    idx = np.arange(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-12:
            continue
        v1 = idx >= m_idx
        if np.any(v1):
            jv = idx[v1]
            g[v1] += wm * (1.0 - h[jv - m_idx])
        v2 = (idx + m_idx) < n
        if np.any(v2):
            jv = idx[v2]
            g[v2] -= wm * h[jv + m_idx]
    g *= (2.0 / n)
    if penalty > 0:
        g += 2 * penalty * (h.sum() - n / 2.0)
    return val, g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def _lgbfs_optimize(h0, n, alpha, penalty, maxiter, ftol, time_limit):
    t_start = time.time()
    best_h = h0.copy()
    best_c5 = _c5(h0, 2.0/n)
    
    def obj_grad(x):
        if time.time() - t_start > time_limit:
            raise ValueError("timeout")
        h = np.clip(x, 0.0, 1.0)
        return _smooth_obj_and_grad(h, n, alpha, penalty)
    
    try:
        res = minimize(obj_grad, h0, jac=True, method='L-BFGS-B',
                      bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': 5000})
        hc = _repair(res.x, n)
        c5 = _c5(hc, 2.0/n)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    except Exception:
        pass
    
    return best_h, best_c5

def _binary_local_search(h, n, max_passes=50):
    """Local search on binary solutions"""
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    # Ensure sum = n/2
    ones = np.sum(h_bin)
    target = n // 2
    if ones != target:
        # Fix
        if ones > target:
            # Set some ones to zeros
            idx1 = np.where(h_bin == 1)[0]
            for i in idx1[:ones-target]:
                h_bin[i] = 0
        else:
            idx0 = np.where(h_bin == 0)[0]
            for i in idx0[:target-ones]:
                h_bin[i] = 1
    
    best_c5 = _c5(h_bin, dx)
    best_h = h_bin.copy()
    
    for _pass in range(max_passes):
        improved = False
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        for i in ones_idx:
            for j in zeros_idx:
                h_bin[i] = 0
                h_bin[j] = 1
                c5 = _c5(h_bin, dx)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h_bin.copy()
                    improved = True
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
        if not improved:
            break
    return best_h, best_c5

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return np.max(_corr(h)) * dx


def _smooth_val_grad(h, n, alpha, penalty=0.0):
    c = _corr(h)
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    s_exps = np.sum(exps)
    val = (m + np.log(s_exps) / alpha) * (2.0 / n)
    if penalty > 0:
        s = h.sum()
        val += penalty * (s - n / 2.0) ** 2
    w = exps / s_exps
    g = np.zeros(n)
    idx = np.arange(n)
    for mi in range(2 * n - 1):
        wm = w[mi]
        if wm < 1e-14:
            continue
        v1 = idx >= mi
        if np.any(v1):
            jv = idx[v1]
            g[v1] += wm * (1.0 - h[jv - mi])
        v2 = (idx + mi) < n
        if np.any(v2):
            jv = idx[v2]
            g[v2] -= wm * h[jv + mi]
    g *= (2.0 / n)
    if penalty > 0:
        g += 2.0 * penalty * (h.sum() - n / 2.0)
    return val, g


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for ix in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[ix])
                    h[ix] += add
                    diff -= add
                else:
                    sub = min(-diff, h[ix])
                    h[ix] -= sub
                    diff += sub
                if abs(diff) < 1e-12:

            for _it in range(300):
                el = time.time() - t0
                if el > budget_s - 60:
                    break
                c = _corr(h)
                mi = int(np.argmax(c))
                g = np.zeros(n)
                v1 = idx >= mi
                if np.any(v1):
                    g[v1] = (1.0 - h[idx[v1] - mi])
                v2 = (idx + mi) < n
                if np.any(v2):
                    g[v2] -= h[idx[v2] + mi]
                hn = _repair(h - step * g, n)
                c5n = _c5(hn, dx)
                if c5n < c5c:
                    h, c5c = hn, c5n
                    if c5c < best_c5:
                        best_c5, best_h = c5c, h.copy()

    # Phase 6: Pair coordinate descent
    el = time.time() - t0
    if el < budget_s - 30:
        h = best_h.copy()
        c5c = _c5(h, dx)
        deltas = [0.002, 0.005, 0.01, 0.03, -0.002, -0.005, -0.01, -0.03]
        for _pass in range(40):
            el = time.time() - t0
            if el > budget_s - 30:
                break
            any_imp = False
            for i in range(n):
                if el > budget_s - 30:
                    break
                for j in range(i + 1, n):
                    hi, hj = h[i], h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5c - 1e-15:
                            c5c, hi, hj = c, ni, nj
                            any_imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i] = hi
            if not any_imp:
                break
        if c5c < best_c5:
            best_c5, best_h = c5c, h.copy()

    # Phase 7: Annealing
    el = time.time() - t0
    if el < budget_s - 15:
        h = best_h.copy()
        c5c = _c5(h, dx)
        T, cool = 0.002, 0.997
        step_opts = np.array([0.005, 0.01, 0.02, 0.05, 0.1, -0.005, -0.01, -0.02, -0.05, -0.1])
        while time.time() - t0 < budget_s - 15:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = step_opts[rng.randint(len(step_opts))]
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            h = _repair(h, n)
            c5n = _c5(h, dx)
            if c5n < c5c or (T > 1e-8 and rng.rand() < np.exp(-(c5n - c5c) / T)):
                c5c = c5n
                if c5c < best_c5:
                    best_c5, best_h = c5c, h.copy()
            else:
                h[i], h[j] = hi, hj
                h = _repair(h, n)
            T = max(T * cool, 1e-7)

    # Phase 8: Final micro-polish
    el = time.time() - t0
    if el < budget_s - 5:
        h = best_h.copy()
        c5c = _c5(h, dx)
        for _ in range(5000):
            if time.time() - t0 > budget_s - 5:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice(np.array([0.0002, 0.0005, 0.001, -0.0002, -0.0005, -0.001]))
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                h = _repair(h, n)
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                    best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = hi, hj
                    h = _repair(h, n)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
