def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b
                diff += b
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_obj_grad(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * tgt
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-10:
        o = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in o:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                b = min(-diff, h[i])
                h[i] -= b
                diff += b
            if abs(diff) < 1e-10:
                break
    return h

def _sobg(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(c)
    s = np.clip(alpha * (c - mx), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        wt = w[n - 1 + k]
        if wt < 1e-12:
            continue
        if k >= 0:
            g[:n - k] += wt * (1.0 - h[k:])
            g[k:] -= wt * h[:n - k]
        else:
            j = -k
            g[j:] += wt * (1.0 - h[:n - j])
            g[:n - j] -= wt * h[j:]
    return (mx + np.log(e.sum()) / alpha) * dx, g * dx

def _sobj(x, alpha, dx):
    return _sobg(x, alpha, dx)[0]

def _sjac(x, alpha, dx):
    return _sobg(x, alpha, dx)[1]

def _scon(x, tgt):
    return np.sum(x) - tgt

def _slsqp(h0, n, dx, alpha, mi, ft):
    bd = [(0.0, 1.0)] * n
    tgt = n / 2.0
    try:
        r = minimize(_sobj, h0.copy(), jac=_sjac, args=(alpha, dx),
                     method='SLSQP', bounds=bd,
                     constraints=[{'type': 'eq', 'fun': _scon, 'args': (tgt,)}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h0.copy()

    bc = _c5(best, dx)
    sp = [0.003, 0.007, 0.015, 0.03, 0.06, 0.1]

            d = sp[int(rng.randint(len(sp)))]
            if rng.rand() < 0.5:

            ni, nj = best[i] + d, best[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < bc - 1e-12:

                imp = True
            else:

            else:
                j = int(rng.randint(n + k))
                i = j - k
            d = sp[int(rng.randint(len(sp)))]
            if rng.rand() < 0.5:
                d = -d
            ni, nj = best[i] + d, best[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            oi, oj = best[i], best[j]
            best[i], best[j] = ni, nj
            nc = _c5(best, dx)
            if nc < bc - 1e-12:
                bc = nc
                imp = True
            else:
                best[i], best[j] = oi, oj
            if imp:

        if not imp:

                i, j = int(rng.randint(n)), int(rng.randint(n))

                if imp:

        st = 0 if imp else st + 1

def _bin_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _sine_init(n, fr, ph, rng):

    h = 0.5 + 0.35 * np.sin(2 * np.pi * fr * t + ph) + 0.05 * rng.randn(n)
    return _repair(h, n)

def _pgd(h, n, dx, alpha, iters, dl):
    for _ in range(iters):
        if time.time() > dl:
            break
        _, g = _sobg(h, alpha, dx)
        oc = _c5(h, dx)
        lr = 2.0 / (alpha + 10.0)
        for _ in range(12):
            hnew = _repair(h - lr * g, n)
            if _c5(hnew, dx) < oc:
                h = hnew
                break
            lr *= 0.5
    return h
