import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _smooth_obj(h, alpha):
    c = _corr(h)
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(c - m, -500, 0) * alpha))) / alpha

def _smooth_grad(h, alpha):
    c = _corr(h)
    m = np.max(c)
    p = np.exp(np.clip(c - m, -500, 0) * alpha)
    p /= p.sum()
    # Compute gradient
    n = len(h)
    g = np.zeros(n)
    nc = len(c)
    for k in range(nc):
        if p[k] < 1e-14:
            continue
        pk = p[k]
        for j in range(n):
            v = 0.0
            if 0 <= j - k < n:
                v += (1.0 - h[j - k])
            if 0 <= j + k < n:
                v -= h[j + k]
            g[j] += pk * v
    return g, _smooth_obj(h, alpha)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
