import numpy as np
from scipy.optimize import minimize
import time

_GT = 0.0

def _sum_eq(x):
    return x.sum() - _GT

def _project(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - x.sum()
    if abs(d) > 1e-13:
        m = (x < 1.0) if d > 0 else (x > 0.0)
        if m.sum() > 0:
            x[m] += d / m.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, dx):
    return float(np.max(_corr(h)) * dx)

def _G_all(h):
    n = len(h)
    v = 1.0 - h
    ti = np.arange(n)[:, None]
    mi = np.arange(-(n - 1), n)[None, :]
    tm = ti - mi
    tp = ti + mi
    G1 = np.where((tm >= 0) & (tm < n), v[np.clip(tm, 0, n - 1)], 0.0)
    G2 = np.where((tp >= 0) & (tp < n), h[np.clip(tp, 0, n - 1)], 0.0)
    return G1 - G2

def _grad_s(h, alpha):
    c = _corr(h)
    cm = c.max()
    e = np.exp(np.clip(alpha * (c - cm), -500, 500))
    w = e / e.sum()
    return _G_all(h).T @ w

def _smooth_f(x, alpha, M, target):
    c = _corr(x)
    cm = c.max()
    e = np.exp(np.clip(alpha * (c - cm), -500, 500))
    return cm + np.log(e.sum()) / alpha + M * (x.sum() - target) ** 2

def _slsqp(h0, n, alpha, maxiter):
    global _GT
    _GT = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        r = minimize(_smooth_f, h0, args=(alpha, 1e-4, n / 2.0), method='SLSQP',
                     bounds=bnds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _project(r.x, n / 2.0)
    except Exception:
        return h0.copy()

def _cd(h, n, dx, tlim, t0, ng, mp):
    bc = _c5v(h, dx)
    for _ in range(mp):
        if time.time() - t0 > tlim:
            break
        imp = False
        for i in range(n):
            if time.time() - t0 > tlim:
                break
            for j in range(i + 1, n):
                hs = h[i] + h[j]
                lo = max(0.0, hs - 1.0)
                hi = min(1.0, hs)
                if hi - lo < 1e-10:
                    continue
                oi, oj = h[i], h[j]
                bt, bcc = oi, bc
                for t in np.linspace(lo + 1e-9, hi - 1e-9, ng):
                    h[i], h[j] = t, hs - t
                    c = _c5v(h, dx)
                    if c < bcc:
                        bcc, bt = c, t
                if bt != oi:
                    lo2 = max(lo, bt - (hi - lo) / ng)
                    hi2 = min(hi, bt + (hi - lo) / ng)
                    for t in np.linspace(lo2, hi2, 11):
                        h[i], h[j] = t, hs - t
                        c = _c5v(h, dx)
                        if c < bcc:
                            bcc, bt = c, t
                    if bcc < bc - 1e-14:
                        h[i], h[j] = bt, hs - bt
                        bc = bcc
                        imp = True
                    else:
                        h[i], h[j] = oi, oj
                else:
                    h[i], h[j] = oi, oj
        if not imp:
            break
    return h, bc

def _nest(h, n, dx, tlim, t0, alpha, mi):

            tk = tkn

            tk = 1.0
        if it % 50 == 0:

            if c < bc:

        if it > 0 and it % 5000 == 0:
            lr *= 0.95
    c = _c5v(xk, dx)
    if c < bc:

    return bh, bc
