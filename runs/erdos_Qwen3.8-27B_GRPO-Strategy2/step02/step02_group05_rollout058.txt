    c = _corr(h)

            g[:n - mm] += (1.0 - h[mm:])
            g[mm:] -= h[:n - mm]

            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best, best_c5

def _slsqp_obj(x, alpha, dx):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _slsqp_grad(x, alpha, dx):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    n = len(x)
    g = np.zeros(n)
    for p in range(len(c)):
        if w[p] < 1e-8:
            continue
        lag = p - (n - 1)
        g += w[p] * _grad_lag(x, lag, n)
    return g * dx

def _con_fn(x, target):
    return x.sum() - target

def _con_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, n, dx, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': _con_fn, 'jac': _con_jac, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, dx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _anneal(h, n, dx, rng, deadline):
    best, best_c5 = h.copy(), _c5(h, dx)
    T = 0.02
    for gen in range(300):
        if time.time() > deadline:
            break
        h = best.copy()
        T = max(0.002, 0.02 * (0.985 ** gen))
        for _ in range(300):
            if time.time() > deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = float(rng.uniform(-0.08, 0.08))
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            cur = _c5(h, dx)
            h[i], h[j] = ni, nj
            new = _c5(h, dx)
            if new < cur or rng.rand() < np.exp((cur - new) / T):
                if new < best_c5:
                    best_c5, best = new, h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best, best_c5

def _coarse_search(h, n, dx, rng, deadline):
    best, best_c5 = h.copy(), _c5(h, dx)
    stall = 0
    while time.time() < deadline and stall < 60:
        improved = False
        for _ in range(60):
            if time.time() > deadline:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            steps = np.array([0.02, 0.04, 0.06, 0.1, -0.02, -0.04, -0.06, -0.1])
            for d in steps:
                ni, nj = best[i] + d, best[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                oi, oj = best[i], best[j]
                best[i], best[j] = ni, nj
                nc = _c5(best, dx)
                if nc < best_c5 - 1e-12:
                    best_c5, improved = nc, True
                    break
                best[i], best[j] = oi, oj
        if improved:
            stall = 0
        else:
            stall += 1
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

    # Phase 1: Subgradient attack on max lags
    dl = t0 + budget_s * 0.2
    h1, c1 = _attack_max(h0, n, dx, 8000, dl)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: SLSQP refinement
    if time.time() < t0 + budget_s * 0.42:
        for alpha in [80, 200, 500, 1200]:
            if time.time() > t0 + budget_s * 0.42:
                break
            hc = _slsqp(best_h, n, dx, alpha, 400)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5

    # Phase 3: Block local search
    dl = t0 + budget_s * 0.58
    best_h, best_c5 = _block_search(best_h, n, dx, rng, dl)

    # Phase 4: SLSQP + block search alternation
    for _ in range(3):
        if time.time() > t0 + budget_s * 0.78:
            break
        hc = _slsqp(best_h, n, dx, 600, 350)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
        dl = min(t0 + budget_s * 0.78, time.time() + 60)
        best_h, best_c5 = _block_search(best_h, n, dx, rng, dl)

    # Phase 5: Annealing to escape local optima
    if time.time() < t0 + budget_s * 0.88:
        dl = t0 + budget_s * 0.88
        ha, ca = _anneal(best_h, n, dx, rng, dl)
        if ca < best_c5:
            best_h, best_c5 = ha, ca

    # Phase 6: Coarse coordinate search
    if time.time() < t0 + budget_s * 0.95:
        dl = t0 + budget_s * 0.95
        best_h, best_c5 = _coarse_search(best_h, n, dx, rng, dl)

    # Phase 7: Final SLSQP polish
    if time.time() < t0 + budget_s - 5:
        hc = _slsqp(best_h, n, dx, 2000, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5

    # Final block search with tiny remaining time
    if time.time() < t0 + budget_s - 3:
        best_h, best_c5 = _block_search(best_h, n, dx, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
