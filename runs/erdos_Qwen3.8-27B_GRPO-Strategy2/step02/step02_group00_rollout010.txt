def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lse_obj(x, alpha, n):
    corr = _corr(x)
    m = float(np.max(corr))
    s = np.exp(np.clip(alpha * (corr - m), -60.0, 60.0))
    return (m + float(np.sum(s)) / alpha) * (2.0 / n)

def _topk_obj(x, k, n):
    corr = _corr(x)
    kk = min(k, len(corr))
    idx = np.argpartition(corr, -kk)[-kk:]
    return float(np.sum(corr[idx])) * (2.0 / n)

def _sum_diff(x, target):
    return float(np.sum(x) - target)

def _repair(h, n):

                diff -= add

                h[idx] -= sub
                diff += sub
