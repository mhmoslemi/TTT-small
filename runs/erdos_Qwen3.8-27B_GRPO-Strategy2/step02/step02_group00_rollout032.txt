    target = n / 2.0
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())

        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:

    cur_max = float(np.max(C))
    deltas_large = np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    deltas_small = np.array([0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    no_improve = 0
    phase = 0

    while time.time() < t_end:
        dset = deltas_large if phase == 0 else deltas_small
        best_gain = 0.0
        best_a, best_b, best_d = -1, -1, 0.0
        for a in range(n):
            if h[a] + dset[0] > 1.0 + 1e-9 and h[a] - dset[0] < -1e-9:
                continue
            for d in dset:
                if h[a] + d > 1.0 + 1e-9:
                    continue
                diffs = M[a] - M
                new_maxs = np.max(C[None, :] + d * diffs, axis=1)
                gains = cur_max - new_maxs
                feasible = (h - d) >= -1e-9
                feasible[a] = False
                if not np.any(feasible):
                    continue
                gains[~feasible] = -1.0
                bi = int(np.argmax(gains))
                if gains[bi] > best_gain + 1e-14:
                    best_gain = gains[bi]
                    best_a, best_b, best_d = a, bi, d
                # Also try negative d (move from a to b)
                nd = -d
                if h[b] + nd > 1.0 + 1e-9:
                    pass
            # Try reverse direction
            for d in dset:
                if h[a] - d < -1e-9:
                    continue
                diffs = M[a] - M
                new_maxs = np.max(C[None, :] + (-d) * diffs, axis=1)
                gains = cur_max - new_maxs
                feasible = (h + d) <= 1.0 + 1e-9
                feasible[a] = False
                if not np.any(feasible):
                    continue
                gains[~feasible] = -1.0
                bi = int(np.argmax(gains))
                if gains[bi] > best_gain + 1e-14:
                    best_gain = gains[bi]
                    best_a, best_b, best_d = a, bi, -d

        if best_a < 0:
            no_improve += 1
            if no_improve >= 4:
                phase = 1
                no_improve = 0
                deltas_large = deltas_small
            if no_improve >= 8:
                # Kick: random perturbation
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    d = rng.uniform(0.005, 0.03) * rng.choice([1, -1])
                    if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                        h[i] += d
                        h[j] -= d
                        _upd_M(M, i, d, n)
                        _upd_M(M, j, -d, n)
                        C = _corr(h)
                        cur_max = float(np.max(C))
                no_improve = 0
            if no_improve >= 20:
                break
            continue

        h[best_a] += best_d
        h[best_b] -= best_d
        C += best_d * (M[best_a] - M[best_b])
        _upd_M(M, best_a, best_d, n)
        _upd_M(M, best_b, -best_d, n)
        cur_max = float(np.max(C))
        no_improve = 0

    return h, cur_max * dx


def _binary_local(h, n, dx, t_end, rng):
    target = n // 2
    sort_idx = np.argsort(h)[::-1]
    bh = np.zeros(n)
    bh[sort_idx[:target]] = 1.0
    C = _corr(bh)
    cur_max = float(np.max(C))
    no_improve = 0

    while time.time() < t_end and no_improve < 30:
        best_gain = 0.0
        best_pair = None
        for a in range(n):
            if bh[a] != 1.0:
                continue
            for b in range(n):
                if bh[b] != 0.0:
                    continue
                # Swap: bh[a]=0, bh[b]=1
                C_new = C.copy()
                for s in range(-(n-1), n):
                    si = s + n - 1
                    old = 0.0
                    if 0 <= a - s < n:
                        old += bh[a] * (1 - bh[a - s])
                    if 0 <= a + s < n:
                        old += bh[a + s] * (1 - bh[a])
                    new = 0.0
                    if 0 <= a - s < n:
                        new_a = 0.0 if (a - s != a) else 0.0
                        # h[a] becomes 0
                        if a - s == a:
                            pass  # h[a]*(1-h[a-s]) = 0*(...) = 0
                        else:
                            new += 0.0
                    # This is getting complex, let me just recompute
                    pass
                # Recompute C for this swap (simpler for binary)
                bh[a] = 0.0
                bh[b] = 1.0
                C_try = _corr(bh)
                max_try = float(np.max(C_try))
                gain = cur_max - max_try
                bh[a] = 1.0
                bh[b] = 0.0
                if gain > best_gain + 1e-14:
                    best_gain = gain
                    best_pair = (a, b)
                    C_best = C_try.copy()

        if best_pair is None:
            no_improve += 1
            continue
        a, b = best_pair
        bh[a] = 0.0
        bh[b] = 1.0
        C = C_best
        cur_max = cur_max - best_gain
        no_improve = 0

    return bh, cur_max * dx


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP from initial (10% budget)
    p1 = t0 + min(budget_s * 0.10, 100)
    h, c5 = _slsqp_phase(h, n, dx, bounds, cons, p1)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 2: Incremental local search (45% budget)
    p2 = t0 + min(budget_s * 0.55, 550)
    h, c5 = _local_search(h, n, dx, p2, rng)
    h = _repair(h, n)
    c5 = _c5(h, dx)
    if c5 < best_c5:
        best_c5, best_h = c5, h

    # Phase 3: Multi-restart local search (25% budget)
    p3 = t0 + min(budget_s * 0.82, 860)
    for restart in range(10):
        if time.time() > p3:
            break
        # Create perturbed copy
        h_r = best_h.copy()
        for _ in range(n // 5):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                d = rng.uniform(0.02, 0.08) * rng.choice([1, -1])
                if 0 <= h_r[i] + d <= 1 and 0 <= h_r[j] - d <= 1:
                    h_r[i] += d
                    h_r[j] -= d
        h_r = _repair(h_r, n)
        h_r2, c5_r = _local_search(h_r, n, dx, min(p3, time.time() + 30), rng)
        h_r2 = _repair(h_r2, n)
        c5_r = _c5(h_r2, dx)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r2

    # Phase 4: Binary local search (10% budget)
    p4 = t0 + min(budget_s * 0.92, 980)
    if time.time() < p4:
        bh, c5_b = _binary_local(best_h, n, dx, p4, rng)
        if c5_b < best_c5:
            # Use binary as starting point for continuous refinement
            h_b = bh.astype(float)
            h_b, c5_b2 = _local_search(h_b, n, dx, min(p4 + 20, t0 + budget_s - 10), rng)
            h_b = _repair(h_b, n)
            c5_b2 = _c5(h_b, dx)
            if c5_b2 < best_c5:
                best_c5, best_h = c5_b2, h_b

    # Phase 5: Final SLSQP polish
    p5 = t0 + budget_s - 5
    if time.time() < p5:
        h_f, c5_f = _slsqp_phase(best_h, n, dx, bounds, cons, p5)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
