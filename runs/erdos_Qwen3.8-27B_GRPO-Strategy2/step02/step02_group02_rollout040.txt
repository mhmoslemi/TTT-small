                bc = c
    
    # Phase: SLSQP polish
    bnds = [(0.0, 1.0)] * n
    for a_pol in [3000, 8000]:
        if time.time() - t0 > budget_s - 15:
            break
        try:
            r = minimize(_slsqp_obj, bh, args=(a_pol, 3), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                         options={'maxiter': 500, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass
    
    # Final CD
    if time.time() - t0 > budget_s - 8:
        bh, c = _cd_sweep(bh, dx, tgt, n, rng, passes=300, t0=t0, budget=budget_s)
        if c < bc:
            bc = c
    
    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
