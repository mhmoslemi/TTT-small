    """Gradient step using average of top-k lag gradients"""
    c = _corr(h)
    m = c.max()
    # Find top lags within tol of max
    idx = np.arange(len(c))
    active = idx[c >= m - tol]
    if len(active) == 0:
        active = [int(np.argmax(c))]
    g = np.zeros(n)
    for k in active:
        g += _grad_lag(h, k, n)
    g /= len(active)
    return h - step * g, c, int(active[0])
