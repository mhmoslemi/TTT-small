    try:
        res = minimize(of, h, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        hc = _repair(res.x, n, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_opt(h, n, dx, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def of(x):
        v, g = _smooth_obj_grad(x, alpha)
        return v, np.column_stack([g, np.ones(n)])
    def eqc(x):
        return np.sum(x) - target, np.ones(n)
    try:
        res = minimize(of, h, method='SLSQP', jac=True, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(res.x, n, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coord_descent(h, n, dx, target, rng, deadline, n_iter=300):
    best = _c5(h, dx)
    h = h.copy()
    for _ in range(n_iter):
        if time.time() >= deadline:
            break
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        d_grid = np.linspace(-0.05, 0.05, 101)
        best_d = 0.0
        best_c = best
        for d in d_grid:
            hj, hk = h[j] + d, h[k] - d
            if hj < -1e-10 or hj > 1.0 + 1e-10 or hk < -1e-10 or hk > 1.0 + 1e-10:
                continue
            h[j], h[k] = hj, hk
            c = _c5(h, dx)
            if c < best_c - 1e-14:
                best_c = c
                best_d = d
            h[j], h[k] = hj - d, hk + d
        h[j] += best_d
        h[k] -= best_d
        best = best_c
    return h, best

def _random_ls(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    target = n / 2.0
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for scale in scales:
        stalled = 0
        while time.time() < deadline and stalled < 8:
            if time.time() >= deadline:
                break
            improved = False
            for _ in range(300):
                if time.time() >= deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                hi, hj = h[i] + d, h[j] - d
                if hi < -1e-9 or hi > 1.0 + 1e-9 or hj < -1e-9 or hj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = hi, hj
                c = _c5(h, dx)
                if c < best - 1e-12:
                    best = c
                    improved = True
                else:
                    h[i], h[j] = hi - d, hj + d
            stalled = 0 if improved else stalled + 1
    return h, best

def _eval_n(n, h_init, dx, rng, t0, budget_s, frac_time):
    t_left = budget_s - (time.time() - t0)
    t_alloc = frac_time * t_left
    if t_alloc < 30:
        return h_init, _c5(h_init, dx), n
    target = n / 2.0
    h = _repair(h_init.copy(), n, target)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    # L-BFGS-B phase
    alphas = [20, 60, 150, 400, 1000, 3000]
    t_p1 = time.time() + 0.4 * t_alloc
    for alpha in alphas:
        if time.time() > t_p1:
            break
        best_h, c5 = _lbfgs_opt(best_h, n, dx, alpha, 400, t_p1)
        if c5 < best_c5:
            best_c5 = c5
    # Local search
    if time.time() < t0 + budget_s - 20:
        t_ls = time.time() + 0.15 * t_alloc
        t_ls = min(t_ls, t0 + budget_s - 15)
        best_h, best_c5 = _coord_descent(best_h, n, dx, target, rng, t_ls, n_iter=200)
    # More L-BFGS-B
    if time.time() < t0 + budget_s - 20:
        t_p2 = time.time() + 0.3 * t_alloc
        t_p2 = min(t_p2, t0 + budget_s - 15)
        for alpha in [500, 2000, 8000]:
            if time.time() > t_p2:
                break
            best_h, c5 = _lbfgs_opt(best_h, n, dx, alpha, 500, t_p2)
            if c5 < best_c5:
                best_c5 = c5
    # Final local
    if time.time() < t0 + budget_s - 10:
        t_ls2 = min(time.time() + 10, t0 + budget_s - 5)
        best_h, best_c5 = _random_ls(best_h, n, dx, rng, t_ls2)
    # SLSQP polish
    if time.time() < t0 + budget_s - 5:
        t_p3 = min(time.time() + 8, t0 + budget_s - 3)
        best_h, c5 = _slsqp_opt(best_h, n, dx, 5000, 200, t_p3)
        if c5 < best_c5:
            best_c5 = c5
    best_h = _repair(best_h, n, target)
    best_c5 = _c5(best_h, dx)
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n_parent = len(initial_h_values)
    dx_parent = 2.0 / n_parent
    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), n_parent, n_parent / 2.0)

    best_h = h_parent.copy()
    best_c5 = _c5(best_h, dx_parent)
    best_n = n_parent

    # Try parent n with full budget allocation
    h_r, c5_r, _ = _eval_n(n_parent, h_parent, dx_parent, rng, t0, budget_s, 0.55)
    if c5_r < best_c5:
        best_h, best_c5, best_n = h_r, c5_r, n_parent

    # Try other n values
    other_ns = [50, 60, 80]
    t_check = time.time()
    time_per_n = min(180, (budget_s - (t_check - t0) - 30) / len(other_ns))
    for n_alt in other_ns:
        if time.time() - t0 > budget_s - 40:
            break
        dx_alt = 2.0 / n_alt
        target_alt = n_alt / 2.0
        # Interpolate parent to new size
        x_orig = np.linspace(0, 2, n_parent, endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        h_interp = _repair(np.interp(x_new, x_orig, h_parent), n_alt, target_alt)
        # Add small random perturbation
        h_interp = _repair(h_interp + 0.02 * rng.randn(n_alt), n_alt, target_alt)
        h_r, c5_r, _ = _eval_n(n_alt, h_interp, dx_alt, rng, t0, budget_s, time_per_n / (budget_s - (time.time() - t0) + 1))
        if c5_r < best_c5:
            best_h, best_c5, best_n = h_r, c5_r, n_alt
            # Update parent for further interpolation
            h_parent = _repair(np.interp(np.linspace(0, 2, n_alt, endpoint=False),
                                          np.linspace(0, 2, n_parent, endpoint=False), best_h),
                               n_alt, n_alt / 2.0)
            n_parent = n_alt

    # Final polish on best
    dx_b = 2.0 / best_n
    target_b = best_n / 2.0
    if time.time() - t0 < budget_s - 10:
        t_fin = min(time.time() + 8, t0 + budget_s - 3)
        best_h, c5 = _slsqp_opt(best_h, best_n, dx_b, 10000, 300, t_fin)
        if c5 < best_c5:
            best_c5 = c5
    if time.time() - t0 < budget_s - 5:
        t_fin2 = min(time.time() + 5, t0 + budget_s - 2)
        best_h, best_c5 = _random_ls(best_h, best_n, dx_b, rng, t_fin2)
    if time.time() - t0 < budget_s - 3:
        t_fin3 = min(time.time() + 3, t0 + budget_s - 1)
        best_h, best_c5 = _coord_descent(best_h, best_n, dx_b, target_b, rng, t_fin3, n_iter=100)

    best_h = _repair(best_h, best_n, target_b)
    best_c5 = _c5(best_h, dx_b)
    return (best_h, best_c5, best_n)
