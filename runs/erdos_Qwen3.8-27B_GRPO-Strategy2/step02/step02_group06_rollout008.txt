```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _g_all(h, n):
    return np.array([_g_vec(h, p, n) for p in range(n)])

def _optimal_d(corr, gd, d_min, d_max):
    if d_min >= d_max:
        return 0.0
    lo, hi = d_min, d_max
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * gd
        mx = np.max(vals)
        active = vals >= mx - 1e-12
        min_g = np.min(gd[active])
        max_g = np.max(gd[active])
        if min_g > 1e-15:
            hi = mid
        elif max_g < -1e-15:
            lo = mid
        else:
            lo = hi = mid
            break
    return 0.5 * (lo + hi)

def _sweep_bisect(h, corr, n, dx):
    g_all = _g_all(h, n)
    improved = False
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = g_all[i] - g_all[j]
            d = _optimal_d(corr, gd, d_min, d_max)
            new_max = np.max(corr + d * gd)
            cur_max = np.max(corr)
            if new_max < cur_max - 1e-14 and abs(d) > 1e-12:
                corr += d * gd
                h[i] += d
                h[j] -= d
                improved = True
    return h, corr, improved

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha, dx):
    return _smooth_max(_corr(x), alpha) * dx

def _eq_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, target, bounds, alpha, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level, no closures)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    k = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + k
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _g_all(h, n):
    return np.array([_g_vec(h, p, n) for p in range(n)])

def _bisect_d(corr, gd, d_min, d_max):
    if d_min >= d_max - 1e-15:
        return 0.0
    lo, hi = d_min, d_max
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        v = corr + mid * gd
        mx = v.max()
        act = v >= mx - 1e-12
        if gd[act].min() > 1e-15:
            hi = mid
        elif gd[act].max() < -1e-15:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def _sweep(h, corr, n):
    ga = _g_all(h, n)
    imp = False
    for i in range(n):
        for j in range(n):
            if i == j: continue
            dlo = max(-h[i], h[j] - 1.0)
            dhi = min(1.0 - h[i], h[j])
            if dhi - dlo < 1e-14: continue
            gd = ga[i] - ga[j]
            d = _bisect_d(corr, gd, dlo, dhi)
            if abs(d) < 1e-14: continue
            nm = np.max(corr + d * gd)
            if nm < np.max(corr) - 1e-13:
                corr += d * gd
                h[i] += d
                h[j] -= d
                imp = True
    return h, corr, imp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for k in idx:
            if d > 0:
                a = min(d, 1 - h[k]); h[k] += a; d -= a
            else:
                a = min(-d, h[k]); h[k] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _smax(c, a):
    m = c.max()
    s = np.clip(a * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / a

def _obj(x, a):
    return _smax(_corr(x), a)

def _slsqp(h, n, a, mi):
    t = n * 0.5
    bd = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj, h, args=(a,), method='SLSQP', bounds=bd,
                     constraints=[{'type':'eq','fun':lambda x: x.sum()-t}],
                     options={'maxiter':mi,'ftol':1e-15})
        return _repair(r.x, n)
    except:
        return h
```

```python
    v = np.zeros(n)

        # Smooth max gradient

        w = np.exp(a * (corr - m))

        grad = np.zeros(n)

        grad = ga.T @ w  # shape (n,)
        
        v = momentum * v - lr * grad
        h = h + v
        h = _repair(h, n)
    return h
```

For n points h[0], h[1], ..., h[n-1] with dx = 2/n:
np.correlate(h, 1-h, mode='full') gives an array of length 2n-1.

The element at index m (0 ≤ m ≤ 2n-2) is:
c[m] = sum over i of h[i + m - (n-1)] * (1 - h[i])
where the sum is over all i such that 0 ≤ i < n and 0 ≤ i + m - (n-1) < n.

Let lag = m - (n-1), so m = lag + n - 1, and lag ranges from -(n-1) to n-1.
c[lag + n - 1] = sum over i of h[i + lag] * (1 - h[i])
where 0 ≤ i < n and 0 ≤ i + lag < n.

The C₅ value is max over all lags of c[lag + n - 1] * dx.
