    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _smax_grad(h, alpha):
    c = _corr(h)
    n = len(h)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for li in range(2 * n - 1):
        wl = w[li]
        if wl < 1e-14:
            continue
        l = li - (n - 1)

            g[:n - l] += wl * (1.0 - h[l:])
            g[l:] -= wl * h[:n - l]
        else:
            g[-l:] += wl * (1.0 - h[:n + l])
            g[:n + l] -= wl * h[-l:]
    return g

    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:

                h[idx] += a
                diff -= a
            else:

                h[idx] -= a
                diff += a

    bounds = [(0.0, 1.0)] * n
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _c5(h, dx)

    for alpha in [15, 50, 150, 500, 1500, 4000]:
        if time.time() - t0 > budget_s - 500:
            break
        hc = _slsqp(best_h, n, dx, target, bounds, alpha, 600)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    for alpha in [80, 300, 1000, 2500]:
        if time.time() - t0 > budget_s - 450:
            break
        hc = _lbfgs(best_h, n, dx, target, alpha, 5.0, 5000)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    if time.time() - t0 < budget_s - 280:
        hc, c5 = _pair_descent(best_h.copy(), n, dx, rounds=50, giter=60)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    for alpha in [3000, 8000]:
        if time.time() - t0 > budget_s - 250:
            break
        hc = _slsqp(best_h, n, dx, target, bounds, alpha, 800)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    if time.time() - t0 < budget_s - 180:
        hc, c5 = _pair_descent(best_h.copy(), n, dx, rounds=70, giter=70)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    for alpha in [500, 1500, 4000]:
        if time.time() - t0 > budget_s - 160:
            break
        hc = _lbfgs(best_h, n, dx, target, alpha, 10.0, 5000)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    rng = np.random.RandomState(seed)
    for _ in range(8):
        if time.time() - t0 > budget_s - 60:
            break
        hr = best_h + 0.05 * rng.randn(n)
        hr = _repair(hr, n)
        hr = _lbfgs(hr, n, dx, target, 400, 5.0, 3000)
        hr, cr = _pair_descent(hr, n, dx, rounds=25, giter=55)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    if time.time() - t0 < budget_s - 20:
        hc, c5 = _pair_descent(best_h.copy(), n, dx, rounds=60, giter=80)
        if c5 < best_c5:
            best_c5, best_h = c5, hc

    if time.time() - t0 < budget_s - 5:
        try:
            hc = _slsqp(best_h, n, dx, target, bounds, 6000, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
