            lo = mid
        else:

        else:

                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:

            idxs = [start, start+1 % n, start+2 % n]
            j = rng.randint(n)
            if j in idxs:
                continue
            spec = [(k, h[k]) for k in idxs]
            jv = h[j]
            for d in [0.01, 0.02, 0.03, 0.04, -0.01, -0.02, -0.03, -0.04]:
                valid = True
                for k, _ in spec:
                    h[k] = _ + d
                    if h[k] < 0 or h[k] > 1:
                        valid = False
                        break
                if not valid:
                    for k, v in spec:
                        h[k] = v
                    continue
                h[j] = jv - d * 3
                if h[j] < 0 or h[j] > 1:
                    for k, v in spec:
                        h[k] = v
                    h[j] = jv
                    continue
                c = _c5(h, dx)
                if c < bc - 1e-13:
                    bc, imp = c, True
                    break
            else:
                for k, v in spec:
                    h[k] = v
                h[j] = jv
        if not imp and p > 3:
            break
    return h, bc

def _slsqp_fun(x, a):
    return _sobj(x, a)

def _slsqp_con(x, tgt):
    return float(np.sum(x)) - tgt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = _proj(np.array(initial_h_values, dtype=float), tgt)
    bh, bc = h0.copy(), _c5(h0, dx)

    dc = [0.005, 0.01, 0.02, 0.04, 0.06, 0.1, -0.005, -0.01, -0.02, -0.04, -0.06, -0.1]
    df = [0.001, 0.002, 0.003, 0.005, 0.007, 0.01, -0.001, -0.002, -0.003, -0.005, -0.007, -0.01]
    dv = [0.0003, 0.0006, 0.001, 0.0015, 0.002, 0.003, -0.0003, -0.0006, -0.001, -0.0015, -0.002, -0.003]
    du = [0.0001, 0.0002, 0.0004, 0.0006, 0.0008, -0.0001, -0.0002, -0.0004, -0.0006, -0.0008]

    # Phase 1: FISTA alpha sweep
    for a in [40, 80, 160, 320, 640, 1280, 2560, 5120, 10000]:
        if time.time() > dl - 20:
            break
        h, c = _fista(bh, a, tgt, 2500, dl - 3)
        if c < bc:
            bc, bh = c, h

    # Phase 2: Coarse CD then Fine CD
    if time.time() < dl - 12:
        h, c = _cd(bh, dx, n, 200, rng, dl - 3, dc)
        if c < bc:
            bc, bh = c, h
    if time.time() < dl - 12:
        h, c = _cd(bh, dx, n, 250, rng, dl - 3, df)
        if c < bc:
            bc, bh = c, h

    # Phase 3: Block CD
    if time.time() < dl - 15:
        h, c = _cd_block(bh, dx, n, 150, rng, dl - 3)
        if c < bc:
            bc, bh = c, h
        h, c = _cd(bh, dx, n, 100, rng, dl - 3, df)
        if c < bc:
            bc, bh = c, h

    # Phase 4: Random restarts
    for s in [0.1, 0.2, 0.3, 0.15, 0.25, 0.4, 0.5, 0.08, 0.35, 0.2, 0.45, 0.3]:
        if time.time() > dl - 55:
            break
        hp = _proj(bh + s * rng.randn(n), tgt)
        h, c = _fista(hp, 400, tgt, 2000, dl - 8)
        if c < bc:
            bc, bh = c, h
            if time.time() < dl - 25:
                h2, c2 = _cd(bh, dx, n, 100, rng, dl - 3, df)
                if c2 < bc:
                    bc, bh = c2, h2

    # Phase 5: Binarized restarts (force near-binary structure)
    for nf in [4, 6, 8]:
        if time.time() > dl - 45:
            break
        hr = np.where(bh > 0.5, 1.0, 0.0)
        # Perturb nf random positions to create fractional freedom
        pos = rng.choice(n, size=nf, replace=False)
        for k in pos:
            hr[k] = rng.uniform(0.2, 0.8)
        hb = _proj(hr, tgt)
        h, c = _fista(hb, 500, tgt, 1500, dl - 8)
        if c < bc:
            bc, bh = c, h
            h2, c2 = _cd(bh, dx, n, 100, rng, dl - 3, df)
            if c2 < bc:
                bc, bh = c2, h2

    # Phase 6: Fine and ultra-fine CD
    if time.time() < dl - 10:
        h, c = _cd(bh, dx, n, 300, rng, dl - 3, dv)
        if c < bc:
            bc, bh = c, h
    if time.time() < dl - 6:
        h, c = _cd(bh, dx, n, 500, rng, dl - 2, du)
        if c < bc:
            bc, bh = c, h

    # Phase 7: SLSQP polish
    bnds = [(0.0, 1.0)] * n
    for ap in [3000, 8000, 15000]:
        if time.time() > dl - 18:
            break
        try:
            r = minimize(_slsqp_fun, bh, args=(ap,), method='SLSQP', bounds=bnds,
                        constraints=[{'type':'eq', 'fun': _slsqp_con, 'args': (tgt,)}],
                        options={'maxiter': 400, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc, bh = cf, hf
        except Exception:
            pass

    # Phase 8: Post-SLSQP CD
    if time.time() < dl - 5:
        h, c = _cd(bh, dx, n, 400, rng, dl - 2, du)
        if c < bc:
            bc, bh = c, h

    # Final: one more ultra-fine pass + projection
    if time.time() < dl - 2:
        h, c = _cd(bh, dx, n, 200, rng, dl - 1, du)
        if c < bc:
            bc, bh = c, h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
