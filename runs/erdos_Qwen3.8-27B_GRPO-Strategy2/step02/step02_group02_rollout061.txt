            best_idx = np.argmin(fvals)
            if fvals[best_idx] < best_f:
                best_f = float(fvals[best_idx])
                best_x = float(cand_xs[best_idx])
    
    return best_x, best_f

import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return x

def _exact_coord(h, p):
    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = np.zeros(m)
    B = np.zeros(m)
    hp = h[p]
    for k in range(m):
        s = k - (n - 1)
        coeff = 0.0
        const = corr[k]
        if 0 <= p - s < n:
            idx1 = p - s
            c = 1.0 - h[idx1]
            coeff += c
            const -= c * hp
        if 0 <= p + s < n:
            idx2 = p + s
            c = -h[idx2]
            coeff += c
            const -= c * hp
        A[k] = coeff
        B[k] = const
    f0 = float(np.max(B))
    f1 = float(np.max(A + B))
    best_x = 0.0
    best_f = f0
    if f1 < best_f:
        best_f = f1
        best_x = 1.0
    # Vectorized intersection points
    Da = A[:, None] - A[None, :]
    Db = B[None, :] - B[:, None]
    valid = (np.abs(Da) > 1e-14) & np.triu(np.ones((m, m), dtype=bool), 1)
    if valid.any():
        safe = np.where(Da != 0, Da, 1.0)
        xs = np.where(valid, Db / safe, 0.0)
        ir = valid & (xs > 1e-12) & (xs < 1.0 - 1e-12)
        if ir.any():
            cx = xs[ir]
            fv = np.max(A[:, None] * cx[None, :] + B[:, None], axis=0)
            bi = np.argmin(fv)
            if fv[bi] < best_f:
                best_f = float(fv[bi])
                best_x = float(cx[bi])
    return best_x

def _sweep(h, target):
    n = len(h)
    dx = 2.0 / n
    best_c = _c5(h, dx)
    for sweep in range(500):
        changed = False
        for p in range(n):
            nv = _exact_coord(h, p)
            if abs(nv - h[p]) > 1e-12:
                h[p] = nv
                changed = True
        h = _proj(h, target)
        c = _c5(h, dx)
        if c < best_c:
            best_c = c
        elif not changed:
            break
    return h, best_c

def _pair_step(h, p, q):
    """Try to improve by jointly optimizing h[p] and h[q]."""
    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    hp, hq = h[p], h[q]
    # Coefficients for h[p] and h[q] in each corr[k]
    Ap = np.zeros(m)
    Bp = np.zeros(m)
    Aq = np.zeros(m)
    Bq = np.zeros(m)
    for k in range(m):
        s = k - (n - 1)
        cp = cq = 0.0
        bp = bq = corr[k]
        # h[p] contributions
        if 0 <= p - s < n:
            c = 1.0 - h[p - s]
            cp += c
            bp -= c * hp
        if 0 <= p + s < n:
            c = -h[p + s]
            cp += c
            bp -= c * hp
        # h[q] contributions
        if 0 <= q - s < n:
            c = 1.0 - h[q - s]
            cq += c
            bq -= c * hq
        if 0 <= q + s < n:
            c = -h[q + s]
            cq += c
            bq -= c * hq
        Ap[k] = cp
        Bp[k] = bp - cq * hq  # absorb hq into constant for p-optimization
        Aq[k] = cq
        Bq[k] = bq - cp * hp  # absorb hp into constant for q-optimization
    
    # Feasible region: x_p + x_q = hp + hq, 0<=x_p<=1, 0<=x_q<=1
    # Parametrize: x_p = t, x_q = hp + hq - t
    # t in [max(0, hp+hq-1), min(1, hp+hq)]
    t_lo = max(0.0, hp + hq - 1.0)
    t_hi = min(1.0, hp + hq)
    if t_hi - t_lo < 1e-14:
        return h  # No freedom
    
    # f(t) = max_k (Ap[k]*t + Aq[k]*(hp+hq-t) + Bp[k] + cq_part)
    # = max_k ((Ap[k]-Aq[k])*t + (Aq[k]*(hp+hq) + Bp[k]))
    At = Ap - Aq
    Bt = Aq * (hp + hq) + Bp
    
    # minimize max_k (At[k]*t + Bt[k]) for t in [t_lo, t_hi]
    f_lo = float(np.max(At * t_lo + Bt))
    f_hi = float(np.max(At * t_hi + Bt))
    best_t = t_lo
    best_f = f_lo
    if f_hi < best_f:
        best_f = f_hi
        best_t = t_hi
    
    # Check intersections in (t_lo, t_hi)
    Da = At[:, None] - At[None, :]
    Db = Bt[None, :] - Bt[:, None]
    valid = (np.abs(Da) > 1e-14) & np.triu(np.ones((m, m), dtype=bool), 1)
    if valid.any():
        safe = np.where(Da != 0, Da, 1.0)
        xs = np.where(valid, Db / safe, 0.0)
        ir = valid & (xs > t_lo + 1e-12) & (xs < t_hi - 1e-12)
        if ir.any():
            cx = xs[ir]
            fv = np.max(At[:, None] * cx[None, :] + Bt[:, None], axis=0)
            bi = np.argmin(fv)
            if fv[bi] < best_f:
                best_f = float(fv[bi])
                best_t = float(cx[bi])
    
    if best_f < _c5(h, 2.0/n) - 1e-14:
        h[p] = best_t
        h[q] = hp + hq - best_t
    return h

def _slsqp_obj(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    mval = float(c.max())
    s = np.clip(a * (c - mval), -500.0, 500.0)
    return mval + np.log(float(np.exp(s).sum())) / a

def _slsqp_con(x, target):
    return float(np.sum(x)) - target

                else:

Actually, let me reconsider. For corr[k]:
corr[k] = ∑_i h[i]*(1-h[i-s])

When I vary both h[p] and h[q], the terms involving them are:
- h[p] as first factor: (1-h[p-s])*h[p] if valid
- h[p] as second factor: h[p+s]*(1-h[p]) if valid  
- h[q] as first factor: (1-h[q-s])*h[q] if valid
- h[q] as second factor: h[q+s]*(1-h[q]) if valid
- Cross terms: if p and q are in the same term, e.g., h[p]*(1-h[q]) or h[q]*(1-h[p])

The cross terms make this NOT simply separable! If p-s=q or q-s=p for some lag, then there's a term h[p]*(1-h[q]) which is bilinear.

This makes the pair step more complex. The 2D problem becomes:
min max_k (a_k*t² + b_k*t + c_k)  (if there are cross terms)

min max_k (a_k*t + b_k)  (if no cross terms for that k)

The presence of quadratic terms makes it non-trivial. For simplicity, let me skip the pair step or only do it when there are no cross terms (which is the generic case for random p, q).

Actually, let me check: for a cross term to exist, we need p and q to be related by a lag. Specifically, for shift s, if p-s=q (i.e., q=p-s) or q-s=p (i.e., p=q+s), then there's a cross term. Since s ranges from -(n-1) to n-1, this means |p-q| ≤ n-1, which is always true for any pair. So cross terms always exist!

Wait no. For a specific lag k (shift s), the term h[i]*(1-h[i-s]) involves positions i and i-s. A cross term between h[p] and h[q] in corr[k] exists if {p, q} = {i, i-s} for some valid i. That is, if |p-q| = |s| = |k-(n-1)|.

So for a given pair (p,q) with |p-q| = d, there are at most 2 lags that have cross terms (s=d and s=-d, i.e., k=n-1+d and k=n-1-d).

For those 2 lags, the objective is quadratic in t. For the other m-2 lags, it's linear. So the overall problem is:
min_{t ∈ [t_lo, t_hi]} max(max_{k∈L} (linear in t), max_{k∈Q} (quadratic in t))

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

        if diff > 0:
            idx = np.argsort(1.0 - x)
        else:
            idx = np.argsort(x)

                a = min(diff, 1.0 - x[k])
                x[k] += a
                diff -= a
            else:
                a = min(-diff, x[k])
                x[k] -= a
                diff += a
            if abs(diff) < 1e-13:

    return x

    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = np.zeros(m)
    B = np.zeros(m)

        coeff = 0.0

        if 0 <= p - s < n:
            c = 1.0 - h[p - s]
            coeff += c
            const -= c * hp
        if 0 <= p + s < n:
            c = -h[p + s]
            coeff += c
            const -= c * hp
        A[k] = coeff
        B[k] = const
    f0 = float(np.max(B))
    f1 = float(np.max(A + B))
    best_x = 0.0
    best_f = f0
    if f1 < best_f:
        best_f = f1
        best_x = 1.0
    Da = A[:, None] - A[None, :]
    Db = B[None, :] - B[:, None]
    valid = (np.abs(Da) > 1e-14) & np.triu(np.ones((m, m), dtype=bool), 1)
    if valid.any():
        safe = np.where(Da != 0, Da, 1.0)
        xs = np.where(valid, Db / safe, 0.0)
        ir = valid & (xs > 1e-12) & (xs < 1.0 - 1e-12)
        if ir.any():

            bi = int(np.argmin(fv))
            if fv[bi] < best_f:
                best_f = float(fv[bi])
                best_x = float(cx[bi])
    return best_x

        cp = _c5(hp, dx)
        h = hp
        if cp < best_c:

        if not changed:

    mval = float(c.max())
    s = np.clip(a * (c - mval), -500.0, 500.0)
    return mval + np.log(float(np.exp(s).sum())) / a
