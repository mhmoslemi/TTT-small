def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _subgrad_vec(h, k, n):
    g = np.zeros(n)
    if k >= 0 and k < n:
        i1 = np.arange(k, n)
        s1 = np.arange(0, n - k)
        g[i1] += 1.0 - h[s1]
        i2 = np.arange(0, n - k)
        s2 = np.arange(k, n)
        g[i2] -= h[s2]
    elif k < 0 and -k < n:
        kk = -k
        i1 = np.arange(0, n - kk)
        s1 = np.arange(kk, n)
        g[i1] += 1.0 - h[s1]
        i2 = np.arange(kk, n)
        s2 = np.arange(0, n - kk)
        g[i2] -= h[s2]
    return g

def _smooth_max(x, alpha):

                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})

def _binary_init(n, rng):
