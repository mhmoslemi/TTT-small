            break
        h = _slsqp(h, n, mi, 1e-14, p=p, alpha=a)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 2: Coarse CD then fine CD
    if elapsed() < budget_s - 300:
        h, c = _cd_loop(best_h, n, dx, budget_s - 350, t0, n_grid=17, max_passes=300)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
    if elapsed() < budget_s - 200:
        h, c = _cd_loop(best_h, n, dx, budget_s - 250, t0, n_grid=31, max_passes=300)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 3: Simulated annealing with CD local moves
    sa_budget = min(350, budget_s - 200)
    t_sa_end = max(t0 + sa_budget, budget_s - 150)
    temp_start = 0.008
    temp_end = 0.0005
    n_sa_iters = 60
    sa_iter = 0
    current_h = best_h.copy()
    current_c5 = best_c5
    while time.time() < t_sa_end:
        sa_iter += 1
        frac = min(1.0, sa_iter / n_sa_iters)
        temp = temp_start * (temp_end / temp_start) ** frac
        # Perturb
        h_p = _sa_perturb(current_h, n, rng, 8, 0.15 + 0.3 * (1 - frac))
        # Local CD on perturbed
        h_p, c_p, _ = _cd_pass(h_p, n, dx, 21, t0, t_sa_end)
        delta = c_p - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / temp):
            current_h = h_p
            current_c5 = c_p
            if c_p < best_c5:
                best_c5, best_h = c_p, h_p.copy()
        else:
            # Try targeted perturbation
            h_t = _targeted_perturb(current_h, n, dx, rng, 0.1 * (1 - frac))
            h_t, c_t, _ = _cd_pass(h_t, n, dx, 21, t0, t_sa_end)
            delta_t = c_t - current_c5
            if delta_t < 0 or rng.rand() < np.exp(-delta_t / temp):
                current_h = h_t
                current_c5 = c_t
                if c_t < best_c5:
                    best_c5, best_h = c_t, h_t.copy()

    # Phase 4: SLSQP polish on best
    if elapsed() < budget_s - 120:
        for p, mi in [(32, 500), (64, 400)]:
            if elapsed() > budget_s - 120:
                break
            h = _slsqp(best_h, n, mi, 1e-15, p=p, alpha=400)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()

    # Phase 5: Fine CD
    if elapsed() < budget_s - 60:
        h, c = _cd_loop(best_h, n, dx, budget_s - 70, t0, n_grid=41, max_passes=500)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 6: Very fine CD (last resort improvement)
    if elapsed() < budget_s - 30:
        h, c = _cd_loop(best_h, n, dx, budget_s - 35, t0, n_grid=51, max_passes=300)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 7: Final SLSQP
    if elapsed() < budget_s - 10:
        h = _slsqp(best_h, n, 400, 1e-15, p=64, alpha=500)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
