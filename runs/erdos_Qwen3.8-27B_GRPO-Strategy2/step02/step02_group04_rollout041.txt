import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

        return h, _compute_c5(h, dx)
    try:
        res = minimize(
            lambda x, a=alpha: _smooth_obj(x, a), h,
            method='SLSQP', bounds=bounds, constraints=[eq_con],

        c5 = _compute_c5(hc, dx)
        return hc, c5

                cur.copy()  # keep
        else:
            cur[i], cur[j] = ni - d, nj + d
        elapsed += 1e-5
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP warm-start (30s)
    t_end = t_start + 30
    for alpha in [80, 200]:
        if time.time() > t_end:
            break
        best_h, best_c5 = _slsqp(best_h, n, dx, bounds, eq_con, alpha, 400, t_end)

    # Phase 2: Multi-restart SLSQP (180s)
    t_end = t_start + 210
    alphas = [50, 100, 200, 500, 1000]
    for alpha in alphas:
        if time.time() > t_end or remaining() < 400:
            break
        for restart in range(3):
            if time.time() > t_end:
                break
            if restart == 0:
                start = best_h.copy()
            else:
                start = _repair(best_h + (0.03 + 0.01 * restart) * rng.randn(n), n)
            cand, c5 = _slsqp(start, n, dx, bounds, eq_con, alpha, 600, t_end)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand

    # Phase 3: Simulated annealing exploration (120s)
    if remaining() > 500:
        sa_deadline = time.time() + min(120, remaining() - 450)
        sa_h, sa_c5 = _sa_search(best_h, n, dx, rng, sa_deadline)
        if sa_c5 < best_c5:
            best_c5 = sa_c5
            best_h = sa_h

    # Phase 4: Intensive local search (400s)
    if remaining() > 250:
        ls_deadline = time.time() + min(400, remaining() - 230)
        best_h, best_c5 = _local_search(best_h, n, dx, rng, ls_deadline)

    # Phase 5: SLSQP refinement (150s)
    if remaining() > 100:
        t_end = time.time() + min(150, remaining() - 80)
        for alpha in [200, 500, 1500, 3000]:
            if time.time() > t_end:
                break
            best_h, best_c5 = _slsqp(best_h, n, dx, bounds, eq_con, alpha, 1000, t_end)

    # Phase 6: Fine local search (60s)
    if remaining() > 40:
        ls_deadline = time.time() + min(60, remaining() - 25)
        best_h, best_c5 = _local_search(best_h, n, dx, rng, ls_deadline)

    # Phase 7: Final SLSQP + micro local search
    if remaining() > 20:
        t_end = time.time() + min(15, remaining() - 8)
        best_h, best_c5 = _slsqp(best_h, n, dx, bounds, eq_con, 5000, 2000, t_end)
    if remaining() > 8:
        ls_deadline = time.time() + min(6, remaining() - 2)
        best_h, best_c5 = _local_search(best_h, n, dx, rng, ls_deadline)

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
