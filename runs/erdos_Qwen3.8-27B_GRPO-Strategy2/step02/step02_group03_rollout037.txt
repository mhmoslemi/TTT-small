            for j in oz:
                h[i], h[j] = 0.0, 1.0
                new = cost_fn(h, n)
                d = new - curr
                h[i], h[j] = 1.0, 0.0
                if d < best_d:
                    best_d = d
                    best_pair = (int(i), int(j))
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = 0.0, 1.0
        curr += best_d
        if curr < best:
            best = curr
            best_h = h.copy()
    return best_h, best

def _dswap(h, n, deadline, seed, cost_fn):
    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    curr = cost_fn(h, n)
    best = curr
    best_h = h.copy()
    while time.time() < deadline:
        oi = np.where(h == 1)[0]
        oz = np.where(h == 0)[0]
        found = False
        for _ in range(100):
            if time.time() > deadline:
                return best_h, best
            i1 = int(oi[rng.randint(len(oi))])
            i2 = int(oi[rng.randint(len(oi))])
            j1 = int(oz[rng.randint(len(oz))])
            j2 = int(oz[rng.randint(len(oz))])
            if len({i1, i2, j1, j2}) < 4:
                continue
            h[i1], h[i2], h[j1], h[j2] = 0.0, 0.0, 1.0, 1.0
            new = cost_fn(h, n)
            if new < curr - 1e-12:
                curr = new
                found = True
                if new < best:
                    best = new
                    best_h = h.copy()
            else:
                h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0
    return best_h, best

def _rand_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h
