    lo, hi = -10.0, 10.0

        else:

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx]); h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx]); h[idx] -= sub; diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_grad(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    return e / np.sum(e)

def _slsqp(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    n_c = 2 * n - 1
    def obj(x):
        return _smooth_max(_corr(x), alpha) * dx
    def obj_grad(x):
        c = _corr(x)
        w = _sm_grad(c, alpha)
        grad = np.zeros(n)
        for k in range(n_c):
            if w[k] < 1e-12:
                continue
            sh = k - (n - 1)
            wk = w[k]
            for j in range(n):
                if 0 <= j - sh < n:
                    grad[j] += wk * (1.0 - x[j - sh])
                if 0 <= j + sh < n:
                    grad[j] -= wk * x[j + sh]
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
