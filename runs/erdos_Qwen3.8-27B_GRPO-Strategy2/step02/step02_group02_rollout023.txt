                g[lag:] -= wk * z[:n-lag]

                -0.01, -0.005, -0.003, -0.002, -0.001, -0.0005, -0.0003,
                0.015, 0.02, -0.015, -0.02]
        h, c = _cd(bh, dx, tgt, 3000, rng, deadline - 15, dvals=fine)
        if c < bc:
            bc = c
            bh = h

    # Phase 7: SLSQP polish
    if time.time() < deadline - 10:
        try:
            bnds = [(0.0, 1.0)] * n
            r = minimize(_obj_slsqp, bh, args=(dx,), method='SLSQP', bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': _con_slsqp, 'args': (tgt,)}],
                         options={'maxiter': 800, 'ftol': 1e-15})
            hf = _proj(r.x, tgt)
            cf = _c5(hf, dx)
            if cf < bc:
                bc = cf
                bh = hf
        except Exception:
            pass

    # Phase 8: Final CD after SLSQP
    if time.time() < deadline:
        h, c = _cd(bh, dx, tgt, 1000, rng, deadline)
        if c < bc:
            bc = c
            bh = h

    # Phase 9: Very fine CD
    if time.time() < deadline:
        vfine = [0.002, 0.001, 0.0005, 0.0003, 0.0001, -0.002, -0.001, -0.0005, -0.0003, -0.0001]
        h, c = _cd(bh, dx, tgt, 800, rng, deadline, dvals=vfine)
        if c < bc:
            bc = c
            bh = h

    bh = _proj(bh, tgt)
    bc = _c5(bh, dx)
    return (bh, bc, n)
