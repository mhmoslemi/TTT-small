import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = (h / s) * target
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _smooth_max_val(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_single(h, l, n):
    grad = np.zeros(n)
    idx = np.arange(n)
    vp = (idx + l) < n
    grad[vp] += (1.0 - h[idx[vp] + l])
    vn = (idx - l) >= 0
    grad[vn] -= h[idx[vn] - l]
    return grad

def _smooth_grad_obj(h, n, alpha, dx):
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    val = (m + np.log(np.sum(e)) / alpha) * dx
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        l = k_idx - (n - 1)
        grad += wk * _grad_single(h, l, n)
    return val, grad * dx

def _con_fun(x, target):
    return np.sum(x) - target

def _obj_slsqp(x, alpha, dx):
    return _smooth_max_val(_corr(x), alpha) * dx

def _slsqp(h, n, dx, alpha, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'args': (target,)}]
    try:
        res = minimize(_obj_slsqp, h, args=(alpha, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _gradient_search(h, n, dx, alpha, deadline, rng):
    best = h.copy()
    best_c5 = _c5(best, dx)
    eta = 0.004
    stall = 0
    for it in range(30000):
        if time.time() > deadline:
            break
        val, grad = _smooth_grad_obj(best, n, alpha, dx)
        grad_proj = grad - np.mean(grad)
        if np.linalg.norm(grad_proj) < 1e-13:
            break
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = _repair(best - eta * factor * grad_proj, n)
            c5_try = _c5(h_try, dx)

                best = h_try
                best_c5 = c5_try
                improved = True
                break
        if improved:

            eta = min(eta * 1.04, 0.025)
        else:
            stall += 1
            eta *= 0.65
            if stall > 60 or eta < 1e-11:
                break
    return best, best_c5

    if fine:
