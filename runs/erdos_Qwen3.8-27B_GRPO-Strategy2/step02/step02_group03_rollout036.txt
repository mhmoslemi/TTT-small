def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

                    if nv > best + 0.01:
                        h[i] -= d; h[j] += d
                    else:
                        best = min(best, nv)
        if rng.rand() < 0.15:
            h = _repair(h + 0.005*rng.randn(n), n)
    return h, best
