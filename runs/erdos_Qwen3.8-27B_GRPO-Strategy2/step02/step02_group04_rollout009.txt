def _smag_grad(h, alpha, k):
    corr = _corr(h)
    n = len(h)
    n1 = n - 1
    idx = np.argpartition(corr, -k)[-k:]
    t = corr[idx]
    m = float(t.max())
    s = np.clip(alpha * (t - m), -300, 0)
    e = np.exp(s)
    w = e / e.sum()
    fval = m + float(np.log(e.sum())) / alpha
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        off = ki - n1
        g = np.zeros(n)
        lo1, hi1 = max(0, -off), min(n, n - off)
        if lo1 < hi1:
            g[lo1:hi1] += 1.0 - h[lo1 + off:hi1 + off]
        lo2, hi2 = max(0, off), min(n, n + off)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 - off:hi2 - off]
        grad += wi * g
    return fval, grad

def _obj_fn(x, alpha, k):
    return _smag_grad(x, alpha, k)[0]

def _grad_fn(x, alpha, k):
    return _smag_grad(x, alpha, k)[1]

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        o = np.argsort(-h) if d < 0 else np.argsort(h)
        for idx in o:
            a = min(abs(d), h[idx] if d < 0 else 1.0 - h[idx])
            if d > 0:
                h[idx] += a
                d -= a
            else:
                h[idx] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _slsqp_fast(h, n, alpha, k, maxit, deadline):
    cons = {'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}
    try:
        r = minimize(_obj_fn, h, args=(alpha, k), jac=_grad_fn,
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        if time.time() < deadline:
            return _repair(r.x, n), _c5(r.x, 2.0 / n)
    except Exception:
        pass
    return h, _c5(h, 2.0 / n)

def _lbfgs_pen(h, n, alpha, k, deadline):
    tgt = n / 2.0
    cur_best = h.copy()
    cur_c5 = _c5(h, 2.0 / n)
    lam = 1.0
    for inner in range(40):
        if time.time() >= deadline:
            break
        def _pen_obj(x):
            f, g = _smag_grad(x, alpha, k)
            v = x.sum() - tgt
            return f + lam * v * v
        def _pen_grad(x):
            f, g = _smag_grad(x, alpha, k)
            v = x.sum() - tgt
            return g + 2.0 * lam * v
        r = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 50, 'gtol': 1e-14})
        h = _repair(r.x, n)
        c = _c5(h, 2.0 / n)
        if c < cur_c5:
            cur_c5 = c
            cur_best = h.copy()
        viol = abs(h.sum() - tgt)
        if viol > 0.05:
            lam = min(lam * 5.0, 1e4)
        elif viol < 1e-5:
            lam = max(0.05, lam * 0.5)
    return cur_best, cur_c5

def _guided_local(h, n, dx, rng, deadline, topk=1):
    best = _c5(h, dx)
    h = h.copy()
    n1 = n - 1
    alphas_ls = [50, 100, 200]
    for ls_alpha in alphas_ls:
        if time.time() >= deadline:
            break
        scale = 0.003
        for _outer in range(60000):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            corr = _corr(h)
            idx = np.argpartition(corr, -topk)[-topk:]
            s = np.clip(ls_alpha * (corr[idx] - corr[idx].max()), -300, 0)
            w = np.exp(s)
            w /= w.sum()
            gi = 0.0
            gj = 0.0
            for wi, ki in zip(w, idx):
                off = ki - n1
                if 0 <= i + off < n:
                    gi += wi * (1.0 - h[i + off])
                if 0 <= i - off < n:
                    gi -= wi * h[i - off]
                if 0 <= j + off < n:
                    gj += wi * (1.0 - h[j + off])
                if 0 <= j - off < n:
                    gj -= wi * h[j - off]
            diff = gi - gj
            if abs(diff) < 1e-10:
                d = scale * (1.0 if rng.random() < 0.5 else -1.0)
            else:
                d = -np.sign(diff) * scale
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                d *= 0.1
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best

def _micro_local(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    for scale in [0.0005, 0.0002, 0.0001, 5e-05, 2e-05]:
        if time.time() >= deadline:
            break
        found = False
        for _ in range(30000):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
                found = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not found:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def remain():
        return budget_s - (time.time() - t0)

    n = len(initial_h_values)
    dx = 2.0 / n
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP with analytical gradient, low alpha (60s)
    d = time.time() + 60
    for alpha in [30, 80]:
        if time.time() > d or remain() < 700:
            break
        h, c = _slsqp_fast(h, n, alpha, 3, 400, d)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 2: L-BFGS with penalty, medium alpha (120s)
    d = time.time() + 120
    for alpha in [60, 150, 400]:
        if time.time() > d or remain() < 500:
            break
        h, c = _lbfgs_pen(h, n, alpha, 3, d)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 3: SLSQP high alpha polish (100s)
    d = time.time() + 100
    for alpha in [500, 1500, 4000]:
        if time.time() > d or remain() < 300:
            break
        h, c = _slsqp_fast(h, n, alpha, 3, 500, d)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 4: Guided local search top-1 (250s)
    if remain() > 200:
        d = time.time() + min(250, remain() - 150)
        best_h, best_c5 = _guided_local(best_h, n, dx, rng, d, topk=1)

    # Phase 5: Guided local search top-3 (150s)
    if remain() > 100:
        d = time.time() + min(150, remain() - 80)
        best_h, best_c5 = _guided_local(best_h, n, dx, rng, d, topk=3)

    # Phase 6: Micro local search (80s)
    if remain() > 60:
        d = time.time() + min(80, remain() - 40)
        best_h, best_c5 = _micro_local(best_h, n, dx, rng, d)

    # Phase 7: Final SLSQP + micro (remaining)
    if remain() > 40:
        d = time.time() + max(10, min(80, remain() - 25))
        h, c = _slsqp_fast(best_h, n, 8000, 3, 300, d)
        if c < best_c5:
            best_c5 = c
            best_h = h
    if remain() > 20:
        d = time.time() + max(5, min(20, remain() - 5))
        best_h, best_c5 = _micro_local(best_h, n, dx, rng, d)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
