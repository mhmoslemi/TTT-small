import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return np.max(_corr(h)) * dx


def _lag_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        ij = j - lag
        if 0 <= ij < n:
            g += 1.0 - h[ij]
        jk = j + lag
        if 0 <= jk < n:
            g -= h[jk]
        grad[j] = g
    return grad


def _topk_grad(h, topk=3, sharpness=200):
    c = _corr(h)
    n = len(h)
    top_idx = np.argsort(c)[::-1][:topk]
    c_max = c[top_idx[0]]
    weights = np.exp(sharpness * (c[top_idx] - c_max))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        if w < 1e-8:
            continue
        lag = int(idx) - (n - 1)
        grad += w * _lag_grad(h, lag)
    return c_max, grad


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

    c = _corr(x)
    m = np.max(c)
