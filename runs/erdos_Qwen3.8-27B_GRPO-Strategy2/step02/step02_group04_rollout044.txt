    for i in range(n):
        if d[i] > 1e-14:
            alphas.append(min(1.0, (1.0 - h[i]) / d[i]))
        elif d[i] < -1e-14:
            alphas.append(max(-1.0, -h[i] / d[i]))
    lo, hi = max(min(alphas), -1.0), min(max(alphas), 1.0)
    if lo >= hi: return h, _c5(h, dx)
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    dpt = a + gr * (b - a)
    fc = _c5(h + c * d, dx)
    fd = _c5(h + dpt * d, dx)
    for _ in range(max_iter):
        if fc < fd:
            b, dpt, fd = dpt, c, fc
            c = b - gr * (b - a)
            fc = _c5(h + c * d, dx)
        else:
            a, c, fc = c, dpt, fd
            dpt = a + gr * (b - a)
            fd = _c5(h + dpt * d, dx)
    opt = (a + b) / 2
    return h + opt * d, _c5(h + opt * d, dx)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _obj(x, dx, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def _obj_grad(x, dx, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        off = j - (n - 1)
        lo = max(0, -off); hi = min(n - 1, n - 1 - off)
        for i in range(lo, hi + 1):
            g[i] += w[j] * (1.0 - x[i + off])
            g[i + off] -= w[j] * x[i]
    return g * dx

def _slsqp(h, n, dx, bounds, con, alpha, maxiter, deadline):
    if time.time() >= deadline:
        return h, _c5(h, dx)
    try:
        res = minimize(_obj, h, args=(dx, alpha), jac=_obj_grad,
                       method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        sc = _c5(hc, dx)
        return hc, sc
    except Exception:
        return h, _c5(h, dx)

def _grad_direction(h, dx, top_k=3):
    """Return negative gradient direction of top-k correlations, normalized."""
    n = len(h)
    c = _corr(h)
    top_idx = np.argsort(c)[-top_k:]
    w = c[top_idx] - c[top_idx].min() + 1e-10
    w = w / w.sum()
    g = np.zeros(n)
    for k_idx, j in enumerate(top_idx):
        off = j - (n - 1)
        lo = max(0, -off); hi = min(n - 1, n - 1 - off)
        for i in range(lo, hi + 1):
            g[i] += w[k_idx] * (1.0 - h[i + off])
            g[i + off] -= w[k_idx] * h[i]
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return np.zeros(n)
    return -g / norm

def _line_search(h, d, n, dx):
    """Golden section search for best step along direction d."""
    lo_a, hi_a = -1.0, 1.0
    for i in range(n):
        if d[i] > 1e-14:
            hi_a = min(hi_a, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            lo_a = max(lo_a, -h[i] / d[i])
    if lo_a >= hi_a:
        return h, _c5(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_a, hi_a
    c = b - gr * (b - a)
    dp = a + gr * (b - a)
    fc = _c5(h + c * d, dx)
    fd = _c5(h + dp * d, dx)
    best_a, best_f = 0.0, _c5(h, dx)
    for _ in range(40):
        if fc < fd:
            b, dp, fd = dp, c, fc
            c = b - gr * (b - a)
            fc = _c5(h + c * d, dx)
        else:
            a, c, fc = c, dp, fd
            dp = a + gr * (b - a)
            fd = _c5(h + dp * d, dx)
        if fc < best_f:
            best_f = fc; best_a = c
        if fd < best_f:
            best_f = fd; best_a = dp
        if b - a < 1e-7:
            break
    if best_f < _c5(h, dx) - 1e-13:
        return h + best_a * d, best_f
    return h, _c5(h, dx)

def _peak_local(h, n, dx, rng, deadline):
    """Gradient-informed local search with random perturbations."""
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for scale in scales:
        stall = 0
        while time.time() < deadline and stall < 8:
            improved = False
            for _ in range(3000):
                if time.time() >= deadline:
                    return h, best_c5
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = scale * (2.0 * rng.random() - 1.0)
                ni, nj = h[i] + d, h[j] - d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c5 = _c5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i], h[j] = ni - d, nj + d
            if improved:
                stall = 0
            else:
                stall += 1
    return h, best_c5

def _grad_local(h, n, dx, deadline):
    """Local search using gradient directions."""
    best_c5 = _c5(h, dx)
    h = h.copy()
    for _ in range(200):
        if time.time() >= deadline:
            return h, best_c5
        d = _grad_direction(h, dx, top_k=3)
        for scale in [0.05, 0.02, 0.01, 0.005, 0.001]:
            hh, c5 = _line_search(h, d, n, dx)
            if c5 < best_c5 - 1e-12:
                h = hh.copy()
                best_c5 = c5
                break
        # Also try random direction mixed with gradient
        d2 = d * 0.5 + 0.5 * (_repair(h + 0.01 * np.random.RandomState(hash(time.time()) & 0xffff).randn(n), n) - h)
        d2 /= (np.linalg.norm(d2) + 1e-14)
        for scale in [0.02, 0.01, 0.005]:
            hh, c5 = _line_search(h, d2, n, dx)
            if c5 < best_c5 - 1e-12:
                h = hh.copy()
                best_c5 = c5
                break
    return h, best_c5
