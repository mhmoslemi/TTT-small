        if w[k] < 1e-14:
            continue
        sk = k - (n - 1)
        wk = w[k]
        if sk >= 0:
            g[:n-sk] += wk * (1.0 - h[sk:])
            g[sk:] += -wk * h[:n-sk]
        else:
            sp = -sk
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] += -wk * h[sp:]
    return g * (2.0 / n)

def _sumfunc(x):
    return x.sum() - len(x) * 0.5

def _sumjac(x):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter, t0, budget):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_so_func, h, jac=_so_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _sumfunc, 'jac': _sumjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_descend(h, ntop, max_rounds, t0, budget):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval_c5(h)
    for _r in range(max_rounds):
        if time.time() - t0 > budget - 10:
            break
        c = _overlaps(h)
        top = np.argsort(c)[-ntop:]
        g = _topk_grad(h, top, dx)
        improved = False
        for step in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            h_new = _proj(h - step * g, n * 0.5)
            c5n = _eval_c5(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5

def _coord_descent(h, n_steps, rng, t0, budget):
    n = len(h)
    c5 = _eval_c5(h)
    best = h.copy()
    best_c = c5
    for _ in range(n_steps):
        if time.time() - t0 > budget - 5:
            break
        i = rng.randint(n)
        j = (i + rng.randint(-6, 7)) % n
        if i == j:
            continue
        for d in [8e-4, 4e-4, 2e-4, 1e-4, 5e-5, -8e-4, -4e-4, -2e-4, -1e-4, -5e-5]:
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval_c5(h)
                if c5n < c5 - 1e-18:
                    c5 = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best = h.copy()
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
    return best, best_c

def _fft_perturb(h, rng, n_trials, t0, budget):
    n = len(h)
    best = h.copy()
    best_c = _eval_c5(h)
    half = n // 2
    for _ in range(n_trials):
        if time.time() - t0 > budget - 5:
            break
        H = np.fft.rfft(h)
        n_mod = rng.randint(1, 5)
        for _m in range(n_mod):
            mf = rng.randint(1, max(2, half))
            sc = 1.0 + 0.03 * rng.randn()
            H[mf] *= sc
        h_new = np.fft.irfft(H, n)
        h_new = np.clip(h_new, 0, 1)
        h_new = _proj(h_new, n * 0.5)
        c5n = _eval_c5(h_new)
        if c5n < best_c:
            best_c = c5n
            best = h_new.copy()
    return best, best_c

def _nesterov_opt(h, alpha, steps, step0, t0, budget):
    n = len(h)
    f = _so_func(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() - t0 > budget - 5:
            break
        g = _so_grad(h, alpha)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _so_func(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 3)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _so_func(h, alpha)
    return _repair(h, n)

def _fine_polish(h, n_iters, rng, t0, budget):
    n = len(h)
    c5 = _eval_c5(h)
    best = h.copy()
    best_c = c5
    for _ in range(n_iters):
        if time.time() - t0 > budget - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 4e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval_c5(h)
            if c5n < c5 - 1e-19:
                c5 = c5n
                if c5n < best_c:
                    best_c = c5n
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, best_c

def _block_opt(h, n_blocks, block_size, rng, t0, budget):
    n = len(h)
    c5 = _eval_c5(h)
    best = h.copy()
    best_c = c5
    for _ in range(n_blocks):
        if time.time() - t0 > budget - 5:
            break
        i0 = rng.randint(0, max(1, n - block_size))
        vals = h[i0:i0+block_size].copy()
        dv = 5e-5 * rng.randn(block_size)
        for scale in [1.0, 0.5, 0.2, 2.0]:
            trial = h.copy()
            trial[i0:i0+block_size] = np.clip(vals + scale * dv, 0, 1)
            trial = _proj(trial, n * 0.5)
            c5n = _eval_c5(trial)
            if c5n < c5 - 1e-16:
                h = trial
                c5 = c5n
                if c5n < best_c:
                    best_c = c5n
                    best = h.copy()
                break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval_c5(best_h)
    best_n = n0
    elapsed = time.time() - t0
    # Phase 1: SLSQP warmup at full resolution
    for alpha in [5000, 50000, 300000]:
        if time.time() - t0 > 70:
            break
        h_c = _slsqp_opt(best_h, alpha, 800, t0, budget_s)
        c5v = _eval_c5(h_c)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_c.copy()
    # Phase 2: Active-set top-k descent (multiple k values)
    for nk in [3, 5, 8, 12, 20, 30, 50]:
        if time.time() - t0 > budget_s - 500:
            break
        h2, c52 = _topk_descend(best_h.copy(), nk, 500, t0, budget_s)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
    # Phase 3: Coordinate descent (mass transfer between nearby points)
    if time.time() - t0 < budget_s - 250:
        n_cd = min(int((budget_s - 250 - (time.time() - t0)) / 0.003), 200000)
        best_h, best_c5 = _coord_descent(best_h, n_cd, rng, t0, budget_s)
        best_n = len(best_h)
    # Phase 4: FFT perturbation (explore global structure)
    if time.time() - t0 < budget_s - 150:
        n_fft = min(int((budget_s - 150 - (time.time() - t0)) / 0.001), 50000)
        h3, c53 = _fft_perturb(best_h, rng, n_fft, t0, budget_s)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()
    # Phase 5: Nesterov acceleration on smooth max
    if time.time() - t0 < budget_s - 120:
        h4 = _nesterov_opt(best_h, 100000, 400, 0.002, t0, budget_s)
        c54 = _eval_c5(h4)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4.copy()
    # Phase 6: More top-k descent rounds with varying k
    for nk in [5, 10, 15, 25]:
        if time.time() - t0 > budget_s - 80:
            break
        h5, c55 = _topk_descend(best_h.copy(), nk, 400, t0, budget_s)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5.copy()
    # Phase 7: Block optimization
    if time.time() - t0 < budget_s - 40:
        nb = min(int((budget_s - 40 - (time.time() - t0)) / 0.005), 10000)
        best_h, best_c5 = _block_opt(best_h, nb, 8, rng, t0, budget_s)
    # Phase 8: SLSQP final polish
    if time.time() - t0 < budget_s - 30:
        h6 = _slsqp_opt(best_h, 500000, 1000, t0, budget_s)
        c56 = _eval_c5(h6)
        if c56 < best_c5:
            best_c5 = c56
            best_h = h6.copy()
    # Phase 9: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 2:
        n_pol = min(int((budget_s - 2 - (time.time() - t0)) / 0.001), 50000)
        best_h, best_c5 = _fine_polish(best_h, n_pol, rng, t0, budget_s)
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return (best_h, best_c5, len(best_h))
