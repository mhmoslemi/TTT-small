```python
import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_val(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return dx * (m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha)

def _lse_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _penalty_lse(x, alpha, M, n):
    val = _lse_val(x, alpha)
    constraint = x.sum() - n / 2.0
    return val + M * constraint * constraint

def _penalty_lse_grad(x, alpha, M, n):
    g = _lse_grad(x, alpha)
    constraint = x.sum() - n / 2.0
    g += 2.0 * M * constraint
    return g

def _lbfgs_opt(h, alpha, M, maxiter=20000, time_limit=120):
    n = len(h)
    t_start = time.time()
    bnds = [(0.0, 1.0)] * n
    args = (alpha, M, n)
    try:
        res = fmin_l_bfgs_b(_penalty_lse, h.astype(float), fprime=_penalty_lse_grad,
                           args=args, bounds=bnds, maxiter=maxiter,
                           gtol=1e-12, iprint=0, pgtol=1e-12)
    except Exception:
        res = h.copy()
    return _repair(res, n)

def _slsqp_poly(h, alpha, maxiter=3000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _lse_val(x, alpha)
    
    def grad(x):
        return _lse_grad(x, alpha)
    
    def cons_eq(x):
        return x.sum() - n / 2.0
    
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=bnds, constraints=[{'type': 'eq', 'fun': cons_eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old, kind='linear')

def _coord_descent(h, n_iter, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.01 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _flat_opt(h, n_active=5, n_iter=5000, rng=None):
    """Focus optimization on the worst n_active shifts."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-n_active:]
        # Compute gradient only from top shifts
        g = np.zeros(n)
        for k in top_idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += dx * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= dx * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += dx * (1.0 - h[:n-sp])
                g[:n-sp] -= dx * h[sp:]
        g /= n_active
        # Subtract mean for constraint
        g -= g.mean()
        # Line search
        for step in [0.05, 0.02, 0.01, 0.005, 0.001]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c = h_new, c5n
                break
    return h, c5c
```

```python
import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_val(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return dx * (m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha)

def _lse_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _pen_obj(x, alpha, M, n):
    val = _lse_val(x, alpha)
    c = x.sum() - n / 2.0
    return val + M * c * c

def _pen_grad(x, alpha, M, n):
    g = _lse_grad(x, alpha)
    g += 2.0 * M * (x.sum() - n / 2.0)
    return g

def _lbfgs(h, alpha, M, maxiter=15000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(_pen_obj, h.astype(float), fprime=_pen_grad,
                            args=(alpha, M, n), bounds=bnds,
                            maxiter=maxiter, gtol=1e-13, iprint=0, pgtol=1e-13)
    except Exception:
        res = h.copy()
    return _repair(res, n)

def _slsqp(h, alpha, maxiter=3000):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _o(x):
        return _lse_val(x, alpha)

    def _g(x):
        return _lse_grad(x, alpha)

    def _c(x):
        return x.sum() - n / 2.0

    def _cj(x):
        return np.ones(n)

    try:
        res = minimize(_o, h, jac=_g, method='SLSQP',
                       bounds=bnds, constraints=[{'type': 'eq', 'fun': _c, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _worst_shift_step(h, max_rounds=50):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(max_rounds):
        c = np.correlate(h, 1.0 - h, mode='full')
        ki = int(np.argmax(c))
        sh = (n - 1) - ki
        g = np.zeros(n)
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
        g -= g.mean()
        moved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c, moved = h_new, c5n, True
                break
        if not moved:
            break
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _make_smooth_start(n, rng):

    h = 0.5 + 0.3 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi)) + 0.15 * np.sin(2 * np.pi * x + rng.uniform(0, 2 * np.pi))
    return _repair(h, n)

def _make_binary_pattern(n, pat, rng):
    h = np.zeros(n)
    if pat == 0:
        h[::2] = 1.0
    elif pat == 1:
        per = 3
        h[np.arange(n) % per < 2] = 1.0
    elif pat == 2:
        per = 5
        h[np.arange(n) % per < 3] = 1.0
    elif pat == 3:
        g = max(int(np.sqrt(n)), 10)
        for i in range(n):
            r, c_ = divmod(i, g)
            if (r + c_) % 2 == 0:
                h[i] = 1.0
    else:
        idx = np.random.permutation(n, random_state=rng)
        h[idx[:n // 2]] = 1.0
    h = _repair(h, n)
    return _repair((h > 0.5).astype(float), n)

def _coord_descent(h, n_iter, rng):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.008 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c
```
