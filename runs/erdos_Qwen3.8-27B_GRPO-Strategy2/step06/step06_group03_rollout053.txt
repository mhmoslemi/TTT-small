def _nesterov_subgrad(h, n_iter, step0, top_k_init, top_k_max):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    y = best.copy()
    mu = 0.9
    step = step0
    stuck = 0
    
    for it in range(n_iter):
        # Nesterov lookahead
        y = best + mu * (best - y) if it > 0 else best.copy()
        
        # Compute gradient at y (top-k)
        c = np.correlate(y, 1.0 - y, mode='full')
        top_k = min(top_k_init + it * max(top_k_max - top_k_init, 1) // n_iter, top_k_max, 2 * n - 1)

            if sh >= 0:

                g[sp:] += (1.0 - y[:n - sp])

    best = h.copy()
    best_c = c_cur

        else:

            h[j] += d

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _smax(h, alpha):
    c = _correlate(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _correlate(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lbfgsb(h, alpha, lam, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        c = _correlate(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha + lam * (x.sum() - 0.5 * n) ** 2

    def grad(x):
        n2 = len(x)
        c = _correlate(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n2
        g = np.zeros(n2)
        for k in range(2 * n2 - 1):
            if w[k] < 1e-10:
                continue
            sh = (n2 - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n2 - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n2 - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n2 - sp])
                g[:n2 - sp] -= wk * x[sp:]
        g += 2.0 * lam * (x.sum() - 0.5 * n2)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-13})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_grad(h, top_k):
    n = len(h)
    c = _correlate(h)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _nesterov(h, n_iter, step0):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    y = best.copy()
    mu = 0.85
    step = step0
    stuck = 0
    rng_local = np.random.RandomState(0)
    for it in range(n_iter):
        if it > 0:
            y = best + mu * (best - y)
        top_k = min(2 + it // 100, 30)
        g = _topk_grad(y, top_k)
        z = best - step * g
        new_h = _proj(z, 0.5 * n)
        c_new = _eval(new_h)
        if c_new < best_c - 1e-15:
            y = new_h + mu * (new_h - best)
            best = new_h
            best_c = c_new
            step = min(step * 1.03, 0.3)
            stuck = 0
        else:
            step *= 0.75
            stuck += 1
            if stuck > 40:
                best = _repair(best + 0.015 * rng_local.randn(n), n)
                best_c = _eval(best)
                y = best.copy()
                step = step0
                stuck = 0
    return best, best_c

def _fine(h, n_tries, rng, sa_temp=0.0):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for t in range(n_tries):
        temp = sa_temp * (1.0 - t / n_tries) if sa_temp > 0 else 0.0
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i] = ni
        h[j] = nj
        cn = _eval(h)
        diff = cn - c_cur
        accept = diff < 0 or (temp > 0 and rng.rand() < np.exp(-diff / temp))
        if accept:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best, best_c
