def _grad_w(h, w):

    for k in range(2 * n - 1):

        sh = (n - 1) - k

            g[:n-sh] += w[k] * (1.0 - h[sh:])

        else:
            sp = -sh
            g[sp:] += w[k] * (1.0 - h[:n-sp])
            g[:n-sp] -= w[k] * h[sp:]
    return g

    m = c.max()
    s = np.clip(a * (c - m), -400, 400)

    m = c.max()

    w = np.exp(s - s.max())
