import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    lse = m + np.log(float(np.sum(np.exp(s)))) / alpha
    penalty = beta * (x.sum() - len(x) * 0.5) ** 2
    return lse + penalty

def _grad(x, alpha, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    sev = x.sum() - n * 0.5
    g += 2.0 * beta * sev
    return g

def _lbfgsb(h, alpha, beta, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    x = h.copy()
    best_x = x.copy()
    best_c = _eval(x)
    t0 = time.time()
    # Run in chunks to check deadline
    chunk = min(maxiter, 2000)
    iters_done = 0
    while iters_done < maxiter:
        this_chunk = min(chunk, maxiter - iters_done)
        try:
            res = minimize(_obj, x, jac=_grad, args=(alpha, beta),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': this_chunk, 'ftol': 1e-20,
                                   'gtol': 1e-12, 'maxfun': this_chunk * 3})
            x = _repair(res.x, n)
        except Exception:

        iters_done += this_chunk
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
        if time.time() > deadline:
            break
        # Reset to best for next chunk
        x = best_x.copy()
    return best_x, best_c

def _slsqp(h, alpha, beta, maxiter, ftol):

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,

    try:

def _adam_opt(h0, alpha, n_iter, lr, beta_pen, seed, deadline):

        t += 1

        if c < best_c:

    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

import numpy as np
from scipy.optimize import minimize
import time

_gn = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    lse = m + np.log(float(np.sum(np.exp(s)))) / alpha
    sev = x.sum() - len(x) * 0.5
    return lse + beta * sev * sev

def _grad(x, alpha, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    sev = x.sum() - n * 0.5
    g += 2.0 * beta * sev
    return g

def _sum_eq(x):
    return x.sum() - _gn * 0.5

def _sum_jac(x):
    return np.ones(_gn)

def _lbfgsb(h, alpha, beta, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    x = h.copy()
    best_x = x.copy()
    best_c = _eval(x)
    chunk = min(maxiter, 2000)
    done = 0
    while done < maxiter:
        this = min(chunk, maxiter - done)
        try:
            res = minimize(_obj, x, jac=_grad, args=(alpha, beta),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': this, 'ftol': 1e-20,
                                   'gtol': 1e-12, 'maxfun': this * 4})
            x = _repair(res.x, n)
        except Exception:
            break
        done += this
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
        if time.time() > deadline:
            break
        x = best_x.copy()
    return best_x, best_c

def _slsqp(h, alpha, beta, maxiter, ftol):
    global _gn
    n = len(h)
    _gn = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha, beta),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_opt(h0, alpha0, alpha1, n_iter, lr0, lr1, beta_pen, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if t % 800 == 0:
            x = _repair(x + 0.003 * rng.randn(n), n)
        g = _grad(x, alpha, beta_pen)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        mhat = m_arr / (1 - b1 ** t)
        vhat = v_arr / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _targeted_cd(h, n, rng, deadline, scale_init, n_iters):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    scale = scale_init
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_shift = (n - 1) - int(np.argmax(corr))
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        if scale < 1e-9:
            break
        i = rng.randint(n)
        if rng.random() < 0.6:
            j = (i + max_shift) % n
        else:
            j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < c_cur - 1e-18:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
                corr = np.correlate(x, 1.0 - x, mode='full')
                max_shift = (n - 1) - int(np.argmax(corr))
            scale = max(scale * 0.9997, 1e-9)
        else:
            x = best_x.copy()
            c_cur = best_c
            scale *= 0.985
    return best_x, best_c

def _binary_sa(h0, n, rng, n_iters, deadline):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    T = 0.004
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        x[i] = 1.0 - x[i]
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / T):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
        else:
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
        T = max(T * 0.9999, 0.0001)
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    dl1 = t0 + 150
    for (a0, a1, ni, l0, l1) in [(30, 500, 3000, 0.03, 0.001),
                                   (500, 10000, 3000, 0.015, 0.0005),
                                   (10000, 200000, 2000, 0.005, 0.0002)]:
        if time.time() > dl1:
            break
        x, c = _adam_opt(best_h, a0, a1, ni, l0, l1, 1e4, seed, dl1)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    dl2 = t0 + 280
    for alpha, beta in [(20000, 1e4), (100000, 1e4), (500000, 1e4)]:
        if time.time() > dl2:
            break
        x, c = _lbfgsb(best_h, alpha, beta, 5000, dl2)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    dl3 = t0 + 380
    for alpha in [200000, 2000000]:
        if time.time() > dl3:
            break
        h_new = _slsqp(best_h, alpha, 1e4, 2500, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    dl4 = t0 + 530
    for dn in [-10, -5, 5, 10, 15, 20, -20, 25, 30, -30, 35, -35]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 80 or n_try > 600 or n_try % 2 != 0:
            continue
        h_r = _repair(_resample(best_h, n_try), n_try)
        x, c = _lbfgsb(h_r, 50000, 1e4, 3000, dl4)
        if c < best_c5:
            best_c5, best_h, best_n = c, x.copy(), n_try
        h_s = _slsqp(x, 500000, 1e4, 1500, 1e-20)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_s.copy(), n_try

    dl5 = t0 + 640
    h_comp = _repair(1.0 - best_h, best_n)
    c5c = _eval(h_comp)
    if c5c < best_c5:
        best_c5, best_h = c5c, h_comp.copy()
    for sp in [0.02, 0.05, 0.1]:
        if time.time() > dl5:
            break
        h_p = _repair(best_h + sp * rng.randn(best_n), best_n)
        x, c = _lbfgsb(h_p, 50000, 1e4, 2000, dl5)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    dl6 = t0 + 780
    h_bin = (best_h > 0.5).astype(float)
    h_bin = _repair(h_bin, best_n)
    x_b, c_b = _binary_sa(h_bin, best_n, rng, 200000, dl6)
    if c_b < best_c5:
        best_c5, best_h = c_b, x_b.copy()
    if time.time() < dl6:
        x_p, c_p = _lbfgsb(x_b, 100000, 1e4, 3000, dl6)
        if c_p < best_c5:
            best_c5, best_h = c_p, x_p.copy()

    dl7 = t0 + 950
    x, c = _adam_opt(best_h, 50000, 5000000, 4000, 0.003, 0.00005, 1e4, seed+777, dl7)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl7:
        x, c = _lbfgsb(best_h, 5000000, 1e4, 3000, dl7)
        if c < best_c5:
            best_c5, best_h = c, x.copy()
    if time.time() < dl7:
        h_fin = _slsqp(best_h, 5000000, 1e4, 3000, 1e-22)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    dl8 = t0 + 1080
    x_cd, c_cd = _targeted_cd(best_h, best_n, rng, dl8, 0.001, 80000)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, x_cd.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
