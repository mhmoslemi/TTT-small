import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:

            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

    # Phase 3: L-BFGS-B with growing penalty
    for mu in [50, 500, 5000, 50000]:
        if rem() < 50:
            break
        hp = _lbfgs_opt(best_h, 50000, mu, maxiter=3000)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Phase 4: Subgradient with decreasing step
    if rem() > 100:
        nsim = int(rem() * 0.3)
        best_h, best_c5 = _subgrad_step(best_h, best_n, nsim, 0.05, t0)

    # Phase 5: Active-set SLSQP again (post-subgradient)
    if rem() > 80:
        c = _corr(best_h)
        ti = np.argsort(c)[-3:]
        hp = _slsqp_opt(best_h, 200000, top_idx=ti, maxiter=3000, ftol=1e-19)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Phase 6: Coordinate descent (many small steps)
    if rem() > 40:
        n_cd = int(rem() * 0.15)
        best_h, best_c5 = _coord_descent(best_h, best_n, rng, n_cd, t0)

    # Phase 7: Multi-resolution refinement
    if rem() > 200:
        for dn in [-5, 5, -10, 10, -20, 20, 30, 50]:
            if rem() < 80:
                break
            nn = best_n + dn
            if nn < 50 or nn > 2000:
                continue
            hn = _repair(_resample(best_h, nn), nn)
            for alpha in [20000, 200000]:
                if rem() < 40:
                    break
                c = _corr(hn)
                ti = np.argsort(c)[-3:]
                hn = _slsqp_opt(hn, alpha, top_idx=ti, maxiter=2000, ftol=1e-18)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hn.copy(), nn

    # Phase 8: L-BFGS-B fine-tune at best resolution
    if rem() > 60:
        hp = _lbfgs_opt(best_h, 200000, 50000, maxiter=5000)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Phase 9: Final coordinate descent
    if rem() > 20:
        n_cd = int(rem() * 0.3)
        best_h, best_c5 = _coord_descent(best_h, best_n, rng, n_cd, t0)

    # Phase 10: Final SLSQP polish
    if rem() > 15:
        c = _corr(best_h)
        ti = np.argsort(c)[-2:]
        hp = _slsqp_opt(best_h, 500000, top_idx=ti, maxiter=2000, ftol=1e-19)
        c5 = _eval(hp)
        if c5 < best_c5:
            best_c5, best_h = c5, hp.copy()

    # Final one-point coordinate polish
    if rem() > 5:
        c5c = _eval(best_h)
        for _ in range(20000):
            if time.time() - t0 > budget_s - 2:
                break
            i = rng.randint(best_n)
            for d in [0.0001, -0.0001, 0.00003, -0.00003]:
                best_h[i] += d
                if best_h[i] < 0 or best_h[i] > 1:
                    best_h[i] -= d
                    continue
                hr = _repair(best_h, best_n)
                c5n = _eval(hr)
                if c5n < c5c - 1e-15:
                    best_h = hr
                    c5c = c5n
                    break
                best_h[i] -= d
        best_c5 = c5c

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
