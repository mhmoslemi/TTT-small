import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 500)
    s -= s.max()
    wexp = np.exp(s)
    ws = wexp.sum()
    obj = (cm + np.log(ws) / alpha) * dx
    w = wexp / ws
    g = np.zeros(n)
    active = w > 1e-14
    if not active.any():
        return obj, g
    for k in np.where(active)[0]:
        sh = (n - 1) - k
        wk = w[k] * dx / alpha
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

def _smax_obj(h, alpha):
    return _smax_obj_grad(h, alpha)[0]

def _smax_jac(h, alpha):
    return _smax_obj_grad(h, alpha)[1]

def _cons_eq(x, n):
    return np.array([x.sum() - 0.5 * n])

def _cons_eq_jac(x, n):
    return np.array([np.ones(n)])

def _adam(h, alpha, n_iter, lr, budget_end):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(best_h)
    h = h.copy()
    for t in range(1, n_iter + 1):
        if time.time() > budget_end:
            break
        _, g = _smax_obj_grad(h, alpha)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        h_new = _proj(h - lr * mhat / (np.sqrt(vhat) + eps), n * 0.5)
        sn = len(h_new)
        c_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * 2.0 / sn)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    c_final = _eval(h)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def _slsqp(h, alpha, maxit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smax_obj, h, jac=_smax_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, n_iter, step0, budget_end):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stuck = 0
    for _ in range(n_iter):
        if time.time() > budget_end:
            break
        c = np.correlate(best, 1.0 - h[:0].__class__(1), mode='full') if False else np.correlate(best, 1.0 - best, mode='full')
        idx = np.argsort(c)[-5:]
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - best[sh:])
                if sh > 0:
                    g[sh:] -= best[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - best[:n - sp])
                g[:n - sp] -= best[sp:]
        g *= 2.0 / n
        v = mu * v + g
        improved = False
        for f in [1.0, 0.7, 0.4, 0.2, 0.1]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new
                best_c = c_new
                step = min(step * 1.15, 0.5)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.6
            stuck += 1
            if stuck > 80:
                best = _repair(best + 0.02 * np.random.randn(n), n)
                best_c = _eval(best)
                step = step0
                stuck = 0
                v = np.zeros(n)
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c
