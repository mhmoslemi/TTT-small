from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - x.sum()
    if d > 1e-12:
        idx = np.argmax(1.0 - x)
        x[idx] = min(1.0, x[idx] + d)
    elif d < -1e-12:
        idx = np.argmax(x)
        x[idx] = max(0.0, x[idx] + d)
    return x

def _grad_lse(h, alpha):
    """Gradient of (2/n)*LSE(correlate(h,1-h), alpha) w.r.t. h."""
    n = len(h)

        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])

def _worst_desc(h, n_rounds, t_limit, k_worst):
    """Descent on the worst k shifts."""
    n = len(h)
    target = 0.5 * n
    h = h.copy()
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    for _ in range(n_rounds):
        if time.time() - t0 > t_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(c, -k_worst)[-k_worst:]
        g = np.zeros(n)
        for k in idx:
            sh = k - (n - 1)
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= (2.0 / n) / k_worst
        for lr in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj(h - lr * g, target)
            c5_new = _eval(h_new)
            if c5_new < c5 - 1e-16:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
    return best_h, best_c5

def _upsample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return _proj(np.interp(xn, xo, h_old), 0.5 * n_new)

def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_jac(x, alpha):
    return _grad_lse(x, alpha)

def _sum_eq(x):
    return x.sum() - 0.5 * len(x)

def _sum_eq_jac(x):
    return np.ones(len(x))

def _slsqp_polish(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_jac, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _proj(np.asarray(res.x), 0.5 * n)
    except Exception:
        return _proj(h.copy(), 0.5 * n)

import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - x.sum()
    if d > 1e-10:
        i = int(np.argmax(1.0 - x))
        x[i] = min(1.0, x[i] + d)
    elif d < -1e-10:
        i = int(np.argmax(x))
        x[i] = max(0.0, x[i] + d)
    return x


def _grad_lse(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g


def _pgd(h, alpha, n_iter, t_limit, lr0, momentum=0.0):
    n = len(h)
    tgt = 0.5 * n
    h = _proj(h.copy(), tgt)
    c5 = _eval(h)

    lr = lr0

    vel = np.zeros(n)

        if time.time() - t0 > t_limit:
            break
        g = _grad_lse(h, alpha)
        if ph is not None:
            dy = float(np.dot(g - pg, h - ph))
            yy = float(np.dot(g - pg, g - pg))

                lr = float(np.clip(dy / yy, 1e-6, 0.1))
        if momentum > 0.01:

            step = vel
        else:
            step = lr * g
        step_ok = False
        for _b in range(15):

            cn = _eval(hn)
            if cn < c5 - 1e-16:
                h, c5, step_ok = hn, cn, True
                break
            step *= 0.4
        if not step_ok:
            step *= 0.3
            lr *= 0.5
        if c5 < bc:
            bc, bh = c5, h.copy()
        ph, pg = h.copy(), g.copy()
    return bh, bc


def _wdesc(h, n_rnd, t_lim, kw):
    n = len(h)
    tgt = 0.5 * n
    h = h.copy()
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    t0 = time.time()

        c = np.correlate(h, 1.0 - h, mode='full')

def _ups(h, nn):
    xo = np.linspace(0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0, 2.0, nn, endpoint=False)
    return _proj(np.interp(xn, xo, h), 0.5 * nn)


def _ls_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _ls_jac(x, alpha):
    return _grad_lse(x, alpha)


def _se_fun(x):
    return x.sum() - 0.5 * len(x)


def _se_jac(x):
    return np.ones(len(x))


def _slsqp(h, alpha, mi, ft):
    n = len(h)
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se_fun, 'jac': _se_jac}]
    try:
        r = minimize(_ls_obj, h, jac=_ls_jac, args=(alpha,),
                     method='SLSQP', bounds=b, constraints=cons,
                     options={'maxiter': mi, 'ftol': ft})
        x = np.asarray(r.x, dtype=float)
        if np.all(np.isfinite(x)):
            return _proj(x, 0.5 * n)
    except Exception:
        pass
    return _proj(h.copy(), 0.5 * n)
