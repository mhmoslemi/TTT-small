```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(1.0 + np.sum(np.exp(s))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = (c * dx).max()
    s = np.clip(alpha * (c * dx - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2*n-1)
        w[(c*dx).argmax()] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        wk = w[k] * dx
        sh = (n-1) - k
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n-sh])
            g[:n-sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n-sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter=3000, ftol=1e-18):
    n = len(h)
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':_sum_eq,'jac':_sum_jac}],
                       options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_opt(h0, a0, a1, niter, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.astype(float).copy()
    m_ = np.zeros(n); v_ = np.zeros(n)
    t = 0; b1, b2, eps = 0.85, 0.999, 1e-8
    best = x.copy(); best_c = _eval(x)
    for i in range(niter):
        if time.time() > deadline: break
        t += 1
        f = i / max(niter-1, 1)
        alpha = a0 * (a1/a0)**f
        lr = lr0 * (lr1/lr0)**f
        if t % 800 == 0:
            x = _repair(x + 0.003*rng.randn(n), n)
        g = _grad(x, alpha)
        m_ = b1*m_ + (1-b1)*g
        v_ = b2*v_ + (1-b2)*g*g
        x = x - lr * (m_/(1-b1**t)) / (np.sqrt(v_/(1-b2**t)) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best = x.copy()
    return best, best_c

def _resample(h, nn):
    return np.interp(np.linspace(0,2,nn,endpoint=False),
                     np.linspace(0,2,len(h),endpoint=False), h)

def _peak_shave(h, nsteps, deadline):
    n = len(h)
    dx = 2.0/n
    best = h.copy(); best_c = _eval(h)
    step = 0.005
    for _ in range(nsteps):
        if time.time() > deadline: break
        c = _eval_full(h)
        kstar = int(c.argmax())
        sh = (n-1) - kstar
        g = np.zeros(n)
        if sh >= 0:

            g[:n-sh] -= h[sh:]
        else:
            sp = -sh
            g[:n-sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n-sp]
        g *= dx
        g -= g.mean()
        improved = False
        for _ in range(8):

            ct = _eval(ht)
            if ct < _eval(h):
                h = ht; improved = True; step = min(step*1.1, 0.02); break
            step *= 0.5
        if not improved:

        if step < 1e-8: step = 0.003
        c = _eval(h)
        if c < best_c:

    return best, best_c

def _paired_cd(h, deadline, scale0=0.003):
    n = len(h)
    rng = np.random.RandomState(123)
    x = h.copy(); c_cur = _eval(x)
    best = x.copy(); best_c = c_cur
    scale = scale0
    while time.time() < deadline and scale > 1e-7:
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = scale * rng.randn()
        x[i] += d; x[j] -= d
        x = _repair(x, n)
        cn = _eval(x)
        if cn < c_cur - 1e-16:
            c_cur = cn
            if cn < best_c: best_c = cn; best = x.copy()
            scale = min(scale*1.001, 0.01)
        else:
            x = best.copy(); c_cur = best_c
            scale *= 0.95
    return best, best_c

def _block_slsq
