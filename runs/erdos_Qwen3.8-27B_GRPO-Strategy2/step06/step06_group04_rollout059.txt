import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _lse_val(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, deadline):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        if time.time() > deadline:
            return h.copy()
        res = minimize(_lse_val, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _project(res.x, target)
    except Exception:
        return _project(h, target)

def _adam(h0, a0, a1, nit, lr0, lr1, seed, deadline):
    n = len(h0)
    target = n * 0.5
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(np.float64)
    m = np.zeros(n); v = np.zeros(n)
    t = 0; b1, b2, eps = 0.8, 0.999, 1e-8
    best_x, best_c = x.copy(), _eval(x)
    for i in range(nit):
        if time.time() > deadline:
            break
        t += 1
        f = i / max(nit - 1, 1)
        alpha = a0 * (a1 / a0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        g = _lse_grad(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        x = x - lr * (m / (1 - b1**t)) / (np.sqrt(v / (1 - b2**t)) + eps)
        x = _project(x, target)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _resample(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return _project(np.interp(xn, xo, h), n_new * 0.5)

def _continuous_sa(h0, n, rng, niter, deadline, T0=0.003, cool=0.99997):

    best_x, best_c = x.copy(), c_cur

        if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(T, 1e-8)):

            if c_new < best_c:

        else:

This is quadratic in h, so it's not an LP. It's a QP feasibility problem.

So for binary h: it's a set of linear inequalities + cardinality constraint, which is an ILP. We could potentially solve this with a solver, but n=310 is too large for most solvers.

    n = len(x)
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)

import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _project(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _grad(x, alpha):
    n = len(x)
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        sh = (n - 1) - k
        wk = w[k] * dxv
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g


def _lse_f(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _lse_g(x, alpha):
    return _grad(x, alpha)


def _sum_f(x):
    return x.sum() - len(x) * 0.5


def _sum_g(x):
    return np.ones(len(x))


def _slsqp(h, alpha, mx, dl):
    n = len(h)
    tgt = n * 0.5
    try:
        if time.time() > dl:
            return _project(h, tgt)
        r = minimize(_lse_f, h, jac=_lse_g, args=(alpha,),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sum_f, 'jac': _sum_g}],
                     options={'maxiter': mx, 'ftol': 1e-22})
        return _project(r.x, tgt)
    except Exception:
        return _project(h, tgt)


def _adam(h0, a0, a1, nit, l0, l1, sd, dl):
    n = len(h0)
    tgt = n * 0.5
    rng = np.random.RandomState(sd)
    x = h0.copy().astype(np.float64)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    bx, bc = x.copy(), _eval(x)
    for i in range(nit):
        if time.time() > dl:
            break
        t += 1
        f = i / max(nit - 1, 1)
        a = a0 * (a1 / a0) ** f
        lr = l0 * (l1 / l0) ** f
        g = _grad(x, a)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        x = x - lr * (m / (1 - b1 ** t)) / (np.sqrt(v / (1 - b2 ** t)) + eps)
        x = _project(x, tgt)
        c = _eval(x)
        if c < bc:
            bc = c
            bx = x.copy()
        if t % 2000 == 0:
            x = _project(x + 0.01 * rng.randn(n), tgt)
    return bx, bc
