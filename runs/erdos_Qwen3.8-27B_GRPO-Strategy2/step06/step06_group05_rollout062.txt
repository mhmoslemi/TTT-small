            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize
import time

def _project(g, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(g - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) * 0.5, 0.0, 1.0)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_w(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    e = np.exp(s)
    e /= e.sum()
    return e

def _lssm_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum() + 1e-300) / alpha

def _grad_lssm(w, h, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _pg_solve(h0, alpha, n_iters, lr, n, threshold=1e-13):
    h = h0.copy()
    h_prev = h0.copy()
    for it in range(n_iters):
        if it > 0:
            t = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
            beta = (t - 1.0) / t
            y = h + beta * (h - h_prev)
        else:
            y = h
        c = np.correlate(y, 1.0 - y, mode='full')
        w = _corr_w(c, alpha)
        g = _grad_lssm(w, y, n)
        h_prev = h
        h = _project(y - lr * g, n)
    return h

def _sum_eq(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def _slsqp_polish(h, alpha, maxiter, n):
    def obj(x):
        return _lssm_val(np.correlate(x, 1.0 - x, mode='full'), alpha)
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return _grad_lssm(_corr_w(c, alpha), x, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _project(res.x, n)
    except Exception:
        return _project(h, n)

def _make_binary(n, pat, rng):
    h = np.zeros(n)
    if pat == 0:
        h[::2] = 1.0
    elif pat == 1:
        for i in range(n):
            h[i] = 1.0 if (i % 3) < 2 else 0.0
    elif pat == 2:
        for i in range(n):
            h[i] = 1.0 if (i % 5) < 3 else 0.0
    elif pat == 3:
        g = int(np.sqrt(n))
        for i in range(n):
            r, c = divmod(i, g)
            h[i] = 1.0 if (r + c) % 2 == 0 else 0.0
    else:
        h[:n // 2] = 1.0
    h = _project(h, n)
    h = (h > 0.5).astype(float)
    cnt = int(h.sum())
    idx_1 = np.where(h == 1)[0]
    idx_0 = np.where(h == 0)[0]
    if cnt > n // 2:
        h[rng.choice(idx_1, cnt - n // 2, replace=False)] = 0.0
    elif cnt < n // 2:
        h[rng.choice(idx_0, n // 2 - cnt, replace=False)] = 1.0
    return h

def _anneal_binary(h, t0, t1, n_steps, rng):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = _project(h, n)
    c5c = _eval(h)
    best = c5c
    best_h = h.copy()
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h = _project(h, n)
        c5n = _eval(h)
        if c5n < c5c or (c5n - c5c) < max(t, 1e-7) * 20.0 * (rng.rand() - 0.3):
            c5c = c5n
            if c5n < best:
                best = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            h = _project(h, n)
    return best_h, best

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return _project(np.interp(xn, xo, h_old), n_new)

def _try_sinusoids(n, n_freqs, n_phases, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    best_h = None
    best_c = 1.0
    for f in range(1, n_freqs + 1):
        for ph in range(n_phases):
            phi = 2.0 * np.pi * ph / n_phases + rng.rand() * 0.1
            for amp in [0.3, 0.4, 0.5, 0.6, 0.7]:
                h = 0.5 + amp * np.sin(np.pi * f * x + phi)
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, n)
                c = _eval(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    if best_h is not None:
        return best_h, best_c
    return None, 1.0
