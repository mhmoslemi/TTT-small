def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                if diff > 0:

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

def _adam_optimize(h, n_iter, lr=0.01):
    n = len(h)
    dx = 2.0 / n
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best = _eval(h)
    best_h = h.copy()
    
    for t in range(1, n_iter + 1):
        c = _corr(h)
        # Focus on top shifts
        top_k = np.argsort(c)[-5:]  # top-5 worst shifts
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            wt = c[k] / max(c.sum(), 1e-15)
            if sh >= 0:
                g[:n-sh] += wt * dx * (1.0 - h[sh:])
                if sh > 0: g[sh:] -= wt * dx * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += wt * dx * (1.0 - h[:n-sp])
                g[:n-sp] -= wt * dx * h[sp:]

        h = _repair(h, n)
        
        c5 = _eval(h)
        if c5 < best:

            hb[idx[:n_use // 2]] = 1.0
        hb = (hb > 0.5).astype(float)
        hb = _repair(hb, n_use)
        patterns.append(hb.copy())

    for hb in patterns:
        if elapsed() > 450: break
        c5b = _eval(hb)
        if c5b < best_c5:
            best_c5, best_h, best_n = c5b, hb.copy(), n_use
        hb2 = _slsqp(hb, 80000, 2000)
        c5b2 = _eval(hb2)
        if c5b2 < best_c5:
            best_c5, best_h, best_n = c5b2, hb2.copy(), best_n

    # Phase 4: Intensive SLSQP with very high alpha (250s)
    if elapsed() < budget_s - 250:
        for alpha in [200000, 1000000, 5000000]:
            if elapsed() > budget_s - 200: break
            hn = _slsqp(best_h, alpha, 8000)
            c5 = _eval(hn)
            if c5 < best_c5:
                best_c5, best_h = c5, hn.copy()
            hcomp = _repair(1.0 - best_h, len(best_h))
            hn2 = _slsqp(hcomp, alpha, 4000)
            c5c = _eval(hn2)
            if c5c < best_c5:
                best_c5, best_h = c5c, hn2.copy()

    # Phase 5: Adam on best at multiple scales (150s)
    if elapsed() < budget_s - 100:
        for lr in [0.003, 0.001, 0.0005]:
            if elapsed() > budget_s - 80: break
            ha, c5a = _adam_opt(best_h.copy(), 15000, lr, 5, elapsed())
            if c5a < best_c5:
                best_c5, best_h = c5a, ha.copy()

    # Phase 6: Binary local search on best (150s)
    if elapsed() < budget_s - 60:
        hb = (best_h > 0.5).astype(float)
        hb = _repair(hb, len(hb))
        n_iter_bs = min(50000, int((budget_s - 80 - elapsed()) * 800))
        hb, c5bs = _binary_local(hb, n_iter_bs, rng, elapsed())
        if c5bs < best_c5:
            best_c5, best_h = c5bs, hb.copy()
        hb2 = _slsqp(hb, 200000, 3000)
        c5bs2 = _eval(hb2)
        if c5bs2 < best_c5:
            best_c5, best_h = c5bs2, hb2.copy()

    # Phase 7: Final Adam + SLSQP squeeze (50s)
    if elapsed() < budget_s - 5:
        ha, c5a = _adam_opt(best_h.copy(), 20000, 0.0005, 3, elapsed())
        if c5a < best_c5:
            best_c5, best_h = c5a, ha.copy()
        hn = _slsqp(best_h, 10000000, 5000)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 8: Coordinate micro-descent (20s)
    if elapsed() < budget_s - 3:
        h8 = best_h.copy()
        c5c = _eval(h8)
        nc = len(h8)
        for scale in [0.001, 0.0003, 0.0001]:
            if elapsed() > budget_s - 3: break
            for _it in range(5000):
                if elapsed() > budget_s - 3: break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j: continue
                d = scale * rng.randn()
                ni, nj = h8[i] + d, h8[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
                h8[i], h8[j] = ni, nj
                c5n = _eval(h8)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                else:
                    h8[i], h8[j] = h8[i] - d, h8[j] + d
        if c5c < best_c5 - 1e-14:
            best_c5, best_h = c5c, h8.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
