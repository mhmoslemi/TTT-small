        while h.sum() < half:

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lssm(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

    n = len(h)
    c = _corr(h)
    m = c.max()

    w = np.exp(s - s.max())

    dx = 2.0 / n
    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lssm(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for ki in nz:
        sh = (n - 1) - ki
        wk = w[ki] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _c_eq(h):
    return h.sum() - len(h) * 0.5

def _c_eq_j(h):
    return np.ones(len(h))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_eq, 'jac': _c_eq_j}]
    try:
        res = minimize(_lssm, h, jac=_lssm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _bin_ls(h, n_iter, rng, tl):
    nc = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, nc)
    c5c = _eval(h)
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > tl:
            break
        c = _corr(h)
        top3 = np.argsort(c)[::-1][:3]
        found = False
        for ki in top3:
            if found:
                break
            sh = (nc - 1) - ki
            n_try = max(30, nc // 3)
            for trial in range(n_try):
                if abs(sh) >= nc:
                    continue
                if sh >= 0:
                    i = rng.randint(nc - sh)
                    j = i + sh
                else:
                    sp = -sh
                    i = rng.randint(sp, nc)
                    j = i - sp
                if h[i] < 0.5 or h[j] > 0.5:
                    continue
                h[i], h[j] = 0.0, 1.0
                c5n = _eval(h)
                if c5n < c5c - 1e-15:
                    c5c = c5n
                    found = True
                    break
                h[i], h[j] = 1.0, 0.0
        if not found:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            elif rng.rand() < 0.015:
                c5c = c5n
            else:
                h[i], h[j] = 1.0, 0.0
    return h, c5c

def _make_bin(n, pat, rng):
    h = np.zeros(n)
    half = n // 2
    if pat == 0:
        h[::2] = 1.0
    elif pat == 1:
        for i in range(n):
            h[i] = 1.0 if i % 3 < 2 else 0.0
    elif pat == 2:
        phi = (1 + np.sqrt(5)) / 2
        for i in range(n):
            h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    elif pat == 3:
        idx = rng.permutation(n)
        h[idx[:half]] = 1.0
    elif pat == 4:
        i = np.arange(n)
        bc = np.zeros(n, dtype=int)
        for b in range(10):
            bc += ((i >> b) & 1)
        h = (bc % 2).astype(float)
    elif pat == 5:
        per = rng.randint(7, 13)
        ono = rng.randint(per // 2, per - per // 2 + 1)
        for i in range(n):
            h[i] = 1.0 if (i % per) < ono else 0.0
    else:
        per = rng.randint(11, 21)
        ono = rng.randint(per // 2, per - per // 2 + 1)
        for i in range(n):
            h[i] = 1.0 if (i % per) < ono else 0.0
    while h.sum() > half:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < half:
        h[rng.choice(np.where(h == 0)[0])] = 1
    return h

def _two_swap(h, n_iter, rng, tl):
    nc = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, nc)
    c5c = _eval(h)
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > tl:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if i1 == i2 or j1 == j2:
            continue
        h[i1] = 0.0; h[j1] = 1.0
        h[i2] = 0.0; h[j2] = 1.0
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
        else:
            h[i1] = 1.0; h[j1] = 0.0
            h[i2] = 1.0; h[j2] = 0.0
    return h, c5c
