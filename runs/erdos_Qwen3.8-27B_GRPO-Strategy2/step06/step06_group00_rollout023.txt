            h1 = h[np.clip(idx1, 0, n - 1)]
            h2 = h[np.clip(idx2, 0, n - 1)]
            coeff = np.where(valid1, 1.0 - h1, 0.0) - np.where(valid2, h2, 0.0)
            A = np.zeros(n2)
            A[s + n - 1] = dx * coeff
            A[n - 1 - s] = dx * coeff
            dv = v_grid[:, None] - old_v
            C = c_cur[None, :] + A[None, :] * dv
            C[:, n - 1] += dx * (v_grid * (1.0 - v_grid) - old_v * (1.0 - old_v))
            maxes = C.max(axis=1)
            bi = maxes.argmin()
            if maxes[bi] < cur_max - 1e-14:
                center = v_grid[bi]
                vf = np.clip(center + v_fine, 0, 1)
                dv2 = vf[:, None] - old_v
                C2 = c_cur[None, :] + A[None, :] * dv2
                C2[:, n - 1] += dx * (vf * (1.0 - vf) - old_v * (1.0 - old_v))
                m2 = C2.max(axis=1)
                bi2 = m2.argmin()
                h[j] = vf[bi2]
                c_cur = C2[bi2].copy()
                cur_max = c_cur.max()
                improved = True
        if not improved:
            break
    return h, cur_max

def _nesterov(h, alpha, steps, budget_check):
    n = len(h)
    dx = 2.0 / n
    n2 = 2 * n - 1
    h = h.copy()
    y = h.copy()
    t_k = 1.0
    step = 0.003
    for it in range(steps):
        if budget_check():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        s_arr = np.clip(alpha * (c - c.max()), -500, 500)
        w = np.exp(s_arr - s_arr.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(n2):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - h[i1 + sh])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h + (t_k - 1.0) / t_new * (h - y)
        h_new = _proj_to_sum(y - step * g, n * 0.5)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
        sn = np.clip(alpha * (c_new - c_new.max()), -500, 500)
        wn = np.exp(sn - sn.max())
        wn /= wn.sum()
        f_new = c_new.max() + np.log(wn.sum()) / alpha
        f_old = c.max() + np.log(w.sum()) / alpha
        if f_new < f_old:
            h = h_new
            step = min(step * 1.03, 0.02)
        else:
            step *= 0.6
            h = _proj_to_sum(y - step * g, n * 0.5)
        t_k = t_new
    return h

def _local_2pt(h, n_iter, rng, budget_check):
    n = len(h)
    dx = 2.0 / n
    c_cur = np.correlate(h, 1.0 - h, mode='full') * dx
    cur_max = c_cur.max()
    best_h = h.copy()
    best_c = cur_max
    n2 = 2 * n - 1
    for _it in range(n_iter):
        if budget_check():
            break
        j = rng.randint(n)
        s_off = 1 + rng.randint(0, min(4, n - 1))
        k_idx = s_off + n - 1
        if k_idx >= n2:
            k_idx = n - 1 - s_off
        idx_a = j
        idx_b = j + s_off if j + s_off < n else j - s_off
        if idx_b < 0 or idx_b >= n or idx_b == j:
            continue
        d = 3e-5 * rng.randn()
        nv_a, nv_b = h[j] + d, h[idx_b] - d
        if 0 <= nv_a <= 1 and 0 <= nv_b <= 1:
            h[j], h[idx_b] = nv_a, nv_b
            c_new = c_cur.copy()
            # Update only affected lags
            for kk in range(max(0, k_idx - 1), min(n2, k_idx + 2)):
                sh = (n - 1) - kk
                total = 0.0
                if sh >= 0:
                    for ii in range(n - sh):
                        total += h[ii] * (1.0 - h[ii + sh])
                else:
                    sp = -sh
                    for ii in range(sp, n):
                        total += h[ii] * (1.0 - h[ii - sp])
                c_new[kk] = total * dx
            nm = c_new.max()
            if nm < cur_max - 1e-18:
                cur_max = nm
                if nm < best_c:
                    best_c = nm
                    best_h = h.copy()
            else:
                h[j], h[idx_b] = nv_a - d, nv_b + d
    return best_h, best_c

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    m = len(h) // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coordinate descent at base resolution
    h_cd, c_cd = _coord_descent(h, n0, 60, _bc, grid_pts=40)
    if c_cd < best_c5:
        best_c5 = c_cd
        best_h = h_cd.copy()

    # Phase 2: More coordinate descent with finer grid
    if _el() < budget_s - 200:
        h_cd2, c_cd2 = _coord_descent(best_h, n0, 80, _bc, grid_pts=60)
        if c_cd2 < best_c5:
            best_c5 = c_cd2
            best_h = h_cd2.copy()

    # Phase 3: Nesterov at base res
    if _el() < budget_s - 300:
        h_nr = _nesterov(best_h, 30000, 500, _bc)
        c_nr = _eval(h_nr)
        if c_nr < best_c5:
            best_c5 = c_nr
            best_h = h_nr.copy()
        h_nr2 = _nesterov(best_h, 100000, 400, _bc)
        c_nr2 = _eval(h_nr2)
        if c_nr2 < best_c5:
            best_c5 = c_nr2
            best_h = h_nr2.copy()

    # Phase 4: Upsample to 2x and optimize
    if _el() < budget_s - 500:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        h_hi, c_hi = _coord_descent(h_hi, n1, 30, _bc, grid_pts=35)
        if c_hi < best_c5:
            best_c5 = c_hi
            best_h = h_hi.copy()
        if _el() < budget_s - 350:
            h_nr3 = _nesterov(best_h, 50000, 400, _bc)
            c_nr3 = _eval(h_nr3)
            if c_nr3 < best_c5:
                best_c5 = c_nr3
                best_h = h_nr3.copy()

    # Phase 5: Coordinate descent at higher res
    if _el() < budget_s - 200:
        h_cd3, c_cd3 = _coord_descent(best_h, len(best_h), 40, _bc, grid_pts=50)
        if c_cd3 < best_c5:
            best_c5 = c_cd3
            best_h = h_cd3.copy()

    # Phase 6: 2-point local search
    if _el() < budget_s - 20:
        n_tries = min(int((budget_s - 20 - _el()) / 0.005), 20000)
        best_h, best_c5 = _local_2pt(best_h, n_tries, rng, _bc)

    # Final repair and eval
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
