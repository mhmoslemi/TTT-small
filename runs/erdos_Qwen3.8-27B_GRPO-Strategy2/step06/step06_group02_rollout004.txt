import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_max(h, n, top_m=5):
    """Gradient of sum of top-m correlations (approximation to max)."""
    c = np.correlate(h, 1.0 - h, mode='full')

def _grad_topm(h, n, top_m, temperature):
    # Gradient of soft-max of top-m correlations
    c = np.correlate(h, 1-h, mode='full')
    # Weight by softmax of top values
    c_sorted = np.sort(c)[::-1]
    # Use top-m with softmax weighting
    idx = np.argsort(c)[-top_m:]
    c_vals = c[idx]
    c_max = c_vals.max()
    w = np.exp(temperature * (c_vals - c_max))
    w /= w.sum()
    # Compute gradient
    dx = 2.0/n
    g = np.zeros(n)
    for mk, wk in zip(idx, w):
        sh = (n-1) - mk
        if abs(wk) < 1e-15: continue
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += wk*dx*(1-h[i1+sh])
            if sh > 0:
                g[sh:] -= wk*dx*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*dx*(1-h[:n-sp])
            g[:n-sp] -= wk*dx*h[sp:]
    return g

def _mgd(h, n, n_iter, lr, momentum, top_m, temperature, t_end):
    # Momentum gradient descent
    h = h.copy()
    v = np.zeros(n)
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end: break
        g = _grad_topm(h, n, top_m, temperature)
        # Project gradient to satisfy sum constraint
        g -= g.mean()  # project onto sum-preserving subspace
        v = momentum * v - lr * g
        h_new = h + v
        h_new = _repair(h_new, n)
        cnew = _eval(h_new)
        if cnew < cur:
            h = h_new
            cur = cnew
        else:
            v *= 0.5  # reduce momentum on rejected step
    return h, cur

def _anneal(h, n, n_iter, t_end, t_start=1.0, t_end_param=0.01):
    # Simulated annealing on h
    h = h.copy()
    cur = _eval(h)
    rng = np.random.RandomState(123)
    for it in range(n_iter):
        if time.time() > t_end: break
        t = t_start * (t_end_param/t_start) ** (it/max(n_iter,1))
        # Random perturbation
        delta = rng.randn(n) * 0.05
        delta -= delta.mean()  # preserve sum
        h_new = _repair(h + delta, n)
        cnew = _eval(h_new)
        if cnew < cur or rng.random() < np.exp((cur - cnew)/t):
            h = h_new
            cur = cnew

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_soft(h, n, top_m, temperature):
    c = _corr(h)
    idx = np.argsort(c)[-top_m:]
    cv = c[idx]
    cmx = cv.max()
    w = np.exp(temperature * (cv - cmx))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for mi in range(len(idx)):
        kk = idx[mi]; wk = w[mi]
        if wk < 1e-14:
            continue
        sh = (n - 1) - kk
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * dx * (1.0 - h[i1 + sh])
            if sh > 0:
                g[sh:] -= wk * dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * dx * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * dx * h[sp:]
    return g

def _adam_mgd(h, n, n_iter, lr, top_m, temp, t_end):
    h = h.copy()
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n); v = np.zeros(n)
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _grad_soft(h, n, top_m, temp)
        g -= g.mean()
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** (it + 1))
        vhat = v / (1 - b2 ** (it + 1))
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step -= step.mean()
        h_new = _repair(h - step, n)
        cnew = _eval(h_new)
        if cnew < cur - 1e-15:
            h = h_new; cur = cnew
        else:
            h_new2 = _repair(h - 0.1 * step, n)
            cnew2 = _eval(h_new2)
            if cnew2 < cur - 1e-15:
                h = h_new2; cur = cnew2
    return h, cur

def _anneal(h, n, n_iter, t_end, rng_state=42):
    h = h.copy()
    cur = _eval(h)
    rng = np.random.RandomState(rng_state)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        t = max(0.005, 0.5 * (0.01 / 0.5) ** (it / max(n_iter, 1)))
        sigma = 0.03 * (1 + 2 * t)
        delta = rng.randn(n) * sigma
        delta -= delta.mean()
        h_new = _repair(h + delta, n)
        cnew = _eval(h_new)
        if cnew < cur or rng.random() < np.exp(min(0, (cur - cnew) / t)):
            h = h_new; cur = cnew
    return h, cur

def _two_pt(h, n, n_iter, t_end, rng_state=99):
    h = h.copy()
    cur = _eval(h)
    rng = np.random.RandomState(rng_state)
    c = _corr(h)
    top_idx = np.argsort(c)[-8:]
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.15, 0.15)
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d; h[j] -= d
        cnew = _eval(h)
        if cnew < cur - 1e-14:
            cur = cnew
            c = _corr(h)
            top_idx = np.argsort(c)[-8:]
        else:
            h[i] -= d; h[j] += d
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.clip(np.interp(xn, xo, h_old), 0.0, 1.0)

def _mk_periodic(n, period_idx):
    h = np.zeros(n)
    for i in range(n):
        if (i * period_idx) % (2 * n) < n:
            h[i] = 1.0
    h = _repair(h, n)
    return h
