        if i == j:
            continue
        d = 3e-5 * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return h_cur, c5c

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i*factor:(i+1)*factor] = h[i]
    return _repair(h_hi, n * factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def remaining():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Projected gradient at base res with progressive alpha
    for alpha in [200, 1000, 5000, 20000, 80000, 300000]:
        if remaining() < 300:
            break
        h_cand = _proj_grad(best_h, alpha, 800, t0 + remaining() - 299)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 2: SLSQP polish
    if remaining() > 280:
        h_cand = _slsqp_opt(best_h, 50000, 3000, t0 + remaining() - 279)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 3: Multi-lag descent
    if remaining() > 250:
        h_cand, c5_cand = _multi_lag_descent(best_h, 500, t0 + remaining() - 249)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 4: Alternative initial structures + optimize
    if remaining() > 200:
        freqs = [3, 5, 7, 11]
        for f in freqs:
            if remaining() < 190:
                break
            x = np.arange(n0)
            alt = 0.5 + 0.25 * np.cos(2.0 * np.pi * f * x / n0)
            alt = 0.5 + 0.15 * np.sin(2.0 * np.pi * (f + 2) * x / n0)
            alt = _repair(alt, n0)
            c5_alt = _eval(alt)
            if c5_alt < best_c5 + 0.02:
                alt = _proj_grad(alt, 10000, 300, t0 + remaining() - 185)
                c5_alt = _eval(alt)
                if c5_alt < best_c5:
                    best_c5 = c5_alt
                    best_h = alt.copy()

    # Phase 5: Upsample to 2x
    if remaining() > 400:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        for alpha in [500, 5000, 30000, 100000]:
            if remaining() < 360:
                break
            h_hi = _proj_grad(h_hi, alpha, 400, t0 + remaining() - 359)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if remaining() > 320:
            h_hi = _slsqp_opt(best_h, 30000, 2000, t0 + remaining() - 319)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if remaining() > 280:
            h_cand, c5_cand = _multi_lag_descent(best_h, 400, t0 + remaining() - 279)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
                best_n = n1

    # Phase 6: More projected gradient with different alphas
    if remaining() > 150:
        for alpha in [50000, 200000, 500000]:
            if remaining() < 140:
                break
            h_cand = _proj_grad(best_h, alpha, 600, t0 + remaining() - 139)
            c5 = _eval(h_cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()

    # Phase 7: SLSQP final polish
    if remaining() > 100:
        h_cand = _slsqp_opt(best_h, 200000, 5000, t0 + remaining() - 99)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 8: SA pair swaps
    if remaining() > 80:
        n_iter = min(int((remaining() - 70) / 0.003), 50000)
        best_sa, c5_sa = _sa_pairs(best_h, best_n, t0 + remaining() - 70, n_iter, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = best_sa.copy()

    # Phase 9: Coordinate descent
    if remaining() > 20:
        h_cand, c5_cand = _coord_descent(best_h, best_n, t0 + remaining() - 15,
                                          min(int((remaining() - 15) / 0.002), 80000), rng)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 10: Final micro-refinement
    if remaining() > 10:
        for alpha in [100000, 500000]:
            if remaining() < 8:
                break
            h_cand = _proj_grad(best_h, alpha, 200, t0 + remaining() - 7)
            c5 = _eval(h_cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
