import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlate(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_full(x, alpha):
    # Gradient of soft-max objective
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)

            continue

            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:

            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _adam(h0, alpha_start, alpha_end, n_iter, lr_start, lr_end, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        g = _grad_full(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

def _swap_sa(h0, n, rng, n_iters, deadline, T_start, T_end):
    """Binary swap-based simulated annealing"""
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    ones = np.where(x > 0.5)[0]
    zeros = np.where(x < 0.5)[0]
    c_cur = _eval(x)
    corr_cur = _correlate(x)
    best_x = x.copy()
    best_c = c_cur
    T = T_start
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / n_iters)
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oi, oz = ones[i], zeros[j]
        if oi == oz:

        x[oi] = 0.0
        x[oz] = 1.0
        corr_new = _correlate(x)
        c_new = corr_new.max()
        if c_new <= c_cur or rng.random() < np.exp(-(c_new - c_cur) / max(T, 1e-10)):
            ones[i] = oz
            zeros[j] = oi
            c_cur = c_new
            corr_cur = corr_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
        else:
            x[oi] = 1.0
            x[oz] = 0.0
    return best_x, best_c

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def _grad(x):
        return _grad_full(x, alpha)
    def _se(x):
        return x.sum() - n * 0.5
    def _sj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_soft(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _corr_arr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _obj_soft(x, alpha):
    c = _corr_arr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _adam(h0, a0, a1, nit, l0, l1, seed, dl):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.astype(float).copy()
    m = np.zeros(n); v = np.zeros(n); t = 0
    bx = x.copy(); bc = _eval(x)
    for i in range(nit):
        if time.time() > dl:
            break
        t += 1
        f = i / max(nit - 1, 1)
        a = a0 * (a1 / a0) ** f
        lr = l0 * (l1 / l0) ** f
        g = _grad_soft(x, a)
        m = 0.9 * m + 0.1 * g
        v = 0.99 * v + 0.01 * g * g
        mh = m / (1 - 0.9 ** t)
        vh = v / (1 - 0.99 ** t)
        x = x - lr * mh / (np.sqrt(vh) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < bc:
            bc = c; bx = x.copy()
    return bx, bc

def _swap_sa(h0, n, rng, nit, dl, T0, T1):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    cc = _eval(x)
    bx = x.copy(); bc = cc
    for it in range(nit):
        if time.time() > dl:
            break
        T = T0 * (T1 / T0) ** (it / max(nit - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or x[i] == x[j]:
            continue
        x[i], x[j] = x[j], x[i]
        cn = _eval(x)
        if cn <= cc or rng.random() < np.exp(-(cn - cc) / max(T, 1e-12)):
            cc = cn
            if cn < bc:
                bc = cn; bx = x.copy()
        else:
            x[i], x[j] = x[j], x[i]
    return bx, bc

def _greedy_binary(h0, n, rng, dl, max_rounds=200):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    bx = x.copy(); bc = c_cur
    for r in range(max_rounds):
        if time.time() > dl:
            break
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        best_c = c_cur
        best_pair = None
        n_try = min(5000, len(ones) * len(zeros))
        for _ in range(n_try):
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            a, b = ones[i], zeros[j]
            x[a] = 0; x[b] = 1
            cn = _eval(x)
            x[a] = 1; x[b] = 0
            if cn < best_c - 1e-15:
                best_c = cn
                best_pair = (a, b)
        if best_pair is None:
            break
        x[best_pair[0]] = 0
        x[best_pair[1]] = 1
        c_cur = best_c
        if c_cur < bc:
            bc = c_cur; bx = x.copy()
    return bx, bc

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        return _obj_soft(x, alpha)
    def _gf(x):
        return _grad_soft(x, alpha)
    def _se(x):
        return x.sum() - n * 0.5
    def _sj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_f, h, jac=_gf, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
