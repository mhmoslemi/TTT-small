                h = _repair(h - 0.1 * lr * g, n)
                cn = _eval(h)
            lr *= decay
        if cn < best: best = cn
        if lr < 1e-12: break
    return h, best

def _lbfgs_opt(h, alpha, maxit, t_end):
    n = len(h)
    pen = 1e10
    def _obj(x):
        c = _corr(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        sm = s - s.max()
        val = m + (float(np.log(np.sum(np.exp(sm)))) + s.max()) / alpha
        val += pen * (x.sum() - n * 0.5) ** 2
        return val
    def _ojac(x):
        c = _corr(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14: continue
            g += w[k] * _grad_shift(x, k, n)
        g += 2.0 * pen * (x.sum() - n * 0.5)
        return g
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj, h, jac=_ojac, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': maxit * 3})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp_opt(h, alpha, maxit):
    n = len(h)
    def _f(x):
        c = _corr(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        sm = s - s.max()
        return m + (float(np.log(np.sum(np.exp(sm)))) + s.max()) / alpha
    def _jf(x):
        c = _corr(x) * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14: continue
            g += w[k] * _grad_shift(x, k, n)
        return g
    def _se(x): return x.sum() - n * 0.5
    def _sej(x): return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sej}]
    try:
        r = minimize(_f, h, jac=_jf, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal_binary(h, t0, t1, t_floor, n_steps, rng, t_end):
    n = len(h)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5 = _eval(h)
    best = c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    c_arr = _corr(h)
    for step in range(n_steps):
        if time.time() > t_end: break
        t = max(t0 * (t1 / t0) ** (step / max(n_steps - 1, 1)), t_floor)
        if step % 200 == 0:
            c_arr = _corr(h)
            worst_shift = (n - 1) + int(np.argmax(np.delete(c_arr, n - 1)))
            ws = int(np.delete(np.argsort(c_arr), np.where(np.arange(len(c_arr)) == n - 1)[0][0])[-1])
            sh = (n - 1) - ws
            if rng.rand() < 0.3 and abs(sh) > 0 and abs(sh) < n:
                si = sh if sh > 0 else -sh
                if sh > 0:
                    cands_i = zeros[zeros + sh < n]
                    cands_j = ones[ones < n - sh]
                    if len(cands_i) > 0 and len(cands_j) > 0:
                        i = cands_i[rng.randint(len(cands_i))]
                        j = cands_j[rng.randint(len(cands_j))]
                        if j - i == sh:
                            h[i], h[j] = 1.0, 0.0
                            h2 = _repair(h, n)
                            c2 = _eval(h2)
                            if c2 < c5 or rng.rand() < np.exp(-(c2 - c5) / t):
                                c5 = c2; h = h2
                                ones = np.where(h > 0.5)[0]; zeros = np.where(h <= 0.5)[0]
                                if c2 < best: best = c2; best_h = h.copy()
                                continue
                            else:
                                h[i], h[j] = 0.0, 1.0
                                h = _repair(h, n)
                else:
                    sp = -sh
                    cands_i = zeros[zeros + sp > 0]
                    cands_j = ones[ones - sp >= 0]
                    if len(cands_i) > 0 and len(cands_j) > 0:
                        i = cands_i[rng.randint(len(cands_i))]
                        j = cands_j[rng.randint(len(cands_j))]
                        if i - j == sp:
                            h[i], h[j] = 1.0, 0.0
                            h2 = _repair(h, n)
                            c2 = _eval(h2)
                            if c2 < c5 or rng.rand() < np.exp(-(c2 - c5) / t):
                                c5 = c2; h = h2
                                ones = np.where(h > 0.5)[0]; zeros = np.where(h <= 0.5)[0]
                                if c2 < best: best = c2; best_h = h.copy()
                                continue
                            else:
                                h[i], h[j] = 0.0, 1.0
                                h = _repair(h, n)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        h2 = _repair(h, n)
        c2 = _eval(h2)
        if c2 < c5 or rng.rand() < np.exp(-(c2 - c5) / max(t, 1e-9)):
            c5 = c2; h = h2
            ones = np.where(h > 0.5)[0]; zeros = np.where(h <= 0.5)[0]
            if c2 < best: best = c2; best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            h = _repair(h, n)
    return best_h, best

def _coord_desc(h, n_it, t_end, rng):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for it in range(n_it):
        if time.time() > t_end: break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue
        d = 0.003 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c5 - 1e-15:
            c5 = cn
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

def _make_binary(n, pat, rng):
    h = np.zeros(n)
    if pat == 0:
        h[::2] = 1.0
    elif pat == 1:
        for i in range(n): h[i] = 1.0 if (i % 5) < 3 else 0.0
    elif pat == 2:
        for i in range(n): h[i] = 1.0 if (i % 7) < 4 else 0.0
    elif pat == 3:
        g = int(np.sqrt(n))
        for i in range(n):
            r, c_ = divmod(i, g)
            h[i] = 1.0 if (r + c_) % 2 == 0 else 0.0
    else:
        h = (rng.rand(n) < 0.5).astype(float)
    h = (h > 0.5).astype(float)
    return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def t_rem(): return max(budget_s - (time.time() - t_start), 0)

    # Phase 1: Quick PGD polish on initial (20s)
    t1 = t_start + 20
    h1, c1 = _pgd_topk(best_h, 5000, 5, 0.02, 0.999, t1)
    if c1 < best_c5: best_c5, best_h = c1, h1.copy()

    # Phase 2: L-BFGS with increasing alpha (120s)
    t2 = t_start + 140
    for alpha in [5000, 20000, 100000, 500000]:
        if time.time() > t2: break
        h2 = _lbfgs_opt(best_h, alpha, 800, t2)
        c2 = _eval(h2)
        if c2 < best_c5 - 1e-12:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: PGD with tightened topk (80s)
    t3 = t_start + 220
    for tk in [3, 5, 8, 15]:
        if time.time() > t3: break
        h3, c3 = _pgd_topk(best_h, 8000, tk, 0.01, 0.998, t3)
        if c3 < best_c5 - 1e-12:
            best_c5, best_h = c3, h3.copy()

    # Phase 4: Binary annealing with structure-aware moves (250s)
    t4 = t_start + 470
    h_bin = (best_h > 0.5).astype(float)
    h_bin = _repair(h_bin, n0)
    n_steps = 30000
    ha, ca = _anneal_binary(h_bin, 1.0, 0.01, 0.0001, n_steps, rng, t4)
    if ca < best_c5 - 1e-12:
        best_c5, best_h = ca, ha.copy()
    # Polish the annealed result
    hp = _lbfgs_opt(ha, 100000, 500, min(t4 + 30, t_start + budget_s - 100))
    cp = _eval(hp)
    if cp < best_c5 - 1e-12:
        best_c5, best_h = cp, hp.copy()

    # Phase 5: Try small resolutions (150s)
    t5 = t_start + 620
    for ntry in [100, 150, 200, 250]:
        if time.time() > t5: break
        ht = _repair(_resample(best_h, ntry), ntry)
        ht = _lbfgs_opt(ht, 50000, 600, min(time.time() + 25, t5))
        ct = _eval(ht)
        if ct < best_c5 - 5e-5:
            best_c5, best_h, best_n = ct, ht.copy(), ntry

    # Phase 6: SLSQP fine-tune (100s)
    t6 = t_start + 720
    for alpha in [200000, 1000000, 5000000]:
        if time.time() > t6: break
        h6 = _slsqp_opt(best_h, alpha, 3000)
        c6 = _eval(h6)
        if c6 < best_c5 - 1e-12:
            best_c5, best_h = c6, h6.copy()

    # Phase 7: Second annealing round on best (100s)
    t7 = t_start + 820
    h7b = (best_h > 0.5).astype(float)
    h7b = _repair(h7b, len(best_h))
    ha7, ca7 = _anneal_binary(h7b, 0.3, 0.005, 0.00005, 20000, rng, t7)
    ha7 = _lbfgs_opt(ha7, 500000, 400, min(t7 + 20, t_start + budget_s - 60))
    ca7 = _eval(ha7)
    if ca7 < best_c5 - 1e-12:
        best_c5, best_h = ca7, ha7.copy()

    # Phase 8: Final PGD + coordinate descent (60s)
    t8 = t_start + budget_s - 10
    h8, c8 = _pgd_topk(best_h, 10000, 3, 0.005, 0.999, t8)
    if c8 < best_c5 - 1e-12:
        best_c5, best_h = c8, h8.copy()
    h8b, c8b = _coord_desc(best_h, 20000, t_start + budget_s - 2, rng)
    if c8b < best_c5 - 1e-12:
        best_c5, best_h = c8b, h8b.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
