def _shrink_direction(h, active_lags):
    """Find direction that reduces all active lags simultaneously."""
    n = len(h)
    # For each active lag, compute gradient
    # Find d that minimizes max_k grad_k . d
    # This is: minimize t s.t. grad_k . d <= t for all k
    #          ||d||_inf <= 1 (for simplicity)
    grads = []
    for k in active_lags:
        sh = (n-1) - k
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
        grads.append(g)
    grads = np.array(grads)  # (K, n)
    
    # Find d minimizing max_k grads[k] @ d
    # Using scipy linprog: min t s.t. grads @ d <= t, -1 <= d <= 1
    # This has K + n + 1 = K + n + 1 variables and K constraints + 2n bounds
    # For n=560, K=5: 566 variables, 5 constraints - feasible
    
    from scipy.optimize import linprog

    c5c = _eval(h)
    for _r in range(max_rounds):
        if el_fn() > budget_s - 10:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-n_lags:]
        d = _lp_direction(h, top_k)
        if d is None:
            break
        improved = False
        for t in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = _proj_sum(h + t * d, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _nesterov(h, alpha, steps, step0, el_fn, budget_s):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _s in range(steps):
        if el_fn() > budget_s - 5:
            break
        g = _smooth_max_grad(h, alpha)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 3)
        else:
            step *= 0.5
            h = _proj_sum(h - step * v, n * 0.5)
            f = _eval(h)
    return h, f

def _sim_anneal(h, n_iters, T0, Tf, rng, el_fn, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _it in range(n_iters):
        if el_fn() > budget_s - 5:
            break
        T = T0 * (Tf / T0) ** (_it / max(n_iters, 1))
        i = rng.randint(n)
        blk = 3 + rng.randint(0, 10)
        j = rng.randint(blk)
        idx = np.arange(i, min(i + blk, n))
        d = rng.randn(len(idx)) * 0.02
        h_old = h[idx].copy()
        h[idx] = np.clip(h[idx] + d, 0, 1)
        h = _proj_sum(h, n * 0.5)
        c5n = _eval(h)
        delta = c5n - c5c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-9)):
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h = _proj_sum(np.clip(h - d[np.argsort(np.argsort(idx))[:len(idx)]], 0, 1), n * 0.5)
            h = best_h.copy() if c5c > best_c5 + 0.001 else _repair(
                np.clip(h, 0, 1), n)
            h[idx] = h_old
            h = _proj_sum(h, n * 0.5)
            c5c = _eval(h)
    return best_h, best_c5

def _local_fine(h, n_iters, rng, el_fn, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    scales = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6]
    si = 0
    for _it in range(n_iters):
        if el_fn() > budget_s - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = scales[si] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n0 = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP at low resolution
    n_low = max(60, n0 // 4)
    f_low = n0 // n_low
    h_low = _downsample(h, f_low)
    h_low = _repair(h_low, n_low)
    for alpha in [5000, 50000]:
        if _el() > 40:
            break
        h_low = _slsqp(h_low, alpha, 500, 1e-17)
        h_hi = _repair(_upsample(h_low, f_low), n0)
        c5h = _eval(h_hi)
        if c5h < best_c5:
            best_c5 = c5h
            best_h = h_hi.copy()
            best_n = n0

    # Phase 2: SLSQP at full resolution
    for alpha in [10000, 100000, 500000]:
        if _el() > 180:
            break
        hc = _slsqp(best_h.copy(), alpha, 1500, 1e-18)
        c5v = _eval(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()
            best_n = len(hc)

    # Phase 3: Nesterov refinement
    if _el() < budget_s - 300:
        for alpha in [50000, 200000, 500000]:
            if _el() > budget_s - 250:
                break
            hn, cn = _nesterov(best_h.copy(), alpha, 500, 0.003, _el, budget_s)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
                best_n = len(hn)

    # Phase 4: LP-based active-lag refinement (key innovation)
    for nl in [3, 5, 7, 4, 6]:
        if _el() > budget_s - 200:
            break
        h_lp, c5_lp = _lp_refine(best_h.copy(), nl, 200, _el, budget_s)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            best_n = len(h_lp)
        # Follow with Nesterov after LP
        if _el() < budget_s - 100:
            hn, cn = _nesterov(best_h.copy(), 300000, 200, 0.002, _el, budget_s)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
                best_n = len(hn)

    # Phase 5: Simulated annealing with structured perturbations
    if _el() < budget_s - 150:
        n_sa = min(int((budget_s - 150 - _el()) / 0.05), 2000)
        h_sa, c5_sa = _sim_anneal(best_h.copy(), n_sa, 5e-4, 1e-7, rng, _el, budget_s)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = len(h_sa)

    # Phase 6: More LP refinement after SA
    if _el() < budget_s - 80:
        for nl in [3, 5, 4]:
            if _el() > budget_s - 30:
                break
            h_lp, c5_lp = _lp_refine(best_h.copy(), nl, 300, _el, budget_s)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
                best_n = len(h_lp)

    # Phase 7: Final multi-lag gradient descent
    if _el() < budget_s - 40:
        n = len(best_h)
        c5c = _eval(best_h)
        for _r in range(500):
            if _el() > budget_s - 10:
                break
            c = _corr(best_h)
            top_k = np.argsort(c)[-5:]
            g = np.zeros(n)
            for k in top_k:
                g += _lag_grad(best_h, k)
            g /= len(top_k)
            improved = False
            for t in [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
                h_new = _proj_sum(best_h - t * g, n * 0.5)
                c5n = _eval(h_new)
                if c5n < c5c - 1e-15:
                    best_h = h_new
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_n = n
                    improved = True
                    break
            if not improved:
                break
        best_h = _repair(best_h, best_n)
        best_c5 = _eval(best_h)

    # Phase 8: Ultra-fine local search
    if _el() < budget_s - 3:
        n_tries = min(int((budget_s - 3 - _el()) / 0.0005), 100000)
        best_h, best_c5 = _local_fine(best_h, n_tries, rng, _el, budget_s)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
