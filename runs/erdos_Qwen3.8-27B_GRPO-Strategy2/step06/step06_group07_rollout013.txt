import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vectors(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lagged(h, lam):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if lam[k] < 1e-10:
            continue
        s = (n - 1) - k
        w = lam[k] * dx
        if s >= 0:
            i1 = np.arange(n - s)
            g[i1] += w * (1.0 - h[i1 + s])
            if s > 0:
                i2 = np.arange(s, n)
                g[i2] -= w * h[i2 - s]
        else:
            sp = -s
            i1 = np.arange(sp, n)
            g[i1] += w * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w * h[i2 + sp]
    return g

def _proj_sum(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _exp_weights(h0, n_inner=15, n_outer=200, tau_init=0.05, tau_end=0.001, seed=42):
    n = len(h0)
    n_lags = 2 * n - 1
    h = h0.copy()
    lam = np.ones(n_lags) / n_lags
    best_c = _eval(h)
    best_h = h.copy()
    
    for oit in range(n_outer):
        frac = oit / max(n_outer - 1, 1)
        tau = tau_init * (tau_end / tau_init) ** frac
        
        # Inner: projected gradient to minimize E = sum_k lam[k] * f_k
        lr = 0.05 / (1 + 0.1 * oit)
        for iit in range(n_inner):
            g = _grad_lagged(h, lam)
            h = _proj_sum(h - lr * g, n)
        
        c = _corr_vectors(h)
        if c.max() < best_c:
            best_c = c.max()
            best_h = h.copy()
        
        # Update lam
        c_clipped = np.clip(c / tau, -500, 500)
        lam = np.exp(c_clipped - c_clipped.max())
        lam /= lam.sum()
    
    return best_h, best_c

def _slsqp_active(h, top_k=10, maxiter=3000):
    """SLSQP optimizing for the top-k active lags."""
    n = len(h)
    dx = 2.0 / n
    c = _corr_vectors(h)
    top_indices = np.argsort(c)[-top_k:]
    
    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full') * dx
        return cc[top_indices].sum() + 0.1 * cc.max()
    
    def _single_lag_grad(x, k):
        s = (n - 1) - k
        g = np.zeros(n)
        if s >= 0:
            i1 = np.arange(n - s)
            g[i1] += (1.0 - x[i1 + s])
            if s > 0:
                i2 = np.arange(s, n)
                g[i2] -= x[i2 - s]
        else:
            sp = -s
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= x[i2 + sp]
        return g * dx
    
    def grad(x):
        cc = np.correlate(x, 1.0 - x, mode='full') * dx
        g = np.zeros(n)
        for k in top_indices:
            g += _single_lag_grad(x, k)
        # gradient of max term (approximate: use the argmax)
        kmax = cc.argmax()
        g += 0.1 * _single_lag_grad(x, kmax)
        return g
    
    bnds = [(0.0, 1.0)] * n
    def sum_eq(x): return x.sum() - n * 0.5
    def sum_jac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
