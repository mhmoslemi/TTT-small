def _lsexp_on_lifted(x, alpha, n):
    """x has n-1 elements. h has n elements where h[-1] = n/2 - sum(x)."""
    h = np.empty(n)
    h[:n-1] = x
    h[n-1] = n*0.5 - x.sum()
    if h[n-1] < -1e-10 or h[n-1] > 1 + 1e-10:
        return 1e10, np.ones(n-1)
    return _lsexp_obj_and_grad_from_h(h, alpha)

def _lsexp_obj_and_grad_from_h(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()

    # Gradient w.r.t. h
    g_h = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        g_h += w[k] * _grad_overlap(h, k, dx)
    # Gradient w.r.t. x (first n-1 elements)
    # h[i] = x[i] for i < n-1, h[n-1] = n/2 - sum(x)
    # dh/dx[i] = e_i for i < n-1, h[n-1] has -1 for all i
    g_x = np.zeros(n-1)
    g_x[:n-1] = g_h[:n-1] - g_h[n-1]  # Wait, this isn't quite right
    # Actually: d(obj)/d(x[i]) = sum_j d(obj)/d(h[j]) * dh[j]/dx[i]
    # dh[j]/dx[i] = delta(j,i) for j < n-1, dh[n-1]/dx[i] = -1
    # d(obj)/d(x[i]) = g_h[i] - g_h[n-1] for i < n-1
    g_x = g_h[:n-1] - g_h[n-1]
    return obj, g_x

But wait, the bound 0 ≤ h[n-1] ≤ 1 needs to be enforced. L-BFGS-B can't handle that directly since it's a function of all other variables. I'll need to handle it in the projection step.

For projected gradient: project x onto {0 ≤ x_i ≤ 1 for all i, 0 ≤ n/2 - sum(x) ≤ 1}.

The last constraint is equivalent to (n/2 - 1) ≤ sum(x) ≤ n/2.

So the feasible set for x is: [0,1]^{n-1} ∩ {n/2 - 1 ≤ sum(x) ≤ n/2}.

Projection onto this set:
1. Clip x to [0,1]
2. If sum(x) > n/2: need to reduce. Find the largest x_i's and reduce them.
3. If sum(x) < n/2 - 1: need to increase. Find the smallest x_i's and increase them.

This is similar to the repair function but for n-1 variables with a sum constraint.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _all_ovl(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _grad_ovl(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    s = (n - 1) - k
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - h
            g[:] -= 0.0
            g = (1.0 - h) - np.zeros(n)
            g = 2.0*(1.0 - h) - h - h
            # Actually for s=0: grad[i] = (1-h[i]) + (-h[i]) = 1-2h[i]
            g = 1.0 - 2.0 * h
        else:
            m = n - s
            g[:m] = 1.0 - h[s:]
            g[s:] -= h[:m]
    else:
        sp = -s
        m = n - sp
        g[sp:] = 1.0 - h[:m]
        g[:m] -= h[sp:]
    return g * dx

def _ls_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s_vals = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s_vals - s_vals.max())
    w /= w.sum()
    obj = m + np.log(float(np.sum(np.exp(s_vals)))) / alpha
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-13:
            continue
        g += w[k] * _grad_ovl(h, k, dx)
    return obj, g

def _slsqp(h, alpha, maxiter=2000, ftol=1e-17):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def obj_fn(x):
        return _ls_obj_grad(x, alpha)[0]

    def jac_fn(x):
        return _ls_obj_grad(x, alpha)[1]

    def se(x):
        return x.sum() - n * 0.5

    def se_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': se, 'jac': se_j}]
    try:
        res = minimize(obj_fn, h, jac=jac_fn, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _subgrad(h, top_k, max_iter, base_lr):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    lr = base_lr
    momentum = np.zeros(n)
    mu = 0.7
    for it in range(max_iter):
        c = _all_ovl(h)
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for kk in top_idx:
            g += _grad_ovl(h, kk, dx)
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        momentum = mu * momentum - lr * g
        improved = False
        for step in [1.0, 0.5, 0.25, 0.1]:
            h_new = _repair(h + step * momentum, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            lr *= 0.95
            if lr < 1e-8:
                break
    return h, c5c

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fine_cd(h, n_iter, scales):
    nc = len(h)
    c5c = _eval(h)
    rng_cd = np.random.RandomState(12345)
    for scale in scales:
        for _ in range(n_iter):
            i = rng_cd.randint(nc)
            d = scale * rng_cd.randn()
            h[i] += d
            h_new = _repair(h, nc)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
            else:
                h[i] -= d
                h = _repair(h, nc)
    return h, c5c

def _binary_search(n_bb, h_init, rng, t_rem):
    h_b = _repair(np.array(h_init, dtype=float).copy(), n_bb)
    h_b = (h_b > 0.5).astype(float)
    h_b = _repair(h_b, n_bb)
    c5c = _eval(h_b)
    stuck = 0
    for _ in range(30000):
        if t_rem < 20:
            break
        i = rng.randint(n_bb)
        h_b[i] = 1.0 - h_b[i]
        h2 = _repair(h_b, n_bb)
        c5n = _eval(h2)
        if c5n < c5c - 1e-14:
            c5c = c5n
            h_b = h2
            stuck = 0
        else:
            h_b[i] = 1.0 - h_b[i]
            stuck += 1
            if stuck > 8000:
                break
    return h_b, c5c

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
