def _topk_obj(x, k, use_max=True):
    c = np.correlate(x, 1.0 - x, mode='full')
    if use_max:
        return float(c.max()), None
    idx = np.argsort(c)[-k:]
    return float(c[idx].sum()), idx

def _topk_grad(x, k):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(c)[-k:]
    dx = 2.0 / n
    a = 1.0 - 2.0 * x
    # For each active shift m, add its gradient
    g = np.zeros(n)
    for m in idx:
        s = m - (n - 1)
        if s >= 0:
            i1 = np.arange(n - s)
            g[i1] += dx * (1.0 - x[i1 + s])
            if s > 0:
                i2 = np.arange(s, n)
                g[i2] -= dx * x[i2 - s]
        else:
            sp = -s
            i1 = np.arange(sp, n)
            g[i1] += dx * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= dx * x[i2 + sp]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _soft_grad(x, alpha, idx2=None):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    a = 1.0 - 2.0 * x
    if idx2 is None:
        idx2 = np.arange(n)[None, :] - np.arange(n)[:, None] + (n - 1)
    g = dx * (a @ w[idx2])
    return g, idx2

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_period(h_p, T, n_full, alpha, maxiter):
    m = n_full // T
    idx2 = np.arange(T)[None, :] - np.arange(T)[:, None] + (T - 1)
    # But we need to evaluate on the full tiled array!
    # Actually, for the periodic pattern, the correlate of the tiled array
    # can be computed from the period's correlate
    # For now, let's just use the full array
    
    # Reshape: h_full = tile(h_p, m)
    # The objective is _soft_obj(h_full, alpha)
    # But we only optimize h_p (T variables)
    # Gradient: average over periods
    
    def obj(x):
        h_f = np.tile(x, m)
        return _soft_obj(h_f, alpha)
    
    def grad(x):
        h_f = np.tile(x, m)
        n = len(h_f)
        c = np.correlate(h_f, 1.0 - h_f, mode='full')
        cm = c.max()
        s = np.clip(alpha * (c - cm), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        a_full = 1.0 - 2.0 * h_f
        # Gradient of full array
        idx2_full = np.arange(n)[None, :] - np.arange(n)[:, None] + (n - 1)
        g_full = dx * (a_full @ w[idx2_full])
        # Average over periods
        g_p = g_full.reshape(m, T).mean(axis=0)
        return g_p
    
    bnds = [(0.0, 1.0)] * T
    # Constraint: sum(x) = T/2
    def eq_fun(x):
        return x.sum() - T / 2
    def eq_jac(x):
        return np.ones(T)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h_p, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, T)
    except:
        return _repair(h_p, T)

- Each gradient evaluation: O(n) for correlate + O(n²) for matrix product
