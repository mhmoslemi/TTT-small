import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_smax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad_smax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:

                g[i2] -= wk * h[i2 - sh]

            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj_smax, h, jac=_grad_smax, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:

def _sa_binary(h, n_steps, t_start, t_end, rng, el_fn, budget_limit):
    """Simulated annealing on binary-like solutions."""
    n = len(h)
    # Binarize: threshold and repair
    h_b = (h > 0.5).astype(float)
    # Ensure sum = n/2
    ones_count = h_b.sum()
    target = n // 2
    if ones_count > target:
        idx = np.where(h_b == 1)[0]
        rng.shuffle(idx)
        h_b[idx[:ones_count - target]] = 0
    elif ones_count < target:
        idx = np.where(h_b == 0)[0]
        rng.shuffle(idx)
        h_b[idx[:target - ones_count]] = 1
    h_b = _repair(h_b, n)
    
    best_h = h_b.copy()
    best_c = _eval(h_b)
    cur_c = best_c
    cur_h = h_b.copy()
    
    for step in range(n_steps):
        if el_fn() > budget_limit:
            break
        # Temperature schedule (geometric)
        t = t_start * (t_end / t_start) ** (step / max(n_steps, 1))
        
        # Move: swap a 1 and a 0
        ones_idx = np.where(cur_h > 0.9)[0]
        zeros_idx = np.where(cur_h < 0.1)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        cur_h = _repair(cur_h, n)

            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0

def _nesterov_opt(h, alpha, steps, step0, el_fn, budget_limit):
    n = len(h)
    f = _obj_smax(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _s in range(steps):
        if el_fn() > budget_limit:
            break
        g = _grad_smax(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_smax(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 3)
        else:
            step *= 0.5
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smax(h, alpha)
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_smax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad_smax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj_smax, h, jac=_grad_smax, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _ga_coarse(n_pop, n_gen, n_pts, rng, el_fn, t_limit, parent=None):

    if parent is not None:
        f = len(parent) // n_pts

def _sa_swap(h, n_steps, t0, t1, rng, el_fn, t_limit):

    hb = (h > 0.5).astype(float)
    ones = int(hb.sum()); tgt = n // 2
    if ones > tgt:
        idx = np.where(hb == 1)[0]; rng.shuffle(idx)
        hb[idx[:ones - tgt]] = 0
    elif ones < tgt:
        idx = np.where(hb == 0)[0]; rng.shuffle(idx)
        hb[idx[:tgt - ones]] = 1
    hb = _repair(hb, n)
    best_h, best_c, cur_h, cur_c = hb.copy(), _eval(hb), hb.copy(), _eval(hb)
    for s in range(n_steps):
        if el_fn() > t_limit:
            break
        t = t0 * (t1 / t0) ** (s / max(n_steps, 1))
        oi = np.where(cur_h > 0.9)[0]
        zi = np.where(cur_h < 0.1)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        cur_h[i], cur_h[j] = 0.0, 1.0
        cur_h = _repair(cur_h, n)
        nc = _eval(cur_h)
        if nc < cur_c or rng.rand() < np.exp(-(nc - cur_c) / max(t, 1e-12)):
            cur_c = nc
            if nc < best_c:
                best_c, best_h = nc, cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_c

def _multi_lag(h, n_top, max_rounds, el_fn, t_limit):
    n = len(h); dx = 2.0 / n; c5c = _eval(h)
    for _ in range(max_rounds):
        if el_fn() > t_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n); g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp); g[i2] -= h[i2 + sp]
        g *= dx / n_top
        improved = False
        for step in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h, c5c, improved = h_new, c5n, True
                break
        if not improved:
            break
    return h, c5c

def _nesterov(h, alpha, steps, step0, el_fn, t_limit):
    n = len(h)
    f = _obj_smax(h, alpha); v = np.zeros(n); mu = 0.85; step = step0
    for _ in range(steps):
        if el_fn() > t_limit:
            break
        g = _grad_smax(h, alpha)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        hn = _proj_to_sum(h - step * v, n * 0.5)
        fn = _obj_smax(hn, alpha)
        if fn < f:
            h, f, step = hn, fn, min(step * 1.05, step0 * 4)
        else:
            step *= 0.5
            h = _proj_to_sum(hp - step * v, n * 0.5)
            f = _obj_smax(h, alpha)
    return _repair(h, n)
