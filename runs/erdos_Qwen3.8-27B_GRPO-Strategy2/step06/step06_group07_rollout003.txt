import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _obj_sm(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max())) + 1e-300) / alpha + s.max() / alpha

def _grad_sm(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shift = (n - 1) - k
        wk = w[k]
        if shift >= 0:
            cnt = n - shift
            g[:cnt] += wk * (1.0 - x[shift:shift + cnt])
            if shift > 0:
                g[shift:] -= wk * x[:n - shift]
        else:
            sp = -shift
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp_opt(h0, alpha, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_sm, h0, jac=_grad_sm, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0.copy(), n)

def _nm_obj(x, n):
    x = _repair(x, n)
    return _eval(x)

def _nm_opt(h0, maxiter=5000, seed=0):
    n = len(h0)
    try:
        res = minimize(lambda x: _nm_obj(x, n), h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12,
                                'fatol': 1e-14, 'initial_simplex_size': 0.1})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0.copy(), n)

def _proj_sum(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _pgd_period(h_p, T, n, alphas, max_iters, lr_init, seed):
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    momentum = np.zeros(T)
    beta = 0.8
    total_iters = 0
    for alpha in alphas:
        iters = max_iters // len(alphas)
        for _ in range(iters):

            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            mm = c.max()
            s = np.clip(alpha * (c - mm), -500, 500)
            w = np.exp(s - s.max())
            w /= w.sum()
            dx = 2.0 / n
            g_full = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-14:
                    continue
                shift = (n - 1) - k
                wk = w[k] * dx
                if shift >= 0:
                    g_full[:n - shift] += wk * (1.0 - h_full[shift:n])
                    if shift > 0:
                        g_full[shift:] -= wk * h_full[:n - shift]
                else:
                    sp = -shift
                    g_full[sp:] += wk * (1.0 - h_full[:n - sp])
                    g_full[:n - sp] -= wk * h_full[sp:]
            g_p = g_full.reshape(m, T).mean(axis=0)
            momentum = beta * momentum + (1.0 - beta) * g_p
            h_new = _proj_sum(h_p - lr * momentum, T)

            else:
                lr *= 0.55

                    lr = lr_init

    return best, best_c

def _coord_descent(h, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        ni = min(max(ni, 0.0), 1.0)
        nj = min(max(nj, 0.0), 1.0)
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _single_coord(h, n_iter, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        d = 0.001 * rng.randn()
        old = h[i]
        h[i] = min(max(old + d, 0.0), 1.0)
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h[i] = old
    return h, c5c

    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
