def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _subgrad(h, lag):
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    if lag >= 0:
        ma = j >= lag
        g[ma] += 1.0 - h[ma - lag]
        mb = j <= n - 1 - lag
        g[mb] -= h[mb + lag]
    else:
        p = -lag
        ma = j <= n - 1 - p
        g[ma] += 1.0 - h[ma + p]
        mb = j >= p
        g[mb] -= h[mb - p]
    return g

def _project(x, n):
    tgt = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _sgd(h, n_iter, lr0, seed):
    n = len(h)
    h = h.copy()
    mom = np.zeros(n)
    beta = 0.9
    lr = lr0
    bc = _eval(h)
    bh = h.copy()
    stall = 0
    for it in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        ks = int(c.argmax())
        g = _subgrad(h, ks - (n - 1))
        mom = beta * mom + (1.0 - beta) * g

            stall += 1
            if stall > 200:
                lr *= 0.5
                mom *= 0.5
                stall = 0
    return bh, bc

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _subgrad(h, lag):
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    if lag >= 0:
        ma = j >= lag
        g[ma] += 1.0 - h[ma - lag]
        mb = j <= n - 1 - lag
        g[mb] -= h[mb + lag]
    else:
        p = -lag
        ma = j <= n - 1 - p
        g[ma] += 1.0 - h[ma + p]
        mb = j >= p
        g[mb] -= h[mb - p]
    return g

        else:

        if c2 < bc - 1e-15:
            bc, bh, h = c2, h2.copy(), h2
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-7:
                lr, mom = lr0, np.zeros(n)
    return bh, bc

def _sgd_period(hp, T, m, n_iter, lr0, topk, alpha, seed):
    hp = hp.copy()
    mom = np.zeros(T)
    beta = 0.85
    lr = lr0
    bh = np.tile(hp, m)
    bc = _eval(bh)
    for it in range(n_iter):
        h = np.tile(hp, m)
        n = len(h)
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = c.max()
        idx = np.argsort(c)[-topk:]
        s = np.clip(alpha * (c[idx] - cm), -500, 0)
        w = np.exp(s)
        w /= w.sum()
        gf = np.zeros(n)
        for ii in range(len(idx)):
            gf += w[ii] * _subgrad(h, idx[ii] - (n - 1))
        gp = np.zeros(T)
        for p in range(m):
            gp += gf[p * T:(p + 1) * T]
        gp /= m
        mom = beta * mom + (1.0 - beta) * gp
        hp2 = _project(hp - lr * mom, T)
        c2 = _eval(np.tile(hp2, m))
        if c2 < bc - 1e-15:
            bc, bh, hp = c2, np.tile(hp2, m).copy(), hp2
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-7:
                lr, mom = lr0, np.zeros(T)
    return hp, bc

def _coord(h, n_iter, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    bc = _eval(h)
    scale = 0.02
    for it in range(n_iter):
        if it > 0 and it % 5000 == 0:
            scale *= 0.6
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c2 = _eval(h)
        if c2 < bc - 1e-15:
            bc = c2
        else:
            h[i], h[j] = ni - d, nj + d
    return h, bc

def _resample(h, n_new):
    xo = np.linspace(0.0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h0 = _project(h0, n0)
    bh = h0.copy()
    bc = _eval(bh)

    # Phase 1: Fast subgradient (top-1 and top-3)
    if time.time() - t0 < budget_s * 0.12:
        h, c = _sgd_sub(h0, 50000, 0.03, 1, seed)
        if c < bc:
            bh, bc = h, c
    if time.time() - t0 < budget_s * 0.20:
        h, c = _sgd_sub(bh, 30000, 0.02, 3, seed + 1)
        if c < bc:
            bh, bc = h, c

    # Phase 2: Sparse log-sum-exp on full vector
    if time.time() - t0 < budget_s * 0.40:
        for alpha in [3000, 15000, 60000, 200000]:
            if time.time() - t0 > budget_s * 0.40:
                break
            h, c = _sgd_lse(bh, 15000, 25, alpha, 0.004, seed + 10 + alpha)
            if c < bc:
                bh, bc = h, c
                if _eval(bh) < 0.3809:
                    break

    # Phase 3: Periodic T=31 (10 repeats of 31)
    if n0 % 31 == 0 and time.time() - t0 < budget_s * 0.65:
        m = n0 // 31
        hp = bh.reshape(m, 31).mean(axis=0)
        hp = _project(hp, 31)
        for alpha in [5000, 30000, 100000, 400000]:
            if time.time() - t0 > budget_s * 0.65:
                break
            hp, c = _sgd_period(hp, 31, m, 15000, 0.004, 20, alpha, seed + 100)
            if c < bc:
                bh, bc = np.tile(hp, m).copy(), c

    # Phase 4: Periodic T=62 (5 repeats)
    if n0 % 62 == 0 and time.time() - t0 < budget_s * 0.75:
        m = n0 // 62
        hp = bh.reshape(m, 62).mean(axis=0)
        hp = _project(hp, 62)
        hp, c = _sgd_period(hp, 62, m, 10000, 0.003, 15, 50000, seed + 200)
        if c < bc:
            bh, bc = np.tile(hp, m).copy(), c

    # Phase 5: Full-resolution LSE polish after periodic
    if time.time() - t0 < budget_s * 0.82:
        h, c = _sgd_lse(bh, 10000, 40, 200000, 0.003, seed + 300)
        if c < bc:
            bh, bc = h, c

    # Phase 6: Coordinate descent
    if time.time() - t0 < budget_s * 0.92:
        bh, bc = _coord(bh, 80000, seed + 400)

    # Phase 7: Perturbation restarts
    if time.time() - t0 < budget_s * 0.97:
        rng = np.random.RandomState(seed + 999)
        for trial in range(4):
            if time.time() - t0 > budget_s * 0.97:
                break
            hp = bh + 0.04 * rng.randn(len(bh))
            hp = _project(hp, len(bh))
            hp, c = _sgd_sub(hp, 20000, 0.02, 3, seed + 500 + trial * 10)
            if n0 % 31 == 0 and len(hp) == n0:
                mm = n0 // 31
                hpp = hp.reshape(mm, 31).mean(axis=0)
                hpp = _project(hpp, 31)
                hpp, c2 = _sgd_period(hpp, 31, mm, 8000, 0.003, 20, 100000, seed + 600 + trial * 10)
                if c2 < c:
                    c, hp = c2, np.tile(hpp, mm).copy()
            hp, c = _coord(hp, 20000, seed + 700 + trial)
            if c < bc:
                bh, bc = hp, c

    bh = _project(bh, len(bh))
    return (bh, _eval(bh), len(bh))
