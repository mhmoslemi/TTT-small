def _project_z(z, n):
    """Add constant to z so that sum(sigmoid(z+c)) = n/2."""
    target = n / 2.0
    def f(c):
        return np.sum(_sigmoid(z + c)) - target
    # bisection
    lo, hi = -20.0, 20.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    return z + (lo + hi) / 2

        for i in range(n):
            idx_p = i + shifts
            idx_m = i - shifts
            mp = (idx_p >= 0) & (idx_p < n)
            mm = (idx_m >= 0) & (idx_m < n)
            g_i = np.zeros(2 * n - 1)
            g_i[mp] += (1.0 - h[idx_p[mp]])
            g_i[mm] -= h[idx_m[mm]]
            old = h[i]
            if old <= 1e-9 or old >= 1.0 - 1e-9:
                continue
            deltas = np.linspace(-old, 1.0 - old, n_grid)
            c5_vals = np.max(r[None, :] + deltas[:, None] * g_i[None, :], axis=1)
            best_idx = np.argmin(c5_vals)
            if c5_vals[best_idx] < c5 - 1e-13:
                h[i] = old + deltas[best_idx]
                r += deltas[best_idx] * g_i
                c5 = r.max() * dx
                improved = True
        if not improved:

    return h, c5

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        sh = shifts[k]
        idx_p = np.arange(n) + sh
        idx_m = np.arange(n) - sh
        mp = (idx_p >= 0) & (idx_p < n)
        mm = (idx_m >= 0) & (idx_m < n)
        g[mp] += wk * (1.0 - x[idx_p[mp]])
        g[mm] -= wk * x[idx_m[mm]]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _focused_descent(h, nc, max_rounds, t0=None, t_limit=None):
    c5c = _eval(h)
    dx = 2.0 / nc
    shifts = np.arange(2 * nc - 1) - (nc - 1)
    for r in range(max_rounds):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        ov = np.correlate(h, 1.0 - h, mode='full')
        n_top = min(3, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(nc)
        for kk in top_idx:
            sh = shifts[kk]
            idx_p = np.arange(nc) + sh
            idx_m = np.arange(nc) - sh
            mp = (idx_p >= 0) & (idx_p < nc)
            mm = (idx_m >= 0) & (idx_m < nc)
            g[mp] += (1.0 - h[idx_p[mp]]) * dx
            g[mm] -= h[idx_m[mm]] * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        # Line search
        best_h, best_c5 = h, c5c
        step = 0.05
        for _ in range(20):
            h_new = _repair(h - step * g, nc)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-15:
                best_c5, best_h = c5n, h_new
            else:
                step *= 0.5
        if best_c5 >= c5c - 1e-15:
            break
        h, c5c = best_h, best_c5
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_flip_search(h, n, rng, n_iters=50000, t0=None, t_limit=None):
    h = h.copy()
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    stuck = 0
    for _it in range(n_iters):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
        h2 = _repair(h, n)
        c5n = _eval(h2)
        if c5n < c5c - 1e-14:
            c5c = c5n
            h = h2
            stuck = 0
            if c5c < best_c5 - 1e-14:
                best_c5 = c5c
                best_h = h.copy()
        else:
            h = h2.copy()  # keep the flip (for diversity)
            # Actually, revert
            h[i] = 1.0 - h[i]
            stuck += 1
            if stuck > 10000:
                break
    return best_h, best_c5
