    def _f(x):
        return _smooth_val(x, alpha)
    
    def _g(x):
        return _smooth_grad(x, alpha)
    
    def _cf(x):
        return x.sum() - n * 0.5

        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cg}]
    try:

def _nesterov(h, alpha, steps, lr):
    n = len(h)
    v = np.zeros(n)
    mu = 0.8
    f = _smooth_val(h, alpha)
    for _ in range(steps):
        g = _smooth_grad(h, alpha)
        h_new = _proj_sum(h - lr * (mu * v + (1 - mu) * g), n * 0.5)
