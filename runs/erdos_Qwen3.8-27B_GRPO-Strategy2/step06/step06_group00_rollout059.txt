                best_c5, best_h = _eval(h_sh), h_sh

    # Phase 5: Multi-scale gradient refinement
    for L, st, s0 in [(2, 400, 0.003), (4, 400, 0.002), (10, 400, 0.001)]:
        if time.time() - t0 > 850:
            break
        hc = _mom_descent(best_h.copy(), L, st, s0, 0.9, t0, 850)
        if _eval(hc) < best_c5:
            best_c5, best_h = _eval(hc), hc

    # Phase 6: Very high alpha SLSQP
    if time.time() - t0 < budget_s - 80:
        hc = _slsqp(best_h.copy(), 5000000, 2000, t0, budget_s - 80)
        if _eval(hc) < best_c5:
            best_c5, best_h = _eval(hc), hc

    # Phase 7: Fine local search with log-uniform steps
    rem = budget_s - 3 - (time.time() - t0)
    n_iter = max(500, int(rem / 0.002))
    best_h, best_c5 = _fine_search(best_h, n_iter, rng, t0, budget_s - 2)

    # Phase 8: Second pass local search (different RNG seed)
    if time.time() - t0 < budget_s - 2:
        rng2 = np.random.RandomState(seed + 12345)
        rem2 = budget_s - 2 - (time.time() - t0)
        n_iter2 = max(200, int(rem2 / 0.002))
        best_h, best_c5 = _fine_search(best_h, n_iter2, rng2, t0, budget_s - 1)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
