from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _nesterov(h0, alpha_seq, n_steps, lr, seed, deadline):
    """Nesterov-accelerated projected gradient descent"""
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    y = x.copy()  # Nesterov lookahead
    best_x = x.copy()
    best_c = _eval(x)
    beta = 0.9  # momentum
    for i in range(n_steps):
        if time.time() > deadline: break
        # Nesterov step
        y = x + beta * (x - y if i > 0 else 0)  # Actually y = x + beta*(x - prev_y)
        # Hmm, let me use the standard Nesterov: y = x + beta*(x - y_prev)
        # Then gradient at y
        g = _grad(y, alpha_seq[i % len(alpha_seq)])
        step = lr * g
        x_new = y - step
        x_new = _repair(x_new, n)
        # Nesterov update of y for next iteration
        y_new = x_new + beta * (x_new - y)  # This is the standard form
        x, y = x_new, y_new
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

    # For each i, forward delta: (1-h[i+s]) for s=-(n-1)..n-1

    #       backward[i,k] = h[i-s] (s=n-1-k) if valid

    rows = np.arange(n)

    s_vals = (n-1) - cols  # shape (2n-1,)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _flip_deltas(h):
    n = len(h)
    s_vals = (n - 1) - np.arange(2 * n - 1)
    rows = np.arange(n)
    fwd_idx = rows[:, None] + s_vals[None, :]
    bwd_idx = rows[:, None] - s_vals[None, :]
    m1 = (fwd_idx >= 0) & (fwd_idx < n)
    m2 = (bwd_idx >= 0) & (bwd_idx < n)
    fv = np.where(m1, 1.0 - h[np.clip(fwd_idx, 0, n - 1)], 0.0)
    bv = np.where(m2, h[np.clip(bwd_idx, 0, n - 1)], 0.0)
    return fv - bv

def _local_search(h, n, n_iters, deadline, rng):
    h = h.copy()
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h < 0.5)[0])
    T = 0.01
    for it in range(n_iters):
        if time.time() > deadline: break
        # Compute flip deltas
        deltas = _flip_deltas(h)
        c = np.correlate(h, 1.0 - h, mode='full')
        # Try steepest pair swap
        best_gain = 1e-10
        best_i, best_j = -1, -1
        on_list = list(ones)
        off_list = list(zeros)
        if len(on_list) > 30 and len(off_list) > 30:
            # Subsample for speed
            rng_ss = np.random.RandomState(it)
            on_sample = rng_ss.choice(on_list, 30, replace=False)
            off_sample = rng_ss.choice(off_list, 30, replace=False)
        else:
            on_sample = on_list
            off_sample = off_list
        for i in on_sample:
            # new_c = c + deltas[j] - deltas[i]
            new_c_max = (c[None, :] + deltas[off_sample][:, :] - deltas[i]).max(axis=1)
            j_idx = np.argmin(new_c_max)
            j = off_sample[j_idx]
            gain = c_cur - new_c_max[j_idx] * 2.0 / n
            if gain > best_gain:
                best_gain = gain
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones.discard(best_i); ones.add(best_j)
            zeros.discard(best_j); zeros.add(best_i)
            c_new = _eval(h)
            if c_new <= c_cur + T:
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                # Revert
                h[best_i] = 1.0; h[best_j] = 0.0
                ones.discard(best_j); ones.add(best_i)
                zeros.discard(best_i); zeros.add(best_j)
        T *= 0.999
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _flip_deltas(h):
    n = len(h)
    s_vals = (n - 1) - np.arange(2 * n - 1)
    rows = np.arange(n)
    fwd_idx = rows[:, None] + s_vals[None, :]
    bwd_idx = rows[:, None] - s_vals[None, :]
    m1 = (fwd_idx >= 0) & (fwd_idx < n)
    m2 = (bwd_idx >= 0) & (bwd_idx < n)
    fv = np.where(m1, 1.0 - h[np.clip(fwd_idx, 0, n - 1)], 0.0)
    bv = np.where(m2, h[np.clip(bwd_idx, 0, n - 1)], 0.0)
    return fv - bv

def _local_search(h, n, n_iters, deadline, seed):
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c_cur = _eval(h)
    best_h = h.copy(); best_c = c_cur
    T = 0.008
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        deltas = _flip_deltas(h)
        c = np.correlate(h, 1.0 - h, mode='full')
        # Sample candidates for speed
        nos = len(ones_idx); nzs = len(zeros_idx)
        k_sample = min(40, nos, nzs)
        on_s = rng.choice(ones_idx, k_sample, replace=False)
        zf_s = rng.choice(zeros_idx, k_sample, replace=False)
        best_gain = 1e-12
        best_i = best_j = -1
        for ii in range(k_sample):
            i = on_s[ii]
            new_max = (c[None, :] + deltas[zf_s] - deltas[i]).max(axis=1)
            jj = np.argmin(new_max)
            j = zf_s[jj]
            gain = c_cur - new_max[jj] * 2.0 / n
            if gain > best_gain:
                best_gain = gain; best_i = i; best_j = j
        if best_i >= 0:
            h[best_i] = 0.0; h[best_j] = 1.0
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
            c_new = _eval(h)
            if c_new <= c_cur + T:
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[best_i] = 1.0; h[best_j] = 0.0
                ones_idx = np.where(h > 0.5)[0]
                zeros_idx = np.where(h < 0.5)[0]
        T *= 0.9995
    return best_h, best_c

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dxv = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dxv
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _adam_nest(h0, a0, a1, n_steps, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    y = x.copy()
    m = np.zeros(n); v = np.zeros(n)
    t = 0; b1, b2, eps = 0.9, 0.999, 1e-8
    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_steps - 1, 1)
        alpha = a0 * (a1 / a0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if t % 800 == 0:
            x += 0.003 * rng.randn(n)
            x = _repair(x, n); y = x.copy()
        y = x + (y - x) * (-0.9)  # Nesterov: y = 2x - y_prev
        y = np.clip(y, 0.0, 1.0)
        g = _grad(y, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t); vhat = v / (1 - b2 ** t)
        x_new = y - lr * mhat / (np.sqrt(vhat) + eps)
        x_new = _repair(x_new, n)
        x = x_new
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _fe(x):
        return x.sum() - n * 0.5

    def _fj(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _fe, 'jac': _fj}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
