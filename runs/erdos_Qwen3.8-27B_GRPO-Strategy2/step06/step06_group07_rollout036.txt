from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _circ_corr(h_p):
    T = len(h_p)
    return np.real(ifft(fft(h_p) * np.conj(fft(1.0 - h_p))))

def _pgd_periodic(h_p, alpha_schedule, max_iters, lr=0.01, beta=0.9, seed=0):
    """Optimize periodic pattern using FFT-based gradient."""
    T = len(h_p)
    rng = np.random.RandomState(seed)
    h_p = np.array(h_p, dtype=float).copy()
    h_p = np.clip(h_p, 0, 1)
    h_p *= (T / 2.0) / h_p.sum()
    h_p = np.clip(h_p, 0, 1)
    best = h_p.copy()
    best_c = float('inf')
    mom = np.zeros(T)
    
    for alpha in alpha_schedule:
        n_it = max_iters // len(alpha_schedule)
        for _ in range(n_it):
            c = _circ_corr(h_p)
            m = c.max()
            s = np.clip(alpha * (c - m), -500, 500)
            s -= s.max()
            w = np.exp(s)
            w /= w.sum()
            # Gradient via FFT
            u_flip = np.empty(T)
            u_flip[0] = w[0]
            for j in range(1, T):
                u_flip[j] = w[T - j]
            g = np.real(ifft(fft(u_flip) * fft(1.0 - h_p))) - np.real(ifft(fft(w) * fft(h_p)))
            # Projected gradient step with projection onto sum constraint
            mom = beta * mom + (1 - beta) * g
            h_new = h_p - lr * mom
            # Project: clip to [0,1] and adjust sum
            h_new = np.clip(h_new, 0, 1)
            target = T / 2.0
            h_new += (target - h_new.sum()) / T
            h_new = np.clip(h_new, 0, 1)
            # Check
            c_new = _circ_corr(h_new)
            score = c_new.max()
            if score < best_c:
                best_c = score
                best = h_new.copy()
            h_p = h_new
    
    return best, best_c

So: c_full at lag s = (m-1) * circ_corr[s] + ∑_{r=0}^{T-1-s} h_p[r] * (1-h_p[(r+s) mod T])

The second term is the circular correlation at lag s minus the last s terms:
∑_{r=0}^{T-1-s} h_p[r] * (1-h_p[(r+s) mod T]) = circ_corr[s] - ∑_{r=T-s}^{T-1} h_p[r] * (1-h_p[(r+s) mod T])

So: c_full[s] = (m-1) * circ_corr[s] + circ_corr[s] - ∑_{r=T-s}^{T-1} h_p[r] * (1-h_p[(r+s) mod T])
= m * circ_corr[s] - ∑_{r=T-s}^{T-1} h_p[r] * (1-h_p[(r+s) mod T])

The correction term ∑_{r=T-s}^{T-1} h_p[r] * (1-h_p[(r+s) mod T]) has at most s ≤ T-1 terms, each ≤ 1, so it's at most T-1.

For m=10, T=31: circ_corr[s] is O(15) (roughly), so m*circ_corr is O(150), and the correction is O(30). The correction is about 20% of the main term. Not negligible!

import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_full(h, alpha):
    n = len(h)
    v = 1.0 - h
    c = np.correlate(h, v, 'full')
    mm = c.max()
    s_vec = np.clip(alpha * (c - mm), -500, 500)
    s_vec -= s_vec.max()
    w = np.exp(s_vec)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                g[:n-sh] += wk * v[sh:]
                if sh > 0:
                    g[sh:] -= wk * h[:n-sh]
        else:
            p = -sh
            if p < n:
                g[:n-p] += wk * v[p:]
                g[:n-p] -= wk * h[p:]
    return g

def _proj_sum_bnds(h, target_sum, n):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.array(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _pgd_periodic(h_p, m, alpha_schedule, max_iters, lr_init=0.005, beta=0.85, seed=0):
    """Optimize periodic pattern h_p (length T) for full signal of length T*m."""
    T = len(h_p)
    n = T * m
    h_p = np.array(h_p, dtype=float).copy()
    h_p = _proj_sum_bnds(h_p, T/2.0, T)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    mom = np.zeros(T)
    lr = lr_init
    rng = np.random.RandomState(seed)
    
    for alpha in alpha_schedule:
        n_it = max_iters // len(alpha_schedule)
        for it in range(n_it):
            h_full = np.tile(h_p, m)
            g_full = _grad_full(h_full, alpha)
            # Average gradient over periods
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p*T:(p+1)*T]
            g_p /= m
            
            mom = beta * mom + (1.0 - beta) * g_p
            h_new = h_p - lr * mom
            h_new = _proj_sum_bnds(h_new, T/2.0, T)
            
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                lr = min(lr * 1.02, 0.05)
            else:
                lr *= 0.7
                if lr < 1e-6:
                    lr = lr_init
                    mom = np.zeros(T)
    return best, best_c

def _slsqp_periodic(h_p, alpha, maxiter, ftol):

    def obj(x):

    # Actually, let me use the soft-max objective for smoothness

        s = np.clip(alpha*(c-m), -500, 500)

    def sum_eq(x):

    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    try:
