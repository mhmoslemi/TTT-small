def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        if diff > 0:
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    return h

def _cvals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return float(np.max(_cvals(h)) * 2.0 / len(h))

def _grad_k(h, k, dx):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _lsexp_obj(x, alpha, penalty):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    base = m + np.log(float(np.sum(np.exp(s)))) / alpha
    pen = penalty * (x.sum() - len(x) * 0.5) ** 2
    return base * (2.0 / len(x)) + pen

def _lsexp_grad(x, alpha, penalty):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-11:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            if sh < n:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    g += penalty * 2.0 * (x.sum() - n * 0.5) / n
    return g

def _slsqp_wrap(h0, alpha, penalty, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lsexp_obj, h0, jac=_lsexp_grad,
                       args=(alpha, penalty), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': ftol, 'gtol': 1e-12, 'maxfun': maxiter * 3})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0.copy(), n)

def _resample(h, n_new):
    xo = np.linspace(0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h), n_new)

def _fourier_init(n, K, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-0.25, 0.25)
        b = rng.uniform(-0.25, 0.25)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)

def _coord_descent(h, n, rng, scale, n_iter):
    c5c = _c5(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _c5(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c
