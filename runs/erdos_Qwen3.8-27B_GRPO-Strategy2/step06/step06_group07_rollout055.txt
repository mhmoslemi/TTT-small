    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter // len(scales)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Nesterov PGD full resolution (multi-alpha)
    if elapsed() < budget_s * 0.20:
        for alpha_set in [[3000, 10000, 40000, 150000],
                          [8000, 30000, 100000, 400000, 1000000]]:
            if elapsed() > budget_s * 0.22:
                break
            h_new, c5 = _nesterov_pgd(best_h.copy(), 6000, alpha_set,
                                       lr_init=0.006, seed=seed)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 2: Periodic T=31 PGD (exploit n=310=10*31)
    if n0 % 31 == 0 and elapsed() < budget_s * 0.40:
        m = n0 // 31
        h_p = best_h.reshape(m, 31).mean(axis=0)
        h_p = _proj(h_p, 31)
        for alpha_set in [[5000, 20000, 80000, 300000, 800000],
                          [20000, 80000, 300000, 1000000, 3000000]]:
            if elapsed() > budget_s * 0.42:
                break
            h_p, c5 = _nesterov_pgd(h_p, 8000, alpha_set, lr_init=0.008,
                                     seed=seed + 1)
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()
                best_n = n0

    # Phase 3: SLSQP on T=31 periodic pattern
    if n0 % 31 == 0 and elapsed() < budget_s * 0.60:
        m = n0 // 31
        h_p = best_h[:31].copy() if best_n == n0 else _repair(_resample(best_h, n0)[:31], 31)
        for alpha in [50000, 200000, 800000, 2000000]:
            if elapsed() > budget_s * 0.62:
                break
            h_p = _slsqp_solve(h_p, alpha, 4000, 1e-20)
            c5 = _eval(np.tile(h_p, m))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h_p, m).copy()
                best_n = n0

    # Phase 4: SLSQP at full resolution
    if elapsed() < budget_s * 0.78:
        for alpha in [20000, 100000, 500000, 2000000]:
            if elapsed() > budget_s * 0.80:
                break
            h_new = _slsqp_solve(best_h, alpha, 4000, 1e-20)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase 5: Random perturbation + re-optimize
    rng = np.random.RandomState(seed + 50)
    if elapsed() < budget_s * 0.88:
        for trial in range(3):
            if elapsed() > budget_s * 0.90:
                break
            for eps in [0.03, 0.08, 0.15]:
                h_pert = _repair(best_h + eps * rng.randn(best_n), best_n)
                h_n, c5 = _nesterov_pgd(h_pert, 3000, [50000, 300000],
                                         lr_init=0.004, seed=seed + 100 + trial * 10)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_n.copy()

    # Phase 6: Coordinate descent polish (many iterations, fine scales)
    if elapsed() < budget_s - 15:
        scales = [0.01, 0.003, 0.001, 0.0003, 0.0001]
        h9, c5 = _coord_descent(best_h.copy(), 25000, scales, seed=seed + 200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h9.copy()

    # Phase 7: Try slightly different n
    if elapsed() < budget_s - 25:
        for dn in [31, -31, 62, 100]:
            if elapsed() > budget_s - 12:
                break
            n_try = best_n + dn
            if n_try < 50 or n_try > 800:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _slsqp_solve(h_r, 200000, 2000, 1e-18)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                best_n = n_try
            h_r2, c5b = _coord_descent(h_r.copy(), 5000, [0.001, 0.0003],
                                        seed=seed + 300)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = h_r2.copy()
                best_n = best_n

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
