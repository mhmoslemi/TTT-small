import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Iteratively adjust
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _soft_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _adam(h0, n, alphas, steps, lr, beta1, beta2, eps, seed):
    rng = np.random.RandomState(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h0.copy()
    best_c = _eval(h0)
    h = h0.copy()
    for alpha in alphas:
        for _ in range(steps):
            t += 1
            g = _soft_grad(h, alpha)
            # Add constraint gradient (push toward sum = n/2)
            d = h.sum() - n / 2.0
            g += 0.01 * d * np.ones(n) / n
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = _repair(h_new, n)
            c_new = _eval(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def _slsqp_periodic(h_p, T, n_full, alpha, maxiter):
    """SLSQP on T-var periodic pattern, evaluating at full resolution."""
    m = n_full // T
    dx = 2.0 / n_full
    def obj(x):
        h_full = np.tile(x, m)
        return _soft_obj(h_full, alpha)
    def grad(x):
        h_full = np.tile(x, m)
        g_full = _soft_grad(h_full, alpha)
        g_p = np.zeros(T)
        for i in range(m):
            g_p += g_full[i*T:(i+1)*T]
        return g_p / m
    def sum_fun(x):
        return x.sum() - T/2.0
    def sum_jac(x):
        return np.ones(T)
    
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h_p, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, T)
    except:
        return _repair(h_p, T)
