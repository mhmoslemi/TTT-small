        else:

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):

    for _ in range(45):

        if np.clip(h - mid, 0, 1).sum() > target:

        else:

            stall = 0
            if c5 < best_c5:
                best_c5, best_h = c5, h_cur.copy()
        else:
            found = False
            for f in [0.5, 0.25, 0.1, 0.05]:
                h2 = _project(h_cur - step * f * v, target)
                c2 = _eval(h2)
                if c2 < c5 - 1e-16:
                    h_cur, c5 = h2, c2
                    step *= f
                    stall = 0
                    found = True
                    if c2 < best_c5:
                        best_c5, best_h = c2, h_cur.copy()
                    break
            if not found:
                step *= 0.4
                stall += 1
                if stall > 300 or step < 5e-10:
                    break
    return best_h, best_c5

def _smax_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(x, alpha, n, M=30):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    top = np.argsort(w)[::-1][:M]
    wt = w[top].copy()
    wt /= wt.sum()
    g = np.zeros(n)
    for i, idx in enumerate(top):
        g += wt[i] * _grad_lag(x, idx - (n - 1), n)
    return g * (2.0 / n)

def _slsqp(h, alpha, maxit, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return x.sum() - n * 0.5
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    f0 = _smax_obj(h, alpha, n)
    try:
        res = minimize(_smax_obj, h, jac=lambda x: _smax_grad(x, alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-14, 'disp': False})
        if res.fun < f0 - 1e-12:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _perturb(h, rng, n):
    h2 = h.copy()
    bl = rng.randint(n // 8, n // 3)
    st = rng.randint(0, max(1, n - bl))
    mode = rng.randint(3)
    if mode == 0:
        h2[st:st + bl] = 1.0 - h2[st:st + bl]
    elif mode == 1:
        h2[st:st + bl] = h2[st:st + bl][::-1]
    else:
        h2 = np.roll(h2, rng.randint(1, n // 4))
    h2 += rng.randn(n) * 0.02
    return _repair(h2, n)

def _coord_descent(h, t_end, rng, n_iter):
    n = len(h)
    h_c = h.copy()
    c5c = _eval(h_c)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 2e-4 * rng.randn()
        ni, nj = h_c[i] + d, h_c[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_c[i], h_c[j] = ni, nj
        c5n = _eval(h_c)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h_c[i], h_c[j] = ni - d, nj + d
    return h_c, c5c

def _multi_lag_descent(h, t_end, n_rounds):
    n = len(h)
    h_c = h.copy()
    c5c = _eval(h_c)
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_c, 1.0 - h_c, mode='full')
        top = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        for idx in top:
            g += _grad_lag(h_c, idx - (n - 1), n)
        g *= (2.0 / n) / len(top)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.008, 0.004, 0.001]:
            h2 = _project(h_c - step * g, n * 0.5)
            c2 = _eval(h2)
            if c2 < c5c - 1e-15:
                h_c, c5c = h2, c2
                improved = True
                break
        if not improved:
            break
    return h_c, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_end = t0 + budget_s

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h0.copy(), _eval(h0)
    best_n = n0

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Fast top-lag projected gradient
    if rem() > 70:
        h1, c1 = _proj_grad(best_h, t0 + rem() - 70, 3, 80000)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Multi-lag descent
    if rem() > 55:
        h2, c2 = _multi_lag_descent(best_h, t0 + rem() - 55, 500)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: SLSQP with increasing alpha
    if rem() > 60:
        for alpha in [200, 1000, 5000, 20000, 80000]:
            if rem() < 55:
                break
            hc = _slsqp(best_h, alpha, 2000, t0 + rem() - 55)
            c5c = _eval(hc)
            if c5c < best_c5:
                best_c5, best_h = c5c, hc

    # Phase 4: Perturb + re-optimize (escape local optima)
    if rem() > 120:
        for _ in range(12):
            if rem() < 100:
                break
            hp = _perturb(best_h, rng, best_n)
            hc, cc = _proj_grad(hp, t0 + rem() - 95, 3, 15000)
            if cc < best_c5:
                best_c5, best_h = cc, hc
            hc2, cc2 = _multi_lag_descent(hc, t0 + rem() - 92, 200)
            if cc2 < best_c5:
                best_c5, best_h = cc2, hc2

    # Phase 5: SLSQP fine-tune
    if rem() > 45:
        hc = _slsqp(best_h, 200000, 5000, t0 + rem() - 45)
        c5c = _eval(hc)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc

    # Phase 6: More projected gradient with higher M
    if rem() > 40:
        h61, c61 = _proj_grad(best_h, t0 + rem() - 35, 5, 50000)
        if c61 < best_c5:
            best_c5, best_h = c61, h61

    # Phase 7: Tiny step projected gradient
    if rem() > 30:
        h71, c71 = _proj_grad(best_h, t0 + rem() - 25, 10, 80000)
        if c71 < best_c5:
            best_c5, best_h = c71, h71

    # Phase 8: Coordinate descent polish
    if rem() > 20:
        n_iter = min(int((rem() - 15) * 800), 80000)
        h8, c8 = _coord_descent(best_h, t0 + rem() - 15, rng, n_iter)
        if c8 < best_c5:
            best_c5, best_h = c8, h8

    # Phase 9: Final multi-lag + SLSQP
    if rem() > 25:
        h9, c9 = _multi_lag_descent(best_h, t0 + rem() - 22, 300)
        if c9 < best_c5:
            best_c5, best_h = c9, h9
    if rem() > 15:
        hc = _slsqp(best_h, 500000, 3000, t0 + rem() - 14)
        c5c = _eval(hc)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc

    # Final repair and eval
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
