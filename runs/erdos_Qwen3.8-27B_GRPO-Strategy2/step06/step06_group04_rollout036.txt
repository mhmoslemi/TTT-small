            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _grad_topk(h, k_top=5):
    """Gradient focusing only on top-k correlation values"""
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    # Find top k indices
    top_idx = np.argsort(c)[-k_top:]
    w = np.zeros(2*n-1)
    w[top_idx] = 1.0 / k_top
    g = np.zeros(n)
    for k in top_idx:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _obj_topk(h, k_top=5):
    c = _corr(h)
    return float(np.sum(np.sort(c)[-k_top:]) / k_top) * 2.0 / len(h)

def _adam_refine(h0, n_iters, lr, alpha_start, alpha_end, k_start, k_end, seed, deadline):
    """Adam with adaptive alpha and top-k focus"""
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    
    for i in range(n_iters):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        k_top = max(1, int(k_start + (k_end - k_start) * frac))
        g = _grad_lse(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
        # Perturbation every 2000 steps
        if t % 2000 == 0:
            x = x + 0.01 * rng.randn(n)
            x = _repair(x, n)
    
    return best_x, best_c

def _slsqp_refine(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    
    def _obj2(x, a):
        c = _corr(x)
        m = c.max()
        s = np.clip(a * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / a
    
    def _grad2(x, a):
        return _grad_lse(x, a)
    
    def _se(x):
        return x.sum() - n * 0.5
    
    def _sj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_obj2, h, jac=_grad2, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_local_search(h, n, deadline, max_rounds=50):
    """Try each bit flip, keep if improves (hill climbing)"""
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    improved = True
    round_num = 0
    while improved and round_num < max_rounds:
        if time.time() > deadline:
            break
        improved = False
        round_num += 1
        for i in range(n):
            if time.time() > deadline:
                break
            old = x[i]
            x[i] = 1.0 - old
            x = _repair(x, n)
            c_new = _eval(x)
            if c_new <= c_cur - 1e-15:
                c_cur = c_new
                improved = True
                if c_new < best_c:
                    best_c = c_new
                    best_x = x.copy()
            else:
                x[i] = old
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _pop_search(h_init, n, pop_size, n_rounds, deadline, seed):
    """Simple population-based search: maintain population, mutate, select"""
    rng = np.random.RandomState(seed)
    pop = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_p = h_init + 0.05 * rng.randn(n)
        pop.append(_repair(h_p, n))
    best_x = h_init.copy()
    best_c = _eval(h_init)
    
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        new_pop = []
        # Keep best
        idx_best = min(range(pop_size), key=lambda i: _eval(pop[i]))
        new_pop.append(pop[idx_best].copy())
        # Mutate some
        for i in range(pop_size - 1):
            src = pop[rng.randint(pop_size)]
            if rng.random() < 0.3:
                # Large perturbation
                h_p = src + 0.1 * rng.randn(n)
            else:
                # Small perturbation
                h_p = src + 0.01 * rng.randn(n)
            new_pop.append(_repair(h_p, n))
        
        # Evaluate and keep best pop_size
        scored = [(_eval(h), i) for i, h in enumerate(new_pop)]
        scored.sort()
        pop = [new_pop[i] for _, i in scored[:pop_size]]
        
        for c_s, i in scored:
            if c_s < best_c:
                best_c = c_s
                best_x = new_pop[i].copy()
    
    return best_x, best_c

def _paired_cd(h, n, rng, deadline, scale_init, max_iters=200000):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    scale = scale_init
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if scale < 1e-8:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        x[i] += d
        x[j] -= d
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < c_cur - 1e-16:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
            scale = max(scale * 0.9995, 1e-9)
        else:
            x[i] -= d
            x[j] += d
            x = _repair(x, n)
            c_cur = _eval(x)
            scale *= 0.95
    return best_x, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)

        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])

    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

        res = minimize(_obj_lse, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
