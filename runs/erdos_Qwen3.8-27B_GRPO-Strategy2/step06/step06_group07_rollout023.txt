    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h0, alpha, maxiter=5000, ftol=1e-20):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h0, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0.copy(), n)

def _proj_sum(h, T):
    h = np.array(h, dtype=float).copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        d = T * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / T
    return np.clip(h, 0.0, 1.0)

def _nesterov_31(h_p, T, n, n_iter, lr_init, seed):
    m = n // T
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    x = h_p.astype(float).copy()
    y = x.copy()
    lr = lr_init
    for t in range(n_iter):
        h_full = np.tile(y, m)
        gf = _lse_grad(h_full, 30000)
        g_p = np.zeros(T)
        for p in range(m):
            g_p += gf[p * T:(p + 1) * T]
        g_p /= m
        y_new = y - lr * g_p
        y_new = _proj_sum(y_new, T)
        momentum = t / (t + 2.0)
        x_new = y_new + momentum * (y_new - x)
        x_new = _proj_sum(x_new, T)
        c_new = _eval(np.tile(x_new, m))
        if c_new < best_c:
            best_c = c_new
            best = x_new.copy()
        x = x_new
        y = y_new
        lr = lr_init / (1.0 + 0.005 * t)
    return best, best_c

def _sa_31(h_p, T, n, n_iter, T_init, T_end, seed):
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    cur = h_p.astype(float).copy()
    cur_c = best_c
    for it in range(n_iter):
        T_sa = T_init * (T_end / T_init) ** (float(it) / max(float(n_iter), 1.0))
        i, j = rng.randint(T), rng.randint(T)
        if i == j:
            continue
        d = 0.02 * rng.randn()
        ci, cj = cur[i] + d, cur[j] - d
        if ci < 0 or ci > 1 or cj < 0 or cj > 1:
            continue
        cur[i], cur[j] = ci, cj
        cn = _eval(np.tile(cur, m))
        if cn < cur_c or (cn - cur_c) < 0 or rng.rand() < np.exp(-(cn - cur_c) / max(T_sa, 1e-15)):
            cur_c = cn
            if cn < best_c:
                best_c = cn
                best = cur.copy()
        else:
            cur[i], cur[j] = ci - d, cj + d
    return best, best_c

def _coord_descent(h, n, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    c5c = _eval(h)
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: T=31 SLSQP - main optimization with many restarts
    if n0 % 31 == 0 and elapsed() < budget_s * 0.40:
        m31 = n0 // 31
        h31 = h0.reshape(m31, 31).mean(axis=0)
        h31 = _proj_sum(h31, 31)
        # Progressive alpha refinement
        for a in [2000, 10000, 50000, 200000, 800000, 3000000]:
            if elapsed() > budget_s * 0.25:
                break
            h31 = _slsqp_opt(h31, a, 4000)
            c5 = _eval(np.tile(h31, m31))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h31, m31).copy()
        # Multiple random restarts
        for trial in range(20):
            if elapsed() > budget_s * 0.40:
                break
            h31r = _proj_sum(h31 + 0.06 * rng.randn(31), 31)
            for a in [50000, 500000]:
                h31r = _slsqp_opt(h31r, a, 2500)
                c5 = _eval(np.tile(h31r, m31))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = np.tile(h31r, m31).copy()
                    h31 = h31r.copy()

    # Phase 2: T=31 Nesterov accelerated gradient
    if n0 % 31 == 0 and elapsed() < budget_s * 0.55:
        m31 = n0 // 31
        h31 = best_h[:31].copy() if best_n == n0 else _proj_sum(best_h[:31], 31)
        for lr0 in [0.004, 0.002, 0.008]:
            if elapsed() > budget_s * 0.50:
                break
            h31, c5 = _nesterov_31(h31, 31, n0, 4000, lr0, seed + lr0 * 1000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h31, m31).copy()

    # Phase 3: T=31 Simulated Annealing
    if n0 % 31 == 0 and elapsed() < budget_s * 0.70:
        m31 = n0 // 31
        h31 = best_h[:31].copy() if best_n == n0 else _proj_sum(best_h[:31], 31)
        for ti in [0.002, 0.005, 0.015]:
            if elapsed() > budget_s * 0.65:
                break
            h31, c5 = _sa_31(h31, 31, n0, 40000, ti, 1e-9, seed + 100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h31, m31).copy()
        # Final SLSQP after SA
        h31 = best_h[:31].copy() if best_n == n0 else _proj_sum(best_h[:31], 31)
        for a in [300000, 2000000]:
            if elapsed() > budget_s * 0.69:
                break
            h31 = _slsqp_opt(h31, a, 3000)
            c5 = _eval(np.tile(h31, m31))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h31, m31).copy()

    # Phase 4: T=10 alternative periodicity
    if n0 % 10 == 0 and elapsed() < budget_s * 0.78:
        m10 = n0 // 10
        h10 = best_h[:n0].reshape(m10, 10).mean(axis=0) if best_n == n0 else _proj_sum(best_h[:10], 10)
        h10 = _proj_sum(h10, 10)
        for a in [10000, 50000, 200000]:
            if elapsed() > budget_s * 0.76:
                break
            h10 = _slsqp_opt(h10, a, 3000)
            c5 = _eval(np.tile(h10, m10))
            if c5 < best_c5:
                best_c5 = c5
                best_h = np.tile(h10, m10).copy()

    # Phase 5: Full-resolution SLSQP
    if elapsed() < budget_s * 0.90:
        for a in [50000, 300000, 1500000]:
            if elapsed() > budget_s * 0.88:
                break
            h_new = _slsqp_opt(best_h.copy(), a, 3000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Coordinate descent polish
    if elapsed() < budget_s - 20:
        for sc in [0.003, 0.001, 0.0004]:
            if elapsed() > budget_s - 12:
                break
            h_cd, c5 = _coord_descent(best_h.copy(), best_n, 5000, sc, seed + 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd.copy()

    # Phase 7: Try small n perturbations
    if elapsed() < budget_s - 15:
        for dn in [2, -2, 5, -5, 10]:
            if elapsed() > budget_s - 10:
                break
            n_try = best_n + dn
            if n_try < 50 or n_try > 500 or n_try % 2 != 0:
                continue
            xo = np.linspace(0, 2, best_n, endpoint=False)
            xn = np.linspace(0, 2, n_try, endpoint=False)
            h_r = _repair(np.interp(xn, xo, best_h), n_try)
            h_r = _slsqp_opt(h_r, 200000, 2000)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                best_n = n_try

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
