    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:

        room = 1.0 - h
        order = np.argsort(-room)
        idx = order[0]
        a = min(diff, room[idx])
        h[idx] += a

def _project(h, n):
    target = n / 2.0
    # Find λ such that sum(clip(h + λ, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2, 0, 1)
    # Final adjustment for floating point
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        # Distribute to positions not at boundary
        if diff > 0:
            free = np.where(h_new < 1 - 1e-6)[0]
        else:
            free = np.where(h_new > 1e-6)[0]
        if len(free) > 0:
            h_new[free] += diff / len(free)
            h_new = np.clip(h_new, 0, 1)
    return h_new

    target = n / 2.0

    if s < 1e-12:
        return np.full(n, 0.5)

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a

                cur = c2
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                break
            else:
                h[oi] = 1.0
                h[oj] = 0.0
            tried += 1
            if tried >= min(80, len(ones)):
                break
        if not improved:
            break
    return h, cur

def _slsqp_fine(h_init, alpha, maxiter, t_end, topk):
    n = len(h_init)
    def obj_fn(x):
        _, f = _grad_lse(x, alpha, topk)
        return f
    def grad_fn(x):
        g, _ = _grad_lse(x, alpha, topk)
        return g
    bnds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return x.sum() - n * 0.5
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_fn, h_init, jac=grad_fn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h_init.copy()

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _anneal_binary(h, n, t_start, t_end_sim, t_end, rng, n_rounds):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    temp = t_start
    decay = 0.97
    ones_cnt = n // 2
    for it in range(n_rounds):
        if time.time() > t_end:
            break
        temp *= decay
        oi = rng.randint(n)
        oj = rng.randint(n)
        if h[oi] == h[oj]:
            continue
        h[oi] = 0.0
        h[oj] = 1.0
        c_new = _eval(h)
        delta = c_new - cur
        if delta < 0 or (temp > 1e-10 and rng.rand() < np.exp(-delta / temp)):
            cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def _rem():
        return budget_s - (time.time() - t0)

    def _check_end(pad=20.0):
        return time.time() > (t0 + budget_s - pad)

    # Phase 1: PG optimization at increasing alphas
    for alpha in [1000, 5000, 25000, 100000]:
        if _check_end(120):
            break
        tr = min(_rem() - 100, 150)
        if tr < 30:
            break
        h_new, c5 = _pg_optimize(best_h, alpha, 3000, 0.02, t0 + tr, topk=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: SLSQP fine-tune at current resolution
    if _rem() > 120:
        tr = min(_rem() - 100, 100)
        h_s = _slsqp_fine(best_h, 50000, 3000, t0 + tr, topk=8)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s.copy()

    # Phase 3: Try different resolutions
    if _rem() > 200:
        for n_try in [250, 350, 450, 600]:
            if _check_end(150):
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            for alpha in [10000, 100000]:
                if _check_end(130):
                    break
                hr, c5 = _pg_optimize(hr, alpha, 2000, 0.015, t0 + _rem() - 110, topk=8)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hr.copy()
                best_n = n_try

    # Phase 4: Binary local search
    if _rem() > 100:
        for nb in [best_n, min(best_n, 400)]:
            if _check_end(80):
                break
            hb = _resample(best_h, nb)
            hb, c5 = _binary_flip(hb, nb, 800, t0 + _rem() - 60, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hb.copy()
                best_n = nb

    # Phase 5: Annealing on binary
    if _rem() > 120:
        hb, c5 = _anneal_binary(best_h, best_n, 0.005, 0, t0 + _rem() - 100, rng, 50000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hb.copy()

    # Phase 6: SLSQP polish at high alpha
    if _rem() > 100:
        tr = min(_rem() - 80, 90)
        h_p = _slsqp_fine(best_h, 500000, 5000, t0 + tr, topk=5)
        c5 = _eval(h_p)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_p.copy()

    # Phase 7: Final PG with very high alpha
    if _rem() > 60:
        tr = min(_rem() - 40, 50)
        h_f, c5 = _pg_optimize(best_h, 1000000, 3000, 0.005, t0 + tr, topk=4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f.copy()

    # Phase 8: Final binary flip
    if _rem() > 40:
        hb, c5 = _binary_flip(best_h, best_n, 500, t0 + _rem() - 25, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hb.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
