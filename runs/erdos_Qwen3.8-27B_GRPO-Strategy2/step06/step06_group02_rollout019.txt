def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _fk(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_fk(h)) * 2.0 / n)

def _compute_grad(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-16:
            continue
        tau = m - (n - 1)
        wm = w[m] * dx
        if tau >= 0:
            g[:n - tau] += wm * (1.0 - h[tau:])
            if tau > 0:
                g[tau:] -= wm * h[:n - tau]
        else:
            t2 = -tau
            g[t2:] += wm * (1.0 - h[:n - t2])
            g[:n - t2] -= wm * h[t2:]
    return g

def _grad_desc(h, t_end, anneal_per=2000, max_iter=80000):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    best_h, best_c = h.copy(), cur
    step_sizes = [0.2, 0.1, 0.05, 0.025, 0.0125, 0.006, 0.003, 0.0015]
    it = 0
    while it < max_iter and time.time() < t_end:
        c = _fk(h)
        temp = 0.02 + 0.03 * max(0.0, 1.0 - it / 20000.0)
        maxc = c.max()
        w = np.exp(-(maxc - c) / temp)
        w_total = w.sum()
        if w_total < 1e-30:
            w = np.zeros(2 * n - 1)
            top = np.argsort(c)[-20:]
            w[top] = 1.0
            w /= w.sum()
        else:
            w /= w_total
        g = _compute_grad(h, w, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        moved = False
        for step in step_sizes:
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-14:
                h = ht; cur = ct; moved = True
                break
        if not moved:
            if it % anneal_per == anneal_per - 1:
                rng = np.random.RandomState(1000 + it // anneal_per)
                noise = 0.02 * rng.randn(n)
                h = _repair(h + noise, n)
                cur = _eval(h)
            it += 1
        else:
            it += 1
            if cur < best_c:
                best_c = cur; best_h = h.copy()
    return best_h, best_c

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    def obj(x):
        c = _fk(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return float(m + np.log(np.sum(np.exp(s))) / alpha)
    def grad(x):
        c = _fk(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        return _compute_grad(x, w, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_project(h):
    n = len(h)
    target = n / 2
    vals, counts = np.unique(np.round(h * 1000) / 1000, return_counts=True)
    order = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[order[:int(target)]] = 1.0
    return out

def _local_binary_swap(h, t_end, max_rounds=200):
    n = len(h)
    h = h.copy()
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in ones.copy():
            for j in zeros.copy():
                h[i] = 0.0; h[j] = 1.0
                c2 = _eval(h)
                if c2 < cur - 1e-13:
                    cur = c2; improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h <= 0.5)[0]
                    break
                else:
                    h[i] = 1.0; h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur
