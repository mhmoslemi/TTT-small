def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 5e-10:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-13: break
    return h

        else: hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 5e-10:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_sm(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (h.sum() - n * 0.5) ** 2

def _grad_sm(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * lam * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, alpha, lam, mi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_sm, h, jac=_grad_sm, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': -1})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _mlag(h, ntop, nrounds, el, bc):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _r in range(nrounds):
        if bc():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[-ntop:]
        g = np.zeros(n)
        for k in top:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:n - sp] -= h[sp:]
        g *= dx / ntop
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        improved = False
        for st in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5]:
            h2 = _proj(h - st * g / gn, n * 0.5)
            c2 = _eval(h2)
            if c2 < c5 - 1e-15:
                h = h2; c5 = c2; improved = True; break
        if not improved:
            break
    return h, c5

def _ls2(h, niter, rng, el, bc, scale):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc5 = c5c
    for _ in range(niter):
        if bc():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0 <= a <= 1 and 0 <= b <= 1:
            h[i], h[j] = a, b
            c2 = _eval(h)
            if c2 < c5c - 1e-17:
                c5c = c2
                if c2 < bc5:
                    bc5 = c2; bh = h.copy()
            else:
                h[i], h[j] = a - d, b + d
    return bh, bc5

def _ls3(h, niter, rng, el, bc, scale):

    bh = h.copy(); bc5 = c5c
    for _ in range(niter):

        a, b, c = h[i] + d1, h[j] + d2, h[k] - d1 - d2

                    bc5 = c2; bh = h.copy()
            else:

    return bh, bc5
