def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_weighted(x, alpha, topk):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if topk > 0 and topk < 2 * n - 1:
        idx = np.argpartition(c, -topk)[-topk:]
        vals = c[idx] - m
        s = np.clip(alpha * vals, -40, 0)
        e = np.exp(s)
        wsel = e / e.sum()
        w = np.zeros(2 * n - 1)
        for ii, ki in enumerate(idx):
            if wsel[ii] > 1e-14:
                w[ki] = wsel[ii]
    else:
        s = np.clip(alpha * (c - m), -40, 0)
        e = np.exp(s)
        w = e / e.sum()
    obj = m + np.log(max(e.sum(), 1e-30)) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k

        if sh >= 0:
