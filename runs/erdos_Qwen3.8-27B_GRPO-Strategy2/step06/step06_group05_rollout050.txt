    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

        if c5s < best_c5 - 1e-14: best_h, best_c5 = hs.copy(), c5s
        hs = _slsqp(hs, 2000000, 1500, 1e-19)
        c5s = _eval(hs)
        if c5s < best_c5 - 1e-14: best_h, best_c5 = hs.copy(), c5s

    # Phase 5: Binary anneal + smooth cycles (120s)
    d5 = t0 + 1000
    for cyc in range(3):
        if time.time() > d5: break
        hb = (best_h > 0.5).astype(float)
        hb = _repair(hb, len(best_h))
        if _eval(hb) > best_c5 + 0.05: break
        hb, _ = _anneal(hb, 0.15, 0.0005, 4000, rng, d5)
        hb = _slsqp(hb, 1000000, 2000, 1e-19)
        c5a = _eval(hb)
        if c5a < best_c5 - 1e-14: best_h, best_c5 = hb.copy(), c5a
        hc = _repair(1.0 - hb, len(hb))
        hc = _slsqp(hc, 1000000, 1500, 1e-19)
        c5c = _eval(hc)
        if c5c < best_c5 - 1e-14: best_h, best_c5 = hc.copy(), c5c

    # Phase 6: Coordinate descent polish (50s)
    d6 = t0 + budget_s - 3
    for sc in [0.004, 0.001, 0.0002]:
        if time.time() > d6: break
        h6, c56 = _coord_desc(best_h, 5000, sc, rng, d6)
        if c56 < best_c5 - 1e-16: best_h, best_c5 = h6.copy(), c56

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
