import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _slsqp_solve(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    def _smax(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _sgrad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        nz = np.where(w > 1e-12)[0]
        for k in nz:
            wk = w[k] * dx
            sh = k - (n - 1)
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
            else:
                p = -sh
                g[p:] += wk * (1.0 - h[:n - p])
                g[:n - p] -= wk * h[p:]
        return g
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _topk_grad(h, n, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    for k in idx:
        sh = k - (n - 1)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            p = -sh
            g[p:] += (1.0 - h[:n - p])
            g[:n - p] -= h[p:]
    return g * (2.0 / n) / len(idx)

def _subgrad_run(h, n_iter, step0, top_k, time_limit):
    n = len(h)
    tgt = 0.5 * n
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stuck = 0
    t_start = time.time()
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        g = _topk_grad(best, n, top_k)
        v = mu * v + g
        gv = np.max(np.abs(v))
        if gv < 1e-14:
            break
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj(best - step * f * v / gv, tgt)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new
                best_c = c_new
                step = min(step * 1.15, 0.8)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.55
            stuck += 1
            if stuck > 80:
                best = _repair(best + 0.02 * np.random.randn(n), n)
                best_c = _eval(best)
                step = step0
                stuck = 0
                v = np.zeros(n)
    return best, best_c

def _fista_smooth(h0, alpha, n_iter, base_step, time_limit):
    n = len(h0)
    tgt = 0.5 * n
    x = h0.copy()
    y = h0.copy()
    tk = 1.0
    best = x.copy()
    best_c = _eval(best)
    step = base_step
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        c = np.correlate(y, 1.0 - y, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 30)
        e = np.exp(s)
        w = e / e.sum()
        val = m + np.log(e.sum()) / alpha
        dx = 2.0 / n
        g = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for k in nz:
            wk = w[k] * dx
            sh = k - (n - 1)
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - y[sh:])
                if sh > 0:
                    g[sh:] -= wk * y[:n - sh]
            else:
                p = -sh
                g[p:] += wk * (1.0 - y[:n - p])
                g[:n - p] -= wk * y[p:]
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        beta = (tk - 1.0) / tk_new
        y_new = x + beta * (x - y)
        accepted = False
        for ls in [1.0, 0.5, 0.25, 0.1, 0.05]:
            z = _proj(y_new - step * ls * g, tgt)
            fn = _eval(z)
            if fn < best_c:
                best = z.copy()
                best_c = fn
                x = z
                accepted = True
                break
        if not accepted:
            x = _proj(y_new - step * 0.02 * g, tgt)
        y = x
        tk = tk_new
    return best, best_c

def _pairwise_refine(h, n_tries, rng, time_limit):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    t0 = time.time()
    for _ in range(n_tries):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        ni = np.clip(ni, 0, 1)
        nj = np.clip(nj, 0, 1)
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = h[i] - (ni - h[i]) - d + d, h[j] + (nj - h[j]) + d - d
            h[i] = np.clip(h[i] - d, 0, 1)
            h[j] = np.clip(h[j] + d, 0, 1)
    return best, best_c

def _make_structured(n, rng):
    h = np.full(n, 0.5)
    pattern = rng.randint(4)
    if pattern == 0:
        for i in range(n):
            h[i] = 1.0 if (i % 4 < 2) else 0.0
    elif pattern == 1:
        blocks = rng.randint(3, 8)
        for b in range(blocks):
            s = (b * n // blocks)
            e = ((b + 1) * n // blocks)
            h[s:e] = 1.0 if b % 2 == 0 else 0.0
    elif pattern == 2:
        freq = rng.randint(5, 20)
        x = np.arange(n)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        h = np.clip(h, 0, 1)
    else:
        for i in range(n):
            h[i] = 1.0 if rng.rand() > 0.5 else 0.0
    return _repair(h, n)
