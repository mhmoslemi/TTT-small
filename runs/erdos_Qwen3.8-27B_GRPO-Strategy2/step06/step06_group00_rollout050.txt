import numpy as np
import time

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -200, 200)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad(h, alpha, n_active=20):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -200, 200)
    w = np.exp(s - s.max())
    w /= w.sum()
    # Only process top-weighted lags
    top = np.argsort(w)[-n_active:]
    g = np.zeros(n)
    for k in top:
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            t = -sh
            g[t:] += wk * (1.0 - h[:n - t])
            g[:n - t] -= wk * h[t:]
    return g

def _nesterov(h, alpha, max_steps, step0, t_end):
    n = len(h)
    f = _obj(h, alpha)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    best_h, best_f = h.copy(), f
    stall = 0
    for _ in range(max_steps):
        if time.time() > t_end:
            break
        g = _grad(h, alpha)
        h_old = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _obj(h_new, alpha)
        if f_new < f - 1e-15:
            h, f = h_new, f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
                stall = 0
            else:
                stall += 1
            step = min(step * 1.015, step0 * 5.0)
        else:
            step *= 0.5
            h = _proj(h_old - step * v, n * 0.5)
            f = _obj(h, alpha)
            if step < 1e-12:
                break
    return best_h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _coord_polish(h, n_rounds, t_end, rng):
    n = len(h)
    c5 = _eval(h)
    offsets = [1, 3, 7, 13, 23, 37, 53]
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            j = (i + offsets[_r % len(offsets)]) % n
            for d in [1e-4, -1e-4, 5e-5, -5e-5, 2e-4, -2e-4]:
                h[i] += d; h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    c5n = _eval(h)
                    if c5n < c5 - 1e-14:
                        c5 = c5n; improved = True
                    else:
                        h[i] -= d; h[j] += d
                else:
                    h[i] -= d; h[j] += d
        if not improved:
            break
    return h, c5

    return (best_h, best_c5, n0)
