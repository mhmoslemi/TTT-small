import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lse(h, alpha, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    wm = s - s.max()
    w = np.exp(wm); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] -= wk * h[sh:]
            g[sh:] += wk * (1.0 - h[:n-sh])
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n-sp]
    return g

def _pgd(h, n, alpha, max_iter, t_end):
    cur = _eval(h)
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = _grad_lse(h, alpha, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        accepted = False
        step = lr
        for _ in range(35):
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-13:
                h = ht; cur = ct; accepted = True
                break
            step *= 0.55
        if accepted:
            stall = 0
            lr = min(step * 1.15, 0.08)
        else:
            stall += 1
            lr *= 0.8
            if stall > 80:
                break
    return h, cur

def _binary_search(h, n, t_end, n_rounds=30, n_tries_per=30000, seed=0):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    h = _repair(h, n)
    cur = _eval(h)
    rng = np.random.RandomState(seed)
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(n_tries_per):
            if time.time() > t_end:
                return h, cur
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0; h[j] = 1.0
            ct = _eval(h)
            if ct < cur - 1e-14:
                cur = ct; improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i] = 1.0; h[j] = 0.0
        if not improved:
            break
    return h, cur

def _slsqp_polish(h, alpha, n, maxiter=2000):
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def jac(x):
        return _grad_lse(x, alpha, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)
