import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Water-filling projection
    if s > target:
        # Need to decrease sum
        order = np.argsort(-h)  # largest first
        for i in order:
            d = min(h[i], s - target)
            h[i] -= d
            s -= d
            if s <= target + 1e-12:
                break
    else:
        # Need to increase sum
        order = np.argsort(h)  # smallest first
        for i in order:
            d = min(1.0 - h[i], target - s)
            h[i] += d
            s += d
            if s >= target - 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        g[sp:] += dx * (1.0 - h[:n-sp])

    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        order = np.argsort(-h)

            d = min(h[i], s - target)
            h[i] -= d
            s -= d
