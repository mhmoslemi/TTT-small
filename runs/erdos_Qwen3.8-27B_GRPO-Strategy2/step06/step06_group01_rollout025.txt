```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0 and sh < n:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    elif sh < 0:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _ls_weights(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w, m

def _top_grad(h, n_top, dx):
    n = len(h)
    ov = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(ov)[-n_top:]
    g = np.zeros(n)
    for kk in top:
        g += _grad_lag(h, kk, dx)
    return g, ov, top

def _lbfgsb(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    st = n * 0.5
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        pen = (x.sum() - st) ** 2 * 0.05
        return m + np.log(np.sum(np.exp(s))) / alpha + pen
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        w, _ = _ls_weights(c, alpha)
        dx_ = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12: continue
            g += w[k] * _grad_lag(x, k, dx_)
        g += 0.1 * (x.sum() - st)
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14, 'maxfun': maxiter * 3})
        return _repair(np.clip(res.x, 0, 1), n)
    except:
        return _repair(h, n)

def _fista(h, n_top, n_iter, lr0):
    n = len(h)
    dx = 2.0 / n
    st = n * 0.5
    x = h.copy()
    z = h.copy()
    c5c = _eval(x)
    for t in range(1, n_iter + 1):
        g = np.zeros(n)
        for kk in np.argsort(np.correlate(z, 1.0 - z, mode='full'))[-n_top:]:
            g += _grad_lag(z, kk, dx)
        lr = lr0 / (1.0 + t * 0.01)
        x_new = z - lr * g
        x_new = np.clip(x_new, 0, 1)
        # Projection onto sum constraint
        diff = st - x_new.sum()
        if abs(diff) > 1e-12:
            x_new += diff / n
            x_new = np.clip(x_new, 0, 1)
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = x_new + ((t - 1) / theta) * (x_new - x)
        x = x_new
```

```python
def _top_grad(h, n_top, dx):
    n = len(h)
    ov = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(ov)[-n_top:]
    g = np.zeros(n)
    for kk in top:
        g += _grad_lag(h, kk, dx)
    return g, ov

def _ls_obj(x, alpha, st):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 0.05 * (x.sum() - st) ** 2

def _ls_grad(x, alpha, st):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx_ = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12: continue
        g += w[k] * _grad_lag(x, k, dx_)
    g += 0.1 * (x.sum() - st)
    return g

def _lbfgsb(h, alpha, maxiter, ftol):
    n = len(h); st = n * 0.5
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha, st),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return _repair(h, n)

def _sum_fun(x, st): return x.sum() - st
def _sum_jac(x, st): return np.ones(len(x))

def _sls_obj(x, alpha, st):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sls_grad(x, alpha, st):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx_ = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12: continue
        g += w[k] * _grad_lag(x, k, dx_)
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h); st = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (st,)}]
    try:
        res = minimize(_sls_obj, h, jac=_sls_grad, args=(alpha, st),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _gd_descent(h, n_top, n_iter, lr0, decay):
    n = len(h); dx = 2.0 / n
    c5c = _eval(h)
    for t in range(1, n_iter + 1):
        g, ov = _top_grad(h, n_top, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        g /= gn
        lr = lr0 / (1.0 + t * decay)
        h_new = _repair(h - lr * g, n)
        c5n = _eval(h_new)
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            if shrink < 1.0:
                h_new = _repair(h - lr * shrink * g, n)
                c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                c5c = c5n; h = h_new; improved = True; break
        if not improved and t > 100:
            continue
    return h, c5c

def _coord3(h, nc, rng, n_iter, scale):
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = (i + 1) % nc
        k = (i + 2) % nc
        d = scale * rng.randn(3)
        ni, nj, nk = h[i] + d[0], h[j] + d[1], h[k] + d[2]
        if ni < 0 or ni > 1 or nj < 0 or nj > 1 or nk < 0 or nk > 1: continue
        h[i], h[j], h[k] = ni, nj, nk
        h2 = _repair(h, nc)
        c5n = _eval(h2)
        if c5n < c5c - 1e-16:
            h = h2; c5c = c5n
        else:
            h[i], h[j], h[k] = ni - d[0], nj - d[1], nk - d[2]
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_anneal(n, rng, n_iter, T0, T1):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    c5c = _eval(h)
    for t in range(n_iter):
        T = T0 * (T1 / T0) ** (t / max(1, n_iter - 1))
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
        c5n = _eval(h)
        d = c5n - c5c
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-10)):
            c5c = c5n
        else:
            h[i] = 1.0 - h[i]
    return h, c5c

def _elapsed(t0, budget_s):
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS-B initial polish
    if _elapsed(t0, budget_s) > 200:
        h1 = _lbfgsb(best_h, 15000, 3000, 1e-18)
        c5 = _eval(h1)
        if c5 < best_c5: best_c5, best_h = c5, h1.copy()

    # Phase 2: GD on top lags
    if _elapsed(t0, budget_s) > 250:
        h2, c5 = _gd_descent(best_h, 5, 8000, 0.03, 0.004)
        if c5 < best_c5: best_c5, best_h = c5, h2.copy()

    # Phase 3: SLSQP multiple alphas
    for alpha in [30000, 100000, 300000]:
        if _elapsed(t0, budget_s) < 150: break
        h3 = _slsqp(best_h, alpha, 3000, 1e-18)
        c5 = _eval(h3)
        if c5 < best_c5: best_c5, best_h = c5, h3.copy()

    # Phase 4: L-BFGS-B higher alpha
    if _elapsed(t0, budget_s) > 200:
        h4 = _lbfgsb(best_h, 80000, 3000, 1e-18)
        c5 = _eval(h4)
        if c5 < best_c5: best_c5, best_h = c5, h4.copy()

    # Phase 5: GD finer
    if _elapsed(t0, budget_s) > 200:
        h5, c5 = _gd_descent(best_h, 3, 12000, 0.01, 0.002)
        if c5 < best_c5: best_c5, best_h = c5, h5.copy()

    # Phase 6: Coord descent
    if _elapsed(t0, budget_s) > 100:
        for scale in [0.01, 0.003, 0.001, 0.0003, 0.0001]:
            if _elapsed(t0, budget_s) < 30: break
            h6, c5 = _coord3(best_h, best_n, rng, 10000, scale)
            if c5 < best_c5: best_c5, best_h = c5, h6.copy()

    # Phase 7: Multi-resolution
    if _elapsed(t0, budget_s) > 350:
        for n_try in [500, 560, 640, 700, 780, 840]:
            if _elapsed(t0, budget_s) < 80: break
            hr = _repair(_resample(best_h, n_try), n_try)
            hr = _lbfgsb(hr, 20000, 2000, 1e-17)
            hr, c5 = _gd_descent(hr, 5, 10000, 0.02, 0.003)
            hr = _slsqp(hr, 80000, 2000, 1e-18)
            c5 = _eval(hr)
            if c5 < best_c5: best_c5, best_h, best_n = c5, hr.copy(), n_try

    # Phase 8: Perturbation + re-optimize
    if _elapsed(t0, budget_s) > 200:
        for _ in range(6):
            if _elapsed(t0, budget_s) < 50: break
            noise = rng.randn(best_n) * 0.008
            hp = _repair(best_h + noise, best_n)
            hp = _lbfgsb(hp, 20000, 1500, 1e-17)
            hp, c5 = _gd_descent(hp, 5, 15000, 0.015, 0.003)
            hp = _slsqp(hp, 100000, 1500, 1e-18)
            c5 = _eval(hp)
            if c5 < best_c5: best_c5, best_h = c5, hp.copy()

    # Phase 9: Binary annealing at low res then upsample
    if _elapsed(t0, budget_s) > 250:
        for n_b in [60, 80, 100]:
            if _elapsed(t0, budget_s) < 60: break
            hb, _ = _binary_anneal(n_b, rng, 50000, 0.02, 0.001)
            hr = _repair(_resample(hb, best_n), best_n)
            hr = _lbfgsb(hr, 20000, 2000, 1e-17)
            hr, c5 = _gd_descent(hr, 5, 10000, 0.02, 0.003)
            hr = _slsqp(hr, 80000, 2000, 1e-18)
            c5 = _eval(hr)
            if c5 < best_c5: best_c5, best_h = c5, hr.copy()

    # Phase 10: Final SLSQP + micro coord
    if _elapsed(t0, budget_s) > 40:
        hf = _slsqp(best_h, 500000, 3000, 1e-19)
        c5 = _eval(hf)
        if c5 < best_c5: best_c5, best_h = c5, hf.copy()
    if _elapsed(t0, budget_s) > 15:
        h9, c5 = _coord3(best_h, best_n, rng, 20000, 0.0001)
        if c5 < best_c5: best_c5, best_h = c5, h9.copy()
    if _elapsed(t0, budget_s) > 10:
        h10, c5 = _gd_descent(best_h, 1, 5000, 0.003, 0.001)
        if c5 < best_c5: best_c5, best_h = c5, h10.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```
