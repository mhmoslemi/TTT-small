```python
def _topk_grad(h, topk):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -topk)[-topk:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0: g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[sp:])
            g[:n - sp] -= h[sp:]
    return g * dx / topk
```

This computes the sum of gradients of the top-k correlations, divided by topk (so it's the average gradient of the top-k), times dx. This is the gradient of (1/topk) * sum of top-k c_k values * dx. Good.

Actually wait - the topk sum is not exactly the max, but it's a good approximation when k is small relative to the number of lags. The gradient of the top-k sum is the sum of the gradients of the individual top-k lags (assuming the set of top-k doesn't change, which is true almost everywhere). This is a valid subgradient of the max function when k=1, and a valid gradient of the sum-of-top-k function for general k.

The sum-of-top-k is a concave function, so minimizing it is not convex. But as a surrogate for the max, it's reasonable.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _obj_fn(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad_fn(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[sp:])
            g[:n - sp] -= wk * x[sp:]
    return g

def _ceq_fn(x, n):
    return x.sum() - n * 0.5

def _ceq_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

        return _repair(h.copy(), n)
```
