    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k] * dx
        if wk < 1e-15:
            continue
        shift = (n - 1) - k  # correlate index to actual shift
        if shift >= 0:
            # c_k = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])

                d2 = -h[i2 - shift]  # d(c_k)/d(h[i]) for i in [shift, n-1]

            i1 = slice(s, n)
            d1 = (1.0 - h[i1 - s])
            g[i1] += wk * d1
            i2 = slice(0, n - s)
            d2 = -h[i2 + s]
            g[i2] += wk * d2

    return (best_h, best_c, best_n)
