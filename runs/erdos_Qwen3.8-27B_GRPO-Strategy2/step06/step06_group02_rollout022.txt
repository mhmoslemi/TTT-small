import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr_full(h)) * 2.0 / n)

def _grad_h(h, shifts, weights):
    n = len(h)
    g = np.zeros(n)
    for s_idx, w in zip(shifts, weights):
        if abs(w) < 1e-15:
            continue
        shift = s_idx - (n - 1)
        if shift >= 0:
            g[:n - shift] += w * (1.0 - h[shift:])
            g[shift:] -= w * h[:n - shift]
        else:
            sp = -shift
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _grad_descent(h, n_iter, top_m, t_end, lr_init=0.02, beta=20.0):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    mu = 0.85
    best_h, best_c = h.copy(), cur
    stall = 0
    top_m_cur = top_m
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        top_m_cur = min(top_m, 2 * n - 1)
        top_idx = np.argpartition(c, -top_m_cur)[-top_m_cur:]
        c_top = c[top_idx]
        c_max, c_min_top = c.max(), c_top.min()
        if c_max - c_min_top > 1e-12:
            weights = np.exp(beta * (c_top - c_min_top) / (c_max - c_min_top))
        else:
            weights = np.ones(top_m_cur)
        weights /= weights.sum()
        g = _grad_h(h, top_idx, weights) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_prev = h.copy()
        h = _project(h + mu * vel - lr * g, n)
        vel = h - h_prev
        c_new = _eval(h)
        if c_new < cur - 1e-15:
            cur = c_new
            lr = min(lr * 1.05, 0.06)
            stall = 0
        else:
            h = h_prev
            lr *= 0.6
            stall += 1
            if stall > 80:
                break
        if cur < best_c:
            best_c, best_h = cur, h.copy()
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    xo = np.linspace(0, 2.0, n, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _binary_search(h, n, n_iter, t_end):

    h[h <= 0.5] = 0.0

            h[cand_i] = 0.0
            h[cand_j] = 1.0
            c2 = _eval(h)
            if c2 < cur - 1e-14:

        else:

def _lse_obj(x, n, alpha):
    c = _corr_full(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        return _lse_obj(x, n, alpha)
    def _cons_fun(x):
        return x.sum() - n * 0.5
    def _cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_f, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _pair_cd(h, n, n_iter, t_end, scales):
    rng = np.random.RandomState(999)
    cur = _eval(h)
    dx = 2.0 / n
    for scale in scales:
        for _ in range(n_iter // len(scales)):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, cur
