            if i == j:
                continue
            d = 8e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c_cur - 1e-15:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        elif p < 0.75:
            b = max(3, min(rng.randint(3, n // 8 + 3), n // 4))
            i = rng.randint(0, n - b)
            j = rng.randint(0, n - b)
            if abs(i - j) < b + 2:
                continue
            d = 6e-5 * rng.randn()
            h[i:i + b] += d
            h[j:j + b] -= d
            if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
                h[i:i + b] -= d
                h[j:j + b] += d
                continue
            cn = _eval(h)
            if cn < c_cur - 1e-15:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = h.copy()
            else:
                h[i:i + b] -= d
                h[j:j + b] += d
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_w = np.argmax(corr)
            sh = (n - 1) - k_w
            if sh >= 0:
                for _ in range(3):
                    i = rng.randint(0, n - sh)
                    d = 5e-6 * rng.uniform(0.5, 1.5)
                    if h[i] > d and h[i + sh] < 1 - d:
                        h[i] -= d
                        h[i + sh] += d
                        cn = _eval(h)
                        if cn < c_cur:
                            c_cur = cn
                            if cn < best_c:
                                best_c = cn
                                best = h.copy()
                        else:
                            h[i] += d
                            h[i + sh] -= d
            else:
                sp = -sh
                for _ in range(3):
                    i = rng.randint(0, n - sp)
                    d = 5e-6 * rng.uniform(0.5, 1.5)
                    if h[i] < 1 - d and h[i + sp] > d:
                        h[i] += d
                        h[i + sp] -= d
                        cn = _eval(h)
                        if cn < c_cur:
                            c_cur = cn
                            if cn < best_c:
                                best_c = cn
                                best = h.copy()
                        else:
                            h[i] -= d
                            h[i + sp] += d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def _el():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [2000, 10000, 50000, 200000]:
        if _el() > 180:
            break
        hc = _slsqp(best_h, alpha, 2000)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 2: Momentum gradient descent (primary optimizer)
    if _el() < budget_s - 350:
        for alpha in [5000, 20000, 100000, 500000]:
            if _el() > budget_s - 350:
                break
            rem = budget_s - 350 - _el()
            n_it = min(3000, int(rem / 0.06))
            if n_it < 200:
                break
            best_h, best_c = _momentum(best_h, alpha, n_it, 0.008, 0.8)

    # Phase 3: Up-resolve and re-optimize
    if _el() < budget_s - 200:
        n1 = min(best_n * 2, 1000)
        if n1 > best_n:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n1, endpoint=False)
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = _repair(h_hi, n1)
            for alpha in [10000, 100000]:
                if _el() > budget_s - 120:
                    break
                hc = _slsqp(h_hi, alpha, 1200)
                c = _eval(hc)
                if c < best_c:
                    best_c = c
                    best_h = hc
                    best_n = n1
            if _el() < budget_s - 100:
                n_it = min(1500, int((budget_s - 100 - _el()) / 0.07))
                if n_it > 100:
                    best_h, best_c = _momentum(best_h, 50000, n_it, 0.004, 0.8)
                    best_n = len(best_h)

    # Phase 4: Final SLSQP polish at high alpha
    if _el() < budget_s - 60:
        hc = _slsqp(best_h, 1000000, 2500)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc
            best_n = len(hc)

    # Phase 5: One more momentum pass
    if _el() < budget_s - 40:
        n_it = min(1000, int((budget_s - 40 - _el()) / 0.08))
        if n_it > 50:
            best_h, best_c = _momentum(best_h, 200000, n_it, 0.003, 0.7)
            best_n = len(best_h)

    # Phase 6: Local search with block + lag-targeted moves
    if _el() < budget_s - 5:
        avail = budget_s - 5 - _el()
        n_tr = min(int(avail / 0.005), 100000)
        best_h, best_c = _local_search(best_h, n_tr, rng)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
