        n_iter = min(60000, int((budget_s - 150 - _el()) / 0.004))
        h, c = _mom_grad_descent(h, n_iter, 0.008, 10, 0.88)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Try lower resolution then upsample
    if _el() < budget_s - 200:
        h_lo = _down_up(best_h, 2)
        n_lo = len(h_lo)
        c_lo = _eval(h_lo)
        if c_lo < 0.382:
            for alpha in [20000, 100000]:
                if _el() > budget_s - 150:
                    break
                h_lo = _slsqp(h_lo, alpha, 800)
            h_hi = np.repeat(h_lo, 2)
            h_hi = _repair(h_hi, len(h_hi))
            if _el() < budget_s - 100:
                h_hi = _slsqp(h_hi, 50000, 600)
            c_hi = _eval(h_hi)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi.copy()

    # Phase 6: Final SLSQP polish
    if _el() < budget_s - 80:
        h_new = _slsqp(best_h, 1000000, 1500)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 7: Fine pairwise search
    if _el() < budget_s - 15:
        n_tries = min(200000, int((budget_s - 15 - _el()) / 0.0008))
        h, c = _fine_search(best_h, n_tries, rng)
        if c < best_c:
            best_c = c
            best_h = h

    # Final
    best_h = _repair(best_h, len(best_h))
    best_c = _eval(best_h)
    return (best_h, best_c, len(best_h))
