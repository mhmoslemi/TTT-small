```python
def _coord_1d_solve(a, b, x_lo, x_hi):
    """minimize max_k (a[k] + b[k]*x) over x in [x_lo, x_hi]"""
    # f(x) = max(a + b*x) is convex and piecewise linear
    # Use golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    x1 = x_hi - gr * (x_hi - x_lo)
    x2 = x_lo + gr * (x_hi - x_lo)
    f1 = np.max(a + b * x1)
    f2 = np.max(a + b * x2)
    for _ in range(40):
        if f1 < f2:
            x_hi = x2
            x2 = x1
            f2 = f1
            x1 = x_hi - gr * (x_hi - x_lo)
            f1 = np.max(a + b * x1)
        else:
            x_lo = x1
            x1 = x2
            f1 = f2
            x2 = x_lo + gr * (x_hi - x_lo)
            f2 = np.max(a + b * x2)
    return (x_lo + x_hi) / 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lag_grad(h, k, n, dx):
    """Compute ∂g_k/∂h for the k-th lag in full correlation"""
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

            b = grads[:, i] - grads[:, j]  # shape (T,)
            a = c[top_idx]  # shape (T,)
            
            # Bounds on δ

            d_hi = min(1.0 - h[i], h[j])
```

```python
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov(h, alpha, steps, step0, t0, t_end):
    n = len(h)
    f = _obj_smooth(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_smooth(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smooth(h, alpha)
    return h

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    n_pairs_per_sweep = n * 2
    no_improve = 0
    while time.time() < min(end_t, t_end - 8):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        sweep_improved = False
        for _ in range(n_pairs_per_sweep):
            if time.time() > min(end_t, t_end - 8):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 10))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(22):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 40:
            h += 1e-5 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 4e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov smooth optimization
    for alpha in [5000, 30000, 150000, 600000]:
        t_end = min(t0 + 25, t0 + budget_s * 0.1)
        h = _nesterov(h, alpha, 400, 0.004, t0, t_end)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coordinate descent (pair-swap, exact 1D subproblem)
    for top_T in [3, 5, 8]:
        if time.time() > t0 + budget_s * 0.35:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.1, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Nesterov with higher precision
    if time.time() < t0 + budget_s * 0.6:
        h = best_h.copy()
        for alpha in [400000, 1500000, 5000000]:
            t_end = min(time.time() + 30, t0 + budget_s * 0.58)
            h = _nesterov(h, alpha, 300, 0.002, t0, t_end)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 4: More coordinate descent
    if time.time() < t0 + budget_s * 0.82:
        for top_T in [4, 6, 3, 5]:
            if time.time() > t0 + budget_s * 0.8:
                break
            h = best_h.copy()
            h, c5v = _coord_descent(h, n, budget_s * 0.05, top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 5: Final ultra-fine local search
    if time.time() < t0 + budget_s - 5:
        remaining = (t0 + budget_s) - time.time() - 3
        n_tries = min(int(remaining / 0.004), 40000)
        if n_tries > 100:
            h = best_h.copy()
            best_h, best_c5 = _local_search(h, n_tries, rng, t0 + budget_s - 2)

    # Phase 6: Final coordinate polish
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, max(1, budget_s - (time.time() - t0) - 3), 5, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
