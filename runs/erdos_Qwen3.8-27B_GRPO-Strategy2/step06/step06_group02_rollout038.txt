OK now let me think about the actual optimization strategy that will be different and better.

**Key insight**: The gradient of c[k] w.r.t. h[m] is 1-2h[k-m] for m in S(k). This means:
- If h[k-m] < 0.5, the gradient is positive → increasing h[m] increases c[k]
- If h[k-m] > 0.5, the gradient is negative → decreasing h[m] increases c[k]

So to DECREASE c[k], we want to:
- Decrease h[m] where h[k-m] < 0.5 (make h[m] smaller)
- Increase h[m] where h[k-m] > 0.5 (make h[m] larger)

This is essentially: make h[m] and h[k-m] more "aligned" (both large or both small) to reduce h[m]*(1-h[k-m]).

**New strategy: Direct "alignment" optimization**

For the worst shift k*, the move is:
- For each m in S(k*): if h[m] ≠ h[k-m], push them together.
- Specifically: h[m] -= step * sign(h[m] - h[k-m]) * 0.5 (and correspondingly adjust h[k-m])

But we need to maintain the constraint ∑h = n/2. So we can't just push pairs together independently.

Actually, the gradient already accounts for this via the Lagrange multiplier in SLSQP. But for a manual approach, we can:
1. Compute g = _grad_shift(h, n, k*)
2. Do h_new = h - step * g, then project: h_new = _repair(h_new, n)

The repair handles the constraint.

        w = np.exp(s - s.max()); w /= w.sum()
        gg = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            gg += w[k] * _gs(x, n, k)
        return gg
    try:
        r = minimize(f, h, jac=g, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})
        return _repair(r.x, n)
    except Exception:
        return h.copy()

def _subgrad_mom(h, n_iter, beta, t_end):
    n = len(h); cur = _eval(h); mom = np.zeros(n); step = 0.05
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        c = _eval_vec(h); k = int(c.argmax())
        g = _gs(h, n, k)
        mom = beta * mom + (1.0 - beta) * g
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h2 = _repair(h - s * mom, n)
            c2 = _eval(h2)
            if c2 < cur - 1e-15:
                h, cur = h2, c2; improved = True; break
        if not improved:
            step *= 0.6
            if step < 1e-7:
                break
    return h, cur

def _reweight_adaptive(h, n_iter, top_m, t_end):
    n = len(h); cur = _eval(h); step = 0.04
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        c = _eval_vec(h)
        idx = np.argsort(c)[-top_m:]
        cmin, cmax = c[idx].min(), c[idx].max()
        w = np.zeros(2 * n - 1)
        if cmax - cmin > 1e-12:
            w[idx] = np.exp(10.0 * (c[idx] - cmin) / (cmax - cmin))
        else:
            w[idx] = 1.0
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            g += w[k] * _gs(h, n, k)
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h2 = _repair(h - s * g, n)
            c2 = _eval(h2)
            if c2 < cur - 1e-15:
                h, cur = h2, c2; improved = True; break
        if not improved:
            step *= 0.65
            if step < 1e-7:
                break
    return h, cur

def _binary_quick(h, n, max_rounds, t_end):
    h = h.copy(); h[h > 0.5] = 1.0; h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h); c = np.correlate(h, 1.0 - h, mode='full'); dx = 2.0 / n
    ones = np.where(h == 1.0)[0]; zeros = np.where(h == 0.0)[0]
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False; k_star = int(c.argmax())
        for i in ones:
            if time.time() > t_end:
                break
            pi = k_star - i
            if not (0 <= pi < n) or h[pi] != 0:
                continue
            for j in zeros:
                pj = k_star - j
                if not (0 <= pj < n) or h[pj] != 1:
                    continue
                h[i] = 0.0; h[j] = 1.0
                c_new = np.correlate(h, 1.0 - h, mode='full')
                c5 = float(c_new.max() * dx)
                if c5 < cur - 1e-14:
                    cur = c5; c = c_new
                    ones = np.where(h == 1.0)[0]; zeros = np.where(h == 0.0)[0]
                    improved = True; break
                h[i] = 1.0; h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pair_cd(h, n, n_iter, scales, t_end):
    rng = np.random.RandomState(2024)
    cur = _eval(h)
    for scale in scales:
        for _ in range(n_iter // len(scales)):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i], h[j] = ni - d, nj + d
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h); best_n = n0

    # Phase 1: SLSQP with increasing alpha
    for alpha in [20000, 100000, 500000, 2000000, 5000000]:
        if time.time() > t_end - 250:
            break
        hn = _slsqp(best_h, alpha, 4000, t_end - 30)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5 = c5; best_h = hn.copy(); best_n = len(hn)

    # Phase 2: Subgradient with momentum (new approach)
    if time.time() < t_end - 180:
        h2 = best_h.copy()
        h2, c5 = _subgrad_mom(h2, 15000, 0.92, t_end - 120)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2.copy(); best_n = len(h2)

    # Phase 3: Adaptive reweighting multi-pass
    for top_m in [8, 4, 2, 1, 2, 3]:
        if time.time() > t_end - 100:
            break
        h3 = best_h.copy()
        h3, c5 = _reweight_adaptive(h3, 2500, top_m, t_end - 40)
        if c5 < best_c5:
            best_c5 = c5; best_h = h3.copy(); best_n = len(h3)

    # Phase 4: Multi-resolution
    if time.time() < t_end - 100:
        for nn in [best_n, best_n - 50, best_n + 50, 400, 600]:
            if time.time() > t_end - 50:
                break
            if nn < 100:
                continue
            hr = _repair(_resample(best_h, nn), nn)
            hr = _slsqp(hr, 300000, 2500, t_end - 30)
            c5 = _eval(hr)
            if c5 < best_c5:
                best_c5 = c5; best_h = hr.copy(); best_n = nn

    # Phase 5: Binary local search
    if time.time() < t_end - 50:
        nb = min(best_n, 400)
        hb = _repair(_resample(best_h, nb), nb)
        hb, c5 = _binary_quick(hb, nb, 200, t_end - 15)
        if c5 < best_c5:
            best_c5 = c5; best_h = hb.copy(); best_n = nb

    # Phase 6: Subgradient again on refined solution
    if time.time() < t_end - 30:
        h6 = best_h.copy()
        h6, c5 = _subgrad_mom(h6, 10000, 0.95, t_end - 15)
        if c5 < best_c5:
            best_c5 = c5; best_h = h6.copy(); best_n = len(h6)

    # Phase 7: Pair coordinate descent polish
    if time.time() < t_end - 10:
        h7 = best_h.copy()
        h7, c5 = _pair_cd(h7, len(h7), 8000, [0.003, 0.001, 0.0003], t_end - 5)
        if c5 < best_c5:
            best_c5 = c5; best_h = h7.copy(); best_n = len(h7)

    # Phase 8: Final SLSQP polish
    if time.time() < t_end - 8:
        hf = _slsqp(best_h, 8000000, 8000, t_end - 2)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5 = c5; best_h = hf.copy(); best_n = len(hf)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
