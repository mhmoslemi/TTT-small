import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _top_grad(h, m):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-m:]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * dx

def _accel_grad(h, n_iter, step0, m_shifts, time_limit, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.9
    step = step0
    t_start = time.time()
    stuck = 0
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        g = _top_grad(best, m_shifts)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-14:
                best = h_new
                best_c = c_new
                step = min(step * 1.12, 0.25)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.5
            stuck += 1
            if stuck > 40:
                best = _repair(best + 0.03 * rng.randn(n), n)
                best_c = _eval(best)
                step = step0
                v = np.zeros(n)
                stuck = 0
    return best, best_c

def _fine_search(h, n_tries, rng):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur

            else:
                h[i] = save
        elif r < 0.75:

            j = rng.randint(n)
            if i == j: continue
            d = rng.choice([0.005, 0.01, 0.02, 0.05]) * rng.choice([-1.0, 1.0])
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c2 = _eval(h)
                if c2 < c_cur - 1e-14:
                    c_cur = c2
                    if c2 < best_c: best = h.copy(); best_c = c2
                else:

        else:
            g = _top_grad(h, 15)

            d = -rng.choice([0.005, 0.01, 0.02, 0.05]) * np.sign(g[idx])
            ni = h[idx] + d

                    h = h_r; c_cur = c2
                    if c2 < best_c: best = h.copy(); best_c = c2
                else:
                    h[idx] = save
    return best, best_c

def _restart_opt(h, rng, alpha, maxiter, time_limit):
    n = len(h)
    t_start = time.time()
    for scale in [0.05, 0.1, 0.2]:
        if time.time() - t_start > time_limit:
            break
        pert = h + scale * rng.randn(n)
        pert = _repair(pert, n)
        hc = _slsqp(pert, alpha, maxiter, 1e-16)
        c = _eval(hc)
        c0 = _eval(h)
        if c < c0:
            h = hc
    return h, _eval(h)
