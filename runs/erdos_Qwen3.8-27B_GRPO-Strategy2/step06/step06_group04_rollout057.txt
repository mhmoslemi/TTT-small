import numpy as np
from scipy.optimize import minimize

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        # Distribute the difference
        diff = target - s
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = abs(diff)
        step = remaining / n
        for idx in order:
            if diff > 0:
                add = min(step, remaining, 1.0 - h[idx])
                h[idx] += add
                remaining -= add
            else:
                sub = min(step, remaining, h[idx])
                h[idx] -= sub
                remaining -= sub
            if remaining < 1e-12:
                break
    # Final exact fix
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

        h += diff / n
        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute correction
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    remaining = diff
    for idx in order:
        if remaining > 0:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
        else:
            sub = min(-remaining, h[idx])
            h[idx] -= sub
            remaining += sub
        if abs(remaining) < 1e-14:
            break
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def grad_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-s] = 1.0 - h[s:]
            g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] = 1.0 - h[:n-t]
        g[:n-t] -= h[t:]
    return g * dx

def _sobj(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    val += pen * (x.sum() - n * 0.5) ** 2
    return val

def _sgrad(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(len(c)):
        wi = w[ki]
        if wi < 1e-16:
            continue
        sh = ki - (n - 1)
        if sh >= 0:
            if sh == 0:
                g += wi * dx * (1.0 - 2.0 * x)
            else:
                g[:n-sh] += wi * dx * (1.0 - x[sh:])
                g[sh:] -= wi * dx * x[:n-sh]
        else:
            t = -sh
            g[t:] += wi * dx * (1.0 - x[:n-t])
            g[:n-t] -= wi * dx * x[t:]
    g += 2.0 * pen * (x.sum() - n * 0.5) * np.ones(n)
    return g

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def fo(x):
        return _sobj(x, alpha, pen)
    def go(x):
        return _sgrad(x, alpha, pen)
    try:
        res = minimize(fo, h, jac=go, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def _subgrad(h, n_iter, lr0, deadline, seed):
    n = len(h)
    rng = np.random.RandomState(seed)

    best_x = x.copy()

from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    remaining = diff
    for idx in order:
        if abs(remaining) < 1e-14:
            break
        if remaining > 0:
            a = min(remaining, 1.0 - h[idx]); h[idx] += a; remaining -= a
        else:
            a = min(-remaining, h[idx]); h[idx] -= a; remaining += a
    return h

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def grad_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-s] = 1.0 - h[s:]
            g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] = 1.0 - h[:n-t]
        g[:n-t] -= h[t:]
    return g * dx

def _sobj(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    val += pen * (x.sum() - n * 0.5) ** 2
    return val

def _sgrad(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(len(c)):
        wi = w[ki]
        if wi < 1e-16:
            continue
        sh = ki - (n - 1)
        if sh >= 0:
            if sh == 0:
                g += wi * dx * (1.0 - 2.0 * x)
            else:
                g[:n-sh] += wi * dx * (1.0 - x[sh:])
                g[sh:] -= wi * dx * x[:n-sh]
        else:
            t = -sh
            g[t:] += wi * dx * (1.0 - x[:n-t])
            g[:n-t] -= wi * dx * x[t:]
    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g

def _lbfgs(h, alpha, pen, maxit, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def fo(x): return _sobj(x, alpha, pen)
    def go(x): return _sgrad(x, alpha, pen)
    try:
        res = minimize(fo, h, jac=go, method='L-BFGS-B', bounds=bnds,
                      options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def _subgrad(h, n_iter, lr0, deadline, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    x = h.copy()
    best_x = x.copy()
    best_c = eval_c5(x)
    lr = lr0
    v = np.zeros(n)
    beta = 0.8
    for i in range(n_iter):
        if time.time() > deadline:
            break
        if i % 500 == 0 and i > 0:
            x = x + 0.002 * rng.randn(n)
            x = repair(x, n)
        corr = np.correlate(x, 1.0 - x, mode='full')
        k = int(corr.argmax())
        g = grad_shift(x, k)
        v = beta * v + (1.0 - beta) * g
        x = x - lr * v
        x = repair(x, n)
        c = eval_c5(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
        lr *= 0.99995
    return best_x, best_c

def _resample(h, n_new):
    n = len(h)
    xo = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _coord_descend(h, deadline, scale0):
    n = len(h)
    rng = np.random.RandomState(42)
    x = h.copy()
    c_cur = eval_c5(x)
    best_x = x.copy()
    best_c = c_cur
    scale = scale0
    for i in range(800000):
        if time.time() > deadline:
            break
        if scale < 1e-10:
            break
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = scale * rng.randn()
        x[i1] += d
        x[i2] -= d
        x = repair(x, n)
        c_new = eval_c5(x)
        if c_new < c_cur - 1e-18:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
            scale = min(scale * 1.002, 0.05)
        else:
            x[i1] -= d
            x[i2] += d
            x = repair(x, n)
            scale *= 0.97
    return best_x, best_c

def _slsqp(h, alpha, pen, maxit, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def fo(x): return _sobj(x, alpha, pen)
    def go(x): return _sgrad(x, alpha, pen)
    def so(x): return x.sum() - n * 0.5
    def sj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': so, 'jac': sj}]
    try:
        res = minimize(fo, h, jac=go, method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxit, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

            break
        h_new = _lbfgs(best_h, alpha, 1e4, 600, dl1)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Subgradient + L-BFGS-B (170-350s)
    dl2 = t0 + 350
    x, c = _subgrad(best_h, 60000, 0.004, dl2, seed)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl2:
        h_new = _lbfgs(best_h, 50000000, 1e5, 1000, dl2)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Multi-n exploration (350-550s)
    dl3 = t0 + 550
    for n_try in [300, 304, 306, 310, 312, 314, 316, 318, 320, 290, 280]:
        if time.time() > dl3:
            break
        if n_try % 2 != 0:
            n_try += 1
        h_r = repair(_resample(best_h, n_try), n_try)
        h_new = _lbfgs(h_r, 100000, 1e4, 500, min(dl3, time.time()+80))
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_new.copy(), n_try
        if time.time() < dl3 - 30:
            x2, c2 = _subgrad(h_new, 15000, 0.003, min(dl3, time.time()+50), seed + n_try)
            if c2 < best_c5:
                best_c5, best_h, best_n = c2, x2.copy(), n_try

    # Phase 4: Very high alpha L-BFGS-B (550-720s)
    dl4 = t0 + 720
    for alpha in [100000000, 500000000, 2000000000]:
        if time.time() > dl4:
            break
        h_new = _lbfgs(best_h, alpha, 1e5, 1500, dl4)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: SLSQP polish with equality constraint (720-850s)
    dl5 = t0 + 850
    for alpha in [50000000, 500000000]:
        if time.time() > dl5:

        h_new = _slsqp(best_h, alpha, 0, 2000, dl5)
        c5 = eval_c5(h_new)
        if c5 < best_c5:

    dl6 = t0 + 1000

        if c_cd2 < best_c5:
            best_c5, best_h = c_cd2, x_cd2.copy()

    best_h = repair(best_h, best_n)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, best_n)
