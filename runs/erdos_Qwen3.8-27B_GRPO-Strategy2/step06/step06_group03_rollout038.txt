def _pnorm_obj(h, p, lam, target):
    c = _corr(h)
    # p-norm smooth max
    c_pos = np.maximum(c, 1e-10)
    obj = (np.sum(c_pos**p))**(1.0/p)

    return obj

def _pnorm_grad(h, p, lam, target):

    weights = c_pos**(p-1) / cnorm**(p-1)  # gradient weights

    # penalty gradient

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _cvec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(_cvec(h).max() * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

    S = np.sum(c**p)
    Sp = S**(1.0/p)

            continue

    shs = np.arange(2*n-1) - (n-1)

            for k in range(2*n-1):

                bk = 0.0
                j1 = idx + sh
                if 0 <= j1 < n:

                j2 = idx - sh

        c = _cvec(h)
        mc = c.max()
        if mc < best_c:

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

            hi = mid

    c = _cvec(h)
    S = np.sum(c**p)
    Sp = S**(1.0/p)
    w = c**(p-1) / S**((p-1)/p)
    dx = 2.0/n
    g = np.zeros(n)
    shs = np.arange(2*n-1) - (n-1)
    for k in range(2*n-1):

        if abs(wkv) < 1e-18:
            continue
        sh = shs[k]
        wk = wkv * dx
        if sh == 0:

        elif sh > 0:

            g[sh:] -= wk*h[:n-sh]
        else:

            g[sp:] += wk*(1.0-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    g += 2.0*lam*(h.sum()-0.5*n)
    return Sp, g

def _slsqp(h, p, lam, maxiter, ftol):

    bnds = [(0.0,1.0)]*n
    def _f(x): return _pnorm_og(x, p, lam)[0]
    def _g(x): return _pnorm_og(x, p, lam)[1]
    cons = [{'type':'eq','fun':lambda x: x.sum()-0.5*n,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
