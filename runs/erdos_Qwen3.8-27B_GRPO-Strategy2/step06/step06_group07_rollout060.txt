import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(np.abs(h - 0.5))
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    sm = s.max()
    es = np.exp(s - sm)
    return (m + (sm + np.log(es.sum())) / alpha) * dx + lam * (x.sum() - n * 0.5) ** 2

def _grad(x, alpha, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    sm = s.max()
    es = np.exp(s - sm)
    w = es / es.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    g += 2.0 * lam * (x.sum() - n * 0.5)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _sobj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    sm = s.max()
    es = np.exp(s - sm)
    return (m + (sm + np.log(es.sum())) / alpha) * dx

def _sgrad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    sm = s.max()
    es = np.exp(s - sm)
    w = es / es.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h0, alphas, lam, maxiter, t_end):
    n = len(h0)
    h = h0.copy()
    best_c = _eval(h)
    bnds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > t_end: break
        try:
            res = minimize(_obj, h, jac=_grad, args=(alpha, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
            h = _repair(res.x, n)
            c = _eval(h)
            if c < best_c: best_c = c
        except Exception: pass
    return h, best_c

def _slsqp(h0, alphas, maxiter, t_end):
    n = len(h0)
    h = h0.copy()
    best_c = _eval(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    for alpha in alphas:
        if time.time() > t_end: break
        try:
            res = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-18})
            h = _repair(res.x, n)
            c = _eval(h)
            if c < best_c: best_c = c
        except Exception: pass
    return h, best_c

def _resample(h, n_new):
    n = len(h)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _coord_descent(h, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-17: c5 = c5n
        else: h[i], h[j] = ni - d, nj + d
    return h, c5

def _targeted_shift(h, t_end, lr, n_steps):
    """Minimize the worst shift overlap via targeted gradient steps."""
    n = len(h)
    best_c = _eval(h)
    for _ in range(n_steps):
        if time.time() > t_end: break
        c = np.correlate(h, 1.0 - h, mode='full')
        kstar = c.argmax()
        sh = (n - 1) - kstar
        dx_v = 2.0 / n
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += dx_v * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= dx_v * h[:-sh]
        else:
            sp = -sh
            g[sp:] += dx_v * (1.0 - h[:-sp])
            g[:-sp] -= dx_v * h[sp:]
        h_new = _repair(h - lr * g, n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-17:
            h = h_new; best_c = c_new
        else:
            lr *= 0.7
            if lr < 1e-8: lr = 0.01
    return h, best_c
