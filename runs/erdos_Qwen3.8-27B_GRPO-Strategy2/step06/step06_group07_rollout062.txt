def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sm_obj(x, T, n, alpha):
    m = n // T
    hf = np.tile(x, m)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    return mx + np.log(float(np.sum(np.exp(s)))) / alpha

def _sm_grad(x, T, n, alpha):
    m = n // T
    hf = np.tile(x, m)
    c = np.correlate(hf, 1.0 - hf, mode='full')
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    n1 = n - 1
    gf = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - n1
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            gf[:cnt] += wk * (1.0 - hf[sh:sh + cnt])
            if sh > 0:
                gf[sh:] -= wk * hf[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            gf[sp:] += wk * (1.0 - hf[:cnt])
            gf[:cnt] -= wk * hf[sp:sp + cnt]
    return gf.reshape(m, T).sum(axis=0) / m

def _se(x):
    return x.sum() - len(x) * 0.5

def _sj(x):
    return np.ones(len(x))

def _opt_per(h_p, T, n, alphas, maxit=4000):
    best = h_p.copy()
    for a in alphas:
        bnds = [(0.0, 1.0)] * T
        cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
        try:
            r = minimize(_sm_obj, best, jac=_sm_grad, args=(T, n, a),
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': maxit, 'ftol': 1e-20})
            best = _repair(r.x, T)
        except Exception:
            pass
    return best

def _pgd(h_p, T, n, alpha, iters, lr=0.004, seed=0):
    m = n // T
    rng = np.random.RandomState(seed)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    mom = np.zeros(T)
    beta = 0.82
    for _ in range(iters):
        hf = np.tile(h_p, m)
        c = np.correlate(hf, 1.0 - hf, mode='full')
        mx = c.max()
        s = np.clip(alpha * (c - mx), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / n
        nf = _sm_grad(h_p, T, n, alpha)
        mom = beta * mom + (1.0 - beta) * nf
        h_new = np.clip(h_p - lr * mom, 0, 1)

        else:

def _resample(h_old, n_new):
    xo = np.linspace(0, 2, len(h_old), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _cd(h, n_iter, scales, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            oi, oj = h[i], h[j]
            ni, nj = oi + d, oj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-16:
                c5c = cn
            else:
                h[i], h[j] = oi, oj
    return h, c5c

def _polarize(h, n):

    b = np.zeros(n)

    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
