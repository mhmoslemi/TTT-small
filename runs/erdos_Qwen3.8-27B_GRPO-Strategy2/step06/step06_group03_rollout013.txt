import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sgrad(h, alpha, top_m=100):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    idx = np.argsort(w)[-top_m:]
    for k in idx:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol, top_m):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=lambda x: _sgrad(x, alpha, top_m), args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h, alpha0, n_iter, lr, decay):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c = _eval(best)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    alpha = alpha0
    for i in range(1, n_iter + 1):
        g = _sgrad(best, alpha, top_m=50)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        mhat = m / (1 - beta1 ** i)
        vhat = v / (1 - beta2 ** i)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        h_new = _proj(best - step, target)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-15:
            best = h_new
            best_c = c_new
        else:
            lr *= 0.8
        alpha *= decay
    return best, best_c

def _guided_pairs(h, n_tries, rng, delta0):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy()
    best_c = c_cur
    for t in range(n_tries):
        # Guided: use gradient to select pairs
        g = _sgrad(h, 5000, top_m=30)
        # Want to decrease h where g>0, increase where g<0
        pos = np.where(g > 0.1 * np.max(np.abs(g)))[0]
        neg = np.where(g < -0.1 * np.max(np.abs(g)))[0]
        if len(pos) > 0 and len(neg) > 0:
            i = pos[rng.randint(len(pos))]
            j = neg[rng.randint(len(neg))]
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d = delta0 * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return best, best_c

    t0 = time.time()
    def _el(): return time.time() - t0
    rng = np.random.RandomState(seed)
    
    n0 = len(initial_h_values)

        hc = _slsqp(best_h, alpha, it, 1e-15, 80)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc
    
    # Phase 2: Adam with adaptive alpha
    if _el() < budget_s - 350:
        n_it = min(4000, int((budget_s - 350 - _el()) / 0.03))
        best_h, best_c = _adam(best_h, 30000, n_it, 0.008, 0.9997)
        best_n = len(best_h)
    
    # Phase 3: SLSQP refinement with high alpha
    for alpha, it in [(50000, 1200), (200000, 1200), (1000000, 1200)]:
        if _el() > budget_s - 200: break
        hc = _slsqp(best_h, alpha, it, 1e-16, 50)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc
    
    # Phase 4: Second Adam pass with different params
    if _el() < budget_s - 150:
        n_it = min(3000, int((budget_s - 150 - _el()) / 0.03))
        h2, c2 = _adam(best_h, 100000, n_it, 0.005, 0.9998)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    
    # Phase 5: Final SLSQP
    if _el() < budget_s - 60:
        hc = _slsqp(best_h, 5000000, 1500, 1e-18, 30)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc
            best_n = len(hc)
    
    # Phase 6: Guided pairwise search
    if _el() < budget_s - 15:
        n_tr = min(int((budget_s - 15 - _el()) / 0.005), 20000)
        best_h, best_c = _guided_pairs(best_h, n_tr, rng, 5e-5)
        best_n = len(best_h)
    
    # Final repair and eval
    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)

import numpy as np
from scipy.optimize import minimize
import time

_GN = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(x, p):
    alpha = p[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _sgrad(x, p):
    alpha, top_m = p[0], p[1]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()

    dx = 2.0 / n
    g = np.zeros(n)
    idx = np.argsort(w)[-top_m:]
    for k in idx:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:

            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
