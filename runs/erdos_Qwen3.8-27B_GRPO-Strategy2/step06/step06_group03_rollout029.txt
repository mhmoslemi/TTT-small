import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h.copy(), jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _pair_delta(h, i, j, n, c):
    """Compute optimal δ to transfer from j to i, given current c."""
    shifts = np.arange(2*n-1) - (n-1)
    di = np.zeros(2*n-1)
    v1 = (shifts >= -i) & (shifts < n - i)
    di[v1] += 1.0 - h[i + shifts[v1]]
    v2 = (shifts >= i - n + 1) & (shifts <= i)
    di[v2] -= h[i - shifts[v2]]
    dj = np.zeros(2*n-1)
    v1j = (shifts >= -j) & (shifts < n - j)
    dj[v1j] += 1.0 - h[j + shifts[v1j]]
    v2j = (shifts >= j - n + 1) & (shifts <= j)
    dj[v2j] -= h[j - shifts[v2j]]
    d = di - dj
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    if lo >= hi - 1e-14:
        return 0.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        vals = c + mid * d
        imax = np.argmax(vals)
        if d[imax] > 1e-12:
            hi = mid
        elif d[imax] < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def _local_search(h, n, time_limit, rng):
    """Local search with pair moves and exact 1D optimization."""
    t_start = time.time()
    best = h.copy()
    best_c = _eval(best)
    no_improve = 0
    pass_count = 0
    while time.time() - t_start < time_limit:
        c = np.correlate(best, 1.0 - best, mode='full')
        improved = False
        # Try n random pairs
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = _pair_delta(best, i, j, n, c)
            if abs(delta) < 1e-12:
                continue
            best[i] += delta
            best[j] -= delta
            # Clip to be safe
            best[i] = np.clip(best[i], 0, 1)
            best[j] = np.clip(best[j], 0, 1)
            # Re-evaluate
            new_c = _eval(best)
            if new_c < best_c - 1e-14:
                best_c = new_c
                c = np.correlate(best, 1.0 - best, mode='full')
                improved = True
                no_improve = 0
            else:
                # Undo
                best[i] -= delta
                best[j] += delta
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
    return best, best_c

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    c = np.correlate(h, 1.0 - h, mode='full')

        else:

            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]

def _pair_opt(h, i, j, n, c):
    """Find optimal transfer δ from j to i. Returns (delta, d_vec)."""
    shifts = np.arange(2 * n - 1) - (n - 1)
    di = np.zeros(2 * n - 1)
    # First term: (1-h[i-shift]) valid when 0 <= i-shift < n
    v1 = (shifts >= i - n + 1) & (shifts <= i)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n

            g[:n-sp] -= wk * h[sp:]

def _pair_opt(h, i, j, n, c, shifts):

    v1 = (shifts >= i - n + 1) & (shifts <= i)
    if np.any(v1): di[v1] += 1.0 - h[i - shifts[v1]]
    v2 = (shifts >= -i) & (shifts < n - i)
    if np.any(v2): di[v2] -= h[i + shifts[v2]]
    dj = np.zeros(sz)
    v1j = (shifts >= j - n + 1) & (shifts <= j)
    if np.any(v1j): dj[v1j] += 1.0 - h[j - shifts[v1j]]
    v2j = (shifts >= -j) & (shifts < n - j)
    if np.any(v2j): dj[v2j] -= h[j + shifts[v2j]]
    d = di - dj
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    if lo >= hi - 1e-14:
        return 0.0, d
    imax = np.argmax(c)
    if d[imax] < -1e-12:
        return lo if lo > -1e-13 else 0.0, d
    if d[imax] > 1e-12:
        return hi if hi < 1e-13 else 0.0, d
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        im = np.argmax(c + mid * d)
        dk = d[im]
        if dk > 1e-10: hi = mid
        elif dk < -1e-10: lo = mid
        else: break
    return 0.5 * (lo + hi), d

def _local_search(h, n, t_limit, rng):
    t0 = time.time()
    best = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)
    c = np.correlate(best, 1.0 - best, mode='full')
    best_c = float(c.max() * 2.0 / n)
    no_imp = 0
    moves = 0
    while time.time() - t0 < t_limit:
        improved = 0
        for _ in range(n):
            if time.time() - t0 > t_limit: break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j: continue
            delta, d = _pair_opt(best, i, j, n, c, shifts)
            if abs(delta) < 1e-13: continue
            oi, oj = best[i], best[j]
            ni, nj = oi + delta, oj - delta
            clipped = ni < -1e-14 or ni > 1+1e-14 or nj < -1e-14 or nj > 1+1e-14
            best[i] = min(max(ni, 0.0), 1.0)
            best[j] = min(max(nj, 0.0), 1.0)
            moves += 1
            if not clipped:
                c += delta * d
            else:
                c = np.correlate(best, 1.0 - best, mode='full')
            nc = c.max() * 2.0 / n
            if nc < best_c - 1e-13:
                best_c = nc
                improved += 1
            else:
                best[i], best[j] = oi, oj
                if not clipped:
                    c -= delta * d
                else:
                    c = np.correlate(best, 1.0 - best, mode='full')
        if moves % 1000 == 0 and moves > 0:
            c = np.correlate(best, 1.0 - best, mode='full')
            best_c = float(c.max() * 2.0 / n)
        if improved == 0:
            no_imp += 1
            if no_imp >= 12: break
        else:
            no_imp = 0
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el(): return time.time() - t0
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = _eval(best_h)

    # Phase 1: SLSQP moderate alpha
    for alpha in [300, 2000, 15000]:
        if _el() > 100: break
        hc = _slsqp(best_h, alpha, 2500, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 2: Main local search with exact pair moves
    if _el() < budget_s - 350:
        best_h, best_c = _local_search(best_h, n, budget_s - 350 - _el(), rng)

    # Phase 3: SLSQP high alpha polish
    for alpha in [50000, 300000]:
        if _el() > budget_s - 200: break
        hc = _slsqp(best_h, alpha, 2000, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 4: Final local search
    if _el() < budget_s - 40:
        best_h, best_c = _local_search(best_h, n, budget_s - 40 - _el(), rng)

    # Phase 5: Ultra-fine SLSQP
    if _el() < budget_s - 20:
        hc = _slsqp(best_h, 1000000, 2000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Final local search with remaining time
    if _el() < budget_s - 3:
        best_h, best_c = _local_search(best_h, n, budget_s - 3 - _el(), rng)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
