import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    logsum = np.log(float(np.sum(np.exp(s - s.max())))) + s.max()
    return (2.0 / n) * (m + logsum / alpha)

def _soft_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _adam(h0, a0, a1, nit, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m_ = np.zeros(n)
    v_ = np.zeros(n)
    t = 0
    best_x, best_c = x.copy(), _eval(x)
    for i in range(nit):
        if time.time() > deadline:
            break
        t += 1
        f = i / max(nit - 1, 1)
        alpha = a0 * (a1 / a0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        if t % 800 == 0:
            x = x + 0.003 * rng.randn(n)
            x = _repair(x, n)
        g = _soft_grad(x, alpha)
        m_ = 0.9 * m_ + 0.1 * g
        v_ = 0.999 * v_ + 0.001 * g * g
        mh = m_ / (1 - 0.9 ** t)
        vh = v_ / (1 - 0.999 ** t)
        x = x - lr * mh / (np.sqrt(vh) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c, best_x = c, x.copy()
    return best_x, best_c

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _coord_descend(h, n, n_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = _eval(h)
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        idx = rng.permutation(n)
        for i in idx:
            c = np.correlate(h, 1.0 - h, mode='full')
            # Compute B_k = d(c_k)/d(h[i])
            B = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):

                b = 0.0
                if 0 <= i + shift < n:
                    b += (1.0 - h[i + shift])
                if 0 <= i - shift < n:
                    b -= h[i - shift]
                B[k] = b
            # f(t) = max_k (c_k + B_k * (t - h[i]))
            # Find min of f on [0,1]
            # Grid + refine
            ts = np.linspace(0.0, 1.0, 50)
            vals = np.max(c + B * (ts[:, None] - h[i])[None, :], axis=1)  # wrong shape
            # Actually: f(t) = max_k (c_k + B_k*(t - h[i]))
            # Vectorized: for each t, compute c + B*(t - h[i]), then max
            ts_col = (ts[:, None] - h[i])  # shape (50, 1)
            vals = np.max(c[None, :] + B[None, :] * ts_col, axis=1)  # shape (50,)
            bi = np.argmin(vals)
            t_best = ts[bi]
            # Refine: bisection around ts[bi]
            lo, hi = max(0.0, ts[max(bi-1, 0)]), min(1.0, ts[min(bi+1, 49)])
            for _ in range(20):
                mid = (lo + hi) / 2
                # Evaluate at mid and at lo, hi
                f_mid = np.max(c + B * (mid - h[i]))
                # Determine direction by subgradient at mid
                active = np.where(np.abs(c + B * (mid - h[i]) - f_mid) < 1e-12)[0]
                if len(active) == 0:
                    f_mid = np.max(c + B * (mid - h[i]))
                    active = np.where(np.abs(c + B * (mid - h[i]) - f_mid) < 1e-10)[0]
                if len(active) > 0:
                    min_b = B[active].min()
                    max_b = B[active].max()
                    if min_b > 1e-10:
                        hi = mid  # go left
                    elif max_b < -1e-10:
                        lo = mid  # go right
                    else:
                        break  # optimal
                else:
                    break
            t_best = (lo + hi) / 2
            if t_best < 0: t_best = 0
            if t_best > 1: t_best = 1
            if abs(t_best - h[i]) > 1e-10:
                h[i] = t_best
                h = _repair(h, n)
                c_new = _eval(h)
                if c_new < best_c:

    return h, best_c

def _binary_local_search(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    x = (h > 0.5).astype(float)
    x = _repair(x, n)
    best_c = _eval(x)
    best_x = x.copy()
    # Single flip
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            x[i] = 1.0 - x[i]
            x = _repair(x, n)
            c = _eval(x)
            if c < best_c - 1e-14:
                best_c = c
                best_x = x.copy()
                improved = True
            else:
                x[i] = 1.0 - x[i]
                x = _repair(x, n)
    # Pair swap
    if time.time() < deadline:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for trial in range(5000):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                if x[i] == x[j]:
                    continue
                # Swap: x[i]=1,x[j]=0 → x[i]=0,x[j]=1 or vice versa
                trial_x = x.copy()
                trial_x[i], trial_x[j] = x[j], x[i]
                trial_x = _repair(trial_x, n)
                c = _eval(trial_x)
                if c < best_c - 1e-14:
                    best_c = c
                    best_x = trial_x.copy()
                    x = trial_x
                    improved = True
    return best_x, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    ls = np.log(float(np.sum(np.exp(s - s.max())))) + s.max()
    return (2.0 / n) * (m + ls / alpha)

def _soft_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _adam(h0, a0, a1, nit, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m_ = np.zeros(n); v_ = np.zeros(n); t = 0
    best_x, best_c = x.copy(), _eval(x)
    for i in range(nit):
        if time.time() > deadline:
            break
        t += 1
        f = i / max(nit - 1, 1)
        alpha = a0 * (a1 / a0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        if t % 600 == 0:
            x = x + 0.004 * rng.randn(n)
            x = _repair(x, n)
        g = _soft_grad(x, alpha)
        m_ = 0.85 * m_ + 0.15 * g
        v_ = 0.999 * v_ + 0.001 * g * g
        x = x - lr * (m_ / (1 - 0.85 ** t)) / (np.sqrt(v_ / (1 - 0.999 ** t)) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c, best_x = c, x.copy()
    return best_x, best_c

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _coord_hd(h, n, n_passes, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy(); best_c = _eval(h)
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        idx = rng.permutation(n)
        any_change = False
        for i in idx:
            c = np.correlate(h, 1.0 - h, mode='full')
            B = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                sh = k - (n - 1)
                b = 0.0
                if 0 <= i + sh < n:
                    b += (1.0 - h[i + sh])
                if 0 <= i - sh < n:
                    b -= h[i - sh]
                B[k] = b
            base = c - B * h[i]
            ts = np.linspace(0.0, 1.0, 80)
            vals = np.max(base[None, :] + B[None, :] * ts[:, None], axis=1)
            bi = int(np.argmin(vals))
