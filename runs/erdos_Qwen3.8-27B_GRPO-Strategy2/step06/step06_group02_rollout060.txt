```python
def compute_grad(h, top_indices, weights):
    """Compute weighted gradient of sum of top correlation values."""
    n = len(h)
    g = np.zeros(n)
    for s_idx, w in zip(top_indices, weights):
        s = s_idx - (n - 1)  # actual shift
        # (1-h[j-s]) for valid j-s
        lo = max(0, -s)  # j ≥ max(0, -s) so j-s ≥ 0
```

```python
def project(x, n):
    """Project x onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search for lambda
    lo, hi = x.min() - 1.0, x.max()
    for _ in range(50):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fine-tune for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h
```

```python
    target = n / 2.0
```
```python
    h = h.copy()
```

```python
def _slsqp_lse(h, alpha, maxiter, t_end):
    """SLSQP with LSE objective and fast gradient for top shifts."""
    n = len(h)
    target = n / 2.0
    
    def obj(x):

        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha

        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        # Only compute for shifts with significant weight
        thresh = 1e-6
        active = np.where(w > thresh)[0]
        return _grad_weighted(x, active, w[active], n)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(x, target):
    lo = float(x.min()) - 1.0
    hi = float(x.max())
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_weighted(h, indices, weights, n):
    g = np.zeros(n)
    for k_idx, w in zip(indices, weights):
        k = int(k_idx) - (n - 1)
        j1 = max(0, k); j2 = min(n - 1, n - 1 + k)
        if j1 <= j2:
            g[j1:j2 + 1] += w * (1.0 - h[j1 - k:j2 - k + 1])
        j3 = max(0, -k); j4 = min(n - 1, n - 1 - k)
        if j3 <= j4:
            g[j3:j4 + 1] -= w * h[j3 + k:j4 + k + 1]
    return g

def _subgrad(h, n, n_iter, m, mom, step0, t_end):
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max()) * 2.0 / n
    step = step0
    best_cur = cur
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        top_m = np.argsort(c)[-m:]
        c_top = c[top_m]
        rng_w = c_top.max() - c_top.min()
        if rng_w > 1e-12:
            w = np.exp(4.0 * (c_top - c_top.min()) / rng_w)
        else:
            w = np.ones(m)
        w /= w.sum()
        g = _grad_weighted(h, top_m, w, n)
        v = mom * v - step * g
        h_new = _project(h + v, target)
        c = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_new = float(c.max()) * 2.0 / n
        if c_new < cur:
            h = h_new
            cur = c_new
            if cur < best_cur:
                best_cur = cur
                best_h = h.copy()
        else:
            step *= 0.8
            v *= 0.4
    return best_h, best_cur

def _pair_exchange(h, n, n_iter, t_end):
    target = n / 2.0
    h = h.copy()
    cur = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top3 = np.argsort(c)[-3:]
        g = np.zeros(n)
        for k_idx in top3:
            k = int(k_idx) - (n - 1)
            j1 = max(0, k); j2 = min(n - 1, n - 1 + k)
            if j1 <= j2:
                g[j1:j2 + 1] += (1.0 - h[j1 - k:j2 - k + 1])
            j3 = max(0, -k); j4 = min(n - 1, n - 1 - k)
            if j3 <= j4:
                g[j3:j4 + 1] -= h[j3 + k:j4 + k + 1]
        i = int(np.argmax(g)); j = int(np.argmin(g))
        if i == j:
            break
        moved = False
        for delta in [0.04, 0.02, 0.01, 0.005, 0.002]:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = _project(h_new, target)
            c_new = _eval(h_new)
            if c_new < cur - 1e-13:
                h = h_new; cur = c_new; moved = True
                break
        if not moved:
            break
    return h, cur

def _slsqp_lse(h, alpha, maxiter, t_end):
    n = len(h)
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        active = np.where(w > 1e-7)[0]
        if len(active) == 0:
            return np.zeros(len(x))
        return _grad_weighted(x, active, w[active], n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        return _project(res.x, target)
    except Exception:
        return h.copy()
```
