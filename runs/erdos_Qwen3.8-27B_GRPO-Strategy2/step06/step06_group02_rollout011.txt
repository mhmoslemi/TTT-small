    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _lse_and_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 500)
    s -= s.max()
    w = np.exp(s)
    W = w.sum()
    obj = cm + np.log(W) / alpha
    w /= W
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        sh = k - n + 1
        wk = w[k] * 2.0 / n
        if sh >= 0:
            idx1 = np.arange(n - sh)
            grad[idx1] += wk * (1.0 - h[idx1 + sh])
            if sh > 0:

            grad[idx1] += wk * (1.0 - h[idx1 - sp])
            idx2 = np.arange(n - sp)
            grad[idx2] -= wk * h[idx2 + sp]

                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, _g_n)
    except Exception:
        return h.copy()
