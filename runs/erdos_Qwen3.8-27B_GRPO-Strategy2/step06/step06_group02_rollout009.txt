```python
    # Phase 7: Final exact PGD with top_m=1
    if rem() > 30:
        h7 = _pgd(best_h, 8000, 1, 100.0, t0 + rem() - 5)
        c5 = _eval(h7)
        if c5 < best_c5:
            best_c5 = c5; best_h = h7

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
```
