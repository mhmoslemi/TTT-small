def _benders_step(h0, max_lags=15, tol=1e-10):
    """One Benders decomposition step."""
    from scipy.optimize import linprog
    n = len(h0)
    dx = 2.0 / n
    c = _correlate(h0)  # n*2-1 values
    # Get top lags
    top_k = min(max_lags, len(c))
    top_idx = np.argsort(c)[-top_k:]
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize t: c_obj = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag s: sum_i g_i * h_i - t <= rhs
    A_rows = []
    b_rows = []
    
    for k in top_idx:
        s_val = 2 * k - n + 1  # actual lag
        g = _lag_grad(h0, s_val, dx)
        f_s = c[k] * dx  # actual weighted value
        
        # g · h - t <= -f_s + g · h0
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-f_s + np.dot(g, h0))
    
    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.array(b_rows) if b_rows else None
    
    # Equality: sum h_i = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    # Bounds: 0 <= h_i <= 1, t is free (but bounded below by 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return h0, np.max(c) * dx

            best_h, best_c5 = _topk_descent(best_h, 7, 300)

    # Phase 7: Pairwise CD fine-tuning
    if tl() > 80:
        h7, c5 = _pair_cd(best_h, 0.005, 5000, rng)
        if c5 < best_c5: best_c5, best_h = c5, h7.copy()
    if tl() > 50:
        h7, c5 = _pair_cd(best_h, 0.001, 8000, rng)
        if c5 < best_c5: best_c5, best_h = c5, h7.copy()
    if tl() > 30:
        h7, c5 = _pair_cd(best_h, 0.0003, 6000, rng)
        if c5 < best_c5: best_c5, best_h = c5, h7.copy()

    # Phase 8: Multi-resolution
    if tl() > 200:
        for n_try in [310, 400, 500, 620]:
            if tl() < 60: break
            hr = _repair(_resample(best_h, n_try), n_try)
            hr = _lbestep(hr, 20000, 50.0, 400)
            hr, c5 = _topk_descent(hr, 7, 300)
            hr, c5 = _pair_cd(hr, 0.002, 3000, rng)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hr.copy(), n_try

    # Phase 9: Final polish
    if tl() > 30:
        hp, c5 = _topk_descent(best_h, 12, 400)
        if c5 < best_c5: best_c5, best_h = c5, hp.copy()
    if tl() > 15:
        hp, c5 = _pair_cd(best_h, 0.0001, 5000, rng)
        if c5 < best_c5: best_c5, best_h = c5, hp.copy()
    if tl() > 10:
        hp, c5 = _momentum_descent(best_h, 100, 0.378, 0.382, mu=0.9)
        if c5 < best_c5: best_c5, best_h = c5, hp.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
