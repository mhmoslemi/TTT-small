    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    nc = 2*n - 1
    if topk < nc:

    else:
        idx = np.arange(nc)
    cv = c[idx]
    m = cv.max()

            continue
        k = idx[i]

        wk = w[i] * dx
        if lag >= 0:
            # First term: i in [lag, n-1], contribution (1-h[i-lag])
            rng1 = np.arange(lag, n)
            g[rng1] += wk * (1.0 - h[rng1 - lag])
            # Second term: i in [0, n-1-lag], contribution -h[i+lag]
            if lag < n:

            g[rng1] += wk * (1.0 - h[rng1 + L])
            # Second term: i in [L, n-1], contribution -h[i-L]
            rng2 = np.arange(L, n)
            g[rng2] -= wk * h[rng2 - L]
