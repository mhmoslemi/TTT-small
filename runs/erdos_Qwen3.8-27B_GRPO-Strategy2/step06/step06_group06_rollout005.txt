    n = len(h_init)
    h = h_init.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    for t in range(1, n_steps + 1):
        if time.time() > t_end:
            break
        g = _smax_grad(h, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        decay = lr / (1.0 + 0.5 * t / 200.0)
        h_new = _proj(h - decay * m_hat / (np.sqrt(v_hat) + 1e-8), n * 0.5)
        c = _eval(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _nesterov_subgrad(h_init, n_steps, t_end, k_active=6):
    n = len(h_init)
    dx = 2.0 / n
    x = h_init.copy()
    y = x.copy()
    best_h = h_init.copy()
    best_c = _eval(h_init)
    for t in range(1, n_steps + 1):
        if time.time() > t_end:
            break
        c = _corr(y)
        top_idx = np.argsort(c)[::-1][:k_active]
        g = np.zeros(n)
        for kk in top_idx:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - y[sh:])
                if sh > 0:
                    g[sh:] -= y[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - y[:n - sp])
                g[:n - sp] -= y[sp:]
        g *= dx / k_active
        step = 1.5 / (1.0 + t * 0.05)
        x_new = _proj(y - step * g, n * 0.5)
        beta_t = (t - 1.0) / (t + 1.0)
        y_new = x_new + beta_t * (x_new - x)
        y_new = np.clip(y_new, 0, 1)
        c_new = _eval(x_new)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        x = x_new
        y = y_new
    return best_h, best_c

def _topk_penalty_descent(h_init, n_steps, t_end, k_top=10):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c = _eval(h)
    alpha_pen = 200.0
    for t in range(n_steps):
        if time.time() > t_end:
            break
        c = _corr(h)
        m = c.max()
        top_vals = np.sort(c)[::-1][:k_top]
        w = np.exp(alpha_pen * (c - m))
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(2 * n - 1):
            wk = w[kk]
            if wk < 1e-14:
                continue
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        g *= dx
        step = 0.5 / (1.0 + t * 0.01)
        h_new = _proj(h - step * g, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if t % 200 == 0:
            alpha_pen *= 1.5
    return best_h, best_c

def _slsqp_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x):
        return x.sum() - n * 0.5
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _upsample(h, factor):
    n = len(h)
    h_hi = np.zeros(n * factor)
    for i in range(n):
        h_hi[i * factor:(i + 1) * factor] = h[i]
    return h_hi

def _diverse_starts(n, rng):
    out = []
    x = np.linspace(0, 2, n)
    h1 = _repair(np.clip(1.0 - np.abs(2.0 * x - 1.0), 0, 1), n)
    out.append(h1)
    h2 = _repair(np.clip(0.5 + 0.45 * np.sin(np.pi * x), 0, 1), n)
    out.append(h2)
    h3 = _repair(np.clip(0.5 + 0.35 * np.sin(2.0 * np.pi * x + rng.randn()) + 0.2 * np.sin(4.0 * np.pi * x), 0, 1), n)
    out.append(h3)
    h4 = np.zeros(n)
    h4[:n // 4] = 1.0
    h4[3 * n // 4:] = 1.0
    out.append(_repair(h4, n))
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP polish base resolution
    for alpha in [500, 3000, 15000, 60000, 150000]:
        if remaining() < 350:
            break
        h_c = _slsqp_opt(best_h, alpha, 3500, t0 + remaining() - 349)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 2: ADAM with progressive alpha
    if remaining() > 310:
        for alpha in [3000, 15000, 60000, 150000]:
            if remaining() < 290:
                break
            h_c, c5 = _adam_opt(best_h, alpha, 900, t0 + remaining() - 289, lr=0.02)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c.copy()

    # Phase 3: Nesterov subgradient
    if remaining() > 260:
        h_c, c5 = _nesterov_subgrad(best_h, 3000, t0 + remaining() - 255, k_active=8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 4: Top-k penalty descent
    if remaining() > 230:
        h_c, c5 = _topk_penalty_descent(best_h, 4000, t0 + remaining() - 225, k_top=12)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 5: Try diverse starting points
    if remaining() > 550:
        starts = _diverse_starts(n0, rng)
        for sh in starts:
            if remaining() < 510:
                break
            sh = _slsqp_opt(sh, 8000, 1500, t0 + remaining() - 509)
            c5 = _eval(sh)
            if c5 < best_c5:
                best_c5 = c5
                best_h = sh.copy()
            if remaining() < 510:
                break
            sh, c5 = _adam_opt(sh, 40000, 600, t0 + remaining() - 509, lr=0.012)
            if c5 < best_c5:
                best_c5 = c5
                best_h = sh.copy()
            if remaining() < 510:
                break
            sh, c5 = _nesterov_subgrad(sh, 1000, t0 + remaining() - 509, k_active=6)
            if c5 < best_c5:
                best_c5 = c5
                best_h = sh.copy()

    # Phase 6: Upsample to 2x and optimize
    if remaining() > 500:
        n1 = n0 * 2
        h_hi = _repair(_upsample(best_h, 2), n1)
        for alpha in [2000, 10000, 50000, 120000]:
            if remaining() < 440:
                break
            h_hi = _slsqp_opt(h_hi, alpha, 2000, t0 + remaining() - 439)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if remaining() > 420:
            h_hi, c5 = _adam_opt(best_h, 80000, 500, t0 + remaining() - 419, lr=0.008)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if remaining() > 380:
            h_hi, c5 = _nesterov_subgrad(best_h, 1500, t0 + remaining() - 379, k_active=10)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase 7: SA with pair swaps
    if remaining() > 160:
        h_cur = best_h.copy()
        nc = len(h_cur)
        c_cur = _eval(h_cur)
        t0v, t1v = 0.003, 2e-6
        n_iter = min(int((remaining() - 145) / 0.002), 50000)
        for i in range(n_iter):
            if remaining() < 145:
                break
            t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
            i1, i2 = rng.randint(nc), rng.randint(nc)
            if i1 == i2:
                continue
            d = rng.randn() * (0.002 + 0.012 * t)
            o1, o2 = h_cur[i1], h_cur[i2]
            h_cur[i1], h_cur[i2] = o1 + d, o2 - d
            h_cur = _repair(h_cur, nc)
            c_new = _eval(h_cur)
            if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
                c_cur = c_new
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = h_cur.copy()
            else:
                h_cur[i1], h_cur[i2] = o1, o2
                h_cur = _repair(h_cur, nc)

    # Phase 8: Final SLSQP + ADAM polish
    if remaining() > 70:
        h_c = _slsqp_opt(best_h, 300000, 4000, t0 + remaining() - 65)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
    if remaining() > 40:
        h_c, c5 = _adam_opt(best_h, 200000, 300, t0 + remaining() - 35, lr=0.005)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 9: Ultra-fine coordinate descent
    if remaining() > 18:
        h_cur = best_h.copy()
        nc = len(h_cur)
        c5c = _eval(h_cur)
        n_tries = min(int((remaining() - 14) / 0.0015), 80000)
        for _it in range(n_tries):
            if remaining() < 14:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_cur[i], h_cur[j] = ni, nj
            c5n = _eval(h_cur)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
