import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr(h):
    n = len(h)
    r = h.conj(); r[0] = 1.0 - r[0]
    return np.real(np.fft.ifft(np.fft.fft(h) * np.fft.fft(r)))

def _eval(h):
    return float(_corr(h)[0])

def _obj(x, alpha):
    C = _corr(x); m = C.max()
    theta = np.log(np.clip(2.0 / C.max(), 1e-6, 1e6))
    t = theta * (C - m)
    if t.max() > 30:
        t = t - t.max(); t = np.clip(t, -400, 30) + 30
    s = t - theta * C.max()
    s = np.clip(s - 1.0, -400, 40.0)
    denom = float(np.sum(np.exp(s - s.max())))
    m_eff = m + np.log(denom) / theta / np.exp(s.max() - s.max())
    return m_eff

def _grad(x, alpha):
    n = len(x); dx = 2.0 / n
    C = _corr(x); m = C.max()
    theta = np.log(np.clip(2.0 / C.max(), 1e-6, 1e6))
    t = theta * (C - m)
    if t.max() > 30:
        t = t - t.max(); t = np.clip(t, -400, 30.0) + 30.0
    dw = np.exp(t - np.clip(t.max(), 1.0, 30.0))
    tw = dw.sum()
    if tw < 1e-20:
        return np.zeros(n)
    dw *= dx / tw
    rp = np.zeros(n); rn = np.zeros(n)
    rp[1:n] = dw[1:n]; rn[1:] = dw[:0:-1]
    Fh = np.fft.fft(x); Fh1 = np.fft.fft(1.0 - x)
    K1 = np.real(np.fft.ifft(np.fft.fft(rp) * Fh))
    K2 = np.real(np.fft.ifft(np.fft.fft(rn) * Fh1))
    return K2 - rp[0] + K1

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h), n_new)
