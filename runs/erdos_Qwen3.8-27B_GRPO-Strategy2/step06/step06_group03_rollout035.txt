    s = h.sum()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):

    m = c.max()

    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')

            g[:n - sh] += wk * (1.0 - h[sh:])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])

    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return [x.sum() - 0.5 * n]
    def _cj(x): return [np.ones(n)]
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _worst_grad(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(c)
    s = (n - 1) - k_max
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _worst_descent(h, n_iter, step0, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    step = step0; stuck = 0; v = np.zeros(n)
    for _ in range(n_iter):
        g = _worst_grad(best)
        v = 0.7 * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            cand = _proj_sum(best - step * f * v, 0.5 * n)
            c_new = _eval(cand)
            if c_new < best_c - 1e-15:
                best = cand; best_c = c_new
                step = min(step * 1.12, 0.4)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.55; stuck += 1
            if stuck > 70:
                best = _repair(best + 0.015 * rng.randn(n), n)
                best_c = _eval(best)
                step = step0; stuck = 0; v = np.zeros(n)
    return best, best_c

def _sa_search(h, n_tries, rng, t0, deadline):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    T = 2e-4
    for it in range(n_tries):
        if it % 3000 == 2999 and time.time() > deadline - 5:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = 4e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        delta = cn - c_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-16)):
            c_cur = cn
            if cn < best_c - 1e-12:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
        T *= 0.9997
    return best, best_c

def _perturb_reopt(h, rng, alpha):
    n = len(h)
    h_p = h + 0.008 * rng.randn(n)
    h_p = _repair(h_p, n)
    h_p = _slsqp(h_p, alpha, 800, 1e-14)
    return h_p
