from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(v, target):
    n = len(v)
    a, b = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (a + b) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            a = mid
        else:
            b = mid
    return np.clip(v - (a + b) * 0.5, 0.0, 1.0)

def _grad_lse(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    ac = alpha * (c - m)
    ac = np.clip(ac, -500, 500)
    w = np.exp(ac)
    wsum = w.sum()
    if wsum < 1e-300:
        return c.copy(), np.zeros(n)
    w /= wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return c, g

def _pgd(h, alpha, n_iter, step0, t_end):

    h = h.copy()
    cur = _eval(h)
    step = step0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c, g = _grad_lse(h, alpha)
        # Backtracking line search
        accepted = False
        for _ in range(20):

    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

    s = h.sum()

from scipy.optimize import linprog

def _lp_step(h0, n, top_m=5):
    c = np.correlate(h0, 1.0 - h0, mode='full')
    idx = np.argsort(c)[-top_m:]
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t → c_lp = [0, ..., 0, 1]
    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0
    
    # Inequality constraints: g_k · h - t ≤ c[k]_0 - g_k · h0
    # Build A_ub, b_ub
    A_ub_rows = []
    b_ub_vals = []
    
    for ki in idx:
        g = _grad_c_k(h0, ki, n)  # ∂c[k]/∂h
        # g · h - t ≤ c[k]_0 - g · h0
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(c[ki] - g @ h0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 ≤ h_i ≤ 1, t free (or t ≥ 0)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return h0.copy()

        else:

def _grad_correlate(h, w):
    """Compute sum_k w[k] * dc[k]/dh"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

    step = step0

        wsum = w.sum()
        if wsum < 1e-300:

        w /= wsum

                cur = c_new
                step = min(step * 1.08, 2.0)
                accepted = True
                break
            step *= 0.5
        if not accepted and step < 1e-10:
            break
    return h, cur

            g[s:] -= h0[:n - s] * dx

            g[:n - sp] -= h0[sp:] * dx

            h_new = _project_sum(h - step * m, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur - 1e-15:
                h = h_new
                cur = c_new
                step = min(step * 1.05, 5.0)
                stall = 0
                accepted = True
                break
            step *= 0.5
        if not accepted:
            stall += 1
            if stall > 50:
                break
    return h, cur

def _lp_direction(h0, n, top_m):
    c = np.correlate(h0, 1.0 - h0, mode='full')
    idx = np.argsort(c)[-top_m:]
    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0
    A_rows = []
    b_vals = []
    dx = 2.0 / n
    for ki in idx:
        s = ki - (n - 1)
        g = np.zeros(n)
        if s == 0:
            g = (1.0 - 2.0 * h0) * dx
        elif s > 0:
            g[:n - s] = (1.0 - h0[s:]) * dx
            g[n - s:] -= h0[n - s - s if n - s > s else 0:n - s] * dx if s < n else 0
            g[:n - s] += (1.0 - h0[s:]) * dx
            g[s:] -= h0[:n - s] * dx
        else:
            sp = -s
            g[sp:] += (1.0 - h0[:n - sp]) * dx
            g[:n - sp] -= h0[sp:] * dx
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(c[ki] - g @ h0)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return h0.copy()

def _lp_refine(h, n_iter, top_m, t_end):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        h_lp = _lp_direction(h, n, top_m)
        found = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = _project_sum(h + frac * (h_lp - h), n * 0.5)
            c_try = _eval(h_try)
            if c_try < cur - 1e-15:
                h = h_try
                cur = c_try
                found = True
                break
        if not found:
            break
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_single(h, ki, n):
    """Gradient of c[ki] w.r.t. h."""
    g = np.zeros(n)
    s = ki - (n - 1)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n - s] = 1.0 - h[s:]
        g[s:] = -h[:n - s]
    else:
        sp = -s
        g[sp:] = 1.0 - h[:n - sp]
        g[:n - sp] = -h[sp:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() or True:
        try:
            n0 = len(initial_h_values)
            h = _repair(np.array(initial_h_values, dtype=float), n0)
        except (NameError, Exception):
            n0 = 500
            rng = np.random.RandomState(seed)
            idx = rng.choice(n0, size=n0 // 2, replace=False)
            h = np.zeros(n0)
            h[idx] = 1.0

    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = len(best_h)

    # Phase 1: PGD with momentum, geometric alpha
    for alpha, ni, s0 in [(30, 4000, 1.0), (100, 3000, 0.5), (500, 3000, 0.3),
                          (2000, 2500, 0.15), (10000, 2000, 0.08), (50000, 1500, 0.04)]:
        if rem() < 70:
            break
        h_new, c5 = _pgd_momentum(best_h, alpha, ni, s0, t0 + rem() - 30)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    # Phase 2: LP cutting plane
    if rem() > 50:
        for tm in [3, 5, 10, 15, 25]:
            if rem() < 45:
                break
            h_new, c5 = _lp_refine(best_h, 300, tm, t0 + rem() - 25)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)

    # Phase 3: Alternative start (random binary) + optimize
    if rem() > 150:
        rng = np.random.RandomState(seed + 1)
        for trial in range(3):
            if rem() < 130:
                break
            hb = np.zeros(n0)
            idx = rng.choice(n0, size=n0 // 2, replace=False)
            hb[idx] = 1.0
            for alpha in [200, 2000, 20000]:
                if rem() < 110:
                    break
                hb, c5 = _pgd_momentum(hb, alpha, 1500, 0.3, t0 + rem() - 60)
            if rem() > 50:
                hb, c5 = _lp_refine(hb, 100, 10, t0 + rem() - 30)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hb.copy()
                best_n = len(hb)

    # Phase 4: Multi-resolution
    for n_try in [400, 600, 800]:
        if rem() < 120:
            break
        hr = _repair(_resample(best_h, n_try), n_try)
        for alpha in [500, 5000, 50000]:
            if rem() < 100:
                break
            hr, c5 = _pgd_momentum(hr, alpha, 1500, 0.2 / np.sqrt(alpha / 500), t0 + rem() - 50)
        if rem() > 40:
            hr, c5 = _lp_refine(hr, 100, 10, t0 + rem() - 20)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hr.copy()
            best_n = n_try

    # Phase 5: Final ultra-high alpha + LP
    if rem() > 25:
        for alpha in [200000, 1000000]:
            if rem() < 20:
                break
            h_new, c5 = _pgd_momentum(best_h, alpha, 1000, 0.01, t0 + rem() - 10)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = len(h_new)
    if rem() > 12:
        h_new, c5 = _lp_refine(best_h, 200, 3, t0 + rem() - 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            best_n = len(h_new)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
