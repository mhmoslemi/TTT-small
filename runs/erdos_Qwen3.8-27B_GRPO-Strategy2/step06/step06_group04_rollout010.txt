                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_pen(x, alpha, mu):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha*(c-m), -400, 400); s -= s.max()
    e = np.exp(s)
    f = m + np.log(1.0 + e.sum() - 1.0) / alpha
    pen = mu * (x.sum() - 0.5*n)**2
    return f + pen

def _obj_pen_grad(x, alpha, mu):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha*(c-m), -400, 400); s -= s.max()
    e = np.exp(s)
    den = 1.0 + e.sum() - 1.0
    w = e / den
    f = m + np.log(den) / alpha
    # gradient of smooth max part
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        sh = k - (n-1)
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n-sh])
            g[:n-sh] -= wk * x[sh:]
        else:
            p = -sh
            g[:n-p] += wk * (1.0 - x[p:])
            g[p:] -= wk * x[:n-p]
    # penalty gradient
    g += 2.0 * mu * (x.sum() - 0.5*n) * np.ones(n)
    return f, g

    m1 = np.zeros(n); v2 = np.zeros(n)
    best = x.copy(); best_c = _eval(x)
    for i in range(n_iters):
        if time.time() > deadline: break
        t += 1
        fr = i / max(n_iters-1, 1)
        alpha = a0 * (a1/a0)**fr
        lr = lr0 * (lr1/lr0)**fr
        if t % 800 == 0:
            x += 0.01 * rng.randn(n)
        f, g = _obj_pen_grad(x, alpha, mu)
        b1, b2, eps = 0.9, 0.999, 1e-8
        m1 = b1*m1 + (1-b1)*g
        v2 = b2*v2 + (1-b1) if False else b2*v2 + (1-b2)*g*g
        m1h = m1 / (1 - b1**t)
        v2h = v2 / (1 - b2**t)
        x = x - lr * m1h / (np.sqrt(v2h) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best = x.copy()
    return best, best_c

def _lbfgs(x0, alpha, mu, maxiter, deadline):

        f, g = _obj_pen_grad(x, alpha, mu)
        return g

                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18,
                   'gtol': 1e-12, 'maxfun': maxiter*3})
    h = _repair(res.x, n)
    return h, _eval(h)

        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        if x[i] < x[j]:  # try to move mass from j to i
            x[i] = 1.0; x[j] = 0.0
        else:
            x[i] = 0.0; x[j] = 1.0
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new <= c_cur or rng.random() < np.exp(-(c_new-c_cur)/max(T,1e-10)):
            c_cur = c_new
            if c_new < best_c: best_c = c_new; best_x = x.copy()
        else:
            # revert
            x[i], x[j] = 1.0-x[i], 1.0-x[j]
            x = _repair(x, n)
        T *= 0.99998
    return best_x, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h
