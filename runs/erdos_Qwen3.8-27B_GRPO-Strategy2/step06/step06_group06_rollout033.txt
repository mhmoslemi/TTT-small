import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_and_weights(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s_max = s.max()
    e = np.exp(s - s_max)
    w = e / e.sum()
    smax = (m + s_max / alpha + np.log(e.sum()) / alpha)
    return smax, w, c, m

def _smax_obj(h, alpha, mu):
    n = len(h)
    smax, w, c, m = _smax_and_weights(h, alpha)
    penalty = mu * (h.sum() - n * 0.5) ** 2
    return (smax + penalty) * (2.0 / n)

def _smax_grad(h, alpha, mu):
    n = len(h)
    smax, w, c, m = _smax_and_weights(h, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    # Add penalty gradient
    g += 2.0 * mu * (h.sum() - n * 0.5) * np.ones(n)
    g *= dx
    return g

def _lbfgs(h, alpha, mu, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 3})
        if res.x is not None and np.all(np.abs(res.x - h) < 2):
            h_new = _repair(res.x, n)
            if _eval(h_new) < _eval(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-30:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax_w(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    e = np.exp(s - sm)
    w = e / e.sum()
    smax = m + sm / alpha + np.log(e.sum()) / alpha
    return smax, w

def _smax_obj(h, alpha, mu):
    n = len(h)
    smax, w = _smax_w(h, alpha)
    pen = mu * (h.sum() - 0.5 * n) ** 2
    return (smax + pen) * (2.0 / n)

def _smax_grad(h, alpha, mu):
    n = len(h)
    smax, w = _smax_w(h, alpha)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    if mu > 0:
        g += 2.0 * mu * (h.sum() - 0.5 * n)
    g *= 2.0 / n
    return g

def _lbfgs_opt(h, alpha, mu, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    f0 = _smax_obj(h, alpha, mu)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 2, 'iprint': -1})
        hx = res.x
        if np.all(hx >= -0.5) and np.all(hx <= 1.5):
            hx = _repair(hx, n)
            if _eval(hx) < f0 * n / 2 + 1e-6:
                return hx
    except Exception:
        pass
    return _repair(h, n)

def _adam_proj(h, alpha, n_steps, t_end, lr):
    n = len(h)
    h_cur = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    f_prev = _smax_obj(h_cur, alpha, 0.0)
    for t in range(1, n_steps + 1):
        if time.time() > t_end:
            break
        g = _smax_grad(h_cur, alpha, 0.0)
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        step = lr / (1 + t * 0.001)
        accepted = False
        for _bt in range(6):
            h_new = _proj(h_cur - step * m_hat / (np.sqrt(v_hat) + eps), 0.5 * n)
            f_new = _smax_obj(h_new, alpha, 0.0)
            if f_new <= f_prev + 1e-13:
                h_cur = h_new
                f_prev = f_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr = max(lr * 0.5, 1e-8)
    return h_cur

def _topk_descent(h, n_rounds, t_end, k=8):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:k]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= 2.0 / n
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - step * g, 0.5 * n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h_cur = h_new
                c5c = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5c

def _block_cd(h, n_tries, t_end, rng, bs=3):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    for i in range(n_tries):
        if time.time() > t_end:
            break
        idx = rng.randint(n - bs + 1)
        old = h_cur[idx:idx + bs].copy()
        d = 0.002 * rng.randn(bs)
        h_cur[idx:idx + bs] = np.clip(old + d, 0, 1)
        h_cur = _repair(h_cur, n)
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-14:
            c5c = c5n
        else:
            h_cur = _repair(h, n)
            h_cur = h.copy()
            c5c = _eval(h_cur)
            h_cur = old_h = h.copy()
            # restore
            h_cur = h.copy()
    return h_cur, c5c

def _coord_fine(h, n_tries, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    for i in range(n_tries):
        if time.time() > t_end:
            break
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = 3e-5 * rng.randn()
        o1, o2 = h_cur[i1], h_cur[i2]
        h_cur[i1], h_cur[i2] = o1 + d, o2 - d
        if h_cur[i1] < 0 or h_cur[i1] > 1 or h_cur[i2] < 0 or h_cur[i2] > 1:
            h_cur[i1], h_cur[i2] = o1, o2
            continue
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-15:
            c5c = c5n
        else:
            h_cur[i1], h_cur[i2] = o1, o2
    return h_cur, c5c

def _pair_sa(h, n_iter, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    c_cur = _eval(h_cur)
    best_h = h_cur.copy()
    best_c = c_cur
    t_lo, t_hi = 0.003, 2e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t_lo * (t_hi / t_lo) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.002 + 0.015 * t)
        o1, o2 = h_cur[i1], h_cur[i2]
        h_cur[i1] = np.clip(o1 + d, 0, 1)
        h_cur[i2] = np.clip(o2 - d, 0, 1)
        h_cur = _repair(h_cur, n)
        c_new = _eval(h_cur)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
        else:
            h_cur[i1], h_cur[i2] = o1, o2
            h_cur = _repair(h_cur, n)
    return best_h, best_c
