        h = np.clip(h, 0.0, 1.0)

def _grad_max(h, M=3):
    n = len(h)
    c = _corr(h)
    center = n - 1
    tmp = c.copy()
    tmp[center] = -np.inf
    top = np.argsort(tmp)[-M:]
    g = np.zeros(n)
    for ki in top:
        if ki > center:
            k = ki - center
            # b_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            j = np.arange(n)
            mask1 = j < n - k  # j appears as h[i], i=j
            mask2 = j >= k      # j appears as h[i+k], j-k=i
            g[mask1] += (1.0 - h[mask1 + k])
            g[mask2] -= h[mask2 - k]
        elif ki < center:
            kk = center - ki
            # c = sum_{i=0}^{n-1-kk} h[i+kk](1-h[i])
            j = np.arange(n)
            mask1 = j >= kk      # j appears as h[i+kk], i=j-kk
            mask2 = j < n - kk   # j appears as h[i], i=j
            g[mask1] += (1.0 - h[mask1 - kk])
            g[mask2] -= h[mask2 + kk]
    return g
