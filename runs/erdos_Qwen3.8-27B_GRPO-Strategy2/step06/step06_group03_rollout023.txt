def _sgrad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            idx1 = np.arange(n - sh)
            g[idx1] += wk * (1.0 - h[idx1 + sh])
            if sh > 0:
                idx2 = np.arange(sh, n)
                g[idx2] -= wk * h[idx2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _con_fun(x):
    return x.sum() - 0.5 * len(x)

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _top_grad(h, n, top_m):
    c = _corr(h)
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _subgrad(h, n_iter, step0, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    cur = best.copy(); cur_c = best_c
    v = np.zeros(n)
    mu = 0.82; step = step0
    stuck = 0
    top_ks = [3, 5, 7, 10, 5, 3, 8, 4, 6, 12]
    T_start = 0.008; T_end = 0.0003
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(n_iter, 1))
        tk = top_ks[it % len(top_ks)]
        g = _top_grad(cur, n, tk)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.6, 0.35, 0.15, 0.08]:
            h_new = _proj(cur - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            accept = c_new < cur_c or (c_new > cur_c and rng.rand() < np.exp(-(c_new - cur_c) / max(T, 1e-12)))
            if accept:
                cur = h_new; cur_c = c_new
                improved = True
                if c_new < best_c:
                    best = h_new.copy(); best_c = c_new
                    step = min(step * 1.1, 0.35)
                    stuck = 0
                break
        if not improved:
            step *= 0.55; stuck += 1
            if stuck > 45:
                cur = _repair(cur + 0.025 * rng.randn(n), n)
                cur_c = _eval(cur)
                if cur_c < best_c:
                    best = cur.copy(); best_c = cur_c
                step = step0; stuck = 0
                v = np.zeros(n)
    if cur_c < best_c:
        best = cur.copy(); best_c = cur_c
    return best, best_c

def _block_fine(h, n_tries, rng):
    n = len(h); h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        bs = int(rng.choice([2, 3, 4, 5]))
        i0 = rng.randint(0, max(1, n - bs))
        indices = list(range(i0, min(n, i0 + bs)))
        orig = [h[i] for i in indices]
        delta = 0.008 * rng.randn(bs)
        new_vals = [max(0.0, min(1.0, orig[i] + delta[i])) for i in range(bs)]
        sum_diff = sum(orig) - sum(new_vals)
        candidates = [j for j in range(n) if j not in set(indices)]
        j = candidates[rng.randint(len(candidates))]
        if 0 <= h[j] + sum_diff <= 1:
            for ii, idx in enumerate(indices):
                h[idx] = new_vals[ii]
            h[j] += sum_diff
            cn = _eval(h)
            if cn < c_cur:
                c_cur = cn
                if cn < best_c:
                    best_c = cn; best = h.copy()
            else:
                for ii, idx in enumerate(indices):
                    h[idx] = orig[ii]
                h[j] -= sum_diff
    return best, best_c

def _pair_fine(h, n_tries, rng, step_size):
    n = len(h); h = h.copy()
    c_cur = _eval(h)
    best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = step_size * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i] -= d; h[j] += d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c = _eval(best_h); best_n = n0

    # Phase 1: SLSQP with increasing alpha
    alphas1 = [200, 800, 3000, 10000, 40000, 150000]
    for alpha in alphas1:
        if time.time() - t0 > 110:
            break
        hc = _slsqp(best_h, alpha, 2500, 1e-15)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 2: Subgradient with simulated annealing
    if time.time() - t0 < budget_s - 280:
        rem = budget_s - 280 - (time.time() - t0)
        n_it = min(6000, int(rem / 0.03))
        best_h, best_c = _subgrad(best_h, n_it, 0.012, rng)
        best_n = len(best_h)

    # Phase 3: SLSQP medium-high alpha
    for alpha in [30000, 120000, 500000]:
        if time.time() - t0 > budget_s - 220:
            break
        hc = _slsqp(best_h, alpha, 2500, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 4: Second subgradient pass (finer steps)
    if time.time() - t0 < budget_s - 160:
        rem = budget_s - 160 - (time.time() - t0)
        n_it = min(3500, int(rem / 0.03))
        best_h, best_c = _subgrad(best_h, n_it, 0.007, rng)
        best_n = len(best_h)

    # Phase 5: Block fine search
    if time.time() - t0 < budget_s - 80:
        rem = budget_s - 80 - (time.time() - t0)
        n_tr = min(int(rem / 0.003), 40000)
        best_h, best_c = _block_fine(best_h, n_tr, rng)
        best_n = len(best_h)

    # Phase 6: SLSQP very high alpha
    if time.time() - t0 < budget_s - 45:
        hc = _slsqp(best_h, 1000000, 2500, 1e-17)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc; best_n = len(hc)

    # Phase 7: Pair fine search with multiple scales
    if time.time() - t0 < budget_s - 10:
        for ss in [3e-4, 1e-4, 4e-5, 1e-5]:
            if time.time() - t0 > budget_s - 5:
                break
            n_tr = min(20000, int((budget_s - 5 - (time.time() - t0)) / 0.003))
            best_h, best_c = _pair_fine(best_h, n_tr, rng, ss)
            best_n = len(best_h)

    # Phase 8: Final SLSQP polish
    if time.time() - t0 < budget_s - 3:
        hc = _slsqp(best_h, 300000, 1500, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc; best_n = len(hc)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
