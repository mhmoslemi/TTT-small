                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _adam_opt(h, alphas, n_iter):

    h = h.copy().astype(float)
    m_arr = np.zeros(n); v_arr = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_c = _eval(h)
    lr = 0.1
    for t in range(1, n_iter + 1):
        alpha = alphas[min((t * len(alphas)) // n_iter, len(alphas) - 1)]
        g = _sgrad(h, alpha)
        gn = np.linalg.norm(g)
        if gn > 1e-18:
            g = g / gn * min(gn, 1.0)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        mh = m_arr / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        h_new = _proj(h - step, n / 2.0)
        c_new = _eval(h_new)
        if c_new < best_c:
            h = h_new
            best_c = c_new
        else:
            lr = max(lr * 0.9, 0.001)
    return h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            i1 = np.arange(n - sh)

            if sh > 0:
