Let me think about the paired flip search more carefully. For a binary h with n/2 ones:
- Flipping h[i] from 1 to 0 and h[j] from 0 to 1
- The change in c[k] for lag k:
  - Before: h[i] contributes +1*(1-h[...]) terms, h[j] contributes 0
  - After: h[i] contributes 0, h[j] contributes +1*(1-h[...]) terms
  
  Δc[k] = -(contribution of h[i] at position i) + (contribution of h[j] at position j)

For c[k] = Σ_m x[m] * (1-x[m+(n-1-k)]):
- Position i contributes: (1-x[i+(n-1-k)]) if i+(n-1-k) is in [0,n-1]
- Position j contributes: (1-x[j+(n-1-k)]) if j+(n-1-k) is in [0,n-1]

But also, x[i] and x[j] appear as the (1-x[...]) factor:
- (1-x[m]) factor at position m: x[m-(n-1-k)] * (-1) for the contribution to c[k]

So the total change when flipping h[i]: 1→0 and h[j]: 0→1:
Δc[k] = -[1-x[i+lag]] + [1-x[j+lag]] - x[i-lag] + x[j-lag]
where lag = n-1-k, and terms with out-of-range indices are 0.

This is O(1) per k, so O(n) total per pair. With 15000 pairs per sweep, that's 15000*n = 4.7M operations per sweep. Very fast.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_opt(h0, a0, a1, n_iter, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        fr = i / max(n_iter - 1, 1)
        alpha = a0 * (a1 / a0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        if t % 500 == 0:
            x += 0.005 * rng.randn(n)
        g = _grad(x, alpha)
        m_arr = b1 * m_arr + (1 - b1) * g
        v_arr = b2 * v_arr + (1 - b2) * g * g
        m_hat = m_arr / (1 - b1 ** t)
        v_hat = v_arr / (1 - b2 ** t)
        x = _repair(x - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

def _binary_flip(h, n, deadline, max_iters=400000, seed=42):
    x = (h > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    rng = np.random.RandomState(seed)
    ones = np.where(x == 1.0)[0]
    zeros = np.where(x == 0.0)[0]
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        x[i] = 0.0; x[j] = 1.0
        c_new = _eval(x)
        if c_new < c_cur - 1e-14:
            c_cur = c_new
            ones = np.where(x == 1.0)[0]
            zeros = np.where(x == 0.0)[0]
            stall = 0
            if c_new < best_c:
                best_c = c_new; best_x = x.copy()
        else:
            x[i] = 1.0; x[j] = 0.0
            stall += 1
            if stall > 5000:
                break
    return best_x, best_c
