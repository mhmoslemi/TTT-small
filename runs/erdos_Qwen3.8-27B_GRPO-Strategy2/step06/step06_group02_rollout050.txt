def _clip_sum(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h
    h *= target / s
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    return h

def _proj(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""

    s = h.sum()
    if s < 1e-12:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h
