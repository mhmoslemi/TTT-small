        else:
            hi = mid

    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _proj(h, target):
    """Project onto {0<=x<=1, sum=target}"""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _corr(h):
    return np.correlate(h, 1.0-h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0/n)

def _grad_top(h, T):

    n = len(h)
    c = _corr(h)

import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s)))/alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12: continue
        sh = (n-1)-k
        wk = w[k]*dx
        if sh >= 0:
            m2 = n-sh
            g[sh:sh+m2] += wk*(1.0-h[:m2])
            g[:m2] -= wk*h[sh:sh+m2]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol, n):
    bnds = [(0.0, 1.0)]*n
    def cf(x): return x.sum() - 0.5*n
    def cj(x): return np.ones(n)
    cons = [{'type':'eq','fun':cf,'jac':cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter, 'ftol':ftol})
        return _proj(res.x, 0.5*n)
    except:
        return _proj(h, 0.5*n)

def _top_grad_step(h, T, step, n):
    """Take a projected gradient step using top T shifts"""
    c = np.correlate(h, 1.0-h, mode='full')
    idx = np.argpartition(c, -T)[-T:]
    dx = 2.0/n
    g = np.zeros(n)
    for k in idx:
        sh = (n-1)-k
        if sh >= 0:
            m = n-sh
            g[sh:sh+m] += dx*(1.0-h[:m])
            g[:m] -= dx*h[sh:sh+m]
        else:
            sp = -sh
            g[sp:] += dx*(1.0-h[:n-sp])
            g[:n-sp] -= dx*h[sp:]
    return _proj(h - step*g, 0.5*n)

def _momentum_opt(h, n_iter, step0, T, rng, n):

    best = h.copy(); best_c = _eval(best)
    y = best.copy(); v = np.zeros(n)
    step = step0
    for it in range(n_iter):
        c = np.correlate(best, 1.0-best, mode='full')
        idx = np.argpartition(c, -T)[-T:]
        dx = 2.0/n
        g = np.zeros(n)
        for k in idx:
            sh = (n-1)-k
            if sh >= 0:
                m = n-sh
                g[sh:sh+m] += dx*(1.0-best[:m])
                g[:m] -= dx*best[sh:sh+m]
            else:
                sp = -sh
                g[sp:] += dx*(1.0-best[:n-sp])
                g[:n-sp] -= best[sp:]
        v = 0.85*v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj(best - step*f*v, 0.5*n)
            cn = _eval(h_new)
            if cn < best_c - 1e-14:
                # Nesterov: y = best, then best = h_new
                y = h_new.copy()
                best = h_new; best_c = cn
                step = min(step*1.02, 0.15)
                improved = True
                break
        if not improved:

            if step < 1e-5:

                best_c = _eval(best)
                v = np.zeros(n); step = step0
        if it % 50 == 0:
            y = best.copy(); v = np.zeros(n)
    return best, best_c

def _line_search_pair(h, i, j, n_rng, rng, n):
    """Try to improve by adjusting h[i] and h[j] along a line"""
    best = h.copy(); best_c = _eval(h)
    for d in np.linspace(-0.05, 0.05, n_rng):
        if d == 0: continue
        ni, nj = h[i]+d, h[j]-d
        if 0<=ni<=1 and 0<=nj<=1:

            cn = _eval(h)
            if cn < best_c:
                best_c = cn; best = h.copy()
            h[i], h[j] = ni-d, nj+d
    return best, best_c
