from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _lbfgs_optimize(h0, alpha, maxiter=500, maxcor=20):
    n = len(h0)
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s - s.max()
        val = m + np.log(np.sum(np.exp(sm)) + 1e-300) / alpha
        penalty = 1e6 * (x.sum() - n/2.0)**2
        return val + penalty
    def grad(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-14: continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0: g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        g += 2e6 * (x.sum() - n/2.0)
        return g
    bnds = [(0,1)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'maxcor': maxcor, 'ftol': 1e-20})
    return _repair(res.x, n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def g(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        gd = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] < 1e-14: continue
            sh = (n-1) - k
            wk = w[k] * dx
            if sh >= 0:
                gd[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0: gd[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                gd[sp:] += wk * (1.0 - x[:n-sp])
                gd[:n-sp] -= wk * x[sp:]
        return gd
    def cons_eq(x): return x.sum() - n/2.0
    def cons_jac(x): return np.ones(n)
    bnds = [(0,1)]*n
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bnds,
                       constraints=[{'type':'eq','fun':cons_eq,'jac':cons_jac}],
                       options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except:
        return h

def _subgrad_descent(h, n_steps, c5_target):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5c = _eval(h)
    h_prev = h.copy()
    c_prev = None
    for step in range(n_steps):
        c = _corr(h)
        ki = int(np.argmax(c))
        sh = (n-1) - ki
        grad = np.zeros(n)
        if sh >= 0:
            grad[:n-sh] += dx * (1.0 - h[sh:])
            if sh > 0: grad[sh:] -= dx * h[:n-sh]
        else:
            sp = -sh
            grad[sp:] += dx * (1.0 - h[:n-sp])
            grad[:n-sp] -= dx * h[sp:]
        # Step size (BB or fixed)
        if c_prev is not None:
            y = h - h_prev
            s_mag = np.dot(y, y)
            g_mag = np.dot(grad, y)
            if s_mag > 1e-16 and g_mag > 1e-16:
                eta = s_mag / g_mag
            else:
                eta = 0.01
        else:
            eta = 0.01
        eta = np.clip(eta, 0.001, 0.1)
        h_new = _repair(h - eta * grad, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            h, c5c, h_prev = h_new, c5n, h
            c_prev = c
        else:
            h_prev = h
        if c5c <= c5_target: break
    return h, c5c

def _binary_segment_search(h, n_iters, rng):
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, n)
    c5c = _eval(h)
    for _ in range(n_iters):
        # Try segment reversal
        a = rng.randint(n)
        b = a + rng.randint(1, min(20, n - a))
        b = min(b, n)
        h_new = h.copy()
        h_new[a:b] = h_new[a:b][::-1]
        c5n = _eval(h_new)
        if c5n < c5c - 1e-14:
            h, c5c = h_new, c5n
            continue
        # Try block move
        a = rng.randint(n)
        ln = rng.randint(1, 8)
        if a + ln > n: continue
        b = rng.randint(0, n - ln)
        if abs(a - b) <= ln: continue
        h_new = h.copy()
        block = h_new[a:a+ln].copy()
        if b > a:
            h_new[a:b] = np.concatenate([h_new[a+ln:b], block])
        else:
            h_new[b:ln] = np.concatenate([block, h_new[b:a]])
            # This is wrong for overlapping, let me use a different approach
        # Simpler: just do element swaps
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        c5n = _eval(h_new)
        if c5n < c5c - 1e-14:
            h, c5c = h_new, c5n
    return h, c5c

def _coord_descent(h, n_rounds, rng, scales):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_rounds):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j: continue
            d = scale * rng.randn()
            h[i] += d
            h[j] -= d
            h_new = _repair(h, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                c5c = c5n
                h = h_new
            else:
                h[i] -= d
                h
