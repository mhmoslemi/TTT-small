    n = len(h)

    order = np.argsort(w)[::-1][:top_k]
    for k in order:
        g += w[k] * _grad_lag(h, k, dx)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(x, k, dx)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_ls(h, n, max_iter, rng):
    h = h.copy()
    c5 = _eval(h)
    best_c5 = c5
    best_h = h.copy()
    stuck = 0
    for it in range(max_iter):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        if (h[i] > 0.5) != (h[j] > 0.5):
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n; stuck = 0
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
            else:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
                stuck += 1
        if stuck > 3000:
            break
    return best_h, best_c5

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pairwise_cd(h, nc, rng, scale, n_iter):
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(nc); j = rng.randint(nc)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _fourier_init(K, n, rng, amp):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-amp, amp); b = rng.uniform(-amp, amp)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)

def _focused_descent(h, nc, max_rounds):
    c5c = _eval(h)
    dx = 2.0 / nc
    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        n_top = min(5, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        step = 0.04
        improved = False
        for _ in range(20):
            h_new = _repair(h - step * g, nc)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new; c5c = c5n; improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, c5c
