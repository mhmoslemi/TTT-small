        h_f = _slsqp_run(best_h, 10000000, 2000, 1e-22)
        c5f = _eval(h_f)
        if c5f < best_c5: best_c5, best_h = c5f, h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
