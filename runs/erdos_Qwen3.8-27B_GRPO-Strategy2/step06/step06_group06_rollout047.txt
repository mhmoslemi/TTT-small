import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj_box_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    s_max = s.max()
    val = m + (s_max + np.log(np.sum(np.exp(s - s_max)))) / alpha
    val += mu * (x.sum() - n * 0.5) ** 2
    return val * (2.0 / n)

    # Phase 1: L-BFGS-B progressive alpha
    alpha_mu_list = [(50, 10), (200, 20), (1000, 50), (5000, 200), (20000, 500), (100000, 1000)]
    for alpha, mu in alpha_mu_list:
        if remaining() < 150:
            break
        h_cand = _bfgs_opt(best_h, alpha, mu, 3000, t0 + remaining() - 149)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 2: Subgradient descent on top lags
    if remaining() > 200:
        h_cand, c5_cand = _subgrad_descent(best_h, 500, 15, t0 + remaining() - 199, rng)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 3: Binary push + re-optimize
    if remaining() > 250:
        h_bin = _binary_push(best_h, n)
        for alpha, mu in [(3000, 200), (15000, 500), (60000, 2000)]:
            if remaining() < 200:
                break
            h_cand = _bfgs_opt(h_bin, alpha, mu, 2500, t0 + remaining() - 199)
            c5 = _eval(h_cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()

    # Phase 4: Multiple perturbations + re-optimize
    if remaining() > 180:
        for _r in range(8):
            if remaining() < 140:
                break
            scale = 0.05 + 0.1 * _r
            h_rst = best_h + scale * rng.randn(n)
            h_rst = _repair(h_rst, n)
            h_cand = _bfgs_opt(h_rst, 5000, 200, 2000, t0 + remaining() - 139)
            c5 = _eval(h_cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()

    # Phase 5: High-alpha L-BFGS-B polish
    if remaining() > 100:
        h_cand = _bfgs_opt(best_h, 200000, 2000, 5000, t0 + remaining() - 95)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 6: Subgradient refinement
    if remaining() > 80:
        h_cand, c5_cand = _subgrad_descent(best_h, 400, 20, t0 + remaining() - 75, rng)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 7: SA pair swaps
    if remaining() > 60:
        n_sa = min(int((remaining() - 50) / 0.001), 80000)
        h_sa, c5_sa = _sa_pairs(best_h.copy(), t0 + remaining() - 50, n_sa, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 8: Final L-BFGS-B with very high alpha
    if remaining() > 40:
        h_cand = _bfgs_opt(best_h, 500000, 5000, 5000, t0 + remaining() - 35)
        c5 = _eval(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()

    # Phase 9: Ultra-fine coordinate descent
    if remaining() > 20:
        h_cd, c5_cd = _coord_descent(best_h.copy(), 60000, t0 + remaining() - 15, rng, 0.002)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Tiny coordinate adjustments
    if remaining() > 10:
        h_cd, c5_cd = _coord_descent(best_h.copy(), 30000, t0 + remaining() - 8, rng, 0.0005)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
