    best_h = h.copy()
    best_c = c5c
    for _it in range(n_iter):
        if el_fn() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _multi_swap_search(h, n_iter, rng, el_fn, deadline, scale):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_iter):
        if el_fn() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 10)) % n
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _soft_obj(h, alpha):
    n = len(h)
    c = _corl(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_quick(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_soft_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP at low alpha (30s)
    if _el() < budget_s - 200:
        h_s = _slsqp_quick(best_h, 1000, 500)
        c_s = _eval(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

        if _el() < budget_s - 150:
            h_s = _slsqp_quick(best_h, 10000, 500)
            c_s = _eval(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 2: Multi-scale (downsample, optimize, upsample)
    if _el() < budget_s - 300:
        for factor in [4, 3, 2]:
            if _el() > budget_s - 200:
                break
            n_low = n0 // factor
            h_low = _downsample(best_h, factor)
            h_low = _repair(h_low, n_low)
            h_low, c_low = _focus_descent(h_low, 2000, 0.005, 10, 5, _el, budget_s - 100)
            h_hi = _repair(_upsample(h_low, factor), n0)
            h_hi, c_hi = _focus_descent(h_hi, 1000, 0.003, 10, 5, _el, budget_s - 100)
            if c_hi < best_c5:
                best_c5 = c_hi
                best_h = h_hi.copy()

    # Phase 3: Aggressive focus descent at full resolution
    if _el() < budget_s - 200:
        h_c, c_c = _focus_descent(best_h, 5000, 0.004, 8, 5, _el, budget_s - 100)
        if c_c < best_c5:
            best_c5 = c_c
            best_h = h_c.copy()

    # Phase 4: Second SLSQP pass at higher alpha
    if _el() < budget_s - 150:
        h_s = _slsqp_quick(best_h, 50000, 400)
        c_s = _eval(h_s)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s.copy()

    # Phase 5: More focus descent with tighter parameter
    if _el() < budget_s - 100:
        h_c, c_c = _focus_descent(best_h, 3000, 0.002, 5, 3, _el, budget_s - 50)
        if c_c < best_c5:
            best_c5 = c_c
            best_h = h_c.copy()

    # Phase 6: Swap-based local search (medium scale)
    if _el() < budget_s - 30:
        n_tries = min(300000, int((budget_s - 30 - _el()) / 0.0015))
        best_h, best_c5 = _swap_search(best_h, n_tries, rng, _el, budget_s - 15, 0.002)
        best_n = len(best_h)

    # Phase 7: Multi-point swap search
    if _el() < budget_s - 15:
        n_tries = min(200000, int((budget_s - 15 - _el()) / 0.002))
        best_h, best_c5 = _multi_swap_search(best_h, n_tries, rng, _el, budget_s - 5, 0.001)
        best_n = len(best_h)

    # Phase 8: Ultra-fine 2-point polish
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        n_pol = min(500000, int((budget_s - 3 - _el()) / 0.001))
        for _it in range(n_pol):
            if _el() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 3)) % nc
            d = 3e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                        best_n = nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
