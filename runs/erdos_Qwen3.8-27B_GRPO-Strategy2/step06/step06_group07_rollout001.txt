import numpy as np
from scipy.optimize import minimize, fmin
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_sum(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
    return np.clip(h, 0.0, 1.0)

def _obj_ls(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad_ls(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])

                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _pgd_period(h_p, T, n, alphas, max_iters, lr_init, seed):
    m = n // T
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    momentum = np.zeros(T)
    beta = 0.82
    for alpha in alphas:
        iters = max(50, max_iters // len(alphas))
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            mm = c.max()
            s = np.clip(alpha * (c - mm), -500, 500)
            w = np.exp(s - s.max())
            w /= w.sum()
            dx = 2.0 / n
            g_full = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-14:
                    continue
                sh = k - (n - 1)
                wk = w[k] * dx
                if sh >= 0:
                    g_full[sh:] += wk * (1.0 - h_full[:n - sh])
                    if sh > 0:
                        g_full[:n - sh] -= wk * h_full[sh:]
                else:
                    sp = -sh
                    g_full[sp:] += wk * (1.0 - h_full[:n - sp])

            # revert

            # Just keep old cur
            pass

def _resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _slsqp_periodic(h_p, T, m, alphas, maxiter):
    bnds = [(0.0, 1.0)] * T
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    for alpha in alphas:
        try:
            res = minimize(_obj_ls, best, jac=_grad_ls, args=(alpha,),
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = _repair(res.x, T)
            c5 = _eval(np.tile(h_new, m))
            if c5 < best_c:
                best_c = c5
                best = h_new.copy()
        except Exception:
            pass
    return best, best_c
