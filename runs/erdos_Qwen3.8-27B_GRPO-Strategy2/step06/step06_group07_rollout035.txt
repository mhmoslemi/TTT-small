    n = T * m
    h = np.tile(p, m)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = corr.max()

    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n

    # d(corr[k])/dh[i] is tridiagonal-like

        return x.sum() - tgt
    def cjac(x):
        return np.ones(T)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:

                       args=(T, m, alpha), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, T)

        return p.copy()

            if c5a < best_c5:
                best_c5 = c5a
                best_h = np.tile(p_alt, m_alt).copy()

    # Phase 4: SLSQP at full resolution
    if elapsed() < budget_s * 0.80:
        n_f = len(best_h)
        for alpha in [20000, 200000]:
            if elapsed() > budget_s * 0.82:
                break
            h_f = best_h.copy()
            bnds_f = [(0.0, 1.0)] * n_f
            tgt_f = n_f / 2.0
            def cf(x):
                return x.sum() - tgt_f
            def jf(x):
                return np.ones(n_f)
            cons_f = [{'type': 'eq', 'fun': cf, 'jac': jf}]
            try:
                res = minimize(_obj, h_f, method='SLSQP', bounds=bnds_f,
                              constraints=cons_f, args=(n_f, 1, alpha),
                              options={'maxiter': 1000, 'ftol': 1e-16})
                h_f = _repair(res.x, n_f)
                c5f = _eval_full(h_f)
                if c5f < best_c5:
                    best_c5 = c5f
                    best_h = h_f.copy()
            except Exception:
                pass

    # Phase 5: Coordinate descent polish (multiple scales)
    if elapsed() < budget_s * 0.92:
        h_cd = best_h.copy()
        for scale in [0.01, 0.003, 0.001, 0.0003]:
            if elapsed() > budget_s * 0.95:
                break
            n_iter_cd = 8000 if scale > 0.001 else 15000
            h_cd, c5_cd = _coord_hill(h_cd, n_iter_cd, scale, seed + int(1/scale))
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 6: Final SLSQP with very high alpha on best pattern
    if elapsed() < budget_s - 15:
        n_b = len(best_h)
        T_b, m_b = (T, m) if n_b == T * m else (n_b, 1)
        p_b = best_h.reshape(m_b, T_b).mean(axis=0) if m_b > 1 else best_h.copy()
        p_b = _repair(p_b, T_b)
        for alpha in [10000000, 50000000]:
            if elapsed() > budget_s - 8:
                break
            p_b = _slsqp_pattern(p_b, T_b, m_b, alpha, 8000, 1e-20)
            c5b = _eval_full(np.tile(p_b, m_b))
            if c5b < best_c5:
                best_c5 = c5b
                best_h = np.tile(p_b, m_b).copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval_full(best_h)
    return (best_h, best_c5, len(best_h))
