            w[idx] = np.exp(10.0 * (c[idx] - cmin) / (cmax - cmin))
        else:
            w[idx] = 1.0
        w /= w.sum()
        g = _grad_weighted(h, w, dx)
        lr = lr0 * (0.97 ** (it // 200))
        noise = 0.003 * (0.995 ** it) * rng.randn(n)
        h_new = _repair(h - lr * g + noise, n)
        c_new = _eval(h_new)
        if c_new < cur - 1e-14:
            h = h_new
            cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        elif rng.rand() < 0.01:
            h = h_new
            cur = c_new
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Fast momentum subgradient
    if rem() > 70:
        h1, c1 = _proj_subgrad(best_h, 20000, 8, 0.04, 0.93, 0.85, t0 + rem() - 25)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: SLSQP with increasing alpha
    if rem() > 90:
        for alpha in [30000, 100000, 300000]:
            if rem() < 80:
                break
            h2 = _slsqp_lse(best_h, alpha, 2500, 1e-18, t0 + rem() - 20)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 3: Multi-resolution with full pipeline
    if rem() > 250:
        for n_try in [200, 250, 300, 350, 400, 450]:
            if rem() < 180:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            hr, c5 = _proj_subgrad(hr, 10000, 6, 0.06, 0.93, 0.85, t0 + rem() - 80)
            if rem() > 50:
                hr = _slsqp_lse(hr, 150000, 2000, 1e-17, t0 + rem() - 30)
            c5 = _eval(hr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hr.copy()
            if rem() > 30:
                hr2, c5b = _pair_cd(hr, len(hr), 8000, [0.005, 0.0015], t0 + rem() - 10, 42 + n_try)
                if c5b < best_c5:
                    best_c5 = c5b
                    best_h = hr2.copy()

    # Phase 4: Fourier smoothing perturbation
    if rem() > 60:
        nc = len(best_h)
        for nk in [8, 14, 22, 35]:
            if rem() < 30:
                break
            hf = _fourier_smooth(best_h, nk)
            hf, c5 = _proj_subgrad(hf, 6000, 6, 0.05, 0.93, 0.85, t0 + rem() - 12)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hf.copy()

    # Phase 5: Annealed subgradient for escaping local minima
    if rem() > 50:
        h5, c5 = _anneal_subgrad(best_h, 8000, 5, 0.03, t0 + rem() - 15, seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5.copy()

    # Phase 6: Targeted subgradient with different top_m
    if rem() > 40:
        for m in [3, 10]:
            if rem() < 25:
                break
            h6, c5 = _proj_subgrad(best_h, 5000, m, 0.025, 0.94, 0.88, t0 + rem() - 8)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h6.copy()

    # Phase 7: Pairwise CD
    if rem() > 20:
        h7, c5 = _pair_cd(best_h, len(best_h), 20000, [0.004, 0.001, 0.0004], t0 + rem() - 8, seed + 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h7.copy()

    # Phase 8: Final SLSQP polish
    if rem() > 12:
        h8 = _slsqp_lse(best_h, 600000, 4000, 1e-19, t0 + rem() - 4)
        c5 = _eval(h8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h8.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
