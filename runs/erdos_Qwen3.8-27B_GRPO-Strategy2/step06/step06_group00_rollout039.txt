    """Gradient of correlation at specific lag k."""
    n = len(h)

    sh = (n - 1) - k
    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if abs(d) < 1e-12:
                    break
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smval(h, a):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / a

def _smgrad(h, a):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    act = np.where(w > 1e-10)[0]
    for k in act:
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _adam(h, a, steps, lr, dl):
    n = len(h)
    m1 = np.zeros(n); m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    for t in range(1, steps + 1):
        if time.time() > dl:
            break
        g = _smgrad(h, a)
        m1 = b1 * m1 + (1.0 - b1) * g
        m2 = b2 * m2 + (1.0 - b2) * g * g
        step = lr * (m1 / (1.0 - b1 ** t)) / (np.sqrt(m2 / (1.0 - b2 ** t)) + eps)
        h = _proj(h - step, n * 0.5)
    return _repair(h, n)

def _laggrad(h, k):
    n = len(h); dx = 2.0 / n; sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g * dx

def _actlag(h, nt, nr, dl):
    n = len(h)
    bc = _eval(h); bh = h.copy()
    for _ in range(nr):
        if time.time() > dl:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        tp = np.argsort(c)[-nt:]
        g = np.zeros(n)
        for k in tp:
            g += _laggrad(h, k)
        g /= float(nt)
        imp = False
        for st in [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            hn = _proj(h - st * g, n * 0.5)
            cv = _eval(hn)
            if cv < bc - 1e-15:
                h = hn; bc = cv; bh = h.copy(); imp = True; break
        if not imp:
            break
    return bh, bc

def _ls2(h, it, rng, dl):
    n = len(h); c0 = _eval(h); bh = h.copy(); bc = c0
    for _ in range(it):
        if time.time() > dl:
            break
        i = rng.randint(n); j = (i + 1 + rng.randint(0, 3)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c0 - 1e-19:
                c0 = cn
                if cn < bc:
                    bc = cn; bh = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _ls3(h, it, rng, dl):
    n = len(h); c0 = _eval(h); bh = h.copy(); bc = c0
    for _ in range(it):
        if time.time() > dl:

        i = rng.randint(n)
        j = (i + rng.randint(2, 5)) % n
        k = (i + rng.randint(6, 10)) % n
        d1 = 1e-5 * rng.randn(); d2 = 1e-5 * rng.randn()
        ni = h[i] + d1; nj = h[j] + d2; nk = h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < c0 - 1e-18:
                c0 = cn
                if cn < bc:
                    bc = cn; bh = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return bh, bc

def _cf(x):
    return x.sum() - len(x) * 0.5

def _cj(x):
    return np.ones(len(x))

def _sObj(h, a):
    return _smval(h, a)

def _ds(h, f):
    mn = len(h) // f
    return h[:mn * f].reshape(mn, f).mean(axis=1).copy()

def _us(h, f):
    return np.repeat(h, f).copy()

    bh = h.copy(); bc = _eval(bh); bn = n0
