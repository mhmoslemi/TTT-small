import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_shift(h, k):
    """Gradient of corr[k] w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += dx * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx * h[:n-sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:n-sp])
        g[:n-sp] -= dx * h[sp:]
    return g

def _grad_worst(h):
    n = len(h)
    c = _corr(h)
    ki = int(np.argmax(c))
    return _grad_shift(h, ki)

def _logsumexp_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _logsumexp_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        g += w[k] * _grad_shift(h, k)
    return g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x): return _logsumexp_obj(x, alpha)
    def obj_grad(x): return _logsumexp_grad(x, alpha)
    def con_fun(x): return x.sum() - n * 0.5
    def con_jac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _proj_grad(h, n_iter, lr0, beta):
    """Projected gradient with momentum on the worst-shift objective"""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    c5 = _eval(h)
    for t in range(n_iter):
        g = _grad_worst(h)
        lr = lr0 / (1 + t * lr0 / 10)
        v = beta * v - lr * g
        h_new = h + v
        h = _repair(h_new, n)
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
        else:
            v *= 0.5  # reduce momentum if stuck
    return h, c5

    """Local search on binary: flip or swap"""
    n = len(h)
    h = (h > 0.5).astype(float)
    h = _repair(h, n)
    c5 = _eval(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0: break
        if rng.rand() < 0.5 and len(ones) > 1 and len(zeros) > 1:
            # Swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            h2 = _repair(h, n)
            c5n = _eval(h2)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

    return h

    n = len(h)

    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:n - sp])
        g[:n - sp] -= dx * h[sp:]
    return g
