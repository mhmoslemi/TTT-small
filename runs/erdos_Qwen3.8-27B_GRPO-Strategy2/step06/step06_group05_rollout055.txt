import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            if diff > 0:
                for idx in order:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                    if abs(diff) < 1e-12: break
            else:
                for idx in order:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                    if abs(diff) < 1e-12: break
    return h

def _c5(h):
    return float(np.max(np.correlate(h, 1-h, mode='full')) * 2.0 / len(h))

def _slsqp_polish(h, alpha, nit):
    n = len(h)
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    def obj(x):
        cc = np.correlate(x, 1-x, mode='full')
        mm = cc.max()
        s = np.clip(alpha*(cc-mm), -500, 500)
        s -= s.max()
        return mm + np.log(np.sum(np.exp(s))+1e-30)/alpha
    def grad(x):
        cc = np.correlate(x, 1-x, mode='full')
        mm = cc.max()
        s = np.clip(alpha*(cc-mm), -500, 500)
        w = np.exp(s-s.max()); w /= w.sum()
        dx = 2.0/n
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-14: continue
            sh = (n-1)-k; wk = w[k]*dx
            if sh >= 0:
                g[:n-sh] += wk*(1-x[sh:])
                if sh > 0: g[sh:] -= wk*x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk*(1-x[:n-sp])
                g[:n-sp] -= wk*x[sp:]
        return g
    def sef(x): return x.sum() - n*0.5
    def sej(x): return np.ones(n)
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':sef,'jac':sej}],
                       options={'maxiter':nit,'ftol':1e-19})
        return _repair(res.x, n)
    except:
        return _repair(h.copy(), n)

            jd = j - d  # j-d for each d

            mask_pos = (jdp >= 0) & (jdp < n)  # 0<=j+d<n
