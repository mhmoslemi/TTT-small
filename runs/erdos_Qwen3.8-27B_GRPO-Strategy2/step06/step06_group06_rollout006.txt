    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _smax_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _c_fun(x, n):
    return x.sum() - n * 0.5

def _c_jac(x, n):
    return np.ones_like(x)

def _slsqp_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac, 'args': (n,)}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if time.time() < t_end and res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _nesterov_grad_descent(h, alpha, n_steps, t_end):
    n = len(h)
    h_arr = h.copy()
    v = np.zeros(n)
    step = 0.005
    mu = 0.75
    f_prev = _smax_obj(h_arr, alpha)
    best_h = h_arr.copy()
    best_f = f_prev
    for _s in range(n_steps):
        if time.time() > t_end:
            break
        h_look = _proj(h_arr + mu * v, n * 0.5)
        g = _smax_grad(h_look, alpha)
        v = mu * v + g
        v_norm = np.linalg.norm(v)
        if v_norm > 1.0:
            v /= v_norm
        for _bt in range(10):
            h_new = _proj(h_arr - step * v, n * 0.5)
            f_new = _smax_obj(h_new, alpha)
            if f_new <= f_prev + 1e-13:
                h_arr = h_new
                f_prev = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h_arr.copy()
                step = min(step * 1.04, 0.05)
                break
            step *= 0.5
        else:
            step = max(step, 0.002)
            v *= 0.5
    return best_h

def _topk_descent(h, k_count, n_rounds, t_end):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    dx = 2.0 / n
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = _corr(h_cur)
        top_k = np.argsort(c)[::-1][:k_count]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _upsample_interp(h, factor):
    n = len(h)
    n_new = n * factor
    x_old = np.linspace(0.0, 2.0, n + 1)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old[:-1], np.concatenate([h, [h[-1]]]))
    return h_new

def _sa_optimize(h, t_end, n_iter, rng):
    n = len(h)
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    t0v, t1v = 0.006, 5e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.003 + 0.04 * t)
        old_i1, old_i2 = h[i1], h[i2]
        h[i1] = np.clip(old_i1 + d, 0, 1)
        h[i2] = np.clip(old_i2 - d, 0, 1)
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[i1], h[i2] = old_i1, old_i2
            h = _repair(h, n)
    return best_h, best_c

def _coord_descent(h, t_end, n_tries, rng):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return h_cur, c5c

def _perturb_restart(h, rng, scale):
    n = len(h)
    h_new = h + scale * rng.randn(n)
    return _repair(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP progressive alpha at base res
    for alpha in [400, 1500, 6000, 20000, 70000, 200000]:
        if _rem() < 220:
            break
        t_deadline = t0 + _rem() - 219
        h_c = _slsqp_opt(best_h, alpha, 4000, t_deadline)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 2: Nesterov projected gradient at base res
    if _rem() > 200:
        h_c = _nesterov_grad_descent(best_h, 50000, 2000, t0 + _rem() - 199)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 3: Top-k descent at base res
    if _rem() > 180:
        for kc in [3, 5, 8, 12]:
            if _rem() < 175:
                break
            h_c, c5_c = _topk_descent(best_h, kc, 500, t0 + _rem() - 174)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Phase 4: Upsample to 2x with interpolation
    if _rem() > 400:
        n1 = n0 * 2
        h_hi = _upsample_interp(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [400, 1500, 6000, 25000, 80000]:
            if _rem() < 350:
                break
            t_deadline = t0 + _rem() - 349
            h_hi = _slsqp_opt(h_hi, alpha, 3000, t_deadline)
        c5 = _eval(h_hi)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_hi.copy()
            best_n = n1
        if _rem() > 300:
            h_hi = _nesterov_grad_descent(best_h, 30000, 1500, t0 + _rem() - 299)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1
        if _rem() > 250:
            for kc in [3, 5, 8]:
                if _rem() < 245:
                    break
                h_c, c5_c = _topk_descent(best_h, kc, 400, t0 + _rem() - 244)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()
                    best_n = n1

    # Phase 5: Perturbation restarts + re-optimize
    if _rem() > 200:
        for _r in range(6):
            if _rem() < 190:
                break
            for scale in [0.01, 0.03, 0.05]:
                if _rem() < 185:
                    break
                h_try = _perturb_restart(best_h, rng, scale)
                h_try = _slsqp_opt(h_try, 30000, 2000, t0 + _rem() - 184)
                h_try, c5_try = _topk_descent(h_try, 5, 200, t0 + _rem() - 183)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()

    # Phase 6: Upsample to 4x if budget allows
    if _rem() > 350 and best_n == n0:
        n2 = n0 * 4
        h_hi2 = _upsample_interp(best_h, 4)
        h_hi2 = _repair(h_hi2, n2)
        for alpha in [500, 3000, 15000]:
            if _rem() < 300:
                break
            t_deadline = t0 + _rem() - 299
            h_hi2 = _slsqp_opt(h_hi2, alpha, 2000, t_deadline)
        c5 = _eval(h_hi2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_hi2.copy()
            best_n = n2
        if _rem() > 250:
            h_hi2 = _nesterov_grad_descent(best_h, 20000, 1000, t0 + _rem() - 249)
            c5 = _eval(h_hi2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi2.copy()
                best_n = n2

    # Phase 7: SA optimization
    if _rem() > 100:
        n_iter = min(int((_rem() - 90) * 300), 80000)
        h_sa, c5_sa = _sa_optimize(best_h, t0 + _rem() - 89, n_iter, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 8: Final polish - top-k + SLSQP
    if _rem() > 60:
        h_c, c5_c = _topk_descent(best_h, 5, 300, t0 + _rem() - 55)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
    if _rem() > 40:
        h_c = _slsqp_opt(best_h, 300000, 4000, t0 + _rem() - 35)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 9: Coordinate descent polish
    if _rem() > 15:
        n_tries = min(int((_rem() - 12) / 0.002), 120000)
        h_c, c5_c = _coord_descent(best_h, t0 + _rem() - 12, n_tries, rng)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Final: short top-k descent
    if _rem() > 8:
        h_c, c5_c = _topk_descent(best_h, 3, 100, t0 + _rem() - 7)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
