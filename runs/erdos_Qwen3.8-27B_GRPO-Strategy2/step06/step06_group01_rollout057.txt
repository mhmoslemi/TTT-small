    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()  # unscaled
    c5 = corr.max()
    for _pass in range(n_passes):
        for i in rng.permutation(n):
            hi = h[i]
            # Compute slope for each corr entry
            slopes = np.zeros(2*n - 1)
            for sh in range(-(n-1), n):
                s = n - 1 + sh
                if sh == 0:
                    continue  # handle separately
                sl = 0.0
                if 0 <= i + sh < n:
                    sl += (1.0 - h[i + sh])
                if 0 <= i - sh < n:
                    sl -= h[i - sh]
                slopes[s] = sl
            # For s=0 (index n-1): quadratic, handle separately
            
            dmin = -hi
            dmax = 1.0 - hi
            
            # Candidate deltas
            cands = [0.0, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.001, -0.001, dmin, dmax]
            cands = [c for c in cands if dmin - 1e-12 <= c <= dmax + 1e-12]
            
            best_c5 = c5
            best_d = 0.0
            for d in cands:
                if abs(d) < 1e-15:

        for i in rng.permutation(n):
            hi = h[i]
            dmin, dmax = -hi, 1.0 - hi
            slopes = np.zeros(full_n)
            sh_lo = max(-i, -n1)
            sh_hi = min(n1 - i, n1)
            if sh_lo <= sh_hi:
                shs = np.arange(sh_lo, sh_hi + 1)
                m = shs != 0
                if m.any():
                    sm = shs[m]
                    slopes[n1 + sm] += (1.0 - h[i + sm])
            sh2_lo = max(i - n1, -n1)
            sh2_hi = min(i, n1)
            if sh2_lo <= sh2_hi:
                shs2 = np.arange(sh2_lo, sh2_hi + 1)
                m2 = shs2 != 0
                if m2.any():
                    sm2 = shs2[m2]
                    slopes[n1 + sm2] -= h[i - sm2]
            cands = []
            for s in scales:
                cands.extend([s, -s])
            cands.extend([dmin, dmax])
            cands = [c for c in cands if dmin - 1e-12 <= c <= dmax + 1e-12 and abs(c) > 1e-14]
            best_c5v, best_d = c5, 0.0
            base0 = corr[n1]
            for d in cands:
                vals = corr + d * slopes
                vals[n1] = base0 + d * (1.0 - 2.0 * hi) - d * d
                mc = vals.max()
                if mc < best_c5v - 1e-14:
                    best_c5v, best_d = mc, d
            if best_d != 0.0:
                h[i] = hi + best_d
                corr += best_d * slopes
                corr[n1] = base0 + best_d * (1.0 - 2.0 * hi) - best_d * best_d
                c5 = best_c5v
    return h, c5 * dx

def _ls_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _ls_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    n1 = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = n1 - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _lbfgs(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: Coordinate descent (coarse -> fine scales)
    if tl() > 200:
        h, c5 = _coord_desc(best_h.copy(), best_n, 80, rng,
                            scales=[0.05, 0.02, 0.01, 0.005, 0.002])
        h = _repair(h, best_n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: L-BFGS-B global structure
    if tl() > 250:
        for alpha in [5000, 20000, 80000]:
            if tl() < 60:
                break
            h2 = _lbfgs(best_h.copy(), alpha, 300)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 3: Fine coordinate descent
    if tl() > 200:
        h, c5 = _coord_desc(best_h.copy(), best_n, 120, rng,
                            scales=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        h = _repair(h, best_n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: More L-BFGS at higher alpha
    if tl() > 200:
        for alpha in [150000, 400000, 1000000]:
            if tl() < 50:
                break
            h2 = _lbfgs(best_h.copy(), alpha, 400)
            h2, c5 = _coord_desc(h2, best_n, 60, rng,
                                 scales=[0.005, 0.002, 0.001, 0.0005, 0.0002])
            h2 = _repair(h2, best_n)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()

    # Phase 5: Ultra-fine coordinate descent
    if tl() > 150:
        h, c5 = _coord_desc(best_h.copy(), best_n, 200, rng,
                            scales=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        h = _repair(h, best_n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: Multi-resolution search
    if tl() > 200:
        for n_try in [310, 400, 500, 632, 750, 900]:
            if tl() < 60:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r, c5 = _coord_desc(h_r, n_try, 80, rng,
                                  scales=[0.02, 0.01, 0.005, 0.002, 0.001])
            if tl() < 30:
                break
            h_r = _lbfgs(h_r, 50000, 200)
            h_r, c5 = _coord_desc(h_r, n_try, 100, rng,
                                  scales=[0.005, 0.002, 0.001, 0.0005, 0.0002])
            h_r = _repair(h_r, n_try)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 7: Perturbation + re-optimize
    if tl() > 150:
        for _rep in range(4):
            if tl() < 50:
                break
            noise = rng.randn(best_n) * 0.008
            h_p = _repair(best_h + noise, best_n)
            h_p, c5 = _coord_desc(h_p, best_n, 100, rng,
                                  scales=[0.01, 0.005, 0.002, 0.001, 0.0005])
            h_p = _lbfgs(h_p, 100000, 250)
            h_p, c5 = _coord_desc(h_p, best_n, 80, rng,
                                  scales=[0.002, 0.001, 0.0005, 0.0002, 0.0001])
            h_p = _repair(h_p, best_n)
            c5 = _eval(h_p)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 8: Structured periodic initializations
    if tl() > 150:
        for period in [3, 4, 5, 6, 7, 8]:
            if tl() < 40:
                break
            h_p = np.zeros(best_n)
            for i in range(best_n):
                phase = (i * period) / best_n
                h_p[i] = 0.5 + 0.4 * np.cos(2.0 * np.pi * phase)
            h_p = _repair(h_p, best_n)
            h_p, c5 = _coord_desc(h_p, best_n, 80, rng,
                                  scales=[0.02, 0.01, 0.005, 0.002, 0.001])
            h_p = _lbfgs(h_p, 30000, 200)
            h_p, c5 = _coord_desc(h_p, best_n, 60, rng,
                                  scales=[0.002, 0.001, 0.0005, 0.0002])
            h_p = _repair(h_p, best_n)
            c5 = _eval(h_p)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 9: Final ultra-fine polish
    if tl() > 20:
        h_f, c5 = _coord_desc(best_h.copy(), best_n, 300, rng,
                              scales=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        h_f = _repair(h_f, best_n)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    # Final repair and eval
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
