def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):

        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k_idx, n):

        return x.sum() - n * 0.5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k_idx, n):
    dx = 2.0 / n
    shift = k_idx - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        j1 = np.arange(shift, n)
        g[j1] += (1.0 - h[j1 - shift])
        j2 = np.arange(n - shift)
        g[j2] -= h[j2 + shift]
    else:
        sp = -shift
        j1 = np.arange(n - sp)
        g[j1] += (1.0 - h[j1 + sp])
        j2 = np.arange(sp, n)
        g[j2] -= h[j2 - sp]
    return g * dx

def _topk_descent(h, topk, n_iters, el_fn, budget_check):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    step = 0.015
    stall = 0
    for it in range(n_iters):
        if budget_check():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[-topk:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, k, n)
        g /= topk
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for _bt in range(18):
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h = h_new
                c5 = c5n
                stall = 0
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                step = min(step * 1.04, 0.06)
                improved = True
                break
            step *= 0.55
        if not improved:
            stall += 1
            if stall % 200 == 0:
                h = _proj_to_sum(h + 1e-4 * np.random.randn(n), n * 0.5)
                c5 = _eval(h)
            step = 0.015
            if stall > 500:
                break
    return best_h, best_c5

def _coord_search(h, n_iters, rng, el_fn, budget_check):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    base_step = 2e-4
    for it in range(n_iters):
        if budget_check():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = base_step * (0.5 + rng.random()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-18:
                c5 = c5n
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                base_step = min(base_step * 1.05, 8e-4)
            else:
                h[i], h[j] = ni - d, nj + d
                base_step = max(base_step * 0.97, 2e-6)
    return best_h, best_c5

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    def _objf(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def _objg(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                g += w[k] * _grad_lag(x, k, n)
        return g
    try:
        res = minimize(_objf, h, jac=_objg, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
