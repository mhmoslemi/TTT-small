Let me also make the gradient computation faster. Instead of looping over all 2n-1 shifts, I can use the relationship between the correlation and the signal to compute the gradient more efficiently.

For c[k] = Σ_j h[j+d] * (1-h[j]) where d = k-(n-1):

∂c/∂h[i] = (1-h[i-d]) - h[i+d] (times dx, with appropriate boundary conditions)

So the full gradient of c (for all k) at index i is:
g[i] = Σ_d w[d] * [(1-h[i-d]) - h[i+d]] * dx

This can be computed as:
g[i] = Σ_d w[d]*(1-h[i-d]) * dx - Σ_d w[d]*h[i+d] * dx
     = Σ_d w[d] * dx - (Σ_d w[d]*h[i-d]) * dx - (Σ_d w[d]*h[i+d]) * dx
     = sum(w)*dx - conv(w, h)[i] * dx - conv(w_flipped, h)[i] * dx

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_c(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _soft_obj(h, alpha):
    c = _eval_c(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -600, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_weighted(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    nn2 = 2 * n - 1
    for k in range(nn2):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * dx * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * dx * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * dx * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * dx * h[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def objf(x):
        return _soft_obj(x, alpha)
    def gradf(x):
        c = _eval_c(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -600, 0)
        w = np.exp(s)
        w /= w.sum()
        return _grad_weighted(x, w)
    def ceq(x):
        return x.sum() - n * 0.5
    def jeq(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': jeq}]
    try:
        res = minimize(objf, h, jac=gradf, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _active_descent(h, topk, n_rounds, max_step, t0, budget_s):
    n = len(h)
    c5 = _eval(h)
    for r in range(n_rounds):
        if time.time() - t0 > budget_s - 15:
            break
        c = _eval_c(h)
        idx = np.argsort(c)[-topk:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / topk
        g = _grad_weighted(h, w)
        step = max_step
        improved = False
        for _ in range(10):
            h_new = _proj(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                improved = True
                break
            step *= 0.4
            if step < 1e-9:
                break
        if not improved:
            break
    return h, c5

def _perturb_restart(h, amp, freq, t0, budget_s):
    n = len(h)
    x = np.arange(n)
    h_p = h.copy() + amp * np.sin(2 * np.pi * freq * x / n)
    h_p = _proj(h_p, n)
    c5p = _eval(h_p)
    if time.time() - t0 > budget_s - 30:
        return h, _eval(h)
    h_d, c5d = _active_descent(h_p, 3, 100, 0.03, t0, budget_s)
    if c5d < c5p:
        return h_d, c5d
    return h, _eval(h)

def _coord_descent(h, n_passes, t0, budget_s, rng):
    n = len(h)
    c5 = _eval(h)
    for p in range(n_passes):
        if time.time() - t0 > budget_s - 20:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() - t0 > budget_s - 30:
                break
            j = (i + 1) % n
            for d in [1e-4, -1e-4, 5e-5, -5e-5, 2e-5, -2e-5]:
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5 - 1e-16:
                        c5 = c5n
                        break
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d
    return h, c5

def _local_2pt(h, n_iters, rng, t0, budget_s):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    for _ in range(n_iters):
        if time.time() - t0 > budget_s - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-17:
                c5 = c5n
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5

def _block_sa(h, n_steps, t0, budget_s, rng):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    t_cur = 0.005
    t_end = 0.0003
    for i in range(n_steps):
        if time.time() - t0 > budget_s - 15:
            break
        t_cur = 0.005 * (0.0003 / 0.005) ** (i / max(n_steps, 1))
        blen = 3 + int(rng.randint(0, 12))
        start = int(rng.randint(0, max(n - blen, 1)))
        noise = rng.randn(blen)
        scale = 0.005 + t_cur * 20
        h[start:start + blen] += scale * noise
        h_new = _proj(h, n)
        c5n = _eval(h_new)
        if c5n < c5 or rng.rand() < np.exp(-(c5n - c5) / max(t_cur, 1e-12)):
            h = h_new
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[start:start + blen] -= scale * noise
            h = _proj(h, n)
    return best_h, best_c5
