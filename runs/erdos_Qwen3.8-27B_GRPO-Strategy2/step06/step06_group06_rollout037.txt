```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smax_val(c, alpha, dx_factor):
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx_factor

def _smax_grad_full(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _lag_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g
```

```python
import numpy as np
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > target: lo = m
        else: hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _sm_f(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500., 500.)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _sm_g(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500., 500.)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]

    return g
```
