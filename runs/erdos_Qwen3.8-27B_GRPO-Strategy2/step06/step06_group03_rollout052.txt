import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_topk(h, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad(h, n_iter, step0, top_k, rng):
    n = len(h)
    best = h.copy(); bc = _eval(best)
    v = np.zeros(n); mu = 0.85; step = step0
    stuck = 0
    for i in range(n_iter):
        g = _grad_topk(best, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = mu * v + g / gn
        vn = np.linalg.norm(v)
        if vn < 1e-16:
            v = g / gn; vn = 1.0
        else:
            v /= vn
        improved = False
        for f in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_new = _proj_sum(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < bc - 1e-18:
                best = h_new; bc = c_new
                step = min(step * 1.12, 0.15)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.6; stuck += 1
            if stuck > 50:
                pert = _repair(best + 0.002 * rng.randn(n), n)
                best = pert; bc = _eval(best)
                step = step0; stuck = 0
                v = np.zeros(n)
    return best, bc

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        dxv = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dxv
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    idx_list = np.arange(n)

                        break
                    else:
