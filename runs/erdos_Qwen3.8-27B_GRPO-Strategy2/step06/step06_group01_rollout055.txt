def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _calc_sens(h, i, n):
    """Compute sensitivity vector a[k] for coordinate i."""
    ov_len = 2 * n - 1
    a = np.zeros(ov_len)

        if 0 <= ri < n:
            val += 1.0 - h[ri]
        li = i - v
        if 0 <= li < n:
            val -= h[li]
        a[k] = val
    return a

        if time.time() > deadline:
            break
        g, _ = _grad_top_lag(h, n_top)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        v = mu * v + (1.0 - mu) * g / gn
        # Line search
        step = lr
        found = False
        for _ in range(20):
            h_new = _repair(h - step * v * gn, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                lr = min(lr * 1.02, 0.5)
                found = True
                break
            step *= 0.5
        if lr < 1e-8:
            break
    return h, c5c

def _lsexp_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            if sp < n:
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, t0, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if time.time() < deadline - 5:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    # Start from parent
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP polish
    if time.time() < deadline - 200:
        h1 = _slsqp(best_h, 10000, 500, t0, deadline)
        c5 = _eval(h1)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Fast coordinate descent (the key improvement)
    if time.time() < deadline - 150:
        h2 = best_h.copy()
        h2, c5 = _ternary_cd(h2, best_n, 30, t0, deadline - 150)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: Momentum gradient
    if time.time() < deadline - 100:
        h3 = best_h.copy()
        h3, c5 = _momentum_grad(h3, 5, 200, t0, deadline - 100)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 4: Another round of coord descent
    if time.time() < deadline - 80:
        h4 = best_h.copy()
        h4, c5 = _ternary_cd(h4, best_n, 40, t0, deadline - 80)
        if c5 < best_c5:
            best_c5, best_h = c5, h4.copy()

    # Phase 5: SLSQP with higher alpha
    if time.time() < deadline - 80:
        h5 = _slsqp(best_h, 50000, 1000, t0, deadline)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()

    # Phase 6: Try smaller n values
    for n_try in [310, 200, 160, 130]:
        if time.time() > deadline - 40:
            break
        h_r = _repair(_resample(best_h, n_try), n_try)
        # Coordinate descent on small n
        h_r, c5 = _ternary_cd(h_r, n_try, 50, t0, deadline - 30)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_r.copy(), n_try
        # Quick gradient polish
        if time.time() < deadline - 20:
            h_r2 = best_h.copy()
            h_r2, c5 = _momentum_grad(h_r2, 4, 100, t0, deadline - 15)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r2.copy(), len(h_r2)

    # Phase 7: Multi-start perturbation
    if time.time() < deadline - 60:
        for _rep in range(4):
            if time.time() > deadline - 30:
                break
            noise = rng.randn(best_n) * 0.005
            h_p = _repair(best_h + noise, best_n)
            h_p, c5 = _ternary_cd(h_p, best_n, 20, t0, deadline - 25)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()

    # Phase 8: Final SLSQP
    if time.time() < deadline - 5:
        hf = _slsqp(best_h, 100000, 800, t0, deadline)
        c5 = _eval(hf)
        if c5 < best_c5:
            best_c5, best_h = c5, hf.copy()

    # Phase 9: Final coord descent
    if time.time() < deadline - 2:
        hf2 = best_h.copy()
        hf2, c5 = _ternary_cd(hf2, best_n, 15, t0, deadline)
        if c5 < best_c5:
            best_c5, best_h = c5, hf2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
