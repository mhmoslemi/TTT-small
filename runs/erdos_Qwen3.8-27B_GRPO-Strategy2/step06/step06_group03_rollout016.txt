    for i in range(len(idx)):
        k = idx[i]
        sh = (n - 1) - k
        wk = w[i] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _subgrad(h, n_iter, step0, top_k):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    mu = 0.88
    step = step0
    stuck = 0
    for _ in range(n_iter):
        g = _top_k_grad(best, top_k, 0.8 / best_c)
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj(best - step * f * v, 0.5 * n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-16:
                best = h_new
                best_c = c_new
                step = min(step * 1.12, 0.4)
                improved = True
                stuck = 0
                break
        if not improved:
            step *= 0.65
            stuck += 1
            if stuck > 100:
                best = _repair(best + 0.008 * np.random.randn(n), n)
                best_c = _eval(best)
                step = step0
                stuck = 0
                v = np.zeros(n)
    return best, best_c

def _coord_search(h, n_trials, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni, nj = best[i] + d, best[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        best[i], best[j] = ni, nj
        c_new = _eval(best)
        if c_new < best_c:
            best_c = c_new
        else:
            best[i], best[j] = ni - d, nj + d
    return best, best_c

def _block_opt(h, rng, n_blocks, n_trials_per_block):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    bs = max(4, n // n_blocks)
    starts = list(range(0, n, bs))
    for _ in range(n_trials_per_block):
        for s in starts:
            e = min(s + bs, n)
            for _t in range(3):
                pert = 0.02 * rng.randn(e - s)
                h_new = best.copy()
                h_new[s:e] = np.clip(h_new[s:e] + pert, 0, 1)
                h_new = _repair(h_new, n)
                c_new = _eval(h_new)
                if c_new < best_c:
                    best = h_new
                    best_c = c_new
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c = _eval(best_h)
    best_n = n0

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [300, 3000, 30000, 300000]:
        if _el() > 130:
            break
        hc = _slsqp(best_h, alpha, 2000, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 2: Subgradient with multiple top_k settings
    if _el() < budget_s - 350:
        n_it = min(3500, int((budget_s - 350 - _el()) / 0.028))
        for tk in [3, 5, 7]:
            if _el() > budget_s - 320:
                break
            hc, cc = _subgrad(best_h.copy(), n_it // 3 + 100, 0.018, tk)
            if cc < best_c:
                best_c = cc
                best_h = hc

    # Phase 3: Re-refine with SLSQP after subgradient
    for alpha in [50000, 500000, 2000000]:
        if _el() > budget_s - 220:
            break
        hc = _slsqp(best_h, alpha, 2500, 1e-17)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 4: Try doubled resolution
    if _el() < budget_s - 180:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [10000, 100000, 800000]:
            if _el() > budget_s - 120:
                break
            h_hi = _slsqp(h_hi, alpha, 1500, 1e-16)
            c = _eval(h_hi)
            if c < best_c:
                best_c = c
                best_h = h_hi
                best_n = n1
        if _el() < budget_s - 70:
            best_h, best_c = _subgrad(best_h, 1200, 0.01, 5)

    # Phase 5: Block-structured perturbation search
    if _el() < budget_s - 60:
        n_b = min(int((budget_s - 60 - _el()) / 0.5), 200)
        best_h, best_c = _block_opt(best_h, rng, 12, n_b)

    # Phase 6: Final SLSQP polish
    for alpha in [5000000, 1000000]:
        if _el() > budget_s - 25:
            break
        hc = _slsqp(best_h, alpha, 3000, 1e-18)
        c = _eval(hc)
        if c < best_c:
            best_c = c
            best_h = hc

    # Phase 7: Coordinate descent final polish
    if _el() < budget_s - 8:
        n_tr = min(int((budget_s - 8 - _el()) / 0.0035), 60000)
        best_h, best_c = _coord_search(best_h, n_tr, rng)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
