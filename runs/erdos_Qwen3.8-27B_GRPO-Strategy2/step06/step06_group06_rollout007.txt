```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _cvals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _lp_active(h, top_m, delta):
    n = len(h)
    c = _cvals(h)
    active = np.argsort(c)[::-1][:top_m]
    dx = 2.0 / n
    A = np.zeros((top_m, n))
    c_act = np.zeros(top_m)
    for idx, k in enumerate(active):
        sh = (n - 1) - k
        c_act[idx] = c[k]
        if sh >= 0:
            A[idx, :n - sh] += (1.0 - h[sh:])
            if sh > 0:
                A[idx, sh:] -= h[:n - sh]
        else:
            sp = -sh
            A[idx, sp:] += (1.0 - h[:n - sp])
            A[idx, :n - sp] -= h[sp:]
    A *= dx
    rhs = A @ h - c_act
    A_ineq = np.hstack([A, -np.ones((top_m, 1))])
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    lb = np.clip(h - delta, 0.0, 1.0)
    ub = np.clip(h + delta, 0.0, 1.0)
    bounds = list(zip(lb, ub))
    bounds.append((-1.0, 1.0))
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x[-1] < c.max() * dx * 0.5 + 1e-6:
            return _proj(res.x[:n], n * 0.5)
    except Exception:
        pass
    return h

def _slsqp_step(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    f0 = _smax(h, alpha)
    try:
        res = minimize(_smax, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': False})
        if res.fun < f0 - 1e-16 and time.time() < t_end:
            return _proj(res.x, n * 0.5)
    except Exception:
        pass
    return h

def _nesterov(h, alpha, n_steps, t_end):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    mu = 0.75
    step = 0.02
    f_prev = _smax(x, alpha)
    for s in range(n_steps):
        if time.time() > t_end:
            break
        g = _smax_grad(y, alpha)
        x_new = _proj(y - step * g, target)
        f_new = _smax(x_new, alpha)
        if f_new <= f_prev + 1e-15:
            y = x_new + mu * (x_new - x)
            y = _proj(y, target)
            x = x_new
            f_prev = f_new
            step = min(step * 1.03, 0.3)
        else:
            step *= 0.5
            x_new_chk = _proj(y - step * g, target)
            f_new2 = _smax(x_new_chk, alpha)
            if f_new2 <= f_prev + 1e-15:
                x = x_new_chk
                f_prev = f_new2
            else:
                if step < 0.001:
                    break
    return x

def _coord_desc(h, t_end, rng, n_tries_max=200000):
    n = len(h)
    h_cur = h.copy()
    c5c = _eval(h_cur)
    n_tries = min(n_tries_max, int((t_end - time.time()) / 0.002))
    for _ in range(max(n_tries, 0)):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 2e-5 * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _eval(h_cur)
        if c5n < c5c - 1e-17:
            c5c = c5n
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
    return h_cur, c5c

def _multilag(h, n_rounds, t_end):

    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = _cvals(h_cur)
        top_k = np.argsort(c)[::-1][:6]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                if sh > 0:
                    g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _perturb_restart(h, t_end, rng, budget_each):

        h_p = _slsqp_step(h_p, 50000, 800, t_end - budget_each)
        c5p = _eval(h_p)
        if c5p < best_c:
            best_c = c5p
            best_h = h_p.copy()
        h_p2, c5p2 = _multilag(h_p, 100, t_end - budget_each)
        if c5p2 < best_c:
            best_c = c5p2
            best_h = h_p2.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_deadline = t0 + budget_s

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    n0 = len(initial_h_values)
    h = _proj(np.array(initial_h_values, dtype=float).copy(), n0 * 0.5)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Progressive SLSQP
    for alpha in [500, 3000, 15000, 80000, 300000, 1200000]:
        if remaining() < 200:
            break
        h_c = _slsqp_step(best_h, alpha, 3000, t0 + remaining() - 195)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 2: LP-based active set refinement
    if remaining() > 180:
        for delta in [0.03, 0.015, 0.008, 0.004, 0.002]:
            if remaining() < 175:
                break
            for _ in range(8):
                if remaining() < 172:
                    break
                h_lp = _lp_active(best_h, 8, delta)
                c5lp = _eval(h_lp)
                if c5lp < best_c5:
                    best_c5 = c5lp
                    best_h = h_lp
                else:
                    break

    # Phase 3: Nesterov momentum
    if remaining() > 160:
        for alpha in [50000, 200000, 800000]:
            if remaining() < 155:
                break
            n_steps = min(int((remaining() - 152) * 0.3 / 0.005), 3000)
            h_n = _nesterov(best_h, alpha, max(n_steps, 200), t0 + remaining() - 150)
            c5n = _eval(h_n)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_n.copy()

    # Phase 4: Multi-lag descent
    if remaining() > 140:
        h_ml, c5_ml = _multilag(best_h, 500, t0 + remaining() - 135)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 5: LP again with tighter trust region
    if remaining() > 120:
        for delta in [0.003, 0.001, 0.0005, 0.0002]:
            if remaining() < 116:
                break
            h_lp = _lp_active(best_h, 10, delta)
            c5lp = _eval(h_lp)
            if c5lp < best_c5:
                best_c5 = c5lp
                best_h = h_lp.copy()
            h_ml2, c5_ml2 = _multilag(best_h, 200, t0 + remaining() - 114)
            if c5_ml2 < best_c5:
                best_c5 = c5_ml2
                best_h = h_ml2.copy()

    # Phase 6: SLSQP polish at very high alpha
    if remaining() > 100:
        for alpha in [500000, 2000000, 8000000]:
            if remaining() < 96:
                break
            h_c = _slsqp_step(best_h, alpha, 5000, t0 + remaining() - 94)
            c5 = _eval(h_c)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c.copy()

    # Phase 7: Perturbation restarts
    if remaining() > 80:
        h_pr, c5_pr = _perturb_restart(best_h, t0 + remaining() - 30, rng,
                                       max(int((remaining() - 30) / 5), 1))
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr.copy()

    # Phase 8: Ultra-fine coordinate descent
    if remaining() > 20:
        h_cd, c5_cd = _coord_desc(best_h, t0 + remaining() - 10, rng,
                                  n_tries_max=int((remaining() - 10) / 0.002))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: Final LP + multilag polish
    if remaining() > 15:
        h_lp = _lp_active(best_h, 12, 0.0005)
        c5lp = _eval(h_lp)
        if c5lp < best_c5:
            best_c5 = c5lp
            best_h = h_lp.copy()
        h_ml3, c5_ml3 = _multilag(best_h, 100, t0 + remaining() - 5)
        if c5_ml3 < best_c5:
            best_c5 = c5_ml3
            best_h = h_ml3.copy()

    # Final repair and evaluation
    best_h = _proj(best_h, n0 * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
```
