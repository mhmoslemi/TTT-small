        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _polish_2pt(h, n_iters, rng, el_limit):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_iters):
        if _elapsed() > el_limit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global budget_global, t_start
    t_start = time.time()
    budget_global = budget_s
    rng = np.random.RandomState(seed)
    el_lim = budget_s - 10

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Low-res multi-start (n=112)
    n_low = 112
    h_low = _downsample(h, n0 // n_low)
    h_low = _repair(h_low, n_low)
    candidates = [h_low.copy()]
    for _ in range(5):
        hp = h_low.copy() + 0.08 * rng.randn(n_low)
        candidates.append(_proj(hp, n_low * 0.5))
    # Structured starts
    hb = np.concatenate([np.ones(n_low // 4), np.zeros(n_low // 2), np.ones(n_low - n_low // 4)])
    candidates.append(_proj(hb, n_low * 0.5))
    hm = np.concatenate([np.ones(2 * n_low // 5), np.zeros(n_low - 2 * n_low // 5)])
    candidates.append(_proj(hm, n_low * 0.5))

    for alpha in [500, 2000, 10000, 50000]:
        if _elapsed() > 100:
            break
        for ci in range(len(candidates)):
            if _elapsed() > 140:
                break
            candidates[ci] = _nesterov(candidates[ci], alpha, 150, 0.008, 140)
            c5 = _eval(candidates[ci])
            if c5 < best_c5:
                best_c5 = c5
                best_h = _repair(_upsample(candidates[ci], n0 // n_low), n0)

    # Phase 2: SLSQP at full res, escalating alpha
    for alpha in [1500, 8000, 40000, 150000, 500000]:
        if _elapsed() > budget_s - 350:
            break
        h_c = _slsqp(best_h, alpha, 2000, budget_s - 340)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 3: Nesterov at full res
    if _elapsed() < budget_s - 250:
        h_nr = _nesterov(best_h, 80000, 400, 0.003, budget_s - 240)
        c5 = _eval(h_nr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_nr.copy()

    # Phase 4: Top-lag descent
    for nl in [3, 5, 2, 4]:
        if _elapsed() > budget_s - 150:
            break
        h_ml, c5_ml = _take_top_lags(best_h.copy(), nl, 200, budget_s - 140)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 5: Additional SLSQP + Nesterov cycle
    if _elapsed() < budget_s - 120:
        h_c = _slsqp(best_h, 800000, 2000, budget_s - 110)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
        if _elapsed() < budget_s - 80:
            h_nr = _nesterov(best_h, 200000, 300, 0.002, budget_s - 70)
            c5 = _eval(h_nr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_nr.copy()

    # Phase 6: 3-point local search
    if _elapsed() < budget_s - 5:
        n_iters = min(int((budget_s - 5 - _elapsed()) * 300), 60000)
        best_h, best_c5 = _local_search(best_h, n_iters, rng, budget_s - 5)

    # Phase 7: Ultra-fine 2-point polish
    if _elapsed() < budget_s - 2:
        n_pol = min(int((budget_s - 2 - _elapsed()) * 800), 40000)
        best_h, best_c5 = _polish_2pt(best_h, n_pol, rng, budget_s - 2)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
