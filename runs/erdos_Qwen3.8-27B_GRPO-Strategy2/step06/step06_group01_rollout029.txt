import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    d = min(diff, 1.0 - h[idx])
                    h[idx] += d
                    diff -= d
                else:
                    d = min(-diff, h[idx])
                    h[idx] -= d
                    diff += d
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    x = np.linspace(0, 2.0, n, endpoint=False)

        h += theta[2*(j-1)] * np.cos(j * np.pi * x) + theta[2*(j-1)+1] * np.sin(j * np.pi * x)
    return h

        theta[2*(j-1)+1] = (2.0 / n) * np.dot(h - 0.5, np.sin(j * np.pi * x))

def _grad_lag(h, kk, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - kk
    if sh >= 0 and sh < n:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    elif sh < 0:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _top_lag_descent(h, nc, max_rounds):

    for r in range(max_rounds):
        ov = np.correlate(h, 1.0 - h, mode='full')
        n_top = min(4, len(ov))
        top_idx = np.argsort(ov)[-n_top:]
        g = np.zeros(nc)
        for kk in top_idx:
            g += _grad_lag(h, kk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        best_c = c5c
        best_h = h
        step = 0.06
        for _ in range(22):
            h_new = _repair(h - step * g, nc)
            c5n = _eval(h_new)
            if c5n < best_c - 1e-15:
                best_c = c5n
                best_h = h_new
            else:
                step *= 0.5
        if best_c >= c5c - 1e-15:
            g1 = _grad_lag(h, top_idx[-1], dx)
            gn1 = np.linalg.norm(g1)
            if gn1 < 1e-14:

            g1 /= gn1
            step = 0.04

                    best_h = h_new
                else:
                    step *= 0.5
            if best_c >= c5c - 1e-15:
                break
        h = best_h
        c5c = best_c
    return h, c5c
