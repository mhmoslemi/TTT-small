Great. Now for the 1D optimization:
Given the current correlations c_k and the gradient A_k for coordinate i, I want to minimize:

This is a convex function of δ (max of linear functions). Its minimum can be found by:
- Bisection on the subgradient
- Or: since it's 1D convex, golden section search

The subgradient at δ is A_k for k in argmax(c_k + δ*A_k).
- If all subgradients > 0: f is increasing, decrease δ
- If all subgradients < 0: f is decreasing, increase δ  
- If some ≥ 0 and some ≤ 0: δ is optimal

```python
def _coord_cd(h, nc, n_sweeps=100):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = nc
    dx = 2.0 / n
    h = h.copy()
    # Initial correlations
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c5 = c.max()
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Compute gradient A for coordinate i
            A = np.zeros(2 * n - 1)
            old = h[i]
            
            # s >= 0: k = n-1-s for s = 0, ..., n-1
            for s in range(n):
                k = n - 1 - s
                val = 0.0
                if i < n - s:
                    val += (1.0 - h[i + s])
                if i >= s:
                    val -= h[i - s]
                A[k] = val * dx
            
            # s < 0: s = -p, p = 1, ..., n-1; k = n-1+p
            for p in range(1, n):
                k = n - 1 + p
                val = 0.0
                if i >= p:
                    val += (1.0 - h[i - p])
                if i < n - p:
                    val -= h[i + p]
                A[k] = val * dx
            
            # Now minimize max_k(c[k] + t * A[k]) over t in [0, 1]
            # (where t is the new value of h[i], not the change)
            # Actually, let me work with the change δ = t - old
            # New correlation: c[k] + (t - old) * A[k]
            # Minimize max_k(c[k] + (t - old) * A[k]) over t in [0,1]
            
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                vals = c + (mid - old) * A
                m = vals.max()
                # Subgradient: A[k] for k achieving max
                mask = vals >= m - 1e-14
                sg_min = A[mask].min()
                sg_max = A[mask].max()
                if sg_min > 1e-15:
                    hi = mid
                elif sg_max < -1e-15:
                    lo = mid
                else:
                    break
            
            new_i = (lo + hi) / 2.0
            if abs(new_i - old) > 1e-10:
                delta = new_i - old
                c += delta * A
                h[i] = new_i
                new_c5 = c.max()
                if new_c5 < c5 - 1e-14:
                    c5 = new_c5
                    improved = True
        
        if not improved:
            break
    
    return h, c5
```

```python
def _coord_cd_fast(h, nc, n_sweeps, t_budget):
    n = nc
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    c5 = c.max()
    s_all = np.arange(n)
    
    for sweep in range(n_sweeps):
        if time.time() - t_budget > 0:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Compute A vector
            # Left part: s = 0..n-1, k = n-1-s
            valid1 = s_all < (n - i)
            lhs = np.zeros(n)
            lhs[valid1] = 1.0 - h[i + s_all[valid1]]
            valid2 = s_all <= i
            rhs = np.zeros(n)
            rhs[valid2] = h[i - s_all[valid2]]
            A_left = (lhs - rhs) * dx
            
            # Right part: p = 1..n-1, k = n-1+p
            p_all = s_all[1:]  # 1, 2, ..., n-1
            valid3 = i >= p_all
            lhs2 = np.zeros(n-1)
            lhs2[valid3] = 1.0 - h[i - p_all[valid3]]
            valid4 = i + p_all < n
            rhs2 = np.zeros(n-1)
            rhs2[valid4] = h[i + p_all[valid4]]
            A_right = (lhs2 - rhs2) * dx
            
            A = np.concatenate([A_left[::-1], A_right])
            
            # 1D convex minimization: min over t∈[0,1] of max_k(c[k] + (t-old)*A[k])
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0

                mask = vals >= m - 1e-13

                    hi = mid

                c += delta * A
                h[i] = new_i
                nc5 = c.max()
                if nc5 < c5 - 1e-14:
                    c5 = nc5
                    improved = True
        if not improved:
            break
    return h, c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol, t_budget):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
```

```python
def _compute_A_i(h, i, n, dx):
    """Compute gradient of all correlations w.r.t. h[i]."""
    A = np.zeros(2 * n - 1)
    # Left part: s = 0, 1, ..., n-1 → k = n-1-s
    # A[n-1-s] = dx * [(1-h[i+s]) * 1_{i < n-s} - h[i-s] * 1_{i >= s}]
    for s in range(n):
        val = 0.0
        if i + s < n:  # i < n-s
            val += 1.0 - h[i + s]
        if i >= s:  # i >= s, i-s >= 0
            val -= h[i - s]
        A[n - 1 - s] = val * dx
    
    # Right part: p = 1, 2, ..., n-1 → k = n-1+p
    # A[n-1+p] = dx * [(1-h[i-p]) * 1_{i >= p} - h[i+p] * 1_{i + p < n}]
    for p in range(1, n):
        val = 0.0
        if i >= p:
            val += 1.0 - h[i - p]
        if i + p < n:
            val -= h[i + p]
        A[n - 1 + p] = val * dx
    
    return A
```
