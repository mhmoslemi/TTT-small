import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = _proj(h * target / s, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _smax_weights(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

def _smax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    w = _smax_weights(c, alpha)
    g = np.zeros(n)
    # Zero lag at index n-1
    g += w[n-1] * (1.0 - 2.0 * h)
    for m in range(1, n):
        wp = w[n-1+m] if n-1+m < 2*n-1 else 0.0
        wn = w[n-1-m] if n-1-m >= 0 else 0.0
        if wp > 1e-15:
            g[m:] += wp * (1.0 - h[:n-m])
            g[:n-m] -= wp * h[m:]
        if wn > 1e-15:
            g[:n-m] += wn * (1.0 - h[m:])
            g[m:] -= wn * h[:n-m]
    g *= dx
    return g

def _pgrad_ho(h, alpha, n_steps, t_end):

    h_cur = h.copy()
    f_prev = _smax_obj(h_cur, alpha)
    v = np.zeros(n)
    mu = 0.7
    step = 0.01
    for s in range(n_steps):
        if time.time() > t_end:
            break
        g = _smax_grad(h_cur, alpha)
        v = mu * v + (1.0 - mu) * g
        # BB step
        if s > 0:

            dg = g - g_prev
            den = np.dot(dg, dg) + 1e-15
            step_bb = max(0.01, min(0.3, abs(np.dot(dh, dg) / den)))
