def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    diff -= a
                else:

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    active = w > 1e-10
    idx = np.where(active)[0]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        g += w[k] * _grad_shift(sh, h, n, dx)
    return g

def _nesterov(h, alpha, steps, step0, t0, budget):
    n = len(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    f_cur = _eval(h)
    for _ in range(steps):
        if time.time() - t0 > budget:
            break
        g = _smoothed_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f_cur:
            h, f_cur = h_new, f_new
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f_cur = _eval(h)
    return h

def _subgrad_step(h, t0, budget, n_lags=1):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    tol = 1e-10
    active = np.where(c >= m - tol)[0]
    if len(active) == 0:
        active = np.where(c >= m - 0.01)[0]
    n_eff = min(n_lags, len(active))
    top = active[np.argsort(c[active])[-n_eff:]]
    g = np.zeros(n)
    for k in top:
        sh = (n - 1) - k
        g += _grad_shift(sh, h, n, dx)
    g /= len(top)
    for lr in [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
        h_new = _proj_to_sum(h - lr * g, n * 0.5)
        if _eval(h_new) < _eval(h):
            return h_new
    return h

def _binary_push(h, threshold=0.5):
    h2 = np.where(h > threshold, 1.0, 0.0)
    return _repair(h2, len(h))

def _local_search(h, n_iter, rng, t0, budget):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() - t0 > budget:
            break
        strategy = rng.randint(3)
        if strategy == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = 1e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        elif strategy == 1:
            i = rng.randint(n)
            j = (i + rng.randint(3, 12)) % n
            k = (i + rng.randint(13, 24)) % n
            d1 = 3e-5 * rng.randn()
            d2 = 3e-5 * rng.randn()
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h[i], h[j], h[k] = ni, nj, nk
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
        else:
            i0 = rng.randint(n - 4)
            block = h[i0:i0 + 4].copy()
            delta = 2e-5 * rng.randn(4)
            delta -= delta.mean()
            nb = block + delta
            if np.all(nb >= 0) and np.all(nb <= 1):
                h[i0:i0 + 4] = nb
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i0:i0 + 4] = block
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    def _ok(marg):
        return _el() < budget_s - marg

    # Phase 1: Subgradient warm-up (fast, focuses on worst lag)
    for _ in range(200):
        if not _ok(200):
            break
        h = _subgrad_step(h, t0, budget_s - 200, n_lags=2)
    c5 = _eval(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Nesterov continuation
    for alpha in [2000, 10000, 50000, 200000, 800000, 2000000]:
        if not _ok(100):
            break
        h = _nesterov(h, alpha, 400, 0.02, t0, budget_s - 100)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        h = _repair(h, len(h))

    # Phase 3: More Nesterov at high alpha with smaller steps
    for alpha in [500000, 2000000, 5000000]:
        if not _ok(80):
            break
        h = _nesterov(best_h.copy(), alpha, 300, 0.005, t0, budget_s - 80)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Multi-restart with perturbations
    for trial in range(8):
        if not _ok(60):
            break
        scale = 0.3 * (1.0 - trial / 10.0) + 0.02
        h_p = best_h + scale * rng.randn(len(best_h))
        h_p = _repair(h_p, len(best_h))
        h_p = _nesterov(h_p, 100000, 200, 0.01, t0, budget_s - 60)
        c5 = _eval(h_p)
        if c5 < best_c5:
            best_c5, best_h = c5, h_p.copy()
        h_p = _bullet(h_p)
        c5 = _eval(h_p)
        if c5 < best_c5:
            best_c5, best_h = c5, h_p.copy()

    # Phase 5: Binary push experiment
    for thresh in [0.45, 0.5, 0.55]:
        if not _ok(40):
            break
        h_b = _binary_push(best_h.copy(), thresh)
        h_b = _nesterov(h_b, 50000, 300, 0.01, t0, budget_s - 40)
        c5 = _eval(h_b)
        if c5 < best_c5:
            best_c5, best_h = c5, h_b.copy()

    # Phase 6: Subgradient polish on true objective
    for _ in range(500):
        if not _ok(30):
            break
        h = _subgrad_step(best_h.copy(), t0, budget_s - 30, n_lags=3)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 7: Long local search
    if _ok(5):
        remaining = budget_s - 5 - _el()
        n_ls = int(remaining / 0.001)
        n_ls = min(n_ls, 200000)
        best_h, best_c5 = _local_search(best_h.copy(), n_ls, rng, t0, budget_s - 2)

    # Phase 8: Final ultra-fine polish
    if _ok(3):
        h = best_h.copy()
        n = len(h)
        c5c = _eval(h)
        n_pol = min(int((budget_s - 3 - _el()) / 0.0008), 50000)
        for _ in range(n_pol):
            if time.time() - t0 > budget_s - 2:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 2)) % n
            d = 3e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))

def _bullet(h):
    n = len(h)
    for alph in [5000, 20000, 100000]:
        v = np.zeros(n)
        mu = 0.8
        step = 0.01
        for _s in range(150):
            g = _smoothed_grad(h, alph)
            h_prev = h.copy()
            v = mu * v + (1.0 - mu) * g
            h_new = _proj_to_sum(h - step * v, n * 0.5)
            if _eval(h_new) < _eval(h_prev):
                h = h_new
                step = min(step * 1.05, 0.1)
            else:
                step *= 0.5
                h = _proj_to_sum(h_prev - step * v, n * 0.5)
    return _repair(h, n)
