            sp = -sh

            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _pmobj(x, p):
    c = _corr(x)
    return (np.sum(c ** p) / len(c)) ** (1.0 / p)

def _pmgrad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = len(c)
    base = (np.sum(c ** p) / m) ** (1.0 / p - 1.0) / m
    cpm = c ** (p - 1) * base
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = cpm[k]
        if abs(w) < 1e-20:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= w * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - _GLOBAL_N * 0.5

def _sum_eq_jac(x):
    return np.ones(len(x))

def _slsqp_smooth(h, alpha, maxiter, ftol):
    n = len(h)
    _GLOBAL_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_smobjc, h, jac=_smgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp_power(h, p, maxiter, ftol):
    n = len(h)
    _GLOBAL_N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    try:
        res = minimize(_pmobj, h, jac=_pmgrad, args=(p,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, obj_fn, grad_fn, arg, steps, step0):
    n = len(h)
    f = obj_fn(h, arg)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        g = grad_fn(h, arg)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        hn = _proj_to_sum(h - step * v, n * 0.5)
        fn = obj_fn(hn, arg)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.04, step0 * 3)
        else:
            step *= 0.5
            h = _proj_to_sum(hp - step * v, n * 0.5)
            f = obj_fn(h, arg)
    return _repair(h, n)

def _topk_descent(h, k, max_rounds, el_fn, bc_fn):
    n = len(h)
    c5c = _eval(h)
    for _ in range(max_rounds):
        if bc_fn():
            break
        c = _corr(h)
        top = np.argsort(c)[-k:]
        g = np.zeros(n)
        for kk in top:
            sh = kk - (n - 1)
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= (2.0 / n) / k
        improved = False
        for st in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            hn = _proj_to_sum(h - st * g, n * 0.5)
            cn = _eval(hn)
            if cn < c5c - 1e-15:
                h = hn
                c5c = cn
                improved = True
                break
        if not improved:
            break
    return h, c5c

def _block_search(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(n_iter):
        if bc_fn() or el_fn() > budget_s - 5:
            break
        blen = rng.randint(2, 7)
        i = rng.randint(0, n - blen)
        j = rng.randint(0, n - blen)
        if abs(i - j) < blen:
            continue
        seg1 = h[i:i+blen].copy()
        seg2 = h[j:j+blen].copy()
        h[i:i+blen] = seg2
        h[j:j+blen] = seg1
        h = _repair(h, n)
        cn = _eval(h)
        if cn < c5c - 1e-17:
            c5c = cn
            if cn < bc:
                bc = cn
                bh = h.copy()
        else:
            h[i:i+blen] = seg1
            h[j:j+blen] = seg2
    return bh, bc

def _finer_sear(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(n_iter):
        if bc_fn() or el_fn() > budget_s - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-19:
                c5c = cn
                if cn < bc:
                    bc = cn
                    bh = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _downsample(h, fac):
    n = len(h)
    m = n // fac
    return h[:m * fac].reshape(m, fac).mean(axis=1).copy()

def _upsample(h, fac):
    return np.repeat(h, fac).copy()

budget_s = 1100

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s
    budget_s = budget_s
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Power-mean SLSQP at multiple p values (key difference from previous)
    for p in [20, 50, 100, 200, 500]:
        if _el() > 150:
            break
        hc = _slsqp_power(best_h, p, 1500, 1e-16)
        cv = _eval(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 2: Smooth-max SLSQP with varied alpha
    for alpha in [5000, 50000, 200000, 1000000, 5000000]:
        if _el() > 350:
            break
        hc = _slsqp_smooth(best_h, alpha, 2000, 1e-17)
        cv = _eval(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 3: Nesterov with power-mean
    for p in [30, 80, 200]:
        if _el() > 550:
            break
        hc = _nesterov(best_h, _pmobj, _pmgrad, p, 500, 0.002)
        cv = _eval(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 4: Top-k gradient descent (vary k)
    for k in [2, 3, 5, 8, 4]:
        if _el() > budget_s - 300:
            break
        hc, cv = _topk_descent(best_h.copy(), k, 500, _el, _bc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 5: Block-exchange local search
    if _el() < budget_s - 250:
        n_blk = int(min((budget_s - 250 - _el()) / 0.003, 50000))
        best_h, best_c5 = _block_search(best_h, n_blk, rng, _el, _bc)

    # Phase 6: Nesterov polish with high alpha smooth-max
    if _el() < budget_s - 150:
        hc = _nesterov(best_h, _smobjc, _smgrad, 300000, 600, 0.001)
        cv = _eval(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 7: Final SLSQP with very high alpha
    if _el() < budget_s - 60:
        hc = _slsqp_smooth(best_h, 2000000, 3000, 1e-18)
        cv = _eval(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc.copy()

    # Phase 8: Coarse restart attempt (different basin)
    if _el() < budget_s - 200:
        n_coarse = max(28, n0 // 20)
        for trial in range(5):
            if _el() > budget_s - 210:
                break
            hx = _downsample(best_h, n0 // n_coarse)
            hx += 0.3 * rng.randn(n_coarse)
            hx = _repair(hx, n_coarse)
            hc = _slsqp_power(hx, 100, 800, 1e-14)
            hi = _repair(_upsample(hc, n0 // n_coarse), n0)
            cv = _eval(hi)
            if cv < best_c5:
                best_c5 = cv
                best_h = hi.copy()
            if _el() < budget_s - 180:
                hc2 = _slsqp_smooth(hi, 100000, 1000, 1e-15)
                cv2 = _eval(hc2)
                if cv2 < best_c5:
                    best_c5 = cv2
                    best_h = hc2.copy()

    # Phase 9: Ultra-fine 2-point polish
    if _el() < budget_s - 5:
        n_pol = int(min((budget_s - 5 - _el()) / 0.001, 60000))
        best_h, best_c5 = _finer_sear(best_h, n_pol, rng, _el, _bc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
