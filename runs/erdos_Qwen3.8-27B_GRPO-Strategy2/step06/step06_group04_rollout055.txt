    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_full(h, alpha, topk=None):
    n = len(h)
    dx = 2.0 / n
    cov = np.correlate(h, 1.0 - h, mode='full')
    m = cov.max()
    s = np.clip(alpha * (cov - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    if topk is not None and topk < 2*n-1:
        idx = np.argpartition(w, -topk)[-topk:]
        mask = np.zeros_like(w)
        mask[idx] = 1.0
        w = w * mask
        ws = w.sum()
        if ws > 0: w /= ws
        else: w = np.zeros_like(w); w[n-1] = 1.0
    g = np.zeros(n)
    for k in range(2 * n - 1):

            g[:n-sp] -= wk * h[:n-sp]
            g[sp:] += wk * (1.0 - h[sp:])

        # Let me just use standard momentum instead

        if c < best_c:
            best_c = c
            best_x = x.copy()

Actually, for coordinate descent: if we fix all elements except h[i], the objective is a piecewise linear function of h[i] (since each f_k is linear in h[i] when other elements are fixed... wait, no, f_k = ∑ h[j]*(1-h[j+k]) is quadratic in h if h[i] appears in multiple terms).

Actually, for a fixed h with only h[i] variable:
f_k(h) = ∑_j h[j]*(1-h[j+k])
The terms involving h[i] are:
- h[i] appears as h[j] when j=i: contributes h[i]*(1-h[i+k])
- h[i] appears as (1-h[j+k]) when j+k=i: contributes h[j]*(1-h[i]) = h[i-k]*(1-h[i])

So f_k as a function of h[i] is: h[i]*(1-h[i+k]) + h[i-k]*(1-h[i]) + (terms not involving h[i])
= h[i] - h[i]*h[i+k] + h[i-k] - h[i-k]*h[i] + const
= h[i]*(1 + h[i-k]) - h[i]*h[i+k] + h[i-k] + const
= h[i]*(1 + h[i-k] - h[i+k]) + const

So f_k is LINEAR in h[i]! (Given the other elements fixed.)

This means the max over k is the max of linear functions of h[i], which is a convex function of h[i]. We can find the minimum by ternary search or by finding the kink points.

This is great for coordinate descent! For each i, the optimal h[i] is either:
- At a kink (where two f_k's are equal and maximal)
- At the boundary (0 or 1)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _overlaps(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _grad(h, alpha, topk=0):
    n = len(h)
    dx = 2.0 / n
    cov = np.correlate(h, 1.0 - h, mode='full')
    m = cov.max()
    s = np.clip(alpha * (cov - m), -600, 600)
    w = np.exp(s - s.max())
    w /= w.sum()
    if topk > 0 and topk < 2 * n - 1:
        idx = np.argpartition(w, -topk)[-topk:]
        mask = np.zeros(2 * n - 1)
        mask[idx] = 1.0
        w = w * mask
        ws = w.sum()
        if ws > 0:
            w /= ws
        else:
            w[:] = 0.0; w[n - 1] = 1.0
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[:n - sp] -= wk * h[:n - sp]
            g[sp:] += wk * (1.0 - h[sp:])
    return g

def _adam(h0, a0, a1, topk, n_iter, lr0, lr1, seed, deadline):
    n = len(h0)
    x = h0.copy().astype(float)
    m_v = np.zeros(n); v_v = np.zeros(n)
    t = 0; b1, b2, eps = 0.85, 0.999, 1e-9
    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = a0 * (a1 / a0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        g = _grad(x, alpha, topk)
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        mhat = m_v / (1 - b1 ** t)
        vhat = v_v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
        if t % 800 == 0:
            rng = np.random.RandomState(t)
            x = _repair(x + 0.003 * rng.randn(n), n)
    return best_x, best_c

def _slsqp(h, alpha, topk, maxiter, ftol):
    n = len(h)
    mat = np.ones((1, n))
    def _se(x):
        return x.sum() - n * 0.5
    def _se_jac(x):
        return mat
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _se, 'jac': _se_jac}]
    try:
        res = minimize(lambda x: _grad(x, alpha, topk).dot(x) + 0, h,
                       jac=lambda x: _grad(x, alpha, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _obj_slsqp(x, alpha, topk):
    n = len(x)
    cov = np.correlate(x, 1.0 - x, mode='full')
    m = cov.max()
    s = np.clip(alpha * (cov - m), -600, 600)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _obj_jac_slsqp(x, alpha, topk):
    return _grad(x, alpha, topk)

def _cd_bisect(h, n, rng, deadline, max_rounds=60):
    """Paired coordinate descent with bisection on delta."""
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            j = (i + n // 2) % n
            if i == j:
                continue
            lo, hi = -x[i], 1.0 - x[i]
            lo = max(lo, -0.5); hi = min(hi, 0.5)
            if hi - lo < 1e-12:
                continue
            for _ in range(40):
                mid = (lo + hi) / 2
                x[i] += mid; x[j] -= mid
                if x[i] < 0 or x[i] > 1 or x[j] < 0 or x[j] > 1:
                    if mid > 0:
                        hi = mid
                    else:
                        lo = mid
                    x[i] -= mid; x[j] += mid
                    continue
                c_mid = _eval(x)
                x[i] -= mid; x[j] += mid
                if c_mid <= c_cur:
                    lo = mid
                else:
                    hi = mid
            delta = (lo + hi) / 2
            if abs(delta) > 1e-12:
                x[i] += delta; x[j] -= delta
                x = _clip_x(x, n)
                c_new = _eval(x)
                if c_new < c_cur - 1e-15:
                    c_cur = c_new; improved = True
                    if c_cur < best_c:
                        best_c = c_cur; best_x = x.copy()
                else:
                    x[i] -= delta; x[j] += delta
                    x = _clip_x(x, n)
            if not improved and r > 8:
                break
        if not improved:
            break
    return best_x, best_c

def _clip_x(x, n):
    target = n / 2.0
    s = x.sum()
    if abs(s - target) > 1e-12:
        x += (target - s) / n
    return np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h), n_new)

def _subgrad_polyak(h0, deadline, max_iter=2000):
    """Subgradient method with Polyak step for min-max."""
    n = len(h0)
    x = h0.copy()
    f_best = _eval(x)
    best_x = x.copy()
    gamma = 0.01
    for it in range(max_iter):
        if time.time() > deadline:
            break
        cov = _overlaps(x)
        m = cov.max()
        # subgradient: pick the argmax shift (or average of top few)
        top3 = np.argsort(cov)[-3:]
        g = np.zeros(n)
        for k in top3:
            sh = (n - 1) - k
            dx = 2.0 / n
            if sh >= 0:
                g[:n - sh] += (1.0 - x[:n - sh])
                if sh > 0:
                    g[sh:] -= x[:n - sh]
            else:
                sp = -sh
                g[:n - sp] -= x[:n - sp]
                g[sp:] += (1.0 - x[sp:])
        g *= dx
        # Polyak step: gamma / (f_current - f_best) but f_current == m
        # Use fixed decreasing step
        step = gamma / (it + 10)
        x = x - step * g
        x = _repair(x, n)
        c = _eval(x)
        if c < f_best:
            f_best = c; best_x = x.copy()
        gamma *= 0.999
    return best_x, f_best
