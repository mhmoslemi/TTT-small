    """Project z onto {0<=h<=1, sum(h)=target} via water-filling."""
    n = len(z)

        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0, 1)
        s = h.sum()

        else:

    diff = target - h.sum()

        # Adjust the element closest to boundary in the right direction
        if diff > 0:

def _proj(z, target):
    n = len(z)
    lo, hi = z.min() - 2.0, z.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
