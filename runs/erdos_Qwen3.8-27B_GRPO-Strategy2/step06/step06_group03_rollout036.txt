    # Subject to: A_dot_h - t <= -b  (i.e., A@h + b <= t)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = 0.5*(lo+hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5*(lo+hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _lp_step(h, n, top_m, max_iter=50):

    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    for it in range(max_iter):
        c = np.correlate(h, 1-h, mode='full')
        active_idx = np.argsort(c)[-top_m:]

        c_vals = np.zeros(len(active_idx))

            if sh == 0:

            elif sh > 0:

                A[ii, sh:] -= h[:n-sh]

                A[ii, p:] += (1 - h[:n-p])
                A[ii, :n-p] -= h[p:n]

        # LP: min t s.t. A@h_new - t <= -rhs, sum(h_new)=n/2, 0<=h_new<=1

from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = 0.5*(lo+hi)
            if np.clip(h-mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5*(lo+hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s-s.max()))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    w = np.exp(s-s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12: continue
        sh = (n-1)-k
        wk = w[k]
        if sh >= 0:
            g[:n-sh] += wk*(1-h[sh:])
            if sh > 0:
                g[sh:] -= wk*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    def _cf(x): return np.array([x.sum()-0.5*n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type':'eq','fun':_cf,'jac':_cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _lp_refine(h, n, top_m, max_iter, time_limit):
    """LP-based active set refinement"""
    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit: break
        c = np.correlate(h, 1-h, mode='full')
        cmax = c.max()
        active_idx = np.argsort(c)[-top_m:]
        A = np.zeros((len(active_idx), n))
        c_vals = c[active_idx].copy()
        for ii, k in enumerate(active_idx):
            sh = (n-1)-k
            if sh == 0:
                A[ii,:] = 1-2*h
            elif sh > 0:
                A[ii,:n-sh] += 1-h[sh:]
                A[ii,sh:] -= h[:n-sh]
            else:
                p = -sh
                A[ii,p:] += 1-h[:n-p]
                A[ii,:n-p] -= h[p:]
        rhs = c_vals - A @ h
        # LP: min t s.t. A@h_new - t <= -rhs, 1@h_new = n/2, 0<=h_new<=1
        c_obj = np.zeros(n+1); c_obj[-1] = 1
        A_ub = np.hstack([A, -np.ones((len(active_idx),1))])
        b_ub = -rhs
        A_eq = np.zeros((1,n+1)); A_eq[0,:n] = 1
        b_eq = np.array([n/2.0])
