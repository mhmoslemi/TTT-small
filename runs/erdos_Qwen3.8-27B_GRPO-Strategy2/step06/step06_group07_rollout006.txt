    n = len(h)

    s = np.clip(alpha*(c - cmax), -400, 0)
    w = np.exp(s)
    wsum = w.sum()

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _sobj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    s = np.clip(alpha * (c - cmax), -500, 0)
    return (cmax + np.log(max(float(np.exp(s).sum()), 1e-300)) / alpha) * (2.0 / n)

def _sgrad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    cmax = c.max()
    s = np.clip(alpha * (c - cmax), -500, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-11:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_small(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_sobj, h0, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x)
    except Exception:
        return _repair(h0.copy())

def _lbfgsb_full(h0, alpha, maxiter):
    n = len(h0)
    pen = 1e4

    def obj_pen(x):
        base = _sobj(x, alpha)
        dev = x.sum() - n * 0.5
        return base + pen * dev * dev

    def grad_pen(x):
        g = _sgrad(x, alpha)
        dev = x.sum() - n * 0.5
        return g + 2.0 * pen * dev * np.ones(n)

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x)
    except Exception:
        return _repair(h0.copy())

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _pgd_period(h_p, T, n, alphas, max_iters, lr_init, seed):
    m = n // T
    rng = np.random.RandomState(seed)
    h_p = _repair(h_p)
    best = h_p.copy()
    best_c = _eval(np.tile(best, m))
    lr = lr_init
    mom = np.zeros(T)
    beta = 0.8
    for alpha in alphas:
        iters = max(50, max_iters // len(alphas))
        for it in range(iters):
            h_full = np.tile(h_p, m)
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            cmax = c.max()
            s = np.clip(alpha * (c - cmax), -500, 0)
            w = np.exp(s)
            ws = w.sum()
            if ws < 1e-300:
                break
            w /= ws
            dx = 2.0 / n
            g_full = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-11:
                    continue
                sh = (n - 1) - k
                wk = w[k] * dx
                if sh >= 0:
                    g_full[:n - sh] += wk * (1.0 - h_full[sh:])
                    if sh > 0:
                        g_full[sh:] -= wk * h_full[:n - sh]
                else:
                    sp = -sh
                    g_full[sp:] += wk * (1.0 - h_full[:n - sp])
                    g_full[:n - sp] -= wk * h_full[sp:]
            g_p = np.zeros(T)
            for p in range(m):
                g_p += g_full[p * T:(p + 1) * T]
            g_p /= m
            mom = beta * mom + (1.0 - beta) * g_p
            h_new = _repair(h_p - lr * mom)
            c_new = _eval(np.tile(h_new, m))
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h_p = h_new
                if lr < 0.02:
                    lr *= 1.02
            else:
                lr *= 0.7
                if lr < 1e-7:
                    lr = lr_init
                    mom = np.zeros(T)
    return best, best_c

def _coord_descent(h, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c
