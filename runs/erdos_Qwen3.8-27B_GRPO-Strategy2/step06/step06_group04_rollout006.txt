import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _wgrad(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n-1), n):
        k = d + n - 1
        wk = w[k] * dx
        if abs(wk) < 1e-12:
            continue
        if d >= 0:
            g[d:] += wk * (1.0 - h[:n-d])
            g[:n-d] -= wk * h[d:]
        else:
            dp = -d
            g[:n-dp] += wk * (1.0 - h[dp:])
            g[dp:] -= wk * h[:n-dp]
    return g

def _primal_dual(h0, lr0, beta0, n_iters, deadline, seed):
    n = len(h0)
    x = h0.copy()
    w = np.ones(2*n-1) / (2*n-1)
    best_x, best_c = x.copy(), _eval(x)
    lr = lr0
    beta = beta0
    for i in range(n_iters):
        if time.time() > deadline:
            break
        c = _corr(x, n)
        g = _wgrad(x, w, n)
        x = _repair(x - lr * g, n)
        c = _corr(x, n)
        mc = c.max()
        w = w * np.exp(np.clip(beta * (c - mc), -50, 50))
        ws = w.sum()
        if ws > 1e-30:
            w /= ws
        cx = _eval(x)
        if cx < best_c:
            best_c = cx
            best_x = x.copy()
        lr *= 0.99998
        if i % 2000 == 1999:
            beta = min(beta * 1.02, 800.0)
    return best_x, best_c

def _lss_fn(x, alpha, n):
    c = _corr(x, n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _lss_df(x, alpha, n):
    c = _corr(x, n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s.max()
    ew = np.exp(s - sm)
    return ew / ew.sum()

def _lss_grad(x, alpha, n):
    c = _corr(x, n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    sm = s.max()
    ew = np.exp(s - sm)
    w = ew / ew.sum()
    return _wgrad(x, w, n)

def _sum_fn(x):
    return x.sum() - len(x) * 0.5

def _sum_jc(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jc}]
    try:
        res = minimize(_lss_fn, h, jac=_lss_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam_smooth(h0, alpha_s, alpha_e, n_iter, lr_s, lr_e, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x, best_c = x.copy(), _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = alpha_s * (alpha_e / alpha_s) ** frac
        lr = lr_s * (lr_e / lr_s) ** frac
        if t % 800 == 0:
            x = _repair(x + 0.003 * rng.randn(n), n)
        g = _lss_grad(x, alpha, n)
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        mhat = m_v / (1 - b1 ** t)
        vhat = v_v / (1 - b2 ** t)
        x = _repair(x - lr * mhat / (np.sqrt(vhat) + eps), n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

def _shift_cd(h0, deadline, seed):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy()
    c_cur = _eval(x)
    best_x, best_c = x.copy(), c_cur
    scale = 0.005
    iters = 0
    while time.time() < deadline and scale > 1e-8:
        iters += 1
        c_vals = _corr(x, n)
        top_k = np.argsort(c_vals)[-min(5, 2*n-1):][::-1]
        for k in top_k:
            d = k - (n - 1)
            if d >= 0:
                idxs = list(range(d, n))
            else:
                idxs = list(range(0, n + d))
            if len(idxs) < 2:
                continue
            i1 = idxs[rng.randint(len(idxs))]
            i2 = idxs[rng.randint(len(idxs))]
            if i1 == i2:
                continue
            j = rng.randint(n)
            delta = scale * rng.randn()
            x[i1] += delta
            x[j] -= delta
            x = _repair(x, n)
            c_new = _eval(x)
            if c_new < c_cur - 1e-15:
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_x = x.copy()
                scale = max(scale * 0.9995, 1e-9)
            else:
                x[i1] -= delta
                x[j] += delta
                x = _repair(x, n)
                scale *= 0.95
                break
        if iters % 500 == 0:
            scale *= 0.98
    return best_x, best_c

def _multi_n_explore(h_best, n_vals, deadline, seed):
    best_h, best_c, best_n = h_best.copy(), _eval(h_best), len(h_best)
