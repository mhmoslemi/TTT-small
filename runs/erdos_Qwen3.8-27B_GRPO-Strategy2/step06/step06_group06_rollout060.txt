    n = len(h)
    f0 = _smax(h, alpha)
    try:
        res = minimize(_smax, h, jac=_smax_g, args=(alpha,), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() > t_end:
            pass
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _proj_grad(h, alphas, t_end):
    n = len(h)
    hc = h.copy()
    fc = _eval(hc)
    bh, bf = hc.copy(), fc
    v = np.zeros(n)
    for alpha in alphas:
        step = 0.02
        for s in range(5000):
            if time.time() > t_end:
                return bh, bf
            g = _smax_g(hc, alpha)
            v = 0.7 * v + 0.3 * g
            ok = False
            for _bt in range(12):
                hn = _proj(hc - step * v, n * 0.5)
                fn = _smax(hn, alpha)
                if fn <= fc + 1e-15:
                    hc = hn
                    fc = fn
                    step = min(step * 1.05, 0.4)
                    ok = True
                    if s % 20 == 0:
                        ec = _eval(hc)
                        if ec < bf:
                            bf, bh = ec, hc.copy()
                    break
                step *= 0.4
            if not ok:
                step *= 0.05
                v *= 0.1
        v *= 0.1
    ec = _eval(hc)
    if ec < bf:
        bf, bh = ec, hc.copy()
    return bh, bf

def _sa(h, n, t_end, n_iter, rng):
    cc = _eval(h)
    bh, bc = h.copy(), cc
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = 0.003 * (1e-4 / 0.003) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.001 + 0.02 * t)
        o1, o2 = h[i1], h[i2]
        h[i1] = o1 + d
        h[i2] = o2 - d
        h = _repair(h, n)
        cn = _eval(h)
        if cn < cc or rng.rand() < np.exp(-(cn - cc) / max(t, 1e-10)):
            cc = cn
            if cn < bc:
                bc, bh = cn, h.copy()
        else:
            h[i1], h[i2] = o1, o2
            h = _repair(h, n)
    return bh, bc

def _coord(h, n, t_end, n_tries, rng):
    hc = h.copy()
    cc = _eval(hc)
    bh, bc = hc.copy(), cc
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-5 * rng.randn()
        ni = hc[i] + d
        nj = hc[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        cn = _eval(hc)
        if cn < cc - 1e-16:
            cc = cn
            if cn < bc:
                bc, bh = cn, hc.copy()
        else:
            hc[i], hc[j] = hc[i] - d, hc[j] + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh, bc5 = h.copy(), _eval(h)

    rem = budget_s - (time.time() - t0)
    for a in [100, 500, 2000, 10000, 50000]:
        if rem < 250:
            break
        hc = _slsqp(bh, a, 4000, t0 + rem - 249)
        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()
        rem = budget_s - (time.time() - t0)

    rem = budget_s - (time.time() - t0)
    if rem > 200:
        hc, c5 = _proj_grad(bh, [500, 5000, 50000], t0 + rem - 190)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    rem = budget_s - (time.time() - t0)
    if rem > 100:
        ni = min(int((rem - 80) / 0.004), 50000)
        hc, c5 = _sa(bh, n0, t0 + rem - 80, ni, rng)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    rem = budget_s - (time.time() - t0)
    if rem > 60:
        hc = _slsqp(bh, 500000, 8000, t0 + rem - 50)
        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        nt = min(int((rem - 10) / 0.003), 30000)
        hc, c5 = _coord(bh, n0, t0 + rem - 10, nt, rng)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        hc = _slsqp(bh, 1000000, 5000, t0 + rem - 5)
        c5 = _eval(hc)
        if c5 < bc5:
            bc5, bh = c5, hc.copy()

    bh = _repair(bh, n0)
    bc5 = _eval(bh)
    return (bh, bc5, n0)
