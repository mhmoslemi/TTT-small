import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g

def _mlgd(h, rounds, k_top, steps_init, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top_k) * dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.4] + [s for s in steps if s < steps[si] * 0.5]
                if len(steps) > 10:
                    steps = steps[:10]
                break
        if not improved:
            steps = [s * 0.4 for s in steps]
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0
    return h, c5c

def _lsx_obj(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 50000.0
    sc = np.clip(a * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 50000.0
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _local_3pt(h, ntry, rng, c5c, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        k = (i + 7 + rng.randint(0, 6)) % nc
        d1 = 8e-6 * rng.randn()
        d2 = 8e-6 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk

            if c5n < c5c - 1e-18:

                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
    return best_h, best_c
