            i2 = np.arange(n - sp)

    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter=5000, ftol=1e-20):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)

    t = 0

    best_x = x.copy()

        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = a0 * (a1 / a0) ** frac

        mhat = m / (1 - b1 ** t)

        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:

            best_x = x.copy()
    return best_x, best_c

            x[xj] = 0.0
    
    return best_x, best_c

def _subgrad_max(h, n_iter, lr, deadline):
    x = _repair(h, len(h)).copy()
    n = len(x)
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(x, 1.0 - x, mode='full')
        k_star = np.argmax(c)
        # Gradient of c[k_star] with respect to x
        sh = (n - 1) - k_star
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= x[i2 + sp]
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g /= gn
        x = x - lr * g
        x = _repair(x, n)
        c_val = _eval(x)
        if c_val < best_c:
            best_c = c_val
            best_x = x.copy()
        lr *= 0.9999
    return best_x, best_c

One more thought: the _soft_grad function is the most expensive part (O(n²) per call). For n=310, that's about 620 correlation terms, each O(n). In numpy, the correlate itself is fast (C implementation), but the weighted sum in the gradient loop is Python-level. Let me optimize this:

Instead of looping over all 2n-1 lags, I can use the fact that:
g[i] = sum_k w[k] * d(c[k])/d(x[i])

The derivative d(c[k])/d(x[i]) depends on the relationship between i and k:
- For the correlation c[k] = sum_j x[j] * (1-x[j+k-n+1]) (for appropriate j range)
- d(c[k])/d(x[i]) = (1-x[i+k-n+1]) - x[i-(k-n+1)] (when both terms exist)

This is what the current code does. The loop over k is the bottleneck. Can I vectorize it?

For a given i:
g[i] = sum_{k: i in valid range for lag k} w[k] * dx * [(1-x[i+sh_k]) if applicable - x[i-sh_k] if applicable]

where sh_k = (n-1-k).

Let me think... For lag k with shift sh = n-1-k:
- If sh >= 0: x[i] contributes (1-x[i+sh]) to g[i] for i in [0, n-sh-1], and -x[i-sh] to g[i] for i in [sh, n-1]
- If sh < 0 (let sp = -sh): x[i] contributes (1-x[i-sp]) to g[i] for i in [sp, n-1], and -x[i+sp] to g[i] for i in [0, n-sp-1]

So:
g[i] = dx * [sum_{k: sh_k >= 0, i < n-sh_k} w[k]*(1-x[i+sh_k]) + sum_{k: sh_k > 0, i >= sh_k} w[k]*(-x[i-sh_k]) + sum_{k: sh_k < 0, i >= -sh_k} w[k]*(1-x[i+sh_k]) + sum_{k: sh_k < 0, i < n+sh_k} w[k]*(-x[i+sh_k])]

This is hard to vectorize easily in numpy without a 2D array. For n=310, the Python loop is 619 iterations, each doing a numpy operation on arrays of size up to 310. This should be fast enough (~0.1ms total).

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _soft_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h0, a0, a1, n_iter, lr0, lr1, seed, deadline):
    n = len(h0)
    rng = np.random.RandomState(seed)
    x = h0.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(n_iter - 1, 1)
        alpha = a0 * (a1 / a0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac
        if t > 1500 and t % 2500 == 0:
            x = _repair(x + 0.004 * rng.randn(n), n)
        g = _soft_grad(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)
        x = x - lr * mhat / (np.sqrt(vhat) + eps)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return best_x, best_c

def _binary_search(h0, n, rng, n_iters, deadline):
    x = (h0 > 0.5).astype(float)
    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    patience = 0
    for i in range(n_iters):
        if time.time() > deadline:
            break
        if patience > 50000:
            break
        oi = rng.randint(n)
        xj = rng.randint(n)
        if oi == xj:
            continue
        if x[oi] == x[xj]:
            continue
        x[oi], x[xj] = x[xj], x[oi]
        c_new = _eval(x)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_x = x.copy()
            c_cur = c_new
            patience = 0
        else:
            x[oi], x[xj] = x[xj], x[oi]
            patience += 1
    return best_x, best_c

def _subgrad_max(h, n_iter, lr0, deadline):
    x = _repair(h.copy(), len(h))
    n = len(x)
    best_x = x.copy()
    best_c = _eval(x)
    lr = lr0
    for i in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(x, 1.0 - x, mode='full')
        k_star = int(np.argmax(c))
        sh = (n - 1) - k_star
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= x[i2 + sp]
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            x = x - lr * g / gn
            x = _repair(x, n)
            c_val = _eval(x)
            if c_val < best_c:
                best_c = c_val
                best_x = x.copy()
        lr *= 0.99995
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
