import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_grad(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        m = n - sh
        if m > 0:
            i1 = np.arange(m)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _smooth_obj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    val = m + np.log(float(np.sum(np.exp(s)))) / alpha
    sm = x.sum() - n * 0.5
    val += 0.5 * lam * sm * sm
    return val

def _smooth_grad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    n_active = int(np.sum(w > 1e-10))
    idxs = np.argsort(w)[::-1][:max(n_active, 10)]
    for k in idxs:
        if w[k] < 1e-12:
            continue
        g += w[k] * _lag_grad(x, k, n)
    g += lam * (x.sum() - n * 0.5)
    return g

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, lam, n), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-15, 'ftol': 1e-19, 'maxcor': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_descent(h, n, max_rounds, top_k, step0):
    c5c = _eval(h)
    step = step0
    for r in range(max_rounds):

        top_idx = np.argsort(ov)[-k_use:]

        ws /= ws.sum()

            g += ws[i] * _lag_grad(h, kk, n)
