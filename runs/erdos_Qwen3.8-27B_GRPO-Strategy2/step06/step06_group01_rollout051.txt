                h_fin, c5f2 = focused_cd(h_fin, n_bb, rng, min(20, rem() - 10))
                if c5f2 < c5f:
                    h_fin, c5f = h_fin, c5f2
                if c5f < best_c5:
                    best_c5, best_h, best_n = c5f, h_fin.copy(), n_bb

    # Phase 8: Final polish
    if rem() > 25:
        hf = slsqp_opt(best_h, 100000, 800, 1e-18)
        c5 = ec5(hf)
        if c5 < best_c5:
            best_c5, best_h = c5, hf.copy()

    if rem() > 8:
        hf2, c52 = focused_cd(best_h, best_n, rng, max(5, rem() - 2))
        if c52 < best_c5:
            best_c5, best_h = c52, hf2.copy()

    best_h = repair(best_h, best_n)
    best_c5 = ec5(best_h)
    return (best_h, best_c5, best_n)
