        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sm_val(h, alpha, mu):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha + mu * (h.sum() - n / 2.0) ** 2

def _sm_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return g

def _opt(h, alpha, mu, maxiter, tlimit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_val, h, jac=_sm_grad, args=(alpha, mu),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-14, 'maxfun': maxiter * 3,
                                'iprint': -1})
        out = _repair(res.x, n)
        return out
    except Exception:
        return _repair(h.copy(), n)

def _focal_step(h, top_k_count, tlimit):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    top_k = np.argsort(c)[::-1][:top_k_count]
    g = np.zeros(n)
    for kk in top_k:
        sh = (n - 1) - kk
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return h, c, g

def _focal_descent(h, max_rounds, tlimit):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    for _ in range(max_rounds):
        if time.time() > tlimit:
            break
        h, c, g = _focal_step(h, 5, tlimit)
        moved = False
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h, c5 = h_new, c5n
                moved = True
                break
        if not moved:
            break
    return h, c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _make_binary_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    perm = rng.permutation(n)
    h[perm[:half]] = 1.0
    return _repair(h, n)

def _make_anti_sym(n, rng):
    half = n // 2
    g = rng.rand(half) * 0.8 - 0.4
    g = np.clip(g, -0.49, 0.49)
    h = np.zeros(n)
    h[:half] = 0.5 + g
    if n % 2 == 0:
        h[half:] = 0.5 - g
    else:
        h[half:] = 0.5 - g[:n - half]
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B polish with increasing smoothness (25% time)
    p1_end = t0 + budget_s * 0.25
    for alpha, mu in [(500, 50.0), (5000, 100.0), (50000, 200.0),
                      (500000, 500.0), (5000000, 2000.0), (50000000, 10000.0)]:
        if elapsed() > p1_end:
            break
        h_new = _opt(best_h, alpha, mu, 4000, p1_end)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Multi-resolution exploration (25% time)
    p2_end = t0 + budget_s * 0.50
    for n_try in [200, 250, 300, 350, 400, 500, 620]:
        if elapsed() > p2_end:
            break
        xo = np.linspace(0.0, 2.0, best_n, endpoint=False)
        xn = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_r = _repair(np.interp(xn, xo, best_h), n_try)
        for alpha, mu in [(10000, 200.0), (200000, 2000.0), (2000000, 20000.0)]:
            if elapsed() > p2_end:
                break
            h_r = _opt(h_r, alpha, mu, 3000, p2_end)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 3: Diverse starts at best resolution (20% time)
    p3_end = t0 + budget_s * 0.70
    diverse_starts = []
    diverse_starts.append((best_h.copy() + 0.02 * rng.randn(best_n), best_n))
    diverse_starts.append((_make_binary_init(best_n, rng), best_n))
    diverse_starts.append((_make_anti_sym(best_n, rng), best_n))
    if best_n % 2 == 0:
        diverse_starts.append((best_h.copy()[::-1], best_n))
    for hs, ns in diverse_starts:
        if elapsed() > p3_end:
            break
        hs = _repair(hs, ns)
        for alpha, mu in [(10000, 200.0), (500000, 5000.0)]:
            if elapsed() > p3_end:
                break
            hs = _opt(hs, alpha, mu, 3000, p3_end)
            c5 = _eval(hs)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hs.copy(), ns

    # Phase 4: Intensive L-BFGS-B on best (20% time)
    p4_end = t0 + budget_s * 0.90
    for alpha, mu in [(500000, 2000.0), (5000000, 20000.0), (50000000, 50000.0), (500000000, 500000.0)]:
        if elapsed() > p4_end:
            break
        h_new = _opt(best_h, alpha, mu, 6000, p4_end)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: Focal subgradient descent (8% time)
    p5_end = t0 + budget_s * 0.98
    if elapsed() < p5_end:
        h_f, c5_f = _focal_descent(best_h, 200, p5_end)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Phase 6: Final L-BFGS polish (2% time)
    if elapsed() < t0 + budget_s - 3:
        h_new = _opt(best_h, 100000000, 100000.0, 3000, t0 + budget_s - 3)
        c5_f = _eval(h_new)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_new.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
