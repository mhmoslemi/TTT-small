def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
