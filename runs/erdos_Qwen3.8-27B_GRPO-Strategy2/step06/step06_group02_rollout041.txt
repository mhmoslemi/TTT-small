def _project(h, n):
    """Project onto {0≤h≤1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Need to reduce: use water-filling from top
        # Find t such that sum(min(h_i, t)) = target
        # Sort h descending
        hs = np.sort(h)[::-1]
        cumsum = np.cumsum(hs)
        # We need to find where to "cut"
        # sum(min(h_i, t)) = sum_{h_i>t} t + sum_{h_i<=t} h_i
        # Binary search for t
        lo, hi = 0.0, hs[0]
        for _ in range(60):
            mid = (lo + hi) / 2
            smid = np.minimum(h, mid).sum()
            if smid > target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
        h = np.minimum(h, t)
    else:
        # Need to increase: use water-filling from bottom
        # Find t such that sum(max(h_i, t)) = target
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2
            smid = np.maximum(h, mid).sum()
            if smid < target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
        h = np.maximum(h, t)
    # Fix any tiny numerical errors
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-bound elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
            # Final fix
            diff2 = target - h.sum()
            if abs(diff2) > 1e-12:
                idx = np.argmax(1 - h) if diff2 > 0 else np.argmin(h)
                h[idx] += diff2
                h = np.clip(h, 0, 1)

def _grad(h, c, beta, n, top_k=0):

    cs = c.copy()

        w = np.zeros(n2)
        vals = c[idx]
        vals -= vals.max()
        w[idx] = np.exp(beta * vals)
        w /= w.sum()

        w /= w.sum()

            sp = -sh
            if sp < n:
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
    return g

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-10:
        return h
    if s > tgt:
        lo, hi = 0.0, h.max() + 1e-6
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.minimum(h, mid).sum() > tgt:
                lo = mid
            else:
                hi = mid
        h = np.minimum(h, (lo + hi) * 0.5)
    else:

            if np.maximum(h, mid).sum() < tgt:
                lo = mid
            else:
                hi = mid

    h = h.copy()

    lr = 0.15
    beta = beta0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        g = _grad_topk(h, c, beta, n, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        OK = False
        for _ in range(25):
            h_new = _proj(h - lr * g / gn, n)
            c_new = _eval(h_new)
            if c_new < cur - 1e-15:
                h = h_new
                cur = c_new
                OK = True
                break
            lr *= 0.5
        if not OK:
            beta = min(beta * 1.3, 500.0)
            if lr < 1e-6:
                break
            continue
        lr = min(lr * 1.2, 0.5)
        if it % 200 == 199:
            beta *= 1.1
    return h, cur

    h = _proj(h, n)
    cur = _eval(h)
    for rnd in range(max_rounds):
