                break
    return best_h, best_c

def _local_search(h, n_iter, rng, t0, budget):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() - t0 > budget - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = 8e-5 * rng.randn()
        h[i] += d
        h[j] -= d
        if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
            c = _eval(h)
            if c < c5c:
                c5c = c
                if c < best_c:
                    best_c = c
                    best = h.copy()
            else:
                h[i] -= d
                h[j] += d
        else:
            h[i] -= d
            h[j] += d
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=np.float64).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    n_low = 200
    fac = n0 // n_low
    starts = []
    for sh in [0, n0 // 6, n0 // 3, 2 * n0 // 5, 3 * n0 // 7]:
        hs = np.roll(h0, sh)
        hs = _repair(_downsample(hs, fac), n_low)
        starts.append(hs)
    starts.append(_repair(_downsample(1.0 - h0, fac), n_low))

    # Phase 1: Optimize each start at n_low
    for si in range(len(starts)):
        if time.time() - t0 > budget_s * 0.22:
            break
        hs = starts[si].copy()
        for alpha in [150, 1500, 15000]:
            if time.time() - t0 > budget_s * 0.25:
                break
            hs = _slsqp_opt(hs, alpha, 500, 1e-13, t0, budget_s)
            h_up = _repair(_upsample_lin(hs, n0), n0)
            c = _eval(h_up)
            if c < best_c5:
                best_c5 = c
                best_h = h_up.copy()
                best_n = n0
        if time.time() - t0 < budget_s * 0.25:
            hs = _nesterov_opt(hs, [3000, 15000], 150, 0.006, t0, budget_s)
            h_up = _repair(_upsample_lin(hs, n0), n0)
            c = _eval(h_up)
            if c < best_c5:
                best_c5 = c
                best_h = h_up.copy()
                best_n = n0

    # Phase 2: PGD at n_low on best
    if time.time() - t0 < budget_s * 0.35:
        h_l = _repair(_downsample(best_h, fac), n_low)
        h_l, _ = _pgd_topk(h_l, 5, 400, 0.01, t0, budget_s)
        h_up = _repair(_upsample_lin(h_l, n0), n0)
        c = _eval(h_up)
        if c < best_c5:
            best_c5 = c
            best_h = h_up.copy()
            best_n = n0

    # Phase 3: Medium resolution n=400
    if time.time() - t0 < budget_s * 0.55:
        n_med = 400
        h_med = _repair(_upsample_lin(best_h, n_med), n_med)
        for alpha in [800, 8000, 60000]:
            if time.time() - t0 > budget_s * 0.52:
                break
            h_med = _slsqp_opt(h_med, alpha, 700, 1e-13, t0, budget_s)
            c = _eval(h_med)
            if c < best_c5:
                best_c5 = c
                best_h = h_med.copy()
                best_n = n_med
        if time.time() - t0 < budget_s * 0.55:
            h_med = _nesterov_opt(h_med, [20000, 80000], 250, 0.003, t0, budget_s)
            c = _eval(h_med)
            if c < best_c5:
                best_c5 = c
                best_h = h_med.copy()
                best_n = n_med

    # Phase 4: PGD at medium res
    if time.time() - t0 < budget_s * 0.62:
        h_med = _repair(_upsample_lin(best_h, 400), 400)
        h_med, _ = _pgd_topk(h_med, 4, 350, 0.005, t0, budget_s)
        c = _eval(h_med)
        if c < best_c5:
            best_c5 = c
            best_h = h_med.copy()
            best_n = len(h_med)

    # Phase 5: Full resolution n0
    if time.time() - t0 < budget_s * 0.78:
        h_f = _repair(_upsample_lin(best_h, n0), n0)
        for alpha in [3000, 30000, 200000]:
            if time.time() - t0 > budget_s * 0.75:
                break
            h_f = _slsqp_opt(h_f, alpha, 1000, 1e-13, t0, budget_s)
            c = _eval(h_f)
            if c < best_c5:
                best_c5 = c
                best_h = h_f.copy()
                best_n = n0
        if time.time() - t0 < budget_s * 0.78:
            h_f = _nesterov_opt(h_f, [80000, 400000], 250, 0.002, t0, budget_s)
            c = _eval(h_f)
            if c < best_c5:
                best_c5 = c
                best_h = h_f.copy()
                best_n = n0

    # Phase 6: PGD at full res
    if time.time() - t0 < budget_s * 0.88:
        h_f, _ = _pgd_topk(best_h.copy(), 3, 300, 0.003, t0, budget_s)
        c = _eval(h_f)
        if c < best_c5:
            best_c5 = c
            best_h = h_f.copy()
            best_n = len(h_f)

    # Phase 7: Local search
    if time.time() - t0 < budget_s * 0.96:
        best_h, best_c5 = _local_search(best_h.copy(),
                                        min(int((budget_s - time.time() - 3) / 0.003), 50000),
                                        rng, t0, budget_s)
        best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
