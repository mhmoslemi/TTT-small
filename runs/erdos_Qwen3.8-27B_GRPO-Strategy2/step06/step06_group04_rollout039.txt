```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _sgrad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            od = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for idx in od:
                if d > 0:
                    a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
                else:
                    a = min(-d, h[idx]); h[idx] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _sgrad(x, alpha):
    n = len(x); dx = 2.0 / n
    c = _corr(x); m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

    return _smax(x, alpha) + pen * (x.sum() - n * 0.5) ** 2

    return _sgrad(x, alpha) + 2.0 * pen * (x.sum() - n * 0.5)

                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'gtol': 1e-14})
        return _repair(r.x, n)
    except Exception:

def _adam(h0, a0, a1, ni, lr0, lr1, rng, dl):
    n = len(h0); x = h0.copy()
    m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    bx = x.copy(); bc = _eval(x)
    for i in range(ni):
        if time.time() > dl: break
        t += 1; fr = i / max(ni - 1, 1)
        al = a0 * (a1 / a0) ** fr; lr = lr0 * (lr1 / lr0) ** fr
        g = _sgrad(x, al)
        m = b1*m + (1-b1)*g; v = b2*v + (1-b2)*g*g
        x = x - lr * (m/(1-b1**t)) / (np.sqrt(v/(1-b2**t)) + eps)
        x = _repair(x, n); c = _eval(x)
        if c < bc: bc = c; bx = x.copy()
        if t % 1500 == 0:

    return bx, bc
```

Actually, I also want to try a slightly different approach for the final polish: **alternating coordinate descent** on individual h[i] values, where at each step I find the optimal value of h[i] given all other h[j]'s fixed.

For a fixed h[j], j≠i, the objective as a function of h[i] is:
f(h[i]) = max_k (c_k - h[i]*a_k)*dx where a_k = ∂c_k/∂h[i]

Since each c_k is linear in h[i] (when everything else is fixed), f is the max of linear functions of h[i], which is a convex piecewise-linear function. The minimum can be found analytically!

For each cord k, c_k = A_k + h[i] * B_k (where A_k, B_k don't depend on h[i]).
f(t) = max_k (A_k + t * B_k) * dx

The minimum of this piecewise-linear convex function on [0,1] can be found by:
1. Evaluating f at 0 and 1
2. Finding all breakpoints (where the argmax changes)
3. The minimum is at a breakpoint or at 0 or 1

The breakpoints are where A_k1 + t*B_k1 = A_k2 + t*B_k2, i.e., t = (A_k1 - A_k2)/(B_k2 - B_k1).

But with 619 lags, there are O(619²) potential breakpoints, which is too many.

Simpler approach: use ternary search or golden section search on [0,1] to find the minimum of f(t). Since f is convex, this converges in O(log(1/eps)) steps.

```python
def _coord_descend(h, n, n_sweeps, dl):
    h = h.copy()
    best_c = _eval(h)
    for sweep in range(n_sweeps):
        if time.time() > dl: break
        improved = False
        for i in range(n):

            # Golden section search on h[i] in [0, 1]
            a, b = 0.0, 1.0

            else:

        if not improved: break
```

```python
import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            od = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for idx in od:
                if d > 0:
                    a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
                else:
                    a = min(-d, h[idx]); h[idx] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smax(x, alpha):
    c = _corr(x); m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _sgrad(x, alpha):
    n = len(x); dx = 2.0 / n
    c = _corr(x); m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

    return _smax(x, al) + pen * (x.sum() - n * 0.5) ** 2

def _pg(x, al, pen, n):
    return _sgrad(x, al) + 2.0 * pen * (x.sum() - n * 0.5)

    n = len(x0); try:
        r = minimize(_pf, x0, _pg, (al, 1e6, n),
                     'L-BFGS-B', [(0.0,1.0)]*n,
                     {'maxiter':mi,'ftol':1e-20,'gtol':1e-14})
        return _repair(r.x, n)
    except Exception: return _repair(x0, n)

    n = len(h0); x = h0.copy()
    m = np.zeros(n); v = np.zeros(n); t = 0

        t += 1; fr = i / max(ni - 1, 1)
        g = _sgrad(x, a0*(a1/a0)**fr)
        m = b1*m+(1-b1)*g; v = b2*v+(1-b2)*g*g
        x = x - l0*(l1/l0)**fr * m/(1-b1**t) / (np.sqrt(v/(1-b2**t))+eps)
        x = _repair(x, n); c = _eval(x)
        if c < bc: bc, bx = c, x.copy()
        if t % 1500 == 0: x += 0.003*rng.randn(n); x = _repair(x, n)
    return bx, bc
```

```python
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _pf(x, al, pen, n):
    return _smax(x, al) + pen * (x.sum() - n * 0.5) ** 2

def _pg(x, al, pen, n):
    return _sgrad(x, al) + 2.0 * pen * (x.sum() - n * 0.5)

def _lfgsb(x0, al, dl, mi=2000):
    n = len(x0)
    try:
        r = minimize(_pf, x0, _pg, (al, 1e6, n), 'L-BFGS-B',
                     [(0.0, 1.0)] * n,
                     {'maxiter': mi, 'ftol': 1e-20, 'gtol': 1e-14})
        return _repair(r.x, n)
    except Exception:
        return _repair(x0, n)

def _adam(h0, a0, a1, ni, l0, l1, rng, dl):
    n = len(h0); x = h0.copy()
    m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    bx, bc = x.copy(), _eval(x)
    for i in range(ni):
        if time.time() > dl: break
        t += 1; fr = i / max(ni - 1, 1)
        g = _sgrad(x, a0 * (a1 / a0) ** fr)
        m = b1*m + (1-b1)*g; v = b2*v + (1-b2)*g*g
        x = x - l0*(l1/l0)**fr * m/(1-b1**t) / (np.sqrt(v/(1-b2**t)) + eps)
        x = _repair(x, n); c = _eval(x)
        if c < bc: bc, bx = c, x.copy()
        if t % 1500 == 0:
            x += 0.003 * rng.randn(n); x = _repair(x, n)
    return bx, bc

def _resample(h, nn):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    return _repair(np.interp(np.linspace(0, 2, nn, endpoint=False), xo, h), nn)

def _bils(h0, n, rng, ni, dl):
    x = _repair((h0 > 0.5).astype(float), n)
    cc = _eval(x); bx, bc = x.copy(), cc
    for _ in range(ni):
        if time.time() > dl: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j or x[i] == x[j]: continue
        x[i], x[j] = x[j], x[i]; cn = _eval(x)
        if cn < cc: cc = cn
        if cn < bc: bc, bx = cn, x.copy()
        else: x[i], x[j] = x[j], x[i]
    return bx, bc

def _pcd(h0, n, rng, ni, dl):
    h = h0.copy(); bc = _eval(h); gr = 0.6180339887
    for _ in range(ni):
        if time.time() > dl: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        hi, hj = h[i], h[j]
        a = max(-hi, hj - 1.0)
        b = min(1.0 - hi, hj)
        if b - a < 1e-12: continue
        aa, bb = a, b
        for _g in range(20):
            t1 = aa + (1-gr)*(bb-aa); t2 = aa + gr*(bb-aa)
            h[i] = hi+t1; h[j] = hj-t1; f1 = _eval(h)
            h[i] = hi+t2; h[j] = hj-t2; f2 = _eval(h)
            if f1 < f2: bb = t2
            else: aa = t1
        to = (aa + bb) * 0.5
        h[i] = hi + to; h[j] = hj - to
        f = _eval(h)
        if f < bc: bc = f
    h = _repair(h, n)
    return h, _eval(h)

def _subg(h0, n, dl):
    x = h0.copy(); bx, bc = x.copy(), _eval(x)
    lr = 0.002; K = 5
    for _ in range(8000):
        if time.time() > dl: break
        c = _corr(x); idx = np.argsort(c)[-K:]
        g = np.zeros(n)
        for km in idx:
            sh = (n - 1) - km
            if sh >= 0:
                g[:n-sh] += (1.0 - x[sh:])
                if sh > 0: g[sh:] -= x[:n-sh]
            else:
                sp = -sh; g[sp:] += (1.0 - x[:n-sp]); g[:n-sp] -= x[sp:]
        g /= K
        x = _repair(x - lr * g, n); cv = _eval(x)
        if cv < bc: bc, bx = cv, x.copy()
        else: lr *= 0.98
        lr = max(lr, 1e-5)
    return bx, bc

def _per(T, nt, rng):
    half = T // 2; bp = None; bm = np.inf; p = np.zeros(T)
    for cb in combinations(range(T), half):
        p[:] = 0; p[list(cb)] = 1
        cc = np.real(np.fft.ifft(np.fft.fft(p) * np.conj(np.fft.fft(1.0 - p))))
        cc[cc < 0] = 0
        if cc.max() < bm - 1e-12: bm = cc.max(); bp = p.copy()
    nu = (nt // T) * T
    if nu < T or nu % 2: nu = max(T, (nu // 2) * 2)
    return _repair(np.tile(bp, nu // T), nu), nu

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh, bc, bn = h0.copy(), _eval(h0), n0
    # Phase 1: periodic patterns (0-55s)
    dl1 = t0 + 55
    for T in [10, 12, 14, 16, 18, 20, 22]:
        if time.time() > dl1: break
        hp, npf = _per(T, n0 + 20, rng)
        cp = _eval(hp)
        if cp < bc: bc, bh, bn = cp, hp, npf
    # Phase 2: L-BFGS-B multi-alpha (55-260s)
    dl2 = t0 + 260
    for al in [500, 5000, 50000, 500000]:
        if time.time() > dl2: break
        hn = _lfgsb(bh, al, dl2, 2000)
        cp = _eval(hn)
        if cp < bc: bc, bh = cp, hn
    # Phase 3: Adam multi-scale (260-440s)
    dl3 = t0 + 440
    for a0, a1, ni, l0, l1 in [(500, 5000, 3000, 0.01, 0.001),
                                (5e3, 5e4, 3000, 0.005, 5e-4),
                                (5e4, 5e5, 3000, 0.003, 3e-4)]:
        if time.time() > dl3: break
        x, c = _adam(bh, a0, a1, ni, l0, l1, rng, dl3)
        if c < bc: bc, bh = c, x
    # Phase 4: binary LS + fractional polish (440-570s)
    dl4 = t0 + 570
    xb, cb = _bils(bh, bn, rng, 80000, dl4)
    if cb < bc: bc, bh = cb, xb
    hn = _lfgsb(bh, 10000, dl4, 1500)
    cp = _eval(hn)
    if cp < bc: bc, bh = cp, hn
    # Phase 5: paired CD with golden section (570-730s)
    dl5 = t0 + 730
    bh, bc = _pcd(bh, bn, rng, 50000, dl5)
    # Phase 6: subgradient + high-alpha L-BFGS-B (730-870s)
    dl6 = t0 + 870
    xs, cs = _subg(bh, bn, dl6)
    if cs < bc: bc, bh = cs, xs
    hn = _lfgsb(bh, 1e6, dl6, 3000)
    cp = _eval(hn)
    if cp < bc: bc, bh = cp, hn
    # Phase 7: multi-n exploration (870-990s)
    dl7 = t0 + 990
    for dn in [-20, -10, 10, 20, 30]:
        if time.time() > dl7: break
        nt = bn + dn
        if nt < 100 or nt > 500 or nt % 2: continue
        hr = _resample(bh, nt)
        hn = _lfgsb(hr, 10000, dl7, 1000)
        x, c = _adam(hn, 1e4, 1e5, 1500, 0.003, 3e-4, rng, dl7)
        if c < bc: bc, bh, bn = c, x, nt
    # Phase 8: final paired CD + L-BFGS-B polish (990-1085s)
    dl8 = t0 + 1085
    bh, bc = _pcd(bh, bn, rng, 40000, dl8)
    if time.time() < dl8 - 30:
        hn = _lfgsb(bh, 5e6, dl8, 2000)
        cp = _eval(hn)
        if cp < bc: bc, bh = cp, hn
    bh = _repair(bh, bn); bc = _eval(bh)
    return (bh, bc, bn)
```
