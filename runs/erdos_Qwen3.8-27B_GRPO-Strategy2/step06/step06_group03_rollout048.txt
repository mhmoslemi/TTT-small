import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_one(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        if sh < n:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        if sp < n:
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g

def _smoothmax_f(h, alpha, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    cv = corr[idx]
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smoothmax_g(h, alpha, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    cv = corr[idx]
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for j, k in enumerate(idx):
        if w[j] < 1e-15:
            continue
        sh = (n - 1) - k
        g += w[j] * _grad_one(h, sh, n)
    return g

def _slsqp(h, alpha, K, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x): return _smoothmax_f(x, alpha, K)
    def _g(x): return _smoothmax_g(x, alpha, K)
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _local_search(h, t_end, rng):

    if time.time() - t0 < budget_s - 250:
        n_it = min(6000, int((budget_s - 250 - (time.time() - t0)) / 0.018))
        h2, c2 = _nesterov(best_h, max(n_it, 1000), 20, 0.012, 0.92, rng)
        if c2 < best_c:
            best_c = c2; best_h = h2

    # Phase 3: SLSQP high alpha
    for alpha, K in [(30000, 150), (100000, 300)]:
        if time.time() - t0 > budget_s - 180:
            break
        hc = _slsqp(best_h, alpha, K, 1500, 1e-15)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 4: Nesterov again with more lags for fine-tuning
    if time.time() - t0 < budget_s - 120:
        h3 = _repair(best_h + 0.003 * rng.randn(best_n), best_n)
        n_it = min(3000, int((budget_s - 120 - (time.time() - t0)) / 0.018))
        h3, c3 = _nesterov(h3, max(n_it, 500), 40, 0.005, 0.95, rng)
        if c3 < best_c:
            best_c = c3; best_h = h3

    # Phase 5: Final SLSQP squeeze
    if time.time() - t0 < budget_s - 60:
        hc = _slsqp(best_h, 500000, 500, 2000, 1e-16)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 6: Local pairwise search
    if time.time() - t0 < budget_s - 10:
        best_h, best_c = _local_search(best_h, budget_s - 5, rng)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
