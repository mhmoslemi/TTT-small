import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    c = np.correlate(h, 1.0 - h, mode='full')
    o = c * 2.0 / n
    return o, o.max()

def _ls_obj(x, hot_idx, lam):
    # Objective: weighted sum of overlaps at hot shifts + constraint penalty
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    o = c * 2.0 / n
    # Sum of squared excess at hot shifts
    excess = o[hot_idx] - 0  # we want all o <= C, so penalize o directly
    val = np.sum(excess ** 2)
    # Constraint penalty
    val += lam * (x.sum() - n / 2.0) ** 2
    return val

def _ls_grad(x, hot_idx, lam):

    c = np.correlate(x, 1.0 - x, mode='full')
    o = c * 2.0 / n
    g = np.zeros(n)
    dx = 2.0 / n

        # gradient of o[k_idx] w.r.t. x
        sh = k_shift
        if sh >= 0:

            g[i1] += 2 * o[k_idx] * dx * (1.0 - x[i1 + sh])
            if sh > 0:

                lr *= 0.7
                if lr < 1e-7:

                    mom = np.zeros(n)
                    break

def _coord_descent(h, n, n_iter, scales, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter // len(scales)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:

            else:

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    idx_sub = (np.arange(T)[:, None] - np.arange(T)[None, :]) % T
    idx_add = (np.arange(T)[:, None] + np.arange(T)[None, :]) % T
    term1 = np.sum(w[None, :] * (1.0 - p[idx_sub]), axis=1)
    term2 = np.sum(w[None, :] * p[idx_add], axis=1)
    return (term1 - term2) * (2.0 / T)

def _pgd_momentum(h0, n, alphas, total_iters, lr_init=0.1, seed=0):
    h = h0.copy()
    best = h.copy()
    best_c = _eval(h)
    iters_per = total_iters // len(alphas)
    for alpha in alphas:
        lr = lr_init
        mom = np.zeros(n)
        beta = 0.88
        for it in range(iters_per):
            g = _lse_grad(h, alpha)
            mom = beta * mom + (1.0 - beta) * g
            h_new = _proj_sum(h - lr * mom, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h = h_new
                lr = min(lr * 1.015, 3.0)
            else:
                lr *= 0.65
                if lr < 1e-7:
                    lr = lr_init
                    mom = np.zeros(n)
    return best, best_c

def _slsqp_opt(h0, alpha, maxiter=2000, ftol=1e-18):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_lse_obj, h0, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _coord_descent(h, n, n_iter, scales, seed=0):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    per_scale = n_iter // len(scales)
    for scale in scales:
        for _ in range(per_scale):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: Projected gradient with momentum (fast, good initial progress)
    if elapsed() < budget_s * 0.30:
        alphas_pgd = [2000, 10000, 50000, 200000, 800000]
        h1, c1 = _pgd_momentum(best_h, n0, alphas_pgd, 5000, lr_init=0.08, seed=seed)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: SLSQP on full signal with increasing alpha
    if elapsed() < budget_s * 0.60:
        for alpha in [50000, 200000, 800000, 3000000]:
            if elapsed() > budget_s * 0.60:
                break
            h2 = _slsqp_opt(best_h, alpha, 1500, 1e-20)
            c2 = _eval(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
                best_n = len(h2)

    # Phase 3: Periodic pattern T=31 with CORRECT pattern gradient
    if best_n % 31 == 0 and elapsed() < budget_s * 0.78:
        m = best_n // 31
        h_p = best_h.reshape(m, 31).mean(axis=0)
        h_p = _proj_sum(h_p, 31)
        bnds_p = [(0.0, 1.0)] * 31
        cons_p = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
        for alpha in [50000, 300000, 1500000, 8000000]:
            if elapsed() > budget_s * 0.78:
                break
            try:
                res = minimize(_pat_obj, h_p, jac=_pat_grad, args=(alpha,),
                               method='SLSQP', bounds=bnds_p, constraints=cons_p,
                               options={'maxiter': 3000, 'ftol': 1e-20})
                h_p_new = _repair(res.x, 31)
                c3 = _eval(np.tile(h_p_new, m))
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = np.tile(h_p_new, m).copy()
                    best_n = m * 31
                    h_p = h_p_new
                else:
                    h_p = h_p_new  # still use improved pattern for next alpha
            except Exception:
                pass

    # Phase 4: Random perturbation restarts to escape local minima
    if elapsed() < budget_s * 0.88:
        for trial in range(3):
            if elapsed() > budget_s * 0.88:
                break
            h_pert = best_h + 0.03 * rng.randn(best_n)
            h_pert = _repair(h_pert, best_n)
            h_pert = _slsqp_opt(h_pert, 300000, 800, 1e-18)
            c4 = _eval(h_pert)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h_pert.copy()
                best_n = len(h_pert)

    # Phase 5: Coordinate descent polish (multiple scales)
    if elapsed() < budget_s - 20:
        scales = [0.002, 0.0005, 0.0001]
        h5, c5 = _coord_descent(best_h.copy(), best_n, 8000, scales, seed=seed + 7)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h5.copy()

    # Phase 6: Try nearby n values
    if elapsed() < budget_s - 40:
        for dn in [10, -10, 20, -20]:
            if elapsed() > budget_s - 35:
                break
            n_try = best_n + dn
            if n_try < 100 or n_try > 600:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _slsqp_opt(h_r, 200000, 1000, 1e-18)
            c_try = _eval(h_r)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_r.copy()
                best_n = n_try

    # Final polish
    if elapsed() < budget_s - 5:
        scales = [0.0003, 0.0001, 0.00003]
        h_f, c_f = _coord_descent(best_h.copy(), best_n, 3000, scales, seed=seed + 99)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
