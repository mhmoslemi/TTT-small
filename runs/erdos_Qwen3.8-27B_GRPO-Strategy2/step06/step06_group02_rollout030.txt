def _eval(h):

    return float(np.max(np.correlate(h, 1-h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    v = np.zeros(n)
    t = 1.0
    best = _eval(x)
    best_x = x.copy()
    no_improve = 0

    """Return the full correlation vector (scaled)."""
    n = len(h)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)

                h[idx] -= a

            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if cur < best_c:

                best_x = x.copy()
        else:
            x[i] = 1.0
            x[j] = 0.0

def _reweight_descend(h, n_iter, top_k, t_end):
    """Gradient descent weighted by top-k worst shifts."""
    n = len(h)
    x = h.copy()
    cur = _eval(x)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(x, 1 - x, mode='full') * 2.0 / n
        top_idx = np.argsort(c)[-top_k:]
        cmin = c[top_idx].min()
        cmax = c[top_idx].max()
        if cmax - cmin < 1e-14:
            w = np.ones(top_k) / top_k
        else:
            w = np.exp(5.0 * (c[top_idx] - cmin) / (cmax - cmin))
            w /= w.sum()
        
        g = np.zeros(n)
        for kk in range(len(top_idx)):
            shift = (n - 1) - top_idx[kk]
            g += w[kk] * _grad_shift(x, shift)
        
        for lr in [0.05, 0.02, 0.01, 0.005, 0.002]:
            x_new = _repair(x - lr * g, n)
            c_new = _eval(x_new)
            if c_new < cur - 1e-14:
                x = x_new
                cur = c_new
                break
    
    return x, cur

def _eval(h):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])

        return budget_s - (time.time() - t0)
    try:
        h_init = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    except Exception:
        h_init = np.full(500, 0.5)
        h_init[:250] = 1.0
        h_init = _repair(h_init, 500)
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    # Phase 1: Quick binary greedy
    if rem() > 80:
        hb, cb = _binary_greedy_limited(best_h, 2000, t0 + rem() - 20)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
    # Phase 2: Nesterov subgradient (main)
    if rem() > 300:
        hs, cs = _accel_subgrad(best_h, t0 + rem() * 0.35, lr0=0.08)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()
    # Phase 3: Reweighted descent (top-5)
    if rem() > 200:
        hr, cr = _reweight_descend(best_h, 8000, 5, t0 + rem() * 0.3)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
    # Phase 4: Binary simulated annealing
    if rem() > 150:
        ha, ca = _binary_sa(best_h, t0 + rem() * 0.35, t_max=0.02, t_min=0.00003)
        if ca < best_c5:
            best_c5 = ca
            best_h = ha.copy()
    # Phase 5: Multi-resolution exploration
    if rem() > 120:
        for n_try in [300, 400, 600, 750]:
            if rem() < 80:
                break
            hr = _repair(_resample(best_h, n_try), n_try)
            hr2, cr2 = _accel_subgrad(hr, t0 + min(rem() - 20, 35), lr0=0.05)
            if cr2 < best_c5:
                best_c5 = cr2
                best_h = hr2.copy()
                best_n = n_try
                n0 = n_try
    # Phase 6: Reweight top-3
    if rem() > 80:
        h6, c6 = _reweight_descend(best_h, 5000, 3, t0 + rem() - 30)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()
    # Phase 7: Final binary SA (short)
    if rem() > 50:
        h7, c7 = _binary_sa(best_h, t0 + rem() - 10, t_max=0.005, t_min=0.00001)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7.copy()
    # Phase 8: Final subgradient polish
    if rem() > 20:
        h8, c8 = _accel_subgrad(best_h, t0 + rem() - 3, lr0=0.02)
        if c8 < best_c5:
            best_c5 = c8
            best_h = h8.copy()
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
