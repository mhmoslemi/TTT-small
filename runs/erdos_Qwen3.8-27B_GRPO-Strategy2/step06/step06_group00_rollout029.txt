            if sh == 0: g += wk*(1.0-2.0*h)
            else:
                g[:n-sh] += wk*(1.0-h[sh:])
                g[sh:] -= wk*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    return f, g

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    def _f(x):
        f,_ = _fg(x, alpha)
        return f + 500*(x.sum()-n*0.5)**2
    def _g(x):
        _,gr = _fg(x, alpha)
        gr = gr.copy()
        gr += 1000*(x.sum()-n*0.5)
        return gr
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _nesterov(h, alphas, steps_per_alpha, step0):
    n = len(h); best = h.copy(); best_f = _eval(h)
    for al in alphas:
        h = best.copy()
        v = np.zeros(n)
        mu = 0.83; step = step0
        for _s in range(steps_per_alpha):
            _, g = _fg(h, al)
            h_prev = h.copy()
            v = mu*v + (1-mu)*g
            h_new = _proj(h - step*v, n*0.5)
            f_new = _eval(h_new)
            if f_new < _eval(h):
                h = h_new
                step = min(step*1.02, step0*5)
            else:
                step *= 0.5
                h = _proj(h_prev - step*v, n*0.5)
        best = _repair(h, n)
        best_f = _eval(best)
    return best
