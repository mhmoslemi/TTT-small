A key trick: **anti-symmetric structure**. If h(x) is structured so that h(x) + h(x+1) ≈ 1 for all x, then the overlap at shift 1 is ≈ 0. But we can't satisfy this for all shifts simultaneously.

E[h(X)] - E[h(X)h(X+k)] = 0.5 - R(h)(k)

where R(h)(k) is the autocorrelation at lag k.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.5, 2.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _obj_lse(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_weighted(x, w, dx):
    n = len(x)
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:sp + cnt] += wk * (1.0 - x[:cnt])
            g[:cnt] -= wk * x[sp:sp + cnt]
    return g

def _grad_lse(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    w = np.exp(s)
    w /= w.sum()
    return _grad_weighted(x, w, dx)

def _grad_topk(x, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / top_k
    return _grad_weighted(x, w, dx)

def _cf(x):
    return x.sum() - len(x) * 0.5

def _cj(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj_lse, h, jac=_grad_lse, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, steps, step0):
    n = len(h)
    f = _obj_lse(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        g = _grad_lse(h, alpha)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _obj_lse(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.05, step0 * 4.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _obj_lse(h, alpha)
    return _repair(h, n)
