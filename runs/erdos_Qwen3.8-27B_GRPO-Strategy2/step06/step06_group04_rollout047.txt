import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - h, mode='full') if False else np.correlate(x, 1.0 - x, mode='full')

    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxiter, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha,), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h0, alphas, n_per, lr, seed, deadline):
    n = len(h0); rng = np.random.RandomState(seed)
    x = h0.copy().astype(float); m = np.zeros(n); v = np.zeros(n); t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x, best_c = x.copy(), _eval(x)
    for alpha in alphas:
        for i in range(n_per):
            if time.time() > deadline:
                return best_x, best_c
            t += 1
            g = _grad(x, alpha)
            m = b1*m + (1-b1)*g; v = b2*v + (1-b2)*g*g
            mh = m/(1-b1**t); vh = v/(1-b2**t)
            x = _repair(x - lr*mh/(np.sqrt(vh)+eps), n)
            c = _eval(x)
            if c < best_c: best_c, best_x = c, x.copy()
            if t % 800 == 0:
                x = _repair(x + 0.003*rng.randn(n), n)
    return best_x, best_c

def _resample(h, n_new):
    n_old = len(h)
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, n_old, endpoint=False), h)

def _binary_ls(h0, n, rng, n_iters, deadline):
    x = (h0 > 0.5).astype(float); x = _repair(x, n)
    c_cur = _eval(x); best_x, best_c, stuck = x.copy(), c_cur, 0
    for _ in range(n_iters):
        if time.time() > deadline or stuck > 8000:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j or x[i] == x[j]: continue
        x[i], x[j] = 1.0-x[i], 1.0-x[j]
        c_new = _eval(x)
        if c_new < c_cur - 1e-15:
            c_cur = c_new; stuck = 0
            if c_new < best_c: best_c, best_x = c_new, x.copy()
        else:
            x[i], x[j] = 1.0-x[i], 1.0-x[j]; stuck += 1
    return best_x, best_c

def _pair_cd(h, n, rng, deadline, s0):
    x = h.copy(); c_cur = _eval(x)
    best_x, best_c, scale = x.copy(), c_cur, s0
    for _ in range(500000):
        if time.time() > deadline or scale < 1e-8: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue

            if c_new < best_c: best_c, best_x = c_new, x.copy()
            scale = max(scale*0.998, 1e-8)

    return best_x, best_c
