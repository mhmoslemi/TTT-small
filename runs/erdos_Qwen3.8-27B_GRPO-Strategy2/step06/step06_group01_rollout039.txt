def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        return 0.0
    else:

        return 0.0

import time

    target = n / 2.0

    if s < 1e-12:
        return np.full(n, 0.5)

            for idx in order:
                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _projected_nesterov(h, n, alpha, max_iters, t0, budget_s):
    """Nesterov-accelerated projected gradient for LSE approximation."""
    dx = 2.0 / n
    c5 = _eval(h)
    y = h.copy()
    t = 1.0
    step = 1.0 / (alpha * 2.0)  # initial step size
    
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 30:
            break
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new if t_new > 0 else 0.0
        y = y + beta * (y - h)
        
        # Gradient at y (with repair)
        y_r = _repair(y, n)
        c = np.correlate(y_r, 1.0 - y_r, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        
        # Gradient
        g = np.zeros(n)
        for k_idx in range(2*n-1):
            if w[k_idx] < 1e-14:
                continue
            sh = (n-1) - k_idx
            if sh >= 0:
                if sh < n:
                    g[:n-sh] += w[k_idx] * (1.0 - y_r[sh:])
                    if sh > 0:
                        g[sh:] -= w[k_idx] * y_r[:n-sh]
            else:
                sp = -sh
                if sp < n:
                    g[sp:] += w[k_idx] * (1.0 - y_r[:n-sp])
                    g[:n-sp] -= w[k_idx] * y_r[sp:]
        
        # Gradient step
        h_new = _repair(h - step * g, n)
        
        # Check improvement
        c5_new = _eval(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            step *= 0.5
            if step < 1e-10:
                break
        
        t = t_new
    
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, shifts, dx, n):
    g = np.zeros(n)
    for sh in shifts:
        if sh >= 0:
            if sh < n:
                g[:n-sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            if sp < n:
                g[sp:] += (1.0 - h[:n-sp])
                g[:n-sp] -= h[sp:]
    return g * dx

def _nesterov(h, n, top_k, max_iters, deadline):
    dx = 2.0 / n
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    x = h.copy()
    x_prev = h.copy()
    t = 1.0
    lr = 0.03
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(c)[-top_k:]
        shifts = (n - 1) - top_idx
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        beta = (t - 1.0) / t_new if t_new > 0 else 0.0
        y = x + beta * (x - x_prev)
        y = _repair(y, n)
        g = _topk_grad(y, shifts, dx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        x_new = _repair(x - lr * g / (gn * top_k), n)
        c5_new = _eval(x_new)
        if c5_new < c5 - 1e-15:
            x_prev = x.copy()
            x = x_new
            c5 = c5_new
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
        else:
            lr *= 0.6
            stall += 1
            if stall > 100:
                break
        if it % 200 == 199 and lr < 0.03:
            lr = min(lr * 2.0, 0.03)
        t = t_new
    return best_h, best_c5

def _lsexp_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
    m = c.max()
    s = np.clip(alpha*(c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s))))/alpha

def _lsexp_grad(x, alpha):
    n = len(x)
    dx = 2.0/n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha*(c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k_idx in range(2*n-1):
        if w[k_idx] < 1e-12:
            continue
        sh = (n-1) - k_idx
        wk = w[k_idx]*dx
        if sh >= 0:
            if sh < n:
                g[:n-sh] += wk*(1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk*x[:n-sh]
        else:
            sp = -sh
            if sp < n:
                g[sp:] += wk*(1.0 - x[:n-sp])
                g[:n-sp] -= wk*x[sp:]
    return g

def _sum_eq(x):
    return x.sum() - len(x)*0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, deadline):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':_sum_eq,'jac':_sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter, 'ftol':1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _fourier_obj(coeffs, n, K):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for j in range(1, K+1):
        h += coeffs[2*(j-1)]*np.cos(2.0*np.pi*j*x/n)
        h += coeffs[2*(j-1)+1]*np.sin(2.0*np.pi*j*x/n)
    h = _repair(h, n)
    return _eval(h)

def _fourier_extract(h, K):
    n = len(h)
    x = np.arange(n)
    hc = h - 0.5
    coeffs = np.zeros(2*K)
    for j in range(1, K+1):
        coeffs[2*(j-1)] = (2.0/n)*np.sum(hc*np.cos(2.0*np.pi*j*x/n))
        coeffs[2*(j-1)+1] = -(2.0/n)*np.sum(hc*np.sin(2.0*np.pi*j*x/n))
    return coeffs

def _fourier_recon(coeffs, K, n):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for j in range(1, K+1):
        h += coeffs[2*(j-1)]*np.cos(2.0*np.pi*j*x/n)
        h += coeffs[2*(j-1)+1]*np.sin(2.0*np.pi*j*x/n)
    return _repair(h, n)

def _fourier_sa(h, n, K, rng, deadline, n_iter=300):
    best_c = _fourier_extract(h, K).copy()
    best_c5 = _eval(h)
    h_best = h.copy()
    temp = 0.015
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_new = best_c + rng.randn(2*K)*temp
        h_new = _fourier_recon(c_new, K, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c = c_new.copy()
            best_c5 = c5_new
            h_best = h_new.copy()
        elif rng.random() < np.exp(-(c5_new-best_c5)/max(temp*0.005, 1e-12)):
            best_c = c_new.copy()
        temp *= 0.97
    return h_best, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _coord_desc(h, n, rng, deadline):
    c5 = _eval(h)
    for scale in [0.006, 0.003, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        for _ in range(1500):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale*rng.randn()
            ni, nj = h[i]+d, h[j]-d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-16:
                c5 = c5n
            else:
                h[i], h[j] = ni-d, nj+d
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, shifts, dx, n):
    g = np.zeros(n)
    for sh in shifts:
        if sh >= 0:
            if sh < n:
                g[:n-sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            if sp < n:
                g[sp:] += (1.0 - h[:n-sp])
                g[:n-sp] -= h[sp:]
    return g * dx

def _nesterov(h, n, top_k, max_iters, deadline):
    dx = 2.0 / n
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    x = h.copy()
    x_prev = h.copy()
    t = 1.0
    lr = 0.03
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(c)[-top_k:]
        shifts = (n - 1) - top_idx
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = x + beta * (x - x_prev)
        y = _repair(y, n)
        g = _topk_grad(y, shifts, dx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        x_new = _repair(x - lr * g / (gn * top_k), n)
        c5_new = _eval(x_new)
        if c5_new < c5 - 1e-15:
            x_prev = x.copy()
            x = x_new
            c5 = c5_new
            t = t_new
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
        else:
            lr *= 0.6
            stall += 1
            if stall > 100:
                break
        if it % 200 == 199 and lr < 0.03:
            lr = min(lr * 2.0, 0.03)
    return best_h, best_c5

    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

        for K in [8, 15, 25, 40]:
            if time.time() > deadline - 60:
                break
            h_f, c5_f = _fourier_sa(best_h, n0, K, rng, deadline - 40, 400)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, h_f.copy()
            if K <= 20:
                try:
                    co = _fourier_extract(best_h, K)
                    nm = minimize(_fourier_obj, co, args=(n0, K), method='Nelder-Mead',
                                 options={'maxiter': 1500, 'xatol': 1e-9, 'fatol': 1e-12})
                    h_nm = _fourier_recon(nm.x, K, n0)
                    c5_nm = _eval(h_nm)
                    if c5_nm < best_c5:
                        best_c5, best_h = c5_nm, h_nm.copy()
                except Exception:
                    pass

    # Phase 6: Nesterov top-5
    if time.time() < deadline - 200:
        h6, c5 = _nesterov(best_h, n0, 5, 2500, deadline - 150)
        if c5 < best_c5:
            best_c5, best_h = c5, h6.copy()

    # Phase 7: Perturbation restarts + binary start
    if time.time() < deadline - 200:
        for _ in range(3):
            if time.time() > deadline - 80:
                break
            noise = rng.randn(n0) * 0.005
            h_p = _repair(best_h + noise, n0)
            h_p, c5 = _nesterov(h_p, n0, 2, 1500, deadline - 50)
            if c5 < best_c5:
                best_c5, best_h = c5, h_p.copy()
        if time.time() < deadline - 100:
            h_bin = np.zeros(n0)
            idx = rng.choice(n0, n0 // 2, replace=False)
            h_bin[idx] = 1.0
            h_bin, c5 = _nesterov(h_bin, n0, 2, 1000, deadline - 50)
            h_bin = _slsqp(h_bin, 10000, 200, deadline)
            c5 = _eval(h_bin)
            if c5 < best_c5:
                best_c5, best_h = c5, h_bin.copy()

    # Phase 8: Resample to different n
    if time.time() < deadline - 150:
        for n_try in [310, 450, 500, 800, 1000]:
            if time.time() > deadline - 50:
                break
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r, c5_r = _nesterov(h_r, n_try, 3, 1500, deadline - 30)
            h_r = _slsqp(h_r, 15000, 300, deadline)
            c5_r = _eval(h_r)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r.copy()

    # Phase 9: Coordinate descent
    if time.time() < deadline - 30:
        h8, c5 = _coord_desc(best_h.copy(), len(best_h), rng, deadline - 15)
        if c5 < best_c5:
            best_c5, best_h = c5, h8.copy()

    # Phase 10: Final SLSQP + Nesterov
    if time.time() < deadline - 10:
        h_fin = _slsqp(best_h, 30000, 300, deadline)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()
        if time.time() < deadline - 2:
            h10, c5 = _nesterov(best_h, len(best_h), 2, 1000, deadline)
            if c5 < best_c5:
                best_c5, best_h = c5, h10.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
