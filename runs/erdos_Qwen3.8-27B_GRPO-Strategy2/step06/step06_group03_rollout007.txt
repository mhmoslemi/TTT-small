    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_f, h, jac=_slsqp_g, method='SLSQP',
                      bounds=bnds, constraints=[{'type':'eq','fun':_slsqp_cf,'jac':_slsqp_cg}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

_sg_n = 0
_sg_alpha = 0
_sg_ntop = 100

def _slsqp_f(x):
    c = np.correlate(x, 1.0-x, mode='full') * 2.0 / _sg_n
    m = c.max()
    s = np.clip(_sg_alpha * (c - m), -500, 500)
    return m + np.sum(np.exp(s)) / _sg_alpha

def _slsqp_g(x):
    c = np.correlate(x, 1.0-x, mode='full') * 2.0 / _sg_n
    w = _soft_w(c, _sg_alpha)
    if _sg_ntop < len(w):
        idx = np.argsort(w)[::-1][_sg_ntop:]
        w[idx] = 0
    return _grad_w(c, x, w, _sg_n)

def _slsqp_cf(x):
    return np.array([x.sum() - 0.5 * _sg_n])

def _slsqp_cg(x):
    return np.array([np.ones(_sg_n)])
