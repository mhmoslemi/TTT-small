    m = c.max()

def _subgrad_nesterov(h, n_iter, step0, top_k, beta0=0.9):
    n = len(h)
    x = h.copy()
    best = x.copy()
    best_c = _eval(best)
    v = np.zeros(n)
    beta = beta0
    step = step0
    stuck = 0
    for it in range(n_iter):
        c = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(c)[-top_k:]
        dx = 2.0 / n
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += (1.0 - x[sh:]) * dx
                if sh > 0:
                    g[sh:] -= x[:n-sh] * dx
            else:
                sp = -sh
                g[sp:] += (1.0 - x[:n-sp]) * dx
                g[:n-sp] -= x[sp:] * dx
        # Nesterov
        y = x + beta * v
        # Recompute gradient at y (projected)
        y = np.clip(y, 0, 1)
        c2 = np.correlate(y, 1.0 - y, mode='full')
        idx2 = np.argsort(c2)[-top_k:]
        g2 = np.zeros(n)
        for k in idx2:
            sh = (n - 1) - k
            if sh >= 0:
                g2[:n-sh] += (1.0 - y[sh:]) * dx
                if sh > 0:
                    g2[sh:] -= y[:n-sh] * dx
            else:
                sp = -sh
                g2[sp:] += (1.0 - y[:n-sp]) * dx
                g2[:n-sp] -= y[sp:] * dx
        v = beta * v + g2
        x_new = _proj(x - step * v, 0.5 * n)
        c_new = _eval(x_new)
        if c_new < best_c - 1e-15:
            best = x_new.copy()
            best_c = c_new
            stuck = 0
            step = min(step * 1.05, 0.5)
        else:
            stuck += 1
            step *= 0.7
            if stuck > 80:
                x = _repair(x + 0.02 * np.random.randn(n), n)
                v = np.zeros(n)
                step = step0
                stuck = 0
        beta = min(beta + 1.0 / (it + 25.0), 0.95)
        x = x_new
    return best, best_c

def _grad_pair_search(h, n_tries, rng, top_k=20):

def _fft_smooth(h, bandwidth):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return m + np.log(np.sum(np.exp(s))) / alpha

        else:

                       method='SLSQP', bounds=bnds, constraints=cons,

        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _con_fun(x):
    return np.array([x.sum() - 0.5 * len(x)])

def _con_jac(x):
    return np.array([np.ones(len(x))])

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _wgrad(h, top_k=100):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    w = np.exp(150.0 * (c - m))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])

    return g

def _momentum_subgrad(h, n_iter, step0, top_k, rng):

    x = h.copy()

    best_c = _eval(x)
    v = np.zeros(n)

                best_c = _eval(x)
                step = step0
                stuck = 0
                v = np.zeros(n)
        mu = min(mu + 0.0005, 0.92)

    n = len(h)
    x = h.copy()
    best = x.copy()

    cur_c = best_c
    g = _wgrad(x, 100)
    batch = 150

        i = pos[rng.randint(len(pos))]

        if i == j:
            continue
        d = 4e-5 * rng.randn()
        x[i] += d
        x[j] -= d

            x[i] -= d
            x[j] += d
            continue
        cn = _eval(x)

            cur_c = cn
            if cn < best_c:
                best_c = cn
                best = x.copy()
        else:
            x[i] -= d
            x[j] += d
    return best, best_c

def _fft_smooth(h, bw):
    n = len(h)
    H = np.fft.rfft(h)
    f = np.arange(n // 2 + 1)
    att = np.exp(-0.5 * (f / (n / float(bw))) ** 2)
    return _repair(np.fft.irfft(H * att, n), n)

def _finerand(h, n_tries, rng):

    best_c = _eval(x)
    cur_c = best_c

        d = 8e-6 * rng.randn()
