import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def smax_obj(h, alpha, pen):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sx = np.clip(alpha * (c - m), -500, 500)
    val = (m + np.log(np.sum(np.exp(sx))) / alpha) * (2.0 / n)
    s = h.sum() - n * 0.5
    val += pen * s * s
    return val

def smax_grad(h, alpha, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sx = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(sx - sx.max())
    w /= w.sum()
    g = np.zeros(n)
    # Vectorized gradient computation
    # For shift sh >= 0: g[:n-sh] += w*(1-h[sh:]), g[sh:] -= w*h[:n-sh]
    # For shift -sp: g[sp:] += w*(1-h[:n-sp]), g[:n-sp] -= w*h[sp:]
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        sh = (n - 1) - k
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    s = h.sum() - n * 0.5
    g += 2.0 * pen * s
    return g

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    f0 = smax_obj(h, alpha, pen)
    try:
        res = minimize(smax_obj, h, jac=smax_grad, args=(alpha, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-15, 'maxfun': maxiter * 2,
                                'disp': False})
    except Exception:
        return repair(h, n)
    if time.time() > deadline:
        return repair(h, n)
    f1 = smax_obj(res.x, alpha, pen)
    if f1 < f0 - 1e-12:
        return repair(res.x, n)
    return repair(h, n)

    g_prev = smax_grad(h_arr, alpha, 0.0)

def topk_lag_descent(h, n_rounds, k_count, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = eval_c5(h_cur)
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:k_count]
        g = np.zeros(n)
        for kk in top_k:
            sh = (n - 1) - kk
            if sh == 0:
                g += (1.0 - 2.0 * h_cur)
            elif sh > 0:
                g[:n - sh] += (1.0 - h_cur[sh:])
                g[sh:] -= h_cur[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h_cur[:n - sp])
                g[:n - sp] -= h_cur[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:

        g /= gn
        improved = False

            h_new = proj(h_cur - st * g, n * 0.5)
            c5n = eval_c5(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break

def sa_optimize(h, n_iter, deadline, rng):
