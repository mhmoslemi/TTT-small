```python
def _topk_descent(h, n_rounds, t_end, n_top=5):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end: break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:n_top]
        g = np.zeros(n)
        for kk in top_k:
            g += _k_grad(h_cur, kk)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13: break
        g /= gn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new; c5_cur = c5n; improved = True; break
        if not improved:
            for step in [0.0005, 0.0002, 0.0001]:

                c5n = _eval(h_new)
                if c5n < c5_cur - 1e-16:
                    h_cur = h_new; c5_cur = c5n; improved = True; break
            if not improved: break
    return h_cur, c5_cur
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if abs(s - t) < 1e-12: return h
    if s < 1e-12: return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in o:
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
            if abs(d) < 1e-12: break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(h))

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    g *= dx
    return g

def _k_grad(h, k):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0: g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _lbfgs(h, alpha, t_end, maxiter=3000):
    n = len(h)
    target = n * 0.5
    pen = 100.0
    def obj(x):
        return _smax_obj(x, alpha) + pen * (x.sum() - target) ** 2
    def grad(x):
        return _smax_grad(x, alpha) + 2.0 * pen * (x.sum() - target)
    bnds = [(0.0, 1.0)] * n
    f0 = obj(h)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
        if res.fun < f0 - 1e-15:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _momentum(h, alpha, n_steps, t_end):
    n = len(h)
    target = n * 0.5
    h_cur = h.copy()
    h_pool = h.copy()
    v = np.zeros(n)
    g_prev = _smax_grad(h_cur, alpha)
    mu = 0.7
    step = 0.008
    f_prev = _smax_obj(h_cur, alpha)
    for i in range(n_steps):
        if time.time() > t_end: break
        g = _smax_grad(h_cur, alpha)
        v = mu * v + (1.0 - mu) * g
        dh = h_cur - h_pool
        dg = g - g_prev
        den = np.dot(dg, dg) + 1e-15
        step_bb = max(0.002, min(0.2, 0.5 * abs(np.dot(dh, dg) / den)))
        step = 0.5 * step + 0.5 * step_bb
        accepted = False
        for _bt in range(10):
            h_new = _proj(h_cur - step * v, target)
            f_new = _smax_obj(h_new, alpha)
            if f_new <= f_prev + 1e-14:
                h_pool = h_cur.copy()
                h_cur = h_new
                g_prev = g.copy()
                f_prev = f_new
                step = min(step * 1.05, 0.3)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 0.3
        h_pool = h_cur.copy()
        g_prev = _smax_grad(h_cur, alpha)
    return h_cur

def _topk_descent(h, n_rounds, t_end, n_top=5):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5c = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end: break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top = np.argsort(c)[::-1][:n_top]
        g = np.zeros(n)
        for kk in top:
            g += _k_grad(h_cur, kk)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13: break
        g /= gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h_cur = h_new; c5c = c5n; improved = True; break
        if not improved: break
    return h_cur, c5c
```
