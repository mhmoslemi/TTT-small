def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode = 'full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-12:
                break
    return h

def _ceq(x):
    return x.sum() - _ng[0] * 0.5

def _cjq(x):
    return np.ones(_ng[0])

def _obj(x, a):
    c = np.correlate(x, 1.0 - x, mode = 'full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / a

def _grad(x, a):
    n = _ng[0]
    c = np.correlate(x, 1.0 - x, mode = 'full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:

                g[i2] -= wk * x[i2 - sh]
        else:

            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

                     bounds=b, constraints=cs,
                     options={'maxiter': mi, 'ftol': ft, 'iprint': 0})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

            w[idx] = np.exp(4.0 * (c[idx] - cmn) / (cmx - cmn))
        else:
            w[idx] = 1.0

        dx = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):

            sh = (n - 1) - k

                    g[i2] -= wk * h[i2 - sh]
            else:

                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        mv = False

            ct = _eval(ht)

                cur = ct
                mv = True

    return h, cur

            else:

def _grad(x, a):
    n = _ng[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    # Only process shifts with significant weight
    thresh = 1e-10
    ks = np.where(w > thresh)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in ks:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g
