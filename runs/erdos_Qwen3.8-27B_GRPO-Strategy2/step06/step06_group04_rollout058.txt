    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _pw_obj(h, p):
    c = _corr(h)
    c = np.maximum(c, 1e-12)
    return float(np.sum(c**p) ** (1.0/p))

def _pw_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cp = np.maximum(c, 1e-12)
    cp_p = cp ** p
    S = cp_p.sum()
    w_all = (S ** (1.0/p - 1.0)) * cp ** (p-1) * dx
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w_all[k]
        if wk < 1e-18:
            continue
        sh = (n-1) - k
        if sh >= 0:
            i1 = np.arange(n-sh)
            grad[i1] += wk * (1.0 - h[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                grad[i2] -= wk * h[i2-sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            grad[i1] += wk * (1.0 - h[i1-sp])
            i2 = np.arange(n-sp)
            grad[i2] -= wk * h[i2+sp]
    return grad

    x = h0.copy()

    best_x = x.copy(); best_c = _eval(x)
    for i in range(n_iter):
        if time.time() > deadline: break
        frac = i / max(n_iter-1, 1)
        p = int(p_lo + (p_hi - p_lo) * frac)
        p = max(p, 2)
        g = _pw_grad(x, p)
        t = i
        m = 0.9*m + 0.1*g
        v = 0.999*v + 0.001*g*g
        x = x - lr * m / (np.sqrt(v) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _lfb(h0, p, deadline):
    """L-BFGS-B with sum constraint via penalty."""
    n = len(h0)
    def obj(x):
        base = _pw_obj(x, p)
        pen = 1000.0 * (x.sum() - n/2.0)**2
        return base + pen
    def grad(x):
        base_g = _pw_grad(x, p)
        pen_g = 2000.0 * (x.sum() - n/2.0)
        return base_g + pen_g

                       bounds=bnds, options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-14})
        return _repair(res.x, n)
    except:
        return _repair(h0, n)

def _binary_2opt(h0, n, rng, deadline):

    x = _repair(x, n)
    c_cur = _eval(x)
    best_x = x.copy(); best_c = c_cur
    ones = set(np.where(x==1)[0].tolist())
    zeros = set(np.where(x==0)[0].tolist())

        else:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    t = 0.5 * n
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _pw(h, p):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full') * 2.0 / n
    c = np.maximum(c, 1e-15)
    return float(np.sum(c ** p) ** (1.0 / p))

def _pwg(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    cc = np.maximum(c, 1e-15)
    S = float(np.sum(cc ** p))
    w = S ** (1.0 / p - 1.0) * cc ** (p - 1) * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18: continue
        sh = n - 1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

        g = _pwg(x, p)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g

        c = _eval(x)

        r = minimize(f, h0, jac=gf, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': se, 'jac': sj}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return _repair(h0.copy(), n)

def _b2o(h0, n, rng, dl):
    x = _repair((h0 > 0.5).astype(float), n)
    cc = _eval(x); bx = x.copy(); bc = cc
    o1 = np.where(x == 1)[0]; o0 = np.where(x == 0)[0]
    for _ in range(500000):
        if time.time() > dl: break
        if len(o1) == 0 or len(o0) == 0: break
        i = o1[rng.randint(len(o1))]; j = o0[rng.randint(len(o0))]
        x[i] = 0; x[j] = 1
        cn = _eval(x)
        if cn < cc:
            cc = cn
            if cn < bc: bc = cn; bx = x.copy()
            o1 = np.where(x == 1)[0]; o0 = np.where(x == 0)[0]
        else:
            x[i] = 1; x[j] = 0
    return bx, bc

def _rs(h, nn):
    no = len(h)
    xo = np.linspace(0, 2, no, endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(xn, xo, h)

    t0 = time.time()

    n0 = len(initial_h_values)
    hi = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh = hi.copy(); bc = _eval(bh); bn = n0

    # P1: Adam multi-phase (0-220s)
    d1 = t0 + 220
    for (pl, ph, ni, l0, l1) in [(3, 15, 5000, 0.05, 0.005),

        if c < bc: bc, bh = c, x.copy()

        if time.time() > d2: break
        hn = _lfb(bh, p, mu, d2)
        c = _eval(hn)
        if c < bc: bc, bh = c, hn.copy()

    # P3: SLSQP polish (350-450s)

        hn = _slsqp(bh, p, 2000, d3)
        c = _eval(hn)
        if c < bc: bc, bh = c, hn.copy()

    # P4: Binary 2-opt (450-650s)
    d4 = t0 + 650
    xb, cb = _b2o(bh, bn, rng, d4)
    if cb < bc: bc, bh = cb, xb.copy()

    # P5: Adam from binary (650-780s)
    d5 = t0 + 780
    x, c = _adam(bh, 10, 300, 3000, 0.015, 0.001, seed + 1, d5)
    if c < bc: bc, bh = c, x.copy()

    # P6: Multi-n exploration (780-900s)
    d6 = t0 + 900
    for dn in [-20, -10, 10, 20, 30]:
        if time.time() > d6: break
        nt = bn + dn
        if nt < 100 or nt > 600 or nt % 2 != 0: continue
        hr = _repair(_rs(bh, nt), nt)
        x, c = _adam(hr, 10, 200, 1500, 0.01, 0.001, seed + nt, d6)
        hn = _lfb(x, 200, 500, d6)
        c2 = _eval(hn)
        if c2 < bc: bc, bh, bn = c2, hn.copy(), nt

    # P7: Final high-p Adam (900-1030s)
    d7 = t0 + 1030
    x, c = _adam(bh, 500, 10000, 4000, 0.003, 0.0003, seed + 3, d7)
    if c < bc: bc, bh = c, x.copy()

    # P8: Final SLSQP + binary (1030-1085s)
    d8 = t0 + 1085
    if time.time() < d8:
        hf = _slsqp(bh, 5000, 2000, d8)
        c = _eval(hf)
        if c < bc: bc, bh = c, hf.copy()
    if time.time() < d8:
        xb, cb = _b2o(bh, bn, rng, d8)
        if cb < bc: bc, bh = cb, xb.copy()

    # Final
    bh = _repair(bh, len(bh))
    bc = _eval(bh)
    return (bh, bc, len(bh))
