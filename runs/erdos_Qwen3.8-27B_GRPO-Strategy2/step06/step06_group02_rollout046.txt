                h[idx] += a; diff -= a
            else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _ew_grad(h, n, idx, w):
    """Compute weighted gradient of correlate peaks."""
    g = np.zeros(n)
    dx = 2.0 / n
    for wi, ki in zip(w, idx):
        if wi < 1e-12:
            continue
        k = int(ki) - (n - 1)
        if k >= 0:
            g[k:] += wi * (1.0 - h[:n-k])
            g[:n-k] -= wi * h[k:]
        else:
            m = -k
            g[:n-m] += wi * (1.0 - h[m:])
            g[m:] -= wi * h[:n-m]
    g *= dx
    return g

def _proj_subgrad(h, n, top_m, n_iter, t_end, step_init=0.01, beta=10.0):
    h = h.copy()
    cur = _eval(h)
    step = step_init
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-top_m:]
        cmax, cmin = c[top_idx].max(), c[top_idx].min()
        if cmax - cmin > 1e-12:
            w = np.exp(beta * (c[top_idx] - cmin) / (cmax - cmin))
        else:
            w = np.ones(top_m)
        w = w / w.sum()
        g = _ew_grad(h, n, top_idx, w)
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            s = step * f
            h_new = _repair(h - s * g, n)
            c_new = _eval(h_new)
            if c_new < cur - 1e-15:
                h = h_new
                cur = c_new
                improved = True
                step = s
                break
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
    return h, cur

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _ew_grad(x, n, np.arange(len(c)), w)

def _lbfgs_opt(h, alpha, maxiter, t_end):
    n = len(h)
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad,
                       args=(alpha, n),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 2})
        if time.time() > t_end:
            return h.copy()
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old, period=2.0)  # periodic!

def _pairwise_cd(h, n, n_iter, scales, t_end, seed=123):
    rng = np.random.RandomState(seed)
    cur = _eval(h)
    for scale in scales:
        for _ in range(n_iter):
            if time.time() > t_end:
                return h, cur
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur - 1e-15:
                cur = cn
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, cur

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _ew_grad(h, n, idx, w):
    g = np.zeros(n)
    dx = 2.0 / n
    for wi, ki in zip(w, idx):
        if wi < 1e-14:
            continue
        k = int(ki) - (n - 1)
        if k >= 0:
            g[k:] += wi * (1.0 - h[:n - k])
            g[:n - k] -= wi * h[k:]
        else:
            m = -k
            g[:n - m] += wi * (1.0 - h[m:])
            g[m:] -= wi * h[:n - m]
    g *= dx
    return g

                break

            step *= 0.6

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _ew_grad(x, n, np.arange(len(c)), w)
