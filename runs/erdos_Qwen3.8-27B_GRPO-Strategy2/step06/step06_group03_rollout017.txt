import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

    g = _sgrad(h, alpha)
    g += 2.0 * lam * (h.sum() - 0.5 * n)
    return g

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    bnds = [(0.0, 1.0)] * n

        res = minimize(_efunc, h, jac=_egrad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return x.sum() - 0.5 * n
    def _cj(x): return np.ones(n)

def _top_grad(h, n, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)

        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _subgrad(h, n_iter, step0, top_k=5):

def _top_grad(h, top_m):

    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_m:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * 2.0 / n

def _subgrad(h, n_iter, step0, top_k, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); mu = 0.8; step = step0; stuck = 0
    for _ in range(n_iter):
        g = _top_grad(best, top_k)
        v = mu * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            hn = _proj_sum(best - step * f * v, 0.5 * n)
            cn = _eval(hn)
            if cn < best_c - 1e-15:
                best = hn; best_c = cn; step = min(step * 1.15, 0.5)
                improved = True; stuck = 0; break
        if not improved:
            step *= 0.6; stuck += 1
            if stuck > 60:
                best = _repair(best + 0.02 * rng.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0; v = np.zeros(n)
    return best, best_c

def _fine(h, n_tries, rng, scale):
    n = len(h)
    h = h.copy()
    c_cur = _eval(h); best = h.copy(); best_c = c_cur
    for _ in range(n_tries):
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur:
            c_cur = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c

def _ft_perturb(h, strength, rng):
    n = len(h)
    H = np.fft.rfft(h.astype(float))
    r = np.random.RandomState(rng.randint(2**31 - 1))
    H = H * (1.0 + strength * (r.randn(len(H)) + 1j * r.randn(len(H))))
    return _repair(np.real(np.fft.irfft(H, n)), n)

def _try_n(h, n_new, alpha, maxiter, t0, budget_s, frac):
    n_old = len(h)
    h_new = np.interp(np.linspace(0, 2, n_new, endpoint=False),
                      np.linspace(0, 2, n_old, endpoint=False), h)
    h_new = _repair(h_new, n_new)
    mid = budget_s * frac
    if time.time() - t0 > mid:
        return h, _eval(h), n_old
    hc = _slsqp(h_new, alpha, maxiter)
    c = _eval(hc)
    c_old = _eval(h)
    if c < c_old:
        return hc, c, n_new
    return h, c_old, n_old

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c = _eval(best_h); best_n = n0

    # Phase 1: L-BFGS-B penalty warmup (10% budget)
    for alpha, lam, mit in [(300, 50, 500), (1000, 200, 700), (5000, 1000, 1000)]:
        if _el() > budget_s * 0.10:
            break
        hc = _lbfgs_pen(best_h, alpha, lam, mit)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 2: SLSQP medium alpha (15%)
    for alpha, mit in [(3000, 1500), (10000, 2000), (50000, 2000)]:
        if _el() > budget_s * 0.25:
            break
        hc = _slsqp(best_h, alpha, mit)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 3: Subgradient with top-5 (20%)
    if _el() < budget_s * 0.50:
        n_it = min(6000, int((budget_s * 0.50 - _el()) / 0.016))
        best_h, best_c = _subgrad(best_h, n_it, 0.012, 5, rng)
        best_n = len(best_h)

    # Phase 4: SLSQP high alpha (10%)
    for alpha, mit in [(200000, 2000), (800000, 2500)]:
        if _el() > budget_s * 0.60:
            break
        hc = _slsqp(best_h, alpha, mit)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc

    # Phase 5: Fourier perturbation exploration (10%)
    if _el() < budget_s * 0.72:
        for strength in [0.4, 0.15, 0.05]:
            if _el() > budget_s * 0.72:
                break
            h_try = _ft_perturb(best_h, strength, rng)
            hc = _slsqp(h_try, 50000, 1500)
            c = _eval(hc)
            if c < best_c:
                best_c = c; best_h = hc
                best_n = len(best_h)

    # Phase 6: Subgradient polish (5%)
    if _el() < budget_s * 0.80:
        n_it = min(3000, int((budget_s * 0.80 - _el()) / 0.016))
        best_h, best_c = _subgrad(best_h, n_it, 0.006, 3, rng)
        best_n = len(best_h)

    # Phase 7: Final SLSQP (5%)
    if _el() < budget_s * 0.85:
        hc = _slsqp(best_h, 2000000, 2000)
        c = _eval(hc)
        if c < best_c:
            best_c = c; best_h = hc; best_n = len(best_h)

    # Phase 8: Fine coordinate search (15%)
    if _el() < budget_s - 3:
        n_tr1 = min(40000, int((budget_s * 0.5 - _el()) / 0.002))
        if n_tr1 > 100:
            best_h, best_c = _fine(best_h, n_tr1, rng, 3e-5)
            best_n = len(best_h)
        n_tr2 = min(40000, int((budget_s - 3 - _el()) / 0.002))
        if n_tr2 > 100:
            best_h, best_c = _fine(best_h, n_tr2, rng, 8e-6)
            best_n = len(best_h)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
