def _sim_anneal(h0, n_steps, t_init, t_final, scale, seed=0, time_fn=None, time_budget=1e9):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best = h.copy()
    best_c = _eval(h)
    for i in range(n_steps):
        if time_fn is not None and time_fn() > time_budget:
            break
        temp = t_init * (t_final / t_init) ** (i / max(1, n_steps - 1))
        idx = rng.choice(n, size=rng.randint(2, 8), replace=False)
        delta = scale * rng.randn(len(idx))
        move = np.zeros(n)
        move[idx] = delta
        move -= move.mean()
        h_new = _repair(h + move, n)
        c_new = _eval(h_new)
        if c_new < best_c or rng.rand() < np.exp(-(c_new - best_c) / (temp + 1e-12)):
            h = h_new
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
    return best, best_c

def _coord_descent(h, n_iter, scales, seed=0, time_fn=None, time_budget=1e9):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5c = _eval(h)
    for scale in scales:
        for _ in range(n_iter // len(scales)):
            if time_fn is not None and time_fn() > time_budget:
                return h, c5c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rsp = seed - 42 if seed > 0 else 0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0
    tb = budget_s * 0.92

    # Phase 1: PGD + Adam at full resolution, multiple alphas
    tb1 = budget_s * 0.30
    h1, c1 = _pgd_adam(best_h, [5000, 20000, 80000, 300000], 20000,
                        lr_base=0.02, seed=seed, time_fn=elapsed, time_budget=tb1)
    if c1 < best_c5:
        best_c5, best_h, best_n = c1, h1.copy(), len(h1)

    # Phase 2: SLSQP polish
    tb2 = budget_s * 0.45
    for alpha in [20000, 100000, 500000]:
        if elapsed() > tb2:
            break
        h2 = _slsqp(best_h, alpha, 2000)
        c2 = _eval(h2)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), len(h2)

    # Phase 3: Simulated annealing escape
    tb3 = budget_s * 0.60
    h3a, c3a = _sim_anneal(best_h, 5000, 0.005, 0.0001, 0.03,
                            seed=seed+10, time_fn=elapsed, time_budget=tb3)
    if c3a < best_c5:
        best_c5, best_h, best_n = c3a, h3a.copy(), len(h3a)
    if elapsed() < tb3:
        h3b, c3b = _sim_anneal(best_h, 5000, 0.003, 0.00005, 0.015,
                                seed=seed+20, time_fn=elapsed, time_budget=tb3)
        if c3b < best_c5:
            best_c5, best_h, best_n = c3b, h3b.copy(), len(h3b)

    # Phase 4: PGD again from annealed solution
    tb4 = budget_s * 0.72
    h4, c4 = _pgd_adam(best_h, [50000, 200000], 8000,
                        lr_base=0.01, seed=seed+30, time_fn=elapsed, time_budget=tb4)
    if c4 < best_c5:
        best_c5, best_h, best_n = c4, h4.copy(), len(h4)

    # Phase 5: SLSQP final polish
    tb5 = budget_s * 0.82
    for alpha in [200000, 1000000]:
        if elapsed() > tb5:
            break
        h5 = _slsqp(best_h, alpha, 1500)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h5.copy(), len(h5)

    # Phase 6: Coordinate descent
    tb6 = budget_s * 0.90
    h6, c6 = _coord_descent(best_h, 20000, [0.004, 0.001, 0.0003],
                            seed=seed+50, time_fn=elapsed, time_budget=tb6)
    if c6 < best_c5:
        best_c5, best_h, best_n = c6, h6.copy(), len(h6)

    # Phase 7: Try lower resolution (n=62) and upsample
    tb7 = budget_s * 0.95
    if elapsed() < tb7:
        n_low = 62
        h_low = _repair(_resample(best_h, n_low), n_low)
        hl_opt = _slsqp(h_low, 50000, 2000)
        c_low = _eval(hl_opt)
        if elapsed() < tb7:
            hl_opt2, c_low2 = _pgd_adam(hl_opt, [10000, 50000], 10000,
                                         lr_base=0.03, seed=seed+60,
                                         time_fn=elapsed, time_budget=tb7)
            if c_low2 < c_low:
                hl_opt, c_low = hl_opt2, c_low2
        # Upsample and compare
        h_up = _repair(_resample(hl_opt, best_n), best_n)
        h_up = _slsqp(h_up, 100000, 1000)
        c_up = _eval(h_up)
        if c_up < best_c5:
            best_c5, best_h, best_n = c_up, h_up.copy(), len(h_up)

    # Phase 8: Final SLSQP + coordinate descent
    if elapsed() < budget_s - 5:
        h_fin = _slsqp(best_h, 500000, 1000)
        cf = _eval(h_fin)
        if cf < best_c5:
            best_c5, best_h = cf, h_fin.copy()
        h_fin2, cf2 = _coord_descent(best_h, 5000, [0.001, 0.0002],
                                     seed=seed+90, time_fn=elapsed, time_budget=budget_s - 2)
        if cf2 < best_c5:
            best_c5, best_h = cf2, h_fin2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
