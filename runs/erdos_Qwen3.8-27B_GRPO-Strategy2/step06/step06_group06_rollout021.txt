Let me also think about **CVXPY** usage. The problem:
minimize max_k [Σ_i h_i (1-h_{i+k})]
subject to 0 ≤ h_i ≤ 1, Σ h_i = n/2

This is a QP with a max-of-quadratics objective. Not directly solvable by CVXPY (non-convex). But we can:
- Fix the max value to t, and check feasibility: is there h with c_k ≤ t for all k? This is still non-convex (quadratic constraints).
- Linearize: at a point h₀, c_k ≈ c_k(h₀) + ∇c_k(h₀)·(h-h₀). Then minimize Σ max(0, c_k_lin - t) subject to constraints. This is an LP!

Actually, the linearization gives:
c_k(h) ≈ c_k(h₀) + ∑_i [∂c_k/∂h_i](h_i - h_{0,i})

where ∂c_k/∂h_i = (1-h_{i+k}) - h_{i-k} (for valid indices, with appropriate boundary handling)

So we can solve:
minimize t
subject to: c_k(h₀) + ∇c_k(h₀)·(h-h₀) ≤ t for all k
           0 ≤ h_i ≤ 1
           Σ h_i = n/2

This is a **linear program**! And CVXPY with a fast LP solver can handle it.

Then iterate: solve LP → get h_new → relinearize → solve LP again.

This is a **cutting-plane / linearization** approach. Each LP is exact for the linear approximation, and we iterate to convergence.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):

def _linearize_and_solve(h0, solver='CLARABEL'):
    n = len(h0)
    dx = 2.0 / n
    h0v = cp.Variable(n)
    t = cp.Variable()
    
    # Compute current correlation
    c_cur = np.max(np.correlate(h0, 1.0 - h0, mode='full')) * dx
    
    # Compute gradient of each c_k w.r.t. h
    # corr[k] = sum_j h[j] * (1-h[j+p]) where p = k-(n-1), valid when 0<=j+p<n
    # d(corr[k])/d(h_i) = (1-h[i+p])*1[0<=i+p<n] - h[i-p]*1[0<=i-p<n]
    
    p_vals = np.arange(2*n-1) - (n-1)  # p ranges from -(n-1) to n-1
    
    # For LP: minimize t
    # s.t. c_k(h0) + sum_i G[k,i] * (h_i - h0_i) <= t for all k
    #      0 <= h_i <= 1
    #      sum h_i = n/2
    
    constraints = [h0v >= 0, h0v <= 1, cp.sum(h0v) == n/2]
    
    # Build the gradient matrix efficiently
    # G[k,i] = (2/n) * [(1-h0[i+p_k])*1[0<=i+p_k<n] - h0[i-p_k]*1[0<=i-p_k<n]]
    
    # For each k, p = k-(n-1)
    # G[k] = (2/n) * [(1-h0) shifted - h0 shifted]
    
    n_k = 2*n - 1
    # Build A matrix where row k is the gradient of c_k
    # A[k, i] = (2/n) * [(1-h0[(i+p)%n]) * 1[0<=i+p<n] - h0[(i-p)%n] * 1[0<=i-p<n]]
    
    # More carefully:
    # For shift p:
    # (1-h0) shifted: for index i, value is (1-h0[i+p]) if 0<=i+p<n else 0
    # h0 shifted: for index i, value is h0[i-p] if 0<=i-p<n else 0
    
    # Vectorized:
    A = np.zeros((n_k, n))
    for k in range(n_k):
        p = k - (n-1)
        if p >= 0:
            # i+p in [0,n-1] means i in [0, n-1-p]
            A[k, :n-p] += (1.0 - h0[p:])  # (1-h0[i+p]) for i=0..n-1-p
            # i-p in [0,n-1] means i in [p, n-1]
            A[k, p:] -= h0[:n-p]  # h0[i-p] for i=p..n-1
        else:
            q = -p
            # i+p in [0,n-1] means i+(-q) in [0,n-1], i.e., i in [q, n-1]
            A[k, q:] += (1.0 - h0[:n-q])  # (1-h0[i-q]) for i=q..n-1
            # i-p in [0,n-1] means i+q in [0,n-1], i.e., i in [0, n-1-q]
            A[k, :n-q] -= h0[q:]  # h0[i+q] for i=0..n-1-q
    A *= dx
    
    # c_k values
    c_vals = np.correlate(h0, 1.0 - h0, mode='full') * dx
    
    # LP: min t s.t. A @ (h - h0) + c_vals <= t, 0<=h<=1, sum(h)=n/2
    # => A @ h <= t - c_vals + A @ h0
    # => A @ h - t <= A @ h0 - c_vals
    
    rhs = A @ h0 - c_vals
    # A @ h - 1*t <= rhs for each row
    # This is: [A | -1] @ [h; t] <= rhs
    
    AT = np.column_stack([A, -np.ones(n_k)])
    prob = cp.Problem(cp.Minimize(t),
                      [AT @ cp.hstack([h0v, t]) <= rhs,
                       h0v >= 0, h0v <= 1,
                       cp.sum(h0v) == n / 2])
    
    prob.solve(solver=solver, verbose=False)
    
    if h0v.value is not None:
        h_new = np.clip(h0v.value, 0, 1)
        h_new = _repair(h_new, n)
        return h_new, t.value if t.value is not None else np.inf
    return h0.copy(), c_cur

def _slsqp_opt(h, alpha, maxiter, t_end):

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * dx
    
    def smax_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)

                continue

            else:
                q = -p
                g[q:] += wk * (1.0 - x[:n - q])
                g[:n - q] -= wk * x[q:]
        g *= dx
        return g

    f_start = smax_obj(h)
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,

        if res.fun < f_start - 1e-14:

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sa_pairs(h, n, t_end, n_iter, rng):
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    t0v, t1v = 0.003, 2e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.002 + 0.015 * t)
        old_i1, old_i2 = h[i1], h[i2]
        h[i1], h[i2] = old_i1 + d, old_i2 - d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[i1], h[i2] = old_i1, old_i2
            h = _repair(h, n)
    return best_h, best_c

def _multi_lag_descent(h, n_rounds, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        thresh = c.max() - 0.005
        active = np.where(c >= thresh)[0]
        g = np.zeros(n)
        for kk in active:
            p = kk - (n - 1)
            if p >= 0:
                g[:n - p] += (1.0 - h_cur[p:])
                if p > 0:
                    g[p:] -= h_cur[:n - p]
            else:
                q = -p
                g[q:] += (1.0 - h_cur[:n - q])
                g[:n - q] -= h_cur[q:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _lp_iterate(h, n_iters, t_end):
    """Iterate: linearize -> solve LP -> check improvement"""
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        # Try different solvers
        solved = False
        for solver in ['CLARABEL', 'ECOS', 'SCS']:
            try:
                h_new, t_val = _linearize_and_solve(h_cur, solver=solver)
                c5_new = _eval(h_new)
                if c5_new < c5_cur - 1e-10:
                    h_cur = h_new
                    c5_cur = c5_new
                    solved = True
                    break
                elif c5_new < c5_cur:
                    h_cur = h_new
                    break
            except Exception:
                continue
        if not solved and c5_cur == _eval(h_cur):
            # No improvement, try a smaller perturbation
            break
    return h_cur, c5_cur

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        p = k - (n - 1)
        if p >= 0:
            g[:n - p] += wk * (1.0 - x[p:])
            if p > 0:
                g[p:] -= wk * x[:n - p]
        else:
            q = -p
            g[q:] += wk * (1.0 - x[:n - q])
            g[:n - q] -= wk * x[q:]
    g *= dx
    return g

def _sum_cons(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_cons, 'jac': _sum_jac, 'args': (n,)}]
    f0 = _smax_obj(h, alpha, n)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _build_A(h):
    n = len(h)
    dx = 2.0 / n
    nk = 2 * n - 1
    A = np.zeros((nk, n))
    for k in range(nk):
        p = k - (n - 1)
        if p >= 0:
            A[k, :n - p] += (1.0 - h[p:])
            if p > 0:
                A[k, p:] -= h[:n - p]
        else:
            q = -p
            A[k, q:] += (1.0 - h[:n - q])
            A[k, :n - q] -= h[q:]
    A *= dx
    return A

def _lp_solve(h, solver_names):
    n = len(h)
    c_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    A = _build_A(h)
    rhs = A @ h - c_vals
    hv = cp.Variable(n)
    tv = cp.Variable()
    AT = np.column_stack([A, -np.ones(2 * n - 1)])
    for solver in solver_names:
        try:
            prob = cp.Problem(cp.Minimize(tv),
                              [AT @ cp.hstack([hv, tv]) <= rhs,
                               hv >= 0, hv <= 1,
                               cp.sum(hv) == n / 2])
            prob.solve(solver=solver, verbose=False)
            if hv.value is not None and tv.value is not None:
                h_new = np.clip(np.array(hv.value).flatten(), 0, 1)
                h_new = _repair(h_new, n)
                c5 = _eval(h_new)
                if c5 < c_vals.max():
                    return h_new, c5
        except Exception:
            continue
    return h.copy(), float(c_vals.max())

def _lp_iterate(h, n_iters, t_end):
    n = len(h)
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    solvers = ['CLARABEL', 'ECOS', 'SCS']
    for it in range(n_iters):
        if time.time() > t_end:
            break
        h_new, c5_new = _lp_solve(h_cur, solvers)
        if c5_new < c5_cur - 1e-12:
            h_cur = h_new
            c5_cur = c5_new
        else:
            # Try damped step
            h_mid = 0.5 * (h_cur + h_new)
            h_mid = _repair(h_mid, n)
            c5_mid = _eval(h_mid)
            if c5_mid < c5_cur - 1e-12:
                h_cur = h_mid
                c5_cur = c5_mid
            else:
                break
    return h_cur, c5_cur

def _multi_lag(h, n_rounds, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _ in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        thr = c.max() - 0.003
        active = np.where(c >= thr)[0]
        g = np.zeros(n)
        for kk in active:
            p = kk - (n - 1)
            if p >= 0:
                g[:n - p] += (1.0 - h_cur[p:])
                if p > 0:
                    g[p:] -= h_cur[:n - p]
            else:
                q = -p
                g[q:] += (1.0 - h_cur[:n - q])
                g[:n - q] -= h_cur[q:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _sa(h, n, t_end, n_iter, rng):
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    t0v, t1v = 0.003, 2e-6
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.002 + 0.01 * t)
        o1, o2 = h[i1], h[i2]
        h[i1], h[i2] = o1 + d, o2 - d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c, best_h = c_cur, h.copy()
        else:
            h[i1], h[i2] = o1, o2
            h = _repair(h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort((1.0 - h) if diff > 0 else h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        p = k - (n - 1)
        if p >= 0:
            g[:n - p] += wk * (1.0 - x[p:])
            if p > 0:
                g[p:] -= wk * x[:n - p]
        else:
            q = -p
            g[q:] += wk * (1.0 - x[:n - q])
            g[:n - q] -= wk * x[q:]
    g *= dx
    return g

def _sum_cons(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_cons, 'jac': _sum_jac, 'args': (n,)}]
    f0 = _smax_obj(h, alpha, n)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if res.fun < f0 - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _build_A(h):
    n = len(h)
    dx = 2.0 / n
    nk = 2 * n - 1
    A = np.zeros((nk, n))
    for k in range(nk):
        p = k - (n - 1)
        if p >= 0:
            A[k, :n - p] += (1.0 - h[p:])
            if p > 0:
                A[k, p:] -= h[:n - p]
        else:
            q = -p
            A[k, q:] += (1.0 - h[:n - q])
            A[k, :n - q] -= h[q:]
    A *= dx
    return A

def _lp_solve(h):
    n = len(h)
    c_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cmax = float(c_vals.max())
    A = _build_A(h)
    rhs = A @ h - c_vals
    hv = cp.Variable(n)
    tv = cp.Variable()
    AT = np.column_stack([A, -np.ones(2 * n - 1)])
    for solver in ['CLARABEL', 'ECOS', 'SCS']:
        try:

                              [AT @ cp.hstack([hv, tv]) <= rhs,
                               hv >= 0, hv <= 1,

            prob.solve(solver=solver, verbose=False)

                c5 = _eval(h_new)
                if c5 < cmax - 1e-12:
                    return h_new, c5
        except Exception:
            continue
    return h.copy(), cmax

        else:

            if c5_mid < c5_cur - 1e-12:
                h_cur, c5_cur = h_mid, c5_mid

                break

            else:
                q = -p
                g[q:] += (1.0 - h_cur[:n - q])
                g[:n - q] -= h_cur[q:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13:

        g /= gn
        improved = False
