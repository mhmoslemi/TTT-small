def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):

            lo = mid
        else:
            hi = mid

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _block_xover(pa, pb, rng, n, nb=2, bl_min=8, bl_max=35):
    child = pa.copy()
    for _ in range(nb):
        length = rng.randint(bl_min, bl_max + 1)
        start = rng.randint(0, max(1, n - length))
        child[start:start+length] = pb[start:start+length]
    return child

def _ga_mutate(h, rng, n, sigma):
    h = h.copy()
    r = rng.random()
    if r < 0.45:
        nm = rng.randint(4, min(15, n))
        idx = rng.choice(n, nm, replace=False)
        h[idx] += sigma * rng.randn(nm)
    elif r < 0.75:
        bl = rng.randint(8, min(45, n // 3))
        st = rng.randint(0, max(1, n - bl))
        sh = rng.randint(1, 6) * (1 if rng.random() < 0.5 else -1)
        ns = st + sh
        if 0 <= ns and ns + bl <= n:
            seg = h[st:st+bl].copy()
            h[ns:ns+bl] = seg
    else:
        nm = rng.randint(2, 6)
        idx = rng.choice(n, nm, replace=False)
        h[idx] += sigma * 4.0 * rng.randn(nm)
    return _repair(h, n)

def _ga(h_init, n, t0, budget, pop_sz=35, max_gen=500, seed=42):
    rng = np.random.RandomState(seed)
    pop = [h_init.copy()]
    for i in range(1, pop_sz):
        r = rng.random()
        if r < 0.35:
            ind = h_init + 0.12 * rng.randn(n)
        elif r < 0.6:
            ind = h_init + 0.05 * rng.randn(n)
        elif r < 0.8:
            ind = h_init.copy()
            idx = rng.choice(n, n//4, replace=False)
            ind[idx] += 0.25 * rng.randn(len(idx))
        else:
            ind = np.clip(0.5 + 0.25 * rng.randn(n), 0, 1)
        pop.append(_repair(ind, n))
    fit = [_eval(p) for p in pop]
    bi = int(np.argmin(fit))
    bh, bc = pop[bi].copy(), fit[bi]
    elite = 4
    stagnant = 0
    for gen in range(max_gen):
        if time.time() - t0 > budget - 150:
            break
        gf = gen / max_gen
        sig = 0.12 * (1.0 - 0.7 * gf) + 0.015
        sidx = np.argsort(fit)
        npop = [pop[sidx[i]].copy() for i in range(elite)]
        while len(npop) < pop_sz:
            c1 = rng.randint(0, pop_sz, 3)
            p1 = pop[c1[int(np.argmin([fit[c] for c in c1]))]]
            c2 = rng.randint(0, pop_sz, 3)
            p2 = pop[c2[int(np.argmin([fit[c] for c in c2]))]]
            ch = _block_xover(p1, p2, rng, n)
            ch = _ga_mutate(ch, rng, n, sig)
            npop.append(ch)
        pop = npop
        fit = [_eval(p) for p in pop]
        gm = int(np.argmin(fit))
        if fit[gm] < bc - 1e-12:
            bc = fit[gm]
            bh = pop[gm].copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30 and rng.random() < 0.3:
                idx = rng.randint(0, pop_sz)
                pop[idx] = h_init + 0.2 * rng.randn(n)
                pop[idx] = _repair(pop[idx], n)
                fit[idx] = _eval(pop[idx])
    return bh, bc

def _sa(h, n, t0, budget, steps=15000, seed=99):
    rng = np.random.RandomState(seed)
    c = _eval(h)
    bh, bc = h.copy(), c
    for it in range(steps):
        if time.time() - t0 > budget - 40:
            break
        T = max(0.00005, 0.003 * (1.0 - it / float(steps)))
        bl = rng.randint(3, min(25, n // 5))
        st = rng.randint(0, max(1, n - bl))
        old = h[st:st+bl].copy()
        h[st:st+bl] = np.clip(old + rng.randn(bl) * 0.04, 0, 1)
        h = _repair(h, n)
        cn = _eval(h)
        if cn - c < 0 or rng.random() < np.exp(-(cn - c) / T):
            c = cn
            if c < bc:
                bc = c
                bh = h.copy()
        else:
            h[st:st+bl] = old
            h = _repair(h, n)
    return bh, bc

def _topk_grad(h, nt, t0, budget, rounds=200):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(rounds):
        if time.time() - t0 > budget - 20:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-nt:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= dx / nt
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        imp = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4]:
            hn = _proj(h - st * g, n * 0.5)
            cn = _eval(hn)
            if cn < c5 - 1e-14:
                h, c5 = hn, cn
                imp = True
                break
        if not imp:
            break
    return h, c5

def _kick_refine(h, n, t0, budget, n_kicks=15, seed=77):
    rng = np.random.RandomState(seed)
    bc, bh = _eval(h), h.copy()
    for kk in range(n_kicks):
        if time.time() - t0 > budget - 50:
            break
        h2 = h.copy()
        nk = rng.randint(n // 10, n // 4)
        idx = rng.choice(n, nk, replace=False)
        h2[idx] += 0.15 * rng.randn(nk)
        h2 = _repair(h2, n)
        h2, c2 = _topk_grad(h2, 5, t0, budget, 150)
        if c2 < bc:
            bc, bh = c2, h2
    return bh, bc

def _polish(h, n_iter, rng, t0, budget):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(n_iter):
        if time.time() - t0 > budget - 2:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 4)) % n
        d = 8e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-17:
                c5c = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

    if time.time() - t0 < budget_s - 40:
