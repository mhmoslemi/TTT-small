        if i == j:
            j = (i + 1) % n
        d = 0.003 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            c = _corr_full(h)
        else:
            h[i], h[j] = ni - d, nj + d
            h = _repair(h, n)
    return h, c5c

def _complement_check(h):
    h_c = 1.0 - h
    h_c = _repair(h_c, len(h_c))
    c5 = _eval(h_c)
    return h_c, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Complement check
    hc, c5c = _complement_check(best_h)
    if c5c < best_c5:
        best_c5, best_h = c5c, hc

    # Phase 1: SLSQP polish (40s)
    t1 = t0 + 40
    if time.time() < t1:
        for alpha in [5000, 50000, 200000]:
            if time.time() > t1:
                break
            h_new = _slsqp_opt(best_h, alpha, 3000, 1e-16)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 2: Worst-shift descent (120s)
    t2 = t0 + 160
    if time.time() < t2:
        h2, c5_2 = _worst_descent(best_h, 800, 3, t2, t0)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-resolution search (180s)
    t3 = t0 + 340
    for n_try in [250, 280, 300, 320, 350, 400]:
        if time.time() > t3:
            break
        h_r = _resample(best_h, n_try)
        h_r = _slsqp_opt(h_r, 50000, 2000, 1e-16)
        h_r, c5_r = _worst_descent(h_r, 300, 3, time.time() + 25, t0)
        hc_r, c5_rc = _complement_check(h_r)
        if c5_rc < c5_r:
            h_r, c5_r = hc_r, c5_rc
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 4: Shift-pair descent (200s)
    t4 = t0 + 540
    if time.time() < t4:
        n_iter = int((t4 - time.time()) * 4000)
        n_iter = max(n_iter, 10000)
        h4, c5_4 = _shift_pair_descent(best_h, n_iter, rng, t4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: High-alpha SLSQP (120s)
    t5 = t0 + 660
    if time.time() < t5:
        for alpha in [1000000, 5000000, 20000000]:
            if time.time() > t5:
                break
            h_new = _slsqp_opt(best_h, alpha, 5000, 1e-18)
            c5 = _eval(h_new)
            if c5 < best_c5 - 1e-13:
                best_c5, best_h = c5, h_new.copy()

    # Phase 6: Another worst-shift round with more lags (150s)
    t6 = t0 + 810
    if time.time() < t6:
        h6, c5_6 = _worst_descent(best_h, 1000, 5, t6, t0)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Shift-pair again (100s)
    t7 = t0 + 910
    if time.time() < t7:
        n_iter = int((t7 - time.time()) * 4000)
        h7, c5_7 = _shift_pair_descent(best_h, n_iter, rng, t7)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Fine coordinate descent (120s)
    t8 = t0 + 1030
    if time.time() < t8:
        n_iter = int((t8 - time.time()) * 5000)
        h8, c5_8 = _fine_coord(best_h, n_iter, rng, t8)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    # Phase 9: Final SLSQP + complement (50s)
    if time.time() < t0 + 1090:
        h9 = _slsqp_opt(best_h, 50000000, 2000, 1e-18)
        c5_9 = _eval(h9)
        if c5_9 < best_c5:
            best_c5, best_h = c5_9, h9.copy()
        hc9, c5_9c = _complement_check(best_h)
        if c5_9c < best_c5:
            best_c5, best_h = c5_9c, hc9

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
