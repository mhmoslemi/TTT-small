def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    return (m + np.log(np.sum(np.exp(s - sm))) / alpha) * (2.0 / n)

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= dx
    return g

def _lbfgs_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    f_start = _smax_obj(h, alpha)

    def obj(x):
        return _smax_obj(x, alpha) + 1e5 * (x.sum() - n * 0.5) ** 2

    def obj_grad(x):
        g = _smax_grad(x, alpha).copy()
        g += 2e5 * (x.sum() - n * 0.5)
        return g

    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                             'gtol': 1e-12, 'maxls': 50, 'disp': False})
        if time.time() < t_end and res.fun < f_start + 1e-8:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _nesterov(h, alpha, n_steps, t_end):
    n = len(h)
    h_x = h.copy()
    h_v = h.copy()
    m = 0.85
    lr = 0.03
    best_f = _smax_obj(h, alpha)
    best_h = h.copy()
    for i in range(n_steps):
        if time.time() > t_end:
            break
        if i % 50 == 0:
            lr = max(0.01, lr * (1 - 5e-6 * i))
        g = _smax_grad(h_x, alpha)
        h_v = m * h_v + (1 - m) * h_x
        h_new = np.clip(h_v - lr * g, 0, 1)
        h_new = _repair(h_new, n)
        f = _smax_obj(h_new, alpha)
        if f < best_f:
            best_f = f
            best_h = h_new.copy()
        h_x = h_new
    return best_h

def _coord_descent(h, n_tries, t_end, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5

def _sa_opt(h, n_iter, t_end, rng):
    n = len(h)
    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    h_w = h.copy()
    t_max, t_min = 0.04, 0.0005
    s_init, s_final = 0.08, 0.003
    for i in range(n_iter):
        if time.time() > t_end:
            break
        fr = i / max(1, n_iter - 1)
        t = t_max * (t_min / t_max) ** fr
        sc = s_init * (s_final / s_init) ** fr
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * sc
        o1, o2 = h_w[i1], h_w[i2]
        h_w[i1] = o1 + d
        h_w[i2] = o2 - d
        h_w = _repair(h_w, n)
        c_new = _eval(h_w)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-10)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_w.copy()
        else:
            h_w[i1], h_w[i2] = o1, o2
            h_w = _repair(h_w, n)
    return best_h, best_c

def _slsqp_fallback(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def c_fun(x):
        return x.sum() - n * 0.5

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if time.time() < t_end and res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)
