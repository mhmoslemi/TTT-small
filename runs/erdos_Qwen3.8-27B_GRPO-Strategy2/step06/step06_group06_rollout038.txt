def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _smax_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        if k < n:

        else:
            lo = k - n + 1
            g[lo:] += wk * (1.0 - 2.0 * h[n - 1:lo - 1:-1])
    g *= dx
    return g

def _lbfgs_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x):
        return x.sum() - n * 0.5
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'disp': False})
        if res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _proj_grad_descent(h, alpha, n_steps, t_end):
    n = len(h)
    h_cur = h.copy()
    f_prev = _smax_obj(h_cur, alpha)
    g_prev = _smax_grad(h_cur, alpha)
    v = np.zeros(n)
    mu = 0.7
    step = 0.05
    for _s in range(n_steps):
        if time.time() > t_end:
            break
        g = _smax_grad(h_cur, alpha)
        v = mu * v + (1.0 - mu) * g
        # Backtracking line search
        accepted = False
        s = step
        for _bt in range(10):
            h_new = _proj(h_cur - s * v, n * 0.5)
            f_new = _smax_obj(h_new, alpha)
            if f_new <= f_prev + 1e-15:
                h_cur = h_new
                f_prev = f_new
                g_prev = g
                step = min(s * 1.1, 0.3)
                accepted = True
                break
            s *= 0.5
        if not accepted:
            step = 0.3 * step
            if step < 1e-8:
                break
    return h_cur

def _multi_lag_descent(h, n_rounds, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = _eval(h_cur)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(c)[::-1][:8]
        g = np.zeros(n)
        for kk in top_k:
            if kk < n:
                g[:kk + 1] += (1.0 - 2.0 * h_cur[kk::-1][:kk + 1])
            else:
                lo = kk - n + 1
                g[lo:] += (1.0 - 2.0 * h_cur[n - 1:lo - 1:-1])
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj(h_cur - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h_cur, c5_cur

def _upsample(h, factor):
    n = len(h)

    # Noise

    # Fourier perturbation

    for freq in range(1, len(H)):
        H[freq] *= (1.0 - 0.02 * rng.rand())
    h4 = np.fft.irfft(H, n=n)
    variants.append(_repair(h4, n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    return h

        if time.time() > t_end:
            break
        c = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx = np.argsort(c)[::-1][:top_k]
        g = np.zeros(n)
        for kk in idx:
            if kk < n:
                g[:kk + 1] += (1.0 - 2.0 * h_cur[kk::-1][:kk + 1])
            else:
                lo = kk - n + 1
                g[lo:] += (1.0 - 2.0 * h_cur[n - 1:lo - 1:-1])
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = _proj(h_cur - st * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h_cur, c5, improved = h_new, c5n, True
                break
        if not improved:
            break
    return h_cur, c5

def _make_variants(h, n, rng):
    out = []
    sh = rng.randint(1, n // 4)
    out.append(_repair(np.roll(h, sh), n))
    bl = rng.randint(n // 10, max(n // 3, 10))
    st = rng.randint(0, n - bl)
    h2 = h.copy()
    h2[st:st + bl] = 1.0 - h2[st:st + bl]
    out.append(_repair(h2, n))
    out.append(_repair(h + 0.01 * rng.randn(n), n))
    H = np.fft.rfft(h)
    for f in range(1, len(H)):
        H[f] *= (1.0 - 0.015 * rng.rand())
    out.append(_repair(np.fft.irfft(H, n=n), n))
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h.copy(), _eval(h), n0

    # Phase 1: L-BFGS-B progressive alpha at base resolution
    for alpha in [200, 1000, 5000, 20000, 80000, 200000, 500000]:
        if rem() < 220:
            break
        h_c = _lbfgs_opt(best_h, alpha, 4000, t0 + rem() - 215)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5, best_h = c5, h_c.copy()

    # Phase 2: Projected gradient with momentum
    if rem() > 180:
        h_c = _proj_grad(best_h, 10000, 800, t0 + rem() - 175)
        c5 = _eval(h_c)
        if c5 < best_c5:
            best_c5, best_h = c5, h_c.copy()

    # Phase 3: Multi-lag targeted descent
    if rem() > 150:
        h_c, c5c = _multi_lag(best_h, 500, t0 + rem() - 145, top_k=8)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()

    # Phase 4: Variant exploration + re-optimization
    if rem() > 350:
        for vh in _make_variants(best_h, best_n, rng):
            if rem() < 320:
                break
            for alpha in [5000, 50000, 200000]:
                if rem() < 300:
                    break
                ht = _lbfgs_opt(vh, alpha, 2000, t0 + rem() - 290)
                c5 = _eval(ht)
                if c5 < best_c5:
                    best_c5, best_h = c5, ht.copy()
            ht, c5c = _multi_lag(best_h, 300, t0 + rem() - 280, top_k=10)
            if c5c < best_c5:
                best_c5, best_h = c5c, ht.copy()

    # Phase 5: Upsample 2x and optimize
    if rem() > 420:
        n1 = n0 * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [200, 1000, 5000, 20000, 80000, 200000]:
            if rem() < 380:
                break
            h_hi = _lbfgs_opt(h_hi, alpha, 3000, t0 + rem() - 375)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if rem() > 340:
            h_hi = _proj_grad(best_h, 8000, 500, t0 + rem() - 335)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        if rem() > 300:
            ht, c5c = _multi_lag(best_h, 400, t0 + rem() - 295, top_k=10)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, ht.copy(), n1

    # Phase 6: Simulated annealing pair swaps
    if rem() > 120:
        t_end_sa = t0 + (time.time() - t0) + (rem() - 100)
        h_sa = best_h.copy()
        c_cur = _eval(h_sa)
        best_sa_h, best_sa_c = h_sa.copy(), c_cur
        t0v, t1v = 0.005, 3e-6
        n_it = min(int((t_end_sa - time.time()) / 0.003), 40000)
        for i in range(n_it):
            if time.time() > t_end_sa:
                break
            t = t0v * (t1v / t0v) ** (i / max(1, n_it - 1))
            i1, i2 = rng.randint(best_n), rng.randint(best_n)
            if i1 == i2:
                continue
            d = rng.randn() * (0.002 + 0.015 * t)
            o1, o2 = h_sa[i1], h_sa[i2]
            h_sa[i1], h_sa[i2] = o1 + d, o2 - d
            h_sa = _repair(h_sa, best_n)
            c_new = _eval(h_sa)
            if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
                c_cur = c_new
                if c_new < best_sa_c:
                    best_sa_c, best_sa_h = c_new, h_sa.copy()
            else:
                h_sa[i1], h_sa[i2] = o1, o2
                h_sa = _repair(h_sa, best_n)
        if best_sa_c < best_c5:
            best_c5, best_h = best_sa_c, best_sa_h.copy()

    # Phase 7: Final polish - targeted multi-lag + L-BFGS-B
    if rem() > 60:
        ht, c5c = _multi_lag(best_h, 400, t0 + rem() - 55, top_k=12)
        if c5c < best_c5:
            best_c5, best_h = c5c, ht.copy()
    if rem() > 35:
        ht = _lbfgs_opt(best_h, 300000, 5000, t0 + rem() - 30)
        c5 = _eval(ht)
        if c5 < best_c5:
            best_c5, best_h = c5, ht.copy()

    # Phase 8: Ultra-fine coordinate descent
    if rem() > 15:
        h_cd = best_h.copy()
        nc = len(h_cd)
        c5c = _eval(h_cd)
        t_end_cd = t0 + (time.time() - t0) + (rem() - 10)
        while time.time() < t_end_cd:
            i, j = rng.randint(nc), rng.randint(nc)
            if i == j:
                continue
            d = 3e-5 * rng.randn()
            ni, nj = h_cd[i] + d, h_cd[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h_cd[i], h_cd[j] = ni, nj
            c5n = _eval(h_cd)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h_cd.copy()
            else:
                h_cd[i], h_cd[j] = h_cd[i] - d, h_cd[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
