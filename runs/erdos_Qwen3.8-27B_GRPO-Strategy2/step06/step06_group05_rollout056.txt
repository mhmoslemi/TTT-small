import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp_opt(h0, alpha, maxiter):
    n = len(h0)
    bnds = Bounds(np.zeros(n), np.ones(n))
    A = np.ones((1, n))
    lc = LinearConstraint(A, n * 0.5, n * 0.5)
    try:
        res = minimize(_lse_obj, h0.copy(), jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=[lc],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _trust_opt(h0, alpha, maxiter):
    n = len(h0)
    bnds = Bounds(np.zeros(n), np.ones(n))
    A = np.ones((1, n))
    lc = LinearConstraint(A, n * 0.5, n * 0.5)
    try:
        res = minimize(_lse_obj, h0.copy(), jac=_lse_grad, args=(alpha,),
                       method='trust-constr', bounds=bnds, constraints=[lc],
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'verbose': 0})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _binary_greedy(h0, max_time, rng, use_anneal=True):
    n = len(h0)
    h = (h0 > 0.5).astype(np.float64).copy()
    h = _repair(h, n)
    best_c5 = _eval(h)
    best_h = h.copy()
    stuck_count = 0
    T = 0.005
    while time.time() < max_time:
        cor = np.correlate(h, 1.0 - h, mode='full')
        km = int(np.argmax(cor))
        s = (n - 1) - km
        cur_max = cor[km]
        best_improve = -1e9
        best_pair = None
        best_new_max = cur_max
        candidates = []
        if s > 0:
            for i in range(s, n):
                if h[i] == 1.0 and h[i - s] == 0.0:
                    candidates.append((i, i - s))
        elif s < 0:
            sp = -s
            for i in range(n - sp):
                if h[i] == 1.0 and h[i + sp] == 0.0:
                    candidates.append((i, i + sp))
        if len(candidates) > 200:

            candidates = [candidates[i] for i in idxs]

            nm = np.max(np.correlate(h, 1.0 - h, mode='full'))
            h[a], h[b] = h[b], h[a]

                best_improve = cur_max - nm
                best_pair = (a, b)
        if best_pair is not None and best_improve > 0:

def _resample(h, n_new):

def _worst_shift_descent(h0, max_time):
    n = len(h0)
    h = h0.copy()
    best_c5 = _eval(h)
    dx = 2.0 / n
    for _ in range(500):

            break

        else:

                moved = True
