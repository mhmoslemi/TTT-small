    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

def _soft_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n-sh])
            g[:n-sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n-sp]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h0, alpha, maxiter=3000):

                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:
        return _repair(h0, n)

    xn = np.linspace(0, 2, n_new, endpoint=False)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()

            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _proj_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _time_end
    t0 = time.time()
    _time_end = t0 + budget_s
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Momentum subgradient (25% budget)
    t_end1 = t0 + budget_s * 0.25
    h1, c1 = _subgrad_opt(best_h, t_end1, lr_init=0.005, seed=seed)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: SLSQP full resolution, increasing alpha (25% budget)
    t_end2 = t0 + budget_s * 0.50
    for alpha in [10000, 100000, 800000]:
        if time.time() > t_end2:
            break
        h_new = _slsqp(best_h.copy(), alpha, maxiter=2500)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 3: Periodic SLSQP T=31 (15% budget)
    if n0 % 31 == 0 and time.time() < t0 + budget_s * 0.65:
        m = n0 // 31
        h_p = best_h.reshape(m, 31).mean(axis=0)
        h_p = _repair(h_p, 31)
        for alpha in [50000, 500000, 3000000]:
            if time.time() > t0 + budget_s * 0.65:
                break
            h_p = _slsqp(h_p, alpha, maxiter=5000)
            h_full = np.tile(h_p, m)
            c5 = _eval(h_full)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_full

    # Phase 4: Subgradient on periodic pattern then expand (10% budget)
    if n0 % 31 == 0 and time.time() < t0 + budget_s * 0.75:
        m = n0 // 31
        h_full = np.tile(h_p, m) if 'h_p' in dir() else best_h
        h_sub, c_sub = _subgrad_opt(h_full, t0 + budget_s * 0.75, lr_init=0.003, seed=seed+1)
        if c_sub < best_c5:
            best_c5 = c_sub
            best_h = h_sub

    # Phase 5: Try other periodicities (10% budget)
    t_end5 = t0 + budget_s * 0.85
    for T in [10, 5, 62]:
        if n0 % T != 0 or time.time() > t_end5:
            continue
        m = n0 // T
        h_p = best_h[:n0].reshape(m, T).mean(axis=0) if best_n == n0 else _repair(_resample(best_h, n0).reshape(m, T).mean(axis=0), T)
        h_p = _repair(h_p, T)
        h_p = _slsqp(h_p, 300000, maxiter=4000)
        h_full = np.tile(h_p, m)
        c5 = _eval(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_full

    # Phase 6: Coordinate descent polish (10% budget)
    for scale in [0.004, 0.001, 0.0003, 0.0001]:
        if time.time() > t0 + budget_s - 8:
            break
        h_cd, c5 = _coord_descent(best_h.copy(), 8000, scale, t0 + budget_s - 5, seed=seed + int(scale * 1e7))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd

    # Phase 7: Final SLSQP + try nearby n
    if time.time() < t0 + budget_s - 15:
        h_f = _slsqp(best_h.copy(), 2000000, maxiter=2000)
        c5 = _eval(h_f)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_f

    if time.time() < t0 + budget_s - 12:
        for dn in [10, -10, 31]:
            if time.time() > t0 + budget_s - 10:
                break
            n_try = best_n + dn
            if n_try < 50 or n_try > 600:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _slsqp(h_r, 200000, maxiter=2000)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r
                best_n = n_try

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
