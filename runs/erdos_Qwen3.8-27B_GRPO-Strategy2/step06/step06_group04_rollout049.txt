import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k

        if sh >= 0:

            g[i1] += wk * (1.0 - x[i1 + sh])

        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 2: Adam refinement (155-310s)
    dl = t0 + 305
    for (a0, a1, ni, lr0, lr1) in [(2000, 50000, 4000, 0.025, 0.001),
                                    (50000, 500000, 4000, 0.01, 0.0003),
                                    (500000, 5000000, 3000, 0.004, 0.0001)]:
        if time.time() > dl:
            break
        x, c = _adam(best_h, a0, a1, ni, lr0, lr1, seed, dl)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    # Phase 3: Multi-n exploration with L-BFGS-B (305-480s)
    dl = t0 + 475
    n_list = [240, 260, 280, 300, 320, 340, 360, 400, 420, 460, 500, 540]
    for nt in n_list:
        if time.time() > dl:
            break
        hr = _repair(_resample(best_h, nt), nt)
        hn = _lbfgsb(hr, 30000, 2000, dl)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hn.copy(), nt
        x, c = _adam(hn, 5000, 200000, 2000, 0.01, 0.0003, seed + nt, dl)
        if c < best_c5:
            best_c5, best_h, best_n = c, x.copy(), nt
        hn2 = _lbfgsb(x, 200000, 1500, dl)
        c5 = _eval(hn2)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hn2.copy(), nt

    # Phase 4: SLSQP with constraint (475-600s)
    dl = t0 + 595
    for a in [200000, 800000, 3000000]:
        if time.time() > dl:
            break
        hn = _slsqp(best_h, a, 4000, dl)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 5: Multi-start L-BFGS-B on best_n (600-730s)
    dl = t0 + 725
    mh, mc = _multi_start_lbfgsb(best_h, best_n, [50000, 200000, 500000], rng, dl)
    if mc < best_c5:
        best_c5, best_h = mc, mh.copy()

    # Phase 6: Binary SA + Adam (725-860s)
    dl = t0 + 855
    xb, cb = _bin_sa(best_h, best_n, rng, dl)
    if cb < best_c5:
        best_c5, best_h = cb, xb.copy()
    if time.time() < dl:
        xp, cp = _adam(xb, 3000, 200000, 3000, 0.006, 0.0002, seed + 555, dl)
        if cp < best_c5:
            best_c5, best_h = cp, xp.copy()

    # Phase 7: Very high alpha L-BFGS-B (860-990s)
    dl = t0 + 985
    for a in [2000000, 5000000, 10000000, 20000000]:
        if time.time() > dl:
            break
        hn = _lbfgsb(best_h, a, 3000, dl)
        c5 = _eval(hn)
        if c5 < best_c5:
            best_c5, best_h = c5, hn.copy()

    # Phase 8: Paired CD fine-tuning (990-1080s)
    dl = t0 + 1078
    xc, cc = _pair_cd(best_h, best_n, rng, dl, 0.003)
    if cc < best_c5:
        best_c5, best_h = cc, xc.copy()

    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
