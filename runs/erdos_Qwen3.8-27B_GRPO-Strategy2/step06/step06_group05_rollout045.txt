def _grad_single(h, k_idx, n):
    """Gradient of c[k_idx] w.r.t. h, where c = correlate(h, 1-h) * dx"""
    dx = 2.0 / n
    g = np.zeros(n)

    # c[k_idx] = sum_i h[i+shift_valid] * (1-h[i]) for valid i

    s = k_idx - (n - 1)

        # c = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
        # d/dc/dh[j]:
        #   h[j] as h[i+s] → i = j-s, valid if 0<=j-s and j-s <= n-1-s → s<=j<=n-1
        #     contributes (1-h[j-s])
        #   h[j] as h[i] → i = j, valid if 0<=j and j <= n-1-s → 0<=j<=n-1-s
        #     contributes -h[j+s]
        top = n - s
        if top > 0:
            # h[j] as first factor (h[i+s]): j in [s, n-1]
            g[s:n] += (1.0 - h[:top])  # (1-h[j-s]) for j in [s, n-1]
            # h[j] as second factor (1-h[i]): j in [0, n-1-s]
            g[:top] += -h[s:]  # -h[j+s] for j in [0, top-1]
    else:
        sp = -s
        # c = sum_{i=sp}^{n-1} h[i]*(1-h[i-sp])
        # d/dc/dh[j]:
        #   h[j] as h[i] → i=j, valid if sp<=j<=n-1
        #     contributes (1-h[j-sp])
        #   h[j] as h[i-sp] → i=j+sp, valid if sp<=j+sp and j+sp<=n-1 → 0<=j<=n-1-sp
        #     contributes -h[j+sp]
        if sp < n:
            g[sp:n] += (1.0 - h[:n-sp])
            g[:n-sp] += -h[sp:]
    return g * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        if abs(diff) > 1e-12:

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])

                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

def _eval(h):

def _corr(h):

def _grad_ck(h, k_idx, n):
    dx = 2.0 / n
    s = k_idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        top = n - s
        if top > 0:
            g[s:n] += (1.0 - h[:top])

    else:
        sp = -s
        if sp < n:
            g[sp:n] += (1.0 - h[:n - sp])
            g[:n - sp] += -h[sp:]
    return g * dx

def _slsqp_polish(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def _f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip((c - m) * alpha, -400, 400)
        s -= s.max()
        return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

    def _gf(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip((c - m) * alpha, -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            g += w[k] * _grad_ck(x, k, n)
        return g

    def _feq(x):
        return x.sum() - n * 0.5

    def _jeq(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _feq, 'jac': _jeq}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

        return _repair(h, n)

            h7[i], h7[j] = ni, nj
            c5n = _eval(h7)
            if c5n < c5c - 1e-16:
                c5c = c5n
            else:
                h7[i], h7[j] = ni - d, nj + d
        if c5c < best_c5:
            best_c5, best_h = c5c, h7.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
