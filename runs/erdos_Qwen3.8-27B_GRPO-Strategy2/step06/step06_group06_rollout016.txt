    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    """Returns (correlations, list of per-k gradients) - actually just correlations"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_k(h, k, n):
    """Gradient of c_k with respect to h"""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _smoothmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    logsum = np.log(np.sum(np.exp(s))) / alpha
    return (m + logsum) * dx

def _smoothmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    active = w > 1e-13
    for k in np.where(active)[0]:
        wk = w[k]
        g += wk * _grad_k(h, k, n)
    return g

def _penalty_obj(h, M, power=2):
    """Penalty: sum of (c_k - M)^p for c_k > M"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    excess = np.maximum(c - M / (2.0/n), 0)
    return np.sum(excess**power) * (2.0/n)**power

def _topk_focus_obj(h, K):

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.sort(c)[::-1][:K]
    return top.sum() * dx

def _topk_focus_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for k in idx:
        g += _grad_k(h, k, n)
    return g

def _proj_grad_descent(h, obj_fn, grad_fn, n_steps, t_end, lr_init=0.03, mu=0.6):
    n = len(h)
    h_cur = h.copy()
    f_cur = obj_fn(h_cur)
    best_f = f_cur
    best_h = h_cur.copy()
    v = np.zeros(n)
    lr = lr_init
    for i in range(n_steps):
        if time.time() > t_end:
            break
        g = grad_fn(h_cur)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        v = mu * v + (1.0 - mu) * (g / gnorm * gnorm)  # = mu*v + (1-mu)*g
        accepted = False
        for _ in range(12):
            h_new = _proj(h_cur - lr * v, n * 0.5)
            f_new = obj_fn(h_new)
            if f_new <= f_cur + 1e-13:
                h_cur = h_new
                f_cur = f_new
                accepted = True
                lr = min(lr * 1.08, 0.15)
                if f_cur < best_f - 1e-10:
                    best_f = f_cur
                    best_h = h_cur.copy()
                v = 0.9 * v
                break
            lr *= 0.4
        if not accepted:
            lr *= 0.3
            v = 0.5 * v
    return best_h

def _lbfgs(h, obj_fn, grad_fn, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    f0 = obj_fn(h)
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        if res.fun < f0 - 1e-13:
            return _repair(res.x, n)
    except:
        pass
    return _repair(h, n)

def _block_sa(h, t_end, rng, n_max_blocks=50000):

    c_cur = _eval(h)
    best_h = h.copy()
    best_c = c_cur
    t_temps = [0.02, 5e-4]
    n_trials = int((t_end - time.time()) / 0.0025)
    n_trials = max(100, min(n_trials, n_max_blocks))
    for i in range(n_trials):
        if time.time() > t_end:
            break
        frac = i / max(1, n_trials - 1)
        t = t_temps[0] * (t_temps[1] / t_temps[0]) ** frac
        blen = max(2, int(np.exp(rng.randn() * 0.8 + np.log(5))))
        blen = min(blen, n // 4)
        pos = rng.randint(0, n - blen)
        d = rng.randn(blen)
        d -= d.mean()
        d *= (0.02 + 0.06 * t / t_temps[0])
        old = h[pos:pos+blen].copy()
        h[pos:pos+blen] = old + d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new <= c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-12)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[pos:pos+blen] = old
            h = _repair(h, n)
    return best_h, best_c

def _upsample_interp(h, factor):

    x_new = np.linspace(0, n - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:

                h[idx] -= a
                diff += a

        for K in [8, 4, 2]:
            if rem() < 50:
                break
            h_try = _pgd_topk(best_h, K, 400, t0 + rem() - 49, 0.04, 0.7)
            c5 = _eval(h_try)
            if c5 < best_c5:
                best_c5, best_h = c5, h_try.copy()

    # Phase 2: smooth-max base res
    if rem() > 130:
        for alpha in [8000, 40000, 200000]:
            if rem() < 110:
                break
            h_try = _pgd_smoothmax(best_h, alpha, 600, t0 + rem() - 109, 0.02, 0.7)
            c5 = _eval(h_try)
            if c5 < best_c5:
                best_c5, best_h = c5, h_try.copy()

    # Phase 3: upsample to 2x, optimize
    if rem() > 380:
        n1 = n0 * 2
        h_a = _repair(_upsample_interp(best_h, 2), n1)
        h_b = _repair(np.repeat(best_h, 2), n1)
        h_hi = h_a if _eval(h_a) <= _eval(h_b) else h_b
        for alpha in [3000, 15000, 80000, 400000]:
            if rem() < 330:
                break
            h_hi = _pgd_smoothmax(h_hi, alpha, 500, t0 + rem() - 329, 0.02, 0.65)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), n1
        for K in [6, 3, 1]:
            if rem() < 270:
                break
            h_hi = _pgd_topk(best_h, K, 350, t0 + rem() - 269, 0.02, 0.72)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_hi.copy(), best_n
        if rem() > 220:
            for alpha in [80000, 500000]:
                if rem() < 190:
                    break
                h_hi = _lbfgs_sm(best_h, alpha, 2500, t0 + rem() - 189)
                c5 = _eval(h_hi)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h_hi.copy(), n1

    # Phase 4: restart with perturbation + re-optimize
    if rem() > 180:
        h_pert = _repair(best_h + 0.01 * rng.randn(len(best_h)), len(best_h))
        for K in [4, 2]:
            if rem() < 150:
                break
            h_try = _pgd_topk(h_pert, K, 300, t0 + rem() - 149, 0.03, 0.7)
            c5 = _eval(h_try)
            if c5 < best_c5:
                best_c5, best_h = c5, h_try.copy()

    # Phase 5: Block SA
    if rem() > 100:
        h_try, c5_try = _block_sa(best_h, t0 + rem() - 90, rng)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()

    # Phase 6: final top-K polish
    if rem() > 35:
        for K in [2, 1]:
            if rem() < 25:
                break
            h_try = _pgd_topk(best_h, K, 250, t0 + rem() - 24, 0.01, 0.8)
            c5 = _eval(h_try)
            if c5 < best_c5:
                best_c5, best_h = c5, h_try.copy()

    # Phase 7: ultra-fine coordinate descent
    if rem() > 12:
        n_pts = len(best_h)
        h_cur = best_h.copy()
        c5c = _eval(h_cur)
        t_end_cd = t0 + rem() - 6
        while time.time() < t_end_cd:
            g = _tk_grad(h_cur, 2)
            gn = np.linalg.norm(g)
            if gn > 1e-13:
                for lr in [0.008, 0.003, 0.001]:
                    h_try = _proj(h_cur - lr * g / gn, n_pts * 0.5)
                    c_try = _eval(h_try)
                    if c_try < c5c - 1e-14:
                        h_cur, c5c = h_try, c_try
                        break
            i, j = rng.randint(n_pts), rng.randint(n_pts)
            if i != j:
                d = 8e-5 * rng.randn()
                if 0 <= h_cur[i] + d <= 1 and 0 <= h_cur[j] - d <= 1:
                    h_cur[i], h_cur[j] += d, -d
                    c_try = _eval(h_cur)
                    if c_try < c5c:
                        c5c = c_try
                    else:
                        h_cur[i], h_cur[j] -= d, -d
        if c5c < best_c5:
            best_c5, best_h = c5c, h_cur.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
