def _repair(h, n):

    target = n / 2.0

            else:

                h[idx] -= a; diff += a

    h = h.copy()
    cur = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        for j in range(n):
            # Compute A_s(j) for all s = -(n-1), ..., n-1
            # s = m - (n-1) where m is the index in c
            # For each m (0 to 2n-2), s = m - (n-1)
            # A depends on whether j-s and j+s are in [0, n-1]
            
            s_vals = np.arange(-(n-1), n)  # shape (2n-1,)
            js = j - s_vals  # j - s, should be in [0, n-1]
            jp = j + s_vals  # j + s, should be in [0, n-1]
            
            valid_s = (js >= 0) & (js < n)  # j-s valid
            valid_p = (jp >= 0) & (jp < n)  # j+s valid
            
            A = np.zeros(2*n-1)
            # Both valid: A = 1 - h[j-s] - h[j+s]
            both = valid_s & valid_p
            A[both] = 1.0 - h[js[both]] - h[jp[both]]
            # Only s valid: A = 1 - h[j-s]
            only_s = valid_s & ~valid_p
            A[only_s] = 1.0 - h[js[only_s]]
            # Only p valid: A = -h[j+s]
            only_p = ~valid_s & valid_p
            A[only_p] = -h[jp[only_p]]
            # Neither: A = 0 (already)
            
            B = c - h[j] * A
            t_new = _optimal_t(B, A)
            
            if abs(t_new - h[j]) > 1e-14:
                delta = t_new - h[j]
                c += delta * A
                h[j] = t_new
    
    h = _repair(h, n)
    cur = _eval(h)
    return h, cur

import numpy as np
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _optimal_t(B, A):
    lo, hi = 0.0, 1.0
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        vals = B + A * mid
        fval = vals.max()
        active = vals >= fval - 1e-14
        if not active.any():
            active = vals == fval
        gmin = A[active].min()
        gmax = A[active].max()
        if gmin > 1e-14:
            hi = mid
        elif gmax < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def _cd(h, n, n_sweeps, t_end, seed=0):
    h = _repair(h.copy(), n)
    cur = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.RandomState(seed)
    s_vals = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        max_chg = 0.0
        for j in order:
            js = j - s_vals
            jp = j + s_vals
            vs = (js >= 0) & (js < n)
            vp = (jp >= 0) & (jp < n)
            A = np.zeros(2*n-1)
            both = vs & vp
            if both.any():
                A[both] = 1.0 - h[js[both]] - h[jp[both]]
            os_ = vs & ~vp
            if os_.any():
                A[os_] = 1.0 - h[js[os_]]
            op = ~vs & vp
            if op.any():
                A[op] = -h[jp[op]]
            B = c - h[j] * A
            t_new = _optimal_t(B, A)
            d = t_new - h[j]
            if abs(d) > 1e-14:
                c += d * A
                h[j] = t_new
                if abs(d) > max_chg:
                    max_chg = abs(d)
        if sweep % 10 == 0:
            new_c = _eval(h)
            if new_c < cur:
                cur = new_c
            elif sweep > 100 and max_chg < 1e-10:
                break
    h = _repair(h, n)
    return h, _eval(h)

    # Phase 4: Stochastic restarts from best
    if rem() > 100:
        for restart in range(4):
            if rem() < 60:
                break
            rng2 = np.random.RandomState(seed + 500 + restart)
            h_try = best_h.copy()
            # Small perturbation
            noise = rng2.normal(0, 0.02, best_n)
            h_try = _repair(h_try + noise, best_n)
            hc, c5 = _cd(h_try, best_n, 10000, t0 + rem() - 20, seed=300 + restart)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hc.copy(), best_n

    # Phase 5: Binary projection + CD
    if rem() > 60:
        hb = (best_h > 0.5).astype(float)
        hb = _repair(hb, best_n)
        hc, c5 = _cd(hb, best_n, 10000, t0 + rem() - 15, seed=400)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hc.copy(), best_n

    # Phase 6: Final long CD
    if rem() > 30:
        hc, c5 = _cd(best_h, best_n, 30000, t0 + rem() - 5, seed=999)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hc.copy(), best_n

    # Phase 7: Try a few more resolutions near best
    if rem() > 20:
        for n_try in [best_n - 20, best_n + 20, best_n + 50]:
            if rem() < 12 or n_try < 100:
                continue
            hr = _resample(best_h, n_try)
            hr = _repair(hr, n_try)
            hc, c5 = _cd(hr, n_try, 5000, t0 + rem() - 5, seed=800)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, hc.copy(), n_try

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
