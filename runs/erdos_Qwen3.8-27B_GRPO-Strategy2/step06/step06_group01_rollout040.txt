    target = n / 2.0

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = c.max()
        s = np.clip(alpha * (c - cm), -400, 400)
        val = cm + np.log(float(np.sum(np.exp(s)))) / alpha
        pen = 0.1 * (x.sum() - target) ** 2
        return val + pen

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = c.max()
        s = np.clip(alpha * (c - cm), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] <= 1e-10:
                continue
            sh = (n - 1) - k
            wk = w[k] * alpha * dx
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wk * (1.0 - x[sh:sh + cnt])
                if sh > 0:
                    g[:cnt] -= wk * x[:cnt]
            else:
                sp = -sh
                cnt = n - sp
                g[sp:sp + cnt] += wk * (1.0 - x[:cnt])
                g[:cnt] -= wk * x[sp:sp + cnt]
        g += 0.2 * (x.sum() - target) * np.ones(n)
        return g

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _binary_init(n, family, rng):
    h = np.zeros(n)
    if family == 0:
        period = rng.choice([3, 4, 5, 6, 7, 8])
        pattern = [1] * (period // 2) + [0] * (period - period // 2)
        for i in range(n):
            h[i] = pattern[i % period]
    elif family == 1:
        m = n // 4
        idx = rng.choice(n, size=m, replace=False)
        h[idx] = 1.0
    elif family == 2:
        block = max(2, n // 20)
        for s in range(0, n, block * 2):
            h[s:min(s + block, n)] = 1.0
    else:
        h = (np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n) + 1) / 2
        h = (h > 0.5).astype(float)
    return _repair(h, n)

def _coord_descent(h, n, n_sweeps, t0, deadline):
    dx = 2.0 / n
    c = _eval_corr(h)
    c5c = c.max() * dx
    for sweep in range(n_sweeps):
        if time.time() > deadline - 10:
            break
        idx = np.random.permutation(n)
        for i in idx:
            old = h[i]
            best_val = old
            best_obj = c5c
            for t_cand in [0.0, 1.0, old + 0.02, old - 0.02, 0.5, max(0, old + 0.005), min(1, old - 0.005)]:
                if t_cand < 0 or t_cand > 1 or abs(t_cand - old) < 1e-12:
                    continue
                h[i] = t_cand
                c_new = _eval_corr(h)
                obj = c_new.max() * dx
                if obj < best_obj - 1e-15:
                    best_val, best_obj = t_cand, obj
            h[i] = best_val
            c5c = best_obj
        c = _eval_corr(h)
        c5c = c.max() * dx
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: L-BFGS on smoothed objective
    if time.time() < deadline - 200:
        for alpha in [5000, 20000, 80000]:
            if time.time() > deadline - 100:
                break
            h1 = _lbfgs_smooth(best_h, best_n, alpha, t0, deadline, maxiter=1500)
            c5 = _eval(h1)
            if c5 < best_c5:
                best_c5, best_h = c5, h1.copy()

    # Phase 2: Nesterov subgradient with top-3
    if time.time() < deadline - 150:
        h2, c5 = _nesterov_descent(best_h, best_n, 3, 500, t0, deadline - 50, 0.03)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 3: Nesterov with top-5, top-8
    if time.time() < deadline - 120:
        for M in [5, 8]:
            if time.time() > deadline - 50:
                break
            h3, c5 = _nesterov_descent(best_h, best_n, M, 400, t0, deadline - 40, 0.02)
            if c5 < best_c5:
                best_c5, best_h = c5, h3

    # Phase 4: L-BFGS again with higher alpha
    if time.time() < deadline - 180:
        for alpha in [150000, 400000]:
            if time.time() > deadline - 80:
                break
            h4 = _lbfgs_smooth(best_h, best_n, alpha, t0, deadline, maxiter=2000)
            c5 = _eval(h4)
            if c5 < best_c5:
                best_c5, best_h = c5, h4.copy()

    # Phase 5: Nesterov polish (very small steps)
    if time.time() < deadline - 80:
        h5, c5 = _nesterov_descent(best_h, best_n, 3, 800, t0, deadline - 30, 0.008)
        if c5 < best_c5:
            best_c5, best_h = c5, h5
        h5, c5 = _nesterov_descent(best_h, best_n, 1, 1000, t0, deadline - 20, 0.003)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 6: Try smaller n values
    if time.time() < deadline - 200:
        for n_try in [100, 150, 200, 250, 300, 400, 500]:
            if time.time() > deadline - 50:
                break
            # Start from parent resampled
            h_r = _repair(_resample(best_h, n_try), n_try)
            # Quick L-BFGS
            h_r = _lbfgs_smooth(h_r, n_try, 30000, t0, deadline, maxiter=800)
            # Nesterov polish
            h_r, c5 = _nesterov_descent(h_r, n_try, 3, 300, t0, deadline - 20, 0.02)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h_r.copy(), n_try

    # Phase 7: Binary starts at small n
    if time.time() < deadline - 150:
        for n_b in [60, 100, 140, 200]:
            if time.time() > deadline - 40:
                break
            for fam in range(4):
                if time.time() > deadline - 40:
                    break
                hb = _binary_init(n_b, fam, rng)
                hb = _lbfgs_smooth(hb, n_b, 20000, t0, deadline, maxiter=600)
                hb, c5 = _nesterov_descent(hb, n_b, 3, 200, t0, deadline - 20, 0.02)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, hb.copy(), n_b

    # Phase 8: Coordinate descent final polish
    if time.time() < deadline - 20:
        h8, c5 = _coord_descent(best_h, best_n, 3, t0, deadline - 10)
        if c5 < best_c5:
            best_c5, best_h = c5, h8

    # Phase 9: Ultra-fine Nesterov
    if time.time() < deadline - 10:
        h9, c5 = _nesterov_descent(best_h, best_n, 1, 500, t0, deadline, 0.001)
        if c5 < best_c5:
            best_c5, best_h = c5, h9

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
