import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    amt = min(diff, 1.0 - h[idx]); h[idx] += amt; diff -= amt
                else:
                    amt = min(-diff, h[idx]); h[idx] -= amt; diff += amt
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x): return np.array([x.sum() - 0.5 * n])
    def _cj(x): return np.array([np.ones(n)])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smax, h, jac=_sgrad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad(h, n_iter, step0, t0, time_limit):

    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el(): return time.time() - t0
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c = _eval(best_h); best_n = n0

    # Phase 1: SLSQP warm-start
    for alpha in [500, 2000, 20000]:
        if _el() > 80: break
        hc = _slsqp(best_h, alpha, 2000, 1e-16)
        c = _eval(hc)
        if c < best_c: best_c = c; best_h = hc

    # Phase 2: Coordinate descent (main new algorithm)
    t_cd_end = budget_s * 0.55
    best_h, best_c = _coord_descent(best_h, n0, 200, t0, t_cd_end)
    best_n = n0

    # Phase 3: SLSQP polish after coord descent
    for alpha in [50000, 200000]:
        if _el() > budget_s * 0.65: break
        hc = _slsqp(best_h, alpha, 1500, 1e-16)
        c = _eval(hc)
        if c < best_c: best_c = c; best_h = hc

    # Phase 4: Another round of coord descent (finer)
    t_cd2 = budget_s * 0.75
    if _el() < t_cd2:
        best_h, best_c = _coord_descent(best_h, best_n, 100, t0, t_cd2)

    # Phase 5: Subgradient polish
    if _el() < budget_s - 200:
        n_it = min(3000, int((budget_s - 200 - _el()) / 0.02))
        best_h, best_c = _subgrad(best_h, n_it, 0.012, t0, budget_s - 150)

    # Phase 6: Try upscaled resolution
    if _el() < budget_s - 150:
        n1 = best_n * 2
        h_hi = np.repeat(best_h, 2)
        h_hi = _repair(h_hi, n1)
        h_hi, c_hi = _coord_descent(h_hi, n1, 30, t0, budget_s - 100)
        if c_hi < best_c:
            best_c = c_hi; best_h = h_hi; best_n = n1

    # Phase 7: Final SLSQP
    if _el() < budget_s - 40:
        hc = _slsqp(best_h, 500000, 2000, 1e-17)
        c = _eval(hc)
        if c < best_c: best_c = c; best_h = hc

    # Phase 8: Final coord descent
    if _el() < budget_s - 15:
        best_h, best_c = _coord_descent(best_h, best_n, 30, t0, budget_s - 10)

    # Phase 9: Fine pairwise search
    if _el() < budget_s - 5:
        n = best_n; h_cur = best_h.copy(); c_cur = best_c
        n_tr = min(int((budget_s - 5 - _el()) / 0.004), 80000)
        for _ in range(n_tr):
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            d = 2e-5 * rng.randn()
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            h_cur[i], h_cur[j] = ni, nj
            cn = _eval(h_cur)
            if cn < c_cur:
                c_cur = cn
                best_c = cn; best_h = h_cur.copy()
            else:
                h_cur[i], h_cur[j] = ni - d, nj + d

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, best_c, best_n)
