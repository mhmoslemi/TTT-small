def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _sobj(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _sgrad(h, alpha):

            sp = -sh

            g[:n-sp] -= wk * h[sp:]
    return g * dx

def _topk_descent(h, n_rounds, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()

                g[:n-sh] += (1.0 - h_cur[sh:])

                g[:n-sp] -= h_cur[sp:]
