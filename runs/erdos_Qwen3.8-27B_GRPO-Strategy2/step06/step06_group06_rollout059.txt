    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

    """Gradient of soft-max objective. If top_k given, only use top-k lags."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        # Zero out non-top entries
        c_masked = np.full_like(c, c.min() - 1.0)
        c_masked[top_idx] = c[top_idx]
        c_use = c_masked
    else:
        c_use = c
    m = c_use.max()
    s = np.clip(alpha * (c_use - m), -500, 500)
    w = np.exp(s - s.max())
    w_sum = w.sum()
    if w_sum < 1e-30:

    w /= w_sum
    g = np.zeros(n)

        else:

            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]

def slsqp_phases(h, alphas, maxiter_each, t_end):
    """Run SLSQP with increasing alpha."""
    n = len(h)

    for alpha in alphas:
        if time.time() > t_end:

                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14, 'disp': False})
            if res.fun < smax_obj(h_cur, alpha) - 1e-14:
                h_cur = repair(res.x, n)
        except Exception:
            pass
    return h_cur

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)

def _smax_grad(h, alpha, top_k=None):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_use = np.full_like(c, -np.inf)
        c_use[idx] = c[idx]
    else:
        c_use = c
    m = c_use.max()
    s = np.clip(alpha * (c_use - m), -500, 500)
    w = np.exp(s - s.max())
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g *= 2.0 / n
    return g

def _active_grad(h, n_lags=5):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argpartition(c, -n_lags)[-n_lags:]
    g = np.zeros(n)
    for k in top:
        sh = k - (n - 1)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    g *= 2.0 / n
    return g

def _proj_grad(h, alpha, n_iter, t_end, top_k=15, mom=0.75):
    n = len(h)

    fc = _smax_obj(hc, alpha)
    v = np.zeros(n)

        return np.ones(n)
