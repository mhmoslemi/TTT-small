import numpy as np
from scipy.optimize import minimize
import time

_N_CONST = [0]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _bit_threshold(h, n):
    h = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        sel = np.argsort(h[idx])[:n_ones - target]
        h[idx[sel]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        sel = np.argsort(-h[idx])[:target - n_ones]
        h[idx[sel]] = 1
    return h

def _block_mutate(h, rng, n_blocks, len_frac, mag):
    h = h.copy()
    n = len(h)
    for _ in range(n_blocks):
        s = rng.randint(n)
        l = max(1, rng.randint(1, max(2, int(n * len_frac)) + 1))
        e = min(s + l, n)
        h[s:e] += mag * rng.randn()
    return _proj_to_sum(np.clip(h, 0, 1), n * 0.5)

def _topk_descent(h, n_top, max_rounds, bc):

    dx = 2.0 / n
    c5c = _eval(h)
    mom = np.zeros(n)
    beta = 0.85
    for r in range(max_rounds):
        if bc():

        c = np.correlate(h, 1.0 - h, mode='full') * dx

        g = np.zeros(n)

            w = max(c[k] - c.max() + 0.005, 0.005)
            if sh >= 0:

                g[i1] += w * (1.0 - h[i1 + sh])
                if sh > 0:

                    g[i2] -= w * h[i2 - sh]

                i1 = np.arange(sp, n)
                g[i1] += w * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w * h[i2 + sp]

        x = np.linspace(0, 2, n_low)
        if i < 5:
            p = h_low + 0.2 * rng.randn(n_low)
        elif i < 10:
            freq = 1 + rng.randint(4)
            p = 0.5 + 0.25 * np.sin(2 * np.pi * freq * x / 2 + 2 * np.pi * rng.rand())
            p += 0.1 * rng.randn(n_low)
        else:
            p = np.full(n_low, 0.5)
            for _ in range(rng.randint(3, 8)):
                s = rng.randint(n_low)
                l = rng.randint(5, max(6, n_low // 4))
                p[s:s + l] += rng.choice([-0.3, 0.3])
            p += 0.05 * rng.randn(n_low)
        pop.append(_repair(p, n_low))
    if _el() < budget_s - 500:
        ga_b, ga_c = _ga_run(pop, rng, 350, _bc, 4)
        h_up = _repair(np.repeat(ga_b, f_d), n0)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: GA at full resolution
    if _el() < budget_s - 350:
        pop2 = [best_h.copy()]
        for i in range(14):
            if i < 4:
                p = _block_mutate(best_h, rng, 2, 0.05, 0.05)
            elif i < 8:
                p = _block_mutate(best_h, rng, 4, 0.1, 0.1)
            else:
                p = _block_mutate(best_h, rng, 6, 0.15, 0.15)
            pop2.append(p)
        ga_b2, ga_c2 = _ga_run(pop2, rng, 250, _bc, 3)
        if ga_c2 < best_c5:
            best_c5 = ga_c2
            best_h = ga_b2.copy()

    # Phase 3: Multi-k top gradient descent
    for nl in [3, 5, 8, 4, 10, 2, 6]:
        if _el() > budget_s - 200:
            break
        h_ml, c5_ml = _topk_descent(best_h.copy(), nl, 600, _bc)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 4: 0-1 combinatorial swap search
    if _el() < budget_s - 150:
        n_swap = min(int((budget_s - 150 - _el()) / 0.004), 200000)
        h_bit, c5_bit = _swap_search(best_h, n_swap, _bc, rng)
        if c5_bit < best_c5:
            best_c5 = c5_bit
            best_h = h_bit.copy()

    # Phase 5: SLSQP on (possibly 0-1) solution
    if _el() < budget_s - 100:
        for alpha in [50000, 300000, 2000000]:
            if _el() > budget_s - 30:
                break
            h_s = _slsqp_refine(best_h, alpha, len(best_h), 400)
            c5s = _eval(h_s)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = h_s.copy()

    # Phase 6: Another 0-1 search on SLSQP result
    if _el() < budget_s - 60:
        h_bit2, c5_bit2 = _swap_search(best_h, 100000, _bc, rng)
        if c5_bit2 < best_c5:
            best_c5 = c5_bit2
            best_h = h_bit2.copy()

    # Phase 7: Final fine continuous search
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        n_pol = min(int((budget_s - 3 - _el()) / 0.003), 100000)
        for _it in range(n_pol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 7)) % nc
            d = 8e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 1e-4 <= ni <= 1 - 1e-4 and 1e-4 <= nj <= 1 - 1e-4:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
