    # Compute gradient of softmax max
    c = np.correlate(h, 1-h, mode='full')
    m_val = c.max()
    s = alpha * (c - m_val)
    s = np.clip(s, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()

    n = len(h)
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(2*n-1):

        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])

                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:

                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

def _sg(h, alpha):
    """Softmax max: returns (value, gradient, weights)"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    sm = s.max()
    es = np.exp(s - sm)
    se = es.sum()
    if se < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = es / se
    val = m + (sm + np.log(se)) / alpha
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return val, g, w

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    bc5 = _eval(h)
    bh = h.copy()
    while time.time() < t_end:
        t += 1
        frac = min(1.0, (time.time() - t_start) / max(t_end - t_start, 0.1))
        alpha = alpha_s * (alpha_e / alpha_s) ** frac
        lr = lr_s * (lr_e / lr_s) ** frac
        val, g, w = _sg(h, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        mhat = m / (1.0 - 0.9 ** t)
        vhat = v / (1.0 - 0.999 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
        h_new = _repair(h_new, n)
        c5 = _eval(h_new)
        if c5 < bc5:
            bc5 = c5
            bh = h_new.copy()
        h = h_new
    return bh, bc5
