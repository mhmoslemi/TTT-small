import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k, dx):
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        if m > 0:
            g[:m] += (1.0 - h[sh:sh+m])
            if sh > 0:
                g[sh:] -= h[:m]
    else:
        sp = -sh; m = n - sp
        if m > 0:
            g[sp:] += (1.0 - h[:m])
            g[:m] -= h[sp:sp+m]
    return g * dx

def _lsexp_obj(x, alpha):
    c = _corr(x) * (2.0 / len(x))
    m = c.max()
    z = np.clip(alpha * (c - m), -500, 500)
    z2 = z - z.max()
    return m + np.log(float(np.sum(np.exp(z2))) + 1e-300) / alpha

def _lsexp_grad(x, alpha):
    n = len(x); dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    z = np.clip(alpha * (c - m), -500, 500)
    z -= z.max()
    w = np.exp(z); w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-14:
            g += w[k] * _grad_lag(x, k, dx)
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _projected_subgrad(h, n_iters, alpha_lo, alpha_hi):
    n = len(h); dx = 2.0 / n
    y = h.copy(); v = np.zeros(n)
    c5c = _eval(h)
    for it in range(n_iters):

        c = _corr(y) * dx
        m = c.max()

        z -= z.max()
        w = np.exp(z); w /= w.sum()

        step = 0.002 * (n_iters / float(it + 1)) ** 0.5
        v = 0.9 * v - step * g
        y_new = y + v
        h_new = _repair(y_new, n)
        c5n = _eval(h_new)
        if c5n < c5c:

            c5n = _eval(h_new)

            else:

        y = h

remaining_global = 1100

def _single_lag_descent(h, k, n_iters):
    n = len(h); dx = 2.0 / n
    c5c = _eval(h)
    step = 0.05
    for _ in range(n_iters):
        g = _grad_lag(h, k, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        h_try = _repair(h - step * g, n)
        c5n = _eval(h_try)
        if c5n < c5c:
            h = h_try; c5c = c5n; step = min(step * 1.05, 0.1)
        else:

def _coord_descent(h, n_iters, rng, scale):

    c5c = _eval(h)

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d

        h[i], h[j] = ni, nj
        h_r = _repair(h, n)
        c5n = _eval(h_r)

            h = h_r; c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c
