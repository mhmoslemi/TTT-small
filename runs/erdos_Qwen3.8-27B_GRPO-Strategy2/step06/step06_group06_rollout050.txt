import numpy as np
from scipy.optimize import minimize
import time

def _correl(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_correl(h)) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_constrained(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_fg(h, alpha, top_k):
    n = len(h)
    c = _correl(h)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = c[idx]
    else:
        idx = np.arange(len(c))
        c_use = c
    m = c_use.max()
    s = alpha * (c_use - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    f = (m + np.log(w.sum()) / alpha) * dx
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wi = w[ii]
        if wi < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wi * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wi * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wi * (1.0 - h[:n - sp])
            g[:n - sp] -= wi * h[sp:]
    g *= dx
    return f, g

def _proj_grad_nest(h_init, alpha, top_k, n_steps, t_end):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    f_cur = _smax_fg(h, alpha, top_k)[0]
    step = 0.005
    mu = 0.7
    g_prev = np.zeros(n)
    h_pool = h.copy()
    for _s in range(n_steps):
        if time.time() > t_end:
            break
        f, g = _smax_fg(h, alpha, top_k)
        dh = h - h_pool
        dg = g - g_prev
        den = np.dot(dg, dg) + 1e-15
        step_bar = max(0.002, min(0.2, 0.5 * abs(np.dot(dh, dg) / den)))
        step = 0.5 * step + 0.5 * step_bar
        v = mu * v + (1.0 - mu) * g
        gn = np.linalg.norm(v)
        if gn < 1e-14:
            break
        accepted = False
        for _bt in range(10):
            h_new = _proj_constrained(h - step * v, n * 0.5)
            f_new = _smax_fg(h_new, alpha, top_k)[0]
            if f_new <= f_cur + 1e-14:
                h_pool = h.copy()
                h = h_new
                g_prev = g.copy()
                f_cur = f_new
                accepted = True
                step = min(step * 1.08, 0.4)
                break
            step *= 0.5
        if not accepted:
            step = 0.3 * step
            h_pool = h.copy()
            g_prev = _smax_fg(h, alpha, top_k)[1]
    return h

def _active_lag_descent(h_init, n_rounds, n_top, t_end):
    n = len(h_init)
    h = h_init.copy()
    c5_cur = _eval(h)
    for _r in range(n_rounds):
        if time.time() > t_end:
            break
        c = _correl(h)
        top_idx = np.argsort(c)[::-1][:n_top]
        g = np.zeros(n)
        for kk in top_idx:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:n - sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
        g *= 2.0 / n
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.025, 0.01, 0.005, 0.002, 0.001]:
            h_new = _proj_constrained(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            for step in [0.0005, 0.0002, 0.0001]:
                h_new = _proj_constrained(h - step * g, n * 0.5)
                c5n = _eval(h_new)
                if c5n < c5_cur - 1e-16:
                    h = h_new
                    c5_cur = c5n
                    improved = True
                    break
            if not improved:
                break
    return h, c5_cur

def _lbfgs_run(h_init, alpha, top_k, maxiter, t_end):
    n = len(h_init)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        hx = np.clip(x, 0, 1)
        f, _ = _smax_fg(hx, alpha, top_k)
        return f + 1e5 * (hx.sum() - n * 0.5) ** 2
    def grad(x):
        hx = np.clip(x, 0, 1)
        _, g = _smax_fg(hx, alpha, top_k)
        return g + 2e5 * (hx.sum() - n * 0.5)
    f0 = obj(h_init)
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'disp': False})
        if res.fun < f0 - 1e-10:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h_init, n)

def _make_pattern(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    w1, w2 = n // 4, n // 4
    s1 = rng.randint(0, max(1, n - w1))
    s2 = rng.randint(0, max(1, n - w2))
    h[s1:s1+w1] = 1.0
    h[s2:s2+w2] = 1.0
    h = _repair(h, n)
    kern = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    h = np.convolve(h, kern[n//2:] * kern[:n//2+1], mode='same')
    return _repair(h, n)

def _coord_descent(h_init, n_tries, t_end, rng, scale):
    n = len(h_init)
    h = h_init.copy()
    c5c = _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, c5c

def _sa_refine(h_init, n_iter, t_end, rng):
    n = len(h_init)
    c_cur = _eval(h_init)
    best_h, best_c = h_init.copy(), c_cur
    t0, t1 = 0.003, 5e-7
    for i in range(n_iter):
        if time.time() > t_end:
            break
        t = t0 * (t1 / t0) ** (i / max(1, n_iter - 1))
        i1, i2 = rng.randint(n), rng.randint(n)
        if i1 == i2:
            continue
        d = rng.randn() * (0.002 + 0.015 * t)
        o1, o2 = h_init[i1], h_init[i2]
        h_init[i1], h_init[i2] = o1 + d, o2 - d
        h_init = _repair(h_init, n)
        c_new = _eval(h_init)
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-9)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_init.copy()
        else:
            h_init[i1], h_init[i2] = o1, o2
            h_init = _repair(h_init, n)
    return best_h, best_c
