def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                    diff -= a

                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w[0] += 1.0
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

    rng = np.random.RandomState(seed)
    x = h0.copy()

    t = 0
    best_x = x.copy(); best_c = _eval(x)
    for i in range(niter):
        if time.time() > dl:
            break
        t += 1
        fr = i / max(niter - 1, 1)
        al = a0 * (a1 / a0) ** fr
        lr = lr0 * (lr1 / lr0) ** fr
        if t % 800 == 0:
            x += 0.003 * rng.randn(n)
        g = _grad(x, al)
        gn = np.linalg.norm(g)
        if gn > 1e2:
            g /= gn
        m = 0.85 * m + 0.15 * g
        v = 0.999 * v + 0.001 * g * g
        mh = m / (1 - 0.85 ** t)
        vh = v / (1 - 0.999 ** t)
        x = x - lr * mh / (np.sqrt(vh) + 1e-8)
        x = _repair(x, n)
        c = _eval(x)
        if c < best_c:
            best_c = c; best_x = x.copy()
    return best_x, best_c

def _binary_sa(h0, n, rng, niter, dl, T0):

    x = _repair(x, n)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        c = _corr(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 0)
        return (m + np.log1p(float(np.sum(np.exp(s)))) / alpha) * 2.0 / n
    def _g(x):
        return _grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_f, h, jac=_g, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)
