    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:

            else:

                    h[idx] += amt
                    diff -= amt
                else:

                    h[idx] -= amt
                    diff += amt

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    amt = min(diff, 1.0 - h[idx])
                    h[idx] += amt
                    diff -= amt
                else:
                    amt = min(-diff, h[idx])
                    h[idx] -= amt
                    diff += amt
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _solve_1d(a, b, x_init):
    v = a * x_init + b
    mv = v.max()
    act = v >= mv - 1e-14
    if not act.any():
        act = np.zeros(len(v), dtype=bool)
        act[v.argmax()] = True
    if a[act].min() <= 0.0 <= a[act].max():
        return x_init
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        v = a * mid + b
        mv = v.max()
        act = v >= mv - 1e-14
        if not act.any():
            act = np.zeros(len(v), dtype=bool)
            act[v.argmax()] = True
        ld = a[act].min()
        rd = a[act].max()
        if ld <= 0.0 <= rd:
            break
        if ld > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def _coord_sweep(h, n, rng, reverse):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    order = np.arange(n)
    if rng is not None:
        perm = order.copy()
        rng.shuffle(perm)
        order = perm
    elif reverse:
        order = order[::-1]
    k_arr = np.arange(m)
    sh_arr = (n - 1) - k_arr
    for i in order:
        hi_old = h[i]
        a = np.zeros(m)
        pm = sh_arr >= 0
        if pm.any():
            sv = sh_arr[pm].astype(np.intp)
            tmp = np.zeros(len(sv))
            m1 = sv < (n - i)
            m2 = sv <= i
            if m1.any():
                tmp[m1] += (1.0 - h[i + sv[m1]])
            if m2.any():
                tmp[m2] -= h[i - sv[m2]]
            a[pm] = tmp
        nm = ~pm
        if nm.any():
            sv = (-sh_arr[nm]).astype(np.intp)
            tmp = np.zeros(len(sv))
            m1 = sv <= i
            m2 = (i + sv) < n
            if m1.any():
                tmp[m1] += (1.0 - h[i - sv[m1]])
            if m2.any():
                tmp[m2] -= h[i + sv[m2]]
            a[nm] = tmp
        a *= dx
        b = c - a * hi_old
        x_opt = _solve_1d(a, b, hi_old)
        h[i] = x_opt
        c += a * (x_opt - hi_old)
    return _repair(h, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _lsexp_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(float(np.sum(np.exp(s - s.max())))) / alpha

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _do_slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h
