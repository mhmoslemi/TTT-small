    n = len(h)

    """Return (value, gradient) of soft-max of correlation."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()

    n = len(h)

    m_v = np.zeros(n)
    v_v = np.zeros(n)

        val, g = _fg(h_cur, alpha)
        m_v = b1 * m_v + (1 - b1) * g

        c_new = _eval(h_new)
        if c_new < best_c:

            best = h_new.copy()
        h_cur = h_new
    return best, best_c

    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for it in range(n_iter):
        temp = base_temp * max(0.1, 1.0 - it / n_iter)
        h_pert = best + temp * rng.randn(n)
        h_pert = _repair(h_pert, n)
        # Local optimize
        h_loc, c_loc = _adam_proj(h_pert, 5000, 200, lr=0.005)
        if c_loc < best_c:
            best_c = c_loc
            best = h_loc
    return best, best_c

def _coord_descent(h, n_iter, rng):
    """Pairwise coordinate descent: move mass from i to j."""
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_c = _eval(best)
    c_cur = best_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try moving small amount from j to i (or vice versa)
        d = 1e-4 * rng.random() * (0.5 + 0.5 * rng.random()) * (1 if rng.random() < 0.5 else -1)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < c_cur - 1e-15:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best = h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best, best_c
