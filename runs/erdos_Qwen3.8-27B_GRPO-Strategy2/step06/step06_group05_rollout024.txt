import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                for i in np.argsort(1.0 - h):
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                    if diff < 1e-12: break
            else:
                for i in np.argsort(h):
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                    if diff > -1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0: g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g / k

def _obj_fn(x, alpha, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return dx * (m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha)

def _grad_fn(x, alpha, dx):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15: continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _cfun(x):
    return x.sum() - len(x) * 0.5

def _cjac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha, 2.0 / n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _pgd(h, k, lr, n_steps, mom=0.9):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best = _eval(h)
    best_h = h.copy()
    stagnant = 0
    for _ in range(n_steps):
        g = _topk_grad(h, k)
        v = mom * v - lr * g
        moved = False
        for dec in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _repair(h + dec * v, n)
            val = _eval(h_new)
            if val < best - 1e-16:
                h = h_new; best = val; best_h = h.copy()
                moved = True; stagnant = 0
                break
        if not moved:
            v *= 0.5
            stagnant += 1
            if stagnant > 500:
                h = _repair(h + 0.005 * np.random.randn(n), n)
                lr /= 2.0
                stagnant = 0
                v = np.zeros(n)
    return best_h, best

def _anneal(h, t0, t1, n_steps, rng):
    nc = len(h)
    h = (h > 0.5).astype(float).copy()
    h = _repair(h, nc)
    c5 = _eval(h)
    best = c5
    best_h = h.copy()
    for step in range(n_steps):
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        i = rng.randint(nc)
        j = rng.randint(nc)
        if i == j: continue
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        h2 = _repair(h, nc)
        c5n = _eval(h2)
        if c5n <= c5 or rng.rand() < np.exp(-(c5n - c5) / max(t, 1e-8)):
            c5 = c5n; h = h2
            if c5n < best:
                best = c5n; best_h = h.copy()
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            h = _repair(h, nc)
    return best_h, best

def _resample(h, n_new):
    n = len(h)
    return np.interp(np.linspace(0, 2.0, n_new, endpoint=False),
                     np.linspace(0, 2.0, n, endpoint=False), h)

def _coord_desc(h, rng, n_steps, scale):
    nc = len(h)
    h = h.copy()
    best = _eval(h)
    for _ in range(n_steps):
        i = rng.randint(nc)

        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
        h[i], h[j] = ni, nj
        val = _eval(h)
        if val < best - 1e-16:
            best = val
        else:
            h[i], h[j] = ni - d, nj + d
    return h, best
