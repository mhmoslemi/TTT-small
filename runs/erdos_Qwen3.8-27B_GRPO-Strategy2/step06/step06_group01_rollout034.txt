import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _coord_opt(h, i):
    n = len(h)
    vs = h[i]
    h[i] = 0.0
    b0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    b1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = vs
    d = b1 - b0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(b0 + m1 * d))
        f2 = float(np.max(b0 + m2 * d))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _coord_descent(h, n, max_passes, rng, time_fn=None, min_time=10.0):
    c5 = _eval(h)
    for p in range(max_passes):
        if time_fn is not None and time_fn() < min_time:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            vn = _coord_opt(h, i)
            if abs(vn - h[i]) > 1e-12:
                h[i] = vn
                improved = True
        if not improved:
            break
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5:
            c5 = c5n
        else:
            break
    return h, c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h_old), n_new)

def _fourier_init(K, n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        a = rng.uniform(-0.25, 0.25)
        b = rng.uniform(-0.25, 0.25)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)

def _slsqp_polish(h, n, alpha, maxiter):
    def _lsexp_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def _lsexp_grad(x):
        nn = len(x)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        dx = 2.0 / nn
        g = np.zeros(nn)
        for k in range(2 * nn - 1):
            if w[k] < 1e-12:
                continue
            sh = (nn - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(nn - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, nn)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, nn)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(nn - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    def _sum_eq(x):
        return x.sum() - len(x) * 0.5
    def _sum_jac(x):
        return np.ones(len(x))
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():

    n0 = len(initial_h_values)
