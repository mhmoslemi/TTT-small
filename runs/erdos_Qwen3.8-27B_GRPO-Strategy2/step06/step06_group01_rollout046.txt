def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

    return h

output[k] for k=2: full overlap (no shift).
output[k] for k=4: a[2]*v[0], meaning a is shifted right by 2 relative to v.

So the "shift" parameter: for output index k, v is shifted by (k - (n-1)) relative to a.
- k < n-1: v shifted right (negative lag)
- k = n-1: no shift
- k > n-1: v shifted left (positive lag)

The integral we want: ∫ h(x)(1-h(x+k)) dx
This is the cross-correlation of h with (1-h) at lag k.

In terms of numpy correlate:
If we compute correlate(h, 1-h, 'full'), then:
output[k] = sum_i h[i] * (1-h)[i + (k-(n-1))] (roughly)

output[k] = sum_i a[i] * v[i + (n-1-k)] for valid i (when k < n-1)

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _grad_one(h, k_idx, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0 and sh < n:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    elif sh < 0:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

def _ls_obj(x, alpha, mu):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    base = m + np.log(float(np.sum(np.exp(s)))) / alpha
    pen = mu * (x.sum() - len(x) * 0.5) ** 2
    return base + pen

def _ls_grad(x, alpha, mu):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx_v = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        g += w[k] * _grad_one(x, k, dx_v)
    g += 2.0 * mu * (x.sum() - n * 0.5)
    return g

def _lbfgs_opt(h, alpha, mu, maxiter, t0, budge):
    n = len(h)
    try:
        res = fmin_l_bfgs_b(_ls_obj, h, fprime=_ls_grad, args=(alpha, mu),
                            pgtol=1e-14, maxiter=maxiter, iprint=0,
                            bounds=[(0.0, 1.0)] * n)
        h = _repair(res, n)
    except Exception:
        pass
    if time.time() - t0 > budge - 5:
        h = _repair(h, n)
    return h

def _slsqp_opt(h, alpha, mu, maxiter, t0, budge):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _se(x): return x.sum() - n * 0.5
    def _sj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _se, 'jac': _sj}]
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha, mu),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha, mu, n_iter, t0, budge):
    n = len(h)
    h_n = h.copy(); vel = np.zeros(n)
    best_h = h.copy(); best_c5 = _eval(h)
    base_lr = 0.03; mom = 0.92
    for i in range(n_iter):
        if time.time() - t0 > budge - 20: break
        g = _ls_grad(h_n, alpha, mu)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        g /= gn
        lr = base_lr / (1.0 + 0.005 * i)
        vel = mom * vel - lr * g
        h_new = _repair(h_n + vel, n)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
        h_n = h_new
    return best_h, best_c5

def _focus_subgrad(h, n_iter, t0, budge):
    n = len(h); dx_v = 2.0 / n
    c5c = _eval(h); vel = np.zeros(n); mom = 0.88
    for i in range(n_iter):
        if time.time() - t0 > budge - 15: break
        c = _corr(h)
        top_k = max(4, min(len(c), n // 15))
        top_idx = np.argsort(c)[-top_k:]
        w = np.maximum(c[top_idx] - c[top_idx][0] + 1.0, 0.1)
        w /= w.sum()
        g = np.zeros(n)
        for j, kk in enumerate(top_idx):
            g += w[j] * _grad_one(h, kk, dx_v)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        g /= gn
        lr = 0.04 / (1.0 + 0.003 * i)
        vel = mom * vel - lr * g
        h_new = _repair(h + vel, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            h = h_new; c5c = c5n
        else:
            vel *= 0.5
    return h, c5c

def _equalize_obj(h, k_top):
    c = _corr(h)
    top = np.sort(c)[-k_top:]
    m = top.max()
    return m + 0.5 * np.var(top)

def _binary_sa(h, n_flips, t0, budge, temp0=0.002, cool=0.9999):
    n = len(h)
    h = (h > 0.5).astype(float); h = _repair(h, n)
    c5c = _eval(h); T = temp0
    for _ in range(n_flips):
        if time.time() - t0 > budge - 10: break
        i = np.random.randint(n)
        h[i] = 1.0 - h[i]; h = _repair(h, n)
        c5n = _eval(h)
        d = c5n - c5c
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-10)):
            if d < 0: c5c = c5n
        else:
            h[i] = 1.0 - h[i]; h = _repair(h, n)
        T *= cool
    return h, c5c

def _resample(h_old, n_new):
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_init(K, n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(1, K + 1):
        h += rng.uniform(-0.4, 0.4) * np.cos(np.pi * j * x)
        h += rng.uniform(-0.4, 0.4) * np.sin(np.pi * j * x)
    return _repair(h, n)
