            h_cf = _repair(_resample(best_h, n_cf), n_cf)
            for alpha in [2000, 10000, 50000, 200000]:
                if elapsed() > 60 + 15:
                    break
                h_cf = _lbfgs(h_cf, alpha, 1e4, 2000)
            c5_cf = _eval(h_cf)
            if n_cf == 200 and c5_cf < best_c5:
                best_c5, best_h, best_n = c5_cf, h_cf.copy(), n_cf
            # Upsample back
            if n_cf < n0:
                h_up = _resample(h_cf, n0)
                h_up = _lbfgs(_repair(h_up, n0), 100000, 1e4, 1500)
                c5_up = _eval(h_up)
                if c5_up < best_c5:
                    best_c5, best_h = c5_up, h_up.copy()

    # Phase 3: Aggressive multi-restart L-BFGS-B (250s)
    if elapsed() < budget_s - 300:
        alphas_seq = [20000, 50000, 100000, 300000, 1000000, 3000000, 10000000]
        for it_outer in range(12):
            if elapsed() > budget_s - 300:
                break
            alpha = alphas_seq[min(it_outer, len(alphas_seq) - 1)]
            h_try = _perturb(best_h, rng, 0.02)
            h_try = _lbfgs(h_try, alpha, 1e4, 3000)
            c5_try = _eval(h_try)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()
            else:
                h_try2 = _lbfgs(_perturb(best_h, rng, 0.005), alpha, 1e4, 2000)
                c5_try2 = _eval(h_try2)
                if c5_try2 < best_c5:
                    best_c5, best_h = c5_try2, h_try2.copy()

    # Phase 4: Worst-shift guided descent (150s)
    if elapsed() < budget_s - 200:
        h4 = best_h.copy()
        c5_4 = best_c5
        for _ in range(30):
            if elapsed() > budget_s - 200:
                break
            h4, c5_4 = _worst_shift_step(h4, 30, [0.01, 0.005, 0.002, 0.001, 0.0005])
            if c5_4 < best_c5 - 1e-14:
                best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Binary annealing + polish (200s)
    if elapsed() < budget_s - 150:
        t_rem = max(budget_s - 150 - elapsed(), 10)
        n_steps_a = max(int(t_rem * 500), 2000)
        h_a, c5_a = _anneal_binary(best_h, n_steps_a, 0.3, 0.001, rng)
        h_a = _lbfgs(h_a, 100000, 1e4, 2000)
        c5_a = _eval(h_a)
        if c5_a < best_c5 - 1e-14:
            best_c5, best_h = c5_a, h_a.copy()

    # Phase 6: Final high-alpha L-BFGS + coord descent (120s)
    if elapsed() < budget_s - 30:
        h6 = _lbfgs(best_h, 5000000, 1e4, 3000)
        c5_6 = _eval(h6)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()
        if elapsed() < budget_s - 10:
            n_cd = max(int((budget_s - 10 - elapsed()) * 800), 1000)
            h6, c5_6 = _coord_descent(best_h, n_cd, rng, [0.005, 0.002, 0.001, 0.0005])
            if c5_6 < best_c5:
                best_c5, best_h = c5_6, h6.copy()

    # Phase 7: One more L-BFGS pass + tiny coord descent (20s)
    if elapsed() < budget_s - 5:
        h7 = _lbfgs(best_h, 20000000, 1e4, 2000)
        c5_7 = _eval(h7)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()
        if elapsed() < budget_s - 2:
            n_cd2 = max(int((budget_s - 2 - elapsed()) * 1200), 500)
            h7, c5_7 = _coord_descent(best_h, n_cd2, rng, [0.001, 0.0005, 0.0002])
            if c5_7 < best_c5:
                best_c5, best_h = c5_7, h7.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
