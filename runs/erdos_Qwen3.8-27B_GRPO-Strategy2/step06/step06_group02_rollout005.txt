import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_w(h, w, shifts):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in range(len(shifts)):
        wk = w[ki] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - shifts[ki]
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _pgd(h, n_iters, top_m, beta, step0, t_end):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    step = step0
    stall = 0
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(c, -top_m)[-top_m:]
        tc = c[idx]
        diff = tc.max() - tc.min()
        if diff > 1e-12:
            w = np.exp(beta * (tc - tc.max()) / diff)
        else:
            w = np.ones(top_m)
        w /= w.sum()
        g = _grad_w(h, w, idx)
        ok = False
        s = step
        for _ in range(15):
            ht = _repair(h - s * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-15:
                h = ht; cur = ct; stall = 0; ok = True
                break
            s *= 0.5
        if ok:
            step = min(s * 1.3, 0.3)
        else:
            step *= 0.5
            stall += 1
            if stall > 30:
                break
    return h, cur

def _binary_search(h, n_iters, t_end):
    n = len(h)
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _repair(h, n)
    cur = _eval(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        improved = False
        for i in ones[:]:
            for j in zeros[:]:
                h[i], h[j] = 0.0, 1.0
                c2 = _eval(h)
                if c2 < cur - 1e-13:
                    cur = c2
                    ones.remove(i); ones.append(j)
                    zeros.remove(j); zeros.append(i)
                    improved = True
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur

def _fractional_from_binary(hb, n_iters, t_end):
    """Optimize starting from a binary-like solution, allowing fractional values."""
    n = len(hb)
    h = hb.copy()
    # Slightly perturb to allow gradient flow
    rng = np.random.RandomState(123)
    h += rng.randn(n) * 0.02
    h = _repair(h, n)
    return _pgd(h, n_iters, 5, 8.0, 0.08, t_end)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old, period=2.0)

def _lse_slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    t_chk = time.time()
    def obj(x):
        if time.time() > t_chk + t_end:
            return 1e10
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(float(np.sum(np.exp(s)))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        nz = w > 1e-12
        return _grad_w(x, w[nz], np.where(nz)[0])
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()
