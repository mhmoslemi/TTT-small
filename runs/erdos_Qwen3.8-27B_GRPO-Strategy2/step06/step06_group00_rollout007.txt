import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_single_lag(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _smm_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smm_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _grad_single_lag(h, k, n)
    return g

def _worst_lag_momentum(h, n_top, n_rounds, el_fn, bc_fn):
    n = len(h)
    c5 = _eval(h)
    v = np.zeros(n)
    mu = 0.75
    step = 0.01
    for _r in range(n_rounds):
        if bc_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * c5  # dummy, reuse
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in top_k:
            g += _grad_single_lag(h, k, n)
        g /= n_top
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        v = mu * v + (1.0 - mu) * g
        h_prev = h.copy()
        for scale in [1.0, 0.5, 0.25, 0.1, 0.04, 0.01]:
            s_try = step * scale
            v_try = mu * v + (1.0 - mu) * g
            h_new = _proj_to_sum(h - s_try * v_try, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h = h_new
                v = v_try
                c5 = c5n
                step = min(step * 1.05, 0.05)
                break
        else:
            step *= 0.5
            if step < 1e-7:
                break
    return h, c5

def _nesterov_smm(h, alpha, steps, step0, el_fn, bc_fn):
    n = len(h)
    f = _smm_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _s in range(steps):
        if bc_fn():
            break
        g = _smm_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _smm_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 5)
        else:
            step *= 0.5
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _smm_obj(h, alpha)
    return h

def _local_search_guided(h, n_tries, rng, el_fn, bc_fn, scale=5e-5):
    n = len(h)
    h = h.copy()
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_idx = np.argsort(c)[-3:]
    for _it in range(n_tries):
        if bc_fn():
            break
        if _it % 5000 == 0:
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top_idx = np.argsort(c)[-3:]
        k = top_idx[rng.randint(3)]
        sh = (n - 1) - k
        if sh >= 0:
            i = rng.randint(max(1, n - sh))
            j = (i + sh) % n if sh < n else i
        else:
            sp = -sh
            i = sp + rng.randint(max(1, n - sp))
            j = (i - sp) % n if sp < n else i
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        if _it % 2000 == 0:
            i2 = rng.randint(n)
            j2 = (i2 + 1 + rng.randint(0, 3)) % n
            d2 = scale * 0.5 * rng.randn()
            ni2, nj2 = h[i2] + d2, h[j2] - d2
            if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
                h[i2], h[j2] = ni2, nj2
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i2], h[j2] = ni2 - d2, nj2 + d2
    return best_h, best_c

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_smm_obj, h, jac=_smm_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()
