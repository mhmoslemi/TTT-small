            best_h = np.tile(hp, m).copy()

    # Phase 2: Full-scale PGD round 1
    if el() < d2:
        ap = [3000, 15000, 80000, 400000, 1500000, 5000000]
        best_h, best_c5 = _pgd(best_h, ap, 20000, 0.005, 0.82, t0 + d2, seed)
        best_n = best_h.shape[0]

    # Phase 3: SLSQP refinement on periodic pattern
    if best_n % 31 == 0 and el() < d3:
        m = best_n // 31
        hp = best_h.reshape(m, 31).mean(axis=0)
        hp = _proj(hp, 31 * 0.5)
        for a in [500000, 2000000, 8000000, 30000000]:
            if el() > d3 - 2:
                break
            hp = _slsqp(hp, a, 3000, 1e-19, t0 + d3)
        c5 = _eval(np.tile(hp, m))
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.tile(hp, m).copy()
            best_n = best_h.shape[0]

    # Phase 4: PGD round 2 (higher alpha, lower lr)
    if el() < d4:
        ap = [100000, 500000, 2000000, 8000000, 30000000]
        best_h, best_c5 = _pgd(best_h, ap, 15000, 0.002, 0.85, t0 + d4, seed + 11)
        best_n = best_h.shape[0]

    # Phase 5: Multi-n exploration
    if el() < d5:
        for dn in [0, -10, 10, -31, 31, -100, 100, 200]:
            if el() > d5 - 1:
                break
            nt = best_n + dn
            if 60 < nt < 500:
                hr = _repair(_resample(best_h, nt), nt)
                hr, c5 = _pgd(hr, [200000, 1000000, 5000000], 6000, 0.003, 0.8,
                              min(t0 + d5, t0 + budget_s * 0.90), seed + 30)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hr.copy()
                    best_n = nt

    # Phase 6: Polish
    if el() < d6:
        for sc in [0.005, 0.0015, 0.0004]:
            if el() > d6 - 3:
                break
            best_h, best_c5 = _coord(best_h.copy(), best_n, 4000, sc, seed + 200)
        if el() < d6:
            best_h, best_c5 = _peak_csd(best_h, 3000, 0.005, t0 + d6, seed + 300)
        if el() < d6:
            for sc in [0.0008, 0.0002]:
                if el() > d6 - 2:
                    break
                best_h, best_c5 = _coord(best_h.copy(), best_n, 3000, sc, seed + 400)

    # Final: one more tiny SLSQP on periodic if applicable
    if best_n % 31 == 0 and el() < budget_s * 0.97:
        m = best_n // 31
        hp = best_h.reshape(m, 31).mean(axis=0)
        hp = _proj(hp, 31 * 0.5)
        hp = _slsqp(hp, 50000000, 2000, 1e-19, t0 + budget_s * 0.98)
        c5 = _eval(np.tile(hp, m))
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.tile(hp, m).copy()
            best_n = best_h.shape[0]

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
