def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_val(h, alpha, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_m >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_m)[-top_m:]
    c_top = c[idx]
    m = c_top.max()
    s = alpha * (c_top - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    s_shifted = s - s.max()
    w = np.exp(s_shifted)
    w /= w.sum()
    return val, w, idx

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_m >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_m)[-top_m:]
    c_top = c[idx]
    m = c_top.max()
    s = alpha * (c_top - m)
    s = np.clip(s, -500, 500)
    s_shifted = s - s.max()
    w = np.exp(s_shifted)
    w /= w.sum()
    dx = 2.0 / n

        wk = w[ii]

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _obj_penalty(h, alpha, top_m, lam):
    n = len(h)
    val, _, _ = _smax_val(h, alpha, top_m)
    s = h.sum() - 0.5 * n
    return val + lam * s * s

def _jac_penalty(h, alpha, top_m, lam):
    g = _grad_h(h, alpha, top_m)
    s = h.sum() - 0.5 * len(h)
    g += 2.0 * lam * s * np.ones(len(h))
    return g

def _optimize_lbfgs(h, alpha, top_m, lam, maxiter):
    n = len(h)
    try:
        res = minimize(_obj_penalty, h, jac=_jac_penalty,
                       args=(alpha, top_m, lam),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_step(h, top_m, step):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -top_m)[-top_m:]

            g[sp:] += (1.0 - h[:n - sp])

    return g * step * 2.0 / n / len(idx)

        if s > target:

        else:
            hi = mid

        else:

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                a = min(abs(diff), abs(1.0 - h[idx]) if diff > 0 else h[idx])
                h[idx] += a if diff > 0 else -a
                diff -= a if diff > 0 else -a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_and_w(h, alpha, top_m):
    c = np.correlate(h, 1.0 - h, mode='full')
    if top_m >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_m)[-top_m:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    ss = s - s.max()
    w = np.exp(ss)
    w /= w.sum()
    val = m + np.log(w.sum() * np.exp(ss.max())) / alpha
    return val, w, idx

    ss = s - s.max()
    w = np.exp(ss)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)

        k = int(idx[ii])

            g += wk * (1.0 - 2.0 * h)

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _obj_pen(x, alpha, top_m, lam):

    if top_m >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_m)[-top_m:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    ss = s - s.max()
    val = m + np.log(np.sum(np.exp(ss)) * np.exp(ss.max())) / alpha
    n = len(x)
    dv = x.sum() - 0.5 * n
    return val + lam * dv * dv

def _jac_pen(x, alpha, top_m, lam):
    n = len(x)
    g = _grad_func(x, alpha, top_m)
    dv = x.sum() - 0.5 * n
    g += 2.0 * lam * dv * np.ones(n)
    return g

def _opt_lbfgs(h, alpha, top_m, lam, maxiter):
    n = len(h)
    try:
        res = minimize(_obj_pen, h, jac=_jac_pen,
                       args=(alpha, top_m, lam),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)
