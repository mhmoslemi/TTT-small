def _eval(h):

    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-10:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_top(h, idxs, weights):
    """Weighted gradient from specific correlation indices"""
    n = len(h)
    g = np.zeros(n)
    for i, k in enumerate(idxs):

        else:
            sp = -m
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5

    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:

    """Gradient of c at shift m=w.r.t. h, times dx=2/n"""
