def _slsqp(h, alpha, maxiter, ftol, K=0):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    
    def _obj_k(x):
        return _soft_obj(x, alpha)
    
    def _grad_k(x):
        return _grad(x, alpha, K)
    
    try:
        res = minimize(_obj_k, h, jac=_grad_k, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

def _coord_cd(h, n, rng, deadline, scale0=0.005, max_it=300000):
    x = h.copy()
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    scale = scale0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if scale < 1e-9:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        x[i] += d; x[j] -= d
        x = _repair(x, n)
        c_new = _eval(x)
        if c_new < c_cur - 1e-17:
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()
            scale = max(scale * 0.9997, 1e-10)
        else:
            x[i] -= d; x[j] += d
            x = _repair(x, n)
            scale *= 0.93
    return best_x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Top-K Adam to settle (0-60s)
    dl1 = t0 + 60
    x, c = _adam_topk(best_h, [(300, 3000, 1500), (3000, 30000, 1500)], 0.015, seed, dl1, K=12)
    if c < best_c5:
        best_c5, best_h = c, x.copy()

    # Phase 2: SLSQP polish with increasing alpha (60-220s)
    dl2 = t0 + 220
    for alpha in [10000, 100000, 1000000, 10000000, 100000000]:
        if time.time() > dl2:
            break
        h_new = _slsqp(best_h, alpha, 3000, 1e-20)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Multi-n exploration (220-520s)
    dl3 = t0 + 520
    n_list = sorted(set([n0, 156, 200, 240, 300, 360, 420, 500]))
    for n_try in n_list:
        if time.time() > dl3:
            break
        if n_try < 100 or n_try > 600:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_r = _resample(best_h, n_try)
        x, c = _adam_topk(h_r, [(1000, 80000, 2500)], 0.008, seed + n_try, dl3, K=10)
        h_s = _slsqp(x, 5000000, 2500, 1e-20)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h_s.copy(), n_try

    # Phase 4: Deep SLSQP on best (520-680s)
    dl4 = t0 + 680
    for alpha in [50000000, 200000000, 1000000000]:
        if time.time() > dl4:
            break
        h_new = _slsqp(best_h, alpha, 4000, 1e-21)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: Perturbation cycles (680-880s)
    dl5 = t0 + 880
    for trial in range(6):
        if time.time() > dl5:
            break
        pert = best_h + 0.015 * rng.randn(best_n)
        pert = _repair(pert, best_n)
        x, c = _adam_topk(pert, [(5000, 200000, 1200)], 0.004, seed + 2000 + trial * 7, dl5, K=15)
        h_s = _slsqp(x, 50000000, 2000, 1e-20)
        c5 = _eval(h_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h_s.copy()

    # Phase 6: Coordinate descent fine-tune (880-1050s)
    dl6 = deadline - 10
    x_cd, c_cd = _coord_cd(best_h, best_n, rng, dl6, scale0=0.003, max_it=400000)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, x_cd.copy()

    # Final SLSQP polish
    if time.time() < deadline:
        h_fin = _slsqp(best_h, 500000000, 2000, 1e-21)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5, best_h = c5, h_fin.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
