        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        if not improved:
            break
    return h

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    obj = m + np.log(float(np.sum(np.exp(s)))) / alpha
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return obj, g

        obj, _ = _lse_obj_grad(x, alpha, n)
        return obj

        return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _coord_descent(h, n, max_sweeps, top_k, deadline):
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode='full').copy()
    if top_k > 2 * n - 1:
        top_k = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        top_idx = np.argpartition(lags, -top_k)[-top_k:]
        improved = False
        for i in range(n):
            betas = np.zeros(top_k)
            for j in range(top_k):
                shift = top_idx[j] - (n - 1)
                b = 0.0
                if 0 <= i + shift < n:
                    b += 1.0 - h[i + shift]
                if 0 <= i - shift < n:
                    b -= h[i - shift]
                betas[j] = b
            alphas = lags[top_idx] - betas * h[i]
            candidates = [0.0, 1.0, h[i]]
            for a in range(top_k):
                for b in range(a + 1, top_k):
                    db = betas[a] - betas[b]
                    if abs(db) > 1e-14:
                        ts = (alphas[b] - alphas[a]) / db
                        if 0.0 <= ts <= 1.0:
                            candidates.append(ts)
            best_t = h[i]
            best_v = np.max(alphas + betas * h[i])
            for t in candidates:
                v = np.max(alphas + betas * t)
                if v < best_v - 1e-13:
                    best_v = v
                    best_t = t
            if abs(best_t - h[i]) > 1e-10:
                delta = best_t - h[i]
                h[i] = best_t
                lags[top_idx] += betas * delta
                improved = True
        if sweep % 3 == 0:
            lags = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    return h

def _lse_obj_and_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s_shifted = s - s.max()
    w = np.exp(s_shifted)
    w_sum = w.sum()
    w /= w_sum
    obj = m + float(s.max() + np.log(w_sum)) / alpha
    dx = 2.0 / n
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for k in nz:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            if sh < n:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            if sp < n:
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
    return obj, g

def _lbfgs(h, alpha, maxiter, deadline):
    n = len(h)
    def fun(x):
        return _lse_obj_and_grad(x, alpha, n)[0]
    def jac(x):
        return _lse_obj_and_grad(x, alpha, n)[1]

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'maxls': 40})
        return res.x
    except Exception:
        return h

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _fourier_init(K, n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)

        b = rng.uniform(-0.2, 0.2)
        h += a * np.cos(np.pi * j * x) + b * np.sin(np.pi * j * x)
    return _repair(h, n)

def _binary_flip_search(h, n, rng, n_iters, deadline):

    h_b = _repair(h_b, n)
    c5c = _eval(h_b)

        c5n = _eval(h_b)

            stuck = 0
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
            stuck += 1

                break
    return h_b, c5c
