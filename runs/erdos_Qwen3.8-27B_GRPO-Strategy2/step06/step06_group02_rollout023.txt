import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

    dx = 2.0 / n

        # Focus on top-m shifts
        idx = np.argsort(c)[-top_m:]
        csub = c[idx]
        m = csub.max()
        s = np.clip(alpha * (csub - m), -500, 500)

        w /= w.sum()
        # Only compute gradient for these shifts

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _grad_topm(h, alpha, m):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    dx = 2.0 / n
    nc = 2 * n - 1
    if m < nc:
        idx = np.argpartition(c, -m)[-m:]
        cs = c[idx]
        cm = cs.max()
        w = np.exp(np.clip(alpha * (cs - cm), -500, 500))
        w /= w.sum()
    else:
        idx = np.arange(nc)
        w = np.exp(np.clip(alpha * (c - c.max()), -500, 500))
        w /= w.sum()
    g = np.zeros(n)
    for j, k in enumerate(idx):
        if w[j] < 1e-12:
            continue
        sh = k - (n - 1)
        wk = w[j] * dx
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj_grad(h, n, t_end, alpha0, alpha1, rng):
    h = h.copy()
    cur = _eval(h)
    lr = 0.5
    alpha = alpha0
    stuck = 0
    while time.time() < t_end:
        m = max(3, min(2 * n - 1, int(5.0 * np.sqrt(alpha) + 3)))
        g = _grad_topm(h, alpha, m)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        lr_l = lr
        for _ in range(25):
            h_new = _repair(h - lr_l * g, n)
            c_new = _eval(h_new)
            if c_new < cur - 1e-14:
                h = h_new; cur = c_new
                lr = min(lr_l * 1.15, 2.0)
                improved = True; stuck = 0
                break
            lr_l *= 0.75
        if not improved:
            stuck += 1
            lr = max(lr_l, 1e-10)
            if stuck % 100 == 0:
                alpha = min(alpha * 2.0, alpha1)
            if stuck > 800:
                h = _repair(h + 0.005 * rng.randn(n), n)
                cur = _eval(h)
                stuck = 0
                alpha = min(alpha * 0.5, alpha1)
    return h, cur

def _binary_search(h, n, t_end, rng):
    hb = np.where(h > 0.5, 1.0, 0.0).copy()
    hb = _repair(hb, n)
    cur = _eval(hb)
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    for _ in range(50000):
        if time.time() > t_end:
            break
        improved = False
        for trial in range(3000):
            if time.time() > t_end:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            hb[i], hb[j] = 0.0, 1.0
            c_new = np.correlate(hb, 1.0 - hb, 'full').max() * 2.0 / n
            if c_new < cur - 1e-13:
                cur = c_new
                ones = np.where(hb == 1.0)[0]
                zeros = np.where(hb == 0.0)[0]
                improved = True
            else:
                hb[i], hb[j] = 1.0, 0.0
        if not improved:
            break
    return hb, cur

def _resample(h_old, n_new):
    if len(h_old) == n_new:
        return h_old.copy()
    xo = np.linspace(0, 2.0, len(h_old), endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def _slsqp_quick(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, 'full')
        m = c.max()
        return m + np.log(float(np.sum(np.exp(np.clip(alpha * (c - m), -500, 500))))) / alpha
    def con_fun(x):
        return x.sum() - n * 0.5
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=lambda x: _grad_topm(x, alpha, 2*n-1),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _repair(res.x, n)
    except:
        return h.copy()
