def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _coord_descent(h, n, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

            h_new = _lbfgs(h_pert, 100000, 500, 2000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 7: Coordinate descent polish
    if time.time() - t0 < budget_s - 15:
        for scale in [0.003, 0.001, 0.0003]:
            if time.time() - t0 > budget_s - 10:
                break
            h_cd, c5 = _coord_descent(best_h.copy(), best_n, 4000, scale, seed + 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd.copy()

    # Phase 8: Try nearby n values
    if time.time() - t0 < budget_s - 30:
        for dn in [10, -10, 20]:
            if time.time() - t0 > budget_s - 20:
                break
            n_try = best_n + dn
            if n_try < 50 or n_try > 800:
                continue
            h_r = _repair(_resample(best_h, n_try), n_try)
            h_r = _lbfgs(h_r, 20000, 200, 3000)
            c5 = _eval(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()
                best_n = n_try

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
