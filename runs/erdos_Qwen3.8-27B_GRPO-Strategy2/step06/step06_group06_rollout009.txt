    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(x))

def _smax_grad(x, alpha):
    n = len(x)
    c = _correl(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15: continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g * dx

def _lbfgs_smax(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    lam = 1e6
    def obj(x):
        return _smax_obj(x, alpha) + lam * (x.sum() - n * 0.5) ** 2
    def grad(x):
        return _smax_grad(x, alpha) + 2.0 * lam * (x.sum() - n * 0.5)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 40})
        res_x = _repair(res.x, n)
        if time.time() < t_end and _eval(res_x) < _eval(h):
            return res_x
    except: pass
    return h

def _proj_grad_max(h, n_steps, t_end, top_k_init=5):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    f_cur = _eval(h_cur)
    step = 0.02
    for i in range(n_steps):
        if time.time() > t_end: break
        c = _correl(h_cur)
        # Adaptive top-k
        top_idx = np.argsort(c)[::-1]
        c_max = c[top_idx[0]]
        # Count how many are within 1% of max
        k_adapt = min(n, max(top_k_init, int(np.sum(c > c_max * 0.999))))
        k_adapt = min(k_adapt, 20)
        g = np.zeros(n)
        for kk in top_idx[:k_adapt]:
            g += _c_grad(h_cur, kk)
        g *= dx / k_adapt
        gn = np.linalg.norm(g)
        if gn < 1e-14: break
        g /= gn
        for step_try in [step, step*0.5, step*0.25, step*0.1, step*0.05]:
            h_new = _proj(h_cur - step_try * g, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f_cur - 1e-16:
                h_cur = h_new; f_cur = f_new
                step = min(step * 1.1, 0.1)
                break
        else:
            step *= 0.7
    return h_cur, f_cur

def _coord_descent(h, n_sweeps, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    f_cur = _eval(h_cur)
    for _s in range(n_sweeps):
        if time.time() > t_end: break
        improved = False
        for i in range(0, n, 1):
            if time.time() > t_end: break
            for d in [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]:
                j = (i + 1) % n  # pair with next
                hi = h_cur[i] + d; hj = h_cur[j] - d
                if hi < 0 or hi > 1 or hj < 0 or hj > 1: continue
                h_cur[i], h_cur[j] = hi, hj
                f_new = _eval(h_cur)
                if f_new < f_cur - 1e-16:
                    f_cur = f_new; improved = True
                    break
                h_cur[i], h_cur[j] = hi - d, hj + d
        if not improved: break
    return h_cur, f_cur

def _slsqp_smax(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x): return x.sum() - n * 0.5
    def c_jac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'disp': False})
        if time.time() < t_end:
            res_x = _repair(res.x, n)
            if _eval(res_x) < _eval(h):
                return res_x
    except: pass
    return h

def _sa_pairs(h, t_end, n_iter, rng):
    n = len(h)
    h_cur = h.copy()
    f_cur = _eval(h_cur)
    best_h = h_cur.copy(); best_f = f_cur
    t0v, t1v = 0.003, 1e-6
    for i in range(n_iter):
        if time.time() > t_end: break
        t = t0v * (t1v / t0v) ** (i / max(1, n_iter - 1))
        i1 = rng.randint(n); i2 = rng.randint(n)
        if i1 == i2: continue
        d = rng.randn() * (0.002 + 0.01 * t)
        hi, hj = h_cur[i1] + d, h_cur[i2] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1: continue
        h_cur[i1], h_cur[i2] = hi, hj
        f_new = _eval(h_cur)
        if f_new < f_cur or rng.rand() < np.exp(-(f_new - f_cur) / max(t, 1e-10)):
            f_cur = f_new
            if f_cur < best_f: best_f = f_cur; best_h = h_cur.copy()
        else:
            h_cur[i1], h_cur[i2] = h_cur[i1] - d, h_cur[i2] + d
    return best_h, best_f

def _slsqp_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def c_fun(x): return x.sum() - n * 0.5
    def c_jac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    f_start = _smax_obj(h, alpha)
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
        if time.time() > t_end:
            pass
        if res.fun < f_start - 1e-14:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h, n)

def _proj_grad_max(h, n_steps, t_end, top_k_init=5):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    f_cur = _eval(h_cur)
    step = 0.015
    vn = np.zeros(n)
    for i in range(n_steps):
        if time.time() > t_end: break
        c = _correl(h_cur)
        top_idx = np.argsort(c)[::-1]
        g = np.zeros(n)
        kk = top_idx[0]
        for j in range(1, min(top_k_init, 2*n-1)):
            if c[top_idx[j]] > c[kk] * 0.995: g += _c_grad(h_cur, top_idx[j])
            else: break
        g *= dx / max(1, np.sum(c[top_idx[:top_k_init]] > c[kk]*0.995))
        vn = 0.7 * vn + 0.3 * g
        accepted = False
        for st in [step, step*0.5, step*0.25, step*0.1]:
            h_new = _proj(h_cur - st * vn, n * 0.5)
            f_new = _eval(h_new)
            if f_new < f_cur - 1e-16:
                h_cur = h_new; f_cur = f_new
                step = min(step * 1.05, 0.08)
                accepted = True
                break
        if not accepted: step *= 0.6
    return h_cur, f_cur

def _lbfgs_refine(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.5, 0.5)] * 0  # dummy, will use real bounds
    bnds = [(0.0, 1.0)] * n
    LAM = 5e6
    def obj(x):
        return _smax_obj(x, alpha) + LAM * (x.sum() - n * 0.5) ** 2
    def grad(x):
        return _smax_grad(x, alpha) + 2.0 * LAM * (x.sum() - n * 0.5)
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
        if _eval(res.x) < _eval(h) - 1e-10:
            return _repair(res.x, n)
    except: pass
    return h

def _coord_pair(h, n_iters, t_end, rng):
    n = len(h)
    h_cur = h.copy()
    f_cur = _eval(h_cur)
    for _ in range(n_iters):
        if time.time() > t_end: break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue
        d = rng.choice([0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002, 0.0001, -0.0001])
        hi, hj = h_cur[i] + d, h_cur[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1: continue
        h_cur[i], h_cur[j] = hi, hj
        f_new = _eval(h_cur)
        if f_new < f_cur - 1e-17:
            f_cur = f_new
        else:
            h_cur[i], h_cur[j] = hi - d, hj + d
    return h_cur, f_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    
    # Phase 1: SLSQP with moderate alpha
    for alpha in [500, 3000, 15000, 80000]:
        if rem() < 300: break
        h_new = _slsqp_smax(best_h, alpha, 3000, t0 + rem() - 299)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
    
    # Phase 2: L-BFGS-B with high alpha
    for alpha in [30000, 100000, 400000]:
        if rem() < 200: break
        h_new = _lbfgs_refine(best_h, alpha, 5000, t0 + rem() - 199)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
    
    # Phase 3: Projected gradient on exact max with adaptive top-k
    if rem() > 150:
        h_new, c5_new = _proj_grad_max(best_h, 800, t0 + rem() - 149, top_k_init=8)
        if c5_new < best_c5:
            best_c5 = c5_new; best_h = h_new.copy()
    
    # Phase 4: SLSQP with very high alpha
    for alpha in [200000, 500000, 1000000]:
        if rem() < 150: break
        h_new = _slsqp_smax(best_h, alpha, 4000, t0 + rem() - 149)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
    
    # Phase 5: Coordinate descent
    if rem() > 100:
        h_new, c5_new = _coord_pair(best_h, min(int((rem()-100)/0.003), 50000), t0 + rem() - 99, rng)
        if c5_new < best_c5:
            best_c5 = c5_new; best_h = h_new.copy()
    
    # Phase 6: Projected gradient again
    if rem() > 100:
        h_new, c5_new = _proj_grad_max(best_h, 500, t0 + rem() - 99, top_k_init=12)
        if c5_new < best_c5:
            best_c5 = c5_new; best_h = h_new.copy()
    
    # Phase 7: SA
    if rem() > 80:
        h_new, c5_new = _sa_pairs(best_h, t0 + rem() - 79, min(int((rem()-80)/0.003), 40000), rng)
        if c5_new < best_c5:
            best_c5 = c5_new; best_h = h_new.copy()
    
    # Phase 8: Final polish
    if rem() > 50:
        h_new = _slsqp_smax(best_h, 500000, 3000, t0 + rem() - 49)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5; best_h = h_new.copy()
    
    if rem() > 30:
        h_new, c5_new = _coord_pair(best_h, min(int((rem()-30)/0.003), 30000), t0 + rem() - 29, rng)
        if c5_new
