import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Distribute the difference
    h *= target / max(s, 1e-15)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _binary_sa(h_start, n, rng, n_iters, deadline, T0=0.02):

    x = (h_start > 0.5).astype(float)

    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    c_cur = _eval(x)
    best_x = x.copy()
    best_c = c_cur
    T = T0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pick a random 1 and random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        x[i] = 0.0
        x[j] = 1.0

            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()

            x[j] = 0.0
        T *= 0.99997

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lse(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shk = (n - 1) - k
        wk = w[k] * dx
        if shk >= 0:
            i1 = np.arange(shk, n)
            g[i1] += wk * (1.0 - x[i1 - shk])
            i2 = np.arange(0, n - shk)
            g[i2] -= wk * x[i2 + shk]
        else:
            sp = -shk
            i1 = np.arange(0, n - sp)
            g[i1] += wk * (1.0 - x[i1 + sp])
            i2 = np.arange(sp, n)
            g[i2] -= wk * x[i2 - sp]
    return g

def _grad_active(x, tol):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    active = np.where(c > m - tol)[0]
    w = np.ones(len(active)) / len(active)
    g = np.zeros(n)
    for idx in range(len(active)):
        k = active[idx]
        shk = (n - 1) - k
        wk = w[idx] * dx
        if shk >= 0:
            i1 = np.arange(shk, n)
            g[i1] += wk * (1.0 - x[i1 - shk])

            sp = -shk
            i1 = np.arange(0, n - sp)
            g[i1] += wk * (1.0 - x[i1 + sp])
            i2 = np.arange(sp, n)

def _adam_opt(h0, alphas, n_iter_total, lr_start, lr_end, deadline):
    n = len(h0)
    x = h0.copy().astype(float)
    m_val = np.zeros(n)
    v_val = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_x = x.copy()
    best_c = _eval(x)
    n_iter = max(n_iter_total, 1)
    for i in range(n_iter_total):
        if time.time() > deadline:
            break
        t += 1
        frac = i / (n_iter_total - 1) if n_iter_total > 1 else 0

        if len(alphas) == 1:
            alpha = alphas[0]
        else:
            seg = frac * (len(alphas) - 1)
            a0 = int(seg)

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _slsqp_grad(x, alpha):
    return _grad_lse(x, alpha)

def _slsqp(h, alpha, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return h

    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

            best_c5, best_h = c_s2, h_s2.copy()

    # Phase 5: Multi-n exploration (620-780s)
    dl5 = t0 + 770
    for n_try in [n0 - 10, n0 + 10, n0 + 20, 350, 400]:
        if time.time() > dl5:
            break
        if n_try < 100 or n_try > 800 or n_try % 2 != 0:
            continue
        h_r = _repair(_resample(best_h, n_try), n_try)
        x, c = _adam_opt(h_r, [5000, 100000, 1000000], 3000, 0.01, 0.0003, dl5)
        if c < best_c5:
            best_c5, best_h, best_n = c, x.copy(), n_try
        if time.time() < dl5:
            h_sp = _slsqp(x, 500000, 2000)
            c_sp = _eval(h_sp)
            if c_sp < best_c5:
                best_c5, best_h, best_n = c_sp, h_sp.copy(), n_try

    # Phase 6: Perturbation + re-optimize (780-920s)
    dl6 = t0 + 910
    rng = np.random.RandomState(seed + 99)
    for _ in range(4):
        if time.time() > dl6:
            break
        h_pert = _repair(best_h + 0.02 * rng.randn(best_n), best_n)
        x, c = _adam_opt(h_pert, [10000, 500000], 2000, 0.008, 0.0003, dl6)
        if c < best_c5:
            best_c5, best_h = c, x.copy()

    # Phase 7: Final active-set + SLSQP (920-1070s)
    dl7 = t0 + 1060
    x, c = _active_set_opt(best_h, dl7, n_outer=300)
    if c < best_c5:
        best_c5, best_h = c, x.copy()
    if time.time() < dl7 - 30:
        h_fin = _slsqp(best_h, 10000000, 4000)
        c_fin = _eval(h_fin)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin.copy()

    # Final repair and eval
    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
