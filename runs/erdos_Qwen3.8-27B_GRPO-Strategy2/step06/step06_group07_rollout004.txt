def _obj(x, alpha, topk):
    c = np.correlate(x, 1.0-x, mode='full')
    mm = c.max()
    nk = len(c)
    if topk < nk:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha*(ct-mm), -500, 500)
    return mm + np.log(float(np.sum(np.exp(s))))/alpha

def _grad(x, alpha, topk):
    n = len(x)
    c = np.correlate(x, 1.0-x, mode='full')
    mm = c.max()
    nk = len(c)
    if topk < nk:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(nk)
        ct = c
    s = np.clip(alpha*(ct-mm), -500, 500)
    w = np.exp(s-s.max()); w /= w.sum()
    wfull = np.zeros(nk)
    wfull[idx] = w
    dx = 2.0/n
    g = np.zeros(n)
    for k in range(nk):
        if wfull[k] < 1e-12: continue
        sh = (n-1)-k
        wk = wfull[k]*dx
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += wk*(1.0-x[i1+sh])
            if sh > 0:
                g[i1+sh] -= wk*x[i1]
        else:
            sp = -sh
            i2 = np.arange(sp, n)
            g[i2] += wk*(1.0-x[i2-sp])
            g[i2-sp] -= wk*x[i2]
    return g

def _sum_eq(x): return x.sum() - len(x)*0.5
def _sum_jac(x): return np.ones(len(x))

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _compute_weights(c, alpha, topk):
    mm = c.max()
    nk = len(c)
    if topk < nk:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(nk)
        ct = c
    s = np.clip(alpha * (ct - mm), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    wfull = np.zeros(nk)
    wfull[idx] = w
    return wfull

def _obj(x, alpha, topk):
    c = np.correlate(x, 1.0 - x, mode='full')
    mm = c.max()
    nk = len(c)
    if topk < nk:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - mm), -500, 500)
    return mm + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha, topk):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    wfull = _compute_weights(c, alpha, topk)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):

            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * x[i1]
        else:
            sp = -sh
            i2 = np.arange(sp, n)
            g[i2] += wk * (1.0 - x[i2 - sp])
            g[i2 - sp] -= wk * x[i2]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_solve(h0, alpha, topk, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h0, jac=_grad, args=(alpha, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n), float(res.fun)
    except Exception:
        return _repair(h0, n), _obj(h0, alpha, topk)

def _nm_solve(h0, alpha, topk, maxiter):
    n = len(h0)
    # Nelder-Mead doesn't handle constraints well, so use penalty
    def _pen_obj(x):
        x = np.clip(x, 0.0, 1.0)
        o = _obj(x, alpha, topk)
        pen = 100.0 * (x.sum() - n * 0.5) ** 2
        return o + pen
    try:
        res = minimize(_pen_obj, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    r = np.interp(xn, xo, h_old)
    return _repair(r, n_new)

def _coord_descent(h, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    n = len(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj(x, alpha, topk):
    c = np.correlate(x, 1.0 - x, mode='full')
    mm = c.max()
    nk = len(c)
    if topk < nk:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - mm), -500, 500)
    return mm + np.log(float(np.sum(np.exp(s)))) / alpha

def _grad(x, alpha, topk):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    mm = c.max()
    nk = len(c)
    if topk < nk:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(nk)
        ct = c
    s = np.clip(alpha * (ct - mm), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    wfull = np.zeros(nk); wfull[idx] = w
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(nk):
        if wfull[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = wfull[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * x[i1]
        else:
            sp = -sh
            i2 = np.arange(sp, n)
            g[i2] += wk * (1.0 - x[i2 - sp])
            g[i2 - sp] -= wk * x[i2]
    return g

def _sum_eq(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h0, alpha, topk, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    try:
        res = minimize(_obj, h0, jac=_grad, args=(alpha, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _nm_sine_init(T, m, seed):
    rng = np.random.RandomState(seed)
    h_p = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(T) / T * rng.randint(2, 6))
    h_p += 0.05 * rng.randn(T)
    return _repair(h_p, T)

def _coord_descent(h, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    n = len(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-16:
            c5c = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5c

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return _repair(np.interp(xn, xo, h_old), n_new)
