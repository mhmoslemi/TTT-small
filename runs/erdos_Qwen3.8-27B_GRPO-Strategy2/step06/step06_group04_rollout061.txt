    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()

        s = np.clip(s, -500, 500)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * 2.0/n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n/2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s-target) > 1e-9:
        h *= target/s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _se(x):
    return x.sum() - len(x)*0.5

def _sj(x):
    return np.ones(len(x))

def _sobj(x, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 0)
    return m + float(np.log(np.sum(np.exp(s))))/alpha

def _sgrad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 0)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        wk = w[k]*2.0/n
        sh = (n-1)-k
        if sh >= 0:
            g[:n-sh] += wk*(1.0-x[sh:])
            if sh > 0: g[sh:] -= wk*x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-x[:n-sp])
            g[:n-sp] -= wk*x[sp:]
    return g

    n = len(h)

                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':_se,'jac':_sj}],
                       options={'maxiter':maxiter,'ftol':1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)
