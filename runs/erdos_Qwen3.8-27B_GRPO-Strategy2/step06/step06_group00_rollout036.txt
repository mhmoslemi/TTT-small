from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1100.0
_N = 0

def _elapsed():
    return time.time() - _T0

def _oob():
    return _elapsed() > _BUDGET - 5

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cf(x):
    return x.sum() - _N * 0.5

def _cj(x):
    return np.ones(_N)

def _sobj(x, alpha):
    n = _N
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.exp(s - s.max()).sum()) / alpha

def _sgrad(x, alpha):
    n = _N
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp(h, alpha, maxit):
    global _N
    n = len(h)
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _grad_lag(h, top_m):
    """Gradient descent on top lags"""
    n = len(h)
    dx = 2.0 / n
    c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    bh = h.copy(); bc = c5
    for _ in range(300):
        if _oob(): break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in top:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]
        g *= dx / top_m
        g -= g.mean()
        ok = False
        for st in [0.1, 0.05, 0.01, 0.001, 1e-4]:
            hn = _proj(h - st * g, n * 0.5)
            cn = _eval(hn)
            if cn < c5 - 1e-15:
                h = hn; c5 = cn; ok = True
                if c5 < bc: bc = c5; bh = h.copy()
                break
        if not ok: break
    return bh, bc

def _fs2(h, n, n_iter, rng):
    """Fast 2-point local search with incremental corr update"""
    dx = 2.0 / n
    mlen = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    c5 = c.max() * dx
    bh = h.copy(); bc = c5
    s_vals = np.arange(-(n - 1), n)
    step = 5e-4
    for _ in range(n_iter):
        if _oob(): break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        d = step * rng.randn()
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            step *= 0.98
            continue
        # Compute delta corr
        ip = i + s_vals
        im = i - s_vals
        vip = (ip >= 0) & (ip < n)
        vim = (im >= 0) & (im < n)
        di = np.zeros(mlen)
        if vip.any(): di[vip] = 1.0 - h[ip[vip]]
        if vim.any(): di[vim] -= h[im[vim]]
        jp = j + s_vals

        vjp = (jp >= 0) & (jp < n)
        vjm = (jm >= 0) & (jm < n)
        dj = np.zeros(mlen)
        if vjp.any(): dj[vjp] = 1.0 - h[jp[vjp]]
        if vjm.any(): dj[vjm] -= h[jm[vjm]]
        dc = d * (di - dj) * dx

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1100.0
_N = 0

def _elapsed():
    return time.time() - _T0

def _oob():
    return _elapsed() > _BUDGET - 5

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cf(x):
    return x.sum() - _N * 0.5

def _cj(x):
    return np.ones(_N)

def _sobj(x, alpha):
    n = _N
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.exp(s - s.max()).sum()) / alpha

def _sgrad(x, alpha):
    n = _N
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n); g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n); g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp); g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp(h, alpha, maxit):
    global _N
    n = len(h); _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _grad_lag(h, top_m):
    n = len(h); dx = 2.0 / n
    c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    bh = h.copy(); bc = c5
    for _ in range(400):
        if _oob(): break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in top:
            sh = (n - 1) - k
