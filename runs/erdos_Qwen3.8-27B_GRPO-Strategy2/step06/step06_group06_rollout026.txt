c[3] = a[1]*v[2] + a[2]*v[1]

For our problem, a = h, v = 1-h:

For k = n-1+1 = n: c[n] = sum_{i=1}^{n-1} h[i] * (1-h[i-1]) (shifted by +1)
For k = n-1-1 = n-2: c[n-2] = sum_{i=0}^{n-2} h[i] * (1-h[i+1]) (shifted by -1)

So the lag interpretation:
- Index n-1 + j corresponds to: sum_{i=j}^{n-1} h[i] * (1-h[i-j]) for j > 0

  Or equivalently: ∫ h(x+j) * (1-h(x)) dx, i.e., the overlap when h is shifted by -j

- Index n-1 - j corresponds to: sum_{i=0}^{n-1-j} h[i] * (1-h[i+j]) for j > 0
  This is ∫ h(x) * (1-h(x+j)) dx

    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lag_g(h, kk):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - kk
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _sm_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    logsum = smax + np.log(np.sum(np.exp(s - smax)))
    return (m + logsum / alpha) * 2.0 / n

def _sm_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()
    g = np.zeros(x.size)
    for k in range(2 * x.size - 1):
        if w[k] > 1e-13:
            g += w[k] * _lag_g(x, k)
    g *= 2.0 / n
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(x.size)

def _slsqp_opt(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    f0 = _sm_obj(h, alpha, n)
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'maxls': 50, 'disp': False})
        if time.time() < t_end and res.fun < f0 - 1e-15:
            return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def _focused_subgrad(h, k_lags, n_iter, t_end):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    c5 = _eval(h_c)
    step = 0.03
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h_c, 1.0 - h_c, mode='full')
        active = np.argsort(c)[::-1][:k_lags]
        g = np.zeros(n)
        for kk in active:
            g += _lag_g(h_c, kk)
        g *= dx / len(active)
        v = 0.7 * v + 0.3 * g
        improved = False
        for _bt in range(15):
            h_new = _proj(h_c - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n <= c5 + 1e-16:
                h_c = h_new
                c5 = c5n
                step = min(step * 1.06, 0.35)
                improved = True
                break
            step *= 0.5
        if not improved:
            step = max(step * 0.3, 1e-7)
            v *= 0.5
    return h_c, c5

def _pair_cd(h, n_passes, t_end):
    n = len(h)
    h_c = h.copy()
    c5 = _eval(h_c)
    offsets = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.05, -0.05, 0.03, -0.03]
    for _p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n - 1):
            j = i + 1
            hi0, hj0 = h_c[i], h_c[j]
            best_c5 = c5
            best_t = 0.0
            for t in offsets:
                ni, nj = hi0 + t, hj0 - t
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h_c[i], h_c[j] = ni, nj
                c5n = _eval(h_c)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_t = t
                h_c[i], h_c[j] = hi0, hj0
            if best_t != 0.0:
                h_c[i], h_c[j] = hi0 + best_t, hj0 - best_t
                c5 = best_c5
                improved = True
        if not improved:
            break
    return h_c, c5

def _fine_coord(h, n_iters, t_end, rng):
    n = len(h)
    h_c = h.copy()
    c5 = _eval(h_c)
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 2e-5 * rng.randn()
        ni, nj = h_c[i] + d, h_c[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h_c[i], h_c[j] = ni, nj
        c5n = _eval(h_c)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h_c[i], h_c[j] = h_c[i] - d, h_c[j] + d
    return h_c, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)

    def rem():

    for alpha in [500, 5000, 50000, 200000, 500000]:

    # Phase 2: Focused subgradient with momentum (main workhorse)

    if rem() > 300:
