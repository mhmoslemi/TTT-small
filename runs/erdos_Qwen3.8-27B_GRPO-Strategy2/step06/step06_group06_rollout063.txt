```python
def _smax_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    val = (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n)
    ssum = x.sum() - n * 0.5
    val += 1e6 * ssum * ssum
    return val

def _smax_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:

            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    g *= dx
    ssum = x.sum() - n * 0.5
    g += 2.0 * 1e6 * ssum
    return g

            c5_cur = c5_new
```

```python
def _optimal_pair_swap(h, i, j, n):
    """Find optimal d for h[i]+=d, h[j]-=d to minimize C₅."""
    c0 = np.correlate(h, 1-h, mode='full')
    m0 = c0.max()
    
    # Compute dc_k/dd for all k
    # h[i] += d: affects c_k where i appears
    # h[j] -= d: affects c_k where j appears
    
    # For shift sh (where output index k = (n-1)-sh):
    # dc_k/dd from h[i]+=d: (1-h[i-sh]) if sh<=i<n-sh... 
    # This is getting complex. Let me just use numerical derivative.
    
    # Actually, the change is:
    # delta_k = contribution from h[i]+=d minus contribution from h[j]+=d
    # For h[i]+=d: adds d*(1-h[m]) for each k where m+lag=i, subtracts d*h[m] for each k where m=i
    # ...
    
    # Simpler: just compute c at d=0 and d=eps, get slope, then find min of max.
    eps = 1e-8
    h[i] += eps
    h[j] -= eps
    c1 = np.correlate(h, 1-h, mode='full')
    h[i] -= eps
    h[j] += eps
    
    slope = (c1 - c0) / eps  # dc_k/dd for each k
    
    # C₅(d) = max_k (c0[k] + slope[k]*d) * 2/n
    # Minimize this over d such that 0≤h[i]+d≤1 and 0≤h[j]-d≤1
    d_min = max(-h[i], -(1-h[j]))  # h[i]+d≥0, h[j]-d≤1 → d≤1-h[j]... 
    # h[i]+d ≥ 0 → d ≥ -h[i]
    # h[i]+d ≤ 1 → d ≤ 1-h[i]  
    # h[j]-d ≥ 0 → d ≤ h[j]
    # h[j]-d ≤ 1 → d ≥ h[j]-1
    d_lo = max(-h[i], h[j]-1)
    d_hi = min(1-h[i], h[j])
    
    # Minimize max_k (c0[k] + slope[k]*d) over d in [d_lo, d_hi]
    # This is a 1D convex problem (max of linear functions)
    # Use bisection: find d where the max is balanced
    # Or just evaluate at a few points
    
    # The optimal d is where the derivative changes sign
    # f(d) = max_k (c0[k] + slope[k]*d) is convex piecewise linear
    # f'(d) = slope[argmax_k (c0[k]+slope[k]*d)] (at non-kink points)
    # Optimal: f'(d) = 0, i.e., slope of active constraint = 0
    # Or at a kink point where two constraints are tied
    
    # Simple approach: ternary search
    a, b = d_lo, d_hi
    if a >= b:
        return 0.0
    for _ in range(30):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        f1 = np.max(c0 + slope*m1)
        f2 = np.max(c0 + slope*m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a+b)/2
```

The terms that change are those involving index i or j (either as m or as m+lag).

For a term h[m]*(1-h[m+lag]):
- If m=i: (h[i]+d)*(1-h[i+lag]) → change is d*(1-h[i+lag])
- If m+lag=i, i.e., m=i-lag: h[i-lag]*(1-(h[i]+d)) → change is -d*h[i-lag]
- If m=j: (h[j]-d)*(1-h[j+lag]) → change is -d*(1-h[j+lag])
- If m+lag=j, i.e., m=j-lag: h[j-lag]*(1-(h[j]-d)) → change is d*h[j-lag]

If i and j don't interfere (i≠j, i+lag≠j, j+lag≠i for the relevant lag), then:
dc_k/dd = (1-h[i+lag]) - h[i-lag] - (1-h[j+lag]) + h[j-lag]

This IS linear in d (the slope is constant). If i and j DO interfere (e.g., i+lag=j), then there's a cross term that makes it quadratic. But for generic i,j and most lags, they don't interfere.

For safety, I'll handle the general case by just computing the slope numerically (as I did above).

**Revised algorithm:**

1. L-BFGS-B with smooth max (coarse optimization)
2. Active constraint subgradient descent (medium optimization)
3. Pair-swap local search with exact line search (fine optimization)
4. Repeat 2-3 several times
5. Final polish

The pair-swap local search is the key novelty. Each step:

```python
        nj = h_c[j:j + bl] - d
        if np.any(ni < 0) or np.any(ni > 1) or np.any(nj < 0) or np.any(nj > 1):
            continue
        h_c[i:i + bl] = ni
        h_c[j:j + bl] = nj
        c1 = np.correlate(h_c, 1.0 - h_c, mode='full')
        if np.max(c1) < np.max(c0) - 1e-14:
            c0 = c1
            c5 = float(np.max(c0) * 2.0 / n)
        else:
            h_c[i:i + bl] -= d
            h_c[j:j + bl] += d
    return h_c, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B coarse optimization
    for alpha in [500, 3000, 20000, 100000]:
        if rem() < 80:
            break
        hc = _lbfgs(best_h, alpha, 5000, t0 + rem() - 79)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 2: Pair-swap local search (exact line search per pair)
    if rem() > 60:
        hc, c5 = _pair_search(best_h, 100000, t0 + rem() - 59, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 3: Subgradient refinement on active constraints
    if rem() > 40:
        hc, c5 = _subgrad(best_h, 300, t0 + rem() - 39)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 4: L-BFGS-B fine optimization
    for alpha in [500000, 2000000]:
        if rem() < 60:
            break
        hc = _lbfgs(best_h, alpha, 5000, t0 + rem() - 59)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 5: Extended pair search
    if rem() > 60:
        hc, c5 = _pair_search(best_h, 150000, t0 + rem() - 59, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 6: Block search for structural changes
    if rem() > 50:
        hc, c5 = _block_search(best_h, 30000, t0 + rem() - 49, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 7: Subgradient + pair combo
    if rem() > 50:
        hc, c5 = _subgrad(best_h, 200, t0 + rem() - 45)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
        if rem() > 45:
            hc, c5 = _pair_search(best_h, 80000, t0 + rem() - 44, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 8: Ultra-fine L-BFGS-B
    if rem() > 40:
        hc = _lbfgs(best_h, 5000000, 3000, t0 + rem() - 35)
        c5 = _eval(hc)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 9: Final extended pair search
    if rem() > 30:
        hc, c5 = _pair_search(best_h, 200000, t0 + rem() - 10, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 10: Last subgradient polish
    if rem() > 20:
        hc, c5 = _subgrad(best_h, 100, t0 + rem() - 8)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
        if rem() > 10:
            hc, c5 = _pair_search(best_h, 50000, t0 + rem() - 5, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
