def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

- x_{k+1} = proj_C(y_k - step * g_k)

This is the "heavy ball" method or "momentum SGD." It's not exactly Nesterov but works well in practice.

Let me use this simpler momentum formulation:

```python
def _nesterov(h, n_iter, step0, top_m):
    n = len(h)
    best = h.copy()
    best_c = _c5val(best)
    y = h.copy()
    v = np.zeros(n)
    mu = 0.9
    step = step0
    stuck = 0
    for it in range(n_iter):
        c = np.correlate(y, 1.0 - y, mode='full')
        top = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in top:
            sh = n - 1 - k
            if sh >= 0:
                g[:n-sh] += 1.0 - y[sh:]
                if sh > 0:
                    g[sh:] -= y[:n-sh]
            else:
                sp = -sh
                g[sp:] += 1.0 - y[:n-sp]
                g[:n-sp] -= y[sp:]
        g *= 2.0 / n
        v = mu * v + (1 - mu) * g
        yn = _repair(y - step * v, n)
        cn = _c5val(yn)
        if cn < best_c - 1e-16:
            best = yn.copy()
            best_c = cn
            step = min(step * 1.02, 0.15)
            stuck = 0
        else:
            step *= 0.85
            stuck += 1
            if stuck > 100:
                v = np.zeros(n)
                step = step0
                stuck = 0
        y = yn
    return best, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

_GN = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = 0.5 * n
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _c5val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _smax_f(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(alpha * (c - m), -500, 500)))) / alpha

def _smax_g(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    w = np.exp(np.clip(alpha * (c - m), -500, 500))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = n - 1 - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _cfun(x):
    return np.array([x.sum() - 0.5 * _GN])

def _cjn(x):
    return np.array([np.ones(_GN)])

def _slsqp_opt(h, alpha, maxiter, ftol):
    global _GN
    _GN = len(h)
    n = _GN
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjn}]
    try:
        res = minimize(_smax_f, h, jac=_smax_g, args=(alpha,),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, n_iter, step0, top_m, mu):
    n = len(h)
    best = h.copy()
    best_c = _c5val(best)
    y = h.copy()
    v = np.zeros(n)
    step = step0
    stuck = 0
    for it in range(n_iter):
        c = np.correlate(y, 1.0 - y, mode='full')
        top = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in top:
            sh = n - 1 - k
            if sh >= 0:
                g[:n-sh] += 1.0 - y[sh:]
                if sh > 0:
                    g[sh:] -= y[:n-sh]
            else:
                sp = -sh
                g[sp:] += 1.0 - y[:n-sp]
                g[:n-sp] -= y[sp:]
        g *= 2.0 / n
        v = mu * v + (1.0 - mu) * g
        yn = _proj(y - step * v, 0.5 * n)
        cn = _c5val(yn)
        if cn < best_c - 1e-16:
            best = yn.copy()
            best_c = cn
            step = min(step * 1.02, 0.12)
            stuck = 0
        else:
            step *= 0.85
            stuck += 1
            if stuck > 120:
                v = np.zeros(n)
                step = step0
                stuck = 0
        y = yn
    return best, best_c

def _fine_search(h, n_tries, rng, step_scale):
    n = len(h)
    h = h.copy()
    cc = _c5val(h)
    best = h.copy()
    best_c = cc
    for _ in range(n_tries):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step_scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i] = ni; h[j] = nj
        cn = _c5val(h)
        if cn < cc:
            cc = cn
            if cn < best_c:
                best_c = cn; best = h.copy()
        else:
            h[i] = ni - d; h[j] = nj + d
    return best, best_c

    t0 = time.time()

        return time.time() - t0

    for alpha, iters in [(200, 500), (2000, 600), (20000, 600), (100000, 800)]:

        best_h, best_c = _nesterov(best_h, n_it, 0.025, 5, 0.88)

    # Phase 3: SLSQP refinement
    for alpha, iters in [(300000, 800), (2000000, 1000), (10000000, 1200)]:
        if _el() > budget_s - 200:
            break
        h = _slsqp_opt(best_h, alpha, iters, 1e-16)
        c = _c5val(h)
        if c < best_c:
            best_c = c; best_h = h.copy()

    # Phase 4: Nesterov with wider top
    if _el() < budget_s - 120:
        n_it = min(250000, int((budget_s - 120 - _el()) * 600))
        best_h, best_c = _nesterov(best_h, n_it, 0.015, 8, 0.90)

    # Phase 5: Final SLSQP at very high alpha
    if _el() < budget_s - 80:
        h = _slsqp_opt(best_h, 50000000, 2500, 1e-17)
        c = _c5val(h)
        if c < best_c:
            best_c = c; best_h = h.copy()

    # Phase 6: Structured Fourier perturbations + short SLSQP
    if _el() < budget_s - 60:
        for freq in [1, 2, 3, 5]:
            if _el() > budget_s - 40:
                break
            pert = 0.01 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
            hp = _proj(best_h + pert, 0.5 * n)
            hp = _slsqp_opt(hp, 2000000, 500, 1e-16)
            cp = _c5val(hp)
            if cp < best_c:
                best_c = cp; best_h = hp.copy()
            hp2 = _proj(best_h - pert, 0.5 * n)
            hp2 = _slsqp_opt(hp2, 2000000, 500, 1e-16)
            cp2 = _c5val(hp2)
            if cp2 < best_c:
                best_c = cp2; best_h = hp2.copy()

    # Phase 7: Fine coordinate search
    if _el() < budget_s - 10:
        n_tr = min(int((budget_s - 10 - _el()) * 400), 300000)
        best_h, best_c = _fine_search(best_h, n_tr, rng, 1.5e-5)

    # Phase 8: Tiny final polish
    if _el() < budget_s - 3:
        h = _slsqp_opt(best_h, 100000000, 500, 1e-18)
        c = _c5val(h)
        if c < best_c:
            best_c = c; best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c = _c5val(best_h)
    return (best_h, best_c, n)
