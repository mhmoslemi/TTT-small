def _repair(h, n):

    target = n / 2.0

    if s < 1e-12:

                diff -= a

    for k in idx:
        s = k - (n - 1)
        wk = 1.0 / top_m
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    v = np.zeros(n)

    best = cur
    best_h = h.copy()

        if time.time() > t_end:
            break
        t += 1
        g = _grad_lse(h, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        mh = m / (1 - 0.9**t)
        vh = v / (1 - 0.999**t)
        h_new = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < cur:
            h = h_new
            cur = c_new
            if c_new < best:
                best = c_new
                best_h = h.copy()
    return best_h, best

```python
def _flip_effect(h, n):
    """Compute effect of decreasing h[i] by 1 on each lag k of correlate(h, 1-h)."""
    eff = np.zeros((n, 2*n-1))
    for i in range(n):
        for s in range(n):
            # forward lag k = n-1+s
            k = n - 1 + s
            if i + s < n:
                eff[i, k] += (1.0 - h[i+s])  # from h[i]*(1-h[i+s])
            if s > 0 and i >= s:
                eff[i, k] -= h[i-s]  # from h[i-s]*(1-h[i])
        for s in range(1, n):
            k = n - 1 - s
            if i >= s:
                eff[i, k] += (1.0 - h[i-s])
            if i + s < n:
                eff[i, k] -= h[i+s]
    return eff
```

This is O(n³) for the outer loop. For n=500, that's 125M operations in a Python loop - way too slow.

Let me vectorize:

For forward shifts, the contribution to eff[i, n-1+s] is:
- (1-h[i+s]) if i+s < n → for all i, add to positions where i+s < n
- -h[i-s] if i ≥ s and s > 0 → for all i, subtract where i ≥ s

Let me think of it as:
Forward shift s (k = n-1+s):
  For each i where i < n-s: eff[i, k] += (1-h[i+s])
  For each i where i ≥ s (and s > 0): eff[i, k] -= h[i-s]

So for a fixed s:
  eff[:n-s, n-1+s] += (1 - h[s:])     # (1-h[i+s]) for i=0..n-1-s
  eff[s:, n-1+s] -= h[:n-s]           # h[i-s] for i=s..n-1

Backward shift s (k = n-1-s):
  For each i where i ≥ s: eff[i, k] += (1-h[i-s])
  For each i where i < n-s: eff[i, k] -= h[i+s]

  eff[s:, n-1-s] += (1 - h[:n-s])
  eff[:n-s, n-1-s] -= h[s:]

So the full computation is:
```python
def _flip_effect(h, n):
    eff = np.zeros((n, 2*n-1))
    # Forward shifts
    for s in range(n):
        k = n - 1 + s
        if s == 0:
            # eff[:n, k] += (1-h[0:]) = 1-h
            # eff[0:, k] -= h[:n] = h (but s>0 condition, so skip second)
            eff[:, k] += (1.0 - h)  # from (1-h[i+0]) for i < n
            # No backward part for s=0 in forward
        else:
            eff[:n-s, k] += (1.0 - h[s:])
            eff[s:, k] -= h[:n-s]
    # Backward shifts
    for s in range(1, n):
        k = n - 1 - s
        eff[s:, k] += (1.0 - h[:n-s])
        eff[:n-s, k] -= h[s:]
    return eff
```

This is O(n) loops each doing O(n) vectorized ops = O(n²) total. For n=500, that's 250K vectorized ops. Fast!

Then for binary optimization:
- Precompute eff (O(n²))
- For swap (i: h[i]-=1, j: h[j]+=1): Δc = -eff[i] + eff[j]
- New c = c + Δc, new max = max(new_c)
- Trying all pairs is O(n² * n) = O(n³) which is 125M for n=500. Still slow.

But I can be smarter:
- For each i that's a 1 (to turn off), and each j that's a 0 (to turn on):
  - Δc = -eff[i] + eff[j]
  - new_max = max(c - eff[i] + eff[j])
  
  The max operation is O(n) per pair. Total: O((n/2)² * n) = O(n³/4). For n=500: ~15.6M. Still slow in Python.

Alternative: For each i, precompute c - eff[i] (the correlation if we turn off i). Then for each j, new_c = (c - eff[i]) + eff[j]. Finding the j that minimizes max(new_c) is still O(n) per j.

Total: O(n * n * n) = O(n³). Same thing.

For n=500, n³/4 ≈ 15.6M comparisons. In NumPy, if I vectorize the inner loop:
- For fixed i: compute c_i = c - eff[i] (O(n))
- For all j simultaneously: new_c[j] = c_i + eff[j] (O(n) each, O(n²) total)
- max of each new_c[j]: O(n²) using np.max(axis=1)

So for fixed i: O(n²) to evaluate all j.
For all i: O(n³) total.

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lse(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    sc = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for k in np.where(w > 1e-10)[0]:
        s = k - (n - 1)
        wk = w[k]
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    idx = np.argsort(c)[-top_m:]

        wk = 1.0 / top_m
        if s == 0:

        elif s > 0:

            g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    v = np.zeros(n)

        t += 1
        g = _grad_lse(h, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        h_new = h - lr * m / (1 - 0.9**t) / (np.sqrt(v / (1 - 0.999**t)) + 1e-8)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < cur:

            cur = c_new

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_lse(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sc = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(sc)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    for k in np.where(w > 1e-10)[0]:
        s = k - (n - 1)
        wk = w[k]
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n-s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n-s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:n-sp])

    n = len(h0)
    h = h0.copy()

    v = np.zeros(n)
    t = 0
    cur = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        t += 1
        g = _grad_lse(h, alpha)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        h_new = h - lr * m / (1.0 - 0.9**t) / (np.sqrt(v / (1.0 - 0.999**t)) + 1e-8)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)

            cur = c_new
