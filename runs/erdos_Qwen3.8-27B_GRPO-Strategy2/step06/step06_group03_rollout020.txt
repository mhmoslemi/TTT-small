```python
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import sigmoid
import time

_alpha_g = 1.0
_lam_g = 0.0

def _repair(h, n=None):
    if n is None: n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smax_val(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    half = n - 1
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15: continue
        d = half - i
        if d >= 0:
            if d < n:
                g[:n-d] += wi * (1.0 - h[d:])
                g[d:] -= wi * h[:n-d]
        else:
            e = -d
            if e < n:
                g[e:] += wi * (1.0 - h[:n-e])
                g[:n-e] -= wi * h[e:]
    return g

def _z_obj(z):
    h = sigmoid(z)
    return _smax_val(h, _alpha_g) + _lam_g * (h.sum() - len(h) / 2.0)

def _z_grad(z):
    h = sigmoid(z)
    gh = _smax_grad(h, _alpha_g)
    s = h * (1.0 - h)
    return gh * s + _lam_g * s

def _optimize_z(h0, alpha, maxiter, t_end):
    global _alpha_g, _lam_g
    n = len(h0)
    h_safe = np.clip(h0, 1e-6, 1.0 - 1e-6)
    z0 = np.log(h_safe / (1.0 - h_safe))
    z0 -= z0.mean()
    best_h = h0.copy(); best_c = _eval(h0)
    for lam in [0.0, 0.5, 2.0, 8.0, 30.0]:
        if time.time() > t_end: break
        _alpha_g = alpha; _lam_g = lam
        try:
            res = minimize(_z_obj, z0, jac=_z_grad, method='L-BFGS-B',
                         options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = sigmoid(res.x)
            h_new = _repair(h_new, n)
            c_new = _eval(h_new)
            if c_new < best_c:
                best_c = c_new; best_h = h_new
                z0 = res.x.copy()
        except Exception:
            pass
    return best_h, best_c

def _proj_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad_vec(h, n_iter, step0, n_top, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    v = np.zeros(n); mu = 0.8; step = step0; stuck = 0
    half = n - 1
    for it in range(n_iter):
        c = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argsort(c)[-n_top:]
        g = np.zeros(n)
        for k in idx:
            d = half - k
            if d >= 0:
                if d < n:
                    g[:n-d] += (1.0 - best[d:])
                    g[d:] -= best[:n-d]
            else:
                e = -d
                if e < n:
                    g[e:] += (1.0 - best[:n-e])
                    g[:n-e] -= best[e:]
        g *= 2.0 / n
        v = mu * v + g
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj_box_sum(best - step * f * v, n / 2.0)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best = h_new; best_c = c_new
                step = min(step * 1.15, 0.3)
                improved = True; stuck = 0
                break
        if not improved:
            step *= 0.5; stuck += 1
            if stuck > 60:
                best = _repair(best + 0.004 * rng.randn(n), n)
                best_c = _eval(best); step = step0; stuck = 0; v = np.zeros(n)
    return best, best_c

def _binary_nudge(h, rng):
    n = len(h)
    best = h.copy(); best_c = _eval(best)
    for _ in range(50):
        i = rng.randint(n)
        h[i] = 1.0 if h[i] > 0.5 else 0.0
        h = _repair(h, n)
        c = _eval(h)
        if c < best_c:
            best = h.copy(); best_c = c
        else:
            h[i] = 0.5 * (best[i] + (1.0 if best[i] > 0.5 else 0.0))
            h = _repair(h, n)
    return best, best_c

def _slsqp_final(h, alpha, maxiter, t_end):
    n = len(h)
    try:
        res = minimize(lambda x: _smax_val(x, alpha), h,
                      jac=lambda x: _smax_grad(x, alpha),
                      method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq',
                                   'fun': lambda x: x.sum() - n / 2.0,
                                   'jac': lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = _repair(res.x, n)
        return h_new, _eval(h_new)
    except Exception:
        return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    best_h = h.copy(); best_c = _eval(h)

    t_end1 = t0 + budget_s * 0.30

        if time.time() > t_end1: break
        best_h, best_c = _optimize_z(best_h, alpha, 600, t_end1)

    # Phase 2: Subgradient with top-k (vectorized)
    t_end2 = t0 + budget_s * 0.55
    if time.time() < t_end2:
        n_it = min(4000, max(100, int((t_end2 - time.time()) / 0.018)))
        best_h, best_c = _subgrad_vec(best_h, n_it, 0.012, 5, rng)

    # Phase 3: L-BFGS-B round 2 with higher alpha
    t_end3 = t0 + budget_s * 0.78
    for alpha in [5000, 20000, 80000, 300000]:
        if time.time() > t_end3: break
        best_h, best_c = _optimize_z(best_h, alpha, 1000, t_end3)

    # Phase 4: Subgradient polish
    t_end4 = t0 + budget_s * 0.88
    if time.time() < t_end4:
        n_it = min(2000, max(100, int((t_end4 - time.time()) / 0.018)))
        best_h, best_c = _subgrad_vec(best_h, n_it, 0.006, 3, rng)

    # Phase 5: SLSQP final polish
    if time.time() < t0 + budget_s * 0.96:
        h_new, c_new = _slsqp_final(best_h, 100000, 2000, t0 + budget_s * 0.96)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 6: Binary nudge
    if time.time() < t0 + budget_s * 0.99:
        hn, cn = _binary_nudge(best_h, rng)
        if cn < best_c:
            best_h, best_c = hn, cn

    best_h = _repair(best_h, len(best_h))
    best_c = _eval(best_h)
    return (best_h, best_c, len(best_h))
```
