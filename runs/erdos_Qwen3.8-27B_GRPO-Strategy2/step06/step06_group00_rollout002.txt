    return g, m + np.log(w.sum()) / alpha_eff, w, top_idx

def _proj_grad_desc(h, alpha_list, top_m, steps_per_alpha, step0, t0, budget):
    n = len(h)
    dx = 2.0 / n
    c5_best = _eval(h)
    best_h = h.copy()
    for alpha in alpha_list:
        step = step0
        mu = 0.8
        v = np.zeros(n)
        for _s in range(steps_per_alpha):
            if time.time() > budget:
                return best_h, c5_best
            c = _cvals(h)
            order = np.argsort(c)
            top_idx = order[-top_m:]
            vals = c[top_idx]
            m = vals.max()
            s = np.clip(alpha * (vals - m), -700, 700)
            w = np.exp(s - s.max())
            w /= w.sum()
            g = np.zeros(n)
            for ii in range(top_m):
                g += w[ii] * _grad_lag(h, top_idx[ii], dx)
            v = mu * v + (1.0 - mu) * g
            for _ls in range(30):
                h_new = _proj(h - step * v, n * 0.5)
                c5n = _eval(h_new)
                if c5n < c5_best - 1e-16:
                    h = h_new
                    c5_best = c5n
                    best_h = h_new.copy()
                    step = min(step * 1.05, 0.05)
                    break
                step *= 0.5
            else:
                step = max(step * 0.5, 1e-8)
                v = v * 0.5
    return best_h, c5_best

def _block_perturb(h, rng, n_blocks, block_size):
    n = len(h)
    h2 = h.copy()
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size)
        end = start + block_size
        shift = 0.02 * rng.randn()
        h2[start:end] = np.clip(h2[start:end] + shift, 0, 1)
    return _repair(h2, n)

def _fine_local(h, n_iters, rng, t0, budget):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_iters):
        if time.time() > budget:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 1e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    budget = t0 + budget_s - 5

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Coarse optimize (n/4) then upsample
    if time.time() < budget - 200:
        ratio = max(1, n0 // 140)
        n_low = n0 // ratio
        h_low = _downsample(h, ratio)
        h_low = _repair(h_low, n_low)
        alphas = [500, 2000, 8000, 30000]
        for a in alphas:
            if time.time() > budget - 180:
                break
            h_low, _ = _proj_grad_desc(h_low, [a], 5, 200, 0.01, t0, budget - 180)
        h_hi = _repair(_upsample(h_low, ratio), n0)
        c5h = _eval(h_hi)
        if c5h < best_c5:
            best_c5 = c5h
            best_h = h_hi.copy()

    # Phase 2: Main projected gradient at full res with increasing alpha
    if time.time() < budget - 150:
        h_c = best_h.copy()
        alphas_coarse = [500, 2000, 8000]
        h_c, c5_coarse = _proj_grad_desc(h_c, alphas_coarse, 5, 150, 0.008, t0, budget - 100)
        if c5_coarse < best_c5:
            best_c5 = c5_coarse
            best_h = h_c.copy()

    # Phase 3: Active-set refinement with large alpha
    if time.time() < budget - 120:
        for top_m in [3, 5, 4, 6]:
            if time.time() > budget - 100:
                break
            alphas_fine = [30000, 100000, 300000]
            h_c = best_h.copy()
            h_c, c5_fine = _proj_grad_desc(h_c, alphas_fine, top_m, 120, 0.003, t0, budget - 80)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_c.copy()

    # Phase 4: Structured restarts
    if time.time() < budget - 80:
        for n_blk in [3, 5, 8]:
            if time.time() > budget - 70:
                break
            h_p = _block_perturb(best_h, rng, n_blk, 15)
            h_p, c5_p = _proj_grad_desc(h_p, [10000, 50000], 5, 100, 0.002, t0, budget - 50)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 5: Nesterov with high alpha
    if time.time() < budget - 60:
        n = len(best_h)
        dx = 2.0 / n
        h_c = best_h.copy()
        v = np.zeros(n)
        mu = 0.85
        step = 0.002
        c5c = _eval(h_c)
        for _s in range(500):
            if time.time() > budget - 30:
                break
            c = _cvals(h_c)
            top_idx = np.argsort(c)[-5:]
            vals = c[top_idx]
            m = vals.max()
            s = np.clip(200000.0 * (vals - m), -700, 700)
            w = np.exp(s - s.max())
            w /= w.sum()
            g = np.zeros(n)
            for ii in range(5):
                g += w[ii] * _grad_lag(h_c, top_idx[ii], dx)
            v = mu * v + (1.0 - mu) * g
            h_new = _proj(h_c - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-16:
                h_c = h_new
                c5c = c5n
                step = min(step * 1.02, 0.01)
            else:
                step *= 0.6
                v *= 0.8
        if c5c < best_c5:
            best_c5 = c5c
            best_h = h_c.copy()

    # Phase 6: Medium local search (3-point)
    if time.time() < budget - 15:
        n = len(best_h)
        c5c = _eval(best_h)
        best_h2 = best_h.copy()
        best_c2 = c5c
        h_c = best_h.copy()
        n_iter = min(int((budget - 15 - time.time()) / 0.004), 50000)
        for _it in range(n_iter):
            if time.time() > budget - 10:
                break
            i = rng.randint(n)
            j = (i + rng.randint(2, 8)) % n
            k = (i + rng.randint(9, 15)) % n
            d1 = 5e-6 * rng.randn()
            d2 = 5e-6 * rng.randn()
            ni, nj, nk = h_c[i] + d1, h_c[j] + d2, h_c[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h_c[i], h_c[j], h_c[k] = ni, nj, nk
                c5n = _eval(h_c)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c2:
                        best_c2 = c5n
                        best_h2 = h_c.copy()
                else:
                    h_c[i], h_c[j], h_c[k] = ni - d1, nj - d2, nk + d1 + d2
        if best_c2 < best_c5:
            best_c5 = best_c2
            best_h = best_h2.copy()

    # Phase 7: Ultra-fine 2-point polish
    if time.time() < budget - 2:
        best_h, best_c5 = _fine_local(best_h, min(int((budget - 2 - time.time()) / 0.002), 80000), rng, t0, budget - 2)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
