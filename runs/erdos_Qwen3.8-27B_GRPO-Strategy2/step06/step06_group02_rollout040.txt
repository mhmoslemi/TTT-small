```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
```

```python
            break
        c = _corr(h)
        # Softmax weighting with adaptive temperature
        c_range = c.max() - c.min()
        if c_range < 1e-12:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            temp = c_range / 10.0  # adaptive temperature
            w = np.exp(-temp * (c.max() - c) / c_range)  # weight worst lags more
            w /= w.sum()
        
        g = _grad_weighted(h, w)
        
        # Nesterov momentum
        m = beta * m - lr * g
        h_new = h + m
        h_new = _repair(h_new, n)
        
        c_new = _eval(h_new)
        if c_new < cur - 1e-15:
            h = h_new
            cur = c_new
            lr = min(lr * 1.02, 0.05)
            plateau = 0
        else:
            plateau += 1
            lr *= 0.9
            if plateau > 200:
                # Reset momentum, try random perturbation
                m = np.zeros(n)
                h = _repair(h + 0.01 * np.random.randn(n), n)
                cur = _eval(h)
                plateau = 0
                if lr < 0.001:
                    break
    
    return h, cur
```

2. It has adaptive step size (increases on success, decreases on failure)
3. It has plateau detection with random perturbation
4. The weighting is different (exponential in the gap from max, not just top-M)

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_weighted(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(max(0, n - sh))
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _lse_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _grad_weighted(x, w)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def eq_fun(x):
        return x.sum() - n * 0.5
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _momentum_grad(h, t_end, max_iter=200000, seed=42):
    n = len(h)
    h = h.copy()
    m = np.zeros(n)
    beta = 0.82
    lr = 0.012
    cur = _eval(h)
    plateau = 0
    rng = np.random.RandomState(seed)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        c_range = c.max() - c.min()
        if c_range < 1e-12:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            gap = c.max() - c
            w = np.exp(-5.0 * gap / c_range)
            w /= w.sum()
        g = _grad_weighted(h, w)
        m = beta * m - lr * g
        h_new = _repair(h + m, n)
        c_new = _eval(h_new)
        if c_new < cur - 1e-15:
            h = h_new
            cur = c_new
            lr = min(lr * 1.03, 0.06)
            plateau = 0
        else:
            plateau += 1
            lr *= 0.92
            if plateau > 150:
                m = np.zeros(n)
                h = _repair(h + 0.008 * rng.randn(n), n)
                cur = _eval(h)
                plateau = 0
                if lr < 0.001:
                    break
    return h, cur

def _reweight(h, n_iter, top_m, t_end, seed=99):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    rng = np.random.RandomState(seed)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        idx = np.argsort(c)[-top_m:]
        cmin, cmax = c[idx].min(), c[idx].max()
        w = np.zeros(2 * n - 1)
        if cmax - cmin > 1e-12:
            w[idx] = np.exp(6.0 * (c[idx] - cmin) / (cmax - cmin))
        else:
            w[idx] = 1.0
        w /= w.sum()
        g = _grad_weighted(h, w)
        moved = False
        for step in [0.025, 0.012, 0.006, 0.003, 0.0015]:
            ht = _repair(h - step * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-14:
                h = ht; cur = ct; moved = True
                break
        if not moved:
            h = _repair(h + 0.003 * rng.randn(n), n)
            cur = _eval(h)
    return h, cur

def _pair_targeted(h, n, max_rounds, t_end, seed=777):
    h = h.copy()
    rng = np.random.RandomState(seed)
    cur = _eval(h)
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[-8:]
        ci = c[top_idx]
        pw = np.exp(3.0 * (ci - ci.min()) / max(ci.max() - ci.min(), 1e-12))
        pw /= pw.sum()
        chosen = top_idx[rng.choice(len(top_idx), p=pw)]
        sh = (n - 1) - chosen
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = old_j, old_i
        h = _repair(h, n)
        c2 = _eval(h)
        if c2 < cur - 1e-14:
            cur = c2
        else:
            h[i], h[j] = old_i, old_j
            h = _repair(h, n)
    return h, cur

def _resample(h_old, n_new):
    n_old = len(h_old)
    xo = np.linspace(0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)

    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP smoothed optimization

        hn = _slsqp(best_h, alpha, 4000, 1e-20)

    # Phase 2: Momentum-based projected gradient (novel)

        h2, c5 = _momentum_grad(h2, t0 + rem() - 250, 150000, seed=seed)

    # Phase 3: Multi-scale reweighting
```
