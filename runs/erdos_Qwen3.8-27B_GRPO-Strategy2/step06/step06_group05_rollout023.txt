import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _rb(h, seed=0):
    """Repair to binary."""
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    ones = int(h.sum())
    if ones != target:
        r = np.random.RandomState(seed)
        if ones > target:
            h[r.choice(np.where(h==1)[0], ones-target, replace=False)] = 0
        else:
            h[r.choice(np.where(h==0)[0], target-ones, replace=False)] = 1
    return h

def _lssm(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sm = s - s.max()
    return m + float(np.log(np.sum(np.exp(sm))) + s.max()) / alpha

def _lssm_g(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, mx, ftol):
    n = len(h)
    try:
        res = minimize(_lssm, h, jac=_lssm_g, args=(alpha,),
                       method='SLSQP', bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-n*0.5,
                                     'jac':lambda x: np.ones(n)}],
                       options={'maxiter':mx,'ftol':ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _anneal(h, t0, t1, nsteps, rng):
    n = len(h)
    h = _rb(h, 0).copy()
    c5c = _eval(h)

    ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]

        j = zeros[rng.randint(len(zeros))]
