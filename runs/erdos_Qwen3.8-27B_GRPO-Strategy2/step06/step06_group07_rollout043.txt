def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _grad_correlate(h, weights, n):
    """Compute gradient of weighted sum of correlate values."""
    dx = 2.0 / n
    g = np.zeros(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if abs(w) < 1e-15:
            continue
        sh = (n - 1) - k  # shift amount
        if sh >= 0:
            # c[k] = sum_{i=0}^{n-1-sh} h[i] * (1-h[i+sh])
            # dh[i]/dh[i] = (1-h[i+sh]) for i in [0, n-1-sh]
            # dh[i]/dh[i] = -h[i-sh] for i in [sh, n-1]
            i1 = np.arange(n - sh) if sh > 0 else np.arange(n)
            if sh > 0:
                g[:n-sh] += w * (1.0 - h[sh:])
                g[sh:] -= w * h[:n-sh]
            else:
                g[:] += w * (1.0 - h[:])
                g[:] -= w * h[:]
        else:
            sp = -sh
            # c[k] = sum_{i=0}^{n-1-sp} (1-h[i]) * h[i+sp]
            # = sum_{i=sp}^{n-1} (1-h[i-sp]) * h[i]
            g[sp:] += w * (1.0 - h[:n-sp])
            g[:n-sp] -= w * h[sp:]
    return g

def _logsumexp_obj(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sx = s - s.max()
    obj = m + np.log(np.sum(np.exp(sx))) / alpha
    obj += lam * (h.sum() - n / 2.0) ** 2
    return obj

def _logsumexp_grad(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    sx = s - s.max()
    w = np.exp(sx)
    w /= w.sum()
    g = _grad_correlate(h, w, n)
    g += 2.0 * lam * (h.sum() - n / 2.0) * np.ones(n)
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _pgd(h0, alphas, iters_per_alpha, lr_init, seed=0):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best = h.copy()
    best_c = _eval(h)
    momentum = np.zeros(n)
    beta = 0.8
    lr = lr_init
    for alpha in alphas:
        for it in range(iters_per_alpha):
            c = np.correlate(h, 1.0 - h, mode='full')
            m = c.max()
            s = np.clip(alpha * (c - m), -500, 500)
            sx = s - s.max()
            w = np.exp(sx)
            w /= w.sum()
            g = _grad_correlate(h, w, n)
            # Project gradient onto feasible set (sum constraint)
            g -= g.mean() * np.ones(n)  # project out sum direction
            momentum = beta * momentum + (1.0 - beta) * g
            h_new = h - lr * momentum
            h_new = _repair(h_new, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h = h_new
                lr *= 1.02
            else:
                lr *= 0.7
                h = best.copy()
                if lr < 1e-7:
                    lr = lr_init
                    momentum = np.zeros(n)
    return best, best_c

def _lbfgsb(h0, alpha, lam, maxiter=200):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    res = minimize(_logsumexp_obj, h0, jac=_logsumexp_grad,
                   args=(alpha, lam), method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-12})
    return _repair(res.x, n)

def _coord_descent(h, n_iter, scale, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def _grad_weighted(h, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            g[:n - sh] -= wk * h[sh:]
        else:
            s = -sh
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
    return g

def _soft_obj(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    dx = 2.0 / n
    s = np.clip(alpha * (c - m), -500, 500)
    sx = s - s.max()
    return dx * (m + np.log(np.sum(np.exp(sx))) / alpha) + lam * (h.sum() - n / 2.0) ** 2

def _soft_grad(h, alpha, lam):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    dx = 2.0 / n
    s = np.clip(alpha * (c - m), -500, 500)
    sx = s - s.max()
    w = np.exp(sx)
    w /= w.sum()
    g = _grad_weighted(h, w, n)
    g += 2.0 * lam * (h.sum() - n / 2.0)
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _lbfgsb(h0, alpha, lam, maxiter=300):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_soft_obj, h0, jac=_soft_grad, args=(alpha, lam),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _pgd_momentum(h0, alphas, iters_per_alpha, lr_init, beta, seed):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best = h.copy()
    best_c = _eval(h)
    mom = np.zeros(n)
    lr = lr_init
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            c = np.correlate(h, 1.0 - h, mode='full')
            m = c.max()
            s = np.clip(alpha * (c - m), -500, 500)
            sx = s - s.max()
            w = np.exp(sx)
            w /= w.sum()
            g = _grad_weighted(h, w, n)
            mom = beta * mom + (1.0 - beta) * g
            h_new = _repair(h - lr * mom, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = h_new.copy()
                h = h_new
                lr = min(lr * 1.03, 0.05)
            else:
                lr *= 0.65
                if lr < 1e-8:
                    lr = lr_init
                    mom = np.zeros(n)
    return best, best_c

def _coord_desc(h, n_iter, scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-16:
            c5 = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, c5

def _resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h)
