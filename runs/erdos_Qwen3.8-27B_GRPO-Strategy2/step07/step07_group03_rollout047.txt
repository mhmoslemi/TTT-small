    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        top = c[idx]
    else:
        top = c
    m = top.max()
    s = np.clip(alpha * (top - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha, n, top_k):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        idx = np.arange(len(c))
    top = c[idx]
    m = top.max()
    s = np.clip(alpha * (top - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for wi, k in zip(w, idx):
        if wi > 1e-14:
            g += wi * _lag_grad(h, int(k), n, dx)
    return g

def _nesterov(h, alpha, steps, step0, top_k, t_end):
    n = len(h)
    f = _smooth_obj(h, alpha, top_k)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    stall = 0
    for _ in range(steps):
        if time.time() > t_end: break
        g = _smooth_grad(h, alpha, n, top_k)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new, alpha, top_k)
        if f_new < f:
            h = h_new; f = f_new; stall = 0
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-11: break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _smooth_obj(h, alpha, top_k)
            stall += 1
            if stall > 30: break
    return h

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        w_enc = np.exp(-(a_vals.max() - a_vals) / 0.008)
        w_enc /= w_enc.sum()
        pressure = (w_enc[:, None] * np.abs(grads)).sum(axis=0)
        p_base = pressure + 0.005
        p_base /= p_base.sum()
        sweep_improved = False
        for _ in range(n * 2):
            if time.time() > min(end_t, t_end - 5): break
            if rng.random() < 0.6:
                i = rng.randint(n)
            else:
                i = rng.choice(n, p=p_base)
            window = max(3, min(25, n // 25))
            off = rng.randint(-window, window + 1)
            if off == 0: off = 1
            j = (i + off) % n
            if i == j: continue
            b = grads[:, i] - grads[:, j]
            if np.max(np.abs(b)) < 1e-14: continue
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi: continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(20):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = 0.5 * (d_lo + d_hi)
            if abs(d_opt) > 1e-14:
                h[i] += d_opt; h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5; no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 30:
            mode = rng.randint(2, 20)
            amp = 0.03 * rng.random()
            phase = rng.random() * 2.0 * np.pi
            h += amp * np.sin(2.0 * np.pi * mode * np.arange(n) / n + phase)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _fine_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy(); best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Multi-alpha smooth optimization (top-k focused)
    phases1 = [(1500, 25), (8000, 18), (40000, 12), (200000, 8), (700000, 5)]
    for alpha, top_k in phases1:
        t_ep = min(time.time() + 22, t0 + budget_s * 0.13)
        h = _nesterov(h, alpha, 400, 0.003, top_k, t_ep)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 2: Coordinate descent - increasing then decreasing top_T
    for top_T in [3, 5, 8, 5]:
        t_lim = t0 + budget_s * 0.48
        if time.time() > t_lim - 10: break
        h = best_h.copy()
        dur = min(budget_s * 0.09, t_lim - time.time() - 5)
        if dur < 5: break
        h, c5v = _coord_descent(h, n, dur, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 3: High-alpha smooth + coord
    if time.time() < t0 + budget_s * 0.72:
        for alpha, top_k in [(400000, 6), (1500000, 4), (5000000, 3)]:
            t_ep = min(time.time() + 18, t0 + budget_s * 0.66)
            if time.time() > t_ep: break
            h = _nesterov(best_h.copy(), alpha, 250, 0.0015, top_k, t_ep)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()
        for top_T in [4, 6, 3, 5]:
            t_lim = t0 + budget_s * 0.72
            if time.time() > t_lim - 8: break
            h = best_h.copy()
            dur = min(budget_s * 0.04, t_lim - time.time() - 3)
            if dur < 3: break
            h, c5v = _coord_descent(h, n, dur, top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()

    # Phase 4: Fine local search
    if time.time() < t0 + budget_s - 15:
        remaining = (t0 + budget_s) - time.time() - 12
        n_tries = min(int(remaining / 0.003), 60000)
        if n_tries > 200:
            h = best_h.copy()
            best_h, best_c5 = _fine_search(h, n_tries, rng, t0 + budget_s - 8)

    # Phase 5: Final coordinate polish
    if time.time() < t0 + budget_s - 4:
        h = best_h.copy()
        dur = max(2.0, budget_s - (time.time() - t0) - 5)
        h, c5v = _coord_descent(h, n, dur, 4, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Final ultra-fine
    if time.time() < t0 + budget_s - 2:
        h = best_h.copy()
        best_h, best_c5 = _fine_search(h, 5000, rng, t0 + budget_s - 1)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
