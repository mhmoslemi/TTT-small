import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lse_obj(h, tau, pen):
    c = _corrs(h)
    cm = c.max()
    ex = np.exp(np.clip(tau * (c - cm), -500, 500))
    val = cm + np.log(ex.sum()) / tau
    n = len(h)
    return val + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, tau, pen):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    w = np.exp(np.clip(tau * (c - cm), -500, 500))
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    if pen > 0:
        g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, tau, pen, maxi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_lse_obj, h, jac=_lse_grad, args=(tau, pen),
                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxi, 'ftol': 1e-16,
                              'gtol': 1e-13, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, tau, niter, el, bc):
    n = len(h)
    hc = h.copy()
    v = np.zeros(n)
    beta = 0.75
    step = 0.015
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    for _ in range(niter):
        if bc():
            break
        g = _lse_grad(hc, tau, 0.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 1e-16:
            hc = ht
            best_c = ct
            best_h = ht.copy()
            stall = 0
            step = min(step * 1.015, 0.06)
        else:
            step *= 0.5
            stall += 1
            if stall > 60 or step < 1e-10:
                v = np.zeros(n)
                step = 0.01
                stall = 0
    return best_h, best_c

def _spectral_search(h, rng, el, bc):
    n = len(h)

    best_c = _eval(hc)

        ct = _eval(ht)
        if ct < best_c - 1e-15:
            best_c = ct
            best_h = ht.copy()
            hc = ht.copy()
    return best_h, best_c
