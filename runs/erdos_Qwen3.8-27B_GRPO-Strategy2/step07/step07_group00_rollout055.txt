import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Water-filling projection
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s_mid = np.clip(h - mid, 0, 1).sum()
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj_simplex(h, target):
    """Project onto {x: 0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov_alpha(h, alpha, topk, steps, step0, deadline):
    """Nesterov accelerated gradient with soft-max."""
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for i in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_simplex(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.5
            h = _proj_simplex(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, min_block, max_block):
    n = len(h)

    bh = h.copy()
    bc = c5c

        i0 = rng.randint(0, max(1, n - bs))

        seg += d

        hn[i0:i0+bs] = seg

        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            no_improve = 0
            if c5n < bc - 1e-12:
                bc = c5n
                bh = h.copy()
        else:
            no_improve += 1
    return bh, bc

def _downsample(h, factor):
    """Downsample by averaging blocks."""
    n = len(h)
    new_n = n // factor
    h2 = h[:new_n*factor].reshape(new_n, factor).mean(axis=1)
    return h2

def _upsample(h, factor, n):
    """Upsample by repeating and interpolating."""
    n2 = len(h)
    h2 = np.repeat(h, factor)
    if len(h2) > n:
        h2 = h2[:n]
    elif len(h2) < n:
        h2 = np.concatenate([h2, h[n2:]*0 + np.array([h[-1]]*(n-len(h2)))])
    return h2

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    # Bisection water-fill
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    k = min(topk, len(c))
    idx = np.argpartition(c, -k)[-k:] if k < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    k = min(topk, len(c))
    idx = np.argpartition(c, -k)[-k:].copy() if k < len(c) else np.arange(len(c)).copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-14:
            continue
        lag = idx[ii]
        sh = (n - 1) - lag
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            h = _proj(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, bmin, bmax):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = bmin + rng.randint(bmax - bmin + 1)
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        sd = 0.008 * max(T, 1e-8)**0.4
        d = sd * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc - 1e-14:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 4e-6
    n_accept = 0
    for it in range(niter):
        if time.time() > deadline:
            break
        # Adaptive step
        r1 = rng.choice([1, 2, 3, 5, 8, 13, 21, 34, 55])
        r2 = rng.choice([1, 2, 3, 5, 8, 13])
        i = rng.randint(nc)
        j = (i + r1) % nc
        k = (i + r1 + r2) % nc
        d = step * rng.randn(3)
        d -= d.mean()
        ni, nj, nk = h[i]+d[0], h[j]+d[1], h[k]+d[2]
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                n_accept += 1
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j], h[k] = ni-d[0], nj-d[1], nk-d[2]
        # Adapt step every 200 iters
        if (it + 1) % 200 == 0:
            rate = n_accept / 200.0
            if rate > 0.5:
                step = min(step * 1.3, 3e-5)
            elif rate < 0.1:
                step = max(step * 0.6, 5e-7)
            n_accept = 0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    dl1 = min(deadline, t0 + 100)
    for alpha, tk, st, ns in [(5000, 50, 0.004, 3000), (30000, 30, 0.002, 3000), (100000, 15, 0.001, 2500)]:
        if time.time() > dl1:
            break
        h_new = _nesterov(best_h, alpha, tk, ns, st, dl1)
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()
    
    # Phase 2: Coarse-to-fine (60s)
    dl2 = min(deadline, t0 + 160)
    for fac in [4, 2]:
        if time.time() > dl2:
            break
        nc = n0 // fac
        h_c = best_h[:nc*fac].reshape(nc, fac).mean(axis=1)
        h_c = _nesterov(h_c, 5000, 8, 800, 0.003, dl2)
        h_up = np.repeat(_repair(h_c, nc), fac)
        if len(h_up) > n0:
            h_up = h_up[:n0]
        elif len(h_up) < n0:
            h_up = np.pad(h_up, (0, n0-len(h_up)))
        h_up = _repair(h_up, n0)
        h_up = _nesterov(h_up, 100000, 10, 800, 0.001, dl2)
        c5v = _eval(h_up)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_up.copy()
    
    # Phase 3: Block annealing (250s)
    dl3 = min(deadline, t0 + 410)
    h_a, c5_a = _anneal(best_h.copy(), 80000, rng, dl3, 0.004, 1e-10, 2, 12)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()
    
    # Phase 4: Second anneal with different params (100s)
    dl4 = min(deadline, t0 + 510)
    h_a2, c5_a2 = _anneal(best_h.copy(), 40000, rng, dl4, 0.002, 1e-10, 1, 6)
    if c5_a2 < best_c5:
        best_c5 = c5_a2

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    k = min(topk, len(c))
    idx = np.argpartition(c, -k)[-k:] if k < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    k = min(topk, len(c))
    idx = np.argpartition(c, -k)[-k:].copy() if k < len(c) else np.arange(len(c)).copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-14:
            continue
        lag = idx[ii]
        sh = (n - 1) - lag
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            h = _proj(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, bmin, bmax):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = bmin + rng.randint(bmax - bmin + 1)
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        sd = 0.008 * max(T, 1e-8) ** 0.4
        d = sd * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc - 1e-14:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 5e-6
    n_accept = 0
    win = 0
    for it in range(niter):
        if time.time() > deadline:
            break
        r1 = rng.randint(1, 60)
        r2 = rng.randint(1, 20)
        i = rng.randint(nc)
        j = (i + r1) % nc
        k2 = (i + r1 + r2) % nc
        d = step * rng.randn(3)
        d -= d.mean()
        ni, nj, nk = h[i] + d[0], h[j] + d[1], h[k2] + d[2]
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k2] = ni, nj, nk
            c5n = _eval(h)
            win += 1
            if c5n < c5c - 1e-19:
                c5c = c5n
                n_accept += 1
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j], h[k2] = ni - d[0], nj - d[1], nk - d[2]
        if win >= 300:
            rate = n_accept / float(win)
            if rate > 0.45:
                step = min(step * 1.25, 4e-5)
            elif rate < 0.08:
                step = max(step * 0.55, 3e-7)
            n_accept = 0
            win = 0
    return bh, bc

def _two_pt(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 3e-6
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        r = rng.randint(1, 30)
        j = (i + r) % nc
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
