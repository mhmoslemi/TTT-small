def _repair(h, n):

        h = np.clip(h, 0.0, 1.0)

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

        else:

        if wk < 1e-15:

        m = n - sp
        g[:m] += wk * (1.0 - h[sp:])
        g[sp:] -= wk * h[:m]
