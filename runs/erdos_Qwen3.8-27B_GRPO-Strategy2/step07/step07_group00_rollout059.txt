def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    """Gradient of correlation at lag k with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k

            g[sh:] -= dx * h[:-sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:-sp])
        g[:n - sp] -= dx * h[sp:]
    return g

    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha

    return g

    h = np.clip(h, 0.0, 1.0).copy()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-10:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-13:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-13: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t: lo = m
        else: hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    s = (n - 1) - k
    if s >= 0:
        g[:n - s] += dx * (1.0 - h[s:])
        g[s:] -= dx * h[:n - s]
    else:
        p = -s
        g[p:] += dx * (1.0 - h[:n - p])
        g[:n - p] -= dx * h[p:]
    return g

def _smax_obj(h, a, tk):
    c = _corr(h)
    m = c.max()
    if tk < len(c):
        ix = np.argpartition(c, -tk)[-tk:]
        ct = c[ix]
    else:
        ct = c
    s = np.clip(a * (ct - m), -400, 400)
    mx = s.max()
    return m + (mx + np.log(np.sum(np.exp(s - mx)))) / a

def _smax_grad(h, a, tk):
    c = _corr(h)
    m = c.max()
    if tk < len(c):
        ix = np.argpartition(c, -tk)[-tk:].copy()
    else:
        ix = np.arange(len(c))
    s = np.clip(a * (c[ix] - m), -400, 400)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    g = np.zeros(len(h))
    n = len(h)
    for i in range(len(ix)):
        if w[i] > 1e-14:
            g += w[i] * _grad_k(h, int(ix[i]), n)
    return g

def _nesterov(h, a, tk, steps, s0, dl):
    n = len(h)
    f = _smax_obj(h, a, tk)
    v = np.zeros(n)
    bh, bf = h.copy(), f
    beta, st = 0.83, s0
    for _ in range(steps):
        if time.time() > dl: break
        g = _smax_grad(h, a, tk)
        mv = beta * v + (1.0 - beta) * g
        hn = _proj(h - st * mv, n * 0.5)
        fn = _smax_obj(hn, a, tk)
        if fn < f:
            v, h, f = mv, hn, fn
            st = min(st * 1.015, s0 * 15.0)
            if f < bf: bf, bh = f, h.copy()
        else:
            st *= 0.4
            hn = _proj(h - st * mv, n * 0.5)
            f = _smax_obj(hn, a, tk)
            h = hn if f <= bf else bh.copy()
            f = min(f, bf)
    return bh, _eval(bh)

def _slsqp_opt(h, a, tk, mi, dl):
    n = len(h)
    if time.time() > dl:
        return _repair(h.copy(), n)
    try:
        r = minimize(lambda x: _smax_obj(x, a, tk), h,
                     jac=lambda x: _smax_grad(x, a, tk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _lag_cd(h, niter, rng, dl):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for _ in range(niter):
        if time.time() > dl: break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: j = (i + 7) % n
        for d in [0.004, -0.004, 0.0015, -0.0015, 0.0004, -0.0004, 0.0001, -0.0001]:
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5 - 1e-18:
                    c5 = c5n
                    if c5 < bc: bc, bh = c5, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _polish(h, niter, rng, dl):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for _ in range(niter):
        if time.time() > dl: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n

        ni, nj = h[i] + d, h[j] - d

            h[i], h[j] = ni, nj

            if c5n < c5 - 1e-19:
                c5 = c5n
                if c5n < bc: bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _anneal(h, niter, rng, dl, t0, t1, bsz):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for it in range(niter):
        if time.time() > dl: break
        T = t0 * (t1 / t0) ** (it / max(niter, 1))
        bs = bsz[rng.randint(len(bsz))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        seg += 0.01 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        seg -= seg.mean()
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5 or rng.random() < np.exp(-(c5n - c5) / max(T, 1e-12)):
            h, c5 = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _multi_nesterov(hg, a, tk, dl):
    """Run nesterov with 3 momentum settings, return best."""
    results = []
    for beta_over in [0.80, 0.85, 0.90]:
        n = len(hg)
        f = _smax_obj(hg, a, tk)
        v = np.zeros(n)
        bh, bf = hg.copy(), f
        st = 0.002
        for _ in range(2000):
            if time.time() > dl: break
            g = _smax_grad(hg, a, tk)
            mv = beta_over * v + (1.0 - beta_over) * g
            hn = _proj(hg - st * mv, n * 0.5)
            fn = _smax_obj(hn, a, tk)
            if fn < f:
                v, hg, f = mv, hn, fn
                st = min(st * 1.02, 0.03)
                if f < bf: bf, bh = f, hg.copy()
            else:
                st *= 0.5
                v *= 0.7
                hn = _proj(hg - st * v, n * 0.5)
                f = _smax_obj(hn, a, tk)
                hg = hn if f <= bf else bh.copy()
                f = min(f, bf)
        c = _eval(bh)
        results.append((c, bh))
    results.sort(key=lambda x: x[0])
    return results[0][1], results[0][0]
