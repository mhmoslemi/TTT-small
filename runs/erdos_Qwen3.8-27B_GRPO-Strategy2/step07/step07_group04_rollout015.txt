import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # project to sum constraint
        h = _proj_to_sum(h, target)
    return h

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_full(h, w):
    """Full gradient of weighted sum of correlations."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[n-1+s]
        if ws < 1e-18:
            continue
        if s >= 0:
            idx = np.arange(n - s)
            g[idx] += ws * (1.0 - h[idx + s])
            if s > 0:
                idx2 = np.arange(s, n)
                g[idx2] -= ws * h[idx2 - s]
        else:
            sp = -s
            idx = np.arange(sp, n)
            g[idx] += ws * (1.0 - h[idx - sp])
            idx2 = np.arange(n - sp)
            g[idx2] -= ws * h[idx2 + sp]
    return g * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_to_sum(h, target)
    return h

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_wei(h, w):
    """Gradient of weighted sum of correlations. w has 2n-1 entries (correlate indexing)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = w[n - 1 + s]
        if ws < 1e-18:
            continue
        if s >= 0:
            g[:n - s] += ws * (1.0 - h[s:n])
            if s > 0:
                g[s:] -= ws * h[:n - s]
            else:
                g[:] -= ws * h[:]
        else:
            sp = -s
            g[sp:] += ws * (1.0 - h[:n - sp])
            g[:n - sp] -= ws * h[sp:]
    return g * dx

def _smooth_val(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    a = alpha * c
    amax = a.max()
    return amax / alpha + np.log(np.exp(a - amax).sum()) / alpha

def _softmax_weights(c, alpha):
    a = alpha * c
    amax = a.max()
    w = np.exp(a - amax)
    w /= w.sum()
    return w

            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 4e-6 * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d
            h[j] -= d
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5c < best_c5:
                    best_c5, best_h = c5c, h.copy()
            else:
                h[i] -= d
                h[j] += d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
