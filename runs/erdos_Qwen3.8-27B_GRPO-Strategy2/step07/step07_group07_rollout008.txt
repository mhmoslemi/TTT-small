        nj = h2[j] + eps
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            c_new = _eval(h2)
            if c_new < best_c:
                best_c = c_new
                best_h = h2.copy()
            else:
                h2[i], h2[j] = ni + eps, nj - eps
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0
    def _bc(margin):
        return (time.time() - t0) > budget_s - margin

    # Phase 1: Coarse-to-fine block optimization
    for M in [5, 10, 20, 40]:
        if _bc(500):
            break
        W = n0 // M
        b = np.array([h[m * W:(m + 1) * W].mean() for m in range(M)])
        target_b = (n0 / 2.0) / W
        b = _proj_sum(b, target_b)
        for beta in [50, 500]:
            if _bc(500):
                break
            h_try = _block_opt(b, M, n0, beta, 1e5, 200)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try
            b = np.array([h_try[m * W:(m + 1) * W].mean() for m in range(M)])

    # Phase 2: Full LSE optimization with increasing beta
    for beta in [100, 500, 2000, 8000]:
        if _bc(350):
            break
        pen = max(1e4, 1e3 * beta)
        h_try = _lbfgs(best_h.copy(), beta, pen, 500)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 3: Binary push - freeze and re-optimize
    if not _bc(300):
        for thr in [0.08, 0.15, 0.25]:
            if _bc(300):
                break
            h2 = best_h.copy()
            h2[h2 < thr] = 0.0
            h2[h2 > 1.0 - thr] = 1.0
            h2 = _proj_sum(h2, n0 / 2.0)
            h_try = _lbfgs(h2, 3000, 1e6, 300)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 4: Simulated annealing flip search
    if not _bc(250):
        deadline = time.time() + min(100, budget_s - 250 - (time.time() - t0))
        h_try, c_try = _flip_sa(best_h, 300000, rng, t0, deadline)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
        # Re-refine after SA
        h_try = _lbfgs(best_h.copy(), 5000, 1e6, 300)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 5: Smooth perturbation search
    if not _bc(150):
        deadline = time.time() + min(60, budget_s - 150 - (time.time() - t0))
        h_try, c_try = _smooth_perturb(best_h, 40000, rng, t0, deadline)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
        h_try = _lbfgs(best_h.copy(), 8000, 1e7, 200)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 6: Mass transfer fine search
    if not _bc(80):
        deadline = time.time() + min(40, budget_s - 80 - (time.time() - t0))
        h_try, c_try = _mass_transfer(best_h, 80000, rng, t0, deadline)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 7: Second SA round at lower temperature
    if not _bc(50):
        deadline = time.time() + min(25, budget_s - 50 - (time.time() - t0))
        h_try, c_try = _flip_sa(best_h, 100000, rng, t0, deadline)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 8: Final LSE polish
    if not _bc(10):
        h_try = _lbfgs(best_h.copy(), 10000, 1e7, 150)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
