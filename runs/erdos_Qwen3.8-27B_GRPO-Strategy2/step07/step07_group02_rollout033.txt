    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    xg = np.arange(n) * (2.0 / n)

    # Phase 1: Fourier pre-optimization for basin exploration
    for M in [8, 14, 22]:
        if time.time() > deadline - 650:

        A = np.zeros((n, 2*M))
        for k in range(1, M+1):
            A[:, 2*k-2] = np.cos(k * np.pi * xg)
            A[:, 2*k-1] = np.sin(k * np.pi * xg)
        theta0, _, _, _ = np.linalg.lstsq(A, h0 - 0.5, rcond=None)
        for alpha_f in [200, 800]:
            if time.time() > deadline - 650:

            h_f = _repair(_fourier_h(th, xg, M), n)
            c5f = _eval(h_f)
            if c5f < best_c5:
                best_c5, best_h = c5f, h_f.copy()
        # Perturbed restart
        theta_p = theta0 + 0.08 * rng.randn(2*M)
        th2 = _fourier_opt(theta_p, xg, M, 500, 2000, deadline)
        h_f2 = _repair(_fourier_h(th2, xg, M), n)
        c5f2 = _eval(h_f2)
        if c5f2 < best_c5:
            best_c5, best_h = c5f2, h_f2.copy()

    # Phase 2: L-BFGS full grid with expanding alpha
    h = best_h.copy()
    for alpha in [100, 500, 2000, 8000, 30000, 100000, 400000, 1500000, 5000000]:
        if time.time() > deadline - 180:
            break
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha,),
                           method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': 2500, 'ftol': 1e-15, 'gtol': 1e-10})
            h = _repair(res.x, n)
        except Exception:
            continue
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 3: Multi-config top-lag descent
    for nl in [2, 3, 5, 4, 3]:
        if time.time() > deadline - 100:
            break
        h_ml, c5_ml = _toplag(best_h.copy(), nl, 1000, deadline)
        if c5_ml < best_c5:
            best_c5, best_h = c5_ml, h_ml.copy()

    # Phase 4: Shimmy - perturb and re-optimize

        for trial in range(4):

            try:
                res = minimize(_ls_obj, h_sh, jac=_ls_grad, args=(200000,),
                               method='L-BFGS-B', bounds=bnds,
                               options={'maxiter': 1500, 'ftol': 1e-15, 'gtol': 1e-10})
                h_sh = _repair(res.x, n)
            except Exception:
                continue
            c5s = _eval(h_sh)
            if c5s < best_c5:
                best_c5, best_h = c5s, h_sh.copy()

    # Phase 5: Final high-alpha L-BFGS
    if time.time() < deadline - 60:
        h = _lbfgs_final = best_h.copy()
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(_ls_obj, h, jac=_ls_grad, args=(10000000,),
                           method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': 4000, 'ftol': 1e-15, 'gtol': 1e-10})
            h = _repair(res.x, n)
        except Exception:
            pass
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 6: Top-lag final pass
    if time.time() < deadline - 40:
        h_ml, c5_ml = _toplag(best_h.copy(), 2, 600, deadline)
        if c5_ml < best_c5:
            best_c5, best_h = c5_ml, h_ml.copy()

    # Phase 7: 3-point local search
    if time.time() < deadline - 10:
        h_ls, c5_ls = _local3(best_h, 50000, rng, deadline)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    # Phase 8: Ultra-fine 2-point polish
    if time.time() < deadline - 2:
        h_ls, c5_ls = _local2(best_h, 100000, rng, deadline)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
