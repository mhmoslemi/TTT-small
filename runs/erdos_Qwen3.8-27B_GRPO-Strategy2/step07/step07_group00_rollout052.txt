import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c)).copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        sh = (n - 1) - idx[ii]
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _fista(h0, alpha, topk, time_budget, lr_init):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    lr = lr_init
    f = _smax_obj(h, alpha, topk)
    while time.time() < time_budget:
        g = _smax_grad(y, alpha, topk)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        step = lr / gnorm
        y_new = _proj(y - step * g, target)
        f_new = _smax_obj(y_new, alpha, topk)
        if f_new > f + 1e-12:
            lr *= 0.5
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = y_new + ((t - 1.0) / t_new) * (y_new - h)
        h = y_new
        t = t_new
        f = f_new
        lr = min(lr * 1.01, lr_init * 8.0)

def _maxgrad_cd(h, niter, rng, deadline, lr_init):

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c

        topk = 3
        k_idx = np.argpartition(c, -topk)[-topk:]

                break

                bh = h.copy()

def _slsqp_refine(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP quick refine (0-80s)
    dl1 = min(deadline, t0 + 80)
    for alpha, tk, mi in [(8000, 15, 200), (50000, 8, 150)]:
        if time.time() > dl1:
            break
        h_r = _slsqp_refine(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h_r)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_r.copy()

    # Phase 2: FISTA with increasing alpha (80-250s)
    dl2 = min(deadline, t0 + 250)
    for alpha, tk, lr in [(20000, 12, 0.003), (200000, 6, 0.0015), (2000000, 3, 0.0008)]:
        if time.time() > dl2:
            break
        tsub = min(dl2, time.time() + 50)
        h_f, c5_f = _fista(best_h.copy(), alpha, tk, tsub, lr)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 3: Fourier perturb + FISTA restarts (250-400s)
    dl3 = min(deadline, t0 + 400)
    for trial in range(10):
        if time.time() > dl3:
            break
        h_p = _fourier_perturb(best_h, rng, 3, 0.025)
        tsub = min(dl3, time.time() + 12)
        h_f, c5_f = _fista(h_p, 100000, 5, tsub, 0.001)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 4: Block + Fourier annealing (400-600s)
    dl4 = min(deadline, t0 + 600)
    h_a, c5_a = _anneal(best_h.copy(), 50000, rng, min(dl4, time.time() + 70), 0.003, 1e-10, 'block')
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()
    h_a2, c5_a2 = _anneal(best_h.copy(), 50000, rng, min(dl4, time.time() + 60), 0.002, 1e-10, 'fourier')
    if c5_a2 < best_c5:
        best_c5 = c5_a2
        best_h = h_a2.copy()

    # Phase 5: Very high alpha FISTA (600-780s)
    dl5 = min(deadline, t0 + 780)
    for alpha, tk, lr in [(10000000, 3, 0.0004), (50000000, 2, 0.0002)]:
        if time.time() > dl5:
            break
        tsub = min(dl5, time.time() + 80)
        h_f, c5_f = _fista(best_h.copy(), alpha, tk, tsub, lr)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Max-gradient coordinate descent (780-960s)
    dl6 = min(deadline, t0 + 960)
    best_h, best_c5 = _maxgrad_cd(best_h, 300000, rng, dl6, lr_init=3e-5)

    # Phase 7: SLSQP final touch
    if time.time() < deadline - 30:
        h_r = _slsqp_refine(best_h.copy(), 1000000, 3, 100, deadline - 30)
        c5v = _eval(h_r)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_r.copy()

    # Phase 8: Ultra-fine polish
    dl7 = deadline
    remaining = max(0, dl7 - time.time())
    n_pol = min(int(remaining / 0.001), 500000)
    best_h, best_c5 = _fine_polish(best_h, n_pol, rng, dl7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
