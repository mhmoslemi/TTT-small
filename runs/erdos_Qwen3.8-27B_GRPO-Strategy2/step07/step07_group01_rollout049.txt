    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_full(x) * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_full(x) * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, steps, step0):
    n = len(h)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _smooth_grad(h, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 8.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 8.0)
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step, n_pert=3):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    best_c5 = c5cur
    for _ in range(n_iter):
        step = base_step * (0.5 + rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
                best_c5 = c5n
    return best_h, best_c5

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution coarse optimization
    for ratio in [4, 8]:
        if _el() > 60:
            break
        nc = max(40, n // ratio)
        hc = _downsample(h, n // nc)
        for alpha in [500, 3000, 20000]:
            if _el() > 80:
                break
            hc = _slsqp(hc, alpha, 30, 500, 1e-14)
        h_up = _upsample(hc, n // nc)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: SLSQP alphadive at full resolution
    alpha_seq = [3000, 10000, 50000, 200000, 800000, 3000000, 10000000, 50000000, 200000000]
    for alpha in alpha_seq:
        if _el() > 400:
            break
        lift = min(80, max(20, int(40 * np.log10(alpha / 3000) + 20)))
        h_new = _slsqp(best_h, alpha, lift, 1500, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient with top-lag focus
    if _el() < budget_s - 250:
        configs = [
            (5000, 40, 800, 0.005),
            (30000, 50, 600, 0.003),
            (150000, 60, 500, 0.002),
            (600000, 80, 400, 0.001),
            (2000000, 100, 300, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if _el() > budget_s - 180:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Another SLSQP pass with very high alpha
    if _el() < budget_s - 120:
        for alpha in [500000000, 2000000000, 8000000000]:
            if _el() > budget_s - 60:
                break
            h_new = _slsqp(best_h, alpha, 120, 600, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Gradient-informed local search
    if _el() < budget_s - 30:
        ntries = min(int((budget_s - 30 - _el()) / 0.002), 500000)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 4e-6, 3)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 2 - _el()) / 0.001), 400000)
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 5)) % nc2
            d = 3e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
