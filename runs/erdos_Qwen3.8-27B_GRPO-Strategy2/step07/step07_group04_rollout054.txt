    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        else:

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_grad(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    a = np.clip(alpha * (c - m), -700, 0)
    w = np.exp(a)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    g = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for k in active:
        kk = int(k)
        sh = (n - 1) - kk
        wk = w[kk] * dx
        if sh >= 0:
            r = n - sh
            i1 = np.arange(r)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov(h, steps, step0, alpha0, alpha1, mu, t0, deadline):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n)
    stall = 0
    for s in range(steps):
        if time.time() > deadline:
            break
        frac = min(s / max(steps - 1, 1), 1.0)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        step = step0 / (1.0 + 0.003 * s)
        g = _smooth_grad(h, alpha, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-17:
            h, f = h_new, f_new
            stall = 0
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            stall += 1
            step *= 0.6
            if step < 1e-12:
                break
            h_new = _proj(h_prev - step * v, target)
            f_new = _eval(h_new)
            if f_new < f:
                h, f = h_new, f_new
                if f < best_f:
                    best_f, best_h = f, h.copy()
                stall = 0
            if stall > 80:
                break
    return best_h, best_f

def _pair_ls(h, n_rounds, rng, t0, deadline, scale):
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 1e-6 < ni < 1 - 1e-6 and 1e-6 < nj < 1 - 1e-6:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-17:
                f = fn
                if fn < best_f:
                    best_f, best_h = fn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_f

def _targeted_ls(h, n_rounds, rng, t0, deadline, scale):
    """Focus on coordinates in the top correlation lags."""
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    c = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(c)[-20:]
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        k = top_lags[rng.randint(len(top_lags))]
        sh = (n - 1) - int(k)
        if sh >= 0:
            cand_idx = np.arange(0, n - sh)
        else:
            sp = -sh
            cand_idx = np.arange(sp, n)
        i = cand_idx[rng.randint(len(cand_idx))]
        j = (i + rng.randint(1, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 1e-6 < ni < 1 - 1e-6 and 1e-6 < nj < 1 - 1e-6:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < f - 1e-17:
                f = fn
                if fn < best_f:
                    best_f, best_h = fn, h.copy()
                c = np.correlate(h, 1.0 - h, mode='full')
                top_lags = np.argsort(c)[-20:]
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Nesterov smooth-max, moderate alpha
    dl = t0 + 180
    h2, f2 = _nesterov(h, 200000, 0.005, 100, 3000, 0.9, t0, dl)
    if f2 < best_c5:
        best_h, best_c5 = h2, f2

    # Phase 2: Higher alpha
    if tl() > 180:
        dl = t0 + 350
        h2, f2 = _nesterov(best_h.copy(), 200000, 0.003, 3000, 30000, 0.88, t0, dl)
        if f2 < best_c5:
            best_h, best_c5 = h2, f2

    # Phase 3: Very high alpha (near-exact max gradient)
    if tl() > 150:
        dl = t0 + 510
        h2, f2 = _nesterov(best_h.copy(), 200000, 0.001, 30000, 200000, 0.85, t0, dl)
        if f2 < best_c5:
            best_h, best_c5 = h2, f2

    # Phase 4: Ultra-high alpha convergence
    if tl() > 120:
        dl = t0 + 640
        h2, f2 = _nesterov(best_h.copy(), 200000, 0.0004, 200000, 1000000, 0.82, t0, dl)
        if f2 < best_c5:
            best_h, best_c5 = h2, f2

    # Phase 5: Perturbation escape
    if tl() > 100:
        h_p = best_h + 0.008 * rng.randn(n)
        h_p = _proj(h_p, n * 0.5)
        dl = t0 + tl() - 80
        h2, f2 = _nesterov(h_p, 100000, 0.002, 1000, 50000, 0.88, t0, dl)
        if f2 < best_c5:
            best_h, best_c5 = h2, f2

    # Phase 6: Targeted local search on active lags
    if tl() > 50:
        dl = t0 + tl() - 30
        nr = min(int((tl() - 30) / 0.003), 100000)
        best_h, best_c5 = _targeted_ls(best_h, nr, rng, t0, dl, 8e-6)

    # Phase 7: Pair local search (wider)
    if tl() > 30:
        dl = t0 + tl() - 12
        nr = min(int((tl() - 12) / 0.003), 80000)
        best_h, best_c5 = _pair_ls(best_h, nr, rng, t0, dl, 4e-6)

    # Phase 8: Ultra-fine pair
    if tl() > 8:
        dl = t0 + tl() - 2
        nr = min(int((tl() - 2) / 0.003), 50000)
        best_h, best_c5 = _pair_ls(best_h, nr, rng, t0, dl, 1.5e-6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
