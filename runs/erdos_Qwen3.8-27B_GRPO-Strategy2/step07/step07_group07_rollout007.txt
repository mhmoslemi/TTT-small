def _adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step, in-place."""
    if 'm' not in state:
        state['m'] = np.zeros_like(params)
        state['v'] = np.zeros_like(params)
        state['t'] = 0
    state['t'] += 1
    t = state['t']
    state['m'] = beta1 * state['m'] + (1 - beta1) * grads
    state['v'] = beta2 * state['v'] + (1 - beta2) * grads**2
    m_hat = state['m'] / (1 - beta1**t)
    v_hat = state['v'] / (1 - beta2**t)
    return params - lr * m_hat / (np.sqrt(v_hat) + eps)

import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _grad_lag(h, k_idx):
    """Gradient of c_k w.r.t. h for a specific lag k_idx in full correlation array."""
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g

def _adam_state_init(n):
    return {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}

def _adam_update(h, g, state, lr, b1=0.85, b2=0.999, eps=1e-8):
    state['t'] += 1
    t = state['t']
    state['m'] = b1 * state['m'] + (1 - b1) * g
    state['v'] = b2 * state['v'] + (1 - b2) * (g * g)
    m_hat = state['m'] / (1.0 - b1**t)
    v_hat = state['v'] / (1.0 - b2**t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps)

def _adam_phase(h, T, lr, n_iter, bp, el_fn, bc_fn):
    """Adam with log-sum-exp smoothed max + binary pressure."""
    n = len(h)
    h_c = h.copy().astype(float)
    state = _adam_state_init(n)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        # Log-sum-exp gradient
        c_max = c.max()
        ec = np.exp(T * (c - c_max))
        w = ec / ec.sum()
        g = np.zeros(n)
        # Only compute gradient for lags with significant weight
        thresh = 0.01 / len(c)
        for k in range(len(c)):
            if w[k] < thresh:
                continue
            g += w[k] * _grad_lag(h_c, k)
        g *= T
        if bp > 0:
            g += bp * (1.0 - 2.0 * h_c)
        h_new = _adam_update(h_c, g, state, lr)
        h_new = _proj_sum(h_new, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c

def _subgrad_phase(h, k_top, lr0, n_iter, beta, el_fn, bc_fn):
    """Nesterov-like momentum on top-k subgradient."""
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    lr = lr0
    best_c = _eval(h_c)
    best_h = h_c.copy()
