import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        d = k - (n - 1)
        if d >= 0:
            if d == 0:
                g += 1.0 - 2.0 * h
            else:
                g[d:] += (1.0 - h[:n-d])
                g[:n-d] -= h[d:]
        else:
            sp = -d
            g[:n-sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n-sp]
    return g

def _mlgd(h, rounds, k_top, steps_init, check_fn, use_momentum=True):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top_k) * dx / k_top
        if use_momentum:
            v = mu * v + (1 - mu) * g
            g = v
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                if c5n < c5c - 0.5 * steps[si] * np.mean(np.abs(g)):
                    mu = min(mu + 0.02, 0.95)
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.5] + [s for s in steps if s < steps[si] * 0.5]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            v *= 0.5
            stall += 1
            if stall > 10:
                v = np.zeros(n)
                break
        else:
            stall = 0
    return h, c5c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lags(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        d = k - (n - 1)
        if d >= 0:
            if d == 0:
                g += 1.0 - 2.0 * h
            else:
                g[d:] += (1.0 - h[:n - d])
                g[:n - d] -= h[d:]
        else:
            sp = -d
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g

def _mlgd(h, rounds, k_top, steps_init, check_fn, momentum):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = list(steps_init)
    stall = 0
    v = np.zeros(n)
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top_k) * dx / k_top
        if momentum > 0:
            v = momentum * v + (1.0 - momentum) * g
            g_step = v
        else:
            g_step = g
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * g_step, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.4] + [s for s in steps if s < steps[si] * 0.5]
                if len(steps) > 10:
                    steps = steps[:10]
                break
        if not improved:
            steps = [s * 0.4 for s in steps]
            v *= 0.5
            stall += 1
            if stall > 12:
                v = np.zeros(n)
                break
        else:
            stall = 0
    return h, c5c

def _lsx_obj(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 30000.0
    sc = np.clip(a * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 30000.0
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    top_idx = np.argsort(w)[-5:]
    g = _grad_lags(x, [(n - 1) + (k - (n - 1)) for k in range(2 * n - 1)][:0])
    g = np.zeros(n)
    for idx in top_idx:
        wk = w[idx]
        d = idx - (n - 1)
        if d >= 0:
            if d == 0:
                g += wk * (1.0 - 2.0 * x)
            else:
                g[d:] += wk * (1.0 - x[:n - d])
                g[:n - d] -= wk * x[d:]
        else:
            sp = -d
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g * (2.0 / n)

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _local_2pt(h, ntry, rng, c5c, check_fn, step_init):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    step = step_init
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                step = min(step * 1.2, 0.05)
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                step *= 0.7
                if step < 1e-7:
                    step = step_init * 0.3
    return best_h, best_c

def _local_swap(h, ntry, rng, c5c, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, nc - 1)) % nc
        amt = 0.02 * rng.rand()
        if amt > h[i] or amt > 1.0 - h[j]:
            amt = min(h[i], 1.0 - h[j]) * 0.5
        if amt < 1e-6:
            continue
        h[i] -= amt
        h[j] += amt
        c5n = _eval(h)
        if c5n < c5c - 1e-18:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i] += amt
            h[j] -= amt
    return best_h, best_c

    # Phase 1: Corrected multi-lag gradient with momentum
    k_cycle = [5, 7, 3, 5, 4, 6, 3, 5, 8, 4]
    for ki in range(len(k_cycle)):
        if _bc():
            break
        kt = k_cycle[ki]
        mom = 0.85 if ki < 6 else 0.7
        h2, c52 = _mlgd(best_h.copy(), 800, kt,
                        [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 5e-4], _bc, mom)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Fine gradient refinement, no momentum
    for kt in [3, 5, 4, 3, 6, 3]:
        if _bc():
            break
        h2, c52 = _mlgd(best_h.copy(), 600, kt,
                        [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4], _bc, 0.0)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: SLSQP with smooth max
    if _el() < budget_s - 200:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        for _att in range(3):
            if _bc():
                break
            try:
                res = minimize(_lsx_obj, best_h, args=(n,), jac=_lsx_grad,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-16})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 4: Perturbation restarts
    n = len(best_h)
    for trial in range(5):
        if _bc():
            break
        scale = [0.005, 0.01, 0.003, 0.008, 0.004][trial]
        h_sh = best_h.copy() + scale * rng.randn(n)
        h_sh = _repair(h_sh, n)
        h2, c52 = _mlgd(h_sh, 400, 4,
                        [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4], _bc, 0.8)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Final gradient polish
    for kt in [3, 5, 3, 4, 5, 3]:
        if _bc():
            break
        h2, c52 = _mlgd(best_h.copy(), 500, kt,
                        [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4], _bc, 0.5)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Swap-based local search
    if _el() < budget_s - 50:
        nt = min(int((budget_s - 50 - _el()) * 0.6 / 0.004), 150000)
        best_h, best_c5 = _local_swap(best_h, nt, rng, best_c5, _bc)

    # Phase 7: Adaptive 2-point local search
    if _el() < budget_s - 10:
        nt = min(int((budget_s - 10 - _el()) / 0.003), 200000)
        best_h, best_c5 = _local_2pt(best_h, nt, rng, best_c5, _bc, 0.01)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
