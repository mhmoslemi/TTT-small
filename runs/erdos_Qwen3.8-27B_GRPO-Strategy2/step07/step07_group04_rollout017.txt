```python
import numpy as np
from scipy.optimize import minimize
import time

_GP = {'alpha': 1.0, 'n': 1}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sm_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    thr = 1e-12
    for k in range(2 * n - 1):
        if w[k] < thr:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _grad_topk(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _nesterov_topk(h, k_top, steps, step0, t0, budget_s):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _s in range(steps):
        if time.time() - t0 > budget_s - 8:
            break
        g = _grad_topk(h, k_top)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 8.0)
        else:
            step *= 0.45
            if step < 1e-11:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _nesterov_sm(h, alpha, steps, step0, t0, budget_s):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    for _s in range(steps):
        if time.time() - t0 > budget_s - 8:
            break
        g = _sm_grad(h, alpha, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 6.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_wrap_f(x):
    return _sm_obj(x, _GP['alpha'], _GP['n'])

def _slsqp_wrap_g(x):
    return _sm_grad(x, _GP['alpha'], _GP['n'])

def _cons_f(x):
    return x.sum() - _GP['n'] * 0.5

def _cons_j(x):
    return np.ones(_GP['n'])

def _slsqp_opt(h, alpha, maxiter, ftol, t0, budget_s):
    n = _GP['n'] = len(h)
    _GP['alpha'] = alpha
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
    try:
        res = minimize(_slsqp_wrap_f, h, jac=_slsqp_wrap_g,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _lbfgs_opt(h, alpha, maxiter, t0, budget_s):
    n = len(h)
    pen = 1e8
    target = n * 0.5
    def _pen_f(x):
        return _sm_obj(x, alpha, n) + 0.5 * pen * (x.sum() - target) ** 2
    def _pen_g(x):
        return _sm_grad(x, alpha, n) + pen * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(_pen_f, h, jac=_pen_g, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search_sys(h, t0, budget_s, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    it = 0
    while time.time() - t0 < budget_s - 3:
        it += 1
        n_pert = 2 if it % 3 == 0 else rng.randint(2, 5)
        idxs = rng.choice(n, n_pert, replace=False)
        d = 4e-6 * np.randn(n_pert)
        d -= d.mean()
        orig = h[idxs].copy()
        new = h[idxs] + d
        if np.all(new >= -1e-15) and np.all(new <= 1.0 + 1e-15):
            h[idxs] = np.clip(new, 0, 1)
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[idxs] = orig
    return best_h, best_c

def _local_search_fine(h, t0, budget_s, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() - t0 < budget_s - 2:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 2e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _multi_scale(h_best, t0, budget_s, rng):
    n = len(h_best)
    best = h_best.copy()
    best_c = _eval(best)
    for factor in [4, 2]:
        if time.time() - t0 > budget_s - 40:
            break
        n_s = n // factor
        if n_s < 40:
            continue
        h_s = h_best[:n_s * factor].reshape(n_s, factor).mean(axis=1)
        h_s = _repair(h_s, n_s)
        h_s = _nesterov_topk(h_s, 5, 2000, 0.005, t0, budget_s - 20)
        h_s = _nesterov_sm(h_s, 500, 1000, 0.003, t0, budget_s - 15)
        h_up = np.repeat(h_s, factor)
        if len(h_up) < n:
            h_up = np.concatenate([h_up, h_best[n - len(h_up):]])
        h_up = _repair(h_up, n)
        c5v = _eval(h_up)
        if c5v < best_c:
            best_c = c5v
            best = h_up.copy()
    return best, best_c
```
