def _coord_opt(h, i, n):
    """Find optimal h[i] given all other coordinates fixed."""
    # For each lag k, c_k = a_k + b_k * h[i] where:
    # a_k = c_k with h[i] = 0
    # b_k = dc_k/dh[i] (gradient contribution)
    
    # Compute base correlations and gradients
    h[i] = 0.0
    c_base = _corr(h)  # c_k with h[i] = 0
    
    # Gradient of c_k w.r.t. h[i]:
    # dc_k/dh[i] = (1 - h[i+shift_k]) - h[i-shift_k] (with boundary)
    # But h[i] = 0 now, so:
    b = np.zeros(2*n-1)
    for k in range(2*n-1):
        sh = (n-1) - k
        # h[i] appears at position i in the "forward" sum
        # and at position i - sh in the "backward" part
        val = 0.0
        # As h[j] where j=i: contributes (1 - h[i+sh]) if i+sh < n
        if 0 <= i+sh < n:
            val += (1.0 - h[i+sh])
        # As h[j+sh] where j+sh=i, j=i-sh: contributes -h[i-sh] if 0<=i-sh<n
        if 0 <= i-sh < n:
            val -= h[i-sh]
        b[k] = val
    
    # Now c_k(h[i]) = c_base[k] + b[k] * h[i] for h[i] in [0,1]
    # Minimize max_k (c_base[k] + b[k]*t) for t in [0,1]
    # This is a convex piecewise linear function in t
    # Minimum is at a "knee" or at boundary
    
    # Evaluate at t=0 and t=1
    f0 = c_base.max()  # max of c_base + 0
    f1 = (c_base + b).max()  # max of c_base + b*1
    
    # Find the crossing points
    # The minimum of max of lines is where two lines cross at the peak
    # Simple approach: ternary search on [0,1]
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f_m1 = (c_base + b * m1).max()
        f_m2 = (c_base + b * m2).max()
        if f_m1 < f_m2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2
    h[i] = t_opt
    return h

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

            lo = mid
        else:
            hi = mid

                c5c = c5n

            step = step0
            if stall > 15:
                break
        else:
            stall = 0
            step = min(step * 1.3, 0.5)
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _project_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, k_top, n_iter, step0, check_fn):
    n = len(h)
    h_c = h.copy()
    c5c = _eval(h_c)
    step = step0
    stall = 0
    best = h_c.copy()
    best_c = c5c
    for _ in range(n_iter):
        if check_fn():
            break
        c = _corr(h_c)
        top = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in top:
            g += _grad_lag(h_c, k)
        g /= k_top
        ok = False
        for _b in range(14):
            h_n = _project_sum(h_c - step * g, n * 0.5)
            c5n = _eval(h_n)
            if c5n < c5c - 1e-15:
                h_c = h_n
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best = h_c.copy()
                ok = True
                break
            step *= 0.45
        if ok:
            stall = 0
            step = min(step * 1.3, 0.5)
        else:
            stall += 1
            step = step0
            if stall > 18:
                break
    return best, best_c

def _coord_pass(h, n, c, check_fn):
    """One full coordinate descent pass with exact 1D minimization."""
    n_lags = 2 * n - 1
    sh_arr = np.arange(-(n - 1), n)
    for i in range(n):
        if check_fn():
            return h, c
        jp = i + sh_arr
        jm = i - sh_arr
        b = np.zeros(n_lags)
        mp = (jp >= 0) & (jp < n)
        b[mp] += 1.0 - h[jp[mp]]
        mm = (jm >= 0) & (jm < n)
        b[mm] -= h[jm[mm]]
        t = h[i]
        c_base = c - t * b
        lo, hi = 0.0, 1.0
        for _ in range(45):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if (c_base + b * m1).max() < (c_base + b * m2).max():
                hi = m2
            else:
                lo = m1
        t_new = (lo + hi) * 0.5
        if abs(t_new - t) > 1e-15:
            h[i] = t_new
            c = c_base + b * t_new
    return h, c

def _coord_descent(h, n, max_passes, check_fn):
    h = h.copy()
    c = _corr(h)
    best_c = c.max()
    best = h.copy()
    for _p in range(max_passes):
        if check_fn():
            break
        h, c = _coord_pass(h, n, c, check_fn)
        h = _project_sum(h, n * 0.5)
        c = _corr(h)
        cm = c.max()
        if cm < best_c:
            best_c = cm
            best = h.copy()
    return best, best_c
