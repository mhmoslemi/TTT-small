def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:

        m3 = np.arange(lo3, hi3 + 1)
        g[lo3:hi3+1] -= (1.0 - h[j + m3 - (n-1)]) * dx
    lo4 = max(0, j); hi4 = min(2 * n - 2, j + n - 1)
    if lo4 <= hi4:
        m4 = np.arange(lo4, hi4 + 1)
        g[lo4:hi4+1] += h[j - m4 + (n-1)] * dx
    return g

def _smart_cd(h, n_iter, rng, deadline, temp0=0.002, cool_rate=0.9997):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full').copy() * (2.0 / n)
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    temp = temp0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 500 == 0 and stable > 2000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            stable += 1
            continue
        d = j - i
        k1 = n - 1 + d
        k2 = n - 1 - d
        g = _grad_diff(h, i, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            stable += 1
            continue
        for _ in range(42):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt; h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        delta = c5_new - c5_cur
        if delta < -1e-14:
            c5_cur = c5_new; stable = 0
            if c5_cur < best_c:
                best_c = c5_cur; best_h = h.copy()
        else:
            accept = temp > 0 and rng.rand() < np.exp(-delta / temp)
            if accept:
                c5_cur = c5_new
            else:
                h[i] -= d_opt; h[j] += d_opt
                c -= d_opt * g
                if 0 <= k1 < 2 * n - 1:
                    c[k1] -= d_opt * d_opt
                if 0 <= k2 < 2 * n - 1:
                    c[k2] -= d_opt * d_opt
                stable += 1
        temp *= cool_rate
    return best_h, best_c

def _subgrad(h, rng, deadline, top_k=3):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy(); best_c = c5_cur
    for it in range(3000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            sh = k - (n - 1)
            if sh >= 0:
                for p in range(n - sh):
                    g[p] += (1.0 - h[p + sh])
                if sh > 0:
                    for p in range(sh, n):
                        g[p] -= h[p - sh]
            else:
                sp = -sh
                for p in range(sp, n):
                    g[p] += (1.0 - h[p - sp])
                for p in range(n - sp):
                    g[p] -= h[p + sp]
        g *= dx / top_k
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.02, 0.5 / gn)
        improved = False
        for sm in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_sum(h - sm * step * g, 0.5 * n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new; c5_cur = c5n; improved = True
                break
        if c5_cur < best_c:
            best_c = c5_cur; best_h = h.copy()
        if not improved:
            h = h + 0.01 * rng.randn(n)
            h = _repair(h, n)
    return best_h, best_c

def _fine_polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy(); best_c = c5_cur
    scales = [5e-5, 2e-5, 1e-5]
    for scale in scales:
        for it in range(80000):
            if time.time() > deadline:
                return best_h, best_c
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5_cur - 1e-18:
                    c5_cur = c5n
                    if c5n < best_c:
                        best_c = c5n; best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    # Multi-start perturbations
    for ps in [0.05, 0.1, 0.2, 0.4]:
        hp = np.clip(h + ps * rng.randn(n0), 0, 1)
        hp = _proj_sum(hp, 0.5 * n0)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()
    # Phase 1: Subgradient (10%)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _subgrad(best_h.copy(), rng, dl1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1
    # Phase 2: Main CD with annealing (45%)
    dl2 = t0 + budget_s * 0.55
    h2, c5_2 = _smart_cd(best_h.copy(), 500000, rng, dl2, temp0=0.003, cool_rate=0.9998)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2
    # Phase 3: Second CD round with different params (20%)
    rng2 = np.random.RandomState(seed + 1000)
    dl3 = t0 + budget_s * 0.78
    h3, c5_3 = _smart_cd(best_h.copy(), 300000, rng2, dl3, temp0=0.001, cool_rate=0.9999)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3
    # Phase 4: Subgradient refinement (8%)
    dl4 = t0 + budget_s * 0.88
    h4, c5_4 = _subgrad(best_h.copy(), rng, dl4, top_k=5)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4
    # Phase 5: Fine polish (10%)
    dl5 = t0 + (budget_s - 3)
    h5, c5_5 = _fine_polish(best_h.copy(), rng, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5
    # Final repair and verify
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
