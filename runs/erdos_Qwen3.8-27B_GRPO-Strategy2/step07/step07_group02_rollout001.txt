def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _grad_c(h):
    """Compute gradient of all c_k with respect to h. Returns (2n-1, n) matrix."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        sh = (n-1) - k
        if sh >= 0:
            np_ = n - sh
            G[k, :np_] += (1.0 - h[sh:])
            if sh > 0:
                G[k, sh:] -= h[:np_]
        else:
            sp = -sh
            G[k, sp:] += (1.0 - h[:-sp])
            G[k, :-sp] -= h[sp:]
    G *= dx
    return G

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _slp_iter(h, deadline):
    """One SLP iteration: linearize c_k at h, solve LP, return (h_new, improved)."""
    n = len(h)
    dx = 2.0 / n
    c = _cvals(h)
    
    # Compute gradient matrix G (2n-1, n)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        sh = (n-1) - k
        if sh >= 0:
            np_ = n - sh
            G[k, :np_] += (1.0 - h[sh:])
            if sh > 0:
                G[k, sh:] -= h[:np_]
        else:
            sp = -sh
            G[k, sp:] += (1.0 - h[:-sp])
            G[k, :-sp] -= h[sp:]
    G *= dx
    
    # LP: min t s.t. G @ h_new - t <= G @ h - c, 0<=h_new<=1, sum(h_new)=n/2
    # Variables: [h_new (n), t (1)]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    A_corr = np.column_stack([G, np.full(2*n-1, -1.0)])
    b_corr = G @ h - c
    
    # Box: h_new <= 1, -h_new <= 0
    A_box = np.zeros((2*n, n+1))
    A_box[:n, :n] = np.eye(n)
    A_box[n:, :n] = -np.eye(n)
    b_box = np.concatenate([np.ones(n), np.zeros(n)])
    
    A_ub = np.vstack([A_corr, A_box])
    b_ub = np.concatenate([b_corr, b_box])
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    
    bounds = [(0, 1)] * n + [(None, None)]

            best_h = h_trial

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n * 0.5])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0, 1)] * n + [(None, None)], method='highs',
                          options={'presolve': True, 'time_limit': 8})
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = res.x[:n]
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_t = _repair((1 - alpha) * h + alpha * h_lp, n)
            c_t = _eval(h_t)
            if c_t < best_c - 1e-14:
                h = h_t
                best_c = c_t
                best = h.copy()
                improved = True
                break
        if improved:
            stagn = 0
        else:
            stagn += 1
            if stagn > 15:
                rng2 = np.random.RandomState(it * 7 + n)
                h = _repair(h + 0.008 * rng2.randn(n), n)
                stagN = 0
                stagn = 0
    return best, best_c

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            nt = n - sh
            g[:nt] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:nt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=[(0, 1)] * n,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _dual_pg(h, n_iter, beta, step0, deadline):
    n = len(h)
    w = np.ones(2 * n - 1) / (2 * n - 1)
    step = step0
    for _ in range(n_iter):
        if time.time() > deadline - 20:
            break
        c = _cvals(h)
        w *= np.exp(beta * (c - c.max()))
        w /= w.sum()
        G = _grad_mat(h)
        g = (w[:, None] * G).sum(axis=0)
        h = _repair(h - step * g, n)
        step *= 0.97
    return h

def _local2(h, niter, rng, deadline):
    n = len(h)
    bc5 = _eval(h)
    bh = h.copy()
    for _ in range(niter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 4e-6 * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c = _eval(h)
            if c < bc5:
                bh = h.copy()
                bc5 = c
            else:
                h[i], h[j] = hi - d, hj + d
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLP (primary new approach)
    h_s, c_s = _slp_run(best_h, 300, deadline)
    if c_s < best_c5:
        best_c5 = c_s
        best_h = h_s

    # Phase 2: Dual projected gradient (complementary exploration)
    if time.time() < deadline - 200:
        h_d = _dual_pg(best_h.copy(), 2000, 5.0, 0.05, deadline)
        cd = _eval(h_d)
        if cd < best_c5:
            best_c5 = cd
            best_h = h_d

    # Phase 3: L-BFGS with log-sum-exp
    for alpha in [3000, 30000, 200000]:
        if time.time() > deadline - 150:
            break
        h = _lbfgs(best_h.copy(), alpha, 3000, deadline)
        cv = _eval(h)
        if cv < best_c5:
            best_c5 = cv
            best_h = h

    # Phase 4: SLP again from improved point
    if time.time() < deadline - 100:
        h_s, c_s = _slp_run(best_h, 300, deadline)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s

    # Phase 5: Resolution reduction (downsample, SLP, upsample, refine)
    if time.time() < deadline - 120:
        nh = n0 // 2
        hh = _repair(best_h[:nh * 2].reshape(nh, 2).mean(axis=1), nh)
        hh_s, ch_s = _slp_run(hh, 200, deadline)
        if time.time() < deadline - 60:
            hu = _repair(np.repeat(hh_s, 2), n0)
            hu = _lbfgs(hu, 50000, 2000, deadline)
            cv = _eval(hu)
            if cv < best_c5:
                best_c5 = cv
                best_h = hu

    # Phase 6: Dual PG with different params
    if time.time() < deadline - 60:
        h_d = _dual_pg(best_h.copy(), 3000, 10.0, 0.03, deadline)
        cd = _eval(h_d)
        if cd < best_c5:
            best_c5 = cd
            best_h = h_d

    # Phase 7: Final SLP
    if time.time() < deadline - 40:
        h_s, c_s = _slp_run(best_h, 200, deadline)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s

    # Phase 8: High-alpha L-BFGS polish
    if time.time() < deadline - 20:
        h = _lbfgs(best_h, 2000000, 4000, deadline)
        cv = _eval(h)
        if cv < best_c5:
            best_c5 = cv
            best_h = h

    # Phase 9: Local search
    if time.time() < deadline - 3:
        hl, cl = _local2(best_h, 80000, rng, deadline)
        if cl < best_c5:
            best_c5 = cl
            best_h = hl

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
