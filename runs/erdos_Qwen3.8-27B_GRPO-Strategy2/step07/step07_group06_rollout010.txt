def _grad_all(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _grad_top(h, top_lags, n):
    """Gradient of sum of top_lags correlations w.r.t. h"""
    n2 = 2 * n - 1
    g = np.zeros(n)

        if shift == 0:

        elif shift > 0:

            g[shift:] -= h[:n - shift]
        else:
            sp = -shift
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g / len(top_lags)

def _downsample(h, n_new):
    """Downsample h to n_new points by averaging"""
    n_old = len(h)
    ratio = n_old / n_new
    idx = np.arange(n_new) * ratio
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n_old - 1)
    w = idx - i0
    h_new = (1 - w) * h[i0] + w * h[i1]
    return _repair(h_new, n_new)

        if 0 <= bk2 < 2 * n - 1:
            c[bk2] += bd * bd
        c5_cur = bc5

        if c5_cur < best_c - 1e-15:
            best_c = c5_cur
            best_h = h.copy()
            stable = 0
        else:
            stable += 1

    return best_h, best_c

def _downsample(h, n_new):
    n_old = len(h)
    if n_new >= n_old:
        return _repair(h.copy(), n_new) if n_new == n_old else _repair(np.interp(
            np.linspace(0, 1, n_new), np.linspace(0, 1, n_old), h), n_new)
    idx = (np.arange(n_new) + 0.5) * n_old / n_new
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n_old - 1)
    w = idx - i0
    h_new = (1.0 - w) * h[i0] + w * h[i1]
    return _repair(h_new, n_new)

def _fine_polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(300000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 1e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-20:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Multi-resolution: try n0 and coarser versions
    res_list = [n0, int(n0 * 0.75), int(n0 * 0.6), int(n0 * 0.5)]
    res_list = sorted(set(r for r in res_list if r >= 100), reverse=True)
    
    # Time allocation: give more time to larger n (higher quality starts)
    # but also try smaller n for potential lower C5
    n_res = len(res_list)
    # Allocate time: 40% to n0, split rest among others
    time_alloc = [(budget_s * 0.38) if r == n0 else (budget_s * 0.18) for r in res_list]
    # Normalize
    total_t = sum(time_alloc)
    time_alloc = [t * (budget_s - 60) / total_t for t in time_alloc]

    for ri, res in enumerate(res_list):
        elapsed = time.time() - t0
        if elapsed > budget_s - 50:
            break
        phase_budget = min(time_alloc[ri], budget_s - 40 - (time.time() - t0))
        if phase_budget < 30:
            break

        # Get initial for this resolution
        if res == n0:
            h_cand = h0.copy()
        else:
            h_cand = _downsample(h0, res)

        # Run CD on this resolution
        dl = time.time() + phase_budget * 0.85
        h_opt, c5_opt = _cd_optimize(h_cand.copy(), 200000, rng, dl, top_m=3, explore_mag=0.04)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # Second pass with different params on best so far
        elapsed = time.time() - t0
        if elapsed < budget_s - 40:
            dl2 = time.time() + min(phase_budget * 0.4, budget_s - 40 - elapsed)
            h_opt2, c5_opt2 = _cd_optimize(best_h.copy(), 100000, rng, dl2, top_m=2, explore_mag=0.02)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    # Final targeted polish
    elapsed = time.time() - t0
    if elapsed < budget_s - 15:
        dl_pol = time.time() + min(20, budget_s - 12 - elapsed)
        h_pol, c5_pol = _fine_polish(best_h.copy(), rng, dl_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final repair
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
