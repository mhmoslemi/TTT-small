At the optimum, we expect the top lags to be approximately equal (otherwise we could redistribute). So the algorithm should:
1. Compute current correlations
2. Identify the top few lags
3. Try to REDUCE the maximum by equalizing the top lags

So we want to move h in a direction that decreases the max lag but doesn't increase other lags too much. This is essentially what the coordinate descent does.

∂c_k/∂h[i] = (1-h[j]) - h[i-s_k]

And ∂c_k/∂h[j] = (1-h[j+s_k]) - h[j-s_k] = (1-h[j+s_k]) - h[i]

So the change in c_k for h[i]+=d, h[j]-=d is:
d*(1-h[j]) - d*h[i-s_k] - d*(1-h[j+s_k]) + d*h[i]
= d*((1-h[j]) + h[i]) - d*(h[i-s_k] + (1-h[j+s_k]))

And the total second-order term: the h[i]*h[j] term in c_k changes by:
Original: h[i]*h[j] (if both contribute to c_k)
After: (h[i]+d)*(h[j]-d) = h[i]*h[j] + d*(h[j]-h[i]) - d²
Change: d*(h[j]-h[i]) - d²

But the linear approximation gives: d*(h[j]-h[i]) (from the partials). The d² term is second order and small for small d. Since we're doing golden section with small steps (bounded by h[i], 1-h[i], h[j], 1-h[j]), the error is at most d² which is tiny. So the linear approximation is excellent.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, n_steps, step0, t_end):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.88
    step = step0
    for i in range(n_steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        g = np.where(mask, g, 0.0)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.015, step0 * 12.0)
        else:
            step *= 0.45
            if step < 1e-13:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _smooth_obj(h, alpha)
    return h

def _coord_dcs(h, n, duration_s, rng, t_end):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    temp = 3e-5
    h_pad = np.zeros(2 * n)
    h_pad[:n] = h
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    gr_c = (np.sqrt(5) - 1) / 2
    while time.time() < min(end_t, t_end - 4):
        g = _smooth_grad(h, 5000)
        neg_g = np.maximum(-g, 0)
        pos_g = np.maximum(g, 0)
        tn = neg_g.sum()
        tp = pos_g.sum()
        n_pairs = 2500
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_end - 4):
                break
            if tn < 1e-12 or tp < 1e-12 or rng.random() < 0.15:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, max(3, n // 10))) % n
            else:
                i = rng.choice(n, p=neg_g / tn)
                j = rng.choice(n, p=pos_g / tp)
            if i == j:
                continue
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            col_i_p = h_pad[i - (n-1): i + n]
            col_i_n = col_i_p[::-1]
            col_j_p = h_pad[j - (n-1): j + n]
            col_j_n = col_j_p[::-1]
            delta = (col_j_p - col_i_p) - (col_i_n - col_j_n)
            x1 = d_hi - gr_c * (d_hi - d_lo)
            x2 = d_lo + gr_c * (d_hi - d_lo)
            f1 = np.max(c + x1 * delta)
            f2 = np.max(c + x2 * delta)
            for _ in range(28):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr_c * (d_hi - d_lo)
                    f1 = np.max(c + x1 * delta)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr_c * (d_hi - d_lo)
                    f2 = np.max(c + x2 * delta)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = np.max(c + d_opt * delta)
            accept = (f_opt < best_c5 + temp) or (f_opt < best_c5 * 1.0001)
            if accept:
                h[i] += d_opt
                h[j] -= d_opt
                h_pad[i] += d_opt
                h_pad[j] -= d_opt
                c = c + d_opt * delta
                if f_opt < best_c5:
                    best_c5 = f_opt
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        if no_improve > 12:
            scale = min(0.008, temp * 20)
            h = _proj_to_sum(h + scale * rng.randn(n), n * 0.5)
            h_pad[:n] = h
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            no_improve = 0
        temp = max(temp * 0.999, 5e-8)
        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj_to_sum(h, n * 0.5)
            h_pad[:n] = h
            c = np.correlate(h, 1.0 - h, mode='full') * dx
    return best_h, best_c5

    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    c = np.correlate(h, 1.0 - h, mode='full') * dx if False else _eval(h)
