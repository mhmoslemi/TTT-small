import numpy as np
import scipy.optimize as sop
import time


def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = 0.0, 1.0
    if s > target:
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
                lo = mid
            else:
                hi = mid
        return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h + mid, 0.0, 1.0).sum()) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)


def _snap_sum(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-9:
        return h
    h = _repair(h, n)
    diff = target - float(h.sum())
    if abs(diff) > 1e-11:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size:
            i = int(idx[0])
            h[i] += diff
            h[i] = min(1.0, max(0.0, float(h[i])))
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _grad_lag(h, sh):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += dx * (1.0 - h[sh:])
        g[sh:] -= dx * h[:m]
    else:
        s = -sh
        m = n - s
        g[s:] += dx * (1.0 - h[:-s])
        g[:-s] -= dx * h[s:]
    return g


def _feasible_dir(d, h):
    for _ in range(4):
        d[((h <= 1e-12) & (d > 0.0)) | ((h >= 1.0 - 1e-12) & (d < 0.0))] = 0.0
        d = d - float(d.mean())
    return d


def _soft_obj(x, alpha, pen):
    c = _cvals(x)
    m = float(c.max())
    if alpha <= 0.0:
        f = m
    else:
        s = alpha * (c - m)
        s = np.clip(s, -700.0, 0.0)
        f = m + float(np.log(np.sum(np.exp(s)))) / alpha
    if pen:
        f += float(pen) * (float(np.mean(x)) - 0.5) ** 2
    return float(f)


def _soft_grad(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = float(c.max())
    if alpha <= 0.0:
        w = np.zeros(len(c))
        imax = int(np.argmax(c))
        w[imax] = 1.0
    else:
        s = alpha * (c - m)
        s = np.clip(s, -700.0, 0.0)
        w = np.exp(s)
        w /= w.sum()
    g = np.zeros(n)
    nz = np.nonzero(w > 1e-13)[0]
    for k in nz:
        wk = float(w[k])
        sh = (n - 1) - int(k)
        if sh >= 0:
            m = n - sh
            g[:m] += wk * dx * (1.0 - x[sh:])
            g[sh:] -= wk * dx * x[:m]
        else:
            sp = -sh
            m = n - sp
            g[sp:] += wk * dx * (1.0 - x[:-sp])
            g[:-sp] -= wk * dx * x[sp:]
    if pen:
        g += 2.0 * float(pen) * (float(np.mean(x)) - 0.5)
    return g


def _lbfgs_soft(h, alpha, pen, maxit):
    n = len(h)
    maxit = int(maxit)
    if maxit <= 0:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = sop.minimize(
            _soft_obj, h, jac=_soft_grad, args=(float(alpha), float(pen)),
            method="L-BFGS-B", bounds=bnds,
            options={"maxiter": maxit, "ftol": 1e-16, "gtol": 1e-12, "maxls": 40}
        )
        x = res.x if res.x is not None else h
        return _repair(np.asarray(x, dtype=float), n)
    except Exception:
        return h


def _lambda_fw(G, iters=70):
    m = G.shape[0]
    lam = np.full(m, 1.0 / m)
    if m == 1:
        return lam
    r = G.T.dot(lam)
    for t in range(1, iters + 1):
        grad = 2.0 * (G.dot(r))
        s = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        lam *= (1.0 - gamma)
        lam[s] += gamma
        r = G.T.dot(lam)
    ss = float(lam.sum())
    if ss > 0.0:
        lam /= ss
    return lam


def _equalize(h, end_wall, tols=(3e-4, 1e-4, 3e-5, 1e-5, 3e-6, 1e-6)):
    n = len(h)
    best = float(_eval(h))
    for tol in tols:
        idle = 0
        it = 0
        while idle < 18:
            if time.time() >= end_wall:

            it += 1
            c = _cvals(h)
            m = float(c.max())
            if m < best - 1e-15:
                best = m
            elif m > best + 1e-15:
                best = m

            order = np.argsort(c)[::-1][:40]
            act = []
            for kk in order:
                if float(c[kk]) >= m - tol:
                    act.append(int(kk))
            if not act:
                act.append(int(order[0]))

            mact = len(act)
            G = np.empty((mact, n))
            ii = 0
            for kk in act:
                G[ii] = _grad_lag(h, (n - 1) - kk)
                ii += 1

            if it % 3 == 0:
                lam = np.full(mact, 1.0 / mact)
            else:
                lam = _lambda_fw(G, 70)

            d = G.T.dot(lam)
            d = _feasible_dir(d, h)
            maxd = float(np.max(np.abs(d)))
            if maxd < 1e-9:
                idle += 1
                continue

            base = 0.05 / maxd
            quick_best = best
            quick_h = None
            target_sum = 0.5 * n
            for fac in (0.5, 0.15, 0.05, 0.015, 0.005, 0.0015, 0.0005):
                hq = np.clip(h - (base * fac) * d, 0.0, 1.0)

            else:

    return h, best


def _down(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()


def _coarse_trial(h, f, end_wall, rng):

    if rc < 24:
