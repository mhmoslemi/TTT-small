                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, n, k, temp):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    idx = np.argpartition(c, -k)[-k:]
    ct = c[idx]
    sc = (ct - ct.max()) / temp
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(k):
        kx = idx[i]
        sh = (n - 1) - kx
        wk = w[i] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g, c.max() * dx

def _grad_desc(h, n, max_iter, k_lags, temp, steps, check_fn):
    c5 = _eval(h)
    stall = 0
    for it in range(max_iter):
        if check_fn():
            break
        g, _ = _topk_grad(h, n, k_lags, temp)
        improved = False
        for s in steps:
            h_new = _project(h - s * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h = h_new
                c5 = c5n
                improved = True
                steps = [s * 1.5] + [x for x in steps if x < s * 0.5]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [x * 0.4 for x in steps]
            stall += 1
            if stall > 10:
                break
            h_new = _project(h + 0.005 * np.random.randn(n), n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5:
                h = h_new
                c5 = c5n
                stall = 0
        else:
            stall = 0
    return h, c5

def _sim_anneal(h, n, max_iter, T_init, T_final, k_lags, temp, check_fn):
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    n_imp = 0
    for it in range(max_iter):
        if check_fn():
            break
        T = T_init * (T_final / T_init) ** (it / max(1, max_iter - 1))
        g, _ = _topk_grad(h, n, k_lags, temp)
        alpha = 0.05 * (1.0 - 0.7 * it / max(1, max_iter))
        noise = 0.003 * (1.0 + 0.5 * it / max(1, max_iter)) * np.random.randn(n)
        h_new = _project(h - alpha * g + noise, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5 or np.random.rand() < np.exp(-(c5n - c5) / max(T, 1e-18)):
            h = h_new
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                n_imp += 1
    return best_h, best_c5

def _double_res(h, n, budget_frac, check_fn):
    n2 = n * 2
    h2 = np.repeat(h, 2).astype(float)
    h2 = _repair(h2, n2)
    c5_2 = _eval(h2)
    h2, c5_2 = _grad_desc(h2, n2, 300, 30, 0.005,
                          [0.04, 0.02, 0.01, 0.005, 0.002, 0.001], check_fn)
    # Downsample by averaging pairs
    h_ds = (h2[0::2] + h2[1::2]) / 2.0
    h_ds = _repair(h_ds, n)
    c5_ds = _eval(h_ds)
    return h_ds, c5_ds, h2, c5_2

def _local_search(h, n, ntry, rng, c5c, check_fn, scale=1.0):
    best_h = h.copy()
    best_c = c5c
    base = h.copy()
    for _ in range(ntry):
        if check_fn():
            break
        trial = base.copy()
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 3e-5 * scale * rng.randn()
        ni, nj = trial[i] + d, trial[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            trial[i], trial[j] = ni, nj
            c5n = _eval(trial)
            if c5n < best_c - 1e-18:
                base = trial.copy()
                best_c = c5n
                best_h = trial.copy()
                if scale < 5.0:
                    scale *= 1.1
        else:
            scale = max(0.3, scale * 0.9)
    return best_h, best_c

def _smooth_max_obj(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 30000.0
    sc = np.clip(a * (c - m), -200, 0)
    return m + np.log(np.sum(np.exp(sc))) / a

def _smooth_max_grad(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 30000.0
    sc = np.clip(a * (c - m), -200, 0)
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def timeup():
        return elapsed() > budget_s - 20

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Wide-window gradient descent (top-40 lags, softmax)
    for temp in [0.003, 0.005, 0.008, 0.004, 0.002]:
        if timeup():
            break
        h2, c52 = _grad_desc(best_h.copy(), n, 800, 40, temp,
                             [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001], timeup)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Simulated annealing for global exploration
    if elapsed() < budget_s - 300:
        h2, c52 = _sim_anneal(best_h.copy(), n, 5000, 0.002, 1e-6, 30, 0.005, timeup)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: Resolution doubling perturbation
    if elapsed() < budget_s - 200:
        h_ds, c5_ds, h_hi, c5_hi = _double_res(best_h, n, 0.15, timeup)
        if c5_ds < best_c5:
            best_c5 = c5_ds
            best_h = h_ds
        # Also try optimizing at high res then projecting
        if elapsed() < budget_s - 100:
            h_hi2, c5_hi2 = _grad_desc(h_hi, n * 2, 200, 25, 0.004,
                                       [0.03, 0.015, 0.008, 0.004, 0.002], timeup)
            h_hp = (h_hi2[0::2] + h_hi2[1::2]) / 2.0
            h_hp = _repair(h_hp, n)
            c5_hp = _eval(h_hp)
            if c5_hp < best_c5:
                best_c5 = c5_hp
                best_h = h_hp

    # Phase 4: SLSQP polish with smooth max
    if elapsed() < budget_s - 120:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        for attempt in range(3):
            if timeup():
                break
            try:
                res = minimize(_smooth_max_obj, best_h, args=(n,), jac=_smooth_max_grad,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 5: More annealing rounds with different params
    if elapsed() < budget_s - 100:
        for T_mid in [0.001, 0.0005]:
            if timeup():
                break
            perturbed = best_h + 0.004 * rng.randn(n)
            perturbed = _repair(perturbed, n)
            h2, c52 = _sim_anneal(perturbed, n, 3000, T_mid, 1e-7, 25, 0.003, timeup)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2

    # Phase 6: Fine gradient polish
    for temp in [0.001, 0.0005, 0.002]:
        if timeup():
            break
        h2, c52 = _grad_desc(best_h.copy(), n, 500, 30, temp,
                             [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4], timeup)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: Local search
    if elapsed() < budget_s - 10:
        remaining = budget_s - 10 - elapsed()
        ntry = min(int(remaining / 0.001), 300000)
        best_h, best_c5 = _local_search(best_h, n, ntry, rng, best_c5, timeup, scale=1.0)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
