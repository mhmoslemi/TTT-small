                bh = hc.copy()
            else:
                hc[j1], hc[j2], hc[j3] = a-d1, b-d2, c+d1+d2
        if i % 3500 == 0:
            scale *= 0.88
    return bh, bc2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def el():
        return time.time() - t0
    def bc():
        return el() > budget_s - 8

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS with growing p and perturbations between runs
    schedule = [(32, 1e4, 450), (64, 1e5, 450), (128, 1e6, 400),
                (256, 1e7, 350), (512, 1e8, 300)]
    for idx, (p, pen, mi) in enumerate(schedule):
        if el() > 120 or el() > budget_s - 350:
            break
        if idx == 0:
            h_curr = best_h.copy()
        else:
            h_curr = best_h + 0.008*rng.randn(n0)
            h_curr = _repair(h_curr, n0)
        ht = _lbfgs(h_curr, p, pen, mi)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 2: Subgradient descent with top-k (main workhorse)
    if el() < 200:
        ht, ct = _subgrad(best_h, 50000, el, bc, beta=0.88, step0=0.006, kstart=5)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 3: L-BFGS high-p polish
    if el() < budget_s - 280:
        ht = _lbfgs(best_h.copy(), 512, 1e8, 400)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 4: Second subgradient pass (finer)
    if el() < budget_s - 200:
        ht, ct = _subgrad(best_h, 35000, el, bc, beta=0.92, step0=0.003, kstart=2, kmax=15)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 5: Simulated annealing local search
    if el() < budget_s - 80:
        nt = min(55000, int((budget_s - 80 - el()) / 0.0045))
        best_h, best_c5 = _sa_search(best_h, nt, rng, el, bc)

    # Phase 6: 3-point local search
    if el() < budget_s - 30:
        nt = min(25000, int((budget_s - 30 - el()) / 0.006))
        best_h, best_c5 = _3pt_search(best_h, nt, rng, el, bc)

    # Phase 7: Final subgradient micro-polish
    if el() < budget_s - 4:
        ht, ct = _subgrad(best_h, 8000, el, bc, beta=0.9, step0=0.001, kstart=1, kmax=3)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
