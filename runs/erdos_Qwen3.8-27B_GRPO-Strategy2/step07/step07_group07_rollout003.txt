    c = np.correlate(h, 1.0-h, mode='full') * dx

    constraint = (h.sum() - n*0.5)**2

def _grad_softmax(h, beta, penalty):

    c_max = c.max()
    weights = np.exp(beta * (c - c_max))

    # Phase 3: L-BFGS very high beta
    for b in [20000, 50000]:
        if el() > budget_s - 250:
            break
        h_try = _lbfgs_run(best_h.copy(), b, 1e8, 600)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 4: Fine Nesterov (small k, small steps)
    if el() < budget_s - 200:
        h_try, c_try = _nesterov_bt(best_h, 15000, 2, 25, 0.004, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 5: L-BFGS polish
    if el() < budget_s - 120:
        h_try = _lbfgs_run(best_h.copy(), 30000, 1e9, 400)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 6: 3-point local search
    if el() < budget_s - 60:
        nt = min(int((budget_s - 60 - el()) / 0.005), 60000)
        best_h, best_c5 = _greedy_h3(best_h, nt, rng, bc)

    # Phase 7: 2-point local search (finer)
    if el() < budget_s - 15:
        nt = min(int((budget_s - 15 - el()) / 0.002), 100000)
        best_h, best_c5 = _greedy_hj(best_h, nt, rng, bc)

    # Phase 8: Final L-BFGS
    if el() < budget_s - 3:
        h_try = _lbfgs_run(best_h.copy(), 50000, 1e9, 300)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
