import numpy as np
import time
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

```python
def _softmax_obj_grad(x, M, n, lam):
    h = np.clip(x, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    Mc = M * c - M * c.max()
    exp_c = np.exp(Mc)
    w = exp_c / exp_c.sum()
    f = (c.max() + np.log(exp_c.sum())) / M
    # Add penalty for sum constraint
    s = h.sum()
    f += lam * (s - n / 2.0) ** 2
    
    # Gradient of soft-max part
    g_sm = np.zeros(n)
    u = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        # First term: (1-h[p+sh]) for valid p
        if sh >= 0:
            p_idx = np.arange(0, n - sh)
            g_sm[p_idx] += wk * u[p_idx + sh]
            if sh > 0:
                p_idx2 = np.arange(sh, n)
                g_sm[p_idx2] -= wk * h[p_idx2 - sh]
        else:
            sp = -sh
            p_idx = np.arange(sp, n)
            g_sm[p_idx] += wk * u[p_idx - sp]
            p_idx2 = np.arange(0, n - sp)
            g_sm[p_idx2] -= wk * h[p_idx2 + sp]
    g_sm *= (2.0 / n)
    
    # Gradient of penalty
    g_pen = 2.0 * lam * (s - n / 2.0)
    g = g_sm + g_pen
    return f, g
```

            h[i] -= d
            h[j] += d

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for idx in o:
                if d > 0:
                    a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
                else:
                    a = min(-d, h[idx]); h[idx] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)

def _gpos(h, p, n):
