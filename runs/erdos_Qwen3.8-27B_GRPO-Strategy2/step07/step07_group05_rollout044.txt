    # Phase 2: SLSQP with smooth max
    if time.time() < t_limit - 150:
        h2, c2 = _slsqp_solve(best_h, n0, t_limit - 60, rng)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Perturbation restarts + gradient
    for attempt in range(6):
        if time.time() > t_limit:
            break
        scale = 0.004 * (1.0 - 0.1 * attempt)
        hn = best_h + scale * rng.randn(n0)
        hn = _proj(hn, n0)
        e0, e1, s0 = 0.008 * (0.9 ** attempt), 0.001 * (0.9 ** attempt), 0.25
        h2, c2 = _gd_momentum(hn, n0, 1200, e0, e1, s0, 0.90, t0, t_limit)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 4: Final tight SLSQP
    if time.time() < t_limit - 80:
        h2, c2 = _slsqp_solve(best_h, n0, t_limit - 30, rng)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 5: Fine momentum gradient
    if time.time() < t_limit - 30:
        h2, c2 = _gd_momentum(best_h.copy(), n0, 2000, 0.001, 0.0003, 0.1, 0.93, t0, t_limit)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 6: Pair search polish
    if time.time() < t_limit - 5:
        best_h, best_c5 = _pair_search(best_h, n0, 3000, rng, t_limit)

    best_h = _proj(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
