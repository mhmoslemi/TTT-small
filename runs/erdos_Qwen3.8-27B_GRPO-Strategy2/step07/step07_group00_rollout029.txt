import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, m, n):
    shift = m - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        g[:n-shift] += (1.0 - h[shift:n])
        g[shift:] -= h[:n-shift]
    else:
        sp = -shift
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = alpha * (ct - m)
    s = np.clip(s, -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], n)
    return g

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)]*n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _lp_refine(h0, K, deadline, max_iters=15):
    n = len(h0)
    h = h0.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_all = _correl(h)
        top_idx = np.argpartition(c_all, -K)[-K:]
        c0 = c_all[top_idx].copy()
        
        hvar = cp.Variable(n)
        Mvar = cp.Variable()
        cons = [hvar >= 0, hvar <= 1, cp.sum(hvar) == n/2.0]
        
        for j in range(K):
            m = top_idx[j]
            g = _grad_lag(h, m, n)
            # c_m(h) ≈ c0[j] + g·(h - h0 cur)
            # constraint: c0[j] + g·(hvar - h) <= Mvar
            # => g·hvar <= Mvar - c0[j] + g·h
            cons.append(g @ hvar <= Mvar - c0[j] + g @ h)
        
        # Add small ball constraint for stability
        # ||hvar - h||_inf <= 0.05
        cons.append(hvar - h <= 0.05)
        cons.append(h - hvar <= 0.05)
        
        obj = cp.Minimize(Mvar)
        prob = cp.Problem(obj, cons)
        try:
            prob.solve(solver=cp.ECOS, max_iters=20000)
        except Exception:
            break
        
        if hvar.value is None:
            break
        h_new = np.clip(hvar.value, 0, 1)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        c_old = _eval(h)
        if c_new < c_old:
            h = h_new
        else:
            # Try half step
            h_half = _repair(0.5*h + 0.5*h_new, n)
            if _eval(h_half) < c_old:
                h = h_half
            else:
                break
    return h

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.4
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _targeted_polish(h, niter, rng, deadline):
    """Polish using gradient-directed moves on top lags"""
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    c_all = _correl(h)
    top_idx = np.argpartition(c_all, -5)[-5:]
    
    for it in range(niter):

            break
        # Re-compute top lags every 200 iterations
        if it % 200 == 0:

        # Pick a random top lag
        m = top_idx[rng.randint(len(top_idx))]
        g = _grad_lag(h, m, nc)
        
        # Find indices where gradient suggests most improvement
        # We want to decrease c_m, so move h in direction -g
        # But we need to preserve sum, so use pairwise swaps
        # Pick i where -g[i] is large (decreasing h[i] helps) and j where -g[j] is small
        pos = g < 0  # decreasing h[i] increases c_m... wait
        # Actually: to decrease c_m, we want ∂c_m/∂h[i] * Δh[i] < 0
        # So if g[i] > 0, we want Δh[i] < 0 (decrease h[i])
        # If g[i] < 0, we want Δh[i] > 0 (increase h[i])
        
        # Greedy: pick i with largest g[i] (decrease it) and j with smallest g[j] (increase it)
        # But to be stochastic, pick from top/bottom 10%
        n_sample = max(10, nc // 20)
        top_candidates = np.argsort(-g)[:n_sample]  # largest g
        bot_candidates = np.argsort(g)[:n_sample]   # smallest g
        
        i = top_candidates[rng.randint(n_sample)]
        j = bot_candidates[rng.randint(n_sample)]
        
        # Determine step size
        d = rng.uniform(1e-5, 5e-5)
        ni, nj = h[i] - d, h[j] + d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni + d, nj - d
    return bh, bc

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):

            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.001 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:

                bh = h.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, m, n):
    shift = m - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:n])
        g[shift:] -= h[:n - shift]
    else:
        sp = -shift
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-13:
            g += w[ii] * _grad_lag(h, idx[ii], n)
    return g

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _lp_refine(h0, K, deadline, max_iters=20):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = _eval(h)
    ball = 0.03
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_all = _correl(h)
        top_idx = np.argpartition(c_all, -K)[-K:]
        c0 = c_all[top_idx].copy()
        hvar = cp.Variable(n)
        Mvar = cp.Variable()
        cons = [hvar >= 0, hvar <= 1, cp.sum(hvar) == n / 2.0]
        for j in range(K):
            g = _grad_lag(h, top_idx[j], n)
            cons.append(g @ hvar <= Mvar - c0[j] + g @ h)
        cons.append(hvar - h <= ball)
        cons.append(h - hvar <= ball)
        prob = cp.Problem(cp.Minimize(Mvar), cons)
        try:
            prob.solve(solver=cp.ECOS, max_iters=30000, reltol=1e-10)
        except Exception:
            break
        if hvar.value is None:
            break
        h_new = _repair(np.clip(hvar.value, 0, 1), n)
        c_new = _eval(h_new)
        c_old = _eval(h)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
        elif c_new < c_old:
            h = h_new
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                hd = _repair((1 - frac) * h + frac * h_new, n)
                if _eval(hd) < c_old:
                    h = hd
                    improved = True
                    break
            if not improved:
                ball *= 0.7
                if ball < 0.001:
                    break
    return best_h

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.4
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _targeted_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    c_all = _correl(h)
    top_idx = np.argpartition(c_all, -5)[-5:]
    n_cand = max(15, nc // 25)
    for it in range(niter):
        if time.time() > deadline:
            break
        if it % 300 == 0:
            c_all = _correl(h)
            top_idx = np.argpartition(c_all, -5)[-5:]
        m = top_idx[rng.randint(len(top_idx))]
        g = _grad_lag(h, m, nc)
        top_c = np.argsort(-g)[:n_cand]
        bot_c = np.argsort(g)[:n_cand]
        i = top_c[rng.randint(n_cand)]
        j = bot_c[rng.randint(n_cand)]
        d = rng.uniform(2e-5, 8e-5)
        ni, nj = h[i] - d, h[j] + d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni + d, nj - d
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = 1e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.001 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc
