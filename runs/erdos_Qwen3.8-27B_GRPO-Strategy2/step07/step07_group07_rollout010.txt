    dx = 2.0 / n
    c = _raw_corr(h) * dx
    cs = c - c.max()
    return c.max() + np.log(np.sum(np.exp(T * cs))) / T + pen * (h.sum() - n * 0.5)**2

def _lse_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = _raw_corr(h) * dx
    cs = c - c.max()
    w = np.exp(T * cs)
    w /= w.sum()
    g = _grad_from_weights(h, w * dx)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _topk_obj(h, K, pen):
    n = len(h)
    dx = 2.0 / n
    c = _raw_corr(h) * dx
    idx = np.argpartition(c, -K)[-K:]
    return np.mean(c[idx]) + pen * (h.sum() - n * 0.5)**2

def _topk_grad(h, K, pen):
    n = len(h)
    dx = 2.0 / n
    c = _raw_corr(h) * dx
    idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / K
    g = _grad_from_weights(h, w * dx)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _weighted_topk_obj(h, K, weights, pen):
    n = len(h)
    dx = 2.0 / n
    c = _raw_corr(h) * dx
    idx = np.argpartition(c, -K)[-K:]
    val = np.dot(c[np.sort(idx)], weights) / np.sum(weights)
    return val + pen * (h.sum() - n * 0.5)**2

def _weighted_topk_grad(h, K, weights, pen):
    n = len(h)
    dx = 2.0 / n
    c = _raw_corr(h) * dx
    idx = np.argpartition(c, -K)[-K:]
    sw = np.sort(idx)
    w = np.zeros(2 * n - 1)
    w[sw] = weights / np.sum(weights)
    g = _grad_from_weights(h, w * dx)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, obj, grad, args, maxi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), jac=grad, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxi, 'ftol': 1e-18,
                       'gtol': 1e-12, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sinusoid_init(n, freq, phase, rng_state):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.28 * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    h = _proj_sum(h, n * 0.5)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def done():
        return elapsed() > budget_s - 20

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    pen = 1e6

    # Phase 1: LSE with increasing temperature (smooth -> sharp)
    for T in [15, 50, 200, 800]:
        if done():
            break
        h_try = _lbfgs(best_h, _lse_obj, _lse_grad, (T, pen), 350)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 2: Top-k optimization, multiple passes
    for K in [1, 2, 3, 2, 1]:
        if done():
            break
        h_try = _lbfgs(best_h, _topk_obj, _topk_grad, (K, pen), 300)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 3: Very high T LSE (near-exact max)
    for T in [3000, 10000]:
        if done():
            break
        h_try = _lbfgs(best_h, _lse_obj, _lse_grad, (T, pen), 250)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 4: Weighted top-k (emphasize the worst lag)
    w2 = np.array([0.7, 0.3])
    if not done():
        h_try = _lbfgs(best_h, _weighted_topk_obj, _weighted_topk_grad, (2, w2, pen), 250)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
    w3 = np.array([0.6, 0.25, 0.15])
    if not done():
        h_try = _lbfgs(best_h, _weighted_topk_obj, _weighted_topk_grad, (3, w3, pen), 250)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 5: Perturbation-based restart to escape local optima
    for scale in [0.03, 0.015, 0.008]:
        if done():
            break
        noise = rng.randn(n0) * scale
        h_pert = _repair(np.clip(best_h + noise, 0, 1), n0)
        if _eval(h_pert) < best_c5 + 0.003:
            h_try = _lbfgs(h_pert, _topk_obj, _topk_grad, (1, pen), 300)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 6: Alternative sinusoidal start + optimize
    if not done() and elapsed() < budget_s * 0.5:
        for freq in [3, 5, 7]:
            if done():
                break
            h_alt = _sinusoid_init(n0, freq, rng.rand() * 6.28, rng)
            h_alt = _lbfgs(h_alt, _lse_obj, _lse_grad, (200, pen), 200)
            h_alt = _lbfgs(h_alt, _topk_obj, _topk_grad, (1, pen), 200)
            c_alt = _eval(h_alt)
            if c_alt < best_c5:
                best_c5 = c_alt
                best_h = h_alt

    # Phase 7: Fine local search (adaptive scale pairwise moves)
    if not done():
        h_c = best_h.copy()
        scale = 1.2e-4
        for _ in range(40000):
            if done():
                break
            i = rng.randint(n0)
            j = (i + 1 + rng.randint(0, 5)) % n0
            d = rng.randn() * scale
            ni, nj = h_c[i] + d, h_c[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h_c[i], h_c[j] = ni, nj
                c_new = _eval(h_c)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_c.copy()
                else:
                    h_c[i], h_c[j] = ni - d, nj + d
            if _ % 8000 == 0:
                scale *= 0.7

    # Phase 8: Final top-1 polish
    if not done():
        h_try = _lbfgs(best_h, _topk_obj, _topk_grad, (1, pen), 400)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 9: Another LSE pass at very high T for final smoothing
    if not done():
        h_try = _lbfgs(best_h, _lse_obj, _lse_grad, (50000, pen), 200)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
