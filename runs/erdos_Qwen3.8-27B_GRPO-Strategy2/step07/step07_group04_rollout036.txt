import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    pen = _state['pen']
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)

    s = np.clip(alpha * (c - m), -500, 500)
    obj = m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    obj += pen * (x.sum() - n * 0.5) ** 2
    return obj

def _grad_fn(x):

    alpha = _state['alpha']
    pen = _state['pen']
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()

    return g

    _state['alpha'] = alpha
    _state['pen'] = 0

def _nesterov_topk(h, k_top, steps, step0, t0, time_limit):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _s in range(steps):
        if time.time() - t0 > time_limit:
            break
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = dx / k_top
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g

            f = f_new
