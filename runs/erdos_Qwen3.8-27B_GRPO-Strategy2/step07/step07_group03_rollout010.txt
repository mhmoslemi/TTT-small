        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(hp - lr * v, n * 0.5)
        fn, _ = _obj_grad(h_new, alpha)
        if fn < f0:
            h, f0, lr = h_new, fn, min(lr * 1.03, lr0 * 10)
        else:
            lr *= 0.5
            if lr < 1e-11:
                break
            h_new = _proj_to_sum(hp - lr * v, n * 0.5)
            f0, _ = _obj_grad(h_new, alpha)
            h = h_new
    return h

def _adam(h, t_budget, t_end, alpha_init=5000):
    n = len(h)
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.003
    it = 0
    alpha = alpha_init
    no_imp = 0
    best_f = _eval(h)
    while time.time() < min(t0_global + t_budget, t_end - 3):
        if time.time() > t_end:
            break
        it += 1
        f, g = _obj_grad(h, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1 ** it)
        vhat = v / (1 - b2 ** it)
        h_new = _proj_to_sum(h - lr * mhat / (np.sqrt(vhat) + eps), n * 0.5)
        cn = _eval(h_new)
        if cn < best_f:
            best_f = cn
            no_imp = 0
            alpha = min(alpha * 1.5, 20000000)
            if cn < best_f * 0.9999 and it > 50:
                lr = lr * 1.1
        else:
            no_imp += 1
            if no_imp > 80:
                h_new = _proj_to_sum(h + 1e-4 * np.random.RandomState(it).randn(n), n * 0.5)
                alpha = max(alpha * 0.5, 3000)
                no_imp = 0
        h = h_new
    return h, best_f

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _coord_descent(h, T, t_budget, t_end, rng, wide=False):
    n = len(h)
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = min(time.time() + t_budget, t_end - 5)
    max_dist = n // 3 if wide else n // 8
    no_improve = 0
    while time.time() < end_t:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        sweep_improved = False
        n_pairs = n * 3
        for _ in range(n_pairs):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(3, max_dist))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(20):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 60:
            h = _proj_to_sum(h + 1e-5 * rng.randn(n), n * 0.5)
            no_improve = 0
        h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _local_search(h, n_iter, t_end, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

t0_global = 0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0_global = time.time()
    t_end = t0_global + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov warmup
    for alpha in [3000, 15000, 80000, 300000]:
        t_e = min(t0_global + 12, t0_global + budget_s * 0.08)
        h = _nesterov(h, alpha, 300, 0.004, t_e)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Adam optimization
    t_e2 = t0_global + budget_s * 0.22
    h, c5v = _adam(best_h.copy(), budget_s * 0.12, t_e2, alpha_init=10000)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 3: Coordinate descent with T=10, nearby pairs
    t_e3 = t0_global + budget_s * 0.45
    h, c5v = _coord_descent(best_h.copy(), 10, budget_s * 0.18, t_e3, rng, wide=False)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 4: Coordinate descent with T=15, wide pairs
    t_e4 = t0_global + budget_s * 0.62
    h, c5v = _coord_descent(best_h.copy(), 15, budget_s * 0.14, t_e4, rng, wide=True)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 5: Adam with higher alpha
    if time.time() < t0_global + budget_s * 0.75:
        t_e5 = t0_global + budget_s * 0.75
        h, c5v = _adam(best_h.copy(), budget_s * 0.08, t_e5, alpha_init=200000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Restart with perturbation
    if time.time() < t0_global + budget_s * 0.82:
        t_e6 = t0_global + budget_s * 0.82
        h2 = _proj_to_sum(best_h + 0.01 * rng.randn(n), n * 0.5)
        h2 = _nesterov(h2, 100000, 200, 0.003, t_e6)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 7: Final coordinate descent T=12
    if time.time() < t0_global + budget_s * 0.92:
        t_e7 = t0_global + budget_s * 0.93
        h, c5v = _coord_descent(best_h.copy(), 12, budget_s * 0.06, t_e7, rng, wide=False)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 8: Local search polish
    if time.time() < t_end - 5:
        remaining = (t_end - time.time()) - 3
        n_tries = min(int(remaining / 0.004), 60000)
        if n_tries > 200:
            best_h, best_c5 = _local_search(best_h.copy(), n_tries, t_end - 2, rng)

    # Final fix
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
