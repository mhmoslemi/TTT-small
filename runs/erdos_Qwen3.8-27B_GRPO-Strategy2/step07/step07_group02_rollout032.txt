import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, s, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if s == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif s > 0:
        m = n - s
        g[:m] += (1.0 - h[s:])
        g[s:s+m] -= h[:m]
        g *= dx
    else:
        t = -s
        m = n - t
        g[t:t+m] += (1.0 - h[:m])
        g[:m] -= h[t:t+m]
        g *= dx
    return g

def _fw_step(h, g, n):
    """Frank-Wolfe: find vertex minimizing g·s, then line search."""
    # Vertex: s[i]=1 for the n//2 indices with smallest g[i]
    idx = np.argsort(g)[:n // 2]
    s = np.zeros(n)
    idx = np.sort(idx)  # for deterministic behavior
    s[idx] = 1.0
    # Line search
    best_f = _eval(h)
    best_gamma = 0.0
    for gamma in [2.0/3, 0.5, 0.3, 0.2, 0.7, 0.9, 0.1, 0.4]:
        h_new = (1 - gamma) * h + gamma * s
        h_new = _repair(h_new, n)
        f_new = _eval(h_new)
        if f_new < best_f - 1e-15:
            best_f = f_new
            best_gamma = gamma
    if best_gamma > 0:
        h = (1 - best_gamma) * h + best_gamma * s
        h = _repair(h, n)
    return h, best_f

def _proj_subgrad(h, g, n, max_steps=20, deadline=None):
    """Projected subgradient with backtracking."""
    dx = 2.0 / n
    f_cur = _eval(h)
    step = 0.1
    for _ in range(max_steps):
        if deadline and time.time() > deadline - 10:
            break
        h_new = _repair(h - step * g, n)
        f_new = _eval(h_new)
        if f_new < f_cur - 1e-14:
            h = h_new
            f_cur = f_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.5
            if step < 1e-8:
                break
    return h, f_cur

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, s, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if s == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif s > 0:
        m = n - s
        g[:m] += (1.0 - h[s:])
        g[s:s + m] -= h[:m]
        g *= dx
    else:
        t = -s
        m = n - t
        g[t:] += (1.0 - h[:m])
        g[:m] -= h[t:]
        g *= dx
    return g

def _topk_grad(x, k, alpha):
    n = len(x)
    c = _cvals(x)
    top_idx = np.argsort(c)[-k:]
    top_c = c[top_idx]
    e = np.exp(alpha * (top_c - top_c.max()))
    w = e / e.sum()
    g = np.zeros(n)
    for i, ki in enumerate(top_idx):
        s_val = ki - (n - 1)
        g += w[i] * _grad_lag(x, s_val, n)
    return g

def _topk_obj(x, k, alpha):
    n = len(x)
    c = _cvals(x)
    top_idx = np.argsort(c)[-k:]
    top_c = c[top_idx]
    m = top_c.max()
    logsum = m + np.log(np.sum(np.exp(alpha * (top_c - m)))) / alpha
    rest = c.copy()
    rest[top_idx] = -np.inf
    r = rest.max()
    overall = max(logsum, m + np.log(np.exp(alpha * (r - m))) / alpha) if r > m else logsum
    return overall

def _fw_run(h, n, iters, deadline):
    f_cur = _eval(h)
    for i in range(iters):
        if time.time() > deadline - 15:
            break
        c = _cvals(h)
        k_max = np.argmax(c)
        g = _grad_lag(h, k_max - (n - 1), n)
        idx = np.argsort(g)[:n // 2]
        s = np.zeros(n)
        s[idx] = 1.0
        best_f = f_cur
        best_gamma = 0.0
        for gamma in [0.35, 0.5, 0.2, 0.6, 0.15, 0.75, 0.8, 0.1]:
            h_new = _repair((1 - gamma) * h + gamma * s, n)
            f_new = _eval(h_new)
            if f_new < best_f - 1e-15:
                best_f = f_new
                best_gamma = gamma
                break
        if best_gamma == 0:
            break
        h = _repair((1 - best_gamma) * h + best_gamma * s, n)
        f_cur = best_f
    return h, f_cur

def _lbfgs_topk(h, k, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_topk_obj, h, jac=_topk_grad, args=(k, alpha),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 40, 'maxfun': maxit * 3})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_run(h, n, iters, k, step0, deadline):

                break
            step *= 0.4
            if step < 1e-10:

        if not improved:

    return h, f_cur
