def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:

    H = np.fft.rfft(h, N)

The _delta_corr function:
```python
def _delta_corr(h, i, j, d, n):
    """Compute Δout[k] (before multiplying by dx) for 2-point change h[i]+=d, h[j]-=d"""
    dx = 2.0/n
    dc = np.zeros(2*n-1)
    for k in range(2*n-1):
        shift = k - (n-1)
        contrib_i = 0.0
        contrib_j = 0.0
        if shift >= 0:
            if shift <= i < n:
                contrib_i = (1.0 - h[i-shift])
            if shift <= j < n:
                contrib_j = (1.0 - h[j-shift])
        else:
            s = -shift
            if s <= i < n:
                contrib_i = (1.0 - h[i+s])
            if s <= j < n:
                contrib_j = (1.0 - h[j+s])
        dc[k] = (contrib_i - contrib_j) * d * dx
    return dc
```

def _delta_c(h, i, j, d, n):
    dx = 2.0 / n
    # Precompute shift array
    # For index m, affected k: from m to min(m+n-1, 2n-2)
    # Actually: k from max(0, m-(n-1)) to min(m+(n-1), 2n-2)
    # Simplifies to: k from m to m+n-1 (since m < n, m-(n-1) < 0, and m+n-1 < 2n-1)

    full_s = full_k - (n-1)  # shifts, range -(n-1) to n-1

        c = np.zeros(2*n-1)
        # s >= 0 part
        pmask = full_s >= 0
        sp = full_s[pmask]
        idx = m + sp  # for c1: need m+sp < n
        valid1 = idx < n
        vals1 = np.zeros_like(sp, dtype=float)

        vals2 = np.zeros_like(sp, dtype=float)

        idx = m - sn  # for c1: need m >= sn (m-sn >= 0) and m-sn < n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _ss_obj(x, alpha):

    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()

def _ss_grad(x, alpha):
