def _proj(h, target):
    """Project onto {0<=x<=1, sum(x)=target}"""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_exact(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _nesterov(h, mode, steps, step0, deadline, **kw):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    beta = kw.get('beta', 0.85)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        if mode == 'exact':
            g, _ = _grad_exact(h, kw.get('tol', 0.003))
        else:
            g, _ = _grad_smax(h, kw.get('alpha', 30000), kw.get('topk', 10))
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.015, step0 * 15.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _slsqp_refine(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_smax(x, alpha, topk)[0],
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c, h[i], h[j] = c5n, ni, nj
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, step0, tol, deadline):
    n = len(h)
    c5c = _eval(h)
    d = 0.001 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _nesterov(hn, 'exact', 500, step0, deadline, tol=tol, beta=0.9)
    if c5n < c5c:
        return hn2, c5n
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h)

    # Phase 1: SLSQP with soft-max (quick local refinement)
    dl1 = min(deadline, t0 + 120)
    for alpha, tk, mi in [(8000, 30, 400), (30000, 15, 350)]:
        if time.time() > dl1:
            break
        h = _slsqp_refine(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Nesterov with exact subgradient (multiple configs)
    dl2 = min(deadline, t0 + 400)
    configs = [
        (0.004, 0.005, 0.85, 4000),
        (0.002, 0.002, 0.88, 4000),
        (0.001, 0.001, 0.90, 4000),
    ]
    for step0, tol, mom, ns in configs:
        if time.time() > dl2:
            break
        h2, c5v = _nesterov(best_h.copy(), 'exact', ns, step0, dl2, tol=tol, beta=mom)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Soft-max Nesterov (smooth, different basin)
    dl3 = min(deadline, t0 + 580)
    for alpha, topk, step0, ns in [(50000, 8, 0.002, 3000), (200000, 5, 0.001, 3000)]:
        if time.time() > dl3:
            break
        h2, c5v = _nesterov(best_h.copy(), 'smax', ns, step0, dl3, alpha=alpha, topk=topk, beta=0.88)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Jitter + re-optimize (escape local min)
    dl4 = min(deadline, t0 + 720)
    for _ in range(5):
        if time.time() > dl4:
            break
        hj, cj = _jitter_reopt(best_h.copy(), rng, 0.002, 0.002, dl4)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 5: Block annealing
    dl5 = min(deadline, t0 + 880)
    blocks = [2, 3, 4, 5, 7, 10]
    best_h, best_c5 = _anneal(best_h.copy(), 60000, rng, dl5, blocks, 0.003, 1e-10)

    # Phase 6: Final Nesterov (very fine)
    dl6 = min(deadline, t0 + budget_s - 40)
    h2, c5v = _nesterov(best_h.copy(), 'exact', 5000, 0.0005, dl6, tol=0.0005, beta=0.92)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Fine polish
    dl7 = deadline
    rem = max(0, dl7 - time.time())
    n_pol = min(int(rem / 0.0015), 50000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, dl7, 1e-6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
