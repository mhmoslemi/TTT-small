from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    return _proj_sum(h, target)

def _softplus_obj(h, T, penalty):
    n = len(h)
    c = _corrs(h)
    # log-sum-exp for numerical stability
    c_max = c.max()
    log_sum = c_max + np.log(np.exp(T * (c - c_max)).sum()) / T
    return log_sum + penalty * (h.sum() - n * 0.5) ** 2

def _softplus_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h)
    # softmax weights
    c_shifted = T * (c - c.max())
    w = np.exp(c_shifted) / np.exp(c_shifted).sum()
    
    # Gradient: weighted sum of per-lag gradients
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx] * dx
        if wk < 1e-15:
            continue
        sh = idx - (n - 1)  # shift: negative = h shifted left, positive = h shifted right
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            # corr = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])
            g[:n-sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            # corr = sum_{i=0}^{n-1-sp} h[i+sp]*(1-h[i])
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    
    g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

    dx = 2.0/n

            g[:m] -= wk*h[s:]

def _subgrad(h, n_iter, eta0, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    bc = _eval(hc)
    bh = hc.copy()
    eta = eta0
    momentum = np.zeros(n)
    for it in range(n_iter):
        if bc_fn(): break
        c = _corrs(hc)
        # Use top-3 lags for smoothed gradient
        top3 = np.argsort(c)[-3:]
        T_sm = 50.0
        ca = c[top3]
        cm = np.max(ca)
        e = np.exp(T_sm*(ca-cm))
        wa = e/np.sum(e)
        dx = 2.0/n
        g = np.zeros(n)
        for i, k in enumerate(top3):
            wk = wa[i]*dx
            sh = k-(n-1)
            if sh == 0:
                g += wk*(1.0-2.0*hc)
            elif sh > 0:
                m = n-sh
                g[:m] += wk*(1.0-hc[sh:])
                g[sh:] -= wk*hc[:m]
            else:
                s = -sh; m = n-s
                g[s:] += wk*(1.0-hc[:m])
                g[:m] -= wk*hc[s:]
        gn = np.linalg.norm(g)
        if gn < 1e-16: break
        momentum = 0.8*momentum + 0.2*(g/gn)
        step = eta
        hc_new = _proj_sum(hc - step*momentum, n*0.5)
        cn = _eval(hc_new)
        if cn < bc:
            bc = cn; bh = hc_new.copy(); hc = hc_new
            step = min(step*1.01, 0.03)
        else:
            step *= 0.6
            if step < 1e-10: break
    return bh, bc

            else:

        if it % 5000 == 4999:

    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h):
    return _proj_sum(np.clip(h, 0, 1), len(h) * 0.5)

def _se_obj(h, T, pen):
    c = _corrs(h)
    cm = np.max(c)
    return float(cm + np.log(np.sum(np.exp(T * (c - cm)))) / T + pen * (np.sum(h) - len(h) * 0.5) ** 2)

def _se_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h)
    cm = np.max(c)
    e = np.exp(T * (c - cm))
    w = e / np.sum(e)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            m = n - sh
            g[:m] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:m]
        else:
            s = -sh
            m = n - s
            g[s:] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[s:]
    g += 2.0 * pen * (np.sum(h) - n * 0.5)
    return g

def _lbfgs_full(h0, T, pen, maxiter):
    bnds = [(0.0, 1.0)] * len(h0)
    try:
        res = minimize(_se_obj, h0, jac=_se_grad, args=(T, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 80})
        return _repair(res.x)
    except Exception:
        return _repair(h0)

def _ase_obj(h, act, T, pen):
    c = _corrs(h)
    ca = c[act]
    cm = np.max(ca)
    return float(cm + np.log(np.sum(np.exp(T * (ca - cm)))) / T + pen * (np.sum(h) - len(h) * 0.5) ** 2)

def _ase_grad(h, act, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h)
    ca = c[act]
    cm = np.max(ca)
    e = np.exp(T * (ca - cm))
    wa = e / np.sum(e)
    g = np.zeros(n)
    for i, k in enumerate(act):
        wk = wa[i] * dx
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            m = n - sh
            g[:m] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:m]
        else:
            s = -sh
            m = n - s
            g[s:] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[s:]
    g += 2.0 * pen * (np.sum(h) - n * 0.5)
    return g

def _lbfgs_ase(h0, act, T, pen, maxiter):
    bnds = [(0.0, 1.0)] * len(h0)
    try:
        res = minimize(_ase_obj, h0, jac=_ase_grad, args=(act, T, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 80})
        return _repair(res.x)
    except Exception:
        return _repair(h0)

def _coord_sweep(hc, n, bc_fn):
    dx = 2.0 / n
    center = n - 1
    for j in range(n):
        if bc_fn():
            return hc, False
        hc[j] = 0.0
        c_lo = np.correlate(hc, 1.0 - hc, mode='full') * dx
        hc[j] = 1.0
        c_hi = np.correlate(hc, 1.0 - hc, mode='full') * dx
        slope = c_hi - c_lo
        base = c_lo[center]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(45):
            m1 = lo_t + (hi_t - lo_t) / 3.0
            m2 = hi_t - (hi_t - lo_t) / 3.0
            f1 = np.max(c_lo + m1 * slope)
            q1 = base + m1 * (1.0 - m1)
            if q1 > f1:
                f1 = q1
            f2 = np.max(c_lo + m2 * slope)
            q2 = base + m2 * (1.0 - m2)
            if q2 > f2:
                f2 = q2
            if f1 < f2:
                hi_t = m2
            else:
                lo_t = m1
        hc[j] = (lo_t + hi_t) * 0.5
    return hc, True

def _subgrad(h, n_iter, eta0, n_active, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    bc = _eval(hc)
    bh = hc.copy()
    eta = eta0
    mom = np.zeros(n)
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(hc)
        top = np.argsort(c)[-n_active:]
        T_sm = 80.0
        ca = c[top]
        cm = np.max(ca)
        e = np.exp(T_sm * (ca - cm))
        wa = e / np.sum(e)
        dx = 2.0 / n
        g = np.zeros(n)
        for i, k in enumerate(top):
            wk = wa[i] * dx
            if wk < 1e-14:
                continue
            sh = k - (n - 1)
            if sh == 0:
                g += wk * (1.0 - 2.0 * hc)
            elif sh > 0:
                m = n - sh
                g[:m] += wk * (1.0 - hc[sh:])
                g[sh:] -= wk * hc[:m]
            else:
                s = -sh
                m = n - s
                g[s:] += wk * (1.0 - hc[:m])
                g[:m] -= wk * hc[s:]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        mom = 0.75 * mom + 0.25 * (g / gn)
        hc_new = _proj_sum(hc - eta * mom, n * 0.5)
        cn = _eval(hc_new)
        if cn < bc:
            bc = cn
            bh = hc_new.copy()
            hc = hc_new
            eta = min(eta * 1.015, 0.04)
        else:
            eta *= 0.65
            if eta < 1e-11:
                break
    return bh, bc

def _finetune(h, rng, eta0, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    bc = _eval(hc)
    bh = hc.copy()
    scale = 2e-5
    stall = 0
    it = 0
    while not bc_fn() and it < 200000:
        it += 1
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = rng.randn() * scale
        ni, nj = hc[i] + d, hc[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[i], hc[j] = ni, nj
            cn = _eval(hc)
            if cn < bc - 1e-18:
                bc = cn
                bh = hc.copy()
                stall = 0
                scale = min(scale * 1.005, 0.02)
            else:
                hc[i], hc[j] = ni - d, nj + d
                stall += 1
        if stall > 200:
            scale *= 0.5
            stall = 0
            if scale < 1e-9:
                break
    return bh, bc
