            k2 = n - 2
            gi = _grad_pos(h, i, n, shifts, n2)
            gj = _grad_pos(h, j, n, shifts, n2)
            g = gi - gj
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-15:
                continue
            for _ in range(20):
                dm = 0.5 * (d_lo + d_hi)
                v = c + dm * g
                c[k1] += dm * dm
                c[k2] += dm * dm
                if c.max() < c5 - 1e-15:
                    d_lo = dm
                else:
                    d_hi = dm
                c[k1] -= dm * dm
                c[k2] -= dm * dm
            d = 0.5 * (d_lo + d_hi)
            if abs(d) < 1e-13:
                continue
            h[i] += d
            h[j] -= d
            c[k1] += d * d
            c[k2] += d * d
            c5n = float(c.max())
            if c5n <= c5:
                c5 = c5n
                if c5 < best_c:
                    best_c = c5
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
                c[k1] -= d * d
                c[k2] -= d * d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    shifts = np.arange(-(n - 1), n)
    n2 = 2.0 / n
    rng = np.random.RandomState(seed)
    h = _proj(np.array(initial_h_values, dtype=float).copy(), n * 0.5)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    for sc in [0.04, 0.08, 0.15, 0.3, 0.5]:
        hp = _proj(np.clip(h + sc * rng.randn(n), 0, 1), n * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 1: main CD descent (40%)
    dl = t0 + budget_s * 0.40
    h1, c51 = _cd(best_h.copy(), 800000, rng, dl, T0=0, T1=0, shifts=shifts, n2=n2)
    if c51 < best_c5:
        best_c5, best_h = c51, h1

    # Phase 2: annealed CD (25%)
    dl = t0 + budget_s * 0.65
    rng2 = np.random.RandomState(seed + 100)
    h2, c52 = _cd(best_h.copy(), 600000, rng2, dl, T0=0.0004, T1=0, shifts=shifts, n2=n2)
    if c52 < best_c5:
        best_c5, best_h = c52, h2

    # Phase 3: perturb + CD (15%)
    dl = t0 + budget_s * 0.80
    hp3 = _proj(best_h + 0.02 * rng.randn(n), n * 0.5)
    rng3 = np.random.RandomState(seed + 200)
    h3, c53 = _cd(hp3, 400000, rng3, dl, T0=0.0008, T1=0.0001, shifts=shifts, n2=n2)
    if c53 < best_c5:
        best_c5, best_h = c53, h3

    # Phase 4: fine polish CD (12%)
    dl = t0 + budget_s * 0.92
    rng4 = np.random.RandomState(seed + 300)
    h4, c54 = _cd(best_h.copy(), 1200000, rng4, dl, T0=0, T1=0, ls_n=35, shifts=shifts, n2=n2)
    if c54 < best_c5:
        best_c5, best_h = c54, h4

    # Phase 5: cyclic neighbor passes (5%)
    dl = t0 + budget_s * 0.97
    rem = int(max(1, (budget_s * 0.05) / (2.0 / n / 50)))
    h5, c55 = _cyclic_cd(best_h.copy(), rem, dl, shifts=shifts, n2=n2)
    if c55 < best_c5:
        best_c5, best_h = c55, h5

    # Final micro-restarts
    if time.time() - t0 < budget_s - 10:
        for r in range(4):
            if time.time() - t0 > budget_s - 5:
                break
            sc = 0.005 + 0.01 * r
            hp = _proj(best_h + sc * rng.randn(n), n * 0.5)
            hl, cl = _cd(hp, 80000, rng, t0 + budget_s - 5, T0=0.0001, T1=0, shifts=shifts, n2=n2)
            if cl < best_c5:
                best_c5, best_h = cl, hl

    best_h = _proj(best_h, n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
