def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _softmax_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    cmax = c.max()
    val = cmax + T * np.log(np.sum(np.exp((c - cmax) / T)))
    return val + penalty * (h.sum() - n * 0.5)**2

def _softmax_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    exp_c = np.exp((c - cmax) / T)
    weights = exp_c / exp_c.sum()  # softmax weights
    g = np.zeros(n)
    for k in range(2 * n - 1):
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            i1 = np.arange(cnt)
            g[i1] += weights[k] * (1.0 - h[i1 + sh]) * dx
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= weights[k] * h[i2 - sh] * dx

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _lsexp_obj(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    val = cmax + T * np.log(np.sum(np.exp((c - cmax) / T)))
    return val + penalty * (h.sum() - n * 0.5) ** 2

def _lsexp_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    exp_c = np.exp((c - cmax) / T)
    w = exp_c / exp_c.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

def _topk_grad(h, topk_idx, dx):
    n = len(h)
    g = np.zeros(n)
    for k in topk_idx:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:cnt]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    return g * dx / max(len(topk_idx), 1)

def _lbfgs_lsexp(h, T, penalty, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lsexp_obj, h, jac=_lsexp_grad,
                       args=(T, penalty), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_adapt(h, n_iter, k_min, k_max, step0, el_fn, bc_fn):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    v = np.zeros(n)
    beta = 0.8
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        cmax = c.max()
        spread = cmax - np.sort(c)[-int(0.1 * len(c)):-1].mean()
        if spread < 1e-4:
            k_eff = k_max
        elif spread < 0.05:
            k_eff = max(k_min, min(k_max, int(0.05 * len(c))))
        else:
            k_eff = max(k_min, len(c) * 0.01)
        topk = np.argsort(c)[-int(k_eff):]
        g = _topk_grad(h_c, topk, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-17:

            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.025, 0.025)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-10:
                break
    return best_h, best_c

            else:

        if _ % 6000 == 0:
            scale *= 0.6
    return best_h, best_c
