        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj_to_sum(h, n * 0.5)
    return h, best

def _local_search(h, n_iters, rng, t_end):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 5e-6 * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c = _eval(h)
            if c < cur_c - 1e-19:
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)
    t_final = t0 + budget_s
    
    # Phase 1: Quick Nesterov smooth
    for alpha in [5000, 50000, 300000]:
        te = min(time.time() + 12, t_final * 0.08)
        if time.time() > te:
            break
        h = _nesterov(h, alpha, 300, 0.003, te)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Main coordinate descent
    for T, dur_f, pairs in [(5, 0.18, 1200), (8, 0.18, 1200), (6, 0.12, 1500)]:
        if time.time() > t_final * 0.6:
            break
        dur = budget_s * dur_f
        h2 = best_h.copy()
        h2, c5 = _coord_desc(h2, n, dur, rng, t_final, T, pairs)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()
    
    # Phase 3: Nesterov refinement
    if time.time() < t_final * 0.68:
        for alpha in [500000, 2000000]:
            te = min(time.time() + 18, t_final * 0.67)
            if time.time() > te:
                break
            h2 = best_h.copy()
            h2 = _nesterov(h2, alpha, 250, 0.001, te)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
    
    # Phase 4: More coordinate descent
    for T, dur_f, pairs in [(10, 0.1, 1500), (5, 0.08, 2000), (8, 0.08, 1500)]:
        if time.time() > t_final * 0.88:
            break
        dur = budget_s * dur_f
        h2 = best_h.copy()
        h2, c5 = _coord_desc(h2, n, dur, rng, t_final, T, pairs)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2.copy()
    
    # Phase 5: Fine local search
    if time.time() < t_final - 30:
        rem = t_final - 25 - time.time()
        ntr = min(int(rem / 0.003), 60000)
        if ntr > 100:
            h2 = best_h.copy()
            best_h, best_c5 = _local_search(h2, ntr, rng, t_final - 25)
    
    # Phase 6: Final coord polish
    if time.time() < t_final - 8:
        rem = t_final - 6 - time.time()
        if rem > 5:
            h2 = best_h.copy()
            h2, c5 = _coord_desc(h2, n, rem, rng, t_final, 8, 1000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
    
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
