            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _sum_obj(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, ftol, top_k=None):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_obj, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                      args=(alpha, top_k), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _block_obj(x, h_full, s0, s1, bsz):
    ht = h_full.copy()
    ht[s0:s1] = x
    return _eval(ht)

def _block_con(x, bsum):
    return x.sum() - bsum

def _block_jac(x, bsum):
    return np.ones(len(x))

def _block_refine(h, bsz, n_passes):
    n = len(h)
    nb = (n + bsz - 1) // bsz
    for _pass in range(n_passes):
        for b in range(nb):
            s0 = b * bsz
            s1 = min(s0 + bsz, n)
            x0 = h[s0:s1].copy()
            bsum = x0.sum()
            bn = s1 - s0
            try:
                cons = [{'type': 'eq', 'fun': _block_con, 'jac': _block_jac, 'args': (bsum,)}]
                res = minimize(_block_obj, x0, method='SLSQP',
                             args=(h, s0, s1, bn),
                             bounds=[(0, 1)] * bn, constraints=cons,
                             options={'maxiter': 80, 'ftol': 1e-15})
                h[s0:s1] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    stagnation = 0
    for _ in range(n_iter):
        if c5cur < best_c - 1e-15:
            stagnation = 0
        else:
            stagnation += 1
        step = base_step * (1.0 + 0.3 * min(stagnation // 3000, 10))
        step = min(step, base_step * 8.0)
        r = rng.random()
        m = 2 if r < 0.25 else (3 if r < 0.55 else (4 if r < 0.8 else 5))
        i0 = rng.randint(n)
        indices = []
        used = set()
        for _ in range(m):
            j = (i0 + rng.randint(0, max(2, m * 4))) % n
            if j not in used:
                indices.append(j)
                used.add(j)
        if len(indices) < 2:
            continue
        deltas = rng.randn(m)
        deltas -= deltas.mean()
        deltas *= step
        feasible = True
        for ii in range(m):
            v = h[indices[ii]] + deltas[ii]
            if v < -1e-10 or v > 1 + 1e-10:
                feasible = False
                break
        if not feasible:
            continue
        orig = np.array([h[idx] for idx in indices])
        for ii in range(m):
            h[indices[ii]] += deltas[ii]
        c5n = _eval(h)
        if c5n < c5cur:
            c5cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            for ii in range(m):
                h[indices[ii]] = orig[ii]
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution coarse
    for ratio in [4, 8]:
        if time.time() - t0 > 50:
            break
        nc = max(32, n // ratio)
        hc = _repair(_downsample(h, n // nc), nc)
        for alpha in [1000, 5000, 25000]:
            if time.time() - t0 > 60:
                break
            hc = _slsqp(hc, alpha, 250, 1e-14)
        h_up = _repair(_upsample(hc, n // nc), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Full-res SLSQP with progressive alpha
    p2_end = budget_s * 0.35
    for alpha in [5000, 20000, 100000, 500000, 2000000]:
        if time.time() - t0 > p2_end:
            break
        h_new = _slsqp(best_h, alpha, 1200, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-13:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Top-k focused SLSQP (faster per iter)
    p3_end = budget_s * 0.55
    for tk in [3, 5, 8]:
        if time.time() - t0 > p3_end:
            break
        for alpha in [50000, 300000]:
            if time.time() - t0 > p3_end:
                break
            h_new = _slsqp(best_h, alpha, 2000, 1e-15, top_k=tk)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-13:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Block coordinate
    p4_end = budget_s * 0.70
    for bsz in [12, 20]:
        if time.time() - t0 > p4_end:
            break
        h_new = _block_refine(best_h.copy(), bsz, 3)
        c5n = _eval(h_new)
        if c5n < best_c5 - 1e-13:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 5: Multi-point local search (3 passes, decreasing step)
    remaining = budget_s - 8 - (time.time() - t0)
    if remaining > 5:
        n1 = min(int(remaining * 0.35 / 0.0014), 250000)
        h_tmp, c_tmp = _local_search(best_h.copy(), n1, rng, 4e-5)
        if c_tmp < best_c5:
            best_c5 = c_tmp
            best_h = h_tmp.copy()

    remaining = budget_s - 5 - (time.time() - t0)
    if remaining > 3:
        n2 = min(int(remaining * 0.60 / 0.0013), 300000)
        h_tmp, c_tmp = _local_search(best_h.copy(), n2, rng, 1.5e-5)
        if c_tmp < best_c5:
            best_c5 = c_tmp
            best_h = h_tmp.copy()

    remaining = budget_s - 3 - (time.time() - t0)
    if remaining > 2:
        n3 = min(int(remaining * 0.7 / 0.0012), 250000)
        h_tmp, c_tmp = _local_search(best_h.copy(), n3, rng, 4e-6)
        if c_tmp < best_c5:
            best_c5 = c_tmp
            best_h = h_tmp.copy()

    # Phase 6: Ultra-fine 2-point polish
    remaining = budget_s - 1 - (time.time() - t0)
    if remaining > 1:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int(remaining / 0.0005), 150000)
        for _ in range(npol):
            if time.time() - t0 > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 4)) % nc2
            d = 8e-8 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
