import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _cgrad_one(h, k):
    """Gradient of c_k w.r.t. h, where k is the full-correlate index (0..2n-2)."""
    n = len(h)
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        nproper = n - sh
        g[:nproper] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:nproper]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * _cgrad_one(x, k)
    return g * dx

def _hinge_obj(x, C, margin):
    c = np.maximum(_cvals(x), 0.0)
    exc = np.maximum(c - C + margin, 0.0)
    return 0.5 * np.sum(exc ** 2)

def _hinge_grad(x, C, margin):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cvals(x), 0.0)
    exc = np.maximum(c - C + margin, 0.0)
    w = exc / max(exc.sum(), 1e-30)
    active = np.where(exc > 1e-15)[0]
    g = np.zeros(n)
    for k in active:
        g += w[k] * _cgrad_one(x, k)
    return g * dx

def _lbfgs_lp(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 15:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10, 'factr': 100})
        return _repair(res.x, n)
    except Exception:
        return h

def _hinge_pg(h, C, margin, steps, step0, deadline):
    """Projected gradient with backtracking on hinge objective."""
    n = len(h)
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.05 * i)
        g = _hinge_grad(h, C, margin)
        f_cur = _hinge_obj(h, C, margin)
        for _ in range(20):
            h_new = _repair(h - t * g, n)
            f_new = _hinge_obj(h_new, C, margin)
            if f_new <= f_cur - 1e-12 * t * np.sum(g ** 2):
                break
            t *= 0.5
        h = h_new
    return h

def _topk_iter(h, n_lags, steps, step0, deadline):
    """Iterative projected gradient focusing on top-k lags."""
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += _cgrad_one(h, k)
        g *= dx / n_lags
        t = step0 / (1.0 + 0.02 * i)
        for _ in range(10):
            h_new = _repair(h - t * g, n)
            c5n = _eval(h_new)
            c5c = _eval(h)
            if c5n <= c5c + 1e-14:
                break
            t *= 0.5
        h = h_new
    return h

def _local_2pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _local_block(h, n_iter, rng, deadline):
    """Perturb small blocks of consecutive values."""
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n - 4)
        sz = rng.randint(1, 4)
        d = 2e-5 * rng.randn(sz)
        block = h[i:i+sz] + d
        if np.all(block >= 0) and np.all(block <= 1):
            h[i:i+sz] = block
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i:i+sz] = h[i:i+sz] - d
    return best_h, best_c

def _coord_descent(h, n_sweeps, deadline):
    """Cycle through positions, do 1D minimization (with integral fix via partner)."""
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline - 10:
            break
        for i in range(n):
            j = (i + 7) % n
            # Optimize h[i] and h[j] jointly (sum preserved)
            hi, hj = h[i], h[j]
            s = hi + hj
            # Try a few splits
            for frac in [0.3, 0.4, 0.5, 0.6, 0.7, 0.2, 0.8]:
                ni = frac * s
                nj = s - ni
                if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= hi <= 1 and 0 <= hj <= 1:
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    old_i, old_j = h[i], h[j]
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < _eval(best_h) - 1e-18:
                        if c5n < best_c:
                            best_c = c5n
                            best_h = h.copy()
                    else:
                        h[i], h[j] = old_i, old_j
    return best_h, best_c
