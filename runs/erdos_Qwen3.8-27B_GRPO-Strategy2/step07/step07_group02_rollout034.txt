    dx = 2.0 / n

    return g * dx

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lag_g(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g

def _sob(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sog(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] >= 1e-14:
            g += _lag_g(x, k, n) * w[k]
    return g

def _lbfgs(h, alpha, maxit, dl):
    n = len(h)
    if time.time() > dl - 20:
        return h
    try:
        r = minimize(_sob, h, jac=_sog, args=(alpha,),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-15,
                              'gtol': 1e-12, 'maxfun': 2 * maxit})
        return _repair(r.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    if abs(h.sum() - t) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lg(h, k, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

def _sob(x, a):
    c = _cvals(x)
    m = c.max()
    s = np.clip(a * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / a

def _sog(x, a):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(a * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] >= 1e-14:
            g += _lg(x, k, n) * w[k]
    return g

    c = _cvals(x)
    c = np.maximum(c, 1e-30)
    return (np.sum(c ** p)) ** (1.0 / p)

    n = len(x)
    c = np.maximum(_cvals(x), 1e-30)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] >= 1e-14:
            g += _lg(x, k, n) * w[k]
    return g

        return x

                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-12})
        return _repair(r.x, n)
    except Exception:
        return x

            idx = [i, j, k]; delta = [d1, d2, -d1-d2]

        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:m]
    else:

        g[sp:] += (1.0 - h[:-sp])
        g[:-sp] -= h[sp:]
    return g * dx

        x, c5x = _ls(bh, 30000, rng, dl, 3)
        if c5x < bc5:
            bc5 = c5x; bh = x.copy()
    if time.time() < dl - 2:
        x, c5x = _ls(bh, 80000, rng, dl, 2)
        if c5x < bc5:
            bc5 = c5x; bh = x.copy()

    bh = _repair(bh, n0)
    bc5 = _eval(bh)
    return (bh, bc5, n0)
