import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n


def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    if abs(h.sum() - t) > 1e-9:
        h = _proj_sum(h, t)
    return h


def _lse_obj(h, tau, pen):
    c = _corrs(h)
    cm = c.max()
    val = (cm + np.log(np.sum(np.exp(tau * (c - cm))))) / tau
    return val + pen * (h.sum() - len(h) * 0.5) ** 2


def _lse_grad(h, tau, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    w = np.exp(tau * (c - cm))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(c) / len(c)
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s == 0:
            g += wk * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] += -wk * h[:n - s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] += -wk * h[sp:]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g


def _lbfgs(h, tau, pen, maxiter, el_fn, bc_fn):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(tau, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)


def _subgrad(h, n_iter, el_fn, bc_fn):
    n = len(h)
    h_c = h.astype(float).copy()
    best_c = _eval(h_c)
    best_h = h_c.copy()
    dx = 2.0 / n
    step = 0.015
    stall = 0
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        k_act = max(1, min(20, int(8 * np.exp(-it * 0.003))))
        idx = np.argsort(c)[-k_act:]
        g = np.zeros(n)
        for k in idx:
            s = k - (n - 1)
            if s == 0:
                g += dx * (1.0 - 2.0 * h_c)
            elif s > 0:
                g[:n - s] += dx * (1.0 - h_c[s:])
                g[s:] += -dx * h_c[:n - s]
            else:

                g[sp:] += dx * (1.0 - h_c[:-sp])
                g[:n - sp] += -dx * h_c[sp:]
        g /= len(idx)
        h_try = _proj_sum(h_c - step * g, n * 0.5)

        if c_int < best_c5 + 0.003:
            for tau in [100, 800]:
                if el() > budget_s - 80:
                    break
                h_t = _lbfgs(h_int.copy(), tau, 1e6, 200, el, bc)
                c_t = _eval(h_t)
                if c_t < c_int:
                    c_int = c_t
                    h_int = h_t
        if c_int < best_c5:
            best_c5 = c_int
            best_h = h_int
            n0 = n_try

    # Phase 6: Subgradient refinement
    if el() < budget_s - 80:
        h_try, c_try = _subgrad(best_h, 15000, el,
                                lambda: el() > budget_s - 80)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 7: Triple-point local search
    if el() < budget_s - 40:
        nt = min(int((budget_s - 40 - el()) / 0.005), 50000)
        best_h, best_c5 = _triple_local(best_h, nt, rng, el, bc)

    # Phase 8: Fine 2-point local search
    if el() < budget_s - 10:
        nt = min(int((budget_s - 10 - el()) / 0.0025), 80000)
        best_h, best_c5 = _fine_local(best_h, nt, rng, el,
                                      lambda: el() > budget_s - 10)

    # Phase 9: Final L-BFGS polish
    if el() < budget_s - 3:
        h_try = _lbfgs(best_h.copy(), 3000, 1e8, 100, el, bc)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
