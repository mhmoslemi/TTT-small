def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid

            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _lse_obj(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    e = np.exp(beta * (c - cm))
    return (1.0 / beta) * np.log(e.sum()) + cm + penalty * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    e = np.exp(beta * (c - cm))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g + 2.0 * penalty * (h.sum() - n * 0.5)

                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    best_c = _eval(h_c)
    best_h = h_c.copy()
    step = step0
    stall = 0
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in topk:
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += (1.0 - h_c[i1 + sh])
                if sh > 0:

            else:
                sp = -sh
                g[sp:] += (1.0 - h_c[:n - sp])
                g[:n - sp] -= h_c[sp:]
        g *= dx / len(topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:

        h_try = _proj_sum(h_c - step * g / gn, n * 0.5)
        c_try = _eval(h_try)

        else:

            stall += 1

    return best_h, best_c

def _swap_search(h, n_iter, rng, el_fn, bc_fn):
