def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, float(c.max())

def _grad_single(h, p, n):
    g = np.zeros(2 * n - 1)
    m_lo = max(0, p + 1 - n)
    m_hi = min(2 * n - 2, p + n - 2)
    if m_lo <= m_hi:
        j_vals = np.arange(m_lo, m_hi + 1) - p + n - 1
        g[m_lo:m_hi + 1] = 1.0 - 2.0 * h[j_vals]
    return g

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
