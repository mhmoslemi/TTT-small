What about h(x) = 1 for x in a carefully chosen set of measure 1 in [0,2]? The key is to make ∫h(x)h(x+k)dx as uniform as possible (and as large as possible) over all k.

This is related to "difference sets" in additive combinatorics. We want a set S ⊂ [0,2] with |S|=1 such that the autocorrelation |S ∩ (S+k)| is as uniform as possible and as large as possible.

For the continuous case, if h is the indicator of a set S, then:
c[k] = |S| - |S ∩ (S+k)| = 1 - |S ∩ (S+k)|

We want to maximize min_k |S ∩ (S+k)|.

This is a continuous version of the "friendship theorem" or "additive energy" problems.

For a well-distributed set (like a Cantor-type set), the autocorrelation is smooth. For a finite union of intervals, it's piecewise linear.

I think the optimal solution for this problem is NOT binary but rather a carefully shaped continuous function. The previous solution with C₅ ≈ 0.381 is likely in this regime.

                    diff -= a
                else:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

                    break

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _subgrad_step(h, step, n):
    """One subgradient step: descend in direction of argmax correlation."""
    c = _correl(h)
    k_max = c.argmax()
    sh = (n - 1) - k_max
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return _proj(h - step * g, n * 0.5)

def _smax_objs(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_objs(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_objs(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            h = _proj(h - step * v, n * 0.5)
            f = _smax_objs(h, alpha, topk)
    return _repair(h, n)

    n = len(h)

    bh = h.copy()
    bc = c5c
    step0 = 1e-4 if mode == 'fine' else 1e-3
