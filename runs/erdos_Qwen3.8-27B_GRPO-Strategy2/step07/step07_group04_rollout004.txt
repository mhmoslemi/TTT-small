def _local_search(h, n_tries, rng, t0, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > budget_s - 5:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 6)) % n
        k = (i + rng.randint(7, 15)) % n
        d1 = 4e-5 * rng.randn()
        d2 = 4e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
    return best_h, best_c

def _ultra_fine(h, n_tries, rng, t0, budget_s):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _it in range(n_tries):
        if time.time() - t0 > budget_s - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 8e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _lowres_escape(best_h, best_c5, t0, budget_s, rng):
    n = len(best_h)
    for factor in [2, 3]:
        if time.time() - t0 > budget_s - 50:
            break
        n_lo = n // factor
        h_lo = best_h[:n_lo * factor].reshape(n_lo, factor).mean(axis=1)
        h_lo += 0.15 * rng.randn(n_lo)
        h_lo = _proj_to_sum(h_lo, n_lo * 0.5)
        for kt in [5, 20]:
            h_lo, _ = _nesterov(h_lo, n_lo, kt, 1500, 0.003, t0, budget_s - 30)
        h_hi = np.repeat(h_lo, factor)
        h_hi = _repair(h_hi, n)
        c5v = _eval(h_hi)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_hi.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-scale Nesterov (k=3, 10, 50)
    for kt, ms, st in [(3, 4000, 0.003), (10, 2500, 0.0015), (50, 1500, 0.0008)]:
        if time.time() - t0 > budget_s - 400:
            break
        h, f = _nesterov(best_h.copy(), n, kt, ms, st, t0, budget_s - 380)
        if f < best_c5:
            best_c5, best_h = f, h.copy()

    # Phase 2: SLSQP sweep with increasing alpha
    for alpha in [3000, 15000, 60000, 250000]:
        if time.time() - t0 > budget_s - 250:
            break
        h_c = _slsqp(best_h.copy(), alpha, 700, t0, budget_s)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_c.copy()

    # Phase 3: Low-resolution escape
    if time.time() - t0 < budget_s - 150:
        best_h, best_c5 = _lowres_escape(best_h, best_c5, t0, budget_s - 40, rng)

    # Phase 4: High-alpha SLSQP (sharper max approximation)
    for alpha in [600000, 2000000, 5000000]:
        if time.time() - t0 > budget_s - 80:
            break
        h_c = _slsqp(best_h.copy(), alpha, 500, t0, budget_s)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_c.copy()

    # Phase 5: Final Nesterov polish (very high k)
    if time.time() - t0 < budget_s - 50:
        h, f = _nesterov(best_h.copy(), n, 80, 1200, 0.0004, t0, budget_s - 35)
        if f < best_c5:
            best_c5, best_h = f, h.copy()

    # Phase 6: 3-point local search
    if time.time() - t0 < budget_s - 15:
        rem = budget_s - 15 - (time.time() - t0)
        ntr = min(int(rem / 0.004), 45000)
        best_h, best_c5 = _local_search(best_h, ntr, rng, t0, budget_s - 5)

    # Phase 7: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 3:
        rem = budget_s - 3 - (time.time() - t0)
        ntr = min(int(rem / 0.002), 35000)
        best_h, best_c5 = _ultra_fine(best_h, ntr, rng, t0, budget_s)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
