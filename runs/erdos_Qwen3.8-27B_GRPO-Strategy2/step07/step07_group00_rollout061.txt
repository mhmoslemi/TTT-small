        t += 1

        m_v = beta1 * m_v + (1 - beta1) * g

            f = fn

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _softmax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-14:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wi * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wi * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wi * (1.0 - h[:-sp])
            g[:n - sp] -= wi * h[sp:]
    return g

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _softmax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_topk(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h, alpha, topk, steps, lr, deadline):
    n = len(h)
    target = n * 0.5
    b1, b2, eps = 0.85, 0.999, 1e-8
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    f = _softmax_obj(h, alpha, topk)
    for _ in range(steps):
        if time.time() > deadline:
            break
        t += 1
        g = _grad_topk(h, alpha, topk)
        mv = b1 * mv + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        mh = mv / (1 - b1 ** t)
        vh = vv / (1 - b2 ** t)
        sv = lr * mh / (np.sqrt(vh) + eps)
        hn = _proj(h - sv, target)
        fn = _softmax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _softmax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        wi = w[ii]
        if wi < 1e-14:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wi * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wi * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wi * (1.0 - h[:-sp])
            g[:n - sp] -= wi * h[sp:]
    return g

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: _softmax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_topk(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _adam(h, alpha, topk, steps, lr, deadline):
    n = len(h)
    target = n * 0.5
    b1, b2, eps = 0.85, 0.999, 1e-8
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    f = _softmax_obj(h, alpha, topk)
    for _ in range(steps):
        if time.time() > deadline:
            break
        t += 1
        g = _grad_topk(h, alpha, topk)
        mv = b1 * mv + (1.0 - b1) * g
        vv = b2 * vv + (1.0 - b2) * g * g
        mh = mv / (1.0 - b1 ** t)
        vh = vv / (1.0 - b2 ** t)
        sv = lr * mh / (np.sqrt(vh) + eps)
        hn = _proj(h - sv, target)
        fn = _softmax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
    return _repair(h, n)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _softmax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    target = n * 0.5
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _softmax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 6.0)
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _softmax_obj(h, alpha, topk)
    return _repair(h, n)

def _multi_restart(h_best, c5_best, rng, deadline, n_restart):
    n = len(h_best)
    elapsed = deadline - time.time()
    per = max(20, elapsed / max(n_restart, 1))
    for r in range(n_restart):
        if time.time() > deadline:
            break
        sigma = 0.035 / (1.0 + 0.6 * r)
        h_pert = h_best + sigma * rng.randn(n)
        h_pert = _repair(h_pert, n)
        sub_dl = min(deadline, time.time() + per)
        h_opt = _adam(h_pert, 80000, 12, 4000, 0.0015, sub_dl)
        c5_opt = _eval(h_opt)
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt.copy()
    return h_best, c5_best

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _pair_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _single_polish(h, niter, rng, deadline):

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n

        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
