import numpy as np
import time
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _slsqp_quick(h, n, alphas, t0, budget_frac):
    def _obj(x, alpha):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _sum_f(x):
        return x.sum() - n * 0.5
    def _sum_j(x):
        return np.ones(len(x))
    best = h.copy()
    best_c = _eval(best)
    for alpha in alphas:
        if time.time() - t0 > budget_frac:
            break
        cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(_obj, best, args=(alpha,), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14})
            cand = _repair(res.x, n)
            cc = _eval(cand)
            if cc < best_c:
                best_c = cc
                best = cand.copy()
        except Exception:
            pass
    return best
