    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if abs(diff) < 1e-12:
                    break
                i = order[idx]
                if diff > 0:
                    a = min(diff, 1.0 - h[i])
                    h[i] += a
                    diff -= a
                else:
                    a = min(-diff, h[i])
                    h[i] -= a
                    diff += a
    return h

            lo = mid
        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        else:
            sp = -sh

        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 1 + rng.randint(7, 18)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            fn = _eval(h)
            if fn < fc:
                fc = fn
                stall = 0
                step = min(step * 1.03, 2e-4)
                if fn < best_c:
                    best_c, best_h = fn, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
                stall += 1
                if stall > 3000:
                    step *= 0.4
                    stall = 0
    return best_h, best_c

def _ls2(h, n_tries, rng, t0, budget_s):
    n = len(h)
    fc = _eval(h)
    best_h, best_c = h.copy(), fc
    step = 3e-6
    stall = 0
    for _ in range(n_tries):
        if time.time() - t0 > budget_s - 4:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            if fn < fc:
                fc = fn
                stall = 0
                step = min(step * 1.02, 2e-5)
                if fn < best_c:
                    best_c, best_h = fn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                stall += 1
                if stall > 5000:
                    step *= 0.4
                    stall = 0
    return best_h, best_c

def _ls_block(h, n_tries, rng, t0, budget_s):
    n = len(h)
    fc = _eval(h)
    best_h, best_c = h.copy(), fc
    step = 4e-5
    stall = 0
    for _ in range(n_tries):
        if time.time() - t0 > budget_s - 10:
            break
        bl = rng.randint(2, 7)
        i = rng.randint(n - bl)
        idx = np.arange(i, i + bl)
        d = step * rng.randn(bl)
        d -= d.mean()
        h_new = h[idx] + d
        if h_new.min() >= 0 and h_new.max() <= 1:
            h[idx] = h_new
            fn = _eval(h)
            if fn < fc:
                fc = fn
                stall = 0
                step = min(step * 1.04, 1.5e-4)
                if fn < best_c:
                    best_c, best_h = fn, h.copy()
            else:
                h[idx] = h_new - d
                stall += 1
                if stall > 2000:
                    step *= 0.4
                    stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Nesterov top-k=15
    h = _nesterov(h, 15, 16000, 0.0015, 80)
    c = _eval(h)
    if c < best_c5:
        best_c5, best_h = c, h.copy()

    # Phase 2: SLSQP alpha annealing
    for al in [5000, 25000, 100000, 400000]:
        if time.time() - t0 > budget_s * 0.35:
            break
        hc = _slsqp(best_h.copy(), al, 1500, 1e-18)
        c = _eval(hc)
        if c < best_c5:
            best_c5, best_h = c, hc.copy()

    # Phase 3: Nesterov top-k=30 (more faithful max)
    if time.time() - t0 < budget_s - 320:
        h = _nesterov(best_h.copy(), 30, 8000, 0.0006, 55)
        c = _eval(h)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

    # Phase 4: High-alpha SLSQP for precise max
    for al in [1000000, 3000000, 8000000]:
        if time.time() - t0 > budget_s - 380:
            break
        hc = _slsqp(best_h.copy(), al, 1000, 1e-19)
        c = _eval(hc)
        if c < best_c5:
            best_c5, best_h = c, hc.copy()

    # Phase 5: Block local search (medium-scale)
    if time.time() - t0 < budget_s - 150:
        remain = budget_s - 150 - (time.time() - t0)
        nt = min(int(remain * 1800), 250000)
        best_h, best_c5 = _ls_block(best_h, nt, rng, t0, budget_s - 140)

    # Phase 6: 3-point adaptive search
    if time.time() - t0 < budget_s - 80:
        remain = budget_s - 80 - (time.time() - t0)
        nt = min(int(remain * 3000), 500000)
        best_h, best_c5 = _ls3(best_h, nt, rng, t0, budget_s - 70)

    # Phase 7: 2-point ultra-fine polish
    if time.time() - t0 < budget_s - 5:
        remain = budget_s - 5 - (time.time() - t0)
        nt = min(int(remain * 5000), 1000000)
        best_h, best_c5 = _ls2(best_h, nt, rng, t0, budget_s - 3)

    # Final micro-polish with even finer steps
    if time.time() - t0 < budget_s - 2:
        h_micro = best_h.copy()
        step = 1e-7
        for _ in range(200000):
            if time.time() - t0 > budget_s - 1:
                break
            i = rng.randint(n0)
            j = (i + 1 + rng.randint(0, 3)) % n0
            d = step * rng.randn()
            ni, nj = h_micro[i] + d, h_micro[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_micro[i], h_micro[j] = ni, nj
                fn = _eval(h_micro)
                if fn < best_c5 - 1e-20:
                    best_c5 = fn
                    best_h = h_micro.copy()
                else:
                    h_micro[i], h_micro[j] = ni - d, nj + d
    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
