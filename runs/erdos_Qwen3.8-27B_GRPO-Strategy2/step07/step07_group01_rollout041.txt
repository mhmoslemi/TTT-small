from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n // 2
    s = h.sum()
    if s < 1e-15:
        h[:] = 0.5
        return h
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for ii in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[ii])
                h[ii] += a; diff -= a
            else:
                a = min(-diff, h[ii])
                h[ii] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h
