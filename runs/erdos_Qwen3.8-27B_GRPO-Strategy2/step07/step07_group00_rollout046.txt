        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    h[idx] += a
                    diff -= a
                else:

                    diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    n = len(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        j = idx[ii]
        offset = j - (n - 1)
        wk = w[ii] * dx
        if offset >= 0:
            g[offset:] += wk * (1.0 - h[:n - offset])
            g[:n - offset] -= wk * h[offset:]
        else:
            q = -offset
            g[:n - q] += wk * (1.0 - h[q:])
            g[q:] -= wk * h[:n - q]
    return g

def _proj_grad(h, alpha, topk, max_iter, deadline):
    n = len(h)
    target = n * 0.5
    f = _smax_obj(h, alpha, topk)
    step = 0.01
    best_local = h.copy()
    best_f = f
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        gnorm = np.sqrt((g * g).sum())
        if gnorm < 1e-12:
            break
        # Backtracking line search
        for ls in range(30):
            hn = _proj(h - step * g, target)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f - 1e-12:
                break
            step *= 0.6
            if step < 1e-8:
                hn = h
                break
        else:
            step *= 0.5
            if step < 1e-10:
                break
            hn = h
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.1, 0.1)
            if f < best_f:
                best_f = f
                best_local = h.copy()
        else:
            step *= 0.5
    return best_local
