```python
def _pair_grad(h, i, j, n):
    """Compute g_lin[p] = dc_p/dh[i] - dc_p/dh[j] for all p."""
    # dc/dh[i] as function of p
    # For p = n-1+shift:
    #   if shift >= 0: (1-h[i-shift]) if 0<=shift<=i, minus h[i+shift] if shift<=n-1-i
    #   if shift < 0: (1-h[i-s]) if s<=n-1-i, minus h[i-s] if s<=i where s=-shift
    
    gi = np.zeros(2*n-1)
    gj = np.zeros(2*n-1)
    
    # For point i:
    # shift in [0, i]: gi[n-1+shift] += 1-h[i-shift]
    if i >= 0:
        shifts = np.arange(0, i+1)
        gi[n-1+shifts] += (1.0 - h[i-shifts])
    # shift in [0, n-1-i]: gi[n-1+shift] -= h[i+shift]
    shifts2 = np.arange(0, n-i)
    gi[n-1+shifts2] -= h[i+shifts2]
    # shift in [-i, -1] i.e., s in [1, i]: gi[n-1-s] -= h[i-s]
    if i > 0:
        s_vals = np.arange(1, i+1)
        gi[n-1-s_vals] -= h[i-s_vals]
    # shift in [-(n-1-i), -1] i.e., s in [1, n-1-i]: gi[n-1-s] += 1-h[i+s]
    if n-1-i > 0:
        s_vals2 = np.arange(1, n-i)
        gi[n-1-s_vals2] += (1.0 - h[i+s_vals2])
    
    # Similarly for j
    if j >= 0:
        shifts = np.arange(0, j+1)
        gj[n-1+shifts] += (1.0 - h[j-shifts])
    shifts2 = np.arange(0, n-j)
    gj[n-1+shifts2] -= h[j+shifts2]
    if j > 0:
        s_vals = np.arange(1, j+1)
        gj[n-1-s_vals] -= h[j-s_vals]
    if n-1-j > 0:
        s_vals2 = np.arange(1, n-j)
        gj[n-1-s_vals2] += (1.0 - h[j+s_vals2])
    
    return gi - gj
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _subgrad(h, m_top=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-m_top:]
    g = np.zeros(n)
    for p in top_idx:
        shift = p - (n - 1)
        if shift >= 0:
            if shift == 0:
                g += 1.0 - 2.0 * h
            else:
                g[shift:] += (1.0 - h[:n - shift])
                g[:n - shift] -= h[shift:]
        else:
            s = -shift
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    g *= dx / m_top
    return g, c

def _pair_grad(h, i, j, n):
    """g_lin[p] = dc_p/dh[i] - dc_p/dh[j] for all p (without dx factor)."""
    res = np.zeros(2 * n - 1)
    for pt, sign in ((i, 1.0), (j, -1.0)):
        # Positive shifts: p = n-1+shift, shift in [0, ...]
        # (1-h[pt-shift]) for shift in [0, pt]
        sh_max = pt
        if sh_max >= 0:
            idx = np.arange(sh_max + 1)
            res[n - 1 + idx] += sign * (1.0 - h[pt - idx])
        # -h[pt+shift] for shift in [0, n-1-pt]
        sh_max2 = n - 1 - pt
        if sh_max2 >= 0:
            idx = np.arange(sh_max2 + 1)
            res[n - 1 + idx] += sign * (-h[pt + idx])
        # Negative shifts: p = n-1-s, s in [1, ...]
        # -h[pt-s] for s in [1, pt]
        if pt > 0:
            idx = np.arange(1, pt + 1)
            res[n - 1 - idx] += sign * (-h[pt - idx])
        # (1-h[pt+s]) for s in [1, n-1-pt]
        if n - 1 - pt > 0:
            idx = np.arange(1, n - pt)
            res[n - 1 - idx] += sign * (1.0 - h[pt + idx])
    return res

def _line_search(h, i, j, c, n, dx):
    """Find optimal d for h[i]+=d, h[j]-=d. Returns (d, c_new)."""
    k1 = n - 1 + (j - i)
    k2 = n - 1 - (j - i)
    g_lin = _pair_grad(h, i, j, n) * dx
    quad = np.zeros(2 * n - 1)
    if 0 <= k1 < 2 * n - 1:
        quad[k1] += 1.0
    if 0 <= k2 < 2 * n - 1:
        quad[k2] += 1.0
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-14:
        return 0.0, c
    for _ in range(30):
        d_mid = 0.5 * (d_lo + d_hi)
        val = c + d_mid * g_lin + d_mid * d_mid * quad
        p_star = np.argmax(val)
        sg = g_lin[p_star] + 2.0 * d_mid * quad[p_star]
        if sg > 1e-12:
            d_hi = d_mid
        elif sg < -1e-12:
            d_lo = d_mid
        else:
            break
    d_opt = 0.5 * (d_lo + d_hi)
    if abs(d_opt) < 1e-12:
        return 0.0, c
    c_new = c + d_opt * g_lin + d_opt * d_opt * quad
    return d_opt, c_new

def _smart_cd(h, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    order = np.arange(n)
    for it in range(500000):
        if time.time() >= deadline:
            break
        if it % 50 == 0:
            g, _ = _subgrad(h, 5)
            order = np.argsort(g)
        if stable > 3000 and it % 200 == 0:
            break
        r = it % 7
        if r < 3:
            i = rng.randint(n)
            j = rng.randint(n)
        elif r == 3:
            i = order[rng.randint(0, n // 4)]
            j = order[rng.randint(3 * n // 4, n)]
        elif r == 4:
            i = order[rng.randint(0, n // 3)]
            j = order[rng.randint(2 * n // 3, n)]
        elif r == 5:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
        else:
            i = rng.randint(0, n // 2)
            j = (i + n // 2 + rng.randint(-3, 4)) % n
        if i == j:
            continue
        c_saved = c.copy()
        d, c_new = _line_search(h, i, j, c, n, dx)
        if abs(d) < 1e-13:
            stable += 1
            continue
        h[i] += d
        h[j] -= d
        c5_new = c_new.max()
        if c5_new < c5_cur - 1e-15:
            c = c_new
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
            c = c_saved
            stable += 1
    return best_h, best_c

def _proj_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad_descent(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(2000):
        if time.time() >= deadline:
            break
        g, c = _subgrad(h, 5)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = min(0.05, 1.0 / gn)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _proj_sum(h - mult * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def _fine_polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for it in range(1000000):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-16:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    n = len(initial_h_values)

    best_h = h.copy()
    best_c5 = _eval(best_h)

    for ps in [0.02, 0.05, 0.1, 0.2]:

        hp = _proj_sum(hp, n * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:

            best_h = hp.copy()
    
    # Phase 1: Subgradient (40s)

    # Phase 3: Another CD round (250s)
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _subgrad(h, m_top=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-m_top:]
    g = np.zeros(n)
    for p in top_idx:
        shift = p - (n - 1)
        if shift >= 0:
            if shift == 0:
                g += 1.0 - 2.0 * h
            else:
                g[shift:] += (1.0 - h[:n - shift])
                g[:n - shift] -= h[shift:]
        else:
            s = -shift
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    g *= dx / m_top
    return g, c

def _pair_grad(h, i, j, n):
    res = np.zeros(2 * n - 1)
    for pt, sign in ((i, 1.0), (j, -1.0)):
        sh_max = pt
        if sh_max >= 0:
            idx = np.arange(sh_max + 1)
            res[n - 1 + idx] += sign * (1.0 - h[pt - idx])
        sh_max2 = n - 1 - pt
        if sh_max2 >= 0:
            idx = np.arange(sh_max2 + 1)
            res[n - 1 + idx] += sign * (-h[pt + idx])
        if pt > 0:
            idx = np.arange(1, pt + 1)
            res[n - 1 - idx] += sign * (-h[pt - idx])
        if n - 1 - pt > 0:
            idx = np.arange(1, n - pt)
            res[n - 1 - idx] += sign * (1.0 - h[pt + idx])
    return res

def _line_search(h, i, j, c, n, dx):
    k1 = n - 1 + (j - i)
    k2 = n - 1 - (j - i)
    g_lin = _pair_grad(h, i, j, n) * dx
    quad = np.zeros(2 * n - 1)
    if 0 <= k1 < 2 * n - 1:
        quad[k1] += 1.0
    if 0 <= k2 < 2 * n - 1:
        quad[k2] += 1.0
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-14:
        return 0.0, c
    for _ in range(32):
        d_mid = 0.5 * (d_lo + d_hi)
        val = c + d_mid * g_lin + d_mid * d_mid * quad
        p_star = np.argmax(val)
        sg = g_lin[p_star] + 2.0 * d_mid * quad[p_star]
        if sg > 1e-13:
            d_hi = d_mid
        elif sg < -1e-13:
            d_lo = d_mid
        else:
            break
    d_opt = 0.5 * (d_lo + d_hi)
    if abs(d_opt) < 1e-12:
        return 0.0, c
    c_new = c + d_opt * g_lin + d_opt * d_opt * quad
    return d_opt, c_new

        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        if i == j:
            continue
        d = 1e-4 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-16:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    for ps in [0.01, 0.03, 0.07, 0.12, 0.2, 0.35]:
        hp = np.clip(h + ps * rng.randn(n), 0, 1)
        hp = _proj_sum(hp, n * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    dl = t0 + min(30, budget_s * 0.03)
    ha, ca = _subgrad_descent(best_h.copy(), rng, dl)
    if ca < best_c5:
        best_c5, best_h = ca, ha

    dl = t0 + min(400, budget_s * 0.40)
    hb, cb = _smart_cd(best_h.copy(), rng, dl)
    if cb < best_c5:
        best_c5, best_h = cb, hb

    dl = t0 + min(650, budget_s * 0.65)
    hc, cc = _smart_cd(best_h.copy(), rng, dl)
    if cc < best_c5:
        best_c5, best_h = cc, hc

    dl = t0 + min(700, budget_s * 0.7)
    hd, cd = _subgrad_descent(best_h.copy(), rng, dl)
    if cd < best_c5:
        best_c5, best_h = cd, hd

    dl = t0 + min(850, budget_s * 0.85)
    he, ce = _smart_cd(best_h.copy(), rng, dl)
    if ce < best_c5:
        best_c5, best_h = ce, he

    dl = t0 + min(950, budget_s * 0.92)
    hf, cf = _sweep_cd(best_h.copy(), dl)
    if cf < best_c5:
        best_c5, best_h = cf, hf

    dl = t0 + (budget_s - 10)
    hg, cg = _fine_polish(best_h.copy(), rng, dl)
    if cg < best_c5:
        best_c5, best_h = cg, hg

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
