def _lsx_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = alpha * (c - m)
    return m + np.log(np.sum(np.exp(sc))) / alpha

def _lsx_jac(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sc = alpha * (c - m)
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:

        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            if sp < n:
                g[sp:] -= wk * x[:n - sp]
    return g

    h4, c4 = _anneal_gd(bh, t0 + budget_s - 80, 5000, 100, 3, 500)
    if c4 < bc:
        bc, bh = c4, h4

    # Phase 5: Perturbation restart to escape local minima
    if time.time() < t0 + budget_s - 50:
        hp = bh.copy()
        idx = rng.choice(n0, size=max(1, n0 // 3), replace=False)
        hp[idx] += 0.003 * rng.randn(len(idx))
        hp = _project(hp, 0.5 * n0)
        h5, c5 = _anneal_gd(hp, t0 + budget_s - 25, 8000, 300, 3, 400)
        if c5 < bc:
            bc, bh = c5, h5

    # Phase 6: Shift-and-redo
    if time.time() < t0 + budget_s - 30:
        hs = np.roll(bh, 2).copy()
        hs = _project(hs, 0.5 * n0)
        h6, c6 = _anneal_gd(hs, t0 + budget_s - 15, 5000, 400, 2, 300)
        if c6 < bc:
            bc, bh = c6, h6

    # Phase 7: Fine local search with two sigma levels
    if time.time() < t0 + budget_s - 5:
        bh, bc = _local_2(bh, 500000, rng, t0 + budget_s - 8, 4e-6)
        if time.time() < t0 + budget_s - 5:
            bh, bc = _local_2(bh, 500000, rng, t0 + budget_s - 3, 1.5e-6)

    bh = _project(bh, 0.5 * n0)
    bc = _eval(bh)
    return (bh, bc, len(bh))
