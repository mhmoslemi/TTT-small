So dc_k/dh[j] = dx * [(1-h[j+s]) * 1(j+s<n) - h[j-s] * 1(j-s≥0)] for the given shift s.

This means the gradient at each j is a weighted sum over shifts. For top-k only:
g[j] = sum_{k in topk} p_k * dx * [(1-h[j+s_k]) * 1(j+s_k<n) - h[j-s_k] * 1(j-s_k≥0)]

This is O(k*n) which is very fast for small k.

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    n = len(h)
    return float(np.max(_corrs(h)) * 2.0 / n)

def _lse_obj(h, T, pen):
    n = len(h)
    c = _corrs(h) * 2.0 / n
    cm = c.max()
    lse = cm + T * np.log(np.sum(np.exp((c - cm) / T)))
    return lse + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h) * dx
    cm = c.max()
    ec = np.exp((c - cm) / T)
    sw = ec / ec.sum()
    top_idx = np.argsort(sw)[-20:]
    sw = sw[top_idx]
    sw = sw / sw.sum()
    g = np.zeros(n)
    for kk in range(len(top_idx)):
        k = top_idx[kk]
        wk = sw[kk] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _topk_grad_direct(h, k_count):
    n = len(h)
    dx = 2.0 / n
    c = _corrs(h)
    topk = np.argsort(c)[-k_count:]
    g = np.zeros(n)
    for k in topk:
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            g[:n - sp] -= h[sp:]
    return g * dx / k_count

def _lbfgs_lse(h, T, pen, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(T, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_descent(h, n_iter, k, step0, el_fn, bc_fn):
    n = len(h)
    hc = h.copy()
    best_c = _eval(hc)
    best_h = hc.copy()
    step = step0
    stall = 0
    for _ in range(n_iter):

        else:

    return best_h, best_c

def _anneal(h, n_iter, T0, T1, rng, el_fn, bc_fn):

        ni, nj = hc[i] + d, hc[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[i], hc[j] = ni, nj
            c_new = _eval(hc)
            if c_new < best_c:
                best_c = c_new
                best_h = hc.copy()
            elif (c_new - best_c) < T * 0.01 * rng.rand():
                pass
            else:
                hc[i], hc[j] = ni - d, nj + d
    return best_h, best_c
