import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _min_1d(a, b, cuc):
    if len(a) == 0:
        return 0.0
    g0 = max(cuc, b.max())
    g1 = max(cuc, (a + b).max())
    act0 = b >= g0 - 1e-14
    if cuc >= g0 - 1e-14:
        if not act0.any() or a[act0].max() <= 1e-14:
            return 0.0
    else:
        if not act0.any() or a[act0].max() <= 1e-14:
            return 0.0
    v1 = a + b
    act1 = v1 >= g1 - 1e-14
    if cuc >= g1 - 1e-14:
        if not act1.any() or a[act1].min() >= -1e-14:
            return 1.0
    else:
        if not act1.any() or a[act1].min() >= -1e-14:
            return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        gm = max(cuc, vals.max())
        act = vals >= gm - 1e-13
        has_neg = (cuc >= gm - 1e-13)
        has_pos = has_neg
        if act.any():
            if (a[act] < -1e-13).any():
                has_neg = True
            if (a[act] > 1e-13).any():
                has_pos = True
        if has_neg and has_pos:
            break
        if has_pos and not has_neg:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def _coord_descent(h, n_sweeps, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    best_c5 = _eval(h)
    bh = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        for i in order:
            klo1, khi1 = max(0, n - 1 - i), 2 * n - 2
            klo2, khi2 = i, min(2 * n - 2, i + n - 1)
            klo, khi = min(klo1, klo2), max(khi1, khi2)
            ks = np.arange(klo, khi + 1)
            jr = i + ks - (n - 1)
            jl = i + (n - 1) - ks
            valid_r = (jr >= 0) & (jr < n)
            valid_l = (jl >= 0) & (jl < n)
            a_k = np.zeros(len(ks))
            a_k[valid_r] += (1.0 - h[jr[valid_r]]) * dx
            a_k[valid_l] -= h[jl[valid_l]] * dx
            b_k = c[ks] - a_k * h[i]
            if klo > 0:
                muel = c[:klo].max()
            else:
                muel = -1e30
            if khi < 2 * n - 2:
                muer = c[khi + 1:].max()
            else:
                muer = -1e30
            cuc = max(muel, muer)
            x_star = _min_1d(a_k, b_k, cuc)
            h[i] = x_star
        c5c = _eval(h)
        if c5c < best_c5 - 1e-15:
            best_c5 = c5c
            bh = h.copy()
        elif c5c > best_c5 + 1e-10:
            h = bh.copy()
    return bh, best_c5

def _slsqp_quick(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    dx = 2.0 / n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-13:
                continue
            k = idx[ii]
            sh = (n - 1) - k
            wk = w[ii] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:n - sp] -= wk * x[sp:]
        return g

    try:
        r = minimize(_obj, h, jac=_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc
