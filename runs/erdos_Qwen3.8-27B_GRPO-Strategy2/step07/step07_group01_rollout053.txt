import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _c5_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        g[i1 - sp] -= h[i1]
    return g * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_step(h, n, alpha, top_k):
    """One projected gradient step using top-k soft max."""
    c = _c5_vec(h)
    top_idx = np.argsort(c)[-top_k:]
    c_top = c[top_idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s) / np.exp(s).sum()
    g = np.zeros(n)
    for ii, k in enumerate(top_idx):
        if w[ii] > 1e-12:
            g += w[ii] * _grad_lag(h, k)
    return g

def _grad_descent(h, n, n_steps, alpha, top_k, base_step):
    c5 = _eval(h)
    v = np.zeros(n)
    step = base_step
    mu = 0.7
    for _ in range(n_steps):
        g = _grad_step(h, n, alpha, top_k)
        v = mu * v + (1.0 - mu) * g
        norm = np.linalg.norm(v)
        if norm < 1e-15:
            break
        v /= norm
        taken = False
        for _ in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5:
                h = h_new
                c5 = c5n
                step = min(step * 1.25, base_step * 8)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.3, base_step * 8)
    return _repair(h, n), c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _c5_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        g[i1 - sp] -= h[i1]
    return g * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    v = np.zeros(n)
    step = base_step
    mu = 0.7
    for it in range(n_steps):
        if time.time() > t_end:
            break
        c = _c5_vec(h)
        top_idx = np.argsort(c)[-top_k:]
        c_top = c[top_idx]
        m = c_top.max()
        s = np.clip(alpha * (c_top - m), -400, 400)
        s -= s.max()
        w = np.exp(s) / np.exp(s).sum()
        g = np.zeros(n)
        for ii, k in enumerate(top_idx):
            if w[ii] > 1e-12:
                g += w[ii] * _grad_lag(h, k)
        v = mu * v + (1.0 - mu) * g
        norm = np.linalg.norm(v)
        if norm < 1e-15:
            break
        v /= norm
        taken = False
        for _ in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5:
                h = h_new
                c5 = c5n
                step = min(step * 1.25, base_step * 8)
                taken = True
                break
            step *= 0.5
        if not taken:
