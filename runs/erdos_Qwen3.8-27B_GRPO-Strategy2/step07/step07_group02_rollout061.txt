        else:
            hi = mid

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_w(h, lam, dx):
    n = len(h)
    g = np.zeros(n)
    nz = np.where(lam > 1e-14)[0]
    for k in nz:
        sh = (n - 1) - k
        wk = lam[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _saddle_opt(h, T, n_iter, deadline):
    """Saddle-point: multiplicative weights on lags + proximal gradient on h."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = _eval(h)
    gamma = T  # temperature for λ update
    eta = 0.01  # initial step size
    
    for it in range(n_iter):
        if time.time() > deadline - 30:
            break
        c = _cvals(h)
        # Multiplicative weights update
        c_shifted = c - c.max()
        lam = np.exp(gamma * c_shifted)
        lam_sum = lam.sum()
        if lam_sum > 0:
            lam /= lam_sum
        else:
            lam = np.ones(2*n-1) / (2*n-1)
        
        # Primal: proximal gradient
        g = _grad_w(h, lam, dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm  # normalize for stable step
        
        # Adaptive step size
        for _ in range(10):

            c5_new = _eval(h_new)
            if c5_new <= c5 + eta * g_norm * 1.01:  # Armijo-like
                h = h_new
                c5 = c5_new
                break
            eta *= 0.5
        
        # Decay temperature
        gamma *= 0.995
    return h, c5

def _lbfgs_smooth(h, alpha, maxit, deadline):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = _cvals(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def grad(x):
        c = _cvals(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        w = np.exp(s)
        w /= w.sum()
        return _grad_w(x, w, dx)

    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
    return _repair(res.x, n)

def _local_search(h, n_iter, rng, deadline, scale=1e-4):

    best_c = c5c

        top_k = np.argsort(c)[-K:]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_w(h, lam, dx):
    n = len(h)
    g = np.zeros(n)
    nz = np.where(lam > 1e-12)[0]
    for k in nz:
        sh = (n - 1) - k
        wk = lam[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _saddle(h, g0, g1, nit, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = _eval(h)
    eta = 0.03
    for it in range(nit):
        if time.time() > deadline - 25:
            break
        frac = it / max(nit - 1, 1)
        gamma = g0 + (g1 - g0) * frac
        c = _cvals(h)
        cs = c - c.max()
        lam = np.exp(gamma * cs)
        ls = lam.sum()
        lam = lam / ls if ls > 0 else np.ones(2*n-1) / (2*n-1)
        g = _grad_w(h, lam, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        best_h, best_c = h, c5
        for s in [eta, 0.5*eta, 0.25*eta, 2.0*eta]:
            ht = _repair(h - s * g, n)
            ct = _eval(ht)
            if ct < best_c:
                best_h, best_c = ht, ct
                if ct < c5 - 1e-15:
                    eta = min(eta * 1.3, 0.1)
        if best_c < c5:
            h, c5 = best_h, best_c
        else:
            eta *= 0.7
    return h, c5

def _lbfgs(h, alpha, mi, deadline):
    n = len(h)
    dx = 2.0 / n
    def ob(x):
        c = _cvals(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def gr(x):
        c = _cvals(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        w = np.exp(s)
        w /= w.sum()
        return _grad_w(x, w, dx)
    bnds = [(0.0, 1.0)] * n
    res = minimize(ob, h, jac=gr, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
    return _repair(res.x, n)

def _topk(h, K, mr, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(mr):
        if time.time() > deadline - 15:

        c = _cvals(h)
        tk = np.argsort(c)[-K:]
        lam = np.zeros(2*n-1)
        lam[tk] = 1.0 / K
        g = _grad_w(h, lam, dx)
        imp = False

            hn = _repair(h - st * g, n)
            cn = _eval(hn)
            if cn < c5c - 1e-15:
                h, c5c, imp = hn, cn, True
                break
        if not imp:
            break
    return h, c5c

def _loc(h, ni, rng, dl, sc):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_w(h, lam, dx):
    n = len(h)
    g = np.zeros(n)
    nz = np.where(lam > 1e-12)[0]
    for k in nz:
        sh = (n - 1) - k
        wk = lam[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _saddle(h, g0, g1, nit, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = _eval(h)
    eta = 0.03
    for it in range(nit):
        if time.time() > deadline - 25:
            break
        frac = it / max(nit - 1, 1)
        gamma = g0 + (g1 - g0) * frac
        c = _cvals(h)
        cs = c - c.max()
        lam = np.exp(gamma * cs)
        ls = lam.sum()
        lam = lam / ls if ls > 0 else np.ones(2 * n - 1) / (2 * n - 1)
        g = _grad_w(h, lam, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        best_h, best_c = h, c5
        for s in [eta, 0.5 * eta, 0.25 * eta, 2.0 * eta, 4.0 * eta]:
            ht = _repair(h - s * g, n)
            ct = _eval(ht)
            if ct < best_c:
                best_h, best_c = ht, ct
        if best_c < c5 - 1e-16:
            h, c5 = best_h, best_c
            eta = min(eta * 1.15, 0.1)
        else:
            eta *= 0.75
    return h, c5

def _lbfgs(h, alpha, mi, deadline):
    n = len(h)
    dx = 2.0 / n
    def ob(x):
        c = _cvals(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def gr(x):
        c = _cvals(x)
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 0)
        w = np.exp(s)
        w /= w.sum()
        return _grad_w(x, w, dx)
    bnds = [(0.0, 1.0)] * n
    res = minimize(ob, h, jac=gr, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
    return _repair(res.x, n)

def _topk(h, K, mr, deadline):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    for _ in range(mr):
        if time.time() > deadline - 15:
            break
        c = _cvals(h)
        tk = np.argsort(c)[-K:]
        lam = np.zeros(2 * n - 1)
        lam[tk] = 1.0 / K
        g = _grad_w(h, lam, dx)
        imp = False
        for st in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
            hn = _repair(h - st * g, n)
            cn = _eval(hn)
            if cn < c5c - 1e-15:
                h, c5c, imp = hn, cn, True
                break
        if not imp:
            break
    return h, c5c

def _loc(h, ni, rng, dl, sc):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(ni):
        if time.time() > dl - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = sc * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
            h[i], h[j] = a, b
            c = _eval(h)
            if c < c5c:
                c5c = c
                if c < bc:
                    bc, bh = c, h.copy()
            else:
                h[i], h[j] = a - d, b + d
    return bh, bc

def _shake_reopt(h, rng, dl):

    pert = _repair(h + 0.02 * rng.randn(n), n)
    c_p = _eval(pert)

        h2, c2 = _saddle(pert, 100, 2000, 200, dl)
        if c2 < _eval(h):
            return h2, c2
    return h, _eval(h)
