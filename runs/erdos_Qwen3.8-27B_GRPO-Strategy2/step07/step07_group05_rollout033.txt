def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _coord_descent_step(h, n, seed):
    """One pass of coordinate descent, optimizing each h[i] in turn."""
    rng = np.random.RandomState(seed)

        a_vals = i + ks - (n - 1)
        # Exclude k where a == i (self), i.e., k == n-1
        mask = ks != (n - 1)
        delta[ks[mask]] += (1.0 - h[a_vals[mask]])
    # Case 2: h[i] in (1-h[i]), partner b = i-k+(n-1)
    k2_lo = max(0, i)  # k > i-1 → k >= i
    k2_hi = min(nz1 - 1, i + n - 1)

        # Exclude k where b == i (self), i.e., k == n-1

        delta[ks[mask]] -= h[b_vals[mask]]
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _delta_for_coord(i, h, n):
    nz1 = 2 * n - 1
    delta = np.zeros(nz1)
    k1_lo = max(0, n - 1 - i)
    k1_hi = min(nz1 - 1, 2 * n - 2 - i)
    if k1_lo <= k1_hi:
        ks = np.arange(k1_lo, k1_hi + 1)
        a_vals = i + ks - (n - 1)
        mask = ks != (n - 1)
        delta[ks[mask]] += (1.0 - h[a_vals[mask]])
    k2_lo = max(0, i)
    k2_hi = min(nz1 - 1, i + n - 1)
    if k2_lo <= k2_hi:
        ks = np.arange(k2_lo, k2_hi + 1)
        b_vals = i - ks + (n - 1)
        mask = ks != (n - 1)
        delta[ks[mask]] -= h[b_vals[mask]]
    return delta

def _coord_sweep(h, n, dx, seed, nsweeps, check_fn):
    rng = np.random.RandomState(seed)
    for sweep in range(nsweeps):
        if check_fn():
            break
        c_curr = np.correlate(h, 1.0 - h, mode='full') * dx
        order = rng.permutation(n)
        for ii in range(n):
            i = order[ii]
            old = h[i]
            delta = _delta_for_coord(i, h, n) * dx
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(c_curr + (m1 - old) * delta)
                f2 = np.max(c_curr + (m2 - old) * delta)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            if abs(t_opt - old) > 1e-14:
                c_curr += (t_opt - old) * delta
                h[i] = t_opt
    return h

def _grad_lags(h, lags):
    n = len(h)
    g = np.zeros(n)
    for k in lags:
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mlgd(h, rounds, k_top, check_fn):
    n = len(h)
    dx = 2.0 / n
    c5c = _eval(h)
    steps = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c5c = c.max()
        top_k = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top_k) * dx / k_top
        improved = False
        for si in range(len(steps)):
            h_new = _project_sum(h - steps[si] * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-15:
                h = h_new
                c5c = c5n
                improved = True
                steps[si] = min(steps[si] * 1.5, 0.12)
                break
        if not improved:
            steps = [s * 0.45 for s in steps]
            if min(steps) < 5e-6:
                break
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _lsx_obj(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 100000.0
    sc = np.clip(a * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / a

def _lsx_grad(x, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = 100000.0
    sc = np.clip(a * (c - m), -300, 300)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))
