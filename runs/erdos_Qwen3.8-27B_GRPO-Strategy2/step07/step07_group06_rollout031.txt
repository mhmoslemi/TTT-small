def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    g[n-1-i : 2*n-1-i] += (1.0 - h)
    g[i : n+i] -= h[::-1]
    return g * (2.0 / n)

    x = np.linspace(0, 2, n0, endpoint=False)
    for freq in [1, 2, 3, 5, 7, 11, 15, 23, 31]:
        for amp in [0.005, 0.015, 0.04, 0.08, 0.15]:
            hp = _proj_to_sum(h + amp * np.sin(2 * np.pi * freq * x), n0 * 0.5)
            cp = _eval(hp)
            if cp < best_c5:
                best_c5 = cp; best_h = hp.copy()
    for amp in [0.03, 0.08, 0.15, 0.3, 0.6]:
        hp = _proj_to_sum(np.clip(h + amp * rng.randn(n0), 0, 1), n0 * 0.5)
        cp = _eval(hp)
        if cp < best_c5:
            best_c5 = cp; best_h = hp.copy()

    # Phase 1: Annealed CD (main exploration)
    dl1 = t0 + min(350, budget_s * 0.40)
    h1, c1 = _cd_anneal(best_h.copy(), n0, 500000, rng, t0, dl1, top_k=6, temp_init=0.005)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Sweep CD (systematic)
    dl2 = t0 + min(550, budget_s * 0.60)
    h2, c2 = _sweep_cd(best_h.copy(), n0, 100, rng, t0, dl2)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Another annealed CD round
    dl3 = t0 + min(750, budget_s * 0.78)
    h3, c3 = _cd_anneal(best_h.copy(), n0, 400000, rng, t0, dl3, top_k=8, temp_init=0.002)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Sweep again
    dl4 = t0 + min(900, budget_s * 0.90)
    h4, c4 = _sweep_cd(best_h.copy(), n0, 80, rng, t0, dl4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Fine polish
    dl5 = t0 + (budget_s - 10)
    h5, c5 = _fine_polish(best_h.copy(), n0, 200000, rng, t0, dl5)
    if c5 < best_c5:
        best_c5, best_h = c5, h5

    # Final repair and verify
    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
