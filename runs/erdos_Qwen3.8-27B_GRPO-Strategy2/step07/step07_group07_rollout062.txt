    n = len(h)

        w = 1.0 if weights is None else weights[idx]
        g += w * _grad_for_shift(h, t)
    if weights is None:
        g /= len(topk_shifts)
    dx = 2.0 / n
    return g * dx

def _corrs(h):

    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:

            hi = mid

    h = np.clip(h, 0.0, 1.0).copy()

    g = np.zeros(n)
    if t == 0:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _grad_raw(h, t):
    n = len(h)
    g = np.zeros(n)
    if t == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[t:] += 1.0 - h[:n - t]
        g[:n - t] -= h[t:]
    return g

def _sg_run(h, n_iter, k0, kmax, step0, beta, temp, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy()
    v = np.zeros(n)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    step = step0
    stall = 0
    k = k0
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-k:]
        cw = np.exp(temp * (c[topk] - c[topk].max()))
        cw /= cw.sum()
        g = np.zeros(n)
        for j in range(k):
            t = abs(int(topk[j]) - (n - 1))
            g += cw[j] * _grad_raw(h_c, t)
        g *= 2.0 / n
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            best_c = c_try
            best_h = h_try.copy()
            h_c = h_try
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            stall += 1
            if stall > 25:
                k = min(k + 2, kmax)
                # Small random perturbation to escape

                    h_c = _proj_sum(h_c + 1e-4 * np.random.randn(n), n * 0.5)
                    v *= 0.5
            if step < 1e-10:
                break
    return best_h, best_c

def _obj_pn(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.sum(c**p)**(1.0/p) + pen*(h.sum()-n*0.5)**2

def _grad_pn(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    cp = np.maximum(c, 1e-30)
    S = cp.sum()
    fac = S**(1.0/p - 1.0)
    w = fac * cp**(p-1)
    g = np.zeros(n)
    for ki in range(2*n-1):
        wk = w[ki] * dx
        if wk < 1e-20:
            continue
        g += wk * _grad_raw(h, abs(ki-(n-1)))
    g += 2.0*pen*(h.sum()-n*0.5)
    return g

def _lbfgs(h, p, pen, mi):
    n = len(h)
    try:
        r = minimize(_obj_pn, h, jac=_grad_pn, args=(p, pen),
                     method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                     options={'maxiter':mi,'ftol':1e-18,'gtol':1e-14,'maxls':50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _loc2(h, niter, rng, el_fn, bc_fn):

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _grad_raw(h, t):
    n = len(h)
    g = np.zeros(n)
    if t == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[t:] += 1.0 - h[:n - t]
        g[:n - t] -= h[t:]
    return g

def _sg_run(h, n_iter, k0, kmax, step0, beta, temp, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy()
    v = np.zeros(n)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    step = step0
    stall = 0
    k = k0
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-k:]
        cw = np.exp(temp * (c[topk] - c[topk].max()))
        cw /= cw.sum()
        g = np.zeros(n)
        for j in range(k):
            t = abs(int(topk[j]) - (n - 1))
            g += cw[j] * _grad_raw(h_c, t)
        g *= 2.0 / n
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            best_c = c_try
            best_h = h_try.copy()
            h_c = h_try
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            stall += 1
            if stall > 25:
                k = min(k + 2, kmax)
                if stall > 60:
                    h_c = _proj_sum(h_c + 1e-4 * np.random.randn(n), n * 0.5)
                    v *= 0.3
            if step < 1e-11:
                break
    return best_h, best_c

def _obj_pn(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.sum(c**p)**(1.0/p) + pen*(h.sum()-n*0.5)**2

def _grad_pn(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    cp = np.maximum(c, 1e-30)
    S = cp.sum()
    fac = S**(1.0/p - 1.0)
    w = fac * cp**(p-1)
    g = np.zeros(n)
    for ki in range(2*n-1):
        wk = w[ki] * dx
        if wk < 1e-20:
            continue
        g += wk * _grad_raw(h, abs(ki-(n-1)))
    g += 2.0*pen*(h.sum()-n*0.5)
    return g

def _lbfgs(h, p, pen, mi):
    n = len(h)
    try:
        r = minimize(_obj_pn, h, jac=_grad_pn, args=(p, pen),
                     method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                     options={'maxiter':mi,'ftol':1e-18,'gtol':1e-14,'maxls':50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _loc2(h, niter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy()
    bcst = _eval(h_c)
    bh = h_c.copy()
    sc = 5e-5
    for i in range(niter):
        if bc_fn():
            break
        a = rng.randint(n)
        b = (a+1+rng.randint(0,10))%n
        d = rng.randn()*sc
        na, nb = h_c[a]+d, h_c[b]-d
        if 0<=na<=1 and 0<=nb<=1:
            h_c[a],h_c[b] = na,nb
            cn = _eval(h_c)
            if cn < bcst:
                bcst = cn; bh = h_c.copy()
            else:
                h_c[a],h_c[b] = na-d, nb+d
        if i%4000==0:
            sc *= 0.65
    return bh, bcst

def _loc3(h, niter, rng, el_fn, bc_fn):

        if i%6000==0:
            sc *= 0.6
    return bh, bcst

    def el():
        return time.time() - t0
