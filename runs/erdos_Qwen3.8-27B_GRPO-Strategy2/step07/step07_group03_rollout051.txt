import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, n, dx):
    """Gradient of c_k with respect to h, where k is index in full correlation."""
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _nesterov_topk(h, topk, alpha, steps, step0, t_end):
    """Nesterov optimization focusing on top-k lags."""
    n = len(h)
    dx = 2.0 / n
    mu = 0.82
    step = step0
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _eval(h)
    
    for it in range(steps):
        if time.time() > t_end:
            break
        c = _corr_all(h)
        top_idx = np.argsort(c)[-topk:]
        # Smooth objective: log-sum-exp of top-k values
        cv = c[top_idx]
        m = cv.max()
        s = alpha * (cv - m)
        s = np.clip(s, -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        
        # Gradient: weighted sum of top-k lag gradients
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            if wi < 1e-10:
                continue
            g += wi * _grad_lag(h, ki, n, dx)
        
        # Nesterov step
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.02, step0 * 10)
        else:
            step *= 0.5
            if step < 1e-11:
                break
            h = _proj_to_sum(h - step * v, n * 0.5)
    
    return best_h, best_c5

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    n_pairs_per_sweep = n * 3
    no_improve = 0

        a_vals = c[top_idx]

            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j:

            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:

            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(20):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True

            new_c5 = _eval(h)

                best_c5 = new_c5

        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:

    best_c5 = _eval(h)
    best_h = h.copy()
    
    while time.time() < t_end:
        c = _corr_all(h)
        top_idx = np.argsort(c)[-8:]
        grads = np.array([_grad_lag(h, k, n, dx=2.0/n) for k in top_idx])
        a_vals = c[top_idx]
        
        i = rng.randint(n)
        # Try pushing h[i] towards 0 or 1
        target_val = 0.0 if h[i] < 0.5 else 1.0
        d_target = target_val - h[i]
        if abs(d_target) < 1e-6:
            continue
        
        # Find j to compensate
        j_candidates = [(i + 1 + rng.randint(0, 20)) % n for _ in range(10)]
        best_d = 0
        best_f = np.max(a_vals)
        
        for j in j_candidates:
            if j == i:
                continue
            b = grads[:, i] - grads[:, j]
            # h[i] -> h[i] + d, h[j] -> h[j] - d
            # Check bounds
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if abs(d_target) > d_hi - d_lo:
                d_target_clipped = np.clip(d_target, d_lo, d_hi)
            else:
                d_target_clipped = d_target
            if d_lo + 1e-13 >= d_hi - 1e-13:
                continue
            f_new = np.max(a_vals + b * d_target_clipped)
            if f_new < best_f:
                best_f = f_new
                best_d = d_target_clipped
                best_j = j
        
        if best_d != 0:
            h[i] += best_d
            h[best_j] -= best_d
            c5_new = _eval(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            else:
                h[i] -= best_d
                h[best_j] += best_d
    
    return best_h, best_c5

def _local_search(h, n_iter, rng, t_end):

    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):

                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_all(h):

def _proj_to_sum(h, target):

    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        g[i1] += (1.0 - h[i1 + sh]) * dx

        g[i1] += (1.0 - h[i1 - sp]) * dx

        cv = c[top_idx]

        s = np.clip(alpha * (cv - m), -400, 400)

        w = np.exp(s)

        d_tgt = tgt - h[i]
        best_d = 0.0
        best_f = np.max(a_vals)
        best_j = -1
        for _ in range(12):
            j = (i + 1 + rng.randint(0, 30)) % n
            if j == i:
                continue
            b = grads[:, i] - grads[:, j]
            cl = max(-h[i], h[j] - 1.0)
            ch = min(1.0 - h[i], h[j])
            if cl + 1e-12 >= ch - 1e-12:
                continue
            d_c = np.clip(d_tgt, cl + 1e-12, ch - 1e-12)
            f_n = np.max(a_vals + b * d_c)
            if f_n < best_f:
                best_f = f_n
                best_d = d_c
                best_j = j
        if best_j >= 0 and abs(best_d) > 1e-8:
            h[i] += best_d
            h[best_j] -= best_d
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
            else:
                h[i] -= best_d
                h[best_j] += best_d
    return best_h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 2e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)
    T = budget_s

    # Phase 1: Top-k Nesterov (broad to narrow)
    for topk, alpha, step0, frac in [(30, 200, 0.004, 0.05), (15, 1000, 0.003, 0.05), (8, 5000, 0.002, 0.05), (5, 20000, 0.0015, 0.05)]:
        if time.time() > t0 + T * 0.24:
            break
        h = best_h.copy()
        h, c5v = _nesterov_topk(h, topk, alpha, 400, step0, t0 + T * (0.05 + 0.05 * (topk / 30)), 0.82)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coordinate descent (multiple top-T)
    for top_T in [4, 6, 3, 8]:
        if time.time() > t0 + T * 0.42:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, T * 0.055, top_T, rng, t0 + T)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: High-precision Nesterov
    for topk, alpha, step0 in [(10, 10000, 0.001), (20, 2000, 0.0015), (50, 300, 0.002), (5, 50000, 0.0008)]:
        if time.time() > t0 + T * 0.62:
            break
        h = best_h.copy()
        h, c5v = _nesterov_topk(h, topk, alpha, 300, step0, min(time.time() + 20, t0 + T * 0.62), 0.85)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: More coordinate descent
    for top_T in [5, 3, 7]:
        if time.time() > t0 + T * 0.78:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, T * 0.05, top_T, rng, t0 + T)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Binary push
    if time.time() < t0 + T * 0.90:
        h = best_h.copy()
        h, c5v = _binary_push(h, n, rng, min(time.time() + T * 0.10, t0 + T * 0.90))
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Fine local search
    if time.time() < t0 + T - 8:
        remaining = max(0.0, (t0 + T) - time.time() - 5)
        n_tries = min(int(remaining / 0.003), 60000)
        if n_tries > 300:
            h = best_h.copy()
            best_h, best_c5 = _local_search(h, n_tries, rng, t0 + T - 3)

    # Phase 7: Final CD polish
    if time.time() < t0 + T - 2:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, max(0.5, t0 + T - time.time() - 1), 4, rng, t0 + T)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
