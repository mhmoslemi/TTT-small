from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj_sum(h, n / 2.0)

def _lse_obj(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    val = cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta
    return val + pen * (h.sum() - n * 0.5)**2

def _lse_grad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    ec = np.exp(beta * (c - cm))
    w = ec / ec.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            if sh < n:
                g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, beta, pen, maxiter, t0, budget):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h.copy(), jac=_lse_grad,
                       args=(beta, pen), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 30})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _topk_nesterov(h, n_iter, k, step0, beta_mom, t0, budget):

    h_c = h.copy().astype(float)
    v = np.zeros(n)
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 10:
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-k:]
        # Gradient of sum of top-k correlations
        g = np.zeros(n)
        dx = 2.0 / n

            if sh >= 0:
                if sh < n:
                    g[:n-sh] += dx * (1.0 - h_c[sh:])
                if sh > 0:
                    g[sh:] -= dx * h_c[:n-sh]
            else:
                sp = -sh
                g[sp:] += dx * (1.0 - h_c[:n-sp])
                g[:n-sp] -= dx * h_c[sp:]
        g /= len(topk)
        gn = np.linalg.norm(g)

            if step < 1e-8 or stall > 100:
                break
    return best_h, best_c

def _coord_descent(h, n_iter, rng, t0, budget):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    c = _corrs(h_c)
    top = np.argsort(c)[-20:]
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        # Find most violated constraint
        k = top[rng.randint(len(top))]
        sh = (n - 1) - k
        # Try to reduce c[k] by adjusting h
        # c[k] = sum of h(i)*(1-h(i+sh)) for valid i
        # To reduce: increase h(i+sh) where h(i) is large, or decrease h(i)
        if sh >= 0:
            idx_a = np.arange(n - sh)
            idx_b = idx_a + sh
        else:
            sp = -sh
            idx_b = np.arange(sp, n)
            idx_a = idx_b - sp
        # Find pair (a, b) that most reduces c[k]
        # d c[k] / d h(a) = (1-h(b)) - h(a-sh_offset)... 
        # For simplicity, do random pair swap
        i = rng.randint(n)
        j = (i + rng.randint(1, 10)) % n
        # Binary search for optimal transfer
        d_max = min(h_c[i], 1.0 - h_c[j])
        if d_max <= 0:
            continue
        # Try a few transfer amounts
        for frac in [0.3, 0.5, 0.7, 1.0]:
            d = d_max * frac * (0.5 + 0.5 * rng.rand())
            ni, nj = h_c[i] + d, h_c[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_c[i], h_c[j] = ni, nj
                c_new = _eval(h_c)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_c.copy()
                    break
                else:
                    h_c[i], h_c[j] = h_c[i] - d, h_c[j] + d
        # Re-update top set periodically
        if it % 500 == 0:
            c = _corrs(h_c)
            top = np.argsort(c)[-20:]
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - n * 0.5) > 1e-10:
        h = _proj_sum(h, n * 0.5)
    return h

def _lse_obj(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    val = cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta
    return val + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    ec = np.exp(beta * (c - cm))
    w = ec / ec.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            if sh < n:
                g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs_lse(h, beta, pen, maxiter, t0, budget):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    if time.time() - t0 > budget - 10:
        return _repair(h, n)
    try:
        res = minimize(_lse_obj, h.copy(), jac=_lse_grad,
                       args=(beta, pen), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _topk_grad(h, topk):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in topk:
        sh = (n - 1) - k
        if sh >= 0:
            if sh < n:
                g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return g / len(topk)

def _nesterov_topk(h, n_iter, k_start, k_max, step0, t0, budget):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)

            k = min(k + 2, k_max)
        topk = np.argsort(c)[-k:]
        g = _topk_grad(h_c, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = 0.8 * v + 0.2 * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try

        else:

            stall += 1
            if step < 1e-9 or stall > 120:
                break
    return best_h, best_c
