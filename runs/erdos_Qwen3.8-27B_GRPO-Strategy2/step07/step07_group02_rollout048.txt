            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _local_4pt(h, n_iter, rng, deadline, scale=2e-5):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i1 = rng.randint(n)
        i2 = (i1 + rng.randint(2, 5)) % n
        i3 = (i1 + rng.randint(6, 10)) % n
        i4 = (i1 + rng.randint(11, 20)) % n
        if len({i1, i2, i3, i4}) < 4:
            continue
        d1, d2 = scale * rng.randn(), scale * rng.randn()
        vals = {i1: h[i1] + d1, i2: h[i2] + d2, i3: h[i3] - d1, i4: h[i4] - d2}
        if all(0.0 <= v <= 1.0 for v in vals.values()):
            for idx, v in vals.items():
                h[idx] = v
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                for idx in (i1, i2, i3, i4):
                    h[idx] = h[idx] - vals[idx] + h[idx]
                h[i1] -= d1; h[i2] -= d2; h[i3] += d1; h[i4] += d2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: LP-based linearized descent
    if time.time() < deadline - 500:
        h_lp, c5_lp = _lp_descent(best_h.copy(), n0, 300, deadline)
        if c5_lp < best_c5:
            best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 2: L-BFGS with increasing alpha (annealing)
    h = best_h.copy()
    for alpha in [1000, 5000, 20000, 80000, 300000, 1000000]:
        if time.time() > deadline - 350:
            break
        h, c5v = _lbfgs_smooth(h, alpha, 3000, deadline)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 3: More LP iterations from improved point
    if time.time() < deadline - 250:
        h_lp, c5_lp = _lp_descent(best_h.copy(), n0, 200, deadline)
        if c5_lp < best_c5:
            best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 4: Subgradient with momentum
    if time.time() < deadline - 200:
        h_sg, c5_sg = _subgrad_mom(best_h.copy(), 800, 0.015, 1e-8, 5, deadline)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 5: Top-lag descent multiple configs
    if time.time() < deadline - 150:
        for nl in [2, 3, 4, 6]:
            if time.time() > deadline - 150:
                break
            h_tl, c5_tl = _top_lag(best_h.copy(), nl, 400, deadline)
            if c5_tl < best_c5:
                best_c5, best_h = c5_tl, h_tl.copy()

    # Phase 6: Perturbation + brief optimization
    if time.time() < deadline - 100:
        for _ in range(4):
            if time.time() > deadline - 100:
                break
            h_p = _repair(best_h + 0.005 * rng.randn(n0), n0)
            if _eval(h_p) > best_c5 + 0.002:
                continue
            h_p = _lbfgs_smooth(h_p, 200000, 1500, deadline)[0]
            c5p = _eval(h_p)
            if c5p < best_c5:
                best_c5, best_h = c5p, h_p.copy()

    # Phase 7: Final LP
    if time.time() < deadline - 80:
        h_lp, c5_lp = _lp_descent(best_h.copy(), n0, 150, deadline)
        if c5_lp < best_c5:
            best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 8: Final subgradient tight
    if time.time() < deadline - 50:
        h_sg, c5_sg = _subgrad_mom(best_h.copy(), 1000, 0.004, 1e-10, 3, deadline)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 9: Top-lag final
    if time.time() < deadline - 30:
        h_tl, c5_tl = _top_lag(best_h.copy(), 3, 200, deadline)
        if c5_tl < best_c5:
            best_c5, best_h = c5_tl, h_tl.copy()

    # Phase 10: 4-point local search
    if time.time() < deadline - 15:
        h_ls, c5_ls = _local_4pt(best_h.copy(), 15000, rng, deadline, scale=3e-5)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    # Phase 11: 2-point ultra-fine polish
    if time.time() < deadline - 3:
        h_ls, c5_ls = _local_2pt(best_h.copy(), 60000, rng, deadline, scale=2e-6)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
