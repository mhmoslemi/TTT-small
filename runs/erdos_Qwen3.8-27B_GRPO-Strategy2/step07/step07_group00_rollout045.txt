import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        k = idx[ii]; sh = (n-1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _lbfgs(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    try:
        r = minimize(lambda x: _smax(x, alpha, topk), h,
                     jac=lambda x: _grad_smax(x, alpha, topk),
                     method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'gtol': 1e-14, 'ftol': 1e-18,
                               'maxls': 20, 'maxfun': maxit * 3})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _pgrad(h, alpha, topk, steps, step0, deadline):
    """Projected gradient with backtracking."""
    n = len(h); f = _smax(h, alpha, topk)
    v = np.zeros(n); beta = 0.9; step = step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        for _bt in range(20):
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax(hn, alpha, topk)
            if fn < f:
                h = hn; f = fn; step = min(step * 1.1, step0 * 10); break
            step *= 0.5
        else:
            step = max(step, 1e-10)
    return _repair(h, n)

def _upsample(h, factor):

    x = np.arange(m) * n / m
    i0 = np.floor(x).astype(int); i1 = (i0 + 1) % n
    fr = x - i0
    return h[i0] * (1 - fr) + h[i1] * fr

    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _corr(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy(); ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _pgrad(h, alpha, topk, steps, step0, deadline):
    n = len(h); f = _smax(h, alpha, topk)
    v = np.zeros(n); beta = 0.88; step = step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        improved = False
        for _bt in range(25):
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax(hn, alpha, topk)
            if fn < f:
                h = hn; f = fn; step = min(step * 1.08, step0 * 20); improved = True; break
            step *= 0.45
        if not improved:
            step = max(step, 1e-12)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    try:
        r = minimize(lambda x: _smax(x, alpha, topk), h,
                     jac=lambda x: _grad_smax(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, factor):
    n = len(h); m = n * factor
    x = np.arange(m) * n / m
    i0 = np.floor(x).astype(int); i1 = (i0 + 1) % n
    fr = x - i0
    return h[i0] * (1 - fr) + h[i1] * fr

def _downsample(h, factor):
    n = len(h); m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn; c5c = c5n
            if c5n < bc: bc = c5n; bh = h.copy()
    return bh, bc

def _polish2(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    scale = 3e-6
    for it in range(niter):
        if time.time() > deadline: break

        i = rng.randint(nc); j = (i + 1 + rng.randint(0, 6)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj; c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc: bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _polish3(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline: break
        i = rng.randint(nc)
        j = (i + rng.randint(2, 5)) % nc
        k = (i + rng.randint(6, 12)) % nc
        d1 = 1.5e-6 * rng.randn(); d2 = 1.5e-6 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0<=ni<=1 and 0<=nj<=1 and 0<=nk<=1:
            h[i],h[j],h[k] = ni,nj,nk; c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc: bc = c5n; bh = h.copy()
            else:
                h[i],h[j],h[k] = ni-d1, nj-d2, nk+d1+d2
    return bh, bc

def _basin_jump(h, rng, deadline, n_jumps, jump_scale):
    n = len(h)
    for _ in range(n_jumps):
        if time.time() > deadline: break
        hp = h.copy()
        mask = rng.rand(n) < 0.15
        hp[mask] += jump_scale * rng.randn(mask.sum())
        hp = _repair(np.clip(hp, 0, 1), n)
        hp = _pgrad(hp, 50000, 8, 800, 0.003, deadline)
        hp = _pgrad(hp, 300000, 4, 500, 0.001, deadline)
        if _eval(hp) < _eval(h):
            h = hp
    return h
