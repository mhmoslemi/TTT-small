                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_2pt(h, n_tries, step, rng, t0, tlimit):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_tries):
        if time.time() - t0 > tlimit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c:
                c5c = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    def _update(hc, fc):
        nonlocal best_h, best_c5
        if fc < best_c5:
            best_c5 = fc
            best_h = hc.copy()

    # Phase 1: FISTA with increasing alpha (smooth start)
    for alpha, niter in [(1000, 8000), (5000, 6000), (20000, 5000), (80000, 5000)]:
        if time.time() - t0 > budget_s * 0.35:
            break
        h2, f2 = _fista(best_h.copy(), alpha, niter, t0, budget_s * 0.35)
        _update(h2, f2)

    # Phase 2: FISTA with high alpha (sharpen)
    for alpha in [200000, 500000]:
        if time.time() - t0 > budget_s * 0.55:
            break
        h2, f2 = _fista(best_h.copy(), alpha, 6000, t0, budget_s * 0.55)
        _update(h2, f2)

    # Phase 3: SLSQP on active lags
    if time.time() - t0 < budget_s * 0.7:
        c_full = _corr_vec(best_h)
        top_idx = np.argsort(c_full)[-8:]
        h2 = _slsqp_active(best_h.copy(), top_idx, 50000, 800, t0, budget_s * 0.7)
        _update(h2, _eval(h2))
        c_full = _corr_vec(best_h)
        top_idx = np.argsort(c_full)[-5:]
        h2 = _slsqp_active(best_h.copy(), top_idx, 100000, 600, t0, budget_s * 0.75)
        _update(h2, _eval(h2))

    # Phase 4: Nesterov top-k polish
    if time.time() - t0 < budget_s * 0.85:
        h2, f2 = _nesterov_topk(best_h.copy(), 5, 6000, 0.001, t0, budget_s * 0.88)
        _update(h2, f2)
        h2, f2 = _nesterov_topk(best_h.copy(), 3, 4000, 0.0005, t0, budget_s * 0.92)
        _update(h2, f2)

    # Phase 5: Low-res escape and retry
    if time.time() - t0 < budget_s * 0.92:
        n_lo = n // 4
        h_lo = _downsample(best_h, 4)
        h_lo = _repair(h_lo, n_lo)
        for kt in [3, 8]:
            if time.time() - t0 > budget_s * 0.94:
                break
            h_lo, f_lo = _nesterov_topk(h_lo, kt, 3000, 0.01, t0, budget_s * 0.94)
        h_hi = _repair(_upsample(h_lo, 4), n)
        _update(h_hi, _eval(h_hi))

    # Phase 6: Fine local search
    if time.time() - t0 < budget_s - 20:
        remaining = budget_s - 25 - (time.time() - t0)
        ntries = int(remaining / 0.005)
        h2, f2 = _local_2pt(best_h.copy(), ntries, 5e-5, rng, t0, budget_s - 20)
        _update(h2, f2)

    # Phase 7: Ultra-fine 2-point
    if time.time() - t0 < budget_s - 5:
        remaining = budget_s - 5 - (time.time() - t0)
        ntries = int(remaining / 0.003)
        h2, f2 = _local_2pt(best_h.copy(), ntries, 1e-5, rng, t0, budget_s - 3)
        _update(h2, f2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
