        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad(w, h):
    """Gradient of weighted sum of correlations, given lag weights w."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s >= 0:
            m = n - s
            g[:m] += wk * (1.0 - h[s:])
            if s > 0:
                g[s:] -= wk * h[:m]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, n):
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _grad(w, h):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            g[:m] += wk * (1.0 - h[s:])
            if s > 0:
                g[s:] -= wk * h[:m]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
    return g

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    return _grad(w, x)

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        return _proj(res.x, n)
    except Exception:
        return h

def _subgrad(h, steps, step0, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.03 * i)
        c = _cvals(h)
        m = c.max()
        tol = max(5e-9, m * 1e-8)
        active = np.where(c >= m - tol)[0]
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = _grad(w, h)
        h = _proj(h - t * g, n)
    return h

            best_c5 = c5_ann
            best_h = best_h.copy()

    best_h = _proj(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
