        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = _proj(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
        f_new = _eval(h_new)
        if best_f is None or f_new < best_f:
            h = h_new
            best_f = f_new
    return h

def _coord_desc(h, n, dur_s, topT, rng, t_end):
    dx = 2.0 / n
    best_c5 = _eval(h)
    end_t = time.time() + dur_s
    noimp = 0
    while time.time() < min(end_t, t_end - 4):
        c = _eval_arr(h)
        top_idx = np.argsort(c)[-topT:]
        a_vals = c[top_idx]
        grads = np.array([_corr_grad(h, k, n, dx) for k in top_idx])
        imp = False
        for _ in range(max(200, n)):
            if time.time() > min(end_t, t_end - 4):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(3, n // 6))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            dlo = max(-h[i], h[j] - 1.0) + 1e-15
            dhi = min(1.0 - h[i], h[j]) - 1e-15
            if dlo >= dhi:
                continue
            lo, hi = dlo, dhi
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(a_vals + b * m1) < np.max(a_vals + b * m2):
                    hi = m2
                else:
                    lo = m1
            d = (lo + hi) * 0.5
            if abs(d) > 1e-15:
                h[i] += d
                h[j] -= d
                imp = True
        if imp:
            nc = _eval(h)
            if nc < best_c5 - 1e-13:
                best_c5 = nc
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1
        if noimp > 25:
            h += 5e-5 * rng.randn(n)
            h = _proj(h, n * 0.5)
            noimp = 0
        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj(h, n * 0.5)
    return h, best_c5

def _fine_tune(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    deltas = [1e-5, -1e-5, 3e-6, -3e-6, 1e-6, -1e-6, 5e-7, -5e-7]
    for _ in range(n_iter):

        j = (i + 1 + rng.randint(0, 3)) % n

            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _eval(h)
                if c < c5c - 1e-18:
                    c5c = c
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
                    break
    return best_h, best_c

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_arr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _corr_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] = (1.0 - h[sh:]) * dx
        if sh > 0: g[sh:] -= h[:cnt] * dx
    else:
        sp = -sh
        g[sp:] = (1.0 - h[:n - sp]) * dx
        g[:n - sp] -= h[sp:] * dx
    return g

def _adam_opt(h, alpha, steps, lr0, decay, t_end):
    n = len(h)
    m = np.zeros(n); v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    dx = 2.0 / n
    for t in range(1, steps + 1):
        if time.time() > t_end: break
        lr = lr0 / (1.0 + t * decay)

        mc = c.max()
        s = np.clip(alpha * (c - mc), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12: continue
            wk = w[k] * dx
            sh = (n - 1) - k
            if sh >= 0:
                cnt = n - sh
                g[:cnt] += wk * (1.0 - h[sh:])
                if sh > 0: g[sh:] -= wk * h[:cnt]
            else:
                sp = -sh
