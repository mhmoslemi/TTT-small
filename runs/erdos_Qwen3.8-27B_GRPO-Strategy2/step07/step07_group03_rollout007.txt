def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

    best_c5 = _eval(h); end_t = time.time() + dur; no_imp = 0
    while time.time() < min(end_t, t_end - 3):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        grads = np.zeros((top_T, n))
        for t, k in enumerate(top_idx):
            sh = k - (n - 1)
            if sh >= 0:
                i1 = np.arange(n - sh); grads[t, i1] = (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n); grads[t, i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n); grads[t, i1] = (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp); grads[t, i2] -= h[i2 + sp]
        a_vals = c[top_idx].copy(); improved = False
        n_pairs = n * 4
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_end - 3): break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n // 5)) % n
            if i == j: continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi: continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1); f2 = np.max(a_vals + b * x2)
            for _ in range(28):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt; h[j] -= d_opt
                a_vals += b * d_opt; improved = True
        if improved:
            nc = _eval(h)
            if nc < best_c5: best_c5 = nc; no_imp = 0
            else: no_imp += 1
        else: no_imp += 1
        if no_imp > 20:
            h += 1e-4 * rng.randn(n); h = _proj_to_sum(h, n * 0.5); no_imp = 0
        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _local_refine(h, n, rng, t_end, dur):
    dx = 2.0 / n
    best_c5 = _eval(h); best_h = h.copy()
    end_t = time.time() + dur; step_scale = 2e-3
    while time.time() < min(end_t, t_end - 2):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        for k_max in np.argsort(c)[-3:]:
            sh = k_max - (n - 1); g = np.zeros(n)
            if sh >= 0:
                i1 = np.arange(n - sh); g[i1] = (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n); g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n); g[i1] = (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp); g[i2] -= h[i2 + sp]
            for _ in range(15):
                h_new = _proj_to_sum(h - step_scale * g, n * 0.5)
                nc = _eval(h_new)
                if nc < best_c5:
                    h = h_new; best_c5 = nc; best_h = h.copy(); break
                step_scale *= 0.5
                if step_scale < 1e-11: break
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 4)) % n
                d = 5e-6 * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    if _eval(h) < best_c5: best_c5 = _eval(h); best_h = h.copy()
                    else: h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy(); best_c5 = _eval(h)
    # Phase 1: Nesterov with increasing alpha
    for alpha in [2000, 10000, 50000, 200000, 800000, 3000000]:
        if time.time() > t0 + budget_s * 0.15: break
        h = _nesterov(h, alpha, 600, 0.003, min(time.time() + 20, t0 + budget_s * 0.14))
        c5v = _eval(h)
        if c5v < best_c5: best_c5 = c5v; best_h = h.copy()
    h = best_h.copy()
    # Phase 2: Coordinate descent with high top_T
    for top_T in [16, 24, 32]:
        if time.time() > t0 + budget_s * 0.38: break
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, budget_s * 0.1, top_T, rng, t_end)
        if c5v < best_c5: best_c5 = c5v; best_h = h.copy()
    # Phase 3: Restarts with structured perturbations
    for r in range(3):
        if time.time() > t0 + budget_s * 0.60: break
        scale = 0.008 * (0.5 ** r)
        h = best_h + scale * rng.randn(n)
        h = _proj_to_sum(h, n * 0.5)
        h = _nesterov(h, 500000, 400, 0.002, min(time.time() + 18, t0 + budget_s * 0.58))
        h, c5v = _coord_desc(h, n, 20, 16, rng, t_end)
        if c5v < best_c5: best_c5 = c5v; best_h = h.copy()
    # Phase 4: Extended coordinate descent
    if time.time() < t0 + budget_s * 0.80:
        for top_T in [32, 48, 24]:
            if time.time() > t0 + budget_s * 0.78: break
            h = best_h.copy()
            h, c5v = _coord_desc(h, n, budget_s * 0.07, top_T, rng, t_end)
            if c5v < best_c5: best_c5 = c5v; best_h = h.copy()
    # Phase 5: Local refine on critical lags
    if time.time() < t0 + budget_s - 15:
        rem = (t0 + budget_s) - time.time() - 8
        h = best_h.copy()
        h, c5v = _local_refine(h, n, rng, t_end, max(5, rem * 0.35))
        if c5v < best_c5: best_c5 = c5v; best_h = h.copy()
    # Phase 6: Final CD polish
    if time.time() < t0 + budget_s - 3:
        rem = max(1, (t0 + budget_s) - time.time() - 2)
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, rem, 24, rng, t_end)
        if c5v < best_c5: best_c5 = c5v; best_h = h.copy()
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
