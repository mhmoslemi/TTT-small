    if t_rem > 130:
        h2 = _slsqp_fast(best_h, 50000, t0, t0 + min(100, t_rem * 0.2), 500)
        f2 = _eval(h2)
        if f2 < best_c5:
            best_c5, best_h = f2, h2.copy()

    # Phase 3: Adam-soft high alpha
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 80:
        h3, f3 = _nesterov(best_h, 200000, t0, t0 + min(60, t_rem * 0.12))
        if f3 < best_c5:
            best_c5, best_h = f3, h3

    # Phase 4: SLSQP high alpha
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 120:
        h4 = _slsqp_fast(best_h, 300000, t0, t0 + min(80, t_rem * 0.15), 400)
        f4 = _eval(h4)
        if f4 < best_c5:
            best_c5, best_h = f4, h4.copy()

    # Phase 5: Perturb-refine cycles
    t_rem = budget_s - (time.time() - t0)
    for _ in range(min(6, int(t_rem // 50))):
        if time.time() > budget_s - 80:
            break
        h5, f5 = _perturb_refine(best_h, rng, t0, budget_s - 60)
        if f5 < best_c5:
            best_c5, best_h = f5, h5.copy()

    # Phase 6: 3-point local search
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 30:
        best_h, best_c5 = _local_search(best_h, t0, t0 + t_rem * 0.55, rng, 3)

    # Phase 7: 2-point ultra-fine
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 10:
        best_h, best_c5 = _local_search(best_h, t0, t0 + t_rem - 3, rng, 2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
