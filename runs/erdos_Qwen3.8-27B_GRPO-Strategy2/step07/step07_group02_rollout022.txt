def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lp(h, alpha):

    n = len(h)
    dx = 2.0 / n
    c = _cvals(h)
    m = c.max()

def _grad_from_weights(h, w):

    n = len(h)

    grad = np.zeros(n)

    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_weights(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += wk * (1.0 - h[s:])
            if s > 0:
                g[s:] -= wk * h[:n - s]
        else:
            sp = -s
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lp_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lp_grad(x, alpha):
    n = len(x)
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= max(w.sum(), 1e-30)
    return _grad_weights(x, w)

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 1e-30)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    c = np.maximum(_cvals(x), 1e-30)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    w /= max(w.sum(), 1e-30)
    return _grad_weights(x, w)

def _tk_obj(x, nk, alpha):
    c = _cvals(x)
    m = c.max()
    idx = np.argsort(c)[-nk:]
    s = np.clip(alpha * (c[idx] - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _tk_grad(x, nk, alpha):
    n = len(x)
    c = _cvals(x)
    idx = np.argsort(c)[-nk:]
    w = np.zeros(2 * n - 1)
    s = np.clip(alpha * (c[idx] - c.max()), -500, 0)
    w[idx] = np.exp(s)
    w /= max(w.sum(), 1e-30)
    return _grad_weights(x, w)

def _lbfgs(h, obj, gfun, args, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 30:
        return h
    try:
        res = minimize(obj, h, jac=gfun, args=args, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _adam(h, obj, gfun, args, steps, lr, deadline):
    n = len(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for t in range(1, steps + 1):
        if time.time() > deadline - 10:
            break
        g = gfun(h, *args)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = _repair(h - lr * mh / (np.sqrt(vh) + eps), n)
    return h
