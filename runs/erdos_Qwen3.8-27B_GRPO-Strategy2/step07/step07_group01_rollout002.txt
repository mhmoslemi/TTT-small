    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        g += w[k] * _grad_lag(h, k, n)
    return g

def _nesterov(h0, top_k, sharp, n_steps, step0, t_end):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c = _eval(h0)
    step = step0
    for t in range(n_steps):
        if time.time() > t_end:
            break
        y = h + 0.75 * v
        g = _subgrad_top(y, top_k, sharp)
        h_new = _proj(y - step * g, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-16:
            v = h_new - h
            h = h_new
            best_h = h.copy()
            best_c = c_new
            step = min(step * 1.04, step0 * 6.0)
        else:
            took = False
            for _ in range(8):
                h_new = _proj(y - step * g, n * 0.5)
                c_new = _eval(h_new)
                if c_new < best_c - 1e-16:
                    v = h_new - h
                    h = h_new
                    best_h = h.copy()
                    best_c = c_new
                    step = min(step * 1.02, step0 * 6.0)
                    took = True
                    break
                step *= 0.5
            if not took:
                v = 0.5 * v
                step = min(step * 1.2, step0 * 6.0)
    return best_h, best_c

def _slsqp_smooth(h, alpha, maxiter, t_end):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            g += w[k] * _grad_lag(x, k, n)
        return g
    def _cons_fn(x):
        return x.sum() - n * 0.5
    def _cons_jac(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fn, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() < t_end:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    out = np.empty(len(h) * factor)
    for i in range(len(h)):
        out[i * factor:(i + 1) * factor] = h[i]
    return out

def _local_search(h, n_iter, rng, step, t_end):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        k = (i + 4 + rng.randint(0, 7)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution global optimization
    t_phase1_end = t0 + 0.15 * budget_s
    for ratio in [4, 2]:
        nc = n // ratio
        hc = _repair(_downsample(h, ratio), nc)
        for alpha in [500, 3000, 20000]:
            if time.time() > t_phase1_end:
                break
            hc = _slsqp_smooth(hc, alpha, 300, t_phase1_end)
        h_up = _repair(_upsample(hc, ratio), n)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Nesterov subgradient - aggressive top-2 focus
    t_phase2_end = t0 + 0.40 * budget_s
    for (tk, sh_, st, ns) in [(1, 20, 0.006, 2000), (2, 50, 0.004, 1500), (2, 100, 0.003, 1000)]:
        if time.time() > t_phase2_end:
            break
        h_new, c_new = _nesterov(best_h.copy(), tk, sh_, ns, st, t_phase2_end)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 3: SLSQP smooth max polish
    t_phase3_end = t0 + 0.65 * budget_s
    for alpha in [3000, 15000, 80000, 400000, 1500000, 8000000]:
        if time.time() > t_phase3_end:
            break
        h_new = _slsqp_smooth(best_h.copy(), alpha, 1500, t_phase3_end)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: More Nesterov after SLSQP (cycling)
    if _el() < 0.72 * budget_s:
        for (tk, sh_, st, ns) in [(1, 30, 0.002, 800), (3, 80, 0.001, 600)]:
            if time.time() > t_end - 0.20 * budget_s:
                break
            h_new, c_new = _nesterov(best_h.copy(), tk, sh_, ns, st, t_end - 0.20 * budget_s)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 5: Intensive 3-point local search (decreasing step)
    t_local_end = t_end - 0.08 * budget_s
    remaining = max(0, t_local_end - time.time())
    for scale in [8e-6, 4e-6, 2e-6, 1e-6]:
        if time.time() > t_local_end:
            break
        ntries = int(remaining * 0.25 / max(0.001, t_local_end - time.time()))
        ntries = max(ntries, 20000)
        best_h, best_c5 = _local_search(best_h, ntries, rng, scale, t_local_end)
        remaining = t_local_end - time.time()

    # Phase 6: Ultra-fine 2-point polish
    t_polish_end = t_end - 1.0
    if _el() < 0.92 * budget_s:
        nc2 = len(best_h)
        h2 = best_h.copy()
        c5c = best_c5
        n_polish = int(min((t_polish_end - time.time()) / 0.0006, 300000))
        step_polish = 3e-7
        for _ in range(n_polish):
            if time.time() > t_polish_end:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 2)) % nc2
            d = step_polish * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                        step_polish *= 0.9
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
