        ht = _lbfgs_lse(best_h.copy(), mu, 1e7, 300)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 3: More p-norm at higher p
    for p in [512, 1024, 2048, 4096]:
        if el() > 350:
            break
        ht = _lbfgs_pnorm(best_h.copy(), p, 1e8, 350)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 4: Top-k momentum descent
    if el() < budget_s - 250:
        ht, ct = _topk_momentum(best_h, 6000, 5, 60, 0.02, 0.7, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 5: LSE polish
    if el() < budget_s - 200:
        for mu in [3000, 5000]:
            if el() > budget_s - 200:
                break
            ht = _lbfgs_lse(best_h.copy(), mu, 1e9, 250)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht

    # Phase 6: Another momentum pass with finer params
    if el() < budget_s - 180:
        ht, ct = _topk_momentum(best_h, 8000, 3, 25, 0.008, 0.6, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 7: Directed local search
    if el() < budget_s - 100:
        nt = min(int((budget_s - 100 - el())/0.006), 150000)
        best_h, best_c5 = _directed_search(best_h, nt, rng, el, bc)

    # Phase 8: Fine local search
    if el() < budget_s - 20:
        nt = min(int((budget_s - 20 - el())/0.003), 200000)
        best_h, best_c5 = _fine_local(best_h, nt, rng, el, bc)

    # Phase 9: Final L-BFGS polish
    if el() < budget_s - 3:
        ht = _lbfgs_pnorm(best_h.copy(), 4096, 1e9, 150)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    best_h = _repair(best_h)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
