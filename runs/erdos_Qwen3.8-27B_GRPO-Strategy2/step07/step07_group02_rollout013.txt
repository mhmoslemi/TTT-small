import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lp_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lp_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _pnorm_obj(x, p):
    c = np.maximum(_cvals(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cvals(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs_lp(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    try:
        res = minimize(_lp_obj, h, jac=_lp_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=[(0,1)]*n,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
        return _repair(res.x, n)
    except:
        return h

def _lbfgs_pnorm(h, p, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p,),

def _thue_morse(n):
    """Generate Thue-Morse sequence of length n, balanced to have n/2 ones."""
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Ensure exactly n//2 ones
    target = n // 2
    n_ones = int(tm.sum())
    if n_ones > target:
        idx = np.where(tm == 1)[0]
        rng = np.random.RandomState(0)
        to_remove = rng.choice(idx, n_ones - target, replace=False)
        tm[to_remove] = 0
    elif n_ones < target:
        idx = np.where(tm == 0)[0]
        rng = np.random.RandomState(0)
        to_add = rng.choice(idx, target - n_ones, replace=False)
        tm[to_add] = 1
    return tm

def _sa_01(h_init, n, n_steps, deadline, rng):

    h = h_init.copy()
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        to_rm = rng.choice(idx, n_ones - target, replace=False)
        h[to_rm] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        to_add = rng.choice(idx, target - n_ones, replace=False)
        h[to_add] = 1
    
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    
    T = 0.05
    T_min = 0.0005
    cool = (T_min / T) ** (1.0 / n_steps)
    
    for step in range(n_steps):
        if time.time() > deadline - 30:
            break
        T = max(T * cool, T_min)
        # Sample swap positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 1.0, 0.0
        new_c = _eval(h)
        delta = new_c - cur_c

            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
    
    return best_h, best_c

def _top_lag(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 20:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_lags
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _local_2pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:

                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c
