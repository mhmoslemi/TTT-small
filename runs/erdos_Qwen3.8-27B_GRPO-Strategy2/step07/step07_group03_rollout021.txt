def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

def _proj_to_sum(h, target):

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corrs(h):

- With T lags, there are at most T*(T-1)/2 kinks
- We evaluate f at all kinks (within [t_lo, t_hi]) and at t_lo, t_hi
- The minimum is the best of these

import numpy as np
import time

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj_to_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _obj_smooth(h, alpha):
    c = _eval_corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _eval_corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp+cnt]
    return g

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    f = _obj_smooth(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_smooth(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smooth(h, alpha)
    return h

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:sh+cnt]) * dx
        if sh > 0:
            g[sh:] -= h[:cnt] * dx
    else:
        sp = -sh
        cnt = n - sp
        g[sp:] += (1.0 - h[:cnt]) * dx
        g[:cnt] -= h[sp:sp+cnt] * dx
    return g

def _exact_line_search(a_vals, b, d_lo, d_hi):
    T = len(a_vals)
    candidates = [d_lo, d_hi]
    for i in range(T):
        for j in range(i + 1, T):
            denom = b[i] - b[j]
            if abs(denom) > 1e-15:
                tk = (a_vals[j] - a_vals[i]) / denom
                if d_lo < tk < d_hi:
                    candidates.append(tk)
    best_t = d_lo
    best_f = np.max(a_vals + b * d_lo)
    for t in candidates:
        fv = np.max(a_vals + b * t)
        if fv < best_f:
            best_f = fv
            best_t = t
    return best_t

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    stall_limit = 50
    while time.time() < min(end_t, t_end - 5):
        c = _eval_corrs(h)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        n_pairs = n * 2
        improved = False
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 12))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            d_opt = _exact_line_search(a_vals, b, d_lo, d_hi)
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > stall_limit:
            h += 1e-4 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
            stall_limit = max(25, stall_limit - 5)
        if abs(h.sum() - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        hi0, hj0 = h[i], h[j]
        ni, nj = hi0 + d, hj0 - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = hi0, hj0
    return best_h, best_c

def _downsample(h, n_new):
    n_old = len(h)
    idx = (np.arange(n_new) + 0.5) * n_old / n_new
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n_old - 1)
    frac = idx - i0
    h_new = h[i0] * (1 - frac) + h[i1] * frac
    return _proj_to_sum(h_new, n_new / 2.0)

def _quick_opt(h, dur, rng, t_end):
    n = len(h)
    h = h.copy()
    best_c5 = _eval(h)
    t_ph1 = min(time.time() + dur * 0.2, t_end - 20)
    for alpha in [8000, 60000]:
        if time.time() > t_ph1:
            break
        h = _nesterov(h, alpha, 150, 0.003, t_ph1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
    t_ph2 = min(time.time() + dur * 0.6, t_end - 10)
    for tt in [4, 7, 5]:
        if time.time() > t_ph2:
            break
        hw = h.copy()
        hw, c5v = _coord_descent(hw, n, dur * 0.15, tt, rng, t_end)
        if c5v < best_c5:
            best_c5 = c5v
            h = hw.copy()
    if time.time() < t_end - 5:
        rem = t_end - 5 - time.time()
        nt = min(int(rem / 0.002), 15000)
        if nt > 100:
            hw, c5v = _local_search(h.copy(), nt, rng, t_end - 2)
            if c5v < best_c5:
                best_c5 = c5v
                h = hw.copy()
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_abs = t0 + budget_s
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
