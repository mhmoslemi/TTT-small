    # d/dh[j] corr[k] = (1-h[j-(k-(n-1))]) - h[j+(k-(n-1))]
    # grad[j] = sum_k w[k] * [(1-h[j-s_k]) - h[j+s_k]] where s_k = k-(n-1)

    # This is cross-correlation of w_pos with (1-h)
    # Second term: sum_s w[n-1-s] * h[j+s] for all s where valid
    # This is cross-correlation of w_neg with h

        if w[k] == 0:
            continue

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_corr(h, w):
    n = len(h)
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        if abs(w[k]) < 1e-18:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[s:] += w[k] * omh[:n - s]
            if s > 0:
                g[:n - s] -= w[k] * h[s:]
        else:
            s2 = -s
            g[:n - s2] += w[k] * omh[s2:]
            g[s2:] -= w[k] * h[:n - s2]
    return g

def _pnorm_obj(h, p, pen):
    n = len(h)
    c = np.maximum(_corrs(h), 1e-30)
    return float(np.sum(c**p)**(1.0/p) + pen * (h.sum() - n*0.5)**2)

def _pnorm_grad(h, p, pen):
    n = len(h)
    c = np.maximum(_corrs(h), 1e-30)
    cp = c**p
    s = cp.sum()
    return _grad_corr(h, c**(p-1) * p / s) * 2.0/n + 2.0*pen*(h.sum()-n*0.5)

def _sm_obj(h, beta, pen):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    return float(cm + np.log(np.exp(c-cm).sum())/beta + pen*(h.sum()-n*0.5)**2)

def _sm_grad(h, beta, pen):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    ec = np.exp(c - cm)
    w = ec / ec.sum()
    return _grad_corr(h, w) * 2.0/n + 2.0*pen*(h.sum()-n*0.5)

def _lbfgs(h, obj, grad, mx, bnds):
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': mx, 'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 30})
        return r.x
    except Exception:
        return h

def _nest_topk(h, nit, k0, k1, st0, bc):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.65
    st = st0
    bc5 = _eval(hc)
    bh = hc.copy()
    stall = 0
    for it in range(nit):
        if bc():
            break
        c = _corrs(hc)
        k = min(k0 + int((k1-k0)*it/max(nit,1)), k1)
        tk = np.argsort(c)[-k:]
        w = np.zeros(2*n-1)
        w[tk] = 1.0/k
        g = _grad_corr(hc, w) * 2.0/n
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta*v + (1.0-beta)*(g/gn)
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        ht = _proj_sum(hc - st*v, n*0.5)
        ct = _eval(ht)
        if ct < bc5 - 1e-18:
            hc = ht
            bc5 = ct
            bh = hc.copy()
            stall = 0
            st = min(st*1.015, 0.025)
        else:
            st *= 0.6
            stall += 1
            if stall > 150:
                st = st0*0.05
                v *= 0.3
                stall = 0
            if st < 1e-10:
                break
    return bh, bc5

def _loc2(h, nit, rng, scale, bc):
    n = len(h)
    hc = h.copy()
    bc5 = _eval(hc)
    bh = hc.copy()
    for _ in range(nit):
        if bc():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = rng.randn() * scale
        ni, nj = hc[i]+d, hc[j]-d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[i], hc[j] = ni, nj
            cn = _eval(hc)
            if cn < bc5:
                bc5 = cn
                bh = hc.copy()
            else:
                hc[i], hc[j] = ni-d, nj+d
        if _ % 8000 == 0:
            scale *= 0.65
    return bh, bc5

def _loc3(h, nit, rng, scale, bc):
    n = len(h)
    hc = h.copy()
    bc5 = _eval(hc)
    bh = hc.copy()
    for _ in range(nit):
        if bc():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        m = (j + 1 + rng.randint(0, 12)) % n
        d1 = rng.randn()*scale
        d2 = rng.randn()*scale
        ni, nj, nm = hc[i]+d1, hc[j]+d2, hc[m]-d1-d2
        if 0<=ni<=1 and 0<=nj<=1 and 0<=nm<=1:
            hc[i], hc[j], hc[m] = ni, nj, nm
            cn = _eval(hc)
            if cn < bc5:
                bc5 = cn
                bh = hc.copy()
            else:
                hc[i], hc[j], hc[m] = ni-d1, nj-d2, nm+d1+d2
        if _ % 12000 == 0:
            scale *= 0.7
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    def bc():
        return el() > budget_s - 12

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    bnds = [(0.0, 1.0)] * n0
    pen = 1e6

    # Phase 1: p-norm L-BFGS, aggressive schedule
    p_sched = [16, 64, 256, 512, 1024, 2048, 4096]
    for p in p_sched:
        if el() > 350:
            break
        iters = 600 if p <= 512 else 400
        ht = _lbfgs(best_h.copy(), _pnorm_obj, _pnorm_grad, iters, bnds)
        ht = _repair(ht, n0)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()
        pen *= 3

    # Phase 2: Softmax L-BFGS with increasing beta
    if el() < budget_s - 350:
        for beta in [40, 80, 160, 320, 640]:
            if el() > budget_s - 350:
                break
            ht = _lbfgs(best_h.copy(), _sm_obj, _sm_grad, 400, bnds)
            ht = _repair(ht, n0)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()

    # Phase 3: Nesterov top-k
    if el() < budget_s - 280:
        ht, ct = _nest_topk(best_h, 20000, 2, 40, 0.012, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 4: Multi-restart with perturbation
    if el() < budget_s - 200:
        for trial in range(8):
            if el() > budget_s - 200:
                break
            hp = best_h + rng.randn(n0) * 0.005
            hp = _repair(hp, n0)
            p = 1024
            ht = _lbfgs(hp, _pnorm_obj, _pnorm_grad, 300, bnds)
            ht = _repair(ht, n0)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
            ht = _lbfgs(best_h.copy(), _sm_obj, _sm_grad, 200, bnds)
            ht = _repair(ht, n0)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()

    # Phase 5: High-p polish
    if el() < budget_s - 120:
        for p in [4096, 8192]:
            if el() > budget_s - 80:
                break
            ht = _lbfgs(best_h.copy(), _pnorm_obj, _pnorm_grad, 350, bnds)
            ht = _repair(ht, n0)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()

    # Phase 6: Nesterov fine
    if el() < budget_s - 100:
        ht, ct = _nest_topk(best_h, 25000, 1, 15, 0.004, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 7: 3-point local search
    if el() < budget_s - 50:
        nt = min(int((budget_s - 50 - el()) / 0.004), 60000)
        best_h, best_c5 = _loc3(best_h, nt, rng, 4e-5, bc)

    # Phase 8: 2-point local search
    if el() < budget_s - 20:
        nt = min(int((budget_s - 20 - el()) / 0.003), 100000)
        best_h, best_c5 = _loc2(best_h, nt, rng, 3e-5, bc)

    # Phase 9: Final softmax polish
    if el() < budget_s - 3:
        ht = _lbfgs(best_h.copy(), _sm_obj, _sm_grad, 150, bnds)
        ht = _repair(ht, n0)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
