def _obj_smooth(h, alpha):
    n = len(h)
    c = _corrs(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    f = _obj_smooth(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_smooth(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smooth(h, alpha)
    return h

            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:

            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)

                    x1 = d_hi - gr * (d_hi - d_lo)

                else:

                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                # Update grads incrementally
                for t in range(top_T):

                        if i < n - sh:

                        # This is getting complicated, just recompute
                sweep_improved = True
        # Recompute grads each outer iteration (simpler and correct)
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 50:
            h += 1e-4 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] -= wk * h[i1 + sh]
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] += wk * (1.0 - h[i2 - sh])
        else:
            sp = -sh
            i1 = np.arange(n - sp)
            g[i1] += wk * (1.0 - h[i1 + sp])
            i2 = np.arange(sp, n)
            g[i2] -= wk * h[i2 - sp]
    return g

def _obj_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] -= h[i1 + sh] * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] += (1.0 - h[i2 - sh]) * dx
    else:
        sp = -sh
        i1 = np.arange(n - sp)
        g[i1] += (1.0 - h[i1 + sp]) * dx
        i2 = np.arange(sp, n)
        g[i2] -= h[i2 - sp] * dx
    return g

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    f = _obj_smooth(h, alpha)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _grad_smooth(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _obj_smooth(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _obj_smooth(h, alpha)
        mu *= 0.9995
    return h

def _coord_descent(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = _corrs(h)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_lag_grad(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        sweep_improved = False
        for _ in range(n * 3):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 7))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(26):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 40:
            h += 1e-4 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _local_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    scales = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7]
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        scale = scales[rng.randint(len(scales))]
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coordinate descent with corrected gradients (30%)
    for top_T in [3, 5, 8, 12]:
        if time.time() > t0 + budget_s * 0.5:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.07, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Nesterov high-precision polish (15%)
    if time.time() < t0 + budget_s * 0.7:
        h = best_h.copy()
        for alpha in [500000, 2000000, 8000000]:
            t_end = min(time.time() + budget_s * 0.04, t0 + budget_s * 0.69)
            h = _nesterov(h, alpha, 250, 0.001, t_end)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 4: More coordinate descent (15%)
    if time.time() < t0 + budget_s * 0.85:
        for top_T in [5, 10, 15]:
            if time.time() > t0 + budget_s * 0.84:
                break
            h = best_h.copy()
            h, c5v = _coord_descent(h, n, budget_s * 0.04, top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 5: Multi-scale local search (12%)
    if time.time() < t0 + budget_s * 0.95:
        remaining = (t0 + budget_s * 0.95) - time.time() - 2
        n_tries = min(int(remaining / 0.003), 60000)
        if n_tries > 200:
            h = best_h.copy()
            h, c5v = _local_search(h, n_tries, rng, t0 + budget_s * 0.95 - 1)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 6: Final coordinate polish
    if time.time() < t0 + budget_s - 3:
        rem = max(1, budget_s - (time.time() - t0) - 3)
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, rem, 6, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
