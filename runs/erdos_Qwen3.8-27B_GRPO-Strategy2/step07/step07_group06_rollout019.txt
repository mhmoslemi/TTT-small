def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smooth_grad_step(h, rng, t0, deadline, n_top=5):
    """Gradient step using smoothed max (top-k softmax)."""
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h, best_c = h.copy(), c5_cur
    temp = 0.01
    for it in range(2000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        # Softmax weights over top correlations
        c_sorted_idx = np.argsort(c)[::-1]
        top_idx = c_sorted_idx[:n_top]
        c_top = c[top_idx]
        c_max = c_top.max()
        w = np.exp((c_top - c_max) / temp)
        w /= w.sum()
        # Compute weighted gradient
        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            shift = ki - (n - 1)
            gr_k = np.zeros(n)
            # h[p] in first factor
            lo1 = max(0, n - 1 - 0)  # for each p, but we need per-p
            # Vectorized: for shift s, grad at p is (1-h[p+s])*dx if p+s<n, minus h[p-s]*dx if p-s>=0

                gr_k[:m] += (1.0 - h[shift:])
                if shift > 0:
                    gr_k[shift:] -= h[:m]
            else:
                s2 = -shift
                m = n - s2
                gr_k[s2:] += (1.0 - h[:m])
                if s2 > 0:
                    gr_k[:m] -= h[s2:]
            g += wi * gr_k * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        # Backtracking line search
        best_step = 0.0
        best_f = c5_cur
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = mult * 0.02 / gn
            h_new = _proj_to_sum(h - step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < best_f:
                best_f = c5n
                best_step = step
                h = h_new
        if best_step > 0:
            c5_cur = best_f
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            temp *= 0.9
        if c5_cur >= best_c:
            pass
    return best_h, best_c

def _fine_polish(h, rng, t0, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h, best_c = h.copy(), c5_cur
    scales = [1e-4, 5e-5, 1e-5, 5e-6]
    for scale in scales:
        if time.time() > deadline:
            break
        stall = 0
        for it in range(50000):
            if time.time() > deadline:
                break
            if stall > 2000:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5_cur - 1e-18:
                    c5_cur = c5n
                    stall = 0
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
                    stall += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Try perturbations for potentially better start
    for ps in [0.05, 0.1, 0.2]:
        hp = _proj_to_sum(h + ps * rng.randn(n0), n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
            h = hp.copy()

    # Phase 1: Smooth gradient descent (20s)
    dl1 = t0 + min(25, budget_s * 0.04)
    h1, c5_1 = _smooth_grad_step(best_h.copy(), rng, t0, dl1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Main random pairwise CD (400s)
    dl2 = t0 + min(budget_s * 0.40, budget_s - 350)
    h2, c5_2 = _pair_cd(best_h.copy(), 200000, rng, t0, dl2, restart_every=10000)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Cyclic structured CD (250s)
    dl3 = t0 + min(budget_s * 0.62, budget_s - 300)
    h3, c5_3 = _cyclic_cd(best_h.copy(), rng, t0, dl3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Another round of random pairwise CD (200s)
    dl4 = t0 + min(budget_s * 0.80, budget_s - 200)
    h4, c5_4 = _pair_cd(best_h.copy(), 200000, rng, t0, dl4, restart_every=12000)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Smooth gradient on refined solution (40s)
    dl5 = t0 + min(budget_s * 0.88, budget_s - 120)
    h5, c5_5 = _smooth_grad_step(best_h.copy(), rng, t0, dl5, n_top=8)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Final pairwise CD (70s)
    dl6 = t0 + min(budget_s * 0.95, budget_s - 50)
    h6, c5_6 = _pair_cd(best_h.copy(), 100000, rng, t0, dl6, restart_every=15000)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    # Phase 7: Fine polish
    dl7 = t0 + (budget_s - 3)
    h7, c5_7 = _fine_polish(best_h.copy(), rng, t0, dl7)
    if c5_7 < best_c5:
        best_c5, best_h = c5_7, h7

    # Final repair and verification
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
