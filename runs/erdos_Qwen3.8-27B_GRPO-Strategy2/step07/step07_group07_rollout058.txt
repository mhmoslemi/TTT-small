    lo, hi = -2.0, 2.0
    for _ in range(50):

        if np.clip(h - mid, 0, 1).sum() > target:

    return h

        g[:n - sh] += (1.0 - h[sh:])

    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    e = np.exp(beta * (c - cm))
    return cm + np.log(e.sum()) / beta + pen * (x.sum() - n * 0.5) ** 2

    w = e / e.sum()
    g = _wgrad(x, w)
    g += 2.0 * pen * (x.sum() - n * 0.5)

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval_h(h):
    return float(np.max(_corrs(h)))

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def _wgrad(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g += w[k] * dx * _lag_grad(h, k)
    return g

def _lse_f(x, beta, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    e = np.exp(beta * (c - cm))
    return cm + np.log(e.sum()) / beta + pen * (x.sum() - n * 0.5) ** 2

def _lse_jac(x, beta, pen):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    e = np.exp(beta * (c - cm))
    w = e / e.sum()
    g = _wgrad(x, w)
    g += 2.0 * pen * (x.sum() - n * 0.5)
    return g

def _lse_opt(h, beta, pen, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_lse_f, h, jac=_lse_jac, args=(beta, pen),
                     method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-20,
                              'gtol': 1e-15, 'maxls': 80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _mom_descent(h, nit, b0, s0, deadline):
    n = len(h)
    h_c = h.copy()
    v = np.zeros(n)
    mu = 0.82
    step = s0
    beta = b0
    best_c = _eval_h(h_c)
    best_h = h_c.copy()
    stall = 0
    for it in range(nit):
        if time.time() > deadline:
            break
        c = _corrs(h_c)
        cm = c.max()
        e = np.exp(beta * (c - cm))
        w = e / e.sum()
        g = _wgrad(h_c, w)
        g += 5e-4 * (h_c.sum() - n * 0.5)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mu * v + g / gn
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval_h(h_try)
        if c_try < best_c - 1e-17:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.008, 0.012)
            beta = min(beta * 1.002, 4000.0)
        else:
            step *= 0.48
            stall += 1
            if stall > 50 or step < 1e-11:
                break
    return best_h, best_c

def _topk_descent(h, nit, kinit, smax, deadline):
    n = len(h)
    h_c = h.copy()
    v = np.zeros(n)
    mu = 0.8
    step = 0.006
    kcount = kinit
    best_c = _eval_h(h_c)
    best_h = h_c.copy()
    stall = 0
    for it in range(nit):
        if time.time() > deadline:
            break
        c = _corrs(h_c)
        topk = np.argsort(c)[-kcount:]
        dx = 2.0 / n
        g = np.zeros(n)
        for kk in topk:
            g += _lag_grad(h_c, kk)
        g *= dx / len(topk)
        g += 3e-4 * (h_c.sum() - n * 0.5)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mu * v + g / gn
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval_h(h_try)
        if c_try < best_c - 1e-17:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.006, 0.01)
        else:
            step *= 0.48
            stall += 1
            if stall > 25:
                kcount = min(kcount + 1, smax)
                step = 0.004
                stall = 0
            if step < 1e-11:

    return best_h, best_c

def _fine_search(h, nit, rng, deadline):

    h_c = h.copy()
    best_c = _eval_h(h_c)
    best_h = h_c.copy()
    scale = 3e-5
    for it in range(nit):

        i = rng.randint(n)

        d = rng.randn() * scale
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            cn = _eval_h(h_c)
            if cn < best_c:
                best_c = cn
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
        if it % 12000 == 0:
            scale *= 0.55
            if scale < 1e-8:

    return best_h, best_c

def _triple_search(h, nit, rng, deadline):
    n = len(h)
    h_c = h.copy()
    best_c = _eval_h(h_c)
    best_h = h_c.copy()
    scale = 2e-5
    for it in range(nit):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 10)) % n
        m = (i + 12 + rng.randint(0, 15)) % n
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni = h_c[i] + d1
        nj = h_c[j] + d2
        nm = h_c[m] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nm <= 1:
            h_c[i], h_c[j], h_c[m] = ni, nj, nm
            cn = _eval_h(h_c)
            if cn < best_c:

                best_h = h_c.copy()
            else:
                h_c[i], h_c[j], h_c[m] = ni - d1, nj - d2, nm + d1 + d2
        if it % 8000 == 0:
            scale *= 0.55
            if scale < 5e-9:
                break
    return best_h, best_c
