    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                    diff -= a

        if s >= 0:
            g[s:] += wk * (1.0 - x[:n - s])
            g[:n - s] -= wk * x[s:]
        else:
            p = -s
            g[:n - p] += wk * (1.0 - x[p:])
            g[p:] -= wk * x[:n - p]
    return g

def _sum_con(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _mgd(h, rounds, k_top, steps_init, check_fn, mu=0.6):
    """Momentum projected gradient with correct gradient."""
    n = len(h)
    c5c = _eval(h)
    steps = list(steps_init)
    v = np.zeros(n)
    stall = 0
    for _r in range(rounds):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top = np.argsort(c)[-k_top:]
        g = _grad_lags(h, top, n)
        v = mu * v + g
        improved = False
        for si in range(len(steps)):
            h_new = _project(h - steps[si] * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-14:
                h = h_new
                c5c = c5n
                improved = True
                steps = [steps[si] * 1.3] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 10:
                    steps = steps[:10]
                break
        if not improved:
            v *= 0.3
            steps = [s * 0.5 for s in steps]
            stall += 1
            if stall > 12:
                break
        else:
            stall = 0
    return h, c5c

def _local_adapt(h, ntry, rng, c5c, check_fn, step0=5e-5, npts=3):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    step = step0
    for _ in range(ntry):
        if check_fn():
            break
        idxs = np.sort(rng.randint(0, nc, npts))
        d = rng.randn(npts)
        d -= d.mean()
        d *= step
        nv = h[idxs] + d
        if np.all(nv >= -1e-15) and np.all(nv <= 1 + 1e-15):
            h[idxs] = np.clip(nv, 0, 1)
            c5n = _eval(h)
            if c5n < best_c - 1e-17:
                if c5n < c5c:
                    c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                step = min(step * 1.05, 2e-3)
            else:
                h[idxs] = h[idxs] - d
                step *= 0.9
        else:
            step *= 0.7
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    
    def _el():
        return time.time() - t0
    def _bc():
        return _el() > budget_s - 20

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Momentum gradient - aggressive
    for kt in [5, 3, 7, 4, 5, 3]:
        if _bc():
            break
        h2, c52 = _mgd(best_h.copy(), 500, kt, [0.05, 0.025, 0.01, 0.005, 0.002, 0.001], _bc, mu=0.6)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: SLSQP
    if _el() < budget_s - 180:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
        for _ in range(2):
            if _bc():
                break
            try:
                res = minimize(_smooth_max_obj, best_h, args=(n,),
                               jac=_smooth_max_grad, method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-18})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 3: Fine momentum gradient
    for kt in [3, 4, 5, 3, 4]:
        if _bc():
            break
        h2, c52 = _mgd(best_h.copy(), 400, kt, [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4], _bc, mu=0.5)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: Fourier perturbation restarts
    for _ in range(3):
        if _bc():
            break
        n = len(best_h)
        pert = np.zeros(n)
        for m in range(1, 6):
            amp = 0.01 / m
            ph = rng.uniform(0, 2 * np.pi)
            pert += amp * np.cos(2 * np.pi * m * np.arange(n) / n + ph)
        h_p = _repair(best_h + pert, n)
        h2, c52 = _mgd(h_p, 300, 4, [0.03, 0.01, 0.005, 0.002, 0.001], _bc, mu=0.5)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Adaptive local search
    if _el() < budget_s - 20:
        nt = min(int((budget_s - 20 - _el()) / 0.003), 120000)
        best_h, best_c5 = _local_adapt(best_h, nt, rng, best_c5, _bc, 5e-5, 3)

    # Phase 6: Ultra-fine 2-point
    if _el() < budget_s - 4:
        nt = min(int((budget_s - 4 - _el()) / 0.002), 120000)
        best_h, best_c5 = _local_adapt(best_h, nt, rng, best_c5, _bc, 2e-5, 2)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
