import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(-h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                diff -= a
                if abs(diff) < 1e-13:
                    break
    return h

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def _grad_lag(h, lag_idx, n, dx):
    sh = lag_idx - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        g[:cnt] += (1.0 - h[sh:n]) * dx
        if sh > 0:
            g[sh:] -= h[:cnt] * dx
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp]) * dx
        g[:n - sp] -= h[sp:n] * dx
    return g

def _nesterov_topk(h, n, top_k, steps, step0, t_end):
    dx = 2.0 / n
    best_c = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, k, n, dx)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        fn = _eval(h_new)
        if fn < best_c:
            h = h_new
            best_c = fn
            step = min(step * 1.04, step0 * 20.0)
        elif fn < _eval(h) + 1e-15:
            h = h_new
            step *= 0.75
        else:
            step *= 0.45
            if step < 1e-13:
                break
    return h, best_c

def _cd_pair(h, n, top_T, rng, dur, t_end, wide=False):
    dx = 2.0 / n
    best_c = _eval(h)
    gr = (np.sqrt(5) - 1) / 2
    end_t = min(time.time() + dur, t_end - 6)
    span = max(4, n // 4) if wide else max(3, n // 8)
    no_imp = 0

        top_idx = np.argsort(c)[-top_T:]
        c_top = c[top_idx]
        grads = np.array([_grad_lag(h, k, n, dx) for k in top_idx])
        improved = False
        n_pairs = 5 * n if wide else 4 * n
        for _ in range(n_pairs):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, span)) % n
            if i == j:

            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(c_top + b * x1)
            f2 = np.max(c_top + b * x2)
            for _ in range(24):

                    x2 = x1
                    f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(c_top + b * x1)
                else:
                    d_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(c_top + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        h = _proj(h, n * 0.5)
        if improved:
            nc = _eval(h)
            if nc < best_c:

                no_imp = 0
            else:

            no_imp += 1
        if no_imp > 60:
            h += 1e-5 * rng.randn(n)

def _fine_local(h, n_iter, rng, t_end):
    best_c = _eval(h)
    best_h = h.copy()
    n = len(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 5)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d

            c = _eval(h)
            if c < best_c:

            else:

def _try_mirror(best_h, best_c5, n, rng, t_end):
    h_m = 1.0 - best_h.copy()
    h_m = _repair(h_m, n)
    c_m = _eval(h_m)

    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov top-k (progressive)
    p1cfg = [(3, 300, 0.006), (5, 250, 0.004), (8, 200, 0.0025), (15, 150, 0.001)]
    for top_k, steps, step0 in p1cfg:
        if time.time() > t0 + budget_s * 0.12:
            break
        h, c = _nesterov_topk(h, n, top_k, steps, step0, t0 + budget_s * 0.12)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Try mirror
    best_h, best_c5 = _try_mirror(best_h, best_c5, n, rng, t_end)

    # Phase 2: CD pass 1
    for top_T in [3, 5, 8, 12]:
        if time.time() > t0 + budget_s * 0.42:
            break
        h = best_h.copy()
        h, c = _cd_pair(h, n, top_T, rng, budget_s * 0.1, t_end, wide=False)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 3: Nesterov refinement
    if time.time() < t0 + budget_s * 0.55:
        for top_k, steps, step0 in [(5, 200, 0.001), (10, 150, 0.0005), (20, 100, 0.0003)]:
            if time.time() > t0 + budget_s * 0.54:
                break
            h = best_h.copy()
            h, c = _nesterov_topk(h, n, top_k, steps, step0, t0 + budget_s * 0.54)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Phase 4: CD pass 2 (wider)
    if time.time() < t0 + budget_s * 0.75:
        for top_T in [6, 10, 15, 20]:
            if time.time() > t0 + budget_s * 0.74:
                break
            h = best_h.copy()
            h, c = _cd_pair(h, n, top_T, rng, budget_s * 0.06, t_end, wide=True)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Phase 5: Fine local search
    if time.time() < t_end - 15:
        remaining = t_end - time.time() - 10
        ntries = min(int(remaining / 0.006), 80000)
        if ntries > 100:
            h = best_h.copy()
            h, c = _fine_local(h, ntries, rng, t_end - 5)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Phase 6: Final CD polish (multiple top_T)
    if time.time() < t_end - 4:
        for top_T in [10, 5, 15]:
            if time.time() > t_end - 4:
                break
            h = best_h.copy()
            h, c = _cd_pair(h, n, top_T, rng, max(1.0, (t_end - time.time()) / 3 - 2), t_end, wide=False)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Final mirror check
    best_h, best_c5 = _try_mirror(best_h, best_c5, n, rng, t_end)
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
