import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _softmax_grad(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    sc = np.clip(alpha * (c - c.max()), -500, 500)
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g, c

def _pair_opt(h, i, j, c, n):
    f = h
    dlo = -f[i]
    dhi = min(1.0 - f[i], f[j])
    if dhi <= dlo + 1e-12:
        return 0.0, c.max()
    sh = np.arange(2 * n - 1) - (n - 1)
    A = np.zeros(2 * n - 1)
    vip = (i + sh >= 0) & (i + sh < n)
    vjp = (j + sh >= 0) & (j + sh < n)
    vim = (i - sh >= 0) & (i - sh < n)
    vjm = (j - sh >= 0) & (j - sh < n)
    A[vip] += 1.0 - f[i + sh[vip]]
    A[vjp] -= 1.0 - f[j + sh[vjp]]
    A[vim] -= f[i - sh[vim]]
    A[vjm] += f[j - sh[vjm]]
    B = A * (2.0 / n)
    best_d, best_v = 0.0, c.max()
    for d in np.linspace(dlo, dhi, 15):
        v = np.max(c + d * B)
        if v < best_v:

    span = (dhi - dlo) / 15
    lo2 = max(dlo, best_d - span)
    hi2 = min(dhi, best_d + span)
    for d in np.linspace(lo2, hi2, 15):
        v = np.max(c + d * B)
        if v < best_v:
            best_v, best_d = v, d
    return best_d, best_v

def _pair_descent(h, n_iters, rng, t0, deadline):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c5 = c.max()
    best_h = h.copy()
    best_c5 = c5
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        delta, new_c5 = _pair_opt(h, i, j, c, n)
        if abs(delta) > 1e-10 and new_c5 < c5 - 1e-12:
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = _repair(h, n)
            c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            c5 = c.max()
            if c5 < best_c5:

                best_h = h.copy()
    return best_h, best_c5

def _fista(h, alpha_max, n_steps, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    h_prev = h.copy()
    v = np.zeros(n)
    step = 0.001
    best_h = h.copy()
    best_c5 = _eval(h)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        alpha = min(alpha_max, 100 * (1 + it * 0.5))
        g, c = _softmax_grad(h, alpha, dx)
        f_val = c.max()
        h_prev = h.copy()
        h_new = _proj_to_sum(h - step * g, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f_val:
            h = h_new
            step = min(step * 1.05, 0.01)
            if f_new < best_c5:
                best_c5 = f_new
                best_h = h.copy()
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h, best_c5

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s - s.max()
        return m + np.log(np.sum(np.exp(sm))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pair_descent(h, n_iters, rng, t0, deadline, greedy_k=1):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = c.max()
    best_h = h.copy()
    best_c5 = c5
    sh_arr = np.arange(2 * n - 1) - (n - 1)
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        best_dlt, best_v, best_i, best_j = 0.0, c5, -1, -1
        for _g in range(greedy_k):
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n - 1)) % n
            f = h
            dlo = -f[i]
            dhi = min(1.0 - f[i], f[j])
            if dhi <= dlo + 1e-12:
                continue
            A = np.zeros(2 * n - 1)
            vip = (i + sh_arr >= 0) & (i + sh_arr < n)
            vjp = (j + sh_arr >= 0) & (j + sh_arr < n)
            vim = (i - sh_arr >= 0) & (i - sh_arr < n)
            vjm = (j - sh_arr >= 0) & (j - sh_arr < n)
            A[vip] += 1.0 - f[i + sh_arr[vip]]
            A[vjp] -= 1.0 - f[j + sh_arr[vjp]]
            A[vim] -= f[i - sh_arr[vim]]
            A[vjm] += f[j - sh_arr[vjm]]
            B = A * dx
            bd, bv = 0.0, c5
            span = dhi - dlo
            for d in np.linspace(dlo, dhi, 11):
                v = np.max(c + d * B)
                if v < bv:
                    bv, bd = v, d
            lo2 = max(dlo, bd - span / 11)
            hi2 = min(dhi, bd + span / 11)
            for d in np.linspace(lo2, hi2, 11):
                v = np.max(c + d * B)
                if v < bv:
                    bv, bd = v, d
            if bd != 0.0 and bv < best_v:

            h[best_i] += best_dlt
            h[best_j] -= best_dlt

def _full_grad_and_c(h, alpha, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    sc = np.clip(alpha * (c - c.max()), -500, 500)
    w = np.exp(sc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g, c

def _fista_run(h, t0, deadline, alpha_start=2000, alpha_end=50000):
    n = len(h)
    dx = 2.0 / n
    step = 0.002
    best_h = h.copy()
    best_c5 = _eval(h)
    h_prev = h.copy()
    t_ = 1.0
    for it in range(20000):
        if time.time() > deadline:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / 20000.0)
        g, c = _full_grad_and_c(h, alpha, dx)
        f_val = c.max()
        h_prev = h.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_ * t_))
        h_new = _proj_to_sum(h_prev + (t_ - 1.0) / t_new * (h_prev - (h_prev - step * g)), n * 0.5)
        h_new = _proj_to_sum(h - step * g, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f_val:
            h = h_new
            step = min(step * 1.03, 0.02)
            if f_new < best_c5:
                best_c5 = f_new
                best_h = h.copy()
        else:
            step *= 0.6
            if step < 1e-10:
                break
        t_ = t_new
    return best_h, best_c5

def _slsqp_opt(h, alpha, maxiter):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        sm = s - s.max()
        return m + np.log(np.sum(np.exp(sm))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wk * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wk * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * x[i2 + sp]
        return g
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_2pt(h, n_tries, rng, t0, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
