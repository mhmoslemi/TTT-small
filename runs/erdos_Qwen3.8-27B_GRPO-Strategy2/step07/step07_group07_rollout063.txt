    bb = _eval(hc)
    bh = hc.copy()
    sc = 1.5e-5
    for it in range(ni):
        if time.time() - t0 > bs - 25:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 8)) % n
        m = (i + 10 + rng.randint(0, 20)) % n
        d1 = rng.randn() * sc
        d2 = rng.randn() * sc
        a, b, c = hc[i] + d1, hc[j] + d2, hc[m] - d1 - d2
        if 0 <= a <= 1 and 0 <= b <= 1 and 0 <= c <= 1:
            hc[i], hc[j], hc[m] = a, b, c
            cn = _eval(hc)
            if cn < bb:
                bb, bh = cn, hc.copy()
            else:
                hc[i], hc[j], hc[m] = a - d1, b - d2, c + d1 + d2
        if it % 10000 == 0 and it > 0:
            sc *= 0.5
    return bh, bb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh = h.copy()
    bc5 = _eval(bh)

    for T in [20, 50, 100, 200]:
        if time.time() - t0 > 85:
            break
        ht = _lbfgs(bh.copy(), T, 1e6, 600)
        ct = _eval(ht)
        if ct < bc5:
            bc5, bh = ct, ht

    for T in [500, 1000, 2000]:
        if time.time() - t0 > 185:
            break
        ht = _lbfgs(bh.copy(), T, 1e7, 500)
        ct = _eval(ht)
        if ct < bc5:
            bc5, bh = ct, ht

    for T in [4000, 7000]:
        if time.time() - t0 > 270:
            break
        ht = _lbfgs(bh.copy(), T, 1e7, 400)
        ct = _eval(ht)
        if ct < bc5:
            bc5, bh = ct, ht

    if time.time() - t0 < budget_s - 270:
        bh, bc5 = _gn(bh, 6000, 5, 0.02, t0, budget_s)

    if time.time() - t0 < budget_s - 215:
        bh, bc5 = _greedy2(bh, 14000, 3, t0, budget_s)

    if time.time() - t0 < budget_s - 165:
        bh, bc5 = _gn(bh, 5000, 3, 0.004, t0, budget_s)

    for amp in [0.025, 0.01, 0.004]:
        if time.time() - t0 > budget_s - 135:
            break
        for _ in range(3):
            if time.time() - t0 > budget_s - 135:
                break
            hp = _repair(bh + rng.randn(n) * amp, n)
            ht = _lbfgs(hp, 5000, 1e7, 200)
            ct = _eval(ht)
            best_pt = (ht, ct)
            ht2 = _gn(hp, 1500, 3, 0.004, t0, budget_s)
            ct2 = _eval(ht2)
            if ct2 < ct:
                best_pt = (ht2, ct2)
            if best_pt[1] < bc5:
                bc5, bh = best_pt[1], best_pt[0]

    if time.time() - t0 < budget_s - 95:
        bh, bc5 = _greedy2(bh, 8000, 2, t0, budget_s)

    if time.time() - t0 < budget_s - 75:
        bh, bc5 = _gn(bh, 3000, 2, 0.002, t0, budget_s)

    if time.time() - t0 < budget_s - 40:
        nt = min(int((budget_s - 75 - (time.time() - t0)) / 0.004), 60000)
        bh, bc5 = _triplet(bh, nt, rng, t0, budget_s)

    if time.time() - t0 < budget_s - 28:
        nt = min(int((budget_s - 28 - (time.time() - t0)) / 0.003), 120000)
        bh, bc5 = _rnd2(bh, nt, rng, t0, budget_s)

    if time.time() - t0 < budget_s - 5:
        ht = _lbfgs(bh.copy(), 8000, 1e8, 150)
        ct = _eval(ht)
        if ct < bc5:
            bc5, bh = ct, ht

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
