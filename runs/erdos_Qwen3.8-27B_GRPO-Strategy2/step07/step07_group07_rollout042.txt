from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj_sum(h, n / 2.0)

def _lse_obj(h, beta, penalty):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    # Log-sum-exp for smooth max
    c_max = c.max()
    lse = c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta
    return lse + penalty * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    ec = np.exp(beta * (c - c_max))
    w = ec / ec.sum() / beta  # softmax weights
    g = np.zeros(n)

        if w[k] < 1e-12:
            continue

            g[i1] += wk * (1.0 - h[i1 + sh])

from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj_sum(h, n / 2.0)

def _corr_grad(h, weights):
    """Gradient weighted sum of correlations c_k with weights."""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if abs(weights[k]) < 1e-14:
            continue
        wk = weights[k]
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _lse_obj(h, beta, pen):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cm = c.max()
    lse = cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta
    return lse + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    ec = np.exp(beta * (c - cm))
    w = ec / ec.sum() / beta
    g = _corr_grad(h, w) * dx
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _pobj(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(c**p)**(1.0/p)) + pen * (h.sum() - n * 0.5)**2

def _pgrad(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.maximum(np.correlate(h, 1.0 - h, mode='full') * dx, 1e-30)
    s = np.sum(c**p)
    w = s**(1.0/p - 1.0) * c**(p - 1)
    g = _corr_grad(h, w) * dx
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs(h, obj_fn, grad_fn, args, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=args,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

            best_c = c_try

    idx = rng.choice(n, size=n_pert, replace=False)
