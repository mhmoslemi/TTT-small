import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    if abs(h.sum() - t) > 1e-9:
        h = _proj_sum(h, t)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_rl(h, weights):
    """Gradients: dot product of weights with each correlation gradient."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k] * dx
        if abs(wk) < 1e-16:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                g[i1 + sh] -= wk * h[i1]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            g[i1 - sp] -= wk * h[i1]
    return g

def _lse_obj(h, kappa, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    val = kappa * np.log(np.sum(np.exp((c - cm) / kappa))) + cm
    return val + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, kappa, pen):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    w = np.exp((c - cm) / kappa)
    w /= w.sum()
    g = _grad_rl(h, w)
    return g + 2.0 * pen * (h.sum() - n * 0.5)

def _pnorm_obj(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(np.maximum(c, 0)**p)**(1.0/p)) + pen * (h.sum() - n * 0.5) ** 2

def _pnorm_grad(h, p, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cp = np.maximum(c, 1e-30)**p
    s = cp.sum()
    fac = s**(1.0/p - 1.0)
    w = fac * np.maximum(c, 1e-30)**(p - 1)
    g = _grad_rl(h, w)
    return g + 2.0 * pen * (h.sum() - n * 0.5)

def _lbfgs(h, obj_fn, grad_fn, args, maxiter=200):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, args=args,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-13, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _proj_grad(h, kappa, pen, n_iter, step0, mom, el_fn, bc_fn):
    """Momentum projected gradient on LSE objective with adaptive step."""
    n = len(h)

def _coord_search(h, n_iter, rng, el_fn, bc_fn):
    """Adaptive coordinate descent with gradient guidance."""
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 0.004
    for it in range(n_iter):
        if bc_fn():
            break
        # Occasionally use gradient direction
        if it % 200 == 0 and scale < 0.001:
            c = _corrs(h_c)
            topk = np.argpartition(c, -5)[-5:]
            g = _grad_rl(h_c, np.ones_like(c) * np.where(np.isin(np.arange(2*n-1), topk), 1.0, 0.0))
            g -= g.mean()
            idx = np.argsort(np.abs(g))[-2:]
            i, j = int(idx[0]), int(idx[1])
            d = np.sign(g[i]) * scale
        else:

            j = (i + 1 + rng.randint(0, 8)) % n

        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

            c_new = _eval(h_c)
            if c_new < best_c - 1e-18:
                best_c = c_new
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
        if it % 5000 == 4999:

            best_c5 = c_try
            best_h = h_try
        pen *= 2.5

    # Phase 2: p-norm L-BFGS (70-200s)
    for p in [8, 16, 32, 64, 128, 256]:
        if el() > 200:
            break
        h_try = _lbfgs(best_h.copy(), _pnorm_obj, _pnorm_grad,
                       (p, 1e5), maxiter=150)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 3: Projected gradient with momentum (200-380s)
    if el() < budget_s - 300:
        for kappa in [0.01, 0.005, 0.003, 0.002]:
            if el() > budget_s - 300:
                break
            h_try, c_try = _proj_grad(best_h, kappa, 1e5,
                                       4000, 0.012, 0.7, el, bc)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 4: High p L-BFGS (380-520s)
    if el() < budget_s - 200:
        for p in [512, 1024, 2048]:
            if el() > budget_s - 200:
                break
            h_try = _lbfgs(best_h.copy(), _pnorm_obj, _pnorm_grad,
                           (p, 1e7), maxiter=300)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 5: Focused descent on active lags (520-750s)
    if el() < budget_s - 150:
        h_try, c_try = _focused_descent(best_h, 8000, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 6: More projected gradient at very small kappa
    if el() < budget_s - 100:
        h_try, c_try = _proj_grad(best_h, 0.001, 1e6,
                                   5000, 0.004, 0.8, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 7: Long coordinate search (most of remaining time)
    if el() < budget_s - 25:
        nt = min(int((budget_s - 25 - el()) / 0.002), 300000)
        best_h, best_c5 = _coord_search(best_h, nt, rng, el, bc)

    # Phase 8: Final L-BFGS polish
    if el() < budget_s - 5:
        h_try = _lbfgs(best_h.copy(), _pnorm_obj, _pnorm_grad,
                       (4096, 1e8), maxiter=100)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
