def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                if diff > 0:

                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    c_sh = c - c_max
    exp_c = np.exp(eps * c_sh)
    w = exp_c / exp_c.sum()
    a = 1.0 - 2.0 * h
    a_flip = a[::-1].copy()
    conv = np.convolve(w, a_flip, mode='full')
    grad = conv[n - 1:2 * n - 1] * dx
    return grad, c

def _smooth_opt(h, rng, deadline):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval(best_h)
    eps_list = [8.0, 6.0, 4.5, 3.0, 2.0, 1.5, 1.0]
    for eps in eps_list:
        if time.time() > deadline:
            break
        step = 0.04

            if time.time() > deadline:
                break
            g, c = _smooth_grad(h, eps)
            gn = np.linalg.norm(g)
            if gn < 1e-13:

            g_dir = g / gn
            accepted = False
            for bk in range(10):
                h_try = _proj_to_sum(h - step * g_dir, target)
                c5_try = _eval(h_try)

                    accepted = True
                    if c5_try < best_c5:

                        best_h = h.copy()
                    break
                step *= 0.5
            if not accepted:
                step *= 0.4
                if step < 1e-7:
                    break
    return best_h, best_c5

def _pair_grad(h, i, j, n):
    g = np.zeros(2 * n - 1)
    lo_j = j
    hi_j = min(j + n - 1, 2 * n - 2)
    if lo_j <= hi_j:
        g[lo_j:hi_j + 1] += (1.0 - 2.0 * h[0:hi_j - lo_j + 1])
    lo_i = i
    hi_i = min(i + n - 1, 2 * n - 2)
    if lo_i <= hi_i:
        g[lo_i:hi_i + 1] -= (1.0 - 2.0 * h[0:hi_i - lo_i + 1])
    return g

def _cd_line_search(c, g, d_lo, d_hi, c5_cur, n):
    for _ in range(35):
        d_mid = 0.5 * (d_lo + d_hi)
        c_try = c + d_mid * g
        c_try_max = c_try.max()
        if c_try_max < c5_cur - 1e-13:
            d_lo = d_mid
        elif c_try_max > c5_cur + 1e-13:

            kmax_old = np.argmax(c)
            if c_try[kmax_old] < c[kmax_old] - 1e-14:

            elif c_try[kmax_new] > c[kmax_new] + 1e-14:
                d_hi = d_mid
            else:

    return 0.5 * (d_lo + d_hi)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
