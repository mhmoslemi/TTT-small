import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    k_lo1 = max(-(n - 1), p - n + 1)
    k_hi1 = min(n - 1, p)
    if k_lo1 <= k_hi1:
        kr = np.arange(k_lo1, k_hi1 + 1)
        g[n - 1 + kr] += (1.0 - h[p - kr])
    k_lo2 = max(-(n - 1), -p)
    k_hi2 = min(n - 1, n - 1 - p)
    if k_lo2 <= k_hi2:
        kr = np.arange(k_lo2, k_hi2 + 1)
        g[n - 1 + kr] -= h[p + kr]
    return g

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = _project_sum(h, target)
    return h

def _cd_focused(h, rng, deadline, peak_w=6, n_iter=300000, anneal_temp=0.0):
    n = len(h)
    c = _corr(h)
    c5 = float(c.max())
    best_h = h.copy()
    best_c5 = c5
    stable = 0
    nsamples = 2 * n - 1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 2000 == 0 and stable > 15000:
            break
        m = min(peak_w, nsamples)
        top = np.argpartition(c, -m)[-m:]
        c_med = c[int(nsamples * 0.5)]
        w = np.array([max(0.05, c[ki] - c_med) for ki in top])

        for ii, ki in enumerate(top):
            k = ki - (n - 1)
            ww = w[ii]
            if k >= 0:
                g_agg[k:] += ww * (1.0 - h[:n - k])
                g_agg[:n - k] -= ww * h[k:]
            else:
                kk = -k
                g_agg[:n - kk] += ww * (1.0 - h[kk:])
                g_agg[kk:] -= ww * h[:n - kk]
        sorted_idx = np.argsort(g_agg)

            j = j_pool[rng.randint(pool_sz)]
            if i == j:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-14:
                continue
            gi = _grad_at_pos(h, i, n)
            gj = _grad_at_pos(h, j, n)
            g = gi - gj
            d0 = j - i
            k1 = n - 1 + d0

def _subgrad(h, rng, deadline, n_iter=3000):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    top_m = 4
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for ki in top_k:
            k = ki - (n - 1)
            if k >= 0:
                g[k:] += (1.0 - h[:n - k])
                g[:n - k] -= h[k:]
            else:
                kk = -k
                g[:n - kk] += (1.0 - h[kk:])
                g[kk:] -= h[:n - kk]
        g *= dx / top_m
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = min(0.02, 0.5 / gn)
        for sm in [1.0, 0.5, 0.2, 0.1]:
            h_new = _project_sum(h - sm * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
    return best_h, best_c

def _fine_cd(h, rng, deadline, n_iter=200000):
    n = len(h)
    c = _corr(h)
    c5 = float(c.max())
    best_h = h.copy()
    best_c5 = c5
    nsamples = 2 * n - 1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-14:
            continue
        d_scale = (d_hi - d_lo) * rng.uniform(0.01, 0.3) * (1 if rng.random() > 0.5 else -1)
        d_opt = d_lo + (d_hi - d_lo) * 0.5 + d_scale
        d_opt = np.clip(d_opt, d_lo, d_hi)
        if abs(d_opt) < 1e-14:
            continue
        gi = _grad_at_pos(h, i, n)
        gj = _grad_at_pos(h, j, n)
        g = gi - gj
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < nsamples:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < nsamples:
            c[k2] += d_opt * d_opt
        c5_new = float(c.max())
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < nsamples:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < nsamples:
                c[k2] -= d_opt * d_opt
    return best_h, best_c5
