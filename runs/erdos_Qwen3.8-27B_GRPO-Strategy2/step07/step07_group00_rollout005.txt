Let me think about the smooth-max more carefully. The objective is:
f(h) = max_k [correlate(h, 1-h, k)] * dx

The soft-max approximation with alpha and topk:
smax = (max(ct - m) + log(sum(exp(alpha*(ct-m)))))/alpha + m

where ct is the top few correlation values.

For very large alpha, this approaches the true max. But the gradient becomes concentrated on just the max element, which can make optimization harder.

Alternative: Use a "top-k mean" approach - minimize the sum of the top k correlation values. This is different from soft-max and can give better results because it's smoother.

Let me also think about the problem structure. The correlation at lag k is:
c(k) = sum_i h[i] * (1-h[i+k]) * dx (with appropriate wrapping/indexing)

For the Erdős problem, we want all lags to have small overlap. The structure suggests that h should be somewhat "spread out" - not too concentrated.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    stall = 0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.03, step0 * 8.0)
            stall = 0
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
            stall += 1
            if stall > 300:
                break
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_smax(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sizes = [1, 2, 3, 4, 5, 7, 10, 15]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        mt = rng.randint(5)
        hn = h.copy()
        if mt == 0:
            bs = sizes[rng.randint(len(sizes))]
            i0 = rng.randint(0, max(1, n - bs))
            d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
            d -= d.mean()
            hn[i0:i0+bs] = np.clip(hn[i0:i0+bs] + d, 0, 1)
        elif mt == 1:
            bs = sizes[rng.randint(5)]
            i0 = rng.randint(0, max(1, n - bs))
            hn[i0:i0+bs] = 1.0 - hn[i0:i0+bs]
        elif mt == 2:
            bs = sizes[rng.randint(4)]
            i0 = rng.randint(0, max(1, n - bs))
            sh = rng.randint(-4, 5)
            seg = h[i0:i0+bs].copy()
            ns = max(0, min(n - bs, i0 + sh))
            hn[ns:ns+bs] = seg
            if 0 <= i0 < n - bs and i0 != ns:
                hn[i0:i0+bs] = 0.5
        elif mt == 3:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            d = 8e-4 * max(T, 1e-9) ** 0.5 * rng.randn(2)
            d -= 0.5 * d.mean()
            hn[i] = np.clip(hn[i] + d[0], 0, 1)
            hn[j] = np.clip(hn[j] + d[1], 0, 1)
        else:
            bs = sizes[rng.randint(3)]
            i0 = rng.randint(0, max(1, n - bs))
            j0 = range(i0, i0+bs)
            perm = list(j0)
            rng.shuffle(perm)
            for k, pi in zip(j0, perm):
                hn[k] = h[pi]
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = 4e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _3pt_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + rng.randint(1, 5)) % nc
        k = (i + rng.randint(6, 15)) % nc
        d1 = 2e-6 * rng.randn()
        d2 = 2e-6 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return bh, bc

def _multires(h, alpha, topk, maxit, deadline):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for fac in [8, 4, 2]:
        if time.time() > deadline:
            break
        m = n // fac
        if m < 10:
            continue
        hd = np.array([h[i*m:(i+1)*m].mean() for i in range(m)])
        hd = _repair(hd, m)
        hd = _slsqp(hd, alpha, max(topk, m//4), maxit, deadline)
        # Upsample via linear interpolation
        xs = np.linspace(0, 1, m)
        xs_fine = np.linspace(0, 1, n)
        hu = np.interp(xs_fine, xs, hd)
        hu = _repair(hu, n)
        # Refine on fine grid
        hu = _slsqp(hu, alpha * 2, topk, maxit // 2, deadline)
        c5 = _eval(hu)
        if c5 < best_c:
            best, best_c = hu, c5
    return best, best_c
