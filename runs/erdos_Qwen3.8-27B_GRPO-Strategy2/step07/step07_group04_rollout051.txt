import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    tgt = 0.5 * n
    if abs(h.sum() - tgt) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lags(h, k_list):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 / len(k_list)
    g = np.zeros(n)
    for k in k_list:
        if k >= 0:
            g[k:] += w * dx * (1.0 - h[:n - k])
            g[:n - k] -= w * dx * h[k:]
        else:
            s = -k
            g[:n - s] += w * dx * (1.0 - h[s:])
            g[s:] -= w * dx * h[:n - s]
    return g

def _fista_topk(h, k_top, momentum, max_iter, t0, t_end):
    n = len(h)
    dx = 2.0 / n
    step = 0.9 / (2.0 * dx)  # slightly less than 1/L
    h_cur = h.copy()
    y = h_cur.copy()
    t = 1.0
    best_f = _eval(h_cur)
    best_h = h_cur.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(y)
        top_k = np.argsort(c)[-k_top:]
        g = _grad_lags(y, top_k)
        h_new = _repair(y - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h_cur)
        y = _repair(y, n)
        h_cur = h_new
        t = t_new
        if it % 500 == 0:
            f = _eval(h_cur)
            if f < best_f:
                best_f = f
                best_h = h_cur.copy()
    f = _eval(h_cur)
    if f < best_f:
        best_f, best_h = f, h_cur.copy()
    return best_h, best_f

def _slsqp(h, alpha, maxiter, t0, t_end):
    n = len(h)
    dx = 2.0 / n
    tgt = 0.5 * n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        p = np.exp(s)
        p /= p.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-12:
                continue
            if k >= 0:
                g[k:] += p[k] * dx * (1.0 - x[:n - k])
                g[:n - k] -= p[k] * dx * x[k:]
            else:
                s2 = -k
                g[:n - s2] += p[k] * dx * (1.0 - x[s2:])
                g[s2:] -= p[k] * dx * x[:n - s2]
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _directed_search(h, n_steps, t0, t_end):
    """Target the active maximum lag specifically"""
    n = len(h)
    h_cur = h.copy()
    best_f = _eval(h_cur)
    best_h = h_cur.copy()
    for it in range(n_steps):
        if time.time() > t_end:
            break
        c = _corr(h_cur)
        k_star = int(np.argmax(c))
        g = _grad_lags(h_cur, [k_star])
        # Try several step sizes
        accepted = False
        for step_scale in [1.0, 0.5, 0.25, 0.1]:
            step = step_scale * (n / 8.0) * 0.01
            h_try = _repair(h_cur - step * g, n)
            f_try = _eval(h_try)
            if f_try < best_f - 1e-15:
                h_cur = h_try
                best_f = f_try
                best_h = h_cur.copy()
                accepted = True
                break
        if not accepted:
            break
    return best_h, best_f

def _local_search(h, n_tries, step_size, rng, t0, t_end, n_pts=2):
    n = len(h)
    h_cur = h.copy()
    best_f = _eval(h_cur)
    best_h = h_cur.copy()
    for it in range(n_tries):
        if time.time() > t_end:

        idx = rng.randint(n, size=n_pts)
        d = np.zeros(n_pts)
        d[:-1] = step_size * rng.randn(n_pts - 1)
        d[-1] = -d[:-1].sum()
        h_try = h_cur.copy()
        h_try[idx] += d
        if (h_try >= 0).all() and (h_try <= 1).all():
            h_try = _repair(h_try, n)
            f_try = _eval(h_try)
            if f_try < best_f - 1e-16:
                best_f = f_try
                best_h = h_try.copy()
                h_cur = h_try
    return best_h, best_f
