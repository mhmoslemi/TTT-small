import numpy as np

from scipy.optimize import linprog

def _proj(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid,0.0,1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-0.5*(lo+hi),0.0,1.0)

def _repair(h,n):
    h = np.asarray(h,dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h,0.0,1.0)
    target = 0.5*n
    if abs(h.sum()-target) > 1e-9:
        h = _proj(h,target)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for ii in order:
            if diff > 0:
                a = min(diff, 1.0 - h[ii])
                h[ii] += a; diff -= a
            else:
                a = min(-diff, h[ii])
                h[ii] -= a; diff += a
            if abs(diff) < 1e-11:
                break
    return h

import numpy as np
import time
from scipy.optimize import linprog


def _proj(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    if abs(h.sum() - target) > 1e-9:
        h = _proj(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for ii in order:
            if diff > 0:
                a = min(diff, 1.0 - h[ii])
                h[ii] += a
                diff -= a
            else:
                a = min(-diff, h[ii])
                h[ii] -= a
                diff += a
            if abs(diff) < 1e-11:
                break
    return h


def _counts(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _eval(h):
    n = len(h)
    return float(np.max(_counts(h)) * (2.0 / n))


def _default_h(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.20 * np.sin(2.0 * np.pi * x) + 0.10 * np.sin(4.0 * np.pi * x + 0.3)
    return _repair(np.clip(h, 0.0, 1.0), n)


def _to_binary(h, n):
    b = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1]
    b[idx[:n // 2]] = 1.0
    return b


def _upsample(h, n2):
    n = len(h)
    if n == n2:
        return _repair(h, n2)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(n2) + 0.5) * (2.0 / n2)
    y = np.interp(x1, x0, h)
    return _repair(y, n2)


def _grad_index(p, h, n):
    g = np.zeros(n, dtype=float)
    sh = n - 1 - p
    if sh >= 0:
        if sh == 0:
            g += 1.0 - 2.0 * h
        else:
            g[:n - sh] += 1.0 - h[sh:]
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:-sp]
        g[:n - sp] -= h[sp:]
    return g


def _slp_candidate(h, c, active, n):
    if len(active) == 0:
        return None
    rows = []
    rhs = []
    for p in active:
        p = int(p)
        g = _grad_index(p, h, n)
        rows.append(g)
        rhs.append(float(np.dot(g, h)) - float(c[p]))
    A = np.vstack(rows)
    A = np.concatenate((A, -np.ones((len(rows), 1))), axis=1)
    obj = np.zeros(n + 1)
    obj[n] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, np.inf)]
    try:
        res = linprog(
            obj,
            A_ub=A,
            b_ub=np.asarray(rhs, dtype=float),
            A_eq=Aeq,
            b_eq=np.asarray([0.5 * n]),
            bounds=bounds,
            method='highs',
            options={'time_limit': 6.0}
        )
    except Exception:
        return None
    if res is None or getattr(res, 'status', 1) not in (0, 1):
        return None
    x = np.asarray(res.x[:n], dtype=float)
    if not np.all(np.isfinite(x)):
        return None
    return x


def _slp(h, n, rng, deadline, cap):
    f = _eval(h)
    best = h.copy()
    bestf = f
    cap = max(32, int(cap))
    cap_cur = cap
    max_cap = 800 if n < 800 else 600
    stall = 0
    resets = 0

    while time.time() < deadline - 6.0:
        c = _counts(h)
        m = float(c.max())
        idx = np.argsort(c)[::-1]
        max_idx = np.where(c >= m - 1e-7)[0]

        if max_idx.size <= max_cap:
            act = max_idx.copy()
            if act.size < 48:
                ct = idx[:min(cap_cur, max_cap)]
                act = np.unique(np.concatenate((act, ct)))

        else:
            act = idx[:min(cap_cur, max_cap)]

        x = _slp_candidate(h, c, act, n)
        accepted = False
        if x is not None and float(np.max(np.abs(x - h))) > 1e-9:

                cand = _repair(h + lam * (x - h), n)

                if nf < f - 1e-14:
                    h = cand
                    f = nf
                    accepted = True
                    break

        if accepted:
            if f < bestf:
                bestf = f
                best = h.copy()
            stall = 0
            cap_cur = max(32, int(cap_cur * 0.85))

                stall = 0

            bestf = f
            best = h.copy()

    return best, bestf


def _weighted_grad(h, c, topk, alpha):
    n = len(h)
    topk = min(int(topk), len(c))
    idx = np.argsort(c)[::-1][:topk]
    sm = c[idx]
    s = np.clip(alpha * (sm - sm.max()), -400.0, 400.0)
    w = np.exp(s - s.max())
    ws = w.sum()
    if ws <= 0.0:
        return np.zeros(n, dtype=float)
    w /= ws
    g = np.zeros(n, dtype=float)
    for ii in range(idx.size):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_index(int(idx[ii]), h, n)
    return g


def _momentum_polish(h, n, rng, deadline, alpha, topk, step0):
    f = _eval(h)
    best = h.copy()
    bestf = f
    v = np.zeros(n, dtype=float)
    step = float(step0)
    stall = 0

    while time.time() < deadline - 2.0:
        c = _counts(h)
        g = _weighted_grad(h, c, topk, alpha)
        gn = float(np.max(np.abs(g)))
        if gn < 1e-12:
            break
        g /= gn
        v = 0.85 * v + 0.15 * g
        vn = float(np.max(np.abs(v)))
        if vn < 1e-12:
            break
        v /= vn

        accepted = False
        for lam in (step, 0.7 * step, 0.5 * step, 0.35 * step, 0.25 * step, 0.15 * step):
            cand = _repair(h - lam * v, n)
            nf = _eval(cand)
            if nf < f - 1e-15:
                h = cand
                f = nf
                accepted = True
                break

        if accepted:
            step = min(step * 1.03, 0.06)
            stall = 0
        else:
            step = max(step * 0.6, 1e-6)
            stall += 1
            if stall > 30:
                h = _repair(h + 0.003 * rng.randn(n), n)
                f = _eval(h)
                v = np.zeros(n, dtype=float)
                step = float(step0)
                stall = 0

        if f < bestf:
            bestf = f
            best = h.copy()

    return best, bestf


def _binary_search(hb, n, rng, deadline):

    bestc = float(np.max(_counts(hb)))
    center = n - 1
    ones = np.nonzero(hb > 0.5)[0]

            if ones.size and zeros.size:

                j = int(zeros[rng.randint(zeros.size)])
            else:
                continue
        else:
            j = i + sh

        if i == j:
            continue

        hb[i], hb[j] = hb[j], hb[i]
        nm = float(np.max(_counts(hb)))
        if nm < m - 1e-12:
            ones = np.nonzero(hb > 0.5)[0]
            zeros = np.nonzero(hb < 0.5)[0]
            if nm < bestc:
                bestc = nm
                best = hb.copy()
        else:
            hb[i], hb[j] = hb[j], hb[i]

    return best, bestc * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 18.0

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = _default_h(560, rng)

    n0 = int(len(h0))
    if n0 < 8:
        h0 = _default_h(560, rng)
        n0 = 560
    if n0 % 2:
        n0 -= 1
        h0 = h0[:n0]
    h0 = _repair(h0, n0)

    best_h = h0.copy()
    best_c = _eval(best_h)
    b = float(budget_s)

    d1 = min(deadline, t0 + 0.28 * b)
    if d1 > time.time() + 5.0:
        h, c = _slp(best_h, n0, rng, d1, 120)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    d2 = min(deadline, t0 + 0.52 * b)
    if d2 > time.time() + 5.0:
        h, c = _slp(best_h, n0, rng, d2, 280)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    d3 = min(deadline, t0 + 0.62 * b)
    if d3 > time.time() + 5.0:
        h, c = _momentum_polish(best_h, n0, rng, d3, 80, 40, 0.012)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    d4 = min(deadline, t0 + 0.70 * b)
    if d4 > time.time() + 5.0:
        hb = _to_binary(best_h, n0)
        hb, c = _binary_search(hb, n0, rng, d4)
        if c < best_c:
            best_c = c
            best_h = hb.copy()

    d5 = min(deadline, t0 + 0.76 * b)
    if d5 > time.time() + 5.0:
        h, c = _momentum_polish(best_h, n0, rng, d5, 600, 12, 0.008)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    if n0 < 900 and deadline - time.time() > max(160.0, 0.25 * b):
        n2 = 960
        hu = _upsample(best_h, n2)
        h, c = _slp(hu, n2, rng, deadline - 55.0, 220)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if deadline - time.time() > 35.0:
            h, c = _momentum_polish(best_h, n2, rng, deadline - 15.0, 120, 24, 0.008)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    if deadline - time.time() > 20.0:
        h, c = _momentum_polish(best_h, len(best_h), rng, deadline - 3.0, 200, 24, 0.006)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = _repair(best_h, int(len(best_h)))
    best_c = float(_eval(best_h))
    return (best_h, best_c, int(len(best_h)))
