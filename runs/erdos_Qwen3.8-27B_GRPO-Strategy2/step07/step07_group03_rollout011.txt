def _proj_to_sum(x, target):
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target:

        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    target = n * 0.5
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    h = _proj_to_sum(h, target)
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            if h[idx] >= 1.0:
                break
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 0.0:
                break
            a = min(-diff, h[idx])
            h[idx] -= a
    return h

def _cfull(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _eval(h):
    return float(np.max(_cfull(h)))

def _grad_value(h, idx, m):
    n = len(h)
    dx = 2.0 / n
    shift = (n - 1) - idx
    if shift == 0:
        return dx * (1.0 - 2.0 * h[m])
    p = abs(shift)
    if shift > 0:
        v = 0.0
        if m <= n - 1 - p:
            v += 1.0 - h[m + p]
        if m >= p:
            v -= h[m - p]
        return dx * v
    v = 0.0
    if m >= p:
        v += 1.0 - h[m - p]
    if m <= n - 1 - p:
        v -= h[m + p]
    return dx * v

def _grad_vec(h, idx):
    n = len(h)
    dx = 2.0 / n
    shift = (n - 1) - idx
    g = np.zeros(n)
    if shift == 0:
        return dx * (1.0 - 2.0 * h)
    p = abs(shift)
    if shift > 0:
        g[:n - p] += 1.0
        g[:n - p] -= h[p:]
        g[p:] -= h[:n - p]
    else:
        g[p:] += 1.0
        g[p:] -= h[:n - p]
        g[:n - p] -= h[p:]
    return g * dx

def _smooth_fun(x, alpha, pen):
    n = x.size
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    s -= s.max()
    val = float(m + np.log(np.sum(np.exp(s))) / alpha)
    if pen:
        dsum = x.sum() - n * 0.5
        val += pen * dsum * dsum
    return val

def _smooth_jac(x, alpha, pen):
    n = x.size
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi <= 1e-12:
            continue
        shift = center - idx
        wv = dx * wi
        if shift == 0:
            g += wv * (1.0 - 2.0 * x)
        elif shift > 0:
            p = shift
            g[:n - p] += wv
            g[:n - p] -= wv * x[p:]
            g[p:] -= wv * x[:n - p]
        else:
            p = -shift
            g[p:] += wv
            g[p:] -= wv * x[:n - p]
            g[:n - p] -= wv * x[p:]
    if pen:
        g += (2.0 * pen) * (x.sum() - n * 0.5)
    return g

        if res.x is not None and np.all(np.isfinite(res.x)):

    return h

        if abs(d) <= 1e-13:
            continue
        h[i] = oi + d
        h[j] = oj - d
        cn = _cfull(h)
        v = float(cn.max())
        if v < best_val - 1e-14:
            best_val = v
            best_d = d
            best_c = cn
    if best_d != 0.0:
        h[i] = oi + best_d
        h[j] = oj - best_d
        return h, best_c, best_val, True
    h[i] = oi
    h[j] = oj
    return h, None, c5, False

def _coordinate_phase(h, t_end, T, P, Rt, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    c = _cfull(h)
    c5 = float(c.max())
    best_h = h.copy()
    best_c5 = c5
    T = min(T, 2 * n - 1)
    P = max(3, min(P, n // 4))
    stall = 0
    while time.time() < t_end:
        if c is None:
            c = _cfull(h)
            c5 = float(c.max())
        idx = np.argsort(c)[-T:]
        top_c = c[idx]
        w = np.exp(20.0 * (top_c - top_c.max()))
        w /= w.sum()
        g = np.zeros(n)
        for k, wv in zip(idx, w):
            g += wv * _grad_vec(h, k)
        low = np.argpartition(g, P - 1)[:P]
        high = np.argpartition(g, n - P)[-P:]
        improved = False
        for ii in low:
            if time.time() > t_end:
                break
            ii = int(ii)
            for jj in high:
                if time.time() > t_end:
                    break
                jj = int(jj)
                if ii == jj:
                    continue
                h, c, c5, acc = _pair_search(h, ii, jj, idx, top_c, c5)
                if acc:
                    improved = True
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h.copy()
                    idx = np.argsort(c)[-T:]
                    top_c = c[idx]
        for _ in range(Rt):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(1, n // 10))) % n
            h, c, c5, acc = _pair_search(h, i, j, idx, top_c, c5)
            if acc:
                improved = True
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                idx = np.argsort(c)[-T:]
                top_c = c[idx]
        if not improved:
            stall += 1
            if stall > 4:
                scale = 1e-4 + 1e-5 * stall
                h = _repair(best_h + scale * rng.randn(n), n)
                c = _cfull(h)
                c5 = float(c.max())
                stall = 0
        else:
            stall = 0
    return best_h, best_c5

def _tiny_phase(h, t_end, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = _eval(h)
    step = 1e-4
    stall = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        v = _eval(h)
        if v < best_c5 - 1e-14:
            best_c5 = v
            best_h = h.copy()
            stall = 0
        else:
            h[i] = h[i] - d
            h[j] = h[j] + d
            stall += 1
            if stall > 4000:
                step = min(step * 2.0, 5e-3)
                stall = 0
                h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 2.0
    rng = np.random.RandomState(seed)
    try:
        n = len(initial_h_values)
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        n = 560
        h0 = np.full(n, 0.5, dtype=float)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(h)
    best_h, best_c5 = _transforms(best_h, n)

    if budget_s > 120:
        t_lb = min(t0 + min(240.0, budget_s * 0.28), deadline - 70.0)
        for alpha, iters in ((20, 45), (100, 32), (500, 22), (2000, 14)):
            if time.time() > t_lb:
                break
            h1 = _lbfgs(best_h.copy(), float(alpha), 120.0, iters)
            v = _eval(h1)
            if v < best_c5 - 1e-13:
                best_c5 = v
                best_h = h1.copy()
        best_h, best_c5 = _transforms(best_h, n)

        t_fw_start = time.time()
        t_fw_end = min(t_fw_start + min(90.0, max(0.0, (deadline - t_fw_start) * 0.12)), deadline - 60.0)
        for alpha in (8.0, 40.0, 160.0):
            while time.time() < t_fw_end:
                h, c, c5, acc = _fw_step(best_h.copy(), 10, alpha)
                if acc and c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    break
        best_h, best_c5 = _transforms(best_h, n)

    if time.time() < deadline - 50:
        h1, c1 = _coordinate_phase(best_h.copy(), deadline - 65, 28, 11, 14, seed + 1)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1
        best_h, best_c5 = _transforms(best_h, n)

    if time.time() < deadline - 40:
        h1, c1 = _coordinate_phase(best_h.copy(), deadline - 40, 18, 15, 20, seed + 2)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1
        best_h, best_c5 = _transforms(best_h, n)

    if time.time() < deadline - 5:
        h1, c1 = _tiny_phase(best_h.copy(), deadline - 2.0, seed + 3)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return best_h, float(best_c5), int(n)
