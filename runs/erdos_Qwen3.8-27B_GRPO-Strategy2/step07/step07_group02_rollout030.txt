def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = 0.5 * n
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)

    G = np.zeros(n)

            continue

        wk = w[k] * dx
        if sh >= 0:
            G[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:

        else:
            sp = -sh
            G[sp:] += wk * (1.0 - x[:-sp])
            G[:-sp] -= wk * x[sp:]
    return G

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-11,
                                'maxfun': maxit * 3})
        return _repair(res.x, n)
    except Exception:
        return h

def _active_descent(h, n_active, n_iter, deadline):
    """Subgradient descent targeting top-n_active lags."""
    n = len(h)
    dx = 2.0 / n
    t = 0.05
    for i in range(n_iter):
        if time.time() > deadline - 5:
            break
        c = _cvals(h)
        top = np.argsort(c)[-n_active:]
        G = np.zeros(n)
        for k in top:

            if sh >= 0:

                if sh > 0:
                    G[sh:] -= h[:n-sh]
            else:
                sp = -sh
                G[sp:] += (1.0 - h[:-sp])
                G[:-sp] -= h[sp:]
        G *= dx / n_active
        h = _repair(h - t * G, n)
        t *= 0.995
    return h

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    G = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            G[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                G[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            G[sp:] += wk * (1.0 - x[:-sp])
            G[:-sp] -= wk * x[sp:]
    return G

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 25:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-17, 'gtol': 1e-12,
                                'maxfun': maxit * 4, 'factr': 1e8})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad(h, steps, step0, tol, deadline):
    n = len(h)
    dx = 2.0 / n
    t = step0
    for i in range(steps):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        G = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                G[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    G[sh:] -= h[:n - sh]
            else:
                sp = -sh
                G[sp:] += (1.0 - h[:-sp])
                G[:-sp] -= h[sp:]
        G *= dx / len(active)
        h = _repair(h - t * G, n)
        t = max(t * 0.997, step0 * 0.01)
    return h

def _adaptive_2pt(h, n_iter, rng, deadline):

    c5c = _eval(h)

def _adaptive_3pt(h, n_iter, rng, deadline):

def _dual_proj(h, g, mu, deadline):

    c = _cvals(h)
    m = c.max()

    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()
