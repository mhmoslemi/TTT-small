    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _solve_1d(a, b):
    """Solve min_{t in [0,1]} max_k (a[k]*t + b[k])."""
    L = len(a)
    best_t = 0.0
    best_v = np.inf
    
    # Evaluate at 0 and 1
    for tc in [0.0, 1.0]:
        v = np.max(a * tc + b)
        if v < best_v:
            best_v = v
            best_t = tc
    
    # Pairwise intersections
    # t*(p,q) = (b[q] - b[p]) / (a[p] - a[q])
    for p in range(L):
        for q in range(p + 1, L):
            da = a[p] - a[q]
            if abs(da) < 1e-15:
                continue
            t_int = (b[q] - b[p]) / da
            if 0.0 < t_int < 1.0:
                v = np.max(a * t_int + b)
                if v < best_v - 1e-15:
                    best_v = v
                    best_t = t_int
    
    return best_t

    n = len(h)

    best_h = h.copy()

        if time.time() > deadline - 60:
            break

            else:
                top_lags = np.arange(m)

            # Compute a_k, b_k
            a = np.zeros(top_L.size)
            b = np.zeros(top_L.size)
            for idx, k in enumerate(top_lags):
                lag = int(k) - (n - 1)
                val = 0.0
                j1 = i + lag
                if 0 <= j1 < n:
                    val += (1.0 - h[j1])
                j2 = i - lag
                if 0 <= j2 < n:
                    val -= h[j2]
                a[idx] = val
                b[idx] = C[k] - val * t_old
            
            t_new = _solve_1d(a, b)

                    j2 = i - lag
                    if 0 <= j2 < n:
                        dc -= delta * h[j2]
                    C[k] += dc

        c5 = _eval(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

    n = len(h)
    if time.time() > deadline - 30:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n_iter, rng, deadline, pts=2):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        if pts == 2:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = 8e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        else:
            i = rng.randint(n)
            j = (i + rng.randint(2, 7)) % n
            k = (i + rng.randint(8, 16)) % n
            d1 = 4e-5 * rng.randn()
            d2 = 4e-5 * rng.randn()
            ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
            if 0<=ni<=1 and 0<=nj<=1 and 0<=nk<=1:
                h[i], h[j], h[k] = ni, nj, nk
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return best_h, best_c

def _down(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _up(h, f):
    return np.repeat(h, f).copy()

def _make_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 1, n)
    h = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.05 * rng.randn(n)
    return _repair(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)
    best_n = n0

    # Phase 1: Coordinate descent on initial (main new method)
    h = best_h.copy()
    h, c5v = _coord_descent(h, 150, 40, deadline, seed=seed)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()

    # Phase 2: L-BFGS refinement
    if time.time() < deadline - 350:
        h = _lbfgs(best_h.copy(), 5000, 3000, deadline)
        h = _repair(h, best_n)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 3: More coordinate descent (finer, more sweeps)
    if time.time() < deadline - 300:
        h = _coord_descent(best_h.copy(), 200, 25, deadline, seed=seed + 1)
        if _eval(h) < best_c5:
            best_c5 = _eval(h)
            best_h = h.copy()

    # Phase 4: L-BFGS with higher alpha
    if time.time() < deadline - 250:
        h = _lbfgs(best_h.copy(), 50000, 4000, deadline)
        h = _repair(h, best_n)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 5: Multi-resolution - downsample, optimize, upsample
    if time.time() < deadline - 200:
        nh = n0 // 2
        hh = _repair(_down(best_h, 2), nh)
        hh, c5h = _coord_descent(hh, 200, 30, deadline, seed=seed + 2)
        hu = _repair(_up(hh, 2), n0)
        c5u = _eval(hu)
        if c5u < best_c5:
            best_c5, best_h = c5u, hu.copy()
        # Also try a fresh init at half res
        h_fresh = _make_initial(nh, seed + 100)
        h_fresh, c5f = _coord_descent(h_fresh, 300, 30, deadline, seed=seed + 3)
        hu2 = _repair(_up(h_fresh, 2), n0)
        c5u2 = _eval(hu2)
        if c5u2 < best_c5:
            best_c5, best_h = c5u2, hu2.copy()

    # Phase 6: Coordinate descent at full res with very tight top
    if time.time() < deadline - 150:
        h = _coord_descent(best_h.copy(), 300, 15, deadline, seed=seed + 4)
        if _eval(h) < best_c5:
            best_c5 = _eval(h)
            best_h = h.copy()

    # Phase 7: L-BFGS final polish
    if time.time() < deadline - 100:
        h = _lbfgs(best_h.copy(), 200000, 5000, deadline)
        h = _repair(h, best_n)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 8: 3-point local search
    if time.time() < deadline - 50:
        h_ls, c5_ls = _local_search(best_h.copy(), 50000, rng, deadline, pts=3)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    # Phase 9: 2-point ultra-fine polish
    if time.time() < deadline - 5:
        h_ls, c5_ls = _local_search(best_h.copy(), 100000, rng, deadline, pts=2)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
