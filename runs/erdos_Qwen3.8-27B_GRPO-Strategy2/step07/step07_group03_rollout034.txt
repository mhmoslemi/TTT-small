                no_improve += 1
            if no_improve > 50:
                freq = rng.choice(np.arange(2, 20, 1))
                phase = rng.uniform(0, 2 * np.pi)
                h += 3e-4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                h = _proj_to_sum(h, n * 0.5)
                no_improve = 0
        if sweep_count % 200 == 0:
            h = _repair(h, n)
    final_c5 = _eval(h)
    return h, min(best_c5, final_c5)

def _subgrad_polish(h, n, duration_s, t_end):
    dx = 2.0 / n
    end_t = time.time() + duration_s
    step = 0.002
    h_prev = h.copy()
    g_prev = np.zeros(n)
    f = _eval(h)
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = _correlate(h)
        m = c.max()
        s_arr = np.clip(2000.0 * (c - m), -500, 500)
        w = np.exp(s_arr - s_arr.max())
        w /= w.sum()
        top_k = np.where(w > 1e-10)[0]
        if len(top_k) == 0:
            top_k = np.argsort(c)[-5:]
        g = np.zeros(n)
        for k in top_k:
            if w[k] < 1e-10:
                continue
            s = k - (n - 1)
            wk = w[k] * dx
            if s >= 0:
                i1 = np.arange(n - s)
                g[i1] += wk * (1.0 - h[i1 + s])
                if s > 0:
                    i2 = np.arange(s, n)
                    g[i2] -= wk * h[i2 - s]
            else:
                sp = -s
                i1 = np.arange(sp, n)
                g[i1] += wk * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wk * h[i2 + sp]
        h_new = _proj_to_sum(h - step * g, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            y = g - g_prev
            sy = np.dot(h_new - h_prev, y)
            yy = np.dot(y, y)
            if yy > 1e-16 and sy > 1e-16:
                step = min(step * 1.5, 0.02)
            h = h_new; f = f_new; g_prev = g; h_prev = h_new
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1
            if no_improve > 80:
                h = h_prev.copy()
                f = _eval(h)
                step = 0.001
                no_improve = 0
    return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Brief Nesterov warm-up
    for alpha in [8000, 40000, 200000]:
        t_end = min(time.time() + 12, t0 + budget_s * 0.06)
        h = _nesterov_brief(h, alpha, 200, 0.003, t_end)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 2: Main coordinate descent (larger active set, exact 1D solve)
    t_end_cd1 = t0 + budget_s * 0.45
    h = best_h.copy()
    h, c5v = _coord_descent_v2(h, n, max(2, t_end_cd1 - time.time()), 12, rng, t0 + budget_s)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h.copy()

    # Phase 3: Subgradient polish
    if time.time() < t0 + budget_s * 0.65:
        h = best_h.copy()
        h, c5v = _subgrad_polish(h, n, max(2, (t0 + budget_s * 0.65) - time.time()), t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 4: More coordinate descent with different params
    if time.time() < t0 + budget_s * 0.85:
        for top_T in [8, 15, 10]:
            if time.time() > t0 + budget_s * 0.83:
                break
            h = best_h.copy()
            h, c5v = _coord_descent_v2(h, n, max(2, (t0 + budget_s * 0.84) - time.time()), top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()

    # Phase 5: Final fine local search
    if time.time() < t0 + budget_s - 10:
        remaining = (t0 + budget_s) - time.time() - 8
        n = len(best_h)
        c5c = _eval(best_h)
        best_local = best_h.copy()
        best_local_c = c5c
        tries = 0
        while time.time() < t0 + budget_s - 8 and tries < 60000:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 7)) % n
            d = 3e-6 * rng.randn()
            ni, nj = best_local[i] + d, best_local[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_local[i], best_local[j] = ni, nj
                c5n = _eval(best_local)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_local_c:
                        best_local_c = c5n
                else:
                    best_local[i], best_local[j] = best_local[i] - d, best_local[j] + d
            tries += 1
        if best_local_c < best_c5:
            best_c5 = best_local_c; best_h = best_local

    # Phase 6: Final coordinate polish
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        h, c5v = _coord_descent_v2(h, n, max(1, (t0 + budget_s - 3) - time.time()), 10, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
