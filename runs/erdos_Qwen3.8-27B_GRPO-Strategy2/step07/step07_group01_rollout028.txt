    """Gradient of c at shift s (where s = k-(n-1), range [-(n-1), n-1])."""
    dx = 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12: return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s; h = np.clip(h, 0, 1); s = h.sum()
        if abs(s - t) < 1e-14: break
    d = t - h.sum()
    if abs(d) > 1e-14:
        for i in (np.argsort(1-h) if d > 0 else np.argsort(h)):
            if d > 0:
                a = min(d, 1-h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-15: break
    return h

        else: hi = m
    return np.clip(h-(lo+hi)*.5, 0, 1)

        return m + np.log(e.sum())/al
    def gr(x):
        c = np.correlate(x, 1.-x, mode='full')*dx; m = c.max()
        e = np.exp(np.clip(al*(c-m), -500, 500)); w = e/e.sum()

                if sh > 0:
                    i2 = np.arange(sh, n); g[i2] -= wk*x[i2-sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n); g[i1] += wk*(1-x[i1-sp])
                i2 = np.arange(n-sp); g[i2] -= wk*x[i2+sp]
        return g
    def sf(x): return x.sum()-n*.5
    def sj(x): return np.ones(n)
    try:

                     constraints=[{'type':'eq','fun':sf,'jac':sj}],
