    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    g += 2.0 * penalty * (h.sum() - n * 0.5)
    return g

def _lse_optimize(h, T, penalty, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    def ob(x):
        return _lse_obj(x, T, penalty, n)
    def gr(x):
        return _lse_grad(x, T, penalty, n)
    try:
        res = minimize(ob, h, jac=gr, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descent(h, n_passes, n_trials, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy()
    best_c = _eval(h_c)
    best_h = h_c.copy()
    for p in range(n_passes):
        if bc_fn():
            break
        span = max(0.05, 0.3 * (0.85 ** p))
        improved = 0
        order = np.random.RandomState(42 + p).permutation(n)
        for idx in range(n):
            if bc_fn():
                break
            i = int(order[idx])
            old = h_c[i]
            if old <= 0.001 or old >= 0.999:
                continue
            lo_s = max(0.0, old - span)
            hi_s = min(1.0, old + span)
            if hi_s - lo_s < 1e-6:
                continue
            best_val = best_c
            best_xi = old
            for t in range(n_trials):
                xi = lo_s + (hi_s - lo_s) * t / max(1, n_trials - 1)
                if abs(xi - old) < 1e-7:
                    continue
                h_c[i] = xi
                c_new = _eval(h_c)
                if c_new < best_val:
                    best_val = c_new
                    best_xi = xi
            h_c[i] = best_xi
            h_c = _proj_sum(h_c, n * 0.5)
            cv = _eval(h_c)
            if cv < best_c:
                best_c = cv
                best_h = h_c.copy()
                improved += 1
        if improved == 0:
            break
    return best_h, best_c

def _nesterov_topk(h, n_iter, k_start, k_max, step0, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.7
    step = step0
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_start
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        if stall > 40:
            k = min(k + 1, k_max)
        topk = np.argsort(c)[-k:]
        g = np.zeros(n)
        for ki in topk:
            sh = (n - 1) - ki
            if sh >= 0:
                g[:n - sh] += (1.0 - h_c[sh:]) * 2.0 / n
                if sh > 0:
                    g[sh:] -= h_c[:n - sh] * 2.0 / n
            else:
                sp = -sh
                g[sp:] += (1.0 - h_c[:n - sp]) * 2.0 / n
                g[:n - sp] -= h_c[sp:] * 2.0 / n
        g /= k
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            stall = 0
            step = min(step * 1.04, 0.06)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _fine_local(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 5e-5
    for _ in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = rng.randn() * scale
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h_c[i], h_c[j] = ni, nj
            c_new = _eval(h_c)
            if c_new < best_c:
                best_c = c_new
                best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
        if _ % 3000 == 0:
            scale *= 0.75
    return best_h, best_c

def _perturb(h, sigma, rng, n):
    t = np.linspace(0, 2, n, endpoint=False)
    noise = np.zeros(n)
    for freq in [1, 2, 3, 5, 7, 11]:
        noise += sigma * rng.randn() * np.sin(np.pi * freq * t)
    h2 = np.clip(h + noise, 0, 1)
    return _proj_sum(h2, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    def bc():
        return el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: LSE optimization with multiple temperatures
    for T in [0.3, 0.15, 0.07, 0.03, 0.015, 0.008]:
        if el() > 80:
            break
        h_try = _lse_optimize(best_h.copy(), T, 1e6, 400, n0)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 2: Coordinate descent (wide scan)
    if el() < budget_s - 200:
        nt = min(int((budget_s - 200 - el()) / 8.0), 40)
        best_h, best_c5 = _coord_descent(best_h, max(nt, 5), 25, el, bc)

    # Phase 3: LSE polish
    if el() < budget_s - 150:
        for T in [0.01, 0.005, 0.002]:
            if el() > budget_s - 150:
                break
            h_try = _lse_optimize(best_h.copy(), T, 1e7, 300, n0)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 4: Coordinate descent (narrower scan)
    if el() < budget_s - 120:
        nt = min(int((budget_s - 120 - el()) / 6.0), 60)
        best_h, best_c5 = _coord_descent(best_h, max(nt, 10), 20, el, bc)

    # Phase 5: Nesterov top-k
    if el() < budget_s - 100:
        h_try, c_try = _nesterov_topk(best_h, 6000, 3, 50, 0.01, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 6: Multi-restart coordinate descent
    if el() < budget_s - 80:
        for ri in range(3):
            if el() > budget_s - 80:
                break
            h_pert = _perturb(best_h, 0.02 * (ri + 1), rng, n0)
            h_try, c_try = _coord_descent(h_pert, 15, 15, el, bc)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    # Phase 7: Nesterov fine
    if el() < budget_s - 60:
        h_try, c_try = _nesterov_topk(best_h, 8000, 2, 30, 0.004, el, bc)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try

    # Phase 8: Final coordinate descent (very narrow)
    if el() < budget_s - 40:
        nt = min(int((budget_s - 40 - el()) / 5.0), 50)
        best_h, best_c5 = _coord_descent(best_h, max(nt, 10), 15, el, bc)

    # Phase 9: Fine local search
    if el() < budget_s - 10:
        nt = min(int((budget_s - 10 - el()) / 0.001), 150000)
        best_h, best_c5 = _fine_local(best_h, nt, rng, el, bc)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
