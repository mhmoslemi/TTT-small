    c = np.correlate(h, 1-h, mode='full') * dx
    top = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top:
        s = k - (n-1)
        # grad[j] += dx * [(1-h[j+s]) if j+s in [0,n) else 0] - dx * [h[j-s] if j-s in [0,n) else 0]
        if s >= 0:
            # (1-h)[j+s] for j in [0, n-s-1]
            g[:n-s] += dx * (1 - h[s:])
            # -h[j-s] for j in [s, n-1]
            g[s:] -= dx * h[:n-s]
        else:

            # (1-h)[j-sp] for j in [sp, n-1]
            g[sp:] += dx * (1 - h[:n-sp])
            # -h[j+sp] for j in [0, n-sp-1]
            g[:n-sp] -= dx * h[sp:]
    g /= top_k
    return g, c.max()

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    h = h * target / max(s, 1e-15)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx

    best_h, best_c = h.copy(), c5_cur
    stall = 0
    while time.time() < deadline:
        g, _, _ = _grad(h, n, top_k)
        i, j = np.argmin(g), np.argmax(g)
        if i == j:
            break
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            break
        direction = _pos_grad(h, i, n) - _pos_grad(h, j, n)
        k1, k2 = n - 1 + j - i, n - 1 + i - j
        for _ in range(35):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * direction
            if 0 <= k1 < 2 * n - 1:
                vals[k1] -= d_mid * d_mid * dx
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid * dx
            kmax = np.argmax(vals)
            sg = direction[kmax]
            if kmax == k1:
                sg -= 2.0 * d_mid * dx
            if kmax == k2:
                sg += 2.0 * d_mid * dx
            if sg > 1e-13:
                d_hi = d_mid
            elif sg < -1e-13:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-14:
            stall += 1
            top_k = min(top_k + 2, 30)
            if stall > 2000:
                break
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * direction
        if 0 <= k1 < 2 * n - 1:
            c[k1] -= d_opt * d_opt * dx
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt * dx
        c5_new = c.max()
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            stall = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * direction
            if 0 <= k1 < 2 * n - 1:
                c[k1] += d_opt * d_opt * dx
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt * dx
            stall += 1
            if stall > 2000:
                break
    return best_h, best_c

def _random_cd(h0, n, rng, t0, deadline, top_k=10):
    h = h0.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h, best_c = h.copy(), c5_cur
    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:

        direction = _pos_grad(h, i, n) - _pos_grad(h, j, n)
        k1, k2 = n - 1 + j - i, n - 1 + i - j
        for _ in range(30):

            vals = c + d_mid * direction
            if 0 <= k1 < 2 * n - 1:
                vals[k1] -= d_mid * d_mid * dx
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid * dx
            kmax = np.argmax(vals)
            sg = direction[kmax]
            if kmax == k1:
                sg -= 2.0 * d_mid * dx
            if kmax == k2:

            if sg > 1e-13:

            if 0 <= k1 < 2 * n - 1:

def _adam_descent(h0, n, rng, t0, deadline, tk0=5, tk_max=20):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h, best_c = h.copy(), _eval(h)
    c5_cur = best_c
    lr = 0.04
    t = 0
    no_imp = 0
    while time.time() < deadline:
        t += 1
        tk = min(tk0 + t // 150, tk_max)
        g, c5_val, _ = _grad(h, n, tk)
        if c5_val < best_c - 1e-15:
            best_c = c5_val
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 80:
                lr *= 0.75
                no_imp = 0
                if lr < 1e-6:
                    break
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj_sum(h - step, n * 0.5)
        c5n = _eval(h_new)
        if c5n <= c5_cur + 1e-14:
            h = h_new
            c5_cur = c5n
        else:
            for mult in [0.5, 0.2, 0.05]:
                h_try = _proj_sum(h - mult * step, n * 0.5)
                c5t = _eval(h_try)
                if c5t <= c5_cur + 1e-14:
                    h = h_try
                    c5_cur = c5t
                    break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    for ps in [0.02, 0.05, 0.1, 0.2, 0.35]:
        r = np.random.RandomState(seed + int(ps * 997))
        hp = _proj_sum(h + ps * r.randn(n0), n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 1: Adam coarse (10%)
    dl1 = t0 + budget_s * 0.10
    hh, cc = _adam_descent(best_h.copy(), n0, rng, t0, dl1)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 2: Greedy CD (25%)
    dl2 = t0 + budget_s * 0.35
    hh, cc = _greedy_cd(best_h.copy(), n0, t0, dl2, top_k=8)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 3: Random CD (25%)
    dl3 = t0 + budget_s * 0.60
    hh, cc = _random_cd(best_h.copy(), n0, rng, t0, dl3, top_k=10)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 4: Greedy CD again with wider top_k (15%)
    dl4 = t0 + budget_s * 0.75
    hh, cc = _greedy_cd(best_h.copy(), n0, t0, dl4, top_k=15)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 5: Adam fine (8%)
    dl5 = t0 + budget_s * 0.83
    hh, cc = _adam_descent(best_h.copy(), n0, rng, t0, dl5, tk0=10, tk_max=25)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 6: Final greedy + random polish (14%)
    dl6 = t0 + budget_s * 0.92
    hh, cc = _greedy_cd(best_h.copy(), n0, t0, dl6, top_k=20)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    dl7 = t0 + budget_s - 5
    hh, cc = _random_cd(best_h.copy(), n0, rng, t0, dl7, top_k=15)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
