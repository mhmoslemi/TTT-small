                    a_lo = a_mid
                else:
                    break
            a = best_a
            if a < 1e-12:
                continue
            h[i_dec] += a; h[i_inc] -= a
            c += a * gc
            if 0 <= k1 < total:
                c[k1] += a * a
            if 0 <= k2 < total:
                c[k2] += a * a
            c5 = float(c.max())
            made_move = True
            if c5 < best_c5:
                best_c5 = c5; best_h = h.copy()
        if not made_move:
            break
    return best_h, best_c5

def _local_polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for it in range(200000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-16:
                c5_cur = c5n
                if c5n < best_c5:
                    best_c5 = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    for ps in [0.02, 0.06, 0.12]:
        hp = _proj_sum(np.clip(h + ps * rng.randn(n0), 0, 1), n0 * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p; best_h = hp.copy()

    dl1 = t0 + budget_s * 0.15
    h1, c5_1 = _subgrad_phase(best_h.copy(), rng, dl1, m_top=5)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    dl2 = t0 + budget_s * 0.50
    c = _eval_full(best_h)
    h2, c5_2 = _grad_cd(best_h.copy(), c, n0, dl2, rng)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    dl3 = t0 + budget_s * 0.65
    h3, c5_3 = _subgrad_phase(best_h.copy(), rng, dl3, m_top=8)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    dl4 = t0 + budget_s * 0.82
    c = _eval_full(best_h)
    h4, c5_4 = _multi_cd(best_h.copy(), c, n0, dl4, rng)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    dl5 = t0 + budget_s * 0.92
    c = _eval_full(best_h)
    h5, c5_5 = _grad_cd(best_h.copy(), c, n0, dl5, rng)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    dl6 = t0 + budget_s - 5
    h6, c5_6 = _local_polish(best_h.copy(), rng, dl6)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
