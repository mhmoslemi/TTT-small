            except Exception:
                pass
    return _repair(h, n)

def _perturb_restart(h, rng, frac):
    n = len(h)
    h_try = h.copy()
    mask = rng.random(n) < frac
    h_try[mask] += 0.03 * rng.randn(mask.sum())
    return _repair(h_try, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Nesterov projected gradient (medium alpha, top-6)
    t_lim1 = t0 + 220
    h1, c1 = _nesterov_phase(h.copy(), 3000, 0.004, 6, 0.85, 50000, t_lim1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Nesterov with higher alpha, top-4 (sharper max focus)
    t_lim2 = t0 + 380
    h2, c2 = _nesterov_phase(best_h.copy(), 30000, 0.002, 4, 0.8, 40000, t_lim2)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 3: Nesterov with very high alpha, top-3 (near-exact max)
    t_lim3 = t0 + 500
    h3, c3 = _nesterov_phase(best_h.copy(), 200000, 0.001, 3, 0.75, 30000, t_lim3)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()

    # Phase 4: SLSQP block coordinate on active regions
    t_lim4 = t0 + 620
    if _el() < t_lim4 - 60:
        for bsz in [12, 20]:
            h4 = _slsqp_block(best_h.copy(), bsz, 60)
            c4 = _eval(h4)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4
            if time.time() > t_lim4:
                break

    # Phase 5: Coordinate descent (long, fine)
    t_lim5 = t0 + 850
    n_cd = int(max(1, (t_lim5 - _el()) / 0.0004))
    h5, c5 = _coord_search(best_h.copy(), n_cd, rng, 2e-4, t_lim5)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()

    # Phase 6: Multi-restart local search
    t_lim6 = t0 + 950
    if _el() < t_lim6 - 30:
        for _r in range(4):
            if time.time() > t_lim6:
                break
            h_try = _perturb_restart(best_h, rng, 0.08)
            h_try, c_try = _coord_search(h_try, 30000, rng, 1e-4, t_lim6)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()

    # Phase 7: Ultra-fine 2-point polish
    t_lim7 = t0 + budget_s - 2
    h2f = best_h.copy()
    c5c = _eval(h2f)
    step2 = 5e-7
    npol = int(max(1, (t_lim7 - _el()) / 0.0005))
    for _ in range(npol):
        if time.time() > t_lim7:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step2 * rng.randn()
        ni, nj = h2f[i] + d, h2f[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2f[i], h2f[j] = ni, nj
            cn = _eval(h2f)
            if cn < c5c:
                c5c = cn
                if cn < best_c5:
                    best_c5 = cn
                    best_h = h2f.copy()
                step2 = min(step2 * 1.02, 1e-6)
            else:
                h2f[i], h2f[j] = h2f[i] - d, h2f[j] + d
                step2 = max(step2 * 0.98, 1e-8)

    # Phase 8: 3-point very fine polish
    t_lim8 = t0 + budget_s - 0.5
    if _el() < t_lim8 - 1:
        h3f = best_h.copy()
        c3c = _eval(h3f)
        step3 = 3e-7
        npol3 = int(max(1, (t_lim8 - _el()) / 0.0005))
        for _ in range(npol3):
            if time.time() > t_lim8:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            k = (i + 6 + rng.randint(0, 8)) % n
            d1 = step3 * rng.randn()
            d2 = step3 * rng.randn()
            ni, nj, nk = h3f[i] + d1, h3f[j] + d2, h3f[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h3f[i], h3f[j], h3f[k] = ni, nj, nk
                cn = _eval(h3f)
                if cn < c3c:
                    c3c = cn
                    if cn < best_c5:
                        best_c5 = cn
                        best_h = h3f.copy()
                else:
                    h3f[i], h3f[j], h3f[k] = ni - d1, nj - d2, nk + d1 + d2

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
