        try:
            res = minimize(fo, c_try, jac=fg, method='L-BFGS-B',
                          options={'maxiter': 1500, 'ftol': 1e-13, 'gtol': 1e-9})
            if res.fun < best_v:
                best_v = res.fun
                best_c = res.x.copy()
        except Exception:
            pass
        if time.time() > deadline - 30:
            break
    return _repair(_fourier_eval(best_c, M, n), n)

def _top_lag_descent(h, n_lags, n_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(n_rounds):
        if time.time() > deadline - 20:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_lags
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5, 2e-5]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _local_search(h, n_iter, scale, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 2:
            break
        i = rng.randint(n)
        dist = 1 + rng.randint(0, 4)
        j = (i + dist) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _upscale(h, f):
    return np.repeat(h, f).copy()

def _downscale(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine optimization
    # Start at n/4, optimize, then upsample and refine
    n_coarse = n0 // 4
    h_c = _downscale(best_h, 4)
    h_c = _repair(h_c, n_coarse)
    for alpha in [1000, 5000, 20000]:
        if time.time() > deadline - 700:
            break
        h_c = _lbfgs_obj(h_c, _smooth_obj, _smooth_grad, (alpha,), 3000, deadline)
        c5c = _eval(h_c)
    # Upscale to n/2
    h_mid = _repair(_upscale(h_c, 2), n0 // 2)
    for alpha in [2000, 10000, 50000]:
        if time.time() > deadline - 600:
            break
        h_mid = _lbfgs_obj(h_mid, _smooth_obj, _smooth_grad, (alpha,), 2000, deadline)
    # Upscale to full
    h_up = _repair(_upscale(h_mid, 2), n0)
    c5v = _eval(h_up)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_up.copy()

    # Phase 2: Fourier optimization at full resolution
    for M in [12, 18, 25]:
        if time.time() > deadline - 450:
            break
        h_f = _fourier_opt(best_h, n0, M, deadline - 200, rng)
        c5f = _eval(h_f)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = h_f.copy()

    # Phase 3: L-BFGS with increasing alpha on best
    h = best_h.copy()
    for alpha in [5000, 20000, 80000, 300000]:
        if time.time() > deadline - 350:
            break
        h = _lbfgs_obj(h, _smooth_obj, _smooth_grad, (alpha,), 4000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: p-norm optimization (smoother, different basin)
    h = best_h.copy()
    for p in [8, 15, 40, 100]:
        if time.time() > deadline - 250:
            break
        h = _lbfgs_obj(h, _pnorm_obj, _pnorm_grad, (p,), 3000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Random perturbations to escape local optima
    for ntries in range(8):
        if time.time() > deadline - 200:
            break
        scale = 0.03 + 0.02 * ntries
        h_p = _repair(best_h + scale * rng.randn(n0), n0)
        for alpha in [5000, 50000]:
            if time.time() > deadline - 180:
                break
            h_p = _lbfgs_obj(h_p, _smooth_obj, _smooth_grad, (alpha,), 2000, deadline)
            c5v = _eval(h_p)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_p.copy()

    # Phase 6: Top-lag descent
    if time.time() < deadline - 120:
        for nl in [2, 4, 6, 3]:
            if time.time() > deadline - 100:
                break
            h_ml, c5_ml = _top_lag_descent(best_h.copy(), nl, 400, deadline)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()

    # Phase 7: Subgradient with adaptive steps
    if time.time() < deadline - 80:
        h = best_h.copy()
        n = n0
        dx = 2.0 / n
        c5_cur = _eval(h)
        for outer in range(5):
            if time.time() > deadline - 40:
                break
            tol = 1e-5 * (0.5 ** outer)
            for step in [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]:
                improved = False
                for _ in range(100):
                    if time.time() > deadline - 30:
                        break
                    c = _cvals(h)
                    m = c.max()
                    active = np.where(c >= m - tol)[0]
                    if len(active) == 0:
                        break
                    g = np.zeros(n)
                    for k in active:
                        sh = (n - 1) - k
                        if sh >= 0:
                            np_ = n - sh
                            g[:np_] += (1.0 - h[sh:])
                            if sh > 0:
                                g[sh:] -= h[:np_]
                        else:
                            sp = -sh
                            g[sp:] += (1.0 - h[:-sp])
                            g[:-sp] -= h[sp:]
                    g *= dx / len(active)
                    h_new = _repair(h - step * g, n)
                    c5n = _eval(h_new)
                    if c5n < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5n
                        improved = True
                    else:
                        break
                if not improved:
                    break
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()

    # Phase 8: Final high-alpha L-BFGS
    if time.time() < deadline - 40:
        h = _lbfgs_obj(best_h, _smooth_obj, _smooth_grad, (800000,), 5000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 9: Local search with medium steps
    if time.time() < deadline - 20:
        h_ls, c5_ls = _local_search(best_h, 40000, 1e-4, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 10: Ultra-fine polish
    if time.time() < deadline - 3:
        h_ls, c5_ls = _local_search(best_h, 100000, 1e-5, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
