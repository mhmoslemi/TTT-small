import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _lse_obj(h, beta, pen):
    n = len(h)
    c = _corr(h)
    mc = c.max()
    val = mc + np.log(np.sum(np.exp(beta * (c - mc)))) / beta
    return val + pen * (h.sum() - n * 0.5) ** 2

def _lse_obj_grad(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = c.max()
    e = np.exp(beta * (c - mc))
    s = e.sum()
    w = e / s
    val = mc + np.log(s) / beta
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        wdk = w[k] * dx
        sh = k - (n - 1)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wdk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wdk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wdk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)

def _topk_obj(h, k, pen):

    c = _corr(h)
    topk = np.sum(np.sort(c)[::-1][:k]) / k
    return topk + pen * (h.sum() - n * 0.5) ** 2

def _topk_obj_grad(h, k, pen):

    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c_full)[::-1][:k]

    val = c_top.mean()
    g = np.zeros(n)

        wdk = dx / k
        sh = kidx - (n - 1)
        if sh >= 0:
            i1 = np.arange(n - sh)

def _lbfgs_lse(h0, beta, pen, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def f(x):
        v, _ = _lse_obj_grad(x, beta, pen)
        return v
    def jf(x):
        _, g = _lse_obj_grad(x, beta, pen)
        return g
    try:
        res = minimize(f, h0, jac=jf, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 80})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def _lbfgs_topk(h0, k, pen, maxiter):

    bnds = [(0.0, 1.0)] * n

        v, _ = _topk_obj_grad(x, k, pen)
        return v

        _, g = _topk_obj_grad(x, k, pen)
        return g

        return _repair(res.x, n)

        g *= 2.0 / n
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        v = beta_mom * v + (1.0 - beta_mom) * (g / gn)
        v /= np.linalg.norm(v)

def _fast_local_search(h0, n_iter, rng, el, bc):
