import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_corr(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        elif sh < 0:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:-sp] -= wk * h[sp:]
        else:
            g += wk * (1.0 - 2.0 * h)
    return g

def _lse_obj(x, beta):
    c = _cvals(x)
    m = c.max()
    s = np.clip(beta * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / beta

def _lse_grad(x, beta):
    c = _cvals(x)
    m = c.max()
    s = np.clip(beta * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    return _grad_corr(x, w)

def _flat_obj(x, beta, lam):
    """max + lam*std: flatten correlation profile."""
    c = _cvals(x)
    m = c.max()
    s = np.clip(beta * (c - m), -400, 0)
    lse = m + np.log(np.sum(np.exp(s))) / beta
    return lse + lam * np.std(c)

def _flat_grad(x, beta, lam):
    c = _cvals(x)
    m = c.max()
    s = np.clip(beta * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = _grad_corr(x, w)
    # Gradient of std(c) w.r.t. h
    cc = c - c.mean()
    g_std = np.zeros(len(c))
    for k in range(len(c)):
        if abs(cc[k]) > 1e-14:
            g_std[k] = cc[k] / (len(c) * np.sqrt(np.mean(c**2) + c.mean()**2) * np.sqrt(len(c)))
    # Simpler: grad of mean((c-mean)^2) / (2*std)

        g += lam * g_std2
    return g

        return h

        else:

                           options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h

def _adam(h, beta, steps, lr, deadline):
    n = len(h)
    mv = np.zeros(n); vv = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_c = _eval(h); best_h = h.copy()
    for t in range(1, steps + 1):
        if time.time() > deadline - 25:
            break
        g = _lse_grad(h, beta)
        mv = b1*mv + (1-b1)*g
        vv = b2*vv + (1-b2)*g*g
        step = lr * (mv/(1-b1**t)) / (np.sqrt(vv/(1-b2**t)) + eps)
        h = _repair(h - step, n)
        c = _eval(h)
        if c < best_c:
            best_c = c; best_h = h.copy()
    return best_h, best_c

def _top_lag(h, nl, rounds, deadline):
    n = len(h)
    c5 = _eval(h)
    for _ in range(rounds):
        if time.time() > deadline - 25:
            break
        c = _cvals(h)
        top = np.argsort(c)[-nl:]
        w = np.zeros(len(c)); w[top] = 1.0/nl
        g = _grad_corr(h, w)
        ok = False
        for s in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5, 1e-5]:
            hn = _repair(h - s*g, n)
            cn = _eval(hn)
            if cn < c5 - 1e-15:
                h, c5, ok = hn, cn, True
                break
        if not ok:
            break
    return h, c5

def _local2(h, iters, rng, deadline):
    n = len(h); c5 = _eval(h); bh = h.copy(); bc = c5
    scales = [5e-5, 3e-5, 1e-5, 3e-6, 1e-6]
    si = 0
    for _ in range(iters):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = scales[si] * rng.randn()
        if 0 <= h[i]+d and h[i]+d <= 1 and 0 <= h[j]-d and h[j]-d <= 1:
            h[i] += d; h[j] -= d
            cn = _eval(h)
            if cn < c5 - 1e-18:
                c5 = cn; si = 0
                if cn < bc: bc = cn; bh = h.copy()
            else:
                h[i] -= d; h[j] += d
                si = min(si + 1, len(scales)-1)
    return bh, bc

def _local4(h, iters, rng, deadline):
    n = len(h); c5 = _eval(h); bh = h.copy(); bc = c5
    for _ in range(iters):
        if time.time() > deadline - 3:
            break
        idx = rng.choice(n, size=4, replace=False)
        d = 1e-4 * rng.randn(4)
        d -= d.mean()
        if np.all(h[idx] + d >= 0) and np.all(h[idx] + d <= 1):
            h[idx] += d
            cn = _eval(h)
            if cn < c5 - 1e-18:
                c5 = cn
                if cn < bc: bc = cn; bh = h.copy()
            else:
                h[idx] -= d
    return bh, bc

def _perturb_reopt(h, rng, beta, maxit, deadline):
    n = len(h)
    k = rng.randint(3, 15)
    idx = rng.choice(n, size=k, replace=False)
    h2 = h.copy()
    h2[idx] += rng.randn(k) * rng.choice([0.03, 0.05, 0.08, 0.12])
    h2 = _repair(h2, n)
    h2 = _lbfgs(h2, beta, maxit, deadline)
    return h2

def _spectral_nudge(h, rng, deadline):
    """Modify h in frequency domain to reduce peak correlations."""
    n = len(h)
    hs = h - h.mean()  # zero-mean part
    H = np.fft.rfft(hs)
    # Attenuate modes that contribute most to correlations
    mag = np.abs(H)
    if mag.max() > 0:
        mag_norm = mag / mag.max()
        H *= (1.0 - 0.1 * mag_norm**2)  # dampen strong modes
    hs_new = np.fft.irfft(H, n=n)
    h2 = h + 0.3 * hs_new * (1 - (hs/hs_new * 0 + 1) * 0)  # careful
    h2 = h + 0.3 * (hs_new - hs)
    return _repair(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS with increasing sharpness
    for beta in [200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > deadline - 450:
            break
        h = _lbfgs(best_h, beta, 3000, deadline)
        c = _eval(h)
        if c < best_c5:
            best_c5 = c; best_h = h.copy()

    # Phase 2: Flatness-regularized optimization (novel)
    if time.time() < deadline - 350:
        for lam in [0.05, 0.1, 0.2]:
            if time.time() > deadline - 350:
                break
            h = _lbfgs(best_h, 50000, 2000, deadline, lam=lam)
            c = _eval(h)
            if c < best_c5:
                best_c5 = c; best_h = h.copy()

    # Phase 3: Adam optimizer (different trajectory)
    if time.time() < deadline - 250:
        h2, c2 = _adam(best_h.copy(), 30000, 8000, 0.005, deadline)
        if c2 < best_c5:
            best_c5 = c2; best_h = h2.copy()

    # Phase 4: Top-lag descent
    if time.time() < deadline - 200:
        for nl in [2, 3, 5, 7, 4]:
            if time.time() > deadline - 200:
                break
            h3, c3 = _top_lag(best_h.copy(), nl, 600, deadline)
            if c3 < best_c5:
                best_c5 = c3; best_h = h3.copy()

    # Phase 5: Perturbation + re-optimization (escape local optima)
    for attempt in range(10):
        if time.time() > deadline - 130:
            break
        h2 = _perturb_reopt(best_h, rng, 200000, 800, deadline)
        c2 = _eval(h2)
        if c2 < best_c5:
            best_c5 = c2; best_h = h2.copy()

    # Phase 6: High-beta L-BFGS
    if time.time() < deadline - 100:
        h = _lbfgs(best_h, 5000000, 5000, deadline)
        c = _eval(h)
        if c < best_c5:
            best_c5 = c; best_h = h.copy()

    # Phase 7: Spectral nudge + reopt
    if time.time() < deadline - 60:
        h2 = _spectral_nudge(best_h, rng, deadline)
        h2 = _lbfgs(h2, 500000, 2000, deadline)
        c2 = _eval(h2)
        if c2 < best_c5:
            best_c5 = c2; best_h = h2.copy()

    # Phase 8: Top-lag refinement
    if time.time() < deadline - 40:
        for nl in [2, 3]:
            if time.time() > deadline - 40:
                break
            h3, c3 = _top_lag(best_h.copy(), nl, 400, deadline)
            if c3 < best_c5:
                best_c5 = c3; best_h = h3.copy()

    # Phase 9: 4-point local search
    if time.time() < deadline - 15:
        h4, c4 = _local4(best_h, 30000, rng, deadline)
        if c4 < best_c5:
            best_c5 = c4; best_h = h4.copy()

    # Phase 10: 2-point ultrafine polish
    if time.time() > 5:
        h2f, c2f = _local2(best_h, 120000, rng, deadline)
        if c2f < best_c5:
            best_c5 = c2f; best_h = h2f.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
