Let me think about the softmax gradient more carefully. The objective is:
f(h) = max_k c_k * dx where c_k = correlate_k(h, 1-h)

A smooth approximation: f_α(h) = (dx/α) * log(sum_k exp(α * c_k * dx))

The gradient: ∂f_α/∂h_i = dx * sum_k w_k * (∂c_k/∂h_i) where w_k = exp(α*c_k*dx)/sum_j exp(α*c_j*dx)

With α → 0: uniform weights (average of all lags)
With α → ∞: only max lag (recovery to subgradient)

For the gradient computation, we need ∂c_k/∂h_i for all k weighted by w_k.

Actually, this is expensive if we compute all 2n-1 separate gradients. But we can be smarter:

c_k for lag s (where s = k-(n-1)):
c_s = sum_j h[j+s]*(1-h[j])

∂c_s/∂h_i = sum over all (j, j+s) pairs involving i

For a fixed i:
- i appears as the "first index" h[j+s] when j+s = i, i.e., j = i-s
- i appears as the "second index" h[j] when j = i

So ∂c_s/∂h_i = (1-h[i-s]) * [s <= i <= n-1] - h[i+s] * [0 <= i <= n-1-s]

The total gradient with softmax weights:
∇f_α[i] = dx * sum_s w_s * [(1-h[i-s])*[s<=i<n] - h[i+s]*[0<=i<n-s]]

This is a sum over s of weighted terms. Each term is a shifted version of h or 1-h.

def _repair(h, n):

                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:

        else:
            hi = mid

def _grad_weighted(h, k_w, n):
    g = np.zeros(n)
    for k, w in k_w:
        if w < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += w * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= w * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += w * (1.0 - h[sp:])
            g[sp:] -= w * h[:n - sp]
    return g

def _momentum_grad(h, n, max_it, alpha0, alpha1, beta, check_fn):
    dx = 2.0 / n
    c5 = _eval(h)
    v = np.zeros(n)
    lr = 0.1
    for it in range(max_it):
        if check_fn():
            break
        alpha = alpha0 + (alpha1 - alpha0) * it / max(max_it - 1, 1)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        sc = alpha * c - alpha * c.max()
        w = np.exp(sc); w /= w.sum()
        k_w = [(k, w[k]) for k in range(2 * n - 1) if w[k] > 1e-10]
        g = _grad_weighted(h, k_w, n) * dx / len(k_w)
        v = beta * v + g
        improved = False

            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:

                lr = min(lr * 1.3, 0.5)
                break

        if not improved:

            lr = max(lr * 0.3, 1e-6)
    return h, c5

    c5 = _eval(h)
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]

def _slsqp_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    a = alpha

    # Phase 4: Block coordinate descent
    if _el() < budget_s - 120:
        nt = min(int((budget_s - 120 - _el()) / 0.004), 80000)
        best_h, best_c5 = _block_cd(best_h, n0, nt, rng, best_c5, _bc)

    # Phase 5: Targeted pair search on top lag
    if _el() < budget_s - 80:
        nt = min(int((budget_s - 80 - _el()) / 0.005), 100000)
        best_h, best_c5 = _pair_search(best_h, n0, nt, rng, best_c5, _bc)

    # Phase 6: Alternative structured starts
    if _el() < budget_s - 100:
        starts = []
        xs = np.linspace(0, 2, n0, endpoint=False)
        for freq in [1, 2, 3]:
            trial = 0.5 + 0.3 * np.cos(2 * np.pi * freq * xs) * np.exp(-0.3 * (xs - 1.0)**2)
            trial = _repair(trial, n0)
            starts.append(trial)
        for st in starts:
            if _bc():
                break
            h2, c52 = _topk_grad(st, n0, 300, 7, _bc)
            if c52 < best_c5:
                best_c5 = c52; best_h = h2

    # Phase 7: Final top-k polish
    for kt in [5, 7, 4, 6, 5]:
        if _bc():
            break
        h2, c52 = _topk_grad(best_h.copy(), n0, 350, kt, _bc)
        if c52 < best_c5:
            best_c5 = c52; best_h = h2

    # Phase 8: Ultra-fine pair search
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.004), 200000)
        best_h, best_c5 = _pair_search(best_h, n0, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
