def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = max(diff, -h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

min_h max_k f_k(h) = min_h max_{λ ∈ Δ_K} Σ_k λ_k f_k(h)

The inner max over λ is a linear program (maximize a linear function over the simplex). At the optimum, λ* is concentrated on the active shifts.

2. Given λ*, optimize h to minimize Σ_k λ_k f_k(h) (a quadratic program)
3. Repeat

import numpy as np
from scipy.optimize import minimize
import time
import functools

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

def _proj_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _full_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        wk *= dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _nesterov_full(h, alpha, step0, t0, deadline):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.75 + 0.10 * min(alpha / 1000.0, 1.0)
    step = step0
    best_f = f
    stall = 0
    while time.time() < deadline:
        g, m = _full_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h = h_new
            stall = 0
            if f_new < best_f:
                best_f = f_new
            f = f_new
            step = min(step * 1.04, step0 * 30.0)
        else:
            stall += 1
            step *= 0.5
            if step < 1e-11:
                break
            h = _proj_sum(h_prev - step * v, target)
            f = _eval(h)
        if stall > 800:
            v *= 0.2
            step = step0 * 0.3
            stall = 0
    return _repair(h, n), best_f

def _slsqp_obj(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    g, _ = _full_grad(x, alpha)
    return g

def _slsqp_constraint(x):
    return x.sum() - len(x) * 0.5

def _slsqp_constraint_jac(x):
    return np.ones(len(x))

def _slsqp_run(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    obj_f = functools.partial(_slsqp_obj, alpha=alpha, n=n)
    grad_f = functools.partial(_slsqp_grad, alpha=alpha, n=n)
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_constraint_jac}]
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_2pt(h, n_tries, rng, t0, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = rng.randn() * 5e-6
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_3pt(h, n_tries, rng, t0, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 10)) % n
        k = (i + rng.randint(11, 25)) % n
        d1, d2 = rng.randn() * 2e-5, rng.randn() * 2e-5
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _symmetrize(h, n):
    hs = 0.5 * (h + h[::-1])
    return _repair(hs, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(initial_h_values.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Nesterov full-softmax, alpha ramp
    alphas = [30, 100, 300, 1000, 3000, 10000]
    for i_al, alpha in enumerate(alphas):
        frac = 0.08 + 0.04 * i_al
        deadline = t0 + budget_s * frac
        step0 = min(2.0 / max(alpha, 1.0), 0.1)
        h2, f2 = _nesterov_full(h, alpha, step0, t0, deadline)
        if f2 < best_c5:
            best_c5, best_h = f2, h2
            h = h2
        else:
            h = best_h.copy()

    # Phase 2: Symmetrize and re-optimize (exploit symmetry)
    h_sym = _symmetrize(best_h, n)
    c5v = _eval(h_sym)
    if c5v < best_c5 - 1e-9:
        best_c5, best_h = c5v, h_sym
        h = h_sym
        h2, f2 = _nesterov_full(h, 3000, 0.001, t0, t0 + budget_s * 0.6)
        if f2 < best_c5:
            best_c5, best_h = f2, h2

    # Phase 3: SLSQP with increasing alpha
    for alpha_sl in [3000, 30000, 100000, 500000]:
        deadline = t0 + budget_s * 0.82
        if time.time() > deadline:
            break
        h2 = _slsqp_run(best_h.copy(), alpha_sl, 2000, 1e-16)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2

    # Phase 4: Nesterov with very high alpha (near-exact max)
    deadline = t0 + budget_s * 0.88
    if time.time() < deadline:
        h2, f2 = _nesterov_full(best_h.copy(), 50000, 5e-5, t0, deadline)
        if f2 < best_c5:
            best_c5, best_h = f2, h2

    # Phase 5: 3-point local search
    deadline = t0 + budget_s * 0.95
    if time.time() < deadline:
        n_tries = int((deadline - time.time()) / 0.005)
        best_h, best_c5 = _local_3pt(best_h, n_tries, rng, t0, deadline)

    # Phase 6: 2-point ultra-fine polish
    deadline = t0 + budget_s * 0.99
    if time.time() < deadline:
        n_tries = int((deadline - time.time()) / 0.003)
        best_h, best_c5 = _local_2pt(best_h, n_tries, rng, t0, deadline)

    # Final repair
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
