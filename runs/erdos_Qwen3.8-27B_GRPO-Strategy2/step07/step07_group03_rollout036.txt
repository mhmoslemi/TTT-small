        a_vals = c[top_idx].copy()
        # Compute effective gradient (weighted by top lags)
        eff_g = np.sum(grads, axis=0)
        # Find best pair by gradient difference
        eff_g_mean = eff_g - eff_g.mean()
        sweep_improved = False
        n_pairs = n * 2
        for _ in range(n_pairs):
            if time.time() > t_end:
                break
            r = rng.random()
            if r < 0.3:
                # Gradient-guided pair
                i = int(np.argmax(eff_g_mean))
                j = int(np.argmin(eff_g_mean))
                if h[i] < 1e-8 or h[j] > 1.0 - 1e-8:
                    i = rng.randint(n)
                    j = (i + 1 + rng.randint(0, 10)) % n
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, max(4, n // 6))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(30):
                if f1 < f2:
                    d_hi, f2 = x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, f1 = x1, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                sweep_improved = True
        if sweep_improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 40:
            for _ in range(8):
                idx = rng.randint(n)
                h[idx] += 1e-4 * rng.randn()
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-8:
            h = _proj_to_sum(h, n * 0.5)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov (10% time)
    for a in [5000, 30000, 150000, 800000, 3000000]:
        t_rem = t0 + budget_s * 0.10 - time.time()
        if t_rem < 2:
            break
        h, c_new = _nesterov(h, a, t_rem, step0=0.003, momentum=0.83)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()

    # Phase 2: Long coordinate descent (25%)
    t_p2 = t0 + budget_s * 0.35
    for top_T in [5, 8, 3]:
        if time.time() > t_p2:
            break
        h = best_h.copy()
        avail = t_p2 - time.time()
        if avail < 10:
            break
        h, c_new = _coord_descent(h, n, avail * 0.95, top_T, rng, t0 + budget_s)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()

    # Phase 3: Gradient-informed coordinate descent (20%)
    t_p3 = t0 + budget_s * 0.55
    for top_T in [6, 4, 8]:
        if time.time() > t_p3:
            break
        h = best_h.copy()
        avail = t_p3 - time.time()
        if avail < 10:
            break
        h, c_new = _grad_coord_descent(h, n, avail * 0.95, top_T, rng, t0 + budget_s)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()

    # Phase 4: Nesterov + coord alternating (20%)
    t_p4 = t0 + budget_s * 0.75
    for a in [2000000, 8000000]:
        if time.time() > t_p4:
            break
        h = best_h.copy()
        h, c_new = _nesterov(h, a, 12, step0=0.0015, momentum=0.86)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()
        if time.time() > t_p4:
            break
        h = best_h.copy()
        avail = t_p4 - time.time()
        if avail < 5:
            break
        h, c_new = _coord_descent(h, n, avail * 0.9, 6, rng, t0 + budget_s)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()

    # Phase 5: Ultra-fine coordinate + local search (25%)
    t_p5 = t0 + budget_s - 5
    if time.time() < t_p5:
        h = best_h.copy()
        avail = t_p5 - time.time()
        half = avail * 0.5
        h, c_new = _coord_descent(h, n, half, 4, rng, t0 + budget_s)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()
        if time.time() < t_p5:
            h = best_h.copy()
            n_tries = min(int((t_p5 - time.time()) * 300), 60000)
            if n_tries > 200:
                best_h, best_c5 = _fine_search(h, n_tries, rng, t_p5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
