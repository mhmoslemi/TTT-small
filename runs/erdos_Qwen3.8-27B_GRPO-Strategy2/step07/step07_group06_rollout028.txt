```python
def _lse_gradient(h, mu, n):
    """Compute gradient of smooth max of correlations w.r.t. h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / mu)
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        m = k - (n - 1)
        a = max(0, -m)
        b = min(n - 1, n - 1 - m)
        if a > b:
            continue
        # Term 1: (1-h[p+m]) for p in [a, b]
        ps = np.arange(a, b + 1)
        indices = ps + m
        valid = indices < n
        grad[ps[valid]] += w[k] * (1.0 - h[indices[valid]])
        # Term 2: -h[p-m] for p-m in [a, b], i.e., p in [a+m, b+m]
        p_lo = max(a + m, 0)
        p_hi = min(b + m, n - 1)
        if p_lo <= p_hi:
            ps2 = np.arange(p_lo, p_hi + 1)
            idx2 = ps2 - m
            grad[ps2] -= w[k] * h[idx2]
    
    grad *= dx
    return grad
```

This is O(n * |K_significant|) per call. With n=560 and maybe 50-100 significant k values, that's about 30K-56K operations per gradient evaluation. Very fast!

```python
    # m ranges from -(n-1) to n-1
```

```python
    for m_idx in range(len(m_all)):
        if not valid_all[m_idx]:
            continue
        wk_val = wk[m_idx]
        if wk_val < 1e-10:
            continue
        m_val = m_all[m_idx]
        a_val = a_all[m_idx]
        b_val = b_all[m_idx]
        # Term 1: p in [a,b], contributes (1-h[p+m])
        p_idx = np.arange(a_val, b_val + 1)
        grad[p_idx] += wk_val * (1.0 - h[p_idx + m_val])
        # Term 2: p-m in [a,b], p in [a+m, b+m]
        p_lo = max(a_val + m_val, 0)
        p_hi = min(b_val + m_val, n - 1)
        if p_lo <= p_hi:
            p_idx2 = np.arange(p_lo, p_hi + 1)
            grad[p_idx2] -= wk_val * h[p_idx2 - m_val]
    return grad * dx

def _smooth_max_descent(h, deadline, start_mu=0.05, end_mu=0.002, n_stages=8, iters_per_mu=30):

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    for m in range(-(n-1), n):
        k = n - 1 + m
        a = max(0, -m)
        b = min(n - 1, n - 1 - m)
        if a > b:
            continue

            pm = p + m
            if 0 <= pm < n:
                g[k] += (1.0 - h[pm])
        pm2 = p - m
        if a <= pm2 <= b:
            g[k] -= h[pm2]
    return g * (2.0 / n)
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

                if c5n < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5n
                    ok = True
                    break
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
            if not ok:

    return best_h, best_c

def _pair_cd(h, n_iter, rng, deadline, tgt_frac=0.5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stable > 6000:
            break
        if it % 2000 == 0:
            c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
            c5_cur = c.max()
        r = rng.random()
        if r < tgt_frac:
            peak_k = int(np.argmax(c))
            lag = peak_k - (n - 1)
            i = rng.randint(n)
            j_cand = (i + lag) % n
            if j_cand == i:
                j_cand = (i + 1) % n
            j = j_cand
        elif r < tgt_frac + 0.15:
            top_k = np.argsort(c)[-min(5, 2*n-1):]
            lag = top_k[rng.randint(len(top_k))] - (n - 1)
            i = rng.randint(n)
            j_cand = (i + lag) % n
            if j_cand == i:
                j_cand = (i + 1) % n
            j = j_cand
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        for _ in range(40):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = int(np.argmax(vals))
            sg = g[kmax]
            if kmax == k1:
                sg += 2.0 * d_mid
            if kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _polish(h, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    for ss in [1e-4, 5e-5, 1e-5]:
        for _ in range(100000):
            if time.time() > deadline:
                return best_h, best_c
            i = rng.randint(n)
            j = (i + rng.randint(1, 5)) % n
            d = ss * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                v = _eval(h)
                if v < c5_cur - 1e-16:
                    c5_cur = v
                    if v < best_c:
                        best_c = v
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c
```

```python
import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, n - 1 + p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _smooth_grad(h, mu, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    ec = np.exp((c - cmax) / mu)
    w = ec / ec.sum()
    grad = np.zeros(n)
    m_arr = np.arange(-(n - 1), n)
    a_arr = np.maximum(0, -m_arr)
    b_arr = np.minimum(n - 1, n - 1 - m_arr)
    wk_arr = w[n - 1 + m_arr]
    mask = (a_arr <= b_arr) & (wk_arr > 1e-10)
    for mi in np.where(mask)[0]:
        m_v = m_arr[mi]; a_v = a_arr[mi]; b_v = b_arr[mi]; wk = wk_arr[mi]
        p1 = np.arange(a_v, b_v + 1)
        grad[p1] += wk * (1.0 - h[p1 + m_v])
        lo = max(a_v + m_v, 0); hi = min(b_v + m_v, n - 1)
        if lo <= hi:
            p2 = np.arange(lo, hi + 1)
            grad[p2] -= wk * h[p2 - m_v]
    return grad * dx

def _smooth_descent(h, deadline, mus=None, iters=25):
    if mus is None:
        mus = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for mu in mus:
        if time.time() > deadline:
            break
        c5c = _eval(h)
        for _ in range(iters):
            if time.time() > deadline:
                break
            g = _smooth_grad(h, mu, n)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            ok = False
            for mult in [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1]:
                st = mult * mu / gn
                h_new = _proj_to_sum(h - st * g, n * 0.5)
                v = _eval(h_new)
                if v < c5c - 1e-13:
                    h, c5c, ok = h_new, v, True
                    break
            if c5c < best_c:
                best_c, best_h = c5c, h.copy()
            if not ok:
                break
    return best_h, best_c

def _pair_cd(h, n_iter, rng, deadline, tf=0.5):

    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx

        if it % 2000 == 0:

            c5c = c.max()
        r = rng.random()
        if r < tf:

            lag = pk - (n - 1)
            i = rng.randint(n)

            if j == i: j = (i + 1) % n
        elif r < tf + 0.15:
            tk = np.argsort(c)[-5:]
            lag = tk[rng.randint(5)] - (n - 1)
            i = rng.randint(n)
            j = (i + lag) % n

        else:

        if i == j: continue
        d0 = j - i
        k1, k2 = n - 1 + d0, n - 1 - d0
        dlo, dhi = max(-h[i], h[j] - 1), min(1 - h[i], h[j])
        if dlo >= dhi - 1e-15: continue
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        for _ in range(40):

            vals = c + dm * g

            km = int(np.argmax(vals))
            sg = g[km]
            if km == k1: sg += 2.0 * dm
            if km == k2: sg += 2.0 * dm
            if sg > 1e-14: dhi = dm
            elif sg < -1e-14: dlo = dm
            else: break
        do = 0.5 * (dlo + dhi)
        if abs(do) < 1e-13:

            continue
        h[i] += do; h[j] -= do
        c += do * g
        if 0 <= k1 < 2*n - 1: c[k1] += do * do
        if 0 <= k2 < 2*n - 1: c[k2] += do * do
        if c.max() < c5c - 1e-14:

            if c5c < best_c: best_c, best_h = c5c, h.copy()
        else:

            c -= do * g
            if 0 <= k1 < 2*n - 1: c[k1] -= do * do
            if 0 <= k2 < 2*n - 1: c[k2] -= do * do
```

```python
def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)
```

This finds a scalar shift such that sum(clip(h - shift, 0, 1)) = target. The function f(shift) = sum(clip(h - shift, 0, 1)) is monotonically decreasing in shift. We want f(shift) = target.

The bisection is: if f(mid) > target, we need to increase shift (lo = mid). If f(mid) < target, decrease shift (hi = mid). After 50 iterations, the error is (4/2)^50 ≈ 4*2^(-50) ≈ 3.5e-15. Very accurate.

This is the Euclidean projection onto the constraint set {0 ≤ x_i ≤ 1, sum(x) = target}. ✓

One concern: the bisection uses `lo` and `hi` as the shift values, but the function is decreasing. If f(mid) > target, we need mid to be larger (shift more), so lo = mid. If f(mid) < target, we need mid to be smaller, so hi = mid. ✓
