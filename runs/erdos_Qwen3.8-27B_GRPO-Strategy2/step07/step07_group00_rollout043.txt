    w /= w.sum()

    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons = [{'type': 'eq', 'fun': _sumfun, 'jac': _sumjac}]
    try:
        r = minimize(lambda x: _smapack(x, alpha, topk), h,
                     jac=lambda x: _gradpack(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _sumfun(x):
    return x.sum() - len(x) * 0.5

def _sumjac(x):
    return np.ones(len(x))

def _smapack(x, alpha, topk):
    return _smax_obj(x, alpha, topk)

def _gradpack(x, alpha, topk):
    return _grad_topk(x, alpha, topk)

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += dx * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx * h[:-sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:-sp])
        g[:n - sp] -= dx * h[sp:]
    return g

def _greedy_cd(h, deadline, rng, n_iter_max):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    c = _correl(h)
    stall = 0
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        if it % 200 == 0:
            c = _correl(h)
        topk = np.argpartition(c, -6)[-6:]
        w = c[topk] - c[topk].min() + 1e-6
        w = w / w.sum()
        g = np.zeros(n)
        for ii, kk in enumerate(topk):
            if w[ii] > 1e-8:
                g += w[ii] * _grad_lag(h, kk)
        improved = False
        n_trial = 40
        best_imp = 0
        best_move = None
        for _ in range(n_trial):
            i = rng.randint(n)
            j = (i + rng.randint(1, 15)) % n
            if g[i] > g[j]:
                d = 0.015
            else:
                d = -0.015
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                imp = c5c - c5n
                if imp > best_imp:
                    best_imp = imp
                    best_move = (i, j, d)
                h[i], h[j] = ni - d, nj + d
        if best_move is not None:
            i, j, d = best_move
            for scale in [1.0, 0.5, 0.25, 2.0, 0.75]:
                d2 = d * scale
                ni, nj = h[i] + d2, h[j] - d2
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-15:
                        c5c = c5n
                        improved = True
                        if c5n < bc:
                            bc = c5n
                            bh = h.copy()
                        stall = 0
                        break
                    else:
                        h[i], h[j] = ni - d2, nj + d2
                else:
                    h[i], h[j] = ni - d2, nj + d2
        if not improved:
            stall += 1
            if stall > 200:
                # Perturb to escape
                for _ in range(5):
                    a, b = rng.randint(n, 2)
                    d = 0.05 * rng.randn()
                    na, nb = h[a] + d, h[b] - d
                    if 0 <= na <= 1 and 0 <= nb <= 1:
                        h[a], h[b] = na, nb
                        h = _repair(h, n)
                        c5c = _eval(h)
                        if c5c < bc:
                            bc = c5c
                            bh = h.copy()
                stall = 0
    return bh, bc

def _fine_polish(h, deadline, rng):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 5e-5
    while time.time() < deadline:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                step = min(step * 1.01, 2e-4)
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.98
                if step < 1e-7:
                    step = 5e-5
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP refinement
    dl1 = min(deadline, t0 + 150)
    for alpha, tk, mi in [(3000, 40, 400), (15000, 20, 350), (60000, 10, 300)]:
        if time.time() > dl1:
            break
        h = _slsqp(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Nesterov with varying alpha
    dl2 = min(deadline, t0 + 350)
    for alpha, tk, st, ns in [(15000, 20, 0.005, 3000), (60000, 12, 0.002, 2500),
                              (300000, 8, 0.001, 2500), (1500000, 5, 0.0005, 2000)]:
        if time.time() > dl2:
            break
        h = _nesterov(best_h.copy(), alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Greedy coordinate descent with gradient guidance
    dl3 = min(deadline, t0 + 600)
    best_h, best_c5 = _greedy_cd(best_h.copy(), dl3, rng, 500000)

    # Phase 4: Another Nesterov round with very high alpha
    dl4 = min(deadline, t0 + 750)
    for alpha, tk, st, ns in [(5000000, 4, 0.0003, 2000), (20000000, 3, 0.00015, 1500)]:
        if time.time() > dl4:
            break
        h = _nesterov(best_h.copy(), alpha, tk, ns, st, dl4)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Block anneal (short, for escape)
    dl5 = min(deadline, t0 + 850)
    blocks = [2, 3, 4, 5, 6, 8]
    h_a, c5_a = _block_anneal(best_h.copy(), 20000, rng, dl5, 0.002, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 6: Final greedy CD
    dl6 = min(deadline, t0 + budget_s - 60)
    best_h, best_c5 = _greedy_cd(best_h.copy(), dl6, rng, 500000)

    # Phase 7: Fine polish with remainder
    best_h, best_c5 = _fine_polish(best_h.copy(), t0 + budget_s - 3, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
