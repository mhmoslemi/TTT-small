import numpy as np
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sca_solve(h0, n, top_T):
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    c = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(c)[-top_T:]
    for k in top_idx:
        s = k - (n - 1)
        coef = np.zeros(n)
        if s >= 0:
            m = n - s
            coef[:m] += (1.0 - h0[s:n])
            coef[s:] -= h0[:m]
            const = np.dot(h0[:m], h0[s:n])
        else:
            p = -s
            m = n - p
            coef[p:] += (1.0 - h0[:m])
            coef[:m] -= h0[p:n]
            const = np.dot(h0[p:n], h0[:m])
        constraints.append(coef @ h + const <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if h.value is not None and not np.any(np.isnan(h.value)):
                return np.array(h.value).flatten()
        except Exception:
            continue
    return h0

def _coord_descent(h, n, duration_s, top_T, rng, t_end):

            d_hi = min(1.0 - h[i], h[j]) - 1e-12
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)

def _lag_grad(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g
