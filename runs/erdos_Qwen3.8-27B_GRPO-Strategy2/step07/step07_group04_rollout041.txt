import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_for_lag(h, l):
    """Gradient of c_l (lag l) w.r.t. h, NOT including dx factor."""
    n = len(h)
    grad = np.zeros(n)
    if l == 0:
        grad[:] = 1.0 - 2.0 * h
    elif l > 0:
        grad[l:] += 1.0 - h[:n-l]
        grad[:n-l] -= h[l:]
    else:
        m = -l
        grad[m:] -= h[:n-m]
        grad[:n-m] += 1.0 - h[m:]
    return grad

def _nesterov_topk(h, k_top, steps, step0, t0, budget_end):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    for _s in range(steps):
        if time.time() - t0 > budget_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in top_k:
            l = k - (n - 1)
            g += _grad_for_lag(h, l)
        g *= dx / k_top
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_obj_grad(x, alpha, n, lags):
    """Compute softmax-smoothed objective and gradient using only specified lags."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_sub = c[lags]
    m = c_sub.max()
    s = np.clip(alpha * (c_sub - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for wk, k in zip(w, lags):
        if wk < 1e-15:
            continue
        l = k - (n - 1)
        g += wk * _grad_for_lag(x, l) 
    g *= dx
    return obj, g
