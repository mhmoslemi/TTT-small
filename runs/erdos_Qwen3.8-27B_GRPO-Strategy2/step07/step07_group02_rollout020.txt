from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Binary search for the right shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _proj(h, n):
    return _repair(h, n)

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)

                                'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

    return h, c5_cur

def _local_search(h, n_iter, rng, deadline, n_pts=3, scale=1e-4):
    """Local search with variable-size moves"""
    n = len(h)
    c5c = _eval(h)

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    # Fast projection onto {0<=h<=1, sum(h)=n/2}
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_cvals(h)))

def _sobj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 10:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except:
        return h

def _toplag(h, nl, rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for _ in range(rounds):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        top = np.argsort(c)[-nl:]
        g = np.zeros(n)
        for k in top:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / nl
        imp = False
        for st in np.logspace(-1, -6, 12):
            hn = _repair(h - st * g, n)
            cn = _eval(hn)
            if cn < c5 - 1e-18:
                h, c5, imp = hn, cn, True
                break
        if not imp:
            break
    return h, c5

def _lsearch(h, iters, rng, deadline, np3=3, scale=1e-4):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5

        s = scale * (0.5 + rng.random())

    n = len(h)
    c5 = _ev(h)
    bh = h.copy()
    bc = c5
    sc = sc0
    na = 0
    for i in range(it):
        if time.time() > dl - 2:
            break
        s = sc * (0.4 + 0.8 * rg.random())
        idx = rg.randint(n, size=npts)
        mv = rg.randn(npts) * s
        mv -= mv.mean()
        old = h[idx].copy()
        h[idx] += mv
        if h[idx].min() < -1e-15 or h[idx].max() > 1.0 + 1e-15:
            h[idx] = old
            continue
        cn = _ev(h)
        if cn < c5 - 1e-19:
            c5 = cn
            na += 1
            if c5 < bc:
                bc, bh = c5, h.copy()
        else:
            h[idx] = old
        if i % 350 == 349:
            r = na / 350.0
            if r > 0.35:
                sc = min(sc * 1.35, 0.04)
            elif r < 0.04:
                sc = max(sc * 0.6, 3e-8)
            na = 0
            h = _repair(h, n)
    return bh, bc

def _pnorm_obj(x, p):
    c = np.maximum(_cv(x), 0.0)
    return (np.sum(c ** p)) ** (1.0 / p)

def _pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    c = np.maximum(_cv(x), 0.0)
    fp = (np.sum(c ** p)) ** (1.0 / p)
    if fp < 1e-30:
        return np.zeros(n)
    w = (c / fp) ** (p - 1)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = n - 1 - k
        wk = w[k] * dx
        if sh >= 0:
            L = n - sh
            g[:L] += wk * (1.0 - x[sh:])
            if sh:
                g[sh:] -= wk * x[:L]
        else:
            p2 = -sh
            g[p2:] += wk * (1.0 - x[:-p2])
            g[:-p2] -= wk * x[p2:]
    return g

def _lb_pnorm(h, p, mi, dl):
    n = len(h)
    if time.time() > dl - 10:
        return h
    try:
        r = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p,),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mi, 'ftol': 1e-16,
                              'gtol': 1e-12, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rg = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh = h0.copy()
    bc5 = _ev(bh)
    bn = n

    # Phase 1: L-BFGS log-sum-exp alpha sweep
    h = h0.copy()
    for a in [200, 1000, 5000, 20000, 80000, 400000]:
        if time.time() > dl - 400:
            break
        h = _lb(h, a, 3000, dl)
        c = _ev(h)
        if c < bc5:
            bc5, bh = c, h.copy()

    # Phase 2: p-norm optimization (different smoothing structure)
    if time.time() < dl - 300:
        for p in [8, 20, 60]:
            if time.time() > dl - 300:
                break
            h = _lb_pnorm(bh.copy(), p, 2500, dl)
            c = _ev(h)
            if c < bc5:
                bc5, bh = c, h.copy()

    # Phase 3: Top-lag focused descent
    if time.time() < dl - 220:
        for nl in [2, 3, 5, 4]:
            if time.time() > dl - 220:
                break
            h, c = _tl(bh.copy(), nl, 400, dl)
            if c < bc5:
                bc5, bh = c, h.copy()

    # Phase 4: L-BFGS high-alpha round
    if time.time() < dl - 150:
        for a in [200000, 1000000, 5000000]:
            if time.time() > dl - 150:
                break
            h = _lb(bh.copy(), a, 4000, dl)
            c = _ev(h)
            if c < bc5:
                bc5, bh = c, h.copy()

    # Phase 5: Random restarts from best
    if time.time() < dl - 100:
        for i in range(6):
            if time.time() > dl - 100:
                break
            scale = 0.02 + 0.01 * i
            hp = _repair(bh + scale * rg.randn(n), n)
            hp = _lb(hp, 20000, 2500, dl)
            c = _ev(hp)
            if c < bc5:
                bc5, bh = c, hp.copy()

    # Phase 6: Top-lag refinement
    if time.time() < dl - 60:
        for nl in [2, 3]:
            if time.time() > dl - 60:
                break
            h, c = _tl(bh.copy(), nl, 300, dl)
            if c < bc5:
                bc5, bh = c, h.copy()

    # Phase 7: Local search with adaptive scale
    if time.time() < dl - 15:
        for sc in [8e-4, 3e-4, 1e-4, 3e-5, 8e-6]:
            if time.time() > dl - 4:
                break
            h, c = _ls(bh, 12000, rg, dl, 3, sc)
            if c < bc5:
                bc5, bh = c, h.copy()

    # Phase 8: Ultra-fine 2-point polish
    if time.time() < dl - 2:
        h, c = _ls(bh, 35000, rg, dl, 2, 3e-6)
        if c < bc5:
            bc5, bh = c, h.copy()

    bh = _repair(bh, bn)
    bc5 = _ev(bh)
    return (bh, bc5, bn)
