            h2 = bh.copy()
            h2[i:i+bl] = h[j:j+bl] if False else h[i:i+bl]  # this is getting messy
    return bh, bc

def _finish_local(h, ntry, rng, c5c, check_fn, scale):
    nc = len(h)
    bh, bc = h.copy(), c5c
    for _ in range(ntry):
        if check_fn():
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                h[i], h[j] = ni, nj  # undo
                h[i] = ni - d
                h[j] = nj + d
    return bh, bc

def _down(h, f):
    nn = len(h) // f
    hp = np.zeros(nn * f)
    hp[:len(h)] = h
    return hp.reshape(nn, f).mean(axis=1)

def _up(h, f):
    nn = len(h) * f
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, nn, endpoint=False)
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0
    def _bc():
        return _el() > budget_s - 25

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Top-k gradient descent, varying k
    for kt in [5, 3, 7, 4, 5, 6]:
        if _bc(): break
        h2, c2 = _grad_descend(best_h.copy(), 500, 1, kt,
                                [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001], _bc)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 2: Full-softmax gradient descent
    for _ in range(3):
        if _bc(): break
        h2, c2 = _grad_descend(best_h.copy(), 400, 0, 50,
                                [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4], _bc)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Perturb + refine cycles
    for ns in [0.03, 0.06, 0.01, 0.05, 0.02, 0.04]:
        if _bc(): break
        hn = _repair(best_h + ns * rng.randn(n0), n0)
        h2, c2 = _grad_descend(hn.copy(), 400, 1, 5,
                                [0.05, 0.02, 0.01, 0.005, 0.002, 0.001], _bc)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        # Also try softmax after perturb
        h3, c3 = _grad_descend(h2.copy(), 200, 0, 50,
                                [0.01, 0.005, 0.002, 0.001], _bc)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Multi-resolution
    if _el() < budget_s - 200:
        for f in [2, 4, 2]:
            if _bc(): break
            hc = _down(best_h, f)
            hc = _repair(hc, len(hc))
            hc, cc = _grad_descend(hc.copy(), 300, 1, 3,
                                    [0.08, 0.04, 0.02, 0.01, 0.005], _bc)
            hu = _up(hc, f)
            hu = _repair(hu, n0)
            h2, c2 = _grad_descend(hu.copy(), 300, 1, 5,
                                    [0.03, 0.01, 0.005, 0.002, 0.001], _bc)
            if c2 < best_c5:
                best_c5, best_h = c2, h2

    # Phase 5: Fine gradient polish
    for kt in [3, 5, 4, 3]:
        if _bc(): break
        h2, c2 = _grad_descend(best_h.copy(), 350, 1, kt,
                                [0.008, 0.004, 0.002, 0.001, 5e-4, 2e-4], _bc)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        h3, c3 = _grad_descend(best_h.copy(), 200, 0, 50,
                                [0.005, 0.002, 0.001, 5e-4], _bc)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 6: Local search
    if _el() < budget_s - 30:
        nt = min(int((budget_s - 30 - _el()) / 0.003), 250000)
        best_h, best_c5 = _finish_local(best_h, nt, rng, best_c5, _bc, 4e-6)

    # Phase 7: Ultra-fine
    if _el() < budget_s - 3:
        nt = min(int((budget_s - 3 - _el()) / 0.002), 200000)
        best_h, best_c5 = _finish_local(best_h, nt, rng, best_c5, _bc, 1e-6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
