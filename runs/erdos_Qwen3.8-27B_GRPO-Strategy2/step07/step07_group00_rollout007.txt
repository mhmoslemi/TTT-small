import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += dx * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx * h[:-sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:-sp])
        g[:n - sp] -= dx * h[sp:]
    return g

def _mseq_binary(n, rng):
    reg = [1] * 9
    seq = []
    for _ in range(511):
        fb = reg[0] ^ reg[4]
        seq.append(fb)
        reg = [fb] + reg[:-1]
    all_bits = (seq * ((n // 511) + 1))[:n]
    b = np.array(all_bits, dtype=float)
    n1 = int(b.sum())
    tgt = n // 2
    if n1 > tgt:
        ones = np.where(b == 1)[0]
        b[rng.choice(ones, n1 - tgt, replace=False)] = 0
    elif n1 < tgt:
        zeros = np.where(b == 0)[0]
        b[rng.choice(zeros, tgt - n1, replace=False)] = 1
    return b

def _nesterov_adv(h, alpha_s, alpha_e, tk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha_s, tk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        frac = it / max(steps - 1, 1)
        alpha = alpha_s * (alpha_e / alpha_s) ** frac
        g = _grad_smax(h, alpha, tk)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _smax_obj(h_new, alpha, tk)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.04, step0 * 12.0)
        else:
            step *= 0.45
            if step < 1e-12:
                break
        fe = _eval(h)
        if fe < best_f:
            best_f, best_h = fe, h.copy()
    return _repair(best_h, n)
