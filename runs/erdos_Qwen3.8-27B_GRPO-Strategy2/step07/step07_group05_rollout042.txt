        c = _corr(h)
        temp_sm = T * 0.5 + 0.01
        w = _softmax_weights(c, temp_sm)
        g = _grad_full(h, w)
        g -= g.mean()
        noise = 0.01 * T * rng.randn(n)
        noise -= noise.mean()
        move = 0.1 * (-g + noise)
        h_new = _project_sum(h + move, target)
        c5n = _eval(h_new)
        delta = c5n - c5c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T * 0.01, 1e-10)):
            h = h_new
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
    return best_h, best_c5

def _local_2pt(h, ntry, rng, c5c, check_fn):
    nc = len(h)
    best_h = h.copy()
    best_c = c5c
    arr = h.copy()
    scales = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    si = 0
    for _ in range(ntry):
        if check_fn():
            break
        for _inner in range(8):
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 4)) % nc
            d = scales[si] * rng.randn()
            ni, nj = arr[i] + d, arr[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                arr[i], arr[j] = ni, nj
                c5n = _eval(arr)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = arr.copy()
                    si = min(si + 1, len(scales) - 1)
                    break
                else:
                    arr[i], arr[j] = arr[i] - d, arr[j] + d
        else:
            si = max(si - 1, 0)
    return best_h, best_c

def _lsx_obj(x, n):
    c = _corr(x)
    m = c.max()
    alpha = 80000.0
    sc = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / alpha

def _lsx_grad(x, n):
    c = _corr(x)
    m = c.max()
    alpha = 80000.0
    sc = np.clip(alpha * (c - m), -350, 350)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    return _grad_full(x, w)

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _bc():
        return _el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adaptive temperature gradient descent
    for (th, tc, ni) in [(1.0, 0.05, 300), (0.05, 0.01, 300), (0.01, 0.003, 400)]:
        if _bc():
            break
        h2, c52 = _smlgd(best_h.copy(), th, tc, ni, rng, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 2: Simulated annealing for escaping local minima
    for (th, tc, ni) in [(0.5, 0.01, 4000), (0.1, 0.001, 6000)]:
        if _bc():
            break
        h2, c52 = _sim_anneal(best_h.copy(), th, tc, ni, rng, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: SLSQP with smooth max
    if _el() < budget_s - 200:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        for _att in range(3):
            if _bc():
                break
            try:
                res = minimize(_lsx_obj, best_h, args=(n,), jac=_lsx_grad,
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-18})
                hr = _repair(res.x, n)
                c5r = _eval(hr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = hr
            except Exception:
                pass

    # Phase 4: Second round of gradient descent (colder)
    for (th, tc, ni) in [(0.01, 0.001, 500), (0.001, 0.0003, 500)]:
        if _bc():
            break
        h2, c52 = _smlgd(best_h.copy(), th, tc, ni, rng, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 5: Perturbation + re-optimize (multiple restarts)
    for attempt in range(4):
        if _bc():
            break
        h_p = best_h.copy()
        h_p += 0.003 * rng.randn(n0)
        h_p = _repair(h_p, n0)
        h2, c52 = _smlgd(h_p, 0.5, 0.01, 400, rng, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Final gradient polish
    if _bc() is False:
        h2, c52 = _smlgd(best_h.copy(), 0.001, 0.0001, 600, rng, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: Local 2-point search
    if _el() < budget_s - 10:
        nt = min(int((budget_s - 10 - _el()) / 0.002), 500000)
        if nt > 100:
            best_h, best_c5 = _local_2pt(best_h, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
