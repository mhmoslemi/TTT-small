```python
import numpy as np
import time
from scipy.optimize import minimize


def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _eval(h):
    n = len(h)
    if n <= 0:
        return 1.0
    return float(np.max(_corr(h, n)))


def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    if len(h) != n:
        if n == 0:
            return h
        h = h[:n] if n < len(h) else np.pad(h, (0, n - len(h)))
    s = h.sum()
    if s <= 1e-14:
        h.fill(0.5)
        return h
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                room = 1.0 - h[idx]
                if room <= 1e-14:
                    continue
                add = room if room < diff else diff
                h[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
            else:
                sub = h[idx]
                if sub <= 1e-14:
                    continue
                dec = sub if sub < -diff else -diff
                h[idx] -= dec
                diff += dec
                if diff >= -1e-12:
                    break
    return h


def _proj_sum_box(u, target):
    u = np.asarray(u, dtype=float).copy()
    n = len(u)
    if n == 0:
        return u
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = np.clip(u - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(u - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(1.0 - h[idx], diff)
                if add > 1e-14:
                    h[idx] += add
                    diff -= add
                    if diff <= 1e-12:
                        break
            else:
                dec = min(h[idx], -diff)
                if dec > 1e-14:
                    h[idx] -= dec
                    diff += dec
                    if diff >= -1e-12:
                        break
    return h


def _soft_weights(c, alpha):
    m = float(np.max(c))
    z = alpha * (c - m)
    z = z - float(np.max(z))
    w = np.exp(z)
    s = float(np.sum(w))
    if s <= 1e-300:
        w = np.full_like(w, 1.0 / max(1, len(w)), dtype=float)
    else:
        w /= s
    return w


def _grad_from_weights(h, n, weights):
    g = np.zeros(n, dtype=float)
    dx = 2.0 / n
    for k in range(len(weights)):
        w = float(weights[k])
        if w <= 1e-13:
            continue
        wk = w * dx
        p = (n - 1) - k
        if p >= 0:
            L = n - p
            if L > 0:
                g[:L] += wk * (1.0 - h[p:])
                g[p:] += -wk * h[:L]
        else:
            q = -p
            L = n - q
            if L > 0:
                g[q:] += wk * (1.0 - h[:L])
                g[:L] += -wk * h[q:]
    return g


def _obj_soft(x, n, alpha):
    c = _corr(x, n)
    m = float(np.max(c))
    z = alpha * (c - m)
    z = z - float(np.max(z))
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        return 1.0
    return float(m + np.log(s) / alpha)


def _grad_soft(x, n, alpha):
    c = _corr(x, n)
    return _grad_from_weights(x, n, _soft_weights(c, alpha))


def _obj_pen(x, n, alpha, beta):
    v = _obj_soft(x, n, alpha)
    d = float(x.sum() - 0.5 * n)
    return v + 0.5 * beta * d * d


def _grad_pen(x, n, alpha, beta):
    g = _grad_soft(x, n, alpha)
    d = float(x.sum() - 0.5 * n)
    g += beta * d
    return g


def _cons_eq_fun(x):
    return float(x.sum() - 0.5 * len(x))


def _cons_eq_jac(x):
    return np.ones((1, len(x)), dtype=float)


def _bounds(n):
    b = np.zeros((n, 2), dtype=float)
    b[:, 1] = 1.0
    return b


def _slsqp_soft(h, n, alpha, maxiter, ftol):
    try:
        res = minimize(
            _obj_soft,
            h.copy(),
            method="SLSQP",
            jac=_grad_soft,
            args=(n, alpha),
            bounds=_bounds(n),
            constraints=[{"type": "eq", "fun": _cons_eq_fun, "jac": _cons_eq_jac}],
            options={"maxiter": max(1, int(maxiter)), "ftol": ftol},
        )
        hh = res.x
    except Exception:
        hh = h
    return _repair(hh, n)


def _lbfgs_once(h, n, alpha, beta, maxiter, deadline):
    if time.time() > deadline:
        return h.copy()
    try:
        res = minimize(
            _obj_pen,
            h.copy(),
            method="L-BFGS-B",
            jac=_grad_pen,
            args=(n, alpha, beta),
            bounds=_bounds(n),
            options={"maxiter": max(1, int(maxiter)), "maxfun": max(10, 3 * int(maxiter)),
                     "ftol": 1e-13, "gtol": 1e-9},
        )
        hh = res.x
    except Exception:
        hh = h
    return _repair(hh, n)


def _iter_cap(n, time_left):
    if time_left < 1.5:
        return 0
    if n <= 128:
        per = 0.035
    elif n <= 256:
        per = 0.08
    else:
        per = 0.28
    cap = int(time_left / per)
    if cap < 5:
        return 0
    return min(cap, 220)


def _nesterov_topk(h, k, steps, step0, deadline):
    n = len(h)
    h = _repair(h.copy(), n)
    f = _eval(h)
    v = np.zeros(n, dtype=float)
    step = step0
    mu = 0.85
    for _ in range(int(steps)):
        if time.time() > deadline:
            break
        c = _corr(h, n)
        top = np.argsort(c)[-int(k):]
        w = np.zeros(2 * n - 1, dtype=float)
        w[top] = 1.0 / float(k)
        g = _grad_from_weights(h, n, w)
        old = h.copy()
        v = mu * v + (1.0 - mu) * g
        hn = _proj_sum_box(h - step * v, 0.5 * n)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.04, step0 * 10.0)
        else:
            step *= 0.48
            if step < 1e-8:
                break
            h = _proj_sum_box(old - step * v, 0.5 * n)
            f = _eval(h)
    return _repair(h, n)


def _local_2pt(h, best_c, ntries, rng, deadline):
    n = len(h)
    h = h.copy()
    c = float(best_c)
    for _ in range(int(ntries)):
        if time.time() > deadline:
            break
        i = int(rng.randint(n))
        j = (i + 1 + int(rng.randint(0, 5))) % n
        d = 2.0e-4 * float(rng.randn())
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            oi, oj = h[i], h[j]
            h[i], h[j] = hi, hj
            cn = _eval(h)
            if cn < c - 1e-15:
                c = cn
            else:
                h[i], h[j] = oi, oj
    return h, c


def _resample(h, target):
    n = len(h)
    if n == target:
        return h.copy()
    if n == 0:
        return np.full(target, 0.5, dtype=float)
    xo = np.arange(n, dtype=float) * (2.0 / n) + 1.0 / n
    xn = np.arange(target, dtype=float) * (2.0 / target) + 1.0 / target
    return np.interp(xn, xo, h, left=float(h[0]), right=float(h[-1]))


def _random_h(n, rng, kind):
    kind = int(kind) % 4
    if kind == 0:
        return rng.uniform(0.0, 1.0, size=n).copy()
    if kind == 1:
        m = int(rng.randint(4, 26))
        vals = rng.uniform(0.0, 1.0, size=m)
        if rng.rand() < 0.35:
            vals = (vals > 0.5).astype(float)
        out = np.empty(n, dtype=float)
        for r in range(m):
            a = int(r * n // m)
            b = int((r + 1) * n // m)
            if b > a:
                out[a:b] = vals[r]
        return out
    if kind == 2:
        x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h = np.full(n, 0.5, dtype=float)
        for _ in range(int(rng.randint(2, 6))):
            f = int(rng.randint(1, 7))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            a = float(rng.uniform(0.05, 0.35))
            h += a * np.sin(np.pi * f * x + ph)
        return np.clip(h, 0.0, 1.0)
    b = (rng.rand(n) < 0.5).astype(float)
    k = np.ones(5, dtype=float) / 5.0
    b = np.convolve(b, k, mode="same")
    return np.clip(0.5 + 1.6 * (b - 0.5), 0.0, 1.0)


def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float) * (2.0 / n) + 1.0 / n
    nz = np.zeros(n, dtype=float)
    for _ in range(3):
        f = int(rng.randint(1, 8))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a = float(rng.uniform(0.35, 1.0))
        nz += a * np.sin(np.pi * f * x + ph)
    nm = float(np.linalg.norm(nz))
    if nm > 1e-14:
        nz /= nm
    return _repair(np.clip(h + float(amp) * nz, 0.0, 1.0), n)


def _search_one(h, n, deadline, rng):
    h = _repair(np.asarray(h, dtype=float).copy(), n)
    best_h = h.copy()
    best_c = _eval(best_h)

    if n <= 120:
        a_seq = (5.0, 20.0, 80.0, 300.0, 1500.0, 6000.0)
        it_seq = (160, 130, 100, 75, 55, 40)
    elif n <= 240:
        a_seq = (10.0, 40.0, 160.0, 650.0, 2600.0, 10000.0)
        it_seq = (120, 100, 80, 60, 45, 30)
    elif n <= 360:
        a_seq = (10.0, 40.0, 160.0, 650.0, 2600.0, 10000.0)
        it_seq = (85, 70, 60, 45, 35, 25)
    else:
        a_seq = (10.0, 40.0, 160.0, 650.0, 2600.0, 10000.0, 25000.0)
        it_seq = (60, 55, 45, 35, 28, 22, 16)

    for idx, a in enumerate(a_seq):
        if time.time() > deadline - 1.0:
            break
        it = min(int(it_seq[idx]), _iter_cap(n, deadline - time.time() - 1.5))
        if it < 6:
            break
        hh = _slsqp_soft(best_h, n, a, it, 1e-13)
        cc = _eval(hh)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 12.0:
        a = 8000.0 if n <= 256 else 16000.0
        it = min(45, _iter_cap(n, deadline - time.time() - 10.0))
        if it >= 6:
            hh = _lbfgs_once(best_h, n, a, 120.0, it, deadline - 2.0)
            cc = _eval(hh)
            if cc + 1e-15 < best_c:
                best_c, best_h = cc, hh

    if deadline - time.time() > 30.0:
        kk = 8 if n <= 256 else 5
        st = 90 if n <= 256 else 35
        hh = _nesterov_topk(best_h, kk, st, 3.0e-3, deadline - 2.0)
        cc = _eval(hh)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 6.0:
        nt = int(min(2500, max(40, (deadline - time.time()) / 0.006)))
        best_h, best_c = _local_2pt(best_h, best_c, nt, rng, deadline - 1.0)

    best_h = _repair(best_h, n)
    return best_h, _eval(best_h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + max(0.0, total) - 15.0
    rng = np.random.RandomState(int(seed))

    h0 = None
    try:
        vol = initial_h_values
        if vol is not None:
            arr = np.asarray(vol, dtype=float).copy()
            if arr.size > 0:
                h0 = _repair(arr, arr.size)
    except Exception:
        h0 = None

    if h0 is None:
        h0 = _repair(_random_h(256, rng, 0), 256)

    best_h = h0.copy()
    best_c = _eval(best_h)
    n0 = len(best_h)

    if deadline - t0 < 30.0:
        best_h = _repair(best_h, n0)
        return best_h, _eval(best_h), n0

    for cand in (1.0 - best_h, best_h[::-1], 1.0 - best_h[::-1]):
        ch = _repair(cand, n0)
        cc = _eval(ch)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, ch.copy()

    parent_alloc = 260.0 if n0 > 300 else 130.0
    parent_alloc = max(30.0, min(parent_alloc, deadline - t0 - 60.0))
    hh, cc = _search_one(best_h.copy(), n0, min(deadline, t0 + parent_alloc), rng)
    if cc + 1e-15 < best_c:
        best_c, best_h = cc, hh

    if deadline - time.time() > 80.0:
        htr = _repair(1.0 - best_h.copy(), n0)
        dd = min(deadline, time.time() + 70.0)
        hh, cc = _search_one(htr, n0, dd, rng)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 80.0:
        htr = _repair(best_h[::-1].copy(), n0)
        dd = min(deadline, time.time() + 60.0)
        hh, cc = _search_one(htr, n0, dd, rng)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 70.0:
        htr = _repair(np.roll(best_h, 1).copy(), n0)
        dd = min(deadline, time.time() + 55.0)
        hh, cc = _search_one(htr, n0, dd, rng)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, hh

    max_n = max(560, n0)
    n_list = [96, 128, 160, 192, 224, 256, 320, 416]
    n_list = sorted(set(x for x in n_list if x <= max_n))

    for pos, nf in enumerate(n_list):
        if time.time() > deadline - 45.0:
            break
        time_left = deadline - time.time()
        per_n = max(18.0, min(65.0, time_left / max(1.0, float(len(n_list) - pos)) * 0.75))

        if nf != len(best_h):
            hs = _resample(best_h, nf)
            dd = min(deadline, time.time() + per_n)
            if dd - time.time() > 9.0:
                hh, cc = _search_one(hs, nf, dd, rng)
                if cc + 1e-15 < best_c:
                    best_c, best_h = cc, hh

        if time.time() > deadline - 30.0:
            break
        hs = _random_h(nf, rng, int(rng.randint(0, 4)))
        per_r = per_n * 0.75
        dd = min(deadline, time.time() + per_r)
        if dd - time.time() > 9.0:
            hh, cc = _search_one(_repair(hs, nf), nf, dd, rng)
            if cc + 1e-15 < best_c:
                best_c, best_h = cc, hh

    if deadline - time.time() > 90.0:
        for amp in (0.03, 0.08, 0.16):
            if time.time() > deadline - 45.0:
                break
            hp = _perturb(best_h, amp, rng)
            dd = min(deadline, time.time() + 70.0)
            if dd - time.time() < 20.0:
                break
            hh, cc = _search_one(hp, len(hp), dd, rng)
            if cc + 1e-15 < best_c:
                best_c, best_h = cc, hh

    if deadline - time.time() > 35.0:
        best_n = len(best_h)
        dd = min(deadline, time.time() + min(160.0, deadline - time.time()))
        hh, cc = _search_one(best_h.copy(), best_n, dd, rng)
        if cc + 1e-15 < best_c:
            best_c, best_h = cc, hh

    if deadline - time.time() > 5.0:
        nt = int(min(9000, max(300, (deadline - time.time()) / 0.005)))
        best_h, best_c = _local_2pt(best_h, best_c, nt, rng, deadline - 1.0)

    best_h = _repair(best_h, len(best_h))
    best_c = _eval(best_h)
    return best_h, float(best_c), len(best_h)
```
