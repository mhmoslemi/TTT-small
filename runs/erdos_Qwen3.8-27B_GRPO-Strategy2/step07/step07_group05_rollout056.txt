import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    amt = min(diff, 1.0 - h[idx])
                    h[idx] += amt
                    diff -= amt
                else:
                    amt = min(-diff, h[idx])
                    h[idx] -= amt
                    diff += amt
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        h2, c2 = _gd_annealed(best_h.copy(), n0, iters, a_s, a_e, _bc)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 2: SLSQP with multiple smoothing levels
    if _el() < budget_s - 180:
        for a_slx in [15000, 40000, 80000]:
            if _bc():
                break
            hr, cr = _slsqp_run(best_h.copy(), n0, a_slx, _bc)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr

    # Phase 3: Second round of gradient descent after SLSQP
    if _el() < budget_s - 120:
        for a_s, a_e, iters in [(500, 30000, 300), (3000, 200000, 150)]:
            if _bc():
                break
            h2, c2 = _gd_annealed(best_h.copy(), n0, iters, a_s, a_e, _bc)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2

    # Phase 4: Perturbation + re-optimize (escape local minima)
    rng = np.random.RandomState(seed)
    perturb_dims = [0.006, 0.003, 0.001, 0.004]
    for pd in perturb_dims:
        if _bc():
            break
        if _el() > budget_s - 80:
            break
        h_p = best_h + pd * rng.randn(n0)
        h_p = _repair(h_p, n0)
        h2, c2 = _gd_annealed(h_p, n0, 200, 100, 20000, _bc)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 5: Final SLSQP polish
    if _el() < budget_s - 80:
        hr, cr = _slsqp_run(best_h.copy(), n0, 100000, _bc)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr

    # Phase 6: Triple-point local search
    if _el() < budget_s - 30:
        nt = min(int((budget_s - 30 - _el()) / 0.004), 150000)
        best_h, best_c5 = _local_triple(best_h, n0, nt, rng, best_c5, _bc)

    # Phase 7: Pairwise ultra-fine polish
    if _el() < budget_s - 5:
        nt = min(int((budget_s - 5 - _el()) / 0.0015), 400000)
        best_h, best_c5 = _local_pair(best_h, n0, nt, rng, best_c5, _bc)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
