import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _pg(h, j, n):
    """Point gradient: gradient of full corr w.r.t. h[j]"""
    g = np.zeros(2 * n - 1)
    g[j:j + n] = 1.0 - 2.0 * h
    g *= 2.0 / n
    return g

def _pair_cd(h, n, rng, t0, deadline, max_iter=200000):
    best_h = h.copy()
    best_c = _eval(h)
    c5_cur = best_c
    c = _corr(h)
    stable = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        g_diff = _pg(h, i, n) - _pg(h, j, n)
        k_cross = i + j  # position of quadratic cross term
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        lo_b, hi_b = d_lo, d_hi
        for _ in range(35):
            d_mid = 0.5 * (lo_b + hi_b)
            # Directional derivative of max(c + d*g_diff + d^2*e_{k_cross})
            # Find argmax
            vals = c + d_mid * g_diff
            if 0 <= k_cross < 2 * n - 1:
                vals[k_cross] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g_diff[kmax]
            if 0 <= k_cross < 2 * n - 1 and kmax == k_cross:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                hi_b = d_mid
            elif sg < -1e-14:
                lo_b = d_mid
            else:
                break
        d_opt = 0.5 * (lo_b + hi_b)
        if abs(d_opt) < 1e-13:
            stable += 1
            if stable > 8000:
                break
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g_diff
        if 0 <= k_cross < 2 * n - 1:
            c[k_cross] += d_opt * d_opt
        c5_new = c.max()
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g_diff
            if 0 <= k_cross < 2 * n - 1:
                c[k_cross] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _grad_descent(h, n, rng, t0, deadline, top_k=5, temp=0.02):
    best_h = h.copy()
    best_c = _eval(h)
    c5_cur = best_c
    for it in range(5000):
        if time.time() > deadline:
            break
        c = _corr(h)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        ct = c[top_idx]
        w = np.exp((ct - ct.max()) / temp)
        w /= w.sum()
        g = np.zeros(n)
        for ii, ww in zip(top_idx, w):
            # dc[p]/dh[j] = 1-2h[p-j] for p in [j, j+n-1]
            # So dg[j] = sum_p w_k * (1-2h[p-j]) for p in top set intersect [j, j+n-1]
            # For a given top lag idx, it contributes to g[j] for j in [idx-(n-1), idx] ∩ [0, n-1]
            # with value 1-2h[idx-j]
            jlo = max(0, ii - (n - 1))
            jhi = min(n - 1, ii)
            if jlo <= jhi:
                js = np.arange(jlo, jhi + 1)
                g[js] += ww * (1.0 - 2.0 * h[ii - js])
        g *= 2.0 / n
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = 0.3 / gn
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = _proj(h - alpha * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if c5_cur < best_c:
            best_c = c5_cur
            best_h = h.copy()
        if not improved and it % 200 == 0:
            # Nudge
            h = np.clip(h + 0.01 * rng.randn(n), 0, 1)
            h = _proj(h, n * 0.5)
    return best_h, best_c

def _fine_polish(h, n, rng, t0, deadline):
    best_h = h.copy()
    best_c = _eval(h)
    c5_cur = best_c
    step = 1e-4
    for it in range(500000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5_cur - 1e-18:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _interp_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    n0 = len(initial_h_values)

    best_h = h0.copy()
    best_c5 = _eval(best_h)

    for n_try in [n0//2, n0*3//4, n0*5//4, n0*3//2, n0*2]:
        if 100 <= n_try <= 1000 and n_try != n0 and n_try % 2 == 0:
            n_candidates.append(n_try)
    
    # Optimize for each n candidate with time budget
    time_per_n = budget_s / (len(n_candidates) + 2)
    
    for n_cand in n_candidates:
        if time.time() > t0 + budget_s * 0.75:
            break
        h_cand = _repair(_interp_h(h0, n_cand) if n_cand != n0 else h0.copy(), n_cand)
        c5_cand = _eval(h_cand)
        
        # Phase A: Pairwise CD (60% of allocated time)
        dl_a = min(t0 + (time.time() - t0) + time_per_n * 0.6, t0 + budget_s * 0.75)
        h_a, c5_a = _pair_cd(h_cand, n_cand, rng, t0, dl_a)
        if c5_a < best_c5:
            best_c5, best_h, best_n = c5_a, h_a, n_cand
        
        # Phase B: Gradient descent (20%)
        dl_b = min(dl_a + time_per_n * 0.2, t0 + budget_s * 0.85)
        h_b, c5_b = _grad_descent(best_h if best_n == n_cand else h_a, n_cand, rng, t0, dl_b)
        if c5_b < best_c5:
            best_c5, best_h, best_n = c5_b, h_b, n_cand
        
        # Phase C: More pairwise CD (20%)
        dl_c = min(dl_b + time_per_n * 0.2, t0 + budget_s * 0.92)
        h_c, c5_c = _pair_cd(best_h if best_n == n_cand else h_b, n_cand, rng, t0, dl_c)
        if c5_c < best_c5:
            best_c5, best_h, best_n = c5_c, h_c, n_cand
    
    # Final: Optimize the best solution further
    if time.time() < t0 + budget_s * 0.93:

        h_f, c5_f = _pair_cd(best_h, best_n, rng, t0, dl_f)
        if c5_f < best_c5:
