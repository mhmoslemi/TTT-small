from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lse(h, alpha, n_significant=50):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    # Only use top n_significant
    idx = np.argsort(w)[-n_significant:]
    g = np.zeros(n)
    for k_idx in idx:
        wk = w[k_idx] * dx
        if wk < 1e-16:
            continue
        s_val = (n - 1) - k_idx
        if s_val >= 0:
            g[:n-s_val] += wk * (1.0 - h[s_val:n])
            g[s_val:] -= wk * h[:n-s_val]
        else:
            sp = -s_val
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _lse_cost(h, alpha):
    n = len(h)
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _nesterov_lse(h, alpha, steps, t0, budget_end, step0=0.001, mu=0.9, max_nsig=50):
    n = len(h)
    f = _lse_cost(h, alpha)
    v = np.zeros(n)
    step = step0
    best_h = h.copy()
    best_f = f
    for _s in range(steps):
        if time.time() > budget_end:
            break
        g = _grad_lse(h, alpha, max_nsig)
        h_prev = h.copy()
        y = h + mu * v  # Nesterov lookahead
        v = mu * v + (1 - mu) * _grad_lse(y, alpha, max_nsig)
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _lse_cost(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
            step = min(step * 1.02, step0 * 10)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            h = h_prev
    return best_h

def _coord_descent(h, n_tries, rng, t0, budget_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    c = _corr(h)
    for _it in range(n_tries):
        if time.time() > budget_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try adjusting h[i] and h[j] to maintain sum
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _perturb_reopt(h, alpha, strength, steps, t0, budget_end, rng):

    h_p = h + strength * rng.randn(n)
    h_p = _repair(h_p, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    def elapsed():
        return time.time() - t0

    # Phase 1: Multi-alpha momentum LSE (progressive refinement)
    alphas = [2000, 5000, 15000, 40000, 80000]
    for ai, alpha in enumerate(alphas):
        rem = budget_s - elapsed()
        if rem < 80:
            break
        be = t0 + budget_s - max(60, (len(alphas) - ai - 1) * 40)
        ss = 0.001 * (1.0 - 0.1 * ai)
        h = _momentum_lse(h, alpha, 4000, t0, be, ss, 0.85, 50, rng)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: SLSQP refinement
    if elapsed() < budget_s - 180:
        for al in [15000, 60000]:
            if elapsed() > budget_s - 80:
                break
            h_c = _slsqp_opt(best_h.copy(), al, 800, 1e-16)
            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_c.copy()

    # Phase 3: Perturbation escapes (try different basins)
    for trial in range(4):
        if elapsed() > budget_s - 60:
            break
        be = t0 + budget_s - 40
        h_p = _perturb_reopt(best_h, 30000, 0.006 + 0.004 * trial, 1500, t0, be, rng, 0.0004)
        c5v = _eval(h_p)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_p.copy()

    # Phase 4: High-alpha fine momentum
    if elapsed() < budget_s - 50:
        be = t0 + budget_s - 25
        h = _momentum_lse(best_h.copy(), 150000, 2500, t0, be, 0.0002, 0.8, 30, rng)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 5: Another SLSQP pass if time allows
    if elapsed() < budget_s - 60:
        h_c = _slsqp_opt(best_h.copy(), 100000, 600, 1e-17)
        c5v = _eval(h_c)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_c.copy()

    # Phase 6: Adaptive local search (most of remaining time)
    if elapsed() < budget_s - 5:
        be = t0 + budget_s - 2
        best_h, best_c5 = _adaptive_local_search(best_h, rng, t0, be)

    # Phase 7: Ultra-fine polish with tiny steps
    if elapsed() < budget_s - 2:
        n = len(best_h)
        c5c = _eval(best_h)
        for _it in range(200000):
            if time.time() > t0 + budget_s - 1:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = 1e-7 * rng.randn()
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                c5n = _eval(best_h)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                else:
                    best_h[i], best_h[j] = ni - d, nj + d
        best_c5 = c5c

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
