import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Project to sum constraint using bisection on Lagrange multiplier
    if abs(s - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            s = np.clip(h - mid, 0, 1).sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        # Fix any residual error
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp(h, alpha, maxiter, ftol, target):

    bnds = [(0.0, 1.0)] * n
    obj_f = lambda x: _lse_obj(x, alpha, n)
    grad_f = lambda x: _lse_grad(x, alpha, n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8

    for _s in range(steps):

                    g[i2] -= wk * h[i2 - sh]

                g[i2] -= wk * h[i2 + sp]

        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)

            h = _repair(h_prev - step * v, n)
            f = _eval(h)

    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c

        i = rng.randint(n)

                    best_h = h.copy()
            else:

    return best_h, best_c

        best_h, best_c5 = _local_2pt(best_h, ntr, rng, t_global_end - 0.5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
