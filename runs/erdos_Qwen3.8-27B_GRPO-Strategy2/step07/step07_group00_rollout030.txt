def _coord_descent(h, n_sweeps, deadline, bisection_iters=25):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = c.max()
    best_h = h.copy()
    # Precompute s_k = n-1-k for k=0,...,2n-2
    s_vals = (n - 1) - np.arange(2 * n - 1).astype(float)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # For each k, determine I1 and I2
            # I1: 0 <= j - s_k < n  →  j-n < s_k <= j  (i.e., s_k in (j-n, j])
            # I2: 0 <= j + s_k < n  →  -j <= s_k < n-j  (i.e., s_k in [-j, n-j))
            I1 = ((s_vals > j - n) & (s_vals <= j)).astype(np.float64)
            I2 = ((s_vals >= -j) & (s_vals < n - j)).astype(np.float64)
            
            # h[j-s_k] for I1 positions
            # h[j+s_k] for I2 positions
            # B[k] = dx * ((1-h[j+s_k])*I2 - h[j-s_k]*I1)
            
            # Compute b_j_plus = h[j+s_k] for all k, masked by I2
            # j+s_k is valid only where I2>0
            jp = (j + s_vals).astype(int)
            jm = (j - s_vals).astype(int)
            
            # Safe indexing: clip to [0, n-1]
            b_plus = np.where(I2 > 0, h[np.clip(jp, 0, n-1)], 0.0)
            b_minus = np.where(I1 > 0, h[np.clip(jm, 0, n-1)], 0.0)
            
            B = dx * ((1.0 - b_plus) * I2 - b_minus * I1)
            A = c - B * h[j]
            
            # Minimize max(A + B*x) over x in [0,1] via bisection
            # f(x) = max_k(A[k] + B[k]*x), f'(x) = B[argmax]
            lo, hi = 0.0, 1.0
            for _ in range(bisection_iters):
                mid = (lo + hi) * 0.5
                vals = A + B * mid
                kmax = vals.argmax()
                slope = B[kmax]
                if slope > 1e-14:
                    hi = mid
                elif slope < -1e-14:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            x_opt = (lo + hi) * 0.5
            if x_opt < 1e-12: x_opt = 0.0
            elif x_opt > 1.0 - 1e-12: x_opt = 1.0
            
            if abs(x_opt - h[j]) > 1e-10:
                c += B * (x_opt - h[j])
                h[j] = x_opt
                improved = True
                cm = c.max()
                if cm < best_c5:
                    best_c5 = cm
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()

        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
