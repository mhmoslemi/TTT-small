            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _local_3pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        k2 = (j + 1 + rng.randint(0, 4)) % n
        d1 = 2e-5 * rng.randn()
        d2 = 2e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k2] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k2] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k2] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _alt_start(n_target, seed):
    rng = np.random.RandomState(seed)
    n_low = 64
    f1, f2, f3 = rng.uniform(1, 4), rng.uniform(2, 6), rng.uniform(4, 8)
    p1, p2, p3 = rng.uniform(0, 2*np.pi, 3)
    x = np.arange(n_low) / n_low * 2.0
    h = 0.5 + 0.35 * np.sin(np.pi * f1 * x + p1) + 0.25 * np.sin(np.pi * f2 * x + p2) + 0.15 * np.sin(np.pi * f3 * x + p3)
    h = np.clip(h, 0, 1)
    h = _repair(h, n_low)
    x_l = (np.arange(n_low) + 0.5) / n_low * 2.0
    x_r = (np.arange(n_target) + 0.5) / n_target * 2.0
    h_ext = np.concatenate([h, h[:1]])
    x_ext = np.concatenate([x_l, [x_l[-1] + 2.0/n_low]])
    h_t = np.interp(x_r, x_ext, h_ext)
    h_t = _repair(h_t, n_target)
    return h_t

def _segment_flip(h, n, rng, n_tries=200):
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        L = 2 + rng.randint(0, 6)
        start = rng.randint(0, n - L)
        h_new = h.copy()
        h_new[start:start+L] = 1.0 - h[start:start+L]
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-12:
            c5c = c5n
            best = h_new.copy()
            best_c = c5n
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Alternative starts, quick optimize
    for s in [seed + 100, seed + 200]:
        if time.time() > deadline - 700:
            break
        h_alt = _alt_start(n0, s)
        h_alt = _lbfgs_run(h_alt, 2000, 800, deadline)
        c5v = _eval(h_alt)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_alt.copy()

    # Phase 2: Progressive log-sum-exp L-BFGS
    h = best_h.copy()
    for alpha in [1000, 5000, 20000, 100000, 500000]:
        if time.time() > deadline - 450:
            break
        h = _lbfgs_run(h, alpha, 4000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: p-norm bridge
    h = best_h.copy()
    for p in [10, 30, 100, 300, 1000]:
        if time.time() > deadline - 300:
            break
        h = _lbfgs_run(h, 0, 3000, deadline, use_pnorm=True, p=p)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Active set descent with various n_lags
    if time.time() < deadline - 200:
        for nl in [1, 2, 3, 5, 8, 2, 3]:
            if time.time() > deadline - 150:
                break
            h_ml, c5_ml = _active_descent(best_h.copy(), nl, 1500, deadline)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()

    # Phase 5: Very high alpha L-BFGS
    if time.time() < deadline - 120:
        h = _lbfgs_run(best_h, 2000000, 5000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Subgradient polish
    if time.time() < deadline - 80:
        h = _subgrad(best_h, 5000, 0.008, 1e-9, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 7: Active descent round 2
    if time.time() < deadline - 60:
        for nl in [1, 2, 3]:
            if time.time() > deadline - 60:
                break
            h_ml, c5_ml = _active_descent(best_h.copy(), nl, 2000, deadline)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()

    # Phase 8: Segment flips
    if time.time() < deadline - 30:
        h_s, c5_s = _segment_flip(best_h, n0, rng, 300)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 9: 3-point local search
    if time.time() < deadline - 8:
        h_ls, c5_ls = _local_3pt(best_h, 50000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 10: 2-point ultra-fine polish
    if time.time() < deadline - 2:
        h_ls, c5_ls = _local_2pt(best_h, 150000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
