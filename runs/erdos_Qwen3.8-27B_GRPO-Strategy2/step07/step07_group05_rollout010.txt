from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                for idx in np.argsort(1.0 - h):
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                    if abs(diff) < 1e-12: break
            else:
                for idx in np.argsort(h):
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                    if abs(diff) < 1e-12: break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):

        if np.clip(h - mid, 0, 1).sum() > target:

        else:

    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _mlgd(h, iters, k_top, steps0, momentum, check):
    """Multi-lag gradient descent with momentum"""
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    steps = list(steps0)
    v = np.zeros(n)  # momentum
    stall = 0
    for it in range(iters):
        if check():
            break
        c = _full_corr(h)
        top = np.argsort(c)[-k_top:]
        g = np.sum([_grad_lag(h, k) for k in top], axis=0) * dx / k_top
        improved = False
        for si in range(len(steps)):
            v = momentum * v + (1 - momentum) * g
            h_new = _proj(h - steps[si] * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h = h_new; c5 = c5n; improved = True
                steps = [steps[si] * 1.3] + [s for s in steps[:si]] + [s for s in steps[si+1:] if s < steps[si]*0.5]
                if len(steps) > 8: steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            v *= 0.5
            stall += 1
            if stall > 12: break
        else:
            stall = 0
    return h, c5

def _smooth_step(h, step, alpha, check):
    """Single projected gradient step on smooth max"""
    n = len(h)

    c_raw = np.correlate(h, 1.0 - h, mode='full')

    sc = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(sc - sc.max())

        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    h_new = _proj(h - step * g, n * 0.5)
    return h_new

def _local_search(h, ntry, rng, step_size, check, guided_g=None):
    """2-point local search with optional gradient guidance"""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    # Precompute gradient for guidance
    if guided_g is None:
        c = _full_corr(h)
        top = np.argsort(c)[-3:]
        guided_g = np.sum([_grad_lag(h, k) for k in top], axis=0)

        if check(): break
        # Choose coordinate weighted by |gradient|
        w = np.abs(guided_g)
        w /= w.sum() + 1e-30
        i = rng.choice(n, p=w)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step_size * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

            c5n = _eval(h)
            if c5n < best_c - 1e-18:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                for idx in np.argsort(1.0 - h):
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                    if abs(diff) < 1e-12: break
            else:
                for idx in np.argsort(h):
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                    if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        if sh > 0:
            g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _mlgd(h, iters, k_top, steps0, momentum, check):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    steps = list(steps0)
    v = np.zeros(n)
    stall = 0
    for _ in range(iters):
        if check():
            break
        c = _full_corr(h)
        top = np.argsort(c)[-k_top:]
        g = sum(_grad_lag(h, kk) for kk in top)
        g *= dx / k_top
        improved = False
        for si in range(len(steps)):
            v = momentum * v + (1.0 - momentum) * g
            h_new = _proj(h - steps[si] * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h[:] = h_new
                c5 = c5n
                improved = True
                steps = [steps[si] * 1.25] + [s for s in steps[:si]] + [s for s in steps[si+1:] if s < steps[si] * 0.4]
                if len(steps) > 8:
                    steps = steps[:8]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            v *= 0.4
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0
    return h, c5

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = c.max()
    sc = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    g *= dx
    return g, m + np.log(w.sum()) / alpha

def _smooth_phase(h, iters, alpha_start, alpha_end, steps0, check):
    n = len(h)
    c5 = _eval(h)
    steps = list(steps0)
    v = np.zeros(n)
    for it in range(iters):
        if check():
            break
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(iters - 1, 1))
        g, _ = _smooth_grad(h, alpha)
        improved = False
        for si in range(len(steps)):
            v = 0.9 * v + 0.1 * g
            h_new = _proj(h - steps[si] * v, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-16:
                h[:] = h_new
                c5 = c5n
                improved = True
                steps = [steps[si] * 1.2] + [s for s in steps if s < steps[si] * 0.4]
                if len(steps) > 6:
                    steps = steps[:6]
                break
        if not improved:
            steps = [s * 0.5 for s in steps]
            v *= 0.5
    return h, c5

def _local2(h, ntry, rng, step_sz, check, guidance=None):

    best_h = h.copy()
    best_c = _eval(h)
    if guidance is None:

    gw_sum = gw.sum()
    if gw_sum < 1e-30:
        gw = np.full(n, 1.0)
    else:
        gw /= gw_sum
    for _ in range(ntry):
        if check():
            break
        i = rng.choice(n, p=gw)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step_sz * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c - 1e-19:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _local3(h, ntry, rng, step_sz, check):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(ntry):
        if check():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 7 + rng.randint(0, 5)) % n
        d1 = step_sz * rng.randn()
        d2 = step_sz * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < best_c - 1e-19:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j], h[k] = h[i] - d1, h[j] - d2, h[k] + d1 + d2
    return best_h, best_c

def _slsqp_polish(h, check):
    n = len(h)
    dx = 2.0 / n
    alpha = 200.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, nn=n: x.sum() - nn * 0.5,
             'jac': lambda x, nn=n: np.ones(nn), 'args': (n,)}]
    def obj(x):
        c = _full_corr(x)
        m = c.max()
        sc = np.clip(10000.0 * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(sc - sc.max()))) / 10000.0
    def grad(x):
        c = _full_corr(x)
        m = c.max()
        sc = np.clip(10000.0 * (c - m), -300, 300)
        w = np.exp(sc - sc.max())
        w /= w.sum()
        g = np.zeros(n)
        for kk in range(2 * n - 1):
            if w[kk] < 1e-13:
                continue
            gk = _grad_lag(x, kk)
            g += w[kk] * gk
        return g * dx
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 600, 'ftol': 1e-18})
        hr = _repair(res.x, n)
        return hr, _eval(hr)
    except Exception:
        return h, _eval(h)
