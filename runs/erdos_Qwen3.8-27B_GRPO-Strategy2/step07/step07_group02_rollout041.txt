            freq = rng.randint(2, 25)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.002, 0.015)
            pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
        elif mode == 1:
            c0 = rng.randint(0, n)
            w = rng.uniform(5, 30)
            amp = rng.uniform(0.005, 0.03)
            pert = amp * np.exp(-((x - c0) % n)**2 / (2 * w * w))
            pert = amp * np.exp(-0.5 * ((np.minimum(np.abs(x - c0), n - np.abs(x - c0))) / w)**2)
        else:
            c0 = rng.randint(0, n)
            c1 = rng.randint(0, n)
            w = rng.uniform(3, 15)
            amp = rng.uniform(0.005, 0.02)
            d0 = np.minimum(np.abs(x - c0), n - np.abs(x - c0))
            d1 = np.minimum(np.abs(x - c1), n - np.abs(x - c1))
            pert = amp * (np.exp(-0.5 * (d0 / w)**2) - np.exp(-0.5 * (d1 / w)**2))
        pert -= pert.mean()
        h_new = _repair(h + pert, n)
        c5 = _eval(h_new)
        if c5 < best_c:
            best_c = c5
            best_h = h_new.copy()
            h = h_new.copy()
    return best_h, best_c

def _exchange_refine(h, n_iter, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for i in range(n_iter):
        if time.time() > deadline - 3:
            break
        npts = rng.randint(2, 5)
        pts = rng.choice(n, size=npts, replace=False)
        d = rng.uniform(0, 0.015) * rng.randn(npts)
        d -= d.mean()
        h_new = h.copy()
        for j, p in enumerate(pts):
            h_new[p] += d[j]
        if h_new.min() >= -1e-10 and h_new.max() <= 1 + 1e-10:
            h_new = _repair(h_new, n)
            c5 = _eval(h_new)
            if c5 < cur_c - 1e-18:
                h = h_new
                cur_c = c5
                if c5 < best_c:
                    best_c = c5
                    best_h = h_new.copy()
        if i % 500 == 0 and cur_c > best_c + 5e-9:
            h = best_h.copy()
            cur_c = best_c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Sweep p-norm from moderate to very high
    for p in [100, 300, 1000, 3000, 10000, 30000]:
        if time.time() > deadline - 350:
            break
        h = _lbfgs_pnorm(best_h.copy(), p, 6000, deadline)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Log-sum-exp with very high alpha
    for alpha in [200000, 1000000, 5000000, 20000000]:
        if time.time() > deadline - 280:
            break
        h = _lbfgs_lse(best_h.copy(), alpha, 6000, deadline)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Multi-resolution (downsample, opt, upsample, refine)
    if time.time() < deadline - 220:
        for factor in [2, 4]:
            if time.time() > deadline - 220:
                break
            n_small = n0 // factor
            h_small = best_h[:n_small * factor].reshape(n_small, factor).mean(axis=1)
            h_small = _repair(h_small, n_small)
            for p in [500, 5000, 50000]:
                if time.time() > deadline - 220:
                    break
                h_small = _lbfgs_pnorm(h_small, p, 4000, deadline)
            h_up = np.repeat(h_small, factor)[:n0]
            h_up = _repair(h_up, n0)
            h_ref = _lbfgs_pnorm(h_up, 5000, 4000, deadline)
            c5 = _eval(h_ref)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()

    # Phase 4: Top-k descent with varying k
    if time.time() < deadline - 150:
        for topk in [3, 5, 8, 12]:
            if time.time() > deadline - 150:
                break
            h_top, c5_top = _topk_descent(best_h.copy(), topk, 400, deadline)
            if c5_top < best_c5:
                best_c5 = c5_top
                best_h = h_top.copy()

    # Phase 5: Stochastic subgradient
    if time.time() < deadline - 80:
        h_sg = _stochastic_subgrad(best_h.copy(), 5000, 0.008, deadline, rng)
        c5 = _eval(h_sg)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sg.copy()

    # Phase 6: Multi-modal perturbation (sinusoidal, gaussian bumps)
    if time.time() < deadline - 40:
        h_mp, c5_mp = _multi_perturb(best_h.copy(), rng, deadline, 3000)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()

    # Phase 7: Exchange search fine polish
    if time.time() < deadline - 15:
        h_ex, c5_ex = _exchange_refine(best_h.copy(), 60000, rng, deadline)
        if c5_ex < best_c5:
            best_c5 = c5_ex
            best_h = h_ex.copy()

    # Phase 8: Final ultra-high p polish
    if time.time() < deadline - 5:
        h_fin = _lbfgs_pnorm(best_h.copy(), 50000, 4000, deadline)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
