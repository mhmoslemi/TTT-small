import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0., 1.).astype(float)
    tgt = n / 2.
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= tgt / s
        h = np.clip(h, 0., 1.)
        s = h.sum()
        if abs(s - tgt) < 1e-14:
            break
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.-h, mode='full')) * 2.0 / n)

def _proj(h, tgt):
    lo, hi = -4., 4.
    for _ in range(60):
        mid = (lo+hi)*.5
        if np.clip(h-mid, 0, 1).sum() > tgt: lo = mid
        else: hi = mid
    return np.clip(h-(lo+hi)*.5, 0, 1)

def _softmax_obj(x, al):
    n = len(x)
    dx = 2./n
    c = np.correlate(x, 1.-x, mode='full') * dx
    m = c.max()
    s = np.clip(al*(c-m), -300, 300) - np.clip(al*(c-m), -300, 300).max()
    return m + np.log(np.exp(s).sum())/al

def _softmax_grad(x, al):
    n = len(x)
    dx = 2./n
    c = np.correlate(x, 1.-x, mode='full') * dx
    m = c.max()
    raw = np.clip(al*(c-m), -300, 300)
    s = raw - raw.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(2*n-1)  # gradient w.r.t. c, then chain rule
    # Actually, gradient w.r.t. x:
    g2 = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15: continue
        sh = (n-1) - k
        wk = w[k] * dx
        if sh >= 0:
            i1 = np.arange(n-sh)
            g2[i1] += wk*(1.-x[i1+sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g2[i2] -= wk*x[i2-sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g2[i1] += wk*(1.-x[i1-sp])
            i2 = np.arange(n-sp)
            g2[i2] -= wk*x[i2+sp]
    return g2

def _sum_con(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, al, mx, ft):
    n = len(h)
    try:
        r = minimize(_softmax_obj, h, jac=_softmax_grad, args=(al,),
                     method='SLSQP', bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':_sum_con,'jac':_sum_jac,'args':(n*0.5,)}],
                     options={'maxiter':mx,'ftol':ft})

    except: return h

    """Find optimal h[i] given all other h[j] fixed. Returns (best_x, best_c5)."""

    hi = h[i]

    #   first: h[i]*(1-h[i+k-n+1]) if valid

    A = np.zeros(2*n-1)

        m_hi = n-1

The issue is that after changing h[i], c_full becomes stale. For exact coordinate descent, we need the current c_full. But recomputing it after each change makes it O(n³).

Alternative: Use the stale c_full (computed at the start of the sweep). This is the "Gauss-Seidel" vs "Jacobi" distinction. With stale c_full, each coordinate update is based on the previous state, which is fine for convergence (just slower).

Or: Update c_full incrementally. When h[i] changes from old to new, the change in c(k) is:
Δc(k) = (new - old) * coeff[k] (approximately, ignoring the quadratic term at k=n-1)

So I can update c_full in O(2n) per coordinate change. Total per sweep: O(n * 2n) = O(2n²).

    n = len(h)
    dx = 2.0/n
    c_full = np.correlate(h, 1.-h, mode='full')
    for sw in range(n_sweeps):
        for i in range(n):
            hi = h[i]
            # Compute A and coeff
            A = c_full.copy()
            coeff = np.zeros(2*n-1)
            for k in range(2*n-1):
                m_lo = max(0, n-1-k)
                if m_lo <= i <= n-1:
                    s = i + k - (n-1)
                    if 0 <= s <= n-1 and s != i:
                        A[k] -= hi * (1-h[s])
                        coeff[k] += (1-h[s])
                    elif s == i:
                        A[k] -= hi*(1-hi)
                        # quadratic: A[k] + x - x^2
                m2 = i - k + (n-1)
                if m2 != i and m_lo <= m2 <= n-1:
                    A[k] -= h[m2]*(1-hi)
                    coeff[k] -= h[m2]
            
            # Find best x
            best_x = hi
            best_v = c_full.max()
            for x in [0,1,0.5]:
                v = np.max(A + coeff*x)
                if k == n-1: v = max(v, A[n-1]+x-x*x)  # hmm, this is already in A+coeff*x approximately
                # Actually for k=n-1: c = A[n-1] + x - x^2
                # A+coeff*x gives A[n-1] + coeff[n-1]*x which is the linear approximation
                # The exact is A[n-1] + x - x^2
                vals = A + coeff*x
                vals[n-1] = A[n-1] + x - x*x
                v = vals.max()
                if v < best_v: best_v = v; best_x = x

            if best_x != hi:
                delta = best_x - hi
                for k in range(2*n-1):
                    if k == n-1:
                        c_full[k] += (best_x - best_x*best_x) - (hi - hi*hi)
                    else:
                        c_full[k] += delta * coeff[k]
                h[i] = best_x
    
    return _repair(h, n)
