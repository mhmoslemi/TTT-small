import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lbe_grad(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbe_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _do_lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbe_obj, h, jac=_lbe_grad, args=(alpha, n),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 60})
        return _repair(res.x, n)
    except Exception:
        return h

def _active_lbfgs(h, top_k, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 20:
        return h
    def obj(x):
        c = _cvals(x)
        idx = np.argsort(c)[-top_k:]
        vals = c[idx]
        m = vals.max()
        s = np.clip(alpha * (vals - m), -400, 0)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        c = _cvals(x)
        idx = np.argsort(c)[-top_k:]
        vals = c[idx]
        m = vals.max()
        s = np.clip(alpha * (vals - m), -400, 0)
        w = np.exp(s)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[ii] * dx
            if sh >= 0:
                np_ = n - sh
                g[:np_] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:np_]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:-sp] -= wk * x[sp:]
        return g
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 60})
        return _repair(res.x, n)
    except Exception:
        return h

def _top_lag_descent(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 25:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_lags
        improved = False
        for step in [0.03, 0.015, 0.007, 0.003, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-17:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _coord_descent(h, n_iter, rng, deadline):
    n = len(h)
    cur = h.copy()
    cur_c = _eval(cur)
    for i in range(n_iter):
        if time.time() > deadline - 3:
            break
        i1 = rng.randint(n)
        i2 = (i1 + 1 + rng.randint(0, 4)) % n
        d = 1e-6 * rng.randn()
        ni, nj = cur[i1] + d, cur[i2] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            cur[i1], cur[i2] = ni, nj
            c_new = _eval(cur)
            if c_new < cur_c - 1e-18:
                cur_c = c_new
            else:
                cur[i1], cur[i2] = ni - d, nj + d
    return cur, cur_c

def _spsa(h, steps, c0, gamma, rng, deadline):
    n = len(h)
    cur = h.copy()
    best = h.copy()
    best_c = _eval(h)
    for i in range(steps):
        if time.time() > deadline - 5:
            break
        t = c0 / (1.0 + gamma * i)
        delta = rng.choice([-1.0, 1.0], size=n)
        hp = _repair(cur + t * delta, n)
        hm = _repair(cur - t * delta, n)
        fp = _eval(hp)
        fm = _eval(hm)
        g_est = (fp - fm) / (2.0 * t) * delta
        a = 0.02 / (1.0 + 0.05 * i)
        cur = _repair(cur - a * g_est, n)
        cn = _eval(cur)
        if cn < best_c:
            best_c = cn
            best = cur.copy()
    return best, best_c

def _low_res_opt(n_low, seed, deadline):
    rng2 = np.random.RandomState(seed)
    x = np.linspace(0, 2, n_low, endpoint=False)
    h = np.zeros(n_low)
    for f in range(1, 5):
        h += 0.25 * np.sin(2 * np.pi * f * x / 2.0 + rng2.rand() * 6.28)
    h += 0.5
    h = _repair(h, n_low)
    p = 30
    def pobj(xx):
        cv = np.maximum(_cvals(xx), 0)
        return (np.sum(cv ** p)) ** (1.0 / p)
    def pgrad(xx):
        cv = np.maximum(_cvals(xx), 0)
        fp = (np.sum(cv ** p)) ** (1.0 / p)
        if fp < 1e-30:
            return np.zeros_like(xx)
        w = (cv / fp) ** (p - 1)
        ws = w.sum()
        if ws < 1e-30:
            return np.zeros_like(xx)
        w /= ws
        dx = 2.0 / n_low
        gg = np.zeros(n_low)
        for k in range(2 * n_low - 1):
            if w[k] < 1e-15:
                continue
            sh = (n_low - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                np_ = n_low - sh
                gg[:np_] += wk * (1.0 - xx[sh:])
                if sh > 0:
                    gg[sh:] -= wk * xx[:np_]
            else:
                sp = -sh
                gg[sp:] += wk * (1.0 - xx[:-sp])
                gg[:-sp] -= wk * xx[sp:]
        return gg
    try:
        res = minimize(pobj, h, jac=pgrad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n_low,
                       options={'maxiter': 2000, 'ftol': 1e-14})
        h = _repair(res.x, n_low)
    except Exception:
        pass
    h = _do_lbfgs(h, 20000, 2000, deadline)
    return h

def _upsample(h_low, n_high):
    n_low = len(h_low)
    x_low = np.linspace(0, 2, n_low, endpoint=False)
    x_high = np.linspace(0, 2, n_high, endpoint=False)
    h_high = np.interp(x_high, x_low, h_low)
    return _repair(h_high, n_high)

def _subgrad_exact(h, steps, step0, tol, deadline):
    n = len(h)
    dx = 2.0 / n
    for i in range(steps):
        if time.time() > deadline - 15:
            break
        t = step0 / (1.0 + 0.01 * i)
        c = _cvals(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / max(1, len(active))
        h = _repair(h - t * g, n)
    return h
