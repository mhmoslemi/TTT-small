import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sg_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            m2 = n - sh
            g[:m2] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:m2]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 30:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sg_obj, h, jac=_sg_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxls': 50, 'maxcor': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _lag_grad(h, k_list):
    """Compute gradient of average of c[k] for k in k_list."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_list:
        sh = (n - 1) - k
        if sh >= 0:
            m2 = n - sh
            g[:m2] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:m2]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:-sp] -= h[sp:]
    g *= dx / len(k_list)
    return g

def _top_lag_descent(h, n_lags, max_rounds, deadline):
    n = len(h)
    c5c = _eval(h)
    best = h.copy()
    best_c = c5c
    for rnd in range(max_rounds):
        if time.time() > deadline - 10:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = _lag_grad(h, top_k)
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
            h_new = _repair(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5c - 1e-17:
                h = h_new
                c5c = c5n
                improved = True
                if c5n < best_c:
                    best_c = c5n
                    best = h.copy()
                break
        if not improved:

            else:

    return best, best_c

def _coord_search(h, n_iter, deadline):
    """Coordinate-wise line search on single points."""
    n = len(h)

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sg_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            m2 = n - sh
            g[:m2] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:m2]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lag_grad(h, k_list):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_list:
        sh = (n - 1) - k
        if sh >= 0:
            m2 = n - sh
            g[:m2] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:m2]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:-sp] -= h[sp:]
    g *= dx / len(k_list)
    return g

def _newton_polish(h, n_lags, n_iters, deadline):

            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 8: Ultra-fine 2-point local search
    if time.time() < deadline - 3:
        h_ls, c5_ls = _fine_2pt(best_h, 150000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final Newton micro-polish
    if time.time() < deadline - 1:
        h_nw, c_nw = _newton_polish(best_h, 2, 50, deadline)
        if c_nw < best_c5:
            best_c5 = c_nw
            best_h = h_nw.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
