    best_h = hc.copy()
    stall = 0
    k = k_s
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(hc)
        if stall > 50:
            k = min(k + 1, k_max)
        elif stall > 0 and it % 500 == 0:
            k = max(k - 1, k_s)
        topk = np.argsort(c)[-k:]
        g = _topk_grad(hc, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = mom * v + (1.0 - mom) * (g / gn)
        hv = _proj_sum(hc - step * v, n * 0.5)
        cv = _eval(hv)
        if cv < best_c - 1e-17:
            hc = hv
            best_c = cv
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            step *= 0.55
            stall += 1
            if step < 1e-11:
                step = s0 * 0.05
                v *= 0.3
        if it % 2000 == 0 and stall > 200:
            mom = 0.6
            step = s0 * 0.02
    return best_h, best_c

def _block_local(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    scale = 8e-5
    for it in range(n_iter):
        if bc_fn():
            break
        bs = rng.choice((2, 3, 4, 6))
        i = rng.randint(max(1, n - bs - 10))
        j = (i + bs + 1 + rng.randint(0, max(2, n // 6))) % n
        d = rng.randn() * scale
        ni = hc[i:i + bs] + d / bs
        nj = hc[j:j + bs] - d / bs
        if np.all(ni >= 0) and np.all(ni <= 1) and np.all(nj >= 0) and np.all(nj <= 1):
            hc[i:i + bs] = ni
            hc[j:j + bs] = nj
            cn = _eval(hc)
            if cn < best_c:
                best_c = cn
                best_h = hc.copy()
            else:
                hc[i:i + bs] -= d / bs
                hc[j:j + bs] += d / bs
        if it % 4000 == 0:
            scale *= 0.75
            if scale < 3e-7:
                break
    return best_h, best_c

def _fine_local(h, n_iter, rng, el_fn, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    scale = 3e-5
    for it in range(n_iter):
        if bc_fn():
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = rng.randn() * scale
        ni, nj = hc[i] + d, hc[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            hc[i], hc[j] = ni, nj
            cn = _eval(hc)
            if cn < best_c:
                best_c = cn
                best_h = hc.copy()
            else:
                hc[i], hc[j] = ni - d, nj + d
        if it % 5000 == 0:
            scale *= 0.7
            if scale < 1e-7:
                break
    return best_h, best_c

def _active_set_descent(h, n_iter, el_fn, bc_fn):
    """Focus on the narrow band of near-max correlations."""
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    step = 0.008
    for it in range(n_iter):
        if bc_fn():
            break
        c = _corrs(hc)
        mx = np.max(c)
        band = np.where(c >= mx - 1e-4)[0]
        if len(band) == 0:
            band = np.array([np.argmax(c)])
        g = _topk_grad(hc, band)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        hv = _proj_sum(hc - step * g / gn, n * 0.5)
        cv = _eval(hv)
        if cv < best_c - 1e-17:
            hc = hv
            best_c = cv
            best_h = hc.copy()
            step = min(step * 1.02, 0.03)
        else:
            step *= 0.7
            if step < 1e-9:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    def bc():
        return el() > budget_s - 15

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS log-sum-exp with increasing beta
    for beta in [30, 80, 200, 500, 1200, 3000]:
        if el() > 60:
            break
        ht = _lbfgs_lseg(best_h.copy(), beta, 1e6, 400)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 2: p-norm L-BFGS (very high p ~ max)
    for p in [64, 256, 1024]:
        if el() > 120:
            break
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(_pnorm_obj, best_h.copy(), jac=_pnorm_grad,
                           args=(p, 1e7), method='L-BFGS-B',
                           bounds=bnds, options={'maxiter': 500,
                           'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 40})
            ht = _repair(res.x, n)
        except Exception:
            ht = best_h.copy()
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 3: Accelerated top-k descent (main)
    if el() < budget_s - 300:
        ht, ct = _accdescent(best_h, 15000, 3, 50, 0.015, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 4: Active-set descent (focus on near-max correlations)
    if el() < budget_s - 250:
        ht, ct = _active_set_descent(best_h, 20000, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 5: L-BFGS polish at very high beta
    if el() < budget_s - 200:
        for beta in [5000, 10000, 20000]:
            if el() > budget_s - 200:
                break
            ht = _lbfgs_lseg(best_h.copy(), beta, 1e8, 200)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht

    # Phase 6: Block local search
    if el() < budget_s - 120:
        nt = min(int((budget_s - 120 - el()) / 0.003), 150000)
        best_h, best_c5 = _block_local(best_h, nt, rng, el, bc)

    # Phase 7: Active-set fine descent
    if el() < budget_s - 80:
        ht, ct = _active_set_descent(best_h, 15000, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 8: Fine 2-point local search
    if el() < budget_s - 30:
        nt = min(int((budget_s - 30 - el()) / 0.002), 100000)
        best_h, best_c5 = _fine_local(best_h, nt, rng, el, bc)

    # Phase 9: Final active-set micro-descent
    if el() < budget_s - 8:
        ht, ct = _active_set_descent(best_h, 10000, el, bc)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 10: Final L-BFGS polish
    if el() < budget_s - 3:
        ht = _lbfgs_lseg(best_h.copy(), 20000, 1e9, 100)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
