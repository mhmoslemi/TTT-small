import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _eval(h):
    return float(np.max(_corrs(h)))

def _weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += w * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g * dx

def _lse_obj(h, alpha, pen):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    sh = alpha * (c - cm)
    es = np.exp(sh)
    return cm + np.log(es.sum()) / alpha + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, alpha, pen):
    n = len(h)
    c = _corrs(h)
    cm = c.max()
    es = np.exp(alpha * (c - cm))
    w = es / es.sum()
    g = _weighted_grad(h, w)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lse_solve(h, alpha, pen, maxit):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=_lse_grad, args=(alpha, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-13})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sg_topk(h, nit, k, s0, el_fn, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    best_c = _eval(hc)
    best_h = hc.copy()
    step = s0
    stall = 0
    for _ in range(nit):
        if bc_fn():
            break
        c = _corrs(hc)
        idx = np.argsort(c)[-k:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / k
        g = _weighted_grad(hc, w)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = 0.5 * v + 0.5 * g / gn
        ht = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(ht)
        if ct < best_c - 5e-17:
            hc = ht
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.03, 0.06)
        else:
            step *= 0.45
            stall += 1
            if stall > 60 or step < 1e-10:
                break
    return best_h, best_c

def _local_search(h, el_fn, bc_fn, rng, mscale):

            else:

        if it % 20000 == 0:
            scale *= 0.85
        if scale < 1e-9:
            break
    return best_h, best_c

def _block_search(h, el_fn, bc_fn, rng, bsize, nblk):
    n = len(h)
    hc = h.copy().astype(float)
    best_c = _eval(hc)
    best_h = hc.copy()
    scale = 3e-4
    for _ in range(nblk):
        if bc_fn():
            break
        idx = rng.choice(n, bsize, replace=False)
        d = rng.randn(bsize) * scale
        d -= d.sum() / bsize
        ok = True
        for i, di in zip(idx, d):
            nv = hc[i] + di
            if nv < -1e-12 or nv > 1.0 + 1e-12:
                ok = False
                break
        if ok:
            for i, di in zip(idx, d):
                hc[i] += di
            cn = _eval(hc)
            if cn < best_c:
                best_c = cn
                best_h = hc.copy()
            else:
                for i, di in zip(idx, d):
                    hc[i] -= di
    return best_h, best_c
