        i0 = rng.randint(0, max(1, n - bs))
        i1 = (i0 + rng.randint(bs, n)) % n
        i1 = min(i1, n - bs)
        if abs(i0 - i1) < bs:
            continue
        h_new = h.copy()
        h_new[i0:i0+bs], h_new[i1:i1+bs] = h[i1:i1+bs].copy(), h[i0:i0+bs].copy()
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            h = h_new; c5c = c5n
            if c5c < bc:
                bc = c5c; bh = h.copy()
    return bh, bc

def _quantize(h, thr):
    n = len(h)
    h_new = h.copy()
    h_new[h_new < thr] = 0.0
    h_new[h_new > 1.0 - thr] = 1.0
    return _repair(h_new, n)

def _fine_polish(h, n_iter, rng, deadline):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5c < bc:
                    bc = c5c; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B with increasing alpha
    dl1 = min(deadline, t0 + 280)
    for alpha, mi in [(3000, 250), (10000, 250), (50000, 200), (200000, 150)]:
        if time.time() > dl1:
            break
        h = _lbfgs(best_h, alpha, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 2: Coordinate descent (directly minimizes max)
    dl2 = min(deadline, t0 + 310)
    h_cd = _coord_descent(best_h.copy(), n_passes=8)
    c5v = _eval(h_cd)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h_cd.copy()

    # Phase 3: Structured gradient annealing
    dl3 = min(deadline, t0 + 620)
    h_a, c5_a = _anneal_grad(best_h.copy(), 60000, rng, dl3, 0.008, 1e-11)
    if c5_a < best_c5:
        best_c5 = c5_a; best_h = h_a.copy()
    # CD after annealing
    h_cd = _coord_descent(best_h.copy(), n_passes=5)
    c5v = _eval(h_cd)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h_cd.copy()

    # Phase 4: Block exploration for diversity
    if time.time() < deadline - 120:
        dl4 = min(deadline, t0 + 800)
        h_s, c5_s = _block_explore(best_h.copy(), 15000, rng, dl4)
        if c5_s < best_c5:
            best_c5 = c5_s; best_h = h_s.copy()
        h_cd = _coord_descent(best_h.copy(), n_passes=5)
        c5v = _eval(h_cd)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h_cd.copy()

    # Phase 5: High-alpha L-BFGS-B polish
    dl5 = min(deadline, t0 + budget_s - 80)
    for alpha, mi in [(1000000, 150), (5000000, 100), (20000000, 80)]:
        if time.time() > dl5:
            break
        h = _lbfgs(best_h, alpha, mi, dl5)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h.copy()

    # Phase 6: Final coordinate descent + quantize trial
    h_cd = _coord_descent(best_h.copy(), n_passes=10)
    c5v = _eval(h_cd)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h_cd.copy()

    # Try quantization at a few thresholds
    for thr in [0.02, 0.01, 0.005]:
        hq = _quantize(best_h, thr)
        hq_cd = _coord_descent(hq, n_passes=3)
        c5q = _eval(hq_cd)
        if c5q < best_c5:
            best_c5 = c5q; best_h = hq_cd.copy()

    # Phase 7: Fine polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.001), 25000)
    best_h, best_c5 = _fine_polish(best_h, n_pol, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
