import numpy as np
from scipy.optimize import minimize
import time

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-9:
        h = _proj(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _Mh(lags, weights, h):
    """Compute M@h where M = sum_k w_k/2 * (S_k + S_k^T)"""
    n = len(h)
    result = np.zeros(n)
    for k, w in zip(lags, weights):
        if k == 0:
            result += w * h
        else:
            result[:n-k] += 0.5 * w * h[k:]
            result[k:] += 0.5 * w * h[:n-k]
    return result

def _quad_M(h, lags, weights):
    """Compute h^T M h = sum_k w_k/2 * sum_i (h[i]*h[i+k] + h[i+k]*h[i]) = sum_k w_k * sum_i h[i]*h[i+k] (for k>0)"""
    # Actually h^T M h where M = sum_k w_k/2 (S_k + S_k^T)
    # h^T S_k h = sum_i h[i] h[i+k]
    # h^T (S_k^T) h = sum_i h[i] h[i-k] = same thing (shift the index)
    # So h^T M h = sum_k w_k/2 * 2 * sum_i h[i]*h[i+k] = sum_k w_k * sum_i h[i]*h[i+k]
    # For k=0: sum_i h[i]^2
    n = len(h)
    total = 0.0
    for k, w in zip(lags, weights):
        if k == 0:
            total += w * np.dot(h, h)
        else:
            total += w * np.dot(h[:n-k], h[k:])
    return total

def _mm_qp_solver(h, lags, weights, n, steps, deadline):
    """Solve max h^T M h s.t. 0<=h<=1, sum(h)=n/2 using projected gradient with momentum."""
    h_cur = h.copy()
    h_prev = h.copy()
    g_prev = 2.0 * _Mh(lags, weights, h_cur)
    beta = 0.9
    step = 1.0 / (2.0 * sum(weights) * max(1, len(lags)))
    for it in range(steps):
        if time.time() > deadline:
            break
        g = 2.0 * _Mh(lags, weights, h_cur)
        grad_mom = g + beta * (g - g_prev)
        h_new = h_cur + step * grad_mom
        h_new = _proj(h_new, n * 0.5)
        # Adaptive step
        obj_new = _quad_M(h_new, lags, weights)
        obj_cur = _quad_M(h_cur, lags, weights)
        if obj_new > obj_cur:
            step = min(step * 1.1, 10.0)
        else:
            step *= 0.7
            h_new = h_cur + step * grad_mom
            h_new = _proj(h_new, n * 0.5)
        h_prev = h_cur
        h_cur = h_new
        g_prev = g
    return h_cur

def _smax_obj(h, alpha, topk):
    c = _correl_full(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl_full(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

import numpy as np
import time

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-9:
        h = _proj(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _Mh(lags, weights, h):
    n = len(h)
    result = np.zeros(n)
    for k, w in zip(lags, weights):
        if k == 0:
            result += w * h
        else:
            result[:n - k] += 0.5 * w * h[k:]
            result[k:] += 0.5 * w * h[:n - k]
    return result

def _quad_M(h, lags, weights):
    n = len(h)
    total = 0.0
    for k, w in zip(lags, weights):
        if k == 0:
            total += w * np.dot(h, h)
        else:
            total += w * np.dot(h[:n - k], h[k:])
    return total

def _mm_qp_solve(h, lags, weights, n, steps, deadline):
    h_cur = h.copy()
    g_prev = 2.0 * _Mh(lags, weights, h_cur)
    beta_mom = 0.9
    L_est = 2.0 * sum(abs(w) for w in weights) * max(1, len(lags))
    step = 1.0 / max(L_est, 0.01)
    obj_prev = _quad_M(h_cur, lags, weights)
    for it in range(steps):
        if time.time() > deadline:
            break
        g = 2.0 * _Mh(lags, weights, h_cur)
        grad_mom = g + beta_mom * (g - g_prev)
        h_new = _proj(h_cur + step * grad_mom, n * 0.5)
        obj_new = _quad_M(h_new, lags, weights)
        if obj_new > obj_prev:
            step = min(step * 1.08, 50.0)
            h_cur = h_new
            obj_prev = obj_new
        else:
            step *= 0.6
            h_new = _proj(h_cur + step * grad_mom, n * 0.5)
            h_cur = h_new
            obj_prev = _quad_M(h_cur, lags, weights)
        g_prev = g
    return h_cur

def _smax_obj(h, alpha, topk):
    c = _correl_full(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl_full(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov_smax(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta_mom = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta_mom * v + (1.0 - beta_mom) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _block_refine(h, rng, deadline, n_blocks=2000):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    blocks = [1, 2, 3, 4, 5, 6, 8, 12]
    for _ in range(n_blocks):
        if time.time() > deadline:
            break
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        t = np.linspace(0, np.pi, bs)
        d = 0.001 * rng.randn() * np.sin(t)
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c:
            h = hn
            c5c = c5n
            if c5n < _eval(bh):
                bh = h.copy()
    return bh, _eval(bh)
