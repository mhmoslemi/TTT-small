    s = np.clip(alpha * (c - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

            h4 = _lbfgs(h4, alpha, 2000, deadline)
        xt = np.linspace(0, 1, n0)
        xs = np.linspace(0, 1, n4)
        h_up = np.interp(xt, xs, h4)
        h_up = _repair(h_up, n0)
        h_up = _lbfgs(h_up, 400000, 3000, deadline)
        c5v = _eval(h_up)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_up.copy()
        h_up = _topk_descent(h_up, 3, 300, deadline)
        c5v = _eval(h_up)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_up.copy()

    # Phase 5: Simulated annealing local search
    if time.time() < deadline - 60:
        h_an, c5_an = _anneal_local(best_h.copy(), 120000, rng, deadline)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an.copy()

    # Phase 6: Final high-alpha L-BFGS
    if time.time() < deadline - 35:
        h = _lbfgs(best_h, 4000000, 6000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 7: Final top-k polish
    if time.time() < deadline - 18:
        h_tk, c5_tk = _topk_descent(best_h.copy(), 3, 500, deadline)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()

    # Phase 8: Ultra-fine 2-point sweep
    if time.time() < deadline - 2:
        n = best_n
        c5c = _eval(best_h)
        bh = best_h.copy()
        best_c2 = c5c
        best_h2 = bh.copy()
        for _ in range(200000):
            if time.time() > deadline - 1:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = 4e-6 * rng.randn()
            ni, nj = bh[i] + d, bh[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                bh[i], bh[j] = ni, nj
                c5n = _eval(bh)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                    if c5n < best_c2:
                        best_c2 = c5n
                        best_h2 = bh.copy()
                else:
                    bh[i], bh[j] = bh[i] - d, bh[j] + d
        if best_c2 < best_c5:
            best_c5 = best_c2
            best_h = best_h2.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
