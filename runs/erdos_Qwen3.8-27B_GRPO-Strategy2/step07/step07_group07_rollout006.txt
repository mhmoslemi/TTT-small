def _lbfgs_pnorm(h, p, pen, maxiter):

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    return _proj_sum(np.clip(h, 0.0, 1.0), n * 0.5)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * dx * (1.0 - h[lag:n])
            g[lag:] -= wk * dx * h[:n - lag]
        else:
            p = -lag
            g[p:] += wk * dx * (1.0 - h[:n - p])
            g[:n - p] -= wk * dx * h[p:]
    return g

def _grad_topk(h, topk):

    g = np.zeros(n)
