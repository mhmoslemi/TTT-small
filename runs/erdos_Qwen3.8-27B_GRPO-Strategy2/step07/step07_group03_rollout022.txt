        g = _topK_grad(h, n, top_K, idx)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = _proj_to_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
        c = _correlate_h(h, n)
    return h

def _coord_cd(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    sweeps = 0
    while time.time() < min(end_t, t_end - 6):
        c = _correlate_h(h, n)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.zeros((top_T, n))
        for t_i, k in enumerate(top_idx):
            sh = k - (n - 1)
            if sh >= 0:
                i1 = np.arange(n - sh)
                grads[t_i, i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    grads[t_i, i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                grads[t_i, i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                grads[t_i, i2] -= h[i2 + sp]
        grads *= dx
        a_vals = c[top_idx]
        improved = False
        n_pairs = n * 3
        for _ in range(n_pairs):
            if time.time() > min(end_t, t_end - 6):
                break
            i = rng.randint(n)
            if rng.random() < 0.5:
                j = (i + 1 + rng.randint(0, max(2, n // 16))) % n
            else:
                j = rng.randint(n)
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(16):
                if f1 < f2:
                    d_hi, x2, f2 = x1, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x2, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-15:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        sweeps += 1
        if improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 25:
            c = _correlate_h(h, n)
            ti = np.argsort(c)[-top_T:]
            g2 = np.zeros(n)
            for k in ti:
                sh = k - (n - 1)
                if sh >= 0:
                    i1 = np.arange(n - sh)
                    g2[i1] += (1.0 - h[i1 + sh])
                    if sh > 0:
                        i2 = np.arange(sh, n)
                        g2[i2] -= h[i2 - sh]
                else:
                    sp = -sh
                    i1 = np.arange(sp, n)
                    g2[i1] += (1.0 - h[i1 - sp])
                    i2 = np.arange(n - sp)
                    g2[i2] -= h[i2 + sp]
            src = np.argsort(g2)[-min(20, n//10):]
            dst = np.argsort(g2)[:min(20, n//10)]
            for a, b in zip(src, dst):
                d = min(0.02, max(0, (h[a] - h[b])) * 0.3)
                if d > 1e-9:
                    h[a] -= d
                    h[b] += d
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _fine_polish(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = _eval(h)
    for _ in range(200000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c5 - 1e-18:
                best_c5 = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    else:
        h = np.full(560, 0.5)
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Adam optimization
    for lr, top_K, steps, frac in [(0.02, 8, 400, 0.12), (0.008, 10, 350, 0.12), (0.003, 12, 250, 0.10)]:
        t_end = min(time.time() + 45, t0 + budget_s * (frac + 0.05))
        h = _adam_opt(h, n, lr, steps, t_end, top_K)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Coordinate descent round 1
    for top_T, frac in [(6, 0.12), (10, 0.10), (15, 0.08)]:
        if time.time() > t0 + budget_s * 0.55:
            break
        h = best_h.copy()
        h, c5v = _coord_cd(h, n, budget_s * frac, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Adam high precision
    if time.time() < t0 + budget_s * 0.62:
        for lr, top_K, steps in [(0.001, 10, 250), (0.0004, 14, 200)]:
            t_end = min(time.time() + 35, t0 + budget_s * 0.60)
            h = _adam_opt(h, n, lr, steps, t_end, top_K)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 4: Coordinate descent round 2
    if time.time() < t0 + budget_s * 0.84:
        for top_T, frac in [(8, 0.06), (12, 0.05), (5, 0.05)]:
            if time.time() > t0 + budget_s * 0.82:
                break
            h = best_h.copy()
            h, c5v = _coord_cd(h, n, budget_s * frac, top_T, rng, t0 + budget_s)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 5: Final fine polish
    if time.time() < t0 + budget_s - 5:
        h = best_h.copy()
        best_h, best_c5 = _fine_polish(h, n, t0 + budget_s - 2, rng)

    # Phase 6: Final coordinate sweep
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        rem = max(1.0, t0 + budget_s - time.time() - 2)
        h, c5v = _coord_cd(h, n, rem, 8, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
