```python
def _lse_obj(h, tau, penalty):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    # Shift for numerical stability
    shifted = (c - c_max) / tau
    log_sum = np.log(np.sum(np.exp(shifted))) + c_max / tau
    val = tau * log_sum + penalty * (h.sum() - n/2)**2
    return val

def _lse_grad(h, tau, penalty):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    shifted = (c - c_max) / tau
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Now compute weighted gradient of c
    g = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15: continue
        sh = (n-1) - k
        if sh >= 0:
            i1 = np.arange(n-sh)
            g[i1] += w * (1-h[i1+sh]) * dx
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= w * h[i2-sh] * dx
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w * (1-h[i1-sp]) * dx
            i2 = np.arange(n-sp)
            g[i2] -= w * h[i2+sp] * dx
    gp = 2.0 * penalty * (h.sum() - n/2)
    return g + gp
```

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = _proj_sum(h, target)
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _lse_val(h, tau, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    sh = (c - cm) / tau
    return cm + tau * np.log(np.sum(np.exp(sh))) + pen * (h.sum() - n * 0.5) ** 2

def _lse_grad(h, tau, pen):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()

    w = ex / ex.sum()
    g = np.zeros(n)

            g[:n-e] -= wk * h[e:e+(n-e)] * dx

                       args=(tau, pen), method='L-BFGS-B',
                       bounds=bnds,

                                'gtol': 1e-14, 'maxls': 50,
                                'maxfun': maxiter * 2})
        return _repair(np.array(res.x), n)
    except Exception:
        return _repair(h.copy(), n)

def _nesterov_lse(h, n_iter, tau, step0, beta, bc_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    for it in range(n_iter):
        if bc_fn():
            break
        g = _lse_grad(hc, tau, 0.0)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        htry = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(htry)
        if ct < best_c - 1e-16:
            hc = htry
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.02, 0.03)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
            if stall > 50:
                break
    return best_h, best_c

            else:

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - n / 2.0) > 1e-10:
        h = _proj_sum(h, n / 2.0)
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _lse_val(h, tau, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    sh = (c - cm) / tau
    return cm + tau * np.log(np.sum(np.exp(sh))) + pen * (h.sum() - 0.5 * n) ** 2

def _lse_grad(h, tau, pen):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    sh = (c - cm) / tau
    ex = np.exp(sh)
    w = ex / ex.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        d = k - (n - 1)
        if d >= 0:
            cnt = n - d
            g[:cnt] += wk * omh[d:d + cnt] * dx
            if d > 0:
                g[d:] -= wk * h[:cnt] * dx
        else:
            e = -d
            g[e:] += wk * omh[:n - e] * dx
            g[:n - e] -= wk * h[e:e + n - e] * dx
    if pen > 0:
        g += 2.0 * pen * (h.sum() - 0.5 * n)
    return g

def _lbfgs(h, tau, pen, maxiter):

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_val, h.copy(), jac=_lse_grad,
                       args=(tau, pen), method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 60, 'maxfun': maxiter * 3})
        return _repair(np.asarray(res.x, dtype=float), n)
    except Exception:
        return _repair(h.copy(), n)

def _nesterov(h, n_iter, tau, step0, beta, bc):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    for it in range(n_iter):
        if bc():
            break
        g = _lse_grad(hc, tau, 0.0)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        htry = _proj_sum(hc - step * v, n * 0.5)
        ct = _eval(htry)
        if ct < best_c - 1e-16:
            hc = htry
            best_c = ct
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.015, 0.025)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9 or stall > 80:
                break
    return best_h, best_c

def _local_search(h, n_iter, rng, bc):

    best_h = hc.copy()

The main difference is the Nesterov gradient. The top-k gradient (previous) is:

where S is the set of top-k lags. This is discontinuous when a lag enters/leaves S.

g = sum_k softmax(c[k]/tau) * grad c[k]

This is smooth and always includes all lags (weighted). For small tau, it approximates the top-1 gradient (just the max lag). For moderate tau, it's a blend.
