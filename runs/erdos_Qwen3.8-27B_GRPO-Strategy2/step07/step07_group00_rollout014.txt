        h = np.clip(h, 0.0, 1.0)

                    h[idx] += a
                    diff -= a

                    a = min(-diff, h[idx])

                    diff += a

        if c5v < best_c5:

            best_h = h.copy()

    # Phase 4: Simulated annealing with geometric cooling
    dl4 = min(hard_deadline, t0 + 880)
    rem4 = max(0.1, dl4 - time.time())
    n_an = int(min(rem4 / 0.0004, 60000))
    sizes = [1, 2, 3, 4, 5, 7, 10]
    bh, bc = _anneal(best_h.copy(), n_an, rng, dl4, 0.004, 1e-10, sizes)
    if bc < best_c5:
        best_c5 = bc
        best_h = bh.copy()

    # Phase 5: Second SLSQP round after annealing
    if time.time() < t0 + budget_s - 60:
        h = _slsqp(best_h, 200000, 12, 200, t0 + budget_s - 60)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Fine 2-point polish
    dl6 = hard_deadline
    rem6 = max(0.1, dl6 - time.time())
    n_p2 = int(min(rem6 / 0.0015, 40000))
    best_h, best_c5 = _polish2(best_h.copy(), n_p2, rng, dl6)

    # Phase 7: 3-point polish if time remains
    rem7 = max(0, hard_deadline - time.time())
    if rem7 > 5:
        n_p3 = int(min(rem7 / 0.002, 20000))
        best_h, best_c5 = _polish3(best_h.copy(), n_p3, rng, hard_deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
