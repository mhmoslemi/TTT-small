                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, topk, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    order = np.argsort(c)
    idx = order[-topk:]
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w = w / w.sum()
    g = np.zeros(n)
    for k in idx:
        wi = w[k]
        if wi < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            m1 = np.arange(sh, n)
            g[m1] += wi * (1.0 - h[m1 - sh])
            m2 = np.arange(0, n - sh)
            g[m2] += -wi * h[m2 + sh]
        else:
            sp = -sh
            m1 = np.arange(0, n - sp)
            g[m1] += wi * (1.0 - h[m1 + sp])
            m2 = np.arange(sp, n)
            g[m2] += -wi * h[m2 - sp]
    return g * dx, c

def _pgd_phase(h, t0, t_end, topk, alpha, step0, mu=0.7):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    it = 0
    while time.time() - t0 < t_end:
        g, c = _topk_grad(h, topk, alpha)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(20):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.04, step0 * 8.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.3, step0 * 8.0)
        it += 1
    return _repair(h, n), cur_c

def _block_opt(h, bsz, t0, t_end, maxit=60):
    n = len(h)
    nb = n // bsz
    def _ob_inner(x, sl, h_ref):
        ht = h_ref.copy()
        ht[sl] = x
        return _eval(ht)
    def _oc(x, bs):
        return x.sum() - bs
    def _oj(x, bs):
        return np.ones(len(x))
    for _pass in range(6):
        if time.time() - t0 > t_end:
            break
        for b in range(nb):
            if time.time() - t0 > t_end:
                break
            sl = slice(b * bsz, (b + 1) * bsz)
            x0 = h[sl].copy()
            bsum = x0.sum()
            try:
                res = minimize(_ob_inner, x0, args=(sl, h),
                             method='SLSQP', bounds=[(0,1)]*bsz,
                             constraints=[{'type':'eq','fun':_oc,'jac':_oj,'args':(bsum,)}],
                             options={'maxiter':maxit,'ftol':1e-16})
                if res.fun < _eval(h) + 1e-18:
                    h[sl] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _targeted_local(h, t0, t_end, topk=4, rng=None):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    if rng is None:
        rng = np.random.RandomState(7)
    n_eval = 0
    while time.time() - t0 < t_end:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-topk:]
        # Build position bias from top lags
        bias = np.zeros(n)
        for k in top_idx:
            sh = k - (n - 1)
            if sh >= 0:
                contrib = h[sh:] * (1.0 - h[:n-sh])
                bias[sh:] += contrib
                bias[:n-sh] += contrib
            else:
                sp = -sh
                contrib = h[:n-sp] * (1.0 - h[sp:])
                bias[:n-sp] += contrib
                bias[sp:] += contrib
        # Sample positions weighted by bias
        bsum = bias.sum()
        if bsum > 1e-15:
            pb = bias / bsum
        else:
            pb = np.full(n, 1.0/n)
        
        # 2-point targeted move
        for _round in range(3):
            if time.time() - t0 > t_end:
                break
            i = rng.choice(n, p=pb)
            # j near i to affect nearby lags
            j = (i + 1 + rng.randint(0, 6)) % n
            d = 3e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                n_eval += 1
                if c5n < c5:
                    c5 = c5n
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        # 3-point move
        if time.time() - t0 > t_end:
            break
        i = rng.choice(n, p=pb)
        j = (i + 1 + rng.randint(0, 4)) % n
        k = (i + 4 + rng.randint(0, 8)) % n
        d1 = 2e-6 * rng.randn()
        d2 = 2e-6 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            n_eval += 1
            if c5n < c5:
                c5 = c5n
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return best_h, best_c5

def _uniform_local(h, t0, t_end, step, rng=None):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    if rng is None:
        rng = np.random.RandomState(99)
    while time.time() - t0 < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: PGD at multiple scales
    pg_configs = [
        (3, 500, 0.003, 400),
        (4, 3000, 0.002, 400),
        (5, 15000, 0.0015, 350),
        (6, 80000, 0.001, 300),
    ]
    for topk, alpha, s0, teu in pg_configs:
        t_end = t0 + teu
        if _el() > 180:
            break
        h_new, c_new = _pgd_phase(h.copy(), t0, t_end, topk, alpha, s0, mu=0.65)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 2: Block coordinate descent
    if _el() < budget_s - 400:
        for bsz in [8, 12, 6]:
            t_end = t0 + min(_el() + 80, budget_s - 300)
            h_new = _block_opt(best_h.copy(), bsz, t0, t_end, maxit=50)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 3: More PGD with higher alpha
    if _el() < budget_s - 350:
        for topk, alpha, s0 in [(4, 50000, 0.0008), (5, 200000, 0.0005)]:
            if _el() > budget_s - 250:
                break
            t_end = t0 + min(_el() + 80, budget_s - 200)
            h_new, c_new = _pgd_phase(best_h.copy(), t0, t_end, topk, alpha, s0, mu=0.7)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 4: Targeted local search (main workhorse)
    if _el() < budget_s - 50:
        t_end = t0 + min(_el() + (budget_s - 50 - _el()) * 0.6, budget_s - 60)
        best_h, best_c5 = _targeted_local(best_h.copy(), t0, t_end, topk=4, rng=rng)

    # Phase 5: Uniform fine local search
    if _el() < budget_s - 8:
        t_end = t0 + min(_el() + (budget_s - 8 - _el()) * 0.7, budget_s - 10)
        best_h, best_c5 = _uniform_local(best_h.copy(), t0, t_end, 1.5e-7, rng=rng)

    # Phase 6: Ultra-fine polish
    if _el() < budget_s - 1:
        t_end = t0 + budget_s - 1
        best_h, best_c5 = _uniform_local(best_h.copy(), t0, t_end, 5e-8, rng=rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
