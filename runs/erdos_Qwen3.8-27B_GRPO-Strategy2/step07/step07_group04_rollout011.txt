    # Phase 2: Multi-scale escape (n/4)
    n_lo = n // 4
    if n_lo >= 40:
        h_lo = _down(best_h, 4)
        h_lo, f = _adam(h_lo, 2, 10, 8000, 0.005, t0, t0 + 25)
        h_hi = _up(h_lo, 4)
        if len(h_hi) < n:
            h_hi = np.concatenate([h_hi, h_lo[-(n - len(h_hi)):]])
        h_hi = h_hi[:n]
        h_hi, f = _adam(h_hi, 3, 20, 5000, 0.002, t0, t0 + 40)
        _try(h_hi, "p2")

    # Phase 3: Adam medium (k=5..60)
    h, f = _adam(best_h.copy(), 5, 60, 30000, 0.0015, t0, t0 + 55)
    _try(h, "p3")

    # Phase 4: SLSQP moderate
    if time.time() - t0 < budget_s - 200:
        hc = _slsqp_opt(best_h.copy(), 20000, 600, 1e-17)
        _try(hc, "p4a")
    if time.time() - t0 < budget_s - 150:
        hc = _slsqp_opt(best_h.copy(), 80000, 400, 1e-17)
        _try(hc, "p4b")

    # Phase 5: Adam high-k (k=20..200)
    h, f = _adam(best_h.copy(), 20, 200, 40000, 0.0008, t0, t0 + 90)
    _try(h, "p5")

    # Phase 6: SLSQP high alpha
    if time.time() - t0 < budget_s - 120:
        hc = _slsqp_opt(best_h.copy(), 300000, 500, 1e-18)
        _try(hc, "p6a")
    if time.time() - t0 < budget_s - 80:
        hc = _slsqp_opt(best_h.copy(), 1000000, 400, 1e-18)
        _try(hc, "p6b")

    # Phase 7: Adam ultra-fine (k=100..500)
    h, f = _adam(best_h.copy(), 100, 500, 50000, 0.0003, t0, t0 + 130)
    _try(h, "p7")

    # Phase 8: Simulated annealing local search
    elapsed = time.time() - t0
    if elapsed < budget_s - 40:
        n_sa = min(int((budget_s - 40 - elapsed) * 2.5), 60000)
        best_h, best_c5 = _local_sa(best_h, rng, t0, t0 + budget_s - 30, n_sa)

    # Phase 9: Ultra-fine 2-point polish
    elapsed = time.time() - t0
    if elapsed < budget_s - 3:
        n_fine = min(int((budget_s - 3 - elapsed) * 6), 50000)
        best_h, best_c5 = _local_fine(best_h, rng, t0, t0 + budget_s - 1, n_fine)

    # Final: one more Adam polish with very small lr
    elapsed = time.time() - t0
    if elapsed < budget_s - 2:
        h, f = _adam(best_h.copy(), 300, 800, 5000, 0.0001, t0, t0 + budget_s - 1)
        _try(h, "final")

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
