def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _build_grads(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        sh = k - (n - 1)
        if sh >= 0:
            G[k, :n - sh] += (1.0 - h[sh:]) * dx
            G[k, sh:] -= h[:n - sh] * dx
        else:
            sp = -sh
            G[k, sp:] += (1.0 - h[:n - sp]) * dx
            G[k, :n - sp] -= h[sp:] * dx
    return G

def _slp_step(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _build_grads(h, n)
    A_ub = np.hstack((G, -np.ones((m, 1))))
    b_ub = G @ h - c
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    obj = np.zeros(n + 1)
    obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n].copy()
    return h.copy()

def _slp_loop(h, n, max_iter, t_end):
    best_h = h.copy()
    best_c5 = _eval(h)
    cur_c5 = best_c5
    for it in range(max_iter):
        if time.time() > t_end:
            break
        h_new = _slp_step(h, n)
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= cur_c5 + 1e-12:
            h = h_new
            cur_c5 = c5_new
        else:
            accepted = False
            for lam in [0.6, 0.4, 0.25]:
                hd = (1.0 - lam) * h + lam * h_new
                hd = _repair(hd, n)
                cd = _eval(hd)
                if cd < cur_c5 - 1e-13:
                    h = hd
                    cur_c5 = cd
                    if cd < best_c5:
                        best_c5 = cd
                        best_h = hd.copy()
                    accepted = True
                    break
            if not accepted:
                if it > 5:
                    break
                h = h_new
                cur_c5 = c5_new
    return best_h, best_c5

def _cd_pair_grad(h, i, j, top_idx, n):
    dx = 2.0 / n
    b = np.zeros(len(top_idx))
    for ti in range(len(top_idx)):
        sh = top_idx[ti] - (n - 1)
        if sh >= 0:
            bi = (1.0 - h[i + sh]) * dx if i + sh < n else 0.0
            bi -= h[i - sh] * dx if sh > 0 and i - sh >= 0 else 0.0
            bj = (1.0 - h[j + sh]) * dx if j + sh < n else 0.0
            bj -= h[j - sh] * dx if sh > 0 and j - sh >= 0 else 0.0
        else:
            sp = -sh
            bi = (1.0 - h[i - sp]) * dx if i - sp >= 0 else 0.0
            bi -= h[i + sp] * dx if i + sp < n else 0.0
            bj = (1.0 - h[j - sp]) * dx if j - sp >= 0 else 0.0
            bj -= h[j + sp] * dx if j + sp < n else 0.0
        b[ti] = bi - bj
    return b

def _coord_descent(h, n, dur, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = min(time.time() + dur, t_end - 4)
    stall = 0
    while time.time() < end_t:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        improved = False
        for _ in range(n * 3):
            if time.time() >= end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(3, n // 8))) % n
            if i == j:
                continue
            b = _cd_pair_grad(h, i, j, top_idx, n)
            d_lo = max(-h[i], h[j] - 1.0) + 1e-13
            d_hi = min(1.0 - h[i], h[j]) - 1e-13
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(28):
                if f1 < f2:
                    d_hi, x2, f2 = x2, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) / 2.0
            if abs(d_opt) > 1e-13:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            h = _repair(h, n)
            nc5 = _eval(h)
            if nc5 < best_c5:
                best_c5 = nc5
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 25:
            h += 4e-5 * rng.randn(n)
            h = _proj_sum(h, n * 0.5)
            stall = 0
    return h, best_c5

def _nesterov_fine(h, n, t_end, alpha=200000, steps=200, step0=0.003):
    dx = 2.0 / n
    mu = 0.82
    step = step0
    v = np.zeros(n)
    def _obj_sm(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    f_cur = _obj_sm(h)
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = k - (n - 1)
            wk = w[k] * dx
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                g[sh:] -= wk * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _obj_sm(h_new)
        if f_new < f_cur:
            h = h_new
            f_cur = f_new
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-11:
                break
            h = _proj_sum(h_prev - step * v, n * 0.5)
            f_cur = _obj_sm(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: SLP (primary new method)
    t_end1 = t0 + min(250, budget_s * 0.25)
    h, c5 = _slp_loop(h, n, 500, t_end1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Nesterov fine-tune
    if time.time() < t0 + budget_s * 0.35:
        t_end2 = t0 + budget_s * 0.35
        h = best_h.copy()
        h = _nesterov_fine(h, n, t_end2, alpha=300000, steps=300, step0=0.002)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Coordinate descent
    if time.time() < t0 + budget_s * 0.5:
        t_end3 = t0 + budget_s * 0.5
        h, c5 = _coord_descent(best_h.copy(), n, 120, 10, rng, t_end3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Perturbed SLP restarts
    if time.time() < t0 + budget_s * 0.65:
        t_end4 = t0 + budget_s * 0.65
        for pert in [8e-4, 4e-4, 2e-4, 1e-4]:
            if time.time() > t_end4:
                break
            hp = best_h.copy() + pert * rng.randn(n)
            hp = _proj_sum(hp, n * 0.5)
            _, c5 = _slp_loop(hp, n, 150, t_end4)
            if c5 < best_c5:
                best_c5 = c5
                best_h = _repair(hp, n)

    # Phase 5: Extended CD with varying top_T
    if time.time() < t0 + budget_s * 0.8:
        t_end5 = t0 + budget_s * 0.8
        for top in [12, 8, 6, 10]:
            if time.time() > t_end5:
                break
            h, c5 = _coord_descent(best_h.copy(), n, 100, top, rng, t_end5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Final SLP
    if time.time() < t0 + budget_s * 0.9:
        t_end6 = t0 + budget_s * 0.9
        h, c5 = _slp_loop(best_h.copy(), n, 200, t_end6)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Final CD polish
    if time.time() < t0 + budget_s - 5:
        rem = (t0 + budget_s - 4) - time.time()
        if rem > 10:
            h, c5 = _coord_descent(best_h.copy(), n, rem, 8, rng, t0 + budget_s - 3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
