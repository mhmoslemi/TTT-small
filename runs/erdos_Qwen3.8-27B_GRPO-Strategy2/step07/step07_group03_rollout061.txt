import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correlate_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _smooth_obj(h, n, alpha):
    c = _correlate_vals(h, n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = _correlate_vals(h, n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    # Gradient: for each shift k, contribution to h[j]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            # (1-h[j+sh]) for j=0..n-1-sh
            grad[:n - sh] += wk * (1.0 - h[sh:])
            # -h[j-sh] for j=sh..n-1
            grad[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            # (1-h[j-sp]) for j=sp..n-1
            grad[sp:] += wk * (1.0 - h[:n - sp])
            # -h[j+sp] for j=0..n-sp-1
            grad[:n - sp] -= wk * h[sp:]
    return grad * dx

    """Nesterov accelerated proximal gradient with multiple alphas."""
    best_h = h.copy()
    best_f = float('inf')
    for alpha in alphas:
        phase_end = min(t_end, time.time() + (t_end - time.time()) / len(alphas))
        h_cur = best_h.copy()
        f = _smooth_obj(h_cur, n, alpha)
        v = np.zeros(n)
        mu = 0.8
        step = step0
        for it in range(5000):
            if time.time() > phase_end:
                break
            g = _smooth_grad(h_cur, n, alpha)
            v = mu * v + (1.0 - mu) * g
            h_new = _proj_to_sum(h_cur - step * v, n * 0.5)
            f_new = _smooth_obj(h_new, n, alpha)
            if f_new < f:
                h_cur = h_new
                f = f_new
                step = min(step * 1.05, step0 * 20)
                if f < best_f:
                    best_f = f
                    best_h = h_cur.copy()
            else:

                    break
                h_new = _proj_to_sum(h_cur - step * v, n * 0.5)
                f = _smooth_obj(h_new, n, alpha)
                h_cur = h_new
    return best_h, best_f

def _coord_descent_fast(h, n, duration_s, top_T, rng, t_end):
    """Active-set coordinate descent with pair swaps."""
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = min(time.time() + duration_s, t_end - 5)
    no_improve = 0

        # Precompute gradients for top k values (without dx, we'll multiply later)
        # Actually for the pair optimization, we need d(c_k)/d(h[i]-h[j])
        # For shift s: d(c_k)/d(h[i]) - d(c_k)/d(h[j])
        # = [(1-h[i+s])1_{valid} - h[i-s]1_{valid}] - [(1-h[j+s])1_{valid} - h[j-s]1_{valid}]

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _nesterov(h0, n, alphas, dur_each, t_end, step0=0.005):
    best = h0.copy()
    best_c = _eval(h0)
    for alpha in alphas:
        end_p = min(time.time() + dur_each, t_end)
        if time.time() > end_p:
            break
        hc = best.copy()
        f = _sobj(hc, n, alpha)
        v = np.zeros(n)
        step = step0
        mu = 0.82
        stall = 0
        for it in range(10000):
            if time.time() > end_p:
                break
            g = _sgrad(hc, n, alpha)
            v = mu * v + (1.0 - mu) * g
            hn = _proj(hc - step * v, n * 0.5)
            fn = _sobj(hn, n, alpha)
            if fn < f - 1e-16:
                hc, f = hn, fn
                step = min(step * 1.04, step0 * 30)
                stall = 0
                c5 = _eval(hc)
                if c5 < best_c:
                    best_c = c5
                    best = hn.copy()
            else:
                step *= 0.35
                stall += 1
                if step < 1e-12 or stall > 200:
                    break
                hn = _proj(hc - step * v, n * 0.5)
                hc = hn
                f = _sobj(hc, n, alpha)
    return best, best_c

def _coord(h, n, dur, T, rng, t_end):
    dx = 2.0 / n
    gr = 0.6180339887
    best_c = _eval(h)
    end_t = min(time.time() + dur, t_end - 5)
    stall = 0
    while time.time() < end_t:
        c = _corr(h)
        top = np.argsort(c)[-T:]
        av = c[top]
        improved = False
        for _ in range(min(3000, n * 4)):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 10))) % n
            if i == j:
                continue
            b = np.zeros(T)
            for t in range(T):
                sh = top[t] - (n - 1)
                di = (1.0 - h[i + sh]) if (0 <= i + sh < n) else 0.0
                di -= h[i - sh] if (0 <= i - sh < n) else 0.0
                dj = (1.0 - h[j + sh]) if (0 <= j + sh < n) else 0.0
                dj -= h[j - sh] if (0 <= j - sh < n) else 0.0
                b[t] = (di - dj) * dx
            dlo = max(-h[i], h[j] - 1.0) + 1e-13
            dhi = min(1.0 - h[i], h[j]) - 1e-13
            if dlo >= dhi:
                continue
            x1 = dhi - gr * (dhi - dlo)
            x2 = dlo + gr * (dhi - dlo)
            f1 = np.max(av + b * x1)
            f2 = np.max(av + b * x2)
            for _ in range(20):
                if f1 < f2:
                    dhi, x2, f2 = x2, x1, f1
                    x1 = dhi - gr * (dhi - dlo)
                    f1 = np.max(av + b * x1)
                else:
                    dlo, x1, f1 = x1, x2, f2
                    x2 = dlo + gr * (dhi - dlo)
                    f2 = np.max(av + b * x2)
            d = (dlo + dhi) * 0.5
            if abs(d) > 1e-14:
                h[i] += d
                h[j] -= d
                improved = True
        nc = _eval(h)
        if nc < best_c:
            best_c = nc
            stall = 0
        else:
            stall += 1
        if stall > 50:
            h += rng.randn(n) * 5e-5
            h = _proj(h, n * 0.5)
            stall = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj(h, n * 0.5)
    return h, best_c

def _local(h, ntr, rng, t_end):
    n = len(h)
    bc = _eval(h)
    bh = h.copy()
    for _ in range(ntr):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < bc - 1e-18:
                bc = c
                bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _resample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return _repair(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov from initial (low alpha → high alpha)
    h1, c1 = _nesterov(best_h, n, [3000, 12000], 10, t_end, 0.005)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
    h1, c1 = _nesterov(best_h, n, [40000, 150000], 10, t_end, 0.004)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
    h1, c1 = _nesterov(best_h, n, [400000, 1500000], 10, t_end, 0.003)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Coordinate descent rounds
    for T in [3, 5, 8, 5, 3, 6, 4]:
        if time.time() > t_end - 80:
            break
        h1, c1 = _coord(best_h.copy(), n, 18, T, rng, t_end)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 3: Perturbation + re-optimize (escape local opt)
    if time.time() < t_end - 60:
        for _ in range(3):
            if time.time() > t_end - 50:
                break
            hp = best_h.copy() + rng.randn(n) * 0.02
            hp = _proj(hp, n * 0.5)
            hp2, cp2 = _nesterov(hp, n, [200000, 800000], 8, t_end, 0.003)
            if cp2 < best_c5:
                best_c5 = cp2
                best_h = hp2.copy()
            h1, c1 = _coord(hp2.copy(), n, 8, 5, rng, t_end)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()

    # Phase 4: High-precision Nesterov
    if time.time() < t_end - 50:
        h1, c1 = _nesterov(best_h, n, [300000, 1500000], 12, t_end, 0.002)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 5: Try binary start (different topology)
    if time.time() < t_end - 80:
        hb = _binary_start(n, np.random.RandomState(seed + 7))
        hb, cb = _coord(hb, n, 25, 8, rng, t_end)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        hb2, cb2 = _nesterov(_proj((hb + 0.5) / 2 + rng.randn(n) * 0.1, n), n,
                             [100000, 500000], 15, t_end, 0.004)
        if cb2 < best_c5:
            best_c5 = cb2
            best_h = hb2.copy()
        hb3, cb3 = _coord(best_h.copy(), n, 15, 6, rng, t_end)
        if cb3 < best_c5:
            best_c5 = cb3
            best_h = hb3.copy()

    # Phase 6: Final coordinate polish
    if time.time() < t_end - 25:
        for T in [4, 6, 3, 5]:
            if time.time() > t_end - 15:
                break
            h1, c1 = _coord(best_h.copy(), n, 10, T, rng, t_end)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()

    # Phase 7: Ultra-fine local search
    if time.time() < t_end - 8:
        rem = t_end - time.time() - 5
        nt = min(int(rem / 0.003), 50000)
        if nt > 200:
            h1, c1 = _local(best_h.copy(), nt, rng, t_end - 3)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()

    # Phase 8: Absolute final coord
    if time.time() < t_end - 3:
        rem = max(1, t_end - time.time() - 2)
        h1, c1 = _coord(best_h.copy(), n, rem, 5, rng, t_end)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
