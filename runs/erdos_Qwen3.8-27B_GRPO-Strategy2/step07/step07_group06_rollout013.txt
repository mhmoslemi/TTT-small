def _repair(h, n):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2 exactly."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    lo, hi = -2.0, 2.0

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    """One subgradient step using precomputed correlation."""
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Get top_k active constraints
    top_idx = np.argsort(c_full)[-top_k:]
    
    # Compute subgradient: average gradient of top_k f_k's
    g = np.zeros(n)
    for k in top_idx:
        shift = k - (n - 1)
        if shift >= 0:
            # f_k = sum_i h[i] * (1-h[i-shift]) for valid i
            # df_k/dh[i] = (1-h[i-shift]) - h[i+shift] (with bounds)
            idx_hi = np.arange(n)
            # 1-h[i-shift] valid when i-shift >= 0, i.e., i >= shift
            mask1 = idx_hi >= shift
            g[mask1] += (1.0 - h[np.clip(idx_hi[mask1] - shift, 0, n-1)])
            # -h[i+shift] valid when i+shift < n, i.e., i < n-shift
            mask2 = idx_hi < n - shift
            g[mask2] -= h[np.clip(idx_hi[mask2] + shift, 0, n-1)]
        else:
            sp = -shift
            # f_k = sum_i h[i+sp] * (1-h[i]) for valid i (equiv to shift = -sp)
            idx_hi = np.arange(n)
            mask1 = idx_hi + sp < n  # h[i+sp] valid
            g[mask1] += (1.0 - h[np.clip(idx_hi[mask1] - sp, 0, n-1)])
            mask2 = idx_hi - sp >= 0  # h[i-sp] valid
            g[mask2] -= h[np.clip(idx_hi[mask2] + sp, 0, n-1)]
    
    g *= dx / top_k
    return h - lr * g, c_full

def _subgrad_loop(h, rng, n_iter, top_k, lr_init, deadline):
    """Iterative subgradient descent."""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    c_cur = best_c
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Compute current objective
        c_full = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        c_cur = c_full.max()
        
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()
        
        # Get active constraints
        top_idx = np.argsort(c_full)[-top_k:]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in top_idx:
            shift = k - (n - 1)
            if shift >= 0:
                for i in range(n):
                    val = 0.0
                    if i - shift >= 0:
                        val += (1.0 - h[i - shift])
                    if i + shift < n:
                        val -= h[i + shift]
                    g[i] += val
            else:
                sp = -shift
                for i in range(n):
                    val = 0.0
                    if i - sp >= 0:
                        val += (1.0 - h[i - sp])
                    if i + sp < n:
                        val -= h[i + sp]
                    g[i] += val
        
        g *= (2.0 / n) / top_k
        
        # Projected step
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = lr_init / (1.0 + 0.1 * it)
        for mult in [1.0, 0.5, 0.25, 0.1]:
            h_new = _proj_to_sum(h - mult * step * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < c_cur:
                h = h_new
                break
        
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

    return h

def _eval(h):
