import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max())

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in top_k:
        sh = (n - 1) - k
        wk = dx / k_top
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g, float(c.max())

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / alpha

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sum_fun(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_call(h, alpha, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    res = minimize(_smooth_obj, h, args=(n, alpha), jac=_smooth_grad,
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 600, 'ftol': 1e-16})
    hr = _repair(res.x, n)
    return hr, _eval(hr)

    c5c = _eval(h_c)
    steps = list(steps_init)
    stall = 0
    for _ in range(max_iter):
        if t_check():

        g, _ = _grad_topk(h_c, k_top)
        improved = False
        for s in steps:

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max())

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_topk(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        wk = dx / k_top
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _sl_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.exp(sc - sc.max()).sum()) / alpha

def _sl_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

    sc = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.exp(sc - sc.max()).sum()) / a

def _sg(x, n, a):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(a * (c - m), -500, 500)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - x[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * x[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - x[sp:])
            g[sp:] -= wk * x[:n - sp]
    return g

def _sf(x, n):
    return x.sum() - n * 0.5

def _sj(x, n):
    return np.ones(n)

def _gp(h, t0, dl, mi, kt, si):
    n = len(h)
    hc = h.copy()
    c5 = _ev(hc)
    st = list(si)
    sl = 0
    for _ in range(mi):
        if time.time() - t0 > dl:
            break
        g = _gt(hc, kt)
        imp = False
        for s in st:
            hn = _pr(hc - s * g, n * 0.5)
            cn = _ev(hn)
            if cn < c5 - 1e-15:
                hc = hn; c5 = cn; imp = True
                break
        if imp:
            sl = 0
        else:
            st = [s * 0.4 for s in st]
            sl += 1
            if sl > 12:
                break
    return hc, c5

def _l2(h, nt, rng, c5, t0, dl, stp):
    n = len(h)
    bh, bc = h.copy(), c5
    for _ in range(nt):
        if time.time() - t0 > dl:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = stp * rng.randn()
        oi, oj = h[i], h[j]
        ni, nj = oi + d, oj - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _ev(h)
            if cn < c5 - 1e-18:
                c5 = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j] = oi, oj
    return bh, bc

def _l3(h, nt, rng, c5, t0, dl, stp):
    n = len(h)
    bh, bc = h.copy(), c5
    for _ in range(nt):
        if time.time() - t0 > dl:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 5)) % n
        d1, d2 = stp * rng.randn(), stp * rng.randn()
        oi, oj, ok = h[i], h[j], h[k]
        ni, nj, nk = oi + d1, oj + d2, ok - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _ev(h)
            if cn < c5 - 1e-18:
                c5 = cn
                if cn < bc:
                    bc, bh = cn, h.copy()
            else:
                h[i], h[j], h[k] = oi, oj, ok
    return bh, bc

def _an(h, nt, rng, c5, t0, dl, T0, Te):
    n = len(h)
    bh, bc = h.copy(), c5
    hc, cc = h.copy(), c5
    for it in range(nt):
        if time.time() - t0 > dl:
            break
        T = T0 * (Te / T0) ** (float(it) / float(max(nt, 1)))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 4e-6 * rng.randn()
        oi, oj = hc[i], hc[j]
        ni, nj = oi + d, oj - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[i], hc[j] = ni, nj
            cn = _ev(hc)
            dd = cn - cc
            if dd < 0 or (T > 1e-18 and rng.random() < np.exp(-dd / T)):
                cc = cn
                if cn < bc:
                    bc, bh = cn, hc.copy()
            else:
                hc[i], hc[j] = oi, oj
    return bh, bc

def _sq(h, a, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n,)}]
    res = minimize(_so, h, args=(n, a), jac=_sg,
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-16})
    hr = _repair(res.x, n)
    return hr, _ev(hr)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = budget_s - 15
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh, bc5 = h.copy(), _ev(h)

    for kt in [3, 5, 7, 4, 6, 3, 5, 8, 4, 5]:
        if time.time() - t0 > dl:
            break
        h2, c2 = _gp(bh.copy(), t0, dl, 500, kt, [0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
        if c2 < bc5:
            bc5, bh = c2, h2

    if time.time() - t0 < budget_s - 150:
        for a in [3000, 8000, 20000, 50000, 100000]:
            if time.time() - t0 > dl:
                break
            try:
                h2, c2 = _sq(bh.copy(), a, n0)
                if c2 < bc5:
                    bc5, bh = c2, h2
            except Exception:
                pass

    if time.time() - t0 < budget_s - 100:
        for kt in [3, 5, 3]:
            if time.time() - t0 > dl:
                break
            h2, c2 = _gp(bh.copy(), t0, dl, 300, kt, [0.02, 0.01, 0.005, 0.002, 0.001])
            if c2 < bc5:
                bc5, bh = c2, h2

    if time.time() - t0 < budget_s - 80:
        for sh in [1, 2, 1]:
            if time.time() - t0 > dl:
                break
            hs = np.roll(bh, sh).copy() + 0.001 * rng.randn(n0)
            hs = _repair(hs, n0)
            h2, c2 = _gp(hs, t0, dl, 300, 3, [0.03, 0.01, 0.005, 0.002])
            if c2 < bc5:
                bc5, bh = c2, h2

    rem = budget_s - 200 - (time.time() - t0)
    if rem > 10:
        bh, bc5 = _l3(bh, min(int(rem * 200), 300000), rng, bc5, t0, dl, 7e-6)

    rem = budget_s - 120 - (time.time() - t0)
    if rem > 10:
        bh, bc5 = _an(bh, min(int(rem * 200), 300000), rng, bc5, t0, dl, 1e-7, 1e-13)

    rem = budget_s - 60 - (time.time() - t0)
    if rem > 10:
        bh, bc5 = _l2(bh, min(int(rem * 200), 400000), rng, bc5, t0, dl, 5e-6)

    rem = budget_s - 5 - (time.time() - t0)
    if rem > 5:
        bh, bc5 = _l2(bh, min(int(rem * 200), 300000), rng, bc5, t0, dl, 2e-6)

    bh = _repair(bh, len(bh))
    bc5 = _ev(bh)
    return (bh, bc5, len(bh))
