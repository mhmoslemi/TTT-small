    t_phase2_end = t0 + budget_s * 0.72
    h = best_h.copy()
    best_h, best_c5 = _sa_phase(h, n, rng, time.time(), t_phase2_end)
    if time.time() < t_phase2_end - 2:
        h = best_h + rng.randn(n) * 0.005
        h = _proj_sum(h, n * 0.5)
        h2, c5v = _sa_phase(h, n, rng, time.time(), t_phase2_end)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2
    t_cd_end = t0 + budget_s * 0.9
    for top_T in [4, 6, 3]:
        if time.time() >= t_cd_end:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, min(time.time() + 25, t_cd_end), top_T, rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    if time.time() < t0 + budget_s - 5:
        h = best_h.copy()
        h = _nesterov(h, 2000000, time.time(), t0 + budget_s - 5)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, t0 + budget_s - 3, 5, rng)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
