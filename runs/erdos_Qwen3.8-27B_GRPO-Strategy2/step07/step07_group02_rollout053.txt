    return float(np.max(_cvals(h)))

def _corr_grad(k, x, n):
    """Gradient of c[k] with respect to x"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        np_ = n - sh
        g[:np_] += dx * (1.0 - x[sh:])
        if sh > 0:

    else:

        g[sp:] += dx * (1.0 - x[:-sp])

    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

        bh, bc = _block_search(bh, rng, 15000, dl)

    # Phase 8: 2-point search with larger initial step
    if time.time() < dl - 50:
        bh, bc = _local_2pt(bh, rng, 40000, dl, 8e-5)

    # Phase 9: Final L-BFGS polish
    if time.time() < dl - 35:
        h2 = _lbfgs(bh, 3000000, 2000, dl)
        c2 = _eval(h2)
        if c2 < bc:
            bh, bc = h2.copy(), c2

    # Phase 10: Ultra-fine 2-point search
    if time.time() < dl - 8:
        bh, bc = _local_2pt(bh, rng, 70000, dl, 2e-6)

    # Phase 11: Final shuffle + block polish
    if time.time() < dl - 4:
        bh, bc = _shuffle_search(bh, rng, 8000, dl)
        bh, bc = _local_2pt(bh, rng, 30000, dl, 1e-6)

    # Try half-resolution as alternative
    if n0 >= 200 and time.time() < dl - 3:
        nh = n0 // 2
        hh = _repair(_down(bh, 2), nh)
        hh = _lbfgs(hh, 100000, 1500, dl)
        ch = _eval(hh)
        if ch < bc:
            bh, bc, n0 = hh, ch, nh

    bh = _repair(bh, n0)
    bc = _eval(bh)
    return (bh, bc, n0)
