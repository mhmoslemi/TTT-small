            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(24):
                if f1 < f2:
                    d_hi = x2; x2 = x1; f2 = f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo = x1; x1 = x2; f1 = f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
        new_c5 = _eval(h)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 35:
            h += 1e-4 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _fine_search(h, n_iter, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        scale = max(1e-8, 1e-5 * (1.0 - it / max(1, n_iter)))
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)
    t_dead = t0 + budget_s

    # Phase 1: Direct subgradient on top lags
    h = _subgrad_loop(h, t0 + budget_s * 0.12, 6, 1500, 0.004, rng)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 2: Nesterov smooth (different alphas than parent)
    for alpha in [8000, 40000, 200000, 1000000]:
        t_end = min(time.time() + 22, t0 + budget_s * 0.32)
        h = _nesterov(best_h.copy(), alpha, 250, 0.003, t_end)
        h2 = _nesterov(h, alpha * 2, 150, 0.002, min(time.time() + 15, t0 + budget_s * 0.34))
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 3: Subgradient refinement with tighter focus
    if time.time() < t0 + budget_s * 0.45:
        h = _subgrad_loop(best_h.copy(), t0 + budget_s * 0.44, 4, 3000, 0.002, rng)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Coordinate descent with pair swaps (multiple top_T)
    for top_T in [3, 5, 8]:
        if time.time() > t0 + budget_s * 0.65:
            break
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, budget_s * 0.07, top_T, rng, t_dead)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: High-alpha Nesterov for sharpness
    if time.time() < t0 + budget_s * 0.78:
        for alpha in [3000000, 8000000]:
            t_end = min(time.time() + 12, t0 + budget_s * 0.77)
            h = _nesterov(best_h.copy(), alpha, 80, 0.0008, t_end)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 6: More coordinate descent
    if time.time() < t0 + budget_s * 0.88:
        for top_T in [4, 6, 3]:
            if time.time() > t0 + budget_s * 0.87:
                break
            h = best_h.copy()
            h, c5v = _coord_descent(h, n, budget_s * 0.035, top_T, rng, t_dead)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 7: Ultra-fine local search
    if time.time() < t_dead - 8:
        remaining = (t_dead - 8) - time.time()
        n_tries = min(int(remaining / 0.003), 60000)
        if n_tries > 200:
            h = best_h.copy()
            best_h, best_c5 = _fine_search(h, n_tries, rng, t_dead - 4)

    # Phase 8: Final coordinate polish
    if time.time() < t_dead - 3:
        h = best_h.copy()
        h, c5v = _coord_descent(h, n, max(0.5, t_dead - time.time() - 3), 5, rng, t_dead)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
