def _repair(h, n):

    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

            break
        h2 = best_h.copy()
        h2, c5v = _coord_descent(h2, n, (ph4_end - time.time()) / 3, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 5: Local polish (10% budget)
    if time.time() < t0 + budget_s - 5:
        remaining = (t0 + budget_s) - time.time() - 3
        h2 = best_h.copy()
        h2, c5v = _local_polish(h2, n, remaining, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
