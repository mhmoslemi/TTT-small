    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max weights
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    w /= w.sum()
    # Weighted gradient
    g = np.zeros(n)
    for idx in range(len(c)):

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _wgrad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    w = np.exp(alpha * (c - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += wi * (1.0 - h[:n - k])
            g[:n - k] -= wi * h[k:]
        else:
            m = -k
            g[:n - m] += wi * (1.0 - h[m:])
            g[m:] -= wi * h[:n - m]
    return g * dx

def _smooth_descend(h, rng, deadline, alpha_init=25.0):
    n = len(h)
    best_h, best_c = h.copy(), _c5(h)
    v = np.zeros(n)
    mu = 0.82
    lr = 0.04
    alpha = alpha_init
    stag = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        g = _wgrad(h, alpha)
        v = mu * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _c5(h_new)
        if c_new < best_c - 1e-15:
            best_c, best_h = c_new, h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag > 80:
                lr *= 0.75
                v *= 0.3
                alpha = min(alpha * 1.2, 400.0)
                stag = 0
                if lr < 5e-7:
                    break
        h = h_new
    return best_h, best_c

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _pair_cd(h, rng, deadline, max_iters=300000):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = float(c.max())
    best_h, best_c = h.copy(), c5_cur
    stable = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 3000 == 0 and stable > 4000:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1, k2 = n - 1 + d0, n - 1 - d0
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(40):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = int(np.argmax(vals))
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-14:
                d_hi = d_mid
            elif sg < -1e-14:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-13:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = float(c.max())
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c, best_h = c5_cur, h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt

            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _polish(h, rng, deadline):
    n = len(h)
    c5_cur = _c5(h)
    best_h, best_c = h.copy(), c5_cur
    for it in range(500000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        scale = max(5e-6, 3e-5 * (1.0 - it / 500000.0))
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _c5(h)
            if c5n < c5_cur - 1e-16:
                c5_cur = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _try_perturb(h, rng, seeds):

    best_h, best_c = h.copy(), _c5(h)
    for s in seeds:

            for _ in range(3):

            hp = np.clip(bp, 0, 1)
        else:
            hp = np.clip(h + 0.12 * rng.randn(n), 0, 1)
        hp = _proj(hp, n * 0.5)
        c5p = _c5(hp)
        if c5p < best_c:
            best_c, best_h = c5p, hp.copy()
