import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _g_shift(h, sh, n, dx):
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _smooth_grad(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    obj = m + np.log(w.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        g += w[k] * _g_shift(h, sh, n, dx)
    return obj, g, c

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    f, _, _ = _smooth_grad(h, alpha, n)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        _, g, _ = _smooth_grad(h, alpha, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new, _, _ = _smooth_grad(h_new, alpha, n)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 12.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj(h_prev - step * v, n * 0.5)
            f, _, _ = _smooth_grad(h, alpha, n)
    return h

def _frank_wolfe(h, alpha, n_iter, t_end):
    n = len(h)
    target = n * 0.5
    f, _, _ = _smooth_grad(h, alpha, n)
    best_f, best_h = f, h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        _, g, _ = _smooth_grad(h, alpha, n)
        idx = np.argpartition(g, int(target))[:int(target)]
        v = np.zeros(n)
        v[idx] = 1.0
        t = 2.0 / (it + 2)
        h_new = _proj((1 - t) * h + t * v, target)
        f_new, _, _ = _smooth_grad(h_new, alpha, n)
        if f_new < f:
            h, f = h_new, f_new
            if f < best_f - 1e-15:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _pair_cd(h, n, duration_s, top_T, rng, t_end):
    dx = 2.0 / n
    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)
    end_t = time.time() + duration_s
    no_improve = 0
    while time.time() < min(end_t, t_end - 5):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_g_shift(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        improved = False
        for _ in range(n):
            if time.time() > min(end_t, t_end - 5):
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(2, n // 8))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-14
            d_hi = min(1.0 - h[i], h[j]) - 1e-14
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(22):
                if f1 < f2:
                    d_hi, x2, f2 = x1, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            nc = _eval(h)
            if nc < best_c5:
                best_c5 = nc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 35:
            h += 2e-5 * rng.randn(n)
            h = _proj(h, n * 0.5)
            no_improve = 0
    return h, best_c5

def _subgrad_peak(h, n_iter, top_k, t_end, rng):
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    best_c5 = _eval(h)
    best_h = h.copy()
    step = 0.005
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            sh = k - (n - 1)
            g += _g_shift(h, sh, n, dx)
        g /= len(top_idx)
        h_new = _proj(h - step * g, target)
        nc = _eval(h_new)
        if nc < best_c5:
            best_c5 = nc
            best_h = h_new.copy()
            step = min(step * 1.08, 0.05)
        else:
            h = h_new
            step *= 0.6
            if step < 1e-9:
                step = 0.005
                h += 5e-5 * rng.randn(n)
                h = _proj(h, target)
        h = best_h if time.time() % 1.0 > 0.5 else h
    return best_h, best_c5
