import numpy as np
from scipy.optimize import minimize
import time, functools

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lse(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    s = beta * (c - c.max()); s -= s.max()
    w = np.exp(s); w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            grad[:n-sh] += wk * (1.0 - h[sh:n])
            if sh > 0: grad[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            grad[sp:] += wk * (1.0 - h[:n-sp])
            grad[:n-sp] -= wk * h[sp:]
    return grad * (beta * dx)

def _nesterov(h, beta0, beta1, steps, step0, t_end):
    n = len(h)
    f = _eval(h); v = np.zeros(n); step = step0
    for s in range(steps):
        if time.time() > t_end: break
        beta = beta0 + (beta1 - beta0) * (s / max(steps - 1, 1))
        g = _grad_lse(h, beta)
        hp = h.copy()
        v = 0.85 * v + 0.15 * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.04, step0 * 10)
        else:
            step *= 0.4
            if step < 1e-13: break
            h = _proj_to_sum(hp - step * v, n * 0.5); f = _eval(h)
    return _repair(h, n), f

def _cons_eq(x):
    return x.sum() - x.size * 0.5

def _cons_eq_jac(x):
    return np.ones(x.size)

def _lse_obj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:n])
            if sh > 0: g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    obj = functools.partial(_lse_obj, alpha=alpha)
    gr = functools.partial(_lse_grad, alpha=alpha)
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        res = minimize(obj, h, jac=gr, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
