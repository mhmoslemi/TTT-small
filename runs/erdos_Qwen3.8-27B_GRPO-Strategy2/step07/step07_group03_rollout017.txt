def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Project to sum constraint
        h = _proj_to_sum(h, target)
    return h

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval(h):

    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    return float(c.max())

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c.max(), c

    b = np.zeros(N)
    b[:2*n-1] = w

    grad = dx * (A_all[:n][::-1] - B_all[:n][::-1])

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _nesterov(h, alpha, n, steps, dx, t_end):

    a_buf = np.zeros(N)
    bf_buf = np.zeros(N)
    hf_buf = np.zeros(N)
    
    f = _obj_smooth(h, alpha, n)
    v = np.zeros(n)
    mu = 0.9
    step = 0.001
    best_f = f

        if time.time() > t_end:
            break
        # Compute weights
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()

        a_buf[n:] = 0
        b = w  # length 2n-1
        A_all = np.fft.irfft(np.fft.rfft(a_buf) * np.fft.conj(np.fft.rfft(b, N)), N)

        bf_buf[:2*n-1] = w[::-1]
        bf_buf[2*n-1:] = 0
        B_all = np.fft.irfft(np.fft.rfft(hf_buf) * np.fft.conj(np.fft.rfft(bf_buf)), N)
        
        grad = dx * (A_all[:n][::-1] - B_all[:n][::-1])

            f = _obj_smooth(h, alpha, n)
