        else:
            hi = mid

    c = _correlate_scaled(h)
    top_idx = np.argsort(c)[-top_k:]

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correlate(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_pooled(h, n, top_k=3):
    c = _correlate(h)
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top_idx:
        shift = int(k) - (n - 1)
        if shift == 0:
            g += 1.0 - 2.0 * h
        elif shift > 0:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
        else:
            s = -shift
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    g /= float(top_k)
    return g, c

def _cd_step(h, i, j, n):
    c5 = _eval(h)
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-15:
        return c5, False
    for _ in range(28):
        d_mid = 0.5 * (d_lo + d_hi)
        h[i] += d_mid
        h[j] -= d_mid
        c5_mid = _eval(h)
        h[i] -= d_mid
        h[j] += d_mid
        if c5_mid < c5:
            d_lo = d_mid
        elif c5_mid > c5:
            d_hi = d_mid
        else:
            break
    d_opt = 0.5 * (d_lo + d_hi)
    if abs(d_opt) < 1e-13:
        return c5, False
    h[i] += d_opt
    h[j] -= d_opt
    return _eval(h), True

def _guided_cd(h, n, n_iter, rng, t0, deadline, top_k=3, guided_frac=0.75):

    best_c = _eval(h)
    stable = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break

            break

            j = int(bot_j[rng.randint(20)])

            j = rng.randint(n)

            continue

        if improved and c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            stable = 0
        else:
            stable += 1
    return best_h, best_c

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)
