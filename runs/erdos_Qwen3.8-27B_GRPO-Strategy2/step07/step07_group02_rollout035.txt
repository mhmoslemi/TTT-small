    n = len(h)
    c5 = _eval(h)
    for _ in range(rnds):
        if time.time() > dl - 20:
            break
        c = _cvals(h)
        tk = np.argsort(c)[-nl:]
        g = np.zeros(n)
        for k in tk:
            sh = (n-1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0: g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= (2.0/n) / nl
        imp = False
        for st in [0.05,0.02,0.01,0.005,0.002,0.001,5e-4,1e-4,5e-5]:
            hn = _repair(h - st*g, n)
            cn = _eval(hn)
            if cn < c5 - 1e-15:
                h, c5, imp = hn, cn, True
                break
        if not imp:
            break
    return h, c5

def _local(h, iters, rng, dl, sc):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for _ in range(iters):
        if time.time() > dl - 2:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = sc * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            oi, oj = h[i], h[j]
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5 - 1e-19:
                c5 = cn
                if cn < bc: bc, bh = cn, h.copy()
            else:
                h[i], h[j] = oi, oj
    return bh, bc

def _lbfgs_smooth(h, alpha, mx, dl):
    n = len(h)
    def f(x):
        c = _cvals(x); m = c.max()
        s = np.clip(alpha*(c-m), -400, 0)
        return m + np.log(np.sum(np.exp(s)))/alpha
    def g(x):
        c = _cvals(x); m = c.max()
        s = np.clip(alpha*(c-m), -400, 0)
        w = np.exp(s); w /= w.sum()
        gx = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15: continue
            sh = (n-1)-k; wk = w[k]*2.0/n
            if sh >= 0:
                np_ = n-sh
                gx[:np_] += wk*(1.0-x[sh:])
                if sh > 0: gx[sh:] -= wk*x[:np_]
            else:
                sp = -sh
                gx[sp:] += wk*(1.0-x[:-sp])
                gx[:-sp] -= wk*x[sp:]
        return gx
    bnds = [(0.0,1.0)]*n
    try:
        r = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bnds,
                    options={'maxiter':mx,'ftol':1e-14,'gtol':1e-11})
        return _repair(r.x, n)
    except Exception:
        return h

def _subgrad(h, steps, s0, tol, dl):
    n = len(h); dx = 2.0/n
    for i in range(steps):
        if time.time() > dl - 15: break
        t = s0 / (1.0 + 0.03*i)
        c = _cvals(h); m = c.max()
        act = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for k in act:
            sh = (n-1)-k
            if sh >= 0:
                np_ = n-sh
                g[:np_] += (1.0-h[sh:])
                if sh > 0: g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0-h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / max(len(act),1)
        h = _repair(h - t*g, n)
    return h

def _rand_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, float).copy(), n0)
    bh, bc, bn = h0.copy(), _eval(h0), n0

    # Phase 1: Fourier optimization at full res
    if time.time() < dl - 500:
        for M in [8, 12, 16, 20, 28]:
            if time.time() > dl - 500: break
            for beta in [5.0, 30.0]:
                hf = _fh_opt(bh, n0, M, 4000, beta, 2500, dl)
                c = _eval(hf)
                if c < bc: bc, bh = c, hf.copy()

    # Phase 2: Random perturb + Fourier
    if time.time() < dl - 400:
        for tr in range(5):
            if time.time() > dl - 400: break
            hp = _repair(bh + 0.08*rng.randn(n0), n0)
            for M in [10, 18]:
                if time.time() > dl - 400: break
                hf = _fh_opt(hp, n0, M, 6000, 15.0, 2000, dl)
                c = _eval(hf)
                if c < bc: bc, bh = c, hf.copy()

    # Phase 3: Try random binary starts
    if time.time() < dl - 350:
        for _ in range(3):
            if time.time() > dl - 350: break
            hb = _rand_binary(n0, rng)
            for M in [12, 20]:
                if time.time() > dl - 350: break
                hf = _fh_opt(hb, n0, M, 5000, 20.0, 2000, dl)
                c = _eval(hf)
                if c < bc: bc, bh = c, hf.copy()

    # Phase 4: Multi-resolution
    if time.time() < dl - 250:
        for fac in [2, 4]:
            if time.time() > dl - 250: break
            nr = n0 // fac
            hr = _repair(_down(bh, fac), nr)
            for M in [4, 8, 12, 16]:
                if time.time() > dl - 250: break
                hf = _fh_opt(hr, nr, M, 4000, 15.0, 1500, dl)
                hu = _repair(_up(hf, fac), n0)
                c = _eval(hu)
                if c < bc: bc, bh = c, hu.copy()
            # L-BFGS at reduced res then upsample
            hs = _lbfgs_smooth(hr, 20000, 1500, dl)
            hu = _repair(_up(hs, fac), n0)
            c = _eval(hu)
            if c < bc: bc, bh = c, hu.copy()

    # Phase 5: Smoothed L-BFGS at full res
    if time.time() < dl - 180:
        for alpha in [10000, 100000, 1000000]:
            if time.time() > dl - 180: break
            hn = _lbfgs_smooth(bh.copy(), alpha, 3000, dl)
            c = _eval(hn)
            if c < bc: bc, bh = c, hn.copy()

    # Phase 6: Subgradient
    if time.time() < dl - 100:
        hs = _subgrad(bh, 1500, 0.01, 1e-9, dl)
        c = _eval(hs)
        if c < bc: bc, bh = c, hs.copy()

    # Phase 7: Top-lag descent
    if time.time() < dl - 60:
        for nl in [2, 3, 5, 7, 10]:
            if time.time() > dl - 60: break
            ht, ct = _top_lag(bh.copy(), nl, 500, dl)
            if ct < bc: bc, bh = ct, ht.copy()

    # Phase 8: Subgradient again (finer)
    if time.time() < dl - 30:
        hs = _subgrad(bh, 2000, 0.003, 1e-10, dl)
        c = _eval(hs)
        if c < bc: bc, bh = c, hs.copy()

    # Phase 9: Top-lag final pass
    if time.time() < dl - 15:
        for nl in [2, 3]:
            if time.time() > dl - 15: break
            ht, ct = _top_lag(bh.copy(), nl, 400, dl)
            if ct < bc: bc, bh = ct, ht.copy()

    # Phase 10: Fine local search
    for sc in [1e-5, 3e-6, 1e-6]:
        if time.time() > dl - 3: break
        hl, cl = _local(bh, 40000, rng, dl, sc)
        if cl < bc: bc, bh = cl, hl.copy()

    bh = _repair(bh, bn)
    bc = _eval(bh)
    return (bh, bc, bn)
