        h *= target / s
        h = np.clip(h, 0.0, 1.0)

            for idx in order:

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

                    diff += a

    s = alpha * (c - m); s -= s.max()
    w = np.exp(s); ws = w.sum()
    return m + np.log(ws) / alpha, w / ws

def _topk_grad(h, k_top, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    w = np.zeros(2 * n - 1)
    for k in top_k:
        w[k] = 1.0 / k_top
    return _corr_grad(w, h, n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov(h, k_top, steps, step0, t0, deadline):
    n = len(h); f = _eval(h); v = np.zeros(n); mu = 0.85; step = step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _topk_grad(h, k_top, n)
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new; f = f_new; step = min(step * 1.02, step0 * 3.0)
        else:
            step *= 0.5
            if step < 1e-10: break
            h = _proj_sum(h - step * v, n * 0.5); f = _eval(h)
    return h

def _sig_h(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -40, 40)))

def _h_sig(h):
    h = np.clip(h, 1e-7, 1 - 1e-7)
    return np.log(h / (1.0 - h))

def _sig_obj(z):
    h = _sig_h(z)
    obj, w = _soft_c(h, _G['alpha'], _G['n'])
    obj += _G['mu'] * (h.sum() - _G['n'] * 0.5) ** 2
    return obj

def _sig_grad(z):
    h = _sig_h(z)
    _, w = _soft_c(h, _G['alpha'], _G['n'])
    g = _corr_grad(w, h, _G['n'])
    g += 2.0 * _G['mu'] * (h.sum() - _G['n'] * 0.5)
    return g * h * (1.0 - h)

def _lbfgs_sig(z, alpha, mu, n, maxiter):
    _G['alpha'] = alpha; _G['mu'] = mu; _G['n'] = n
    try:
        res = minimize(_sig_obj, z, jac=_sig_grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-19, 'gtol': 1e-14})
        return res.x
    except Exception:
        return z

def _slsqp_sig(h, alpha, mu, n, maxiter):
    z = _h_sig(h)
    _G['alpha'] = alpha; _G['mu'] = mu; _G['n'] = n
    bnds = [(-30, 30)] * n
    try:
        res = minimize(_sig_obj, z, jac=_sig_grad, method='SLSQP',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-19})
        return _sig_h(res.x)
    except Exception:
        return h

def _block_search(h, n_tries, bsize, step, deadline):
    n = len(h); best_c = _eval(h); best_h = h.copy()
    rng = np.random.RandomState(789)
    for _ in range(n_tries):
        if time.time() > deadline: break
        start = rng.randint(0, max(1, n - bsize))
        d = step * rng.randn(bsize); d -= d.mean()
        trial = h.copy(); trial[start:start + bsize] += d
        if trial.min() >= -1e-12 and trial.max() <= 1 + 1e-12:
            c_new = _eval(trial)
            if c_new < best_c - 1e-18:
                best_c = c_new; best_h = trial
    return best_h, best_c

def _point_search(h, n_tries, step, deadline):
    n = len(h); best_c = _eval(h); best_h = h.copy()
    rng = np.random.RandomState(321)
    for _ in range(n_tries):
        if time.time() > deadline: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d; h[j] -= d
            c_new = _eval(h)
            if c_new < best_c - 1e-18:
                best_c = c_new; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
    return best_h, best_c

def _fbm_start(n, seed):
    x = np.linspace(0, 2, n)
    s = 0.5
    for oct in range(6):
        freq = 2.0 * (2 ** oct)
        s += 0.5 / (oct + 1) * np.sin(2 * np.pi * freq * x + seed * 0.1 * oct)
    return np.clip(s, 0.01, 0.99)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        n = len(initial_h_values)
        h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    else:
        n = 560
        h = _fbm_start(n, seed)
        h = _repair(h, n)

    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Nesterov settle
    h = _nesterov(h, 5, 3000, 0.003, t0, t0 + 60)
    c = _eval(h)
    if c < best_c5: best_c5, best_h = c, h.copy()

    # Phase 2: L-BFGS continuation (sigmoid space)
    alphas = [300, 1500, 8000, 40000, 150000]
    for alpha in alphas:
        if time.time() > t0 + budget_s - 220: break
        z = _h_sig(best_h) + 0.004 * rng.randn(n)
        z = _lbfgs_sig(z, alpha, 1e6, n, 1000)
        h_new = _repair(_sig_h(z), n)
        c = _eval(h_new)
        if c < best_c5: best_c5, best_h = c, h_new

    # Phase 3: Nesterov with wider top-k
    if time.time() < t0 + budget_s - 130:
        h = _nesterov(best_h.copy(), 25, 2500, 0.0007, t0, t0 + budget_s - 120)
        c = _eval(h)
        if c < best_c5: best_c5, best_h = c, h.copy()

    # Phase 4: SLSQP in sigmoid (handles near-exact max)
    if time.time() < t0 + budget_s - 80:
        for alpha in [200000, 800000]:
            if time.time() > t0 + budget_s - 55: break
            h_new = _repair(_slsqp_sig(best_h, alpha, 1e6, n, 600), n)
            c = _eval(h_new)
            if c < best_c5: best_c5, best_h = c, h_new

    # Phase 5: Coarse-to-fine escape
    if time.time() < t0 + budget_s - 90:
        for fac in [2, 4]:
            if time.time() > t0 + budget_s - 70: break
            n_lo = n // fac
            h_lo = best_h[:n_lo * fac].reshape(n_lo, fac).mean(axis=1)
            h_lo = _repair(h_lo, n_lo)
            for kt in [5, 15]:
                h_lo = _nesterov(h_lo, kt, 1500, 0.003, t0, t0 + budget_s - 60)
            h_hi = np.repeat(_repair(h_lo, n_lo), fac)[:n]
            h_hi = _repair(h_hi, n)
            h_hi = _nesterov(h_hi, 10, 1500, 0.001, t0, t0 + budget_s - 50)
            h_hi = _lbfgs_sig(_h_sig(h_hi), 50000, 1e6, n, 400)
            h_hi = _repair(_sig_h(h_hi), n)
            c = _eval(h_hi)
            if c < best_c5: best_c5, best_h = c, h_hi

    # Phase 6: Final high-alpha L-BFGS
    if time.time() < t0 + budget_s - 35:
        z = _h_sig(best_h)
        z = _lbfgs_sig(z, 2000000, 1e6, n, 500)
        h_new = _repair(_sig_h(z), n)
        c = _eval(h_new)
        if c < best_c5: best_c5, best_h = c, h_new

    # Phase 7: Block search
    if time.time() < t0 + budget_s - 12:
        for bs, st in [(8, 0.012), (4, 0.025), (16, 0.006)]:
            if time.time() > t0 + budget_s - 8: break
            best_h, best_c5 = _block_search(best_h, 2000, bs, st, t0 + budget_s - 6)

    # Phase 8: Point search (ultra-fine)
    if time.time() < t0 + budget_s - 2:
        best_h, best_c5 = _point_search(best_h, 25000, 1.2e-4, t0 + budget_s - 1)
        best_h, best_c5 = _point_search(best_h, 20000, 4e-5, t0 + budget_s - 1)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
