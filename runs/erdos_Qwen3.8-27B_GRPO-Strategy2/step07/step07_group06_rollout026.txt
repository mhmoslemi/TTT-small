import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_pos(h, pos, n):
    g = np.zeros(2 * n - 1)
    s = np.arange(-(n - 1), n)
    v1 = (pos + s >= 0) & (pos + s < n)
    g[v1] = 1.0 - h[pos + s[v1]]
    v2 = (pos - s >= 0) & (pos - s < n)
    g[v2] -= h[pos - s[v2]]
    return g * (2.0 / n)

def _pair_cd(h, deadline, rng, smart_pairs=True):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = float(c.max())
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    it = 0
    top_shifts = np.zeros(0, dtype=int)
    while time.time() < deadline:
        it += 1
        if stable > 12000 and it > 3000:
            break
        if smart_pairs and it % 500 == 0:
            top_kk = np.argsort(c)[-8:]
            top_shifts = top_kk - (n - 1)
        if smart_pairs and len(top_shifts) > 0 and rng.random() < 0.6:
            s_pick = top_shifts[rng.randint(len(top_shifts))]
            i = rng.randint(max(0, -s_pick + 1), min(n, n - s_pick))
            j = i + s_pick
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_pos(h, i, n) - _grad_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        for _ in range(35):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < 2 * n - 1:
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < 2 * n - 1:
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if kmax == k1 or kmax == k2:
                sg += 2.0 * d_mid
            if sg > 1e-13:
                d_hi = d_mid
            elif sg < -1e-13:
                d_lo = d_mid
            else:
                break
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-12:
            stable += 1
            continue
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < 2 * n - 1:
            c[k1] += d_opt * d_opt
        if 0 <= k2 < 2 * n - 1:
            c[k2] += d_opt * d_opt
        c5_new = float(c.max())
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < 2 * n - 1:
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < 2 * n - 1:
                c[k2] -= d_opt * d_opt
            stable += 1
    return best_h, best_c

def _anneal_cd(h, deadline, rng, T0=0.015, T1=0.0003, max_steps=80000):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = float(c.max())
    best_h = h.copy()
    best_c = c5_cur
    for it in range(max_steps):
        if time.time() > deadline:
            break
        frac = it / float(max_steps)
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step_mag = 0.005 + 0.03 * rng.random()
        d = step_mag * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_pos(h, i, n) - _grad_pos(h, j, n)
        c_new = c + d * g
        if 0 <= k1 < 2 * n - 1:
            c_new[k1] += d * d
        if 0 <= k2 < 2 * n - 1:
            c_new[k2] += d * d
        c5_new = float(c_new.max())
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h[i], h[j] = ni, nj
            c = c_new
            c5_cur = c5_new
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = h[i], h[j]
    return best_h, best_c

def _subgrad_quick(h, deadline, rng, m_top=4):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    it = 0
    prev_best = best_c
    while time.time() < deadline:
        it += 1
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top = np.argsort(c)[-m_top:]
        g = np.zeros(n)
        for k in top:
            s = k - (n - 1)
            pr = np.arange(n)
            v1 = (pr + s >= 0) & (pr + s < n)
            g[v1] += 1.0 - h[pr[v1] + s]
            v2 = (pr - s >= 0) & (pr - s < n)
            g[v2] -= h[pr[v2] - s]
        g *= dx / m_top
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = min(0.015, 0.8 / (gn + 1e-10))
        improved = False
        for sm in [1.0, 0.5, 0.25, 0.1]:
            h_new = _proj_to_sum(h - sm * step * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5n
                improved = True
                if c5_cur < best_c:
                    best_c = c5_cur
                    best_h = h.copy()
                break
        if not improved and c5_cur == prev_best:
            break
        prev_best = c5_cur
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Try a few perturbations for better start
    for ps in [0.01, 0.04, 0.08, 0.15]:
        hp = _proj_to_sum(np.clip(h + ps * rng.randn(n), 0, 1), n * 0.5)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 1: Quick subgradient (10%)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _subgrad_quick(best_h.copy(), dl1, rng, m_top=5)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Smart pair CD (30%)
    dl2 = t0 + budget_s * 0.40
    h2, c5_2 = _pair_cd(best_h.copy(), dl2, rng, smart_pairs=True)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Annealing to explore (20%)
    dl3 = t0 + budget_s * 0.60
    h3, c5_3 = _anneal_cd(best_h.copy(), dl3, rng, T0=0.012, T1=0.0004)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: CD after annealing (20%)
    dl4 = t0 + budget_s * 0.80
    h4, c5_4 = _pair_cd(best_h.copy(), dl4, rng, smart_pairs=True)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Kick + CD restart (10%)
    if time.time() < budget_s * 0.90:
        h_kick = _proj_to_sum(np.clip(best_h + 0.03 * rng.randn(n), 0, 1), n * 0.5)
        dl5 = t0 + budget_s * 0.93
        h5, c5_5 = _pair_cd(h_kick, dl5, rng, smart_pairs=True)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5
        # Also try without kick from best
        h5b, c5_5b = _pair_cd(best_h.copy(), t0 + budget_s * 0.95, rng, smart_pairs=True)
        if c5_5b < best_c5:
            best_c5, best_h = c5_5b, h5b

    # Phase 6: Final subgradient + CD polish (7%)
    dl6 = t0 + budget_s * 0.97
    h6, c5_6 = _subgrad_quick(best_h.copy(), dl6, rng, m_top=3)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6
    h7, c5_7 = _pair_cd(best_h.copy(), t0 + budget_s - 3, rng, smart_pairs=True)
    if c5_7 < best_c5:
        best_c5, best_h = c5_7, h7

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
