import numpy as np
import time
from scipy.optimize import minimize as _min

def _rep(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float, copy=False)
    t = 0.5 * n
    s = float(h.sum())
    if s < 1e-15:
        return np.full(n, 0.5, dtype=float)
    if abs(s - t) > 1e-10:
        h = h * (t / s)
        h = np.clip(h, 0.0, 1.0)
        d = t - float(h.sum())
        if abs(d) > 1e-12:
            o = np.argsort(h) if d > 0 else np.argsort(-h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a
                    d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a
                    d += a
                if abs(d) < 1e-13:
                    break
    return h

def _proj(y, t):
    y = np.asarray(y, float)
    n = y.size
    if n == 0:
        return y
    ys = float(y.sum())
    if y.min() >= 0 and y.max() <= 1 and abs(ys - t) < 1e-10:
        return y
    th = (ys - t) / n
    x = y - th
    if x.min() >= -1e-12 and x.max() <= 1 + 1e-12:
        x = np.clip(x, 0.0, 1.0)
        if abs(float(x.sum()) - t) < 1e-8:
            return x
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        if np.clip(y - m, 0.0, 1.0).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _cor(h):
    n = h.size
    if n == 0:
        return np.array([0.0])
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def _ev(h):
    return float(np.max(_cor(h))) if h.size else 0.0

def _top(c, k, a):
    m = c.size
    k = int(k)
    k = 1 if k < 1 else (m if k > m else k)
    idx = np.argpartition(c, -k)[-k:] if k < m else np.arange(m)
    cm = float(c[idx].max())
    z = float(a) * (c[idx] - cm)
    np.clip(z, -50.0, 0.0, out=z)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s < 1e-300:
        w = np.ones_like(idx, dtype=float)
        w /= float(m)
        f = cm
    else:
        w /= s
        f = cm + float(np.log(s)) / float(a)
    return idx, w, f

def _grad(h, idx, w, n):
    g = np.zeros(n, dtype=float)
    o = 1.0 - h
    for kk, wk in zip(idx, w):
        if wk <= 0.0:
            continue
        sh = n - 1 - int(kk)
        if sh > 0:
            g[sh:] += wk * o[:n - sh]
            g[:n - sh] -= wk * h[sh:]
        elif sh == 0:
            g += wk * (o - h)
        else:
            sp = -sh
            g[:n - sp] += wk * o[sp:]
            g[sp:] -= wk * h[:n - sp]
    return g

def _coar(h, m):
    n = h.size
    if m <= 0 or m == n:
        return h.copy()
    if n % m == 0:
        hc = h.reshape(n // m, m).mean(axis=0)
    else:
        hc = np.interp(np.linspace(0.0, 1.0, m), np.linspace(0.0, 1.0, n), h)
    return _rep(hc, m)

def _start(n, rng):
    i = np.arange(n, dtype=float)
    t = 2.0 * np.pi * i / float(n)
    return _rep(0.5 + 0.25 * np.sin(t) + 0.12 * np.sin(2.0 * t) + 0.06 * np.sin(3.0 * t), n)

def _mv(h, a, b, cur, T, rng, mode):
    x = float(h[a])
    y = float(h[b])
    diff = (x - y) if mode else (y - x)
    if diff <= 1e-7:
        return False, None, cur
    md = min(x, 1.0 - y, float(diff))
    if md <= 1e-8:
        return False, None, cur
    for f in (0.3, 0.6, 0.9):
        d = f * md
        if d <= 0.0:
            continue
        h[a] -= d
        h[b] += d
        c = _cor(h)
        m2 = float(c.max())
        dl = m2 - cur
        if dl < -1e-15 or (0.0 < dl < T and rng.rand() < np.exp(-dl / T)):
            return True, c, m2
        h[a] += d
        h[b] -= d
    return False, None, cur

def _pick(sc, cnt, rng):
    m = sc.size
    if m == 0:
        return np.array([], dtype=int)
    cnt = min(cnt, m)
    if rng.rand() < 0.8:
        idx = np.arange(cnt) if m <= cnt else np.argpartition(sc, -cnt)[-cnt:]
        rng.shuffle(idx)
    else:
        idx = rng.randint(0, m, cnt)
    return idx

def _mlgd(h, n, end, k, a, st):
    t = 0.5 * n
    h = _proj(h, t)
    bh = h.copy()
    bc = _ev(h)
    c = _cor(h)
    idx, w, f = _top(c, k, a)
    cm = float(c.max())
    steps = [st, st * 0.5, st * 0.25, st * 0.125]
    stall = 0
    while time.time() < end - 0.3:
        c = _cor(h)
        idx, w, f = _top(c, k, a)
        cm = float(c.max())
        g = _grad(h, idx, w, n)
        g -= g.mean()
        if float(np.dot(g, g)) < 1e-22:
            break
        improved = False
        for s_ in steps:
            if s_ <= 0.0:
                continue
            x = _proj(h - s_ * g, t)
            c2 = _cor(x)
            ___, f2 = _top(c2, k, a)
            m2 = float(c2.max())
            if (f2 < f - 1e-12 and m2 <= cm + 1e-8) or (m2 < cm - 1e-12):
                h = x
                f = f2
                cm = m2
                improved = True
                if m2 < bc:
                    bc = m2
                    bh = h.copy()
                ns = []
                for mult in (1.6, 1.0, 0.6, 0.3):
                    s2 = s_ * mult
                    if 1e-7 < s2 < 0.08:
                        ns.append(s2)
                if ns:
                    steps = ns
                break
        if not improved:
            steps = [s_ * 0.5 for s_ in steps]
            stall += 1
            if stall > 25 or steps[0] < 1e-8:
                break
        else:
            stall = 0
    bh = _rep(bh, n)
    return bh, float(_ev(bh))

def _pgs(h, n, end, fine=False):
    if time.time() >= end - 2.0:
        return h, float(_ev(h))
    c = float(_ev(h))
    total = end - time.time()
    if fine:
        seq = ((4, 400.0, 0.008, 0.25), (6, 800.0, 0.005, 0.25),
               (10, 1500.0, 0.004, 0.25), (12, 300.0, 0.003, 0.25))
    else:
        seq = ((4, 50.0, 0.05, 0.25), (8, 150.0, 0.03, 0.25),
               (12, 300.0, 0.02, 0.25), (20, 600.0, 0.012, 0.25))
    for k, a, s_, fr in seq:
        e = min(end - 1.0, time.time() + max(2.0, fr * total))
        if time.time() < e - 1.0:
            h2, c2 = _mlgd(h, n, e, k, a, s_)
            if c2 < c:
                h, c = h2, float(c2)
    return h, float(c)

def _dl(h, n, end, rng, ms=60):
    bh = h.copy()
    bc = _ev(h)
    c = _cor(h)
    cur = float(c.max())
    T = 5e-6
    stall = 0
    while time.time() < end - 0.3:
        if cur > bc + 1e-4:
            h = bh.copy()
            c = _cor(h)
            cur = float(c.max())
            stall = 0
            continue
        tol = max(2e-6, 1e-4 * (cur + 1e-12))
        act = np.where(c >= cur - tol)[0]
        if act.size == 0 or rng.rand() < 0.55:
            kk = int(np.argmax(c))
        else:
            kk = int(rng.choice(act))
        p = kk - n + 1
        mv = False
        if p == 0:
            if rng.rand() < 0.6:
                lo = int(np.argmin(h))
                hi = int(np.argmax(h))
            else:
                idx = rng.randint(0, n, 12)
                v = h[idx]
                lo = int(idx[int(np.argmin(v))])
                hi = int(idx[int(np.argmax(v))])
            if lo != hi:
                ok, c2, m2 = _mv(h, lo, hi, cur, T, rng, False)
                if ok:
                    c = c2
                    cur = m2
                    mv = True
                    if cur < bc:
                        bh = h.copy()
                        bc = cur
        elif p > 0:
            m = n - p
            sc = h[:m] - h[p:]
            srcadd = 0
            dstadd = p
        else:
            q = -p
            m = n - q
            sc = h[q:] - h[:m]
            srcadd = q
            dstadd = 0
        if p != 0 and m > 0 and float(sc.max()) > 1e-7:
            for r in _pick(sc, 8, rng):
                r = int(r)
                ok, c2, m2 = _mv(h, r + srcadd, r + dstadd, cur, T, rng, True)
                if ok:
                    c = c2
                    cur = m2
                    mv = True
                    if cur < bc:
                        bh = h.copy()
                        bc = cur
                    break
        if mv:
            stall = 0
            T = max(5e-8, T * 0.97)
        else:
            stall += 1
        if stall >= ms:
            break
    bh = _rep(bh, n)
    return bh, float(_ev(bh))

def _lobj(x, n, k, a, lam):
    c = _cor(x)
    ___, f = _top(c, k, a)
    d = float(x.sum()) - 0.5 * n
    return float(f) + lam * d * d

def _lgrad(x, n, k, a, lam):
    c = _cor(x)
    idx, w, ___ = _top(c, k, a)
    g = _grad(x, idx, w, n)
    d = float(x.sum()) - 0.5 * n
    return g + 2.0 * lam * d if d else g

def _lbs(h, n, end):
    c = float(_ev(h))
    if time.time() >= end - 2.0:
        return h, c
    total = end - time.time()
    t = time.time()
    for k, a, lam, fr in ((8, 300.0, 1000.0, 0.34), (14, 800.0, 2000.0, 0.33), (6, 1500.0, 3000.0, 0.33)):
        e = min(end - 2.0, t + max(3.0, fr * total))
        rem = e - time.time()
        if rem < 1.0:
            break
        mx = int(min(60, max(5, int(rem))))
        try:
            res = _min(_lobj, h, args=(n, k, a, lam), jac=_lgrad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': mx, 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 20})
        except Exception:
            res = None
        if res is not None:
            x = _rep(res.x, n)
            cv = float(_ev(x))
            if cv < c:
                h, c = x, cv
        t = e
    return h, c

def _basin(h, n, end, rng):
    c = float(_ev(h))
    strengths = (0.05, 0.025, 0.01, 0.005)
    i = 0
    while time.time() < end - 2.0:
        e1 = min(end - 10.0, time.time() + 12.0)
        if e1 <= time.time() + 1.0:
            break
        x = h.copy()
        cnt = max(5, n // 25)
        cnt = min(cnt, n)
        idx = rng.choice(n, cnt, replace=False)
        x[idx] += strengths[i % 4] * rng.randn(cnt)
        x = _proj(x, 0.5 * n)
        x, c2 = _mlgd(x, n, e1, 6, 400.0, 0.015)
        e2 = min(end - 1.0, time.time() + 8.0)
        if time.time() < e2 - 0.5:
            x, c2 = _dl(x, n, e2, rng, 20)
        if c2 < c:
            h, c = x, float(c2)
        i += 1
    return h, c

def _copol(h, end, rng):
    n = h.size
    bc = float(_ev(h))
    bh = h.copy()
    for m in (n // 2, n // 4, n // 8):
        if time.time() >= end - 3.0 or m < 16 or n % m != 0:
            continue
        x = _coar(h, m)
        cv = float(_ev(x))
        if cv < bc - 1e-10:
            e = min(end - 1.0, time.time() + 5.0)
            if time.time() < e - 0.5:
                x, cv = _dl(x, m, e, rng, 30)
            if cv < bc:
                bh, bc = x, float(cv)
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    D = t0 + max(5.0, B - 5.0)
    S = max(60.0, B)
    rng = np.random.RandomState(seed)
    init = globals().get('initial_h_values', None)
    if init is not None:
        h = np.asarray(init, float).ravel().copy()
    if init is None or h.size < 4:
        h = _start(560, rng)
    n = h.size
    if n > 1000:
        h = _coar(h, 560)
        n = 560
    if n % 2 == 1:
        h = h[:-1]
        n -= 1
    if n < 2:
        h = _start(4, rng)
        n = 4
    h = _rep(h, n)
    best = h.copy()
    bc = float(_ev(best))
    for m in (n // 2, n // 4, n // 8):
        if m >= 16 and n % m == 0:
            hc = _coar(h, m)
            cc = float(_ev(hc))
            if cc < bc:
                best, h, bc, n = hc, hc, cc, m
    tA = min(D, t0 + 0.14 * S)
    if time.time() < tA - 1.0:
        h2, c2 = _pgs(best, n, tA)
        if c2 < bc:
            best, bc = h2, float(c2)
    tB = min(D, t0 + 0.24 * S)
    if time.time() < tB - 1.0:
        h2, c2 = _dl(best, n, tB, rng, 50)
        if c2 < bc:
            best, bc = h2, float(c2)
    tC = min(D, t0 + 0.40 * S)
    if time.time() < tC - 3.0:
        h2, c2 = _lbs(best, n, tC)
        if c2 < bc:
            best, bc = h2, float(c2)
    tD = min(D, t0 + 0.52 * S)
    if time.time() < tD - 1.0:
        h2, c2 = _pgs(best, n, tD, True)
        if c2 < bc:
            best, bc = h2, float(c2)
    tE = min(D, t0 + 0.66 * S)
    if time.time() < tE - 1.0:
        h2, c2 = _dl(best, n, tE, rng, 70)
        if c2 < bc:
            best, bc = h2, float(c2)
    tF = min(D, t0 + 0.86 * S)
    if time.time() < tF - 1.0:
        h2, c2 = _basin(best, n, tF, rng)
        if c2 < bc:
            best, bc = h2, float(c2)
    if time.time() < D - 1.0:
        h2, c2 = _dl(best, n, D, rng, 150)
        if c2 < bc:
            best, bc = h2, float(c2)
    if time.time() < D - 3.0:
        h2, c2 = _copol(best, D - 1.0, rng)
        if c2 < bc:
            best, bc = h2, float(c2)
    best = _rep(best, int(len(best)))
    bc = float(_ev(best))
    return (best, bc, int(len(best)))
