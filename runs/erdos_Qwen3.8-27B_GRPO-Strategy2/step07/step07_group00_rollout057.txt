import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        ct = c[np.argpartition(c, -topk)[-topk:]]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        sh = (n - 1) - idx[ii]
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _coord_descent(h, n_passes, deadline):
    n = len(h)
    target = n * 0.5
    lags = np.arange(2 * n - 1) - (n - 1)
    c5c = _eval(h)
    bh = h.copy()
    for p in range(n_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
        for i in range(n):
            jp = i + lags
            jm = i - lags
            vp = (jp >= 0) & (jp < n)
            vm_ = (jm >= 0) & (jm < n)
            B = np.zeros(2 * n - 1)
            B[vp] = 1.0 - h[jp[vp]]
            B[vm_] -= h[jm[vm_]]
            A = corr - h[i] * B
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) * 0.333
                m2 = hi - (hi - lo) * 0.333
                if np.max(A + B * m1) < np.max(A + B * m2):
                    hi = m2
                else:
                    lo = m1
            xo = (lo + hi) * 0.5
            f0 = np.max(A)
            f1 = np.max(A + B)
            fx = np.max(A + B * xo)
            if f0 <= fx and f0 <= f1:
                xo = 0.0
            elif f1 <= fx and f1 <= f0:
                xo = 1.0
            delta = xo - h[i]

                h[i] = xo
