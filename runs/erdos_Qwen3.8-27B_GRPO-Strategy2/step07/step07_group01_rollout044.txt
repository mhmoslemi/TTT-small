def _adam_opt(h, alpha, steps, lr, top_k=None, beta1=0.85, beta2=0.999):
    n = len(h)
    m_state = np.zeros(n)
    v_state = np.zeros(n)

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smgrad(h, alpha, topk):
    n = len(h); dx = 2.0 / n; L = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if topk < L:
        idx = np.argpartition(c, -topk)[-topk:]
        ca = c[idx]
    else:
        idx = np.arange(L); ca = c
    m = ca.max()
    s = np.clip(alpha * (ca - m), -400, 400)
    se = np.exp(s); se_sum = se.sum()
    val = m + np.log(se_sum) / alpha
    w = se / se_sum
    grad = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx; k = idx[ii]; sh = (n - 1) - k
        if sh >= 0:

            i1 = np.arange(n - sp)
            grad[i1] += wk * (1.0 - h[i1 + sp])
            grad[sp:] -= wk * h[:n-sp]
    return val, grad

def _adam(h, alpha, steps, lr, topk, b1, b2):
    n = len(h); ms = np.zeros(n); vs = np.zeros(n)
    for t in range(steps):
        _, g = _smgrad(h, alpha, topk)
        ms = b1 * ms + (1 - b1) * g
        vs = b2 * vs + (1 - b2) * g * g
        mh = ms / (1.0 - b1 ** (t + 1))
        vh = vs / (1.0 - b2 ** (t + 1))
        h = _proj(h - lr * mh / (np.sqrt(vh) + 1e-8), n * 0.5)
    return h

def _downsample(h, f):
    m = len(h) // f

def _upsample(h, f):

    return out

def _anneal(h, n_iter, rng, t0, t1, base_step, triple_frac):
