def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = h.sum() - target

        return h
    # Binary search for threshold
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5

            lo = mid

            hi = mid

        hcontrib = vm * h[j] * (1.0 - hm) + vp * hp * (1.0 - h[j])
        const = c - hcontrib
        coeff = vm * (1.0 - hm) - hp
        vals = const[None, :] + coeff[None, :] * xs[:, None]
        quad = c[n - 1] - h[j] + h[j] * h[j] + xs - xs * xs
        vals_max = np.maximum(np.max(vals, axis=1), quad)
        h[j] = xs[np.argmin(vals_max)]
    return h

def _smooth_obj(x, alpha):
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            np_ = n - sh
            g[:np_] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:np_]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:-sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline - 25:
        return h
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h

def _top_lag(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 25:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] += -wk_dummy * 0
        g *= dx / n_lags
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4]:
            h_new = _proj(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def _proj(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _local_2pt(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    for _ in range(n_iter):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j2 = (i + 1 + rng.randint(0, 5)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j2] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j2] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j2] = ni - d, nj + d
    return best_h, best_c

def _top_lag2(h, n_lags, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    c5_cur = _eval(h)
    for _ in range(max_rounds):
        if time.time() > deadline - 25:
            break
        c = _cvals(h)
        top_k = np.argsort(c)[-n_lags:]
        g = np.zeros(n)
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                np_ = n - sh
                g[:np_] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:np_]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:-sp] -= h[sp:]
        g *= dx / n_lags
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 1e-4, 5e-5]:
            h_new = _proj(h - step * g, n)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5n
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coordinate descent - coarse grid, many sweeps
    target_time = deadline - 350
    while time.time() < target_time:
        h = _coord_sweep(h, n0, 41)
        h = _repair(h, n0)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: L-BFGS to escape coordinate-descent local opt
    h = best_h.copy()
    for alpha in [5000, 20000, 80000]:
        if time.time() > deadline - 280:
            break
        h = _lbfgs(h, alpha, 4000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Coordinate descent - medium grid
    target_time = deadline - 200
    while time.time() < target_time:
        h = _coord_sweep(h if h is best_h else best_h, n0, 81)
        h = _repair(h, n0)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        else:
            h = best_h.copy()

    # Phase 4: Top-lag descent
    if time.time() < deadline - 150:
        for nl in [2, 3, 5]:
            h_ml, c5_ml = _top_lag2(best_h.copy(), nl, 300, deadline)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()
                h = h_ml.copy()
            else:
                h = best_h.copy()

    # Phase 5: L-BFGS with high alpha
    if time.time() < deadline - 100:
        h = _lbfgs(best_h.copy(), 300000, 5000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Coordinate descent - fine grid
    target_time = deadline - 60
    while time.time() < target_time:
        h = _coord_sweep(best_h, n0, 151)
        h = _repair(h, n0)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()
        else:
            # try small perturbation to escape
            h = best_h.copy()
            mask = rng.rand(n0) < 0.05
            h[mask] = np.clip(h[mask] + 0.02 * rng.randn(mask.sum()), 0, 1)
            h = _repair(h, n0)

    # Phase 7: Final L-BFGS
    if time.time() < deadline - 30:
        h = _lbfgs(best_h.copy(), 800000, 3000, deadline)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 8: Very fine coordinate descent
    if time.time() < deadline - 15:
        h = _coord_sweep(best_h, n0, 251)
        h = _repair(h, n0)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 9: Local 2-point polish
    if time.time() < deadline - 5:
        h_ls, c5_ls = _local_2pt(best_h, 80000, rng, deadline)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
