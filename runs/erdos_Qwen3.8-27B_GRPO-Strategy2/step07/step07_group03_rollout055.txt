import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-16:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 50:
            h += 2e-5 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        s = h.sum()
        if abs(s - n * 0.5) > 1e-7:
            h = _proj_to_sum(h, n * 0.5)
    return h, best_c5

def _fine_search(h, n_tries, rng, t_end):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 10)) % n
        if i == j:
            continue
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            oi, oj = h[i], h[j]
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c - 1e-18:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = oi, oj
    return best_h, best_c

def _freq_perturb(h, rng, n, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(30):
        if time.time() > t_end:
            break
        m = rng.randint(2, 25)
        eps = rng.choice([5e-4, 1e-3, 2e-3, 5e-3])
        phase = rng.rand() * 2 * np.pi
        x = np.arange(n) * 2.0 / n
        pert = eps * np.sin(2 * np.pi * m * x / 2.0 + phase)
        h_try = _proj_to_sum(h + pert, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov smoothing
    for alpha in [3000, 10000, 50000, 200000]:
        t_end = min(time.time() + 30, t0 + budget_s * 0.08)
        c_before = _eval(h)
        h = _nesterov_block(h, alpha, n, t_end, 200)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Exact 1D coordinate descent, top-1 lag (main workhorse)
    if time.time() < t0 + budget_s * 0.40:
        h = best_h.copy()
        h, c5v = _exact_cd(h, n, budget_s * 0.30, 1, rng, t0 + budget_s, 6)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Exact CD, top-3 lags
    if time.time() < t0 + budget_s * 0.60:
        h = best_h.copy()
        h, c5v = _exact_cd(h, n, budget_s * 0.18, 3, rng, t0 + budget_s, 5)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: Gradient-based CD for speed
    if time.time() < t0 + budget_s * 0.72:
        h = best_h.copy()
        h, c5v = _grad_cd(h, n, budget_s * 0.10, 2, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Frequency perturbation exploration
    if time.time() < t0 + budget_s * 0.80:
        h = best_h.copy()
        h, c5v = _freq_perturb(h, rng, n, t0 + budget_s * 0.78)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Fine local search
    if time.time() < t0 + budget_s - 60:
        remaining = (t0 + budget_s - 60) - time.time()
        n_tries = min(int(remaining / 0.003), 50000)
        if n_tries > 200:
            h = best_h.copy()
            h, c5v = _fine_search(h, n_tries, rng, t0 + budget_s - 55)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 7: Final exact CD polish
    if time.time() < t0 + budget_s - 10:
        h = best_h.copy()
        remaining = (t0 + budget_s - 10) - time.time()
        if remaining > 5:
            h, c5v = _exact_cd(h, n, remaining, 1, rng, t0 + budget_s - 3, 8)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
