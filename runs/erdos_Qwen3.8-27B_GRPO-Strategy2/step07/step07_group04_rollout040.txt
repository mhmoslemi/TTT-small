from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lse_weights(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s - s.max())
    return e / e.sum()

def _grad_weighted(h, w):
    """Gradient of weighted sum of correlations."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k]
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g * dx

def _adam_opt(h, steps, lr, t0, deadline):
    """Adam optimizer with LSE smoothing."""
    n = len(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    alpha_schedule = np.linspace(100, 100000, steps)
    for t in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        alpha = alpha_schedule[min(t, steps - 1)]
        w = _lse_weights(c, alpha)
        g = _grad_weighted(h, w)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        h_new = _proj_to_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:

            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - x[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * x[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - x[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * x[i2 + sp]
    return g

def _slsqp_opt(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def _obj(x):
        return _slsqp_obj(x, alpha, n)
    def _grad(x):
        return _slsqp_grad(x, alpha, n)
    def _cons_fun(x):
        return x.sum() - target
    def _cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_descent(h, n_sweeps, rng, t0, deadline):
    """2-point coordinate descent: swap between pairs of indices."""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            j = (i + rng.randint(1, 12)) % n
            if i == j:
                continue
            # Try small perturbation: h[i] += d, h[j] -= d
            d = 1e-4 * rng.randn()
            for scale in [1.0, 0.1, 0.01]:
                dd = d * scale
                ni, nj = h[i] + dd, h[j] - dd
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c_new = _eval(h)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    else:
                        h[i], h[j] = h[i] - dd, h[j] + dd
                        break
    return best_h, best_c
