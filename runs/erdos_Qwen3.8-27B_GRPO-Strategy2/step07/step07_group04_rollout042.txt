import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    """Project h to [0,1]^n with sum = n/2 using bisection."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, max(0, 1.0 - h[idx]))
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _soft_loss_grad(h, alpha):
    """Return (loss, gradient) for softmax-smoothed max of correlations."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 0)
    w = np.exp(s)
    w_sum = w.sum()
    loss = m + np.log(w_sum) / alpha
    p = w / w_sum  # softmax weights over lags
    # Gradient
    gh = np.zeros(n)
    for k in range(len(c)):
        if p[k] < 1e-10:
            continue
        wk = p[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            if sh == 0:
                gh += wk * (1.0 - 2.0 * h)
            else:
                gh[:n - sh] += wk * (1.0 - h[sh:])
                gh[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            gh[sp:] += wk * (1.0 - h[:n - sp])
            gh[:n - sp] -= wk * h[sp:]
    return loss, gh

    n = len(h)
