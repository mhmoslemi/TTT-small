import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Main CD (280s)
    dl2 = t0 + min(325, budget_s * 0.325)
    h2, c5_2 = _pair_cd(best_h.copy(), rng, dl2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: PG again to escape (40s)
    dl3 = t0 + min(365, budget_s * 0.365)
    h3, c5_3 = _proj_grad(best_h.copy(), n0, rng, dl3, max_iters=200000)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: More CD (260s)
    dl4 = t0 + min(625, budget_s * 0.625)
    h4, c5_4 = _pair_cd(best_h.copy(), rng, dl4)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4

    # Phase 5: Another PG (30s)
    dl5 = t0 + min(655, budget_s * 0.655)
    h5, c5_5 = _proj_grad(best_h.copy(), n0, rng, dl5, max_iters=150000)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5

    # Phase 6: Final CD (250s)
    dl6 = t0 + min(905, budget_s * 0.905)
    h6, c5_6 = _pair_cd(best_h.copy(), rng, dl6)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6

    # Phase 7: Fine polish (remainder)
    dl7 = t0 + (budget_s - 8)
    h7, c5_7 = _fine_polish(best_h.copy(), rng, dl7)
    if c5_7 < best_c5:
        best_c5, best_h = c5_7, h7

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
