import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w[k] * (1.0 - h[i1 + sh]) * dx
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= w[k] * h[i2 - sh] * dx
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[k] * (1.0 - h[i1 - sp]) * dx
            i2 = np.arange(n - sp)
            g[i2] -= w[k] * h[i2 + sp] * dx
    return g

def _nesterov_smooth(h, alpha, steps, step0, t_end):
    n = len(h)
    f = _smooth_obj(h, alpha)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _smooth_obj(h_new, alpha)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.025, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _smooth_obj(h, alpha)
    return h

def _coord_far(h, n, duration_s, top_T, rng, t_end):

    gr = (np.sqrt(5) - 1) / 2
    best_c5 = _eval(h)

    no_improve = 0
    while time.time() < end_t:

        top_idx = np.argsort(c)[-top_T:]

                    g[i2] -= h[i2 - sh]

                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= h[i2 + sp]

                a_vals = a_vals + b * d_opt
                improved = True
        if improved:

        else:

        if no_improve > 25:

            h = _proj_to_sum(h, n * 0.5)
            no_improve = 0
        if abs(h.sum() - n * 0.5) > 1e-6:
            h = _proj_to_sum(h, n * 0.5)

def _block_cd(h, n, duration_s, rng, t_end):
    """Block coordinate descent: update blocks of consecutive indices."""
    dx = 2.0 / n
    best_c5 = _eval(h)
    end_t = min(time.time() + duration_s, t_end - 4)
    block_sizes = [5, 8, 12, 20]
    while time.time() < end_t:
        bs = rng.choice(block_sizes)
        si = rng.randint(0, n - bs)
        block = h[si:si+bs].copy()
        c_full = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c_full)[-12:]
        for _ in range(40):
            if time.time() > end_t:
                break
            # Random perturbation of block
            delta = 0.02 * rng.randn(bs)
            new_block = np.clip(block + delta, 0, 1)
            h_trial = h.copy()
            h_trial[si:si+bs] = new_block
            h_trial = _proj_to_sum(h_trial, n * 0.5)
            ct = _eval(h_trial)
            if ct < best_c5 - 1e-15:
                h = h_trial
                best_c5 = ct
                block = h[si:si+bs].copy()
    return h, best_c5

def _fine_local(h, n, n_iter, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
