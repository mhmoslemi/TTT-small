import time

def _proj_sum(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, n, k):
    dx = 2.0 / n
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        a = np.arange(n - sh)
        g[a] += (1.0 - h[a + sh]) * dx
        if sh > 0:
            b = np.arange(sh, n)
            g[b] -= h[b - sh] * dx
    else:
        sp = -sh
        a = np.arange(sp, n)
        g[a] += (1.0 - h[a - sp]) * dx
        b = np.arange(n - sp)
        g[b] -= h[b + sp] * dx
    return g

def _smooth_obj(h, n, alpha, top_k):
    c = _corr(h, n)
    if top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        idx = np.arange(len(c))
    cv = c[idx]
    m = cv.max()
    s = np.clip(alpha * (cv - m), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    obj = m + np.log(w.sum()) / alpha
    return obj, w, idx

def _smooth_grad(h, n, alpha, top_k):
    obj, w, idx = _smooth_obj(h, n, alpha, top_k)
    g = np.zeros(n)
    for j in range(len(idx)):
        if w[j] < 1e-14:
            continue
        g += w[j] * _grad_lag(h, n, idx[j])
    return g

def _fista(h0, n, alpha, top_k, n_iter, t_end):
    h = h0.copy().astype(float)
    y = h.copy()
    t_prev = 1.0
    L = alpha * 4.0 / n * top_k + 1e-6
    step = 1.0 / L
    step = min(step, 0.5)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _smooth_grad(y, n, alpha, top_k)
        y_new = _proj_sum(y - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) * 0.5
        beta = (t_prev - 1.0) / t_new
        y = y_new + beta * (y_new - h)
        h = y_new
        t_prev = t_new
    return h

def _coord_sweep(h, n, top_T, n_sweeps, rng, t_end, step_scale=1.0):
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c5 = _eval_c5(h)
    for sv in range(n_sweeps):
        if time.time() > t_end:
            break
        c = _corr(h, n)
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_grad_lag(h, n, k) for k in top_idx])
        a_vals = c[top_idx].copy()
        improved = False
        for _ in range(n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, max(3, n // 15))) % n
            if i == j:
                continue
            b = grads[:, i] - grads[:, j]
            d_lo = max(-h[i], h[j] - 1.0) + 1e-12
            d_hi = min(1.0 - h[i], h[j]) - 1e-12
            if d_lo >= d_hi:
                continue
            x1 = d_hi - gr * (d_hi - d_lo)
            x2 = d_lo + gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
            f2 = np.max(a_vals + b * x2)
            for _ in range(20):
                if f1 < f2:
                    d_hi, x2, f2 = x1, x1, f1
                    x1 = d_hi - gr * (d_hi - d_lo)
                    f1 = np.max(a_vals + b * x1)
                else:
                    d_lo, x1, f1 = x1, x2, f2
                    x2 = d_lo + gr * (d_hi - d_lo)
                    f2 = np.max(a_vals + b * x2)
            d_opt = (d_lo + d_hi) * 0.5
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
    return h, _eval_c5(h)

def _local_refine(h, n, n_evals, rng, t_end):
    c5 = _eval_c5(h)
    best_h = h.copy()
    best_c = c5
    step = 1e-5
    for _ in range(n_evals):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d
            h[j] -= d
            c5n = _eval_c5(h)
            if c5n < c5 - 1e-20:
                c5 = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                    step = min(step * 1.1, 5e-5)
            else:
                h[i] -= d
                h[j] += d
                if step > 2e-8:
                    step *= 0.85
        else:
            step *= 0.9
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    h = _proj_sum(h, n)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h
