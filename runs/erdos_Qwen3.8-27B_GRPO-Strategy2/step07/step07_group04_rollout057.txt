import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_corr(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = weights[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _softmax_weights(c, T):
    m = c.max()
    s = np.clip(T * (c - m), -300, 300)
    e = np.exp(s - s.max())
    return e / e.sum()

def _nesterov_sm(h, T_start, T_end, steps, step0, t0, budget_s):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.65
    step = step0
    for s in range(steps):
        if time.time() - t0 > budget_s - 5:
            break
        T = T_start + (T_end - T_start) * s / max(steps, 1)
        T = max(T, 10.0)
        c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
        w = _softmax_weights(c, T)
        g = _grad_corr(h, w)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.02, step0 * 8.0)
            mu = min(mu + 0.002, 0.93)
        else:
            step *= 0.55
            if step < 1e-11:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
            mu = max(mu - 0.03, 0.5)
    return _repair(h, n)

def _slsqp_soft(h, alpha, maxiter, t0, budget_s):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        e = np.exp(s - s.max())
        return m + np.log(e.sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        e = np.exp(s - s.max())
        w = e / e.sum()
        return _grad_corr(x, w)
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _upsample(h, f):
    return np.repeat(h, f).copy()

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _local_search(h, n_tries, rng, t0, budget_s, mode, scale):

    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c

                    h[i],h[j] = ni-d, nj+d

def _polyak_step(h, g, f, f_opt_est, n, t0, budget_s):
    """One Polyak-accelerated step"""
    step = max((f - f_opt_est) / (np.dot(g,g) + 1e-20), 1e-8)
    step = min(step, 0.1)
    h_new = _proj_to_sum(h - step * g, n * 0.5)
