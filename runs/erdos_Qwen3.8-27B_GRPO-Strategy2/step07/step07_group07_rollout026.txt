import numpy as np
import time
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    target = n / 2.0
    if abs(s - target) > 1e-9:
        h = _proj_sum(h, target)
    return h

def _grad_corr(h, n, k_idx):
    """Gradient of correlation at lag k_idx"""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh]) * dx
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh] * dx
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _subgrad_active(h, n, active_k):
    """Average subgradient over active lags"""
    g = np.zeros(n)
    for k in active_k:
        g += _grad_corr(h, n, k)
    return g / len(active_k)

def _project_grad(g, h, n):
    """Project gradient onto feasible directions"""
    # Remove component along 1 (sum constraint)
    g = g - g.mean()
    # Zero out where bounds are active
    at_lo = (h <= 1e-8) & (g < 0)
    at_hi = (h >= 1 - 1e-8) & (g > 0)
    g[at_lo | at_hi] = 0
    # Re-project for sum
    g = g - g.mean()
    return g

def _active_descent(h, max_iter, k_init, k_max, eps, el_fn, bc_fn):
    """Active-set subgradient descent with line search"""
    n = len(h)
    h_c = h.copy()
    best_c = _eval(h_c)
    best_h = h_c.copy()
    stall = 0
    k = k_init
    
    for it in range(max_iter):
        if bc_fn():
            break
        
        c = _corrs(h_c)
        c_max = c.max()
        
        # Active set: all lags within eps of max
        if eps > 0:
            active = np.where(c >= c_max - eps)[0]
        else:
            active = np.where(c >= c_max - 1e-10)[0]
        
        # Limit active set size
        if len(active) > 50:
            active = np.argsort(c)[-50:]
        
        g = _subgrad_active(h_c, n, active)
        g = _project_grad(g, h_c, n)
        gn = np.linalg.norm(g)
        
        if gn < 1e-15:
            break
        
        direction = -g / gn
        
        # Line search: try multiple step sizes
        best_step = 0
        best_val = c_max
        steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        
        for alpha in steps:
            h_try = h_c + alpha * direction
            h_try = _proj_sum(h_try, n * 0.5)
            val = _eval(h_try)
            if val < best_val:
                best_val = val
                best_step = alpha
                best_try = h_try.copy()
        
        if best_step > 0:
            h_c = best_try
            if best_val < best_c:
                best_c = best_val
                best_h = h_c.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Adaptive k and eps
        if stall > 20:
            k = min(k + 1, k_max)
            eps = min(eps * 1.5, 0.05)
        if stall > 100:
            # Kick: random perturbation
            idx = np.random.randint(0, n, size=5)
            d = np.random.randn(5) * 0.01
            h_kick = h_c.copy()
            h_kick[idx] += d
            h_kick = _proj_sum(h_kick, n * 0.5)
            if _eval(h_kick) < best_c + 0.001:
                h_c = h_kick
            stall = 0
    
    return best_h, best_c

            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1-sp])
            i2 = np.arange(n-sp)
            g[i2] -= wk * h[i2+sp]
    return g + 2.0 * pen * (h.sum() - n*0.5)

def _lbfgs_pnorm(h0, p, pen, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pnorm_obj, h0, jac=_pnorm_grad, args=(p, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    def bc():
        return el() > budget_s - 8

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS p-norm (compact)
    for p, pen, mi in [(64, 1e5, 400), (256, 1e6, 300), (1024, 1e7, 250)]:
        if el() > 120:
            break
        h_try = _lbfgs_pnorm(best_h.copy(), p, pen, mi)
        c_try = _eval(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 2: Active-set descent (main)
    h2, c2 = _active_descent(best_h, 8000, 0.003, 1.3, 60, 800, el, bc)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Active-set with smaller tau (sharper focus)
    if el() < budget_s - 250:
        h3, c3 = _active_descent(best_h, 8000, 0.001, 1.4, 30, 600, el, bc)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Simulated annealing
    if el() < budget_s - 150:
        na = min(int((budget_s - 150 - el()) / 0.003), 60000)
        h4, c4 = _anneal_search(best_h, na, 0.003, 0.998, rng, el, bc)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Final active-set polish
    if el() < budget_s - 40:
        h5, c5 = _active_descent(best_h, 3000, 0.0005, 1.2, 15, 2000, el, bc)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Phase 6: Final L-BFGS
    if el() < budget_s - 5:
        h6 = _lbfgs_pnorm(best_h.copy(), 2048, 1e8, 150)
        if _eval(h6) < best_c5:
            best_h, best_c5 = h6, _eval(h6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
