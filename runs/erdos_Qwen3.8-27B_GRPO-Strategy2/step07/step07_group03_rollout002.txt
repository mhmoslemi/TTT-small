def _repair(h, n):

    target = n / 2.0

    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

    return h

        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: SA with gradient-guided moves
    t_sa_end = t0 + budget_s * 0.38
    if time.time() < t_sa_end:
        h = best_h.copy()
        best_h, best_c5 = _sa_phase(h, t_sa_end, rng)

    # Phase 3: Nesterov refinement after SA
    if time.time() < t0 + budget_s * 0.46:
        h = best_h.copy()
        for alpha in [1000000, 5000000, 20000000]:
            t_end = min(time.time() + 10, t0 + budget_s * 0.45)
            h = _nesterov(h, alpha, 250, 0.002, t_end)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

    # Phase 4: Intensive coordinate descent
    for top_T in [5, 10, 15]:
        t_cd_end = t0 + budget_s * 0.74
        if time.time() > t_cd_end - 5:
            break
        h = best_h.copy()
        dur = min(budget_s * 0.07, t_cd_end - time.time() - 2)
        if dur < 5:
            break
        h, c5v = _coord_descent(h, n, dur, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: More coord descent with varied params
    for top_T in [3, 8, 12]:
        t_cd2_end = t0 + budget_s * 0.91
        if time.time() > t_cd2_end - 5:
            break
        h = best_h.copy()
        dur = min(budget_s * 0.04, t_cd2_end - time.time() - 2)
        if dur < 5:
            break
        h, c5v = _coord_descent(h, n, dur, top_T, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 6: Fine local search
    if time.time() < t0 + budget_s - 15:
        remaining = (t0 + budget_s) - time.time() - 10
        n_tries = min(int(remaining / 0.004), 80000)
        if n_tries > 200:
            h = best_h.copy()
            best_h, best_c5 = _fine_search(h, n_tries, rng, t0 + budget_s - 8)

    # Phase 7: Final coord polish
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        dur = max(1.0, budget_s - (time.time() - t0) - 2)
        h, c5v = _coord_descent(h, n, dur, 8, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
