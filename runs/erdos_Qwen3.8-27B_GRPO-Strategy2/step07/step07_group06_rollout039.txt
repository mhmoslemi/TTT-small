def _pair_cd(h, n_iter, rng, deadline, top_m=5):
    n = len(h)
    c = _corr(h)
    c5_cur = c.max()
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stable > 3000:
            break
        
        if it % 300 == 0:
            c = _corr(h)
            c5_cur = c.max()
        
        # Select pair based on top lags
        top_idx = np.argsort(c)[-top_m:]
        # Combined gradient for top lags
        # For speed, just use the top lag
        k_star = top_idx[-1]
        shifts = np.arange(-(n-1), n)
        s = k_star - (n-1)
        
        # Gradient at each position for this lag
        # g_p = (1-h[p+s]) if 0<=p+s<n, minus h[p-s] if 0<=p-s<n
        gp = np.zeros(n)
        if s >= 0:
            v = np.arange(n - s)
            gp[v] += (1.0 - h[v + s])
            if s > 0:
                v2 = np.arange(s, n)
                gp[v2] -= h[v2 - s]
        else:
            ss = -s
            v = np.arange(ss, n)
            gp[v] += (1.0 - h[v - ss])
            if ss > 0:
                v2 = np.arange(n - ss)
                gp[v2] -= h[v2 + ss]
        
        # Pick i from positive gradient, j from negative
        pos_mask = gp > 0.01
        neg_mask = gp < -0.01
        if pos_mask.sum() > 0 and neg_mask.sum() > 0:
            i = np.where(pos_mask)[0][rng.randint(pos_mask.sum())]
            j = np.where(neg_mask)[0][rng.randint(neg_mask.sum())]
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if gp[i] < gp[j]:
                i, j = j, i
        
        if i == j:
            continue
        
        # Compute pair gradient
        gi = _grad_pos(h, i, n)
        gj = _grad_pos(h, j, n)
        g = gi - gj
        
        # Bounds on d
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:
            continue
        
        # Quadratic correction positions
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        
        # Bisection
        for _ in range(35):
            d_mid = 0.5 * (d_lo + d_hi)
            vals = c + d_mid * g
            if 0 <= k1 < len(c):
                vals[k1] += d_mid * d_mid
            if 0 <= k2 < len(c):
                vals[k2] += d_mid * d_mid
            kmax = np.argmax(vals)
            sg = g[kmax]
            if (kmax == k1 or kmax == k2):
                sg += 2.0 * d_mid
            if sg > 1e-13:
                d_hi = d_mid
            elif sg < -1e-13:
                d_lo = d_mid
            else:
                break
        
        d_opt = 0.5 * (d_lo + d_hi)
        if abs(d_opt) < 1e-12:
            stable += 1
            continue
        
        h[i] += d_opt
        h[j] -= d_opt
        c += d_opt * g
        if 0 <= k1 < len(c):
            c[k1] += d_opt * d_opt
        if 0 <= k2 < len(c):
            c[k2] += d_opt * d_opt
        
        c5_new = c.max()
        if c5_new < c5_cur - 1e-13:
            c5_cur = c5_new
            stable = 0
            if c5_cur < best_c:
                best_c = c5_cur
                best_h = h.copy()
        else:

            h[j] += d_opt
            c -= d_opt * g
            if 0 <= k1 < len(c):
                c[k1] -= d_opt * d_opt
            if 0 <= k2 < len(c):
                c[k2] -= d_opt * d_opt
            stable += 1
    
    return best_h, best_c

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for ps in [0.02, 0.05, 0.12]:
        hp = _proj_to_sum(np.clip(h + ps * rng.randn(n0), 0, 1), n0 / 2.0)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
    dl1 = t0 + budget_s * 0.08
    h1, c5_1 = _subgrad_pass(best_h.copy(), rng, dl1, m=5)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1
    dl2 = t0 + budget_s * 0.55
    h2, c5_2 = _cd_pass(best_h.copy(), rng, dl2, top_m=3)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2
    rng2 = np.random.RandomState(seed + 7)
    dl3 = t0 + budget_s * 0.82
    h3, c5_3 = _cd_pass(best_h.copy(), rng2, dl3, top_m=5)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3
    dl4 = t0 + budget_s * 0.90
    h4, c5_4 = _subgrad_pass(best_h.copy(), rng, dl4, m=3)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4
    dl5 = t0 + (budget_s - 5)
    h5, c5_5 = _polish(best_h.copy(), rng, dl5)
    if c5_5 < best_c5:
        best_c5, best_h = c5_5, h5
    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
