    n = len(h)
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    # Get top-k lags
    idx = np.argpartition(c, -topk)[-topk:]
    # Compute subgradient: sum of gradients for active lags
    grad = np.zeros(n)
    for k in idx:
        lag = (n-1) - k
        # gradient of c[k] w.r.t. h
        # c[k] = sum_j h[j]*(1-h[j+lag]) for valid j
        # d/dh[i] = (1-h[i+lag]) - h[i-lag] (with bounds)

    # Negative gradient step (minimizing)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_for_lag(h, lag):
    """Gradient of correlation at given lag (m = n-1-k) w.r.t. h, times dx."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n-lag] += (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= h[:n-lag]
    else:
        l = -lag
        if l < n:
            g[l:] += (1.0 - h[:n-l])
            g[:n-l] -= h[l:]
    return g * dx

def _topk_grad(h, topk, weights=None):
    """Weighted sum of gradients for top-k lags."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argpartition(c, -topk)[-topk:]

        w = np.exp(ct[idx] * 100)  # sharp weighting
        w /= w.sum()
    else:
        w = np.array(weights)
    grad = np.zeros(n)
    for ii in range(len(idx)):
        lag = (n - 1) - idx[ii]
        grad += w[ii] * _grad_for_lag(h, lag)
    return c.max(), grad, idx

    step = step0

            h = hn
            c5 = cn

def _smoothened_obj(h, alpha, topk):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smoothened_grad(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())

        lag = (n - 1) - idx[ii]
        grad += w[ii] * _grad_for_lag(h, lag)

def _nesterov_smooth(h, alpha, topk, steps, step0, deadline):

def _grad_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = rng.randint(1, 9)
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        c = _corr(h)
        top5 = np.argpartition(c, -5)[-5:]
        g = np.zeros(n)
        for idx in top5:
            lag = (n-1) - idx
            g += _lag_grad(h, lag)
        grad_seg = -g[i0:i0+bs]
        gn = np.linalg.norm(grad_seg)
        if gn > 1e-14:
            grad_seg /= gn
        d = 0.002 * max(T, 1e-10)**0.5 * rng.randn(bs) + 0.0015 * grad_seg
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _coord_search(h, n_passes, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    half = n // 2
    for p in range(n_passes):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            j = (i + half) % n
            if i == j:
                continue
            for d in [3e-5, -3e-5, 1e-5, -1e-5, 7e-5, -7e-5]:
                if h[i] + d < 0 or h[i] + d > 1:
                    continue
                if h[j] - d < 0 or h[j] - d > 1:
                    continue
                h[i] += d; h[j] -= d
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n; bh = h.copy()
                else:
                    h[i] -= d; h[j] += d
                    break
    return bh, bc

def _micro_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 2.5e-6 * rng.randn()
        ni = h[i] + d; nj = h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP quick refinement
    dl1 = min(deadline, t0 + 90)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n0 * 0.5,
             'jac': lambda x: np.ones(n0)}]
    for alpha, tk, mi in [(5000, 25, 150), (30000, 12, 200)]:
        if time.time() > dl1:
            break
        try:
            r = minimize(lambda x: _smax(x, alpha, tk), h,
                         jac=lambda x: _smax_grad(x, alpha, tk),
                         method='SLSQP', bounds=[(0.0, 1.0)] * n0,
                         constraints=cons, options={'maxiter': mi, 'ftol': 1e-15})
            h = _repair(r.x, n0)
            c5v = _eval(h)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()
        except Exception:
            pass

    # Phase 2: Nesterov with parameter sweep
    dl2 = min(deadline, t0 + 320)
    for alpha, tk, st, ns in [(20000, 15, 0.003, 2500), (100000, 8, 0.0015, 2500),
                               (500000, 5, 0.0008, 2000), (2000000, 3, 0.0004, 1500)]:
        if time.time() > dl2:
            break
        h2 = _nesterov(best_h, alpha, tk, ns, st, dl2)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h2.copy()

    # Phase 3: Gradient-guided annealing
    dl3 = min(deadline, t0 + 550)
    h_a, c5_a = _grad_anneal(best_h.copy(), 70000, rng, dl3, 0.003, 1e-10)
    if c5_a < best_c5:
        best_c5 = c5_a; best_h = h_a.copy()

    # Phase 4: Very sharp Nesterov on top 2-3 lags
    dl4 = min(deadline, t0 + 720)
    for alpha, tk, st, ns in [(10000000, 3, 0.0002, 2500), (50000000, 2, 0.0001, 2000)]:
        if time.time() > dl4:
            break
        h2 = _nesterov(best_h, alpha, tk, ns, st, dl4)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h2.copy()

    # Phase 5: Coordinate descent search
    dl5 = min(deadline, t0 + budget_s - 40)
    if time.time() < dl5:
        best_h, best_c5 = _coord_search(best_h, 4, dl5)

    # Phase 6: Micro polish
    dl6 = t0 + budget_s - 3
    rem = max(0, dl6 - time.time())
    n_pol = min(int(rem / 0.0012), 30000)
    best_h, best_c5 = _micro_polish(best_h, n_pol, rng, dl6)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
