def _active_grad(h,c,n,tau,K):
    L=2*n-1
    cmax = c.max()
    # select top K indices by c (argpartition)
    if K >= L: idx=np.arange(L)
    else: idx=np.argpartition(c, -K)[-K:]
    # weights exp((c[idx]-cmax)/tau); if cmax large? Use tau e.g. 1e-4
    w=np.exp((c[idx]-cmax)/tau)
    s=w.sum()
    if s <=0: w=np.ones_like(c[idx])/K
    else: w/=s
    g=np.zeros(n)
    for kk,ww in zip(idx,w):
       if ww < 1e-15: continue
       sh=kk-(n-1)
       if sh>0:

          g[:ln] += ww*(1.0-h[sh:sh+ln])  # h[sh:n]

          g += ww*(1.0-2.0*h)
    return g

        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def _compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _eval(h):
    return float(np.max(_compute_c(h, len(h))))


def _eval_fixed(h):
    h = _repair(h, len(h))
    return h, float(_eval(h))


def _interp_h(h0, n):
    n0 = len(h0)
    if n0 == n:
        return h0.copy()
    x0 = np.linspace(0.0, 1.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x1, x0, h0)


def _default_start(n):
    x = np.arange(n) / float(n)
    h = 0.5 + 0.45 * np.sin(2.0 * math.pi * x * 3.0) + 0.15 * np.sin(2.0 * math.pi * x * 7.0)
    return _proj_sum(np.clip(h, 0.0, 1.0), 0.5 * n)


def _grad_pair(g, i, j, h, n):
    g.fill(0.0)
    rev = h[::-1]
    s1 = n - 1 - i
    g[s1:s1 + n] += 1.0 - h
    g[i:i + n] -= rev
    s2 = n - 1 - j
    g[s2:s2 + n] -= 1.0 - h
    g[j:j + n] += rev
    g *= 2.0 / n


def _line_search(c, h, i, j, n, L, g, mat):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, float(c.max())

    _grad_pair(g, i, j, h, n)
    q1 = n - 1 + (j - i)
    q2 = n - 1 - (j - i)
    zero = n - 1
    dx = 2.0 / n

    NC = 41
    RF = 13

    Ds = np.linspace(lo, hi, NC)
    if lo < 0.0 < hi:
        Ds = np.concatenate((Ds, np.array([0.0])))
        Ds.sort()

    M = int(Ds.shape[0])
    sub = mat[:M]
    np.multiply(g[None, :], Ds[:, None], out=sub)
    sub += c
    dd = Ds * Ds
    sub[:, q1] += dx * dd
    if q2 != q1:
        sub[:, q2] += dx * dd
    sub[:, zero] -= 2.0 * dx * dd

    row = sub.max(axis=1)
    b0 = int(np.argmin(row))
    best_d = float(Ds[b0])
    best_f = float(row[b0])

    order = np.argsort(row)[:2]
    for idx in order:
        idx = int(idx)
        left = max(0, idx - 1)
        right = min(M - 1, idx + 1)
        if right <= left:
            continue
        D2 = np.linspace(float(Ds[left]), float(Ds[right]), RF)
        sub2 = mat[:RF]
        np.multiply(g[None, :], D2[:, None], out=sub2)
        sub2 += c
        dd2 = D2 * D2
        sub2[:, q1] += dx * dd2
        if q2 != q1:
            sub2[:, q2] += dx * dd2
        sub2[:, zero] -= 2.0 * dx * dd2
        row2 = sub2.max(axis=1)
        b2 = int(np.argmin(row2))
        f2 = float(row2[b2])
        if f2 < best_f:
            best_f = f2
            best_d = float(D2[b2])

    return best_d, best_f


def _apply_pair(h, c, i, j, d, g, q1, q2, zero, dx):
    if abs(d) < 1e-18:
        return
    dd = d * d
    h[i] += d
    h[j] -= d
    c += d * g
    c[q1] += dd * dx
    if q2 != q1:
        c[q2] += dd * dx
    c[zero] -= 2.0 * dd * dx
    if h[i] < 0.0:
        h[i] = 0.0
    elif h[i] > 1.0:
        h[i] = 1.0
    if h[j] < 0.0:
        h[j] = 0.0
    elif h[j] > 1.0:
        h[j] = 1.0


def _active_grad(h, c, n, k, tau):
    L = 2 * n - 1
    if k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -k)[-k:]
    cmax = float(c[idx].max())
    w = np.exp((c[idx] - cmax) / tau)
    sw = float(w.sum())
    if sw <= 0.0 or not math.isfinite(sw):
        w = np.full_like(c[idx], 1.0 / max(1, k))
    else:
        w /= sw

    g = np.zeros(n)
    for kk, ww in zip(idx, w):
        if ww < 1e-15:
            continue
        sh = int(kk) - (n - 1)
        if sh > 0:
            ln = n - sh
            g[:ln] += ww * (1.0 - h[sh:n])
            g[sh:n] -= ww * h[:ln]
        elif sh < 0:
            s = -sh
            ln = n - s
            g[s:n] += ww * (1.0 - h[:ln])
            g[:ln] -= ww * h[s:n]
        else:
            g += ww * (1.0 - 2.0 * h)
    g *= 2.0 / n
    return g


def _candidate_pairs(g, h, k, m):
    n = len(h)
    if n < 2 or k <= 0 or m <= 0:
        return []
    can_up = h < (1.0 - 1e-12)
    can_dn = h > 1e-12
    if not can_up.any() or not can_dn.any():
        return []

    K = min(int(k), n - 1)
    gl = g.copy()
    gl[~can_up] = 1e100
    gi = np.argpartition(gl, K)[:K]

    gh = g.copy()
    gh[~can_dn] = -1e100
    gj = np.argpartition(gh, n - K)[-K:]

    arr = []
    for i in gi:
        if not can_up[i]:
            continue
        for j in gj:
            if i == j or not can_dn[j]:
                continue
            arr.append((float(g[i] - g[j]), int(i), int(j)))

    arr.sort()
    res = []
    seen = set()
    for _, i, j in arr:
        key = i * 10000 + j
        if key not in seen:
            seen.add(key)
            res.append((i, j))
            if len(res) >= m:
                break
    return res


def _merge_pairs(pairs_list, m):
    seen = set()
    res = []
    for lst in pairs_list:
        for i, j in lst:
            if i == j:
                continue
            key = i * 10000 + j
            if key not in seen:
                seen.add(key)
                res.append((i, j))
                if len(res) >= m:
                    return res
    return res


def _anneal_step(h, c, n, L, g, cmax, rng, temp):
    i = rng.randint(n)
    j = rng.randint(n)
    if i == j:
        return False, cmax
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo <= 1e-14:
        return False, cmax

    d = float(rng.uniform(lo, hi))
    if abs(d) < 1e-14:
        d = 0.25 * (hi - lo) * (1.0 if rng.random() < 0.5 else -1.0)
        if d < lo or d > hi:
            d = 0.5 * (lo + hi)
        if abs(d) < 1e-14:
            return False, cmax

    _grad_pair(g, i, j, h, n)
    q1 = n - 1 + (j - i)
    q2 = n - 1 - (j - i)
    zero = n - 1
    dx = 2.0 / n
    dd = d * d

    h[i] += d
    h[j] -= d
    c += d * g
    c[q1] += dd * dx
    if q2 != q1:
        c[q2] += dd * dx
    c[zero] -= 2.0 * dd * dx

    new = float(c.max())
    delta = new - cmax
    if delta <= 1e-12 or (temp > 0.0 and rng.random() < math.exp(-delta / temp)):
        if h[i] < 0.0:
            h[i] = 0.0
        elif h[i] > 1.0:
            h[i] = 1.0
        if h[j] < 0.0:
            h[j] = 0.0
        elif h[j] > 1.0:
            h[j] = 1.0
        return True, new

    h[i] -= d
    h[j] += d
    c -= d * g
    c[q1] -= dd * dx
    if q2 != q1:
        c[q2] -= dd * dx
    c[zero] += 2.0 * dd * dx
    return False, cmax


def _local_search(h, c, cmax, rng, deadline, stall_s=120.0):
    n = len(h)
    L = 2 * n - 1
    g = np.zeros(L)
    mat = np.empty((100, L))

    best_h = h.copy()
    best_c = float(cmax)
    last_best = time.time()
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        if it % 25 == 0 and time.time() >= deadline:
            break

        if it % 300 == 0:
            c = _compute_c(h, n)
            old = float(cmax)
            cmax = float(c.max())
            if cmax < best_c - 1e-12:
                best_c = cmax
                best_h = h.copy()
                last_best = time.time()
                no_improve = 0

        gsub = _active_grad(h, c, n, 24, 5e-4)
        pairs = _candidate_pairs(gsub, h, 18, 8)
        if it % 2 == 0:
            gsub2 = _active_grad(h, c, n, 48, 2e-3)
            pairs = _merge_pairs((pairs, _candidate_pairs(gsub2, h, 18, 8)), 12)

        improved = False
        old_cmax = float(cmax)
        best_f = old_cmax
        best_m = None

        for i, j in pairs[:12]:
            d, f = _line_search(c, h, i, j, n, L, g, mat)
            if f < best_f - 1e-12 and abs(d) > 1e-14:
                best_f = f
                best_m = (i, j, d)

        if best_m is not None:
            i, j, d = best_m
            _grad_pair(g, i, j, h, n)
            q1 = n - 1 + (j - i)
            q2 = n - 1 - (j - i)
            _apply_pair(h, c, i, j, d, g, q1, q2, n - 1, 2.0 / n)
            cmax = float(c.max())
            if cmax < best_c - 1e-12:
                best_c = cmax
                best_h = h.copy()
                last_best = time.time()
                no_improve = 0
            if cmax < old_cmax - 1e-12:
                no_improve = 0
            improved = True

        if not improved:
            for _ in range(4):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo <= 1e-14:
                    continue
                d, f = _line_search(c, h, i, j, n, L, g, mat)
                if f < old_cmax - 1e-12 and abs(d) > 1e-14:
                    _grad_pair(g, i, j, h, n)
                    q1 = n - 1 + (j - i)
                    q2 = n - 1 - (j - i)
                    _apply_pair(h, c, i, j, d, g, q1, q2, n - 1, 2.0 / n)
                    cmax = float(c.max())
                    if cmax < best_c - 1e-12:
                        best_c = cmax
                        best_h = h.copy()
                        last_best = time.time()
                        no_improve = 0
                    improved = True
                    break

        if improved:
            continue

        no_improve += 1

        if no_improve > 20:
            temp = 1e-4 + min(6e-4, no_improve * 1e-6)
            ok, cmax2 = _anneal_step(h, c, n, L, g, cmax, rng, temp)
            if ok:
                cmax = cmax2
                if cmax < best_c - 1e-12:
                    best_c = cmax
                    best_h = h.copy()
                    last_best = time.time()
                    no_improve = 0
                else:
                    no_improve = 1

        if no_improve > 120:
            sig = 0.05 + 0.05 * ((no_improve // 120) % 3)
            hp = np.clip(h + sig * rng.randn(n), 0.0, 1.0)
            hp = _proj_sum(hp, 0.5 * n)
            cp = _compute_c(hp, n)
            cpmax = float(cp.max())
            if cpmax < best_c - 1e-12:
                h = hp.copy()
                c = cp.copy()
                cmax = cpmax
                best_c = cpmax
                best_h = h.copy()
                last_best = time.time()
                no_improve = 0
            elif cpmax <= cmax + max(1e-4, 2e-5 * no_improve) and cpmax < best_c + 1e-3:
                h = hp.copy()
                c = cp.copy()
                cmax = cpmax
                no_improve = 0

        if it > 100 and time.time() - last_best > stall_s:
            break

    return best_h, best_c


def _optimize_start(h_start, n, rng, deadline, stall_s):
    h = np.array(h_start, dtype=float).copy()
    if len(h) != n:
        h = _interp_h(h, n)
    h = _repair(h, n)
    c = _compute_c(h, n)
    cmax = float(c.max())
    return _local_search(h, c, cmax, rng, deadline, stall_s)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + float(budget_s)
    rng = np.random.RandomState(seed)

    h = None
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    if h is None or len(h) == 0:
        h = _default_start(560)

    best_h = _repair(h, len(h))
    best_c = float(_eval(best_h))

    for sig in (0.05, 0.12, 0.22):
        hp = np.clip(best_h + sig * rng.randn(len(best_h)), 0.0, 1.0)
        hp = _proj_sum(hp, 0.5 * len(best_h))
        cp = float(_eval(hp))
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    d1 = min(t_end - 300.0, time.time() + 0.45 * budget_s)
    if d1 - time.time() > 30.0:
        h1, c1 = _optimize_start(best_h, len(best_h), rng, d1, 120.0)
        h1, c1 = _eval_fixed(h1)
        if c1 < best_c:
            best_c, best_h = c1, h1

    rem = t_end - time.time()
    if rem > 240.0:
        hp = np.clip(best_h + 0.10 * rng.randn(len(best_h)), 0.0, 1.0)
        hp = _proj_sum(hp, 0.5 * len(best_h))
        d2 = min(t_end - 160.0, time.time() + 0.22 * budget_s)
        if d2 - time.time() > 60.0:
            h2, c2 = _optimize_start(hp, len(best_h), rng, d2, 80.0)
            h2, c2 = _eval_fixed(h2)
            if c2 < best_c:
                best_c, best_h = c2, h2

    n_res = int(round(len(best_h) * 1.5))
    n_res = max(len(best_h) + 32, min(960, n_res))
    rem = t_end - time.time()
    if n_res > len(best_h) and rem > 260.0:
        hi = _interp_h(best_h, n_res)
        hi = _repair(hi, n_res)
        d3 = min(t_end - 80.0, time.time() + 0.25 * budget_s)
        if d3 - time.time() > 80.0:
            h3, c3 = _optimize_start(hi, n_res, rng, d3, 100.0)
            h3, c3 = _eval_fixed(h3)
            if c3 < best_c:
                best_c, best_h = c3, h3

    rem = t_end - time.time()
    if rem > 60.0:
        df = t_end - 3.0
        h4, c4 = _optimize_start(best_h, len(best_h), rng, df, 30.0)
        h4, c4 = _eval_fixed(h4)
        if c4 < best_c:
            best_c, best_h = c4, h4

    best_h = _repair(best_h, len(best_h))
    best_c = float(_eval(best_h))
    return best_h, best_c, len(best_h)
