import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

            if diff > 0:

            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_active(h, alpha, active):
    n = len(h)
    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c_full[active].max()
    s = np.clip(alpha * (c_full[active] - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(active)):
        if w[ii] < 1e-14:
            continue
        shift = active[ii] - (n - 1)
        wk = w[ii] * dx
        if shift >= 0:
            g[:n - shift] += wk * (1.0 - h[shift:])
            if shift > 0:
                g[shift:] -= wk * h[:n - shift]
        else:
            p = -shift
            g[p:] += wk * (1.0 - h[:n - p])
            if p < n:
                g[:n - p] -= wk * h[p:]
    return g

def _pgd(h, alpha, active, step, mom, vel, n):
    g = _grad_active(h, alpha, active)
    g -= g.mean()
    vel = mom * vel + (1.0 - mom) * g
    return _proj(h - step * vel, n * 0.5), vel

def _run_pgd(h, alpha, ka, max_it, t0, tmax, init_step=0.01, mom=0.7):
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    active = np.argpartition(c_full, -ka)[-ka:]
    vel = np.zeros(n)
    step = init_step
    cur_c = _eval(h)
    best = h.copy()
    best_c = cur_c
