    # Find s that minimizes g·s subject to 0≤s≤1, sum(s)=n/2

    if s < 1e-12:
        return np.full(n, 0.5)

            diff -= a
            if abs(diff) < 1e-12:

    return h

            best_c5, best_h = c5v, h3.copy()
        if time.time() < dl3 - 5:
            h3 = _nesterov(best_h.copy(), 400, 15000, 0.00015, t0, dl3)
            c5v = _eval(h3)
            if c5v < best_c5:
                best_c5, best_h = c5v, h3.copy()

    # Phase 4: Low-res escape (10% budget)
    dl4 = t0 + budget_s * 0.75
    if time.time() < dl4:
        lo = len(best_h) // 4
        if lo >= 64:
            h_lo = _downsample(best_h, len(best_h) // lo)
            h_lo = _repair(h_lo, lo)
            h_lo = _nesterov(h_lo, 10, 5000, 0.01, t0, dl4)
            h_hi = _repair(_upsample(h_lo, len(best_h) // lo), len(best_h))
            h_hi = _nesterov(h_hi, 20, 5000, 0.001, t0, dl4)
            c5v = _eval(h_hi)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_hi.copy()

    # Phase 5: Medium local search (15% budget)
    dl5 = t0 + budget_s * 0.90
    if time.time() < dl5:
        best_h, best_c5 = _local_search(best_h, t0, dl5, seed=42)

    # Phase 6: Heavy fine local search (remaining)
    dl6 = t0 + budget_s * 0.99
    if time.time() < dl6:
        best_h, best_c5 = _local_search_heavy(best_h, t0, dl6, seed=99)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
