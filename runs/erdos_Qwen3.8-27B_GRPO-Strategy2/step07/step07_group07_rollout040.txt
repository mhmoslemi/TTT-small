        v = beta * v + (1.0 - beta) * (g / gn)
        h_try = _proj_sum(h_c - step * v, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-16:
            h_c, best_c, best_h = h_try, c_try, h_c.copy()
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _steepest(h, n_iter, bc_fn):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    step = 0.015
    for _ in range(n_iter):
        if bc_fn():
            break
        c = _corrs(h_c)
        k_max = int(np.argmax(c))
        sh = (n - 1) - k_max
        g = np.zeros(n)
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h_c[i1 + sh])
            if sh > 0:
                g[sh:] -= h_c[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h_c[:n-sp])
            g[:n-sp] -= h_c[sp:]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_try = _proj_sum(h_c - step * g / gn, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c:
            best_c, best_h, h_c = c_try, h_c.copy(), h_try
            step = min(step * 1.04, 0.03)
        else:
            step *= 0.6
            if step < 1e-10:
                break
    return best_h, best_c

def _local2(h, n_iter, rng, bc_fn):
    n = len(h)
    h_c = h.copy().astype(float)
    best_c = _eval(h_c)
    best_h = h_c.copy()
    scale = 2e-5
    for i in range(n_iter):
        if bc_fn():
            break
        a = rng.randint(n)
        b = (a + 1 + rng.randint(0, 12)) % n
        d = rng.randn() * scale
        na, nb = h_c[a] + d, h_c[b] - d
        if 0 <= na <= 1 and 0 <= nb <= 1:
            h_c[a], h_c[b] = na, nb
            cn = _eval(h_c)
            if cn < best_c:
                best_c, best_h = cn, h_c.copy()
            else:
                h_c[a], h_c[b] = na - d, nb + d
        if i % 4000 == 0 and i > 0:
            scale *= 0.75
    return best_h, best_c

def _upsample(h, n_tgt):
    n = len(h)
    xs = np.linspace(0, 2, n, endpoint=False)
    xt = np.linspace(0, 2, n_tgt, endpoint=False)
    return _repair(np.interp(xt, xs, np.concatenate([h, [h[0]]])), n_tgt)

def _coarse_avg(h, n_c):
    n = len(h)
    r = n // n_c
    if r < 1:
        return h.copy()
    out = np.array([h[i*r:(i+1)*r].mean() for i in range(n_c)])
    return _repair(out, n_c)

def _sin_init(n, freq, amp, phase):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return _proj_sum(np.clip(h, 0, 1), n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def el():
        return time.time() - t0
    def bc():
        return el() > budget_s - 15

    n0 = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_c5 = _eval(best_h)

    # Phase 1: Quick L-BFGS passes on parent
    for p in [16, 64, 256, 1024]:
        if el() > 90:
            break
        ht = _lbfgs(best_h.copy(), p, 1e5, 350)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 2: Coarse-to-fine (key new idea)
    for nc in [80, 120, 160]:
        if el() > 320:
            break
        hc = _coarse_avg(best_h, nc)
        for p in [8, 64, 512]:
            if el() > 320:
                break
            hc = _lbfgs(hc, p, 1e4, 250)
        hc = _nesterov(hc, 3000, 2, 20, 0.03, bc)[0]
        hu = _upsample(hc, n0)
        hu = _lbfgs(hu, 128, 1e5, 250)
        ct = _eval(hu)
        if ct < best_c5:
            best_c5, best_h = ct, hu

    # Phase 3: Diverse sinusoidal starts
    for freq in [2, 3, 4, 5, 7, 10]:
        if el() > 450:
            break
        for amp in [0.35, 0.45]:
            if el() > 450:
                break
            hs = _sin_init(n0, freq, amp, rng.uniform(0, 6.28))
            hs = _lbfgs(hs, 32, 1e4, 200)
            hs = _lbfgs(hs, 256, 1e5, 200)
            ct = _eval(hs)
            if ct < best_c5:
                best_c5, best_h = ct, hs

    # Phase 4: Nesterov momentum
    if el() < budget_s - 250:
        ht, ct = _nesterov(best_h, 12000, 3, 60, 0.02, bc)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 5: High-p L-BFGS
    if el() < budget_s - 180:
        for p in [512, 2048, 4096]:
            if el() > budget_s - 180:
                break
            ht = _lbfgs(best_h.copy(), p, 1e7, 350)
            ct = _eval(ht)
            if ct < best_c5:
                best_c5, best_h = ct, ht

    # Phase 6: Steepest descent on max lag
    if el() < budget_s - 120:
        ht, ct = _steepest(best_h, 6000, bc)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 7: Nesterov fine
    if el() < budget_s - 100:
        ht, ct = _nesterov(best_h, 10000, 2, 25, 0.006, bc)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 8: Perturbation + re-optimize (escape local opt)
    if el() < budget_s - 60:
        for _ in range(3):
            if el() > budget_s - 60:
                break
            h_pert = best_h + rng.randn(n0) * 0.02
            h_pert = _repair(h_pert, n0)
            h_pert = _lbfgs(h_pert, 256, 1e6, 150)
            ct = _eval(h_pert)
            if ct < best_c5:
                best_c5, best_h = ct, h_pert

    # Phase 9: Fine local search
    if el() < budget_s - 30:
        nt = min(int((budget_s - 30 - el()) / 0.003), 50000)
        best_h, best_c5 = _local2(best_h, nt, rng, bc)

    # Phase 10: Final polish
    if el() < budget_s - 5:
        ht = _lbfgs(best_h.copy(), 2048, 1e8, 100)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
