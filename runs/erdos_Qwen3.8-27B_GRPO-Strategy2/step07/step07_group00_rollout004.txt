        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _obj_wr(x, alpha, topk):
    return _smax_obj(x, alpha, topk)

def _grad_wr(x, alpha, topk):
    return _grad_topk(x, alpha, topk)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.9
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            hn = _proj(h - step * v, n * 0.5)
            f = _smax_obj(hn, alpha, topk)
            h = hn
    return _repair(h, n)

def _lbfgs(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(_obj_wr, h, args=(alpha, topk),
                     jac=_grad_wr,
                     method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 20})
        return _repair(r.x, n)
    except Exception:
        return h

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    h_coarse = np.array([h[i*factor:(i+1)*factor].mean() for i in range(m)])
    return h_coarse, m

def _upsample(h_coarse, n):
    m = len(h_coarse)
    factor = n // m
    h_fine = np.repeat(h_coarse, factor)
    return h_fine[:n]

def _block_anneal(h, block_sizes, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = block_sizes[rng.randint(len(block_sizes))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        sd = 0.0015 * max(T, 1e-10) ** 0.5
        d = sd * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = 8e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _poly_perturb(h, n, rng, degree, amplitude):
    x = np.linspace(0, 2, n)
    pert = np.zeros(n)
    for d in range(1, degree + 1):
        pert += amplitude * rng.randn() * np.cos(d * np.pi * x)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B refinement (different optimizer than SLSQP)
    dl1 = min(deadline, t0 + 180)
    for alpha, tk, mi in [(3000, 60, 150), (10000, 40, 200), (50000, 20, 200), (200000, 10, 150)]:
        if time.time() > dl1:
            break
        h_new = _lbfgs(best_h, alpha, tk, mi, dl1)
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()

    # Phase 2: Multi-resolution (downsample-optimize-upsample)
    dl2 = min(deadline, t0 + 420)
    for factor in [2, 4]:
        if time.time() > dl2:
            break
        h_coarse, m = _downsample(best_h, factor)
        h_coarse = _repair(h_coarse, m)
        for alpha, tk, st, ns in [(30000, 10, 0.005, 800), (100000, 5, 0.002, 800)]:
            if time.time() > dl2:
                break
            h_coarse = _nesterov(h_coarse, alpha, tk, ns, st, dl2)
        h_up = _upsample(h_coarse, n0)
        for blend in [0.3, 0.5, 0.7]:
            h_blend = (1.0 - blend) * best_h + blend * h_up
            h_blend = _repair(np.clip(h_blend, 0, 1), n0)
            h_try = _nesterov(h_blend, 50000, 8, 400, 0.002, dl2)
            c5v = _eval(h_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_try.copy()

    # Phase 3: Aggressive Nesterov (very high alpha, few lags)
    dl3 = min(deadline, t0 + 680)
    nesterov_cfgs = [
        (50000, 8, 0.001, 1500),
        (300000, 5, 0.0005, 2000),
        (1000000, 4, 0.0003, 2000),
        (5000000, 3, 0.0001, 2500),
        (20000000, 2, 0.00005, 3000),
    ]
    for alpha, tk, st, ns in nesterov_cfgs:
        if time.time() > dl3:
            break
        h_new = _nesterov(best_h, alpha, tk, ns, st, dl3)
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()

    # Phase 4: Diverse structured restarts
    dl4 = min(deadline, t0 + 850)
    for trial in range(6):
        if time.time() > dl4:
            break
        if trial == 0:
            h_try = _poly_perturb(best_h, n0, rng, 7, 0.008)
        elif trial == 1:
            h_try = best_h + 0.004 * rng.randn(n0)
        elif trial == 2:
            h_try = best_h.copy()
            bsz = n0 // 6
            i1 = rng.randint(0, n0 - bsz)
            i2 = rng.randint(0, n0 - bsz)
            tmp = h_try[i1:i1+bsz].copy()
            h_try[i1:i1+bsz] = best_h[i2:i2+bsz]
            h_try[i2:i2+bsz] = tmp
        elif trial == 3:
            h_try = best_h.copy()
            bsz = n0 // 16
            i1 = rng.randint(0, n0 - bsz)
            h_try[i1:i1+bsz] = 1.0 - best_h[i1:i1+bsz]
        elif trial == 4:
            h_try = best_h.copy()
            bsz = n0 // 8
            i1 = rng.randint(0, n0 - bsz)
            h_try[i1:i1+bsz] = best_h[i1:i1+bsz][::-1]
        else:
            h_try = _poly_perturb(best_h, n0, rng, 12, 0.015)
        h_try = _repair(np.clip(h_try, 0, 1), n0)
        h_try = _nesterov(h_try, 100000, 6, 400, 0.001, dl4)
        c5v = _eval(h_try)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_try.copy()

    # Phase 5: Block annealing with varied sizes
    dl5 = min(deadline, t0 + 960)
    blocks = [1, 2, 3, 5, 7, 11, 16]
    h_a, c5_a = _block_anneal(best_h.copy(), blocks, 60000, rng, dl5, 0.004, 1e-10)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 6: Final ultra-aggressive Nesterov
    if time.time() < t0 + budget_s - 25:
        h_new = _nesterov(best_h, 100000000, 2, 3000, 0.00003, t0 + budget_s - 25)
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new.copy()

    # Phase 7: Fine 2-point polish
    if time.time() < t0 + budget_s - 3:
        rem = max(0, t0 + budget_s - 3 - time.time())
        n_pol = min(int(rem / 0.0008), 30000)
        best_h, best_c5 = _fine_polish(best_h, n_pol, rng, t0 + budget_s - 3)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
