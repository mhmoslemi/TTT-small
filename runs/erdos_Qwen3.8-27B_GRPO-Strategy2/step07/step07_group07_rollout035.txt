def _active_set_descent(h, n_iter, tol, step0):
    n = len(h)
    h_c = h.copy()
    best_c = _eval(h_c)
    best_h = h_c.copy()
    step = step0
    stall = 0
    for it in range(n_iter):
        c = _corrs(h_c)
        max_c = c.max()
        active = np.where(c > max_c - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        # Compute minimum-norm convex combination of active gradients
        # ∇c_k for each active k
        grads = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            grads[idx] = _single_corr_grad(h_c, k)
        # Solve: min ||G^T lambda||^2 s.t. lambda >= 0, sum lambda = 1
        # This is a QP: min lambda^T G G^T lambda s.t. sum lambda = 1, lambda >= 0
        M = grads @ grads.T  # m x m
        # Simple: if m is small, use scipy.optimize
        from scipy.optimize import linprog
        # Actually for small m, this is easy
        if len(active) <= 5:
            # Solve the QP with scipy
            from scipy.optimize import minimize as sci_min
            def q_obj(l):
                return l @ M @ l
            def q_grad(l):
                return 2 * M @ l
            cons = {'type': 'eq', 'fun': lambda l: l.sum() - 1, 'jac': lambda l: np.ones(len(l))}
            bnds = [(0, 1)] * len(active)
            l0 = np.ones(len(active)) / len(active)
            res = sci_min(q_obj, l0, jac=q_grad, constraints=cons, bounds=bnds, method='SLSQP', options={'maxiter': 100})
            lam = res.x
            lam = np.clip(lam, 0, 1)
            lam /= lam.sum()
        else:
            lam = np.ones(len(active)) / len(active)
        
        g = grads.T @ lam  # minimum-norm subgradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn  # normalize
        
        # Line search
        h_try = _proj_sum(h_c - step * g, n * 0.5)
        c_try = _eval(h_try)
        if c_try < best_c - 1e-14:
            h_c = h_try
            best_c = c_try
            best_h = h_c.copy()
            step = min(step * 1.05, 0.1)
            stall = 0

                h_c = _proj_sum(h_c + 1e-4 * np.random.randn(n), n * 0.5)

def _single_corr_grad(h, k):
    """Gradient of c_k w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        # c_k = sum_{i=0}^{n-1-sh} h[i]*(1-h[i+sh])*dx
        # dg[i] = (1-h[i+sh])*dx for i in [0, n-1-sh]
        # dg[i+sh] += -h[i]*dx for i in [0, n-1-sh]
        indices = np.arange(n - sh)
        g[indices] += (1.0 - h[indices + sh]) * dx
        if sh > 0:
            g[indices + sh] -= h[indices] * dx
    else:
        sp = -sh
        # c_k = sum_{i=sp}^{n-1} h[i]*(1-h[i-sp])*dx  (or equivalently sum_{i=0}^{n-1-sp} h[i+sp]*(1-h[i])*dx)

        g[indices + sp] += (1.0 - h[indices]) * dx
        g[indices] -= h[indices + sp] * dx

One more idea: **coordinate descent on the active set**. 

At each step:
1. Compute all c_k, find the top m (say m=5)
2. For each j in [0, n), compute the "improvement" if I change h[j] slightly
3. Adjust h[j] in the direction that decreases the max c_k

The improvement of changing h[j] by δ:
Δc_k ≈ δ * ∂c_k/∂h[j] for each active k
We want to choose δ to minimize max_k (c_k + δ * ∂c_k/∂h[j])

This is a 1D minimization for each coordinate. We can do coordinate descent: sweep through j=0,1,...,n-1 and at each, find the optimal δ_j.

The optimal δ_j: we want to minimize max over active k of (c_k + δ * g_kj) where g_kj = ∂c_k/∂h[j].
If all g_kj < 0 (decreasing h[j] helps all), then we want δ > 0 (increase h[j]).
If mixed signs, we need to find the δ that equalizes the active constraints.

This is a small 1D problem that can be solved by bisection or analytically.

But doing this for all n coordinates is O(n * n_active) per sweep, and we need many sweeps. Let me estimate: n=560, n_active=5, one sweep = 2800 ops. 1000 sweeps = 2.8M ops. Very fast!

Let me implement coordinate descent:

```python
def _coord_descent(h, n_sweeps, n_active, el, bc):
    n = len(h)
    dx = 2.0/n
    hc = h.copy()
    bc5 = _eval(hc)
    bh = hc.copy()
    for sw in range(n_sweeps):
        if bc(): break
        # Compute active set
        c = _corrs(hc)
        amax = c.max()
        active = np.argsort(c)[-n_active:]
        # For each active k, precompute the shift
        shifts = [(n-1)-k for k in active]
        # Coordinate sweep
        changed = False
        for j in range(n):
            # Find optimal delta for h[j]
            # c_k(new) = c_k + delta * dc_k/dh[j] for each active k
            # dc_k/dh[j]:
            gjs = np.zeros(n_active)
            for ii, k in enumerate(active):
                sh = shifts[ii]
                g = 0.0
                if sh >= 0:
                    if 0 <= j < n-sh:
                        g += (1.0 - hc[j+sh]) * dx
                    if sh <= j < n:
                        g -= hc[j-sh] * dx
                else:
                    sp = -sh
                    if sp <= j < n:
                        g += (1.0 - hc[j-sp]) * dx
                    if 0 <= j < n-sp:
                        g -= hc[j+sp] * dx
                gjs[ii] = g
            # Minimize max_i (c[active[i]] + delta * gjs[i]) s.t. 0 <= hc[j]+delta <= 1
            # This is a 1D convex problem (max of linear functions)
            lo = -hc[j]
            hi = 1.0 - hc[j]
            # Bisection on the subgradient
            # f(delta) = max_i (c_a[i] + delta * gjs[i])
            # f is convex, minimized where 0 is in subdifferential
            # Subdifferential at delta: conv{gjs[i] : f_i(delta) = f(delta)}
            # f is minimized when min gjs[i] (among active at opt) <= 0 <= max gjs[i]
            # Or all gjs < 0 → optimal at hi
            # Or all gjs > 0 → optimal at lo
            if np.all(gjs < 0):
                delta = hi
            elif np.all(gjs > 0):
                delta = lo
            else:
                # Bisection
                for _ in range(30):
                    mid = 0.5*(lo+hi)
                    vals = c[active] + mid * gjs
                    am = vals.max()
                    # Subgradient at mid: weighted avg of gjs where vals == am
                    # But for bisection, we can use: if all active gjs < 0, go right; else go left
                    # Better: find the point where the max changes
                    # f(mid) derivative (from right): max of gjs[i] where c_a[i] + mid*gjs[i] == max
                    act_at_mid = np.where(np.abs(vals - am) < 1e-12)[0]
                    if len(act_at_mid) == 0:
                        act_at_mid = np.array([np.argmax(vals)])
                    sg_max = gjs[act_at_mid].max()
                    sg_min = gjs[act_at_mid].min()
                    if sg_min <= 0 <= sg_max:
                        lo = hi = mid
                        break
                    elif sg_max < 0:
                        lo = mid
                    else:
                        hi = mid
            new_val = hc[j] + delta
            hc[j] = new_val
            # Quick check
        # After full sweep, check if improved
        c_new = _eval(hc)
        if c_new < bc5:
            bc5 = c_new
            bh = hc.copy()
        else:
            hc = bh.copy()  # revert
    return bh, bc5
```

This is O(n_sweeps * n * n_active * 30) for the bisection. For n_sweeps=2000, n=560, n_active=5, bisection=30: 2000*560*5*30 = 168M operations. That might be too slow in Python.

The inner loop (computing gjs for each j) is the bottleneck. Let me vectorize it:

For a given set of shifts, I can precompute which j values are affected by each shift:
- For shift sh ≥ 0: j in [0, n-sh) has forward contribution, j in [sh, n) has backward
- For shift sh < 0 (sp = -sh): j in [sp, n) has forward, j in [0, n-sp) has backward

I can precompute the gradient matrix G[j, i] = ∂c_{active[i]}/∂h[j] for all j and active i. This is an n × n_active matrix. Computing it once is O(n * n_active).

Then for each j, the 1D problem is: minimize max_i (c_a[i] + δ * G[j,i]) s.t. lo ≤ δ ≤ hi.

This can be vectorized if I do the bisection for all j simultaneously... but the bounds (lo, hi) depend on j.

Actually, let me just precompute G and then do the coordinate sweep more efficiently:

```python
def _compute_grad_matrix(h, active_indices):
    n = len(h)
    dx = 2.0/n
    G = np.zeros((n, len(active_indices)))
    for ii, k in enumerate(active_indices):
        sh = (n-1) - k
        if sh >= 0:
            idx = np.arange(n-sh)
            G[idx, ii] += (1.0 - h[idx+sh]) * dx
            if sh > 0:
                G[idx+sh, ii] -= h[idx] * dx
        else:
            sp = -sh
            idx = np.arange(n-sp)
            G[idx+sp, ii] += (1.0 - h[idx]) * dx
            G[idx, ii] -= h[idx+sp] * dx
    return G
```

This is O(n * n_active) and fast.

Then the coordinate sweep:
For each j:
- lo = -h[j], hi = 1-h[j]
- Solve: min max_i (c_a[i] + δ*G[j,i]) s.t. lo ≤ δ ≤ hi

The 1D problem can be solved in O(n_active * log(1/ε)) by bisection. With n_active ≤ 10 and ε = 1e-10, that's ~30 iterations.

Total: n_sweeps * n * n_active * 30 = 2000 * 560 * 5 * 30 = 168M scalar ops. In Python, this is too slow (~170s+). In numpy with vectorization, much faster.

Let me vectorize the bisection:

```python
def _coord_sweep(hc, c_active, G, lo_j, hi_j):
    """For each coordinate j, find optimal delta."""
    n = len(hc)
    m = len(c_active)
    deltas = np.zeros(n)
    for j in range(n):
        lo, hi = lo_j[j], hi_j[j]
        gj = G[j]  # shape (m,)
        if np.all(gj < 0):
            deltas[j] = hi
            continue
        if np.all(gj > 0):
            deltas[j] = lo
            continue
        # Bisection
        for _ in range(25):
            mid = 0.5*(lo+hi)
            vals = c_active + mid * gj
            am = vals.max()
            act = np.where(np.abs(vals - am) < 1e-10)[0]
            if act.size == 0:
                act = np.array([np.argmax(vals)])
            if gj[act].min() <= 0 <= gj[act].max():
                lo = hi = mid
                break
            elif gj[act].max() < 0:
                lo = mid
            else:
                hi = mid
        deltas[j] = 0.5*(lo+hi)
    return deltas
```
