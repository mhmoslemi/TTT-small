    """Project h onto {0<=x<=1, sum(x)=target}"""
    lo, hi = -3.0, 3.0
    for _ in range(60):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

            if rem < 1e-13:
                break
            if d > 0:
                a = min(rem, 1.0 - h[i])
                h[i] += a
            else:
                a = min(rem, h[i])
                h[i] -= a
            rem -= a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _srb(x, hlf):
    return x.sum() - hlf

def _srbj(x, hlf):
    return np.ones(len(x))

def _objf(x, a):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / a

def _gradf(x, a):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(a * (c - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp(h, a, mi):
    n = len(h)
    try:
        r = minimize(_objf, h, jac=_gradf, args=(a,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _srb, 'jac': _srbj, 'args': (n * 0.5,)}],
                     options={'maxiter': mi, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _proj(h, tgt):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    bc = _eval(h)
    bh = h.copy()

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP with increasing alpha
    for a in [15000, 150000, 1500000]:
        if _el() > 150:
            break
        h2 = _slsqp(bh, a, 1500)
        c2 = _eval(h2)
        if c2 < bc:
            bc = c2
            bh = h2.copy()

    # Phase 2: Gradient-guided 2-point search targeting worst lag
    h2 = bh.copy()
    c2 = bc
    t_end2 = min(420, budget_s - 400)
    steps_done = 0
    while _el() < t_end2:
        c_all = np.correlate(h2, 1.0 - h2, mode='full') * (2.0 / n)
        kst = int(np.argmax(c_all))
        sh = (n - 1) - kst
        if sh == 0:
            gi = 1.0 - 2.0 * h2
        elif sh > 0:
            gi = np.zeros(n)
            gi[:n - sh] = 1.0 - h2[sh:]
            gi[sh:] -= h2[:n - sh]
        else:
            sp = -sh
            gi[sp:] = 1.0 - h2[:n - sp]
            gi[:n - sp] += h2[sp:]
        i = int(np.argmin(gi))
        j = int(np.argmax(gi))
        if i == j:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 9)) % n
        if rng.rand() < 0.35:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 7)) % n
            if i == j:
                continue
        steps_done += 1
        prog = min(steps_done / 100000.0, 1.0)
        step = 5e-5 * (1.0 - 0.6 * prog)
        d = step * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c2:
                c2 = cn
            else:
                h2[i], h2[j] = ni - d, nj + d
    h2 = _repair(h2, n)
    c2 = _eval(h2)
    if c2 < bc:
        bc = c2
        bh = h2.copy()

    # Phase 3: Adaptive-step random 3-point search
    h3 = bh.copy()
    c3 = bc
    t_end3 = min(700, budget_s - 300)
    it3 = 0
    while _el() < t_end3:
        it3 += 1
        prog = min(it3 / 200000.0, 1.0)
        step = 6e-6 * (1.0 - 0.7 * prog)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 10)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h3[i] + d1, h3[j] + d2, h3[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h3[i], h3[j], h3[k] = ni, nj, nk
            cn = _eval(h3)
            if cn < c3:
                c3 = cn
            else:
                h3[i], h3[j], h3[k] = ni - d1, nj - d2, nk + d1 + d2
    h3 = _repair(h3, n)
    if _eval(h3) < bc:
        bc = _eval(h3)
        bh = h3.copy()

    # Phase 4: High-alpha SLSQP polish
    if _el() < budget_s - 120:
        for a in [5000000, 20000000]:
            if _el() > budget_s - 80:
                break
            h4 = _slsqp(bh, a, 600)
            c4 = _eval(h4)
            if c4 < bc:
                bc = c4
                bh = h4.copy()

    # Phase 5: Ultra-fine 2-point polish with multiple scales
    h5 = bh.copy()
    c5 = bc
    scales = [3e-7, 2e-7, 1e-7, 5e-8]
    for sc in scales:
        if _el() > budget_s - 2:
            break
        for _ in range(300000):
            if _el() > budget_s - 1:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            d = sc * rng.randn()
            ni, nj = h5[i] + d, h5[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h5[i], h5[j] = ni, nj
                cn = _eval(h5)
                if cn < c5:
                    c5 = cn
                else:
                    h5[i], h5[j] = h5[i] - d, h5[j] + d
    h5 = _repair(h5, n)
    c5 = _eval(h5)
    if c5 < bc:
        bc = c5
        bh = h5.copy()

    bh = _repair(bh, n)
    bc = _eval(bh)
    return (bh, bc, n)
