def _softmax_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -700, 700)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _softmax_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -700, 700)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * 2.0 / n
        if sh >= 0:

            g[i1] += wk * (1.0 - x[i1 + sh])

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum_box(x, target):
    """Project x onto {0<=x<=1, sum(x)=target} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h = _proj_sum_box(h, target)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order[:500]:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _softmax_obj(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -700, 700)
    sm = s - s.max()
    return m + np.log(np.sum(np.exp(sm))) / alpha

def _softmax_grad(x, alpha, n):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -700, 700)
    sm = s - s.max()
    w = np.exp(sm)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - x[:cnt])
            g[:cnt] -= wk * x[sp:]
    return g

def _fista(h, alpha, n_steps, t0, t_end):
    """FISTA with soft-max objective and projection."""
    n = len(h)
    target = n * 0.5
    y = h.copy()
    x = h.copy()
    t = 1.0
    step = 0.05
    for i in range(n_steps):
        if time.time() > t_end:
            break
        g = _softmax_grad(y, alpha, n)
        # Gradient step + projection
        y_new = _proj_sum_box(y - step * g, target)
        f_new = _eval(y_new)
        f_cur = _eval(x)
        if f_new < f_cur:
            x = y_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.5
            if step < 1e-8:
                break
            y_new = _proj_sum_box(y - step * g, target)
            x = y_new
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = x + (t - 1) / t_new * (x - y)
        y = _proj_sum_box(y, target)  # Keep y feasible
        t = t_new
    return _repair(x, n)

def _eval(h):

def _topk_grad(h, k_top):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]

def _nesterov(h, k_top, n_steps, step0, t0, t_end):

        g = _topk_grad(h, k_top)

import numpy as np
from scipy.optimize import minimize
import time

def _proj_sum_box(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = _proj_sum_box(h, target)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order[:400]:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh; cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt]); g[:cnt] -= wk * h[sp:]
    return g

def _nesterov(h, k_top, step0, t_end):
    n = len(h); target = n * 0.5
    f = _eval(h); v = np.zeros(n); mu = 0.85; step = step0
    while time.time() < t_end:
        g = _topk_grad(h, k_top)
        hs = h.copy()
        v = mu * v + (1.0 - mu) * g
        hnew = _proj_sum_box(h - step * v, target)
        fn = _eval(hnew)
        if fn < f:
            h = hnew; f = fn; step = min(step * 1.02, step0 * 5)
        else:
            step *= 0.5
            if step < 1e-10: break
            h = _proj_sum_box(hs - step * v, target); f = _eval(h)
    return _repair(h, n)

def _full_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -700, 700)
    sm = s - s.max()
    w = np.exp(sm); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        sh = (n - 1) - k; wk = w[k] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - x[sh:])
            if sh > 0: g[sh:] -= wk * x[:cnt]
        else:
            sp = -sh; cnt = n - sp
            g[sp:] += wk * (1.0 - x[:cnt]); g[:cnt] -= wk * x[sp:]
    return g

def _slsqp_smo(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -700, 700); sm = s - s.max()
        return m + np.log(np.sum(np.exp(sm))) / alpha
    def grad(x):
        return _full_grad(x, alpha, n)
    def cfun(x): return x.sum() - target
    def cjac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return _repair(h, n)
