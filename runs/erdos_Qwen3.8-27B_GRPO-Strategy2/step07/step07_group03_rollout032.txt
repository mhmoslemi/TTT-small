def _line_min_candidate(a,b,lo,hi):
    t=len(a)
    pts=[lo,hi]
    if lo < 0.0 < hi: pts.append(0.0)
    for p in range(t):
      ap=a[p]; bp=b[p]
      for q in range(p+1,t):
        db=bp-b[q]
        if abs(db)>1e-12:
           x=(a[q]-ap)/db
           if x > lo and x < hi: pts.append(x)
    best=None; bestv=None
    for x in pts:
       vals=a+b*x
       v=float(np.max(vals))
       if bestv is None or v<bestv-1e-15 or (abs(v-bestv)<=1e-15 and abs(x)<abs(best)):
          best=x; bestv=v
    if best is None: return 0.0
    if best < lo: return lo
    if best > hi: return hi
    return float(best)

def _grad_pair_top(h, shifts, i, j, n, dx):

import numpy as np
import time
import math


def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    h = _proj_to_sum(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                if a <= 0.0:
                    continue
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                if a <= 0.0:
                    continue
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h


def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _cvec(h, q=None, n=None):
    if n is None:
        n = len(h)
    if q is None:
        q = 1.0 - h
    c = np.correlate(h, q, mode='full') * (2.0 / float(n))
    return c, float(c.max())


def _eval(h):
    return _cvec(h)[1]


def _random_balanced(n, rng):
    h = np.zeros(n, dtype=float)
    if n <= 0:
        return h
    idx = np.argsort(rng.rand(n))[::-1]
    h[idx[:n // 2]] = 1.0
    return h


def _binarize(h, n):
    out = np.zeros(n, dtype=float)
    if n <= 0:
        return out
    idx = np.argsort(np.asarray(h, dtype=float))[::-1]
    out[idx[:n // 2]] = 1.0
    return out


def _downsample(h, factor):
    n = len(h)
    m = n // factor
    if m < 1:
        m = 2
        return _random_balanced(m, np.random.RandomState(12345))
    if m * factor < n:
        h = np.asarray(h, dtype=float)[:m * factor]
    else:
        h = np.asarray(h, dtype=float)
    hh = h.reshape(m, factor).mean(axis=1)
    return _repair(hh, m)


def _better(new_h, new_c, old_h, old_c, old_n):
    try:
        new_c = float(new_c)
    except Exception:
        return old_h, old_c, old_n
    if new_h is None or not math.isfinite(new_c):
        return old_h, old_c, old_n
    if new_c < float(old_c):
        return np.asarray(new_h, dtype=float).copy(), new_c, len(new_h)
    return old_h, old_c, old_n


def _grad_pair_top(h, shifts, i, j, n, dx):
    t = len(shifts)
    b = np.empty(t, dtype=float)
    hp = h
    for r in range(t):
        sh = shifts[r]
        vi = 0.0
        p = i + sh
        if 0 <= p < n:
            vi += 1.0 - hp[p]
        p = i - sh
        if 0 <= p < n:
            vi -= hp[p]
        vj = 0.0
        p = j + sh
        if 0 <= p < n:
            vj += 1.0 - hp[p]
        p = j - sh
        if 0 <= p < n:
            vj -= hp[p]
        b[r] = dx * (vi - vj)
    return b


def _line_min_candidate(a, b, lo, hi):
    if lo >= hi:
        return 0.0
    pts = [lo, hi]
    if lo < 0.0 < hi:
        pts.append(0.0)
    t = len(a)
    for p in range(t):
        ap = a[p]
        bp = b[p]
        for q in range(p + 1, t):
            db = bp - b[q]
            if abs(db) > 1e-12:
                x = (a[q] - ap) / db
                if lo < x < hi:
                    pts.append(float(x))
    best_x = 0.0
    best_v = None
    for x in pts:
        v = float(np.max(a + b * x))
        if best_v is None or v < best_v - 1e-15:
            best_v = v
            best_x = x
        elif abs(v - best_v) <= 1e-15 and abs(x) < abs(best_x):
            best_x = x
    if best_x < lo:
        return lo
    if best_x > hi:
        return hi
    return float(best_x)


def _active_local(h, duration_s, rng, top_T=8, init_T=0.004,
                  no_impr_limit=50000, small_cap=0.1, big_cap=0.9,
                  big_frac=0.3, cool=0.9999, min_T=1e-6,
                  pert_scale=0.08, allow_perturb=True):
    h = _repair(np.asarray(h, dtype=float).copy(), len(h))
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    q = 1.0 - h
    c, m = _cvec(h, q, n)
    current = m
    best_h = h.copy()
    best_c = m
    T = float(init_T)
    no_impr = 0
    move_count = 0
    end = time.time() + duration_s

    while time.time() < end:
        if allow_perturb and no_impr > no_impr_limit:
            if no_impr > 4 * no_impr_limit and rng.random() < 0.65:
                order = np.argsort(h)[::-1]
                h = np.zeros(n, dtype=float)
                h[order[:n // 2]] = 1.0
                no_impr = max(0, no_impr - 3 * no_impr_limit)
            else:
                h = h + pert_scale * rng.randn(n)
                h = _proj_to_sum(h, target)
                no_impr = max(0, no_impr - no_impr_limit)
            q = 1.0 - h
            c, m = _cvec(h, q, n)
            current = m
            T = max(T, 0.5 * max(init_T, 1e-6))
            if m < best_c:
                best_c = m
                best_h = h.copy()
                no_impr = 0

        top_idx = np.argsort(c)[-top_T:]
        shifts = (n - 1) - top_idx
        a_top = c[top_idx]

        chosen = False
        for attempt in range(4):
            if rng.random() < 0.18:
                i = rng.randint(n)
                w = max(1, n // 10)
                j = (i + 1 + rng.randint(0, w)) % n
                if i == j:
                    j = (j + 1) % n
            else:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
            if abs(h[i] - h[j]) > 0.02 or attempt == 3:
                chosen = True
                break
        if not chosen:
            continue

        lo = max(-h[i], h[j] - 1.0) + 1e-12
        hi = min(1.0 - h[i], h[j]) - 1e-12
        if lo >= hi:
            continue

        roll = rng.random()
        if roll < 0.20:
            d = rng.uniform(lo, hi)
        elif roll < 0.40:
            d = -0.25 * (h[i] - h[j])
        elif roll < 0.60:
            if rng.random() < 0.5:
                d = min(1.0 - h[i], h[j]) - 1e-12
            else:
                d = -(min(h[i], 1.0 - h[j]) - 1e-12)
        else:
            b = _grad_pair_top(h, shifts, i, j, n, dx)
            d = _line_min_candidate(a_top, b, lo, hi)
            if abs(d) < 1e-10:
                d = (hi - lo) * 0.05 * (1.0 if rng.random() < 0.5 else -1.0)

        cap = big_cap if rng.random() < big_frac else small_cap
        if abs(d) > cap:
            d = math.copysign(cap, d)
        if d >= hi:
            d = hi
        if d <= lo:
            d = lo
        if abs(d) < 1e-13:
            continue

        old_i = h[i]
        old_j = h[j]
        new_i = old_i + d
        new_j = old_j - d
        h[i] = new_i
        h[j] = new_j
        q[i] = 1.0 - new_i
        q[j] = 1.0 - new_j

        c_new, m_new = _cvec(h, q, n)
        delta = m_new - current
        accept = False
        if delta < -1e-14:
            accept = True
        elif T > min_T:
            if delta <= 0.0:
                accept = True
            else:
                z = delta / T
                if z < 50.0 and rng.random() < math.exp(-z):
                    accept = True

        if accept:
            c = c_new
            current = m_new
            if m_new < best_c:
                best_c = m_new
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
            T = max(min_T, T * cool)
        else:
            h[i] = old_i
            h[j] = old_j
            q[i] = 1.0 - old_i
            q[j] = 1.0 - old_j
            no_impr += 1
            T = max(min_T, T * 0.99995)

        move_count += 1
        if move_count % 8192 == 0:
            if abs(h.sum() - target) > 1e-8:
                h = _repair(h, n)
                q = 1.0 - h
                c, m = _cvec(h, q, n)
                current = m
                if m < best_c:
                    best_c = m
                    best_h = h.copy()

    h = _repair(h, n)
    m = _eval(h)
    if m < best_c:
        best_h = h.copy()
        best_c = m
    return best_h, float(best_c)


def _binary_sa(h, duration_s, rng, init_T=0.02, min_T=0.001,
               cool=0.99995, restart_limit=40000):
    n = len(h)
    h = _repair(np.asarray(h, dtype=float).copy(), n)
    h = _binarize(h, n)
    ones = []
    zeros = []
    for k in range(n):
        if h[k] > 0.5:
            ones.append(k)
        else:
            zeros.append(k)
    q = 1.0 - h
    c, m = _cvec(h, q, n)
    current = m
    best_h = h.copy()
    best_c = m
    no_impr = 0
    T = float(init_T)
    end = time.time() + duration_s

    while time.time() < end:
        if no_impr > restart_limit:
            if no_impr > 4 * restart_limit:
                h = _random_balanced(n, rng)
                ones = [k for k in range(n) if h[k] > 0.5]
                zeros = [k for k in range(n) if h[k] <= 0.5]
                no_impr = restart_limit // 2
            else:
                kk = max(1, n // 64)
                for _ in range(kk):
                    p = rng.randint(len(ones))
                    pz = rng.randint(len(zeros))
                    i = ones[p]
                    j = zeros[pz]
                    h[i] = 0.0
                    h[j] = 1.0
                    q[i] = 1.0
                    q[j] = 0.0
                    ones[p] = j
                    zeros[pz] = i
            q = 1.0 - h
            c, m = _cvec(h, q, n)
            current = m
            T = max(T, 0.5 * max(init_T, 1e-5))
            if m < best_c:
                best_c = m
                best_h = h.copy()
                no_impr = 0

        if not ones or not zeros:
            break

        p = rng.randint(len(ones))
        pz = rng.randint(len(zeros))
        i = ones[p]
        j = zeros[pz]

        h[i] = 0.0
        h[j] = 1.0
        q[i] = 1.0
        q[j] = 0.0
        ones[p] = j
        zeros[pz] = i

        c_new, m_new = _cvec(h, q, n)
        delta = m_new - current
        accept = False
        if delta < -1e-14:
            accept = True
        elif delta <= 0.0:
            accept = True
        elif T > min_T:
            z = delta / T
            if z < 50.0 and rng.random() < math.exp(-z):
                accept = True

        if accept:
            c = c_new
            current = m_new
            if m_new < best_c:
                best_c = m_new
                best_h = h.copy()
                no_impr = 0
            else:
                no_impr += 1
            T = max(min_T, T * cool)
        else:
            h[i] = 1.0
            h[j] = 0.0
            q[i] = 0.0
            q[j] = 1.0
            ones[p] = i
            zeros[pz] = j
            no_impr += 1
            T = max(min_T, T * 0.99995)

    h = _repair(h, n)
    m = _eval(h)
    if m < best_c:
        best_h = h.copy()
        best_c = m
    return best_h, float(best_c)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + float(budget_s)
    rng = np.random.RandomState(int(seed))

    base_arr = None
    try:
        base_arr = np.asarray(initial_h_values, dtype=float)
    except Exception:
        base_arr = None

    if base_arr is None or base_arr.size == 0:
        base_arr = _random_balanced(560, rng)

    n0 = int(len(base_arr))
    h0 = _repair(base_arr.copy(), n0)
    best_h = h0.copy()
    best_c = _eval(h0)
    best_n = n0
    b = float(budget_s)

    rem = end - time.time()
    if rem > 22:
        alloc = min(rem, max(8.0, 0.06 * b))
        hb = _binarize(h0, n0)
        hs, cs = _binary_sa(hb, alloc, np.random.RandomState(int(seed) + 11))
        best_h, best_c, best_n = _better(hs, cs, best_h, best_c, best_n)

    rem = end - time.time()
    if rem > 45:
        alloc = min(rem, max(30.0, 0.32 * b))
        hs, cs = _active_local(
            best_h.copy(), alloc, np.random.RandomState(int(seed) + 21),
            top_T=8, init_T=0.005, no_impr_limit=60000,
            small_cap=0.10, big_cap=0.9, big_frac=0.32, pert_scale=0.08
        )
        best_h, best_c, best_n = _better(hs, cs, best_h, best_c, best_n)

    rem = end - time.time()
    if rem > 40:
        alloc = min(rem, max(20.0, 0.16 * b))
        hbin = _binarize(best_h, best_n)
        hs, cs = _active_local(
            hbin, alloc, np.random.RandomState(int(seed) + 31),
            top_T=10, init_T=0.003, no_impr_limit=50000,
            small_cap=0.20, big_cap=1.0, big_frac=0.5, pert_scale=0.10
        )
        best_h, best_c, best_n = _better(hs, cs, best_h, best_c, best_n)

    for fac, frac, seed_off in ((2, 0.15, 100), (4, 0.07, 200)):
        rem = end - time.time()
        if rem < 38:
            break
        nlow = n0 // fac
        if nlow % 2:
            nlow -= 1
        if nlow < 80:
            continue
        sr = np.random.RandomState(int(seed) + seed_off)
        if n0 % fac == 0:
            hlow = _downsample(base_arr, fac)
        else:
            hlow = _random_balanced(nlow, sr)
        nlow = int(len(hlow))
        hlow = _repair(hlow, nlow)

        rem = end - time.time()
        if rem < 24:
            break
        alloc = min(rem, max(15.0, frac * b))
        hs, cs = _active_local(
            hlow, alloc, np.random.RandomState(int(seed) + seed_off + 1),
            top_T=8, init_T=0.006, no_impr_limit=40000,
            small_cap=0.15, big_cap=1.0, big_frac=0.42, pert_scale=0.12
        )
        best_h, best_c, best_n = _better(hs, cs, best_h, best_c, best_n)

        rem = end - time.time()
        if rem > 22:
            alloc2 = min(rem, max(8.0, 0.045 * b))
            hbinlow = _binarize(hs, len(hs))
            hs2, cs2 = _active_local(
                hbinlow, alloc2, np.random.RandomState(int(seed) + seed_off + 2),
                top_T=10, init_T=0.003, no_impr_limit=30000,
                small_cap=0.20, big_cap=1.0, big_frac=0.5, pert_scale=0.10
            )
            best_h, best_c, best_n = _better(hs2, cs2, best_h, best_c, best_n)

    rem = end - time.time()
    if rem > 20:
        hbest = _repair(best_h, best_n)
        hs, cs = _active_local(
            hbest, rem, np.random.RandomState(int(seed) + 333),
            top_T=12, init_T=0.0, no_impr_limit=1000000000,
            small_cap=0.01, big_cap=0.12, big_frac=0.20, allow_perturb=False
        )
        best_h, best_c, best_n = _better(hs, cs, best_h, best_c, best_n)

    best_h = _repair(best_h, best_n)
    best_c = _eval(best_h)
    return (best_h, float(best_c), int(best_n))
