def _gold_section(h, i, j, grads, a_vals):
    gr = (np.sqrt(5) - 1) / 2
    d_lo = max(-h[i], h[j] - 1.0) + 1e-14
    d_hi = min(1.0 - h[i], h[j]) - 1e-14
    if d_lo >= d_hi:
        return 0.0
    b = grads[:, i] - grads[:, j]
    x1 = d_hi - gr * (d_hi - d_lo)
    x2 = d_lo + gr * (d_hi - d_lo)
    f1 = np.max(a_vals + b * x1)
    f2 = np.max(a_vals + b * x2)
    for _ in range(28):
        if f1 < f2:
            d_hi, f2, x2 = x2, f1, x1
            x1 = d_hi - gr * (d_hi - d_lo)
            f1 = np.max(a_vals + b * x1)
        else:
            d_lo, f1, x1 = x1, f2, x2
            x2 = d_lo + gr * (d_hi - d_lo)
            f2 = np.max(a_vals + b * x2)
    return (d_lo + d_hi) / 2.0

def _nesterov(h, alpha, steps, step0, t_end):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            g += w[k] * _grad_k(h, k, n, dx)
        g *= alpha
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _proj_to_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.04, step0 * 12.0)
        else:
            step *= 0.45
            if step < 1e-12:
                break
            h = _proj_to_sum(h_prev - step * v, n * 0.5)
            f = _eval(h)
    return h

def _coord_desc(h, n, duration, top_T, rng, t_end):
    dx = 2.0 / n
    best_c5 = _eval(h)
    end_t = min(time.time() + duration, t_end - 4)
    stall = 0
    while time.time() < end_t:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_T:]
        grads = np.array([_grad_k(h, k, n, dx) for k in top_idx])
        a_vals = c[top_idx]
        improved = False
        n_trials = max(300, n * 3)
        for _ in range(n_trials):
            if time.time() > end_t:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_opt = _gold_section(h, i, j, grads, a_vals)
            if abs(d_opt) > 1e-12:
                h[i] += d_opt
                h[j] -= d_opt
                improved = True
        if improved:
            st = h.sum()
            if abs(st - n * 0.5) > 1e-8:
                h = _proj_to_sum(h, n * 0.5)
            new_c5 = _eval(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 25:
            h += 2e-4 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
            stall = 0
    return h, best_c5

def _peel(h, n, duration, top_k, rng, t_end):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(h)
    end_t = min(time.time() + duration, t_end - 3)
    while time.time() < end_t:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += _grad_k(h, k, n, dx)
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        lo, hi = -1.0, 1.0
        best_f = _eval(h)
        best_step = 0.0
        for _ in range(40):
            mid = -(lo + hi) * 0.5
            h_try = _proj_to_sum(h + mid * g * 0.05, n * 0.5)
            f_try = _eval(h_try)
            if f_try <= best_f:
                best_f = f_try
                best_step = mid
                lo = mid
            else:
                hi = mid
        h_try = _proj_to_sum(h + best_step * g * 0.05, n * 0.5)
        c_new = _eval(h_try)
        if c_new < best_c5 - 1e-15:
            h = h_try
            best_c5 = c_new
            best_h = h.copy()
        else:
            h += 1e-3 * rng.randn(n)
            h = _proj_to_sum(h, n * 0.5)
    return best_h, best_c5

def _local_search(h, n, n_iter, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    c5c = best_c
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 8e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: Nesterov (10% budget)
    p1_end = t0 + budget_s * 0.10
    for alpha in [3000, 10000, 40000, 120000]:
        if time.time() > p1_end:
            break
        h = _nesterov(h, alpha, 500, 0.003, p1_end)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Wide coordinate descent (35% budget)
    p2_end = t0 + budget_s * 0.45
    for top_T in [3, 5, 8]:
        if time.time() > p2_end:
            break
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, budget_s * 0.10, top_T, rng, p2_end)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Peeling (15% budget)
    p3_end = t0 + budget_s * 0.62
    for top_k in [3, 5]:
        if time.time() > p3_end:
            break
        h = best_h.copy()
        h, c5v = _peel(h, n, budget_s * 0.07, top_k, rng, p3_end)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 4: More coordinate descent with different top_T (20% budget)
    p4_end = t0 + budget_s * 0.82
    for top_T in [4, 6, 2, 10]:
        if time.time() > p4_end:
            break
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, budget_s * 0.05, top_T, rng, p4_end)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Fine local search (10% budget)
    p5_end = t0 + budget_s * 0.92
    if time.time() < p5_end:
        remaining = p5_end - time.time()
        n_tries = min(int(remaining / 0.003), 60000)
        if n_tries > 200:
            h = best_h.copy()
            best_h, best_c5 = _local_search(h, n, n_tries, rng, p5_end)

    # Phase 6: Final polish (5% budget)
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        h, c5v = _coord_desc(h, n, max(1.0, t0 + budget_s - time.time() - 3), 5, rng, t0 + budget_s)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
