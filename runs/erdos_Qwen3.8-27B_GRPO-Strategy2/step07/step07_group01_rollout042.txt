This is a linear program in (h, t)! But wait, c_k = sum h_i(1-h_j) is bilinear in h, not linear. So it's actually a quadratic program.

Specifically, c_k = sum h_i - sum h_i * h_j (for the relevant pairs). The first term is linear in h, the second is quadratic. So we have:
- minimize t
- subject to: sum h_i - sum_i h_i * h_{i+k} <= t (for each active k)
- 0 <= h_i <= 1
- sum h_i = n/2

This is a QCQP (quadratically constrained quadratic program). I could try using CVXOPT or another solver for this. But the quadratic constraints make it non-convex in general.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(h)
    dx = 2.0 / n

    if top_k < len(c):

        idx = np.arange(len(c))
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    w /= ws
    g = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-15:
            continue
        k = idx[i]
        sh = (n - 1) - k

        else:
            sp = -sh
            g[:n - sp] += wi * (1.0 - h[sp:])
            g[sp:] -= wi * h[:n - sp]
    return g, c

def _obj_fn(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
    else:
        c_top = c
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _obj_grad(x, alpha, top_k):
    g, _ = _grad_sm(x, alpha, top_k)
    return g

def _slsqp_opt(h, alpha, top_k, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj_fn, h, jac=_obj_grad, args=(alpha, top_k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, n_iter, step0, top_k, mu):

    target = n * 0.5
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    stalled = 0

        v = mu * v + (1.0 - mu) * g
        gn = np.linalg.norm(v)
        if gn < 1e-18:

        v = v / gn * gn
        taken = False

                stalled = 0

def _local_search(h, n_iter, rng, base_step):

    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    stagnant = 0

            stagnant = 0
        else:
            stagnant += 1
        step = base_step * (1.0 + 0.3 * np.sqrt(stagnant / 2000.0))
        step = min(step, base_step * 5.0)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 7 + rng.randint(0, 10)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _local2(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out
