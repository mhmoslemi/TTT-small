import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _repair(h):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    if abs(h.sum() - n * 0.5) > 1e-9:
        h = _proj_sum(h, n * 0.5)
    return h

def _corr_grad_w(h, w):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    nz = np.where(w > 1e-14)[0]
    for k in nz:
        wk = w[k]
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g * dx

def _obj_p(h, p, pen):
    n = len(h)
    c = _corrs(h)
    val = np.sum(c ** p) ** (1.0 / p)
    return val + pen * (h.sum() - n * 0.5) ** 2

def _grad_p(h, p, pen):
    n = len(h)
    c = _corrs(h)
    norm = (np.sum(c ** p)) ** (1.0 / p)
    cw = norm ** (1.0 - p) * np.maximum(c, 1e-30) ** (p - 1)
    g = _corr_grad_w(h, cw)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _obj_sf(h, temp, pen):
    n = len(h)
    c = _corrs(h)
    e = np.exp((c - c.max()) / temp)
    w = e / e.sum()
    return w @ c + pen * (h.sum() - n * 0.5) ** 2

def _grad_sf(h, temp, pen):
    n = len(h)
    c = _corrs(h)
    e = np.exp((c - c.max()) / temp)
    w = e / e.sum()
    g = _corr_grad_w(h, w)
    g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

def _lbfgs_obj_p(x, p, pen):
    return _obj_p(x, p, pen)

def _lbfgs_grad_p(x, p, pen):
    return _grad_p(x, p, pen)

def _lbfgs_obj_sf(x, temp, pen):
    return _obj_sf(x, temp, pen)

def _lbfgs_grad_sf(x, temp, pen):
    return _grad_sf(x, temp, pen)

def _lbfgs_run(h, mode, param, pen, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    try:
        if mode == 'p':
            res = minimize(_lbfgs_obj_p, h, jac=_lbfgs_grad_p, args=(param, pen),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 60})
        else:
            res = minimize(_lbfgs_obj_sf, h, jac=_lbfgs_grad_sf, args=(param, pen),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 60})
        return _repair(res.x)
    except Exception:
        return _repair(h)

def _nesterov_topk(h, n_iter, k_start, k_max, step0, elapsed_fn, budget_fn):
    n = len(h)
    hc = h.copy().astype(float)
    v = np.zeros(n)
    beta = 0.75
    step = step0
    best_c = _eval(hc)
    best_h = hc.copy()
    stall = 0
    k = k_start
    for i in range(n_iter):
        if budget_fn():
            break
        c = _corrs(hc)
        if stall > 40:
            k = min(k + 1, k_max)
        topk = np.argsort(c)[-k:]
        g = _corr_grad_w(hc, np.ones(k) / k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * (g / gn)
        hn = _proj_sum(hc - step * v, n * 0.5)
        cn = _eval(hn)
        if cn < best_c - 1e-15:
            hc = hn
            best_c = cn
            best_h = hc.copy()
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.5
            stall += 1
            if step < 1e-9:
                break
    return best_h, best_c

def _local_search(h, n_iter, rng, elapsed_fn, budget_fn):

    best_c = _eval(hc)
    best_h = hc.copy()

            break

            cn = _eval(trial)

                best_h = trial.copy()
            hc = trial
        else:
            pass
        if i % 3000 == 0:

    return best_h, best_c
