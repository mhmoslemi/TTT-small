import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag_vec(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        cnt = n - lag
        if cnt > 0:
            idx = np.arange(cnt)
            g[idx + lag] += (1.0 - h[idx])
            g[idx] -= h[idx + lag]
    else:
        p = -lag
        cnt = n - p
        if cnt > 0:
            idx = np.arange(cnt)
            g[idx] += (1.0 - h[idx + p])
            g[p:] -= h[:cnt]  # h[m-p] for m in p..n-1 => h[0..n-p-1]
    return g

def _smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -200, 200)
    sm = np.log(np.sum(np.exp(sc - sc.max()))) / alpha
    return m + sm

def _smooth_max_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -200, 200)
    w = np.exp(sc - sc.max())
    w /= w.sum()
    g = np.zeros(n)

        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        g += w[k] * (2.0 / n) * _grad_lag_vec(x, lag)
    return g

    return x.sum() - n * 0.5

    return np.ones(len(x))

def _adam_smooth(h, n, alpha, lr, b1, b2, eps, niter, check_fn):

    v = np.zeros(n)
    c5c = _eval(h)

            break
        g = _smooth_max_grad(h, n, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h_new = _project(h - lr * mh / (np.sqrt(vh) + eps), n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5c:

            c5c = c5n
            lr *= 1.05
            if lr > 0.5:

        else:

    return h, c5c

def _nesterov_worst(h, n, k_lags, alpha0, momentum, niter, check_fn):

    v = np.zeros(n)
    c5c = _eval(h)
    alpha = alpha0

def _local_2pt(h, ntry, rng, c5c, check_fn, scale=5e-6):

    best_c = c5c

import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag_vec(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        cnt = n - lag
        if cnt > 0:
            idx = np.arange(cnt)
            g[idx + lag] += (1.0 - h[idx])
            g[idx] -= h[idx + lag]
    else:
        p = -lag
        cnt = n - p
        if cnt > 0:
            idx = np.arange(cnt)
            g[idx] += (1.0 - h[idx + p])
            g[p:] -= h[:cnt]
    return g

def _coord_desc(h, n, passes, check_fn):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c5c = float(np.max(c) * dx)
    lag_arr = np.arange(2 * n - 1) - (n - 1)
    for p in range(passes):
        if check_fn():
            break
        imp = 0
        for m in range(n):
            v = h[m]
            if v < 1e-15 or v > 1.0 - 1e-15:
                idx1 = m - lag_arr
                v1 = (idx1 >= 0) & (idx1 < n)

                G = np.zeros(2 * n - 1)
                G[v1] = (1.0 - h[idx1[v1]])

def _nesterov_worst(h, n, k_lags, alpha0, momentum, niter, check_fn):
    dx = 2.0 / n
    v = np.zeros(n)
    c5c = _eval(h)
    alpha = alpha0
    stall = 0
    for _ in range(niter):
        if check_fn():
            break
        c = np.correlate(h, 1.0 - h, mode='full')

        g = g * dx / k_lags

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sc = np.clip(alpha * (c - m), -150, 150)
    return m + np.log(np.sum(np.exp(sc - sc.max()))) / alpha

def _smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()

        h2, c52 = _coord_desc(best_h.copy(), n0, 100, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 6: Final coordinate descent
    if _el() < budget_s - 80:
        h2, c52 = _coord_desc(best_h.copy(), n0, 500, _bc)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 7: Block local search
    if _el() < budget_s - 120:
        remain = budget_s - 120 - _el()
        nt = int(min(remain / 0.006, 80000))
        best_h, best_c5 = _local_block(best_h, nt, rng, best_c5, _bc, 3, 2e-5)

    # Phase 8: 2-point scan
    if _el() < budget_s - 50:
        remain = budget_s - 50 - _el()
        nt = int(min(remain / 0.0025, 150000))
        best_h, best_c5 = _local_2pt(best_h, nt, rng, best_c5, _bc, 1e-5)

    # Phase 9: Ultra-fine 2-point
    if _el() < budget_s - 5:
        remain = budget_s - 5 - _el()
        nt = int(min(remain / 0.0025, 100000))
        best_h, best_c5 = _local_2pt(best_h, nt, rng, best_c5, _bc, 5e-7)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
