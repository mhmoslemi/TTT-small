import numpy as np
from scipy.optimize import minimize

import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _slp_solve(h0):
    n = len(h0)
    c_h0 = _corr(h0)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        gk = np.zeros(n)
        if shift >= 0:
            gk[shift:] += (1.0 - h0[:n - shift])
            gk[:n - shift] += -h0[shift:]
        else:
            sp = -shift
            gk[:n - sp] += (1.0 - h0[sp:])
            gk[sp:] += -h0[:n - sp]
        rhs = gk @ h0 - c_h0[k]
        cons.append(cp.sum(gk * h_var) <= rhs + t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solved = False
    for solver in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved:
        try:
            prob.solve()
            solved = True
        except Exception:
            pass
    if not solved or h_var.value is None:
        return h0.copy()
    return h0

def _slp_solve_safe(h0, beta=1.0):
    n = len(h0)
    c_h0 = _corr(h0)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        gk = np.zeros(n)
        if shift >= 0:
            gk[shift:] += (1.0 - h0[:n - shift])
            gk[:n - shift] += -h0[shift:]
        else:
            sp = -shift
            gk[:n - sp] += (1.0 - h0[sp:])
            gk[sp:] += -h0[:n - sp]
        rhs = gk @ h0 - c_h0[k]
        cons.append(cp.sum(gk * h_var) <= rhs + t_var)
    obj = t_var + 1e-6 * cp.sum_squares(h_var - h0)
    prob = cp.Problem(cp.Minimize(obj), cons)
    solved = False
    for solver in [cp.ECOS, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            solved = True
            break
        except Exception:
            continue
    if not solved:
        try:
            prob.solve()
            solved = True
        except Exception:
            pass
    if not solved or h_var.value is None:
        return h0.copy()
    h_slp = _repair(np.array(h_var.value).flatten(), n)
    return beta * h_slp + (1.0 - beta) * h0

def _slp_run(h0, n_iters, beta, t0, deadline):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    prev_c = best_c
    for it in range(n_iters):
        if time.time() - t0 > deadline:
            break
        h_new = _slp_solve_safe(h, beta)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        if abs(c_new - prev_c) < 1e-12 and it > 5:
            break
        prev_c = c_new
    return best_h, best_c

def _grad_smooth_max(h, alpha):
    n = len(h)
    c = _corr(h)
    cm = c.max()
    s = np.clip(alpha * (c - cm), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            g[shift:] += w[k] * (1.0 - h[:n - shift])
            g[:n - shift] -= w[k] * h[shift:]
        else:
            sp = -shift
            g[:n - sp] += w[k] * (1.0 - h[sp:])
            g[sp:] -= w[k] * h[:n - sp]
    return g

def _proj_to_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_descent(h, alpha, n_steps, step0, t0, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    step = step0
    for _ in range(n_steps):
        if time.time() - t0 > deadline:
            break
        g = _grad_smooth_max(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        for ls in range(25):
            h_try = _proj_to_sum(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < best_c or c_try < _eval(h):
                break
            step *= 0.5
        else:
            break
        h = h_try
        c_try = _eval(h)
        if c_try < best_c:
            best_c = c_try
            best_h = h.copy()
        step = min(step * 1.05, step0 * 3.0)
    return best_h, best_c

def _local_search(h, n_tries, rng, t0, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_tries):
        if time.time() - t0 > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.repeat(h, factor).copy()
