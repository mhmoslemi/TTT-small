        return np.full(n, 0.5)
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _obj_smooth(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _grad_smooth(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    v = 1.0 - h
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = k - (n - 1)
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * v[sh:]
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * v[:n - sp]
            g[:n - sp] -= wk * h[sp:]
    return g

def _momentum_opt(h, alpha, t_budget, lr0, n):
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.88
    lr = lr0
    f = _obj_smooth(h, alpha)
    t_end = time.time() + t_budget
    stall = 0
    best_f = f
    while time.time() < t_end:
        g = _grad_smooth(h, alpha, n)
        v = mu * v + (1 - mu) * g
        step = lr
        accepted = False
        for _ in range(15):
            h_new = _proj(h - step * v, target)
            f_new = _obj_smooth(h_new, alpha)
            if f_new < f:
                h = h_new
                f = f_new
                if f < best_f:
                    best_f = f
                    stall = 0
                accepted = True
                break
            step *= 0.45
        if not accepted:
            stall += 1
            if stall > 20:
                break
        else:
            lr = min(lr * 1.02, lr0 * 30)
    return h, best_f

def _grad_lag(h, k, n, dx):
    sh = k - (n - 1)
    g = np.zeros(n)
    v = 1.0 - h
    if sh >= 0:
        g[:n - sh] = dx * v[sh:]
        if sh > 0:
            g[sh:] -= dx * h[:n - sh]
    else:
        sp = -sh
        g[sp:] = dx * v[:n - sp]
        g[:n - sp] -= dx * h[sp:]
    return g

        top_idx = np.argsort(c)[-top_T:]
        a_vals = c[top_idx]
        grads = np.array([_grad_lag(h, k, n, dx) for k in top_idx])
        g_sum = grads.sum(axis=0)
        n_pairs = max(n, 256)
        # Precompute sorted indices for gradient guidance
        if use_grad_guide:
            g_rank = np.argsort(g_sum)
        for _ in range(n_pairs):
            if time.time() > t_end:

            else:

            if i == j:

            b = grads[:, i] - grads[:, j]

def _fine_local(h, t_budget, rng, n):
    t_end = time.time() + t_budget
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c

        ni, nj = h[i] + d, h[j] - d

                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
