This is what the previous code handles with the k1, k2 correction.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_to_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_at_pos(h, p, n):
    g = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, 2 * n - 2 - p)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + p - (n - 1)
        g[lo1:hi1 + 1] += (1.0 - h[idx])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    if lo2 <= hi2:
        idx2 = (n - 1) + p - np.arange(lo2, hi2 + 1)
        g[lo2:hi2 + 1] -= h[idx2]
    return g * (2.0 / n)

def _perturb(h, rng):
    n = len(h)
    op = rng.randint(3)
    if op == 0:
        h = _proj_to_sum(h + 0.015 * rng.randn(n), n * 0.5)
    elif op == 1:
        c = rng.rand() * n
        s = max(3, n // 10)
        x = np.arange(n, dtype=float)
        mask = np.exp(-0.5 * ((x - c) / s) ** 2)
        h = _proj_to_sum(h + 0.025 * mask * rng.randn(n), n * 0.5)
    else:
        f1, f2 = rng.randint(1, 5), rng.randint(5, 15)
        x = np.arange(n, dtype=float)

        p2 = 0.012 * np.sin(2 * np.pi * f2 * x / n + rng.rand() * 6.28)
        h = _proj_to_sum(h + p1 + p2, n * 0.5)
    return h

def _cd_loop(h, rng, deadline):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
    c5_cur = float(c.max())
    best_h = h.copy()
    best_c = c5_cur
    stable = 0
    fail_streak = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 4000 == 0:
            c = np.correlate(h, 1.0 - h, mode='full').copy() * dx
            c5_cur = float(c.max())
        if stable > 5000:
            h_new = _perturb(h, rng)
            ct = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
            if float(ct.max()) < best_c + 0.0015:
                h = h_new; c = ct.copy(); c5_cur = float(c.max())
                fail_streak = 0
            else:
                fail_streak += 1
            stable = 0
            if fail_streak > 15:
                break
            continue
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d0 = j - i
        k1 = n - 1 + d0
        k2 = n - 1 - d0
        g = _grad_at_pos(h, i, n) - _grad_at_pos(h, j, n)
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-15:

                d_hi = d_mid
            elif sg < -1e-14:

            else:

def _subgrad_phase(h, rng, deadline):

    dx = 2.0 / n
    idx = np.arange(n)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    while time.time() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = float(c.max())
        active = np.where(c >= m - 2e-5)[0]
        if len(active) == 0:
            active = np.argsort(c)[-5:]
        elif len(active) > 12:
            active = np.argsort(c)[-12:]
        g = np.zeros(n)
        for k in active:
            sh = k - (n - 1)
            v1 = (idx + sh >= 0) & (idx + sh < n)
            g[v1] += (1.0 - h[idx[v1] + sh])
            v2 = (idx - sh >= 0) & (idx - sh < n)
            g[v2] -= h[idx[v2] - sh]
        g *= dx / len(active)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        base = 0.04 / gn
        accepted = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            s = base * mult
            h_new = _proj_to_sum(h - s * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5_cur - 1e-15:
                h = h_new; c5_cur = c5n; accepted = True

            c5_cur = _eval(h)
            if c5_cur < best_c:
                best_c = c5_cur; best_h = h.copy()
    return best_h, best_c

def _try_coarse(n_fine, h_fine, rng, deadline):
    n_c = n_fine // 2
    h_c = (h_fine[::2] + h_fine[1::2]) * 0.5
