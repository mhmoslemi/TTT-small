def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        hv, cv = _block_anneal(best_h.copy(), 20000, rng, dl3, 0.002, 1e-10, blocks)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()
            dl3b = min(deadline, dl3 + 30)
            hv2, cv2 = _lp_phase(best_h.copy(), 8000, 0.001, 5, dl3b, rng)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    dl4 = deadline
    best_h, best_c5 = _coord_polish(best_h, dl4, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
