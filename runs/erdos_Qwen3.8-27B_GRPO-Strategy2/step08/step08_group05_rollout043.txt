                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _grad_for_lag(c, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - c[sh:])
        if sh > 0:
            g[sh:] -= c[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - c[:-sp])
        g[:n - sp] -= c[sp:]
    return g

def _lp_step(c, n, eps, m_top):
    cor = _correl(c)
    idx_top = np.argpartition(cor, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-cor[idx_top])]
    grads = np.array([_grad_for_lag(c, k, n) for k in idx_top])
    c_obj = -grads[0]
    A_ub = grads[1:] if m_top > 1 else None
    b_ub = np.zeros(m_top - 1) if m_top > 1 else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='revise')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        d = _lp_step(c, n, eps, m_top)
        if d is None or np.max(np.abs(d)) < 1e-14:
            stall += 1
            if stall > 30:
                break
            d = np.random.randn(n) * eps * 0.5
            d -= d.mean()
        else:
            stall = max(0, stall - 1)
        step = 1.0
        improved = False
        for _ in range(10):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-17:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            stall += 2
    return best_h, best_c5

def _local_pair_search(h, n_passes, radius, rng, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h, best_c5 = h.copy(), c5_cur
    for p in range(n_passes):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            for dj in range(1, radius + 1):
                j = (i + dj) % n
                if j == i:
                    continue
                for d in [5e-5, 2e-5, 1e-5, -5e-5, -2e-5, -1e-5]:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c5_new = _eval(h)
                        if c5_new < c5_cur - 1e-18:
                            c5_cur = c5_new
                            if c5_new < best_c5:
                                best_c5, best_h = c5_new, h.copy()
                        else:
                            h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _multi_restart(h, n_restarts, rng, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        scale = 0.001 * (r + 1)
        noise = scale * rng.randn(n)
        pert = _repair(h + noise, n)
        hv = _slsqp_opt(pert, 15000, 20, 250, min(deadline, time.time() + 60))
        hv = _lp_descent(hv, 3000, 0.002, 5, min(deadline, time.time() + 60))
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP polish (40s)
    dl1 = min(deadline, t0 + 40)
    for a, tk, mi in [(3000, 40, 300), (12000, 20, 200)]:
        if time.time() > dl1:
            break
        hv = _slsqp_opt(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent coarse (200s)
    dl2 = min(deadline, t0 + 240)
    hv, cv = _lp_descent(best_h.copy(), 15000, 0.004, 6, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: LP descent fine (180s)
    dl3 = min(deadline, t0 + 420)
    hv, cv = _lp_descent(best_h.copy(), 12000, 0.001, 5, dl3)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 4: Local pair search (200s)
    dl4 = min(deadline, t0 + 620)
    hv, cv = _local_pair_search(best_h.copy(), 30, 6, rng, dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: Multi-restart (150s)
    dl5 = min(deadline, t0 + 770)
    hv, cv = _multi_restart(best_h, 3, rng, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 6: Final LP very fine (120s)
    dl6 = min(deadline, t0 + 890)
    hv, cv = _lp_descent(best_h.copy(), 8000, 0.0005, 4, dl6)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 7: Final pair search + coord descent (remaining)
    dl7 = deadline
    hv, cv = _local_pair_search(best_h.copy(), 20, 4, rng, dl7)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
