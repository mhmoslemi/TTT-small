def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_correl(h)))

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smooth_max(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_obj(h, alpha, pen):
    c = _correl(h)
    val = _smooth_max(c, alpha)
    if pen > 0:
        val += pen * (h.sum() - len(h) * 0.5) ** 2
    return val

def _smax_grad(h, alpha, pen):
    n = len(h)
    c = _correl(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    active = np.where(w > 1e-14)[0]

            g[lag:] += w[ki] * (-h[:-lag]) * dx
        else:
            l2 = -lag
            g[l2:] += w[ki] * (1.0 - h[:-l2]) * dx
            g[:n - l2] += w[ki] * (-h[l2:]) * dx
    if pen > 0:
        g += 2.0 * pen * (h.sum() - n * 0.5)
    return g

        g[lag:] += (-h[:-lag]) * dx
    else:
        l2 = -lag
        g[l2:] += (1.0 - h[:-l2]) * dx
        g[:n - l2] += (-h[l2:]) * dx
    return g

        return _smax_obj(x, alpha, pen)
    def g(x):
        return _smax_grad(x, alpha, pen)
    try:
        r = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxit, 'gtol': 1e-14, 'ftol': 1e-16,
