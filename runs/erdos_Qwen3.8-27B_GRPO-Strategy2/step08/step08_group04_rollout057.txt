import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_h(h, sh, dx):
    n = len(h)
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_h(h, idx[ii] - (n - 1), dx)
    return g

def _nesterov(h, alpha, topk, steps, step0, beta, deadline, perturb_every=0):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    rng = np.random.RandomState(int(time.time() * 1000) % 10000)
    for it in range(steps):
        if time.time() > deadline:
            break
        if perturb_every > 0 and it % perturb_every == 0 and it > 0:
            h = _repair(h + 0.002 * rng.randn(n), n)
            f = _smax_obj(h, alpha, topk)
            v = np.zeros(n)
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn; f = fn
            step = min(step * 1.02, step0 * 20.0)
        else:
            for _ in range(3):
                step *= 0.5
                hn = _proj_sum(h - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn; f = fn; break
            else:
                step = step0
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or (T > 1e-9 and rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12))):
            h = hn; c5c = c5n
            if c5n < bc:
                bc = c5c; bh = h.copy()
    return bh, bc

def _bit_search(h, niter, rng, deadline):
    n = len(h)
    h01 = (h > 0.5).astype(float)
    h01 = _repair(h01, n)
    best_h = h01.copy(); best_c5 = _eval(h01)

        ones = np.where(h01 > 0.5)[0]

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h01[i], h01[j] = h01[j], h01[i]
        c5n = _eval(h01)

        elif rng.random() > 0.005:
            h01[i], h01[j] = h01[j], h01[i]
    return best_h, best_c5

def _fine_tune(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    step0 = 5e-6
    for it in range(niter):

        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 10)) % nc
        d = step0 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < bc:
                    bc = c5n; bh = h.copy()
                step0 = min(step0 * 1.03, 2e-5)
            else:
                h[i], h[j] = ni - d, nj + d
                step0 = max(step0 * 0.96, 5e-10)
    return bh, bc
