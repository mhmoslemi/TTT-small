To minimize max_k (coef_k * x + const_k) over x ∈ [0,1]:

This is a 1D convex optimization. The function f(x) = max_k (coef_k * x + const_k) is convex and piecewise linear.

The minimum occurs where:
- f'(x-) > 0 and f'(x+) < 0 (kink point), or
- f is flat at some point, or
- at x=0 or x=1

We can find the minimum by observing:
- At the minimum, the subgradient [min active slope, max active slope] contains 0
- Active slopes are coef_k for k where coef_k * x + const_k = f(x)

Algorithm:
1. Evaluate f at x=0: f(0) = max_k const_k. Active set A0 = {k: const_k = f(0)}. Min slope = min{coef_k: k∈A0}, max slope = max{coef_k: k∈A0}.

3. Evaluate f at x=1: f(1) = max_k (coef_k + const_k). Active set A1 = {k: coef_k + const_k = f(1)}.
4. If max slope (at x=1) ≤ 0, minimum is at x=1.
5. Otherwise, the minimum is in (0,1). Use bisection:
   - At any x, compute f(x) and the active set
   - The "left derivative" is min{coef_k: active at x}, "right derivative" is max{coef_k: active at x}
   - If left derivative > 0: move x down
   - If right derivative < 0: move x up
   - Otherwise: found the minimum (or a flat region)

```python
import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _coord_opt(h, i, n):
    """Find optimal h[i] given all other h fixed."""
    dx = 2.0 / n
    # For each correlation index k (0..2n-2), compute coef and const
    # c_k = dx * sum_j h[j]*(1-h[j+shift]) where shift = k-(n-1)
    # Coefficient of h[i] in c_k:
    #   Term 1: h[i] * (1-h[i+shift]) if 0<=i+shift<n → coef contribution: dx*(1-h[i+shift])
    #   Term 2: h[i-shift] * (1-h[i]) if 0<=i-shift<n → coef contribution: -dx*h[i-shift]
    shift_range = np.arange(2*n - 1) - (n - 1)  # -n+1 to n-1
    coef = np.zeros(2*n - 1)
    # Term 1: i + shift in [0, n-1]
    m1 = (i + shift_range >= 0) & (i + shift_range < n)
    idx1 = i + shift_range[m1]
    coef[m1] += dx * (1.0 - h[idx1])
    # Term 2: i - shift in [0, n-1], i.e., shift in [i-n+1, i]
    m2 = (i - shift_range >= 0) & (i - shift_range < n)
    idx2 = i - shift_range[m2]
    coef[m2] -= dx * h[idx2]
    
    # Now c_k = coef[k] * h[i] + const_k
    # const_k = c_k - coef[k] * h[i]
    c = _correl(h)  # current correlations
    consts = c - coef * h[i]
    
    # Minimize max_k (coef[k]*x + consts[k]) for x in [0,1]
    x_lo, x_hi = 0.0, 1.0
    for _ in range(40):
        x_mid = 0.5 * (x_lo + x_hi)
        vals = coef * x_mid + consts
        f = vals.max()
        active = vals >= f - 1e-15
        if not active.any():
            active = vals >= f - 1e-12
        slope_lo = coef[active].min()
        slope_hi = coef[active].max()
        if slope_lo >= -1e-15:
            x_hi = x_mid
        elif slope_hi <= 1e-15:
            x_lo = x_mid
        else:
            break
    return 0.5 * (x_lo + x_hi)

def _coord_descent(h, n_passes, deadline, random_order=False, rng=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    for p in range(n_passes):

        order = np.random.permutation(n) if random_order else np.arange(n)

        c5 = _eval(h)
        if c5 < best_c5:

    return best_h, best_c5

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        # Compute gradients
        grads = np.zeros((m_top, n))
        dx = 2.0 / n
        for ii, k in enumerate(idx_top):
            off = k - (n - 1)
            j = np.arange(n)
            i1 = j - off
            mask1 = (i1 >= 0) & (i1 < n)
            grads[ii, mask1] += (1.0 - c[i1[mask1]])
            i2 = j + off
            mask2 = (i2 >= 0) & (i2 < n)
            grads[ii, mask2] -= c[i2[mask2]]
        grads *= dx
        c_obj = -grads[0]
        if m_top > 1:
            A_ub = grads[1:]
            b_ub = np.zeros(m_top - 1)
        else:
            A_ub = None
            b_ub = None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs',
                          options={'presolve': True, 'time_limit': 2.0})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        d = res.x
        if np.max(np.abs(d)) < 1e-14:
            break
        # Line search
        improved = False
        for sf in [1.0, 0.5, 0.25, 0.125]:
            h_new = _repair(c + sf * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-14:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 20:
                break
    return best_h, best_c5
```

```python
    """Find best h[i] using precomputed correlations c."""
    dx = 2.0 / n

    m2 = (i - shifts >= 0) & (i - shifts < n) & (shifts != 0)
    coef[m1] += dx * (1.0 - h[i + shifts[m1]])
    coef[m2] -= dx * h[i - shifts[m2]]

    quad_coeff = dx  # coefficient of (x - x^2)

    # c_quad_new(x) = c_quad + quad_coeff*(x - x^2 - ao + ao^2)

        c_new[quad_k] = c[quad_k] + quad_coeff * (x - x*x - ao + ao*ao)
        f = c_new.max()
        if f < best_f:

    if best_x > 0 and best_x < 1:
        # Narrow to surrounding interval
        idx = np.argmin(np.abs(xs - best_x))
        lo = xs[max(0, idx-1)]
        hi = xs[min(len(xs)-1, idx+1)]
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(25):
            m1_ = hi - gr*(hi-lo)
            m2_ = lo + gr*(hi-lo)
            c1 = c + coef*(m1_-ao); c1[quad_k] = c[quad_k]+quad_coeff*(m1_-m1_**2-ao+ao**2)
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _make_coefs(h, i, n):
    """Compute linear coefs and quad flag for coordinate i."""
    dx = 2.0 / n
    nr = 2 * n - 1
    shifts = np.arange(nr) - (n - 1)
    coef = np.zeros(nr)
    m1 = (i + shifts >= 0) & (i + shifts < n)
    m2 = (i - shifts >= 0) & (i - shifts < n) & (shifts != 0)
    m1_nz = m1 & (shifts != 0)
    coef[m1_nz] += dx * (1.0 - h[i + shifts[m1_nz]])
    coef[m2] -= dx * h[i - shifts[m2]]
    quad_k = n - 1
    quad_coeff = dx
    return coef, quad_k, quad_coeff

def _find_best_x(c, coef, quad_k, quad_coeff, ao, n):
    """Minimize max_k c_k_new(x) for x in [0,1]."""
    nr = len(c)
    xs = np.linspace(0.0, 1.0, 25)
    best_x, best_f = ao, np.inf
    for x in xs:
        c_new = c + coef * (x - ao)
        c_new[quad_k] = c[quad_k] + quad_coeff * (x - x * x - ao + ao * ao)
        f = c_new.max()
        if f < best_f:
            best_f, best_x = f, x
    # Golden section refinement
    if best_x > 0.01 and best_x < 0.99:
        idx = int(round(best_x * 24))
        lo = xs[max(0, idx - 1)]
        hi = xs[min(len(xs) - 1, idx + 1)]
        if hi - lo < 1e-8:
            lo, hi = max(0, best_x - 0.04), min(1, best_x + 0.04)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):
            m1_ = hi - gr * (hi - lo)
            m2_ = lo + gr * (hi - lo)
            c1 = c + coef * (m1_ - ao)
            c1[quad_k] = c[quad_k] + quad_coeff * (m1_ - m1_**2 - ao + ao*ao)
            c2 = c + coef * (m2_ - ao)
            c2[quad_k] = c[quad_k] + quad_coeff * (m2_ - m2_**2 - ao + ao*ao)
            if c1.max() > c2.max():
                lo = m1_
            else:
                hi = m2_
        best_x = 0.5 * (lo + hi)
    return best_x

def _coord_descent(h, n_passes, deadline):
    n = len(h)
    c = _correl(h)
    best_h, best_c5 = h.copy(), _eval(h)
    for p in range(n_passes):
        if time.time() > deadline:
            break
        for i in range(n):
            coef, qk, qc = _make_coefs(h, i, n)
            ao = h[i]
            x_new = _find_best_x(c, coef, qk, qc, ao, n)
            if abs(x_new - ao) > 1e-13:
                c += coef * (x_new - ao)
                c[qk] = c[qk] + qc * (x_new - x_new**2 - ao + ao*ao)
                h[i] = x_new
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c_arr = h.copy()
    c5_cur = best_c5
    stall = 0
    dx = 2.0 / n
    for it in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c_arr)
        if m_top < len(cor):
            idx_top = np.argpartition(cor, -m_top)[-m_top:]
        else:
            idx_top = np.arange(len(cor))
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.zeros((m_top, n))
        for ii, k in enumerate(idx_top):
            off = k - (n - 1)
            j = np.arange(n)
            i1 = j - off
            msk1 = (i1 >= 0) & (i1 < n)
            grads[ii, msk1] += (1.0 - c_arr[i1[msk1]])
            i2 = j + off
            msk2 = (i2 >= 0) & (i2 < n)
            grads[ii, msk2] -= c_arr[i2[msk2]]
        grads *= dx
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stall += 1
            if stall > 30:
                break
            continue
        d = res.x
        improved = False
        for sf in [1.0, 0.7, 0.4, 0.2]:
            h_new = _repair(c_arr + sf * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c_arr, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c_arr.copy()
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 40:
                break
    return best_h, best_c5

def _sweep_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for ss in [2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]:
        for _ in range(niter):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 4)) % nc
            di = ss * (2.0 * rng.random() - 1.0)
            ni, nj = bh[i] + di, bh[j] - di
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                bh[i], bh[j] = ni, nj
                c5n = _eval(bh)
                if c5n < bc - 1e-18:
                    bc = c5n
                else:
                    bh[i], bh[j] = ni - di, nj + di
    return bh, bc
```
