```python
def _pair_optimize(h, i, j, n_evals=30):
    """Find best δ for h[i]+=δ, h[j]-=δ using golden section."""
    n = len(h)
    lo = -h[i]
    hi = 1.0 - h[j]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    def f(d):
        h[i] += d
        h[j] -= d
        c = _eval(h)
        h[i] -= d
        h[j] += d
        return c
    
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(n_evals - 2):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    opt = (a + b) / 2
    return opt, f(opt)
```

import numpy as np
from scipy.optimize import minimize
import time
import math

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    return m + np.log(np.exp(s - s.max()).sum() + 1e-300) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

    return p
