def _slsqp_phase(h, deadline, alpha, topk, maxit=300):
    n = len(h)
    target = n/2.0
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return _repair(h, n)

def _exact_grad(h, topk=5):
    """Subgradient of the exact max."""
    c = _correl(h)
    n = len(h)
    dx = 2.0/n
    g = np.zeros(n)
    kmax = np.argmax(c)
    # Use top-k average for stability
    if topk > 1:
        idx = np.argpartition(c, -topk)[-topk:]
        # Weight by how close to max
        w = np.exp(100*(c[idx]-c[kmax]))
        w /= w.sum()
        for ii in range(len(idx)):
            if w[ii] > 1e-10:
                g += w[ii]*_lag_grad(h, idx[ii])
    else:
        g = _lag_grad(h, kmax)
    return g * dx

def _active_set_search(h, deadline, niter=50000):
    n = len(h)
    target = n/2.0
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    lr = 0.002
    for it in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        kmax = c.argmax()
        # Use top 3 with softmax weights
        top3 = np.argpartition(c, -3)[-3:]
        w = np.exp(200*(c[top3]-c[kmax]))
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(3):
            if w[ii] > 1e-10:
                g += w[ii]*_lag_grad(h, top3[ii])
        g *= 2.0/n
        # Normalized gradient step
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = -lr * g / gn
        # Ensure step is feasible
        max_s = 0.01
        step = np.clip(step, -max_s, max_s)
        hn = _proj(h + step, target)
        c5n = _eval(hn)
        if c5n < c5c:
            h = hn
            c5c = c5n
            if c5c < bc:
                bc = c5c
                bh = h.copy()
            lr = min(lr*1.01, 0.01)
        else:
            lr *= 0.7
            if lr < 1e-8:
                lr = 0.001
    return bh, bc

def _coord_refine(h, deadline, niter=100000):
    n = len(h)
    target = n/2.0
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng = np.random.RandomState(12345)
    step = 5e-5
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        # Find partner for mass transfer
        # Choose j to be near a "hot spot" of correlation
        c = _correl(h)
        kmax = c.argmax()
        # Move mass from i to i+kmax (or i-kmax)
        for j_off in [kmax, (n-1-kmax), 1, -1% n, 2]:
            j = (i + j_off) % n
            if j == i:
                continue
            for d in [step, -step, step*0.5, -step*0.5]:
                ni, nj = h[i]+d, h[j]-d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-18:
                        c5c = c5n
                        if c5n < bc:
                            bc = c5n
                            bh = h.copy()
                        break
                    h[i], h[j] = ni-d, nj+d
            else:
                continue
            break
        # Occasionally re-evaluate c
        if it % 50 == 0:
            c = _correl(h)
    return bh, bc

Objective: smax(c, alpha, topk) ≈ max(c) + (1/alpha) * log(sum(exp(alpha*(c_top - max))))

As alpha → ∞, this approaches the exact max. But for finite alpha, it also "sees" the other top values, which provides gradient information about reducing them too.

Another idea: use a "sum of top-k" objective instead of smax:
obj = sum of top-k values of c
gradient = sum of gradients of the top-k values

This is simpler and might work well. The idea is to reduce the sum of the top correlations, which necessarily reduces the max.

Let me combine approaches:
- Phase 1: Adam with smax (alpha grows from 100 to 50000)
- Phase 2: SLSQP with smax
- Phase 3: Subgradient on exact max (top 3 weighted)
- Phase 4: Paired coordinate with exact eval

import numpy as np
from scipy.optimize import minimize
import time

_OPT_ALPHA = 0.0
_OPT_TOPK = 0
_OPT_TARGET = 0.0

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    dx = 2.0 / n
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _lag_grad(h, idx[ii])
    return g * dx

def _topk_grad(h, kcount, temp):
    n = len(h)
    c = _correl(h)
    dx = 2.0 / n
    top = np.argsort(c)[-kcount:]
    w = np.exp(temp * (c[top] - c.max()))
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(top)):
        g += w[ii] * _lag_grad(h, top[ii])
    return g * dx

def _obj_fun(x):
    return _smax_obj(x, _OPT_ALPHA, _OPT_TOPK)

def _jac_fun(x):
    return _smax_grad(x, _OPT_ALPHA, _OPT_TOPK)

def _con_fun(x):
    return x.sum() - _OPT_TARGET

def _con_jac(x):
    return np.ones(len(x))

def _adam(h, deadline, lr0=0.004, a0=200, a1=200000, topk=70, b1=0.7, b2=0.92):
    n = len(h)
    target = n / 2.0
    m = np.zeros(n); v = np.zeros(n)
    t = 0; f = _smax_obj(h, a0, topk)
    best_h, best_f = h.copy(), f
    while time.time() < deadline:
        t += 1
        frac = min(1.0, (time.time() - (deadline - 280.0)) / 280.0)
        alpha = a0 * (a1 / a0) ** frac
        g = _smax_grad(h, alpha, topk)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        lr = lr0 / (1 + 0.0002 * t)
        step = -lr * mh / (np.sqrt(vh) + 1e-8)
        step = np.clip(step, -0.015, 0.015)
        hn = _proj(h + step, target)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f + 1e-16:
            h = hn; f = fn
            if f < best_f:
                best_f = f; best_h = h.copy()
        else:
            hn = _proj(h + 0.4 * step, target)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f + 1e-16:
                h = hn; f = fn
                if f < best_f:
                    best_f = f; best_h = h.copy()
    return best_h

def _slsqp(h, deadline, alpha, topk, maxit=250):
    n = len(h)
    if time.time() > deadline:
        return h
    _OPT_ALPHA = alpha; _OPT_TOPK = topk; _OPT_TARGET = n / 2.0
    try:
        r = minimize(_obj_fun, h, jac=_jac_fun, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, deadline, iters=60000, topk=4, temp=150.0):
    n = len(h)
    target = n / 2.0
    c5c = _eval(h); bh = h.copy(); bc = c5c
    lr = 0.003
    for it in range(iters):
        if time.time() > deadline:
            break
        g = _topk_grad(h, topk, temp)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        step = -(lr / gn) * g
        step = np.clip(step, -0.008, 0.008)
        hn = _proj(h + step, target)
        c5n = _eval(hn)
        if c5n < c5c:
            h = hn; c5c = c5n
            if c5c < bc:
                bc = c5c; bh = h.copy()
            lr = min(lr * 1.005, 0.008)
        else:
            lr *= 0.65
            if lr < 5e-7:
                lr = 0.002
    return bh, bc

def _pair_descent(h, deadline, rng_seed=99):
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(rng_seed)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    step = 4e-5
    c = _correl(h)
    kmax = c.argmax()
    for it in range(500000):
        if time.time() > deadline:
            break
        if it % 200 == 0:
            c = _correl(h)
            kmax = c.argmax()
        i = rng.randint(n)
        j = (i + kmax) % n
        if j == i:
            j = (i + 1) % n
        for d in (step, -step, step * 0.4, -step * 0.4):
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n; bh = h.copy()
                    break
                h[i], h[j] = ni - d, nj + d
        else:
            step = max(step * 0.92, 1e-8)
    return bh, bc

def _anneal(h, deadline, rng, niter=40000):
    n = len(h)
    target = n / 2.0
    c5c = _eval(h); bh = h.copy(); bc = c5c
    T0, T1 = 0.003, 1e-10
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = 1 + rng.randint(0, 4)
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn; c5c = c5n
            if c5n < bc:
                bc = c5n; bh = h.copy()
    return bh, bc

    if c1 < best_c5:

        h2 = _slsqp(best_h.copy(), dl2, alpha, tk, mi)
        c2 = _eval(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Subgradient exact max (470-750s)
    dl3 = min(deadline, t0 + 750)
    h3, c3 = _subgrad(best_h.copy(), dl3, iters=50000, topk=4, temp=200.0)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Anneal (750-900s)
    dl4 = min(deadline, t0 + 900)
    h4, c4 = _anneal(best_h.copy(), dl4, rng, niter=20000)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Pair descent (900-end)
    dl5 = deadline
    h5, c5 = _pair_descent(best_h.copy(), dl5, rng_seed=seed)
    if c5 < best_c5:
        best_c5, best_h = c5, h5

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
