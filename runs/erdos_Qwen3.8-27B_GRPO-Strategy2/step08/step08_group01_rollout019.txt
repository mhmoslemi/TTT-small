        s = idx[ii] - (n - 1)
        gg = np.zeros(n)
        if s > 0:
            gg[s:] = 1.0 - h[:n - s]
            gg[:n - s] -= h[s:]
        elif s < 0:
            sp = -s
            gg[:n - sp] = 1.0 - h[sp:n]
            gg[sp:] -= h[:n - sp]
        else:
            gg[:] = 1.0 - 2.0 * h
        g += wi * gg
    return g * dx, c

def _pgd(h, n_it, a0, a1, tk, s0):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    st = s0
    stuck = 0
    for it in range(n_it):
        frac = (it + 1) / max(n_it, 1)
        a = a0 + (a1 - a0) * frac
        g, c = _grad_sm(h, a, tk)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm
        ls = st
        for _ in range(15):
            ht = _repair(h - ls * g, n)
            ct = _eval(ht)
            if ct < cur - 1e-13:
                h = ht; cur = ct; stuck = 0
                break
            ls *= 0.4
        else:
            ht = _repair(h - ls * g, n)
            ct = _eval(ht)
            if ct < cur:
                h = ht; cur = ct
            stuck += 1
            if stuck > 200:
                st *= 0.5
                stuck = 0
        if ls > st * 1.5:
            st = min(ls, s0 * 10)
        elif it > 100 and ls < st * 0.3:
            st = max(ls * 0.8, s0 * 0.05)
    return h, cur

def _coord2(h, n_it, step, rng):
    n = len(h)
    cur = _eval(h)
    best = cur
    bh = h.copy()
    for _ in range(n_it):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < best:
                    best = cn; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, best

def _coord3(h, n_it, step, rng):
    n = len(h)
    cur = _eval(h)
    best = cur
    bh = h.copy()
    for _ in range(n_it):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        k = (i + 10 + rng.randint(0, 15)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < best:
                    best = cn; bh = h.copy()
            else:
                h[i], h[j], h[k] = s0, s1, s2
    return bh, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_par = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_par)
    best_h = h_par.copy()
    best_c = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Long PGD with smooth start
    if _el() < budget_s - 500:
        h_t, c_t = _pgd(best_h, 4000, 3, 300, 40, 0.02)
        if c_t < best_c:
            best_c = c_t; best_h = h_t.copy()

    # Phase 2: PGD sharpening
    if _el() < budget_s - 450:
        h_t, c_t = _pgd(best_h, 4000, 300, 5000, 20, 0.008)
        if c_t < best_c:
            best_c = c_t; best_h = h_t.copy()

    # Phase 3: PGD with very sharp alpha
    if _el() < budget_s - 400:
        h_t, c_t = _pgd(best_h, 3000, 5000, 80000, 10, 0.003)
        if c_t < best_c:
            best_c = c_t; best_h = h_t.copy()

    # Phase 4: Coordinate descent (medium step)
    if _el() < budget_s - 300:
        nit = min(int((budget_s - 300 - _el()) * 3000), 120000)
        best_h, best_c = _coord2(best_h, nit, 8e-4, rng)

    # Phase 5: PGD fine
    if _el() < budget_s - 250:
        h_t, c_t = _pgd(best_h, 2000, 20000, 200000, 6, 0.001)
        if c_t < best_c:
            best_c = c_t; best_h = h_t.copy()

    # Phase 6: Perturbation attempts
    if _el() < budget_s - 350:
        for scale in [0.008, 0.015, 0.004]:
            if _el() > budget_s - 300:
                break
            h_p = _repair(best_h + scale * rng.randn(n), n)
            h_p, c_p = _pgd(h_p, 1500, 2000, 30000, 10, 0.001)
            if c_p < best_c:
                best_c = c_p; best_h = h_p.copy()
            h_p, c_p = _coord2(h_p, 20000, 3e-4, rng)
            if c_p < best_c:
                best_c = c_p; best_h = h_p.copy()

    # Phase 7: Coordinate descent (medium)
    if _el() < budget_s - 150:
        nit = min(int((budget_s - 150 - _el()) * 2500), 100000)
        best_h, best_c = _coord2(best_h, nit, 3e-4, rng)

    # Phase 8: 3-point local search
    if _el() < budget_s - 100:
        nit = min(int((budget_s - 100 - _el()) * 1500), 60000)
        best_h, best_c = _coord3(best_h, nit, 4e-4, rng)

    # Phase 9: PGD ultra-fine
    if _el() < budget_s - 80:
        h_t, c_t = _pgd(best_h, 1500, 100000, 1000000, 4, 0.0004)
        if c_t < best_c:
            best_c = c_t; best_h = h_t.copy()

    # Phase 10: Fine coordinate descent
    if _el() < budget_s - 30:
        nit = min(int((budget_s - 30 - _el()) * 4000), 150000)
        best_h, best_c = _coord2(best_h, nit, 1e-4, rng)

    # Phase 11: Ultra-fine polish
    if _el() < budget_s - 3:
        nit = min(int((budget_s - 3 - _el()) * 5000), 100000)
        best_h, best_c = _coord2(best_h, nit, 3e-5, rng)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
