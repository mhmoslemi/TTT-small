def estimate_gradient(h):
    # Calculate gradient for all overlaps using analytical derivatives
    # C[k] = (1/n) * ∑_{i} h[i]*(1-h[i+|k|])
    n = len(h)
    k_max = n-1
    grad = np.zeros(n)
    k_ind = argmax_overlap(h)  # most active lag index
    k = k_ind

    if k >= 0:  # positive lag
        for i in range(n-k):
            # Derivative = (1 - h[i+k]) - h[i]
            grad[i] += (1 - h[i+k]) - h[i]
    else:  # negative lag
        for i in range(k, n):
            # Derivative = (1 - h[i]) - h[i+k]  (k negative)
            grad[i] += (1 - h[i]) - h[i+k]

    return grad

def argmax_overlap(h):
    c_all = np.correlate(h, 1-h, mode='full')
