import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_soft(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-16:
            continue
        wk = w[ii] * dx
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, a0, a1, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    mu = 0.7
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    y = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        alpha = a0 + (a1 - a0) * frac
        g = _grad_soft(h, alpha, top_k)
        y = _proj(h + step * (g + mu * v), target)
        v = mu * v + (1 - mu) * (y - h)
        c_try = _eval(y)
        if c_try < cur_c:
            h = y
            cur_c = c_try
            step = min(step * 1.05, step0 * 10)
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
        else:
            step = max(step * 0.6, step0 * 0.05)
            y = h.copy()
    return _repair(best_h, n), best_c

def _local2(h, n_iter, step, rng, temp=None):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 13)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and temp is not None and temp > 1e-14:
                if rng.rand() < np.exp(min(0.0, -(c_new - cur_c) / temp)):
                    accept = True
            if accept:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_block(h, n_iter, bmax, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        bsz = 1 + rng.randint(0, bmax - 1)
        j = (i + bsz + rng.randint(0, bmax * 3)) % n
        d = step * rng.randn()
        ok = True
        for k in range(bsz):
            v = h[(i + k) % n] + d
            if v < -1e-15 or v > 1.0 + 1e-15:
                ok = False
                break
        if not (0.0 - 1e-15 <= h[j] - bsz * d <= 1.0 + 1e-15):
            ok = False
        if ok:
            for k in range(bsz):
                h[(i + k) % n] += d
            h[j] -= bsz * d
            h = np.clip(h, 0, 1)
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                for k in range(bsz):
                    h[(i + k) % n] -= d
                h[j] += bsz * d
    return best_h, best_c

def _down(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _up(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    target = n * 0.5
    def obj(x):
        c = _corr_scaled(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        return _grad_soft(x, alpha, top_k)
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
