    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        m = -lag
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    return g

def _fista_smooth(h0, alpha, max_t, L0=5.0):
    n = len(h0)
    S = n * 0.5
    dx = 2.0 / n
    t0 = time.time()
    h = _proj(h0, S)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    L = L0
    best_c = _c5(h)
    best_h = h.copy()
    stale = 0
    for it in range(300000):
        if time.time() - t0 > max_t:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            y = _proj(y, S)
        corr = _corr(y)
        mx = float(np.max(corr))
        shifted = np.clip(alpha * (corr - mx), -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.one_hot(int(np.argmax(corr)), len(corr)).astype(float)
            wsum = 1.0
        else:
            weights /= wsum
        g = np.zeros(n)
        for j in range(len(corr)):
            if weights[j] > 1e-15:
                lag = j - (n - 1)
                g += weights[j] * _ovl_grad(y, lag)
        g *= dx
        step = 1.0 / max(L, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            accepted = True
            break
        hp = h
        h = xn
        L = max(L * 0.9, 1.0)
        if it % 15 == 0:
            c = _c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        if stale > 80:
            break
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _fista_topk(h0, k_top, max_t, L0=3.0):
    n = len(h0)
    S = n * 0.5
    dx = 2.0 / n
    t0 = time.time()
    h = _proj(h0, S)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    L = L0
    best_c = _c5(h)
    best_h = h.copy()
    stale = 0
    recompute_cnt = 0
    top_idx = np.argsort(_corr(h))[-k_top:]
    for it in range(300000):
        if time.time() - t0 > max_t:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            y = _proj(y, S)
        recompute_cnt += 1
        if recompute_cnt >= 50:
            top_idx = np.argsort(_corr(y))[-k_top:]
            recompute_cnt = 0
        g = np.zeros(n)
        for j in top_idx:
            lag = j - (n - 1)
            g += _ovl_grad(y, lag)
        g *= dx / len(top_idx)
        step = 1.0 / max(L, 1e-12)
        hp = h
        h = _proj(y - step * g, S)
        L = max(L * 0.92, 1.0)
        if it % 20 == 0:
            c = _c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        if stale > 100:
            break
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _local_search(h0, rng, max_t, step0=3e-5):
    n = len(h0)
    S = n * 0.5
    h = _proj(h0.copy(), S)
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    corr = _corr(h)
    mx = float(np.max(corr))
    tol = max(1e-7, 0.03 * (mx - float(np.min(corr))))
    active = np.nonzero(corr >= mx - tol)[0]
    it = 0
    while time.time() - t0 < max_t:
        it += 1
        if it % 300 == 0:
            c = _c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            corr = _corr(h)
            mx = float(np.max(corr))
            tol = max(1e-8, 0.02 * (mx - float(np.min(corr))))
            active = np.nonzero(corr >= mx - tol)[0]
            if len(active) < 1:
                active = np.array([int(np.argmax(corr))])
        frac = min(1.0, (time.time() - t0) / max_t)
        step = step0 * (0.01) ** frac
        if step < 1e-9:
            step = 1e-9
        r = rng.random()
        if r < 0.55:
            li = int(active[rng.randint(len(active))])
            lag = li - (n - 1)
            g = _ovl_grad(h, lag)
            i = int(np.argmin(g))
            j = int(np.argmax(g))
        elif r < 0.80:
            li = int(active[rng.randint(len(active))])
            lag = li - (n - 1)
            g = _ovl_grad(h, lag)
            vals = g.copy()
            vals[i_save] = 1e9 if False else vals[i_save]
            order = np.argsort(vals)
            i = int(order[0])
            j = int(order[-1])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            continue
        if h[i] >= h[j]:
            dirn = -1
        else:
            dirn = 1
        if dirn > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail < 1e-14:
            continue
        mult = 0.3 + 1.7 * rng.random()
        if rng.random() < 0.05:
            mult *= 5.0
        dv = dirn * min(avail, step * mult)
        h[i] += dv
        h[j] -= dv
        cn = _c5(h)
        if cn < c - 1e-14:
            c = cn
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] -= dv
            h[j] += dv
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, m * 0.5)
    x0 = (np.arange(n) + 0.5) / n
    x1 = (np.arange(m) + 0.5) / m
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _proj(h0, n0 * 0.5)
    best = h0.copy()
    best_c = _c5(best)

    e1 = t0 + min(40.0, 0.04 * budget_s)
    if e1 > time.time():
        for al in [20, 100, 500]:
            if time.time() > e1 - 2:
                break
            h, bh, bc = _fista_smooth(best, al, e1 - time.time() - 1, L0=5.0)
            if bc < best_c:
                best, best_c = bh, bc
            c = _c5(h)
            if c < best_c:
                best, best_c = h, c

    e2 = t0 + min(140.0, 0.14 * budget_s)
    if e2 > time.time():
        for kt in [1, 2, 3, 5]:
            if time.time() > e2 - 5:
                break
            h, bh, bc = _fista_topk(best, kt, e2 - time.time() - 2, L0=3.0)
            if bc < best_c:
                best, best_c = bh, bc
            c = _c5(h)
            if c < best_c:
                best, best_c = h, c

    e3 = t0 + min(260.0, 0.28 * budget_s)
    ms = [n0 // 2, n0 // 3, n0 // 4]
    for m in ms:
        if m < 24 or time.time() > e3 - 15:
            break
        hm = _resample(best, m)
        slot = min(40.0, e3 - time.time() - 5)
        if slot < 5:
            break
        h, bh, bc = _fista_smooth(hm, 200, slot * 0.5, L0=4.0)
        h2, bh2, bc2 = _fista_topk(h, 2, slot * 0.4, L0=3.0)
        cand = bh2 if bc2 <= bc else bh
        c = _c5(cand)
        if len(cand) != n0:
            cu = _resample(cand, n0)
            c = _c5(cu)
            cand = cu
        if c < best_c:
            best, best_c = cand, c

    e4 = t0 + budget_s - 280.0
    for al in [200, 1000, 5000, 20000, 100000]:
        if time.time() > e4 - 5:
            break
        slot = min(25.0, e4 - time.time() - 2)
        if slot < 3:
            break
        h, bh, bc = _fista_smooth(best, al, slot, L0=5.0)
        if bc < best_c:
            best, best_c = bh, bc
        c = _c5(h)
        if c < best_c:
            best, best_c = h, c

    e5 = t0 + budget_s - 180.0
    if e5 > time.time():
        for kt in [1, 2, 3]:
            if time.time() > e5 - 5:
                break
            h, bh, bc = _fista_topk(best, kt, e5 - time.time() - 2, L0=2.0)
            if bc < best_c:
                best, best_c = bh, bc
            c = _c5(h)
            if c < best_c:
                best, best_c = h, c

    e6 = t0 + budget_s - 2.0
    if e6 > time.time():
        best, best_c = _local_search(best, rng, e6 - time.time(), 4e-5)

    def _flip(x):
        return _proj(1.0 - x, len(x) * 0.5)
    def _rev(x):
        return _proj(x[::-1].copy(), len(x) * 0.5)
    def _asym(x):
        return _proj(0.5 + 0.5 * (x - x[::-1]), len(x) * 0.5)
    def _sym(x):
        return _proj(0.5 + 0.5 * (x + x[::-1] - 1.0), len(x) * 0.5)
    for fn in [_flip, _rev, _asym, _sym]:
        hv = fn(best)
        c = _c5(hv)
        if c < best_c:
            best, best_c = hv, c

    best = _proj(best, len(best) * 0.5)
    best_c = _c5(best)
    return best, best_c, len(best)
