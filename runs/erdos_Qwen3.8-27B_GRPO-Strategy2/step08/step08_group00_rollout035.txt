import numpy as np

import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj_box_sum(h, target):
    """Project onto {0<=x<=1, sum(x)=target}"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_vals(h):
    """Compute all correlation values (scaled by dx)"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _subgrad_at_lag(h, k_lag):
    """Subgradient of f_k(h) = sum_j h[j]*(1-h[j+k]) for the given lag index k_lag (0 to 2n-2)"""
    n = len(h)
    g = np.zeros(n)
    L = k_lag - (n - 1)  # actual lag (negative to positive)
    if L >= 0:
        # f = sum_{j=0}^{n-1-L} h[j] * (1 - h[j+L])
        # df/dh[j] for j in [0, n-1-L]: (1 - h[j+L])
        # df/dh[j] for j in [L, n-1]: -h[j-L]
        for j in range(n - L):
            g[j] += (1.0 - h[j + L])
        for j in range(L, n):
            g[j] -= h[j - L]
    else:
        M = -L
        # f = sum_{j=0}^{n-1-M} h[j+M] * (1 - h[j])
        # df/dh[j] for j in [0, n-1-M]: -h[j+M]  (from the (1-h[j]) term)
        # df/dh[j] for j in [M, n-1]: (1 - h[j-M])  (from the h[j+M] term)
        for j in range(n - M):
            g[j] -= h[j + M]
        for j in range(M, n):
            g[j] += (1.0 - h[j - M])
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lag(h, lag_idx):
    n = len(h)
    g = np.zeros(n)
    L = lag_idx - (n - 1)
    if L >= 0:
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    else:
        M = -L
        g[:n - M] -= h[M:]
        g[M:] += (1.0 - h[:n - M])
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
            if a < 1e-14:
                break
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return h

def _sum_fn(x, n):
    return x.sum() - n * 0.5
