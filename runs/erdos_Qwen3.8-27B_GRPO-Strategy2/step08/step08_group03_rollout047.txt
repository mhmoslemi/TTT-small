        # For m >= x and m <= n-1: h[n-1+x-m] is valid (index from x to n-1)
        # So t2_full[n-1+x] = sum_{m=x}^{n-1} g[m]*h[n-1+x-m]
        # = sum_{m=x}^{n-1} u[n-1-m]*h[n-1+x-m]

        # But we only need t2[x] for the valid range, and the FFT linear conv handles zeros correctly
        q = -(t1 + t2)
    else:
        q = np.zeros(n)
    
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

import numpy as np
import time

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(12):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    return _proj(h, n * 0.5)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = float(max(P[0], np.max(P[1:]) if n > 1 else 0.0, np.max(Nv[1:]) if n > 1 else 0.0))
    return P, Nv, mx

def _smooth(h, alpha):
    P, Nv, mx = _pn_fft(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / alpha)

def _grad_w(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        Hf = np.fft.rfft(h, Nf)
        Ur = np.fft.rfft(u[::-1], Nf)
        Uh = np.fft.rfft(u, Nf)
        t1 = np.fft.irfft(Hf * Ur, Nf).real[:n]
        t2f = np.fft.irfft(Hf * Uh, Nf).real
        t2 = t2f[n - 1:2 * n - 1]
        q = -(t1 + t2)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _grad_smooth(h, alpha):
    P, Nv, mx = _pn_fft(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
    if tot < 1e-300:
        wP[0] = 1.0
        return _grad_w(h, wP, wN)
    wP[0] = e0 / tot
    if n > 1:
        wP[1:] = ep / tot

    return _grad_w(h, wP, wN)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _active_grad(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(max(P[0], np.max(P[1:]) if n > 1 else 0.0, np.max(Nv[1:]) if n > 1 else 0.0))
    tol = max(2e-10, 6.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += int(pos.size)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += int(pos.size)
    if cnt == 0:
        return np.zeros(n)
    wP *= 1.0 / cnt
    wN *= 1.0 / cnt
    return _grad_w(h, wP, wN)

def _local_search(h0, rng, deadline, start_step=3.0e-5):
    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    P, Nv, _ = _pn_fft(h)
    g = _active_grad(h, P, Nv)
    order = np.argsort(g)
    q = max(8, n // 6)
    low = order[:q]
    high = order[-q:]
    recompute = 1024
    it = 0
    T0 = 0.02
    while time.time() < deadline:
        if it >= recompute:
            c2 = _c5(h)
            if c2 < best_c:

                best_h = h.copy()
            c = c2
            P, Nv, _ = _pn_fft(h)
            g = _active_grad(h, P, Nv)
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            recompute = it + 1024
        frac = (time.time() - t0) / max(1e-9, deadline - t0)
        base = max(1e-11, start_step * (1e-4) ** frac)
        T = T0 * (1.0 - frac) ** 2
        if rng.random() < 0.75:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            it += 1
            continue
        if g[i] < g[j]:
            s = 1
        elif g[j] < g[i]:
            s = -1
        else:
            s = 1 if rng.random() < 0.5 else -1
        avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
        if avail <= 1e-14:
            it += 1
            continue
        mult = 0.3 + 1.7 * rng.random()
        if rng.random() < 0.02:
            mult *= 6.0
        d = s * min(avail, base * mult)
        h[i] += d
        h[j] -= d
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn
            if c < best_c - 1e-15:

        else:

            if accept and T > 1e-8:

            else:
                h[i] -= d
                h[j] += d
