def _repair(h, n):

    target = n / 2.0

    if abs(s - target) > 1e-9:

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:

    return h

def _proj(h, target):

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

    n = len(h)

    m = c.max()
