def _proj(h, target):

def _smooth_grad(h, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    idx = np.argpartition(c, -m_lags)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()

            g[L:] += wk * (1.0 - h[:n-L])
        else:
            M = -L
            g[M:] -= wk * h[:n-M]
            g[:n-M] += wk * (1.0 - h[M:])
