import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} using bisection on threshold."""
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = _proj_sum(h, target)
    # Fix any tiny numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx] + 1e-15)
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx] - 1e-15)
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_topk(h, k, tol):
    """Exact subgradient using top-k shifts within tol of max."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    # Find shifts within tol of max
    active = np.where(c >= m - tol)[0]
    if len(active) > k:
        # Keep top k
        active = np.argsort(-c[active])[:k] + np.sort(active)
    dx = 2.0 / n
    g = np.zeros(n)
    for kk in active:
        sh = (n - 1) - kk
        if sh >= 0:
            # Positive shift: g[i] gets contribution from (1-h[i+sh]) for i < n-sh
            # and -h[i-sh] for i >= sh
            if n - sh > 0:
                g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n-sp] -= h[sp:]
    g *= dx / max(len(active), 1)
    return g, m

def _smooth_grad(h, alpha, topk):
    """Smooth max gradient."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        kk = idx[ii]; sh = (n - 1) - kk; wk = w[ii] * dx
        if sh >= 0:
            if n - sh > 0:
                g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g, m

def _nesterov_fista(h, mode, nsteps, step0, deadline, alpha=10000, topk=20, tol=0.005, beta=0.85):
    """FISTA-like accelerated gradient with projection."""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    y = h.copy()
    v = np.zeros(n)
    t = 1.0
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(nsteps):
        if time.time() > deadline:
            break
        if mode == 'exact':
            g, _ = _grad_topk(y, topk, tol)
        else:
            g, _ = _smooth_grad(y, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        # Backtracking line search
        h_new = _proj_sum(y - step * v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            # Accept
            t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
            y = h_new + (t/t_new - 1) * (h_new - h)
            h = h_new
            f = f_new
            t = t_new
            step = min(step * 1.02, step0 * 20)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h_new = _proj_sum(y - step * v, target)
            f_new = _eval(h_new)
            h = h_new
            f = f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _slsqp(h, alpha, topk, maxit, deadline):
    """SLSQP refinement with smooth max objective."""
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]
            ct = c[idx]
        else:
            ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + (np.log(np.sum(np.exp(s)))) / alpha
    
    def jac(x):
        return _smooth_grad(x, alpha, topk)[0]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)]*n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _pair_polish(h, niche, rng, deadline, scale=1e-5, niters=100000):
    """Fine pairwise polish."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, niche)) % n
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0)**(it/max(niter,1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.005 * max(T, 1e-10)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n-c5c)/max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    h = _proj_sum(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx] + 1e-15)
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx] - 1e-15)
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_shift(h, kk, n, dx):
    g = np.zeros(n)
    sh = (n - 1) - kk
    if sh >= 0:
        if n - sh > 0:
            g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g

def _grad_active(h, active, n, dx):
    g = np.zeros(n)
    for kk in active:
        g += _grad_shift(h, kk, n, dx)
    g *= dx / max(len(active), 1)
    return g

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        sh = (n - 1) - idx[ii]
        wk = w[ii] * dx
        if sh >= 0:
            if n - sh > 0: g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _fista(h, mode, nsteps, step0, deadline, alpha=10000, topk=15, tol=0.003, beta=0.85):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    y = h.copy()
    for it in range(nsteps):
        if time.time() > deadline:
            break
        if mode == 'exact':
            c = _correl(y)
            m = c.max()
            active = np.where(c >= m - tol)[0]
            if len(active) > topk:
                active = np.argsort(-c[active])[:topk] + np.sort(active)
            g = _grad_active(y, active, n, 2.0/n)
        else:
            g, _ = _grad_smax(y, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(y - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            y = hn + (beta - 1.0) / (1.0 - beta) * (hn - y) * 0.5
            h, f = hn, fn
            step = min(step * 1.012, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            hn = _proj_sum(y - step * v, target)
            h, f = hn, _eval(hn)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _focus(h, ntop, deadline, step0=0.001):
    n = len(h)
    c = _correl(h)
    top_idx = np.argsort(-c)[:ntop]
    dx = 2.0 / n
    best_h, best_f = h.copy(), _eval(h)
    step = step0
    v = np.zeros(n)
    beta = 0.9
    target = n / 2
    for _ in range(8000):
        if time.time() > deadline:
            break
        g = _grad_active(h, top_idx, n, dx)
        v = beta * v + (1 - beta) * g
        hn = _proj_sum(h - step * v, target)
        fn = _eval(hn)
        if fn < _eval(h) - 1e-16:
            h = hn
            step = min(step * 1.015, step0 * 40)
            if fn < best_f:
                best_f, best_h = fn, h.copy()
        else:
            step *= 0.5
            hn = _proj_sum(h - step * v, target)
            h = hn
            fh = _eval(h)
            if fh < best_f:
                best_f, best_h = fh, h.copy()
    return best_h, best_f

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400); s -= s.max()
        return m + np.log(np.sum(np.exp(s))) / alpha
    def jac(x):
        return _grad_smax(x, alpha, topk)[0]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=[(0.0,1.0)]*n,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0)**(it/max(niter,1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-10)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n-c5c)/max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _sweep_polish(h, deadline, scale=1e-6):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    step = 0
    while time.time() < deadline:
        step += 1
        i = (step * 7 + 3) % n
        j = (i + 1 + (step % 5)) % n
        d = scale * (2.0 * ((step % 7) / 7.0) - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
