import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _soft_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _soft_grad(h, alpha, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_lag(idx[ii], h, n)
    return g

def _fista(h0, alpha, n_iter, top_k, t_end):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    tt = 1.0
    step = 1.0 / (alpha * 10.0)
    best_x = x.copy()
    best_f = _soft_obj(x, alpha)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _soft_grad(y, alpha, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        y = x - step * g
        x_new = _proj(y, n * 0.5)
        tt_new = (1.0 + np.sqrt(1.0 + 4.0 * tt * tt)) / 2.0
        y = x_new + (tt / tt_new - 1.0) * (x_new - x)
        x = x_new
        tt = tt_new
        f = _soft_obj(x, alpha)
        if f < best_f:
            best_f = f
            best_x = x.copy()
        if it % 100 == 99:
            step = min(step * 1.2, 5.0 / (alpha + 1e-8))
        else:
            step = min(step * 1.01, 5.0 / (alpha + 1e-8))
    return best_x

def _grad_swap(h, n_iter, rng, t_end):
    n = len(h)
    c5c, _ = _eval_worst(h)
    best_h = h.copy()
    best_c = c5c
    alpha = 50000.0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c5c, k_w = _eval_worst(h)
        g = _grad_lag(k_w, h, n)
        gi = np.argmax(g)
        gj = np.argmin(g)
        if g[gi] - g[gj] < 1e-12:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = 1e-5 * rng.randn()
            if 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1:
                h[i] += d; h[j] -= d
                cn = _eval(h)
                if cn < c5c - 1e-15:
                    c5c = cn
                    if cn < best_c: best_c = cn; best_h = h.copy()
                else: h[i] -= d; h[j] += d
            continue
        maxd = min(h[gi], 1.0 - h[gj])
        if maxd < 1e-12:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = 1e-5 * rng.randn()
            if 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1:
                h[i] += d; h[j] -= d
                cn = _eval(h)
                if cn < c5c - 1e-15:
                    c5c = cn
                    if cn < best_c: best_c = cn; best_h = h.copy()
                else: h[i] -= d; h[j] += d
            continue
        moved = False
        for frac in [0.05, 0.15, 0.30, 0.50, 0.80]:
            d = maxd * frac
            h[gi] -= d; h[gj] += d
            cn, _ = _eval_worst(h)
            if cn < c5c - 1e-15:
                c5c = cn
                moved = True
                if cn < best_c: best_c = cn; best_h = h.copy()
                break
            h[gi] += d; h[gj] -= d
        if not moved:
            top_hi = np.argsort(g)[-8:]
            top_lo = np.argsort(g)[:8]
            for _ in range(4):
                i = top_hi[rng.randint(len(top_hi))]
                j = top_lo[rng.randint(len(top_lo))]
                if i == j: continue
                maxd2 = min(h[i], 1.0 - h[j])
                if maxd2 < 1e-12: continue
                d2 = maxd2 * rng.uniform(0.05, 0.5)
                h[i] -= d2; h[j] += d2
                cn, _ = _eval_worst(h)
                if cn < c5c - 1e-15:
                    c5c = cn; moved = True
                    if cn < best_c: best_c = cn; best_h = h.copy()
                    break
                h[i] += d2; h[j] -= d2

def _fine_2pt(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _slsqp_focus(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx

            idx = np.arange(len(c))

        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)

    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]

            idx = np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(idx[ii], x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_sumfun, 'jac': _slsqp_sumjac}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

def _slsqp_sumfun(x):
    return x.sum() - len(x) / 2.0

def _slsqp_sumjac(x):
    return np.ones(len(x))
