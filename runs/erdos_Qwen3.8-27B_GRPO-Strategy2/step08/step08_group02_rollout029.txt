    if sh >= 0:
        m = n - sh
        i1 = np.arange(m)
        g[i1] += 1.0 - h[i1 + sh]
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
        else:
            i2 = np.arange(n)
            g[i2] -= h[i2]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += 1.0 - h[i1 - sp]
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx
```

import numpy as np

from scipy.optimize import linprog

def _repair(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    s = h.sum()
    for _ in range(80):
        if abs(s - target) < 1e-13:
            break
        if s < 1e-15:
            k = int(target)
            h = np.zeros(h.size, dtype=float)
            h[:k] = 1.0
            if k < h.size:
                h[k] = target - k
            return h
        h = np.clip(h * (target / s), 0.0, 1.0)
        s = h.sum()
    d = target - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0.0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            if d > 0.0:
                a = min(d, 1.0 - h[i])
                if a > 1e-18:
                    h[i] += a
                    d -= a
            else:
                a = min(-d, h[i])
                if a > 1e-18:
                    h[i] -= a
                    d += a
    return h

def _proj(y, target):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        y = np.where(np.isfinite(y), y, 0.0)
    n = y.size
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _corr(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5(h):
    return float(np.max(_corr(h)))

def _top_desc(c, k):
    k = int(k)
    m = c.size
    if k <= 0:
        return np.array([], dtype=int)
    if k >= m:
        return np.argsort(c)[::-1].astype(int)
    idx = np.argpartition(c, m - k)[-k:]
    return idx[np.argsort(c[idx])[::-1]].astype(int)

def _grad_shift(sh, h, n, dx):
    g = np.zeros(n, dtype=float)
    if sh >= 0:
        m = n - sh
        if m > 0:
            i1 = np.arange(m)
            g[i1] += 1.0 - h[i1 + sh]
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
        else:
            g -= h
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += 1.0 - h[i1 - sp]
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _grad_smooth(c, h, alpha, top_k, dx):
    n = h.size
    m = c.size
    if top_k <= 0 or top_k >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, m - int(top_k))[-int(top_k):]
    cs = c[idx]
    s = alpha * (cs - float(cs.max()))
    s = np.clip(s - s.max(), -50.0, 50.0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for ii in range(idx.size):
        wi = float(w[ii])
        if wi < 1e-14:
            continue
        sh = int(n - 1 - idx[ii])
        g += wi * _grad_shift(sh, h, n, dx)
    g -= g.mean()
    return g

def _proj_grad_phase(h, alpha, max_iters, step0, top_k, momentum, stop):

def _line_dir(h, d, cur):
    d = np.asarray(d, dtype=float)
    d = d - d.mean()
    if not np.all(np.isfinite(d)):
        return None, cur, False
    pos = d > 1e-12
    neg = d < -1e-12
    maxa = 1.0
    if np.any(pos):
        maxa = min(maxa, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        maxa = min(maxa, float(np.min((-h[neg]) / d[neg])))

        return None, cur, False

    bestc = cur

            continue

        if cn < bestc - 1e-14:
            bestc = cn
            besth = cand.copy()
    if besth is None:

    return besth, bestc, True

def _lp_dir(c, h, top, dx):
    n = h.size
    order = _top_desc(c, top)
    if order.size == 0:
        return None
    rows = []
    for k in order:
        sh = int(n - 1 - k)
        g = _grad_shift(sh, h, n, dx)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            continue
        row = np.empty(n + 1, dtype=float)
        row[:n] = g / gn
        row[n] = 1.0
        rows.append(row)
    if not rows:
        return None
    A_ub = np.vstack(rows)
    b_ub = np.zeros(A_ub.shape[0], dtype=float)
    A_eq = np.ones((1, n + 1), dtype=float)
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1, dtype=float)
    cvec = np.zeros(n + 1, dtype=float)
    cvec[n] = -1.0
    bounds = []
    for i in range(n):
        hi = float(h[i])
        bounds.append((-hi, 1.0 - hi))
    bounds.append((0.0, 1.0))
    try:
        res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'disp': False})
    except Exception:
        return None
    if res is None or not res.success:
        return None
    x = res.x
    if x is None or not np.all(np.isfinite(x)):
        return None
    if float(x[n]) < 1e-5:
        return None
    d = x[:n].copy()
    d[np.abs(d) < 1e-10] = 0.0
    d = d - d.mean()
    if float(np.linalg.norm(d)) < 1e-10:
        return None
    return d

def _try_pair(h, src, dst, fracs, bestc):
    maxt = min(h[src], 1.0 - h[dst]) * 0.999999
    if maxt <= 1e-12:
        return None, bestc
    besth = None
    orig0 = float(h[src])
    orig1 = float(h[dst])
    for f in fracs:
        t = f * maxt
        if t <= 1e-12:
            continue
        h[src] = orig0 - t
        h[dst] = orig1 + t
        cn = _c5(h)
        if cn < bestc - 1e-14:
            bestc = cn
            besth = h.copy()
        h[src] = orig0
        h[dst] = orig1
    return besth, bestc

        h0, c0 = _proj_grad_phase(h0, 3000.0, 100000, 0.02, 8, 0.65, stop01)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()
        h0, c0 = _active_pair_phase(h0, stop02, 100000, 4, 6, f5)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()
        h0, c0 = _lp_phase(h0, stop03, 100000, (3, 2), 2000.0, 8)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()
        h0, c0 = _rand_pair_phase(h0, stop04, 10000000, rng, 2000.0, 8, -8.0, -3.0)
        if c0 < best_c:
            best_c = c0
            best_h = h0.copy()

        if n0 < 800 and budget_s > 150.0:
            factor = 2
        else:
            factor = 1
        nA = n0 * factor
        if factor == 1:
            hA = h0.copy()
            cA = _c5(hA)
        else:
            hA = _repair(_upsample(h0, factor), 0.5 * nA)
            cA = _c5(hA)
            hAi = _repair(_upsample(h_init, factor), 0.5 * nA)
            cAi = _c5(hAi)
            if cAi < cA:
                hA = hAi
                cA = cAi
        if cA < best_c:
            best_c = cA
            best_h = hA.copy()
            best_n = nA

        stopA1 = t0 + 0.50 * budget_s
        stopA2 = t0 + 0.65 * budget_s
        stopA3 = t0 + 0.80 * budget_s
        stopA4 = t0 + 0.90 * budget_s

        if time.time() < stopA1:
            hA, cA = _proj_grad_phase(hA, 2000.0, 100000, 0.012, 8, 0.7, stopA1)
            if cA < best_c:
                best_c = cA
                best_h = hA.copy()
                best_n = nA
        if time.time() < stopA2:
            hA, cA = _active_pair_phase(hA, stopA2, 100000, 3, 5, f4)
            if cA < best_c:
                best_c = cA
                best_h = hA.copy()
                best_n = nA
        if time.time() < stopA3:
            hA, cA = _lp_phase(hA, stopA3, 100000, (3, 2), 1500.0, 6)
            if cA < best_c:
                best_c = cA
                best_h = hA.copy()
                best_n = nA
        if time.time() < stopA4:
            hA, cA = _rand_pair_phase(hA, stopA4, 10000000, rng, 1500.0, 6, -9.0, -4.0)
            if cA < best_c:
                best_c = cA
                best_h = hA.copy()
                best_n = nA

        final_stop = t0 + budget_s - 3.0
        if time.time() < final_stop:
            best_h, best_c = _active_pair_phase(best_h, final_stop, 100000, 3, 4, f3)
    except Exception:
        pass
    best_h = _repair(best_h, 0.5 * best_n)
    best_c = _c5(best_h)
    return best_h, float(best_c), int(best_n)
