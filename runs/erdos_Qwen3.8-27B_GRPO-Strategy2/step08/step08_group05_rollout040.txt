import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_descent(h, n_steps, eps, m_top, deadline, rng):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy(); c5_cur = best_c5
    stalls = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n)); b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stalls += 1
        else:
            d = res.x
            step = 1.0
            improved = False
            for _ in range(10):
                h_new = _repair(c + step * d, n)
                c5_new = _eval(h_new)
                if c5_new < c5_cur - 1e-16:
                    c, c5_cur = h_new, c5_new
                    improved = True
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, c.copy()
                    break
                step *= 0.5
            if not improved:
                stalls += 1
            else:
                stalls = 0
        if stalls >= 8:
            d2 = rng.randn(n); d2 -= d2.mean()
            d2 *= 0.003 / max(np.max(np.abs(d2)), 1e-15)

                if c5_cur < best_c5:

            stalls = 0
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        k, sh, wk = idx[ii], (n-1)-idx[ii], w[ii]*dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-h[sh:])
            if sh > 0: g[sh:] -= wk*h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta, step = 0.8, step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta*v + (1.0-beta)*g
        hn = _proj(h - step*v, n*0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn; step = min(step*1.04, step0*6.0)
        else:
            step *= 0.5
            if step < 1e-11: break
            h = _proj(hp - step*v, n*0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax_obj(x, alpha, topk)
    def jac(x): return _grad_topk(x, alpha, topk)
    def cons_fun(x): return x.sum() - n*0.5
    def cons_jac(x): return np.ones(n)
    cons = [{'type':'eq','fun':cons_fun,'jac':cons_jac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter':maxit,'ftol':1e-22,'maxls':80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _targeted_polish(h, niter, rng, deadline):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    cor = _correl(h); peak_k = np.argmax(cor)
    for it in range(niter):
        if time.time() > deadline: return bh, bc
        if it % 2000 == 0:
            cor = _correl(h); peak_k = np.argmax(cor)
        sh = (n-1) - peak_k
        if sh >= 0:
            pairs = [(i, i+sh) for i in range(max(1,n//4), min(n-sh, 3*n//4))]
        else:
            sp = -sh
            pairs = [(i+sp, i) for i in range(max(1,n//4), min(n-sp, 3*n//4))]
        if not pairs:
            i = rng.randint(n); j = (i+1+rng.randint(0,5))%n
            pairs = [(i,j)]
        i, j = pairs[rng.randint(len(pairs))]
        d = 1e-4*(2*rng.random()-1)
        ni, nj = h[i]+d, h[j]-d
        if 0<=ni<=1 and 0<=nj<=1:
            h[i],h[j] = ni,nj; c5n = _eval(h)
            if c5n < c5c-1e-19:
                c5c=c5n
                if c5n<bc: bc,bh=c5n,h.copy()
            else:
                h[i],h[j] = ni-d, nj+d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    blocks = [2,3,4,5,6,8,10]
    for it in range(niter):
        if time.time() > deadline: break
        T = T0*(T1/T0)**(it/max(niter,1))

        i0 = rng.randint(0, max(1, n-bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004*max(T,1e-9)**0.5*rng.randn(bs); d -= d.mean()
        seg = np.clip(seg+d, 0, 1)

        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n-c5c)/max(T,1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc
