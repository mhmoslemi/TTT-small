    # Phase 1: L-BFGS-B with adaptive alpha (fast local)
    dl1 = min(deadline, t0 + 70)
    for alpha, tk in [(3000, 40), (15000, 15), (60000, 6)]:
        if time.time() > dl1:
            break
        h, c5v = _lbfgs(best_h, alpha, tk, dl1, maxiter=100)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Targeted GD on worst lags
    dl2 = min(deadline, t0 + 200)
    for step0, tk, bt in [(0.004, 5, 0.80), (0.002, 3, 0.85), (0.003, 8, 0.78)]:
        if time.time() > dl2:
            break
        h2, c5v = _gd_targeted(best_h, 3000, step0, dl2, topk=tk, beta=bt)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Nesterov soft-max (multiple configs)
    dl3 = min(deadline, t0 + 420)
    nst_cfgs = [(30000, 10, 0.004, 0.84), (80000, 5, 0.002, 0.88), (150000, 4, 0.001, 0.90)]
    for alpha, topk, step0, beta in nst_cfgs:
        if time.time() > dl3:
            break
        h2, c5v = _nesterov(best_h, alpha, topk, 5000, step0, dl3, beta=beta)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: L-BFGS again from improved point
    dl4 = min(deadline, t0 + 500)
    for alpha, tk in [(50000, 8), (200000, 4)]:
        if time.time() > dl4:
            break
        h, c5v = _lbfgs(best_h, alpha, tk, dl4, maxiter=80)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 5: Jitter + re-optimize (multiple attempts)
    dl5 = min(deadline, t0 + 700)
    for _ in range(7):
        if time.time() > dl5:
            break
        hj, cj = _jitter_reopt(best_h, rng, dl5, 60000, 6, 0.003)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 6: Targeted annealing
    dl6 = min(deadline, t0 + 920)
    best_h, best_c5 = _anneal(best_h, 100000, rng, dl6, T0=0.007, T1=1e-13)

    # Phase 7: Final L-BFGS + Nesterov
    dl7 = min(deadline, t0 + budget_s - 25)
    h, c5v = _lbfgs(best_h, 100000, 5, dl7, maxiter=60)
    if c5v < best_c5:
        best_c5, best_h = c5v, h.copy()
    h2, c5v = _nesterov(best_h, 80000, 5, 2000, 0.001, dl7, beta=0.91)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 8: Fine polish
    dl8 = deadline
    rem = max(0, dl8 - time.time())
    n_pol = min(int(rem / 0.0018), 40000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, dl8, scale=3e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
