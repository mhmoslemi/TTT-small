    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_shift(h, k, n):
    """Gradient of the k-th correlation term w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _grad_weighted(h, weights, k_indices):
    """Weighted sum of gradients for given shift indices"""
    n = len(h)
    g = np.zeros(n)
    for i, k in enumerate(k_indices):
        g += weights[i] * _grad_shift(h, k, n)
    return g

def _nesterov_bb(h, deadline, tol, step0, beta, n_steps):
    """Nesterov with Barzilai-Borwein step and restart"""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    g = _grad_weighted(h, np.ones(len(active))/len(active), active)
    v = g.copy()
    step = step0
    best_h, best_f = h.copy(), f
    prev_h, prev_g = h.copy(), g.copy()
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        
        if fn < f - 1e-15:
            s = hn - hp
            # BB step
            y = None  # would need new gradient
            h, f = hn, fn
            if f < best_f:
                best_f, best_h = f, h.copy()
            
            # Update gradient
            c = _correl(h)
            m = c.max()
            active = np.where(c >= m - tol)[0]
            g_new = _grad_weighted(h, np.ones(len(active))/len(active), active)
            
            # BB step size
            sg = np.sum(s * (g_new - g))
            gg = np.sum((g_new - g)**2)
            if gg > 1e-20:
                bb_step = max(sg / gg, 1e-6)
                step = 0.5 * step + 0.5 * min(bb_step, step0 * 20)
            
            g = g_new
            # FISTA restart
            if np.sum((h - prev_h) * (g - prev_g)) > 0:
                v = g
            prev_h, prev_g = h.copy(), g.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    
    return _repair(best_h, n), best_f

def _eval(h):

def _correl(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5

            lo = mid
        else:
            hi = mid

    if abs(s - target) > 1e-9:
        h = np.clip(h * target / s, 0.0, 1.0)

            for idx in order:

                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

def _shift_grad(h, k, n):

            g[sh:] -= h[:-sh]
    else:

        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

                g += _shift_grad(h, kk, n)
