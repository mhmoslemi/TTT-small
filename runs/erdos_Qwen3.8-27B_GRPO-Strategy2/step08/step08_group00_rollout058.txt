def _grad_sm(h, alpha, nl):
    n = len(h)
    dx = 2.0/n
    c = _corr(h)*dx
    m = c.max()
    top = np.argsort(c)[-nl:]
    s = np.clip(alpha*(c[top]-m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(top)):
        wk = w[ii]*dx
        if wk < 1e-16:
            continue
        L = top[ii] - (n-1)
        if L >= 0:
            r = n - L
            g[L:] += wk*(1.0-h[:r])
            g[:r] -= wk*h[L:]
        else:
            M = -L
            r = n - M
            g[:r] += wk*(1.0-h[M:])
            g[M:] -= wk*h[:r]
    return g
