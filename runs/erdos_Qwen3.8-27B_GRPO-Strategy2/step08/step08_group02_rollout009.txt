import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c)), c

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _binary_init(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _sa_binary(h, n, n_iter, rng, T0):
    c5c, kw, _ = _eval_worst(h)
    best_h = h.copy()
    best_c = c5c
    T = T0
    for it in range(n_iter):
        T = T0 * (0.0005) ** (float(it) / max(n_iter, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5n, kw, _ = _eval_worst(h)
        d = c5n - c5c
        if d < 0 or (T > 1e-12 and rng.rand() < np.exp(-d / T)):
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

def _targeted_binary(h, n, n_iter, rng):
    c5c, kw, _ = _eval_worst(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        if it % 100 == 0:
            c5c, kw, _ = _eval_worst(h)

                best_h = h.copy()

        if sh >= 0:

        else:
            valid = np.arange(-sh, n)
            idx_shift = valid + sh

        zeros_mask = h[valid] < 0.6

        j_pool = valid[zeros_mask]

            j = j_pool[rng.randint(len(j_pool))]

            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]

    return best_h, best_c

def _proj_grad(h, alpha, steps, step0, top_k, momentum, n):
    v = np.zeros(n)
    cur_c = _eval(h)
    step = step0
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -300, 300)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _grad_lag(active[ii], h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(20):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 20.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.3, step0 * 20.0)
        if it % 500 == 0:
            v *= 0.2
    return _repair(h, n)

def _slsqp_focus(h, alpha, maxiter, top_k, n):
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -300, 300)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -300, 300)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _grad_lag(idx[ii], x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _fine_local(h, n_iter, rng, base_step, n):

    best_h = h.copy()
