def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1); pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _sval(h, alpha):
    P, Nv, mx = _pn(h)
    n = len(h); dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _gfw(h, wP, wN):
    n = len(h); dx = 2.0 / n
    u = np.zeros(n)
    if n > 1: u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        convR = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        convL = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(convR[n - 1:2*n-1].real + convL[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sgrad(h, alpha):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0: tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gfw(h, wP, wN)

def _fista(h0, alpha, max_time, L0=8.0):
    if max_time <= 0.0:
        h = h0.copy(); return h, h.copy(), _c5(h)
    n = len(h0); S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n); hp = h.copy(); y = h.copy()
    t = 1.0; fh = _sval(h, alpha); best_h = h.copy(); best_c = _c5(h)
    stale = 0; Lc = L0
    for it in range(800000):
        if time.time() - t0 > max_time: break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp); t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _sval(y, alpha)
        else:
            y = h; fy = fh
        g = _sgrad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        acc = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _sval(xn, alpha)
            d = xn - y
            if fx <= fy + float(np.dot(g, d)) + 0.5 * np.dot(d, d) / step + 1e-11:
                acc = True; break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if acc:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.9, 1.0)
            if not it or it % 10 == 0:
                c = _c5(h)
                if c < best_c: best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 15: break
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _rg(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _agrad(h, P, Nv):
    n = len(h); dx = 2.0 / n
    mx = float(P[0])
    if n > 1: mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 6.0 * dx)
    wP = np.zeros(n); wN = np.zeros(n); cnt = 0
    if P[0] >= mx - tol: wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1; wP[pos] = 1.0; cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1; wN[pos] = 1.0; cnt += pos.size
    if cnt == 0: return np.zeros(n)
    inv = 1.0 / cnt; wP *= inv; wN *= inv
    return _gfw(h, wP, wN)

def _lsearch(h0, rng, deadline, s0=1e-4):
    n = len(h0); h = h0.copy(); c = _c5(h)
    best_c = c; best_h = h.copy(); tl0 = time.time()
    P, Nv, _ = _pn(h); g = _agrad(h, P, Nv)
    recompute = 512; it = 0
    while time.time() < deadline:
        if it >= recompute:
            c = _c5(h)
            if c < best_c: best_c = c; best_h = h.copy()
            P, Nv, _ = _pn(h); g = _agrad(h, P, Nv)
            recompute = it + 512
        frac = min(1.0, (time.time() - tl0) / max(1e-9, deadline - tl0))
        base = max(1e-11, s0 * (1e-7) ** frac)
        r = rng.random()
        if r < 0.6:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        elif r < 0.8:
            q = max(4, n // 16)
            lo_i = np.argsort(g)[:q]; hi_i = np.argsort(g)[-q:]
            i = int(lo_i[rng.randint(q)]); j = int(hi_i[rng.randint(q)])
        else:
            bl = rng.randint(2, 5)
            i = int(rng.randint(max(1, n - bl))); j = int(rng.randint(max(1, n - bl)))
            if abs(i - j) < bl: i = 0; j = n // 2
        if i == j: it += 1; continue
        if g[i] <= g[j]: s = 1
        else: s = -1
        if s > 0: avail = min(1.0 - h[i], h[j])
        else: avail = min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.3 + 1.7 * rng.random()
            if rng.random() < 0.04: mult *= 10.0
            d = s * min(avail, base * mult)
            h[i] += d; h[j] -= d
            cn = _c5(h)
            if cn < c - 1e-15:
                c = cn
                if c < best_c - 1e-15: best_c = c; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy(); best_c = _c5(best)
    for hh in (1.0 - h0, h0[::-1], 0.5 + 0.5*(h0 - h0[::-1]), 0.5 + 0.5*(h0 + h0[::-1] - 1.0)):
        c = _c5(_proj(hh, n0*0.5))
        if c < best_c: best = _proj(hh, n0*0.5); best_c = c
    end1 = t0 + min(40.0, 0.04 * budget_s)
    if end1 > time.time():
        b, bh, bc = _fista(best, 80, end1 - time.time(), 8.0)
        if bc < best_c: best = bh; best_c = bc
    end2 = t0 + 180.0
    if budget_s > 250 and end2 > time.time():
        for f, al in ((2, [30, 200, 1500, 8000]), (4, [20, 100, 600, 4000])):
            m = n0 // f
            if m < 32: continue
            slot = min(60.0, max(10.0, end2 - time.time() - 5.0) / 2.0)
            if slot < 5.0: break
            hm = _rg(best, m)
            b2, bh2, bc2 = _fista(hm, 30, slot*0.15, 8.0)
            for a in al:
                rem = end2 - time.time() - 5.0
                if rem < 3.0: break
                s2 = min(rem * 0.35, 25.0)
                b2, bh2, bc2 = _fista(b2, a, s2, max(8.0, a * 0.05))
            if bc2 < best_c:
                best = _rg(bh2, n0); best_c = _c5(best)
            hu = _rg(best, n0)
            cu = _c5(hu)
            if cu < best_c: best = hu; best_c = cu
    end3 = t0 + budget_s - 380.0
    if end3 > time.time() + 20.0:
        hmain = best if len(best) == n0 else _rg(best, n0)
        alphas = [15, 60, 200, 800, 3000, 12000, 50000, 200000]
        total_rem = end3 - time.time()
        for i, a in enumerate(alphas):
            rem = end3 - time.time() - 5.0
            if rem < 5.0: break
            frac = 1.0 - (i + 1) / len(alphas)
            slot = min(rem, total_rem * (0.15 + 0.85 * frac) / (len(alphas) - i))
            slot = max(5.0, slot)
            Linit = max(8.0, min(a * 0.02, 500.0))
            b, bh, bc = _fista(hmain, a, slot, Linit)
            cb = _c5(bh)
            if cb < best_c: best = bh; best_c = cb
            cf = _c5(b)
            if cf < best_c: best = b; best_c = cf
            hmain = b if cf <= cb else bh
    end4 = t0 + budget_s - 2.0
    if end4 > time.time():
        best, best_c = _lsearch(best, rng, end4, 5e-5)
    if len(best) != n0:
        best = _rg(best, n0); best_c = _c5(best)
    for hh in (1.0 - best, best[::-1], 0.5 + 0.5*(best - best[::-1]), 0.5 + 0.5*(best + best[::-1] - 1.0)):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c: best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
