            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def _lp_descent(h, n_steps, eps, m_top, deadline, m_top2=None):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stalls = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        mt = m_top
        if m_top2 is not None and it > int(n_steps * 0.6):
            mt = m_top2
        idx_top = np.argpartition(cor, -mt)[-mt:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if mt > 1 else None
        b_ub = np.zeros(mt - 1) if mt > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stalls += 1
            if stalls > 30:
                rng_p = np.random.RandomState(1000 + it)
                d2 = rng_p.randn(n)
                d2 -= d2.mean()
                d2 *= eps * 0.4 / max(np.max(np.abs(d2)), 1e-15)
                h_new = _repair(c + d2, n)
                c5_new = _eval(h_new)
                if c5_new < c5_cur:
                    c, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, c.copy()
                    stalls = 0
                else:
                    stalls = 0
                continue
            continue
        d = res.x
        improved = False
        for ts in [1.0, 0.6, 0.35, 0.2, 0.1]:
            h_new = _repair(c + ts * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                stalls = 0
                break
        if not improved:
            stalls += 1
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k, sh, wk = idx[ii], (n - 1) - idx[ii], w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta = 0.8; step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-11:
                break
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_topk(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _sweep_polish(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    sss = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    per = max(1, niter // len(sss))
    for ss in sss:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 5)) % nc
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _targeted_lp(h, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    for it in range(8000):
        if time.time() > deadline:
            break
        cor = _correl(c)
        k_worst = int(np.argmax(cor))
        g = _sg(c, k_worst, n)
        c_obj = -g
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-0.0025, 0.0025)] * n
        try:
            res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bnds,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        improved = False
        for ts in [1.0, 0.5, 0.25]:
            h_new = _repair(c + ts * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 140)
    for a, tk, mi in [(2000, 50, 350), (8000, 30, 300), (25000, 15, 250)]:
        if time.time() > dl1:
            break
        hv = _slsqp(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 370)
    hv, cv = _lp_descent(best_h.copy(), 35000, 0.003, 12, dl2, m_top2=6)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 500)
    for a, tk, st, ns in [(20000, 20, 0.003, 2000), (80000, 10, 0.0015, 2000)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl4 = min(deadline, t0 + 620)
    hv, cv = _targeted_lp(best_h.copy(), dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 770)
    hv, cv = _lp_descent(best_h.copy(), 25000, 0.0015, 8, dl5, m_top2=4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl6 = min(deadline, t0 + 910)
    blocks = [1, 2, 3, 4, 6, 8, 12]
    hv, cv = _block_anneal(best_h.copy(), 45000, rng, dl6, 0.004, 1e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        if time.time() < dl6:
            hv2, cv2 = _lp_descent(best_h.copy(), 6000, 0.001, 5, dl6, m_top2=3)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    dl7 = deadline
    rem = max(0, dl7 - time.time())
    npol = min(int(rem / 0.0005), 70000)
    best_h, best_c5 = _sweep_polish(best_h, npol, rng, dl7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
