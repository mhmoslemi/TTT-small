def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _subgrad_topk(h, k_top):
    n = len(h)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    f = _eval(h)
    for _pass in range(max_passes):
        if time.time() > t_end:
            break
        for off in offsets:

        f_new = _eval(h)
        if f_new > f + 1e-12:
            h = _repair(h, n)
            f_new = _eval(h)
        f = f_new
    return h

def _subgrad_mom(h, k_top, n_steps, alpha0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.7
    alpha = alpha0
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        dx2 = 2.0 / n
        for k in idx:

            if sh >= 0:
                g[:n-sh] += dx2 * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= dx2 * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += dx2 * (1.0 - h[:n-sp])
                g[:n-sp] -= dx2 * h[sp:]

        h_new = _repair(h - alpha * v, n)

def _slsqp_lse(h, alpha, maxiter, t_end):
    n = len(h)
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -300, 300)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        dx2 = 2.0 / n

            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    bnds = [(0.0, 1.0)] * n

                       options={'maxiter': maxiter, 'ftol': 1e-16})
