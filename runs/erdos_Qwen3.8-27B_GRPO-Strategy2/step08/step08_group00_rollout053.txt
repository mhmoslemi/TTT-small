            g[L:] += wk * (1.0 - h[:n-L])
            g[:n-L] -= wk * h[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n-M]
    return g

def _slsqp_smooth(h, alpha, top_k, maxiter, ftol):

    dx = 2.0 / n
    
    def obj(x):
        c = _corr(x) * dx
        return _smooth_vals(c, alpha, top_k)[0]
    
    def grad(x):
        return _grad_smooth(x, alpha, top_k)
    
    def sum_fn(x):
        return x.sum() - n * 0.5
    
    def sum_jac(x):
        return np.ones(n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_proj_grad(h, alpha, top_k, steps, step0, momentum):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    mu = momentum
    for it in range(steps):
        g = _grad_smooth(h, alpha, top_k)
        # Nesterov lookahead
        y = h + mu * v
        g_y = _grad_smooth(y, alpha, top_k)
        v = mu * v - step * g_y
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            step = min(step * 1.05, step0 * 10.0)
        else:
            step *= 0.5
            v = 0.6 * v
    return _repair(h, n)

def _adam_proj_grad(h, alpha, top_k, steps, lr, b1, b2, eps):

    v = np.zeros(n)
