def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _grad_lse(c, h, n, alpha, top_k):
    nc = len(c)
    if top_k > 0 and top_k < nc:
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
    else:
        idx = np.arange(nc)
        cs = c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-12:
            continue
        g += w[ii] * _grad_lag(int(idx[ii]), h, n)
    return g

def _adam_opt(h, alpha_steps, total_steps, lr0, beta1, beta2, eps, top_k, deadline):
    n = len(h)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 1
    cur_c = _eval(h)
    for step in range(total_steps):
        if time.time() > deadline:
            break
        c = _corr(h)
        alpha = alpha_steps[min(step // max(1, total_steps // len(alpha_steps)), len(alpha_steps) - 1)]
        g = _grad_lse(c, h, n, alpha, top_k)
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf = beta2 * v_buf + (1 - beta2) * g * g
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        lr = lr0 * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        h_new = _proj(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
        t += 1
    return h, cur_c

def _binary_swap_search(h, n, n_iter, rng, deadline):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(b.sum())
    if s > target:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])[:s - target]
        b[ones[order]] = 0.0
    elif s < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])[:target - s]
        b[zeros[order]] = 1.0
    c5_cur = _eval(b)
    b_best = b.copy()
    c5_best = c5_cur
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
        c5_new = _eval(b)
        if c5_new < c5_cur:
            c5_cur = c5_new
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if c5_new < c5_best:
                c5_best = c5_new
                b_best = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    return b_best, c5_best

def _slsqp_opt(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h

def _fine_polish(h, n_iter, rng, base_step, deadline):
    n = len(h)
    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    c = _corr(h)
    k_w = int(np.argmax(c))
    sh = (n - 1) - k_w
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if sh > 0 and sh < n and rng.rand() < 0.6:
            i = rng.randint(n - sh)
            j = i + sh
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 8)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5_new = _eval(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                if c5_new < best_c:
                    best_c = c5_new
                    best_h = h.copy()
                c = _corr(h)
                k_w = int(np.argmax(c))
                sh = (n - 1) - k_w
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    n = len(initial_h_values)
    rng = np.random.RandomState(seed)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP warm-start (40s)
    for alpha, tk, mit in [(5000, 0, 400), (50000, 4, 300), (200000, 2, 200)]:
        if _el() > 40:
            break
        h_new = _slsqp_opt(best_h, alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: Adam optimizer with adaptive rates (180s)
    alpha_seq = [500000, 200000, 100000, 50000, 20000, 10000]
    for trial in range(3):
        deadline = t0 + 40 + trial * 55
        if _el() > deadline - 5:
            break
        h_a, c_a = _adam_opt(best_h.copy(), alpha_seq, 40000, 0.003,
                             0.8, 0.999, 1e-8, 5, deadline)
        h_a = _repair(h_a, n)
        c_a = _eval(h_a)
        if c_a < best_c5:
            best_c5 = c_a
            best_h = h_a.copy()

    # Phase 3: Binary swap search (200s)
    deadline_bin = t0 + 250
    b_best, c_bin = _binary_swap_search(best_h, n, 5000000, rng, deadline_bin)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = b_best.copy()
    # Blend binary back to continuous: start from 0.45-0.55 neighborhood
    h_blend = 0.5 + 0.48 * (2.0 * best_h - 1.0)
    h_blend = _repair(h_blend, n)
    c_bl = _eval(h_blend)
    if c_bl < best_c5:
        best_c5 = c_bl
        best_h = h_blend.copy()

    # Phase 4: Adam fine-tune (150s)
    deadline_adam2 = t0 + 420
    alpha_seq2 = [1000000, 500000, 200000, 100000, 50000, 20000]
    for trial in range(2):
        if _el() > deadline_adam2 - 5:
            break
        h_a, c_a = _adam_opt(best_h.copy(), alpha_seq2, 60000, 0.001,
                             0.85, 0.999, 1e-10, 3, deadline_adam2)
        h_a = _repair(h_a, n)
        c_a = _eval(h_a)
        if c_a < best_c5:
            best_c5 = c_a
            best_h = h_a.copy()

    # Phase 5: SLSQP final (30s)
    if _el() < budget_s - 120:
        for alpha, tk, mit in [(5000000, 1, 150), (15000000, 1, 100)]:
            if _el() > budget_s - 120:
                break
            h_new = _slsqp_opt(best_h, alpha, mit, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 6: Multiple restart perturbations (80s)
    deadline_restart = t0 + budget_s - 60
    for trial in range(10):
        if _el() > deadline_restart:
            break
        perturb = best_h.copy() + rng.randn(n) * 0.005
        perturb = _repair(perturb, n)
        h_a, c_a = _adam_opt(perturb, [500000, 200000, 100000, 50000],
                             8000, 0.002, 0.8, 0.999, 1e-9, 3, deadline_restart)
        h_a = _repair(h_a, n)
        c_a = _eval(h_a)
        if c_a < best_c5:
            best_c5 = c_a
            best_h = h_a.copy()

    # Phase 7: Ultra-fine lag-focused polish (remaining)
    deadline_end = t0 + budget_s - 2
    c = _corr(best_h)
    top_lags = np.argsort(c)[-4:]
    for lag in top_lags:
        if _el() > deadline_end:
            break
        h_p, c_p = _fine_polish(best_h.copy(), 500000, rng, 5e-7, deadline_end)
        # Override: focus on this specific lag
        if c_p < best_c5:
            best_c5 = c_p
            best_h = h_p.copy()
    # General polish
    if _el() < deadline_end:
        best_h, best_c5 = _fine_polish(best_h, 2000000, rng, 3e-7, deadline_end)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
