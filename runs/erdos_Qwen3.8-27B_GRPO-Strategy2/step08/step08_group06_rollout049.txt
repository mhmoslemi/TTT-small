def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lags(h, lag_indices, dx):
    n = len(h)

        else:

            g[sp:] += wk * (1.0 - h[:n-sp])

    return g

class _LseCtx:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
        self.dx = 2.0 / n
    def obj(self, x):
        c = _corr(x)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -300, 300)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = _corr(x)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
