def _slsqp_run(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_run(h, alpha0, grow, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 500 == 0:
            alpha = min(alpha * grow, 5e7)
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.01, step0 * 5.0)
        else:
            step *= 0.5
            for _ in range(3):
                if step < 1e-10:
                    break
                hn = _proj(h - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                    break
                step *= 0.5
            else:
                break
    return _repair(h, n)

    offsets = [o for o in offsets if 0 < o < n]

            if time.time() > deadline:
                break

                if j == i:
                    continue
                for d in [1e-4, -1e-4, 5e-5, -5e-5, 1e-5, -1e-5]:

                        else:
                            h[i], h[j] = ni - d, nj + d
                        if time.time() > deadline:
                            break
    return bh, bc

def _anneal(h, rng, deadline, T0=0.01, T1=1e-12, niter=200000):
    """Simulated annealing with mixed block sizes."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    blocks = [1, 2, 3, 4, 5, 8, 12, 20, 35, 60, 100, 200]
    weights = [10, 10, 10, 8, 8, 6, 5, 5, 4, 3, 2, 1]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = rng.choice(blocks, p=np.array(weights, dtype=float)/sum(weights))
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-10)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-14)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

import numpy as np

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

        else:
