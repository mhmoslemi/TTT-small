from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _topk_gradient(h, k_top, n):
    c = _corr_vec(h)
    top_k = np.argpartition(c, -min(k_top, len(c)))[-min(k_top, len(c)):]
    g = np.zeros(n)
    for ki in top_k:
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n - sh] -= h[sh:]
            g[sh:] += (1.0 - h[:n - sh])
        else:
            sp = -sh
            g[:n - sp] += (1.0 - h[sp:])
            g[sp:] -= h[:n - sp]
    return g / k_top

class _LseCtx:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
        self.target = n / 2.0
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            sh = (self.n - 1) - k
            wk = w[k]
            if sh >= 0:
                g[:self.n - sh] -= wk * x[sh:]
                g[sh:] += wk * (1.0 - x[:self.n - sh])
            else:
                sp = -sh
                g[:self.n - sp] += wk * (1.0 - x[sp:])
                g[sp:] -= wk * x[:self.n - sp]
        return g
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _LseCtx(alpha, n)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]

        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            f_new = _eval(h)
            if f_new < f:
                f = f_new
                if f < best_f:
                    best_f, best_h = f, h.copy()
                step = min(step * 1.03, 5e-3)
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.6
                if step < 1e-9:
                    step = 5e-5
    return best_h, best_f

def _topk_descent(h, k, step0, n_iter, t_end):
    n = len(h)
    f = _eval(h)
    step = step0
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        g = _topk_gradient(h, k, n)
        h_new = _repair(h - step * g, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.04, 0.2)
        else:
            step *= 0.5
            if step < 1e-11:
                step = 0.005
            h_new = _repair(h - step * g, n)
            f_new = _eval(h_new)
            if f_new < f:
                h, f = h_new, f_new
    return _repair(h, n), f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_f = _eval(best_h)

    def _update(h_new):
        return h_new, _eval(h_new)

    # Phase 1: SLSQP moderate alpha (broad) - 15%
    t1 = t0 + budget_s * 0.15
    h_try = _slsqp_opt(best_h.copy(), 2000, 800, 1e-14)
    if _eval(h_try) < best_f:
        best_h, best_f = h_try, _eval(h_try)
    if time.time() < t1 - 5:
        h_try = _slsqp_opt(best_h.copy(), 20000, 800, 1e-14)
        if _eval(h_try) < best_f:
            best_h, best_f = h_try, _eval(h_try)

    # Phase 2: Momentum top-k with growing k - 20%
    t2 = t0 + budget_s * 0.35
    h_try = _mom_topk(best_h.copy(), 5, 50, 3000, 0.003, t2 - 5)
    if h_try[1] < best_f:
        best_h, best_f = h_try

    # Phase 3: SLSQP high alpha - 15%
    t3 = t0 + budget_s * 0.50
    h_try = _slsqp_opt(best_h.copy(), 200000, 1000, 1e-15)
    if _eval(h_try) < best_f:
        best_h, best_f = h_try, _eval(h_try)
    if time.time() < t3 - 10:
        h_try = _slsqp_opt(best_h.copy(), 2000000, 600, 1e-16)
        if _eval(h_try) < best_f:
            best_h, best_f = h_try, _eval(h_try)

    # Phase 4: Momentum top-k large k - 15%
    t4 = t0 + budget_s * 0.65
    h_try = _mom_topk(best_h.copy(), 50, 300, 4000, 0.002, t4 - 5)
    if h_try[1] < best_f:
        best_h, best_f = h_try

    # Phase 5: Perturb and reoptimize (escape local optima) - 20%
    t5 = t0 + budget_s * 0.85
    for sigma in [0.05, 0.02, 0.01]:
        if time.time() > t5 - 10:
            break
        h_pert = best_h + sigma * rng.randn(n)
        h_pert = _repair(h_pert, n)
        h_opt = _slsqp_opt(h_pert, 50000, 500, 1e-14)
        h_opt, f_opt = _mom_topk(h_opt, 20, 150, 3000, 0.003, t5 - 10)
        if f_opt < best_f:
            best_h, best_f = h_opt, f_opt

    # Phase 6: Top-k descent polish - 8%
    t6 = t0 + budget_s * 0.93
    if time.time() < t6 - 5:
        h_try, f_try = _topk_descent(best_h.copy(), 200, 0.005, 5000, t6 - 5)
        if f_try < best_f:
            best_h, best_f = h_try, f_try

    # Phase 7: Swap local search
    if time.time() < t_global_end - 20:
        ntr = max(1000, int((t_global_end - 20 - time.time()) / 0.006))
        h_try, f_try = _swap_local(best_h, ntr, rng, t_global_end - 10)
        if f_try < best_f:
            best_h, best_f = h_try, f_try

    # Phase 8: Fine local search
    if time.time() < t_global_end - 2:
        ntr = max(500, int((t_global_end - 2 - time.time()) / 0.004))
        h_try, f_try = _fine_local(best_h, ntr, rng, t_global_end - 0.5)
        if f_try < best_f:
            best_h, best_f = h_try, f_try

    best_h = _repair(best_h, n)
    best_f = _eval(best_h)
    return (best_h, best_f, n)
