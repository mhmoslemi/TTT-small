```python
def _grad_lag(h, lag, dx):
    """Gradient of lag-`lag` correlation w.r.t. h, times dx."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # c = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
        g[:n-lag] += (1.0 - h[lag:])      # h[i] terms
        g[lag:] -= h[:-lag]                # h[i+lag] terms
    else:
        j = -lag
        # c = sum_{i=0}^{n-1-j} h[i+j](1-h[i])
        g[j:] += (1.0 - h[:-j])            # h[i+j] terms
        g[:n-j] -= h[j:]                   # h[i] terms
    return g * dx
```

```python
        g[lag:] -= h[:-lag]
    else:

        g[j:] += 1.0 - h[:-j]
        g[:n - j] -= h[j:]
```

```python
    if s < 1e-15:

    lo, hi = -1.5, 1.5
    for _ in range(40):
```

```python
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time, math

_SUM_TARGET = 0.0

def _sum_fn(x):
    return x.sum() - _SUM_TARGET

def _sum_jac(x):
    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if abs(s - t) < 1e-12:
        return h
    if s < 1e-15:
        return np.full(n, 0.5)
    lo, hi = -2.0, 2.0
    for _ in range(45):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(x, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(45):
        m = (lo + hi) * 0.5
        if np.clip(x - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(x - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_one(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:-lag]
    else:
        j = -lag
        g[j:] += 1.0 - h[:-j]
        g[:n - j] -= h[j:]
    return g * dx

def _sobj(h, a, tk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -tk)[-tk:] if tk < len(c) else np.arange(len(c))
    s = np.clip(a * (c[idx] - m), -300, 300)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / a

def _sgrad(h, a, tk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -tk)[-tk:].copy() if tk < len(c) else np.arange(len(c))
    s = np.clip(a * (c[idx] - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-13:
            g += w[i] * _grad_one(h, (n - 1) - idx[i], 2.0 / n)
    return g

def _slsqp(h, a, tk, mi, dl):

    global _SUM_TARGET
    _SUM_TARGET = n * 0.5

def _bcd(h, n, dl, bs, ka, s0):

    best = h.copy()

    step = s0
    bad = 0
    while time.time() < dl:

        w = np.exp(s - s.max())

def _anneal(h, n, dl, rng, T0, T1, nit, bszs):
    best = h.copy()
    bc = _eval(h)
    cv = bc
    for it in range(nit):

        T = T0 * (T1 / T0) ** (it / max(nit, 1))
        b = bszs[rng.randint(len(bszs))]
        i0 = rng.randint(0, max(1, n - b))
        seg = h[i0:i0 + b].copy()
        d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(b)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)

        cn = _eval(hn)
        if cn < cv or rng.random() < np.exp(-(cn - cv) / max(T, 1e-12)):
            h = hn; cv = cn
            if cn < bc:

def _pair(h, n, dl, rng, s0):

    cv = bc

            h[j] = nj

            else:
                h[i] = ni - d
                h[j] = nj + d
```
