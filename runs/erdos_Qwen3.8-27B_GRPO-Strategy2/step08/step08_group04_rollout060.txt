            for idx in order:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid

            hi = mid

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, pos, dx):
    """Gradient of the 'full' correlation element at position pos."""
    n = len(h)
    g = np.zeros(n)
    lag = pos - (n - 1)  # actual shift
    if lag >= 0:
        # c = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])*dx
        for i in range(n - lag):
            g[i + lag] += (1.0 - h[i])
            g[i] -= h[i + lag]
    else:
        # lag < 0, shift = -lag
        s = -lag
        for i in range(s, n):
            g[i] += (1.0 - h[i - s])
            g[i - s] -= h[i]
    return g * dx

    n = len(h)
    c = _correl(h)
    best_c = c.max()
    top_idx = np.argpartition(c, -(topk-1))[-topk:]
    v = np.zeros(n)
    beta = 0.85
    step = step0
    dx = 2.0 / n

            if cn.max() < best_c:

                best_c = cn.max()

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -(topk-1))[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -(topk-1))[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _nesterov_smax(h, alpha, grow, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.83
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 500 == 0:
            alpha = min(alpha * grow, 5e7)
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 8.0)
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(hp - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _pair_swap_polish(h, niter, rng, deadline):
    """Polish using pair swaps that preserve sum exactly."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        # Choose a partner at a nearby offset
        j = (i + 1 + rng.randint(0, 15)) % n
        if i == j:
            continue
        # Try a small transfer
        max_d = min(h[i], 1.0 - h[j]) * 0.5
        min_d = max(-h[i], h[j] - 1.0) * 0.5
        d = rng.uniform(min_d, max_d) if max_d > min_d else 0.0
        if abs(d) < 1e-15:
            continue
        h[i] += d
        h[j] -= d
        c5n = _eval(h)
        if c5n < c5c - 1e-18:
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        scale = 0.003 * max(T, 1e-9)**0.5
        d = scale * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]

                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:

    k = pos - (n - 1)  # actual lag
    g = np.zeros(n)
    if k >= 0:

    return g * dx

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, pos, dx):
    n = len(h)
    k = pos - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        s = -k
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -(topk - 1))[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -(topk - 1))[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _nesterov_smax(h, alpha, grow, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.84
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 400 == 0:
            alpha = min(alpha * grow, 8e7)
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-18:
            h = hn
            f = fn
            step = min(step * 1.02, step0 * 10.0)
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(hp - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _nesterov_exact(h, topk, steps, step0, deadline):
    n = len(h)
    c = _correl(h)
    best = c.max()
    top_idx = np.argpartition(c, -(topk - 1))[-topk:]
    v = np.zeros(n)
    beta = 0.82
    step = step0
    dx = 2.0 / n
    for it in range(steps):
        if time.time() > deadline:
            break
        g = np.zeros(n)
        for pos in top_idx:
            g += _grad_lag(h, pos, dx)
        g /= topk
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        cn = _correl(hn)
        if cn.max() < best - 1e-16:
            h = hn
            best = cn.max()
            c = cn
            top_idx = np.argpartition(c, -(topk - 1))[-topk:]
            step = min(step * 1.015, step0 * 6.0)
        else:
            step *= 0.7
            if step < 1e-11:
                break
            hn = _proj(hp - step * v, n * 0.5)
            cn = _correl(hn)
            if cn.max() < best:
                h = hn
                best = cn.max()
                c = cn
                top_idx = np.argpartition(c, -(topk - 1))[-topk:]
    return _repair(h, n)

def _lp_refine(h, deadline, max_rounds=25):
    n = len(h)
    best_c5 = _eval(h)
    c = _correl(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        m = c.max()
        thresh = m - 0.006
        active = np.where(c >= thresh)[0]
        hvar = cp.Variable(n)
        tvar = cp.Variable()
        cons = [hvar >= 0, hvar <= 1, cp.sum(hvar) == n * 0.5]
        for pos in active:
            g = _grad_lag(h, pos, 1.0)
            cons.append(g @ hvar - tvar <= g @ h - c[pos])
        prob = cp.Problem(cp.Minimize(tvar), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.SCIP)
            except Exception:
                break
        if hvar.value is not None:
            h_new = _repair(np.array(hvar.value).flatten(), n)
            c5n = _eval(h_new)
            if c5n < best_c5 - 1e-11:
                h = h_new
                best_c5 = c5n
                c = _correl(h)
            else:
                break
    return h, best_c5

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _pair_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        if i == j:
            continue
        max_d = min(h[i], 1.0 - h[j]) * 0.6
        min_d = max(-h[i], h[j] - 1.0) * 0.6
        if max_d <= min_d:
            continue
        d = rng.uniform(min_d, max_d)
        h[i] += d
        h[j] -= d
        c5n = _eval(h)
        if c5n < c5c - 1e-18:
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc
