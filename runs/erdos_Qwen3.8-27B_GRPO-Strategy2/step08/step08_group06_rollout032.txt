import numpy as np
import time

def _proj(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - out)
        else:
            order = np.argsort(out)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - out[idx])
                out[idx] += a
                diff -= a
            else:
                a = min(-diff, out[idx])
                out[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return out

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _subgrad_topk(h, top_k, n):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c_raw, -top_k)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            jv = np.arange(lo, hi + 1)
            g[lo:hi+1] += 1.0 - 2.0 * h[k - jv]
    return (g * dx) / top_k, c_raw * dx

    best_h, best_f = h.copy(), f
    momentum = np.zeros(n)
    mu = 0.75
    lr = lr0
    step = 0
    stall = 0
    while time.time() < t_end:

        k_cur = max(5, int(k_start + (k_end - k_start) * prog))
        g, _ = _subgrad_topk(h, k_cur, n)

            momentum[:] = 0

import numpy as np
import time

def _proj(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - out) if diff > 0 else np.argmin(out)
        out[idx] = np.clip(out[idx] + diff, 0.0, 1.0)
    return out

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _subgrad(h, top_k, n):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    top = np.argpartition(c_raw, -top_k)[-top_k:]
    g = np.zeros(n)
    for kk in top:
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            jv = np.arange(lo, hi + 1)
            g[lo:hi + 1] += 1.0 - 2.0 * h[kk - jv]
    return (g * dx) / top_k

def _sgd(h, n, t_end, k0, k1, lr0):
    f = _eval(h)
    best_h, best_f = h.copy(), f
    mom = np.zeros(n)
    mu = 0.72
    lr = lr0
    st = 0
    stall = 0
    while time.time() < t_end - 2:

        mom = mu * mom + (1.0 - mu) * g
        h_new = _proj(h - lr * mom, n)
        f_new = _eval(h_new)
        if f_new < f:

            stall = 0
            lr = min(lr * 1.025, lr0 * 40.0)

            mom *= 0.4

                stall = 0
        st += 1
    return best_h, best_f

    best_h, best_f = h.copy(), f

        n_batch = max(200, int(remain * 3000))
