import numpy as np

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    """Gradient of correlation at lag index k (0..2n-2) w.r.t. h."""
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

            break
        # Shrink eps over time
        frac = it / max(max_iter, 1)
        eps_cur = eps * max(0.1, 1.0 - frac)
        
        cor = _correl(c)
        idx_top = np.argpartition(cor, -topk)[-topk:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]

        g0 = _grad_k(c, idx_top[0], n)

        if topk > 1:

            b_ub = np.zeros(topk - 1)
        else:
            A_ub, b_ub = None, None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps_cur, eps_cur)] * n

        except Exception:
            break
        
        if not res.success or res.x is None:

        d = res.x
        if np.max(np.abs(d)) < 1e-14:

        for step in [1.0, 0.75, 0.5, 0.25, 0.125, 0.0625]:

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _proj(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _lp_max_step(h, eps, topk):
    """Solve LP to minimize max of top-k correlations after step d."""
    n = len(h)
    cor = _correl(h)
    idx = np.argpartition(cor, -topk)[-topk:]
    idx = idx[np.argsort(-cor[idx])]
    ct = cor[idx]
    
    nv = n + 1  # n for d, 1 for t
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((topk, nv))
    b_ub = np.zeros(topk)
    for i in range(topk):
        A_ub[i, :n] = _grad_k(h, idx[i], n)
        A_ub[i, n] = -1.0
        b_ub[i] = -ct[i]
    
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bnds = [(-eps, eps)] * n + [(ct[0] - 2*eps, 2.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bnds, method='highs', options={'presolve': True})
    return res

def _lp_descent(h, n_iter, eps0, topk, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter, 1)
        eps = eps0 * max(0.02, 1.0 - 0.95 * frac)
        
        try:
            res = _lp_max_step(c, eps, topk)
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        d = res.x[:n]
        if np.max(np.abs(d)) < 1e-14:
            break
        
        improved = False
        for step in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                stall = 0
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                rs = np.random.RandomState(it * 31 + 17)
                d2 = rs.randn(n)
                d2 -= d2.mean()
                scale = eps * 0.03 / (np.max(np.abs(d2)) + 1e-15)
                h_new = _repair(c + d2 * scale, n)
                c5_new = _eval(h_new)
                if c5_new < c5_cur - 1e-17:
                    c, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, c.copy()
                    stall = 0
                elif stall > 400:
                    break
    
    return best_h, best_c5

def _pair_search(h, max_iter, rng, deadline, d0=0.003):
    n = len(h)
    c5_cur = _eval(h)
    best_h, best_c5 = h.copy(), c5_cur
    d = d0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dd = d * (2.0 * rng.random() - 1.0)
        ni = h[i] + dd
        nj = h[j] - dd
        if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
            continue
        h[i] = min(max(ni, 0.0), 1.0)
        h[j] = min(max(nj, 0.0), 1.0)
        c5_new = _eval(h)
        if c5_new < c5_cur - 1e-18:
            c5_cur = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h.copy()
        else:
            h[i] -= dd  # revert (approximately)
            h[j] += dd
            stall += 1
        if stall > 500:
            d *= 0.65
            stall = 0
            if d < 1e-9:
                break
    return best_h, max(best_c5, _eval(best_h))

def _smooth_grad_desc(h, n_iter, deadline, alpha=20000, topk=10, step0=0.005):
    n = len(h)
    target = n * 0.5
    c5_cur = _eval(h)
    best_h, best_c5 = h.copy(), c5_cur
    v = np.zeros(n)
    beta = 0.8
    step = step0

            break

        idx = np.argpartition(cor, -topk)[-topk:]
        ct = cor[idx]
        amax = ct.max()
        s = alpha * (ct - amax)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()

        for i in range(topk):
            if w[i] < 1e-12:
                continue
            g += w[i] * _grad_k(h, idx[i], n)
        
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, target)
        c5_new = _eval(h_new)
        if c5_new < c5_cur - 1e-16:
            h, c5_cur = h_new, c5_new
            step = min(step * 1.05, step0 * 10)
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h_new = _proj(h - step * v, target)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                h, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                step *= 0.8  # still shrink even if no improvement
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        df = t - h.sum()
        if abs(df) > 1e-12:
            orr = np.argsort(1.0 - h) if df > 0 else np.argsort(h)
            for ix in orr:
                if df > 0:
                    a = min(df, 1.0 - h[ix]); h[ix] += a; df -= a
                else:
                    a = min(-df, h[ix]); h[ix] -= a; df += a
                if abs(df) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _proj(v, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _lp_solve(h, eps, topk):
    n = len(h)
    co = _correl(h)
    ix = np.argpartition(co, -topk)[-topk:]
    ix = ix[np.argsort(-co[ix])]
    ct = co[ix]
    nv = n + 1

    for i in range(topk):
        A_ub[i, :n] = _grad_k(h, ix[i], n)
        A_ub[i, n] = -1.0
        b_ub[i] = -ct[i]
    A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0
    bnds = [(-eps, eps)] * n + [(0.0, 10.0)]

                   bounds=bnds, method='highs', options={'presolve': True})

    n = len(h)
    bh, bc = h.copy(), _eval(h)

    d = d0; st = 0

def _sgrad(h, nit, dl, alpha, topk, s0):

import numpy as np
from scipy.optimize import linprog
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        df = t - h.sum()
        if abs(df) > 1e-12:
            orr = np.argsort(1.0 - h) if df > 0 else np.argsort(h)
            for ix in orr:
                if df > 0:
                    a = min(df, 1.0 - h[ix]); h[ix] += a; df -= a
                else:
                    a = min(-df, h[ix]); h[ix] -= a; df += a
                if abs(df) < 1e-12:
                    break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g


def _proj(v, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def _lp_solve(h, eps, topk):
    n = len(h)
    co = _correl(h)
    ix = np.argpartition(co, -topk)[-topk:]
    ix = ix[np.argsort(-co[ix])]
    ct = co[ix]
    nv = n + 1
    c_obj = np.zeros(nv); c_obj[n] = 1.0
    A_ub = np.zeros((topk, nv)); b_ub = np.zeros(topk)
    for i in range(topk):
        A_ub[i, :n] = _grad_k(h, ix[i], n)
        A_ub[i, n] = -1.0
        b_ub[i] = -ct[i]
    A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n + [(0.0, 10.0)]
    return linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                   bounds=bnds, method='highs', options={'presolve': True})


def _lp_dsc(h, nit, eps0, topk, dl):
    n = len(h)
    bh, bc = h.copy(), _eval(h)
    c = h.copy(); c5 = bc; st = 0
    for it in range(nit):
        if time.time() > dl:
            break
        e = eps0 * max(0.05, 1.0 - 0.9 * it / max(nit, 1))
        try:
            r = _lp_solve(c, e, topk)
        except Exception:
            break
        if not r.success or r.x is None:
            break
        d = r.x[:n]
        if np.max(np.abs(d)) < 1e-14:
            break
        ok = False
        for s in (1.0, 0.7, 0.5, 0.3, 0.15, 0.05):
            hn = _repair(c + s * d, n)
            c5n = _eval(hn)
            if c5n < c5 - 1e-16:
                c, c5 = hn, c5n; ok = True; st = 0
                if c5 < bc:
                    bc, bh = c5, c.copy()
                break
        if not ok:
            st += 1
            if st > 80:
                rs = np.random.RandomState((it * 31 + 7) % 100000)
                d2 = rs.randn(n); d2 -= d2.mean()
                d2 *= e * 0.05 / (np.max(np.abs(d2)) + 1e-15)
                hn = _repair(c + d2, n)
                c5n = _eval(hn)
                if c5n < c5:
                    c, c5 = hn, c5n; st = 0
                    if c5 < bc:
                        bc, bh = c5, c.copy()
                elif st > 300:
                    break
    return bh, _eval(bh)


def _psrch(h, nit, rng, dl, d0):
    n = len(h)
    c5 = _eval(h); bh, bc = h.copy(), c5
    d = d0; st = 0
    for it in range(nit):
        if time.time() > dl:
            break

            continue
        dd = d * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + dd, h[j] - dd
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-18:
            c5 = c5n; st = 0
            if c5n < bc:
                bc, bh = c5n, h.copy()
        else:
            h[i], h[j] = ni - dd, nj + dd
            st += 1
        if st > 400:
            d *= 0.6; st = 0
            if d < 1e-10:
                break
    return bh, max(bc, _eval(bh))


def _sgrad(h, nit, dl, alpha, topk, s0):
    n = len(h); tgt = n * 0.5
    c5 = _eval(h); bh, bc = h.copy(), c5
    v = np.zeros(n); beta = 0.8; s = s0
    for it in range(nit):
        if time.time() > dl:
            break
        co = _correl(h)
        ix = np.argpartition(co, -topk)[-topk:]
        ct = co[ix]; am = ct.max()
        sv = alpha * (ct - am); sv -= sv.max()
        w = np.exp(sv); w /= w.sum()
        g = np.zeros(n)
        for i in range(topk):
            if w[i] >= 1e-12:
                g += w[i] * _grad_k(h, ix[i], n)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - s * v, tgt)
        c5n = _eval(hn)
        if c5n < c5 - 1e-16:
            h, c5 = hn, c5n
            s = min(s * 1.04, s0 * 8.0)
            if c5 < bc:
                bc, bh = c5, h.copy()

            s *= 0.5
            if s < 1e-12:
                break
            hn = _proj(h - s * v, tgt)
            c5n = _eval(hn)
            if c5n < c5:
                h, c5 = hn, c5n
