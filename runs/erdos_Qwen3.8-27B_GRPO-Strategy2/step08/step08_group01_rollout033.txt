```python
    best_c = cur_c
    best_h = h.copy()
    
    for it in range(n_iter):

        c = _corr_full(h) * dx
        # Get the top 5 lag indices
        top_idx = np.argpartition(c, -5)[-5:]
        
        # Pick a random top lag
        k = top_idx[rng.randint(5)]
        shift = k - (n - 1)
        
        # For this shift, find a position to perturb
        # c[k] = sum h[i+shift]*(1-h[i]) for valid i
        if shift >= 0:
            valid_i = np.arange(n - shift)
            # We want to decrease c[k]
            # Option A: decrease h[i+shift] for large (1-h[i])
            # Option B: increase h[i] for large h[i+shift]
            weights = h[valid_i + shift] * (1 - h[valid_i])
            # Pick a position proportional to its contribution
            total = weights.sum()
            if total < 1e-15:
                continue
            probs = weights / total
            pos = rng.choice(len(valid_i), p=probs)
            i = valid_i[pos]
            # Perturb: decrease h[i+shift] and increase h[i] (both reduce c[k])
            # But must maintain sum constraint: delta_h[i+shift] = -delta_h[i]
            # Decreasing h[i+shift] by d and increasing h[i] by d
            d = rng.uniform(0.001, 0.01)
            # Check bounds
            if h[i + shift] > d and h[i] < 1.0 - d:
                h[i + shift] -= d
                h[i] += d

                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i + shift] += d
                    h[i] -= d
        else:
            # Similar for negative shift
            pass
    
    return best_h, best_c
```

```python
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order[:n]:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _fft_corr(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N <<= 1
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(1.0 - h, N)
    r = np.fft.irfft(H * np.conj(V), N)
    return np.concatenate((r[N - n + 1:], r[:n]))
```

```python
    dx = 2.0 / n
```
```python
    c = _fft_corr(h) * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
```
