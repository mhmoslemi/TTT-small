def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, K):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_idx = np.argsort(c)[-K:]
    g = np.zeros(n)
    for k in top_idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    g *= (2.0 / n) / K
    return g

def _nesterov(h, K, steps, step0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.75
    step = step0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _topk_grad(h, K)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.015, step0 * 25.0)
        else:
            step *= 0.25
            if step < 1e-11:
                v = np.zeros(n)
                step = step0
            h = h_prev
            f = _eval(h)
    return h, f

def _coord_pair(h, n_tries, rng, t_end, scale):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d
            h[j] -= d
            c = _eval(h)
            if c < best_c:

            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def _coord_3pt(h, n_tries, rng, t_end, scale):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_tries):

            break

        d2 = scale * rng.randn()

class _LSECtx:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        dx2 = 2.0 / self.n
        for k in range(2 * self.n - 1):
            if w[k] < 1e-14:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * dx2
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n - sp])
                g[:self.n - sp] -= wk * x[sp:]
        return g
    def cfun(self, x):
        return x.sum() - self.n / 2.0
    def cjac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    ctx = _LSECtx(alpha, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.cfun, 'jac': ctx.cjac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < t_end:
            return _repair(res.x, n)
    except Exception:
        pass
    return _repair(h.copy(), n)
