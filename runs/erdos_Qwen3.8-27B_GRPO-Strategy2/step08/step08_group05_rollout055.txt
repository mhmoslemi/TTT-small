def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    m = k - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    jm = j - m
    m1 = (jm >= 0) & (jm < n)
    g[m1] += (1.0 - h[jm[m1]])
    jp = j + m
    m2 = (jp >= 0) & (jp < n)
    g[m2] -= h[jp[m2]]
    return g

def _lp_descent(h, max_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for it in range(max_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_grad_k(c, k, n) for k in idx_top])
        c_obj = grads[0]
        if m_top > 1:
            A_ub = grads[1:]
            b_ub = np.zeros(m_top - 1)
        else:
            A_ub, b_ub = None, None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs',
                          options={'presolve': True, 'time_limit': 3})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        if res.fun >= -1e-14:
            stall += 1
            if stall > 80:
                break
            rng_p = np.random.RandomState(it * 31 + 7)
            d = rng_p.randn(n); d -= d.mean()
            d *= eps * 0.05 / max(np.max(np.abs(d)), 1e-15)
            h_new = _repair(c + d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new; stall = 0
                if c5_cur < best_c5: best_c5, best_h = c5_cur, c.copy()
            continue
        d = res.x
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07]:
            h_new = _repair(c + frac * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-17:
                c, c5_cur = h_new, c5_new
                improved = True
                break
        if improved:
            stall = 0
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, c.copy()
        else:
            stall += 1
            if stall > 120:
                break
    return best_h, best_c5

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_k(h, idx[ii], n)
    return g

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta, step = 0.7, step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _smax(h_new, alpha, topk)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    def obj(x): return _smax(x, alpha, topk)
    def jac(x): return _smax_grad(x, alpha, topk)
    def cf(x): return x.sum() - n * 0.5
    def cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-25, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _struct_polish(h, deadline):
    n = len(h)
    bh = h.copy(); bc = _eval(h)
    nbs = [1, -1, 2, -2, 3, -3, 4, -4]
    for p in range(8):
        if time.time() > deadline:
            break
        step = 2e-5 / (p + 1)
        any_imp = False
        for i in range(n):
            if time.time() > deadline:
                break
            for nb in nbs:
                j = i + nb
                if j < 0 or j >= n:
                    continue
                for sgn in [1, -1]:
                    d = sgn * step
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < bc - 1e-18:
                            bc = c5n; bh = h.copy(); any_imp = True
                        else:
                            h[i], h[j] = ni - d, nj + d
        if not any_imp:
            break
    return bh, bc

def _random_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    scales = [8e-6, 4e-6, 2e-6, 8e-7, 3e-7, 1e-7]
    per = max(niter // len(scales), 20)
    for sc in scales:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
            if i == j:
                continue
            d = sc * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc
