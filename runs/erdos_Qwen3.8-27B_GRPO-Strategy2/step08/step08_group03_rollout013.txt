import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(8):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo = float(np.min(v)) - 1.0
        hi = float(np.max(v)) + 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            s = np.clip(v - mid, 0.0, 1.0).sum()
            if s > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        convR = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _fista(h0, alpha, max_time, L=10.0):
    if max_time <= 0:
        h = h0.copy()
        return h, h.copy(), _c5_direct(h)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _smooth_value(h, alpha)
    fh = fy
    best_h = h.copy()
    best_c = _c5_direct(h)
    stale = 0
    Lcur = L
    for it in range(500000):
        if time.time() - t0 > max_time:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y = h
            fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(14):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h
            h = xn
            fh = fx
            stale = 0
            Lcur = max(Lcur * 0.85, 1.0)
            if it % 8 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t = 1.0
            fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _active_grad(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 5.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += int(pos.size)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += int(pos.size)
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv
    wN *= inv
    return _grad_from_weights(h, wP, wN)

def _local_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    P, Nv, _ = _pn_arrays(h)
    g = _active_grad(h, P, Nv)
    order = np.argsort(g)
    q = max(8, n // 6)
    low = order[:q]
    high = order[-q:]
    recompute = 512
    it = 0
    step = 4e-5
    acc_count = 0
    while time.time() < deadline:
        if it >= recompute:
            c = _c5_direct(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            P, Nv, _ = _pn_arrays(h)
            g = _active_grad(h, P, Nv)
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            recompute = it + 512
            # Adaptive step size
            rate = acc_count / 512.0
            if rate < 0.1:
                step *= 0.7
            elif rate > 0.5:
                step *= 1.3
            step = max(1e-7, min(step, 2e-3))
            acc_count = 0
        if rng.random() < 0.75:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i != j:
            if g[i] <= g[j]:
                s = 1
            elif g[j] < g[i]:
                s = -1
            else:
                s = 1 if rng.random() < 0.5 else -1
            if s > 0:
                avail = min(1.0 - h[i], h[j])
            else:
                avail = min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                if rng.random() < 0.2:
                    bw = int(rng.randint(1, 5))
                    j2 = j + (1 if rng.random() < 0.5 else -1) * bw
                    if 0 <= j2 < n and j2 != i:
                        d = s * min(avail, step * (0.3 + 1.5 * rng.random()))
                        h[i] += d
                        h[j2] -= d
                        cn = _c5_direct(h)
                        if cn < c - 1e-15:
                            c = cn
                            acc_count += 1
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                        else:
                            h[i] -= d
                            h[j2] += d
                else:
                    mult = 0.5 + 1.5 * rng.random()
                    if rng.random() < 0.03:
                        mult *= 12.0
                    d = s * min(avail, step * mult)
                    h[i] += d
                    h[j] -= d
                    cn = _c5_direct(h)
                    if cn < c - 1e-15:
                        c = cn
                        acc_count += 1
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _block_swap_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    while time.time() < deadline:
        blen = rng.randint(2, max(3, n // 15))
        i1 = rng.randint(0, n - blen)
        i2 = rng.randint(0, n - blen)
        if abs(i1 - i2) < blen:
            continue
        b1 = h[i1:i1+blen].copy()
        b2 = h[i2:i2+blen].copy()
        h[i1:i1+blen] = b2
        h[i2:i2+blen] = b1
        h = _repair(h, n)
        cn = _c5_direct(h)
        if cn < c - 1e-14:
            c = cn
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i1:i1+blen] = b1
            h[i2:i2+blen] = b2
    return best_h, best_c

def _perturb_sin(h0, rng):
    n = len(h0)
    x = np.linspace(0, 2, n)
    freq = rng.randint(2, 7)
    amp = rng.uniform(0.08, 0.25)
    phase = rng.uniform(0, 2 * np.pi)
    h = h0 + amp * np.sin(freq * np.pi * x + phase)
    return _proj(h, n * 0.5)

def _perturb_rand(h0, rng, sc=0.03):
    n = len(h0)
    return _proj(h0 + sc * (rng.random(n) - 0.5), n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5_direct(best)

    # Phase 1: Quick multi-start FISTA (5% of budget)
    p1_end = t0 + min(40.0, 0.05 * budget_s)
    cands = [best.copy(),
             _proj(0.5 + 0.5 * (h0 - h0[::-1]), n0 * 0.5),
             _proj(0.5 + 0.5 * (h0 + h0[::-1] - 1.0), n0 * 0.5)]
    for _ in range(5):
        cands.append(_perturb_sin(h0, rng))
    for _ in range(3):
        cands.append(_perturb_rand(h0, rng))
    per_c = max(1.0, (p1_end - t0) / max(1, len(cands)))
    for cnd in cands:
        if time.time() >= p1_end - 1:
            break
        slot = min(per_c, p1_end - time.time() - 1)
        if slot < 1.0:
            break
        hf, bh, bc = _fista(cnd, 300, slot, 10.0)
        cf = _c5_direct(hf)
        if bc < best_c:
            best_c, best = bc, bh
        if cf < best_c:
            best_c, best = cf, hf

    # Phase 2: Resolution cycling (25% of budget)
    p2_end = t0 + 0.25 * budget_s
    res_list = [m for m in [n0//2, n0//3, n0//4, n0//6] if m >= 24]
    if len(res_list) > 0:
        per_res = max(5.0, (p2_end - t0) / max(1, len(res_list)))
        for m in res_list:
            if time.time() >= p2_end - 3:
                break
            slot = min(per_res, p2_end - time.time() - 2)
            if slot < 4:
                break
            hm = _resample(best, m)
            t_mid = time.time() + slot * 0.55
            hf, bh, bc = _fista(hm, 800, max(1.0, t_mid - time.time()), 10.0)
            t_ls = time.time() + slot * 0.3
            if t_ls > time.time():
                b2, c2 = _local_search(hf, rng, t_ls)
                if c2 < bc:
                    bc, bh = c2, b2
            hu = _resample(bh, n0)
            cu = _c5_direct(hu)
            if cu < best_c:
                best_c, best = cu, hu
            # Also try from a perturbed version at this resolution
            if time.time() < p2_end - 5:
                hm2 = _perturb_rand(_resample(best, m), rng, 0.02)
                hf2, bh2, bc2 = _fista(hm2, 300, min(10.0, p2_end - time.time() - 2), 10.0)
                hu2 = _resample(bh2, n0)
                cu2 = _c5_direct(hu2)
                if cu2 < best_c:
                    best_c, best = cu2, hu2

    # Phase 3: Main FISTA at full resolution (40% of budget)
    p3_end = t0 + 0.65 * budget_s
    if p3_end > time.time() + 10:
        alphas = [50, 200, 600, 2000, 8000, 30000, 100000, 300000]
        hcur = best.copy()
        tot = p3_end - time.time()

        for a in alphas:
            if time.time() >= p3_end - 2:

            slot = min(pa, p3_end - time.time() - 1)
            if slot < 1.5:
                break
            hf, bh, bc = _fista(hcur, a, slot, 10.0)
            cf = _c5_direct(hf)
            if bc < best_c:
                best_c, best = bc, bh
            if cf < best_c:
                best_c, best = cf, hf
            hcur = hf

    # Phase 4: Block swap search (10% of budget)
    p4_end = t0 + 0.78 * budget_s
    if p4_end > time.time() + 8:
        b, c = _block_swap_search(best, rng, p4_end)
        if c < best_c:
            best_c, best = c, b

    # Phase 5: Final local search (remaining)
    p5_end = t0 + budget_s - 3.0
    if p5_end > time.time():
        best, best_c = _local_search(best, rng, p5_end)
        # Extra kick: perturb and local search again
        if time.time() < p5_end - 30:
            hk = _perturb_rand(best, rng, 0.01)
            b2, c2 = _local_search(hk, rng, p5_end - 10)
            if c2 < best_c:

    # Final variant checks
    for hh in [best, 1.0 - best, best[::-1]]:
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5_direct(h2)
        if c < best_c:
            best, best_c = h2, c

    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
