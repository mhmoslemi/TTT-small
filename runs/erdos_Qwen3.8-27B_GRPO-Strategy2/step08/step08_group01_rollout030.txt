import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * 2.0 / n))

def _eval_arr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(150):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = _eval_arr(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    logw = alpha * (ct - m)
    logw = logw - logw.max()
    w = np.exp(np.clip(logw, -400, 400))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for ii in range(top_k):
        if w[ii] < 1e-15:
            continue
        shift = idx[ii] - (n - 1)
        wk = w[ii] * dx
        if shift >= 0:
            g[shift:n] += wk * (1.0 - h[:n - shift])
            g[:n - shift] += wk * (-h[shift:n])
        else:
            sp = -shift
            g[:n - sp] += wk * (1.0 - h[sp:n])
            g[sp:n] += wk * (-h[:n - sp])
    return g

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_steps, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.75
    step = step0
    cur_c = _eval(h)
    for _ in range(n_steps):
        g = _grad_lse(h, alpha, top_k)
        v = mu * v + (1.0 - mu) * g
        y = h - step * v
        h_try = _proj_sum(y, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.04, step0 * 5.0)
        else:
            step = max(step * 0.5, step0 * 0.05)
    return h, cur_c

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _eval_arr(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = alpha * (ct - m)
        s = s - s.max()
        return m + np.log(np.exp(np.clip(s, -400, 400)).sum()) / alpha
    def _obj_g(x):
        return _grad_lse(x, alpha, top_k)
    def _sum_eq(x):
        return x.sum() - target
    def _sum_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_j}]
    try:
        res = minimize(_obj, h, jac=_obj_g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h
