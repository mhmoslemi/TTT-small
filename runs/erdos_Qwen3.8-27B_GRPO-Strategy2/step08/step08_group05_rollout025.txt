        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stall += 1
            if stall > 30:
                break
            continue
        stall = 0
        d = res.x
        improved = False
        for _ in range(14):
            h_new = _repair(c + d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-17:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            d *= 0.5
        if not improved:
            d2 = rng.randn(n)
            d2 -= d2.mean()
            d2 *= eps / (np.max(np.abs(d2)) + 1e-15)
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _grad_smooth(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _smooth_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smooth_obj(h, alpha, topk)
    v = np.zeros(n)
    beta, step = 0.8, step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smooth(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        h_try = _proj(h - step * v, n * 0.5)
        fn = _smooth_obj(h_try, alpha, topk)
        if fn < f:
            h, f = h_try, fn
            step = min(step * 1.03, step0 * 6.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj(h - step * v, n * 0.5)
            f = _smooth_obj(h, alpha, topk)
    return _repair(h, n)

def _smax_slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smooth_obj(x, alpha, topk)
    def jac(x):
        return _grad_smooth(x, alpha, topk)
    cons = [
        {'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
         'jac': lambda x: np.ones(n)}
    ]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 100})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _fine_search(h, n_iter, steps_arr, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for ss in steps_arr:
        ni2 = n_iter // len(steps_arr)
        for _ in range(ni2):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = ss * (1.0 if h[i] < h[j] else -1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP (60s)
    dl1 = min(deadline, t0 + 55)
    for a, tk, mi in [(15000, 30, 250), (80000, 12, 200)]:
        if time.time() > dl1:
            break
        hv = _smax_slsqp(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent top-12 (250s)
    dl2 = min(deadline, t0 + 300)
    hv, cv = _lp_descent(best_h.copy(), 25000, 0.003, 12, dl2, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov (120s)
    dl3 = min(deadline, t0 + 420)
    for a, tk, st, ns in [(80000, 20, 0.001, 3000), (400000, 8, 0.0005, 2000)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: LP descent top-8 finer (180s)
    dl4 = min(deadline, t0 + 600)
    hv, cv = _lp_descent(best_h.copy(), 20000, 0.0012, 8, dl4, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: Multi-start perturbation + LP (180s)
    dl5 = min(deadline, t0 + 780)
    for ps in [0.003, 0.0015, 0.0008]:
        if time.time() > dl5:
            break
        h_p = best_h + ps * rng.randn(n)
        h_p = _proj(h_p, n * 0.5)
        hv, cv = _lp_descent(h_p, 6000, 0.002, 10, dl5, rng)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 6: Block anneal (100s)
    dl6 = min(deadline, t0 + 880)
    blocks = [1, 2, 3, 4, 5, 7, 9]
    hv, cv = _block_anneal(best_h.copy(), 40000, rng, dl6, 0.003, 5e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 7: Fine pairwise search (80s)
    dl7 = min(deadline, t0 + 960)
    steps_arr = [2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8]
    best_h, best_c5 = _fine_search(best_h.copy(), 60000, steps_arr, rng, dl7)

    # Phase 8: Final micro-polish
    dl8 = deadline
    steps_arr2 = [2e-7, 1e-7, 5e-8, 2e-8]
    best_h, best_c5 = _fine_search(best_h, 30000, steps_arr2, rng, dl8)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
