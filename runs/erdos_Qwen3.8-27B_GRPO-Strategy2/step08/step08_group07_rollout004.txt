import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_smax(h, alpha, topk=0):
    n = len(h)
    c = _correl(h)
    cm = c.max()
    if topk > 0 and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - cm), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii]); sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh; g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _grad_exact(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0: g[sh:] -= h[:-sh]
        else:
            sp = -sh; g[sp:] += (1.0 - h[:-sp]); g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov(h, mode, steps, step0, deadline, alpha0=5000, alpha1=50000, topk=0, tol=0.002, beta=0.85):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for i in range(steps):
        if time.time() > deadline:
            break
        if mode == 'smax':
            frac = i / max(steps - 1, 1)
            alpha = alpha0 * (alpha1 / alpha0) ** frac
            g = _grad_smax(h, alpha, topk)
        else:
            g, _ = _grad_exact(h, tol)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h + step * v, target)
            f = _eval(h)
    return _repair(best_h, n), best_f

def _polish(h, niter, rng, deadline, scale):

    for _ in range(niter):
        if time.time() > deadline:

        j = (i + 1 + rng.randint(0, 8)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

    # Phase 1: Smoothed Nesterov (alpha ramps up)
    dl1 = min(deadline, t0 + 80)
    h2, c5v = _nesterov(best_h.copy(), 'smax', 3000, 0.004, dl1, alpha0=2000, alpha1=30000, topk=15, beta=0.84)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 2: Exact subgradient Nesterov (multiple configs)
    dl2 = min(deadline, t0 + 300)
    configs = [
        (0.003, 0.004, 0.86, 3000),
        (0.0015, 0.0015, 0.90, 3000),
        (0.0008, 0.0008, 0.92, 3000),
    ]
    for step0, tol, mom, ns in configs:
        if time.time() > dl2:
            break
        h2, c5v = _nesterov(best_h.copy(), 'exact', ns, step0, dl2, tol=tol, beta=mom)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Sharper smoothed Nesterov
    dl3 = min(deadline, t0 + 420)
    for al0, al1, tk, s0 in [(50000, 300000, 5, 0.001), (100000, 500000, 3, 0.0005)]:
        if time.time() > dl3:
            break
        h2, c5v = _nesterov(best_h.copy(), 'smax', 2500, s0, dl3, alpha0=al0, alpha1=al1, topk=tk, beta=0.90)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Coordinate descent (the key new phase)
    dl4 = min(deadline, t0 + 780)
    h2, c5v = _coord_descent(best_h.copy(), n0, 200, dl4)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 5: Jitter + re-optimize (escape local min)
    dl5 = min(deadline, t0 + budget_s - 100)
    for _ in range(4):
        if time.time() > dl5:
            break
        hj, cj = _jitter_reopt(best_h.copy(), rng, 0.001, 0.001, dl5)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()
            h2, c5v = _coord_descent(best_h.copy(), n0, 50, dl5)
            if c5v < best_c5:
                best_c5, best_h = c5v, h2.copy()

    # Phase 6: Final fine Nesterov
    dl6 = deadline - 40
    h2, c5v = _nesterov(best_h.copy(), 'exact', 4000, 0.0004, dl6, tol=0.0004, beta=0.93)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Coordinate descent final
    dl7 = deadline - 20
    h2, c5v = _coord_descent(best_h.copy(), n0, 50, dl7)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 8: Polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.002), 40000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 5e-7)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
