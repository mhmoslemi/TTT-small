import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stall += 1
            if stall > 50:
                break
            continue
        stall = 0
        d = res.x
        step = 1.0
        improved = False
        for _ in range(15):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            d2 = np.random.randn(n)
            d2 -= d2.mean()
            d2 *= eps / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    nm = s.max()
    return m + (nm + np.log(np.sum(np.exp(s - nm)))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    nm = s.max()
    w = np.exp(s - nm); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        k, sh = idx[ii], (n-1)-idx[ii]
        gk = _sg(h, k, n)
        g += w[ii]*dx*gk
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_desc(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta, step = 0.85, step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_smax(h, alpha, topk)
        v = beta*v + (1.0-beta)*g
        hn = _proj(h - step*v, n*0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step*1.02, step0*8.0)
        else:
            step *= 0.5
            if step < 1e-12: break
            h = _proj(h - step*v, n*0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _sweep_polish(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    sss = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8]
    for ss in sss:
        ni2 = niter // len(sss)
        for _ in range(ni2):

            i = rng.randint(nc); j = (i+1+rng.randint(0,4))%nc
            d = ss*(2*rng.random()-1)
            ni, nj = h[i]+d, h[j]-d
            if 0<=ni<=1 and 0<=nj<=1:

                    c5c=c5n
                    if c5n<bc: bc,bh=c5n,h.copy()
                else:
                    h[i],h[j] = ni-d, nj+d
    return bh, bc

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax_obj(x, alpha, topk)
    def jac(x): return _grad_smax(x, alpha, topk)
    def cfun(x): return x.sum()-n*0.5
    def cjac(x): return np.ones(n)
    cons = [{'type':'eq','fun':cfun,'jac':cjac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter':maxit,'ftol':1e-22,'maxls':80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _lp_full(h, n_steps, eps, m_top, deadline):

        if time.time() > deadline: break

        itop = np.argpartition(cor, -m_top)[-m_top:]
        itop = itop[np.argsort(-cor[itop])]
        grads = np.array([_sg(c, k, n) for k in itop])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None

        except Exception:
            break

            stall += 1

            continue

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax(h, alpha, topk):
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    nm = s.max()
    return m + (nm + np.log(np.sum(np.exp(s - nm)))) / alpha

def _g_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    nm = s.max()
    w = np.exp(s - nm); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        g += w[ii] * dx * _sg(h, idx[ii], n)
    return g

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax(x, alpha, topk)
    def jac(x): return _g_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-22, 'maxls': 90})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)
