def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)

    for ii in range(len(idx)):

- For each peak at shift k, the correlation is sum_i h[i] * (1-h[i+k]) * dx (approximately)

  - So to reduce the correlation at shift k, we should decrease h[i] where (1-h[i+k]) - h[i-k] > 0

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

_G = {'n': 0, 'alpha': 1.0, 'topk': 10}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _obj_w(x):
    return _smax_obj(x, _G['alpha'], _G['topk'])

def _jac_w(x):
    return _grad_smax(x, _G['alpha'], _G['topk'])

def _cons_fun(x):
    return x.sum() - _G['n'] * 0.5

def _cons_jac(x):
    return np.ones(_G['n'])

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    _G['n'] = n; _G['alpha'] = alpha; _G['topk'] = topk
    try:
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        r = minimize(_obj_w, h, jac=_jac_w, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h0, alpha0, grow, topk, max_iter, step0, deadline):
    n = len(h0)
    h = h0.copy()
    f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n)
    beta = 0.75
    step = step0
    alpha = alpha0
    best_h = h.copy(); best_f = f
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-15:
            h = hn; f = fn
            step = min(step * 1.02, step0 * 15.0)
            stall = 0
            if f < best_f:
                best_f = f; best_h = h.copy()
        else:
            step *= 0.4
            ok = False
            for _ in range(4):
                hn = _proj(h - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f - 1e-15:
                    h = hn; f = fn; ok = True; break
                step *= 0.4
            if not ok:
                stall += 1
        if it > 0 and it % 400 == 0:
            alpha = min(alpha * grow, 5e7)
            v *= 0.1; step = step0 * 0.3
        if stall > 250:
            v = np.zeros(n); stall = 0
    return _repair(best_h, n)

def _peak_search(h, deadline, n_iter, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = _correl(h)
        topk = 5
        idx_t = np.argpartition(c, -topk)[-topk:]
        k_w = idx_t[np.argmax(c[idx_t])]
        sh = (n - 1) - k_w
        g = np.zeros(n)
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0: g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n-sp] -= h[sp:]
        gc = g - g.mean()
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        delta = 3e-5 * (gc[j] - gc[i])
        delta = np.clip(delta, -6e-5, 6e-5)
        if abs(delta) < 1e-9:
            continue
        ni, nj = h[i] + delta, h[j] - delta
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc: bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - delta, nj + delta
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    step_base = 5e-6
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 15)) % nc
        d = step_base * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc: bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step_base *= 0.97
                if step_base < 1e-9: step_base = 1e-9
    return bh, bc

def _multires_polish(h, deadline):
    n = len(h)
    if n % 2 != 0:
        return _repair(h, n)
    n_lo = n // 2
    h_lo = 0.5 * (h[0::2] + h[1::2]).copy()
    h_lo = _repair(h_lo, n_lo)

    h_hi = _nesterov(h_hi, 20000, 1.5, 8, 2000, 0.001, min(deadline, time.time() + (deadline - time.time()) * 0.5))
    return h_hi
