from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_smax_full(h, alpha):
    """Gradient of log-sum-exp of all correlation values."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _smax_all(h, alpha):
    """Log-sum-exp objective value."""
    c = _correl(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _slsqp_smooth(h, alpha, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_all(x, alpha)
    def grad(x):
        return _grad_smax_full(x, alpha)[0]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:

                     bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:

def _nesterov(h, mode, steps, step0, deadline, **kw):

    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    beta = kw.get('beta', 0.85)
    step = step0

    tol = kw.get('tol', 0.003)
    alpha = kw.get('alpha', 10000)

        if mode == 'exact':

            active = np.where(c >= m - tol)[0]

                else:
                    sp = -sh
                    g[sp:] += (1.0 - h[:-sp])
                    g[:n - sp] -= h[sp:]
            g *= dx / len(active)
        elif mode == 'smax':

def _polish(h, niter, rng, deadline, scale):

    for _ in range(niter):

def _pair_descent(h, niter, rng, deadline):

    n = len(h)

            break
        i = rng.randint(n)
        j = rng.randint(n)

                        break
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d
            if h[i] != h[i]:  # skip
                pass

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_k(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g

        h2, c5v = _nesterov(best_h.copy(), 'smax', ns, step0, dl3, alpha=alpha, beta=mom)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Jitter + re-optimize (escape local minima)
    dl4 = min(deadline, t0 + 600)
    for _ in range(6):
        if time.time() > dl4:
            break
        d = 0.0015 * rng.randn(n0)
        d -= d.mean()
        hj = _repair(np.clip(best_h + d, 0, 1), n0)
        h2, c5v = _nesterov(hj, 'exact', 4000, 0.001, dl4, tol=0.003, beta=0.88)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 5: Extended simulated annealing
    dl5 = min(deadline, t0 + 780)
    blocks = [2, 3, 4, 5, 7, 10, 14, 20]
    best_h, best_c5 = _anneal(best_h.copy(), 150000, rng, dl5, blocks, 0.003, 1e-13)

    # Phase 6: Gradient-guided pair descent
    dl6 = min(deadline, t0 + budget_s - 60)
    best_h, best_c5 = _grad_polish(best_h, 80000, rng, dl6)

    # Phase 7: Fine random polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0015), 80000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 3e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
