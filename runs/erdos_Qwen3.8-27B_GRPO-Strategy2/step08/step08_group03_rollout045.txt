        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        convR = np.fft.irfft(H * Urev, Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * Ur, Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _active_grad(h, P, Nv):
    n = len(h)
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-10, 5.0 * (2.0 / n))
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[1:][mP] = 1.0
            cnt += int(mP.sum())
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[1:][mN] = 1.0
            cnt += int(mN.sum())
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv
    wN *= inv
    return _grad_from_weights(h, wP, wN)

def _fista(h0, alpha, max_time, L=10.0):
    if max_time <= 0.0:
        h = h0.copy()
        return h, h.copy(), _c5_direct(h)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _smooth_value(h, alpha)
    fh = fy
    best_h = h.copy()
    best_c = _c5_direct(h)
    stale = 0
    Lcur = max(L, 1.0)
    for it in range(500000):
        if time.time() - t0 > max_time:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.5 or y.max() > 1.5 or abs(y.sum() - S) > 0.01:
                y = _proj(y, S)
                t = 1.0
            fy = _smooth_value(y, alpha)
        else:
            y = h
            fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(16):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            improvement = fy - fx
            hp = h
            h = xn
            fh = fx
            if improvement < 1e-14:
                stale += 1
            else:
                stale = 0
            Lcur = max(Lcur * 0.9, 1.0)
            if stale > 20:
                # restart
                hp = h.copy()
                y = h.copy()
                t = 1.0
                fy = fh
                Lcur = max(Lcur * 0.7, 1.0)
                stale = 0
            if it % 4 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t = 1.0
            fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _subgrad_descent(h0, max_time, step_init=1e-4):
    """Direct subgradient descent on the max objective."""
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    step = step_init
    while time.time() - t0 < max_time:
        P, Nv, mx = _pn_arrays(h)
        g = _active_grad(h, P, Nv)
        if np.max(np.abs(g)) < 1e-15:
            break
        h_new = _proj(h - step * g, S)
        c_new = _c5_direct(h_new)
        if c_new < c:
            h = h_new
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
            step *= 1.02  # slightly increase
        else:
            step *= 0.7
            if step < 1e-12:
                break
    return best_h, best_c

def _local_search(h0, rng, deadline, start_step=5.0e-5):
    tl0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    P, Nv, _ = _pn_arrays(h)
    g = _active_grad(h, P, Nv)
    order = np.argsort(g)
    q = max(5, n // 6)
    low = order[:q]
    high = order[-q:]
    recompute = 1024
    it = 0
    while time.time() < deadline:
        if it >= recompute:
            c = _c5_direct(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            P, Nv, _ = _pn_arrays(h)
            g = _active_grad(h, P, Nv)
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            recompute = it + 1024
        frac = min(1.0, (time.time() - tl0) / max(1e-9, deadline - tl0))
        base = max(1e-12, start_step * (1e-6) ** frac)
        if rng.random() < 0.7:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i != j:
            if g[i] <= g[j]:
                s = 1
            elif g[j] < g[i]:
                s = -1
            else:
                s = 1 if rng.random() < 0.5 else -1
            if s > 0:
                avail = min(1.0 - h[i], h[j])
            else:
                avail = min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                mult = 0.3 + 1.7 * rng.random()
                if rng.random() < 0.05:
                    mult *= 10.0
                d = s * min(avail, base * mult)
                h[i] += d
                h[j] -= d
                cn = _c5_direct(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)
