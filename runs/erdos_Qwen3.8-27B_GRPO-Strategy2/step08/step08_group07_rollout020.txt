def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    """Project h to [0,1]^n with sum = target using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = _proj_sum(h, target)
    else:
        diff = target - s
        if abs(diff) > 1e-12:
            idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in idx:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return _repair(best_h, n), best_f

def _slsqp(h, alpha, ntop, maxit, deadline):
    """SLSQP with log-sum-exp objective."""
    n = len(h)
    def obj(x):
        c = _correl(x)
        m = c.max()
        if ntop > 0 and ntop < len(c):
            idx = np.argpartition(c, -ntop)[-ntop:]
            ct = c[idx]
        else:
            ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (np.log(np.sum(np.exp(s - s.max()))) + s.max()) / alpha
    
    def jac(x):
        return _grad_lse(x, alpha, ntop)[0]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return h

def _equalize_top(h, n_top, n_iter, deadline, step=0.002):
    """Try to equalize the top n_top correlation values."""
    n = len(h)
    target = n * 0.5
    best = h.copy()
    best_f = _eval(h)
    c5 = best_f
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        c = _correl(h)
        top_idx = np.argsort(c)[-n_top:]
        top_vals = c[top_idx]
        mean_top = top_vals.mean()
        # Gradient to reduce variance of top values
        dx = 2.0 / n
        g = np.zeros(n)
        for k in top_idx:
            diff = c[k] - mean_top
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += diff * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= diff * h[:-sh]
            else:
                sp = -sh
                g[sp:] += diff * (1.0 - h[:-sp])
                if sp < n:
                    g[:n-sp] -= diff * h[sp:]
        g *= dx / n_top
        h_new = _proj_sum(h - step * g, target)
        f_new = _eval(h_new)
        if f_new < c5:
            c5 = f_new
            h = h_new
            if f_new < best_f:
                best_f = f_new
                best = h.copy()
    return _repair(best, n), best_f

def _anneal_structured(h, n_iter, rng, deadline, scale):
    """Structured annealing: moves that shift mass between regions."""
    n = len(h)
    target = n * 0.5
    best = h.copy()
    best_f = _eval(h)
    c5 = best_f
    T = scale
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = scale * 0.999**it
        # Random block move
        bs = rng.randint(2, min(15, n // 4))
        i0 = rng.randint(0, n - bs)
        seg = h[i0:i0+bs].copy()
        noise = T * rng.randn(bs)
        noise -= noise.mean()
        seg += noise
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        f_new = _eval(hn)
        if f_new < c5 or rng.random() < np.exp(-(f_new - c5) / max(T * 0.1, 1e-15)):
            h = hn
            c5 = f_new
            if f_new < best_f:
                best_f = f_new
                best = h.copy()
    return _repair(best, n), best_f

def _polish(h, n_iter, rng, deadline, scale):
    """Fine coordinate polish."""
    n = len(h)
    best = h.copy()
    best_f = _eval(h)
    c5 = best_f
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            f_new = _eval(h)
            if f_new < c5:
                c5 = f_new
                if f_new < best_f:
                    best_f = f_new
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return _repair(best, n), best_f

            lo = mid
        else:

        ti = np.argsort(c)[-ntop:]
        tv = c[ti]; mt = tv.mean(); dx = 2.0 / n
        g = np.zeros(n)
        for k in ti:
            d = c[k] - mt; sh = (n - 1) - k
            if sh >= 0:
                if sh < n: g[:n - sh] += d * (1.0 - h[sh:])
                if sh > 0: g[sh:] -= d * h[:-sh]
            else:
                sp = -sh
                if sp < n: g[sp:] += d * (1.0 - h[:-sp])
                if sp < n: g[:n - sp] -= d * h[sp:]
        g *= dx / ntop
        hn = _proj_sum(h - step * g, target); fn = _eval(hn)
        if fn < c5:
            c5 = fn; h = hn
            if fn < bf: bf, best = fn, h.copy()
    return _repair(best, n), bf

def _polish(h, niter, rng, dl, scale):
    n = len(h); best = h.copy(); bf = _eval(h); c5 = bf
    for _ in range(niter):
        if time.time() > dl:
            break
        i = rng.randint(n); j = (i + 1 + rng.randint(0, 7)) % n
        d = scale * rng.randn(); ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj; fn = _eval(h)
            if fn < c5:
                c5 = fn
                if fn < bf: bf, best = fn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return _repair(best, n), bf

def _anneal(h, niter, rng, dl, S0, S1):
    n = len(h); best = h.copy(); bf = _eval(h); c5 = bf
    for it in range(niter):
        if time.time() > dl:
            break
        T = S0 * (S1 / S0) ** (it / max(niter, 1))
        bs = rng.randint(2, min(12, n // 5))
        i0 = rng.randint(0, n - bs)
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-9) * rng.randn(bs)
        d -= d.mean(); seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg; hn = _repair(hn, n)
        fn = _eval(hn)
        if fn < c5 or rng.random() < np.exp(-(fn - c5) / max(T * 0.05, 1e-15)):
            h, c5 = hn, fn
            if fn < bf: bf, best = fn, h.copy()
    return _repair(best, n), bf

def _multi_n_opt(n, seed, dl, budget_per_n):
    rng = np.random.RandomState(seed)
    h = np.clip(0.5 * np.ones(n) + 0.15 * rng.randn(n), 0, 1)
    h = _repair(h, n)
    h = _slsqp(h, 5000, 40, 200, min(dl, time.time() + budget_per_n))
    h2, c2 = _nesterov_lse(h, 1500, 0.002, 200, 20000, min(dl, time.time() + budget_per_n), 0.88, 0)
    return _repair(h2, n), c2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 3
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh, bc = h.copy(), _eval(h)

    # Phase 1: SLSQP
    d1 = min(dl, t0 + 70)
    for al, nt, mi in [(5000, 50, 350), (20000, 20, 300), (80000, 8, 250)]:
        if time.time() > d1: break
        h1 = _slsqp(bh.copy(), al, nt, mi, d1)
        c1 = _eval(h1)
        if c1 < bc: bc, bh = c1, h1.copy()

    # Phase 2: LSE Nesterov (adaptive alpha)
    d2 = min(dl, t0 + 230)
    for a1, a2, s0, st, bt in [(300, 50000, 0.003, 3000, 0.87), (100, 200000, 0.002, 2500, 0.90), (50, 500000, 0.001, 2000, 0.93)]:
        if time.time() > d2: break
        h2, c2 = _nesterov_lse(bh.copy(), st, s0, a1, a2, d2, bt, 0)
        if c2 < bc: bc, bh = c2, h2.copy()

    # Phase 3: Exact Nesterov
    d3 = min(dl, t0 + 420)
    for s0, tol, bt, st in [(0.004, 0.006, 0.86, 3000), (0.0015, 0.001, 0.91, 3000), (0.0005, 0.0004, 0.94, 3500)]:
        if time.time() > d3: break
        h3, c3 = _nesterov_exact(bh.copy(), st, s0, tol, d3, bt)
        if c3 < bc: bc, bh = c3, h3.copy()

    # Phase 4: Equalize + refine
    d4 = min(dl, t0 + 570)
    for nt, ni, st in [(5, 2500, 0.003), (10, 2000, 0.002), (3, 2000, 0.004), (20, 1500, 0.001)]:
        if time.time() > d4: break
        h4, c4 = _equalize_top(bh.copy(), nt, ni, d4, st)
        if c4 < bc: bc, bh = c4, h4.copy()
        if time.time() < d4:
            h5, c5 = _nesterov_exact(h4.copy(), 1000, 0.0008, 0.001, d4, 0.93)
            if c5 < bc: bc, bh = c5, h5.copy()

    # Phase 5: Multi-restart jitter + LSE
    d5 = min(dl, t0 + 740)
    for js in [0.004, 0.008, 0.015, 0.002, 0.012, 0.006]:
        if time.time() > d5: break
        d = js * rng.randn(n); d -= d.mean()
        hj = _repair(np.clip(bh + d, 0, 1), n)
        hj2, cj2 = _nesterov_lse(hj, 1500, 0.0015, 200, 30000, d5, 0.88, 0)
        if cj2 < bc: bc, bh = cj2, hj2.copy()

    # Phase 6: Annealing
    d6 = min(dl, t0 + 870)
    h6, c6 = _anneal(bh.copy(), 50000, rng, d6, 0.004, 1e-10)
    if c6 < bc: bc, bh = c6, h6.copy()

    # Phase 7: Fine LSE + exact
    d7 = min(dl, t0 + budget_s - 50)
    h7, c7 = _nesterov_lse(bh.copy(), 3000, 0.0008, 10000, 300000, d7, 0.92, 5)
    if c7 < bc: bc, bh = c7, h7.copy()
    h8, c8 = _nesterov_exact(bh.copy(), 4000, 0.0003, 0.0002, d7, 0.95)
    if c8 < bc: bc, bh = c8, h8.copy()

    # Phase 8: Polish
    rem = max(0, dl - time.time())
    npol = min(int(rem / 0.0012), 40000)
    bh, bc = _polish(bh, npol, rng, dl, 4e-7)

    bh = _repair(bh, n)
    bc = _eval(bh)
    return (bh, bc, n)
