import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            step = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += np.sign(diff) * step
            diff -= np.sign(diff) * step
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max())

def _subgrad_topk(h, k, step, n_iter, t_end):
    """Subgradient descent using top-k correlations with backtracking."""
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(c)[-k:]
        g = np.zeros(n)
        wk = 1.0 / k
        for ki in top_k:
            sh = (n - 1) - ki
            if sh >= 0:
                g[:n-sh] += wk * dx * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * dx * h[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * dx * (1.0 - h[:n-sp])
                g[:n-sp] -= wk * dx * h[sp:]
        # Backtracking line search
        alpha = step
        for _bs in range(20):
            h_new = _repair(h - alpha * g, n)
            c5_new = _eval(h_new)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                break
            alpha *= 0.5
        else:
            break
    return _repair(h, n), c5

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_global = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def try_update(h_new):
        nonlocal best_h, best_c5
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_new.copy()
        return c5v

    # Phase 1: SLSQP with moderate alpha (broad navigation)
    t_phase = t0 + budget_s * 0.15
    for alpha in [2000, 20000]:
        if time.time() > t_phase - 10:
            break
        h_try = _slsqp_lse(np.zeros(0), best_h.copy(), n, alpha, 2000, t_end_global)
        try_update(h_try)

    # Phase 2: Coordinate descent (main innovation - exact 1D optimization per coordinate)
    t_phase = t0 + budget_s * 0.35
    h_cd, _ = _coord_descent(best_h.copy(), n, 80, 100, t_phase)
    try_update(_repair(h_cd, n))

    # Phase 3: SLSQP high alpha (focus on worst correlation)
    t_phase = t0 + budget_s * 0.45
    for alpha in [200000, 2000000]:
        if time.time() > t_phase - 15:
            break
        h_try = _slsqp_lse(np.zeros(0), best_h.copy(), n, alpha, 1500, t_end_global)
        try_update(h_try)

    # Phase 4: Fine coordinate descent (narrow window)
    t_phase = t0 + budget_s * 0.60
    h_cd, _ = _coord_fine(best_h.copy(), n, 60, 0.05, t_phase)
    try_update(_repair(h_cd, n))

    # Phase 5: Subgradient k=1 with momentum (directly attack worst correlation)
    t_phase = t0 + budget_s * 0.72
    h_sg, _ = _subgrad_k1(best_h.copy(), n, 30000, 0.0005, t_phase)
    try_update(h_sg)

    # Phase 6: Ultra-fine coordinate descent
    t_phase = t0 + budget_s * 0.80
    h_cd, _ = _coord_fine(best_h.copy(), n, 100, 0.005, t_phase)
    try_update(_repair(h_cd, n))

    # Phase 7: Another subgradient pass
    t_phase = t0 + budget_s * 0.88
    h_sg, _ = _subgrad_k1(best_h.copy(), n, 20000, 0.0002, t_phase)
    try_update(h_sg)

    # Phase 8: Local search (2-point, fine scale)
    t_phase = t0 + budget_s * 0.95
    rem = max(0, int((t_phase - time.time()) / 0.003))
    rem = min(rem, 30000)
    h_ls, _ = _local_search(best_h.copy(), n, rem, 1e-5, 777, t_phase)
    try_update(h_ls)

    # Phase 9: Ultra-fine local search
    rem = max(0, int((t_end_global - 2 - time.time()) / 0.003))
    rem = min(rem, 20000)
    if rem > 100:
        h_ls, _ = _local_search(best_h.copy(), n, rem, 2e-6, 999, t_end_global - 1)
        try_update(h_ls)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
