        s = h.sum()
        if abs(s - S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _proj(v, S):

    x += (S - x.sum()) / x.size

                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
