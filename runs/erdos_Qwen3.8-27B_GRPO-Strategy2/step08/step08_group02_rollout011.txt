        g += w[ii] * _grad_lag(idx[ii], x, n)
    return g

def _slsqp_opt(h, alpha, iters, topk):
    n = len(h)
    try:
        res = minimize(_softmax_obj, h, jac=_softmax_grad, args=(alpha, topk),
                       method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': iters, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_opt(h, alpha, iters, topk, mu):
    n = len(h)
    def f(x):
        return _softmax_obj(x, alpha, topk) + mu * (x.sum() - n / 2.0) ** 2
    def g(x):
        return _softmax_grad(x, alpha, topk) + 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=[(0, 1)] * n,
                       options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10, 'maxfun': iters * 15})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, steps, s0, topk, mom):
    n = len(h)
    v = np.zeros(n)
    step = s0
    cur = _eval(h)
    for it in range(steps):
        _, g = _softmax_grad(h, alpha, topk), None
        v = mom * v + (1.0 - mom) * g
        taken = False
        for _ in range(12):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cur - 1e-16:
                h = hn
                cur = cn
                step = min(step * 1.1, s0 * 20.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, s0 * 20.0)
        if it % 300 == 0:
            v *= 0.3
    return _repair(h, n)

def _anneal(h, ntries, rng, t0, t1, step0):
    n = len(h)
    cur = _eval(h)
    best = h.copy()
    bestc = cur
    for it in range(ntries):
        t = t0 * (t1 / t0) ** (it / max(ntries, 1))
        if rng.rand() < 0.25:
            i = rng.randint(n)
            if 1e-6 < h[i] < 1 - 1e-6:
                tgt = 0.0 if rng.rand() < 0.5 else 1.0
                d = tgt - h[i]
                j = (i + 1 + rng.randint(0, 5)) % n
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    cn = _eval(h)
                    if cn < cur or rng.rand() < np.exp(-(cn - cur) / max(t, 1e-20)):
                        cur = cn
                        if cn < bestc:
                            bestc = cn
                            best = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
            d = step0 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < cur or rng.rand() < np.exp(-(cn - cur) / max(t, 1e-20)):
                    cur = cn
                    if cn < bestc:
                        bestc = cn
                        best = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best, bestc

def _local_fine(h, ntries, rng, step0):
    n = len(h)
    cur = _eval(h)
    best = h.copy()
    bestc = cur
    for _ in range(ntries):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step0 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < bestc:
                    bestc = cn
                    best = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best, bestc

def _local_3pt(h, ntries, rng, step0):
    n = len(h)
    cur = _eval(h)
    best = h.copy()
    bestc = cur
    for _ in range(ntries):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        k = (i + 8 + rng.randint(0, 8)) % n
        d1 = step0 * rng.randn()
        d2 = step0 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < bestc:
                    bestc = cn
                    best = h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best, bestc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with increasing alpha (200s)
    slsqp_cfgs = [
        (3000, 0, 500),
        (20000, 6, 400),
        (100000, 4, 350),
        (500000, 3, 300),
        (2000000, 2, 250),
    ]
    for alpha, tk, mit in slsqp_cfgs:
        if _el() > 200:
            break
        h_new = _slsqp_opt(best_h, alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: L-BFGS-B (200s)
    lbfgs_cfgs = [
        (5000, 0, 500, 10.0),
        (50000, 5, 400, 10.0),
        (500000, 3, 350, 100.0),
        (5000000, 2, 300, 100.0),
        (50000000, 1, 250, 1000.0),
    ]
    for alpha, tk, mit, mu in lbfgs_cfgs:
        if _el() > 420:
            break
        h_new = _lbfgs_opt(best_h, alpha, mit, tk, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Projected gradient (200s)
    pg_cfgs = [
        (5000, 600, 0.005, 8, 0.7),
        (50000, 600, 0.003, 5, 0.75),
        (500000, 500, 0.002, 3, 0.8),
        (5000000, 400, 0.001, 2, 0.85),
        (50000000, 350, 0.0005, 1, 0.9),
    ]
    for alpha, steps, s0, tk, mu in pg_cfgs:
        if _el() > 630:
            break
        h_new = _proj_grad(best_h.copy(), alpha, steps, s0, tk, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: Simulated annealing (150s)
    if _el() < budget_s - 250:
        t_left = budget_s - 250 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        ntry = min(int(t_left / eps * 0.55), 2000000)
        best_h, best_c5 = _anneal(best_h, ntry, rng, 5e-5, 1e-9, 3e-5)
        if best_c5 > _eval(best_h):
            best_c5 = _eval(best_h)

    # Phase 5: 3-point local search (120s)
    if _el() < budget_s - 130:
        t_left = budget_s - 130 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        ntry = min(int(t_left / eps * 0.6), 1500000)
        best_h, best_c5 = _local_3pt(best_h, ntry, rng, 2e-6)

    # Phase 6: Ultra-fine 2-point polish (remaining - 10s)
    if _el() < budget_s - 10:
        t_left = budget_s - 10 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        ntry = min(int(t_left / eps * 0.75), 10000000)
        best_h, best_c5 = _local_fine(best_h, ntry, rng, 5e-8)

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
