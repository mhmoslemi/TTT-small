    c = _corr_vec(h)

def _grad_local(h, step, n_iter, alpha, top_k, rng):
    """Gradient-guided 2-point local search."""
    n = len(h)
    target = n * 0.5
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for it in range(n_iter):
        g = _grad_topk(h, top_k, alpha)
        # Find best (i, j) pair: i has most negative g, j has most positive g
        # Add some randomness to avoid cycling
        neg_mask = g < 0
        pos_mask = g > 0
        if not neg_mask.any() or not pos_mask.any():
            # Random move
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = step * rng.randn()
        else:
            neg_idx = np.where(neg_mask)[0]
            pos_idx = np.where(pos_mask)[0]
            # Weight by |g|
            ng = -g[neg_idx]
            pg = g[pos_idx]
            i = neg_idx[rng.choice(len(neg_idx), p=ng/ng.sum())]
            j = pos_idx[rng.choice(len(pos_idx), p=pg/pg.sum())]

        ni, nj = h[i] + d, h[j] - d

            c_new = _eval(h)

                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:

def _rand_local(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out

def _sine_init(n, f1, f2, a1, a2, phase, rng):
    x = np.arange(n)
    h = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * x / n) + a2 * np.sin(2.0 * np.pi * f2 * x / n + phase)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: MGD from parent with varying params
    configs = [(2000, 120, 6, 0.004, 0.7), (5000, 80, 5, 0.003, 0.75),
               (15000, 60, 4, 0.002, 0.8), (50000, 40, 3, 0.001, 0.85)]
    for alpha, steps, tk, s0, mu in configs:
        if _el() > 150:
            break
        h_t, _ = _mgd(best_h.copy(), steps, alpha, tk, s0, mu, rng)
        c_t = _eval(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()

    # Phase 2: Alternative starting points
    if _el() < budget_s - 500:
        alternatives = [
            _sine_init(n, 3, 7, 0.25, 0.15, 0.7, rng),
            _sine_init(n, 5, 11, 0.2, 0.2, 1.3, rng),
            _sine_init(n, 2, 9, 0.3, 0.1, 2.1, rng),
        ]
        for alt in alternatives:
            if _el() > budget_s - 450:
                break
            h_t, _ = _mgd(alt, 300, 5000, 6, 0.003, 0.7, rng)
            c_t = _eval(h_t)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 3: SLSQP refinement
    if _el() < budget_s - 400:
        for alpha, tk, mxi in [(2000, 10, 500), (8000, 7, 400), (30000, 5, 300)]:
            if _el() > budget_s - 350:
                break
            h_t, c_t = _slsqp(best_h.copy(), alpha, tk, mxi)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 4: Multi-resolution
    if _el() < budget_s - 450:
        for ratio in [2, 4]:
            if _el() > budget_s - 400:
                break
            nc = n // ratio
            hc = _repair(_downsample(best_h, ratio), nc)
            hc, _ = _mgd(hc, 150, 8000, 5, 0.005, 0.7, rng)
            h_up = _repair(_upsample(hc, ratio), n)
            h_up, c_up = _grad_local(h_up, 2000, 5e-5, 5000, 5, rng)
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up.copy()

    # Phase 5: SLSQP after multi-res
    if _el() < budget_s - 300:
        for alpha, tk, mxi in [(5000, 8, 350), (20000, 5, 250), (100000, 4, 200)]:
            if _el() > budget_s - 250:
                break
            h_t, c_t = _slsqp(best_h.copy(), alpha, tk, mxi)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 6: Gradient-guided local search (main improvement)
    if _el() < budget_s - 150:
        remaining = budget_s - 150 - _el()
        n_it = min(int(remaining * 8000), 300000)
        best_h, best_c5 = _grad_local(best_h, n_it, 3e-5, 8000, 5, rng)

    # Phase 7: Random local search for diversity
    if _el() < budget_s - 80:
        remaining = budget_s - 80 - _el()
        n_it = min(int(remaining * 5000), 200000)
        best_h, best_c5 = _rand_local(best_h, n_it, 4e-5, rng)

    # Phase 8: Fine gradient-guided polish
    if _el() < budget_s - 30:
        remaining = budget_s - 30 - _el()
        n_it = min(int(remaining * 12000), 400000)
        best_h, best_c5 = _grad_local(best_h, n_it, 1e-5, 20000, 4, rng)

    # Phase 9: Ultra-fine random polish
    if _el() < budget_s - 5:
        remaining = budget_s - 5 - _el()
        n_it = min(int(remaining * 15000), 500000)
        best_h, best_c5 = _rand_local(best_h, n_it, 2e-6, rng)

    # Phase 10: Final MGD micro-step
    if _el() < budget_s - 2:
        h_t, _ = _mgd(best_h, 200, 50000, 3, 0.0003, 0.85, rng)
        c_t = _eval(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()
        n_fin = min(int((budget_s - 2 - _el()) * 20000), 300000)
        if n_fin > 100:
            best_h, best_c5 = _grad_local(best_h, n_fin, 5e-7, 30000, 3, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
