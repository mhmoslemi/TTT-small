import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        o = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in o:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_lse(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _jac_lse(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w = w / w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in np.where(w > 1e-14)[0]:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

class _SP:
    def __init__(self, alpha, n):
        self.a = alpha
        self.n = n
        self.tgt = n / 2.0
    def obj(self, x):
        return _obj_lse(x, self.a, self.n)
    def jac(self, x):
        return _jac_lse(x, self.a, self.n)
    def cf(self, x):
        return x.sum() - self.tgt
    def cj(self, x):
        return np.ones(self.n)

def _slsqp(h0, alpha, maxiter, n):
    sp = _SP(alpha, n)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(sp.obj, h0.copy(), jac=sp.jac, method='SLSQP',
                     bounds=bnds,
                     constraints=[{'type': 'eq', 'fun': sp.cf, 'jac': sp.cj}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return _repair(h0, n)

def _grad_topk(h, k, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-k:]
    g = np.zeros(n)
    dx = 2.0 / (n * k)
    for ki in idx:
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= dx * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return g

def _mom(h0, k, n_steps, step0, t_end, mu=0.72):
    h = h0.copy()
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        g = _grad_topk(h, k, n)
        hp = h.copy()
        v = mu * v + (1.0 - mu) * g
        h2 = _repair(h - step * v, n)
        f2 = _eval(h2)
        if f2 < f - 1e-16:
            h, f = h2, f2
            step = min(step * 1.02, step0 * 25.0)
        else:
            step *= 0.45
            if step < 1e-14:
                v *= 0.05
                step = step0 * 0.02
            h = _repair(hp - step * v, n)
            f = _eval(h)
    return _repair(h, n), f

def _anneal(h0, n_try, rng, t_end, T0, T1, amp):
    h = h0.copy()
    n = len(h)
    fc = _eval(h)
    best_f, best_h = fc, h.copy()
    for it in range(n_try):
        if time.time() > t_end:
            break
        frac = (it + 1.0) / max(n_try, 1)
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = rng.randn() * amp
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            fn = _eval(h)
            accept = fn <= fc or rng.rand() < np.exp(-(fn - fc) / max(T, 1e-14))
            if accept:
                fc = fn
                if fn < best_f:
                    best_f, best_h = fn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_f

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        o = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in o:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_lse(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _jac_lse(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w = w / w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for k in np.where(w > 1e-14)[0]:
        sh = (n - 1) - k
        wk = w[k] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

class _SP:
    def __init__(self, alpha, n):
        self.a = alpha
        self.n = n
        self.tgt = n / 2.0
    def obj(self, x):
        return _obj_lse(x, self.a, self.n)
    def jac(self, x):
        return _jac_lse(x, self.a, self.n)
    def cf(self, x):
        return x.sum() - self.tgt
    def cj(self, x):
        return np.ones(self.n)

def _slsqp(h0, alpha, maxiter, n):
    sp = _SP(alpha, n)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(sp.obj, h0.copy(), jac=sp.jac, method='SLSQP',
                     bounds=bnds,
                     constraints=[{'type': 'eq', 'fun': sp.cf, 'jac': sp.cj}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return _repair(h0, n)

def _grad_topk(h, k, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-k:]
    g = np.zeros(n)
    dx = 2.0 / (n * k)
    for ki in idx:
        sh = (n - 1) - ki
        if sh >= 0:
            g[:n - sh] += dx * (1.0 - h[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += dx * (1.0 - h[:n - sp])
            g[:n - sp] -= dx * h[sp:]
    return g
