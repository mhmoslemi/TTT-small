import numpy as np
from scipy.optimize import minimize
import time

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_correl(h)))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)
        ct = c[idx[-topk:]]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return float(m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha)

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    nc = len(c)
    if topk < nc:
        idx = np.argpartition(c, -topk)
        idx = idx[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(nc)
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        j = int(idx[ii])
        s_shift = j - (n - 1)
        wk = w[ii] * dx
        if s_shift >= 0:
            ss = s_shift
            if n - ss > 0:
                g[:n - ss] += wk * (1.0 - h[ss:])
            if ss > 0:
                g[ss:] += -wk * h[:n - ss]
        else:
            ss = -s_shift
            if n - ss > 0:
                g[:n - ss] += wk * (1.0 - h[ss:])
                g[:n - ss] += -wk * h[ss:]
            if ss > 0:
                g[ss:] += -wk * h[:n - ss]
                g[ss:] += wk * (1.0 - h[:n - ss])
    return g

def _lbfgs_refine(h, alpha, topk, maxiter, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(_smax_obj, h, args=(alpha, topk),
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-18,
                              'gtol': 1e-12, 'maxfun': maxiter * 4})
        return _repair(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def _nesterov(h, steps, step0, deadline, alpha, topk, beta):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _adam_step(h, m, v, t, lr, g, b1, b2, eps, target):
    m = b1 * m + (1 - b1) * g
    v = b2 * v + (1 - b2) * g * g
    mhat = m / (1 - b1 ** t)
    vhat = v / (1 - b2 ** t)
    return m, v, _proj(h - lr * mhat / (np.sqrt(vhat) + eps), target)

def _adam_opt(h, steps, lr, deadline, alpha, topk, b1, b2, eps):
    n = len(h)
    target = n * 0.5
    m = np.zeros(n)
    v = np.zeros(n)
    best_h, best_f = h.copy(), _eval(h)
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        m, v, h = _adam_step(h, m, v, t, lr, g, b1, b2, eps, target)
        f = _eval(h)
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = int(blocks[rng.randint(len(blocks))])
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, deadline, alpha, topk, step0, beta):
    n = len(h)
    c5c = _eval(h)
    d = 0.002 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _nesterov(hn, 400, step0, deadline, alpha, topk, beta)
    if c5n < c5c:
        return hn2, c5n
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):

    # Phase 2: Nesterov with multiple configs
