import time

def _repair(h, n):

        return np.full(n, 0.5)

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

def _shift_grad(h, k_idx, n):

    sh = (n - 1) - k_idx
    g = np.zeros(n)
    if sh >= 0:
        if sh == 0:

def _slsqp(h, alpha, maxiter, top_k):

    bnds = [(0.0, 1.0)] * n
    target = n * 0.5

        idx = np.argsort(c)[-top_k:]

        m = ct.max()

    def _grad(x):

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _shift_grad(h, k_idx, n):
    sh = (n - 1) - k_idx
    g = np.zeros(n)
    if sh >= 0:
        j1 = np.arange(sh, n)
        g[j1] += (1.0 - h[j1 - sh])
        j2 = np.arange(0, n - sh)
        g[j2] -= h[j2 + sh]
    else:
        sp = -sh
        j1 = np.arange(0, n - sp)
        g[j1] += (1.0 - h[j1 + sp])
        j2 = np.arange(sp, n)
        g[j2] -= h[j2 - sp]
    return g

def _mgd_pow(h, n_iter, p, top_k, step0):
    n = len(h)
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        c = _eval_vec(h)
        idx = np.argsort(c)[-top_k:]
        c_top = np.maximum(c[idx], 1e-15)
        s_p = (c_top ** p).sum()
        if s_p < 1e-30:
            break
        factor = s_p ** (1.0 / p - 1.0) / p
        dx = 2.0 / n
        g = np.zeros(n)
        for ii in range(len(idx)):
            ci = c_top[ii]
            w = factor * ci ** (p - 1) * dx
            g += w * _shift_grad(h, idx[ii], n)
        h_try = _repair(h - step * g, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try

            step *= 0.5

    return _repair(h, n), cur_c

    best_h = h.copy()

    for it in range(n_iter):
        c = _eval_vec(h)
        idx = np.argsort(c)[-4:]
        best_dc = 0
        bi, bj = 0, 1
        for ki in idx:
            gk = _shift_grad(h, ki, n)
            i = np.argmax(gk)
            j = np.argmin(gk)
            if i == j:
                continue

                bi, bj = i, j

    bnds = [(0.0, 1.0)] * n

        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

        c = _eval_vec(x)

        return np.ones(n)
