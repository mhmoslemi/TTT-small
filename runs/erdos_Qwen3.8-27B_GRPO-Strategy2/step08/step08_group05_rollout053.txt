        idx2 = i - s

        a = a - np.where(valid2, h[idx2[valid2]], 0.0)

        h[i] = x_new

            if 0 <= i - s_val < n:

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _solve_1d(a_m, b_m):
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        vals = a_m * mid + b_m
        f_mid = vals.max()
        active = vals >= f_mid - 1e-11
        if active.any():
            aa = a_m[active]
            if aa.min() > 1e-9:
                hi = mid
            elif aa.max() < -1e-9:
                lo = mid
            else:
                break
        else:
            hi = mid
    return (lo + hi) * 0.5

def _coord_pass(h, n, s_all, c, lam):
    for i in range(n):
        idx1 = i + s_all
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - s_all
        v2 = (idx2 >= 0) & (idx2 < n)
        a = np.zeros(2 * n - 1)
        a[v1] = 1.0 - h[idx1[v1]]
        a[v2] -= h[idx2[v2]]
        a += lam
        c_s = c[s_all + (n - 1)]
        b = c_s - (a - lam) * h[i]
        mask = np.abs(a) > 1e-15
        if not mask.any():
            continue
        x_new = _solve_1d(a[mask], b[mask])
        delta = x_new - h[i]
        if abs(delta) < 1e-13:
            continue
        h[i] = x_new
        d = np.zeros(2 * n - 1)
        d[v1] += delta * (1.0 - h[idx1[v1]])
        d[v2] += -delta * h[idx2[v2]]
        c += d
    return h, c

def _cd_block(h, n, n_passes, deadline, lam0=0.0):
    s_all = np.arange(-(n - 1), n)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    best_h = h.copy()
    best_c5 = float(np.max(c))
    lam = lam0
    target = n / 2.0
    for p in range(n_passes):
        if time.time() > deadline:
            break
        h, c = _coord_pass(h, n, s_all, c, lam)
        cur_sum = h.sum()
        lam += 0.005 * (cur_sum - target) / n
        lam = np.clip(lam, -0.5, 0.5)
        if p % 20 == 0:
            cv = float(np.max(c))
            if cv < best_c5:
                best_c5 = cv
                best_h = _repair(h.copy(), n)
    return best_h, best_c5

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _lp_refine(h, n, n_steps, eps, m_top, deadline):
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    dx = 2.0 / n
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = np.correlate(c, 1.0 - c, mode='full') * dx
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        improved = False
        for step_scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = _repair(c + step_scale * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
        if not improved:
            rng2 = np.random.RandomState(int(time.time() * 1e4) % 99999)
            d2 = rng2.randn(n)
            d2 -= d2.mean()
            d2 *= eps / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _fine_polish(h, n, deadline, max_iters=200000):
    rng = np.random.RandomState(12345)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sizes = [5e-7, 2e-7, 1e-7, 5e-8]
    per_size = max_iters // len(sizes)
    for ss in sizes:
        for _ in range(per_size):
            if time.time() > deadline:
                break
            i = rng.randint(0, n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5c < bc:
                        bc, bh = c5c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc
