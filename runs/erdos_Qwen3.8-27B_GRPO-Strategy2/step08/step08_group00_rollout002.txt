            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _c_sum_fun(x, half_n):
    return x.sum() - half_n

def _c_sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_sum_fun, 'jac': _c_sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, nsteps, step0):
    n = len(h)
    cur_c = _eval(h)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    for _ in range(nsteps):
        g = _smooth_grad(h, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(12):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cur_c - 1e-15:
                h = hn
                cur_c = cn
                step = min(step * 1.1, step0 * 10.0)
                taken = True
                break
            step *= 0.35
        if not taken:
            step = min(step * 1.2, step0 * 10.0)
            v *= 0.5
    return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _subspace_opt(h, n, k_vars, alpha, rng):
    c = _eval(h)
    g = _smooth_grad(h, alpha, 80)
    idx = np.argsort(np.abs(g))[-k_vars:]
    idx = np.sort(idx)
    sub = h[idx].copy()
    sub_sum = sub.sum()
    bnds = [(0.0, 1.0)] * k_vars
    cons = [{'type': 'eq', 'fun': _c_sum_fun, 'jac': _c_sum_jac, 'args': (sub_sum,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, 80), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 80, 'ftol': 1e-16})
        x = res.x
        h[idx] += _repair(x[idx], k_vars) - sub
        h = _repair(h, n)
    except Exception:
        pass
    return h, _eval(h)

def _local_search(h, n, ntrials, rng, base_step, n_pert):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(ntrials):
        step = base_step * (0.3 + 0.7 * rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        hn = h.copy()
        hn[idx] += d
        if hn.min() < -0.1 or hn.max() > 1.1:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < c5cur - 1e-15:
            c5cur = cn
            h = hn
            if cn < best_c:
                best_c = cn
                best_h = hn.copy()
    return best_h, best_c

def _two_point_polish(h, n, ntrials, rng, step):
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    for _ in range(ntrials):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5c - 1e-15:
                c5c = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _try_n_points(h0, n_candidates, budget_s, t0):
    n = len(h0)
    best = (h0.copy(), _eval(h0), n)
    for nc in n_candidates:
        if time.time() - t0 > budget_s - 30:
            break
        if nc == n:
            continue
        ratio = n / nc if n > nc else nc // n
        if n > nc:
            hc = _downsample(h0, ratio)
        else:
            hc = _upsample(h0, nc // n)
        alpha_seq = [5000, 50000, 500000]
        for a in alpha_seq:
            if time.time() - t0 > budget_s - 20:
                break
            hc = _slsqp(hc, a, 50, 300, 1e-15)
        hc = _proj_grad(hc, 100000, 80, 200, 0.003)
        cn = _eval(hc)
        if cn < best[1]:
            best = (hc.copy(), cn, nc)
    return best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def _el():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse optimization via downsampling
    for ratio in [4, 8]:
        if _el() > 80:
            break
        nc = max(40, n // ratio)
        hc = _downsample(h, n // nc)
        for a in [1000, 10000, 100000]:
            if _el() > 100:
                break
            hc = _slsqp(hc, a, 40, 400, 1e-14)
        hu = _upsample(hc, n // nc)
        cu = _eval(hu)
        if cu < best_c5:
            best_c5 = cu
            best_h = hu.copy()

    # Phase 2: Full-res SLSQP alpha sweep
    alpha_seq = [5000, 20000, 100000, 500000, 2000000, 8000000, 30000000]
    for a in alpha_seq:
        if _el() > 300:
            break
        ml = min(100, max(20, int(20 * np.log10(a / 5000) + 20)))
        h_new = _slsqp(best_h, a, ml, 800, 1e-16)
        cn = _eval(h_new)
        if cn < best_c5:
            best_c5 = cn
            best_h = h_new.copy()

    # Phase 3: Projected gradient with multiple alphas
    grad_configs = [
        (10000, 60, 1000, 0.008),
        (50000, 70, 800, 0.005),
        (200000, 90, 700, 0.003),
        (1000000, 120, 600, 0.0015),
        (5000000, 150, 500, 0.0008),
    ]
    for a, ml, ns, s0 in grad_configs:
        if _el() > 500:
            break
        h_new = _proj_grad(best_h.copy(), a, ml, ns, s0)
        cn = _eval(h_new)
        if cn < best_c5:
            best_c5 = cn
            best_h = h_new.copy()

    # Phase 4: Subspace SLSQP iterations
    if _el() < budget_s - 250:
        for _round in range(40):
            if _el() > budget_s - 250:
                break
            a = 50000 * (_round + 1)
            h_new, cn = _subspace_opt(best_h, n, 15, a, rng)
            if cn < best_c5 - 1e-15:
                best_c5 = cn
                best_h = h_new.copy()

    # Phase 5: High-alpha SLSQP
    if _el() < budget_s - 150:
        for a in [100000000, 500000000, 2000000000]:
            if _el() > budget_s - 100:
                break
            h_new = _slsqp(best_h, a, 150, 600, 1e-17)
            cn = _eval(h_new)
            if cn < best_c5:
                best_c5 = cn
                best_h = h_new.copy()

    # Phase 6: Multi-strategy local search
    if _el() < budget_s - 60:
        remaining = budget_s - 60 - _el()
        ntrials = min(int(remaining / 0.003), 300000)
        best_h, best_c5 = _local_search(best_h, n, ntrials, rng, 5e-6, 4)

    # Phase 7: Fine 2-point polish
    if _el() < budget_s - 3:
        remaining = budget_s - 3 - _el()
        ntrials = min(int(remaining / 0.0008), 500000)
        best_h, best_c5 = _two_point_polish(best_h, n, ntrials, rng, 2e-7)

    # Phase 8: Try slightly different n
    if _el() < budget_s - 5:
        best_alt = _try_n_points(best_h, [n - 40, n + 40, n + 80], budget_s - 5, t0)
        if best_alt[1] < best_c5:
            best_h, best_c5 = best_alt[0], best_alt[1]

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
