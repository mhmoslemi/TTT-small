import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h + mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

class _Ctx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        dx2 = 2.0 / self.n
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * dx2
            if sh >= 0:
                g[:self.n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n-sp])
                g[:self.n-sp] -= wk * x[sp:]
        return g
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _Ctx(alpha, n, target)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _topk_grad(h, k_top, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = (2.0 / n) / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:

            step = min(step * 1.03, step0 * 20.0)

                v = np.zeros(n)
                step = step0 * 0.5

    return h
