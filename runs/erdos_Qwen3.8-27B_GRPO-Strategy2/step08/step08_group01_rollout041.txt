            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w[ii] * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w[ii] * h[i2 + sp]
    return g * dx

def _grad_top1(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    ki = np.argmax(c)
    sh = (n - 1) - ki
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _pgd(h, n_iter, step0, alpha, top_k):
    n = len(h)
    target = n * 0.5
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        h_try = _proj_hinge(h - step * g, target)
        c_try = _eval(h_try)
        if c_try < cur_c - 1e-15:
            h = h_try
            cur_c = c_try
            step = min(step * 1.02, step0 * 8.0)
        else:
            step *= 0.55
            step = max(step, step0 * 0.01)
    return _repair(h, n), cur_c

def _top1_pgd(h, n_iter, step0):
    n = len(h)
    target = n * 0.5
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _grad_top1(h)
        h_try = _proj_hinge(h - step * g, target)
        c_try = _eval(h_try)
        if c_try < cur_c - 1e-15:
            h = h_try
            cur_c = c_try
            step = min(step * 1.01, step0 * 5.0)
        else:
            step *= 0.65
    return _repair(h, n), cur_c

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.exp(s).sum() / alpha

    def _grad(x):
        return _grad_lse(x, alpha, top_k)

    def _cons_f(x):
        return x.sum() - target

    def _cons_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return h

def _local2(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    stagnant = 0
    max_stag = max(200, n_iter // 50)
    for _ in range(n_iter):
        if stagnant > max_stag:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 11)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c - 1e-15:
                cur_c = c_new
                stagnant = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                stagnant += 1
    return best_h, best_c

def _local3(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    stagnant = 0
    max_stag = max(300, n_iter // 40)
    for _ in range(n_iter):
        if stagnant > max_stag:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        k = (i + 5 + rng.randint(0, 9)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            c_new = _eval(h)
            if c_new < cur_c - 1e-15:
                cur_c = c_new
                stagnant = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = s0, s1, s2
                stagnant += 1
    return best_h, best_c

def _sweep2(h, n_passes, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, 5)) % n
            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < cur_c - 1e-15:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: PGD with increasing focus on top lags
    pgd_configs = [
        (2000, 600, 8, 0.003),
        (10000, 500, 6, 0.002),
        (50000, 400, 5, 0.001),
        (200000, 300, 4, 0.0005),
        (500000, 200, 3, 0.0003),
    ]
    for alpha, steps, tk, s0 in pgd_configs:
        if _el() > 250:
            break
        h_t, c = _pgd(best_h.copy(), steps, s0, alpha, tk)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: Top-1 PGD
    if _el() < budget_s - 250:
        for s0 in [0.001, 0.0005, 0.0002]:
            if _el() > budget_s - 250:
                break
            h_t, c = _top1_pgd(best_h.copy(), 800, s0)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 3: SLSQP
    if _el() < budget_s - 450:
        slsqp_cfgs = [
            (5000, 10, 1000),
            (20000, 6, 800),
            (100000, 4, 600),
            (300000, 3, 400),
        ]
        for alpha, tk, mxi in slsqp_cfgs:
            if _el() > budget_s - 400:
                break
            h_t = _slsqp(best_h.copy(), alpha, tk, mxi)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
            h_t, c = _top1_pgd(h_t, 300, 0.0005)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 4: Multi-start perturbation
    if _el() < budget_s - 500:
        for trial in range(5):
            if _el() > budget_s - 450:
                break
            scale = 0.015 * (trial + 1)
            noise = scale * rng.randn(n)
            h_pert = _repair(best_h + noise, n)
            h_t, c = _pgd(h_pert, 400, 0.002, 50000, 5)
            c = _eval(h_t)
            if c < best_c5:
                h_t, c = _top1_pgd(h_t, 500, 0.0005)
                c = _eval(h_t)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_t.copy()

    # Phase 5: Local search 2-point (coarse)
    if _el() < budget_s - 200:
        n_iter = min(int((budget_s - 200 - _el()) * 1000), 120000)
        best_h, best_c5 = _local2(best_h, n_iter, 8e-5, rng)

    # Phase 6: Top-1 PGD after local
    if _el() < budget_s - 150:
        h_t, c = _top1_pgd(best_h, 1000, 0.0003)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 7: Local search 3-point
    if _el() < budget_s - 120:
        n_iter = min(int((budget_s - 120 - _el()) * 700), 80000)
        best_h, best_c5 = _local3(best_h, n_iter, 4e-5, rng)

    # Phase 8: Sweep-based 2-point
    if _el() < budget_s - 80:
        npass = min(int((budget_s - 80 - _el()) * 200), 3000)
        best_h, best_c5 = _sweep2(best_h, npass, 5e-5, rng)

    # Phase 9: Top-1 PGD fine
    if _el() < budget_s - 50:
        h_t, c = _top1_pgd(best_h, 2000, 0.0002)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 10: Ultra-fine local search
    if _el() < budget_s - 20:
        n_iter = min(int((budget_s - 20 - _el()) * 3000), 200000)
        best_h, best_c5 = _local2(best_h, n_iter, 2e-5, rng)

    # Phase 11: Final Top-1
    if _el() < budget_s - 5:
        h_t, c = _top1_pgd(best_h, 1000, 0.0001)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t

    # Phase 12: Micro local
    if _el() < budget_s - 2:
        n_iter = min(int((budget_s - 2 - _el()) * 5000), 300000)
        best_h, best_c5 = _local2(best_h, n_iter, 5e-6, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
