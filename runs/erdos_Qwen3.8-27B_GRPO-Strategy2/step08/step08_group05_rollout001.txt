def _grad(h):
    n=len(h); L=2*n-1; off=n-1
    U=np.zeros((n,L))
    for r in range(n):
        row=U[r]
        row[off:off+r+1] += (1.0-h[:r+1])[::-1]
        if r+1<n:

        row[off:off+n-r] += -h[r:]
    return U

import numpy as np
from scipy.optimize import linprog

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _eval(h):
    return float(np.max(_corr(h)) * (2.0 / len(h)))

def _repair(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    t = 0.5 * n
    s = float(h.sum())
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-10:
        h *= t / max(s, 1e-300)
        h = np.clip(h, 0.0, 1.0)
        d = t - float(h.sum())
        if abs(d) > 1e-12:
            o = np.argsort(1.0 - h if d > 0 else h)
            for i in o:
                if d > 0:
                    a = min(d, 1.0 - h[i])
                    h[i] += a
                    d -= a
                else:
                    a = min(-d, h[i])
                    h[i] -= a
                    d += a
                if abs(d) < 1e-12:
                    break
    return h

def _grad(h):
    n = len(h)
    L = 2 * n - 1
    off = n - 1
    U = np.zeros((n, L), dtype=float)
    for r in range(n):
        row = U[r]
        row[off:off + r + 1] += (1.0 - h[:r + 1])[::-1]
        if r + 1 < n:
            row[r:off] += (1.0 - h[r + 1:])[::-1]
        if r > 0:
            row[off - r:off] += -h[:r]
        row[off:off + n - r] += -h[r:]
    return U

def _weights(c, alpha, topk):
    L = len(c)
    if int(topk) >= L:
        idx = np.arange(L)
    else:
        idx = np.argsort(c)[-int(topk):]
    ct = c[idx]
    e = np.exp(np.clip(alpha * (ct - ct.max()), -700.0, 700.0))
    ss = float(e.sum())
    if ss <= 1e-300:
        e = np.zeros_like(e)
        e[0] = 1.0
        ss = 1.0
    e /= ss
    w = np.zeros(L, dtype=float)
    w[idx] = e
    return w

def _line(h, c, U, d, cap):
    n = len(h)
    d = np.asarray(d, float).copy()
    if d.shape[0] != n or cap <= 1e-14:
        return None
    d -= d.mean()
    sc = float(np.max(np.abs(d)))
    if sc < 1e-14:
        return None
    dn = d / sc
    amax = float(cap)
    pos = dn > 1e-10

    neg = dn < -1e-10
    if neg.any():
        amax = min(amax, float(np.min(h[neg] / (-dn[neg]))))
    if amax <= 1e-13:

    Lv = U.T @ dn
    Rv = np.correlate(dn, dn, 'full')
    K = 33
    ts = np.linspace(0.0, amax, K)
    vals = c[None, :] + ts[:, None] * Lv[None, :]
    vals -= (ts[:, None] * ts[:, None]) * Rv[None, :]
    f = vals.max(axis=1)
    bi = int(np.argmin(f))
    bt = float(ts[bi])
    bf = float(f[bi])
    if 0 < bi < K - 1:
        lo = float(ts[bi - 1])
        hi = float(ts[bi + 1])
        for _ in range(26):

            m2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(c + m1 * Lv - (m1 * m1) * Rv))
            f2 = float(np.max(c + m2 * Lv - (m2 * m2) * Rv))
            if f1 < f2:

            else:

        t = 0.5 * (lo + hi)

def _line_pair(c, lam, i0, i1, i2, tmax, refine=True):
    if tmax <= 1e-12:
        return None
    K = 13
    ts = np.linspace(0.0, tmax, K)
    vals = c[None, :] + ts[:, None] * lam[None, :]
    t2 = ts * ts
    vals[:, i0] -= 2.0 * t2
    vals[:, i1] += t2
    vals[:, i2] += t2
    f = vals.max(axis=1)
    bi = int(np.argmin(f))
    bt = float(ts[bi])
    bf = float(f[bi])
    cur = float(np.max(c))
    if refine and 0 < bi < K - 1:
        lo = float(ts[bi - 1])
        hi = float(ts[bi + 1])
        for _ in range(24):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = c + m1 * lam
            v1[i0] -= 2.0 * m1 * m1
            v1[i1] += m1 * m1
            v1[i2] += m1 * m1
            v2 = c + m2 * lam
            v2[i0] -= 2.0 * m2 * m2
            v2[i1] += m2 * m2
            v2[i2] += m2 * m2
            f1 = float(v1.max())
            f2 = float(v2.max())
            if f1 < f2:
                hi = m1
            else:
                lo = m2
        t = 0.5 * (lo + hi)
        v = c + t * lam
        v[i0] -= 2.0 * t * t
        v[i1] += t * t
        v[i2] += t * t
        ft = float(v.max())
        if ft < bf:
            bt, bf = t, ft
    if bt <= 1e-13 or bf >= cur - 1e-13:
        return None
    return bt, bf

def _descend(h, alpha, topk, cap, steps, dl):
    n = len(h)
    h = np.array(h, float).copy()
    c = _corr(h)
    U = _grad(h)
    bh = h.copy()
    bf = float(c.max())
    for _ in range(steps):
        if time.time() >= dl:
            break
        pf = float(c.max())
        w = _weights(c, alpha, topk)
        if float(w.sum()) <= 0.0:
            break
        G = U.T @ w
        G -= G.mean()
        if float(np.max(np.abs(G))) < 1e-14:
            break
        res = _line(h, c, U, -G, cap)
        if res is None:
            break
        h2, c2, _ = res
        h2 = _repair(h2, n)
        c2 = _corr(h2)
        f2 = float(c2.max())
        if f2 < pf - 1e-13:
            h, c = h2, c2
            U = _grad(h)
            if f2 < bf:
                bf, bh = f2, h.copy()
        else:
            break
    return bh, bf

def _lp_dir(U, c, n, mode, k1, k2, eps, alpha):
    L = 2 * n - 1
    m = float(c.max())
    if mode == 0:
        if k1 <= 0 or k1 >= L:
            act = np.arange(L)
        else:
            act = np.argsort(c)[-int(k1):]
        G = U.T[act]
        A = np.empty((len(act), n + 1), dtype=float)
        A[:, :n] = G
        A[:, n] = -1.0
        b = np.maximum(m - c[act], 0.0)
        z = np.zeros(n + 1)
        z[n] = -1.0
        Aeq = np.ones((1, n + 1))
        Aeq[0, n] = 0.0
        bnds = [(-eps, eps)] * n
        bnds.append((0.0, None))
        try:
            r = linprog(z, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.zeros(1),
                        bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            return None
        if r is None or not r.success or r.x is None:
            return None
        if float(r.x[n]) <= 1e-11:
            return None
        d = r.x[:-1].copy()
        d -= d.mean()
        return d if float(np.max(np.abs(d))) > 1e-14 else None
    else:
        if k1 <= 0:
            io = np.arange(L)
        else:
            io = np.argsort(c)[-int(k1):]
        ct = c[io]
        e = np.exp(np.clip(alpha * (ct - ct.max()), -700.0, 700.0))
        ss = float(e.sum())
        if ss <= 1e-300:
            return None
        e /= ss
        og = e @ U.T[io]
        if float(np.max(np.abs(og))) < 1e-14:
            return None
        if k2 <= 0:
            ig = np.arange(L)
        else:
            ig = np.argsort(c)[-int(k2):]
        A = U.T[ig]
        b = np.zeros(len(ig))
        Aeq = np.ones((1, n))
        bnds = [(-eps, eps)] * n
        try:
            r = linprog(-og, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=np.zeros(1),
                        bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            return None
        if r is None or not r.success or r.x is None:
            return None
        if float(r.fun) >= -1e-10:
            return None
        d = r.x.copy()
        d -= d.mean()
        return d if float(np.max(np.abs(d))) > 1e-14 else None

def _lp(h, mode, k1, k2, eps, alpha, iters, dl):
    n = len(h)
    h = np.array(h, float).copy()
    c = _corr(h)
    U = _grad(h)
    bh = h.copy()
    bf = float(c.max())
    fail = 0
    for _ in range(iters):
        if time.time() >= dl:
            break
        pf = float(c.max())
        d = _lp_dir(U, c, n, mode, k1, k2, eps, alpha)
        if d is None:
            fail += 1
            if fail > 2:
                break
            continue
        res = _line(h, c, U, d, eps)
        if res is None:
            fail += 1
            if fail > 2:
                break
            continue
        h2, c2, _ = res
        h2 = _repair(h2, n)
        c2 = _corr(h2)
        f2 = float(c2.max())
        if f2 < pf - 1e-13:
            h, c = h2, c2
            U = _grad(h)
            fail = 0
            if f2 < bf:
                bf, bh = f2, h.copy()
        else:
            fail += 1
            if fail > 2:
                break
    return bh, bf

def _pair(h, dl, rng):
    n = len(h)
    off = n - 1
    h = np.array(h, float).copy()
    c = _corr(h)
    U = _grad(h)
    bh = h.copy()
    bf = float(c.max())
    fail = 0
    B = 50
    NM = 1800
    while time.time() < dl:
        pf = float(c.max())
        w = _weights(c, 2000.0, 35)
        G = U.T @ w
        G -= G.mean()
        low = np.argsort(G)[:B]
        high = np.argsort(G)[::-1][:B]
        cand = []
        for ii in low:
            ii = int(ii)
            gi = G[ii]
            am = 1.0 - h[ii]
            if am <= 1e-9:
                continue
            for jj in high:
                jj = int(jj)
                if jj == ii:
                    continue
                sc = G[jj] - gi
                if sc > 1e-10:
                    tm = min(am, h[jj])
                    if tm > 1e-9:
                        cand.append((sc, tm, ii, jj))
        best = None
        if cand:
            cand.sort(reverse=True)
            for sc, tm, i, j in cand[:NM]:
                if time.time() >= dl:
                    break
                res = _line_pair(c, U[i] - U[j], off, off + i - j, off + j - i, tm, False)
                if res is not None and (best is None or res[1] < best[1]):
                    best = (res[0], res[1], int(i), int(j), float(tm))
        if best is None:
            for _ in range(250):
                if time.time() >= dl:
                    break
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                tm = min(1.0 - h[i], h[j])
                if tm <= 1e-8:
                    tm = min(h[i], 1.0 - h[j])
                    if tm <= 1e-8:
                        continue
                    i, j = j, i
                res = _line_pair(c, U[i] - U[j], off, off + i - j, off + j - i, tm, False)
                if res is not None and (best is None or res[1] < best[1]):
                    best = (res[0], res[1], int(i), int(j), float(tm))
        if best is None:
            fail += 1
            if fail > 6:
                break
            continue
        t, _, i, j, tm = best
        rf = _line_pair(c, U[i] - U[j], off, off + i - j, off + j - i, tm, True)
        if rf is not None:
            t = rf[0]
        old_h = h.copy()
        old_c = c.copy()
        h[i] += t
        h[j] -= t
        if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
            h = _repair(h, n)
        c = _corr(h)
        nf = float(c.max())
        if nf < pf - 1e-13:
            U = _grad(h)
            fail = 0
            if nf < bf:
                bf, bh = nf, h.copy()
        else:
            h, c = old_h, old_c
            fail += 1
            if fail > 8:
                break
    return bh, bf

def _rand(h, dl, rng, k):
    n = len(h)
    h = np.array(h, float).copy()
    c = _corr(h)
    U = _grad(h)
    bh = h.copy()
    bf = float(c.max())
    caps = (0.01, 0.03, 0.1)
    for _ in range(k):
        if time.time() >= dl:
            break
        d = rng.randn(n)
        d -= d.mean()
        sc = float(np.max(np.abs(d)))
        if sc < 1e-14:
            continue
        d /= sc
        cap = caps[int(rng.randint(len(caps)))]
        res = _line(h, c, U, d, cap)
        if res is None:
            continue
        h2, c2, _ = res
        pf = float(np.max(c))
        h2 = _repair(h2, n)
        c2 = _corr(h2)
        f2 = float(c2.max())
        if f2 < pf - 1e-13:
            h, c = h2, c2
            U = _grad(h)
            if f2 < bf:
                bf, bh = f2, h.copy()
    return bh, bf

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.asarray(initial_h_values, float)
    except Exception:
        h0 = None
    if h0 is None or len(h0) == 0:
        x = np.arange(560)
        h0 = np.clip(0.5 + 0.45 * np.sin(2.0 * np.pi * x / 11.0) + 0.25 * np.sin(2.0 * np.pi * x / 7.0 + 0.7), 0.0, 1.0)
    n = len(h0)
    h = _repair(h0.copy(), n)
    best = _eval(h)

    def _take(hv):
        nonlocal h, best
        if hv is None:
            return
        hv = _repair(hv, n)
        v = float(_eval(hv))
        if v <= best + 1e-15:
            h = hv
            if v < best:
                best = v

    t1 = min(deadline, t0 + 0.20 * budget)
    for a, tk, cap, st in ((800, 60, 0.5, 400), (3000, 30, 0.3, 350),
                           (12000, 15, 0.15, 300), (40000, 8, 0.08, 300),
                           (120000, 4, 0.04, 250)):
        if time.time() >= t1:
            break
        hv, _ = _descend(h, a, tk, cap, st, t1)
        _take(hv)

    t2 = min(deadline, t0 + 0.50 * budget)
    for mode, k1, k2, eps, alpha, it in (
        (0, 30, 0, 0.05, 0, 80),
        (0, 80, 0, 0.035, 0, 80),
        (0, 160, 0, 0.025, 0, 60),
        (1, 20, 120, 0.05, 2000, 100),
        (1, 40, 220, 0.035, 4000, 100),
        (0, 320, 0, 0.02, 0, 35),
        (1, 80, 512, 0.025, 8000, 70)
    ):
        if time.time() >= t2:
            break
        hv, _ = _lp(h, mode, k1, k2, eps, alpha, it, t2)
        _take(hv)

    t3 = min(deadline, t0 + 0.64 * budget)
    for a, tk, cap, st in ((2000, 25, 0.25, 300), (8000, 12, 0.12, 250), (30000, 6, 0.06, 250)):
        if time.time() >= t3:
            break
        hv, _ = _descend(h, a, tk, cap, st, t3)
        _take(hv)

    t4 = min(deadline, t0 + 0.87 * budget)
    hv, _ = _pair(h, t4, rng)
    _take(hv)

    if deadline - time.time() > 130:
        for pp in (0.004, 0.009, 0.016):
            if deadline - time.time() < 90:
                break
            hp = _repair(h + pp * rng.randn(n), n)
            dl1 = min(deadline, time.time() + 25)
            hv, _ = _descend(hp, 6000, 10, 0.08, 180, dl1)
            hv = _repair(hv, n)
            tmp_best = best
            if float(_eval(hv)) <= tmp_best + 1e-15:
                h = hv
                if float(_eval(hv)) < best:
                    best = float(_eval(hv))
            else:
                hv = hp
            dl2 = min(deadline, time.time() + 25)
            hv2, _ = _pair(hv, dl2, rng)
            _take(hv2)

    if time.time() < deadline:
        dl = min(deadline, time.time() + 35)
        hv, _ = _descend(h, 50000, 5, 0.03, 180, dl)
        _take(hv)

    if time.time() < deadline:
        dl = min(deadline, time.time() + 30)
        hv, _ = _rand(h, dl, rng, 250)
        _take(hv)

    if time.time() < deadline:
        hv, _ = _pair(h, deadline, rng)
        _take(hv)

    h = _repair(h, n)
    best = float(_eval(h))
    return h, best, n
