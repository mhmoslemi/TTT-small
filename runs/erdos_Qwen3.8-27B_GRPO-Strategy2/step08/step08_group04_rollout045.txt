        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        h = np.clip(h, 0.0, 1.0)

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

                    h[idx] += a
                    diff -= a
                else:

    """Gradient of correlation at full index idx."""
    n = len(h)

        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g * dx
