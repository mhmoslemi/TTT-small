def _smax_obj(h, alpha, topk):

    m = c.max()

        idx = np.argpartition(c, -topk)[-topk:]

    else:

    s = np.clip(alpha * (ct - m), -300, 300)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):

    else:

    ct = c[idx]
    s = np.clip(alpha * (ct - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n

    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Quick SLSQP polish (40s)
    dl1 = min(deadline, t0 + 40)
    if time.time() < dl1:
        try:
            def _obj(x):
                c = _correl(x)
                m = c.max()
                s = np.clip(3000 * (c - m), -200, 200)
                return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / 3000
            def _jac(x):
                return _grad_topk(x, 3000, 30)
            r = minimize(_obj, h.copy(), jac=_jac, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                       'jac': lambda x: np.ones(n)}],
                         options={'maxiter': 80, 'ftol': 1e-18})
            hv = _repair(r.x, n)
            cv = _eval(hv)
            if cv < best_c5:
                best_c5, best_h = cv, hv.copy()
        except Exception:
            pass

    # Phase 2: LP descent - main workhorse (320s)
    dl2 = min(deadline, t0 + 360)
    for eps, mt, ns in [(0.006, 8, 12000), (0.004, 10, 8000)]:
        if time.time() > dl2:
            break
        hv, cv = _lp_desc(best_h.copy(), ns, eps, mt, dl2)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov (160s)
    dl3 = min(deadline, t0 + 520)
    for alpha, tk, st, ns in [(8000, 25, 0.003, 3000), (40000, 12, 0.002, 3000), (200000, 8, 0.001, 2000)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), alpha, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: Fine LP descent (240s)
    dl4 = min(deadline, t0 + 760)
    for eps, mt, ns in [(0.002, 10, 15000), (0.001, 12, 15000)]:
        if time.time() > dl4:
            break
        hv, cv = _lp_desc(best_h.copy(), ns, eps, mt, dl4)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 5: Block annealing (100s)
    dl5 = min(deadline, t0 + 860)
    blocks = [1, 2, 3, 4, 5, 6, 8, 12]
    hv, cv = _block_anneal(best_h.copy(), 40000, rng, dl5, 0.004, 1e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        # Quick LP refinement after anneal
        if time.time() < dl5:
            hv2, cv2 = _lp_desc(best_h.copy(), 5000, 0.001, 8, dl5)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    # Phase 6: Coordinate polish (remaining)
    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.0006), 60000)
    best_h, best_c5 = _coord_polish(best_h, npol, rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
