                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - tol:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                    break
                else:
                    h[i], h[j] = ni - signs[0]*step, nj - signs[1]*step
    return bh, bc

def _lp_step(h, m_top, eps, deadline):
    """Single LP descent step. Returns (new_h, improved)."""
    n = len(h)
    cor = _correl(h)
    idx_top = np.argpartition(cor, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-cor[idx_top])]
    grads = np.array([_grad_single(h, k, n) for k in idx_top])
    c_obj = -grads[0]
    n_constraints = m_top - 1
    if n_constraints > 0:
        A_ub = np.array(grads[1:])
        b_ub = np.zeros(n_constraints)
    else:
        A_ub = None
        b_ub = None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True})
    except Exception:
        return h, False
    if not res.success or res.x is None:
        return h, False
    d = res.x
    if np.max(np.abs(d)) < 1e-14:
        return h, False
    c5_cur = _eval(h)
    step = 1.0
    for _ in range(10):
        h_new = _repair(h + step * d, n)
        c5_new = _eval(h_new)
        if c5_new < c5_cur - 1e-15:
            return h_new, True
        step *= 0.5
    return h, False

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    stall = 0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c5_cur = _eval(c)
        h2, improved = _lp_step(c, m_top, eps, deadline)
        if improved:
            c = h2
            stall = 0
            if c5_cur < best_c5:  # note: c5_cur is before the step
                pass
            c5_new = _eval(c)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, c.copy()
        else:
            stall += 1

                d = rng2.randn(n) * eps * 0.5
                d -= d.mean()

                stall = 0
    return best_h, best_c5

def _soft_max_loss(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _soft_max_grad(h, alpha, topk, n):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k, wk = idx[ii], w[ii] * dx
        g += wk * _grad_single(h, k, n)
    return g

def _nesterov(h, alpha, topk, n_steps, step0, deadline):
    n = len(h)
    f = _soft_max_loss(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        g = _soft_max_grad(h, alpha, topk, n)
        if np.isnan(g).any() or np.abs(g).max() > 1e10:
            break
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _soft_max_loss(hn, alpha, topk)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj(h - step * v, n * 0.5)
            f = _soft_max_loss(h, alpha, topk)
    return _repair(h, n)

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:

        seg = h[i0:i0 + bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc = c5n; bh = h.copy()
    return bh, bc

def _pair_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        for ss in [1e-6, 5e-7, 2e-7, 1e-7, 5e-8]:
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n; bh = h.copy()
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _multi_lp(h, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    epochs = [
        (2500, 0.005, 8),
        (3000, 0.002, 6),
        (4000, 0.0008, 5),
        (3000, 0.0003, 4),
    ]
    for steps, eps, mk in epochs:
        if time.time() > deadline:
            break
        hv, cv = _lp_descent(best_h.copy(), steps, eps, mk, deadline)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()
        else:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Nesterov soft-max (0 - 100s)
    dl1 = min(deadline, t0 + 100)
    for a, tk, ns, st0 in [(8000, 25, 2500, 0.004), (50000, 12, 2000, 0.002), (200000, 6, 1500, 0.001)]:
        if time.time() > dl1:
            break
        hv = _nesterov_sm(best_h.copy(), a, tk, ns, st0, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: Multi-epoch LP descent (100 - 400s)
    dl2 = min(deadline, t0 + 400)
    best_h, best_c5 = _multi_lp(best_h, dl2)

    # Phase 3: Nesterov again with different params (400 - 500s)
    dl3 = min(deadline, t0 + 500)
    for a, tk, ns, st0 in [(30000, 10, 1500, 0.002), (300000, 5, 1000, 0.0005)]:
        if time.time() > dl3:
            break
        hv = _nesterov_sm(best_h.copy(), a, tk, ns, st0, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: Fine LP descent (500 - 700s)
    dl4 = min(deadline, t0 + 700)
    hv, cv = _lp_descent(best_h.copy(), 5000, 0.0005, 5, dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
    hv, cv = _lp_descent(best_h.copy(), 4000, 0.0002, 4, dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: Block annealing exploration (700 - 870s)
    dl5 = min(deadline, t0 + 870)
    blocks = [1, 2, 3, 4, 5, 7, 10, 15]
    hv, cv = _block_anneal(best_h.copy(), 25000, rng, dl5, 0.003, 5e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        if time.time() < dl5 - 40:
            hv2, cv2 = _lp_descent(best_h.copy(), 3000, 0.001, 5, dl5)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    # Phase 6: Second Nesterov pass on best (870 - 950s)
    dl6 = min(deadline, t0 + 950)
    hv = _nesterov_sm(best_h.copy(), 100000, 4, 1000, 0.0005, dl6)
    cv = _eval(hv)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 7: Final pair polish (950 - deadline)
    dl7 = deadline
    rem = max(0, dl7 - time.time())
    npol = int(rem / 0.0015)
    best_h, best_c5 = _pair_polish(best_h, npol, rng, dl7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
