import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _corr_full(h):
    return np.correlate(h, 1.0-h, mode='full')

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n/2.0
    for _ in range(100):
        s = h.sum()
        if abs(s-target) < 1e-10: break
        if s > 1e-12: h *= target/s
        h = np.clip(h, 0, 1)
    diff = target-h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-11: break
    return h

def _subgrad(h, n, eps=1e-6):
    """Return (max_val, subgradient) of max_k c[k]*dx"""
    dx = 2.0/n
    c = _corr_full(h)*dx
    m = c.max()
    active = np.where(c > m-eps)[0]
    g = np.zeros(n)
    for ki in active:
        k = ki-(n-1)
        if k == 0:
            g += 1.0-2.0*h
        elif k > 0:
            g[:n-k] += (1.0-h[k:])
            g[k:] -= h[:n-k]
        else:
            M = -k
            g[M:] += (1.0-h[:n-M])
            g[:n-M] -= h[M:]
    g /= max(1, len(active))
    return m, g

def _smooth_obj(x, alpha, nlags):
    n = len(x)
    dx = 2.0/n
    c = _corr_full(x)*dx
    m = c.max()
    idx = np.argsort(c)[-nlags:]
    s = np.clip(alpha*(c[idx]-m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum())/alpha

def _smooth_grad(x, alpha, nlags):
    n = len(x)
    dx = 2.0/n
    c = _corr_full(x)*dx
    m = c.max()
    idx = np.argsort(c)[-nlags:]
    s = np.clip(alpha*(c[idx]-m), -300, 300)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, ki in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii]*dx
        k = ki-(n-1)
        if k >= 0:
            g[:n-k] += wk*(1.0-x[k:])
            g[k:] -= wk*x[:n-k]
        else:
            M=-k
            g[M:] += wk*(1.0-x[:n-M])
            g[:n-M] -= wk*x[M:]
    return g

def _sum_fn(x, half): return x.sum()-half
def _sum_jac(x, half): return np.ones(len(x))

def _slsqp(h, alpha, nlags, maxiter, ftol, t0, tlim):
    n = len(h)
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':_sum_fn,'jac':_sum_jac,'args':(n*0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,nlags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_descent(h, n, steps, t0, tlim, step0=1e-4):
    v = np.zeros(n)
    mu = 0.8
    step = step0
    cur = _eval(h)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        if s > 1e-12:
            h *= target / s
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-11:
                break
    return h

def _subgrad(h, n, eps=1e-6):
    dx = 2.0 / n
    c = _corr(h) * dx
    m = c.max()
    active = np.where(c > m - eps)[0]
    g = np.zeros(n)
    for ki in active:
        k = ki - (n - 1)
        if k == 0:
            g += 1.0 - 2.0 * h
        elif k > 0:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        else:
            M = -k
            g[M:] += (1.0 - h[:n - M])
            g[:n - M] -= h[M:]
    g /= max(1, len(active))
    return m, g

def _smooth_obj(x, alpha, nl):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    idx = np.argsort(c)[-nl:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, nl):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    idx = np.argsort(c)[-nl:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, ki in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        k = ki - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - x[k:])
            g[k:] -= wk * x[:n - k]
        else:
            M = -k
            g[M:] += wk * (1.0 - x[:n - M])
            g[:n - M] -= wk * x[M:]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, nl, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, nl),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_descent(h, n, steps, t0, tlim, lr0=1e-4):

    t = 1.0
    lr = lr0
    cur = _eval(h)

    stagn = 0
    for it in range(steps):

        t_new = (1 + np.sqrt(1 + 4 * t * t)) * 0.5

                h = cand
                cur = c_new
                t = t_new

                if d < 1e-12:
                    continue
                for frac in (1.0, 0.5, 0.25, 0.125):
                    dd = d * frac
                    h[i] = hi_s + dd; h[j] = hj_s - dd
                    c5n = _eval(h)
                    if c5n < c5 - 1e-13:
                        c5 = c5n; improved = True
                        hi_s = h[i]
                        if c5 < best_c:
                            best_c = c5; best_h = h.copy()
                        break
                else:
                    h[i] = hi_s; h[j] = hj_s
                # Try reverse direction
                d2 = min(hi_s, 1.0 - hj_s)
                if d2 < 1e-12:
                    continue
                for frac in (1.0, 0.5, 0.25):
                    dd = d2 * frac
                    h[i] = hi_s - dd; h[j] = hj_s + dd
                    c5n = _eval(h)
                    if c5n < c5 - 1e-13:
                        c5 = c5n; improved = True
                        hi_s = h[i]
                        if c5 < best_c:
                            best_c = c5; best_h = h.copy()
                        break
                else:
                    h[i] = hi_s; h[j] = hj_s
        if not improved:
            break
    return best_h, best_c

def _random_polish(h, n, n_iter, t0, tlim, rng):
    c5 = _eval(h)
    best_h = h.copy(); best_c = c5
    for _ in range(n_iter):
        if time.time() - t0 > tlim:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = 5e-5 * rng.randn()
        ni = h[i] + d; nj = h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i] = ni; h[j] = nj
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i] = ni - d; h[j] = nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP moderate alpha
    for alpha in [10000, 50000, 200000]:
        if _el() > 180:
            break
        nl = min(60, max(25, int(15 + 25 * np.log10(alpha / 10000 + 1))))
        hn = _slsqp(best_h, alpha, nl, 1000, 1e-15)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    # Phase 2: Nesterov subgradient on exact max
    if _el() < budget_s - 300:
        hn, c5n = _nesterov_descent(best_h.copy(), n, 30000, t0, budget_s - 300, lr0=8e-5)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    # Phase 3: SLSQP high alpha
    for alpha in [1000000, 10000000, 100000000]:
        if _el() > budget_s - 220:
            break
        nl = min(80, max(30, int(25 + 20 * np.log10(alpha / 1000000 + 1))))
        hn = _slsqp(best_h, alpha, nl, 800, 1e-16)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    # Phase 4: Another Nesterov pass (finer)
    if _el() < budget_s - 180:
        hn, c5n = _nesterov_descent(best_h.copy(), n, 20000, t0, budget_s - 180, lr0=3e-5)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    # Phase 5: Deterministic 2-point sweep
    if _el() < budget_s - 60:
        hn, c5n = _two_point_sweep(best_h.copy(), n, t0, budget_s - 60, n_sweeps=30, n_off=12)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    # Phase 6: SLSQP final
    if _el() < budget_s - 30:
        hn = _slsqp(best_h, 500000000, 50, 500, 1e-17)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    # Phase 7: Random polish
    if _el() < budget_s - 5:
        ntr = min(int((budget_s - 5 - _el()) / 0.004), 200000)
        best_h, best_c5 = _random_polish(best_h.copy(), n, ntr, t0, budget_s - 5, rng)

    # Phase 8: Final 2-point
    if _el() < budget_s - 1:
        hn, c5n = _two_point_sweep(best_h.copy(), n, t0, budget_s - 1, n_sweeps=5, n_off=15)
        if c5n < best_c5:
            best_c5 = c5n; best_h = hn

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
