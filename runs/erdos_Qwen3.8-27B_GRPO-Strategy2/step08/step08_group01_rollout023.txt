def coord_descent_step(h, j, O):
    """Update h[j] to minimize max O(s). O is the current overlap vector."""
    n = len(h)
    hj = h[j]
    # Compute B_s for all shifts s
    # B_s = (1-h[j+s]) if 0<=j+s<n else 0, minus h[j-s] if 0<=j-s<n else 0
    B = np.zeros(2*n-1)
    # j+s < n and j+s >= 0, s ranges from -(n-1) to n-1
    # j+s >= 0 means s >= -j, j+s < n means s < n-j
    s_start = max(-(n-1), -j)  # j+s >= 0
    s_end = min(n-1, n-1-j)    # j+s < n, so s < n-j, s <= n-1-j
    for s in range(s_start, s_end+1):
        B[s+n-1] += (1 - h[j+s])
    # j-s >= 0 means s <= j, j-s < n means s > j-n
    s_start2 = max(-(n-1), j-n+1)  # j-s < n means s > j-n
    s_end2 = min(n-1, j)           # j-s >= 0 means s <= j
    for s in range(s_start2, s_end2+1):
        B[s+n-1] -= h[j-s]
    # C_s = O(s) - h[j] * B[s]
    C = O - hj * B
    # Now minimize max_s (C[s] + t*B[s]) over t in [0,1]
    # Binary search on V
    lo = C.min()  # lower bound
    hi = O.max()   # upper bound (current value at t=hj)
    for _ in range(50):
        mid = (lo + hi) / 2
        # Check if exists t in [0,1] with all C[s]+t*B[s] <= mid
        # t <= (mid - C[s]) / B[s] for B[s] > 0
        # t >= (mid - C[s]) / B[s] for B[s] < 0
        # C[s] <= mid for B[s] == 0
        t_lo = 0.0
        t_hi = 1.0
        feasible = True
        mask_pos = B > 1e-15
        mask_neg = B < -1e-15
        mask_zero = ~mask_pos & ~mask_neg
        if mask_zero.any() and C[mask_zero].max() > mid:
            feasible = False
        if feasible:
            if mask_pos.any():
                t_hi = min(t_hi, ((mid - C[mask_pos]) / B[mask_pos]).min())
            if mask_neg.any():

            if t_lo > t_hi + 1e-15:
                feasible = False
        if feasible:

        else:
            lo = mid

def compute_O(h):

    return np.correlate(h, 1.0 - h, mode='full')

def proj_sum(h, target):
    """Project h to satisfy sum(h) = target, 0 <= h <= 1."""
    lo, hi = -5.0, 5.0
    for _ in range(80):

    # Phase 3: More coordinate descent on best
    h = best_h.copy()
    O = compute_O(h)
    if elapsed() < budget_s - 100:
        for sweep in range(1500):
            if elapsed() > budget_s - 100:
                break
            order = rng.permutation(n)
            for j in order:
                t_opt, O = coord_step(h, O, j, n)
                h[j] = t_opt
            h = proj_sum(h, n * 0.5)
            if sweep % 10 == 0:
                O = compute_O(h)
            c5 = float(np.max(O) * 2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: 2-point local search (limited)
    if elapsed() < budget_s - 20:
        h = best_h.copy()
        cur_c = best_c5
        max_iter = min(int((budget_s - 20 - elapsed()) * 800), 50000)
        for it in range(max_iter):
            if elapsed() > budget_s - 20:
                break
            i = rng.randint(n)
            j2 = (i + 1 + rng.randint(0, 9)) % n
            d = 1e-4 * rng.randn()
            ni, nj = h[i] + d, h[j2] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j2] = ni, nj
                c_new = eval_c5(h)
                if c_new < cur_c:
                    cur_c = c_new
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j2] = ni - d, nj + d

    best_h = repair(best_h, n)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, n)
