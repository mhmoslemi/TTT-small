    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if abs(s - target) < 1e-12:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:

                break

                h[idx] += a
                diff -= a
            else:

                diff += a
    return h

    best_c = cur_c
    best_h = h.copy()

                c = _eval_vec(h)

                deltas = np.zeros(top_m)
                for t, k_idx in enumerate(top_idx):

                    gi = _deriv_at(h, i, k)
                    gj = _deriv_at(h, j, k)
                    deltas[t] = gi - gj

                c_top = c[top_idx]
                # Piecewise linear: find minimum of max_k (c_top[k] + d * deltas[k])
                # Breakpoints: where c_top[a] + d*deltas[a] = c_top[b] + d*deltas[b]
                # d = (c_top[b] - c_top[a]) / (deltas[a] - deltas[b])
                # Also check boundaries: d ∈ [-h[i], 1-h[i]] ∩ [h[j]-1, h[j]]
                d_lo = max(-h[i], h[j]-1)
                d_hi = min(1-h[i], h[j])

                candidates = [d_lo, d_hi, 0.0]
                for a in range(top_m):

                        denom = deltas[a] - deltas[b]
                        if abs(denom) > 1e-15:
                            d_bp = (c_top[b] - c_top[a]) / denom
                            if d_lo <= d_bp <= d_hi:
                                candidates.append(d_bp)
                
                # Evaluate at each candidate
                best_d = d_lo
                best_val = np.inf
                for d_c in candidates:
                    val = np.max(c_top + d_c * deltas)
                    if val < best_val:
                        best_val = val
                        best_d = d_c
                
                cur_max = c.max()
                if best_val < cur_max - 1e-12:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    new_c = _eval(h)
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
        
        # Periodic repair
        h = _repair(h, n)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if abs(s - target) < 1e-12:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _deriv_c(h, idx, k):
    n = len(h)
    if k >= 0:
        val = 0.0
        if k <= idx <= n - 1:
            val += 1.0 - h[idx - k]
        if 0 <= idx <= n - 1 - k:
            val -= h[idx + k]
        return val
    l = -k
    val = 0.0
    if 0 <= idx <= n - 1 - l:
        val += 1.0 - h[idx + l]
    if l <= idx <= n - 1:
        val -= h[idx - l]
    return val

def _smooth_obj(x, alpha, top_k):
    c = _eval_vec(x)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, m_idx in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        wk = w[ii] * dx
        k = m_idx - (n - 1)
        if k >= 0:
            cnt = n - k
            g[k:n] += wk * (1.0 - x[0:cnt])
            g[0:cnt] -= wk * x[k:n]
        else:
            l = -k
            cnt = n - l
            g[0:cnt] += wk * (1.0 - x[l:n])
            g[l:n] -= wk * x[0:cnt]
    return g

def _adam_opt(h, n_iter, alpha, top_k, lr, target):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n); v = np.zeros(n)
    cur_c = _eval(h)
    for t in range(1, n_iter + 1):
        g = _smooth_grad(h, alpha, top_k)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        m_hat -= m_hat.mean()
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj_sum(h + step, target)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new; cur_c = c_new
    return h, cur_c

def _slsqp_run(h, alpha, top_k, maxiter):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _f(x): return _smooth_obj(x, alpha, top_k)
    def _g(x): return _smooth_grad(x, alpha, top_k)
    def _cf(x): return x.sum() - target
    def _cj(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_f, h.copy(), jac=_g, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_sweep(h, n_sweeps, top_m):
    n = len(h)
    target = n * 0.5
    best_c = _eval(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corr.max()
    for sw in range(n_sweeps):
        for i in range(n):
            for j in [(i+1)%n, (i-1+n)%n, (i+2)%n, (i+5)%n]:
                if i == j:
                    continue
                top_idx = np.argsort(corr)[-top_m:]
                c_top = corr[top_idx].copy()
                slopes = np.zeros(top_m)
                for t, m_idx in enumerate(top_idx):
                    k = m_idx - (n - 1)
                    slopes[t] = _deriv_c(h, i, k) - _deriv_c(h, j, k)
                d_lo = max(-h[i], h[j]-1.0)
                d_hi = min(1.0-h[i], h[j])
                if d_lo >= d_hi:
                    continue
                candidates = [d_lo, d_hi, 0.0]
                for a in range(top_m):
                    for b in range(a+1, top_m):
                        denom = slopes[a] - slopes[b]
                        if abs(denom) > 1e-15:
                            dbp = (c_top[b] - c_top[a]) / denom
                            if d_lo < dbp < d_hi:
                                candidates.append(dbp)
                best_d = 0.0
                best_val = cur_max
                for dc in candidates:
                    val = np.max(c_top + dc * slopes)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_d = dc
                if abs(best_d) > 1e-15:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _clip_sum(h, target)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    new_max = corr.max()
                    if new_max < cur_max:
                        cur_max = new_max
                        if cur_max < best_c:
                            best_c = cur_max
                            best_h = h.copy()
                    else:
                        h[i] -= best_d
                        h[j] += best_d
                        h = _clip_sum(h, target)
                        corr = np.correlate(h, 1.0 - h, mode='full')
                        cur_max = corr.max()
    return best_h, best_c

def _clip_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) > 1e-10:
        h += (target - s) / len(h)
        h = np.clip(h, 0, 1)
    return h

def _local2(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = step * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c

def _local3(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n

        d2 = step * rng.randn()
