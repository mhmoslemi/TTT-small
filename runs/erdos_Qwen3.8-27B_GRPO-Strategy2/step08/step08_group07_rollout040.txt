import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_sub(h, topk=5):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(c, -(topk - 1))[-topk:] if topk < len(c) else np.arange(len(c))
    dx = 2.0 / n
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(idx)
    return g, c.max() * dx

def _nesterov(h, steps, step0, deadline, tol=0.003, beta=0.87, topk=5):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g, _ = _grad_sub(h, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step0 * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step0 *= 0.6
            hn = _proj(h - step0 * v, target)
            f = _eval(hn)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _block_cd(h, n_passes, rng, deadline, T0=0.01, T1=0.0001):

    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    block_sizes = [4, 8, 12, 20, 32, 50]
    total_iters = 0
    for p in range(n_passes):
        T = T0 * (T1 / T0) ** (p / max(n_passes - 1, 1))
        for bs in block_sizes:
            if time.time() > deadline:
                return bh, bc
            for _ in range(3):
                i0 = rng.randint(0, n - bs + 1)
                seg = h[i0:i0+bs].copy()
                # Projected gradient on block
                for _s in range(8):
                    g, _ = _grad_sub(h, topk=4)
                    seg = seg - 0.002 * g[i0:i0+bs] + 0.0003 * T * rng.randn(bs)
                    seg = np.clip(seg, 0.0, 1.0)
                    h[i0:i0+bs] = seg
                hn = _repair(h, n)
                c5n = _eval(hn)

                    h = hn; c5c = c5n
                    if c5c < bc:
                        bc = c5c; bh = h.copy()
                else:

def _coord_polish(h, n_iters, rng, deadline, scale0=2e-5, scale1=1e-7):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scale = scale0 * (scale1 / scale0) ** (it / max(n_iters - 1, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        for d_sign in (1, -1):
            d = d_sign * scale * rng.uniform(0.5, 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c, bc = c5n, min(bc, c5n)
                    if bc > c5n:
                        bc = c5n; bh = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _slsqp_quick(h, deadline, alpha=20000, topk=10, maxit=200):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -(topk-1))[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        g, _ = _grad_sub(x, topk=topk)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-14})
        return _repair(r.x, n)
    except Exception:
        return h
