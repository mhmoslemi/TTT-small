from scipy.optimize import minimize
import time

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):

    target = n / 2.0
    s = h.sum()

        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_obj_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    smax = s.max()
    logsum = np.log(np.sum(np.exp(s - smax)))
    obj = m + logsum / alpha
    w = np.exp(s - smax); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

def _fista_adaptive(h0, deadline, alpha_lo, alpha_hi, topk, step0):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    y = h.copy()
    t = 1.0
    step = step0
    it = 0
    while time.time() < deadline:
        batch_end = time.time() + 30
        alpha = alpha_lo * (alpha_hi / alpha_lo) ** (it / 150000.0)
        for _ in range(2000):
            if time.time() > batch_end or time.time() > deadline:
                break
            g, _, _ = _smax_obj_grad(y, alpha, topk)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            mu = (t - 1.0) / t_new
            y = h + mu * (h - y)
            z = _proj(y - step * g, target)
            fz = _eval(z)
            if fz < f - 1e-16:
                h, f = z, fz
                step = min(step * 1.02, step0 * 50)
            else:
                step *= 0.6
                z = _proj(y - step * g, target)
                h, f = z, _eval(z)
            if f < best_f:
                best_f, best_h = f, h.copy()
            t = t_new
            it += 1
    return _repair(best_h, n), best_f

def _lbfgs_smooth(h0, alpha, topk, deadline):
    n = len(h0)
    target = n * 0.5
    def obj(x):
        o, _ = _smax_obj_grad(x, alpha, topk)
        return o
    def grad(x):
        _, g = _smax_obj_grad(x, alpha, topk)
        return g
    try:
        r = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 30, 'ftol': 1e-18, 'gtol': 1e-12,
                              'maxls': 30, 'maxfun': 5000})
        return _repair(r.x, n)
    except Exception:
        return h0

def _pair_sweep(h, n_passes, deadline, scale):
    n = len(h)
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        for i in range(n - 1):
            if time.time() > deadline:
                break
            for lag in [1, 2, 3, 5, 7, 11, 13, 17, 23, 31]:
                j = i + lag
                if j >= n:
                    break
                d = scale * (h[i] - h[j])
                if abs(d) < 1e-15:
                    continue
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5 - 1e-15:
                        c5 = c5n
                        if c5 < best_c5:
                            best_c5, best_h = c5, h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _jitter_refine(h0, rng, step0, deadline, alpha, topk):
    n = len(h0)
    c5 = _eval(h0)
    for scale in [0.005, 0.002, 0.001]:
        if time.time() > deadline:
            break
        d = scale * rng.randn(n)
        d -= d.mean()
        hn = _repair(np.clip(h0 + d, 0, 1), n)
        hn, c5n = _fista_adaptive(hn, min(deadline, time.time() + 80),
                                   alpha, alpha * 20, topk, step0)
        if c5n < c5:
            h0, c5 = hn, c5n
    return h0, c5

    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)

    dl3 = min(deadline, t0 + 600)
    h2, c5v = _fista_adaptive(best_h, dl3, 5000, 200000, 6, 0.001)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 4: Jitter + re-optimize (escape local min)
    dl4 = min(deadline, t0 + 750)
    for _ in range(4):
        if time.time() > dl4:
            break
        hj, cj = _jitter_refine(best_h, rng, 0.002, dl4, 10000, 10)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 5: L-BFGS final polish on very sharp smooth
    dl5 = min(deadline, t0 + 850)
    h2 = _lbfgs_smooth(best_h, 100000, 8, dl5)
    c5v = _eval(h2)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    dl6 = min(deadline, t0 + 1000)
    best_h, best_c5 = _pair_sweep(best_h, 20, dl6, 1e-5)

    dl7 = deadline
    best_h, best_c5 = _pair_sweep(best_h, 30, dl7, 1e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
