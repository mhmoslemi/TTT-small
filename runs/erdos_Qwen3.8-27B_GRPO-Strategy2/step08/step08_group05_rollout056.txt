    """LP-based descent: find best direction in top-m gradient cone."""
    best_h = h.copy()
    best_c5 = _eval(h)
    c = h.copy()
    c5_cur = best_c5
    dx = 2.0 / n
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        cor = _correl(c)
        if m_top < 2*n-1:

        else:

        idx_top = idx_top[np.argsort(-cor[idx_top])]

        grads = np.zeros((len(idx_top), n))
        for ii, k_idx in enumerate(idx_top):

                ipk = i + k

from scipy.optimize import linprog
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _coord_cd(h, alpha, topk, n_sweeps, deadline, rng):

    cor = _correl(h)
    best_h = h.copy()

    total_k = 2 * n - 1

        if time.time() > deadline:
            break
        order = rng.permutation(n)
        if topk < total_k:
            active_idx = np.argpartition(cor, -topk)[-topk:]
        else:
            active_idx = np.arange(total_k)
        active_k = active_idx - (n - 1)
        active_ct = cor[active_idx].copy()
        nk = len(active_idx)
        for i in order:
            a = np.empty(nk)
            for ii in range(nk):
                k = active_k[ii]
                ak = 0.0
                ipk = i + k
                if 0 <= ipk < n:
                    ak += 1.0 - h[ipk]

                m = vals.max()
                s = np.exp(np.clip(alpha * (vals - m), -400, 400))
                w = s / s.sum()

            fp0 = fp(0.0)

            if fp0 >= 0:

        c5_cur = float(cor.max())

            best_h = h.copy()

def _lp_descent(h, n, eps, m_top, max_iters, deadline):
    best_h = h.copy()
    best_c5 = _eval(h)
    c = h.copy()
    c5_cur = best_c5
    total_k = 2 * n - 1
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        cor = _correl(c)
        if m_top < total_k:
            idx_top = np.argpartition(cor, -m_top)[-m_top:]
        else:
            idx_top = np.arange(total_k)
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.zeros((len(idx_top), n))
        for ii, k_idx in enumerate(idx_top):
            k = k_idx - (n - 1)
            for i in range(n):
                g = 0.0
                ipk = i + k
                if 0 <= ipk < n:
                    g += 1.0 - c[ipk]
                imk = i - k
                if 0 <= imk < n:
                    g -= c[imk]
                grads[ii, i] = g
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(len(idx_top) - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n

        try:

def _polish(h, niter, rng, deadline):

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sss = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
