import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(8):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = float(max(P[0], np.max(Nv)))
    return P, Nv, mx

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    return np.correlate(h, 1.0 - h, mode='full')

        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _power_value(h, p):
    """Lp norm approximation of max."""
    P, Nv, _ = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    # Combine P and Nv into one vector
    if n > 1:
        all_v = np.concatenate([P, Nv[1:]])
    else:
        all_v = P
    return dx * float(np.sum(all_v**p)) ** (1.0/p)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        convR = np.fft.irfft(H * Urev, Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * Ur, Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)

def _fft_size(n):

def _proj(v, S):

                h[i:i+w] -= mv / w; h[j:j+w] += mv / w
                cn = _c5_direct(h)
                if cn < c or (T > 1e-14 and rng.random() < np.exp(-(cn - c) / T)):
                    c = cn
                    if c < best_c: best_c = c; best_h = h.copy()
                else:
                    h[i:i+w] += mv / w; h[j:j+w] -= mv / w
        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0); h0 = _repair(h0, n0)
    best = h0.copy(); best_c = _c5_direct(best)
    for hh in [1.0 - h0, h0[::-1]]:
        c = _c5_direct(_repair(hh, n0))
        if c < best_c: best = _repair(hh, n0); best_c = c

    # Phase 1: FISTA at full res (up to 100s)
    end1 = t0 + min(100.0, 0.12 * budget_s)
    hcur = best.copy()
    for a in [20, 100, 500, 3000, 20000]:
        slot = min(20.0, end1 - time.time() - 2)
        if slot < 3: break
        hf, bh, bc = _fista(hcur, a, slot)
        chf = _c5_direct(hf)
        if bc < best_c: best_c = bc; best = bh
        if chf < best_c: best_c = chf; best = hf
        hcur = hf if chf <= _c5_direct(bh) else bh

    # Phase 2: Multi-res (up to 150s)
    end2 = end1 + min(150.0, 0.18 * budget_s)
    for m in [n0 // 2, n0 // 4]:
        slot = end2 - time.time() - 3
        if slot < 5: break
        hm = _resample(best, m); hcur = hm
        for a in [50, 500, 5000]:
            s2 = min(15.0, end2 - time.time() - 2)
            if s2 < 3: break
            hf, bh, bc = _fista(hcur, a, s2)
            cu = _c5_direct(_resample(hf, n0))
            cbu = _c5_direct(_resample(bh, n0))
            if cu < best_c: best_c = cu; best = _resample(hf, n0)
            if cbu < best_c: best_c = cbu; best = _resample(bh, n0)
            hcur = hf if cu <= cbu else bh

    # Phase 3: FISTA + polish at full res (up to 250s)
    end3 = t0 + budget_s - 300.0
    hcur = best.copy()
    for a in [50, 400, 5000, 50000]:
        slot = min(25.0, end3 - time.time() - 2)
        if slot < 3: break
        hf, bh, bc = _fista(hcur, a, slot)
        chf = _c5_direct(hf)
        if bc < best_c: best_c = bc; best = bh
        if chf < best_c: best_c = chf; best = hf
        hcur = hf if chf <= _c5_direct(bh) else bh

    # Phase 4: Polish (gradient on active set)
    end4 = t0 + budget_s - 250.0
    if end4 > time.time() + 5:
        best, best_c = _polish(best, end4, 3e-5)

    # Phase 5: Local search
    end5 = t0 + budget_s - 50.0
    if end5 > time.time() + 5:
        best, best_c = _local_search(best, rng, end5, 4e-5)

    # Phase 6: Perturb + restart
    end6 = t0 + budget_s - 2.0
    if end6 - time.time() > 30:
        pert = best + rng.randn(n0) * 0.008
        pert = _repair(pert, n0)
        s = min(15.0, end6 - time.time() - 20)
        if s > 5:
            hf, bh, bc = _fista(pert, 10000, s)
            if bc < best_c: best_c = bc; best = bh
        best, best_c = _local_search(best, rng, end6, 2e-5)

    # Final variants
    for hh in [1.0 - best, best[::-1],
               0.5 + 0.5 * (best - best[::-1]),
               0.5 + 0.5 * (best + best[::-1] - 1.0)]:
        h2 = _proj(hh, n0 * 0.5)
        c = _c5_direct(h2)
        if c < best_c: best = h2; best_c = c

    best = _repair(best, n0)
    best_c = _c5_direct(best)
    return best, best_c, n0
