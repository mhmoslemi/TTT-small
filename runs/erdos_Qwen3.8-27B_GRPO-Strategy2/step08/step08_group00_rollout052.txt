    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smx_obj(x, alpha, m):
    dx = 2.0 / len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - c.max()), -400, 400)
    s -= s.max()
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _smx_grad(x, alpha, m):
    dx = 2.0 / len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    mval = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - mval), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(len(x))
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (len(x) - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:len(x) - L])
            g[:len(x) - L] -= wk * x[L:]
        else:
            M = -L
            g[:len(x) - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:len(x) - M]
    return g

def _c_fun(x, hn):
    return x.sum() - hn

def _c_jac(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, niter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac, 'args': (n * 0.5,)}]
    try:
        r = minimize(_smx_obj, h, jac=_smx_grad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': niter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _nesterov_grad(h, alpha, ml, steps, lr0):
    n = len(h)
    half = n / 2.0
    y = h.copy()
    vm = np.zeros(n)
    beta = 0.9
    cur = _eval(h)
    lr = lr0
    for t in range(steps):

        vm = beta * vm + (1 - beta) * g
        y = h - lr * vm
        h_new = _proj(y, half)
        c = _eval(h_new)
        if c < cur:
            beta = min(beta + 0.01, 0.95)
            lr = min(lr * 1.05, lr0 * 10)
            h = h_new
            cur = c
        else:
            lr *= 0.7
            beta = max(beta - 0.01, 0.7)
    return _repair(h, n)

    n = len(h)

    ds = np.array([-0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01])
    for _ in range(niters):

        j = (i + 1 + rng.randint(0, n - 1)) % n

        found = False
        for d in ds:

            if 0 <= ni <= 1 and 0 <= nj <= 1:

                    cur = c
                    found = True
                    break
                h[i], h[j] = hi, hj
        if found:
            h[i], h[j] = hi + ds[ds.tolist().index(d)], hj - ds[ds.tolist().index(d)]
        # also try random d
        d2 = 0.003 * rng.randn()
        ni, nj = hi + d2, hj - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cur:
                cur = c
            else:
                h[i], h[j] = hi, hj
    return _repair(h, n)
