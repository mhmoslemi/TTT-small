        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_obj(x, alpha, m_lags):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _adaptive_proj_grad(h, alpha, m_lags, steps, step0):
    n = len(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    mom = 0.85
    for it in range(steps):
        g = _smooth_grad(h, alpha, m_lags)
        v = mom * v + (1.0 - mom) * g
        sq = 0.95 * sq + 0.05 * g * g
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(14):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.12, step0 * 30.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, step0 * 30.0)
    return _repair(h, n)

def _grad_guided_2pt(h, n_iter, alpha, m_lags):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        g = _smooth_grad(h, alpha, m_lags)
        g_proj = g - g.mean()
        i = int(np.argmax(g_proj))
        j = int(np.argmin(g_proj))
        if i == j:
            continue
        best_d = 0.0
        best_c5 = c5c
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            d = frac * (g_proj[i] - g_proj[j]) * 0.01
            h[i] += d
            h[j] -= d
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= d
                h[j] += d
                continue
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_d = d
            else:
                h[i] -= d
                h[j] += d
                break
        if best_d != 0.0:
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c
                best_h = h.copy()
        else:
            h[i] -= best_d
            h[j] += best_d
    return best_h, best_c

def _local_search(h, n_iter, rng, base_step, n_pert):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        step = base_step * (0.5 + rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution coarse optimization
    for ratio in [4, 8]:
        if time.time() - t0 > 55:
            break
        nc = max(40, n // ratio)
        f = n // nc
        hc = _downsample(h, f)
        for alpha in [500, 3000, 20000]:
            if time.time() - t0 > 75:
                break
            hc = _slsqp(hc, alpha, 30, 400, 1e-14)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: SLSQP alpha dive at full resolution
    alpha_seq = [3000, 10000, 50000, 200000, 800000, 3000000, 10000000, 50000000, 200000000]
    for alpha in alpha_seq:
        if time.time() - t0 > 350:
            break
        lift = min(90, max(20, int(40 * np.log10(alpha / 3000) + 20)))
        h_new = _slsqp(best_h, alpha, lift, 1200, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Adaptive RMSProp projected gradient
    if time.time() - t0 < budget_s - 300:
        configs = [
            (5000, 40, 600, 0.005),
            (30000, 50, 500, 0.003),
            (150000, 70, 400, 0.002),
            (600000, 90, 300, 0.001),
            (2000000, 120, 250, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s - 220:
                break
            h_new = _adaptive_proj_grad(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Very high alpha SLSQP
    if time.time() - t0 < budget_s - 130:
        for alpha in [500000000, 2000000000, 8000000000]:
            if time.time() - t0 > budget_s - 80:
                break
            h_new = _slsqp(best_h, alpha, 130, 600, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Gradient-guided 2-point search
    if time.time() - t0 < budget_s - 50:
        nitr = min(int((budget_s - 50 - (time.time() - t0)) / 0.012), 60000)
        best_h, best_c5 = _grad_guided_2pt(best_h, nitr, 1000000, 80)

    # Phase 6: Random local search
    if time.time() - t0 < budget_s - 25:
        ntries = min(int((budget_s - 25 - (time.time() - t0)) / 0.003), 200000)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 2e-6, 3)

    # Phase 7: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 2 - (time.time() - t0)) / 0.0012), 500000)
        for _ in range(npol):
            if time.time() - t0 > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 8)) % nc2
            d = 1.5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
