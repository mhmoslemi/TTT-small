                     jac=lambda x: _lsem_grad(x, alpha, topk),

                     options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return h

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 7)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, step0, tol, deadline):
    n = len(h)
    c5c = _eval(h)
    for jitter_scale in [0.002, 0.004, 0.001]:
        d = jitter_scale * rng.randn(n)
        d -= d.mean()
        hn = _repair(np.clip(h + d, 0, 1), n)
        h2, c2 = _nesterov(hn, 'exact', 600, step0, deadline, tol=tol, beta=0.9)
        if c2 < c5c:
            return h2, c2
        h3, c3 = _nesterov(hn, 'smax', 400, step0 * 0.5, deadline, alpha=100000, topk=5, beta=0.88)
        if c3 < c5c:
            return h3, c3
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Progressive L-BFGS-B with increasing sharpness
    dl1 = min(deadline, t0 + 120)
    for alpha, topk, mi in [(5000, 25, 200), (20000, 15, 250), (80000, 8, 250), (300000, 4, 200)]:
        if time.time() > dl1:
            break
        h = _lbgfsb(best_h.copy(), alpha, topk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: FISTA with smooth objective
    dl2 = min(deadline, t0 + 220)
    h2, c2 = _fista(best_h.copy(), 'smax', 3000, 0.004, dl2, alpha=50000, topk=10)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 3: Nesterov exact subgradient (multiple configs)
    dl3 = min(deadline, t0 + 420)
    configs = [(0.003, 0.004, 0.85), (0.002, 0.003, 0.88), (0.0015, 0.002, 0.90), (0.0008, 0.001, 0.92)]
    for step0, tol, mom in configs:
        if time.time() > dl3:
            break
        h2, c2 = _nesterov(best_h.copy(), 'exact', 6000, step0, dl3, tol=tol, beta=mom)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 4: Tight smooth-max Nesterov
    dl4 = min(deadline, t0 + 560)
    for alpha, topk, step0 in [(100000, 6, 0.003), (300000, 3, 0.0015), (800000, 2, 0.001)]:
        if time.time() > dl4:
            break
        h2, c2 = _nesterov(best_h.copy(), 'smax', 4000, step0, dl4, alpha=alpha, topk=topk, beta=0.90)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 5: Jitter + re-optimize (escape local minima)
    dl5 = min(deadline, t0 + 720)
    for _ in range(8):
        if time.time() > dl5:
            break
        hj, cj = _jitter_reopt(best_h.copy(), rng, 0.002, 0.002, dl5)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 6: Block annealing
    dl6 = min(deadline, t0 + 900)
    blocks = [2, 3, 4, 5, 7, 10, 14]
    best_h, best_c5 = _anneal(best_h.copy(), 100000, rng, dl6, blocks, 0.004, 1e-10)

    # Phase 7: Final fine Nesterov
    dl7 = min(deadline, t0 + budget_s - 60)
    h2, c2 = _nesterov(best_h.copy(), 'exact', 10000, 0.0004, dl7, tol=0.0004, beta=0.93)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    # Phase 8: Fine polish
    dl8 = deadline
    rem = max(0, dl8 - time.time())
    n_pol = min(int(rem / 0.001), 70000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, dl8, 4e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
