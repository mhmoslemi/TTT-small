    c = _corr_vec(x)
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c[idx].max())
    s -= s.max()
    return c[idx].max() + np.log(np.exp(s).sum()) / alpha

    n = len(x)
    c = _corr_vec(x)
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c[idx].max())
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

    out = np.empty(n)
    for i in range(nc):
        out[i * f:(i + 1) * f] = hc[i]
    return _repair(out, n)

def _eq_lags(h, n_iter, rng, step0):
    """Try to equalize top lags by shifting weight from peak to near-peak."""
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        c = _corr_vec(h)
        top = np.argsort(c)[-5:]
        peak_k = top[np.argmax(c[top])]
        L = peak_k - (n - 1)
        if L >= 0:
            src_range = slice(0, n - L)
            dst_range = slice(L, n)
        else:
            M = -L
            src_range = slice(M, n)
            dst_range = slice(0, n - M)
        i = rng.choice(np.arange(*src_range.indices))
        j = rng.choice(np.arange(*dst_range.indices))
        d = step0 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_search_2pt(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    step = base_step
    stall = 0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                stall = 0
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                step = min(step * 1.05, base_step * 2.0)
            else:
                h[i], h[j] = ni - d, nj + d
                stall += 1
                if stall > 100:
                    step *= 0.5
                    stall = 0
                    if step < 1e-9:
                        break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Diverse exploration
    for it in range(6):
        if _el() > 70:
            break
        hp = best_h.copy()
        scale = 0.03 + 0.02 * it
        hp += rng.randn(n) * scale
        hp = _repair(hp, n)
        hc = _coarse_opt(hp, n, 4, 8000)
        c5c = _eval(hc)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 2: SLSQP alpha sweep (main optimization)
    alpha_seq = [5000, 20000, 80000, 400000, 2000000, 10000000, 50000000, 200000000, 800000000, 3000000000]
    for alpha in alpha_seq:
        if _el() > 380:
            break
        lift = min(120, max(25, int(25 * np.log2(alpha / 5000.0) + 25)))
        h_new = _slsqp(best_h, alpha, lift, 2500, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient with diverse configs
    if _el() < budget_s - 280:
        configs = [
            (10000, 40, 1200, 0.005),
            (50000, 50, 900, 0.003),
            (200000, 60, 700, 0.002),
            (800000, 80, 500, 0.001),
            (3000000, 100, 350, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if _el() > budget_s - 200:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Equalize lags
    if _el() < budget_s - 100:
        ntries = min(int((budget_s - 100 - _el()) / 0.004), 150000)
        best_h, c5e = _eq_lags(best_h, ntries, rng, 2e-6)
        if c5e < best_c5:
            best_c5 = c5e

    # Phase 5: High alpha SLSQP
    if _el() < budget_s - 80:
        for alpha in [10000000000, 50000000000]:
            if _el() > budget_s - 30:
                break
            h_new = _slsqp(best_h, alpha, 150, 1000, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 6: Local search
    if _el() < budget_s - 5:
        ntries = min(int((budget_s - 5 - _el()) / 0.003), 400000)
        best_h, best_c5 = _local_search_2pt(best_h, ntries, rng, 3e-6)

    # Phase 7: Ultra-fine polish
    if _el() < budget_s - 1:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 1 - _el()) / 0.0025), 200000)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 3)) % nc2
            d = 8e-8 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
