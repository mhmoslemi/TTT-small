def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5

        s = h.sum()

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = 0.5 * n
    for _ in range(80):
        s = h.sum()
        if abs(s - t) < 1e-12: break
        h *= t / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11: break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0: break
        x[m] += d / c
    return x

def _pn(h):
    n = len(h)
    N = 1; m = 2*n - 1
    while N < m: N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n+1); pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    P = np.empty(n); P[0] = pref[n] - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n-idx] - ac[idx]
        Nv[idx] = pref[n] - pref[idx] - ac[idx]
    return P, Nv

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _sval(h, a):
    P, Nv = _pn(h)
    mx = max(P[0], np.max(P[1:]), np.max(Nv[1:])) if len(h)>1 else P[0]
    e = np.exp(np.clip(a*(P-mx), -400., 0.)).sum() + np.exp(np.clip(a*(Nv-mx), -400., 0.)).sum()
    return (2.0/len(h)) * (mx + np.log(max(e,1e-300))/a)

def _gw(h, wP, wN):
    n = len(h); dx = 2.0/n
    u = np.zeros(n)
    if n > 1: u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n-1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1; m = 2*n-1
        while Nf < m: Nf <<= 1
        Hh = np.fft.rfft(h, Nf)
        R = np.fft.irfft(Hh * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(Hh * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx*(wP[0]*(1.0-2.0*h) + linP + linN + q)

def _sgrad(h, a):
    P, Nv = _pn(h)
    n = len(h)
    mx = max(P[0], np.max(P[1:]), np.max(Nv[1:])) if n>1 else P[0]
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(a*(P[0]-mx), -400., 0.))
    tot = e0
    if n > 1:
        eP = np.exp(np.clip(a*(P[1:]-mx), -400., 0.))
        eN = np.exp(np.clip(a*(Nv[1:]-mx), -400., 0.))
        tot += eP.sum() + eN.sum()
        if tot <= 0: tot = 1.; e0 = 1.
        wP[0] = e0/tot; wP[1:] = eP/tot; wN[1:] = eN/tot
    else: wP[0] = 1.
    return _gw(h, wP, wN)

def _active_grad(h, tol_frac=0.003):
    P, Nv = _pn(h)
    n = len(h)
    mx = max(P[0], np.max(P[1:]), np.max(Nv[1:])) if n>1 else P[0]
    tol = max(1e-9, tol_frac * mx)
    wP = np.zeros(n); wN = np.zeros(n); cnt = 0
    if P[0] >= mx - tol: wP[0] = 1.; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        mN = Nv[1:] >= mx - tol
        if np.any(mP):
            wP[np.nonzero(mP)[0]+1] = 1.; cnt += mP.sum()
        if np.any(mN):
            wN[np.nonzero(mN)[0]+1] = 1.; cnt += mN.sum()
    if cnt == 0: wP[0] = 1.; cnt = 1
    inv = 1.0/cnt
    wP *= inv; wN *= inv
    return _gw(h, wP, wN)

def _fista(h0, alpha, max_time, L=12.0):
    n = len(h0); S = 0.5*n
    t0 = time.time()
    h = _repair(h0, n); hp = h.copy(); y = h.copy(); t = 1.0
    fy = _sval(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    Lcur = L; stale = 0
    for it in range(500000):
        if time.time() - t0 > max_time: break
        if it:
            tn = 0.5*(1.0+np.sqrt(1.0+4.0*t*t))
            y = h + ((t-1.0)/tn)*(h-hp); t = tn
            if y.min() < -1e-5 or y.max() > 1.+1e-5 or abs(y.sum()-S) > 1e-7:
                y = _proj(y, S)
            fy = _sval(y, alpha)
        else: y = h; fy = fh
        g = _sgrad(y, alpha)
        step = 1.0/max(Lcur, 1e-12); accepted = False
        for _ in range(14):
            xn = _proj(y - step*g, S)
            fx = _sval(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g,d)) + 0.5*(1.0/step)*float(np.dot(d,d))
            if fx <= rhs + 1e-10: accepted = True; break
            step *= 0.5
        Lcur = 1.0/max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lcur = max(Lcur*0.85, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c: best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lcur = max(Lcur*0.5, 1.0)
            if stale > 12: break
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    x0 = (np.arange(n)+0.5)*(2.0/n)
    x1 = (np.arange(m)+0.5)*(2.0/m)
    out = np.interp(x1, x0, h)
    return _proj(out, m*0.5)

def _local_search(h0, rng, deadline):
    tl0 = time.time()
    n = len(h0); h = h0.copy()
    c = _c5(h); best_c = c; best_h = h.copy()
    g = _active_grad(h)
    order = np.argsort(g)
    q = max(5, n//8)
    low = order[:q].copy(); high = order[-q:].copy()
    next_recomp = 1500
    it = 0; T = 1e-6; it_anneal_end = int(0.4*(deadline-tl0)*30000)
    kick_next = 80000
    while time.time() < deadline:
        if it >= next_recomp:
            c = _c5(h)
            if c < best_c: best_c = c; best_h = h.copy()
            g = _active_grad(h)
            order = np.argsort(g)
            low = order[:q].copy(); high = order[-q:].copy()
            next_recomp = it + 1500
        if it == kick_next:
            h += (rng.random(n) - 0.5) * 0.003
            h = _proj(h, 0.5*n)
            c = _c5(h)
            kick_next = it + 100000
        frac = (it) / max(1e-9, it_anneal_end)
        base = max(3e-8, 5e-5 * (0.01)**(it/max(1,it_anneal_end)))
        move = rng.randint(100)
        if move < 70:
            i = int(low[rng.randint(len(low))]); j = int(high[rng.randint(len(high))])
        elif move < 85:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        else:
            i = int(rng.randint(n-2)); j = int(rng.randint(i+2, n))
        if i == j: continue
        sgn = 1 if g[i] <= g[j] else (-1 if g[j] < g[i] else (1 if rng.random()<0.5 else -1))
        if sgn > 0: avail = min(1.0-h[i], h[j])
        else: avail = min(h[i], 1.0-h[j])
        if avail > 1e-14:
            mult = 0.3 + 1.7*rng.random()
            if rng.random() < 0.02: mult *= 10.0
            d = sgn * min(avail, base*mult)
            h[i] += d; h[j] -= d
            cn = _c5(h)
            delta = cn - c
            if delta < -1e-15:
                c = cn
                if c < best_c - 1e-15: best_c = c; best_h = h.copy()
            else:
                accept = (it < it_anneal_end and rng.random() < np.exp(-max(0,delta)/(T*(1.0-0.9*frac)+1e-12)))
                if not accept:
                    h[i] -= d; h[j] += d
        it += 1
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return best_h, best_c
