        if time.time() > deadline:
            break
        candidates = []
        for factor in [2, 4]:
            if time.time() > deadline:
                break
            h_coarse = _downsample(h, factor)
            nc = len(h_coarse)
            h_coarse = _nesterov(h_coarse, 15000, 2.0, 10, 2000, 0.006, deadline)
            h_up = _upsample(h_coarse, n)
            h_up = _repair(h_up, n)
            h_fin = _nesterov(h_up, 40000, 1.5, 8, 2000, 0.003, deadline)
            candidates.append(h_fin)
        for hc in candidates:
            c = _eval(hc)
            if c < best_c:
                best_c = c
                best_h = hc.copy()
                h = hc.copy()
    return best_h, best_c

def _direct_max(h, steps, deadline):
    n = len(h)
    dx = 2.0 / n
    for si in range(steps):
        if time.time() > deadline:
            break
        c = _correl(h)
        topk = np.argpartition(c, -4)[-4:]
        g = np.zeros(n)
        for k in topk:
            g += _grad_lag(h, k, dx)
        g /= len(topk)
        improved = False
        for lr in [0.004, 0.002, 0.001, 0.0005, 0.0002]:
            hn = _proj(h - lr * g, n * 0.5)
            hn = _repair(hn, n)
            if _eval(hn) < _eval(h) - 1e-15:
                h = hn
                improved = True
                break
        if not improved and si > 50:
            break
    return _repair(h, n)

def _pair_search(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 0.003
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        dist = 1 + rng.randint(0, 15)
        j = (i + dist) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
                step = min(step * 1.06, 0.025)
            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.97, 1e-7)
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    dl1 = min(deadline, t0 + 100)
    h1 = _nesterov(h, 15000, 1.8, 12, 4000, 0.005, dl1)
    c1 = _eval(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    dl2 = min(deadline, t0 + 320)
    h2, c2 = _multi_res(best_h.copy(), 4, dl2)
    if c2 < best_c5:
        best_c5, best_h = c2, h2.copy()

    dl3 = min(deadline, t0 + 430)
    h3 = _direct_max(best_h.copy(), 6000, dl3)
    c3 = _eval(h3)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    dl4 = min(deadline, t0 + 550)
    h4 = _nesterov(best_h.copy(), 80000, 1.5, 6, 3000, 0.003, dl4)
    c4 = _eval(h4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4.copy()
    h4b = _nesterov(h4, 300000, 1.3, 4, 2000, 0.002, dl4)
    c4b = _eval(h4b)
    if c4b < best_c5:
        best_c5, best_h = c4b, h4b.copy()

    dl5 = min(deadline, t0 + 730)
    blocks = [1, 2, 3, 5, 7, 12, 20, 35]
    h5, c5 = _block_anneal(best_h.copy(), 100000, rng, dl5, 0.004, 1e-10, blocks)
    if c5 < best_c5:
        best_c5, best_h = c5, h5.copy()

    dl6 = min(deadline, t0 + 870)
    h6, c6 = _multi_res(best_h.copy(), 2, dl6)
    if c6 < best_c5:
        best_c5, best_h = c6, h6.copy()

    dl7 = min(deadline, t0 + 990)
    rem7 = max(0, dl7 - time.time())
    n7 = min(int(rem7 / 0.0008), 150000)
    h7, c7 = _pair_search(best_h.copy(), n7, rng, dl7)
    if c7 < best_c5:
        best_c5, best_h = c7, h7.copy()

    dl8 = deadline
    h8 = _nesterov(best_h.copy(), 500000, 1.2, 4, 2000, 0.001, dl8)
    c8 = _eval(h8)
    if c8 < best_c5:
        best_c5, best_h = c8, h8.copy()
    rem8 = max(0, deadline - time.time())
    n8 = min(int(rem8 / 0.0008), 50000)
    h8b, c8b = _pair_search(best_h.copy(), n8, rng, deadline)
    if c8b < best_c5:
        best_c5, best_h = c8b, h8b.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
