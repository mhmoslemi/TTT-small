    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        if off < n:
            g[:n-off] += (1.0 - h[off:])
            g[off:] -= h[:n-off]
    else:
        sp = -off
        if sp < n:
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g

def _build_G(c, n):
    G = np.zeros((2*n-1, n))
    for off in range(0, n):
        k = off + n - 1
        G[k, :n-off] += (1.0 - c[off:])
        G[k, off:] -= c[:n-off]
    for off in range(1, n):
        k = n - 1 - off
        G[k, off:] += (1.0 - c[:n-off])
        G[k, :n-off] -= c[off:]
    return G

def _tangent_lp(h, deadline, max_iter=400):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    stuck = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        cor_raw = np.correlate(c, 1.0 - c, mode='full')
        G = _build_G(c, n)
        c_obj = np.zeros(n + 1); c_obj[n] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = G; A_ub[:, n] = -1.0
        b_ub = G @ c - cor_raw
        A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
        b_eq = np.array([n * 0.5])
        bnds = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs',
                          options={'presolve': True, 'time_limit': 3})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        h_lp = _repair(res.x[:n], n)
        c5_cur = _eval(c)
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = _repair(c + alpha * (h_lp - c), n)
            c5_try = _eval(h_try)
            if c5_try < c5_cur - 1e-14:
                c = h_try
                if c5_try < best_c5:
                    best_c5, best_h = c5_try, c.copy()
                improved = True
                break
        if not improved:
            stuck += 1
            if stuck > 8:
                rng = np.random.RandomState(hash((it, int(time.time()))) % 2**31)
                d = rng.randn(n) * 0.003
                d -= d.mean()
                c_try = _repair(c + d, n)
                if _eval(c_try) < best_c5 + 0.001:
                    c = c_try
                stuck = 0
        else:
            stuck = 0
    return best_h, best_c5

def _nesterov_sub(h, steps, step0, deadline, K=15):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    beta = 0.9
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        cor = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        m = cor.max()
        idx = np.argpartition(cor, -K)[-K:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(h, k, n)
        g *= (2.0 / n) / K
        hp = h.copy()
        v = beta * v + (1 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _eval(hn)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.03, step0 * 4)
        else:
            step *= 0.6
            if step < 1e-13:
                break
            h = _proj(hp - step * v, n * 0.5)
            f = _eval(h)
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc
