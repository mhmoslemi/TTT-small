            j = (i + 1 + rng.randint(0, 3)) % n

            if sh > 0:

                pairs = h[:n-sh] * (1.0 - h[sh:])

                # If h[i] > h[i+sh], moving mass from i to i+sh helps
                # If h[i] < h[i+sh], moving mass from i+sh to i helps
                diff = h[:n-sh] - h[sh:]
                # We want to equalize: move from higher to lower
                # Impact on this lag: -|diff| * delta_move
                # But impact on other lags unknown
                # Just pick the most imbalanced pair
                idx = np.argmax(np.abs(diff))
                if diff[idx] > 0:
                    i, j = idx, idx + sh
                else:
                    i, j = idx + sh, idx
            elif sh < 0:
                sp = -sh
                pairs = h[sp:] * (1.0 - h[:n-sp])
                diff = h[sp:] - h[:n-sp]
                idx = np.argmax(np.abs(diff))
                if diff[idx] > 0:
                    i, j = idx + sp, idx
                else:
                    i, j = idx, idx + sp
            else:
                i = rng.randint(n)
                j = (i + 1) % n
        
        # Try a small move
        d = base_step * rng.randn()
        # Ensure we move in the right direction (reduce the imbalanced pair)
        if d > 0 and h[i] < h[j]:
            d = -d
        if d < 0 and h[i] > h[j]:
            d = -d
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _lbfgs_focus(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        top = np.sort(c)[-top_k:] if (top_k > 0 and top_k < len(c)) else c
        m = top.max()
        s = np.clip(alpha * (top - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha + 0.5 * 0.005 * (x.sum() - target)**2
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if (top_k > 0 and top_k < len(c)) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(idx[ii], x, n)
        g += 0.005 * (x.sum() - target)
        return g
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, jac=_grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-15, 'maxfun': maxiter * 3})
    return _repair(res.x, n)

def _slsqp_focus(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if (top_k > 0 and top_k < len(c)) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)

        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):

        idx = np.argsort(c)[-top_k:] if (top_k > 0 and top_k < len(c)) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()

        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(idx[ii], x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad_adapt(h, alpha, steps, step0, top_k):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    beta = 0.85
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):

                continue
            g += w[ii] * _grad_lag(active[ii], h, n)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c - 1e-18:
            h = h_new
            cur_c = c_new

        else:
            step *= 0.45
        beta = min(beta + 0.0005, 0.98)
        if it % 300 == 0:
            v *= 0.5
    return _repair(h, n)

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out
