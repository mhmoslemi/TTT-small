def _repair(h, n):

        if abs(s - t) < 1e-12:

        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
