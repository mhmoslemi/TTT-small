    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd_nm(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = np.zeros(n)
    mu = 0.75
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y_new = h + step * g
        v = mu * v + (1.0 - mu) * (y_new - h)
        y = y_new
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.04, step0 * 4.0)
        else:
            step *= 0.6
            step = max(step, step0 * 0.1)
    return _repair(h, n), cur_c

def _slsqp_lse(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def _obj(x):
        c = _eval_vec(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):
        return _grad_lse(x, alpha, top_k)
    def _sum_c(x):
        return x.sum() - target
    def _sum_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _lag_local(h, n_iter, step0, rng):
    """Lag-aware local search: target pairs at critical lags with line search."""
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    rng_step = step0
    for it in range(n_iter):
        # Adaptive step
        rng_step = step0 * (1.0 + 0.5 * np.random.RandomState(it % 1000).rand())
        # Get critical lags
        c_all = _eval_vec(h)
        max_val = c_all.max()
        thresh = max_val - max(0.002, max_val * 0.005)
        crit = np.where(c_all >= thresh)[0]
        if len(crit) == 0:
            crit = np.argmax(c_all).__class__([np.argmax(c_all)])
        k = crit[rng.randint(len(crit))]
        sh = (n - 1) - k
        sh = abs(sh)
        if sh == 0:
            sh = 1
        i = rng.randint(n)
        j = (i + sh) % n
        if i == j:
            continue
        # 1D line search for d
        lo_d, hi_d = -rng_step, rng_step
        best_d = 0.0
        best_cd = cur_c
        for _ in range(8):
            mid = (lo_d + hi_d) * 0.5
            ni = h[i] + mid
            nj = h[j] - mid
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cd = _eval(h)
                if cd < best_cd:
                    best_cd = cd
                    best_d = mid
                h[i], h[j] = h[i] - mid, h[j] + mid
            if mid >= 0:
                hi_d = mid
            else:
                lo_d = mid
        # Apply best
        if best_cd < cur_c:
            h[i] += best_d
            h[j] -= best_d
            cur_c = best_cd
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            rng_step *= 1.1
        else:
            rng_step *= 0.85
            rng_step = max(rng_step, 1e-7)
    return _repair(best_h, n), best_c

def _local2(h, n_iter, step0, rng):

        ni, nj = h[i] + d, h[j] - d

                    best_h = h.copy()
            else:

    return best_h, best_c

def _local3(h, n_iter, step0, rng):

def _structured_perturb(h, n, rng, mode):
    """Apply structured perturbation."""
    if mode == 0:  # sinusoidal
        freq = rng.randint(2, 10)
        phase = rng.rand() * 2 * np.pi
        amp = 0.03
        pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 1:  # shear
        shift_amt = rng.randint(1, 5)
        pert = np.zeros(n)
        pert[:n//2] = shift_amt * 0.01
        pert[n//2:] = -shift_amt * 0.01
    else:  # random smooth
        pert = rng.randn(n) * 0.02
        # smooth
        kernel = np.convolve(np.ones(5)/5, np.ones(5)/5, 'same')
        pert = np.convolve(pert, kernel, 'same')
    pert -= pert.mean()
    return _repair(h + pert, n)

        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):
        return _grad_lse(x, alpha, top_k)
    def _sum_c(x):
        return x.sum() - target
    def _sum_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _lag_local(h, n_iter, step0, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    cur_step = step0
    c_all = _eval_vec(h)
    for it in range(n_iter):
        if it % 40 == 0:
            c_all = _eval_vec(h)
        max_val = c_all.max()
        thresh = max_val - max(0.0015, max_val * 0.004)
        crit = np.where(c_all >= thresh)[0]
        if len(crit) == 0:
            k = np.argmax(c_all)
        else:
            k = crit[rng.randint(len(crit))]
        sh = abs((n - 1) - k)
        if sh == 0:
            sh = 1
        i = rng.randint(n)
        j = (i + sh) % n
        if i == j:
            continue
        d = cur_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                cur_step = min(cur_step * 1.03, step0)
            else:
                h[i], h[j] = ni - d, nj + d
                cur_step = max(cur_step * 0.92, step0 * 0.005)
    return _repair(best_h, n), best_c

def _sweep_lag(h, n_passes, step0, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for p in range(n_passes):
        c_all = _eval_vec(h)
        max_val = c_all.max()
        thresh = max_val - 0.001
        crit = np.where(c_all >= thresh)[0]
        if len(crit) == 0:
            sh = 1
        else:
            k = crit[rng.randint(len(crit))]
            sh = abs((n - 1) - k)
            if sh == 0:
                sh = 1
        order = rng.permutation(n)
        for i in order:
            j = (i + sh) % n
            if i == j:
                continue
            d = step0 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < cur_c:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local2(h, n_iter, step0, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = step0 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local3(h, n_iter, step0, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        k = (i + 6 + rng.randint(0, 12)) % n
        d1 = step0 * rng.randn()
        d2 = step0 * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = s0, s1, s2
    return best_h, best_c

def _structured_perturb(h, n, rng, mode):
    if mode == 0:
        freq = rng.randint(2, 8)
        phase = rng.rand() * 2.0 * np.pi
        amp = 0.025
        pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 1:
        pert = np.zeros(n)
        pert[:n//2] = 0.015
        pert[n//2:] = -0.015
    else:
        pert = rng.randn(n) * 0.02
        k5 = np.ones(5) / 5.0
        pert = np.convolve(np.convolve(pert, k5, 'same'), k5, 'same')
    pert -= pert.mean()
    return _repair(h + pert, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: MGD with varied configs
    for alpha, steps, tk, s0 in [(8000, 2500, 4, 0.001), (2000, 1500, 6, 0.004),
                                   (500, 1200, 8, 0.006), (30000, 1000, 3, 0.0005)]:
        if time.time() - t0 > 220:
            break
        h_t, _ = _mgd_nm(best_h.copy(), steps, alpha, tk, s0, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s - 500:
        for alpha, tk, mxi in [(5000, 8, 500), (50000, 5, 400), (200000, 4, 300)]:
            if time.time() - t0 > budget_s - 450:
                break
            h_t = _slsqp_lse(best_h.copy(), alpha, mxi, tk)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 3: Structured perturbations + MGD to escape local minima
    if time.time() - t0 < budget_s - 500:
        for trial in range(8):
            if time.time() - t0 > budget_s - 450:
                break
            h_p = _structured_perturb(best_h, n, rng, trial % 3)
            h_t, _ = _mgd_nm(h_p, 300, 3000, 6, 0.003, rng)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 4: Lag-aware local search (targeted at critical shifts)
    if time.time() - t0 < budget_s - 350:
        n_lag = min(int((budget_s - 350 - (time.time() - t0)) * 400), 60000)
        best_h, best_c5 = _lag_local(best_h, n_lag, 0.015, rng)

    # Phase 5: Sweep at critical lags
    if time.time() - t0 < budget_s - 250:
        npass = min(int((budget_s - 250 - (time.time() - t0)) * 40), 800)
        best_h, best_c5 = _sweep_lag(best_h, npass, 0.008, rng)

    # Phase 6: Lag-aware again with smaller steps
    if time.time() - t0 < budget_s - 200:
        n_lag2 = min(int((budget_s - 200 - (time.time() - t0)) * 500), 50000)
        best_h, best_c5 = _lag_local(best_h, n_lag2, 0.005, rng)

    # Phase 7: Standard 2-point with decreasing steps
    for step in [3e-4, 1e-4, 5e-5, 2e-5]:
        if time.time() - t0 > budget_s - 150:
            break
        n_it = min(int((budget_s - 150 - (time.time() - t0)) * 2500), 40000)
        if n_it < 200:
            break
        best_h, best_c5 = _local2(best_h, n_it, step, rng)

    # Phase 8: 3-point search
    if time.time() - t0 < budget_s - 100:
        n3 = min(int((budget_s - 100 - (time.time() - t0)) * 1800), 60000)
        best_h, best_c5 = _local3(best_h, n3, 3e-5, rng)

    # Phase 9: Fine lag-aware
    if time.time() - t0 < budget_s - 50:
        nfl = min(int((budget_s - 50 - (time.time() - t0)) * 600), 40000)
        best_h, best_c5 = _lag_local(best_h, nfl, 0.003, rng)

    # Phase 10: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 10:
        nfin = min(int((budget_s - 10 - (time.time() - t0)) * 5000), 150000)
        best_h, best_c5 = _local2(best_h, nfin, 1e-5, rng)

    # Phase 11: Final ultra-fine lag-aware
    if time.time() - t0 < budget_s - 2:
        nfl2 = min(int((budget_s - 2 - (time.time() - t0)) * 800), 50000)
        best_h, best_c5 = _lag_local(best_h, nfl2, 0.001, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
