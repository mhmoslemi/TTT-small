import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < 1e-14:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c)), c

def _grad_lag(k, x, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - x[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= x[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - x[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= x[i2 + sp]
    return g

def _sum_fun(x):
    return x.sum() - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k and top_k < len(c):
            cs = np.sort(c)[-top_k:]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c))
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(idx[ii], x, n)
        return g * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:

        h_new, c_new = _local_informed(best_h, ntries, rng, 4e-6)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new

    # Phase 7: Ultra-fine local search
    if _el() < budget_s - 1:
        tl = budget_s - 1 - _el()
        _t = _el()
        for _ in range(40):
            _eval(best_h)
        eps = max((_el() - _t) / 40.0, 1e-6)
        ntries = min(int(tl * (1.0 / eps) * 0.75), 10000000)
        h_new, c_new = _local_fine(best_h, ntries, rng, 5e-7)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
