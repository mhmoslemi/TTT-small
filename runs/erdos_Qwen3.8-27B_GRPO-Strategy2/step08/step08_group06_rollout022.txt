def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    # Bisection for threshold
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute remainder
        order = np.argsort(np.where(diff > 0, 1.0 - h, h))
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max())

def _grad_topk(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _adam_project(h, g, state, lr):
    n = len(h)
    m = state['m']
    v = state['v']
    step = state['t'] + 1
    b1, b2, e = 0.9, 0.999, 1e-8
    m = b1 * m + (1 - b1) * g
    v = b2 * v + (1 - b2) * g * g
    mh = m / (1 - b1 ** step)
    vh = v / (1 - b2 ** step)

    h_new = _repair(h_new, n)

def _adam_optimize(h, k_top, lr, n_iter, t_end, state=None):

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    d = h.sum() - target
    if abs(d) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    d = target - h.sum()
    if d > 1e-12:
        for idx in np.argsort(1.0 - h):
            a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            if d < 1e-12:
                break
    elif d < -1e-12:
        for idx in np.argsort(h):
            a = min(-d, h[idx]); h[idx] -= a; d += a
            if -d < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lags(h, lags):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    wk = dx / float(len(lags))
    for k in lags:
        ki = int(k)
        sh = (n - 1) - ki
        if sh >= 0:
            if n - sh > 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _adam_phase(h, k_top, lr, t_stop, m, v, t_a):
    n = len(h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best = h.copy()
    best_c = _eval(h)
    stall = 0
    while time.time() < t_stop:
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        idx = np.argsort(corr)[-k_top:]
        g = _grad_lags(h, idx)
        t_a += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mh = m / (1.0 - b1 ** t_a)
        vh = v / (1.0 - b2 ** t_a)
        h = _repair(h - lr * mh / (np.sqrt(vh) + eps), n)
        c = _eval(h)
        if c < best_c - 1e-13:
            best_c = c
            best = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                if lr < 1e-7:
                    m[:] = 0.0
                    v[:] = 0.0
                    t_a = 0
                    lr = lr_init_val
                stall = 0
    return best, best_c, m, v, t_a, lr

lr_init_val = 0.003

def _local_3pt(h, rng, t_stop):
    n = len(h)
    best = h.copy()
    best_c = _eval(h)
    scale = 5e-5
    while time.time() < t_stop:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        k = (i + 13 + rng.randint(0, 30)) % n
        if i == j or i == k or j == k:
            continue
        d = scale * rng.randn()
        h[i] += d
        h[j] += d
        h[k] -= 2.0 * d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1 or h[k] < 0 or h[k] > 1:
            h[i] -= d; h[j] -= d; h[k] += 2.0 * d

        best_h, best_c, m, v, t_a, _ = _adam_phase(best_h, 150, 0.003, t_stop, m, v, t_a)

        h_pert = _repair(h_pert, n)

        if c_pert < best_c + 0.008:

            if mid_c < best_c:

        lr_init_val = 0.001
        best_h, best_c, m, v, t_a, _ = _adam_phase(best_h, 300, 0.001, t_stop, m, v, t_a)
