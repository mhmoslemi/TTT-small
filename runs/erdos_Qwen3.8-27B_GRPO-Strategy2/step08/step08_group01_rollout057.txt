            continue
        k = mi - (n - 1)
        if k >= 0:
            term1 = np.zeros(n)
            term1[k:] = 1.0 - h[:n - k]
            term2 = np.zeros(n)
            term2[:n - k] = h[k:]
            g += w[ii] * dx * (term1 - term2)
        else:
            p = -k
            term1 = np.zeros(n)
            term1[:n - p] = 1.0 - h[p:]
            term2 = np.zeros(n)
            term2[p:] = h[:n - p]
            g += w[ii] * dx * (term1 - term2)
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.75
    step = step0
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for it in range(n_iter):
        g = _grad_topk(h, alpha, top_k)
        v = mu * v + (1.0 - mu) * (-step * g)
        h_try = _proj_target(h + v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.65
            step = max(step, step0 * 0.05)
    return _repair(best_h, n), best_c

def _lp_refine(h, n_lp_iters, top_fraction=0.3, blend=0.5):
    """Linearized min-max LP refinement."""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for it in range(n_lp_iters):
        c = _eval_vec(h)
        m = 2 * n - 1
        # Select top shifts
        thresh = np.percentile(c, (1 - top_fraction) * 100)
        top_mask = c >= thresh
        top_idx = np.where(top_mask)[0]
        if len(top_idx) < 5:
            top_idx = np.argsort(c)[-10:]
        
        # Compute gradients for top shifts
        G = np.zeros((len(top_idx), n))
        for ii, mi in enumerate(top_idx):
            k = mi - (n - 1)
            if k >= 0:
                t1 = np.zeros(n); t1[k:] = 1.0 - h[:n-k]
                t2 = np.zeros(n); t2[:n-k] = h[k:]
                G[ii] = (2.0/n) * (t1 - t2)
            else:
                p = -k
                t1 = np.zeros(n); t1[:n-p] = 1.0 - h[p:]
                t2 = np.zeros(n); t2[p:] = h[:n-p]
                G[ii] = (2.0/n) * (t1 - t2)
        
        # LP: min t s.t. c[top] + G@(h_new - h) <= t, 0<=h_new<=1, sum(h_new)=n/2
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for ii in range(len(top_idx)):
            rhs = c[top_idx[ii]] + G[ii] @ h  # c_k(h*) + g_k·h* - g_k·h* = c_k(h*)... 
            # Actually: c_k(h*) + G[ii]@(h_new - h) <= t
            # => G[ii]@h_new <= t - c_k(h*) + G[ii]@h
            # Wait: c_k(h*) is the current value. The linearization is:
            # c_k(h_new) ≈ c_k(h*) + G[ii] · (h_new - h)
            # So: c_k(h*) + G[ii]@h_new - G[ii]@h <= t
            # => G[ii]@h_new <= t - c_k(h*) + G[ii]@h
            rhs_const = -c[top_idx[ii]] + G[ii] @ h
            constraints.append(G[ii] @ h_var <= t_var + rhs_const)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0, 1)
        h_new = _repair(blend * h_lp + (1 - blend) * h, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _local2(h, n_iter, step_hover, rng, sa_temp=None):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = step_hover * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp is not None and sa_temp > 1e-10:
                if rng.rand() < np.exp(-(c_new - cur_c) / sa_temp):
                    accept = True
            if accept:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
