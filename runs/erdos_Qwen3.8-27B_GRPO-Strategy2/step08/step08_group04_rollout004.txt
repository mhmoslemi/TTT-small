    h = np.clip(h, 0.0, 1.0).copy()

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk, lam):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        ct = np.partition(c, -topk)[-topk:]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    obj = m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    sd = h.sum() - n * 0.5
    obj += lam * sd * sd
    return obj

def _smax_grad(h, alpha, topk, lam):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        idx.sort()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    sd = h.sum() - n * 0.5
    g += 2.0 * lam * sd
    return g

def _lbfgs(h, alpha, topk, lam, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(np.clip(x, 0, 1), alpha, topk, lam)
    def jac(x):
        return _smax_grad(np.clip(x, 0, 1), alpha, topk, lam)
    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-25, 'gtol': 1e-14,
                              'maxls': 30, 'maxfun': maxit * 40})
        return _repair(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def _nesterov(h, alpha0, grow, topk, steps, step0, lam, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk, lam)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 150 == 0:
            alpha = min(alpha * grow, 1e8)
        g = _smax_grad(h, alpha, topk, lam)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk, lam)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.4
            hn = _proj_sum(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk, lam)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.4
                hn = _proj_sum(h - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk, lam)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.006 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _surgical(h, deadline, nstep):
    """Target the single worst lag with projected gradient steps."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 0.001
    for _ in range(nstep):
        if time.time() > deadline:
            break
        c = _correl(h)
        k_star = np.argmax(c)
        g = _grad_lag(h, k_star, 2.0/n)
        # Project: subtract mean to maintain sum
        g -= g.mean()
        hn = _proj_sum(h - step * g, n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c - 1e-18:
            h = hn
            c5c = c5n
            if c5c < bc:
                bc = c5c
                bh = h.copy()
            step = min(step * 1.02, 0.01)
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return bh, bc

            hn = _proj_sum(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk, lam)
            if fn < f:
                h = hn
                f = fn
            else:
                break
    return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.006 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _surgical(h, deadline, nstep):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 0.002
    for _ in range(nstep):
        if time.time() > deadline:
            break
        c = _correl(h)
        k_star = int(np.argmax(c))
        g = _grad_lag(h, k_star, 2.0 / n)
        g -= g.mean()
        hn = _proj_sum(h - step * g, n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c - 1e-18:
            h = hn
            c5c = c5n
            if c5c < bc:
                bc = c5c
                bh = h.copy()
            step = min(step * 1.03, 0.01)
        else:
            step *= 0.4
            if step < 1e-11:
                break
    return bh, bc

def _multi_restart(h, rng, alpha, topk, lam, maxit, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for trial in range(4):
        if time.time() > deadline:
            break
        h_p = best_h + rng.randn(n) * 0.003
        h_p = _repair(h_p, n)
        if _eval(h_p) > best_c5 + 0.005:
            continue
        h_r = _lbfgs(h_p, alpha, topk, lam, maxit, deadline)
        c5r = _eval(h_r)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    dl1 = min(deadline, t0 + 100)
    for alpha, tk, lam, mi in [(1000, 30, 200, 1200), (5000, 15, 500, 1000), (20000, 8, 1000, 800)]:
        if time.time() > dl1:
            break
        h = _lbfgs(h, alpha, tk, lam, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    dl2 = min(deadline, t0 + 260)
    for alpha, tk, lam, mi in [(100000, 5, 1000, 800), (500000, 3, 2000, 700), (2000000, 2, 2000, 600)]:
        if time.time() > dl2:
            break
        h = _lbfgs(best_h.copy(), alpha, tk, lam, mi, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    dl3 = min(deadline, t0 + 420)
    for a0, gr, tk, st, ns in [(5000, 2.0, 10, 0.004, 2500), (50000, 1.5, 5, 0.002, 2500), (500000, 1.3, 3, 0.001, 2000)]:
        if time.time() > dl3:
            break
        hw = _nesterov(best_h.copy(), a0, gr, tk, ns, st, 500, dl3)
        c5v = _eval(hw)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hw.copy()

    dl4 = min(deadline, t0 + 580)
    best_h, best_c5 = _multi_restart(best_h, rng, 200000, 4, 2000, 600, dl4)

    dl5 = min(deadline, t0 + 720)
    blocks = [1, 2, 3, 5, 8, 13, 21]
    ha, ca = _anneal(best_h.copy(), 45000, rng, dl5, 0.008, 1e-13, blocks)
    if ca < best_c5:
        best_c5 = ca
        best_h = ha.copy()

    dl6 = min(deadline, t0 + 840)
    for _ in range(6):
        if time.time() > dl6:
            break
        hs, cs = _surgical(best_h.copy(), dl6, 4000)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()

    dl7 = min(deadline, t0 + 970)
    h = _lbfgs(best_h.copy(), 2000000, 2, 3000, 800, dl7)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()
    h = _nesterov(best_h.copy(), 5000000, 1.1, 2, 2500, 0.0003, 3000, dl7)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    dl8 = deadline
    nc = len(best_h)
    c5c = best_c5
    bh = best_h.copy()
    ss = 3e-6
    for it in range(400000):
        if time.time() > dl8:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 15)) % nc
        d = ss * rng.randn()
        ni, nj = bh[i] + d, bh[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            bh[i], bh[j] = ni, nj
            c5n = _eval(bh)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = bh.copy()
            else:
                bh[i], bh[j] = ni - d, nj + d
                ss = max(ss * 0.985, 1e-10)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
