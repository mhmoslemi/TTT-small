import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, k_top):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    nk = min(k_top, len(c))
    if nk < len(c):
        top_k = np.argsort(c)[-nk:]
    else:
        top_k = np.arange(len(c))
    g = np.zeros(n)
    dx2 = 2.0 / n
    wk = dx2 / nk
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

class _Ctx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        return _lse_obj(x, self.alpha, self.n)
    def grad(self, x):
        return _lse_grad(x, self.alpha, self.n)
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp_run(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _Ctx(alpha, n, target)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _fista(h, k_start, k_end, steps, step0, t_end):
    n = len(h)
    f = _eval(h)
    x = h.copy()
    z = h.copy()
    t_k = 1.0
    for s in range(steps):
        if time.time() > t_end:
            break
        frac = s / max(steps - 1, 1)
        k_cur = max(3, int(k_start + (k_end - k_start) * frac))
        g = _topk_grad(z, k_cur)
        step = step0 / (1.0 + 0.03 * s) ** 0.6
        x_new = _repair(z - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        z_new = x_new + ((t_k - 1.0) / t_new) * (x_new - z)
        z_new = _repair(z_new, n)
        t_k = t_new
        f_new = _eval(x_new)
        if f_new < f:
            f = f_new
        x, z = x_new, z_new
    return _repair(x, n)

def _cvx_qp(h, target_c, n_constraints, t_end):
    try:
        import cvxpy as cp
        n = len(h)
        h0 = np.array(h, dtype=float)
        c = np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / n)
        top_idx = np.argsort(c)[-n_constraints:]
        var = cp.Variable(n)
        constraints = [var >= 0, var <= 1, cp.sum(var) == n / 2.0]
        for k in top_idx:
            sh = (n - 1) - k
            if sh >= 0:
                lin = cp.sum(h0[sh:] * (1 - var[:n - sh])) - 0.0
            else:
                sp = -sh
                lin = cp.sum(h0[:n - sp] * (1 - var[sp:])) - 0.0
            constraints.append(lin <= target_c / (2.0 / n))
        obj = cp.Minimize(cp.sum_squares(var - h0))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CLARABEL, max_time=30)
        if var.value is not None and np.isfinite(var.value).all():
            return _repair(var.value, n)
    except Exception:
        pass
    return h

def _local_sa(h, n_tries, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    T = 2e-5
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        k = (i + 5 + rng.randint(0, 10)) % n
        d1 = T * rng.randn()
        d2 = T * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c or rng.rand() < np.exp(-(c5n - c5c) / (T * 0.3)):
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
        T *= 0.9998
    return best_h, best_c

def _local_2pt(h, n_tries, rng, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _it in range(n_tries):

        j = (i + 1 + rng.randint(0, 3)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d

            c5n = _eval(h)

                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
