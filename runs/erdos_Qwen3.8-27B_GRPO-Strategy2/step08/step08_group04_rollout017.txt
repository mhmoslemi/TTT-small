import time

        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:

        if abs(diff) > 1e-12:

def _proj(h, target):

        if sh == 0:

    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]

        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)

    c = _correl(h)
    m = c.max()

        idx = np.arange(len(c))
        ct = c

    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:

                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return h

def _anneal(h, niter, rng, deadline, T0=0.01, T1=1e-9, blocks=None):

    if blocks is None:
        blocks = [1, 2, 3, 5, 8, 13, 21]

    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))

            if c5n < bc:

    return bh, bc

def _fine_polish(h, niter, rng, deadline):

    bh = h.copy()
    bc = c5c
    step = 5e-6

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh == 0:
        g[:] = 1.0 - 2.0 * h
    elif sh > 0:
        g[sh:] += (1.0 - h[:n - sh])
        g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c.copy()
        ct = ct[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)

    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline, grow=2.0):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    beta = 0.82
    for it in range(steps):
        if time.time() > deadline:
            break

        g = _grad_smax(h, alpha, topk)

    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return h

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return np.array([h[i*factor:(i+1)*factor].mean() for i in range(m)])

def _upsample_linear(h, n_out):
    return np.interp(np.linspace(0, n_out-1, n_out), np.linspace(0, len(h)-1, len(h)), h)

def _anneal(h, niter, rng, deadline, blocks=None):
    n = len(h)
    if blocks is None:
        blocks = [1, 2, 3, 5, 8, 13, 21]
    T0, T1 = 0.006, 1e-10
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5c, h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    step = 4e-6
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 15)) % nc
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5c, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.98, 1e-9)
        else:
            step = max(step * 0.94, 1e-9)
    return bh, bc

def _systematic_polish(h, deadline, max_sweeps=5):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j_off in [1, 2, 3, 5]:
                j = (i + j_off) % n
                if j == i:
                    continue
                for d in [1e-5, -1e-5, 3e-6, -3e-6]:
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < c5c - 1e-18:
                            c5c = c5n
                            if c5n < bc:
                                bc, bh = c5c, h.copy()
                            break
                        h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    n0 = len(initial_h_values)
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h_init.copy(), _eval(h_init)

    # Phase 1: Quick SLSQP (0-70s)
    dl1 = min(deadline, t0 + 70)
    for alpha, tk, mi in [(5000, 40, 400), (20000, 15, 300)]:
        if time.time() > dl1:
            break
        h1 = _slsqp(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h1)
        if c5v < best_c5:
            best_c5, best_h = c5v, h1.copy()

    # Phase 2: Multi-resolution (70-250s)
    dl2 = min(deadline, t0 + 250)
    for factor in [2, 3]:
        if time.time() > dl2:
            break
        n_lo = n0 // factor
        if n_lo < 80:
            continue
        h_lo = _repair(_downsample(best_h, factor), n_lo)
        for a, tk, mi in [(3000, 20, 300), (15000, 10, 250)]:
            if time.time() > dl2:
                break
            h_lo = _slsqp(h_lo, a, tk, mi, dl2)
        h_hi = _upsample_linear(h_lo, n0)
        h_hi = _repair(np.clip(h_hi, 0, 1), n0)
        h_hi = _slsqp(h_hi, 10000, 15, 200, dl2)
        c5v = _eval(h_hi)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_hi.copy()

    # Phase 3: Nesterov multi-start (250-550s)
    dl3 = min(deadline, t0 + 550)
    candidates = [best_h.copy()]
    # Periodic perturbation
    x = np.arange(n0)
    for per in [28, 42]:
        hp = 0.5 + 0.25 * np.sin(2 * np.pi * x / per + rng.uniform(0, 6)) + 0.1 * rng.randn(n0) * 0.1
        hp = _repair(hp, n0)
        hp = _slsqp(hp, 5000, 30, 200, dl3)
        if _eval(hp) < 0.40:
            candidates.append(hp)
    # Random perturbation of best
    h_p = _repair(best_h + 0.03 * rng.randn(n0), n0)
    candidates.append(h_p)

    for h_c in candidates:
        if time.time() > dl3:
            break
        for a0, gr, tk, st, ns in [(5000, 2.0, 25, 0.005, 5000),
                                    (50000, 1.5, 10, 0.003, 3000)]:
            if time.time() > dl3:
                break
            h_c = _nesterov(h_c, a0, tk, ns, st, dl3, grow=gr)
            c5v = _eval(h_c)
            if c5v < best_c5:
                best_c5, best_h = c5v, h_c.copy()

    # Phase 4: Intensive SLSQP (550-820s)
    dl4 = min(deadline, t0 + 820)
    for alpha, tk, mi in [(30000, 8, 500), (80000, 5, 400), (200000, 4, 350), (500000, 3, 300)]:
        if time.time() > dl4:
            break
        h4 = _slsqp(best_h.copy(), alpha, tk, mi, dl4)
        c5v = _eval(h4)
        if c5v < best_c5:
            best_c5, best_h = c5v, h4.copy()

    # Phase 5: Annealing (820-960s)
    dl5 = min(deadline, t0 + 960)
    h_a, c5_a = _anneal(best_h.copy(), 50000, rng, dl5, blocks=[1, 2, 3, 5, 8, 13])
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a.copy()

    # Phase 6: Final SLSQP + systematic polish (960-end)
    dl6 = deadline
    h6 = _slsqp(best_h.copy(), 200000, 3, 300, dl6)
    c5v = _eval(h6)
    if c5v < best_c5:
        best_c5, best_h = c5v, h6.copy()

    rem = max(0, dl6 - time.time())
    if rem > 3:
        best_h, best_c5 = _systematic_polish(best_h.copy(), dl6, max_sweeps=3)

    rem2 = max(0, dl6 - time.time())
    n_pol = min(int(rem2 / 0.002), 30000)
    if n_pol > 100:
        best_h, best_c5 = _fine_polish(best_h.copy(), n_pol, rng, dl6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
