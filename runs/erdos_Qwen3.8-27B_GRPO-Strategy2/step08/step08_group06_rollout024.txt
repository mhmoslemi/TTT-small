from scipy.optimize import minimize
import time

_SLP = {}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - tgt) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > tgt:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        d = tgt - h.sum()
        if abs(d) > 1e-12:
            idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in idx:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    a, n = _SLP['a'], _SLP['n']
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / a

def _soft_grad(x):
    a, n = _SLP['a'], _SLP['n']
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k; wk = w[k] * dx2
        if sh >= 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slp_cf(x):
    return x.sum() - _SLP['t']

def _slp_cj(x):
    return np.ones(_SLP['n'])

def _slsqp(h, alpha, niter, ftol, target):
    _SLP['a'], _SLP['n'], _SLP['t'] = alpha, len(h), target
    n = len(h)
    bnds = [(0.0, 1.0)] * n

        r = minimize(_soft_obj, h, jac=_soft_grad, method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': niter, 'ftol': ftol})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _tkgrad(h, kt, bl, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    nk = 2 * n - 1
    w = np.zeros(nk)
    if kt >= nk:

        w = np.exp(s - s.max()); w /= w.sum()
    else:

        ct = c[ix]
        s = np.clip(bl * (ct - ct.max()), -500, 500)

    return g
