import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)

        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_pg(h, alpha, ml, steps, lr0, momentum):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    lr = lr0
    for it in range(steps):
        g = _smooth_grad(h, alpha, ml)
        v = momentum * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        taken = False
        step = lr
        for _t in range(15):
            h_try = _proj(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < _eval(h):
                h = h_try
                v = momentum * v
                lr = min(step * 1.1, lr0 * 20.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            h = h_new
            lr = min(lr * 1.05, lr0 * 20.0)
    return best_h

def _grad_local_search(h, n_iter, rng, alpha, ml):
    n = len(h)
    c5_cur = _eval(h)
    best_c = c5_cur
    best_h = h.copy()
    g_cache = _smooth_grad(h, alpha, ml)
    for _ in range(n_iter):

        h_new = h.copy()

        h_new[j] -= d
        if h_new.min() < 0 or h_new.max() > 1:

            h_new = h.copy()
            h_new[i] += d
            h_new[j] -= d
            if h_new.min() < 0 or h_new.max() > 1:

        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5_cur:
            h = h_new
            c5_cur = c5n
            g_cache = _smooth_grad(h, alpha, ml)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)
