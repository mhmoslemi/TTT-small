    when h[ci] += t and h[cj] -= t."""

        d = 0.0  # coefficient for h[cj] -= t (we'll add -t * d)
        if shift >= 0:
            if shift <= ci < n: c += (1.0 - h[ci-shift])
            if 0 <= ci < n-shift: c += -h[ci+shift]
            if shift <= cj < n: d += (1.0 - h[cj-shift])
            if 0 <= cj < n-shift: d += -h[cj+shift]

            if tt <= cj < n: d += (1.0 - h[cj-tt])
            if 0 <= cj < n-tt: d += -h[cj+tt]

def _min_t(c, coeff, t_lo, t_hi):
    """Find t in [t_lo, t_hi] minimizing max(c + t*coeff)"""
    # This is a 1D convex minimization (max of linear functions is convex)
    # Use ternary search
    for _ in range(50):
        m1 = t_lo + (t_hi - t_lo) / 3
        m2 = t_hi - (t_hi - t_lo) / 3
        f1 = np.max(c + m1 * coeff)
        f2 = np.max(c + m2 * coeff)
        if f1 < f2:
            t_hi = m2
        else:
            t_lo = m1
    return 0.5 * (t_lo + t_hi)

```python
import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _grad(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for ki in active:
        shift = ki - (n - 1)
        if shift >= 0:
            g[shift:] += (1.0 - h[:n-shift])
            g[:n-shift] -= h[shift:]
        else:
            t = -shift
            g[t:] += (1.0 - h[:n-t])
            g[:n-t] -= h[t:]
    g *= dx / len(active)
    return g, m

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < 2*n-1:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(2*n-1)
        ct = c
    s = np.clip(alpha * (ct - m), -300, 300)
    smax = s.max()
    w = np.exp(s - smax)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        ki = idx[ii]
        shift = ki - (n-1)
        wk = w[ii] * dx
        if shift >= 0:
            g[shift:] += wk * (1.0 - h[:n-shift])
            g[:n-shift] -= wk * h[shift:]
        else:
            t = -shift
            g[t:] += wk * (1.0 - h[:n-t])
            g[:n-t] -= wk * h[t:]
    return g, m

def _nesterov(h, nsteps, step0, deadline, tol=0.003, beta=0.85, smax_alpha=0, smax_topk=10):
    n = len(h)
    target = 0.5 * n
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_f, best_h = f, h.copy()
    for _ in range(nsteps):
        if time.time() > deadline:
            break
        if smax_alpha > 0:
            g, _ = _smax_grad(h, smax_alpha, smax_topk)
        else:
            g, _ = _grad(h, tol)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
