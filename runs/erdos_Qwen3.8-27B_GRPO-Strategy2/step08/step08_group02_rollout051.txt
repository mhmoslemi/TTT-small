import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    """Gradient of the k-th lag correlation w.r.t. h"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        if sh < n:
            i1 = np.arange(n - sh)
            g[i1] += (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        if sp < n:
            i1 = np.arange(sp, n)
            g[i1] += (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= h[i2 + sp]
    return g * dx

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break

        for idx in order[:target - n_ones]:
            h_bin[idx] = 1.0

    """Simulated annealing with binary swaps"""
    n = len(h)
    cur = h.copy()
    cur_c = _eval(cur)
    best = cur.copy()
    best_c = cur_c
    end_time = t_start + duration
    T = t_start  # current temperature
    
    while time.time() < end_time:
        # Update temperature (geometric cooling)
        frac = (time.time() - t_start) / duration
        T = t_start * (t_end / t_start) ** min(frac, 1.0)
        
        # Pick swap
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        else:

                    cur_c = c_test
                else:
                    cur[ai], cur[bi] = 1.0, 0.0
    
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _to_binary(h, n):
    hb = (h > 0.5).astype(np.float64)
    target = n // 2
    no = int(hb.sum())
    if no > target:
        oi = np.where(hb == 1.0)[0]
        for idx in oi[np.argsort(h[oi])][:no - target]:
            hb[idx] = 0.0
    elif no < target:
        zi = np.where(hb == 0.0)[0]
        for idx in zi[np.argsort(-h[zi])][:target - no]:
            hb[idx] = 1.0
    return hb

def _sa_binary(h, rng, T0, T1, duration):
    n = len(h)
    cur = h.copy()
    cur_c = _eval(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    decay = (T1 / T0) / max(duration, 1.0)
    while time.time() - start < duration:
        T = T0 * np.exp(decay * (time.time() - start))
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = 0.0, 1.0
        c_new = _eval(cur)
        if c_new <= cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best = cur.copy()
        elif rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-12)):
            cur_c = c_new
        else:
            cur[i], cur[j] = 1.0, 0.0
    return best, best_c

def _hill_climb(h, rng, duration):
    n = len(h)
    cur = h.copy()
    cur_c = _eval(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    while time.time() - start < duration:
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = 0.0, 1.0
        c_new = _eval(cur)
        if c_new < cur_c:
            cur_c = c_new
            best = cur.copy()
            best_c = cur_c
        else:
            cur[i], cur[j] = 1.0, 0.0
    return best, best_c

def _slsqp_quick(h, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _eval(x), h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cont_polish(h, rng, duration):
    n = len(h)
    cur = h.copy()
    cur_c = _eval(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    step = 2e-4
    while time.time() - start < duration:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = cur[i] - d, cur[j] + d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            cur[i], cur[j] = ni, nj
            c_new = _eval(cur)
            if c_new < cur_c - 1e-16:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best = cur.copy()
            else:
                cur[i], cur[j] = ni + d, nj - d
        if (int(time.time()) - int(start)) % 30 == 0:
            step *= 0.7
    return _repair(best, n), best_c
