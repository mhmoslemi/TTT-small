        top_k = np.argsort(c)[-nfocus:]
        k = top_k[rng.randint(nfocus)]
        sh = (n - 1) - k
        if sh >= 0:
            i = rng.randint(0, n - sh)
            j = i + sh
        else:
            sp = -sh
            i = rng.randint(0, n - sp)
            j = i + sp
        d = rng.randn() * 8e-6
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            oi, oj = best_h[i], best_h[j]
            best_h[i], best_h[j] = ni, nj
            c_new = _eval(best_h)
            if c_new < best_c:
                best_c = c_new
            else:
                best_h[i], best_h[j] = oi, oj
    return best_h, best_c

def _3pt_fine(h, n_tries, rng, t_end):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        c = _corr(best_h)
        top_k = np.argsort(c)[-8:]
        k = top_k[rng.randint(8)]
        sh = (n - 1) - k
        if sh >= 0 and n - sh > 2:
            i = rng.randint(0, n - sh - 1)
            j = i + sh
            m = (j + rng.randint(1, 5)) % n
        elif sh < 0 and n + sh > 2:
            sp = -sh
            i = rng.randint(0, n - sp - 1)
            j = i + sp
            m = (j + rng.randint(1, 5)) % n
        else:
            i = rng.randint(n)
            j = (i + 3) % n
            m = (i + 7) % n
        if i == j or i == m or j == m:
            continue
        d1, d2 = 5e-6 * rng.randn(), 5e-6 * rng.randn()
        ni, nj, nm = best_h[i]+d1, best_h[j]+d2, best_h[m]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nm <= 1:
            oi, oj, om = best_h[i], best_h[j], best_h[m]
            best_h[i], best_h[j], best_h[m] = ni, nj, nm
            c_new = _eval(best_h)
            if c_new < best_c:
                best_c = c_new
            else:
                best_h[i], best_h[j], best_h[m] = oi, oj, om
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def update(h_new):
        if _eval(h_new) < best_c5:
            return h_new, best_c5
        return None, best_c5

    # Phase 1: SLSQP coarse (12% budget)
    t1 = t0 + budget_s * 0.12
    for alpha in [1500, 15000]:
        if time.time() > t1 - 5:
            break
        hc = _slsqp(best_h.copy(), alpha, 1500, 1e-15, target)
        cv = _eval(hc)
        if cv < best_c5:
            best_c5, best_h = cv, hc

    # Phase 2: Subgradient descent top-k (28% budget)
    t2 = t0 + budget_s * 0.40
    for nk, stp, mu in [(5, 0.0012, 0.75), (20, 0.0008, 0.82), (60, 0.0005, 0.88)]:
        if time.time() > t2 - 5:
            break
        h_cand, _ = _proj_subgrad(best_h.copy(), nk, 6000, stp, t2, mu)
        cv = _eval(h_cand)
        if cv < best_c5:
            best_c5, best_h = cv, h_cand

    # Phase 3: SLSQP high-alpha (20% budget)
    t3 = t0 + budget_s * 0.60
    for alpha in [200000, 2000000, 20000000]:
        if time.time() > t3 - 10:
            break
        h_cand = _slsqp(best_h.copy(), alpha, 1200, 1e-17, target)
        cv = _eval(h_cand)
        if cv < best_c5:
            best_c5, best_h = cv, h_cand

    # Phase 4: Fine subgradient (15% budget)
    t4 = t0 + budget_s * 0.75
    h_cand, _ = _proj_subgrad(best_h.copy(), 8, 8000, 0.0003, t4, 0.90)
    cv = _eval(h_cand)
    if cv < best_c5:
        best_c5, best_h = cv, h_cand

    # Phase 5: Targeted 2-point descent at critical shifts (10% budget)
    t5 = t0 + budget_s * 0.87
    nt = min(int((t5 - time.time()) / 0.003), 25000)
    best_h, best_c5 = _targeted_descent(best_h, max(nt, 1000), rng, t5, 6)

    # Phase 6: 3-point fine search (5% budget)
    t6 = t0 + budget_s * 0.94
    nt6 = min(int((t6 - time.time()) / 0.003), 20000)
    best_h, best_c5 = _3pt_fine(best_h, max(nt6, 500), rng, t6)

    # Phase 7: Ultra-fine 2-point on top-3 shifts
    if time.time() < t_global_end - 3:
        c_final = _corr(best_h)
        top3 = np.argsort(c_final)[-3:]
        n_tries = min(int((t_global_end - 3 - time.time()) / 0.002), 30000)
        for _ in range(n_tries):
            if time.time() > t_global_end - 1:
                break
            k = top3[rng.randint(3)]
            sh = (n - 1) - k
            if sh >= 0:
                i = rng.randint(0, n - sh)
                j = i + sh
            else:
                sp = -sh
                i = rng.randint(0, n - sp)
                j = i + sp
            d = 3e-7 * rng.randn()
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                oi, oj = best_h[i], best_h[j]
                best_h[i], best_h[j] = ni, nj
                cnn = _eval(best_h)
                if cnn < best_c5:
                    best_c5 = cnn
                else:
                    best_h[i], best_h[j] = oi, oj

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
