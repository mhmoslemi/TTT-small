    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _lag_grad(h, k, n):
    """Gradient of c_k with respect to h, where k is the index into full correlation"""
    dx = 2.0 / n
    sh = (n - 1) - k  # shift
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _slsqp_alpha(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c))
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _lag_grad(x, idx[ii], n)
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h

def _targeted_local(h, n_iter, rng, steps):
    """Targeted local search: perturb coordinates that contribute most to worst lags"""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    h_work = h.copy()
    cur_c = best_c
    
    for it in range(n_iter):
        c = np.correlate(h_work, 1.0 - h_work, mode='full') * (2.0/n)
        # Top 5 worst lags
        top_idx = np.argsort(c)[-5:]
        # Compute combined gradient
        g = sum(_lag_grad(h_work, k, n) for k in top_idx)
        
        # Find top contributing coordinates
        step = steps[min(it // 1000, len(steps)-1)]
        # Pick 2 random coordinates from top |g| contributors
        top_coords = np.argsort(-np.abs(g))[:20]
        i = top_coords[rng.randint(20)]
        j = top_coords[rng.randint(20)]
        if i == j:
            j = (j + 1) % n
        
        # Determine direction
        d = step * rng.randn()
        if g[i] > g[j]:
            # Move from high grad to low grad
            di, dj = -d, d
        else:
            di, dj = d, -d
        
        ni, nj = h_work[i] + di, h_work[j] + dj
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_work[i], h_work[j] = ni, nj
            c_new = _eval(h_work)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_work.copy()
            else:
                h_work[i], h_work[j] = ni - di, nj - dj
    
    return _repair(best_h, n), best_c

def _proj_clip_sum(h, target):
    """Project h to [0,1]^n with sum = target using bisection on threshold"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pg_dynamics(h, alpha, n_steps, step0, top_k, momentum, n):
    """Projected gradient with dynamic step"""
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    h_work = h.copy()
    
    for it in range(n_steps):
        c = np.correlate(h_work, 1.0 - h_work, mode='full') * (2.0/n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _lag_grad(h_work, active[ii], n)
        
        v = momentum * v + (1.0 - momentum) * g
        step_actual = step
        taken = False
        for _ in range(12):
            h_new = _proj_clip_sum(h_work - step_actual * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h_work = h_new
                cur_c = c_new
                step_actual = min(step_actual * 1.15, step0 * 20.0)
                taken = True
                break
            step_actual *= 0.5
        if not taken:
            step_actual = min(step_actual * 1.3, step0 * 20.0)
        step = step_actual
        if it % 500 == 0:
            v *= 0.5
    
    return _repair(h_work, n), cur_c

        else:
            order = np.argsort(h)
        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a

    """Project h onto {x: 0<=x<=1, sum(x)=target}"""

        else:

def _grad_lag(h, k, n):

    dx = 2.0 / n
    sh = (n - 1) - k  # shift in terms of h indices
    g = np.zeros(n)

        a = n - sh
        i1 = np.arange(a)
        g[:a] += (1.0 - h[sh:sh+a])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[sh:] -= h[:n-sh]

                    best_c = cn; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj - d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    def el():
        return time.time() - t0

    # Phase 1: Quick SLSQP (20s)
    for alpha, tk, mit in [(50000, 3, 200), (500000, 2, 150)]:
        if el() > 20:
            break
        _ST = n * 0.5
        hn = _gslq(best_h, alpha, mit, tk)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5 = cn; best_h = hn.copy()

    # Phase 2: Projected gradient (200s)
    pg = [(5000, 300, 0.004, 5, 0.7), (50000, 300, 0.002, 4, 0.8),
          (500000, 250, 0.001, 3, 0.85), (5000000, 200, 0.0004, 2, 0.9)]
    for al, st, s0, tk, _ in pg:
        if el() > 230:
            break
        hn, cn = _pg(best_h.copy(), al, st, s0, tk, n)
        cn2 = _eval(hn)
        if cn2 < best_c5:
            best_c5 = cn2; best_h = hn.copy()

    # Phase 3: Targeted local search (500s)
    if el() < budget_s - 350:
        for ss in [2e-5, 1e-5, 5e-6]:
            if el() > budget_s - 300:
                break
            hn, cn = _tlocal(best_h.copy(), 80000, rng, ss, False)
            if cn < best_c5:
                best_c5 = cn; best_h = hn.copy()

    # Phase 4: Second SLSQP (40s)
    if el() < budget_s - 120:
        for alpha, tk, mit in [(2000000, 2, 150), (20000000, 1, 100)]:
            if el() > budget_s - 100:
                break
            _ST = n * 0.5
            hn = _gslq(best_h, alpha, mit, tk)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5 = cn; best_h = hn.copy()

    # Phase 5: Ultra-fine random local (remaining - 5s)
    if el() < budget_s - 5:
        rem = budget_s - 5 - el()
        _t = el()
        for _ in range(40):
            _eval(best_h)
        eps = max((el() - _t) / 40.0, 1e-6)
        ntries = min(int(rem / eps * 0.7), 5000000)
        for ss in [2e-6, 1e-6, 4e-7]:
            hn, cn = _rlocal(best_h, ntries // 3, rng, ss)
            if cn < best_c5:
                best_c5 = cn; best_h = hn.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
