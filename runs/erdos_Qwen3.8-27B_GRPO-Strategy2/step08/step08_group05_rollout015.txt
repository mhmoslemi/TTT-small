def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

        m2 = (i2 >= 0) & (i2 < n)
        if m2.any():
            grad[k, m2] -= h[i2[m2]]
    return grad

    n = len(h)

        A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0

            break

        improved = False
        for step in [1.0, 0.5, 0.25, 0.125]:

            if c5_new < best_c5:
                h, best_h, best_c5 = h_new, h_new.copy(), c5_new
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smmax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        j = np.arange(n)
        i1 = j - sh
        m1 = (i1 >= 0) & (i1 < n)
        if m1.any():
            g[m1] += wk * (1.0 - h[i1[m1]])
        i2 = j + sh
        m2 = (i2 >= 0) & (i2 < n)
        if m2.any():
            g[m2] -= wk * h[i2[m2]]
    return g

def _proj_box(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smmax(h, alpha, topk)
        hv = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj_box(hv - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 4.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_box(hv - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp_phase(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smmax(x, alpha, topk)
    def cfun(x):
        return x.sum() - n * 0.5
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.005 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    sss = [5e-7, 2e-7, 1e-7, 5e-8]
    per = max(1, niter // len(sss))
    for ss in sss:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 3)) % nc
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 6
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP refinement
    dl1 = min(deadline, t0 + 80)
    for a, tk, mi in [(2000, 50, 300), (8000, 25, 250), (30000, 12, 200)]:
        if time.time() > dl1:
            break
        hv = _slsqp_phase(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: Full LP descent (main solver)
    dl2 = min(deadline, t0 + 480)
    for eps_val in [0.008, 0.005, 0.003, 0.002, 0.001]:
        if time.time() > dl2:
            break
        hv, cv = _lp_full(best_h.copy(), 20000, eps_val, dl2)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov with p-norm equivalent
    dl3 = min(deadline, t0 + 680)
    for a, tk, st, ns in [(30000, 12, 0.003, 2000), (200000, 6, 0.001, 2000)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: Annealing to escape local minima
    dl4 = min(deadline, t0 + 850)
    blocks = [1, 2, 3, 4, 5, 6, 8, 12]
    hv, cv = _anneal(best_h.copy(), 40000, rng, dl4, 0.004, 1e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        if time.time() < dl4:
            hv2, cv2 = _lp_full(best_h.copy(), 5000, 0.002, dl4)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    # Phase 5: Final fine LP + polish
    dl5 = deadline
    hv, cv = _lp_full(best_h.copy(), 15000, 0.001, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
    rem = max(0, dl5 - time.time())
    npol = min(int(rem / 0.001), 30000)
    if npol > 100:
        best_h, best_c5 = _polish(best_h, npol, rng, dl5)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
