import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1} via bisection on threshold."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_sub(h, k_list):
    """Compute subgradient for the max of correlations at given lags."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_list:

                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])

    v = np.zeros(n)

    for it in range(steps):
        if time.time() > deadline:
            break
        # Compute gradient

            # Smooth max gradient

            ct = c_full[idx]
            s = np.clip(30000.0 * (ct - m), -400, 400)
            w = np.exp(s - s.max()); w /= w.sum()
            # Weighted subgradient
            n = len(h)
            dx = 2.0 / n
            g = np.zeros(n)
            for ii in range(len(idx)):
                if w[ii] < 1e-14:
                    continue
                k = idx[ii]; sh = (n-1) - k; wk = w[ii] * dx
                if sh >= 0:
                    g[:n-sh] += wk * (1.0 - h[sh:])
                    if sh > 0: g[sh:] -= wk * h[:-sh]
                else:
                    sp = -sh

                    g[:n-sp] -= wk * h[sp:]
        else:

            g = _grad_sub(h, active)
        
        # Nesterov step

        v = beta * v + (1.0 - beta) * g

            if f < best_f:

def _polish_hill(h, n_iter, deadline, rng):

    n = len(h)
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Random pair swap with varying scale

        d = scale * rng.randn()

        hn[i0:i0+bs] = seg

            h, c5 = hn, c5n
            if c5n < best_c5:
                best_c5, best_h = c5n, h.copy()

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _subgrad(h, active_k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active_k)
    return g

    target = n * 0.5
    c = _correl(h)
    f = c.max()
    v = np.zeros(n)

    for it in range(max_steps):
        if time.time() > deadline:
            break
        
        m = f
        if smooth_alpha > 0:
            idx = np.argpartition(c, -topk)[-topk:]
            ct = c[idx]
            s = np.clip(smooth_alpha * (ct - m), -400, 400)
            w = np.exp(s - s.max()); w /= w.sum()
            g = np.zeros(n)

                if sh >= 0:
                    g[:n-sh] += wk * (1.0 - h[sh:])
                    if sh > 0: g[sh:] -= wk * h[:-sh]
                else:
                    sp = -sh
                    g[sp:] += wk * (1.0 - h[:-sp])
                    g[:n-sp] -= wk * h[sp:]
        else:
            active = np.where(c >= m - tol)[0]
            g = _subgrad(h, active)

def _grad_polish(h, n_iter, deadline, rng, base_scale=2e-5):
    """Grad-guided polish: use subgradient to pick direction, then search."""
    n = len(h)
    c = _correl(h)
    f = c.max()
    best_h, best_f = h.copy(), f
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Refresh gradient
        m = f
        active = np.where(c >= m - 0.001)[0]
        g = _subgrad(h, active)

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue

        scale = base_scale * (1.0 + 0.5 * rng.randn())
        d = scale * rng.randn()

            if cn < best_f:
                best_f, best_h = cn, h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    
    return best_h, best_f

            if c5n < best_c5:

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _subgrad(h, active_k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active_k)
    return g

def _nesterov(h, deadline, tol, max_steps, step0, beta, smooth_alpha, topk):
    n = len(h)
    target = n * 0.5
    c = _correl(h)
    f = c.max()
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    min_step = 1e-7

    for _ in range(max_steps):
        if time.time() > deadline:
            break
        m = f
        if smooth_alpha > 0:
            idx = np.argpartition(c, -topk)[-topk:]
            ct = c[idx]
            s = np.clip(smooth_alpha * (ct - m), -400, 400)
            w = np.exp(s - s.max()); w /= w.sum()
            g = np.zeros(n)
            dx = 2.0 / n
            for ii in range(len(idx)):
                if w[ii] < 1e-14:
                    continue
                k = int(idx[ii]); sh = (n - 1) - k; wk = w[ii] * dx
                if sh >= 0:
                    g[:n - sh] += wk * (1.0 - h[sh:])
                    if sh > 0:
                        g[sh:] -= wk * h[:-sh]
                else:
                    sp = -sh
                    g[sp:] += wk * (1.0 - h[:-sp])
                    g[:n - sp] -= wk * h[sp:]
        else:
            active = np.where(c >= m - tol)[0]
            g = _subgrad(h, active)

        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(h - step * v, target)
        cn = _correl(hn)
        fn = cn.max()

        if fn < f - 1e-16:
            h, f, c = hn, fn, cn
            step = min(step * 1.03, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step = max(step * 0.4, min_step)
            hn2 = _proj_sum(hp - step * v, target)
            c2 = _correl(hn2); f2 = c2.max()
            h, f, c = hn2, f2, c2
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _anneal(h, n_iter, deadline, rng):
    n = len(h)
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    blocks = np.array([2, 3, 4, 5, 7, 10, 15, 20])
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = 0.01 * (1e-10) ** (it / max(n_iter, 1))
        bs = int(blocks[rng.randint(len(blocks))])
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * np.sqrt(max(T, 1e-14)) * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5 or rng.random() < np.exp(-(c5n - c5) / max(T, 1e-14)):
            h, c5 = hn, c5n
            if c5n < best_c5:
                best_c5, best_h = c5n, h.copy()
    return best_h, best_c5

def _polish(h, n_iter, deadline, rng):
    n = len(h)
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        scale = 0.8e-5 + 0.4e-5 * (1.0 + np.sin(0.003 * it))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5 - 1e-19:
                c5 = cn
                if cn < best_c5:
                    best_c5, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _jitter_reopt(h, rng, deadline, js):
    n = len(h)
    c5 = _eval(h)
    d = js * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    h2, c5v = _nesterov(hn, deadline, 0.002, 1200, 0.002, 0.88, 0, 0)
    if c5v < c5:
        return h2, c5v
    return h, c5

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)

    dl1 = t0 + 180

    # Phase 2: Smooth-max Nesterov (different basin)
    dl2 = t0 + 340

        if time.time() > dl2:

        h2, c5v = _nesterov(best_h.copy(), dl2, 0, 2500, s0, b, sa, tk)
        if c5v < best_c5:
