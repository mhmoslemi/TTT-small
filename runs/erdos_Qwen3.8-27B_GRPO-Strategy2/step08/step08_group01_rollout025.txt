One more idea: **use the structure of the problem to do a "fast" evaluation**. Instead of computing the full correlation every time, we can compute the change in the correlation when we modify 2 elements. This would make the local search much faster.

When we change h[i] by δ and h[j] by -δ:
- Only correlation terms involving i or j change
- For each shift s, the affected terms are:

  - If i-s and i are both valid: c_s changes by -δ*h[i-s]

Actually, I just realized something. Let me re-examine the gradient computation more carefully.

For the LSE approximation:
f(h) ≈ max(c) + (1/alpha) * log(sum(exp(alpha*(c_k - max(c)))) for k in top_k)

The gradient of this is:
∇f ≈ (1/W) * sum_k w_k * ∇c_k

where w_k = exp(alpha * (c_k - max_c)) and W = sum w_k.

This is what the previous code does. Good.

But wait - there's a subtlety. The "max" in the LSE is a local max (over the top_k), not the global max. The other shifts (not in top_k) don't contribute to the gradient. This means we're only pushing down the top_k shifts, not worrying about the others. This is fine when the top_k shifts are well-separated from the rest, but could be problematic if many shifts have similar values.

For our problem, if the solution is good, the top few shifts should be well-separated. So top_k = 3-5 should be sufficient.

Let me also think about whether to use a different smooth max. Instead of LSE, I could use:
- Power mean: (sum c_k^p)^(1/p) for large p → max
- Soft max (LSE is the standard)

LSE is the most natural and I'll stick with it.

Now, for the step size: the gradient magnitude depends on the scale of h (which is [0,1]) and the number of points. A reasonable step size is:
step = step0 / (1 + 0.1 * it) (decreasing over time)

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - out)
        else:
            order = np.argsort(out)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - out[idx])
                out[idx] += a
                diff -= a
            else:
                a = min(-diff, out[idx])
                out[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return out


def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    return float(np.max(_corr(h)))


def _grad_lse(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g


def _mgd(h, nsteps, alpha, topk, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for it in range(nsteps):
        g = _grad_lse(h, alpha, topk)
        v = mu * v + (1.0 - mu) * (-g)
        h_try = _repair(h + step * v, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            if it % 50 == 0:
                step = min(step * 1.05, step0 * 3)
        else:
            step *= 0.7
            step = max(step, step0 * 0.05)
    return h, cur_c

    target = n * 0.5
    bnds = [(0.0, 1.0)] * n

        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

    def _grad(x):
        return _grad_lse(x, alpha, topk)

    def _con_fun(x):
        return x.sum() - target

    def _con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h.copy(), jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

            else:

            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < cur_c:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def _el():
        return time.time() - t0

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: MGD gradient descent (first 250s)
    mgd_cfgs = [
        (500, 10, 0.006, 600),
        (2000, 8, 0.004, 800),
        (8000, 6, 0.002, 1200),
        (30000, 5, 0.001, 2000),
        (100000, 4, 0.0005, 3000),
    ]
    for alpha, tk, s0, steps in mgd_cfgs:
        if _el() > 250:
            break
        h_t, c = _mgd(best_h.copy(), steps, alpha, tk, s0, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: SLSQP refinement (next 150s)
    if _el() < 400:
        for alpha, tk, mxi in [(2000, 6, 300), (10000, 5, 200), (50000, 4, 150), (200000, 4, 100)]:
            if _el() > 395:
                break
            h_t = _slsqp_run(best_h.copy(), alpha, tk, mxi)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 3: Perturbation restarts (next 100s)
    if _el() < 500:
        for trial in range(4):
            if _el() > 490:
                break
            scale = 0.02 * (trial + 1)
            noise = scale * rng.randn(n)
            noise -= noise.mean()
            h_p = _repair(best_h + noise, n)
            h_p, _ = _mgd(h_p, 400, 5000, 6, 0.003, rng)
            c = _eval(h_p)
            if c < best_c5:
                best_c5 = c
                best_h = h_p.copy()

    # Phase 4: Critical-shift local search
    if _el() < budget_s - 400:
        niter = min(int((budget_s - 400 - _el()) * 1500), 200000)
        best_h, best_c5 = _local2(best_h, niter, 4e-5, rng, use_crit=True)

    # Phase 5: General 2-point local search
    if _el() < budget_s - 300:
        niter = min(int((budget_s - 300 - _el()) * 1800), 250000)
        best_h, best_c5 = _local2(best_h, niter, 2e-5, rng)

    # Phase 6: 3-point local search
    if _el() < budget_s - 200:
        niter = min(int((budget_s - 200 - _el()) * 800), 100000)
        best_h, best_c5 = _local3(best_h, niter, 1.5e-5, rng)

    # Phase 7: Sweep-based refinement
    if _el() < budget_s - 100:
        npass = min(int((budget_s - 100 - _el()) * 30), 500)
        best_h, best_c5 = _sweep(best_h, npass, 8e-6, rng)

    # Phase 8: Fine 2-point polish
    if _el() < budget_s - 50:
        niter = min(int((budget_s - 50 - _el()) * 3000), 500000)
        best_h, best_c5 = _local2(best_h, niter, 5e-6, rng)

    # Phase 9: Ultra-fine final polish
    if _el() < budget_s - 3:
        niter = min(int((budget_s - 3 - _el()) * 6000), 1000000)
        best_h, best_c5 = _local2(best_h, niter, 2e-6, rng)

    # Final micro-polish
    if _el() < budget_s - 1:
        niter = min(int((budget_s - 1 - _el()) * 8000), 500000)
        best_h, best_c5 = _local2(best_h, niter, 1e-6, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
