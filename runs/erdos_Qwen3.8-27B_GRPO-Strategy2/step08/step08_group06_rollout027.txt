```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * (2.0 / n)

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _con_fun(x, target):
    return x.sum() - target

def _con_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, ftol, n):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target, n)}]
    try:
        res = minimize(_lse_obj, h.copy(), jac=_lse_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _subgrad(h, n, t_end, max_iter=5000):

    k_top = 20

def _grad_backtrack(h, n, t_end, max_iter=3000):

def _pair_search(h, n, n_tries, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):

            else:

    return best_h, best_c

def _swap_search(h, n, n_tries, rng, t_end):
    """Try swapping values (preserves sum exactly)."""
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):

        if c_new < best_c:
            best_c = c_new
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h, best_c

def _4pt_search(h, n, n_tries, rng, t_end):
    """4-point moves preserving sum."""
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):

        idx = rng.choice(n, 4, replace=False)

            if c_new < best_c:
                best_c = c_new
            else:

    return best_h, best_c
```

- But we only need a fraction: 10,000 random pairs with 2 directions each = 20,000 evals = 80s. That's doable.

**Even better: gradient-guided local search**
- Compute the gradient of the max correlation
- For each coordinate i, the gradient tells us which direction to move
- Do a coordinate descent: for i = 0, 1, ..., n-1:
  - Compute the optimal h[i] that minimizes the objective (given all other coordinates fixed)
  - This is a 1D optimization problem (parabolic if we ignore the max)

The 1D optimization for coordinate i:
- c_k(i) = h[i]*(1-h[i+k-sh]) + other terms (where sh = n-1-k)
- The max over k is a piecewise function of h[i]
- We can find the minimum by trying a few values

This is essentially **cyclic coordinate descent** (CCD), which is known to work well for constrained optimization.

Let me implement CCD:
```python
def _ccd(h, n, sweeps, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        for i in range(n):
            # Try different values for h[i]
            # The optimal value is where the derivative of the max correlation is 0
            # Approximate: try h[i] - d and h[i] + d for a few d values
            best_val = best_h[i]
            best_ci = best_c
            for d in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                nv = best_val + d
                if 0 <= nv <= 1:
                    best_h[i] = nv
                    # Adjust one other point to maintain sum (or just evaluate)
                    ci = _eval(best_h)
                    if ci < best_ci:
                        best_ci = ci
                        best_val = nv
            best_h[i] = best_val
        # Repair after sweep
        best_h = _repair(best_h, n)
        c = _eval(best_h)
        if c < best_c:
            best_c = c
        else:
            # Undo the sweep
            break
    return best_h, best_c
```

But this is O(n * 6) evaluations per sweep = 3360 evaluations. At 0.004s each, that's 13.4s per sweep. In 100s, we can do ~7 sweeps. Not bad.

But wait - when I change h[i], the sum constraint is violated. I need to either:
a) Not enforce the sum during CCD (let it drift slightly) and repair at the end
b) Adjust another coordinate simultaneously

Option (a) is simpler and should work if the drift is small.

Actually, a better CCD approach:
- For coordinate i, the constraint is ∑h = n/2. If I change h[i], I must compensate.
- The natural compensation is to change the "least important" other coordinate.
- Or: parameterize as h[i] + δ and h[j] - δ where j is chosen to minimize impact.

Let me use a simpler approach: just do 2-point moves systematically.

**Systematic 2-point CCD:**

```
for i in range(n):
    for j in [i+1, i+2, ..., i+10] (mod n, excluding i):
        Try h[i]+d, h[j]-d for d in {±0.01, ±0.001}
        Keep best
```

This is 560 * 10 * 4 = 22,400 evals per sweep. At 0.004s, that's ~90s per sweep. Still a lot.

Let me reduce: 560 * 3 neighbors * 2 directions = 3,360 evals per sweep. ~13s per sweep. In 60s, ~4 sweeps.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * (2.0 / n)

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        wk = w[k] * (2.0 / n)
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _con_fun(x, target):
    return x.sum() - target

def _con_jac(x, n):
    return np.ones(n)

def _slsqp(h, alpha, maxiter, ftol, n):
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target, n)}]
    try:
        res = minimize(_lse_obj, h.copy(), jac=_lse_grad, args=(alpha, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _subgrad(h, n, t_end, max_iter=8000):
    best_h = h.copy()
    best_c = _eval(h)
    step = 0.004
    for it in range(max_iter):
        if time.time() > t_end:
            break
        frac = it / max_iter
        k_top = int(10 + 150 * frac)
        step_adj = step * (1.0 + 0.5 * np.cos(np.pi * frac))
        c = _full_corr(best_h)
        idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(best_h, k, n)
        g /= k_top
        h_new = _repair(best_h - step_adj * g, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
            step = min(step * 1.02, 0.04)
        else:
            step *= 0.65
            if step < 1e-15:
                break
    return best_h, best_c

def _backtrack(h, n, t_end, max_iter=3000):
    best_h = h.copy()
    best_c = _eval(h)
    k_top = 40
    base_step = 0.008
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _full_corr(best_h)
        idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(best_h, k, n)
        g /= k_top
        step = base_step
        improved = False
        for _ in range(15):
            h_new = _repair(best_h - step * g, n)
            c_new = _eval(h_new)
            if c_new < best_c:
                best_h = h_new
                best_c = c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            base_step *= 0.85
            if base_step < 1e-14:
                break
    return best_h, best_c

def _ccd(h, n, n_sweeps, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    deltas = [0.008, 0.003, 0.001, -0.008, -0.003, -0.001]
    for _sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        perm = rng.permutation(n)
        for i in perm:
            best_val = best_h[i]
            best_ci = best_c
            for j_off in [1, 2, 3, 5, 8]:
                j = (i + j_off) % n
                for d in deltas:
                    ni, nj = best_val + d, best_h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        best_h[i], best_h[j] = ni, nj
                        ci = _eval(best_h)
                        if ci < best_ci:
                            best_ci = ci
                            best_val = ni
                            best_h[j] = nj
                        else:
                            best_h[i], best_h[j] = ni - d, nj + d
            best_h[i] = best_val
        best_c = _eval(best_h)
    return best_h, best_c

def _pair_search(h, n, n_tries, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        dist = rng.randint(1, 30)
        j = (i + dist) % n
        d = 0.004 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            cn = _eval(best_h)
            if cn < best_c:
                best_c = cn
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c

def _4pt_search(h, n, n_tries, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        idx = rng.choice(n, 4, replace=False)
        d = 0.002 * rng.randn(4)
        d -= d.mean()
        nv = best_h[idx] + d
        if np.all(nv >= 0) and np.all(nv <= 1):
            best_h[idx] = nv
            cn = _eval(best_h)
            if cn < best_c:
                best_c = cn
            else:
                best_h[idx] = nv - d
    return best_h, best_c

def _swap_search(h, n, n_tries, rng, t_end):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i, j = rng.randint(n, 2)
        best_h[i], best_h[j] = best_h[j], best_h[i]
        cn = _eval(best_h)
        if cn < best_c:
            best_c = cn
        else:
            best_h[i], best_h[j] = best_h[j], best_h[i]
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(h)

    # Phase 1: SLSQP moderate alpha
    t = t0 + budget_s * 0.12
    if time.time() < t:
        h2 = _slsqp(best_h, 8000, 900, 1e-14, n)
        c2 = _eval(h2)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 2: Subgradient with cosine schedule
    t = t0 + budget_s * 0.32
    if time.time() < t:
        h2, c2 = _subgrad(best_h, n, t)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP high alpha
    t = t0 + budget_s * 0.44
    if time.time() < t:
        h2 = _slsqp(best_h, 80000, 700, 1e-15, n)
        c2 = _eval(h2)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 4: Backtracking gradient
    t = t0 + budget_s * 0.58
    if time.time() < t:
        h2, c2 = _backtrack(best_h, n, t)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 5: CCD (cyclic coordinate descent)
    t = t0 + budget_s * 0.72
    if time.time() < t:
        ns = int((t - time.time()) / 15.0)
        ns = max(ns, 3)
        h2, c2 = _ccd(best_h, n, ns, rng, t)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 6: Pair search (wider range)
    t = t0 + budget_s * 0.82
    if time.time() < t:
        nt = int((t - time.time()) / 0.004)
        h2, c2 = _pair_search(best_h, n, nt, rng, t)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 7: 4-point + swap
    t = t0 + budget_s * 0.90
    if time.time() < t:
        nt = int((t - time.time()) * 0.5 / 0.004)
        h2, c2 = _4pt_search(best_h, n, nt, rng, t)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    t = t0 + budget_s * 0.94
    if time.time() < t:
        nt = int((t - time.time()) / 0.004)
        h2, c2 = _swap_search(best_h, n, nt, rng, t)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 8: Final fine pair search
    t = t_global - 2
    if time.time() < t:
        nt = int((t - time.time()) / 0.003)
        h2, c2 = _pair_search(best_h, n, nt, rng, t_global - 1)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
