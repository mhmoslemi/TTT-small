                L = max(L * 0.5, 1.0)
                hp = h; y = h; t_mom = 1.0
        c_end = _c5(h)
        if c_end < best_c:
            best_c, best_h = c_end, h.copy()
    return best_h, best_c

def _pn_arrays(h, N):
    n = len(h)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    return P, Nv

def _grad_wt(h, wP, wN, u, N, S):
    n = len(h)
    dx = 2.0 / n
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        H = np.fft.rfft(h, N)
        Urev = np.fft.rfft(u[::-1], N)
        Ur = np.fft.rfft(u, N)
        R = np.fft.irfft(H * Urev, N)[:n].real
        L = np.fft.irfft(H * Ur, N)[::-1][:n].real
        q = -(R[:n] + L[:n])
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.asarray(globals()['initial_h_values'], dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    # Anti-palindromic and palindromic projections
    for fn in (_anti_pal, _pal_dev):
        hh = fn(h0)
        cc = _c5(hh)
        if cc < best_c:
            best, best_c = hh, cc
    # Phase 1: Smooth max optimization (FISTA) - 300s
    d1 = t0 + 300.0
    if d1 > time.time():
        b, c = _smooth_max_phase(best, d1, (40, 150, 600, 2500))
        if c < best_c:
            best, best_c = b, c
    # Phase 2: Subgradient - 150s
    d2 = t0 + 450.0
    if d2 > time.time():
        b, c = _subgrad_phase(best, d2, 0.02)
        if c < best_c:
            best, best_c = b, c
    # Phase 3: Multi-resolution at n/2 - 150s
    d3 = t0 + 600.0
    if d3 > time.time() + 10:
        nm = n0 // 2
        hm = _resample(best, nm)
        b, c = _smooth_max_phase(hm, d3, (100, 500, 3000))
        if c < best_c:
            best, best_c = b, c
        hu = _resample(best, n0)
        cu = _c5(hu)
        if cu < best_c:
            best, best_c = hu, cu
    # Phase 4: Multi-start perturbation + local - 200s
    d4 = t0 + 800.0
    if d4 > time.time() + 10:
        n_starts = 5
        per_start = (d4 - time.time()) / n_starts
        for si in range(n_starts):
            if time.time() > d4 - 5:
                break
            for amp in (0.03, 0.08):
                hp = _perturb(best, rng, amp, 8)
                sl = d4 - time.time() - 1
                if sl < 25:
                    sl = 25
                b2, c2 = _local_search(hp, rng, time.time() + min(sl, per_start * 0.4))
                if c2 < best_c:
                    best, best_c = b2, c2
    # Phase 5: Final directed local search - remaining time minus 30s
    d5 = t0 + budget_s - 30.0
    if d5 > time.time() + 5:
        best, best_c = _local_search(best, rng, d5)
    # Final variants
    for fn in (lambda x: _proj(1.0 - x, len(x) // 2),
               lambda x: _proj(x[::-1], len(x) // 2),
               lambda x: _anti_pal(x),
               lambda x: _pal_dev(x)):
        hh = fn(best)
        cc = _c5(hh)
        if cc < best_c:
            best, best_c = hh, cc
    # Final tiny local search
    df = t0 + budget_s - 1.0
    if df > time.time() + 2:
        best, best_c = _local_search(best, rng, df)
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
