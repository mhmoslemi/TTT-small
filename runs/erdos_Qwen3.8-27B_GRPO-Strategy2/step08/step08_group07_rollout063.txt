    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for i in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i]); h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(x, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0, 1)

def _grad_smmax(h, alpha, topk):
    n = len(h)
    c = _corr_vec(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    wn = w.sum()
    if wn < 1e-30:
        return np.zeros(n), m
    w /= wn
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = idx[i]
        ss = k - (n - 1)
        p = abs(ss)
        wt = wi * dx
        if ss <= 0:
            g[:n - p] += wt * (1.0 - h[p:])
            g[p:] -= wt * h[:n - p]
        else:
            g[p:] += wt * (1.0 - h[:n - p])
            g[:n - p] -= wt * h[p:]
    return g, m

def _nesterov_ac(h, alpha, topk, nsteps, step0, deadline, beta=0.85):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(nsteps):
        if time.time() > deadline:
            break
        g, _ = _grad_smmax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        gstep = np.linalg.norm(step * v)
        if gstep < 1e-15:
            step *= 0.5
            continue
        st = step
        accepted = False
        for ls in range(12):
            if time.time() > deadline:
                break
            hn = _proj(hp - st * v, target)
            fn = _eval(hn)
            if fn < f - 1e-16:
                accepted = True
                break
            st *= 0.4
        if not accepted:
            st = max(st, 1e-12)
            hn = _proj(hp - st * v, target)
            fn = _eval(hn)
        h, f, step = hn, fn, min(st * 1.05, step0 * 25.0)
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _anneal(h, niter, rng, deadline, T0=0.002, T1=1e-12):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    blocks = [2, 3, 4, 5, 7, 10, 15, 20]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5 or rng.random() < np.exp(-(c5n - c5) / max(T, 1e-12)):
            h, c5 = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale=5e-6):
    n = len(h)
    c5 = _eval(h)
    bh, bc = h.copy(), c5
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-19:
                c5 = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _gen_alt_start(n, rng, typ):
    x = np.linspace(0, 2, n, endpoint=False)
    if typ == 0:
        h = 1.0 / (1.0 + np.exp((x - 1.0) * 8.0))
    elif typ == 1:
        h = 0.5 + 0.45 * np.sin(np.pi * x + 0.3)
    elif typ == 2:
        h = (x < 1.0).astype(float) * 0.95 + np.random.RandomState(typ).randn(n) * 0.05
    else:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x)
    return _repair(np.clip(h, 0, 1), n)
