def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        else:

def _smax_obj(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0 / n)
    m = c.max()
    if topk > 0 and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]

def _grad_smax(h, alpha, topk):

    c = np.correlate(h, 1.0-h, mode='full') * (2.0 / n)
    m = c.max()
    if topk > 0 and topk < len(c):

        ct = c[idx]

        idx = np.arange(len(c))
        ct = c

import numpy as np
from scipy.optimize import minimize
import time

_gp = {'alpha': 3000, 'topk': 30}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk > 0 and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        if sp < n:
            g[:n - sp] -= h[sp:]
    return g * dx

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk > 0 and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx_val = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_lag(h, idx[ii], dx_val)
    return g

def _obj_w(x):
    return _smax_obj(x, _gp['alpha'], _gp['topk'])

def _jac_w(x):
    return _grad_smax(x, _gp['alpha'], _gp['topk'])

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _run_slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    _gp['alpha'] = alpha
    _gp['topk'] = topk
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        r = minimize(_obj_w, h, jac=_jac_w, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha0, grow, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 200 == 0:
            alpha = min(alpha * grow, 5e7)
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.015, step0 * 10.0)
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(hp - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _pair_golden(h, i, j, n_eval):
    dlo = max(-h[i], h[j] - 1.0)
    dhi = min(1.0 - h[i], h[j])
    if dhi - dlo < 1e-14:
        return _eval(h)
    a, b = dlo, dhi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    hc = h.copy(); hc[i] += c1; hc[j] -= c1
    fd1 = _eval(hc)
    hd = h.copy(); hd[i] += c2; hd[j] -= c2
    fd2 = _eval(hd)
    for _ in range(n_eval - 2):
        if fd1 < fd2:
            b = c2; c2 = c1; fd2 = fd1
            c1 = b - gr * (b - a)
            hc = h.copy(); hc[i] += c1; hc[j] -= c1
            fd1 = _eval(hc)
        else:
            a = c1; c1 = c2; fd1 = fd2
            c2 = a + gr * (b - a)
            hd = h.copy(); hd[i] += c2; hd[j] -= c2
            fd2 = _eval(hd)
    d_star = 0.5 * (a + b)
    h[i] += d_star
    h[j] -= d_star
    return _eval(h)

def _pair_descent(h, rng, npairs, deadline, offsets, gevals):
    n = len(h)
    best_c5 = _eval(h)
    bh = h.copy()
    for _ in range(npairs):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        off = offsets[rng.randint(len(offsets))]
        j = (i + off) % n
        if j == i:
            continue
        c5v = _pair_golden(h, i, j, gevals)
        if c5v < best_c5:
            best_c5 = c5v
            bh = h.copy()
    return bh, best_c5

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (float(it) / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sb = 4e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = sb * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                sb = max(sb * 0.97, 1e-9)
    return bh, bc

def _interp_n(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return _repair(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    n0 = len(initial_h_values)

        c5v = _eval(h)
        if c5v < best_c5:

    # Phase 3: Pair golden descent (220-420s)

import numpy as np
from scipy.optimize import minimize
import time

_gp = {'alpha': 3000, 'topk': 30}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk > 0 and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        if sp < n:
            g[:n - sp] -= h[sp:]
    return g * dx

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk > 0 and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dxv = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_lag(h, idx[ii], dxv)
    return g

def _obj_w(x):
    return _smax_obj(x, _gp['alpha'], _gp['topk'])

def _jac_w(x):
    return _grad_smax(x, _gp['alpha'], _gp['topk'])

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _run_slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    _gp['alpha'] = alpha
    _gp['topk'] = topk
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        r = minimize(_obj_w, h, jac=_jac_w, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, alpha0, grow, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 200 == 0:
            alpha = min(alpha * grow, 5e7)
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.015, step0 * 10.0)
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(hp - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _pair_golden(h, i, j, nevals):
    dlo = max(-h[i], h[j] - 1.0)
    dhi = min(1.0 - h[i], h[j])
    if dhi - dlo < 1e-14:
        return _eval(h)
    a, b = dlo, dhi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    hc = h.copy(); hc[i] += c1; hc[j] -= c1
    fd1 = _eval(hc)
    hd = h.copy(); hd[i] += c2; hd[j] -= c2
    fd2 = _eval(hd)
    for _ in range(nevals - 2):
        if fd1 < fd2:
            b = c2; c2 = c1; fd2 = fd1
            c1 = b - gr * (b - a)
            hc = h.copy(); hc[i] += c1; hc[j] -= c1
            fd1 = _eval(hc)
        else:
            a = c1; c1 = c2; fd1 = fd2
            c2 = a + gr * (b - a)
            hd = h.copy(); hd[i] += c2; hd[j] -= c2
            fd2 = _eval(hd)
    d_star = 0.5 * (a + b)
    h[i] += d_star
    h[j] -= d_star
    return _eval(h)

def _pair_descent(h, rng, npairs, deadline, offsets, gevals):
    n = len(h)
    best_c5 = _eval(h)
    bh = h.copy()
    for _ in range(npairs):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        off = offsets[rng.randint(len(offsets))]
        j = (i + off) % n
        if j == i:
            continue
        c5v = _pair_golden(h, i, j, gevals)
        if c5v < best_c5:
            best_c5 = c5v
            bh = h.copy()
    return bh, best_c5

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (float(it) / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sb = 4e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = sb * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                sb = max(sb * 0.97, 1e-9)
    return bh, bc

def _interp_n(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return _repair(np.interp(x_new, x_old, h), n_new)
