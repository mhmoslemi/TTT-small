import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, s, dx):
    n = len(h)
    j = abs(s)
    if j == 0:
        return (1.0 - 2.0 * h) * dx
    g = np.zeros(n)
    g[j:] += (1.0 - h[:n - j])
    g[:n - j] -= h[j:]
    return g * dx

def _smooth_topk(h, alpha, tk):
    c = _correl(h)
    if tk < len(c):
        idx = np.argpartition(c, -tk)[-tk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    w = np.exp(s - sm)
    w /= w.sum()
    return val, idx, w, m

def _smax_grad(h, idx, w, dx, n):
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-10:
            g += w[ii] * _grad_lag(h, idx[ii] - (n - 1), dx)
    return g
