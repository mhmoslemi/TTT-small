    best_c = _c5_direct(best)
    # Try structural variants
    for hh in [1.0 - h0, h0[::-1],
               _proj(0.5 + 0.5 * (h0 - h0[::-1]), n0 * 0.5),
               _proj(0.5 + 0.5 * (h0 + h0[::-1] - 1.0), n0 * 0.5)]:
        c = _c5_direct(hh)
        if c < best_c: best, best_c = hh.copy(), c
    # Phase 1: Quick FISTA at moderate alpha
    end1 = t0 + min(45.0, 0.05 * budget_s)
    if end1 > time.time() + 2:
        b, c, _ = _run_alphas(best, [10, 50, 300], end1 - time.time(), end1)
        if c < best_c: best, best_c = b, c
    # Phase 2: Multi-scale
    end2 = t0 + min(180.0, 0.25 * budget_s)
    if budget_s > 300 and end2 > time.time() + 5:
        for m in [140, 280]:
            if end2 - time.time() < 4: break
            hm = _resample(best, m)
            slot = min(40.0, end2 - time.time() - 2)
            if slot < 3: break
            b, c, _ = _run_alphas(hm, [20, 200, 2000, 15000], slot, end2 - 2)
            if c < best_c:
                best, best_c = b, c
            if len(best) != n0:
                hu = _resample(best, n0)
                cu = _c5_direct(hu)
                if cu < best_c: best, best_c = hu, cu
    # Phase 3: Main FISTA at high alpha
    end3 = t0 + budget_s - 250.0
    if end3 > time.time() + 10:
        b, c, _ = _run_alphas(best, [100, 500, 2000, 10000, 60000, 300000, 1500000],
                              end3 - time.time(), end3)
        if c < best_c: best, best_c = b, c
        if len(best) != n0:
            best = _resample(best, n0)
            best_c = _c5_direct(best)
    # Phase 4: Polish (projected gradient on active set)
    end4 = t0 + budget_s - 120.0
    if end4 > time.time() + 3:
        best, best_c = _polish(best, end4)
    # Phase 5: Exhaustive local search
    end5 = t0 + budget_s - 2.0
    if end5 > time.time() + 2:
        best, best_c = _local_pair(best, rng, end5, 4e-5)
    # Final variants
    for hh in [1.0 - best, best[::-1],
               _proj(0.5 + 0.5 * (best - best[::-1]), len(best) * 0.5),
               _proj(0.5 + 0.5 * (best + best[::-1] - 1.0), len(best) * 0.5)]:
        c = _c5_direct(hh)
        if c < best_c: best, best_c = hh.copy(), c
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
