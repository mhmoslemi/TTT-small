```
def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = float(c.max())
    s = alpha * (c - m)
    s = np.clip(s, -500.0, 500.0)
    s = s - s.max()
    return m + math.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
    m = float(c.max())
    s = alpha * (c - m)
    s = np.clip(s, -500.0, 500.0)
    s = s - s.max()
    w = np.exp(s)
    wsum = float(w.sum())
    if wsum <= 0:
       w = np.zeros_like(w)
    else:
       w /= wsum
    g = np.zeros(n)
    dx = 2.0/n
    nm1 = n-1
    eps=1e-15
    for k in range(2*n-1):
       wk = w[k]*dx
       if wk < eps: continue
       sh = nm1 - k
       if sh >= 0:
          g[:n-sh] += wk * (1.0 - x[sh:])
          if sh > 0:
             g[sh:] -= wk * x[:n-sh]
       else:
          sp = -sh
          g[sp:] += wk * (1.0 - x[:n-sp])
          g[:n-sp] -= wk * x[sp:]
    return g
```

```
class _SLSQPCtx:
    def __init__(self, n, alpha, target):
        self.n=n; self.alpha=alpha; self.target=target; self.ones=np.ones(n)
    def obj(self,x): return _lse_obj(x, self.alpha, self.n)
    def grad(self,x): return _lse_grad(x, self.alpha, self.n)
    def con_fun(self,x): return float(np.sum(x)-self.target)
    def con_jac(self,x): return self.ones

def _slsqp_run(h0, alpha, maxiter, ftol, target):
    n=len(h0)
    ctx=_SLSQPCtx(n,alpha,target)
    bnds=[(0.0,1.0)]*n
    cons=[{'type':'eq','fun':ctx.con_fun,'jac':ctx.con_jac}]
    try:
       res=minimize(ctx.obj, h0, jac=ctx.grad, method='SLSQP', bounds=bnds, constraints=cons, options={'maxiter':maxiter,'ftol':ftol})
       return _repair(np.asarray(res.x, dtype=float).copy(), target)
    except Exception:
       return _repair(h0.copy(), target)
```

```
def _score_from_lse(h,c,K,alpha,n,scale):
    L=2*n-1; cs=c*scale
    if K>=L: idx=np.arange(L)
    else:
       part=np.argpartition(cs,-K)[-K:]
       idx=part[np.argsort(cs[part])[::-1]]
    mv=float(cs[idx].max())
    s=alpha*(cs[idx]-mv)
    s=np.clip(s,-500.0,500.0)
    s=s-s.max()
    w=np.exp(s)
    ws=float(w.sum())
    if ws<=0.0: w=np.full(len(w),1.0/len(w))
    else: w=w/ws
    sc=np.zeros(n); nm1=n-1
    for t in range(len(w)):
       wq=w[t]
       if wq <= 1e-15: continue
       q=idx[t]
       d=abs(q-nm1)
       if d==0:
          sc += wq*(1.0-2.0*h)
       elif q < nm1:
          sc[:n-d] += wq*(1.0-h[d:]); sc[d:] -= wq*h[:n-d]
       else:
          sc[d:] += wq*(1.0-h[:n-d]); sc[:n-d] -= wq*h[d:]
    return sc
```

```python
import numpy as np
import time
import math
from scipy.optimize import minimize

def _repair(h, target):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    if h.sum() <= 1e-14:
        h.fill(0.5)
        return h
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -1.0, 1.0
        for _ in range(60):
            m = (lo + hi) * 0.5
            if np.clip(h + m, 0.0, 1.0).sum() > target:
                hi = m
            else:
                lo = m
        h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _fast_eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return c, float(c.max() * (2.0 / n))

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = float(c.max())
    s = alpha * (c - m)
    s = np.clip(s, -500.0, 500.0)
    s = s - s.max()
    return m + math.log(float(np.sum(np.exp(s)))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
    m = float(c.max())
    s = alpha * (c - m)
    s = np.clip(s, -500.0, 500.0)
    s = s - s.max()
    w = np.exp(s)
    ws = float(w.sum())
    if ws > 0.0:
        w /= ws
    else:
        w = np.zeros_like(w)
    g = np.zeros(n)
    dx = 2.0 / n
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk <= 1e-15:
            continue
        sh = nm1 - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            g[sh:] -= wk * x[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

class _Ctx:
    def __init__(self, n, alpha, target):
        self.n = n
        self.alpha = alpha
        self.target = target
        self.ones = np.ones(n)

    def f(self, x):
        return _lse_obj(x, self.alpha, self.n)

    def g(self, x):
        return _lse_grad(x, self.alpha, self.n)

    def cf(self, x):
        return float(np.sum(x) - self.target)

    def cj(self, x):
        return self.ones

def _slsqp_run(h0, alpha, maxiter, ftol, target):
    n = len(h0)
    ctx = _Ctx(n, alpha, target)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": ctx.cf, "jac": ctx.cj}]
    try:
        res = minimize(ctx.f, h0, jac=ctx.g, method="SLSQP", bounds=bnds,
                       constraints=cons, options={"maxiter": maxiter, "ftol": ftol})
        x = getattr(res, "x", None)
        return _repair(x if x is not None else h0, target)
    except Exception:
        return _repair(h0, target)

def _score_from_lse(h, c, K, alpha, n, scale):
    L = 2 * n - 1
    cs = c * scale
    if K >= L:
        idx = np.arange(L)
    else:
        part = np.argpartition(cs, -K)[-K:]
        idx = part[np.argsort(cs[part])[::-1]]
    mv = float(cs[idx].max())
    s = alpha * (cs[idx] - mv)
    s = np.clip(s, -500.0, 500.0)
    s = s - s.max()
    w = np.exp(s)
    ws = float(w.sum())
    if ws > 0.0:
        w /= ws
    else:
        w = np.full(len(w), 1.0 / max(1, len(w)))
    sc = np.zeros(n)
    nm1 = n - 1
    for t in range(len(w)):
        wq = w[t]
        if wq <= 1e-15:
            continue
        q = idx[t]
        d = abs(q - nm1)
        if d == 0:
            sc += wq * (1.0 - 2.0 * h)
        elif q < nm1:
            sc[:n - d] += wq * (1.0 - h[d:])
            sc[d:] -= wq * h[:n - d]
        else:
            sc[d:] += wq * (1.0 - h[:n - d])
            sc[:n - d] -= wq * h[d:]
    return sc

def _build_pair_lags(n):
    n1 = n - 1
    la = [None] * n
    pa = [None] * n
    lb = [None] * n
    pb = [None] * n
    for p in range(n):
        d = np.arange(0, n - p, dtype=np.int32)
        al = [(n1 - d).astype(np.int32)]
        ap = [(p + d).astype(np.int32)]
        if p > 0:
            d = np.arange(1, p + 1, dtype=np.int32)
            al.append((n1 + d).astype(np.int32))
            ap.append((p - d).astype(np.int32))
        la[p] = np.concatenate(al)
        pa[p] = np.concatenate(ap)
        bl = [np.array([n1], dtype=np.int32)]
        bp = [np.array([p], dtype=np.int32)]
        if p > 0:
            d = np.arange(1, p + 1, dtype=np.int32)
            bl.append((n1 - d).astype(np.int32))
            bp.append((p - d).astype(np.int32))
        if n - p > 1:
            d = np.arange(1, n - p, dtype=np.int32)
            bl.append((n1 + d).astype(np.int32))
            bp.append((p + d).astype(np.int32))
        lb[p] = np.concatenate(bl)
        pb[p] = np.concatenate(bp)
    return la, pa, lb, pb

def _local_pair(h, c, la, pa, lb, pb, rng, t_end, target):
    h = _repair(h.copy(), target)
    c, cur_c = _fast_eval(h)
    best_h = h.copy()
    best_c = cur_c
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    q0 = n - 1
    A = np.zeros(L)
    buf = np.zeros(L)
    tmp = np.zeros(L)
    btmp = np.zeros(L)
    step = 0.012
    min_step = 1e-7
    score = _score_from_lse(h, c, 220, 12000.0, n, dx)
    updates = 0
    rejects = 0
    trial = 0
    while time.time() < t_end:
        trial += 1
        if trial % 400 == 0 or (updates > 0 and updates % 35 == 0):
            if abs(h.sum() - target) > 1e-8:
                h = _repair(h, target)
            c_t, C_t = _fast_eval(h)
            if C_t < best_c - 1e-12:
                best_c = C_t
                best_h[:] = h
                c[:] = c_t
                cur_c = C_t
            elif C_t > best_c + 1e-9:
                h[:] = best_h
                c_t, C_t = _fast_eval(h)
                c[:] = c_t
                cur_c = C_t
                updates = 0
                rejects = 0
            else:
                c[:] = c_t
                cur_c = C_t
            score = _score_from_lse(h, c, 220, 12000.0, n, dx)
            if rejects > 1500:
                step = max(min_step, step * 0.55)
                rejects = 0
            elif updates > 400:
                step = min(0.035, step * 1.2)
                updates = 0
            if step <= min_step + 1e-9 and rejects > 3000:
                break
        r = rng.random_sample()
        i = int(rng.randint(n))
        if r < 0.70:
            off = int(rng.randint(1, 31))
            if i + off < n:
                j = i + off
            elif i - off >= 0:
                j = i - off
            else:
                j = int(rng.randint(n))
        else:
            j = int(rng.randint(n))
        if i == j:
            continue
        diff = score[i] - score[j]
        if abs(diff) < 1e-10:
            continue
        sgn = -1.0 if diff > 0.0 else 1.0
        hi = min(1.0 - h[i], h[j])
        lo = max(-h[i], h[j] - 1.0)
        if hi <= lo:
            continue
        mag = hi if sgn > 0.0 else -lo
        if mag <= 1e-15:
            continue
        base = step if step < mag else mag
        if base <= 1e-15:
            continue
        A.fill(0.0)
        lai = la[i]
        A[lai] += 1.0
        A[lai] -= h[pa[i]]
        A[lb[i]] -= h[pb[i]]
        laj = la[j]
        A[laj] -= 1.0
        A[laj] += h[pa[j]]
        A[lb[j]] += h[pb[j]]
        ad = abs(i - j)
        q1 = q0 - ad
        q2 = q0 + ad
        found = False
        bd = 0.0
        for fac in (1.0, 0.5, 0.25):
            d = sgn * base * fac
            dd = d * d
            np.multiply(A, d, out=buf)
            np.add(buf, c, out=tmp)
            tmp[q0] -= 2.0 * dd
            tmp[q1] += dd
            tmp[q2] += dd
            val = dx * float(tmp.max())
            if val < cur_c - 1e-12:
                cur_c = val
                bd = d
                found = True
                btmp[:] = tmp
        if found:
            h[i] += bd
            h[j] -= bd
            c[:] = btmp
            updates += 1
            rejects = 0
    c_h, C_h = _fast_eval(h)
    if C_h <= best_c:
        best_h = h.copy()
        c_b = c_h
        best_c = C_h
    else:
        c_b, best_c = _fast_eval(best_h)
    return best_h, c_b, best_c
```
