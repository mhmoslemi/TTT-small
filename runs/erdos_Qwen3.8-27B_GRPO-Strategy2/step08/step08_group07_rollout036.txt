def _dual_primal(h, n_steps, eta, K, deadline):
    """Primal-dual with multiplicative weights on the dual."""
    n = len(h)
    n_corr = 2*n - 1
    p = np.ones(n_corr) / n_corr  # dual distribution
    h_best = h.copy()
    f_best = _eval(h)
    
    for t in range(n_steps):
        if time.time() > deadline:
            break
        # Compute all correlations
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        
        # Update dual: p(k) ∝ p(k) * exp(c(k)/eta)
        p = p * np.exp(c / eta)
        p /= p.sum()
        
        # Compute weighted gradient
        # ∂C(k)/∂h[j] for lag = k - (n-1)
        g = np.zeros(n)
        for k in range(n_corr):
            if p[k] < 1e-10:
                continue
            lag = k - (n - 1)
            wk = p[k] * (2.0 / n)
            # ∂C(k)/∂h[j] = (1-h[j+lag]) - h[j-lag] (with bounds)
            if lag >= 0:
                # j ranges: 0 to n-1-lag for (1-h[j+lag])
                # j ranges: lag to n-1 for h[j-lag]
                g[:n-lag] += wk * (1.0 - h[lag:])
                g[lag:] -= wk * h[:-lag]
            else:
                lag2 = -lag
                g[lag2:] += wk * (1.0 - h[:-lag2])
                g[:n-lag2] -= wk * h[lag2:]
        
        # Gradient step with projection
        step = eta / (1.0 + t * 0.01)  # decreasing step
        h_new = _proj(h - step * g, n * 0.5)
        f_new = _eval(h_new)
        
        if f_new < f_best:
            f_best = f_new
            h_best = h_new.copy()
        
        h = h_new
    
    return h_best, f_best

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _grad_active(h, active):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        lag = int(k) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= h[:-lag]
        else:
            lag2 = -lag
            g[lag2:] += (1.0 - h[:-lag2])
            g[:n - lag2] -= h[lag2:]
    g *= dx / len(active)
    return g

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii]); lag = k - (n - 1); wk = w[ii] * dx
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= wk * h[:-lag]
        else:
            lag2 = -lag
            g[lag2:] += wk * (1.0 - h[:-lag2])
            g[:n - lag2] -= wk * h[lag2:]
    return g, m

    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    stall = 0
    for t in range(steps):
        if time.time() > deadline:
            break
        frac = min(1.0, t / max(steps - 1, 1))
        beta = beta0 + (beta1 - beta0) * frac
        if mode == 'exact':
            c = _correl(h)
