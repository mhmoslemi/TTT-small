    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_topk(h, K, alpha):
    """Subgradient using top-K shifts with softmax weighting."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    if K < len(c):
        idx = np.argpartition(c, -K)[-K:]
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _fw_step(h, g, target_frac):
    """Frank-Wolfe step: find extreme point minimizing g·z."""
    n = len(h)
    # sort by gradient, take n/2 smallest → set to 1
    order = np.argsort(g)
    s = np.zeros(n)
    s[order[:int(target_frac)]] = 1.0
    return s

def _nesterov_fw(h, steps, step0, deadline, K, alpha, beta=0.85):
    """Nesterov with Frank-Wolfe projection."""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        g, _ = _grad_topk(h, K, alpha)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        # Try nesterov step with projection
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _nesterov_pure(h, steps, step0, deadline, K, alpha, beta=0.85):
    """Standard Nesterov with projection (like before but tuned)."""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        g, _ = _grad_topk(h, K, alpha)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 10.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.4
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _fw_optimize(h, niter, deadline, K=5, alpha=10000):
    """Pure Frank-Wolfe optimization."""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for it in range(niter):
        if time.time() > deadline:
            break
        g, _ = _grad_topk(h, K, alpha)
        s = _fw_step(h, g, int(target))
        # Adaptive step size
        gamma = 2.0 / (it + 2)
        # Also try a line search
        hn = (1 - gamma) * h + gamma * s
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            # Try smaller step
            gamma2 = gamma * 0.1
            hn2 = (1 - gamma2) * h + gamma2 * s
            fn2 = _eval(hn2)
            if fn2 < f:
                h, f = hn2, fn2
                if f < best_f:
                    best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _polish_fine(h, niter, rng, deadline, scales):
    """Multi-scale coordinate descent polishing."""
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        scale = scales[rng.randint(len(scales))]
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 7)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _triple_swap(h, i, j, k, d1, d2):
    """Try a 3-point swap for better local search."""
    ai, aj, ak = h[i] + d1, h[j] - d1 + d2, h[k] - d2
    return (0.0 <= ai <= 1.0) and (0.0 <= aj <= 1.0) and (0.0 <= ak <= 1.0)

def _anneal_2(h, niter, rng, deadline, blocks, T0, T1):
    """Simulated annealing with block moves."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc
