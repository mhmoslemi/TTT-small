import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = _eval_vec(h)
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, m_idx in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        k = abs(m_idx - (n - 1))
        wk = w[ii] * dx
        if k == 0:
            g += wk * (1.0 - 2.0 * h)
        else:
            g[k:] += wk * (1.0 - h[:n-k])
            g[:n-k] -= wk * h[k:]
    return g

def _mgd_bt(h, n_iter, alpha, top_k, step0, rng):
    """MGD with backtracking line search."""
    n = len(h)
    best_c = _eval(h)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        v = mu * v + (1 - mu) * (-step * g)  # negative gradient with momentum
        h_new = _repair(h + v, n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-15:
            h = h_new
            best_c = c_new
            step = min(step * 1.1, step0 * 10)
        else:
            step *= 0.5
            if step < step0 * 0.01:
                step = step0
    return h, best_c

For the coordinate descent:

- Compute the effect on the top K correlation values
- Find δ that minimizes max of those K values

The key insight: for a given pair (i,j) and the top K shifts, each c[m] changes linearly in δ:
c[m] → c[m] + δ * (dc[m]/dh[i] - dc[m]/dh[j])

We want to minimize max_m (c[m] + δ * g_m) where g_m = dc[m]/dh[i] - dc[m]/dh[j].

This is a convex 1D problem. The optimal δ is at the point where the maximum is minimized, which is either:
- At a boundary (δ = -h[i] or δ = 1-h[i] or δ = h[j]-1 or δ = h[j])
- At an interior point where two of the lines c[m] + δ*g_m are equal and form the maximum

For efficiency, I'll just check a grid of δ values around 0 (like δ ∈ [-0.01, 0.01] with 100 points) and pick the best one. This is O(K) per δ value, so O(100K) per candidate pair. For K=5, that's 500 operations per candidate. Very fast!

def _cd_step(h, c_vec, top_idx, n_pairs, rng, delta_range=0.02, n_delta=50):
    """One step of coordinate descent: try n_pairs random 2-pt moves."""
    n = len(h)
    best_c = float(np.max(c_vec))
    best_h = h.copy()
    
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        if i == j:
            continue
        # Compute gradient of each top shift w.r.t. h[i] and h[j]
        di = np.zeros(len(top_idx))
        dj = np.zeros(len(top_idx))
        for t, m_idx in enumerate(top_idx):
            k = abs(m_idx - (n - 1))
            # d(c[m_idx])/dh[j] for the raw correlation (before *dx)
            # For shift k: (1-h[j-k])*[j>=k] - h[j+k]*[j<=n-1-k]
            gi = 0.0
            gj = 0.0
            if k == 0:
                gi = 1.0 - 2.0 * h[i]
                gj = 1.0 - 2.0 * h[j]
            else:
                # For h at position i:
                if i >= k:
                    gi += (1.0 - h[i-k])
                if i <= n-1-k:
                    gi -= h[i+k]
                # For h at position j:
                if j >= k:
                    gj += (1.0 - h[j-k])
                if j <= n-1-k:
                    gj -= h[j+k]
            di[t] = gi
            dj[t] = gj
        
        g = di - dj  # dc/d(h[i]-h[j]) for each top shift
        
        # Try a range of δ values
        best_local = best_c
        best_d = 0.0
        for d in np.linspace(-delta_range, delta_range, n_delta):
            ni = h[i] + d
            nj = h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            # New values of top shifts
            new_top = c_vec[top_idx] + d * g
            local_max = float(np.max(new_top))
            if local_max < best_local - 1e-14:
                # Check global max (approximate: check if any non-top shift exceeds)
                best_local = local_max
                best_d = d
        
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            c_vec = _eval_vec(h)  # recompute
            actual_max = float(np.max(c_vec))
            if actual_max < best_c:
                best_c = actual_max
                best_h = h.copy()
                top_idx = np.argsort(c_vec)[-5:]  # update active set
    
    return best_h, best_c, top_idx

We want R(k) to be as uniformly large as possible. The integral of R is ∫R(k)dk = (∫h)² = 1 (for the continuous case on [0,2] with appropriate boundary handling).

def _cd_phase(h, c_vec, top_K, n_iters, delta_max, rng):
    n = len(h)
    dx = 2.0 / n
    best_c = float(np.max(c_vec))
    best_h = h.copy()
    top_idx = np.argsort(c_vec)[-top_K:]
    for it in range(n_iters):
        if it % 100 == 0:
            top_idx = np.argsort(c_vec)[-top_K:]
            best_c = float(np.max(c_vec))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n
        if i == j:
            continue
        g = np.empty(top_K)
        for t in range(top_K):
            mi = top_idx[t]
            v1 = dx * (1.0 - 2.0 * h[mi - i]) if 0 <= mi - i < n else 0.0
            v2 = dx * (1.0 - 2.0 * h[mi - j]) if 0 <= mi - j < n else 0.0
            g[t] = v1 - v2
        c_top = c_vec[top_idx]
        best_local = best_c
        best_d = 0.0
        for d in np.linspace(-delta_max, delta_max, 15):
            if d == 0.0:
                continue
            if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                continue
            pred = float(np.max(c_top + d * g))
            if pred < best_local - 1e-14:
                best_local = pred
                best_d = d
        if best_d != 0.0:
            h[i] += best_d
            h[j] -= best_d
            c_vec = _eval_vec(h)
            actual = float(np.max(c_vec))
            if actual < best_c:
                best_c = actual
                best_h = h.copy()
    return best_h, best_c

def _local_sa(h, n_iter, step, sa_temp, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp > 1e-12:
                ratio = -(c_new - cur_c) / sa_temp
                if ratio < 500 and rng.rand() < np.exp(ratio):
                    accept = True
            if accept:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep2(h, n_passes, step_hover, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, 8)) % n
            d = step_hover * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < cur_c:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: MGD with progressive sharpness
    for alpha, top_k, steps, s0 in [(2000, 8, 400, 0.004), (10000, 6, 300, 0.002), (100000, 4, 200, 0.001)]:
        if _el() > 100:
            break
        h_t = best_h.copy()
        h_t, c = _mgd(h_t, steps, alpha, top_k, s0, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: Multi-restart (random + structured)
    if _el() < budget_s - 400:
        for trial in range(6):
            if _el() > budget_s - 380:
                break
            if trial < 3:
                scale = 0.015 * (trial + 1)
                noise = scale * rng.randn(n)
                noise -= noise.mean()
                h_p = _repair(best_h + noise, n)
            else:
                x = np.linspace(0, 2, n)
                freq = 1.0 + trial * 0.5
                noise = 0.02 * np.sin(2.0 * np.pi * freq * x)
                noise -= noise.mean()
                h_p = _repair(best_h + noise, n)
            h_p, c = _mgd(h_p, 250, 50000, 5, 0.001, rng)
            c = _eval(h_p)
            if c < best_c5:
                best_c5 = c
                best_h = h_p.copy()

    # Phase 3: CD coarse
    if _el() < budget_s - 500:
        c_vec = _eval_vec(best_h)
        n_cd = min(int((budget_s - 520 - _el()) * 600), 40000)
        best_h, best_c5 = _cd_phase(best_h, c_vec, 6, max(n_cd, 5000), 0.04, rng)

    # Phase 4: MGD polish
    if _el() < budget_s - 350:
        h_t = best_h.copy()
        h_t, c = _mgd(h_t, 400, 200000, 3, 0.0005, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 5: CD medium
    if _el() < budget_s - 250:
        c_vec = _eval_vec(best_h)
        n_cd = min(int((budget_s - 270 - _el()) * 1000), 60000)
        best_h, best_c5 = _cd_phase(best_h, c_vec, 5, max(n_cd, 5000), 0.015, rng)

    # Phase 6: SA
    if _el() < budget_s - 150:
        remaining = budget_s - 170 - _el()
        n_sa = min(int(remaining * 1500), 80000)
        best_h, best_c5 = _local_sa(best_h, max(n_sa, 10000), 4e-5,
                                     max(1e-6, best_c5 * 1e-5), rng)

    # Phase 7: CD fine
    if _el() < budget_s - 100:
        c_vec = _eval_vec(best_h)
        n_cd = min(int((budget_s - 120 - _el()) * 2000), 100000)
        best_h, best_c5 = _cd_phase(best_h, c_vec, 4, max(n_cd, 10000), 0.006, rng)

    # Phase 8: Sweep polish
    if _el() < budget_s - 50:
        npass = min(int((budget_s - 60 - _el()) * 100), 3000)
        best_h, best_c5 = _sweep2(best_h, max(npass, 500), 2e-5, rng)

    # Phase 9: Final CD ultra-fine
    if _el() < budget_s - 15:
        c_vec = _eval_vec(best_h)
        n_cd = min(int((budget_s - 20 - _el()) * 4000), 200000)
        best_h, best_c5 = _cd_phase(best_h, c_vec, 3, max(n_cd, 5000), 0.002, rng)

    # Phase 10: Micro MGD
    if _el() < budget_s - 5:
        h_t = best_h.copy()
        h_t, c = _mgd(h_t, 100, 500000, 3, 0.0002, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
