import numpy as np

def _project_sum(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        # if mismatch, resize? not expected
        h = np.resize(h, n)
    np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - out)
        else:
            order = np.argsort(out)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - out[idx])
                out[idx] += a
                diff -= a
            else:
                a = min(-diff, out[idx])
                out[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return out

    if n ==0: return G

def _proj(h,n):
    t=0.5*n
    h=np.asarray(h,float).copy()
    if h.size!=n: h=np.resize(h,n)
    np.clip(h,0,1,out=h)
    s=h.sum()
    if abs(s-t)<1e-12: return h
    lo,hi=-1.,1.
    for _ in range(60):
       m=0.5*(lo+hi)
       if np.clip(h-m,0,1).sum()>t: lo=m
       else: hi=m
    o=np.clip(h-0.5*(lo+hi),0,1)
    d=t-o.sum()
    if abs(d)>1e-12:
       idx=np.argsort(1-o) if d>0 else np.argsort(o)

          if d>0:

          else:
             a=min(-d,o[p]); o[p]-=a; d+=a
          if abs(d)<1e-12: break

        else:

    o = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = t - o.sum()

                d -= a
