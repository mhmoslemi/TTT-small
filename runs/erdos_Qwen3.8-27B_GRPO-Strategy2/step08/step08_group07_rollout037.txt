    target = n / 2.0
    x = h0.copy()
    y = h0.copy()
    t_k = 1.0

            break
        gy, fy = _grad(y, tol)
        # Backtracking line search
        for _ in range(20):
            x_new = _proj(y - step * gy, target)
            fx_new = _eval(x_new)
            if fx_new <= fy + 0.5 * step * np.dot(gy, x_new - y):
                break
            step *= 0.5

            fx_new = _eval(x_new)

        y = _proj(y, target)
        x = x_new
        t_k = t_new
        step = min(step * 1.05, step0 * 10)
        
        if fx_new < best_f:
            best_f, best_x = fx_new, x.copy()

Let me also add a **dual ascent** component. Consider the epigraph:
min_{h,t} t s.t. f_k(h) ≤ t, 0≤h≤1, ∑h = n/2

Lagrangian with multipliers λ_k ≥ 0:
L = t + ∑_k λ_k(f_k(h) - t)

KKT: 
- ∂L/∂h = ∑_k λ_k ∇f_k(h) = 0 (at optimum, within [0,1])
- ∂L/∂t = 1 - ∑_k λ_k = 0 → ∑_k λ_k = 1
- λ_k(f_k(h) - t) = 0 (complementary slackness)

So at the optimum, the weights λ_k form a probability distribution over the active k's, and h minimizes ∑_k λ_k f_k(h) subject to constraints.

This is exactly the soft-max gradient! The λ_k ≈ softmax(α·f_k) for large α.

So the soft-max approach is actually implementing a primal-dual method. The key is to get α large enough for the weights to concentrate on the truly active constraints.

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-9:
        h = _proj(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_exact(h, tol):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for k in active:
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= h[sh:]
        else:
            s = -sh
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    g *= dx / len(active)
    return g, m

def _grad_smax(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii])
        sh = k - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            s2 = -sh
            g[:n - s2] += wk * (1.0 - h[s2:])
            g[s2:] -= wk * h[:n - s2]
    return g, m

def _nesterov(h, steps, step0, deadline, mode='exact', tol=0.003,
               beta=0.85, alpha=30000, topk=10):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        if mode == 'exact':
            g, _ = _grad_exact(h, tol)
        else:
            g, _ = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
            hn = _proj(h - step * v, target)
            f = _eval(hn)
            h = hn
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _slsqp_refine(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    
    def obj(x):
        c = _correl(x)
        m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]
            ct = c[idx]
        else:
            ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    
    def grad(x):
        return _grad_smax(x, alpha, topk)[0]
    
    try:
        r = minimize(obj, h, jac=grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _anneal_smoothing(h, deadline):
    """Anneal the soft-max parameter from smooth to sharp."""
    n = len(h)
    target = n / 2.0
    best_h, best_f = h.copy(), _eval(h)
    n_steps = 3000
    alphas = np.logspace(3, 5.7, n_steps)  # 1000 to ~500000
    step = 0.003
    v = np.zeros(n)
    beta = 0.88
    for it in range(n_steps):
        if time.time() > deadline:

        alpha = alphas[it]
        topk = max(5, int(300000.0 / alpha))
        g, _ = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < _eval(h) + 1e-14:
            h = hn
            step = min(step * 1.01, 0.03)
        else:
            step *= 0.6
        if fn < best_f:
            best_f, best_h = fn, h.copy()
    return _repair(best_h, n), best_f

def _anneal_block(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _coord_descent(h, niter, rng, deadline):
    """Pair coordinate descent: swap mass between pairs of coordinates."""
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        for scale in [5e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
            d = scale * rng.choice([-1, 1])
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter(h, rng, step0, tol, deadline):
    n = len(h)
    c5c = _eval(h)
    d = 0.003 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _nesterov(hn, 500, step0, deadline, mode='exact', tol=tol, beta=0.9)
    return (hn2, c5n) if c5n < c5c else (h, c5c)

import numpy as np
from scipy.optimize import minimize
import time

# --- Core evaluation ---

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

# --- Projection and repair ---

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-9:
        h = _proj(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

# --- Gradients ---

def _grad_exact(h, tol):
    n = len(h); dx = 2.0 / n
    c = _correl(h); m = c.max()
    active = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for k in active:
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += 1.0 - h[:n - sh]
            if sh > 0: g[:n - sh] -= h[sh:]
        else:
            s = -sh
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
    g *= dx / len(active)
    return g, m

def _grad_smax(h, alpha, topk):
    n = len(h); dx = 2.0 / n
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        sh = int(idx[ii]) - (n - 1); wk = w[ii] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0: g[:n - sh] -= wk * h[sh:]
        else:
            s2 = -sh
            g[:n - s2] += wk * (1.0 - h[s2:])
            g[s2:] -= wk * h[:n - s2]
    return g, m

# --- Optimizers ---

def _nesterov(h, steps, step0, deadline, mode='exact', tol=0.003, beta=0.85, alpha=30000, topk=10):
    n = len(h); target = 0.5 * n
    f = _eval(h); v = np.zeros(n); step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline: break
        if mode == 'exact': g, _ = _grad_exact(h, tol)
        else: g, _ = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target); fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn; step = min(step * 1.02, step0 * 20.0)
        else:
            step *= 0.5
            if step < 1e-10: break
            hn = _proj(h - step * v, target); f = _eval(hn); h = hn
        if f < best_f: best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

At α=200 (very smooth), the soft-max is almost the average of all f_k. The gradient points in the direction that reduces the average overlap, not the maximum. As α increases, the gradient increasingly focuses on the worst shifts.

The risk is: at very small α, the optimization moves h in a direction that's good for the average but bad for the maximum. Then when α increases, we're far from the optimum.

To mitigate: use a faster schedule (more α values in the sharp region) and don't go too far into the smooth region.

import numpy as np
from scipy.optimize import minimize
import time

_ctx = {}

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = 0.5 * n
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - tgt) > 1e-9:
        h = _proj(h, tgt)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        od = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in od:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _g_exact(h, tol):
    n = len(h); dx = 2.0 / n
    c = _correl(h); m = c.max()
    act = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for k in act:
        sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += 1.0 - h[:n - sh]
            if sh > 0: g[:n - sh] -= h[sh:]
        else:
            s = -sh; g[:n - s] += 1.0 - h[s:]; g[s:] -= h[:n - s]
    g *= dx / len(act)
    return g, m

def _g_smax(h, alpha, topk):
    n = len(h); dx = 2.0 / n
    c = _correl(h)
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    m = c.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        sh = int(idx[ii]) - (n - 1); wk = w[ii] * dx
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n - sh])
            if sh > 0: g[:n - sh] -= wk * h[sh:]
        else:
            s2 = -sh; g[:n - s2] += wk * (1.0 - h[s2:]); g[s2:] -= wk * h[:n - s2]
    return g, m

def _nest(h, steps, st0, dl, mode='x', tol=0.003, beta=0.85, alpha=30000, topk=10):
    n = len(h); tgt = 0.5 * n
    f = _eval(h); v = np.zeros(n); st = st0
    bh, bf = h.copy(), f
    for _ in range(steps):
        if time.time() > dl: break
        g, _ = _g_exact(h, tol) if mode == 'x' else _g_smax(h, alpha, topk)
        v = beta * v + (1 - beta) * g
        hn = _proj(h - st * v, tgt); fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn; st = min(st * 1.02, st0 * 20)
        else:
            st *= 0.5
            if st < 1e-10: break
            hn = _proj(h - st * v, tgt); f = _eval(hn); h = hn
        if f < bf: bf, bh = f, h.copy()
    return _repair(bh, n), bf

def _s_obj(x):
    a, tk = _ctx['a'], _ctx['k']
    c = _correl(x); m = c.max()
    ct = c[np.argpartition(c, -tk)[-tk:]] if tk < len(c) else c
    s = np.clip(a * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / a

def _s_jac(x):
    return _g_smax(x, _ctx['a'], _ctx['k'])[0]

def _c_fun(x):
    return x.sum() - _ctx['n'] * 0.5

def _c_jac(x):
    return np.ones(_ctx['n'])

def _slsq(h, a, tk, mi, dl):
    n = len(h)
    if time.time() > dl: return h
    _ctx['a'], _ctx['k'], _ctx['n'] = a, tk, n
    try:
        r = minimize(_s_obj, h, jac=_s_jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _anneal_sm(h, dl):
    n = len(h); tgt = 0.5 * n
    bh, bf = h.copy(), _eval(h); f = bf
    la = np.concatenate([np.linspace(2.7, 4.7, 1000), np.linspace(4.7, 5.7, 2000)])
    st = 0.003; v = np.zeros(n); beta = 0.87

def _abl(h, nit, rng, dl, blks, T0, T1):
    n = len(h); c5c = _eval(h); bh, bc = h.copy(), c5c
    for it in range(nit):
        if time.time() > dl: break
        T = T0 * (T1 / T0) ** (it / max(nit, 1))
        bs = blks[rng.randint(len(blks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.005 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean(); seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n-c5c)/max(T,1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _coord(h, nit, rng, dl):
    n = len(h); c5c = _eval(h); bh, bc = h.copy(), c5c
    scs = [1e-3, 5e-4, 1e-4, 5e-5, 1e-5]
    for _ in range(nit):
        if time.time() > dl: break
        i = rng.randint(n); j = (i+1+rng.randint(0, n-1)) % n
        for sc in scs:
            d = sc * rng.choice([1,-1])
            ni, nj = h[i]+d, h[j]-d
            if 0<=ni<=1 and 0<=nj<=1:
                h[i], h[j] = ni, nj; c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc: bc, bh = c5n, h.copy()
                    break
                h[i], h[j] = ni-d, nj+d
    return bh, bc

def _jitter(h, rng, st0, tol, dl):
    n = len(h); c5c = _eval(h)
    d = 0.003 * rng.randn(n); d -= d.mean()
    hn = _repair(np.clip(h+d, 0, 1), n)
    h2, c2 = _nest(hn, 400, st0, dl, 'x', tol, 0.9)
    return (h2, c2) if c2 < c5c else (h, c5c)
