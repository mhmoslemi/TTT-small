import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        w2 = w[ii]
        if w2 < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w2 * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh+cnt])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:sp+cnt]
    return g

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pgd(h, n_iter, alpha, top_k, step0, rng, t0, tlim):
    n = len(h)
    target = n * 0.5
    cur_c = _eval(h)
    step = step0
    for it in range(n_iter):
        if time.time() - t0 > tlim:
            break
        g = _grad_lse(h, alpha, top_k)
        h_try = _proj_sum(h - step * g, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try; cur_c = c_try
            step = min(step * 1.08, step0 * 10.0)
        else:
            step = max(step * 0.4, step0 * 0.02)
    return h, cur_c

def _block_obj(x, bs, n):
    hfull = np.repeat(x, bs)[:n]
    c = _eval_vec(hfull)
    idx = np.argsort(c)[-8:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(8000.0 * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / 8000.0

def _block_opt(h, B, rng, t0, tlim):
    n = len(h)
    bs = n // B
    hb = h[:bs*B].reshape(B, bs).mean(axis=1).copy()
    hb = np.clip(hb, 0.01, 0.99)
    try:
        res = minimize(_block_obj, hb, args=(bs, n), method='Nelder-Mead',
                       options={'maxiter': 600, 'xatol': 1e-10, 'fatol': 1e-13, 'adaptive': True})
        hb = res.x
    except Exception:
        pass
    hb = _repair(hb, B)
    h_up = np.repeat(hb, bs)[:n]
    h_up = _repair(h_up, n)
    return h_up

def _local2(h, n_iter, step, rng, sa_temp, t0, tlim):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > tlim:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp > 1e-14 and rng.rand() < np.exp(-(c_new-cur_c)/sa_temp):
                accept = True
            if accept:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
            if sa_temp > 1e-14:
                sa_temp *= 0.99995
    return best_h, best_c

    best_c = cur_c
