    n = len(h)

    H = np.fft.rfft(h, N)

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0

            if s > S: lo = mid

def _pn_arrays(h):
    n = len(h)

    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]

    np.cumsum(h, out=pref[1:])

    P = np.empty(n); Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = max(float(np.max(P)), float(np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _c5_direct(h):

def _grad_from_weights(h, wP, wN):
    n = len(h); dx = 2.0 / n
    u = np.zeros(n)
    if n > 1: u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)

    if n > 1 and np.any(u[1:]):

def _resample(h, m):
    n = len(h)
    if m == n:
        return _repair(h.copy(), n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _corr_update(h, i, delta):
    """Compute change in correlations after changing h[i] by delta."""
    n = len(h)
    dh = h[i]
    # corr[k] = sum_m h[m](1-h[m+k])
    # Change: for each valid m, h[i] appears as h[m] (m=i) or h[m+k] (m=i-k)
    # Δcorr[k] = delta*(1-h[i+k]) - delta*h[i-k] (bounds checked)
    # We need the index in the full correlate output
    # full correlate: out[k+n-1] for lag k from -(n-1) to n-1
    # Actually let's just recompute - for n<=1120 it's fast enough
    h[i] += delta
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    h[i] -= delta
    return c

def _coord_sweep(h, deadline, n_steps):

    best_c = _c5_direct(h); best_h = h.copy()
    step_sizes = [5e-5, 3e-5, 1e-5, 5e-6]
    si = 0
    sweeps = 0
    while time.time() < deadline and sweeps < n_steps:
        delta = step_sizes[min(si, len(step_sizes) - 1)]

            if time.time() >= deadline:
                break
            # Try increasing
            if h[i] < 1.0 - 1e-10:

                c = _c5_direct(h)
                if c > best_c:

            # Try decreasing
