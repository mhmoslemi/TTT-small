def _eval(h):

def _correl(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:

    if s < 1e-12:

    h *= target / s
    h = np.clip(h, 0.0, 1.0)

def _proj_sum(h, target):
    """Project h to have sum = target while staying in [0,1]"""
    for _ in range(100):
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h2 = np.clip(h - (lo + hi) * 0.5, 0, 1)
        if abs(h2.sum() - target) < 1e-10:
            return h2
        # Fine adjustment
        diff = target - h2.sum()
        if abs(diff) < 1e-12:
            return h2
        order = np.argsort(1.0 - h2) if diff > 0 else np.argsort(h2)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h2[idx])
                h2[idx] += a
                diff -= a
            else:
                a = min(-diff, h2[idx])
                h2[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
        return h2
    return h

def _grad_shift(h, k, n):
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        # h[i] * (1 - h[i+m]) for i = 0..n-1-m
        g[:n-m] += (1.0 - h[m:])
        if m > 0:
            g[m:] -= h[:-m]
    else:
        p = -m
        g[p:] += (1.0 - h[:-p])
        g[:n-p] -= h[p:]
    return g * dx

def _smax_and_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -50, 50)  # clip to avoid overflow
    logsum = s.max() + np.log(np.sum(np.exp(s - s.max())))
    obj = m + logsum / alpha
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(topk):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_shift(h, idx[ii], n)
    return obj, g

def _nesterov(h, alpha, topk, steps, step0, deadline, beta=0.85):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for i in range(steps):
        if time.time() > deadline:
            break
        _, g = _smax_and_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 30.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.7
            hn = _proj_sum(h - step * v, target)
            f = _eval(hn)
            h = hn
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _proj_sum(best_h, target), best_f

def _slsqp(h, alpha, topk, maxiter, deadline):
    n = len(h)
    def obj(x):
        return _smax_and_grad(x, alpha, topk)[0]
    def jac(x):
        return _smax_and_grad(x, alpha, topk)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _proj_sum(np.clip(r.x, 0, 1), n*0.5)
    except:
        return h

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(len(h), 0.5)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h2.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h2) if diff > 0 else np.argsort(h2)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h2[idx]); h2[idx] += a; diff -= a
            else:
                a = min(-diff, h2[idx]); h2[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h2

def _grad_shift(h, k, n):
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += (1.0 - h[m:])
        if m > 0:
            g[m:] -= h[:-m]
    else:
        p = -m
        g[p:] += (1.0 - h[:-p])
        g[:n - p] -= h[p:]
    return g * dx

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -50, 50)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(topk):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_shift(h, idx[ii], n)
    return g

def _nesterov(h, alpha, topk, step0, deadline, beta0, beta1):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    it = 0
    while time.time() < deadline:
        it += 1
        beta = beta0 + (beta1 - beta0) * min(1.0, it / 4000.0)
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_sum(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.012, step0 * 40.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.72
            if step < step0 * 0.001:
                break
            hn = _proj_sum(h - step * v, target)
            f = _eval(hn)
            h = hn
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _proj_sum(best_h, target), best_f

def _anneal_kick(h, rng, T, nl=6):
    n = len(h)
    h2 = h.copy()
    for _ in range(nl):
        bs = rng.randint(3, 15)
        i0 = rng.randint(0, max(1, n - bs))
        seg = h2[i0:i0 + bs].copy()
        d = 0.008 * T * rng.randn(bs)
        d -= d.mean()
        seg += d
        h2[i0:i0 + bs] = np.clip(seg, 0, 1)
    return _proj_sum(h2, n * 0.5)

def _pairwise_polish(h, max_iters, rng, deadline):
    n = len(h)
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    deltas = np.array([0.001, 0.003, 0.008, 0.02])
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1 + 1e-9 or nj < -1e-9 or nj > 1 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-15:
                c5 = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                break
    return best_h, best_c5

def _slsqp_smooth(h, alpha, topk, maxit, deadline):
    n = len(h)
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        s = np.clip(alpha * (c[idx] - m), -50, 50)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        return _smax_grad(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        from scipy.optimize import minimize
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-14})
        return _proj_sum(np.clip(r.x, 0, 1), n * 0.5)
    except Exception:
        return h
