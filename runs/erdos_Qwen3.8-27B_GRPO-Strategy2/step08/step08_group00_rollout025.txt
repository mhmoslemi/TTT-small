                h[idx] -= a; diff += a

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

    lo, hi = -2.0, 2.0

            lo = mid
        else:

    n = len(h)
    L = k - (n - 1)
    g = np.zeros(n)
    if L > 0:
        g[:n - L] -= h[L:]
        g[L:] += 1.0 - h[:n - L]
    elif L < 0:
        M = -L
        g[:n - M] += 1.0 - h[M:]
        g[M:] -= h[:n - M]
    else:

    return g

    c = _corr(h)
    idx = np.argsort(c)[-K:]
    g = np.zeros(n)
    for k in idx:
        g += _grad_one(h, int(k))
    return g, c, idx

    idx = np.argsort(c)[-K:]

    g, _, _ = _grad_topk(x, K)
