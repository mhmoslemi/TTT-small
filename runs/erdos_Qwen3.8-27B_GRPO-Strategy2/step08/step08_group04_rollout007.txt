        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick SLSQP (0-60s)
    dl1 = min(deadline, t0 + 60)
    h = _slsqp_quick(best_h, 4000, 20, 300, dl1)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 2: FISTA with subgradient (60-300s)
    dl2 = min(deadline, t0 + 300)
    for step0, tk, nits in [(0.006, 3, 12000), (0.003, 5, 10000), (0.0015, 2, 10000)]:
        if time.time() > dl2:
            break
        hr, fr = _fista_subgrad(best_h.copy(), nits, dl2, step0, tk)
        if fr < best_c5:
            best_c5 = fr
            best_h = hr.copy()

    # Phase 3: Multi-scale coordinate descent (300-580s)
    dl3 = min(deadline, t0 + 580)
    best_h, best_c5 = _multi_scale_coord(best_h.copy(), dl3)

    # Phase 4: Pairs annealing (580-820s)
    dl4 = min(deadline, t0 + 820)
    best_h, best_c5 = _pairs_anneal(best_h.copy(), dl4, 50000, 0.015, 0.00005, rng)

    # Phase 5: Second FISTA round (820-950s)
    dl5 = min(deadline, t0 + 950)
    hr, fr = _fista_subgrad(best_h.copy(), 8000, dl5, 0.0008, 2)
    if fr < best_c5:
        best_c5 = fr
        best_h = hr.copy()

    # Phase 6: Fine coordinate descent (950-1060s)
    dl6 = min(deadline, t0 + 1060)
    best_h, best_c5 = _multi_scale_coord(best_h.copy(), dl6,
                                         scales=[0.002, 0.0005, 0.0001, 3e-5])

    # Phase 7: Final polish pairs (1060-end)
    dl7 = deadline
    best_h, best_c5 = _pairs_anneal(best_h.copy(), dl7, 20000, 0.005, 0.00001, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
