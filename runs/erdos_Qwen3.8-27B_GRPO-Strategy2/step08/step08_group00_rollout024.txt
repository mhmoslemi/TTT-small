        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_smooth(h, alpha, n_top, maxiter, ftol):
    n = len(h); dx = 2.0 / n; target = n * 0.5
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argsort(c)[-n_top:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argsort(c)[-n_top:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15: continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n-L])
                g[:n-L] -= wk * x[L:]
            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        return g
    def sf(x): return x.sum() - target
    def sj(x): return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sf, 'jac': sj}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return h

def _pgd(h, n_iter, n_top, step0):
    n = len(h); h = h.copy(); c5 = _eval(h); target = n * 0.5; step = step0
    n2 = n - 1
    for _ in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
        idx = np.argsort(c)[-n_top:]
        sp = c[idx].max() - c[idx].min()
        if sp < 1e-14:
            w = np.ones(n_top) / n_top
        else:
            w = ((c[idx] - c[idx].min()) / sp) ** 6; w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-12: continue
            L = k - n2
            if L >= 0:
                g[L:] += w[ii] * (1.0 - h[:n-L])
                g[:n-L] -= w[ii] * h[L:]
            else:
                M = -L
                g[:n-M] += w[ii] * (1.0 - h[M:])
                g[M:] -= w[ii] * h[:n-M]
        taken = False; s = step
        for _ in range(25):
            h_new = _proj(h - s * g, target)
            c_new = _eval(h_new)
            if c_new < c5 - 1e-16:
                h, c5 = h_new, c_new
                step = min(s * 1.1, step0 * 15); taken = True; break
            s *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 15)
    return _repair(h, n), c5

def _grad_local(h, n_iter, rng, amp0):
    n = len(h); h = h.copy(); c5 = _eval(h); best_h, best_c5 = h.copy(), c5
    n2 = n - 1

        sp = c[idx_t].max() - c[idx_t].min()
        if sp < 1e-14:
            w = np.ones(len(idx_t)) / len(idx_t)
        else:
            w = ((c[idx_t] - c[idx_t].min()) / sp) ** 4; w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx_t):
            if w[ii] < 1e-12: continue

                g[L:] += w[ii] * (1.0 - h[:n-L]); g[:n-L] -= w[ii] * h[L:]
            else:
                M = -L
                g[:n-M] += w[ii] * (1.0 - h[M:]); g[M:] -= w[ii] * h[:n-M]
        np_ = rng.randint(5, 16)
        idx = rng.choice(n, size=np_, replace=False)
        d = -g[idx] * amp + rng.randn(np_) * amp * 0.25
        d -= d.mean()
        h_new = h.copy(); h_new[idx] += d; h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5:
            h, c5 = h_new, c5n
            if c5 < best_c5: best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def _anneal(h, n_iter, rng, T0, T1, plo, phi):
    n = len(h); h = h.copy(); c5 = _eval(h); best_h, best_c5 = h.copy(), c5
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter - 1, 1))
        np_ = rng.randint(plo, phi + 1)
        idx = rng.choice(n, size=np_, replace=False)
        d = rng.randn(np_) * 0.005; d -= d.mean()
        h_new = h.copy(); h_new[idx] += d; h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5 or (T > 1e-20 and rng.rand() < np.exp(-max(0, c5n - c5) / T)):
            h, c5 = h_new, c5n
            if c5 < best_c5: best_c5, best_h = c5, h.copy()
    return best_h, best_c5
