                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values) * 0.5)
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: MGD with progressive alpha
    schedules = [
        (400, 1500, 10, 0.004),
        (1500, 1500, 8, 0.003),
        (6000, 1000, 6, 0.002),
        (25000, 1000, 5, 0.001),
        (100000, 800, 4, 0.0008),
        (500000, 600, 3, 0.0005),
        (2000000, 400, 3, 0.0003),
    ]
    for alpha, steps, tk, s0 in schedules:
        if _el() > 250:
            break
        h_t = best_h.copy()
        h_t, c = _mgd(h_t, steps, alpha, tk, s0, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c; best_h = h_t.copy()

    # Phase 2: SLSQP refinement
    if _el() < budget_s - 400:
        for alpha, tk, mxi in [(5000, 10, 600), (50000, 6, 500), (500000, 4, 400), (5000000, 3, 300)]:
            if _el() > budget_s - 350:
                break
            h_t = _slsqp(best_h.copy(), alpha, tk, mxi)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c; best_h = h_t.copy()

    # Phase 3: Perturbation restarts
    if _el() < budget_s - 350:
        for trial in range(5):
            if _el() > budget_s - 320:
                break
            scale = 0.008 * (trial + 1)
            noise = scale * rng.randn(n)
            noise -= noise.mean()
            h_pert = _repair(best_h + noise, n * 0.5)
            h_t, c = _mgd(h_pert, 400, 50000, 5, 0.002, rng)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c; best_h = h_t.copy()
            h_t2 = _slsqp(h_t.copy(), 100000, 4, 300)
            c2 = _eval(h_t2)
            if c2 < best_c5:
                best_c5 = c2; best_h = h_t2.copy()

    # Phase 4: More SLSQP
    if _el() < budget_s - 250:
        for alpha, tk, mxi in [(100000, 5, 400), (2000000, 3, 300), (20000000, 2, 200)]:
            if _el() > budget_s - 200:
                break
            h_t = _slsqp(best_h.copy(), alpha, tk, mxi)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c; best_h = h_t.copy()

    # Phase 5: SA local search (medium step)
    if _el() < budget_s - 120:
        sa_temp = best_c5 * 3e-5
        remaining = budget_s - 120 - _el()
        nsa = min(int(remaining * 2500), 150000)
        best_h, best_c5 = _local2(best_h, nsa, 8e-5, rng, sa_temp)

    # Phase 6: Sweep (small step)
    if _el() < budget_s - 60:
        remaining = budget_s - 60 - _el()
        npass = min(int(remaining * 250), 4000)
        best_h, best_c5 = _sweep(best_h, npass, 3e-5, rng)

    # Phase 7: Very fine local search
    if _el() < budget_s - 10:
        remaining = budget_s - 10 - _el()
        nfin = min(int(remaining * 5000), 500000)
        best_h, best_c5 = _local2(best_h, nfin, 2e-6, rng)

    # Phase 8: Final ultra-tiny polish
    if _el() < budget_s - 2:
        remaining = budget_s - 2 - _el()
        nfin2 = min(int(remaining * 6000), 300000)
        best_h, best_c5 = _local2(best_h, nfin2, 5e-7, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
